import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def autocorrelation(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def compute_gradient(h, weights):
    """Compute gradient using FFT-based convolution"""
    n = len(h)
    L = 2 * n - 1
    weights_rev = weights[::-1]
    
    # conv_plus = sum_j w_j * h[i-j] (shifted by +j)
    conv_plus = np.fft.ifft(np.fft.fft(weights, L) * np.fft.fft(h, L)).real
    
    # conv_rev = sum_j w_j * h[i+j] (shifted by -j)  
    conv_rev = np.fft.ifft(np.fft.fft(weights_rev, L) * np.fft.fft(h, L)).real
    
    # Take the central parts
    grad = -(conv_plus[n-1:] + conv_rev[n-1:])
    return grad

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    n = len(h)
    dx = 2.0 / n
    S = n / 2
    
    O = autocorrelation(h)
    g = S - O
    return dx * np.max(g)

    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')

    h0_interp = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
    h0_proj = project_to_sum_box(h0_interp, target_sum)
    starts.append(h0_proj)
    
    # 2. Random binary start
    h_rand = np.zeros(n_points)
    ones_indices = np.random.choice(n_points, size=int(target_sum), replace=False)
    h_rand[ones_indices] = 1.0
    h_rand = project_to_sum_box(h_rand, target_sum)
    starts.append(h_rand)
    
    # Run optimization for each starting point
    for h_start in starts:
        h = h_start.copy()
        current_c5 = evaluate_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Gradient descent with smooth surrogate
        alpha = 1.0
        eta = 0.05
        max_iters = 15000
        last_improvement = 0
        
        for iteration in range(max_iters):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Autocorrelation and g_j
            O = autocorrelation(h)
            S = target_sum
            g = S - O  # length 2*n-1
            max_g = np.max(g)
            
            # Smooth-max and weights
            shift = g - max_g  # for numerical stability
            w_raw = np.exp(alpha * shift)
            w = w_raw / np.sum(w_raw)
            
            # Gradient
            grad = compute_gradient(h, w)
            
            # Projected gradient step
            y = h - eta * grad
            h = project_to_sum_box(y, target_sum)
            
            # Evaluate true objective
            c5_now = evaluate_c5(h)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()

            # Schedule updates
            if iteration % 200 == 0:
                if alpha < 100:
                    alpha *= 1.3
                if iteration > 0:
                    eta *= 0.9

            if iteration % 1000 == 0 and iteration > last_improvement + 500:
                noise = np.random.normal(0, 0.02, n_points)
                h = project_to_sum_box(np.clip(h + noise, 0, 1), target_sum)
                current_c5 = evaluate_c5(h)
                if current_c5 < best_c5:

                return dx * np.max(S - O)  # exact C5
            
            constraints = ({'type':'eq', 'fun': lambda x: np.sum(x) - S},)
            bounds = [(0.0,1.0)] * n_points
            
            try:
                res = minimize(obj_full,

                              bounds=bounds,

                    polished_c5 = obj_full(res.x)

                        best_h = res.x.copy()

        best_h = project_to_sum_box(best_h, target_sum)
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
