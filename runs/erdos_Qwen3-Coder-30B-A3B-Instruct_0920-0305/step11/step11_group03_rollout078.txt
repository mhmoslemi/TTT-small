import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    n = len(h)
    dx = 2.0 / n
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_sdp(n, target_sum):
    """Solve SDP relaxation for the autocorrelation vector"""
    # Define the SDP variables
    r = cp.Variable(n)
    t = cp.Variable()
    
    # Build the Toeplitz matrix
    T = cp.Variable((n, n), symmetric=True)
    for i in range(n):
        for j in range(n):
            T[i, j].value = r[abs(i - j)].value if T[i, j].value is None else T[i, j].value

    # Constraints
    constraints = [
        cp.diag(T) == r,  # Toeplitz structure (diagonals)
        T >> 0,  # PSD constraint
        cp.sum((n - cp.arange(n)) * r) == target_sum**2,  # mass constraint
        r >= t,  # lower bound on all r_k
        r >= 0,  # non-negativity
        r <= target_sum,  # upper bound based on max value
    ]
    
    # Objective
    objective = cp.Maximize(t)
    
    # Solve the SDP
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            return r.value, t.value
    except:
        pass
    return None, None

def round_spectral(r_opt, n, target_sum):
    """Round the SDP solution to a feasible step function using spectral factorization"""
    # Compute the discrete Fourier transform
    S = np.fft.fft(r_opt)
    # Take the square root (taking real part to avoid numerical errors)
    A = np.sqrt(np.maximum(S, 0))
    
    best_h = None
    best_c5 = float('inf')
    
    # Try different phase vectors
    for phase_seed in [0, 1, 2, 3]:
        np.random.seed(phase_seed)
        # Generate random phases (but we'll also try zero phase)
        if phase_seed == 0:
            phi = np.zeros(n)
        else:
            phi = np.random.uniform(0, 2*np.pi, n)
        # Construct the signal in frequency domain
        H_freq = A * np.exp(1j * phi)
        # Inverse FFT to get the signal
        h_temp = np.fft.ifft(H_freq).real
        # Project to feasible set
        h_proj = project_to_sum_box(h_temp, target_sum)
        # Evaluate the true C5
        c5 = evaluate_c5(h_proj)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_proj.copy()
    
    return best_h, best_c5

def projected_gradient_ascent(h_start, target_sum, dx, max_iter=50, tol=1e-6):
    """Perform projected gradient ascent on the true max-overlap objective"""
    h = h_start.copy()
    last_improvement = float('inf')
    n = len(h)
    
    for i in range(max_iter):
        # Compute all correlations
        corr = correlate(h, 1-h, mode="full")
        
        # Find the minimum correlation
        min_corr_idx = np.argmin(corr)
        min_corr = corr[min_corr_idx]
        
        # Compute the gradient at minimum correlation
        grad = np.zeros(n)
        for k in range(n):
            if k < n and min_corr_idx >= k:
                grad[k] += h[min_corr_idx - k]
            if k < n and min_corr_idx + k < 2*n - 1:
                grad[k] += h[min_corr_idx + k - (n-1)]
        
        # Step size schedule
        eta = 0.2 / np.sqrt(i + 1)
        
        # Update step
        h_new = h + eta * grad
        # Project to feasible set
        h_new = project_to_sum_box(h_new, target_sum)
        
        # Check for improvement
        new_c5 = evaluate_c5(h_new)
        if new_c5 >= last_improvement - tol:
            break
        last_improvement = new_c5
        h = h_new.copy()
        
    return h, last_improvement

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 80
    dx = 2.0 / n
    target_sum = n / 2
    
    best_c5 = float('inf')
    best_h = None
    
    # Solve SDP relaxation
    r_opt, t_opt = solve_sdp(n, target_sum)
    
    if r_opt is not None and t_opt is not None:
        # Round the solution using spectral factorization
        h_rounded, c5_rounded = round_spectral(r_opt, n, target_sum)
        
        if c5_rounded < best_c5:
            best_c5 = c5_rounded
            best_h = h_rounded.copy()
    
    # If SDP solving or rounding failed, use fallback
    if best_h is None:
        h_rand = np.random.uniform(0, 1, n)
        h_rand = project_to_sum_box(h_rand, target_sum)
        best_h = h_rand.copy()
        best_c5 = evaluate_c5(best_h)
    
    # Local refinement by projected gradient ascent
    remaining_time = budget_s - (time.time() - start_time)
    
    if remaining_time > 20:
        max_iter = min(50, int(remaining_time / 2.0))
        refined_h, refined_c5 = projected_gradient_ascent(best_h, target_sum, dx, max_iter)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h.copy()
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(final_c5), n)
