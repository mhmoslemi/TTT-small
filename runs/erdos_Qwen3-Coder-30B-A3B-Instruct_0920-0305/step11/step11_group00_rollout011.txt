import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_value(h):
    """Compute C5 value"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def triangular_initial(n):
    """Create triangular initial shape"""
    m = n//2
    # linear decreasing from 1 to 0 over first half
    first = np.linspace(1, 0, m, endpoint=False)
    if n % 2 == 0:
        h = np.concatenate([first, first[::-1]])
    else:
        h = np.concatenate([first, first[-2::-1]])
    return h

def random_feasible_vector(n, target):
    """Generate random feasible vector"""
    h = np.random.random(n)
    h = project_to_sum_box(h, target)
    return h

def grad_shift(h, s):
    """Compute gradient for a specific shift s"""
    n = len(h)
    base = np.arange(n) + (n-1)
    padded = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    term1 = 1 - padded[base + s]
    term2 = padded[base - s]
    return term1 - term2

def smooth_optimize(h0, target_sum, dx, time_limit, start_time):
    """Smooth optimization using log-sum-exp approximation"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_value(h)
    
    # Parameters
    alphas = np.linspace(0.5, 30, 15)
    step = 0.3
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        for _ in range(100):
            if time.time() - start_time > time_limit:
                break
            # Compute correlation
            c = compute_corr(h)
            
            # Compute weights
            max_c = np.max(c)
            expvals = np.exp(alpha * (c - max_c))
            w = expvals / expvals.sum()
            
            # Compute gradient
            grad = np.zeros_like(h)
            for idx_shift, weight in enumerate(w):
                if weight == 0:
                    continue
                s = idx_shift - (len(h) - 1)
                shift_grad = grad_shift(h, s)
                grad += weight * shift_grad
            
            # Line search
            step_cur = step
            for _ in range(5):
                h_new = h - step_cur * grad
                h_new = project_to_sum_box(h_new, target_sum)
                c5_new = c5_value(h_new)
                if c5_new < best_c5 - 1e-12:
                    h = h_new
                    best_c5 = c5_new
                    best_h = h.copy()
                    break
                else:
                    step_cur *= 0.5
    
    return best_h, best_c5

def subgradient_optimize(h0, target_sum, dx, time_limit, start_time):
    """Subgradient optimization for fine-tuning"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_value(h)
    
    step = 0.2
    for iteration in range(300):
        if time.time() - start_time > time_limit:
            break
            
        # Compute correlation
        c = compute_corr(h)
        max_c = np.max(c)
        
        # Find indices with maximum correlation
        indices = np.where(np.abs(c - max_c) < 1e-8)[0]
        
        # Compute average gradient
        grad = np.zeros_like(h)
        for idx in indices:
            s = idx - (len(h) - 1)
            shift_grad = grad_shift(h, s)
            grad += shift_grad
            
        grad /= len(indices)
        
        # Line search
        step_cur = step
        for _ in range(5):
            h_new = h - step_cur * grad
            h_new = project_to_sum_box(h_new, target_sum)
            c5_new = c5_value(h_new)
            if c5_new < best_c5 - 1e-12:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                break
            else:
                step_cur *= 0.5
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    best_h = None
    best_c5 = np.inf
    best_n = n_points
    
    # Try different n values
    for n in [120, 200]:
        if time.time() - start_time > budget_s * 0.95:
            break
        dx = 2.0 / n
        target_sum = n / 2
        
        # Try different initialization strategies
        for init_type in ['triangular', 'random']:
            if time.time() - start_time > budget_s * 0.95:
                break
                
            if init_type == 'triangular':
                h0 = triangular_initial(n)
            else:
                h0 = random_feasible_vector(n, target_sum)
                
            # Smooth optimization
            h_smooth, c5_smooth = smooth_optimize(h0, target_sum, dx, budget_s * 0.8, start_time)
            
            if c5_smooth < best_c5:
                best_h = h_smooth.copy()
                best_c5 = c5_smooth
                best_n = n
                
            # Subgradient optimization
            h_sub, c5_sub = subgradient_optimize(h_smooth, target_sum, dx, budget_s * 0.95, start_time)
            
            if c5_sub < best_c5:
                best_h = h_sub.copy()
                best_c5 = c5_sub
                best_n = n
    
    return (best_h.tolist(), float(best_c5), best_n)
