import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 80
    dx = 2.0 / n_points
    M = n_points / 2
    
    # SDP solve with timeout
    max_sdp_time = 250  # seconds
    sdp_time = min(max_sdp_time, budget_s - 100)  # Leave margin for other steps
    
    # Build SDP
    r = cp.Variable(n_points)
    t = cp.Variable()
    
    # Construct Toeplitz matrix
    T = cp.Variable((n_points, n_points), symmetric=True)
    
    # Define Toeplitz constraints
    for i in range(n_points):
        for j in range(n_points):
            k = abs(i - j)
            T[i, j] = r[k]
    
    # Constraints
    constraints = [
        T >> 0,  # PSD
        cp.sum([(n_points - k) * r[k] for k in range(n_points)]) == M**2,
        r >= t,
        r <= M
    ]
    
    # Add box constraint r_k <= n-k
    for k in range(n_points):
        constraints.append(r[k] <= n_points - k)
    
    # Objective
    objective = cp.Maximize(t)
    
    # Solve SDP
    prob = cp.Problem(objective, constraints)
    
    sdp_status = "UNATTAINED"
    r_opt = None
    t_opt = None
    
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, verbose=False, time_limit=sdp_time)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            sdp_status = "OPTIMAL"
            r_opt = r.value
            t_opt = t.value
        elif prob.status in ['infeasible', 'unbounded_inaccurate']:
            sdp_status = "INFEASIBLE"
    except:
        sdp_status = "ERROR"
    
    # If SDP fails, fallback to simple approach
    if sdp_status != "OPTIMAL":
        # Fallback to uniform random with mass constraint
        h_fallback = np.full(n_points, 0.5)
        h_fallback = project_to_sum_box(h_fallback, M)
        h_fallback = np.clip(h_fallback, 0, 1)
        c5_fallback = compute_c5_direct(h_fallback)
        return (h_fallback.tolist(), float(c5_fallback), n_points)
    
    # Extract and round from SDP solution
    def project_to_sum_box(x, target_sum):
        def f(lam):
            projected = np.clip(x + lam, 0, 1)
            return np.sum(projected) - target_sum
        a, b = -1.0, 1.0
        while f(a) * f(b) > 0:
            a -= 0.1
            b += 0.1
        for _ in range(50):
            m = (a + b) / 2
            fm = f(m)
            if abs(fm) < 1e-10:
                break
            if fm * f(a) < 0:
                b = m
            else:
                a = m
        result = np.clip(x + (a + b)/2, 0, 1)
        return result
    
    def compute_c5_direct(h):
        corr = correlate(h, 1 - h, mode="full")
        return dx * np.max(corr)
    
    best_h = None
    best_c5 = float('inf')
    
    # Spectral factorization with multiple phase seeds
    phases = [0, np.pi/4, np.pi/2, np.pi]
    
    for phase_seed in phases:
        # Compute DFT of r_opt
        S = fft(r_opt)
        # Ensure real and non-negative
        S = np.real(S)
        A = np.sqrt(np.maximum(S, 0))
        
        # Define random phase (with seed)
        np.random.seed(int(phase_seed))
        phase = np.exp(1j * np.random.uniform(0, 2*np.pi, n_points))
        
        # Inverse FFT with phase
        h_temp = np.real(ifft(A * phase))
        
        # Project onto feasible set
        h_proj = project_to_sum_box(h_temp, M)
        h_proj = np.clip(h_proj, 0, 1)
        
        # Compute exact C5
        c5_val = compute_c5_direct(h_proj)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_proj.copy()
    
    # Final polishing with projected gradient
    if best_h is not None:
        max_gradient_time = min(150, budget_s / 4)
        gradient_start = time.time()
        
        h_current = best_h.copy()
        prev_c5 = float('inf')
        
        for iter_num in range(1000):  # Max iterations
            if time.time() - gradient_start > max_gradient_time:
                break
                
            # Compute correlations
            corr = correlate(h_current, 1 - h_current, mode="full")
            min_corr_idx = np.argmax(corr)
            min_corr_val = corr[min_corr_idx]
            
            # Compute gradient of the minimum correlation
            grad = np.zeros(n_points)
            k = min_corr_idx - (n_points - 1)
            for i in range(n_points):
                if i + k >= 0 and i + k < n_points:
                    grad[i] += h_current[i + k]
                if i - k >= 0 and i - k < n_points:
                    grad[i] += h_current[i - k]
            
            # Step size
            eta = 0.2 / np.sqrt(iter_num + 1) if iter_num > 0 else 0.2
            
            # Update
            h_new = h_current + eta * grad
            h_new = project_to_sum_box(h_new, M)
            h_new = np.clip(h_new, 0, 1)
            
            # Evaluate
            new_c5 = compute_c5_direct(h_new)
            
            # If improvement, update
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                h_current = h_new.copy()
            else:
                h_current = h_new.copy()
                
            # Convergence check
            if abs(prev_c5 - new_c5) < 1e-6:
                break
            prev_c5 = new_c5
    
    # Final check 
    if best_h is None:
        # Use fallback
        h_fallback = np.full(n_points, 0.5)
        h_fallback = project_to_sum_box(h_fallback, M)
        h_fallback = np.clip(h_fallback, 0, 1)
        best_h = h_fallback
        best_c5 = compute_c5_direct(best_h)
        
    return (best_h.tolist(), float(best_c5), n_points)
