n_points = 128

def thue_morse_binary(n):
    """Generate Thue-Morse binary sequence of length n (exactly n/2 ones)"""
    if n % 2 != 0:
        raise ValueError("n must be even")
    # Generate Thue-Morse up to 2*n and take first n
    tm = [0]
    for i in range(n):
        if i >= len(tm):
            tm.extend([1 - x for x in tm])
        if len(tm) > n:
            tm = tm[:n]
    return np.array(tm, dtype=float)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water filling"""
    # Sort values in descending order
    indices = np.argsort(-x)
    sorted_x = x[indices]
    
    # Binary search for lambda
    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high) / 2
        projected = np.clip(sorted_x + mid, 0, 1)
        total = np.sum(projected)
        if abs(total - target_sum) < 1e-10:
            break
        elif total > target_sum:
            low = mid
        else:
            high = mid
    
    lam = (low + high) / 2
    result = np.clip(x + lam, 0, 1)
    return result

def autocorr(h):
    """Compute full correlation with FFT"""
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h, L) * np.fft.fft(h, L)).real
    return corr * dx

def compute_C5(h):
    """Compute C5 bound from h"""
    corr = autocorr(h)
    return 1.0 - np.min(corr)

def greedy_swap_local_search(h, dx, budget_s, start_time):
    """Perform a greedy local search via swapping 0 and 1 to improve worst shift correlation"""
    h = h.copy()
    corr = autocorr(h)
    min_idx = np.argmin(corr)
    min_corr = corr[min_idx]
    shift = min_idx - (n_points - 1)  # convert index to shift offset
    
    # Start with current worst shift
    while time.time() - start_time < budget_s * 0.99:
        # Set up candidates for improvement
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        best_swap = None
        best_net = -np.inf
        
        # Try all pairs (zero, one) to see how they can improve at shift
        for i_zero in zeros:
            gain = 0
            if i_zero + shift >= 0 and i_zero + shift < n_points:
                gain += h[i_zero + shift]
            if i_zero - shift >= 0 and i_zero - shift < n_points:
                gain += h[i_zero - shift]
            
            for i_one in ones:
                loss = 0
                if i_one + shift >= 0 and i_one + shift < n_points:
                    loss += h[i_one + shift]
                if i_one - shift >= 0 and i_one - shift < n_points:
                    loss += h[i_one - shift]
                
                net = gain - loss
                if net > best_net:
                    best_net = net
                    best_swap = (i_zero, i_one)
        
        if best_net <= 0:
            break
            
        i_zero, i_one = best_swap
        h[i_zero] = 1.0
        h[i_one] = 0.0
        
        # Incremental update of correlation for the affected positions
        corr = autocorr(h)
        min_idx = np.argmin(corr)
        min_corr = corr[min_idx]
        shift = min_idx - (n_points - 1)
    
    return h
