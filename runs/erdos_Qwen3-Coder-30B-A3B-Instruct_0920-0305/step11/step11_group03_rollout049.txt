def project_to_box_and_sum(h, target_sum):
    """Project h onto [0,1] and adjust sum to target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    
    # Binary search for lambda
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        h_test = np.clip(h + m, 0, 1)
        test_sum = np.sum(h_test)
        if abs(test_sum - target_sum) < 1e-10:
            return h_test
        elif test_sum < target_sum:
            a = m
        else:
            b = m
    
    return np.clip(h + (a + b)/2, 0, 1)

def evaluate_erdos_solution(h):
    """Evaluate C5 from h"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)
