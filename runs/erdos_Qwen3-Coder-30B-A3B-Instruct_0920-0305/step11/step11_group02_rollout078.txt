def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def autocorrelation(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def compute_gradient(h, w):
    """Compute gradient using FFT-based convolution"""
    w_rev = w[::-1]
    L = 2 * len(h) - 1
    
    # Convolution with h using FFT
    conv_plus = np.fft.ifft(np.fft.fft(w, L) * np.fft.fft(h, L)).real
    conv_rev = np.fft.ifft(np.fft.fft(w_rev, L) * np.fft.fft(h, L)).real
    
    # Extract the relevant part
    shift = len(h) - 1
    grad = -(conv_plus[shift:] + conv_rev[shift:])
    return grad

def smooth_max_gradient_descent(h0, target_sum, dx, max_iter=10000, alpha_start=1.0, 
                              eta_start=0.05, alpha_max=100.0, verbose=False):
    """Optimize using smooth-max gradient descent"""
    n = len(h0)
    S = target_sum
    
    # Initialize
    h = h0.copy()
    alpha = alpha_start
    eta = eta_start
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Precompute FFT size
    L = 2 * n - 1
    
    for iteration in range(max_iter):
        # Compute autocorrelation
        O = autocorrelation(h)
        g = S - O  # g_j = S - O_j
        
        # Compute smooth max and weights with numerical stability
        max_g = np.max(g)
        shift = g - max_g
        w_raw = np.exp(alpha * shift)
        w = w_raw / np.sum(w_raw)
        
        # Compute gradient via FFT
        grad = compute_gradient(h, w)
        
        # Take gradient step
        y = h - eta * grad[:n]  # Take only the first n elements
        h = project_to_sum_box(y, S)
        
        # Evaluate true objective
        current_C5 = dx * max_g
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
        
        # Adapt parameters
        if iteration % 200 == 0:
            alpha = min(alpha * 1.3, alpha_max)
            eta = eta * 0.9
            
        # Random perturbation to escape local minima
        if iteration % 500 == 0 and iteration > 0:
            h = h + 0.02 * np.random.randn(n)
            h = project_to_sum_box(np.clip(h, 0, 1), S)
        
        if verbose and iteration % 1000 == 0:
            print(f"Iteration {iteration}: C5 = {current_C5:.6f}, alpha = {alpha:.3f}")
    
    return best_h, best_C5
