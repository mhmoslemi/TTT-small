import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_autocorr(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    # For shift k from -(n-1) to n-1, we want positive shifts k=0 to n-1
    # The correlation at index n-1 corresponds to shift 0
    # We want the maximum over all shifts
    return dx * np.max(corr[n-1:])

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_autocorr(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h, min_shifts):
    """Compute gradient of min autocorrelation"""
    grad = np.zeros_like(h)
    for k in min_shifts:
        # Left shift contribution: h[i+k] for i=0..n-k-1
        if k > 0:
            grad[:n_points-k] += h[k:]
        # Right shift contribution: h[i-k] for i=k..n-1
        if k > 0:
            grad[k:] += h[:n_points-k]
        # For k=0, we get h_i + h_i = 2*h_i, but we add only h_i in each term
        # Actually we compute correlation sum_i h_i h_{i+k} for each k.

        # for both positive and negative shifts, but let's focus on non-negative ones

    return grad
