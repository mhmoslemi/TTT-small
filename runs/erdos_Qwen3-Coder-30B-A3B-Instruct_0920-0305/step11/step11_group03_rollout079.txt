import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 80
    dx = 2.0 / n_points
    M = n_points / 2
    
    # Time allocation
    sdpsolve_time = 250
    remaining_time = budget_s - sdpsolve_time - 5  # Leave buffer for rest
    
    # SDP model
    r = cp.Variable(n_points)
    t = cp.Variable()
    
    # Construct Toeplitz matrix
    T = cp.bmat([[r[abs(i-j)] for j in range(n_points)] for i in range(n_points)])
    
    # Constraints
    constraints = [
        T >> 0,  # PSD
        cp.sum((n_points - cp.arange(n_points)) * r) == M**2,  # Mass constraint
        r >= t,  # Lower bound on all r_k
        r >= 0,  # Box constraints
        r <= M,  # Box constraints
        r <= n_points - cp.arange(n_points)  # Box constraints
    ]
    
    # Objective
    objective = cp.Maximize(t)
    
    # Solve SDP
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=10000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            r_opt = r.value
            t_opt = t.value
        else:
            raise Exception("SDP solver failed")
    except:
        # Fallback to gradient-free method (simplified)
        h0 = np.array(initial_h_values)
        h0 = h0 * (M / np.sum(h0))
        return (h0.tolist(), evaluate_erdos_solution(h0), n_points)
    
    # From optimal r to signal h via spectral factorization
    # Compute DFT of r_opt
    S = np.fft.fft(r_opt)
    A = np.sqrt(np.maximum(S, 0))  # Magnitudes
    
    # Try multiple phase seeds
    best_h = None
    best_C5 = float('inf')
    
    for phase_seed in range(4):
        np.random.seed(phase_seed)
        phi = 2 * np.pi * np.random.rand(n_points) if phase_seed > 0 else np.zeros(n_points)
        h_candidate = np.fft.ifft(A * np.exp(1j * phi)).real
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Project onto constraint
        current_sum = np.sum(h_candidate)
        if abs(current_sum - M) > 1e-10:
            # Binary search for lambda
            a, b = -1.0, 1.0
            for _ in range(50):
                m = (a + b) / 2
                test = np.clip(h_candidate + m, 0, 1)
                test_sum = np.sum(test)
                if abs(test_sum - M) < 1e-12:
                    break
                if test_sum > M:
                    b = m
                else:
                    a = m
            h_candidate = np.clip(h_candidate + (a + b) / 2, 0, 1)
        
        # Evaluate C5
        corr_full = correlate(h_candidate, 1 - h_candidate, mode='full')
        C5_candidate = np.max(corr_full) * dx
        if C5_candidate < best_C5:
            best_C5 = C5_candidate
            best_h = h_candidate
    
    # Local refinement with projected gradient ascent
    h_refined = best_h.copy()
    eta = 0.2
    current_C5 = best_C5
    max_iter = min(50, int(remaining_time / 2))
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute autocorrelations
        corr_full = correlate(h_refined, 1 - h_refined, mode='full')
        min_corr_idx = np.argmin(corr_full)
        min_corr_val = corr_full[min_corr_idx]
        
        # Compute subgradient
        subgrad = np.zeros(n_points)
        for j in range(n_points):
            if j + min_corr_idx < n_points:
                subgrad[j] += h_refined[j + min_corr_idx]
            if j >= min_corr_idx:
                subgrad[j] += h_refined[j - min_corr_idx]
        
        # Gradient step
        h_new = h_refined + eta * subgrad
        h_new = np.clip(h_new, 0, 1)
        
        # Project
        current_sum = np.sum(h_new)
        if abs(current_sum - M) > 1e-10:
            a, b = -1.0, 1.0
            for _ in range(50):
                m = (a + b) / 2
                test = np.clip(h_new + m, 0, 1)
                test_sum = np.sum(test)
                if abs(test_sum - M) < 1e-12:
                    break
                if test_sum > M:
                    b = m
                else:
                    a = m
            h_new = np.clip(h_new + (a + b) / 2, 0, 1)
        
        # Evaluate
        corr_full_new = correlate(h_new, 1 - h_new, mode='full')
        new_C5 = np.max(corr_full_new) * dx
        
        if new_C5 < current_C5 - 1e-8:
            h_refined = h_new
            current_C5 = new_C5
        else:
            eta *= 0.9  # Decrease step size
            
    return (h_refined.tolist(), float(current_C5), n_points)
