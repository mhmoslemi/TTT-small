import numpy as np

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Initial bounds for lambda
    lam_low = -np.max(x) - 1
    lam_high = 1 - np.min(x) + 1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        lam = (lam_low + lam_high) / 2
        fm = f(lam)
        if abs(fm) < tol:
            break
        if fm * f(lam_low) < 0:
            lam_high = lam
        else:
            lam_low = lam
    
    result = np.clip(x + lam, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Evaluate C5 value"""
    corr = compute_corr(h)
    return np.max(corr)

def gradient_c_k(h, k):
    """Compute gradient of c_k(h)"""
    n = len(h)
    grad = np.zeros(n)
    
    # Create padded array
    pad = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    
    # Use vectorized indexing
    base_indices = np.arange(n) + (n-1)
    
    if k >= 0:
        grad = (1 - pad[base_indices + k]) - pad[base_indices - k]
    else:
        grad = (1 - pad[base_indices + k]) - pad[base_indices - k]
    
    return grad

def log_sum_exp_surrogate(h, alpha):
    """Compute smooth surrogate for max with log-sum-exp"""
    corr = compute_corr(h)
    max_c = np.max(corr)
    # Avoid numerical issues
    exp_vals = np.exp(alpha * (corr - max_c))
    log_sum = max_c + np.log(np.sum(exp_vals)) / alpha
    return log_sum

def compute_gradient_surrogate(h, alpha):
    """Compute gradient of the log-sum-exp surrogate"""
    corr = compute_corr(h)
    max_c = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_c))
    weights = exp_vals / np.sum(exp_vals)
    
    # Compute average gradient
    n = len(h)
    grad = np.zeros(n)
    
    # Build correlation gradient matrix
    pad = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    base_indices = np.arange(n) + (n-1)
    
    for k in range(-n+1, n):
        if k >= 0:
            grad_k = (1 - pad[base_indices + k]) - pad[base_indices - k]
        else:
            grad_k = (1 - pad[base_indices + k]) - pad[base_indices - k]
        grad += weights[k + n - 1] * grad_k
    
    return grad

        # Project

        max_val = np.max(corr)

        active_indices = np.where(np.abs(corr - max_val) < 1e-12)[0]
        
        if len(active_indices) == 0:
            active_indices = [np.argmax(corr)]
            
        # Compute average gradient
        avg_grad = np.zeros(n_points)
        for k in active_indices:
            grad_k = gradient_c_k(h, k - n_points + 1)
            avg_grad += grad_k

        # Update step
        h_new = project_to_sum_box(h - eta * avg_grad, n_points/2)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < c5_old:
            h = h_new
            c5_old = c5_new
            if eta < 0.15:

        else:
            eta = max(eta * 0.5, 1e-6)

            c5_test = evaluate_c5(h_test)

                h = h_test
    
    return h
