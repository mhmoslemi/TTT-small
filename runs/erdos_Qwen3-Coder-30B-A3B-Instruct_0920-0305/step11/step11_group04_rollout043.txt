from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given h"""
    # Compute correlation using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # Scale by dx
    dx = 2.0 / len(h)
    corr = corr * dx
    # Return maximum correlation (which is C_5)
    return np.max(corr)
