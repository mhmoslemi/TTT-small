import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
n_points = 128
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def autocorr(h):
    """Compute full correlation of h with itself (linear, not circular)."""
    corr = correlate(h, h, mode='full')
    return corr * dx

def compute_C5(h):
    """Compute C5 from h."""
    corr = autocorr(h)
    return 1.0 - np.min(corr)

def project_to_sum_box(x, target_sum):
    """Project x onto [0,1]^n with sum = target_sum using water-filling."""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    lam = (a + b) / 2
    result = np.clip(x + lam, 0, 1)
    return result

def thue_morse_binary(n):
    """Generate Thue-Morse binary sequence of length n."""
    if n <= 0:
        return np.array([])
    if n == 1:
        return np.array([0])
    
    # Generate TM sequence of length 2^k >= n
    k = int(np.ceil(np.log2(n)))
    tm = [0]
    for _ in range(k):
        tm = tm + [1 - x for x in tm]
    
    # Truncate or pad to n
    tm = tm[:n]
    tm = np.array(tm, dtype=float)
    
    # Ensure sum = target_sum by adjusting
    current_sum = np.sum(tm)
    if abs(current_sum - TARGET_SUM) < 1e-10:
        pass
    elif current_sum > TARGET_SUM:
        # Turn some 1's to 0's
        indices = np.where(tm == 1)[0]
        num_to_zero = int(current_sum - TARGET_SUM)
        indices_to_zero = indices[:num_to_zero]
        tm[indices_to_zero] = 0
    else:
        # Turn some 0's to 1's
        indices = np.where(tm == 0)[0]
        num_to_one = int(TARGET_SUM - current_sum)
        indices_to_one = indices[:num_to_one]
        tm[indices_to_one] = 1
    
    return tm

def greedy_swap_local_search(h, dx, budget_s, start_time):
    """Perform greedy local swaps focused on worst shift."""
    # Compute correlation to find worst shift
    corr = correlate(h, h, mode='full') * dx
    k_worst = np.argmin(corr)
    # shift index corresponds to shift = k_worst - (n_points - 1)
    shift = k_worst - (n_points - 1)
    
    # Run iterative swaps
    max_iter = 10000
    iter_count = 0
    while iter_count < max_iter and (time.time() - start_time < budget_s - 0.1):
        gain = -np.inf
        best_i0, best_i1 = -1, -1
        
        # For each zero i0, check how changing it would affect shift
        zeros = np.where(h == 0)[0]
        for i0 in zeros:
            # Calculate potential changes in correlation at shift
            g = 0
            if (i0 + shift >= 0) and (i0 + shift < n_points):
                if h[i0 + shift] == 1:
                    g += 1
            if (i0 - shift >= 0) and (i0 - shift < n_points):
                if h[i0 - shift] == 1:
                    g += 1
            if g <= 0:
                continue
            
            # For each one i1, calculate net change
            ones = np.where(h == 1)[0]
            for i1 in ones:
                # loss from removing 1 at i1
                l = 0
                if (i1 + shift >= 0) and (i1 + shift < n_points):
                    if h[i1 + shift] == 1:
                        l += 1
                if (i1 - shift >= 0) and (i1 - shift < n_points):
                    if h[i1 - shift] == 1:
                        l += 1
                
                net = g - l
                if net > gain:
                    gain = net
                    best_i0, best_i1 = i0, i1
            
        if gain <= 0:
            break  # No improvement found
        
        # Perform the swap
        h[best_i0] = 1.0
        h[best_i1] = 0.0
        
        iter_count += 1
        
        # Update correlation
        # Fast recalculation for shift: only update affected pairs
        # Actually it's not deemed necessary for performance here
        # since the number of iterations is small
    
    return h

            c5_new = 1.0 - np.min(autocorr(h_candidate))

        h_fallback = np.random.rand(n_points)
        best_h = project_to_sum_box(h_fallback, TARGET_SUM)
