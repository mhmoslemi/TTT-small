import numpy as np
from scipy.signal import correlate
from scipy.sparse import diags
import cvxpy as cp
import time

def proj_sum_box(v, target):
    """Project v onto box constraints [0,1] with sum = target"""
    def f(lam):
        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    return np.clip(v + (a + b)/2, 0, 1)

def corr_full(h):
    """Compute full correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def grad_shift(h, k):
    """Gradient of correlation term c_k(h) = sum h_i * (1 - h_{i+k})"""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] = 1 - h[:n-k]
        grad[:k] = 1 - h[:k]
    else:
        grad[:k] = 1 - h[-k:]
        grad[k:] = 1 - h[k:]
    if k != 0:
        grad -= h
    return grad

def max_overlap(h, dx):
    """Return C5 = max_k c_k(h) * dx"""
    corr = corr_full(h)
    return dx * np.max(corr)

def shift_mask(k, n):
    """Generate the (0,1) mask for shift k as a sparse matrix"""
    diags_list = []
    offset_list = []
    if k >= 0:
        if k < n:
            diags_list.append(np.ones(n-k))
            offset_list.append(k)
    else:
        if -k < n:
            diags_list.append(np.ones(n+k))
            offset_list.append(k)
    M = diags(diags_list, offset_list, shape=(n,n))
    return M

def build_sdp(n):
    """Build the SDP problem"""
    h = cp.Variable(n)
    X = cp.Variable((n,n), PSD=True)
    tau = cp.Variable()
    
    diag_con = cp.diag(X) == h
    box_con = [h >= 0, h <= 1]
    sum_con = cp.sum(h) == n/2
    tau_con = tau >= 0
    
    constraints = [diag_con] + box_con + [sum_con] + [tau_con]
    
    for k in range(-(n-1), n):
        M = shift_mask(k, n)
        constraints.append(cp.trace(M @ X) >= (n/2) * (1 - tau))
        
    objective = cp.Minimize(tau)
    prob = cp.Problem(objective, constraints)
    return prob, h, X, tau

def solve_sdp(problem):
    """Solve the SDP problem"""
    try:
        problem.solve(solver=cp.SCS, max_iters=5000, verbose=False)
        if problem.status in ["optimal", "optimal_inaccurate"]:
            return True
        else:
            return False
    except Exception:
        return False

def round_from_sdp(h_sdp, X_sdp, n):
    """Try different rounding methods"""
    candidates = []
    
    # 1. Diagonal rounding
    h_diag = h_sdp.copy()
    candidates.append(h_diag)
    
    # 2. Leading eigenvector rounding
    try:
        eigenvals, eigenvecs = np.linalg.eigh(X_sdp)
        idx = np.argmax(eigenvals)
        v = eigenvecs[:, idx]
        sign = np.sign(np.mean(v)) if np.mean(v) != 0 else 1
        h_eig = sign * np.sqrt(np.abs(eigenvals[idx])) * v
        h_eig = proj_sum_box(h_eig, n/2)
        candidates.append(h_eig)
    except:
        pass
    
    # 3. Gaussian randomization
    try:
        for _ in range(5):
            z = np.random.multivariate_normal(np.zeros(n), X_sdp)
            h_rand = np.abs(z)
            h_rand = proj_sum_box(h_rand, n/2)
            candidates.append(h_rand)
    except:
        pass
    
    # Evaluate all candidates
    best_h = None
    best_c5 = float('inf')
    for h in candidates:
        c5 = max_overlap(h, 2.0/n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    return best_h, best_c5

def polish(h_seed, n, dx, budget_s, start_time):
    """Polish using subgradient and swap moves"""
    h = h_seed.copy()
    best_h = h.copy()
    best_c5 = max_overlap(h, dx)
    
    step = 0.07
    h_prev = h.copy()
    beta = 0.9
    
    for it in range(2000):
        if time.time() - start_time > budget_s - 5:
            break
            
        corr = corr_full(h)
        Cmax = np.max(corr)
        Kmax = np.where(np.abs(corr - Cmax) < 1e-12)[0]
        
        if len(Kmax) == 0:
            Kmax = [0]
            
        g = np.zeros(n)
        for k in Kmax:
            g += grad_shift(h, k)
        g /= len(Kmax)
        
        y = h + beta * (h - h_prev)
        h_new = y - step * g
        h_new = proj_sum_box(h_new, n/2)
        
        c5_new = max_overlap(h_new, dx)
        if c5_new < best_c5:
            step = min(step * 1.08, 0.2)
            best_h = h_new.copy()
            best_c5 = c5_new
            h_prev = h.copy()
            h = h_new.copy()
        else:
            step *= 0.5
            
        # Swap move every 250 steps
        if it % 250 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = np.random.uniform(-0.02, 0.02)
            h_new = h.copy()
            old_i, old_j = h_new[i], h_new[j]
            h_new[i] = np.clip(h_new[i] + delta, 0, 1)
            h_new[j] = np.clip(h_new[j] - delta, 0, 1)
            h_new = proj_sum_box(h_new, n/2)
            c5_new = max_overlap(h_new, dx)
            if c5_new < best_c5:
                h_prev = h.copy()
                h = h_new.copy()
                best_h = h_new.copy()
                best_c5 = c5_new
                
    return best_h, best_c5
