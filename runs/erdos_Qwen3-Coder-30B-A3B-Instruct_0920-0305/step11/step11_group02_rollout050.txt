import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 300
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

            # Find the shift with minimum correlation
            corr = compute_corr_fft(current_h)
            # Exclude the shift 0 (which gives max value that we want to minimize)
            corr_nonzero = np.concatenate([corr[n_points-1:], corr[:n_points-1]])
            min_idx = np.argmin(corr_nonzero)
            min_shift = min_idx - (n_points - 1)  # Convert back to actual shift
            
            # Compute gradient for this shift
            gradient = np.zeros_like(current_h)
            # Add gradient contribution from min_shift
            if min_shift > 0:
                # h[i] * h[i - min_shift] term contributes h[i - min_shift] to gradient[i]
                gradient[min_shift:] += current_h[:-min_shift]
                # h[i] * h[i + min_shift] term contributes h[i + min_shift] to gradient[i]
                gradient[:-min_shift] += current_h[min_shift:]
            elif min_shift < 0:
                # min_shift is negative, so h[i] * h[i + |min_shift|]
                abs_shift = -min_shift
                gradient[abs_shift:] += current_h[:-abs_shift]
                gradient[:-abs_shift] += current_h[abs_shift:]
            else:
                # min_shift = 0, skip this case
                pass

    return (best_h.tolist(), float(best_c5), n_points)
