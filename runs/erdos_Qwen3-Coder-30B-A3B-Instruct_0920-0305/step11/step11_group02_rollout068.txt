def compute_autocorrelation(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def compute_gradient_from_weights(h, weights):
    """Compute gradient via FFT-based convolution of weights and h"""
    L = 2 * len(h) - 1
    w_rev = weights[::-1]
    
    # Convolution with w (forward shift)
    conv_plus = np.fft.ifft(np.fft.fft(weights, L) * np.fft.fft(h, L)).real
    
    # Convolution with w_rev (backward shift)  
    conv_rev = np.fft.ifft(np.fft.fft(w_rev, L) * np.fft.fft(h, L)).real
    
    # The gradient is the sum of these convolutions (with opposite sign and normalization)
    grad = -(conv_plus + conv_rev) / target_sum
    
    # Return the central part corresponding to indices 0 to n-1
    return grad[:len(h)]

def project_to_sum_box(x, target_sum):

        return np.sum(projected) - target_sum

        b += 0.1

            b = m
        else:
