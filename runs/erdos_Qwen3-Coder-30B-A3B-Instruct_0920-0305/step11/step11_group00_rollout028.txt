def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def sigmoid(x):
    """Numerically stable sigmoid function"""
    return 1.0 / (1.0 + np.exp(-x))

def sample_knots(m=7):
    """Sample knots over [0,1]"""
    return np.linspace(0, 1, m, endpoint=False)

def cubic_spline_eval(knots, values, x_grid):
    """Evaluate cubic spline at x_grid points"""
    from scipy.interpolate import PchipInterpolator
    if len(knots) != len(values):
        raise ValueError("knots and values must have same length")
    if x_grid[-1] > knots[-1] or x_grid[0] < knots[0]:
        raise ValueError("x_grid is out of bounds")
    
    # Create piecewise cubic Hermite interpolator
    spl = PchipInterpolator(knots, values, extrapolate=False)
    return spl(x_grid)

def build_h_from_params(params, n_points):
    """Build h from spline parameters with symmetry"""
    # params contains m/2 = 3 values at knots [0, 0.25, 0.5]
    m = 7  # 7 knots
    knots = sample_knots(m)
    
    # Sigmoid transform to ensure [0,1]
    y_vals = sigmoid(params)
    
    # Generate symmetric values
    h_left = cubic_spline_eval(knots[:len(y_vals)], y_vals, np.linspace(0, 1, n_points//2 + 1))
    
    # Mirror to get full profile
    h_full = np.concatenate([h_left, h_left[-2::-1]])
    
    # Scale to have sum = n_points / 2
    s = np.sum(h_full)
    alpha = (n_points / 2) / s if s > 1e-10 else 1.0
    h_scaled = np.clip(alpha * h_full, 0, 1)
    
    return h_scaled

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def gradient_descent_polish(h, max_iter=1200):
    """Polish the solution with projected gradient descent"""
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    eta = 0.08
    for i in range(max_iter):
        if i % 200 == 0:
            # Periodic escape from plateaus
            idx1, idx2 = np.random.choice(len(h), 2, replace=False)
            delta = np.random.uniform(-0.015, 0.015)
            h[idx1] += delta
            h[idx2] -= delta
            h = project_to_sum_box(h, n_points / 2)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
            continue
            
        corr = compute_corr_fft(h)
        k_max = np.argmax(corr)
        # Compute gradient for the worst overlap
        g = np.zeros_like(h)
        # Simple gradient estimate for the specific shift k_max
        # Gradient depends on which indices are shifted
        shifts = list(range(-n_points+1, n_points))
        k_idx = np.argmax(corr)
        if k_idx < len(shifts):
            shift_k = shifts[k_idx]
        else:
            shift_k = 0
        for j in range(n_points):
            if j + shift_k < n_points and j + shift_k >= 0:
                g[j] += (1 - h[j + shift_k])
            if j - shift_k >= 0: 
                g[j] -= h[j - shift_k]
        
        # Normalize gradient
        norm = np.linalg.norm(g)
        if norm > 1e-10:
            g = g / norm
        
        # Take a step
        h_new = h - eta * g
        h_new = project_to_sum_box(h_new, n_points / 2)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
        else:
            # Adapt step size
            eta *= 0.5
            if eta < 1e-6:
                eta = 1e-6
                
    return best_h, best_c5
