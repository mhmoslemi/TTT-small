import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    """Try several different initial configurations"""
    best_c5 = float('inf')

        # Config 2: Four segments  
        [0.3]*30 + [0.6]*30 + [0.2]*30 + [0.8]*30,

        [0.8]*60 + [0.2]*60,
        # Config 4: Gaussian-like peak
        [0.1]*30 + [0.5]*60 + [0.1]*30,
        # Config 5: Single peak in center
        [0.0]*40 + [0.9]*40 + [0.0]*40

        h0 = np.array(config)
        h0 = project_to_sum_box(h0, TARGET_SUM)
        c5 = evaluate_c5(h0)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h0.copy()
    
    return best_h, best_c5

    best_h = h0.copy()

    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        # Try with very tight tolerance
        res = minimize(
            objective, 
            h0, 
            method='L-BFGS-B', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-12}
        )
        if res.success:
            optimized_h = res.x
            optimized_h = project_to_sum_box(optimized_h, TARGET_SUM)
            optimized_c5 = evaluate_c5(optimized_h)
            
            if optimized_c5 < best_c5:
                best_c5 = optimized_c5
                best_h = optimized_h.copy()
    except:
        pass
    
    # Try a few local search refinements at different scales
    if time.time() - start_time < budget_s * 0.8:
        for scale in [0.05, 0.02, 0.01]:
            current_h = best_h.copy()
            current_c5 = best_c5
            
            for _ in range(100):
                if time.time() - start_time >= budget_s:
                    break

                step = np.random.choice([-scale, scale])
                
                h_test = current_h.copy()
                h_test[idx] += step
                
                if h_test[idx] < 0 or h_test[idx] > 1:
                    continue
                
                h_test = project_to_sum_box(h_test, TARGET_SUM)
                test_c5 = evaluate_c5(h_test)
                
                if test_c5 < current_c5:
                    current_h = h_test.copy()
                    current_c5 = test_c5
                    if test_c5 < best_c5:
                        best_c5 = test_c5
                        best_h = current_h.copy()
    
    return best_h, best_c5
