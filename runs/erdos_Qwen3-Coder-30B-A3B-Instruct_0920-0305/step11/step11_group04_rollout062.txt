def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant using the official method"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation for all shifts
    corr = np.correlate(h, 1-h, mode="full")
    # The overlap integral C5 is the maximum of these correlations
    c5 = dx * np.max(corr)
    return c5
