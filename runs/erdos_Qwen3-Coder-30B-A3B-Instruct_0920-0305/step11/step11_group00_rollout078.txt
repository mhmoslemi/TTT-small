import numpy as np
from scipy.signal import correlate
from scipy.interpolate import CubicSpline
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
n_knots = 7  # odd number for symmetric spline
n_params = (n_knots + 1) // 2  # number of free parameters

def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-x))

def unsigmoid(y):
    return np.log(y / (1.0 - y))

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_step_function(params, n=120):
    """Evaluates a step function from parameters using a piecewise cubic spline"""
    # Parameters: values at knots (in [0,1] after sigmoid)
    z = params  # unconstrained
    y = sigmoid(z)  # constrained to [0,1]

    # Define knot positions at [0, 0.25, 0.5, 0.75, 1] on [0,1] 
    # This ensures key features (epicentre, edges) are controlled
    x_knots = np.linspace(0, 1, n_knots)
    x_knots = x_knots[1:-1]  # remove first and last (will be fixed by symmetry)
    x_knots = np.insert(x_knots, 0, 0)
    x_knots = np.append(x_knots, 1)
    
    # Create symmetric knots for [0,2]
    x_full = np.concatenate([x_knots, 2 - x_knots[1:-1][::-1]])
    y_full = np.concatenate([y, y[1:-1][::-1]])
    
    # Interpolate with cubic spline
    cs = CubicSpline(x_full, y_full, bc_type=((1, 0), (1, 0)))  # clamped cubic spline with zero derivatives at endpoints
    
    # Sample at uniform grid
    x_sample = np.linspace(0, 2, n)
    h = cs(x_sample)
    
    # Apply scaling to meet integral constraint
    s = np.sum(h) * dx
    alpha = TARGET_SUM / s
    h = h * alpha
    h = np.clip(h, 0, 1)
    
    # Final repulsion if needed to ensure exactly correct sum
    h = project_to_sum_box(h, TARGET_SUM)
    
    return h

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return compute_c5_from_corr(corr)

def cma_es_optimize():
    """Run CMA-ES optimization"""
    # Initialize parameters
    z = np.zeros(n_params)  # start with all flat
    sigma = 2.0  # initial step size
    C = np.eye(n_params)  # initial covariance matrix
    mue = n_params // 2  # number of parents
    
    # Parameters for CMA-ES
    mu = mue
    lamb = 4 + int(3 * np.log(n_params))
    chi_n = n_params**0.5 * (1 - 1/(4*n_params) + 1/(21*n_params**2))
    
    # Store best
    best_f = float('inf')
    best_z = None
    
    # Iterate for a reasonable number of generations
    for gen in range(100):
        # Sample
        samples = []
        for i in range(lamb):
            z_offspring = z + sigma * np.random.multivariate_normal(np.zeros(n_params), C)
            samples.append(z_offspring)
            
        # Evaluate
        fitnesses = []
        for samp in samples:
            h = evaluate_step_function(samp, n_points)
            f = evaluate_c5(h)
            fitnesses.append(f)
            
        # Rank selection
        indices = np.argsort(fitnesses)
        parents = np.array(samples)[indices[:mu]]
        fitnesses = np.array(fitnesses)[indices[:mu]]
        
        # Update distribution
        # Compute weighted mean of parents
        w = np.log(mu + 1) - np.log(np.arange(1, mu + 1))
        w = w / np.sum(w)
        z_mean = np.sum(w.reshape(-1, 1) * parents, axis=0)
        
        # Update mean
        z = z_mean
        
        # Update covariance matrix
        C = (1 - 0.1) * C + 0.1 * np.outer(z_mean - z, z_mean - z)
        
        # Update step size
        sigma = sigma * np.exp(0.3 * (np.linalg.norm(z_mean - z)/chi_n - 1))
        
        # Update best
        min_f = np.min(fitnesses)
        if min_f < best_f:
            best_f = min_f
            best_z = parents[np.argmin(fitnesses)].copy()
            
        # Check time limit
        if time.time() - start_time > budget_s - 10:
            break
    
    if best_z is not None:
        h = evaluate_step_function(best_z, n_points)
        return h, best_f
    else:
        return None, float('inf')

def polished_refinement(h_init, max_iter=1200):
    """Perform gradient-based polishing on full vector"""
    h = h_init.copy()
    current_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Find worst shifts
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        K_max = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        
        # Build gradient
        g = np.zeros_like(h)
        for k in K_max:
            if k == 0:
                continue  # g_k^i = 0 for k=0
            elif k > 0:
                # Forward difference
                g[k:] += (1 - h[:-k]) - h[k:]
            else:
                # Backward difference  
                k = -k
                g[:-k] += (1 - h[k:]) - h[:-k]
                
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            continue
            
        g = g / g_norm
        
        # Line search
        eta = 0.08
        for _ in range(5):
            h_new = h - eta * g
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            c5_new = evaluate_c5(h_new)
            
            if c5_new < current_c5:
                h = h_new
                current_c5 = c5_new
                break
            else:
                eta *= 0.5
                
        # Possibly rescue from plateaus
        if i % 200 == 0:
            # Small random swap
            i1, i2 = np.random.randint(0, n_points, 2)
            delta = np.random.uniform(-0.015, 0.015)
            h_test = h.copy()
            h_test[i1] += delta
            h_test[i2] -= delta
            h_test = project_to_sum_box(h_test, TARGET_SUM)
            c5_test = evaluate_c5(h_test)
            if c5_test < current_c5:
                h = h_test
                current_c5 = c5_test
                
    return h, current_c5
