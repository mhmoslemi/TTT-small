def project_onto_sum_box(v, target_sum):
    """Project vector v onto [0,1]^n with sum = target_sum using water-filling."""
    n = len(v)
    # Sort indices by value (descending)
    sorted_idx = np.argsort(-v)
    v_sorted = v[sorted_idx]
    
    # Compute cumulative sums
    cumsum = np.cumsum(v_sorted)
    
    # Find the threshold
    threshold = 0.0
    for i in range(n):
        if cumsum[i] <= target_sum:
            continue
        else:
            if i == 0:
                threshold = target_sum
            else:
                threshold = (target_sum - cumsum[i-1]) / (n - i)
            break
    else:
        threshold = target_sum / n

    # Clip and rescale
    result = np.clip(v, 0, 1)
    current_sum = np.sum(result)
    if current_sum > target_sum:
        # Reduce entries with largest values
        diff = current_sum - target_sum
        # Find indices to reduce
        idxs = sorted_idx[:n]  # sorted in descending order of values
        # only modify those that are above threshold (this may not be correct)
        # Actually, we should reduce more precisely
        
        # Let's do it properly
        # Determine how much to reduce each element
        v_scaled = np.array(v)
        v_sorted_idx = np.argsort(-v_scaled)
        v_sorted_vals = v_scaled[v_sorted_idx]
        total_excess = current_sum - target_sum
        
        # This is a better water-filling approach:
        # First ensure all values are at most 1.0
        v_clipped = np.clip(v_scaled, 0, 1)
        sum_clipped = np.sum(v_clipped)
        extra = sum_clipped - target_sum
        
        if extra > 0:
            # Reduce from largest values
            remaining = extra
            for i in range(n):
                idx = v_sorted_idx[i]
                if remaining <= 0:
                    break
                reduce_amount = min(v_clipped[idx], remaining)
                v_clipped[idx] -= reduce_amount
                remaining -= reduce_amount
        else:
            # We need to increase some values, but they shouldn't exceed 1
            # Add missing
            missing = target_sum - sum_clipped
            # Add uniformly
            v_clipped += missing / n
            # Clip to [0, 1]
            v_clipped = np.clip(v_clipped, 0, 1)
        
        # Sorting back
        result = np.empty_like(v_clipped)
        result[v_sorted_idx] = v_clipped
    elif current_sum < target_sum:
        # Increase some values
        missing = target_sum - current_sum
        # Increase from smallest values
        v_sorted_idx = np.argsort(v_clipped)
        v_sorted_vals = v_clipped[v_sorted_idx]
        remaining = missing
        for i in range(n):
            idx = v_sorted_idx[i]
            if remaining <= 0:

            increase = min(1.0 - v_clipped[idx], remaining)
            v_clipped[idx] += increase
            remaining -= increase
        # Sorting back
        result = np.empty_like(v_clipped)
        result[v_sorted_idx] = v_clipped
    else:
        # Already good
        result = v_clipped
    
    return result

def corr_fft(h):
    """FFT-based computation of correlation."""
    # This computes convolution of h with (1 - h), but offset in full mode.
    h_rev = h[::-1]
    b = 1 - h
    
    # Use full mode manual convolution with FFT
    L = 2 * len(h) - 1
    
    # Zero pad
    h_padded = np.zeros(L)
    b_padded = np.zeros(L)
    h_padded[:len(h)] = h_rev
    b_padded[:len(b)] = b
    
    # Compute FFT
    H = fft(h_padded)
    B = fft(b_padded)
    
    # Convolution
    C = H * np.conj(B)
    corr = ifft(C).real
    
    return corr

def compute_c5_from_corr(corr, dx):
    """Convert full correlation result to C5 = max_k int h(x) (1 - h(x+k)) dx"""
    # The correlation is C[s] = sum_i h_{-i} * (1 - h_{-i+s}) = sum_i h_i * h_{i+s}, for shift s.
    # So we want max_i sum_i h_i * h_{i+s}
    # In the non-shifted case, final conv gives shift 0,...,2n-1. The corresponding shifts are -n+1,...,n-1.
    # The -n+1th shift is corr[0], the -n+2th shift is corr[1], ..., 0th shift is corr[n-1]
    
    # Just return max of the values, and multiply by dx.
    return np.max(corr) * dx

def lp_step(y, target_sum, dx):
    """
    Solve linearized problem:
    maximize t
    subject to:
        sum_{i} x_i * y_{i+s} >= t for all shifts s
        sum x_i = target_sum
        0 <= x_i <= 1
    """
    n = len(y)
    # Create variables
    x = cp.Variable(n)
    t = cp.Variable()
    
    # Objective
    objective = cp.Maximize(t)
    
    # Constraints
    constraints = []
    
    # Add correlation constraints
    L = 2 * n - 1
    # proc = CorrelationProcessor(n)
    # Calculate correlation manually with shifts
    
    # Full convolution of reversed x with (1-y)
    # But that's actually conv (x, 1-y)
    # Use zero-padding
    
    # Prepare shifted indices, for each shift s in [-n+1, ..., n-1]
    # For shift s, index i in x corresponds to index i+s in 1-y if valid
    
    # In the sliding window way
    # For shift s, the sum is sum_{i, i+s in [0,n-1]} x_i * y_{i+s}
    
    # extract valid indices
    for s in range(-(n-1), n):
        # Build constraint sum_{i, i+s in [0,n-1]} ... >= t
        constraint_terms = []
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                constraint_terms.append(x[i] * y[j])
        
        # If we have terms, add constraint
        if constraint_terms:
            constraints.append(cp.sum(constraint_terms) >= t)
    
    # Sum constraint
    constraints.append(cp.sum(x) == target_sum)
    
    # Box constraints
    for i in range(n):
        constraints.append(x[i] >= 0)
        constraints.append(x[i] <= 1)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000, abstol=1e-8, reltol=1e-8)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            x_val = x.value
            t_val = t.value
        else:
            # Fallback to some small t
            x_val = np.ones(n) * 0.5
            t_val = 0.0
    except Exception:
        # Fallback to simple projection
        x_val = np.ones(n) * 0.5
        t_val = 0.0
    
    return x_val, t_val
