import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 128
dx = 2.0 / n_points

def thue_morse_binary(n):
    """Generate Thue-Morse binary sequence of length n with exactly n/2 ones."""
    # For power of 2: n = 128, compute Thue-Morse up to 128 terms
    tm = [0]
    for i in range(7):  # Since 2^7 = 128
        tm = tm + [1 - x for x in tm]
    tm = tm[:n]
    # Ensure exactly half ones
    ones_count = sum(tm)
    if ones_count > n//2:
        # Turn some 1s to 0s
        zero_indices = [i for i in range(n) if tm[i] == 1]
        for i in range(ones_count - n//2):
            tm[zero_indices[i]] = 0
    elif ones_count < n//2:
        # Turn some 0s to 1s
        one_indices = [i for i in range(n) if tm[i] == 0]
        for i in range(n//2 - ones_count):
            tm[one_indices[i]] = 1
    return np.array(tm, dtype=float)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    # Water-filling algorithm
    
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def autocorr(h):
    """Compute full linear autocorrelation using FFT."""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr * dx

def compute_C5(h):
    """Compute C5 bound from h."""
    corr = autocorr(h)
    return 1.0 - np.min(corr)

def greedy_swap_local_search(h, dx, budget_s, start_time):
    """Perform a greedy local search to improve based on worst shift."""
    # Compute correlation
    corr = autocorr(h)
    # Find the worst shift
    k_star = np.argmin(corr)
    # For k=0, corr_idx = n-1
    shift = k_star - (n_points - 1)
    
    # Local swap search
    h = h.copy()
    max_iter = 1000
    iter_count = 0
    while iter_count < max_iter:
        iter_count += 1
        if time.time() - start_time > budget_s * 0.95:
            break
        best_net = -1e10
        best_i0 = None
        best_i1 = None
        for i0 in range(n_points):
            if h[i0] != 0:
                continue
            gain = 0.0
            if shift != 0:
                if i0 + shift >= 0 and i0 + shift < n_points:
                    gain += h[i0 + shift]
            if i0 - shift >= 0 and i0 - shift < n_points:
                gain += h[i0 - shift]
            
            for i1 in range(n_points):
                if h[i1] != 1:
                    continue
                loss = 0.0
                if shift != 0:
                    if i1 + shift >= 0 and i1 + shift < n_points:
                        loss += h[i1 + shift]
                if i1 - shift >= 0 and i1 - shift < n_points:
                    loss += h[i1 - shift]
                
                net = gain - loss
                if net > best_net:
                    best_net = net
                    best_i0 = i0
                    best_i1 = i1
                
        if best_net <= 0:
            break
            
        # Update h
        h[best_i0] = 1.0
        h[best_i1] = 0.0
        
        # Update correlation manually for affected terms
        corr = autocorr(h)
        k_star = np.argmin(corr)
        shift = k_star - (n_points - 1)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points // 2
    best_c5 = float('inf')
    best_h = None
    
    # Helper functions
    def compute_corr(h):
        h_rev = h[::-1]
        L = 2 * len(h) - 1
        corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
        return corr * dx
    
    def compute_C5(h):
        corr = compute_corr(h)
        return 1.0 - np.min(corr)
    
    # Seeds configuration
    seeds = []
    # 1) Thue-Morse binary
    seeds.append(thue_morse_binary(n_points))
    
    # 2) Random uniform start
    r = np.random.rand(n_points)
    seeds.append(project_to_sum_box(r, TARGET_SUM))
    
    # 3) Provided initial_h_values
    if 'initial_h_values' in globals() and len(initial_h_values) >= n_points:
        # Interpolate if needed
        r = np.array(initial_h_values, dtype=float)
        if len(r) != n_points:
            # For now just truncate if long or pad with 0 if short
            if len(r) > n_points:
                r = r[:n_points]
            else:
                r = np.pad(r, (0, n_points - len(r)), constant_values=0)
        seeds.append(project_to_sum_box(r, TARGET_SUM))
    else:
        # Fallback same as uniform
        r = np.random.rand(n_points)
        seeds.append(project_to_sum_box(r, TARGET_SUM))
    
    # Optimization loop
    for h0 in seeds:
        if time.time() - start_time > budget_s * 0.95:
            break
        
        h = h0.copy()
        tau = 0.2
        alpha = 0.15
        
        for it in range(2000):
            if time.time() - start_time > budget_s * 0.95:
                break
            
            corr = compute_corr(h)
            c5 = 1.0 - np.min(corr)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Compute gradient of smooth min
            w = np.exp(-corr / tau)
            w /= w.sum()
            g = np.zeros_like(h)
            
            # Compute gradient efficiently
            for shift in range(-(n_points - 1), n_points):
                if shift == 0:
                    # Handle shift 0 - include h_i * 2
                    k = shift + n_points - 1
                    wk = w[k]
                    if wk == 0.0:
                        continue
                    g += wk * 2.0 * h
                else:
                    k = shift + n_points - 1
                    wk = w[k]
                    if wk == 0.0:
                        continue
                    if shift > 0:
                        lo = 0
                        hi = n_points - shift
                        g[lo:hi] += wk * h[lo+shift:hi+shift]
                        g[lo+shift:hi+shift] += wk * h[lo:hi]
                    else:
                        lo = -shift
                        hi = n_points
                        g[lo:hi] += wk * h[lo+shift:hi+shift]
                        g[lo+shift:hi+shift] += wk * h[lo:hi]
            
            g *= dx
            
            # Ascent step + projection
            h_candidate = project_to_sum_box(h + alpha * g, TARGET_SUM)
            
            # Accept or reject
            corr_new = compute_corr(h_candidate)
            c5_new = 1.0 - np.min(corr_new)
            if c5_new < c5:
                h = h_candidate
                alpha = min(alpha * 1.02, 0.5)
            else:
                alpha *= 0.5
                
            tau *= 0.99
            
            # Early convergence check
            if it > 200 and abs(c5 - c5_new) < 1e-7:
                break
    
    # Post-processing with binary swap search if we have a solution
    if best_h is not None:
        # Binarise by keeping the largest target_sum entries as 1
        idx = np.argsort(-best_h)[:TARGET_SUM]
        h_bin = np.zeros(n_points)
        h_bin[idx] = 1.0
        # Perform greedy swap local search
        h_bin = greedy_swap_local_search(h_bin, dx, budget_s, start_time)
        c5_bin = compute_C5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
