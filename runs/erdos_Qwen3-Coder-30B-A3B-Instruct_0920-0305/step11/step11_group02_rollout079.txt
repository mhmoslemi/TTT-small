import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import minimize
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_autocorr(h):
    """Compute autocorrelation using FFT"""
    n = len(h)
    L = 2 * n - 1
    h_rev = h[::-1]
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def gradient_from_weights(h, w):
    """Compute gradient from soft-max weights using FFT"""
    n = len(h)
    L = 2 * n - 1
    
    # w_rev = w[::-1] 
    w_rev = np.concatenate([w[n-1:], w[:n-1]])
    
    # conv_plus = np.convolve(w, h, mode='full')[:n]
    # conv_rev = np.convolve(w_rev, h, mode='full')[:n]
    
    # Use FFT for convolution
    conv_plus = np.fft.ifft(np.fft.fft(w, L) * np.fft.fft(h, L)).real[:n]
    conv_rev = np.fft.ifft(np.fft.fft(w_rev, L) * np.fft.fft(h, L)).real[:n]
    
    # Gradient is negative sum of shifted contributions
    grad = -(conv_plus + conv_rev) / len(h)
    return grad

def compute_c5(h):
    """Compute C5 value"""
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2
    
    # Compute autocorrelation
    O = compute_autocorr(h)
    
    # Compute g_j = S - O_j
    g = target_sum - O
    
    # C5 is dx * max(g)
    return dx * np.max(g)

    np.random.seed(seed)

    # Parameters
    n = 300
    dx = 2.0 / n
    target_sum = n / 2
    
    # Initialize
    best_c5 = float('inf')
    best_h = None
    
    # Two initializations

    h0_upsampled = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h0)), h0)
    h0_projected = project_to_sum_box(h0_upsampled, target_sum)
    initial_solutions.append(h0_projected)
    
    # 2. Random binary start
    h_random = np.zeros(n)
    ones_indices = np.random.choice(n, size=int(target_sum), replace=False)
    h_random[ones_indices] = 1.0
    h_random = project_to_sum_box(h_random, target_sum)
    initial_solutions.append(h_random)
    
    # Process all initial solutions
    for h_start in initial_solutions:
        h = h_start.copy()
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Projected gradient descent with smooth max
        alpha = 1.0
        eta = 0.05
        max_iters = 10000
        stagnation_count = 0
        last_improvement = 0
        
        for iteration in range(max_iters):
            if time.time() - start_time > budget_s * 0.98:
                break
                
            # Compute autocorrelation
            O = compute_autocorr(h)
            
            # Compute g_j = S - O_j
            g = target_sum - O
            
            # Smooth max surrogate: F_α = (1/α)·log( Σ exp(α·g_j) )
            # max_g = np.max(g)
            # shift = g - max_g  # for numerical stability
            # w_raw = np.exp(alpha * shift)
            # w = w_raw / np.sum(w_raw)
            
            # More stable version:
            g_max = np.max(g)
            g_shifted = g - g_max
            exp_g = np.exp(alpha * g_shifted)
            w = exp_g / np.sum(exp_g)
            
            # Compute gradient ∇F_α
            grad = gradient_from_weights(h, w)
            
            # Projected gradient step
            y = h - eta * grad
            h = project_to_sum_box(y, target_sum)
            
            # Evaluate true objective
            c5_now = compute_c5(h)
            
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
                last_improvement = iteration
                stagnation_count = 0
            else:
                stagnation_count += 1
            
            # Adaptive learning rate and smoothing
            if iteration % 200 == 0:
                alpha = min(alpha * 1.3, 100.0)
                eta = eta * 0.9
                
            # Occasionally shake-up to escape flat regions
            if iteration % 500 == 0 and stagnation_count > 200:
                h = h + 0.02 * np.random.randn(n)
                h = project_to_sum_box(np.clip(h, 0, 1), target_sum)
                stagnation_count = 0
        
        # Final polishing with SLSQP
        def obj_full(x):
            O = compute_autocorr(x)
            return dx * np.max(target_sum - O)
            
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        bounds = [(0.0, 1.0)] * n
        
        res = minimize(obj_full, 
                      best_h,
                      method='SLSQP',
                      bounds=bounds,
                      constraints=cons,
                      options={'maxiter': 200, 'ftol': 1e-12})
        
        if res.success and obj_full(res.x) < best_c5:
            best_h = res.x
            best_c5 = obj_full(res.x)
    
    if best_h is None:
        # Fallback to initial solution
        best_h = project_to_sum_box(np.array(initial_h_values), target_sum)
        best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n)
