import numpy as np
from scipy.signal import correlate
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    # Binary search
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def grad_shift(h, k):
    """Compute gradient of shift k"""
    n = len(h)
    pad = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    base = np.arange(n) + (n-1)
    
    grad_k = (1 - pad[base + k]) - pad[base - k]
    return grad_k

def lse_surrogate(h, alpha):
    """Compute log-sum-exp surrogate"""
    c = compute_corr(h)
    max_c = np.max(c)
    exp_vals = np.exp(alpha * (c - max_c))
    log_sum_exp = max_c + np.log(np.sum(exp_vals)) / alpha
    return log_sum_exp

def lse_gradient(h, alpha):
    """Compute gradient of log-sum-exp surrogate"""
    c = compute_corr(h)
    max_c = np.max(c)
    exp_vals = np.exp(alpha * (c - max_c))
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros_like(h)
    for k in range(-len(h)+1, len(h)):
        grad_k = grad_shift(h, k)
        grad += weights[k + len(h) - 1] * grad_k
    return grad

def accelerated_projected_gradient(h0, alpha, steps, eta=0.1, beta=0.9):
    """Accelerated projected gradient with backtracking line search"""
    h = h0.copy()
    h_prev = h0.copy()
    
    for _ in range(steps):
        # Compute gradient
        grad = lse_gradient(h, alpha)
        
        # Nesterov acceleration
        y = h + beta * (h - h_prev)
        h_tmp = y - eta * grad
        h_new = project_to_sum_box(h_tmp, len(h) / 2)
        
        # Backtracking line search
        while lse_surrogate(h_new, alpha) > lse_surrogate(h, alpha) - 0.5 * eta * np.dot(grad, h - h_new):
            eta *= 0.5
            h_tmp = y - eta * grad
            h_new = project_to_sum_box(h_tmp, len(h) / 2)
            if eta < 1e-10:
                break
                
        h_prev = h.copy()
        h = h_new.copy()
        
        # Adaptive learning rate
        if lse_surrogate(h_new, alpha) < lse_surrogate(h, alpha):
            eta = min(eta * 1.1, 1.0)
        else:
            eta = max(eta * 0.5, 1e-6)
            
    return h

def subgradient_polishing(h0, max_steps=1500):
    """Exact sub-gradient polishing"""
    h = h0.copy()
    eta = 0.08
    
    for _ in range(max_steps):
        c = compute_corr(h)
        max_val = np.max(c)
        K_max = np.where(c >= max_val - 1e-12)[0]
        
        if len(K_max) == 0:
            break
            
        # Compute average gradient
        avg_grad = np.zeros_like(h)
        for k in K_max:
            avg_grad += grad_shift(h, k - len(h) + 1)
        avg_grad /= len(K_max)
        
        # Gradient step
        h_new = h - eta * avg_grad
        h_new = project_to_sum_box(h_new, len(h) / 2)
        
        # Accept or reject step
        c_new = compute_corr(h_new)
        new_max = np.max(c_new)
        
        if new_max < np.max(c) - 1e-12:
            h = h_new.copy()
            eta = min(eta * 1.05, 0.15)
        else:
            eta *= 0.5
            
    return h

def random_perturbation_escape(h, threshold=0.02):
    """Small random swap-move to escape local minima"""
    h_new = h.copy()
    i, j = np.random.choice(len(h), 2, replace=False)
    delta = np.random.uniform(-threshold, threshold)
    h_new[i] -= delta
    h_new[j] += delta
    h_new = project_to_sum_box(h_new, len(h) / 2)
    
    c = compute_corr(h)
    c_new = compute_corr(h_new)
    if np.max(c_new) < np.max(c) - 1e-12:
        return h_new
    return h
