import numpy as np
from scipy.fft import fft, ifft
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, multiply
import time

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def project_onto_sum_box(v, target_sum):
    """Project vector v onto box constraints [0,1] with sum = target_sum"""
    # Sort values in descending order to compute water-filling
    sorted_indices = np.argsort(-v)
    sorted_vals = v[sorted_indices]
    
    # Find the threshold that makes the sum equal to target_sum
    cum_sum = np.cumsum(sorted_vals)
    n = len(sorted_vals)
    
    # Find where sum remains below target
    threshold_idx = 0
    for i in range(n):
        if cum_sum[i] < target_sum:
            threshold_idx = i + 1
        else:
            break
    
    # All elements with idx < threshold_idx get 1.0
    # Remaining elements get (target_sum - cum_sum[threshold_idx-1]) / (n - threshold_idx)
    if threshold_idx >= n:
        # All elements are at 1.0, but need to adjust
        threshold = target_sum / n
        projected = np.clip(v, 0, threshold)
        # Adjust the sum
        diff = target_sum - np.sum(projected)
        if diff > 0:
            # Distribute diff equally
            increase = diff / n
            projected += increase
    elif threshold_idx == 0:
        # All elements are below the threshold
        threshold = target_sum / n
        projected = np.clip(v, 0, threshold)
    else:
        # Elements [0, threshold_idx-1] are 1.0, others are at some threshold
        threshold = (target_sum - cum_sum[threshold_idx-1]) / (n - threshold_idx)
        projected = np.zeros_like(v)
        projected[sorted_indices[:threshold_idx]] = 1.0
        projected[sorted_indices[threshold_idx:]] = threshold
    
    return projected

def corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = corr_fft(h)
    return dx * np.max(corr)

def preprocess_seed(h):
    """Apply preprocessing to ensure feasibility"""
    h = np.clip(h, 0, 1)
    h = project_onto_sum_box(h, TARGET_SUM)
    return h

def lp_step(y):
    """
    Solve LP: max t s.t.
    sum_i x_i * y_{i+s} >= t for all s
    sum_i x_i = TARGET_SUM
    0 <= x_i <= 1
    Return (x, t)
    """
    # Create variables
    x = Variable(n_points)
    t = Variable()
    
    # Create constraints
    constraints = [x >= 0, x <= 1, cp_sum(x) == TARGET_SUM, t <= 0]
    
    # Add correlation constraints
    for s in range(-n_points + 1, n_points):
        if s >= 0:
            # Shift: y[i] -> y[i+s]
            corr_term = cp_sum(multiply(x, y[s:]))
        else:
            # Shift: y[i] -> y[i+s] where i+s < 0
            corr_term = cp_sum(multiply(x[:s], y[-s:]))
        
        constraints.append(corr_term >= t)
    
    # Solve problem
    prob = Problem(Minimize(-t), constraints)
    try:
        prob.solve(solver='ECOS', max_iters=1000)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            x_opt = np.array(x.value).flatten()
            t_opt = -prob.objective.value
            return x_opt, t_opt
        else:
            # Fall back to solution with t = -1000 if not optimal
            return y.copy(), -1000
    except:
        return y.copy(), -1000

def alternating_lp(h0, timeout):
    """Run alternating LP iterations"""
    start_time = time.time()
    h = h0.copy()
    
    t_old = -np.inf
    while time.time() - start_time < timeout / 2:
        # Step 1: Solve LP with frozen y = h, get x and t
        x, t = lp_step(h)
        
        if abs(t - t_old) < 1e-8 or t <= t_old + 1e-10:
            break
        t_old = t
        
        # Step 2: Update h = x
        h = x.copy()
        
        # Step 3: Enforce symmetry if beneficial
        h_sym = (h + h[::-1]) / 2
        h = project_onto_sum_box(h_sym, TARGET_SUM)
    
    return h, t_old

# Optimized pairwise exchange local refinement
def refine_solution(h, max_iter=2000):
    """Run a short BlackBox optimization"""
    corr_values = corr_fft(h)
    c5_current = dx * np.max(corr_values)
    
    for _ in range(max_iter):
        # Select two random indices
        i = np.random.randint(0, len(h))
        j = np.random.randint(0, len(h))
        if i == j:
            continue
            
        # Get shift value between indices
        s = j - i
        delta = np.random.uniform(0, 0.01)
        
        # Attempt move
        h_trial = h.copy()
        h_trial[i] += delta
        h_trial[j] -= delta
        
        # Ensure bounds
        if h_trial[i] < 0 or h_trial[i] > 1 or h_trial[j] < 0 or h_trial[j] > 1:
            continue
            
        # Re-project to constraint set
        h_trial = project_onto_sum_box(h_trial, TARGET_SUM)
        
        # Evaluate new c5
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < c5_current:
            h = h_trial
            c5_current = c5_trial
        elif np.random.rand() < np.exp(-(c5_trial - c5_current) * 0.1):
            h = h_trial
            c5_current = c5_trial
            
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = np.inf
    best_h = None
    
    # Seeds to try
    initial_seeds = []
    
    # Use provided initial_h_values if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        initial_seeds.append(np.array(initial_h_values))
    
    # Add random starts
    for _ in range(3):
        initial_seeds.append(np.random.rand(n_points))
    
    # Add a tent-like construction
    tent = np.zeros(n_points)
    for i in range(n_points):
        tent[i] = 1.0 - 2.0 * abs((i * dx) - 1.0)
    tent = np.clip(tent, 0, 1)
    initial_seeds.append(tent)
    
    # Run restart loop
    max_restarts = 8
    for ri, h_seed in enumerate(initial_seeds[:max_restarts]):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Preprocess seed
        h = preprocess_seed(h_seed)
        
        # Run alternating LP
        h_final, t = alternating_lp(h, (budget_s * 0.7) / max_restarts)
        c5 = 1 - dx * t
        
        # Refine the solution with local search
        h_refined = refine_solution(h_final, 1000)
        c5_refined = evaluate_c5(h_refined)
        
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    
    # Final check
    if best_h is not None:
        final_c5 = evaluate_c5(best_h)
        best_c5 = min(best_c5, final_c5)
        best_h = project_onto_sum_box(best_h, TARGET_SUM)
    else:
        # Return safe fallback
        best_h = np.array(initial_h_values[:n_points]) if 'initial_h_values' in globals() and initial_h_values is not None else np.ones(n_points) * 0.5
        best_h = project_onto_sum_box(best_h, TARGET_SUM)
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
