import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_value(h):
    """Compute C5 value"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def triangular_initial(n):
    """Create triangular initial shape"""
    m = n // 2
    # linear decreasing from 1 to 0 over first half
    first = np.linspace(1, 0, m, endpoint=False)
    if n % 2 == 0:
        h = np.concatenate([first, first[::-1]])
    else:
        h = np.concatenate([first, first[-2::-1]])
    return h

def random_feasible_vector(n, target_sum):
    """Generate random feasible vector"""
    h = np.random.random(n)
    h = project_to_sum_box(h, target_sum)
    return h

def grad_shift(h, s):
    """Compute gradient for shift s"""
    n = len(h)
    base = np.arange(n) + (n-1)
    padded = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    term1 = 1 - padded[base + s]
    term2 = padded[base - s]
    return term1 - term2

def smooth_optimize(h0, target_sum, dx, time_limit, start_time):
    """Optimize using smooth approximation"""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_val = c5_value(h)
    
    # Parameters
    alphas = np.linspace(0.5, 30, 15)
    step = 0.5
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        for iter in range(200):
            if time.time() - start_time > time_limit:
                break
            c = compute_corr(h)
            max_c = np.max(c)
            expvals = np.exp(alpha * (c - max_c))
            w = expvals / expvals.sum()
            
            grad = np.zeros_like(h)
            for idx_shift, weight in enumerate(w):
                if weight == 0:
                    continue
                s = idx_shift - (n-1)
                grad += weight * grad_shift(h, s)
            
            step_cur = step
            h_new = h - step_cur * grad
            h_new = project_to_sum_box(h_new, target_sum)
            val_new = c5_value(h_new)
            
            if val_new < best_val - 1e-12:
                h = h_new
                best_val = val_new
                best_h = h.copy()
            else:
                step_cur *= 0.5
                h_new = h - step_cur * grad
                h_new = project_to_sum_box(h_new, target_sum)
                val_new = c5_value(h_new)
                if val_new < best_val - 1e-12:
                    h = h_new
                    best_val = val_new
                    best_h = h.copy()
    
    return best_h, best_val

def subgradient_optimize(h0, target_sum, dx, time_limit, start_time):
    """Optimize using subgradient method"""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_val = c5_value(h)
    
    step_size = 0.1
    max_iter = 500
    
    for iter in range(max_iter):
        if time.time() - start_time > time_limit:
            break
            
        c = compute_corr(h)
        c_max = np.max(c)
        indices = np.where(np.abs(c - c_max) < 1e-8)[0]
        
        grad = np.zeros_like(h)
        for idx in indices:
            s = idx - (n-1)
            grad += grad_shift(h, s)
        
        grad = grad / len(indices)
        h_new = h - step_size * grad
        h_new = project_to_sum_box(h_new, target_sum)
        val_new = c5_value(h_new)
        
        if val_new < best_val - 1e-12:
            h = h_new
            best_val = val_new
            best_h = h.copy()
        else:
            step_size *= 0.9
        
        if iter % 50 == 0:
            step_size = max(step_size * 0.95, 1e-6)
    
    return best_h, best_val

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    # Try different resolutions and restarts
    res_list = [120, 200]
    
    for n in res_list:
        if time.time() - start_time > budget_s * 0.95:
            break
        dx = 2.0 / n
        TARGET_SUM = n / 2
        
        # Try different initializations
        for init_type in ['triangular', 'random']:
            if time.time() - start_time > budget_s * 0.95:
                break
                
            if init_type == 'triangular':
                h0 = triangular_initial(n)
            else:
                h0 = random_feasible_vector(n, TARGET_SUM)
            
            # Smooth optimization
            h_smooth, val_smooth = smooth_optimize(h0, TARGET_SUM, dx, budget_s*0.8, start_time)
            if val_smooth < best_c5:
                best_h, best_c5, best_n = h_smooth, val_smooth, n
            
            # Subgradient optimization 
            h_sub, val_sub = subgradient_optimize(h_smooth, TARGET_SUM, dx, budget_s*0.95, start_time)
            if val_sub < best_c5:
                best_h, best_c5, best_n = h_sub, val_sub, n
    
    # Final refinement
    if best_h is not None:
        h_refined, val_refined = subgradient_optimize(best_h, TARGET_SUM, dx, budget_s*0.95, start_time)
        if val_refined < best_c5:
            best_h, best_c5, best_n = h_refined, val_refined, best_n
    
    return (best_h.tolist(), float(best_c5), int(best_n))
