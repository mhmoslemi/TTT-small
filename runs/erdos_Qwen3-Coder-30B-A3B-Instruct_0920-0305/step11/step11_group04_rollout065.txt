# Global variables
n_points = 128
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_shift_matrices():
    """Compute shift matrices for all possible shifts"""
    shift_matrices = {}
    # Create shift matrices for each shift k from -(n-1) to +(n-1)
    for k in range(-(n_points-1), n_points):
        # Create a sparse diagonal matrix for shift k
        offsets = [k]
        data = [1.0]
        shift_matrices[k] = diags(data, offsets, shape=(n_points, n_points), format='csr')
    return shift_matrices

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_sdp(shift_matrices):
    """Solve SDP relaxation for the Erdos problem"""
    # We will formulate SDP: minimize t subject to:
    # X >> 0, diag(X) = 1, trace(J @ X) = 0, trace(S_k @ X) >= 2n*(1-t) - n_k for all shifts k.
    
    # Matrix J is all ones
    J = np.ones((n_points, n_points))
    
    # Create variable X and t
    X = cp.Variable((n_points, n_points), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        X >> 0,  # X is positive semidefinite
        cp.diag(X) == 1,  # diag(X) = 1
        cp.trace(J @ X) == 0  # Balanced constraint
    ]
    
    # Add shift constraints
    for k in range(-(n_points-1), n_points):
        if k in shift_matrices:
            n_k = n_points - abs(k)
            # Constraint: trace(S_k @ X) >= 2n*(1 - t) - n_k
            # Since S_k is only non-zero on diagonal k, the constraint becomes:
            # sum of shifted elements of X >= 2n*(1 - t) - n_k
            # But trace(S_k @ X) = sum_{i,j} S_k[i,j] * X[i,j] = sum_{i} X[i, i+k]
            constraints.append(cp.trace(shift_matrices[k] @ X) >= 2*n_points*(1-t) - n_k)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=1500, eps=1e-4)
        if prob.status == cp.OPTIMAL:
            return X.value, t.value
    except:
        return None, None
    return None, None

def round_solution(X, R=100):

        # Keep positive eigenvalues only
        positive_indices = eigenvals > 1e-8
        V = eigenvecs[:, positive_indices]
        Lambda = np.diag(eigenvals[positive_indices])
        
        # Compute L = V * sqrt(Lambda)
        L = V @ np.sqrt(Lambda)
        
        best_h = None
        best_c5 = float('inf')
        
        for _ in range(R):
            # Generate random vector g
            g = np.random.randn(L.shape[1])
            
            # Compute signs
            x = np.sign(L.T @ g)
            
            # Transform to [0,1]
            h = (x + 1) / 2
            
            # Ensure correct sum
            current_sum = np.sum(h)
            if current_sum > TARGET_SUM:
                # Need to flip some 1s to 0s
                sorted_indices = np.argsort(np.abs(h - 0.5))
                for i in sorted_indices:
                    if current_sum <= TARGET_SUM:
                        break
                    if h[i] > 0.5:
                        h[i] = 0
                        current_sum -= 1
            elif current_sum < TARGET_SUM:
                # Need to flip some 0s to 1s
                sorted_indices = np.argsort(np.abs(h - 0.5))
                for i in sorted_indices:
                    if current_sum >= TARGET_SUM:
                        break
                    if h[i] < 0.5:
                        h[i] = 1
                        current_sum += 1
            
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        return best_h, best_c5
    except:
        return None, None

    current_c5 = evaluate_c5(current_h)
    
    for _ in range(max_iter):
        # Identify shift that gives current maximum
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        max_corr = corr[k_max]
        
        # Try different swaps
        improved = False
        for i in range(len(current_h)):
            # Try making index i equal 1
            if current_h[i] < 1:
                h_trial = current_h.copy()
                h_trial[i] = 1
                h_trial = project_to_sum_box(h_trial, TARGET_SUM)
                c5_trial = evaluate_c5(h_trial)
                
                if c5_trial < current_c5:
                    current_h = h_trial
                    current_c5 = c5_trial
                    improved = True
                    break
        
        if not improved:
            break
    
    return current_h, current_c5
