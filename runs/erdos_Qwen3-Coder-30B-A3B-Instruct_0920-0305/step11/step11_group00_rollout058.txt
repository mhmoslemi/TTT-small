# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-x))

def inverse_sigmoid(y):
    return np.log(y / (1.0 - y))

    from scipy.interpolate import CubicSpline
    # 1D linear space from 0 to 1
    x = np.linspace(0, 1, len(knots))
    # Use a clamped cubic spline (natural boundary conditions)
    cs = CubicSpline(x, knot_values, bc_type='clamped')
    # Evaluate on uniform grid [0, 2]
    x_eval = np.linspace(0, 2, n_points)
    y_eval = cs(x_eval)
    # Ensure it is symmetric
    y_eval_sym = (y_eval + np.flip(y_eval))/2
    # Return positive values only (since we are dealing with step function)
    return np.clip(y_eval_sym, 0, 1)

    x = np.linspace(0, 1, 4)
    x_full = np.linspace(0, 2, n_points)
    
    # Interpolate with the knot values for the first half
    # and then mirror to get full symmetric function
    cs = CubicSpline(x[:4], knot_values[:4], bc_type='clamped')
    h_half = cs(x_full[:n_points//2+1])
    h_full = np.concatenate([h_half, np.flip(h_half[:-1])])
    
    # Normalize so sum is target_sum
    s = np.sum(h_full)
    alpha = target_sum / s
    h_final = np.clip(alpha * h_full, 0, 1)
    
    # Project to sum
    return project_to_sum_box(h_final, target_sum)

def compute_gradient(h, k):
    """Compute gradient ∇c_k(i) = (1-h(i+k) - h(i-k)) for k shift"""
    n = len(h)
    g = np.zeros_like(h)
    for i in range(n):
        j1 = i + k
        j2 = i - k
        if j1 < n:
            g[i] += (1 - h[j1])
        if j2 >= 0:
            g[i] += (-h[j2])
    return g
