def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def lp_step(y, target_sum):
    """Solve LP to maximize t subject to constraints"""
    n = len(y)
    x = cp.Variable(n)
    t = cp.Variable()
    
    # Constraints
    constraints = [x >= 0, x <= 1, cp.sum(x) == target_sum]
    
    # Build weight vectors for each shift
    for s in range(-n+1, n):
        w = np.zeros(n)
        if s >= 0:
            w[:n-s] = y[s:]
        else:
            w[-s:] = y[:n+s]
        constraints.append(w @ x >= t)
    
    # Solve LP
    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iter=1000)
    except:
        try:
            prob.solve(solver=cp.SCS, max_iters=2000)
        except:
            # Fallback to simpler approach
            return y, 0
    
    if prob.status in ["optimal", "optimal_inaccurate"]:
        x_opt = np.clip(x.value, 0, 1)
        t_opt = t.value
        return x_opt, t_opt
    else:
        return y, 0

def alternating_lp(y0, target_sum, max_outer_iter=20, time_limit=None):

    y = y0.copy()
    best_h = y.copy()
    best_c5 = evaluate_c5(y)
    start = time.time()
    
    for outer in range(max_outer_iter):
        if time_limit is not None and time.time() - start > time_limit:
            break
            
        x, t_opt = lp_step(y, target_sum)
        
        # Evaluate C5
        c5 = evaluate_c5(x)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()
            
        # Check convergence
        if np.linalg.norm(x - y) < 1e-8:
            break
        y = x
    
    return best_h, best_c5

        t = max(0.001, 0.02 * (1 - (time.time() - start) / time_limit))

            continue
            
        # Propose small delta
        max_delta = min(h[i], 1 - h[j])  
        delta = np.random.uniform(-max_delta*0.2, max_delta*0.2)
        
        # Adjust values
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        
        # Evaluate new C5
        new_c5 = evaluate_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()

            prob = np.exp(-(new_c5 - cur_c5) / (t + 1e-9))
            if np.random.rand() < prob:
                h = new_h
                cur_c5 = new_c5
    
    return best_h, best_c5
