import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate
from scipy.sparse import csr_matrix

def proj_sum_box(v, target):
    """Project v onto box constraints [0,1] with sum = target using binary search"""
    v = np.asarray(v)
    low, high = -np.max(v) - 1, 1 - np.min(v) + 1
    for _ in range(30):
        lam = (low + high) / 2
        proj = np.clip(v + lam, 0, 1)
        s = np.sum(proj)
        if s < target:
            high = lam
        else:
            low = lam
    return np.clip(v + (low + high) / 2, 0, 1)

def corr_full(h):
    """FFT-based full correlation"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def max_overlap(h, dx):
    """Compute maximum overlap from correlation"""
    c = corr_full(h)
    return dx * np.max(c)

def shift_mask(k, n):
    """Create sparse matrix for shift k"""
    if k >= 0:
        i = np.arange(n - k)
        j = np.arange(k, n)
    else:
        i = np.arange(-k, n)
        j = np.arange(n + k)
    if len(i) == 0:
        return csr_matrix((n, n))
    data = np.ones(len(i))
    M = csr_matrix((data, (i, j)), shape=(n, n))
    return M

def build_sdp(n):
    """Construct SDP problem"""
    h = cp.Variable(n)
    X = cp.Variable((n, n), PSD=True)
    tau = cp.Variable()
    
    constraints = [
        cp.diag(X) == h,
        h >= 0, h <= 1,
        cp.sum(h) == n / 2,
        tau >= 0
    ]
    
    for k in range(-(n - 1), n):
        M = shift_mask(k, n)
        constraints.append(cp.trace(M @ X) >= (n / 2) * (1 - tau))
    
    prob = cp.Problem(cp.Minimize(tau), constraints)
    return prob, h, X, tau

def solve_sdp(prob):
    """Solve SDP and extract solution"""
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)
        if prob.status == cp.OPTIMAL:
            tau_opt = prob.value
            h_sdp = h.value
            X_sdp = X.value
            return h_sdp, X_sdp, tau_opt
        else:
            return None, None, None
    except Exception:
        return None, None, None

def round_from_sdp(h_sdp, X_sdp, use_gaussian=False):
    """Round SDP solution to feasible step function"""
    # Diagonal rounding
    h1 = h_sdp.copy()
    h1 = proj_sum_box(h1, len(h1) / 2)
    c1 = max_overlap(h1, 2. / len(h1))
    
    # Leading eigenvector rounding
    try:
        evals, evecs = np.linalg.eigh(X_sdp)
        idx = np.argmax(evals)
        lambda1 = max(0, evals[idx])
        u1 = evecs[:, idx]
        h2 = np.sqrt(lambda1) * u1
        h2 = proj_sum_box(h2, len(h2) / 2)
        c2 = max_overlap(h2, 2. / len(h2))
    except:
        h2 = h1.copy()
        c2 = c1
    
    # Gaussian rounding
    if use_gaussian:
        best_c = min(c1, c2)
        best_h = h1 if c1 < c2 else h2
        for _ in range(5):
            try:
                z = np.random.multivariate_normal(np.zeros(len(h_sdp)), X_sdp)
                h3 = np.abs(z)
                h3 = proj_sum_box(h3, len(h3) / 2)
                c3 = max_overlap(h3, 2. / len(h3))
                if c3 < best_c:
                    best_c = c3
                    best_h = h3
            except:
                continue
        return best_h, best_c
    
    return h2 if c2 < c1 else h1, min(c1, c2)

def grad_shift(h, k):
    """Compute gradient of correlation term for shift k"""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        # g[i] = (1 - h[i+k]) - h[i-k]
        g[k:] += 1
        g[k:] -= h[:-k] if k < n else 0
        g[:n-k] += h[k:] if k > 0 else 0
    else:
        g[:k] += 1
        g[:k] -= h[-k:] if -k < n else 0
        g[-k:] += h[:k] if k < 0 else 0
    return g

def polish(h_start, n, dx, time_limit):
    """Polish initial solution using projected gradient descent"""
    start_time = time.time()
    h = h_start.copy()
    
    # Precompute shift masks
    shifted_msk = {}
    for k in range(-(n - 1), n):
        shifted_msk[k] = shift_mask(k, n)
    
    # Initialize
    step_size = 0.07
    beta = 0.9
    y = h.copy()
    h_prev = h.copy()
    
    # Iteration loop
    for it in range(2000):
        if time.time() - start_time > time_limit - 1:
            break
            
        # Evaluate gradients
        c = corr_full(h)
        cmax = np.max(c)
        kmax = np.where(c >= cmax - 1e-12)[0]
        g = np.zeros(n)
        for k in kmax:
            g += grad_shift(h, k)
        g /= len(kmax)
        
        # Nesterov extrapolation
        y = h + beta * (h - h_prev)
        h_new = y - step_size * g
        
        # Project
        h_new = proj_sum_box(h_new, n / 2)
        
        c_new = max_overlap(h_new, dx)
        if c_new < max_overlap(h, dx):
            step_size = min(step_size * 1.08, 0.2)
            h_prev = h
            h = h_new
        else:
            step_size *= 0.5
            
        # Swap-moves every 250 steps to escape plateaus
        if it % 250 == 249:
            # Find best, worst values for swap
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            delta = np.random.uniform(-0.02, 0.02)
            
            # Try to swap small delta between i and j
            h_temp = h.copy()
            if h_temp[i] + delta >= 0 and h_temp[i] + delta <= 1:
                if h_temp[j] - delta >= 0 and h_temp[j] - delta <= 1:
                    h_temp[i] += delta
                    h_temp[j] -= delta
                    h_temp = proj_sum_box(h_temp, n / 2)
                    c_temp = max_overlap(h_temp, dx)
                    if c_temp < max_overlap(h, dx):
                        h = h_temp
    return h, max_overlap(h, dx)
