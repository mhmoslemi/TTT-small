import numpy as np

import time

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def sigmoid_inv(x):
    return np.log(x / (1 - x))

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def create_spline_profile(z, n_points=120):
    """Create a symmetric spline profile from parameters z"""
    # Use 7 knots (3 interior + 2 endpoints), symmetric around 1
    n_knots = 7
    knots = np.array([0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0])  # fixed positions
    
    # Apply sigmoid to ensure [0,1] bounds
    y_vals = sigmoid(z)
    
    # For symmetric profile, use only the first half and mirror
    y_left = y_vals[:4]  # 0, 0.2, 0.4, 0.5
    y_right = y_vals[:3][::-1]  # 0.5, 0.4, 0.2
    
    # Combine to make a full monotonic decreasing profile from [0,1]
    y_full = np.concatenate([y_left, y_right])
    
    # Create spline
    cs = CubicSpline(knots, y_full, bc_type='clamped')
    
    # Sample on grid
    x_grid = np.linspace(0, 1, n_points//2)
    y_vals = cs(x_grid)
    
    # Mirror to full range
    h_full = np.concatenate([y_vals, y_vals[::-1]])
    
    # Scale to meet integral constraint
    s = np.sum(h_full)
    alpha = TARGET_SUM / s
    h_scaled = h_full * alpha
    
    # Handle clipping if needed
    h_clipped = np.clip(h_scaled, 0, 1)
    # Adjust if clipping changed sum significantly
    new_s = np.sum(h_clipped)
    if abs(new_s - TARGET_SUM) > 1e-8:
        alpha2 = TARGET_SUM / new_s
        h_final = np.clip(h_clipped * alpha2, 0, 1)
    else:
        h_final = h_clipped
    
    return h_final

    weights = weights / np.sum(weights)
    
    # Initialize covariance matrix
    C = np.eye(d)

        offspring = []

                c5 = evaluate_c5(h)

            best_h = best_h_cand.copy()
        
        # Select best individuals
        best_individuals = values[:mu]
        
        # Update mean
        z_sum = np.zeros(d)
        for z, _, _ in best_individuals:
            z_sum += weights * (z - mean)
        mean = mean + z_sum
        
        # Update covariance matrix
        z_diffs = np.array([(z - mean) for z, _, _ in best_individuals])
        C = (1 - 0.1) * C + 0.1 * np.dot(z_diffs.T, z_diffs) / np.sum(weights)
        
        # Update step-size
        pc = (1 - 0.05) * pc + np.sqrt(0.05 * (2 - 0.05)) * np.dot(C, weights * z_diffs.T)
        sigma = sigma * np.exp(0.05 * (np.linalg.norm(pc) / np.sqrt(d) - 1))
    
    return best_h, best_c5

def local_refinement(h0, max_iter=1200):
    """Refine solution via projected gradient descent"""
    h = h0.copy()
    current_c5 = evaluate_c5(h)
    
    start_time = time.time()
    for i in range(max_iter):
        if time.time() - start_time > 550:
            break
            
        # Compute correlation vector
        corr = compute_corr_fft(h)
        k_max = np.argmax(corr)
        
        # Compute subgradient
        g = np.zeros_like(h)
        for k in range(len(h)):
            if abs(corr[k] - corr[k_max]) < 1e-12:
                # Compute gradient for this shift
                for i in range(len(h)):
                    j = (i + k) % len(h)
                    k2 = (i - k) % len(h)
                    g[i] += (1 - h[j]) - h[k2]
        
        # Direction is average of gradients for worst shifts
        g = g / np.sum(np.abs(g)) * 1e-4
        
        # Update and project
        h_new = h - g
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Line search: keep if improvement
        c5_new = evaluate_c5(h_new)
        if c5_new < current_c5:
            h = h_new
            current_c5 = c5_new
        else:
            # Optionally increase step size if moves are small
            if i % 200 == 0:
                # Try small random moves
                idx1, idx2 = np.random.randint(0, len(h), 2)
                delta = np.random.uniform(-0.015, 0.015)
                h_test = h.copy()
                h_test[idx1] += delta
                h_test[idx2] -= delta
                h_test = project_to_sum_box(h_test, TARGET_SUM)
                c5_test = evaluate_c5(h_test)
                if c5_test < current_c5:
                    h, current_c5 = h_test, c5_test
    
    return h, current_c5
