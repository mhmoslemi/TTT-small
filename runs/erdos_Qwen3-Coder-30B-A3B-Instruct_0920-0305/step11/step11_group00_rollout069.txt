import numpy as np
from scipy.signal import correlate
from scipy.interpolate import CubicSpline
from scipy.optimize import minimize
import time

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-x))

def inv_sigmoid(y):
    return np.log(y / (1 - y))

def compute_corr_fft(h):
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    return dx * np.max(corr)

def evaluate_c5(h):
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(h, target_sum):
    """Project vector h onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(h + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def evaluate_c5_with_scale(h, s):
    """Apply scaling to h and compute C5"""
    h_scaled = h * s
    return evaluate_c5(h_scaled)

def linspace_with_bounds(a, b, n):
    return np.linspace(a, b, n)

def build_piecewise_profile(z):
    """Build step function from parameters z"""
    # Use 7 knots symmetric about center (0,1)
    knots_x = np.array([0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0])
    m = len(knots_x)
    y_vals = sigmoid(z)
    # Make sure it's symmetric around x=0.5
    y_vals = np.concatenate([y_vals, y_vals[-2::-1]])  # Mirror but exclude middle
    
    # Interpolate with cubic spline
    cs = CubicSpline(knots_x, y_vals[:len(knots_x)])
    x_full = np.linspace(0, 1, n_points//2 + 1)
    y_full = cs(x_full)
    
    # Flipped version
    y_full = np.concatenate([y_full, y_full[-2::-1]])
    
    # Normalize to integral = 1
    integral = np.sum(y_full) * dx
    scale = TARGET_SUM / integral
    h = np.clip(y_full * scale, 0, 1)
    return h

def parametric_c5_objective(z, n):
    h = build_piecewise_profile(z)
    return evaluate_c5(h)

def refine_solution(h, max_iter=1200):
    """Refine solution with gradient descent"""
    h_current = h.copy()
    best_h = h_current.copy()
    best_c5 = evaluate_c5(h_current)
    
    gamma = 0.08
    for i in range(max_iter):
        if i % 200 == 0 and i > 0:
            # Plateau escape
            idx1, idx2 = np.random.randint(0, len(h_current), 2)
            delta = np.random.uniform(-0.015, 0.015)
            h_test = h_current.copy()
            h_test[idx1] += delta
            h_test[idx2] -= delta
            h_test = project_to_sum_box(h_test, TARGET_SUM)
            new_c5 = evaluate_c5(h_test)
            if new_c5 < best_c5:
                h_current = h_test
                best_c5 = new_c5
                best_h = h_current.copy()
        else:
            corr = compute_corr_fft(h_current)
            k_max = np.argmax(corr)
            g = np.zeros_like(h_current)
            if k_max < len(h_current):
                g[:-k_max] += h_current[k_max:]
                g[k_max:] += h_current[:-k_max]
            else:
                g[k_max-len(h_current):] += h_current[:len(h_current)-(k_max-len(h_current))]
                g[:k_max-len(h_current)] += h_current[len(h_current)-(k_max-len(h_current)):]
            g = -g
            
            # Gradient step
            h_test = h_current - gamma * g
            h_test = project_to_sum_box(h_test, TARGET_SUM)
            new_c5 = evaluate_c5(h_test)
            
            if new_c5 < best_c5:
                h_current = h_test
                best_c5 = new_c5
                best_h = h_current.copy()
            else:
                gamma *= 0.5
                if gamma < 1e-6:
                    break
    return best_h, best_c5
