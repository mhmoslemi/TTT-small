import numpy as np

def proj_sum_box(v, target_sum):
    """Project vector v onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(v + (a + b)/2, 0, 1)
    return result

def corr_full(h):
    """Compute full correlation using FFT"""
    n = len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def max_overlap(h, dx):
    """Compute max overlap with dx scaling"""
    corr = corr_full(h)
    return dx * np.max(corr)

def grad_shift(h, k):
    """Compute gradient of correlation term c_k(h)"""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] = 1 - h[:-k]
        if k < n:
            g[:k] = -h[k:]
    else:
        g[:k] = 1 - h[-k:]
        if -k < n:
            g[k:] = -h[:-k]
    return g

def build_sdp(n):
    """Build and return SDP model"""
    h = cp.Variable(n)
    X = cp.Variable((n, n), PSD=True)
    tau = cp.Variable()
    
    # Precompute shift masks
    masks = {}
    for k in range(-(n-1), n):
        mask = np.zeros((n, n))
        if k >= 0:
            mask[np.arange(n-k), np.arange(k, n)] = 1
        else:
            mask[np.arange(-k, n), np.arange(n+k)] = 1
        masks[k] = cp.Constant(mask)
    
    # Constraints
    constraints = [
        cp.diag(X) == h,
        h >= 0, h <= 1,
        cp.sum(h) == n/2,
        tau >= 0
    ]
    
    # Overlap constraints
    for k in range(-(n-1), n):
        constraints.append(cp.trace(masks[k] @ X) >= (n/2)*(1 - tau))
    
    prob = cp.Problem(cp.Minimize(tau), constraints)
    return prob, h, X, tau, masks

def solve_sdp(prob):
    """Solve SDP"""
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            return prob.value, prob.variables()[0].value, prob.variables()[1].value
    except:
        pass
    return None, None, None

def round_from_sdp(h_sdp, X_sdp, n):
    """Rounding strategies"""
    candidates = []
    
    # Strategy 1: Diagonal rounding
    h1 = h_sdp.copy()
    candidates.append(h1)
    
    # Strategy 2: Leading eigenvector
    try:
        eigvals, eigvecs = np.linalg.eigh(X_sdp)
        idx = np.argmax(eigvals)
        v = eigvecs[:, idx]
        if np.sum(v) < 0:
            v = -v
        h2 = np.sqrt(eigvals[idx]) * v
        h2 = proj_sum_box(h2, n/2)
        candidates.append(h2)
    except:
        pass
    
    # Strategy 3: Gaussian randomization
    for _ in range(5):
        z = np.random.multivariate_normal(np.zeros(n), X_sdp)
        h3 = np.abs(z)
        h3 = proj_sum_box(h3, n/2)
        candidates.append(h3)
    
    # Evaluate all candidates
    best_h = None
    best_c5 = float('inf')
    for h in candidates:
        c5 = max_overlap(h, 2.0/n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    return best_h, best_c5

    h = h_start.copy()
    best_h = h.copy()
    best_c5 = max_overlap(h, dx)
    
    step = 0.07
    h_prev = h.copy()
    beta = 0.9
    
    for it in range(2000):
        if time.time() - t_start > budget_s - 2:
            break
            
        corr = corr_full(h)
        cmax = np.max(corr)
        Kmax = np.where(corr >= cmax - 1e-12)[0]
        
        # Compute average gradient
        g = np.zeros(n)
        for k in Kmax:
            g += grad_shift(h, k)
        g /= len(Kmax)
        
        # Nesterov-style extrapolation
        y = h + beta * (h - h_prev)
        h_new = y - step * g
        h_new = proj_sum_box(h_new, n/2)
        
        new_c5 = max_overlap(h_new, dx)
        if new_c5 < best_c5:
            step = min(step * 1.08, 0.2)
            best_c5 = new_c5
            best_h = h_new.copy()
            h_prev = h.copy()
            h = h_new.copy()
        else:
            step *= 0.5
            
        # Swap-move every 250 iterations
        if it % 250 == 0:
            i, j = np.random.randint(0, n, 2)
            delta = np.random.uniform(-0.02, 0.02)
            h_new = h.copy()
            if h_new[i] + delta >= 0 and h_new[i] + delta <= 1:
                h_new[i] += delta
                h_new[j] -= delta
                h_new = proj_sum_box(h_new, n/2)
                new_c5 = max_overlap(h_new, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
    
    return best_h, best_c5
