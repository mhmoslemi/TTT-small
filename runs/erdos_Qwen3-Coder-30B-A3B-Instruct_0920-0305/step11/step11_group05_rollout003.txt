import numpy as np
from scipy.fft import correlate
from scipy.optimize import linprog
import time
import cvxpy as cp

# Global constants
n_points = 144
dx = 2.0 / n_points
TARGET_SUM = n_points / 2
MAX_ITER = 200
TOLERANCE = 1e-8

def evaluate_c5(h):
    """Fast FFT-based evaluation of C5"""
    # Full correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Take the maximum and scale by dx
    max_corr = np.max(corr)
    return dx * max_corr

def project_onto_sum_box(v, target_sum):
    """Projection onto sum = target_sum and box constraints [0,1] using waterfilling"""
    v = np.array(v)
    n = len(v)
    # Sort values in descending order
    sorted_idx = np.argsort(v)[::-1]
    sorted_v = v[sorted_idx]
    
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_v)
    # Find the threshold index where sum > target_sum
    idx = np.searchsorted(cumsum, target_sum, side="right")
    
    if idx == 0:
        # All values above 1, clip everything to 1
        result = np.ones(n)
    elif idx == n:
        # All values below 1, clip everything to 1
        result = np.clip(v, 0, 1)
    else:
        # Convert to elements that are at least threshold
        threshold = (target_sum - cumsum[idx - 1]) / (n - idx)
        # Clamp at 1 and 0
        if threshold > 1:
            result = np.ones(n)
        else:
            # Invert the sort
            result = np.zeros(n)
            # Set first idx elements to threshold
            result[sorted_idx[:idx]] = threshold
            # Set remaining elements to 1 (but they will be clipped later)
            result[sorted_idx[idx:]] = 1
            # Reclip to [0,1]
            result = np.clip(result, 0, 1)
    
    # Adjust to be exactly on target sum
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-10:
        # Increment elements from the bottom in case of large residual
        idxs = np.argsort(result)
        i = 0
        while diff > 1e-10 and i < n:
            incr = min(1 - result[idxs[i]], diff)
            result[idxs[i]] += incr
            diff -= incr
            i += 1
    
    return np.clip(result, 0, 1)

def solve_linearized_step(y):
    """
    Solve the linearized LP given a current 'y':
    max_t, s.t. sum_i x_i y_{i+s} >= t for all shifts s,
                sum_i x_i = target_sum,
                0 <= x_i <= 1
    """
    n = len(y)
    # Create constraint matrix and RHS
    A_list = []
    b_list = []
    
    # For each shift s from -n+1 to n-1
    shifts = list(range(1 - n, n))  # [1-n, 2-n, ..., 0, ..., n-2, n-1]
    
    # Build A matrix rows
    for s in shifts:
        row = np.zeros(n)
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                row[i] = y[j]
        A_list.append(row)
        b_list.append(0)  # We'll set this to t later (it's a variable)
    
    # Add equality constraint: sum x_i = target_sum
    A_eq_row = np.ones(n)
    b_eq = TARGET_SUM
    
    # Build the full constraint matrix and RHS: row-wise block stacking with -1 for t
    # We'll use cvxpy variable t along with x_i as vector
    # We'll solve in terms of augmented system of constraints
    # Variables t and x
    
    # Use CVXPY
    x = cp.Variable(n)
    t = cp.Variable()

    constraints = []
    for i in range(len(shifts)):
        constraints.append(A_list[i] @ x >= t)
    # Equality
    constraints.append(cp.sum(x) == b_eq)
    
    # Box constraints
    constraints.extend([x >= 0, x <= 1])
    
    # Objective
    objective = cp.Maximize(t)
    
    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_opt = x.value
            t_opt = t.value
            return x_opt, t_opt
        else:
            print("warning: LP not solved optimally, status:", prob.status)
            return y, 0   # fallback
    except Exception as e:
        print("warning: exception in solving LP: ", e)
        return y, 0  # fallback

                # Accept with probability
                h_refined = h_test.copy()
                current_c5 = new_c5
                
            iter_count += 1
            
        # Final evaluation
        final_c5 = current_c5
        
        return (h_refined.tolist(), float(final_c5), n_points)
        
    # If no improvement was observed, return the first best

        c5_eval = evaluate_c5(best_h)
        return (best_h.tolist(), float(c5_eval), n_points)
    
    return (h0.tolist(), evaluate_c5(h0), n_points)
