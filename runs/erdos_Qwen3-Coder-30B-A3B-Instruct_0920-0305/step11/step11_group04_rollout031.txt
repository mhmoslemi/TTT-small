import numpy as np

# Global variables
TARGET_SUM = None
n_points = 128
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT for speed"""
    h_rev = h[::-1]
    b = h  # h * h = h^2
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_C5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return 1.0 - np.min(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling algorithm
    x_sorted = np.sort(x)[::-1]
    cumsum = np.cumsum(x_sorted)
    # Find the threshold lambda using binary search
    a, b = -1.0, 1.0
    tol = 1e-10
    max_iter = 50
    
    def check_lambda(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    for _ in range(max_iter):
        m = (a + b) / 2
        fm = check_lambda(m)
        if abs(fm) < tol:
            break
        if fm * check_lambda(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def greedy_swap_local_search(h, dx, budget_s, start_time):
    """Perform greedy swap local search focused on the worst shift"""
    corr = compute_corr(h)
    worst_shift = np.argmin(corr)
    if worst_shift >= len(corr) // 2:
        worst_k = worst_shift - len(corr) // 2
    else:
        worst_k = worst_shift - len(corr) // 2
    
    # Iterate until no improving swaps
    improved = True
    while improved and time.time() - start_time < budget_s * 0.9:
        improved = False
        best_net = -1e-10
        best_i, best_j = None, None

        # Check all pairs (i_zero, j_one)
        for i in range(len(h)):
            if h[i] != 0.0: continue
            for j in range(len(h)):  
                if h[j] != 1.0: continue
                # Calculate gain: quantity of 1s at shifted positions
                gain = 0.0
                loss = 0.0
                if 0 <= i + worst_k < len(h) and h[i + worst_k] == 1:
                    gain += dx
                if 0 <= i - worst_k < len(h) and h[i - worst_k] == 1:
                    gain += dx
                if 0 <= j + worst_k < len(h) and h[j + worst_k] == 1:
                    loss += dx
                if 0 <= j - worst_k < len(h) and h[j - worst_k] == 1:
                    loss += dx

                net = gain - loss
                if net > best_net:
                    best_net = net
                    best_i, best_j = i, j

        if best_net > 1e-10 and best_i is not None and best_j is not None:
            # Perform the swap
            h[best_i], h[best_j] = h[best_j], h[best_i]
            improved = True
            # Update correlation vector with incremental computation
            # For simplicity, recompute the whole thing occasionally, but we'll try to just update
            break  # Just one pass for faster computation

    return h
