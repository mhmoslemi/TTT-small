import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C5 for given step function h"""
    # Convert to numpy array if needed
    if not isinstance(h, np.ndarray):
        h = np.array(h)
    
    # Compute overlap integral using full correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # Normalize by dx = 2.0 / n
    dx = 2.0 / len(h)
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use modest size for tractable MILP
    n = 100
    dx = 2.0 / n
    target_sum = n // 2
    
    # Variables
    h = cp.Variable(n, boolean=True)
    z = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h) == target_sum]
    
    # Add McCormick envelopes for all valid shifts
    # We'll use the full correlation computation
    # For each shift k, create auxiliary variables y[i,k] = h[i] * h[i+k]
    y_vars = {}
    correlations = []
    
    # Create auxiliary variables for each k and i where valid
    for k in range(-(n-1), n):  # All possible shifts
        y_vars[k] = {}
        for i in range(n):
            if 0 <= i + k < n:
                y = cp.Variable()
                y_vars[k][i] = y
                # McCormick envelopes for y = h[i] * h[i+k]
                constraints.extend([
                    y <= h[i],
                    y <= h[i+k],
                    y >= h[i] + h[i+k] - 1,
                    y >= 0
                ])
                
                # Add contribution to correlation for this shift
                correlations.append(y)
    
    # Add the objective constraints
    # We want to maximize the minimum correlation 1 - rho_k
    # So we minimize the maximum of (1 - rho_k) which is just to minimize z
    # Where for each k:
    # rho_k = dx * sum over all valid i of y[i,k]
    # So for each k, we compute the correlation
    for k in range(-(n-1), n):  # All possible shifts
        if k >= 0:
            # For positive k, indices are (i, i+k) where i+k < n
            rho_k = cp.sum([y_vars[k][i] for i in range(n-k)])
        elif k < 0:
            # For negative k, indices are (i+k, i) where i+k >= 0
            rho_k = cp.sum([y_vars[k][i] for i in range(-k, n)])
        else:
            # This shouldn't happen as k runs from -(n-1) to n-1
            continue
            
        # This shift's correlation contribution to our global maximum
        # z = max_k (1 - rho_k) = max_k (1 - dx * rho_k)
        # So z >= 1 - dx * rho_k 
        constraints.append(z >= 1 - dx * rho_k)
    
    # Objective
    prob = cp.Problem(cp.Minimize(z), constraints)
    
    # Solve the MILP
    solver = cp.CBC
    try:
        time_left = budget_s - (time.time() - start_time)
        if time_left > 30:
            prob.solve(solver=solver, TimeLimit=time_left - 10, warm_start=True)
        else:
            prob.solve(solver=solver, TimeLimit=time_left, warm_start=True)
    except Exception as e:
        # Fallback to previous strategy if solver fails
        return run_fallback(seed, budget_s)
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_opt = h.value
        if h_opt is None:
            # Fallback if no solution found
            return run_fallback(seed, budget_s)
    else:
        # Try to get a reasonable solution even from infeasible
        h_opt = h.value
        if h_opt is None:
            # Fallback if still no solution found
            return run_fallback(seed, budget_s)
    
    # Convert to binary
    h_opt = np.rint(h_opt).astype(int)
    
    # Project to sum constraint
    def project_to_sum_box(x, target_sum):
        # Binary search for lambda
        def f(lam):
            projected = np.clip(x + lam, 0, 1)
            return np.sum(projected) - target_sum

        a, b = -1.0, 1.0
        for _ in range(50):
            m = (a + b) / 2
            fm = f(m)
            if abs(fm) < 1e-10:
                break
            if fm * f(a) < 0:
                b = m
            else:
                a = m

        result = np.clip(x + (a + b)/2, 0, 1)
        return result

    # Use the waterfilling approach to satisfy the sum constraint exactly
    # First ensure h_opt sums to target_sum
    old_sum = np.sum(h_opt)
    if abs(old_sum - target_sum) > 1e-6:
        h_opt = project_to_sum_box(h_opt, target_sum)
    
    # Final polishing 
    def polish_solution(h_input):
        h_current = h_input.copy().astype(float)
        iterations = 0
        max_iter = 500
        
        while iterations < max_iter:
            # Compute correlations
            corr = np.correlate(h_current, h_current, mode="full")
            # n is the input size, so corr has 2*n-1 elements
            # We want the correlations starting from shift -(n-1) to (n-1)
            # So we extract the relevant part
            mid = len(corr) // 2
            rho_values = corr[mid - (n-1): mid + n]
            
            # Find worst shift (minimum correlation)
            k_min = np.argmin(rho_values)
            k_best = k_min - (n-1)  # Convert to actual shift
            
            # Compute gains and losses
            best_gain = -np.inf
            best_swap = None
            
            # For each position where h_current=0 (gain from adding 1)
            for i in range(n):
                if h_current[i] < 0.5:  # Rough check
                    gain = 0
                    if 0 <= i + k_best < n:
                        gain += h_current[i + k_best]
                    if 0 <= i - k_best < n and k_best != 0:
                        gain += h_current[i - k_best]
                    # For each position where h_current=1 (loss from removing 1)
                    for j in range(n):
                        if h_current[j] > 0.5:  # Rough check
                            loss = 0
                            if 0 <= j + k_best < n:
                                loss += h_current[j + k_best]
                            if 0 <= j - k_best < n and k_best != 0:
                                loss += h_current[j - k_best]
                            
                            if gain - loss > best_gain:
                                best_gain = gain - loss
                                best_swap = (i, j)
            
            if best_gain <= 0:
                break
                
            i, j = best_swap
            h_current[i] = 1
            h_current[j] = 0
            iterations += 1
            
        # Final rounding to ensure binary values (project to nearest 0/1)
        h_final = np.rint(h_current)  
        return h_final
    
    try:
        h_final = polish_solution(h_opt)
        c5_bound = evaluate_erdos_solution(h_final)
    except Exception as e:
        h_final = h_opt
        c5_bound = evaluate_erdos_solution(h_final)

    return (h_final.tolist(), float(c5_bound), n)
