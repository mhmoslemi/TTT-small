n_points = 120

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def convolve_fft(a, b):
    """Convolve two arrays using FFT with same length as result"""
    L = len(a) + len(b) - 1
    L = 1 << (L - 1).bit_length()  # Next power of 2
    A = fft(a, L)
    B = fft(b, L)
    return ifft(A * B).real[:len(a) + len(b) - 1]

def smooth_max(corr, alpha):
    """Compute smooth max approximation"""
    return (1.0 / alpha) * np.log(np.sum(np.exp(alpha * corr)))

def gradient_of_smooth_max(h, corr, alpha):
    """Compute gradient of smooth max approximation"""
    w = np.exp(alpha * corr)

    # Two convolutions needed
    conv1 = convolve_fft(w, h)
    conv2 = convolve_fft(w[::-1], h)  # flip w first then convolve
    return 1.0 - conv1 - conv2
