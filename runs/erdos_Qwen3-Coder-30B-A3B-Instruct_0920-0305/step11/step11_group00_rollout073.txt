import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Initial bounds
    a = -1.0 - np.max(x)
    b = 1.0 - np.min(x)
    
    # Ensure a and b are valid
    if a > b:
        a, b = b, a
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def grad_shift(h, k):
    """Compute gradient of correlation with shift k"""
    n = len(h)
    # Create padded array
    pad = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    base = np.arange(n) + (n-1)
    
    # Compute gradient entries
    grad_k = (1 - pad[base + k]) - pad[base - k]
    return grad_k

def compute_weights(c, alpha):
    """Compute softmax weights"""
    max_c = np.max(alpha * c)
    exp_vals = np.exp(alpha * c - max_c)
    weights = exp_vals / np.sum(exp_vals)
    return weights

def accelerated_projected_gradient(h, alpha, max_iter=150, eta_init=0.1, beta=0.9):
    """Accelerated projected gradient descent with continuation"""
    h_best = h.copy()
    c_best = compute_corr_fft(h_best)
    best_c5 = c5_from_corr(c_best)
    
    # Initialize
    h_prev = h.copy()
    y = h.copy()
    
    # Line search parameters
    eta = eta_init
    max_eta = 1.0
    backtracking_factor = 0.5
    max_backtracking = 5
    
    for i in range(max_iter):
        # Compute current correlation
        c = compute_corr_fft(y)
        
        # Compute weights for smooth surrogate
        weights = compute_weights(c, alpha)
        
        # Compute gradient of smooth surrogate
        grad = np.zeros_like(h)
        for k in range(-n_points+1, n_points):
            if k == 0:
                continue
            grad_k = grad_shift(h, k)
            grad += weights[k + n_points - 1] * grad_k
        
        # Nesterov acceleration
        h_tmp = y - eta * grad
        h_new = project_to_sum_box(h_tmp, TARGET_SUM)
        
        # Backtracking line search
        for _ in range(max_backtracking):
            c_new = compute_corr_fft(h_new)
            f_new = np.sum(weights * c_new) / alpha
            f_prev = np.sum(weights * c) / alpha
            if f_new <= f_prev - 0.5 * eta * np.dot(grad, h_new - y) or eta < 1e-10:
                break
            eta *= backtracking_factor
            h_tmp = y - eta * grad
            h_new = project_to_sum_box(h_tmp, TARGET_SUM)
        
        # Update variables
        y = h_new + beta * (h_new - h_prev)
        h_prev = h_new.copy()
        
        # Update best if improved
        c_new = compute_corr_fft(h_new)
        c5_new = c5_from_corr(c_new)
        if c5_new < best_c5:
            h_best = h_new.copy()
            best_c5 = c5_new
            
        # Periodic random escape
        if i % 30 == 29:
            # Try a random perturbation to escape local minima
            i1, i2 = np.random.choice(n_points, 2, replace=False)
            delta = 0.02
            if h_best[i1] >= delta:
                h_test = h_best.copy()
                h_test[i1] -= delta
                h_test[i2] += delta
                h_test = project_to_sum_box(h_test, TARGET_SUM)
                c_test = compute_corr_fft(h_test)
                c5_test = c5_from_corr(c_test)
                if c5_test < best_c5:
                    h_best = h_test.copy()
                    best_c5 = c5_test
        
        # Increase eta if successful
        if i % 10 == 9 and eta < max_eta:
            eta *= 1.1
            
    return h_best, best_c5

    best_c5 = c5_from_corr(c_best)
    
    eta = eta_init

    max_no_improvement = 500
    
    for i in range(max_iter):
        # Compute current correlation
        c = compute_corr_fft(h_current)
        
        # Find active indices
        max_val = np.max(c)
        active_indices = np.where(np.abs(c - max_val) < 1e-12)[0]
        
        # Compute average gradient
        grad_avg = np.zeros_like(h_current)
        for k in active_indices:
            # Adjust for zero-based indexing in the correlation
            shift = k - (n_points - 1)
            grad_k = grad_shift(h_current, shift)
            grad_avg += grad_k
        
        grad_avg /= len(active_indices)
        
        # Gradient step
        h_new = h_current - eta * grad_avg
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new solution
        c_new = compute_corr_fft(h_new)
        c5_new = c5_from_corr(c_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 200:
                eta *= 0.5
                no_improvement_count = 0

                    break

                    h_test = project_to_sum_box(h_test, TARGET_SUM)

                    c5_test = c5_from_corr(c_test)

        # Check time limit
        if i % 100 == 99:
            # ABORT early to avoid timeout
            pass
    
    return best_h, best_c5

def triangular_start():
    """Create triangular starting vector"""
    h = np.zeros(n_points)
    half = n_points // 2
    
    for i in range(half):
        h[i] = 1 - (i / half)
        h[n_points - 1 - i] = 1 - (i / half)
        
    # Normalize properly to sum to TARGET_SUM
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def random_start():

    h = np.random.random(n_points)
    h = project_to_sum_box(h, TARGET_SUM)
