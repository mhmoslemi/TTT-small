import numpy as np

from scipy.signal import correlate
import time

def weight_normalise(x, n_points):
    """Normalize vector to sum to n_points / 2"""
    dx = 2.0 / n_points
    scale = (n_points / 2) / np.sum(x)
    return x * scale

def max_defect(h):
    """Compute exact C5 = max_k int h(x)(1-h(x+k)) dx"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr / 2.0  # dx = 2/n_points so multiply by 1/2

def softmax_objective(h, alpha):
    """Soft max approximation to max_k int h(x)(1-h(x+k)) dx"""
    n_points = len(h)
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode="full")  # (2*n_points - 1) entries
    
    # Only consider shifts 1..n_points-1
    corr_vals = corr[n_points-1:]
    exp_corr = np.exp(alpha * corr_vals)
    exp_sum = np.sum(exp_corr)
    soft_max = np.log(exp_sum) / alpha
    
    # Gradient
    weights = exp_corr / exp_sum
    grad = np.zeros(n_points)
    for i in range(n_points):
        for s in range(1, n_points):
            if i + s < n_points:
                grad[i] += weights[s-1] * (-h[i + s])
            if i - s >= 0:
                grad[i] += weights[s-1] * (-h[i - s])
    grad += weights @ (1 - h)  # Scalar term

    return soft_max * dx, grad * dx

def build_milp(n):
    """Build and solve binary MILP for the min-min overlap problem"""
    # n binary variables
    h = Variable(n, boolean=True)
    t = Variable(1)
    
    # Weight constraint: sum h[i] = n/2
    constraints = [sum_expr(h) == n / 2]
    
    # Create auxiliary variables and constraints for AND operations
    # w[i,s] = h[i] AND h[i+s] for each shift s
    w = {}
    for s in range(1, n):
        constraints.append(h[s:n] <= h[0:n-s])
        constraints.append(h[0:n-s] <= h[s:n])
        constraints.append(h[0:n-s] + h[s:n] - 1 <= w[s])
    
    # For each shift, set minimum overlap (sum w[i,s] >= t)
    for s in range(1, n):
        constraints.append(sum_expr(w[s]) >= t)
    
    obj = Minimize(-t)
    prob = Problem(obj, constraints)
    
    solver_opts = {"maxTime": 180}
    prob.solve(solver=solver, verbose=False, **solver_opts)
    
    if prob.status == 'optimal' or prob.status == 'feasible':
        h_sol = np.round(h.value).astype(int)
        return h_sol
    else:
        # Try to get a feasible solution
        try:
            prob.solve(solver=solver, verbose=False)
            h_sol = np.round(h.value).astype(int)
            return h_sol
        except:
            return None
    return None

def refine_continuous(h0, n_points):
    """Refine a binary solution using smooth max approximation"""
    def f_and_grad(h, alpha):
        return softmax_objective(h, alpha)

    h_refined = h0.copy()
    
    # Run 3 stages with increasing alpha
    for alpha in [5, 15, 40]:
        # Use trust-constr for a smooth gradient-based optimization
        result = minimize(
            fun=lambda h: f_and_grad(h, alpha)[0],
            x0=h_refined,
            method='trust-constr',
            jac=lambda h: f_and_grad(h, alpha)[1],
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2},
            options={'maxiter': 200}
        )
        if result.success:
            h_refined = result.x
        else:
            # If optimization fails, fallback to projection
            h_refined = weight_normalise(h_refined, n_points)
            h_refined = np.clip(h_refined, 0, 1)

    # Final postprocessing to enforce constraints
    h_refined = weight_normalise(h_refined, n_points)
    h_refined = np.clip(h_refined, 0, 1)
    return h_refined

def anneal(h, n_points):
    """Improve solution using targeted swap-annealing"""
    # Precompute correlations
    c = correlate(h, 1-h, mode="full")
    # Let c[s] = int h(x)*1-h(x+s) dx
    # So we consider shifts 1 to 2*n - 2 (and get 2*n - 2 correlation terms)
    # BUT correlation function puts the center as [n-1]
    # So c[0] = int h(x)h(x+1) dx for s=1, ..., corr[len] = int h(x)h(x+k) dx for k = 2*n - 2
    # so [n-1] corresponds to s=1, and [2*n-2] corresponds to s=n-1
    # we need to work with [n:] as indices that store the values

    corr = c[n-1:]  # corr[i] = int h(x)(1 - h(x + i+1)) 
    # shift 1 -> corr[0], ...
    # We want equal sized overlapping intervals for correctness
    
    def update_corr(h):
        corr = correlate(h, 1 - h, mode="full")[n-1:]
        return corr  

    def effective_max_defect(h):
        return max_defect(h)  # exact computation

    best_h = h.copy()
    best_c5 = effective_max_defect(best_h)
    initial_c5 = best_c5
    gamma = 0.2
    total_proposals = 0
    while total_proposals < 5000 and (time.time() - start_time) < 550:
        # Find worst shift
        corr = update_corr(best_h)
        s_star = np.argmax(corr)
        # Find indices that contribute to w[s_star]
        contrib = []
        for i in range(n_points):
            if best_h[i] > 0 and (i + s_star < n_points and best_h[i+s_star] < 0.5):
                contrib.append(i)
        
        if not contrib:
            break

        # Select a random move
        i = np.random.choice(contrib)
        # Select target region
        target_idx = []
        for j in range(n_points):
            if best_h[j] < 0.5 and (j + s_star < n_points and best_h[j+s_star] > 0.5):
                target_idx.append(j)
        if not target_idx:
            break
        
        j = np.random.choice(target_idx)
        
        # Propose move
        delta = min(best_h[i], 1 - best_h[j]) * gamma
        if delta < 1e-6:
            break

        # Apply symmetric update
        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        if i + s_star < n_points:
            new_h[i+s_star] -= delta
        if j + s_star < n_points:
            new_h[j+s_star] += delta

        new_c5 = effective_max_defect(new_h)
        # Accept iff new_c5 <= best_c5 (lower is better)
        if new_c5 <= best_c5:
            best_h = new_h
            best_c5 = new_c5
        elif np.random.random() < 0.1:  # Long-term perturbation limit
            best_h = new_h
            best_c5 = new_c5
            
        gamma *= 0.99
        if gamma < 1e-4:
            gamma = 1e-4
        total_proposals += 1

    return best_h
