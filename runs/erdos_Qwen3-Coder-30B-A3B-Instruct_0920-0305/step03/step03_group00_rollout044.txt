import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from cvxpy import *
import time

def weight_normalise(x, n):
    """Rescale x so that sum(x) * (2/n) = 1, i.e. sum(x) = n/2"""
    return x * (n / 2) / np.sum(x)

def max_defect(h):
    """Compute exact C5 and the shift that achieves it"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    max_shift = np.argmax(corr) - len(h) + 1
    return max_corr * dx, max_shift

def softmax_objective(h, alpha):
    """Compute soft max of the defect function and its gradient"""
    n = len(h)
    dx = 2.0 / n
    # Compute all the f_s = sum h[i] * (1 - h[i+s])
    corr_1mh = correlate(h, 1 - h, mode="full")
    # Use log-sum-exp for smooth approximation
    exp_alpha_fs = np.exp(alpha * corr_1mh)
    Z = np.sum(exp_alpha_fs)
    if Z <= 0:
        Z = 1e-12
    weights = exp_alpha_fs / Z
    
    # Soft max value
    soft_max = np.sum(weights * corr_1mh) / alpha
    
    # Gradient computation:
    # d/dh_j of soft_max = sum_s w_s * (1 - 1_{j+s<n} h_{j+s} - 1_{j-s>=0} h_{j-s})
    grad = np.zeros_like(h)
    for i in range(n):
        # Forward differences
        for s in range(1, n):
            if i + s < n:
                grad[i] += weights[s + n - 1] * (-h[i + s])
        # Backward differences  
        for s in range(1, i + 1):
            grad[i] += weights[n - s + i] * (-h[i - s])
        # Add constant term
        grad[i] += np.sum(weights)  # This is actually zero, but leave it for clarity
    
    return soft_max, grad

def build_milp(n):
    """Build and solve the binary MILP for min max overlap"""
    # Binary variables h[i] (0 or 1)
    h = Variable(n, boolean=True)
    # For AND constraint we require h[i] ∧ h[i+s] = w[i,s] 
    w = {}
    t = Variable()
    
    # Essential length: avoid boundary artifacts
    constraints = [
        sum(h) == n/2,
        t <= 0,  # Start with something small  
    ]
    
    for s in range(1, n):
        for i in range(n-s):
            # Temporary variable w[i,s]
            # If h[i] = 1, h[i+s] = 1 then w[i,s] must be 1  
            # If any h[i], h[i+s] = 0 then w[i,s] = 0
            # The idea is that we'll have w[i,s] = h[i] * h[i+s] with constraints below
            w[(i, s)] = Variable(boolean=True) 
            constraints += [
                w[(i, s)] <= h[i],
                w[(i, s)] <= h[i + s],
                w[(i, s)] >= h[i] + h[i + s] - 1,
            ]
            # Now force t <= sum of w[i,s]
            constraints[-1] = t <= sum(w[(i,s)] for i in range(n - s))  # Update with actual inequality
    
    # Objective: maximize t
    objective = Maximize(t)
    prob = Problem(objective, constraints)
    
    # Solve with time limit
    solver_opts = {"maxTime": 180}
    try:
        prob.solve(solver=ECOS_BB, verbose=False, **solver_opts)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            return np.array([float(h_i.value) for h_i in h])
        else:
            return None
    except Exception:
        return None

def refine_continuous(h0, n):
    """Refine binary solution by continuous optimization"""
    dx = 2.0 / n
    h = h0.copy()
    
    # Smooth max approximation with increasing α
    alphas = [5.0, 15.0, 40.0]
    for alpha in alphas:
        def obj_and_grad(h_new):
            val, grad = softmax_objective(h_new, alpha)
            return val, grad
        
        result = minimize(
            fun=lambda x: obj_and_grad(x)[0],
            x0=h,
            method='trust-constr',
            jac=lambda x: obj_and_grad(x)[1],
            bounds=[(0, 1)] * len(h),
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n/2},
            options={'gtol': 1e-6, 'maxiter': 200}
        )
        if result.success:
            h = result.x
        else:
            # If optimization fails, fallback
            h = np.clip(weight_normalise(h, n), 0, 1)
            
    # Final projection back to cube and fix weight
    h = np.clip(h, 0, 1)
    h = weight_normalise(h, n)
    return h

def anneal(h, n):
    """Swap-anneal optimization"""
    start_time = time.time()
    dx = 2.0 / n
    h = h.copy()
    T = 1.0
    gamma = 0.2
    max_iter = 5000
    corr = correlate(h, 1 - h, mode="full")
    
    for _ in range(max_iter):
        if time.time() - start_time > 550:
            break
        # Current worst shift
        max_def = np.max(corr)
        best_s = np.argmax(corr) - len(h) + 1
        
        # Find indices contributing to the worst case
        contribs = []
        for i in range(n):
            if h[i] > 0 and (i + best_s < n) and h[i + best_s] < 0.5:
                contribs.append(i)
        
        if not contribs:
            continue
            
        # Select random indices from both sets
        i1 = np.random.choice(contribs)
        # Find indices with small h[j] but large h[j+best_s] 
        ids = [j for j in range(n) if (j + best_s < n and h[j] < 0.5 and h[j + best_s] > 0.5)]
        if not ids:
            continue
        i2 = np.random.choice(ids)
        
        # Make a move (we want to reduce max_corr)
        delta = min(h[i1], 1 - h[i2]) * gamma
        if delta < 1e-8:
            continue
            
        # Modify the h values
        h[i1] -= delta
        h[i2] += delta
        h[i1 + best_s] += delta
        h[i2 + best_s] -= delta
        
        # Re-adjust to satisfy weight constraint
        h = weight_normalise(h, n)
        
        # Update correlation
        corr = correlate(h, 1 - h, mode="full")
        
        # Cool gamma
        gamma *= 0.99
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 100
    dx = 2.0 / n_points
    
    # Try to solve the MILP first (binary)
    h_binary = build_milp(n_points)
    if h_binary is None:
        # Fall back to continuous optimization
        h_binary = np.random.uniform(0, 1, n_points)
        h_binary = weight_normalise(h_binary, n_points)
        
    h_refined = refine_continuous(h_binary, n_points)
    h_final = anneal(h_refined, n_points)
    
    c5_final, s = max_defect(h_final)
    return (h_final.tolist(), c5_final, n_points)
