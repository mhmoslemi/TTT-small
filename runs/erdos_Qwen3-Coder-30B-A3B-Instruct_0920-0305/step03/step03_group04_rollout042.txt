import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time

# Constants
n_values = [48, 60, 72, 84]
target_bound = 0.3807
dx_list = [2.0 / n for n in n_values]
default_maxiter = 300  # for gradient-based stages

def sigmoid(y):
    return 1.0 / (1.0 + np.exp(-y))

def logit(p):
    return np.log(p / (1.0 - p))

def proj_to_simplex(v, S=0.5):
    # Project on to simplex with sum = S using iterative clipping
    v = np.clip(v, 0, 1)
    v_sum = np.sum(v)
    if np.abs(v_sum - S) < 1e-10:
        return v
    # If v_sum > S, subtract excess from highest values
    # If v_sum < S, add to lowest values
    # Let x be the residual we want to distribute
    x = S - v_sum
    # Normalize each v[i] to be [0,1] and scale linearly
    v = v + x / len(v)
    clamped = np.clip(v, 0, 1)
    if np.sum(clamped) == S:
        return clamped
    # Exact projection: sort and adjust
    sorted_indices = np.argsort(v)[::-1]
    total = np.sum(v)
    if total <= S:
        v = np.clip(v + (S - total) / len(v), 0, 1)
    else:
        # Find smallest z such that sum of max(v_i - z,0) = S
        sorted_v = np.sort(v)[::-1]
        cumsum = np.cumsum(sorted_v)
        n_excess = np.searchsorted(cumsum, S) + 1
        z = (cumsum[n_excess - 1] - S) / n_excess
        v = np.maximum(v - z, 0)
        if np.sum(v) < S:
            v = np.clip(v + (S - np.sum(v))/len(v), 0, 1)
    # Ensure exact sum
    if np.abs(np.sum(v) - S) > 1e-7:
        v = v * S / np.sum(v)
    return np.clip(v, 0, 1)

def evaluate_h(h):
    h = np.array(h)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    dx = 2.0 / len(h)
    c5 = max_correlation * dx
    return c5

def correlation_vec(h):
    correlation = np.correlate(h, 1 - h, mode="full")
    return correlation

def smooth_max(c, alpha=30.0):
    m = np.max(c)
    if np.isinf(m):
        return m
    diff = alpha * (c - m)
    diff = np.clip(diff, -100, 100)  # prevent overflow
    return m + np.log(np.sum(np.exp(diff))) / alpha

def penalty_sum(h, lambda_pen=10000.0):
    target_sum = 0.5
    diff = np.sum(h) - target_sum
    return lambda_pen * diff * diff

def binary_initial(n):
    # Generate a random binary vector with n/2 ones
    v = np.zeros(n)
    ones_idx = np.random.permutation(n)[:n//2]
    v[ones_idx] = 1.0
    return v

def swap_and_evaluate(h, i_one, i_zero):
    # swap h[i_one] and h[i_zero] and compute new C5
    h_new = h.copy()
    h_new[i_one], h_new[i_zero] = h_new[i_zero], h_new[i_one]
    return evaluate_h(h_new)

def binary_hill_climb(h_init, max_iter=10000, stagnation_threshold=5 * len(h_init)):
    n = len(h_init)
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best)
    bit_indices = {0: np.where(h_best == 0)[0], 1: np.where(h_best == 1)[0]}
    iterations_no_improvement = 0
    for i in range(max_iter):
        if iterations_no_improvement > stagnation_threshold:
            # Reinitialize
            h_new = binary_initial(n)
            c5_new = evaluate_h(h_new)
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new
                bit_indices[0] = np.where(h_best == 0)[0]
                bit_indices[1] = np.where(h_best == 1)[0]
            iterations_no_improvement = 0
            continue
        ones = bit_indices[1]
        zeros = bit_indices[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_one = np.random.choice(ones)
        i_zero = np.random.choice(zeros)
        h_new = h_best.copy()
        h_new[i_one], h_new[i_zero] = h_new[i_zero], h_new[i_one]
        c5_new = evaluate_h(h_new)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
            bit_indices[0] = np.where(h_best == 0)[0]
            bit_indices[1] = np.where(h_best == 1)[0]
            iterations_no_improvement = 0
        else:
            iterations_no_improvement += 1
        if c5_best <= target_bound:
            break
    return h_best, c5_best

def initial_interval_family(n_samples):
    best_h = None
    best_c5 = np.inf
    dx = 2.0 / n_samples
    for m in range(2, min(9, n_samples // 2 + 1)):  # We need at least two intervals
        l = 1.0 / m  # interval length
        max_g = (2 - m * l) / (m - 1)
        for g in [0.0, max_g * 0.5, max_g * 0.75, max_g]:
            if m * l + (m - 1) * g > 2:
                continue
            for offset in np.arange(0, g, dx / 2):
                h = np.zeros(n_samples)
                for i in range(m):
                    start_idx = int(np.round((offset + i * (l + g)) / dx))
                    end_idx = min(start_idx + int(np.round(l / dx)), n_samples)
                    if start_idx < end_idx:
                        h[start_idx:end_idx] = 1.0
                # Normalize so that sum = n_samples / 2
                h_sum = np.sum(h)
                if abs(h_sum - n_samples / 2) > 1e-10:
                    h = h * (n_samples / 2) / h_sum
                c5 = evaluate_h(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    return best_h, best_c5

def smooth_max_optimization(h_init, maxiter=300, n_restarts=8, alpha=30.0, lambda_pen=10000):
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h)

    def objective(y):
        h = sigmoid(y)
        c = correlation_vec(h)
        smooth_max_val = smooth_max(c, alpha)
        penalty = penalty_sum(h, lambda_pen)
        return smooth_max_val + penalty

    for i in range(n_restarts):
        # Initialize with a slightly perturbed logit version of best binary vector
        y0 = logit(best_h) + np.random.normal(0, 0.2, n)
        result = minimize(objective, y0, method='L-BFGS-B', options={'maxiter': maxiter})
        if result.success:
            h = sigmoid(result.x)
            # Project to feasible simplex
            h = proj_to_simplex(h, S=0.5)
            c5 = evaluate_h(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
            if best_c5 <= target_bound:
                break
        else:
            # Filter out failed optimizations
            continue

    return best_h, best_c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = np.inf
    best_n = None

    # Stage 1: Interval family search
    for n in n_values:
        dx = 2.0 / n
        h, c5 = initial_interval_family(n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    # Stage 2: Binary hill climbing on multiple n
    for n_idx, n in enumerate(n_values):
        dx = 2.0 / n
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 5:
            break
        # Time budget allocation per n = ...
        time_per_n = remaining_time * 0.5 / len(n_values)
        time_start = time.time()
        h_best = np.zeros(n)
        c5_best = np.inf
        # Multiple random restarts
        for _ in range(5):
            h_init = binary_initial(n)
            h_try, c5_try = binary_hill_climb(h_init, max_iter=5000)
            if c5_try < c5_best:
                c5_best = c5_try
                h_best = h_try.copy()
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best.copy()
            best_n = n
        if best_c5 <= target_bound:
            break
        if time.time() - time_start > time_per_n:
            break

    # Stage 3: Smooth max optimization
    for restart in range(8):
        h_try = binary_hill_climb(best_h, max_iter=1000)[0]
        h_try, c5_try = smooth_max_optimization(h_try, n_restarts=1)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        if best_c5 <= target_bound:
            break

    # Stage 4: Final polishing with SLSQP
    bounds = [(0, 1) for _ in range(best_n)]
    def constraint_eq(h_values):
        return np.sum(h_values) * (2.0 / best_n) - 1.0

    constraints = {'type': 'eq', 'fun': constraint_eq}
    try:
        result = minimize(evaluate_h, best_h, method='SLSQP', bounds=bounds,
                         constraints=constraints, options={'maxiter': 500, 'ftol': 1e-12})
        if result.success:
            best_h = result.x
    except:
        pass

    return list(best_h), evaluate_h(best_h), best_n
