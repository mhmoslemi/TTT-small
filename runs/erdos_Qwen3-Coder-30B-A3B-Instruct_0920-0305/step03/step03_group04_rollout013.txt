import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time

# Constants
n_vals = [48, 60, 72, 84]
dx_vals = [2.0 / n for n in n_vals]
target_bound = 0.3807

# Utilities
def dx(n):
    return 2.0 / n

def evaluate_h(h):
    """Evaluate the C_5 bound exactly."""
    h = np.array(h)
    c = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(c)
    return max_c * dx(len(h))

def correlation_vec(h):
    return np.correlate(h, 1 - h, mode="full")

def smooth_max(c, alpha):
    m = np.max(c)
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

def penalty_sum(h, lam):
    return lam * (np.sum(h) - len(h) / 2) ** 2

def proj_to_simplex(v, S=None):
    if S is None:
        S = len(v) / 2
    v = np.clip(v, 0, 1)
    v_sum = np.sum(v)
    if abs(v_sum - S) < 1e-10:
        return v
    diff = S - v_sum
    if diff > 0:
        # Distribute excess to smallest elements
        idxs = np.argsort(v)
        for i in idxs:
            if diff <= 0:
                break
            take = min(diff, 1 - v[i])
            v[i] += take
            diff -= take
    else:
        # Remove excess from largest elements
        idxs = np.argsort(v)[::-1]
        for i in idxs:
            if diff >= 0:
                break
            take = min(-diff, v[i])
            v[i] -= take
            diff += take
    return v

def logit(p):
    return np.log(p / (1 - p))

def sigmoid(y):
    return 1 / (1 + np.exp(-y))

def binary_initial(n):
    """Generate random binary vector with sum=n/2."""
    h = np.zeros(n)
    ones = np.random.choice(n, n//2, replace=False)
    h[ones] = 1
    return h

def swap_and_evaluate(h, i_one, i_zero):
    """Swap h[i_one] and h[i_zero] and return new C5."""
    h_new = h.copy()
    h_new[i_one], h_new[i_zero] = h_new[i_zero], h_new[i_one]
    return evaluate_h(h_new)

# Exact evaluation
def evaluate(h):
    h = np.array(h)
    c = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(c)
    return max_corr * dx(len(h))

# Grid search over structured intervals: equal interval pattern with gaps
def search_structured_intervals(n_vals, max_m=8):
    best = (None, float('inf'), None)
    for n in n_vals:
        dx_n = dx(n)
        for m in range(2, max_m + 1):
            l = 1.0 / m
            g = (2.0 - m * l) / (m - 1)
            if g < 0:
                continue
            for delta in np.arange(0, g, dx_n / 2):
                h = np.zeros(n)
                for i in range(m):
                    start = int(np.round((delta + i * (l + g)) / dx_n))
                    end = int(np.round((delta + i * (l + g) + l) / dx_n))
                    if start < 0:
                        start = 0
                    if end > n:
                        end = n
                    if start < end:
                        h[start:end] = 1
                h = proj_to_simplex(h, n / 2)
                c5 = evaluate_h(h)
                if c5 < best[1]:
                    best = (h.copy(), c5, n)
    return best

# Binary hill-climb
def binary_hill_climb(n, max_iter=50000, max_restarts=10):
    best_h = binary_initial(n)
    best_c5 = evaluate_h(best_h)

    restart_count = 0
    for _ in range(max_restarts):
        h = binary_initial(n)
        c5 = evaluate_h(h)

        it = 0
        stagnation_count = 0
        old_c5 = float('inf')
        while it < max_iter:
            i_one = np.random.choice(np.where(h == 1)[0])
            i_zero = np.random.choice(np.where(h == 0)[0])
            h_new = h.copy()
            h_new[i_one], h_new[i_zero] = h_new[i_zero], h_new[i_one]
            c5_new = evaluate_h(h_new)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
                stagnation_count = 0
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
                if c5 < target_bound:
                    return best_h, best_c5
            else:
                stagnation_count += 1
                if stagnation_count > 5 * n:
                    # restart with new random binary vector
                    h = binary_initial(n)
                    c5 = evaluate_h(h)
                    stagnation_count = 0
            it += 1

            if it % 1000 == 0:
                pass  # diagnostics

    return best_h, best_c5

# Smooth-max stage using L-BFGS-B
def refine_with_smooth_max(h_binary, n, maxiter=300, num_restarts=8):
    alpha = 30.0
    lambda_sum = 10000.0
    h_best = h_binary.copy()
    c5_best = evaluate_h(h_best)

    for i in range(num_restarts):
        # Initialize with a small noise around the binary vector
        y = logit(h_binary) + np.random.normal(0, 0.2, n)
        def obj(y):
            h = sigmoid(y)
            c = correlation_vec(h)
            return smooth_max(c, alpha) + penalty_sum(h, lambda_sum)

        def grad(y):
            h = sigmoid(y)
            c = correlation_vec(h)
            max_c = np.max(c)
            mask = (c == max_c)
            dc = np.zeros_like(c)
            dc[mask] = 1
            grad_logit = np.zeros_like(y)
            # Hard to compute surrogate gradient precisely
            # Using numerical FD for simplicity
            eps = 1e-6
            h_plus = sigmoid(y + eps)
            h_minus = sigmoid(y - eps)
            c_plus = correlation_vec(h_plus)
            c_minus = correlation_vec(h_minus)
            grad_correlate = (c_plus - c_minus) / (2 * eps) * dx(n)
            return grad_correlate / (dx(n) * alpha * np.exp(alpha * (c - max_c))) - 2 * lambda_sum * (np.sum(h) - n/2) * sigmoid(y) * (1 - sigmoid(y))

        # Slow: compute sub-gradient numerically (simpler and okay at scale)
        result = minimize(obj, y, method='L-BFGS-B', options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-10})
        h_final = sigmoid(result.x)
        h_final = proj_to_simplex(h_final, n / 2)
        c5_final = evaluate_h(h_final)

        if c5_final < c5_best:
            h_best, c5_best = h_final, c5_final
            if c5_best < target_bound:
                return h_best, c5_best

    return h_best, c5_best

# Final polishing
def polish_exact(h, n):
    def obj(h_vec):
        return evaluate_h(h_vec)

    bounds = [(0, 1) for _ in range(n)]
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx(n) - 1.0}

    result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'ftol': 1e-12, 'maxiter': 500})
    if result.success:
        h_final = result.x
        c5_final = evaluate_h(h_final)
        return h_final, c5_final
    else:
        return h, evaluate_h(h)

# Training loop
def run(seed=42, budget_s=600):
    np.random.seed(seed)
    t0 = time.time()

    best_h = None
    best_c5 = float('inf')
    best_n = None

    # Stage 1: Structured interval search
    if time.time() - t0 < 0.15 * budget_s:
        h_struct, c5_struct, n_struct = search_structured_intervals(n_vals)
        if h_struct is not None:
            best_h, best_c5, best_n = h_struct, c5_struct, n_struct

    # Stage 2: Binary hill climbing on several n
    if time.time() - t0 < 0.65 * budget_s:
        for n in n_vals:
            if time.time() - t0 > 0.65 * budget_s - 5:
                break
            start_time = time.time()
            h_bin, c5_bin = binary_hill_climb(n, max_iter=10000, max_restarts=5)
            if c5_bin < best_c5:
                best_h, best_c5, best_n = h_bin, c5_bin, n
            if c5_bin < target_bound:
                break

    # Stage 3: Smooth max refinement
    if time.time() - t0 < 0.9 * budget_s:
        if best_h is not None:
            h_refined, c5_refined = refine_with_smooth_max(best_h, best_n)
            if c5_refined < best_c5:
                best_h, best_c5, best_n = h_refined, c5_refined, best_n

    # Stage 4: Exact polishing
    if time.time() - t0 < 0.95 * budget_s:
        if best_h is not None:
            h_polished, c5_polished = polish_exact(best_h, best_n)
            if c5_polished < best_c5:
                best_h, best_c5, best_n = h_polished, c5_polished, best_n

    return list(best_h), best_c5, best_n
