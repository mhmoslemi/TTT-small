import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time
from scipy.special import expit
from scipy.spatial.distance import cdist

# Constants
n_values = [36, 42, 48, 54, 60]  
dx = 2.0 / 48  # default
target_bound = 0.3807
R = 30  # number of rounds per n
max_swaps = 50000
SLSQP_MAXITER = 500
L_BFGS_MAXITER = 300
SMOOTH_ALPHA = 30
PENALTY_LAMBDA = 1000

def sigmoid(x):
    return expit(x)

def logit(x):
    return np.log(x / (1 - x))

def proj_to_simplex(v, s):
    """Project vector v onto the simplex with sum = s."""
    v = np.clip(v, 0, 1)
    v_sum = np.sum(v)
    if np.abs(v_sum - s) < 1e-10:
        return v
    if v_sum > s:
        # Need to reduce
        sorted_v = np.sort(v)[::-1]
        cumsum = np.cumsum(sorted_v)
        idx = np.argmax(sorted_v * np.arange(1, len(sorted_v)+1) - (cumsum - s)) + 1
        theta = (cumsum[idx-1] - s) / idx
        v = np.maximum(v - theta, 0)
        v = v * s / np.sum(v)
    else:
        # Need to increase
        v = v * s / v_sum
    return v

def evaluate_h(h_values):
    # Evaluate the C_5 bound for given h values
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def smooth_max(h_values, alpha=30):
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    # Use log_sum_exp for numerical stability
    soft_sum = np.log(np.sum(np.exp(alpha * corr))) / alpha
    return soft_sum

def solve_sdp(n):
    """Solve the SDP relaxation for a given n."""
    x = cp.Variable(n)
    X = cp.Variable((n, n), PSD=True)
    t = cp.Variable()

    constraints = []

    # Box constraints and mass constraint
    constraints += [x >= 0, x <= 1]
    constraints += [cp.sum(x) == n / 2]

    # Diagonal consistency
    constraints += [cp.diag(X) == x]

    # McCormick constraints
    for i in range(n):
        for j in range(i + 1, n):
            constraints += [
                X[i, j] >= 0,
                X[i, j] <= x[i],
                X[i, j] <= x[j],
                X[i, j] >= x[i] + x[j] - 1
            ]

    # Overlap constraints
    for k in range(n):
        lhs = cp.sum(x[:n - k]) - cp.sum(cp.diag(X, k))
        constraints += [lhs <= t]

    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver='SCS', max_iters=8000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_solution = x.value
            X_solution = X.value
            lower_bound = t.value
            return x_solution, X_solution, lower_bound
        else:
            return None, None, None
    except Exception:
        return None, None, None

def hyperplane_round(X_star, x_star, n):
    # Perform random hyperplane rounding with post-processing
    V, D, _ = np.linalg.svd(X_star)
    r = min(10, V.shape[1])
    V = V[:, :r]
    D = D[:r]

    for _ in range(R):
        g = np.random.randn(r)
        s = np.sign(V @ g)

        # Post-process to ensure exactly n/2 ones
        mask = s >= 0
        ones_current = np.sum(mask)
        delta = ones_current - n // 2
        if delta > 0:
            # Too many ones, flip the ones with smallest |s[i]|
            flip_idx = np.argsort(np.abs(s[mask]))[:delta]
            mask[flip_idx] = False
        elif delta < 0:
            # Not enough ones, flip the zero indices with smallest |s[i]|
            flip_idx = np.argsort(np.abs(s[~mask]))[:(-delta)]
            mask[flip_idx] = True

        h_vec = mask.astype(float)
        h_vec = proj_to_simplex(h_vec, n / 2)
        yield h_vec

def hill_climb(h_current, n):
    """Perform massive binary hill climb."""
    ones = np.where(h_current > 0.5)[0].tolist()
    zeros = np.where(h_current < 0.5)[0].tolist()
    c5_current = evaluate_h(h_current)

    swaps = 0
    best_h = h_current.copy()
    best_c5 = c5_current
    
    while swaps < max_swaps:
        # Find max shift
        corr = np.correlate(h_current, 1 - h_current, mode="full")
        max_k = 0
        for i in range(len(corr)):
            if corr[i] > corr[max_k]:
                max_k = i
        
        # Identification of mismatches
        if max_k < len(h_current):
            mismatches = np.where((h_current[:len(h_current) - max_k] > 0.5)
                                * (h_current[max_k:] < 0.5))[0]
            if len(mismatches) > 0:
                # Pick one and try to swap with a zero
                for _ in range(100):
                    i = np.random.choice(mismatches)
                    if i in ones:
                        j = np.random.choice(zeros)
                        if i != j:
                            h_new = h_current.copy()
                            h_new[i] = 0
                            h_new[j] = 1
                            c5_new = evaluate_h(h_new)
                            if c5_new < best_c5:
                                best_h = h_new
                                best_c5 = c5_new
                                ones.remove(i)
                                ones.append(j)
                                zeros.remove(j)
                                zeros.append(i)
                        break

        # Normal swap
        if swaps % 1000 == 0:
            # Use a more controlled random swap
            if len(ones) > 0 and len(zeros) > 0:
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                if i != j:
                    h_new = h_current.copy()
                    h_new[i] = 0
                    h_new[j] = 1
                    c5_new = evaluate_h(h_new)
                    if c5_new < best_c5:
                        best_h = h_new
                        best_c5 = c5_new
                        ones.remove(i)
                        ones.append(j)
                        zeros.remove(j)
                        zeros.append(i)
                    swaps += 1
        else:
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            if i != j:
                h_new = h_current.copy()
                h_new[i] = 0
                h_new[j] = 1
                c5_new = evaluate_h(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                    ones.remove(i)
                    ones.append(j)
                    zeros.remove(j)
                    zeros.append(i)
        if best_c5 < target_bound:
            return best_h, best_c5
        swaps += 1

    return best_h, best_c5

def refine_to_continuous(h_binary, n):
    h = np.clip(h_binary, 1e-6, 1 - 1e-6)
    y = logit(h)
    
    def f(y_vec):
        h_new = sigmoid(y_vec)
        regularization = 0.5 * PENALTY_LAMBDA * (np.sum(h_new) - n/2)**2
        return smooth_max(h_new) + regularization

    result = minimize(f, y, method='L-BFGS-B', options={'maxiter': L_BFGS_MAXITER})
    if result.success:
        h_refined = sigmoid(result.x)
        h_refined = proj_to_simplex(h_refined, n / 2)
        return h_refined
    else:
        return h_binary

    np.random.seed(seed)
    global dx
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    best_for_n = {}

    for n in n_values:
        if time.time() - start_time > 0.95 * budget_s:
            break

        # Phase A: SDP relaxation
        x_opt, X_opt, t_opt = solve_sdp(n)
        if t_opt is None:
            continue

        # Phase B: Hyperplane rounding and hill climb
        for h_rounded in hyperplane_round(X_opt, x_opt, n):
            # Phase C: hill climb
            h_final, c5_final = hill_climb(h_rounded, n)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
            if c5_final < target_bound:
                break

            # Phase D: Continuous refinement
            h_cont = refine_to_continuous(h_final, n)
            c5_cont = evaluate_h(h_cont)
            if c5_cont < best_c5:
                best_h = h_cont.copy()
                best_c5 = c5_cont

        # If we have a good solution for this n, store it
        if best_h is not None:
            if n in best_for_n:
                if best_c5 < best_for_n[n][1]:
                    best_for_n[n] = (best_h.copy(), best_c5)
            else:
                best_for_n[n] = (best_h.copy(), best_c5)

    if best_h is not None:
