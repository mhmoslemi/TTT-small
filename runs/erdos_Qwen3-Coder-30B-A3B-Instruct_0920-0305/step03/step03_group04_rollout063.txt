import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time

# Default parameters
n_values = [48, 60, 72, 84]
n_points = 47
dx = 2.0 / n_points
target_bound = 0.3807

def dx(n):
    return 2.0 / n

def evaluate(h):
    h = np.array(h)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    return max_correlation * dx(len(h))

def correlation_vec(h):
    return np.correlate(h, 1 - h, mode="full")

def smooth_max(c, alpha=30.0):
    m = np.max(c)
    if m == -np.inf:
        return 0.0
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

def penalty_sum(h, lam=10000.0):
    s = np.sum(h)
    return lam * (s - 0.5 * len(h))**2

def proj_to_simplex(v, S=0.5 * n_points):
    # Project v onto the simplex using the algorithm from
    # "Projected gradient methods for non-negative matrix factorization"
    v = np.clip(v, 0, 1)
    v_sorted = np.sort(v)[::-1]
    rho = np.argmax(np.arange(1, len(v)+1) * v_sorted > (S - np.cumsum(v_sorted[:-1])))
    theta = (S - np.sum(v_sorted[:rho])) / (rho + 1) if rho >= 0 else 0
    return np.maximum(v - theta, 0)

def sigmoid(y):
    return 1.0 / (1.0 + np.exp(-y))

def logit(p):
    return np.log(p / (1.0 - p))

def binary_initial(n):
    h = np.zeros(n)
    idx_ones = np.random.choice(n, size=n//2, replace=False)
    h[idx_ones] = 1.0
    return h

def swap_and_evaluate(h, i_one, i_zero):
    if h[i_one] != 1 or h[i_zero] != 0:
        return evaluate(h)
    h_new = h.copy()
    h_new[i_one] = 0
    h_new[i_zero] = 1
    return evaluate(h_new)

def binary_hill_climb(n, max_steps=50000, restarts=10):
    best_h = None
    best_c5 = np.inf
    h_ones = []
    h_zeros = []

    # Initialization
    for i in range(restarts):
        h = binary_initial(n)
        h_ones = np.where(h == 1)[0].tolist()
        h_zeros = np.where(h == 0)[0].tolist()
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if best_c5 <= target_bound:
            return best_h, best_c5

    # Local search
    for step in range(max_steps):
        if len(h_ones) == 0 or len(h_zeros) == 0:
            break
        i_one = np.random.choice(h_ones)
        i_zero = np.random.choice(h_zeros)
        h_new = best_h.copy()
        h_new[i_one] = 0
        h_new[i_zero] = 1
        c5_new = evaluate(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            # Update binary indices
            h_ones = np.where(best_h == 1)[0].tolist()
            h_zeros = np.where(best_h == 0)[0].tolist()
        if best_c5 <= target_bound:
            return best_h, best_c5
    return best_h, best_c5

def smooth_max_optimize(h, n, alpha=30.0, lam=10000.0, maxiter=300, nrestarts=8):
    best_f = np.inf
    best_h = h.copy()
    
    for _ in range(nrestarts):
        y = logit(h) + np.random.randn(n) * 0.2
        def f(y):
            h_temp = sigmoid(y)
            c = correlation_vec(h_temp)
            return smooth_max(c, alpha) + penalty_sum(h_temp, lam)
        res = minimize(f, y, method='L-BFGS-B', options={'maxiter': maxiter})
        if not res.success:
            continue
        h_final = sigmoid(res.x)
        h_final = proj_to_simplex(h_final, 0.5*n)
        f_final = evaluate(h_final)
        if f_final < best_f:
            best_f = f_final
            best_h = h_final
        if best_f <= target_bound:
            break
    return best_h, best_f
