import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from cvxpy import *
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100  # resolution
    dx = 2.0 / n_points
    target_weight = n_points / 2
    
    # Time tracking
    sdptime = 150  # seconds for SDP
    roundingtime = 120
    refinetime = 180

    # --- PHASE A: SDP Relaxation ---
    # Note: This will require additional helper functions to solve
    # For now, let's define a basic SDP that uses cvxpy to relax the problem
    # After liquidity, compute relaxed value
    try:
        h = Variable(n_points)
        X = Variable((n_points, n_points), symmetric=True)
        t = Variable()

        # Constraints for relational matrix X (McCormick-type)
        constraints = [diag(X) == h,
                       sum(h) == target_weight,
                       0 <= h, h <= 1,
                       X >> 0]  # Positive semidefinite

        # Overlap inequality constraints (for all shifts s=1 to n-1)
        for s in range(1, n_points):
            o_s = sum(X[i, i+s] for i in range(n_points - s))
            constraints.append(o_s >= t)

        # Objective: maximize t
        objective = Maximize(t)
        prob = Problem(objective, constraints)

        # Set solver parameters for SCS
        prob.solve(solver=SCS, max_iters=10000, verbose=False, eps=1e-5)

        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h_val = h.value
            t_opt = t.value
            C5_bound = 1.0 - (2.0 / n_points) * t_opt
        else:
            # Fall back
            h_val = np.array(initial_h_values)
            h_val = h_val[:n_points]  # Downsample if required
            C5_bound = evaluate_c5(h_val)
            if n_points > 80:
                n_points = 80
                dx = 2.0 / n_points
                h_val = h_val[:n_points]
                C5_bound = evaluate_c5(h_val)
            
    except Exception as e:
        # Use fallback construction
        h_val = np.array(initial_h_values)
        h_val = h_val[:n_points]
        C5_bound = evaluate_c5(h_val)

    # Ensure that h_val still satisfies integral constraint
    h_val = h_val * (target_weight / np.sum(h_val))

    candidates = [h_val.copy()]

    # --- PHASE B: Rounding --- 
    num_rounding_trials = 0
    if (time.time() - start_time) < (budget_s - roundingtime):
        try:
            # Assume we computed eigen-decomp earlier; now simulate it
            # We'll round by using Gaussian projection
            num_rounding_trials = 20  # As a heuristic for SDP speed
            for _ in range(num_rounding_trials):
                if (time.time() - start_time) > (budget_s - roundingtime):
                    break
                # Simulate using the relaxed h
                y = np.random.randn(n_points)
                # Sort indices by y values
                sorted_idx = np.argsort(-y)
                h_rounded = np.zeros(n_points)
                h_rounded[sorted_idx[:n_points // 2]] = 1.0
                # Swaps (heuristic)
                swaps = 200
                for _ in range(swaps):
                    if (time.time() - start_time) > (budget_s - roundingtime):
                        break
                    # Pick a 1 and a 0 to swap if it helps
                    ones = np.where(h_rounded == 1)[0]
                    zeros = np.where(h_rounded == 0)[0]
                    if len(ones) > 0 and len(zeros) > 0:
                        i = np.random.choice(ones)
                        j = np.random.choice(zeros)
                        tmp = h_rounded.copy()
                        tmp[i] = 0.0
                        tmp[j] = 1.0
                        if evaluate_c5(tmp) < evaluate_c5(h_rounded):
                            h_rounded = tmp
                candidates.append(h_rounded)
        except Exception:
            pass

    # --- PHASE C: Local refinement (trust-constr) ---

    def smoothmax_obj(h, alpha):
        # Softmax approximation
        dx = 2.0 / n_points 
        f = np.zeros(n_points - 1)
        for s in range(1, n_points):  # shift s
            f[s-1] = np.sum(h[:-s] * (1 - h[s:]))
        return np.log(np.sum(np.exp(alpha * f))) / alpha

    def smoothmax_grad(h, alpha):
        dx = 2.0 / n_points
        f = np.zeros(n_points - 1)
        w = np.zeros(n_points - 1)
        for s in range(1, n_points):
            f[s-1] = np.sum(h[:-s] * (1 - h[s:]))
        exp_f = np.exp(alpha * f)
        w = exp_f / np.sum(exp_f)
        grad = np.zeros_like(h)
        for s in range(1, n_points):
            grad[:-s] += w[s-1] * (1 - h[s:])
            grad[s:] += w[s-1] * (-h[:-s])
        return grad / alpha

    def refine_candidate(h0):
        h = h0.copy()
        # Normalize
        h = h * (target_weight / np.sum(h))

        # Schedule of alphas (gradient ascent via smooth approximation)
        alphas = [5, 15, 40]
        for alpha in alphas:
            if (time.time() - start_time) > (budget_s - refinetime):
                break
            try:
                res = minimize(lambda x: smoothmax_obj(x, alpha),
                               h, method='trust-constr',
                               bounds=[(0, 1) for _ in range(n_points)],
                               constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_weight},
                               options={'maxiter': 200, 'gtol': 1e-7})
                if res.success:
                    h = res.x
                else:
                    # If no convergence, break early
                    break
            except Exception:
                continue
        return h

    best_h = candidates[0]
    best_c5 = evaluate_c5(best_h)

    # Refine each candidate and keep the best
    for i, h0 in enumerate(candidates):
        if (time.time() - start_time) > (budget_s - 5):
            break
        h_refined = refine_candidate(h0)
        c5_cand = evaluate_c5(h_refined)
        if c5_cand < best_c5:
            best_h = h_refined
            best_c5 = c5_cand

    # --- Final Fallback --- 
    if best_c5 >= 0.380:
        # Try random restart
        best_h = np.array(initial_h_values[:n_points])
        best_h = best_h * (target_weight / np.sum(best_h))
        for _ in range(30):
            if (time.time() - start_time) > (budget_s - 5):
                break
            h = np.random.rand(n_points)
            h = h * (target_weight / np.sum(h))
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5

    return (best_h.tolist(), best_c5, n_points)
