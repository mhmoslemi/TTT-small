```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

# Constants
n_points = 47
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def grid_search_initial():
    """Generate a good initial solution using grid search."""
    # Create a gradually varying function that integrates to 1
    x = np.linspace(0, 2, n_points)
    # Start with a piecewise constant function
    h = np.zeros(n_points)
    # Split into 3 sections
    split1 = n_points // 3
    split2 = 2 * n_points // 3
    
    # Make the first third ~0.5, second ~1.0, last ~0.5
    h[:split1] = 0.6
    h[split1:split2] = 1.2
    h[split2:] = 0.4
    
    # Normalize
    h *= (1.0 / (np.sum(h) * dx))
    return h

def tune_solution(h_values, max_iter=300):
    """Use L-BFGS to tune the solution."""
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    def objective(h):
        return evaluate_h(h)
    
    try:
        result = minimize(objective, h_values, method='L-BFGS-B', 
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12})
        if result.success:
            return result.x
    except:
        pass
    return h_values

def accelerated_local_search(h_values, max_iter=500):
    """Faster more aggressive local search."""
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    def objective(h):
        return evaluate_h(h)
    
    try:
        result = minimize(objective, h_values, method='SLSQP',
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': max_iter, 'ftol': 1e-13, 'gtol': 1e-11})
        if result.success:
            return result.x
    except:
        pass
    return h_values

def run(seed=42, budget_s=600):
    """Main optimization routine."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with a better initial solution
    h_current = grid_search_initial()
    c5_current = evaluate_h(h_current)
    
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # Main optimization loop
    iter_num = 0
    max_iter = 1000
    patience = 0
    max_patience = 20
    
    while time.time() - start_time < budget_s * 0.95 and patience < max_patience:
        # Try different strategies
        
        # Strategy 1: Differential evolution
        if iter_num % 5 == 0:
            try:
                bounds = [(0, 1) for _ in range(n_points)]
                result_de = differential_evolution(evaluate_h, bounds, maxiter=30, 
                                                  popsize=10, seed=seed + iter_num)
                if result_de.success:
                    h_candidate = result_de.x
                    c5_candidate = evaluate_h(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
                        patience = 0
                        if best_c5 < target_bound:
                            break
            except:
                pass
        
        # Strategy 2: Local optimization with different starting points
        h_test = h_current.copy()
        # Add small random noise
        noise = np.random.normal(0, 0.05, n_points)
        h_test += noise
        h_test = np.clip(h_test, 0, 1)
        # Normalize integral
        h_test *= (1.0 / (np.sum(h_test) * dx))
        
        h_test = accelerated_local_search(h_test, max_iter=100)
        c5_test = evaluate_h(h_test)
        
        if c5_test < best_c5:
            best_h = h_test.copy()
            best_c5 = c5_test
            patience = 0
            if best_c5 < target_bound:
                break
        else:
            patience += 1
            
        # Strategy 3: Perturb and re-optimize
        h_perturb = best_h.copy()
        # Select random points to change
        indices = np.random.choice(n_points, size=max(1, int(0.1 * n_points)), replace=False)
        for i in indices:
            h_perturb[i] = np.clip(h_perturb[i] + np.random.uniform(-0.1, 0.1), 0, 1)
        
        h_perturb *= (1.0 / (np.sum(h_perturb) * dx))
        h_perturb = tune_solution(h_perturb, max_iter=50)
        c5_perturb = evaluate_h(h_perturb)
        
        if c5_perturb < best_c5:
            best_h = h_perturb.copy()
            best_c5 = c5_perturb
            patience = 0
            if best_c5 < target_bound:
                break
        
        h_current = best_h.copy()
        c5_current = best_c5
        
        iter_num += 1
    
    # Final tuning
    final_h = tune_solution(best_h, max_iter=200)
    final_c5 = evaluate_h(final_h)
    
    return list(final_h), final_c5, n_points
```
