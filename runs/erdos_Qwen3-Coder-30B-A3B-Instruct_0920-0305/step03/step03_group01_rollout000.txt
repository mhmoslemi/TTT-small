import numpy as np
from scipy.optimize import differential_evolution

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_onto_box_and_hyperplane(y, target):
    # Project onto box [0,1] and hyperplane sum = target using binary search
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0, 1)

    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    max_ind = np.where(corr == max_corr)[0]
    # Use first shift that achieves max
    shift_index = max_ind[0]
    shift = shift_index - (n - 1)
    # compute gradient for this shift
    if shift >= 0:
        term1 = np.empty(n)
        term1[:n-shift] = 1 - h[shift:]
        term1[n-shift:] = 1
        term2 = np.empty(n)
        term2[shift:] = h[:n-shift]
        term2[:shift] = 0
    else:
        s = -shift

        term1[:s] = 1
        term2 = np.empty(n)
        term2[:n-s] = h[s:]
        term2[n-s:] = 0
    grad = term1 - term2
    return grad, shift
