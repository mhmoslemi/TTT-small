```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

# Constants
n_points = 48
dx = 2.0 / n_points
target_bound = 0.379

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def evaluate_h_binary(h_values):
    """Evaluate the C_5 bound for binary h values."""
    h = np.array(h_values, dtype=int)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def local_search(h_values, max_iter=200):
    """Refine solution using local optimization."""
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    def objective(h):
        return evaluate_h(h)
    
    try:
        result = minimize(objective, h_values, method='SLSQP', 
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-10})
        if result.success:
            return result.x
    except:
        pass
    return h_values

def run(seed=42, budget_s=600):
    """Main optimization routine."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with binary construction that has exactly half ones
    h_current = np.zeros(n_points)
    ones_indices = np.random.choice(n_points, size=n_points//2, replace=False)
    h_current[ones_indices] = 1
    
    c5_current = evaluate_h_binary(h_current)
    
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # Run simulated annealing with binary pattern
    # Temperature schedule
    initial_temp = 0.1
    temp = initial_temp
    cooling_rate = 0.9999
    min_temp = 1e-6
    
    # Annealing iterations
    max_iter = 1000000
    iter_num = 0
    accepted = 0
    last_improvement = 0
    
    # Time allocation per n
    time_per_n = budget_s // 3
    
    # Try different n values
    n_candidates = [48, 56, 64]
    best_result = (best_h, best_c5, n_points)
    
    for n in n_candidates:
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Adjustments for n
        n_points = n
        dx = 2.0 / n_points
        h_current = np.zeros(n_points)
        ones_indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h_current[ones_indices] = 1
        h_current = h_current.astype(float)
        
        local_best_h = h_current.copy()
        local_best_c5 = evaluate_h_binary(h_current)
        
        # Faster sub-benchmark
        sub_start = time.time()
        progress_threshold = time.time() + time_per_n * 0.75
        
        while time.time() - sub_start < time_per_n and time.time() - start_time < budget_s * 0.95:
            # Choose two indices: one with 1 and one with 0
            ones_idx = np.where(local_best_h == 1)[0]
            zeros_idx = np.where(local_best_h == 0)[0]
            
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
                
            i = np.random.choice(ones_idx)
            j = np.random.choice(zeros_idx)
            
            # Create new candidate solution
            new_h = local_best_h.copy()
            new_h[i] = 0
            new_h[j] = 1
            
            # Compute new correlation
            new_corr = np.correlate(new_h, 1 - new_h, mode="full")
            new_c5 = np.max(new_corr) * dx
            
            # Accept or reject
            delta_c5 = new_c5 - local_best_c5
            if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temp):
                local_best_h = new_h
                local_best_c5 = new_c5
                accepted += 1
                
                if local_best_c5 < best_c5:
                    best_c5 = local_best_c5
                    best_h = local_best_h.copy()
                    last_improvement = iter_num
                    
                if local_best_c5 < target_bound:
                    break
                    
            temp = max(temp * cooling_rate, min_temp)
            iter_num += 1
            
            # Break early if no improvement for long time
            if iter_num - last_improvement > 10000 and temp < 0.0005:
                break
                
            if time.time() > progress_threshold:
                break
    
        if local_best_c5 < best_result[1]:
            best_result = (local_best_h.copy(), local_best_c5, n_points)
            
        # Post-optimization refinement with continuous method
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': constraint_eq}
            
            # Refinement step
            result = minimize(evaluate_h, local_best_h, method='SLSQP', 
                             bounds=bounds, constraints=constraints,
                             options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-8})
            if result.success:
                refined_h = result.x
                refined_c5 = evaluate_h(refined_h)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = refined_h.copy()
        except:
            pass
    
    # Final refinement
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        result = minimize(evaluate_h, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-12})
        if result.success:
            best_h = result.x
    except:
        pass
    
    # Final evaluation
    final_c5 = evaluate_h(best_h)
    
    return [float(x) for x in best_h], final_c5, n_points
```
