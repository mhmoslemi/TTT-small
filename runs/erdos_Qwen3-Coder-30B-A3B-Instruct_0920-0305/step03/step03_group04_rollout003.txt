import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time

# Parameters
n_values = [48, 60, 72, 84]
dx_values = [2.0/n for n in n_values]
target_bound = 0.3807

def dx(n):
    return 2.0 / n

def evaluate_h(h):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx(len(h))
    return c5

def correlation_vec(h):
    """Return the full correlation array."""
    return np.correlate(h, 1 - h, mode="full")

def smooth_max(c, alpha=30):
    """Soft maximum approximation."""
    m = np.max(c)
    if m == -np.inf:
        return 0.0
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

def penalty_sum(h, lambda_reg=10000):
    """Quadratic penalty for deviation from integral constraint."""
    return lambda_reg * (np.sum(h) * dx(len(h)) - 1.0) ** 2

def proj_to_simplex(v, S=None):
    """Project v onto the simplex {x : x ≥ 0, sum(x) = S}."""
    if S is None:
        S = np.sum(v) * dx(len(v))
    v = np.clip(v, 0, 1)
    v_sum = np.sum(v)
    if abs(v_sum - S) < 1e-12:
        return v
    # Scale so that it sums to S
    return v * S / v_sum

def sigmoid(x):
    """Sigmoid function."""
    return 1.0 / (1.0 + np.exp(-x))

def logit(p):
    """Logit function."""
    return np.log(p / (1.0 - p))

def binary_initial(n):
    """Generate a random binary vector."""
    h = np.zeros(n)
    ones_idx = np.random.choice(n, size=n//2, replace=False)
    h[ones_idx] = 1
    return h

def swap_and_evaluate(h, i_one, i_zero):
    """Swap positions i_one and i_zero and return new C5."""
    h_new = h.copy()
    h_new[i_one] = 0
    h_new[i_zero] = 1
    return evaluate_h(h_new)

        grad_sigmoid = h * (1 - h)
        return 30 * grad_sigmoid * exp_terms / s + 20000 * (np.sum(h) * dx(n) - 1.0) * dx(n)

    # Multiple restarts to avoid local minima

                    best_value = c5
                    h_best = h_opt
            y0 = logit(h_best) + np.random.normal(0, 0.1, n)
        except:
            pass
    
    return h_best, best_value

def binary_hill_climb(n, max_iter=50000):
    """Perform hill climb on binary vectors."""
    h_best = binary_initial(n)
    c5_best = evaluate_h(h_best)
    
    ones = list(np.where(h_best == 1)[0])
    zeros = list(np.where(h_best == 0)[0])
    
    # Early exit if bound is already met
    if c5_best <= target_bound:
        return h_best, c5_best

    it = 0
    stagnation = 0
    stagnation_limit = 5 * n
    
    while it < max_iter:
        if stagnation >= stagnation_limit:
            # Restart
            h_best = binary_initial(n)
            c5_best = evaluate_h(h_best)
            ones = list(np.where(h_best == 1)[0])
            zeros = list(np.where(h_best == 0)[0])
            stagnation = 0
        else:
            i_one = np.random.choice(ones)
            i_zero = np.random.choice(zeros)
            h_new = h_best.copy()
            h_new[i_one] = 0
            h_new[i_zero] = 1
            c5_new = evaluate_h(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
                one_idx = np.where(h_best == 1)[0]
                zero_idx = np.where(h_best == 0)[0]
                ones = list(one_idx)
                zeros = list(zero_idx)
                stagnation = 0
            else:
                stagnation += 1
                # Occasionally make bigger jumps (reassign 10% of bits)
                if np.random.rand() < 0.01:
                    num_to_change = max(1, n // 10)
                    to_change = np.random.choice(n, num_to_change, replace=False)
                    for idx in to_change:
                        if h_best[idx] == 0:
                            zeros.remove(idx)
                        else:
                            ones.remove(idx)
                        h_best[idx] = 1 - h_best[idx]
                        if h_best[idx] == 0:
                            zeros.append(idx)
                        else:
                            ones.append(idx)
        it += 1
        # Early exit check
        if c5_best <= target_bound:
            break
            
    return h_best, c5_best

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float("inf")
    best_n = None

    # Stage 1: Parametric interval family search
    for n in n_values:
        if time.time() - start_time > budget_s * 0.15:
            break
        h, c5 = parametric_interval_family(n)
        if h is not None and c5 < best_c5:
            best_h = h
            best_c5 = c5
            best_n = n

    # Stage 2: Binary hill climb
    for n in n_values:
        if time.time() - start_time > budget_s * 0.65:
            break
        remaining_time = budget_s - (time.time() - start_time)
        run_time = min(0.2 * remaining_time, 300.0)
        start_time_it = time.time()

        h, c5 = binary_hill_climb(n, int(20000 * run_time / 300.0))
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
            best_n = n

        if best_c5 <= target_bound:
            break

    # Stage 3: Smooth-max refinement
    if best_h is not None:
        h_refined, c5_refined = refine_with_smooth_max(best_h, best_n)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined

    # Stage 4: Final polishing with SLSQP
    if best_h is not None:
        bounds = [(0, 1) for _ in range(best_n)]
        def objective(h):
            return evaluate_h(h)
        constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) * dx(best_n) - 1.0}
        try:
            result = minimize(objective, best_h, method='SLSQP', 
                             bounds=bounds, 
                             constraints=constraints,

                best_h = result.x
                best_c5 = evaluate_h(best_h)
        except:
            pass

    # Final check
    return list(best_h), best_c5, best_n
