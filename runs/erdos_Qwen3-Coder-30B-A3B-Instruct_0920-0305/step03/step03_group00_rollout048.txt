def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def greedy_swap_improve(h, max_swaps=200):
    """Improve h by swapping 0's and 1's to reduce max overlap"""
    h = h.copy()
    n = len(h)
    dx = 2.0 / n
    
    # Precompute overlaps for each shift (this can be optimized further)
    def compute_all_overlaps(h):
        overlaps = []
        for s in range(1, n):
            corr = np.correlate(h, 1 - h, mode="full")
            overlap = corr[n-1+s]
            overlaps.append(overlap * dx)
        return np.array(overlaps)
    
    current_overlaps = compute_all_overlaps(h)
    best_overlap = np.max(current_overlaps)
    
    swaps = 0
    while swaps < max_swaps:
        # Find the worst shift
        worst_shift_idx = np.argmax(current_overlaps)
        worst_shift = worst_shift_idx + 1   # Shifts start from 1
        
        # Determine which indices to swap
        indices = []
        for i in range(n - worst_shift):
            indices.append((i, i + worst_shift))
         
        # Find best swap
        best_delta = float('inf')
        best_i, best_j = None, None
        
        for i, j in indices:
            if h[i] == 1 and h[j] == 0:
                delta = h[i] + h[j] - (h[i] - 1) - (h[j] + 1)  # Simplified effect
                if delta < best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
        
        # If no beneficial swap found, break
        if best_i is None or best_delta >= 0:
            break
            
        # Perform the swap
        h[best_i], h[best_j] = h[best_j], h[best_i]
        swaps += 1
        
        # Update overlaps only for affected shifts
        current_overlaps = compute_all_overlaps(h)
        new_best = np.max(current_overlaps)
        if new_best > best_overlap:
            # Swap back if it was worse
            h[best_i], h[best_j] = h[best_j], h[best_i]
            break
        else:
            best_overlap = new_best
    
    return h
