import numpy as np

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_binary_milp(n, time_limit):
    """Solve the binary MILP to find optimal pattern"""
    m = n // 2
    # Define binary variables
    b = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    
    # Constraints
    constraints = [cp.sum(b) == m]
    
    # Overlap variables for each shift
    overlaps = []
    
    # Create Y variables to represent b_i * b_{i+s}
    Y = cp.Variable((n, n), boolean=True)
    
    # Add constraints for each pair of indices
    for i in range(n):
        for j in range(n):
            if i <= j:
                # y[i,j] = b[i] * b[j]
                constraints += [
                    Y[i, j] <= b[i],
                    Y[i, j] <= b[j],
                    Y[i, j] >= b[i] + b[j] - 1
                ]
    
    # Compute overlaps for each shift
    for s in range(n):
        if s == 0:
            # Overlap for shift 0 is just the sum of b
            overlaps.append(cp.sum(b))
        else:
            # Sum over valid overlapping positions
            overl_s = cp.sum([Y[i, i+s] for i in range(n - s)])
            overlaps.append(overl_s)
    
    # Add constraints that each overlap >= t
    for i in range(n):
        constraints.append(overlaps[i] >= t)
    
    # Objective: maximize t
    objective = cp.Maximize(t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CBC, timeLimit=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            optimal_t = int(t.value)
            optimal_b = b.value
            return optimal_t, optimal_b
        else:
            return None, None
    except:
        return None, None
