import numpy as np
from scipy.signal import correlate
import time

def proj_box_simplex(v):
    """Project vector v onto the intersection of [0,1]^n and the simplex with sum=1"""
    v = np.array(v)
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    # Compute cumsum
    cumsum = np.cumsum(u)
    # Find the threshold
    mask = u > (1 - cumsum) / np.arange(1, n+1)
    k = np.sum(mask)
    if k == 0:
        # All elements are too large, set to 1/n
        return np.ones(n) / n
    else:
        # Compute the threshold
        tau = (1 - cumsum[k-1]) / k
        # Project onto simplex with 0-1 bounds
        x = np.maximum(v - tau, 0)
        x = x / np.sum(x) * n / 2  # normalize to sum to 1, 2 in [0,2] range
        # Ensure values are in [0,1]
        x = np.clip(x, 0, 1)
        return x

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def spectral_search(n, budget_s, t0, R=30, it_max=500):
    dx = 2.0 / n
    m = n / 2.0
    best_h = None
    best_c5 = np.inf

    for r in range(R):
        if time.time() - t0 > budget_s * 0.99:
            break
        np.random.seed(r)
        
        # Random start on simplex
        u = np.random.exponential(1.0, n)
        h = u / np.sum(u) * m
        h = np.clip(h, 0, 1)
        h = proj_box_simplex(h)
        
        # Compute initial magnitude
        H = np.fft.fft(h)
        M = np.mean(np.abs(H[1:])**2)**0.5  # Avg magnitude of non-DC terms
        
        for t in range(it_max):
            if time.time() - t0 > budget_s * 0.99:
                break
            # Spectral manipulation
            H = np.fft.fft(h)
            H_new = H.copy()
            H_new[1:] = M * np.exp(1j * np.angle(H[1:]))
            h_hat = np.real(np.fft.ifft(H_new))
            
            # Box and simplex projection
            h = proj_box_simplex(h_hat)
            
            # Convergence check (simple)
            if t > 10:
                c5_new = evaluate_c5(h)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
    
    return best_h, best_c5

def binary_refinement(h):
    """Refine h using binary swap heuristic"""
    n = len(h)
    dx = 2.0 / n
    # Convert to binary using 0.5 threshold
    b = (h > 0.5).astype(float)
    # Adjust to meet sum constraint
    sum_b = np.sum(b)
    if sum_b > n/2:
        # Too many ones, make smallest ones zero
        inds = np.argsort(h)[:int(sum_b - n/2)]
        b[inds] = 0
    elif sum_b < n/2:
        # Too few ones, make largest zeros one
        inds = np.argsort(h)[int(n/2 - sum_b):]
        b[inds] = 1
    
    # Initial C5
    c5 = evaluate_c5(b)
    
    # Local hill climb with swaps
    successes = 0
    attempts = 0
    max_attempts = 5 * n
    while successes < 5 and attempts < max_attempts:
        if attempts > 2 * n:
            break
        # Pick random 1 and 0
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)
        
        # Perform swap
        b[i1], b[i0] = b[i0], b[i1]
        c5_new = evaluate_c5(b)
        if c5_new < c5:
            c5 = c5_new
            successes += 1
        else:
            # Undo swap
            b[i1], b[i0] = b[i0], b[i1]
        
        attempts += 1
    
    return b, c5
