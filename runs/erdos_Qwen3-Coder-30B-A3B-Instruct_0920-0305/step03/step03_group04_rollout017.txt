import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time
from scipy.special import expit
from scipy.spatial.distance import cdist

n_values = [36, 42, 48, 54, 60]
target_bound = 0.3807

def evaluate_h(h_values):
    # returns exact C5 = max_k ∫ h(x)(1 - h(x+k)) dx
    h = np.array(h_values)
    dx = 2.0 / len(h)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def logsumexp_smooth_max(c, alpha=30.0):
    # numerically stable logsumexp
    c_max = np.max(c)
    exp_term = np.exp(alpha * (c - c_max))
    return c_max + np.log(np.sum(exp_term)) / alpha

def smooth_max_objective(y, h0, dx, alpha=30.0, lambda_sum=10.0):
    # y is real-valued unconstrained vector
    h = expit(y)  # sigmoid
    h = np.clip(h, 1e-10, 1 - 1e-10)  # prevent numerical issues
    # assign initial values sometimes
    if h0 is not None:
        h = h0 * (np.sum(h0) / np.sum(h))
    # full correlation vector
    c = np.correlate(h, 1 - h, mode="full")
    # smooth max
    soft_max = logsumexp_smooth_max(c, alpha)
    # penalty for violation of integral constraint
    penalty = lambda_sum * (np.sum(h) - 1.0)**2
    return soft_max + penalty

def proj_to_simplex(v, target_sum=1.0):
    # Project v onto simplex {x: x ≥ 0, sum x = target_sum}
    v = np.clip(v, 0, 1e30)
    v_sorted = np.sort(v)[::-1]
    rho = np.argmax(v_sorted * np.arange(1, len(v) + 1) > (np.sum(v) - target_sum))
    theta = (np.sum(v) - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0)

def cvxpy_sdp_solve(n, max_iters=8000):
    # build SDP with cvxpy
    x = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    constraints = [
        0 <= x,
        x <= 1,
        cp.sum(x) == n/2,
        cp.diag(X) == x,
        X >> 0,  # PSD
    ]
    # add McCormick constraints x_i * x_j <= X_ij
    for i in range(n):
        for j in range(i+1, n):
            w = X[i, j]
            constraints.extend([
                w >= 0,
                w <= x[i],
                w <= x[j],
                w >= x[i] + x[j] - 1
            ])
    # add overlap constraints for all shifts k = 0 to n-1
    for k in range(n):
        # sum of x_i - sum of X[i,i+k] <= t
        left = cp.sum(x[:n-k]) - cp.sum(cp.diag(X, k))
        constraints.append(left <= t)

    # set objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=max_iters, verbose=False, eps=1e-6)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value, X.value, prob.value
    except:
        pass
    return None, None, np.inf

def hyperplane_round(X_star, x_star, n, R=30):
    # use random Gaussian to round
    V = np.linalg.eigh(X_star)[1][:, :10]
    best_h = None
    best_C5 = np.inf
    for _ in range(R):
        g = np.random.randn(V.shape[1])
        s = np.sign(V @ g)
        h = (s >= 0).astype(float)
        # fix sum to exactly n/2
        delta = np.sum(h) - n/2
        if delta > 0:
            # need to flip delta ones
            flip_idx = np.argsort(np.abs(V @ g))[:int(delta)]
            h[flip_idx] = 0
        elif delta < 0:
            # need to flip |delta| zeros
            flip_idx = np.argsort(np.abs(V @ g))[-int(-delta):]
            h[flip_idx] = 1
        h = proj_to_simplex(h, n/2)
        C5 = evaluate_h(h)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h
    return best_h, best_C5

def hill_climb_binary(h_current, max_swaps=50000):
    # h is a binary vector (1s and 0s)
    ones = np.where(h_current == 1)[0].tolist()
    zeros = np.where(h_current == 0)[0].tolist()
    C_current = evaluate_h(h_current)

    for i in range(max_swaps):
        if len(ones) == 0 or len(zeros) == 0:
            break
        idx_swap_i = np.random.choice(ones)
        idx_swap_j = np.random.choice(zeros)
        h_new = h_current.copy()
        h_new[idx_swap_i] = 0
        h_new[idx_swap_j] = 1
        C_new = evaluate_h(h_new)
        if C_new < C_current:
            h_current = h_new
            C_current = C_new
            ones.remove(idx_swap_i)
            zeros.append(idx_swap_i)
            zeros.remove(idx_swap_j)
            ones.append(idx_swap_j)
        # check for massive improvement: restart if no improvement for S steps
        if i % 10000 == 0:
            pass  # placeholder
    return h_current, C_current

def run(seed=42, budget_s=600):
    t0 = time.time()
    np.random.seed(seed)
    best_h = None
    best_c5 = np.inf
    best_n = -1
    best_h_sigmoid = None
    best_c5_sigmoid = np.inf
    best_h_fract = None
    best_c5_fract = np.inf

    # Phase A: Solve SDP for different n values
    for n in n_values:
        if time.time() - t0 > (budget_s * 0.6):
            break
        x_star, X_star, t_star = cvxpy_sdp_solve(n, max_iters=8000)
        if x_star is not None and t_star < best_c5:
            h_binary, c5_rounded = hyperplane_round(X_star, x_star, n, R=30)
            if h_binary is not None:
                # Phase C: Run binary hill-climb from rounded solution
                h_opt, c5_final = hill_climb_binary(h_binary)
                if c5_final < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_final
                    best_n = n
                    if best_c5 <= target_bound:
                        break

    # Phase D: Smooth-max refinement
    if best_h is not None and time.time() - t0 < (budget_s * 0.9):
        # Do soft maximisation
        h0 = best_h.copy()
        dx = 2.0 / len(best_h)
        bounds_y = [(-10, 10)] * len(best_h)
        y0 = np.log(h0 / (1 - h0))  # logit
        y0 = np.clip(y0, -10, 10)  # avoid inf
        def obj(y):
            return smooth_max_objective(y, h0, dx)
        try:
            result = minimize(obj, y0, method='L-BFGSB', bounds=bounds_y, options={'maxiter': 200})
            if result.success:
                h_sigmoid = expit(result.x)
                h_sigmoid = proj_to_simplex(h_sigmoid, 1.0)
                c5_sigmoid = evaluate_h(h_sigmoid)
                if c5_sigmoid < best_c5_sigmoid:
                    best_h_sigmoid = h_sigmoid
                    best_c5_sigmoid = c5_sigmoid
        except:
            pass

    # Phase E: Exact polishing with SLSQP
    h_final = best_h
    c5_final = best_c5
    if best_h_sigmoid is not None and best_c5_sigmoid < best_c5:
        h_final = best_h_sigmoid
        c5_final = best_c5_sigmoid
    try:
        # run final SLSQP with tight tolerances
        dx = 2.0 / len(h_final)
        bounds_h = [(0, 1) for _ in range(len(h_final))]
        def obj_h(h):
            return evaluate_h(h)
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - 1.0}]
        result = minimize(obj_h, h_final, method='SLSQP', bounds=bounds_h, constraints=constraints, options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-10})
        if result.success:
            h_final = result.x
            c5_final = evaluate_h(h_final)
    except:
        pass

    return list(h_final), c5_final, len(h_final)
