import numpy as np
from scipy.signal import correlate
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_onto_box_and_hyperplane(y, target):
    """Project vector y onto box [0,1] and hyperplane sum = target"""
    # Binary search for lambda
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0, 1)

def compute_gradient(h, shift):
    """Compute gradient of correlation for given shift"""
    n = len(h)
    if shift >= 0:
        term1 = np.empty(n)
        term1[:n-shift] = 1 - h[shift:]
        term1[n-shift:] = 1
        term2 = np.empty(n)
        term2[shift:] = h[:n-shift]
        term2[:shift] = 0
    else:
        s = -shift
        term1 = np.empty(n)
        term1[s:] = 1 - h[:n-s]
        term1[:s] = 1
        term2 = np.empty(n)
        term2[:n-s] = h[s:]
        term2[n-s:] = 0
    return term1 - term2

        ramp = np.linspace(0.1, 0.9, n_points)
        ramp = ramp * (n_points/2) / np.sum(ramp)
        initial_shapes.append(ramp)
        
        # Sinusoidal
        sinus = 0.5 + 0.3 * np.sin(np.pi * np.linspace(0, 2, n_points))
        sinus = sinus * (n_points/2) / np.sum(sinus)
        initial_shapes.append(sinus)
        
        # Random
        rand = np.random.uniform(0, 1, n_points)

        initial_shapes.append(rand)

        for h_init in initial_shapes:

            for it in range(max_iter):

                    break
                    
                iteration_count += 1
                
                # Compute all correlations
                corr = correlate(h, 1 - h, mode="full")

                max_ind = np.where(corr == max_corr)[0]
                shift = max_ind[0] - (len(h) - 1)
                
                # Compute gradient for this shift
                grad = compute_gradient(h, shift)
                
                # Step size with decay
                step = base_step / (1 + it / 1000)
                
                # Update
                h_new = h - step * grad
                
                # Project

            if restart % 5 == 0:
                print(f"Restart {restart}, current best C5: {best_c5:.6f}")

    return (best_h.tolist(), best_c5, n_points)
