import numpy as np
from scipy.optimize import minimize
import time

# Constants
n_points = 80
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_to_simplex_box(y, target):
    """Project y onto the set {x : 0 <= x <= 1, sum(x) = target}."""
    lo = min(y) - 1.0
    hi = max(y)
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2, 0, 1)
    # Adjust rounding errors
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0, 1)
    return x

def compute_gradient_smooth_max(h, C, gamma):
    """Compute gradient of smooth maximum of correlation."""
    p = np.exp(gamma * C)
    p = p / p.sum()
    n = len(h)
    grad = np.ones(n)  # constant part corresponds to sum p_s = 1
    for idx, weight in enumerate(p):
        if weight == 0:
            continue
        s = idx - (len(C) // 2)
        if s >= 0:
            grad[:n-s] -= weight * h[s:]
            grad[s:] -= weight * h[:n-s]
        else:
            s_abs = -s
            grad[s_abs:] -= weight * h[:n-s_abs]
            grad[:n-s_abs] -= weight * h[s_abs:]
    return grad

    best_c5 = evaluate_h(h_current)
    
    # Parameters
    step_init = 0.1
    gamma_init = 0.5
    gamma_factor = 1.02
    gamma_max = 50.0
    momentum_beta = 0.9
    max_iter = 2000

    for restart in range(restarts):

            h = np.random.rand(n_points)

        step = step_init
        gamma = gamma_init
        v = np.zeros(n_points)

            if time.time() - start_time > budget_s * 0.95:

            # Compute correlation
            C = np.correlate(h, 1 - h, mode="full")
