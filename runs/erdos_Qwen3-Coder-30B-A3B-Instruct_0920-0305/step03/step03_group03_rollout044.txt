import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def proj_box_simplex(v):
    """Project vector v onto the simplex {x: sum(x) = 1, 0 <= x_i <= 1}"""
    n = len(v)
    # Sort values in descending order
    u = np.sort(v)[::-1]
    # Compute the threshold
    css = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, n+1) > (1 - css)) if np.any(u * np.arange(1, n+1) > (1 - css)) else n-1
    tau = (css[rho] - 1) / (rho + 1)
    # Compute the projection
    x = np.maximum(v - tau, 0)
    # Ensure sum = 1
    x = x / np.sum(x)
    return x

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

    dx = 2.0 / n

    remaining = budget_s - (time.time() - t0)

        return best_h, best_c5

    # Allocate time for restarts
    restart_time = remaining / R
    for r in range(R):
        if time.time() - t0 > budget_s * 0.99:
            break
        # Generate random feasible start
        u = np.random.exponential(1, n)
        h0 = u / np.sum(u)
        h0 = h0 * (1.0 / (np.sum(h0) * dx))  # Scale to sum=1
        h0 = proj_box_simplex(h0)
        # Clip to [0,1] and re-project
        h0 = np.clip(h0, 0, 1)
        h0 = proj_box_simplex(h0)

        # Set target magnitude for spectrum
        H = np.fft.fft(h0)
        M = np.sqrt(np.sum(np.abs(H[1:])**2) / (n - 1))  # Assuming first component follows
        if np.isnan(M) or M == 0:
            M = 1.0

        h_t = h0.copy()
        for t in range(it_max):
            if time.time() - t0 > budget_s * 0.99:
                break
            H = np.fft.fft(h_t)
            # Modify magnitudes
            H_new = H.copy()
            H_new[1:] = M * np.exp(1j * np.angle(H[1:]))
            h_prime = np.real(np.fft.ifft(H_new))

            # Project onto feasible set
            h_t = proj_box_simplex(h_prime)
            h_t = np.clip(h_t, 0, 1)
            h_t = proj_box_simplex(h_t)

            # Check convergence
            c5 = evaluate_c5(h_t)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_t.copy()
            if t % 10 == 0 and t > 0:
                prev_c5 = evaluate_c5(h_t)
                if abs(prev_c5 - c5) < 1e-8:
                    break

        if best_c5 <= 0.379:
            break

    return best_h, best_c5

def binary_refinement(h):
    """Refine the solution by binary approximation followed by local swaps"""
    # Binary approximation
    n = len(h)
    b = (h > 0.5).astype(float)
    # Adjust to match sum = 1 (i.e., n/2 samples)
    current_sum = np.sum(b)
    if current_sum > n/2:
        # Need to make some 1s into 0s
        indices = np.argsort(h)
        for i in range(int(current_sum - n/2)):
            b[indices[i]] = 0
    elif current_sum < n/2:
        # Need to make some 0s into 1s
        indices = np.argsort(-h)
        for i in range(int(n/2 - current_sum)):
            b[indices[i]] = 1
            
    # Local hill-climb swaps
    swp_iter = 5 * n
    no_improvement = 0
    max_no_imp = 2 * n
    best_h = b.copy()
    best_c5 = evaluate_c5(best_h)

    for _ in range(swp_iter):
        if no_improvement > max_no_imp:
            break
        i1 = np.random.choice(n)
        i0 = np.random.choice(n)
        if i0 == i1:
            continue
        # Try swap
        h_try = best_h.copy()
        h_try[i0], h_try[i1] = h_try[i1], h_try[i0]
        
        # Enforce constraint
        h_try = proj_box_simplex(h_try)
        h_try = np.clip(h_try, 0, 1)
        h_try = proj_box_simplex(h_try)
        
        c5_try = evaluate_c5(h_try)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
            no_improvement = 0
        else:
            no_improvement += 1

    return best_h, best_c5
