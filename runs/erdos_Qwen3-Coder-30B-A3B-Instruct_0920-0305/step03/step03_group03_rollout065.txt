import numpy as np
from scipy.fft import fft, ifft
import time

def proj_box_simplex(v):
    """Project vector v onto the intersection of [0,1]^n and {x: sum x = 1}"""
    v = np.array(v)
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    # Compute cumulative sums
    su = np.cumsum(u)
    # Find the largest k such that 1/k * (sum u_i - 1) > u_k
    rho = (su - 1) / np.arange(1, n + 1)
    k = np.argmax(u > rho)
    if k == 0:
        if u[0] > 1:
            tau = np.max(u - 1)
        else:
            tau = 0
    else:
        tau = rho[k-1]
    # Project
    x = np.maximum(v - tau, 0)
    # Renormalization to sum to 1
    return x / np.sum(x)


def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx


def spectral_search(n, restarts=30, max_iter=500, seed=42, budget_s=300):
    """Primary algorithm - alternating projections in the frequency domain"""
    t0 = time.time()
    if n % 2 != 0:
        n += 1  # ensure even for FFT symmetry
    dx = 2.0 / n
    m = n / 2  # desired sum = 1
    best_c5 = float('inf')
    best_h = None
    np.random.seed(seed)
    # Memory for "good things"
    history = []

    for r in range(restarts):
        if time.time() - t0 > budget_s:
            break
        # Generate feasible initial point
        u = np.random.exponential(1, n)
        h0 = u / np.sum(u)
        h0 = h0 * (1 / dx)  # Scale to integral 1
        h0 = proj_box_simplex(h0)
        h = h0.copy()
        # Compute the initial target magnitude
        H = fft(h)
        M = np.mean(np.abs(H[1:])**2)
        if M > 0:
            M = np.sqrt(M)
        else:
            M = 1.0
        # Inner loop
        old_c = float('inf')
        for t in range(max_iter):
            if time.time() - t0 > budget_s:
                break
            H = fft(h)
            H_hat = H.copy()
            H_hat[1:] = M * np.exp(1j * np.angle(H[1:]))
            h_new = ifft(H_hat).real
            # Project to feasible set
            h = proj_box_simplex(h_new)
            # Compute C5
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if abs(c5 - old_c) < 1e-8:
                break
            old_c = c5
    return best_h, best_c5


def binary_refinement(h, niter=5 * len(h)):
    """Refine high-quality solution by locally swapping 0-1 entries"""
    h = np.array(h)
    n = len(h)
    # Convert to binary
    b = (h > 0.5).astype(int)
    # Adjust to correct number of 1's (m)
    m = n // 2
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    # Ensure exactly m ones
    if len(ones) > m:
        to_zeros = ones[:len(ones) - m]
        b[to_zeros] = 0
    elif len(ones) < m:
        to_ones = zeros[:m - len(ones)]
        b[to_ones] = 1
    # Binary hill climb
    current = b.copy()
    current_c5 = evaluate_c5(current.astype(float))
    improved = True
    cycles_without_improvement = 0
    cycles = 0
    while improved and cycles_without_improvement < 2 * n and cycles < niter:
        improved = False
        # Sample a '1' and a '0'
        ones = np.where(current == 1)[0]
        zeros = np.where(current == 0)[0]
        if len(ones) > 0 and len(zeros) > 0:
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            # Swap
            candidate = current.copy()
            candidate[i] = 0
            candidate[j] = 1
            candidate_c5 = evaluate_c5(candidate.astype(float))
            if candidate_c5 < current_c5:
                current = candidate
                current_c5 = candidate_c5
                improved = True
                cycles_without_improvement = 0
            else:
                cycles_without_improvement += 1
        cycles += 1
    # Final polishing with projected gradient on continuous version
    h_refined = current.astype(float)
    for i in range(100):
        if time.time() - time.time() > 30:
            break
        c5_before = evaluate_c5(h_refined)
        # Gradient estimate by finite differences
        eps = 1e-6
        grad = np.zeros_like(h_refined)
        for k in range(n):
            h_plus = h_refined.copy()
            h_plus[k] += eps
            # Re-normalize
            h_plus = h_plus / np.sum(h_plus) * (n/2)
            # Clip
            h_plus = np.clip(h_plus, 0, 1)
            c_plus = evaluate_c5(h_plus)
            grad[k] = (c_plus - c5_before) / eps
        # Update
        lr = 0.001
        h_refined = h_refined - lr * grad
        # Project
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined / np.sum(h_refined) * (n/2)
    return h_refined, evaluate_c5(h_refined)


def fallback_random_search(n, n_samples=200, local_iter=50):
    """Fallback to random search and a few gradient steps"""
    best_c5 = float('inf')
    best_h = None
    np.random.seed(42)
    for _ in range(n_samples):
        u = np.random.exponential(1, n)
        h = u / np.sum(u)
        h = h * (1.0 / (2.0 / n))  # Integral = 1
        h = proj_box_simplex(h)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    if best_h is None:
        return None, float('inf')
    # Local polishing
    h = best_h.copy()
    for _ in range(local_iter):
        c5_before = evaluate_c5(h)
        eps = 1e-6
        grad = np.zeros_like(h)
        for k in range(n):
            h_plus = h.copy()
            h_plus[k] += eps
            h_plus = np.clip(h_plus, 0, 1)
            # Renormalize sum to m=1
            h_plus = h_plus / np.sum(h_plus) * (n/2)
            c_plus = evaluate_c5(h_plus)
            grad[k] = (c_plus - c5_before) / eps
        lr = 0.01
        h = h - lr * grad
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n/2)
        c5_after = evaluate_c5(h)
        if c5_after > c5_before:
            break
    return h, evaluate_c5(h)
