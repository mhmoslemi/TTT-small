import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import SCS
import random


def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

            cp.sum(h) == target_sum,  # ∑ h_i = n/2
            0 <= h, h <= 1,  # box constraints
            X >> 0,  # X is PSD
        ]
        
        # Add McCormick envelopes for bilinear terms (only part of matrix needed)
        # X_ij ≤ h_i, X_ij ≤ h_j, X_ij ≥ h_i + h_j - 1
        for i in range(n):
            for j in range(i, n):  # Only upper triangle
                # XC_i_j = h_i h_j formulation is only valid when h_i,h_j ∈ [0,1]
                # For matrix pairs that make a valid constraint in overlap:
                # we can constrain:
                #   X[i,j] ≤ h[i],
                #   X[i,j] ≤ h[j],
                #   X[i,j] ≥ h[i] + h[j] - 1.
                m = i + 1
                if j >= m:
                    # Specific overlap term (only around valid indices)
                    # Need to include in binary products affinity
                    constraints.append(X[i, j] <= h[i])
                    constraints.append(X[i, j] <= h[j])
                    constraints.append(X[i, j] >= h[i] + h[j] - 1)
        
        # Overlap constraints for all shifts s = 1,...,n-1
        for s in range(1, n):
            overlap_expr = sum(X[i, i + s] for i in range(n - s))
            constraints.append(overlap_expr >= t)
        
        # Objective
        prob = cp.Problem(cp.Maximize(t), constraints)
        
        # Solve with time limit
        time_limit = min(150.0, budget_s * 0.25)
        prob.solve(solver=SCS, max_iters=3000, verbose=False, time_limit=time_limit)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sdp_val = h.value

        print("SDP failed; using fallback strategy")
        h_sdp_val = np.array(initial_h_values)
        h_sdp_val = h_sdp_val * (target_sum / np.sum(h_sdp_val))
        X_sdp_val = None
        t_opt = None
    
    # Phase B: Rounding: Use X_sdp_val if available
    candidates = [h_sdp_val]
    
    if X_sdp_val is not None:
        try:
            # Eigen-decomposition of the relaxed solution to extract low-rank structure
            # Works better with cvxpy solution
            U, s = np.linalg.eigh(X_sdp_val)
            # Keep only significant components
            r = min(20, n)
            U = U[:, s > 1e-8]
            s = s[s > 1e-8]
            if len(s) > 0:
                Y = U @ np.diag(np.sqrt(s))  # Y * Y^T ≈ X
            else:
                Y = np.zeros((n, 1))
            
            for _ in range(max_rounding_trials):
                if time.time() - start_time > budget_s * 0.9:
                    break
                g = np.random.randn(Y.shape[1])
                y = Y @ g
                h_bin = np.zeros(n)
                # We want sum(h_bin) = target_sum. So determine the top (n/2) indices.
                indices = np.argsort(y)[::-1][:int(target_sum)]
                h_bin[indices] = 1
                # Now refined rounding with local edge swapping to improve overlap
                h_refined = h_bin.copy()
                swapped = 0
                last_delta = 0
                for _ in range(200):

                    # Get max index from full output (index 0 from non-zero shift indices)

                    shift_idx = s_max - len(corr)//2

                            old_val = h_refined[other_idx + shift_idx] * (1 - h_refined[other_idx])
                            delta = (old_val + swapped_val) - (new_val + orig_val)
                            if delta > best_delta:
                                best_delta = delta
                                best_i = i
                    if best_i >= 0:
                        # perform swap
                        for j in range(len(h_refined)):

        """
        Smooth max approximation:
        max_i f_i(h) ≈ (1/alpha) * log(sum_i exp(alpha * f_i(h)))
        """
        # Evaluate all shifts
        dx = 2.0 / n
        overlaps = []
        for k in range(1, n):
            total = 0
            for i in range(n):
                if i+k >= n: break
                total += h[i] * (1 - h[i+k])
            overlaps.append(total * dx)
        
        # Softmax version
        if len(overlaps) == 0: return 0
        log_vals = np.array(overlaps) * alpha
        max_log = np.amax(log_vals)
        log_sum = max_log + np.log(np.sum(np.exp(log_vals - max_log)))
        return log_sum / alpha

        """Gradient of smooth max function"""
        dx = 2.0 / n
        overlaps = []
        for k in range(1, n):
            total = 0
            for i in range(n):
                if i+k >= n: break
                total += h[i] * (1 - h[i+k])
            overlaps.append(total * dx)
        
        # softmax weights
        if len(overlaps) == 0:
            return np.zeros(n)
        log_vals = np.array(overlaps) * alpha
        max_log = np.amax(log_vals)
        exp_vals = np.exp(log_vals - max_log)
        weights = exp_vals / np.sum(exp_vals)
        
        # gradient computation
        grad = np.zeros_like(h)
        for k in range(1, n):
            for i in range(n):
                if i+k >= n:
                    break
                if i < len(grad):
                    # derivative w.r.t. to h_i
                    grad[i] += weights[k-1] * (1 - h[i+k])
                    # derivative w.r.t. to h_{i+k}
                    if i+k < len(grad):
                        grad[i+k] -= weights[k-1] * h[i]
        return grad

    best_c5 = np.inf

            res = minimize(
                obj,
                h_trial,
                method='trust-constr',
                jac=grad,
                bounds=[(0, 1) for _ in range(n)],
                constraints=[
                    {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                ],
                options={'maxiter': max_refine_iter}
            )
            if res.success:
                h_refined = res.x
                # Clamp to [0,1]
                h_refined = np.clip(h_refined, 0, 1)
                # Normalize again to weight
                h_refined = h_refined * (target_sum / np.sum(h_refined))
                # Evaluate C5
                c5_val = evaluate_c5(h_refined)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_refined
        except Exception as e:
            # Keep original if no refinement worked
            c5_val = evaluate_c5(h_trial)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_trial

    if best_h is None:
        # Clone previous iterations approach as fallback
        h0 = np.array(initial_h_values)
        n_points = len(h0)
        dx = 2.0 / n_points
        
        # Normalize initial to correct integral
        h0 = h0 * (target_sum / np.sum(h0))
        best_c5 = evaluate_c5(h0)
        best_h = h0
    
    return (best_h.tolist(), best_c5, n)
