import numpy as np

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):

    # Parameterize h by m = 10 blocks
    m = 10

        def objective_block(u):
            # u is m-1 dimensional, last one is computed
            u_full = np.append(u, max(0, min(1, m/2 - np.sum(u))))
            if not (0 <= u_full[-1] <= 1):
                return float('inf')
            # Expand to full vector
            h_expanded = np.repeat(u_full, n_points // m)
            # Pad or truncate to exact n_points
            if len(h_expanded) > n_points:
                h_expanded = h_expanded[:n_points]
            elif len(h_expanded) < n_points:
                h_expanded = np.pad(h_expanded, (0, n_points - len(h_expanded)), mode='constant')
            err = evaluate_c5(h_expanded)
            return err
        
        bounds = [(0, 1) for _ in range(m-1)]
        try:
            # Use a local method for quick initial guess

            # Oversample to higher resolution (n=200)
            h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
            u_guess = np.histogram(np.linspace(0, 2, n_points), bins=m, weights=h0)[0]
            u_guess = u_guess / (n_points / m)

            u_guess = u_guess * (n_points / 2) / np.sum(u_guess)
            # Refine with simple search
            max_evals = 1500
            for _ in range(max_evals):
                # Random perturbation around current best
                u_current = u_guess + np.random.normal(0, 0.05, m-1)
                u_current = np.clip(u_current, 0, 1)
                c5 = objective_block(u_current)
                if c5 < stage_a_best_c5:
                    stage_a_best_c5 = c5
                    stage_a_best_h = u_current.copy()
        except:
            pass
    
    # Final clean-up and reconstruction
    if stage_a_best_h is not None:
        u_full = np.append(stage_a_best_h, max(0, min(1, m/2 - np.sum(stage_a_best_h))))
        if 0 <= u_full[-1] <= 1:
            h_expanded = np.repeat(u_full, n_points // m)
            if len(h_expanded) > n_points:
                h_expanded = h_expanded[:n_points]
            elif len(h_expanded) < n_points:
                h_expanded = np.pad(h_expanded, (0, n_points - len(h_expanded)), mode='constant')
            stage_a_best_h = h_expanded.copy()
            stage_a_best_c5 = evaluate_c5(h_expanded)
            
            if stage_a_best_c5 < best_c5:
                best_c5 = stage_a_best_c5
                best_h = stage_a_best_h.copy()
    
    # Stage B: Full resolution projected gradient descent
    if (time.time() - start_time) < budget_s * 0.6 and best_h is not None:
        def project_h(h, target_sum):
            # Water-filling projection
            y = h.copy()
            eps = 1e-12
            low = -2.0
            high = 3.0
            for _ in range(100):
                mid = (low + high) / 2.0
                x = np.clip(y - mid, 0, 1)
                current_sum = np.sum(x)
                if current_sum > target_sum:
                    low = mid
                else:
                    high = mid
            mid = (low + high) / 2.0
            x = np.clip(y - mid, 0, 1)
            # Normalize to ensure exact sum
            scale = target_sum / np.sum(x)
            x = x * scale
            return x

        # Current best from stage A
        h = best_h.copy()
        target_sum = n_points / 2.0
        max_iter = 3000
        step_size = 0.01
        for t in range(max_iter):
            # Find worst shift
            corr = correlate(h, 1 - h, mode='full')
            max_k_idx = np.argmax(corr)
            max_shift = max_k_idx - (n_points - 1)
            
            # Compute subgradient
            g = np.zeros(n_points)
            # Indices for shift max_shift:
            if max_shift >= 0:
                g[:n_points - max_shift] = (1 - h[max_shift:]) - h[:n_points - max_shift]
            else:
                g[-max_shift:] = (1 - h[:n_points + max_shift]) - h[-max_shift:]
            
            # Gradient step
            h_new = h - step_size * g
            
            # Project
            h = project_h(h_new, target_sum)
            
            # Check improvement
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            if t % 100 == 0:
                if time.time() - start_time > budget_s * 0.9:
                    break

    # Stage C: Binary rounding + local search
    if (time.time() - start_time) < budget_s * 0.9 and best_h is not None:
        def binary_round(h, target_sum):
            # Create probability vector
            probs = h.copy()
            # Generate binary vector with exact weight
            sorted_indices = np.argsort(probs)[::-1]
            binary = np.zeros_like(probs)
            ones_count = int(target_sum)
            for idx in sorted_indices[:ones_count]:
                binary[idx] = 1
            return binary

        def get_overlap_delta(b, i, j, shifts_dict):
            # Compute the change in overlap sums
            # if b[i] = 1 and b[j] = 0
            # Found all shifts k such that contribution of i and/or j to shift k is 1 before swap
            # which means i,j or i+k,j+k are both 1 or 0
            old_val = 0
            delta_val = 0
            for k in shifts_dict:
                # contributions of old bit values
                old_contribution_i = b[i] * b[i + k] if i + k < n_points else 0
                old_contribution_j = b[j] * b[j + k] if j + k < n_points else 0
                # after swap
                new_contribution_i = (1 - b[i]) * b[i + k] if i + k < n_points else 0
                new_contribution_j = b[j] * (1 - b[j + k]) if j + k < n_points else 0
                old_contribution = old_contribution_i + old_contribution_j
                new_contribution = new_contribution_i + new_contribution_j
                # The changing term in 1 - overlap = 1 - sum_ h_i h_{i+k}
                d = (1 - new_contribution) - (1 - old_contribution)
                delta_val = max(delta_val, d)
            return delta_val

        def swap_move(b, i, j, shifts_dict, corr_dict):
            b = b.copy()
            b[i] = 1 - b[i]  # swap
            b[j] = 1 - b[j]  # swap
            return b

        def compute_shifts_dict(b):
            # Calculate all contributions per shift
            shifts_dict = {}
            for k in range(1, n_points):
                shifts_dict[k] = []
                for i in range(n_points):
                    if i + k < n_points:
                        if b[i] == 1 and b[i + k] == 1:
                            shifts_dict[k].append(i)
            return shifts_dict

        # Convert continuous to binary multiple times
        candidates = []
        K = 30
        for _ in range(K):
            binary_h = binary_round(best_h, n_points / 2)
            c5_binary = evaluate_c5(binary_h)
            candidates.append((binary_h.copy(), c5_binary))
        
        candidates.sort(key=lambda x: x[1])
        best_binary = candidates[0][0]
        best_binary_c5 = candidates[0][1]
        
        if best_binary_c5 < best_c5:
            best_c5 = best_binary_c5
            best_h = best_binary.copy()
        
        # Local search with swapping
        bs = best_binary.copy()
        max_iterations = 100
        informed_swap_bit = 100  # Do a few rounds with recent swaps
        
        # Precompute all shift indices where contributions affect the overlap sums
        shifts_dict = compute_shifts_dict(bs)
        corr_vals = correlate(bs, 1 - bs, mode='full')
        max_k_idx = np.argmax(corr_vals)
        max_shift = max_k_idx - (n_points - 1)

        # Local improvement via swaps
        for iter_ in range(max_iterations):
            if time.time() - start_time > budget_s * 0.98:
                break
            improved = False
            for _ in range(1000):  # Sufficient number of pairs sampled
                i = np.random.randint(0, n_points)
                j = np.random.randint(0, n_points)
                if bs[i] == 1 and bs[j] == 0:
                    delta_c5 = get_overlap_delta(bs, i, j, shifts_dict)
                    if delta_c5 < 0:
                        bs = swap_move(bs, i, j, shifts_dict, corr_vals)
                        improved = True
                        break
            if not improved:
                break

        current_c5 = evaluate_c5(bs)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = bs.copy()

    # Stage D: If total time allows, run MILP-based refinement
    if (time.time() - start_time) < budget_s - 10:
        try:
            # Setup MILP for the discrete case with binary variables
            h_vars = [0] * n_points  # dummy declaration
        
            # Reconstruct binary solution from best_h
            if best_h is not None:
                binary_h = (best_h > 0.5).astype(int)
                actual_sum = np.sum(binary_h)
                if actual_sum > n_points / 2:
                    # Find 'actual_sum - n_points/2' elements that can be set to 0 to fix sum
                    # sample indices from those with higher values (to preserve shape)
                    to_zero_idx = np.argsort(best_h)[:actual_sum - n_points // 2]
                    for i in to_zero_idx:
                        binary_h[i] = 0
                elif actual_sum < n_points / 2:
                    # Increase the number of 1s
                    to_one_idx = np.argsort(best_h)[-(n_points // 2 - actual_sum):]
                    for i in to_one_idx:
                        binary_h[i] = 1
            else:
                # Fallback - take from stage A if exists
                binary_h = np.zeros(n_points)
                binary_h[:n_points // 2] = 1
                
            # Re-evaluate the exact binary bound directly with correlation
            # Because the MILP model is nothing but an approximation
            binary_h = np.array(binary_h)
            if np.sum(binary_h) != n_points / 2:
                # Adjustment to enforce equality
                ones = np.where(binary_h == 1)[0]
                zeros = np.where(binary_h == 0)[0]
                # swap some ones for zeros or vice versa
                required_adjustments = int(np.sum(binary_h) - n_points / 2)
                if required_adjustments > 0:
                    # Loss one for every increased adjustment, so we must find zeros
                    # and set them as ones
                    # But we just take the last few zeros and check
                    for i in range(required_adjustments):
                        if len(zeros) > 0:
                            idx = zeros[-1]
                            binary_h[idx] = 1
                            zeros = zeros[:-1]
                if required_adjustments < 0:
                    # Need to make more zeros
                    for i in range(-required_adjustments):
                        if len(ones) > 0:
                            idx = ones[-1]
                            binary_h[idx] = 0
                            ones = ones[:-1]
            # Suppress the calling Correlation again, since double work
            corr = correlate(binary_h, 1 - binary_h, mode="full")
            milp_c5 = np.max(corr) * dx
            if milp_c5 < best_c5:
                best_c5 = milp_c5
                best_h = binary_h
        except:
            pass

    # Final check
    if best_h is None:
        # Fallback to initial values
        best_h = np.array(initial_h_values)
        best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        best_c5 = evaluate_c5(best_h)
    
    # Ensure exactly n_points
    if len(best_h) != n_points:
        best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
    
    return (best_h.tolist(), best_c5, n_points)
