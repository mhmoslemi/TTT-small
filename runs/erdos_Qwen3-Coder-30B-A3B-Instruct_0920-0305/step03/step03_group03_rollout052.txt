def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def binary_milp(n, time_limit_s):
    """Solve the binary integer program for a given n using CVXPY"""
    m = n // 2
    dx = 2.0 / n
    
    # Variables
    b = Variable(n, boolean=True)  # binary values
    t = Variable()  # lower bound on overlaps
    y = Variable((n, n), boolean=True)  # products b_i * b_j
    
    # Constraints
    constraints = [
        sum_entries(b) == m,  # weight constraint
        t >= 0  # t non-negative
    ]
    
    # Product linearization
    for i in range(n):
        for j in range(i, n):
            # y[i,j] <= b[i], y[i,j] <= b[j]
            constraints.append(y[i, j] <= b[i])
            constraints.append(y[i, j] <= b[j])
            # y[i,j] >= b[i] + b[j] - 1
            constraints.append(y[i, j] >= b[i] + b[j] - 1)
            if i != j:
                constraints.append(y[j, i] == y[i, j])
    
    # Overlap constraints for each shift s
    for s in range(n):
        # The overlap for shift s is sum_{i=0}^{n-1-s} y[i, i+s]
        overlap_expr = sum_entries(y[0:n-s, s:n])
        constraints.append(overlap_expr >= t)
    
    # Objective
    objective = Minimize(-t)  # we maximize t, so minimize -t
    prob = Problem(objective, constraints)
    
    # Solve with time limit
    try:
        prob.solve(solver=cp.CBC, time_limit=time_limit_s)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZABLE]:
            t_val = -prob.value
            b_sol = np.round(b.value).astype(int)
            c5_tight = (m - t_val) * dx
            return b_sol, c5_tight
        else:
            return None, None
    except:
        return None, None

def project_and_refine(h0, max_iter=2000, tolerance=1e-8):
    """Refine h0 using projected sub-gradient descent"""
    n = len(h0)
    dx = 2.0 / n
    h_current = h0.copy()
    
    for iter_num in range(max_iter):
        corr = correlate(h_current, 1 - h_current, mode="full")
        max_corr = np.max(corr)
        max_indices = np.where(corr == max_corr)[0]
        
        # Compute sub-gradient for active shifts
        g = np.zeros(n)
        for s in max_indices:
            for i in range(n):
                j = i + s
                if j < n:
                    g[i] += 1 - h_current[j]
                if i >= s:
                    j = i - s
                    if j < n:
                        g[i] -= h_current[j]
        g /= len(max_indices)
        
        # Step size
        alpha = 1e-3 / np.sqrt(iter_num + 1)
        h_new = h_current - alpha * g
        
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project to simplex
        scale_factor = 1.0 / (np.sum(h_new) * dx)
        h_new = h_new * scale_factor
        
        # Check for convergence
        improvement = evaluate_c5(h_current) - evaluate_c5(h_new)
        if improvement < tolerance:
            break
        h_current = h_new
    
    return h_current
