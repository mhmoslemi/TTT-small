import numpy as np

def proj_box_simplex(v):
    """Project v onto the intersection of [0,1]^n and the simplex {x: sum x_i = 1}"""
    # Sort values in descending order
    v_sorted = np.sort(v)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(v_sorted)
    # Find the threshold index
    idx = np.arange(1, len(v_sorted) + 1)
    cond = v_sorted - (1 - cumsum) / idx > 0
    if not np.any(cond):
        # All values must be zero
        return np.zeros_like(v)
    k = np.argmax(cond)
    # Compute threshold
    tau = (1 - cumsum[k-1]) / k
    # Apply thresholding and clip to [0,1]
    result = np.clip(v - tau, 0, 1)
    return result

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

            break

        u = np.random.exponential(1, n)

        last_c5 = float('inf')

            # FFT
            H = np.fft.fft(h_current)
            
            # Modify magnitude to enforce constant non-zero magnitudes
            H_new = H.copy()
            H_new[1:] = M * np.exp(1j * np.angle(H[1:]))
            
            # Inverse FFT
            h_tilde = np.real(np.fft.ifft(H_new))
            
            # Project back onto feasible set
            h_new = proj_box_simplex(h_tilde)
            
            h_current = h_new
            
            # Evaluate

                best_c5 = c5
                best_h = h_current.copy()
                
            if abs(c5 - last_c5) < 1e-8:
                break
            last_c5 = c5

    return best_h, best_c5

def binary_refinement(h):
    """Given a continuous vector h, refine it to a near-binary structure and improve C5"""
    n = len(h)
    # First, sort and get threshold to make sum = n/2
    sorted_indices = np.argsort(h)[::-1]
    target_sum = n / 2
    # Count how many items to set to 1
    count_ones = int(np.round(target_sum))
    binary = np.zeros(n)
    binary[sorted_indices[:count_ones]] = 1
    
    # Do swaps to minimize C5
    h_current = binary.copy()
    c5_best = evaluate_c5(h_current)
    num_swaps = 0
    max_swaps = 5 * n
    no_improve_count = 0
    max_no_improve = 2 * n
    
    while num_swaps < max_swaps and no_improve_count < max_no_improve:
        if np.sum(h_current) != n / 2:
            # Force correct sum by adjusting threshold
            thresh = np.partition(h_current, n - int(n/2))[n - int(n/2)]
            h_current = (h_current >= thresh).astype(float)
            
        # Try a swap
        idx0 = np.where(h_current == 0)[0]
        idx1 = np.where(h_current == 1)[0]
        
        if len(idx0) == 0 or len(idx1) == 0:
            break
            
        i0 = np.random.choice(idx0)
        i1 = np.random.choice(idx1)
        
        # Do the swap
        h_test = h_current.copy()
        h_test[i0] = 1
        h_test[i1] = 0
        
        c5_test = evaluate_c5(h_test)
        
        if c5_test < c5_best:
            h_current = h_test
            c5_best = c5_test
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        num_swaps += 1
    
    return h_current, c5_best
