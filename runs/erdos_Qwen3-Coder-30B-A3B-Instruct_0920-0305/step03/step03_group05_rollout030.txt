```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

# Constants
n_points = 48
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def evaluate_h_binary(h_values):
    """Evaluate the C_5 bound for binary h values."""
    h = np.array(h_values, dtype=int)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def local_search(h_values, max_iter=200):
    """Refine solution using local optimization."""
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    def objective(h):
        return evaluate_h(h)
    
    try:
        result = minimize(objective, h_values, method='SLSQP', 
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-10})
        if result.success:
            return result.x
    except:
        pass
    return h_values

def run(seed=42, budget_s=600):
    """Main optimization routine."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize binary solution with equal number of 1s and 0s
    h_current = np.zeros(n_points, dtype=int)
    ones_indices = np.random.choice(n_points, size=n_points//2, replace=False)
    h_current[ones_indices] = 1
    
    # Ensure that sum is exactly n/2
    assert np.sum(h_current) == n_points // 2
    
    # Evaluate initial solution
    c5_current = evaluate_h_binary(h_current)
    
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # Improvement heuristic: attempt swaps to reduce minimum correlation
    def improve_with_swaps(h, max_iterations=5000):
        h = h.copy()
        best_c5 = evaluate_h_binary(h)
        
        # Precompute pairs for quick access
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        ones_list = list(ones)
        zeros_list = list(zeros)
        
        for _ in range(max_iterations):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Select random one and zero
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
                
            i = np.random.choice(ones_list)
            j = np.random.choice(zeros_list)
            
            # Perform swap
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            
            # Evaluate new solution
            c5_new = evaluate_h_binary(h_new)
            
            # Accept if better or with probability
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                # Update indices for fast lookup
                ones_list.remove(i)
                ones_list.append(j)
                zeros_list.remove(j)
                zeros_list.append(i)
                
                if best_c5 < target_bound:
                    break
            elif np.random.rand() < 0.05:
                h = h_new
                # Update indices for fast lookup after acceptance
                ones_list.remove(i)
                ones_list.append(j)
                zeros_list.remove(j)
                zeros_list.append(i)
                
        return h, best_c5

    # Local improvements
    iter_num = 0
    max_iter = 500

    # Run improvement heuristics
    while time.time() - start_time < budget_s * 0.95:
        h_improved, c5_improved = improve_with_swaps(h_current, max_iterations=3000)
        
        if c5_improved < best_c5:
            best_h = h_improved.copy()
            best_c5 = c5_improved
            
        if best_c5 < target_bound:
            break
        
        # Occasionally try a perturbation before optimization
        h_current = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
        ones_indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h_current[ones_indices] = 1
        
        # Refine with gradient-based method
        opts = {'maxiter': 50, 'ftol': 1e-6, 'gtol': 1e-6}
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_eq}
        
        def obj(h):
            return evaluate_h(h)
        
        try:
            result = minimize(obj, h_current, method='SLSQP', bounds=bounds,
                             constraints=constraints, options=opts)
            if result.success:
                h_current = result.x
        except:
            pass
        
        iter_num += 1

    # Final refinement step using SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}

    def objective(h):
        return evaluate_h(h)

    try:
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds,
                         constraints=constraints, options={'maxiter': 150, 'ftol': 1e-15, 'gtol': 1e-12})
        if result.success:
            best_h = result.x
    except:
        pass

    final_c5 = evaluate_h(best_h)
    
    # Normalize solution to satisfy integral constraint
    final_sum = np.sum(best_h) * dx
    if abs(final_sum - 1.0) > 1e-6:
        best_h *= (1.0 / final_sum)

    return list(best_h), final_c5, n_points
```
