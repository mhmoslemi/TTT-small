import numpy as np
from scipy.optimize import minimize
import time

# Constants
n_points = 50
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values using correlation."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def smooth_max(values, alpha=20.0):
    """Smooth approximation to max using log-sum-exp."""
    max_val = np.max(values)
    # subtract for numerical stability
    exps = np.exp(alpha * (values - max_val))
    return max_val + np.log(np.sum(exps)) / alpha

def smooth_objective(h_values):
    """Smoothed objective for optimization."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Smooth maximum
    smooth_value = smooth_max(correlation)
    return smooth_value

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def run(seed=42, budget_s=600):
    """Main optimization routine."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction (if available)

        best_h[-10:] = 0.0
        # Normalize to integral 1
        best_h = best_h * (1.0 / np.sum(best_h)) * (n_points / 2)  # adjust for target sum = n/2
        best_h = np.clip(best_h, 0, 1)  # Ensure in [0,1]

    best_c5 = evaluate_h(best_h)
    
    # Objective function for optimization (smoothed version)
    def opt_func(h):
        return smooth_objective(h)
    
    # Bounds and constraints for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}

        h_start = np.random.rand(n_points)

        # Optimize using SLSQP on smoothed objective
        try:
            result = minimize(opt_func, h_start, method='SLSQP', bounds=bounds, constraints=constraints,

                h_opt = result.x
                # Adjust for exact constraint satisfaction
