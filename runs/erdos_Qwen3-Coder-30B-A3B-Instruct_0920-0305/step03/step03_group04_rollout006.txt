def evaluate_h(h_values):
    h = np.array(h_values)
    correlation = np.correlate(h, 1-h, mode="full")
    max_correlation = np.max(correlation)
    return max_correlation * dx

def correlation_vec(h):
    h = np.array(h)
    corr = np.correlate(h, 1-h, mode="full")
    return corr

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def logit(x):
    x = np.clip(x, 1e-6, 1-1e-6)
    return np.log(x / (1-x))

def proj_to_simplex(v, s=1.0):
    v = np.array(v)
    v = np.clip(v, 0, 1)
    v_sum = np.sum(v)
    if abs(v_sum - s) < 1e-9:
        return v
    if v_sum > s:
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.argmax(u * np.arange(1, len(u)+1) > (cssv - s))
        theta = (cssv[rho] - s) / (rho + 1)
        return np.maximum(v - theta, 0)
    else:
        return v * s / v_sum

def sdpl_relax_scs(n, max_time=30):
    x = cp.Variable(n)
    X = cp.Variable((n, n), PSD=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        x >= 0,
        x <= 1,
        cp.sum(x) == n/2
    ]

    # Diagonal consistency
    constraints.append(cp.diag(X) == x)
    
    # McCormick constraints (for i<j)
    for i in range(n):
        for j in range(i+1, n):
            constraints.extend([
                X[i,j] >= 0,
                X[i,j] <= x[i],
                X[i,j] <= x[j],
                X[i,j] >= x[i] + x[j] - 1
            ])
    
    # Overlap constraints
    for k in range(n):
        constraints.append(cp.sum(x[:n-k]) - cp.sum(cp.diag(X, k)) <= t)
    
    prob = Problem(Minimize(t), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-5, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value, X.value, prob.value
    except:
        pass
    return None, None, np.inf

def hyperplane_round(x_star, V, n, num_rounds=30):
    best_h = None
    best_c5 = float('inf')

    for _ in range(num_rounds):
        g = np.random.randn(V.shape[1])
        s = np.sign(V @ g)
        h = (s >= 0).astype(float)
        
        # Enforce exactly n/2 ones
        current_sum = np.sum(h)
        diff = int(current_sum - n / 2)
        if diff > 0:
            # Too many ones
            indices = np.argsort(np.abs(V @ g))
            flip_indices = indices[-diff:]
            h[flip_indices] = 0
        elif diff < 0:
            # Too few ones
            indices = np.argsort(np.abs(V @ g))
            flip_indices = indices[:abs(diff)]
            h[flip_indices] = 1
            
        c5 = evaluate_h(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    return best_h, best_c5

def hill_climb_swap_binary(h_start, n, max_swaps=50000):

            corr = correlation_vec(h)
            worst_shift = np.argmax(corr)
            mismatches = []
            for j in range(n - worst_shift):
                if h[j] == 1 and h[j + worst_shift] == 0:
                    mismatches.append(j)
            if mismatches:

                best_swap = None
                best_c5 = current_c5
                for idx in mismatches:
                    for z in zeros:
                        h_try = h.copy()
                        h_try[idx] = 0
                        h_try[z] = 1
                        c5_try = evaluate_h(h_try)
                        if c5_try < best_c5:
                            best_c5 = c5_try
                            best_swap = (idx, z)
                if best_swap:
                    i1, i2 = best_swap
                    zeros.remove(i2)
                    ones.add(i1)
                    h[i1] = 0
                    h[i2] = 1
                    current_c5 = best_c5
                    iter_since_improvement = 0
                else:
                    # Restart
                    h = np.random.rand(n) < 0.5
                    ones = set(np.where(h > 0.5)[0])
                    zeros = set(np.where(h < 0.5)[0])
                    current_c5 = evaluate_h(h)
                    iter_since_improvement = 0
            else:
                # Restart
                h = np.random.rand(n) < 0.5
                ones = set(np.where(h > 0.5)[0])
                zeros = set(np.where(h < 0.5)[0])
                current_c5 = evaluate_h(h)
                iter_since_improvement = 0
        else:
            # Normal swap
            i1 = np.random.choice(list(ones))
            i2 = np.random.choice(list(zeros))
            h_try = h.copy()
            h_try[i1] = 0
            h_try[i2] = 1
            c5_try = evaluate_h(h_try)

                ones.remove(i1)
                zeros.remove(i2)
                ones.add(i2)
                zeros.add(i1)
                h = h_try
                current_c5 = c5_try
                iter_since_improvement = 0
            else:
                iter_since_improvement += 1

    return h, current_c5

def smooth_refine(h_start, n, max_iter=300):
    alpha = 30.0
    lambda_penalty = 200.0

    def soft_max_with_penalty(y):
        h = sigmoid(y)
        corr = correlation_vec(h)
        soft_max = np.log(np.sum(np.exp(alpha * corr))) / alpha
        penalty = lambda_penalty * (np.sum(h) - n/2)**2
        return soft_max + penalty

    def grad_soft_max_with_penalty(y):
        h = sigmoid(y)
        corr = correlation_vec(h)
        exp_c = np.exp(alpha * corr)
        sum_exp = np.sum(exp_c)
        d_soft = (exp_c / sum_exp) @ (alpha * np.eye(len(corr)))
        grad_soft = d_soft @ (alpha * h * (1 - h))
        grad_pen = 2 * lambda_penalty * (np.sum(h) - n/2) * h * (1 - h)
        return grad_soft + grad_pen

    y = logit(h_start)
    res = minimize(soft_max_with_penalty, y, method='L-BFGS-B', jac=grad_soft_max_with_penalty, 
                   options={'maxiter': max_iter})
    h_final = np.clip(sigmoid(res.x), 0, 1)
    h_final = proj_to_simplex(h_final, n/2)
    return h_final, evaluate_h(h_final)
