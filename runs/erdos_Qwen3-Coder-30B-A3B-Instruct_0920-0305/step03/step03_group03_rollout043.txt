import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp
from cvxpy.reductions.solvers.defines import INSTALLED_SOLVERS

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Pre-allocated time budgets (in seconds)
    milp_budget = 0.3 * budget_s
    sa_budget = 0.3 * budget_s 
    proj_grad_budget = 0.3 * budget_s
    fallback_budget = 0.1 * budget_s

    # Initial best solution
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    best_c5 = evaluate_c5(initial_h)
    best_h = initial_h.copy()
    n_best = n_points

    # 1. Try exact MILP for small n (n=42) to get a solid baseline
    if (time.time() - start_time) < milp_budget:
        try:
            n_milp = 42
            m = n_milp // 2  # weight
            dx = 2.0 / n_milp

            # Create variables
            b = cp.Variable(n_milp, boolean=True)
            t = cp.Variable()
            y = cp.Variable((n_milp, n_milp), boolean=True)

            # Constraints
            constraints = [
                cp.sum(b) == m,
                t >= 0
            ]

            for i in range(n_milp):
                for j in range(i, n_milp):
                    constraints.extend([
                        y[i, j] <= b[i],
                        y[i, j] <= b[j],
                        y[i, j] >= b[i] + b[j] - 1
                    ])

            # Overlaps
            for s in range(n_milp):
                overlap = cp.sum(y[range(n_milp - s), range(s, n_milp)])
                constraints.append(overlap >= t)

            # Objective
            objective = cp.Maximize(t)
            prob = cp.Problem(objective, constraints)

            # Run solver
            solve_time = min(milp_budget - (time.time() - start_time), 30)
            prob.solve(solver=cp.GLPK_MI, time_limit=solve_time)

            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                t_val = t.value
                best_t = int(np.round(t_val))
                if t_val is not None:
                    C5_milp = (m - best_t) * dx
                    if C5_milp < best_c5:
                        best_c5 = C5_milp
                        best_h = b.value.astype(float)
                        n_best = n_milp
        except:
            pass

    # 2. Binary Simulated Annealing on larger grids
    sa_lengths = [44, 48, 52, 56, 60]
    sa_budget_per_length = sa_budget / len(sa_lengths)

    for n in sa_lengths:
        if (time.time() - start_time) >= budget_s - 10:
            break

        try:
            dx = 2.0 / n
            m = n // 2
            # Generate a random binary vector with exactly m ones
            h_bin = np.zeros(n, dtype=int)
            h_bin[:m] = 1
            np.random.shuffle(h_bin)

            current_c5 = evaluate_c5(h_bin.astype(float))
            best_h_bin = h_bin.copy()
            best_c5_sa = current_c5

            # Simulated Annealing
            temp = 1.0
            temp_min = 1e-8
            alpha = 0.995
            max_iter = 10000
            k = 0

            while temp > temp_min and (time.time() - start_time) < budget_s - 10:
                # Make a move (swap a 1 with a 0)
                idx1 = np.random.choice(np.where(h_bin == 1)[0])
                idx2 = np.random.choice(np.where(h_bin == 0)[0])

                h_bin[idx1], h_bin[idx2] = h_bin[idx2], h_bin[idx1]

                new_c5 = evaluate_c5(h_bin.astype(float))
                diff = new_c5 - current_c5

                if diff < 0 or np.random.rand() < np.exp(diff / temp):
                    current_c5 = new_c5
                    if current_c5 < best_c5_sa:
                        best_c5_sa = current_c5
                        best_h_bin = h_bin.copy()
                else:
                    # Revert the move
                    h_bin[idx1], h_bin[idx2] = h_bin[idx2], h_bin[idx1]

                temp *= alpha
                k += 1
                if k > max_iter:
                    break

            # Convert back to float and refine
            h_refined = best_h_bin.astype(float)
            h_refined /= np.sum(h_refined) * dx
            h_refined /= dx
            refined_c5 = evaluate_c5(h_refined)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = h_refined
                n_best = n

        except:
            continue

    # 3. Projected gradient descent (short)
    if (time.time() - start_time) < proj_grad_budget:
        def projected_grad_descent(h0, steps=1000, lr=1e-3):
            h = h0.copy()
            best_h = h.copy()
            best_c5 = evaluate_c5(h)
            dx = 2.0 / len(h)
            for i in range(steps):
                # Find active shifts
                corr = correlate(h, 1 - h, mode="full")
                max_val = np.max(corr)
                active_shifts = np.where(np.abs(corr - max_val) < 1e-8)[0]
                
                # Compute gradient using active shifts
                grad = np.zeros_like(h)
                for s in active_shifts:
                    # Compute ∂C_s/∂h_i
                    corr_h = correlate(h, (1 - h), mode="full")
                    for k in range(len(h)):
                        # This part needs careful handling
                        i = k
                        j = (k + s) % len(h)
                        # Simplify by approximate gradient
                        if i >= 0 and i < len(h) and j >= 0 and j < len(h): 
                            grad[i] += 1 - h[j]

                # Normalize gradient and perform step
                lr_eff = lr / (1 + i**0.5)  # Line search
                h = h - lr_eff * grad
                # Clip to [0,1]
                h = np.clip(h, 0, 1) 
                # Project back to simplex
                h = h / (np.sum(h) * dx) * 1.0
                new_c5 = evaluate_c5(h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            return best_h, best_c5

        try:
            # Run multiple restarts
            restarts = 5
            for i in range(restarts):
                h_start = np.random.rand(n_best)  # Random uniform
                h_start /= np.sum(h_start) * dx
                h, c5 = projected_grad_descent(h_start, steps=100, lr=0.01)
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        except:
            pass

    # 4. Final fallback if needed
    if best_c5 > 0.3807:
        try:
            h = best_h.copy()
            bounds = [(0, 1) for _ in range(n_best)]
            def objective(h):
                return evaluate_c5(h)
            
            def constraint_eq(h):
                dx = 2.0 / n_best
                return np.sum(h) * dx - 1.0
            
            cons = {'type': 'eq', 'fun': constraint_eq}
            res = differential_evolution(objective, bounds, seed=seed,
                                         maxiter=500, popsize=10,
                                         mutation=(0.5, 1), recombination=0.7)
            
            if res.success:
                h_candidate = res.x
                dx = 2.0 / n_best
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass

    return (best_h.tolist(), best_c5, n_best)
