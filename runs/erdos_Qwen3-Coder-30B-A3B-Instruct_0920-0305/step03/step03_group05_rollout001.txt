import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

# Constants
n_points = 48  # Even number for binary feasibility
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def evaluate_h_binary(h_values):
    """Evaluate the C_5 bound for binary h values (0/1)."""
    h = np.array(h_values, dtype=int)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def local_search(h_values, max_iter=200):
    """Refine solution using local optimization."""
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    def objective(h):
        return evaluate_h(h)
    
    try:
        result = minimize(objective, h_values, method='SLSQP', 
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-10})
        if result.success:
            return result.x
    except:
        pass
    return h_values

                # Too many 1s, swap some to 0
                n_excess = int(current_sum - n_points / 2)
                for i in range(n_excess):
                    if i < len(ones_indices):
                        idx = ones_indices[i]

                    if i < len(zeros_indices):
                        idx = zeros_indices[i]
                        h_current[idx] = 1
                        
        # Convert to binary (0, 1) if needed
        h_current = np.round(h_current).astype(int)
    else:
        # Initialize binary pattern: mix 0s and 1s to sum to n_points/2 (24)

    # Shuffle to get initial random binary pattern with exactly n/2 ones
    np.random.shuffle(h_current)
    
    best_h = h_current.copy()
    c5_current = evaluate_h_binary(best_h)
    best_c5 = c5_current
    
    # Simulated Annealing with binary values
    temp = 0.1
    cooling_rate = 0.9999
    max_iter = 1000000
