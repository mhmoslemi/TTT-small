import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def proj_box_simplex(v):
    """Project vector v onto the intersection of [0,1]^n and sum = 1"""
    v = np.asarray(v)
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    # Compute the threshold
    rho = 0.
    for i in range(n):
        rho += u[i] - 1.0 / n
        if rho <= 0:
            break
    # This loop computes the threshold tau such that sum of max(0, vi-tau) = 1
    # It is guaranteed to find a valid tau since initial_after_clip sums to 1
    # (for i=0..k: u[i] >= 1/n)
    # Compute sum up to k = i - 1, if k<threshold, then k+1 will also always be >= 1/n
    # Thus we always find that if rho <= 0, then tau = u[i]
    # But here we want the true threshold (k after we found threshold)
    # So compute up to where sum (u[i] - 1/n) <= 0
    i = 0
    while i < n:
        s = (u[:i+1] - 1.0/n).sum()
        if s <= 0:
            i += 1
        else:
            break
    # The correct tau
    if i == 0:
        tau = 1.0
    else:
        tau = u[i-1]

    # Project
    x = np.maximum(v - tau, 0.)
    # Correct the sum
    xsum = np.sum(x)
    if xsum > 0:
        x = x / xsum * 1.0  # force sum to 1
    else:
        # All zeros - set to uniform
        x = np.full_like(x, 1.0 / n)
    return x

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def spectral_search(n, budget_s, t0):
    # Given a fixed size n
    dx = 2.0 / n
    m = n / 2.0
    # R = number of restarts
    R = 20
    best_h = None
    best_c5 = float('inf')
    it_max = 300
    
    for r in range(R):
        if time.time() - t0 > budget_s * 0.98:
            break
            
        # Random initialization and rescale to unit sum
        u = np.random.exponential(1.0, n)
        h0 = u / np.sum(u) * m  # so that they sum to m = n/2
        h0 = np.clip(h0, 0, 1)
        h0 = proj_box_simplex(h0)  # make sure sum = 1 and all values in [0,1]
        
        # Compute M of the initial h0
        H0 = np.fft.fft(h0)
        M0 = np.mean(np.abs(H0[1:])**2)
        if M0 > 0:
            M = np.sqrt(M0)
        else:
            # fallback when all values are constant or moving to zero
            M = 1e-6
        h = h0.copy()
        
        for t in range(it_max):
            if time.time() - t0 > budget_s * 0.98:
                break
            # FFT
            H = np.fft.fft(h)
            # Modify magnitude to be constant in non DC
            H_new = H.copy()
            H_new[1:] = M * np.exp(1j * np.angle(H[1:]))
            # Inverse FFT
            hhat = np.real(np.fft.ifft(H_new))
            # Project back onto feasible domain
            h = proj_box_simplex(hhat)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if t > 10 and np.abs(c5 - evaluate_c5(h)) < 1e-10:
                break
                
    return best_h, best_c5

def binary_refinement(h):
    """Takes continuous vector h from spectral method and refines by first converting to binary and then hill-climbing"""
    n = len(h)
    # Get binary pattern with threshold 0.5
    b = (h > 0.5).astype(float)
    # Adjust to ensure exactly n/2 elements are 1
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    if len(ones) > n//2:
        # We have too many ones; make smallest values become 0
        to_zero = np.argsort(h[ones])[:len(ones) - n//2]
        for i in to_zero:
            b[ones[i]] = 0
    elif len(ones) < n//2:
        # We have too few ones; make largest zeros become 1
        to_one = np.argsort(h[zeros])[-(n//2 - len(ones)):]
        for i in to_one:
            b[zeros[i]] = 1
    # Refine via swap operations
    h_refined = b.copy()
    max_swaps = 5 * n
    max_no_improvement = 2 * n
    no_improvement_count = 0
    c5_current = evaluate_c5(h_refined)
    for _ in range(max_swaps):
        if no_improvement_count >= max_no_improvement:
            break
        # propose swap of 1 and 0 at random positions
        ones_pos = np.where(h_refined == 1)[0]
        zeros_pos = np.where(h_refined == 0)[0]
        if len(ones_pos) == 0 or len(zeros_pos) == 0:
            break
        i1 = np.random.choice(ones_pos)
        i0 = np.random.choice(zeros_pos)
        # perform swap
        h_trial = h_refined.copy()
        h_trial[i1] = 0
        h_trial[i0] = 1
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < c5_current:
            h_refined = h_trial
            c5_current = c5_trial
            no_improvement_count = 0
        else:
            no_improvement_count += 1
    return h_refined, c5_current
