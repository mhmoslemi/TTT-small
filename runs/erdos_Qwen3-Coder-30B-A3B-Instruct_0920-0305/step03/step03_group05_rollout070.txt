import numpy as np
from scipy.optimize import minimize
import time

# Constants
n_points = 48  # Even number to allow exact half ones
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h_binary(h_values):
    """Evaluate the C_5 bound for given binary h values."""
    h = np.array(h_values, dtype=int)
    # Compute correlation of h with (1 - h) using full mode
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def compute_min_corr(h_values):
    """Compute minimum correlation for binary pattern."""
    h = np.array(h_values, dtype=int)
    correlation = np.correlate(h, h, mode="full")
    min_correlation = np.min(correlation)
    return min_correlation

def find_indices(h_values, value):
    """Find all indices with a specific value."""
    return np.where(np.array(h_values) == value)[0]

def simulated_annealing_binary(initial_h, max_iter=5000000, temp_start=0.1, cooling_rate=0.99999):
    """Perform simulated annealing on binary patterns."""
    h_current = np.array(initial_h)
    min_corr_current = compute_min_corr(h_current)
    c5_current = 1 - dx * min_corr_current
    best_h = h_current.copy()
    best_min_corr = min_corr_current
    best_c5 = c5_current
    
    temp = temp_start
    iter_count = 0
    
    indices_ones = find_indices(h_current, 1)
    indices_zeros = find_indices(h_current, 0)
    
    start_time = time.time()
    time_limit = 550  # Leave some margin for final optimization
    
    while iter_count < max_iter and (time.time() - start_time) < time_limit:
        # Generate new candidate by swapping one 1 with one 0
        if len(indices_ones) > 0 and len(indices_zeros) > 0:
            i = np.random.choice(indices_ones)
            j = np.random.choice(indices_zeros)
            
            # Create candidate
            h_candidate = h_current.copy()
            h_candidate[i] = 0
            h_candidate[j] = 1
            
            # Compute the change in min correlation
            min_corr_candidate = compute_min_corr(h_candidate)
            c5_candidate = 1 - dx * min_corr_candidate
            
            # Accept if better or probabilistically
            delta = c5_candidate - c5_current
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                h_current = h_candidate
                c5_current = c5_candidate
                min_corr_current = min_corr_candidate
                
                if c5_candidate < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_candidate
                    best_min_corr = min_corr_current
            
            # Update indices
            indices_ones = find_indices(h_current, 1)
            indices_zeros = find_indices(h_current, 0)
        
        iter_count += 1
        temp *= cooling_rate
        
        # Debugging output every 50k iterations
        if iter_count % 50000 == 0:
            print(f"Iteration {iter_count}: C5 = {best_c5:.6f}")
    
    return best_h, best_c5

def run(seed=42, budget_s=600):
    """Main run function."""
    np.random.seed(seed)
    
    # Start with pattern that has alternating 1s and 0s, with exact half ones
    h_initial = [1 if i % 2 == 0 else 0 for i in range(n_points)]
    # Adjust if needed to make exactly half ones
    n_ones = sum(h_initial)
    if n_ones > n_points // 2:
        # Turn some 1s to 0s
        ones_indices = [i for i in range(n_points) if h_initial[i] == 1]
        for i in range(n_ones - n_points // 2):
            idx = np.random.choice(ones_indices)
            h_initial[idx] = 0
            ones_indices.remove(idx)
    elif n_ones < n_points // 2:
        # Turn some 0s to 1s
        zeros_indices = [i for i in range(n_points) if h_initial[i] == 0]
        for i in range(n_points // 2 - n_ones):
            idx = np.random.choice(zeros_indices)
            h_initial[idx] = 1
            zeros_indices.remove(idx)
    
    # Perform simulated annealing
    h_final, c5_bound = simulated_annealing_binary(h_initial, max_iter=1000000)
    
    # Final local optimization using scipy optimize
    def objective(h_vals):
        return 1 - dx * compute_min_corr(h_vals)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}
    
    try:
        result = minimize(objective, h_final, method='SLSQP', bounds=bounds, constraints=constraints,
                         options={'maxiter': 100, 'ftol': 1e-12})
        if result.success:
            h_final = result.x
            c5_bound = objective(h_final)
    except:
        pass
    
    # Ensure final solution meets constraint
    h_final = np.array(h_final)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    return list(h_final), c5_bound, n_points
