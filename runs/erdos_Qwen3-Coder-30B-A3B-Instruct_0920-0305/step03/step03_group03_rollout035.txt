import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def proj_box_simplex(v):
    """
    Project vector v onto the intersection of the box [0,1]^n and the simplex {x: sum x_i = 1}.
    """
    v = np.array(v)
    n = len(v)
    # Sort v in descending order
    vs = np.sort(v)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(vs)
    # Find the threshold
    rho = (cumsum - 1) / np.arange(1, n + 1)
    # Find the largest k such that rho_k <= vs_k
    k = np.argmax(vs >= rho)
    # If no such k, then all elements are too small, set them to 1/n
    if k == 0 and vs[0] >= rho[0]:
        tau = rho[0]
    else:
        tau = rho[k] if k < n else 0
    # Compute projection
    x = np.maximum(v - tau, 0)
    # Normalize to sum to 1
    x = x / np.sum(x)
    return x

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def spectral_search(n, budget_s, seed=42):
    np.random.seed(seed)
    t0 = time.time()
    dx = 2.0 / n
    m = n / 2
    it_max = 500
    R = 30
    best_c5 = float('inf')
    best_h = None
    best_iter = 0
    
    # Compute target magnitude M from initial state
    h_start = np.random.exponential(1, n)
    h_start = h_start / np.sum(h_start) * m
    h_start = np.clip(h_start, 0, 1)
    h_start = proj_box_simplex(h_start)
    
    # Get first estimate of M from the current h_start
    H = np.fft.fft(h_start)
    M = np.mean(np.abs(H[1:])**2)**0.5
    
    for restart in range(R):
        if time.time() - t0 > budget_s * 0.95:
            break
            
        # Start with a random feasible vector
        u = np.random.exponential(1, n)
        h = u / np.sum(u) * m
        h = np.clip(h, 0, 1)
        h = proj_box_simplex(h)
        
        # Inner iterations
        last_c5 = float('inf')
        for t in range(it_max):
            if time.time() - t0 > budget_s * 0.95:
                break
                
            # FFT
            H = np.fft.fft(h)
            
            # Modify magnitude
            H_new = H.copy()
            H_new[1:] = M * np.exp(1j * np.angle(H[1:]))
            
            # IFFT
            h_new = np.real(np.fft.ifft(H_new))
            
            # Projection
            h = proj_box_simplex(h_new)
            
            # Check for convergence
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_iter = t
                
            # Early exit on no improvement
            if abs(c5 - last_c5) < 1e-8 and t > 10:
                break
            last_c5 = c5
            
    return best_h, best_c5

def binary_refinement(h):
    n = len(h)
    # Threshold and proportional adjustment
    h_sorted = np.sort(h)[::-1]
    threshold_idx = int(n / 2)
    threshold = h_sorted[threshold_idx]
    b = (h >= threshold).astype(float)
    
    # Adjust to required sum
    count = np.sum(b)
    if count < n/2:
        ones_to_flip = n/2 - count
        indices_zero = np.where(b == 0)[0]
        if len(indices_zero) >= ones_to_flip:
            flips = np.random.choice(indices_zero, size=int(ones_to_flip), replace=False)
            b[flips] = 1.0
    elif count > n/2:
        ones_to_flip = count - n/2
        indices_one = np.where(b == 1)[0]
        if len(indices_one) >= ones_to_flip:
            flips = np.random.choice(indices_one, size=int(ones_to_flip), replace=False)
            b[flips] = 0.0
            
    # Local hill-climbing swaps
    for _ in range(10 * n):
        if np.sum(b) != n/2:
            break
        idx0 = np.where(b == 0)[0]
        idx1 = np.where(b == 1)[0]
        if len(idx0) == 0 or len(idx1) == 0:
            break
        i0 = np.random.choice(idx0)
        i1 = np.random.choice(idx1)
        
        # Temporarily swap
        b_test = b.copy()
        b_test[i0] = 1
        b_test[i1] = 0

        # If this improves C5, accept it
        c5_test = evaluate_c5(b_test)
        c5_orig = evaluate_c5(b)
        if c5_test < c5_orig:
            b = b_test.copy()
            
    # Final polishing with gradient-based method
    def loss_func(h):
        # Smooth approximation of max
        return - np.log(np.sum(np.exp(-np.array([evaluate_c5(h) for _ in range(100)]))))
        
    bounds = [(0, 1) for _ in range(n)]
    
    # Optimizing current binary vector
    opt_h = b.copy()
    opt_h = proj_box_simplex(opt_h)
    opt_c5 = evaluate_c5(opt_h)
    
    # If need more, use cheap local optimization
    try:
        # Create constraints 
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) * 2/n - 1.0}
        res = minimize(lambda x: evaluate_c5(x), opt_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 50})
        if res.success:
            x_opt = res.x
            x_opt = np.clip(x_opt, 0, 1)
            scale = 1.0 / (np.sum(x_opt) * 2/n)
            x_opt = x_opt * scale
            c5_opt = evaluate_c5(x_opt)
            if c5_opt < opt_c5:
                opt_h = x_opt
                opt_c5 = c5_opt
    except:
        pass
        
    return opt_h, opt_c5
