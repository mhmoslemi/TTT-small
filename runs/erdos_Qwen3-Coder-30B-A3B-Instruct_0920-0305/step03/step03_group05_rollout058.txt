import numpy as np
import cvxpy as cp
import time

# Constants
n_values = [48, 56, 64, 72]
target_bound = 0.3807
dx = 2.0 / 48 # baseline dx for n=48, will scale appropriately

def solve_milp_binary(n, time_limit=120):
    """
    Solve the binary MILP formulation for n points
    Returns tuple: (h_solution, C5_bound) or (None, inf) if failed.
    """
    # Variable definitions 
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()

    # All constraints as per mathematical model
    constraints = [cp.sum(h) == n // 2, t >= 0]

    # Compute the number of y variables per shift
    y_vars = []
    q_vars = []

    for d in range(n):
        if d == 0:
            # For shift d=0, Q_0 = sum h_i 
            # but can take shortcut: autocorr at 0 is just sum h
            # So do not create new y vars; we don't need to enforce sum h >= t
            # since there is no shift.
            # We only need to enforce that each Q_d >= t
            pass

        # Normal case d >= 1
        # Create auxiliary variables y_{i,d} for i=0..n-1-d
        y_list = cp.Variable(n - d, boolean=True)
        y_vars.append((y_list, d))

        # Add constraints for each y_{i,d} = h_i * h_{i+d}
        eq_constrs = []
        for i in range(n - d):
            eq_constrs.append(y_list[i] <= h[i])
            eq_constrs.append(y_list[i] <= h[i + d])
            eq_constrs.append(y_list[i] >= h[i] + h[i + d] - 1)

        q_d = cp.sum(y_list)
        q_vars.append(q_d)
        constraints.extend(eq_constrs)

    # Enforce that all Q_d >= t
    for q_d in q_vars:
        constraints.append(q_d >= t)

    # Objective
    prob = cp.Problem(cp.Maximize(t), constraints)

    try:
        # Solve with time limit
        prob.solve(solver=cp.CBC, verbose=False, 
                   time_limit=time_limit)

        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h.value
            t_sol = t.value
            C5_bound = 1.0 - (2 / n) * t_sol
            return h_sol, C5_bound
        else:
            # No feasible solution found or hit time limit, return best known
            # May not have worked
            if prob.status in [cp.SOLVER_FAILURE, cp.INFEASIBLE]:
                return None, float('inf')
            elif prob.status == cp.UNBOUNDED:
                return None, float('inf')

    except Exception as e:
        # Likely a solver or internal error
        return None, float('inf')

    return None, float('inf')

def hill_climb_swaps(h_init, max_swaps=200000, time_limit=10):
    """
    Perform greedy hill-climbing on binary h using swaps.
    Returns (h_final, C5_final)
    """
    h = h_init.copy()
    dx = 2.0 / len(h)

    def compute_c5(h_vec):
        # Compute overlaps using convolution
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_overlap = np.max(corr)  
        return max_overlap * dx

    # Precompute initial C5
    c5_current = compute_c5(h)
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1 - h)

    num_swaps = 0
    start_time = time.time()

    while num_swaps < max_swaps and (time.time() - start_time) < time_limit - 1:
        # Pick random index where h = 1 and index where h = 0
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # Temporarily swap
        h[i], h[j] = 0, 1
        c5_new = compute_c5(h)
        if c5_new < c5_current:
            c5_current = c5_new
        else:
            # Revert swap
            h[i], h[j] = 1, 0
        num_swaps += 1
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)

    return h, c5_current

def continuous_refine(h_binary, n_high, time_limit=10):
    """
    Refines a binary pattern to continuous by interpolation + projection descent.
    """
    # Interpolate to higher resolution
    n_original = len(h_binary)
    ratio = n_high // n_original
    h_upsample = np.repeat(h_binary, ratio)
    h_upsample = h_upsample[:n_high]
    
    # Project constant to satisfy sum constraint
    current_sum = np.sum(h_upsample)
    target_sum = n_high / 2.0
    scale = target_sum / current_sum
    h_upsample *= scale
    
    # Simple gradient descent on the unequal integral constraint
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / len(h_vec))
    
    dx = 2.0 / len(h_upsample)
    lr = 0.01

    iterations = 0
    start_time = time.time()

    while iterations < 5000 and (time.time() - start_time) < time_limit - 1:
        corr = np.correlate(h_upsample, 1 - h_upsample, mode='full')
        max_corr = np.max(corr)
        c5_new = max_corr * dx
        
        # Project from gradient descent update
        grad = np.gradient(- (1 - h_upsample) * np.convolve(h_upsample, np.ones_like(h_upsample), mode='same'))
        h_upsample = h_upsample - lr * grad
        h_upsample = np.clip(h_upsample, 0, 1)
        
        # Project onto sum constraint
        old_sum = np.sum(h_upsample)
        if abs(old_sum - target_sum) > 1e-6:
            h_upsample *= target_sum / old_sum
            
        iterations += 1

    # Final evaluation
    c5_final = compute_c5(h_upsample)
    return h_upsample, c5_final

    # Initial best
    best = (None, float('inf'), None)
