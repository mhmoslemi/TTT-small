import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as csum, log, exp

def weight_normalise(x, n):
    """Rescale x so that sum(x) = n/2"""
    dx = 2.0 / n
    scale = (n / 2) / np.sum(x)
    return x * scale

def max_defect(h):
    """Compute exact C5 = max_k int h(x)(1-h(x+k)) dx"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def softmax_objective(h, alpha):
    """Compute log-sum-exp approximation of max with gradient"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation for all shifts (we'll do this efficiently later)
    # Store full correlation vector
    corr_full = correlate(h, 1 - h, mode="full") * dx
    # Compute softmax over shifts
    shifts = len(corr_full)
    shift_indices = np.arange(shifts)
    
    # Soft max
    exp_vals = exp(alpha * corr_full)
    Z = csum(exp_vals)
    phi_alpha = (1.0 / alpha) * log(Z)
    
    # Compute gradient
    weights = exp_vals / Z
    grad = np.zeros_like(h)
    # Gradient with respect to h[i] = sum_{s} weights[s] * (1 - h[i+s] if i+s < n else 0) - h[i-s] if i-s >= 0 else 0
    
    for i in range(n):
        d = 0.0
        for s in range(1, n):
            if i + s < n:
                d += weights[s-1] * (-h[i+s])  # - h[i+s] term
            if i - s >= 0:
                d += weights[n-1+s] * (-h[i-s])  # - h[i-s] term
        
        grad[i] = 1.0 - d  # 1 - (sum of h contributions)
    
    return phi_alpha, grad

def build_milp(n):
    """Build and solve binary MILP for max min overlap"""
    dx = 2.0 / n
    # Variables
    h = Variable(n, boolean=True)
    # Auxiliary AND variables
    w = Variable((n, n-1), nonneg=True)
    t = Variable()
    
    # Constraints
    constraints = [
        csum(h) == n/2,
        t <= 1,
    ]
    
    # AND constraint: w[i,s] = h[i] AND h[i+s] 
    for s in range(1, n):
        for i in range(n):
            if i + s < n:
                constraints += [
                    w[i, s-1] <= h[i],
                    w[i, s-1] <= h[i+s],
                    w[i, s-1] >= h[i] + h[i+s] - 1
                ]
    
    # Min overlap constraint
    for s in range(1, n):
        constraints += [csum(w[:, s-1]) >= t]  # Sum of w over i for shift s
    
    # Objective
    prob = Problem(Minimize(-t), constraints)
    
    # Try solving within time limit
    try:
        prob.solve(solver=cp.CBC, verbose=False, maxTime=180.0)
        if prob.status == "optimal" or prob.status == "feasible":
            return np.array(h.value).astype(float)
    except:
        pass
    
    # Return best feasible solution found
    try:
        prob.solve(solver=cp.CBC, verbose=False, maxTime=180.0, 
                   seconds=180.0, ratioGap=0.0)
        if prob.status == "feasible" and h.value is not None:
            return np.array(h.value).astype(float)
    except:
        return None

def anneal(h, n):
    """Targeted swap annealing for kth max defect improvement"""
    # Precompute full correlation matrix at start for lighter updates
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full") * dx
    
    # Find shift that causes maximum defect
    k_max = np.argmax(corr)
    max_def = corr[k_max]
    
    h = h.copy()
    alpha = 0.2
    
    # Look up the positions of all data structures that contribute to max defect
    # This is expensive but is necessary due to full correlation recall efficiency
    # It’s acceptable in section 3 given remaining time budget
    
    # Track which indices have h[i] > 0 and h[i+k_max] < 1 (or something below a threshold)
    def compute_contributors():
        contribs = []
        for i in range(n):
            if h[i] > 0.01 and i + k_max < n and h[i + k_max] < 0.99:
                contribs.append(i)
        return contribs
    
    # For each materialization of a critical pair and evaluate exchange:
    # Do multiple randomizations, and accept slow decrements in max_def only
    def mate_and_test():
        # Select both contributors
        contribs = compute_contributors()
        if len(contribs) < 1:
            return
        
        for _ in range(5):
            idx = np.random.choice(contribs)
            # Select a target with h[j] small but 1-h[j+k_max] large
            # Set up appropriate probabilistic heuristic
            start = max(0, k_max - n//2)  # Narrow window to avoid bottlenecks
            end = min(n, 2 * n - k_max)
            candidates = [i for i in range(start, end) if i != idx]
            if candidates:
                jdx = np.random.choice(candidates)
                if h[jdx] > 0.01 and h[jdx + k_max] < 0.99:
                    # Decrement h[idx] and increment h[jdx] by small amount
                    delta = min(h[idx], 1.0 - h[jdx]) * alpha
                    if delta < 1e-6:
                        continue  # ess.
                    
                    # Swap perturbation in both directions for exact equality
                    h[idx] -= delta
                    h[jdx] += delta
                    # Flip k_max check for optimization:
                    corr[k_max] = np.sum(h * (1 - np.roll(h, k_max))) * dx  # Recompute for clarity
                    # update full correlation if needed (but typically just do max)
                    
                    # Check whether we improved the max_defect (exact)
                    # Recompute max_corr directly:
                    corr_new = correlate(h, 1 - h, mode="full") * dx
                    max_new = np.max(corr_new)
                    if max_new < max_def:
                        max_def = max_new
                        k_max = np.argmax(corr_new)
                    if max_new >= max_def:
                        # Undo swap if no improvement
                        h[idx] += delta
                        h[jdx] -= delta
                        # Restore corr?
                        
    # Do several alternating moves to get low generalizations
    for _ in range(2000):
        mate_and_test()
        alpha *= 0.99
        if time.time() > 550:
            break  # Timeout guard

    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    n = 100  # If good, switch otherwise fallback to 80
    
    # Stage 0: Initialization
    if (time.time() - start_time) > 5:
        n = 80  # fallback
    dx = 2.0 / n
    
    # Phase 1: Binary MILP 
    h_bin = build_milp(n)
    
    if h_bin is None:
        # Fallback: Random restart with continuous optimization
        best_h = None
        best_c5 = np.inf
        for i in range(30):
            h = np.random.rand(n)
            h_norm = weight_normalise(h, n)
            try:
                c5 = max_defect(h_norm)
                if c5 < best_c5:
                    best_h = h_norm
                    best_c5 = c5
            except:
                continue
        if best_h is None:
            # If immediate success not reasonable, dummy solution
            best_h = np.full(n, 0.5)
            best_c5 = 0.5
    else:
        # Phase 2: Continuous refinement
        h_cont = refine_continuous(h_bin, n)
        
        # Phase 3: Swap Annealing
        h_final = anneal(h_cont, n)
        best_h = h_final
        best_c5 = max_defect(h_final)
    
    return (best_h.tolist(), best_c5, n)
