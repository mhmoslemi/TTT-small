import numpy as np

def proj_box_simplex(v):
    """Project vector v onto the intersection of [0,1]^n and the simplex {sum x_i = 1}"""
    n = len(v)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u > css / np.arange(1, n+1))[0]
    if len(rho) == 0:
        # All elements are 0 -> set all to 1/n
        return np.ones(n) / n
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
        return np.clip(v - theta, 0, 1)

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def spectral_search(n, budget_s, seed=42):

            break

        # Initialize h randomly
        u = np.random.exponential(1.0, n)
        h = u / np.sum(u) * m
        h = proj_box_simplex(h)
        h = np.clip(h, 0, 1)

        # Compute M from current h
        H = np.fft.fft(h)
        M = np.sqrt(np.sum(np.abs(H[1:])**2) / (n - 1)) if (n > 1) else 0.0

        # Inner alternating projection loop
        last_c5s = []
        converged = False
        for t in range(it_max):
            if time.time() - start_t > budget_s * 0.95:
                break

            # FFT
            H = np.fft.fft(h)
            
            # Enforce constant magnitude
            H_proj = H.copy()
            H_proj[1:] = M * np.exp(1j * np.angle(H[1:]))

            h = proj_box_simplex(h_new)

            if t % 10 == 0:
                c5 = evaluate_c5(h)
                last_c5s.append(c5)
                if len(last_c5s) >= 10 and abs(last_c5s[-1] - last_c5s[-2]) < TOL:
                    converged = True
                    break

        # Check final result
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    return best_h, best_c5

def binary_refinement(h, n):
    # Create binary approximation
    m = n / 2.0
    sorted_indices = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[sorted_indices[:int(m)]] = 1
    
    # Local hill-climb swaps
    h_best = h.copy()
    c5_best = evaluate_c5(h_best)
    
    for _ in range(5 * n):
        if np.sum(b) == 0 or np.sum(b) == n:
            break
            
        # Try a swap
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Perform swap
        b_new = b.copy()
        b_new[i] = 0
        b_new[j] = 1
        h_new = np.zeros(n)
        h_new[b_new > 0.5] = 1.0
        h_new[b_new < 0.5] = 0.0
        
        c5_new = evaluate_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()
            b = b_new.copy()
    
    return h_best, c5_best
