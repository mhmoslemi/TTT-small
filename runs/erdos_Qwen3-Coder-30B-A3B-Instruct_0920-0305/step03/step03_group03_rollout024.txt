import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def proj_box_simplex(v):
    """Project vector v onto the intersection of [0,1]^n and the simplex {x: sum(x)=1}"""
    v = np.asarray(v)
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    tmp = np.cumsum(v_sorted) - 1
    rho = np.argmax(v_sorted > tmp / np.arange(1, n + 1)) if len(v_sorted) > 0 else 0
    if rho == 0:
        theta = (np.sum(v) - 1) / n
        return np.maximum(v - theta, 0)
    else:
        theta = tmp[rho - 1] / rho
        return np.maximum(v - theta, 0)

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def spectral_search(n, budget, t0, r=30):

    m = n / 2
    
    best_h = None
    best_c5 = float('inf')

            break
            
        # Start with random projection onto feasible set
        u = np.random.exponential(1, n)
        h = u / np.sum(u)
        h = h * m / np.sum(h)  # Scale to sum = m 
        h = proj_box_simplex(h)
        h = np.clip(h, 0, 1)
        h = proj_box_simplex(h)
        
        if M is None:
            M = np.sqrt(2 * (np.sum(h**2) - (m**2)/n) / (n-1))
            if np.isnan(M) or np.isinf(M):
                M = 1.0

        h_best = h.copy()
        c5_best = evaluate_c5(h_best)

        for t in range(it_max):
            if time.time() - t0 > budget * 0.99:
                break
                
            H = np.fft.fft(h)
            H_new = H.copy()
            H_new[1:] = M * np.exp(1j * np.angle(H[1:]))
            h_tilde = np.real(np.fft.ifft(H_new))
            h = proj_box_simplex(h_tilde)
            h = np.clip(h, 0, 1)
            h = proj_box_simplex(h)
            
            c5 = evaluate_c5(h)
            if c5 < c5_best:
                c5_best = c5
                h_best = h.copy()
            
            if t > 10 and abs(c5 - c5_best) < 1e-8:
                break
                
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best.copy()
            
    return best_h, best_c5

def binary_refinement(h):

    m = int(0.5 * n)
    
    # Create binary by thresholding
    sorted_indices = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[sorted_indices[:m]] = 1.0
    b = b / np.sum(b) * (n/2)
    
    # Local swaps to improve C5
    h_s = b.copy()
    c5 = evaluate_c5(h_s)
    max_iter = 5 * n
    patience = 2 * n
    
    for i in range(max_iter):
        if patience == 0:
            break
        one_indices = np.where(h_s > 0.5)[0]
        zero_indices = np.where(h_s < 0.5)[0]
        
        if len(one_indices) == 0 or len(zero_indices) == 0:
            break
            
        i1 = np.random.choice(one_indices)
        i0 = np.random.choice(zero_indices)
        
        h_swap = h_s.copy()
        h_swap[i1] = 0
        h_swap[i0] = 1
        h_swap = h_swap / np.sum(h_swap) * (n/2)
        
        new_c5 = evaluate_c5(h_swap)
        if new_c5 < c5:
            h_s = h_swap
            c5 = new_c5
            patience = 2 * n
        else:
            patience -= 1

    # Final continuous polishing
    def fun(h_vec):
        return evaluate_c5(h_vec)
    
    def jac(h_vec):
        # Approximate gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = h_plus / np.sum(h_plus) * (n/2)
            h_plus = proj_box_simplex(h_plus)
            grad[i] = (fun(h_plus) - fun(h_vec)) / eps
        return grad
    
    try:
        # Local polish with projected gradient steps
        h_final = h_s.copy()
        for iter in range(200):
            if iter % 10 == 0:
                c5_old = fun(h_final)
            grad = jac(h_final)
            lr = 1e-3
            h_new = h_final - lr * grad
            h_new = h_new / np.sum(h_new) * (n/2)
            h_new = proj_box_simplex(h_new)
            h_new = np.clip(h_new, 0, 1)
            h_new = proj_box_simplex(h_new)
            
            c5_new = fun(h_new)
            if c5_new < c5_old:
                h_final = h_new
            else:
                break
    except:
        h_final = h_s

    return h_final, evaluate_c5(h_final)
