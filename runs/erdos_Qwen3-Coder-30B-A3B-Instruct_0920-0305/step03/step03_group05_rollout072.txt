from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

# Constants
n_points = 47
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def local_search(h_values, max_iter=200):
    """Refine solution using local optimization."""
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    def objective(h):
        return evaluate_h(h)
    
    try:
        result = minimize(objective, h_values, method='SLSQP', 
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-10})
        if result.success:
            return result.x
    except:
        pass
    return h_values

    h_new = h_values.copy()

    current_sum = np.sum(h_new) * dx
    if abs(current_sum - 1.0) > 1e-6:
