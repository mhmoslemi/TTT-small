import numpy as np
from scipy.optimize import minimize
import time

# Constants
n_points = 80
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_to_simplex_box(y, target):
    """Project y onto the set {x : 0 <= x <= 1, sum(x) = target}."""
    lo = min(y) - 1.0
    hi = max(y)
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2, 0, 1)
    # Adjust rounding errors
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0, 1)
    return x

def smooth_max_correlation(h, gamma):
    """Compute correlation and weights for smooth max descent."""
    C = np.correlate(h, 1 - h, mode="full")
    w = np.exp(gamma * C)
    w_sum = w.sum()
    if w_sum > 0:
        w = w / w_sum
    else:
        w = np.ones_like(w) / len(w)
    return C, w

def compute_gradient_smooth_max(h, C, w):
    """Compute gradient for smooth max descent."""
    n = len(h)
    grad = np.ones_like(h)
    for idx, weight in enumerate(w):
        if weight == 0:
            continue
        s = idx - (len(C) // 2)
        if s >= 0:
            grad[:n-s] -= weight * h[s:]
            grad[s:] -= weight * h[:n-s]
        else:
            s_abs = -s
            grad[s_abs:] -= weight * h[:n-s_abs]
            grad[:n-s_abs] -= weight * h[s_abs:]
    return grad

def compute_subgradient(h, s_indices):
    """Compute subgradient for given shift indices."""
    n = len(h)
    grad = np.zeros_like(h)
    for s_idx in s_indices:
        s = s_idx - (len(h) - 1)
        if s >= 0:
            grad[:n-s] -= h[s:]
            grad[s:] -= h[:n-s]
        else:
            s_abs = -s
            grad[s_abs:] -= h[:n-s_abs]
            grad[:n-s_abs] -= h[s_abs:]
    return grad

def run(seed=42, budget_s=600):
    """Main optimization routine."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    total = n_points / 2
    
    # Prepare initial shapes
    init_candidates = []
    
    # From initial_h_values
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values)
        if len(init_h) != n_points:
            old_x = np.linspace(0, 2, len(init_h))
            new_x = np.linspace(0, 2, n_points)
            init_h = np.interp(new_x, old_x, init_h)
    else:
        # Fallback to uniform
        init_h = np.full(n_points, 0.5)
    init_candidates.append(init_h)
    
    # Uniform distribution
    init_candidates.append(np.full(n_points, 0.5))
    
    # Triangular shape
    half = np.arange(n_points//2 + 1) / (n_points//2)
    tri = np.concatenate([half[::-1][1:], half])
    init_candidates.append(tri)
    
    # Random gaussian
    rand = np.random.rand(n_points)
    rand = project_to_simplex_box(rand, total)
    init_candidates.append(rand)
    
    # Find best initial shape
    best_h = None
    best_c5 = float('inf')
    for h0 in init_candidates:
        c5 = evaluate_h(h0)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h0.copy()
    
    # Optimization with restarts
    restarts = 10
    for restart in range(restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Initialize
        if restart % 3 == 0:
            h = best_h.copy()
        else:
            h = np.random.rand(n_points)
            h = project_to_simplex_box(h, total)

        # Parameters
        step = 0.1
        gamma = 0.5
        beta = 0.9
        momentum = np.zeros_like(h)
        
        # Primary smooth max descent
        for iter_num in range(3000):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Compute correlation and weights
            C, w = smooth_max_correlation(h, gamma)
            
            # Compute gradient
            grad = compute_gradient_smooth_max(h, C, w)
            
            # Momentum update
            momentum = beta * momentum + (1 - beta) * grad
            h_new = h - step * momentum
            
            # Project
            h = project_to_simplex_box(h_new, total)
            
            # Evaluate
            current_c5 = evaluate_h(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                if best_c5 <= target_bound:
                    break
                    
            # Schedule update
            step *= 0.997
            gamma = min(gamma * 1.01, 50.0)
            
            # Occasionally add jitter
            if iter_num % 300 == 0 and np.random.rand() < 0.3:
                h = project_to_simplex_box(h + np.random.normal(scale=0.02, size=n_points), total)
        
        # Subgradient refinement step
        for sub_iter in range(500):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            C = np.correlate(h, 1 - h, mode="full")
            max_val = C.max()
            max_indices = np.where(np.isclose(C, max_val, atol=1e-12))[0]
            if len(max_indices) == 0:
                break
                
            # Compute subgradient
            grad = compute_subgradient(h, max_indices)
            grad /= len(max_indices)
            
            # Update
            h = h - 0.05 * grad
            h = project_to_simplex_box(h, total)
            
            # Evaluate
            current_c5 = evaluate_h(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Final local optimization with SLSQP if time allows
    if time.time() - start_time < budget_s * 0.8:
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1}
        try:
            result = minimize(evaluate_h, best_h, method='SLSQP', bounds=bounds, constraints=constraints,
                             options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-10})
            if result.success:
                final_c5 = evaluate_h(result.x)
                if final_c5 < best_c5:
                    best_c5 = final_c5
                    best_h = result.x.copy()
        except:
            pass
    
    # Ensure constraint satisfaction
    best_h = project_to_simplex_box(best_h, total)
    final_c5 = evaluate_h(best_h)
    
    return list(best_h), final_c5, n_points
