import numpy as np
from scipy.optimize import minimize
import time

# Constants
n_points = 80
dx = 2.0 / n_points
target_bound = 0.3807

def project_to_simplex_box(y, target):
    """Project y onto box [0,1]^n with sum = target"""
    lo = min(y) - 1.0
    hi = max(y)
    
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    
    x = np.clip(y - (lo + hi) / 2, 0, 1)
    # Correct small rounding errors
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0, 1)
    return x

def compute_gradient_smooth_max(h, C, gamma):
    """Compute gradient of smooth max correlation"""
    w = np.exp(gamma * C)
    w = w / w.sum()
    n = len(h)
    grad = np.ones(n)
    
    for idx, weight in enumerate(w):
        if weight == 0:
            continue
        s = idx - (len(C) // 2)
        if s >= 0:
            grad[:n-s] -= weight * h[s:]
            grad[s:] -= weight * h[:n-s]
        else:
            s_abs = -s
            grad[s_abs:] -= weight * h[:n-s_abs]
            grad[:n-s_abs] -= weight * h[s_abs:]
    return grad

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

    restarts = 8

        step_size = 0.2

        velocity = np.zeros(n_points)
        
        # Main optimization loop
        max_iter = 3000
        for iteration in range(max_iter):

            # Compute correlation
            C = np.correlate(h, 1 - h, mode="full")
            
            # Compute gradient via smooth max approach
            grad = compute_gradient_smooth_max(h, C, gamma)
            
            # Update with momentum
            velocity = beta * velocity + (1 - beta) * grad
            h_new = h - step_size * velocity
            
            # Project onto feasible set
            h = project_to_simplex_box(h_new, n_points / 2)

                    if final_c5 < best_c5:

        best_h = init_h.copy()
    
    return list(best_h), best_c5, n_points
