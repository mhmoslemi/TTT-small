import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time
import cvxpy as cp
from cvxpy import SCS
import cvxpy as cp
from scipy.special import expit

# Constants
default_n_values = [36, 42, 48, 54, 60]
target_bound = 0.3807
dx = 2.0
max_iter_swap = 50000
max_iter_lbfgs = 300
rounding_reps = 30
initial_temp = 100

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * (dx / len(h))
    return c5

def correlation_vec(h_values):
    """Return all correlations for all possible shifts."""
    n = len(h_values)
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    return corr

def logsumexp(x, axis=None):
    """Numerically stable logsumexp"""
    x_max = np.max(x, axis=axis, keepdims=True)
    return x_max + np.log(np.sum(np.exp(x - x_max), axis=axis, keepdims=True))

def smooth_max(corr_vec, alpha=30.0):
    """Smooth approximation of max."""
    return logsumexp(alpha * corr_vec) / alpha

def proj_to_simplex(v, s=1.0):
    """Project v onto the simplex with sum = s."""
    v = np.asarray(v)
    n = len(v)
    u = np.sort(v)[::-1]
    cumsum = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u - (s - cumsum) / idx > 0
    rho = np.sum(cond)
    theta = (s - np.sum(u[:rho])) / rho
    return np.clip(v - theta, 0, 1)

def sigmoid(x):
    return expit(x)

def logit(x):
    x = np.clip(x, 1e-8, 1 - 1e-8)
    return np.log(x / (1 - x))

def solve_sdp(n):
    """Solve the SDP relaxation for given n."""
    # Variables
    x = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = []
    constraints.append(0 <= x)
    constraints.append(x <= 1)
    constraints.append(cp.sum(x) == n / 2)
    
    # Diagonal constraints
    constraints.append(cp.diag(X) == x)
    
    # McCormick constraints for X[i,j] = x[i] * x[j]
    for i in range(n):
        for j in range(i+1, n):
            # Both x_i, x_j are in [0,1], so we can apply McCormick envelopes:
            constraints.append(X[i, j] >= 0)
            constraints.append(X[i, j] <= x[i])
            constraints.append(X[i, j] <= x[j])
            constraints.append(X[i, j] >= x[i] + x[j] - 1)
    
    # Overlap constraints
    for k in range(n):
        if k == 0:
            # All terms,
            c = cp.sum(x) - cp.sum(cp.diag(X, k))
        else:
            c = cp.sum(x[:-k]) - cp.sum(cp.diag(X, k))
        constraints.append(c <= t)
    
    # PSD constraint
    constraints.append(X >> 0)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=SCS, max_iters=8000, eps=1e-5)
        return prob.status == cp.OPTIMAL, x.value, X.value
    except:
        return False, None, None

def round_from_sdp(x_sol, X_sol, n):
    """Round the SDP solution to binary"""
    # Compute a low-rank factorization
    U, s, Vt = np.linalg.svd(X_sol)
    r = min(10, n)
    V = U[:, :r] @ np.diag(np.sqrt(s[:r]))
    
    best_h = None
    best_cost = float('inf')
    
    for _ in range(rounding_reps):
        g = np.random.randn(r)
        sgn = np.sign(V @ g)
        h = (sgn >= 0).astype(float)
        
        # Adjust to match exactly n/2
        diff = int(np.sum(h) - n/2)
        if diff > 0:
            # Remove diff ones (the ones with smallest |V @ g|)
            idxs = np.argsort(np.abs(V @ g))[:diff]
            h[idxs] = 0
        elif diff < 0:
            # Add (-diff) zeros (the ones with smallest |V @ g|)
            idxs = np.argsort(np.abs(V @ g))[diff:]
            h[idxs] = 1
            
        # Evaluate
        cost = evaluate_h(h)
        if cost < best_cost:
            best_cost = cost
            best_h = h.copy()
            
    return best_h

def hill_climb_binary(h_start, n, max_iter=max_iter_swap):
    """Hill-climb on binary vectors using swaps."""
    h = h_start.copy()
    c5 = evaluate_h(h)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    diverge_counter = 0
    last_improvement = 0
    
    for i in range(max_iter):
        if diverge_counter > 2 * n:
            # Random restart
            h = np.random.rand(n) < 0.5
            h = proj_to_simplex(h, s=n/2)
            c5 = evaluate_h(h)
            ones = np.where(h == 1)[0].tolist()
            zeros = np.where(h == 0)[0].tolist()
            diverge_counter = 0
            last_improvement = 0
        
        if i - last_improvement > 2 * n:
            # Identify the worst shift and try to improve on it.
            corr = correlation_vec(h)
            k_star = np.argmax(corr)
            # Count mismatches for that k
            mismatches = 0
            for idx in range(len(h) - k_star):
                if h[idx] == 1 and h[idx + k_star] == 0:
                    mismatches += 1
            if mismatches > 0:
                # Try to find a good swap, a 1 to swap with a 0 at max
                best_swap = None
                best_improvement = 0
                for oi in ones:
                    # Prefer the most problematic 1s
                    if oi + k_star >= len(h):
                        continue
                    if h[oi] == 1 and h[oi+k_star] == 0:
                        for zi in zeros:
                            h_test = h.copy()
                            h_test[oi] = 0
                            h_test[zi] = 1
                            c5_test = evaluate_h(h_test)
                            if c5_test < c5:
                                if c5_test - c5 < best_improvement:
                                    best_improvement = c5_test - c5
                                    best_swap = (oi, zi)
                        
                if best_swap:
                    i1, i2 = best_swap
                    h[i1] = 0
                    h[i2] = 1
                    ones.remove(i1)
                    ones.append(i2)
                    zeros.remove(i2)
                    zeros.append(i1)
                    c5 = c5 + best_improvement
                    last_improvement = i
                    diverge_counter = 0
                else:
                    diverge_counter += 1
            else:
                diverge_counter += 1
        else:
            # Regular swap
            i1 = np.random.choice(ones)
            i2 = np.random.choice(zeros)
            
            # Try swap
            h_next = h.copy()
            h_next[i1] = 0
            h_next[i2] = 1
            
            c5_new = evaluate_h(h_next)
            
            if c5_new < c5:
                h = h_next
                ones.remove(i1)
                ones.append(i2)
                zeros.remove(i2)
                zeros.append(i1)
                c5 = c5_new
                last_improvement = i
                diverge_counter = 0
            else:
                diverge_counter += 1
    
    return h, c5

def smooth_optimise(h_start, n, max_iter=max_iter_lbfgs):
    # h_start is binary or fractional 
    # use soft max over correlation to make it smooth
    y = logit(h_start)
    
    def objective(y_vec):
        h = sigmoid(y_vec)
        corr = correlation_vec(h)
        # soft max approximation to max
        soft = smooth_max(corr)
        # penalty for sum != 1.0
        pen = 1e3 * (np.sum(h) - n/2) ** 2
        return soft + pen
    
    # Optimize
    try:
        result = minimize(objective, y, method='L-BFGS-B', options={'maxiter': max_iter})
        y_opt = result.x
        h_opt = sigmoid(y_opt)
        h_opt = proj_to_simplex(h_opt, s=n/2)
        return h_opt
    except:
        return h_start

# Main runner function
def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0

    # Loop over n values
    found_target = False
    for n in default_n_values:
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 100:
            break

        # SDP solve
        status, x_sol, X_sol = solve_sdp(n)
        if not status:
            continue

        # Randomized rounding to get a binary vector
        h_binary_rounded = round_from_sdp(x_sol, X_sol, n)
        if h_binary_rounded is None:
            continue

        # Perform hill-climb on the rounded vector
        h_hc, c5_hc = hill_climb_binary(h_binary_rounded, n, max_iter=max_iter_swap)
        if c5_hc < best_c5:
            best_c5 = c5_hc
            best_h = h_hc
            best_n = n

        # Optional smooth refinement if time permits and not already target
        if best_c5 <= target_bound:
            found_target = True
            break
            
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 50:
            break

        # Smooth refinement
        h_smooth = smooth_optimise(h_hc, n, max_iter=max_iter_lbfgs)
        c5_smooth = evaluate_h(h_smooth)
        if c5_smooth < best_c5:
            best_c5 = c5_smooth
            best_h = h_smooth
            best_n = n

        if best_c5 <= target_bound:
            found_target = True
            break

    # Final SLSQP refinement if we have anything
    if best_h is not None and not found_target:
        try:
            bounds = [(0, 1) for _ in range(best_n)]
            constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx / best_n - 1.0}

            def objective_h(x):
                return evaluate_h(x)

            result = minimize(objective_h, best_h, method='SLSQP', bounds=bounds, constraints=constraints,
                              options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-10})
            if result.success:
                best_h = result.x
                best_c5 = evaluate_h(best_h)
        except:
            pass

    if best_h is None:
        # Fallback to initial construction
        h_current = np.array(initial_h_values)
        best_h = h_current.copy()
        best_c5 = evaluate_h(best_h)
        best_n = len(initial_h_values)

    return list(best_h), best_c5, best_n
