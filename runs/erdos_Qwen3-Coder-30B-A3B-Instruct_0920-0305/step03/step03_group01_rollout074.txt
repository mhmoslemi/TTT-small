import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
from scipy.optimize import minimize
from cvxpy import *
import time

def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage A: Low-dim global search
    n = 200
    m = 10  # num blocks
    dx = 2.0 / n
    
    def coarse_to_full(u):
        u = np.clip(u, 0, 1)
        u = np.append(u, max(0, m/2 - np.sum(u[:m-1])))
        u = np.clip(u, 0, 1)
        u = u[:m]
        # repeat each block
        block_size = n // m
        h = np.repeat(u, block_size)
        # if residual, append to last block
        if len(h) < n:
            h = np.append(h, np.full(n - len(h), u[-1]))
        return h
    
    def objective_coarse(u):
        # make full vector
        h = coarse_to_full(u)
        return evaluate_c5(h)
    
    bounds = [(0, 1) for _ in range(m-1)]
    
    try:
        from cma import CMAEvolutionStrategy
        es = CMAEvolutionStrategy(m * [0.5], 0.2)
        max_evals = min(1500, int(120 / (0.1 + 0.001*m)))
        for _ in range(max_evals):
            if (time.time() - start_time) > budget_s * 0.95:
                break
            X = es.ask()
            F = [objective_coarse(x) for x in X]
            es.tell(X, F)
        
        u_best = es.result.xbest
        h_A = coarse_to_full(u_best)
        best_C5_A = evaluate_c5(h_A)
    except:
        h_A = np.array(initial_h_values)
        h_A = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial_h_values)), h_A)
        best_C5_A = evaluate_c5(h_A)
    
    # Stage B: projected sub-gradient descent
    def projected_gradient_descent(h0, max_iter=3000, alpha0=0.1):
        h = h0.copy()
        best_h = h.copy()
        best_c5 = evaluate_c5(h)
        
        for t in range(max_iter):
            if (time.time() - start_time) > budget_s * 0.95:
                break

            # Find active shift constraint
            corr = correlate(h, 1 - h, mode="full")
            k_max = np.argmax(corr)
            c_k = corr[k_max]
            k_idx = k_max
            
            # Compute sub-gradient for this k
            grad = np.zeros_like(h)
            k = k_idx - (n - 1)
            # Positive shift: h_i -> h[i+k], h_{i+k} -> h[i]
            if k > 0:
                # For i: grad[i] += 1 - h[i+k] - h[i-k]
                for i in range(n):
                    if i+k < n:
                        grad[i] += 1 - h[i+k]
                    if i-k >= 0:
                        grad[i] -= h[i-k]
            elif k < 0:
                k = -k
                for i in range(n):
                    if i+k < n:
                        grad[i] -= h[i+k]
                    if i-k >= 0:
                        grad[i] += 1 - h[i-k]
            else:
                grad.fill(0.)
            
            # Step size (decaying)
            alpha = alpha0 / (1 + np.sqrt(t))
            h_new = h - alpha * grad
            # Project to feasible set
            h_proj = project_h(h_new, n, dx)
            h = h_proj
            
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                
        return best_h, best_c5

    def project_h(h_unproj, n, dx):
        # Project to box [0,1] and sum = n/2
        h = np.clip(h_unproj, 0, 1)
        total = np.sum(h)
        # Waterfilling to move sum to exactly n/2
        # Use binary search on lambda: sum_i max(0, h[i] - lambda) = n/2
        # Simplify: sum_i h[i] - sum_i max(0, 0 - lambda) - sum_i max(0, h[i] - lambda)
        # = sum(h) - lambda * num_pos(lambda) - sum_i max(0, h[i] - lambda)
        def sum_filtered(lam):
            val = 0
            for i in range(n):
                val += max(0, h[i] - lam)
            return val
        
        eps = 1e-10
        u, v = h.min() - 1, h.max() + 1
        
        # Just iterate for simplicity
        def loss_fn(lam):
            return sum_filtered(lam) - n/2
        
        # Try to get target value using bisection
        try:
            for _ in range(100):
                x = (u + v)/2
                val = loss_fn(x)
                if val > 0:
                    u = x
                else:
                    v = x
                if abs(val) < eps:
                    break
            lam = (u + v)/2
            h_proj = np.clip(h - lam, 0, 1)
            return h_proj
        except:
            return h
            
    h_B, best_C5_B = projected_gradient_descent(h_A)
    
    # Stage C: Binary rounding and combinatorial local search
    def binary_rounding_local_search(h_cont, n_iter=30):
        best_h = None
        best_c5 = float('inf')
        for _ in range(n_iter):
            # probabilistic binary rounding
            probs = h_cont.copy()
            rand_vals = np.random.rand(n)
            b = (rand_vals < probs).astype(float)
            
            # Make sure exactly weight n/2
            actual_count = np.sum(b)
            diff = n/2 - actual_count
            if diff > 0:
                # Add diff ones
                idx = np.argsort(-probs)
                for i in range(int(diff)):
                    if b[idx[i]] == 0:
                        b[idx[i]] = 1
            elif diff < 0:
                # Remove diff zeros
                idx = np.argsort(probs)
                for i in range(int(-diff)):
                    if b[idx[i]] == 1:
                        b[idx[i]] = 0
                        
            # Local search on binary
            b_out = binary_local_search(b)
            c5_b = evaluate_c5(b_out)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = b_out.copy()
        return best_h, best_c5
    
    def binary_local_search(h0):
        h = h0.copy()
        best_c5 = evaluate_c5(h)
        best_h = h.copy()
        
        # precompute contributions
        # for shift k, compute sum_i h_i * h_{i+k}
        S = np.zeros((n, 2*n-1))  # indices 0..2n-2
        for k in range(1-n, n):
            k_idx = k + n - 1
            s = 0
            for i in range(n):
                if 0 <= i+k < n:
                    s += h[i] * h[i+k]
            S[:, k_idx] = s  # Store sum
        S[:, n-1] = 0  # ignore k=0
            
        # swap loop
        iter_count = 0
        for _ in range(50):  # max local search iterations
            if (time.time() - start_time) > budget_s * 0.95:
                break
            changed = False
            # sample 0.1% of swaps
            n_swaps = max(100, n * n // 1000)
            candidates = []
            for _ in range(n_swaps):
                i = np.random.randint(0, n)
                j = np.random.randint(0, n)
                if h[i] == 1 and h[j] == 0:
                    candidates.append((i, j))
            # If few pairs, try all valid
            if len(candidates) == 0:
                candidates = [(i, j) for i in range(n) for j in range(n) if h[i]==1 and h[j]==0]
            if len(candidates) == 0:
                break
            # precompute all deltas
            delta_best = 0
            best_pair = None
            for i, j in candidates:
                # compute dC5 when swapping h[i], h[j]
                old_max_shift = np.argmax(S.sum(axis=0)) - (n-1)
                # printf(1000) # defensive
                old_k_vals = S[:, n-1 + old_max_shift]
                old_val = np.max(old_k_vals)
                old_val = old_val
                
                # new: after swap
                h_new = h.copy()
                h_new[i] = 0
                h_new[j] = 1
                # update overlaps for affected shifts
                dS = np.zeros_like(S)
                for k in range(1-n, n):
                    k_idx = k + n -1
                    if abs(k) < 30:  # limit to short shifts to save time
                        s_new = 0
                        for iij in range(n):
                            if 0 <= iij+k < n:
                                s_new += h_new[iij] * h_new[iij+k]
                        dS[:, k_idx] = s_new
                    else:
                        dS[:, k_idx] = S[:, k_idx]  # unchanged
                        
                new_shift = np.argmax(dS.sum(axis=0)) - (n-1)
                new_k_vals = dS[:, n-1 + new_shift]
                new_val = np.max(new_k_vals)
                new_val = new_val
                
                diff = new_val - old_val
                if diff < delta_best:
                    delta_best = diff
                    best_pair = (i, j)
                    
            if best_pair and delta_best < 0:
                i, j = best_pair
                h[i] = 0
                h[j] = 1
                changed = True
                # update S too
                for k in range(1-n, n):
                    k_idx = k + n - 1
                    newval = 0
                    for iij in range(n):
                        if 0 <= iij+k < n:
                            newval += h[iij] * h[iij+k]
                    S[:, k_idx] = newval
                if evaluate_c5(h) < best_c5:
                    best_c5 = evaluate_c5(h)
                    best_h = h.copy()
            else:
                # no better move
                break
        return best_h
    
    # Recompute final h
    h_C, best_C5_C = binary_rounding_local_search(h_B, n_iter=30)        
    
    # Stage D: MILP refinement (if time permits)
    try:
        from cvxpy import *
        c5_milp = None
        if (time.time() - start_time) < budget_s - 30:
            # binary variables and constraint only for 200 points
            x = Variable(n, boolean=True)
            # sum constraint
            prob = Problem(Minimize(0), [sum(x) == n/2])
            prob.solve(solver=CPLEX, verbose=False)
            if prob.status == "optimal":
                h_milp = np.array(x.value).astype(float)
                c5_milp = evaluate_c5(h_milp)
            if c5_milp and c5_milp < best_C5_C:
                best_C5_C = c5_milp
                h_C = h_milp
    except:
        pass
    
    final_h = h_C
    final_c5 = best_C5_C
    final_n = n
    
    return (final_h.tolist(), final_c5, final_n)
