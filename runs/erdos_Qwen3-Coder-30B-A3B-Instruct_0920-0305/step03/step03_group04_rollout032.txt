import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time

# Constants
target_bound = 0.3807
dxs = {46: 2/46, 48: 2/48, 50: 2/50, 60: 2/60, 72: 2/72, 84: 2/84}

# Use pre-imported initial_h_values if available
try:
    initial_h_values
except:
    initial_h_values = np.ones(48) * 0.5

def dx(n):
    return 2.0 / n

def evaluate_h(h):
    h = np.array(h)
    # Compute the correlation (full mode) of h with (1 - h)
    c = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value
    max_correlation = np.max(c)
    # Scale by dx
    c5 = max_correlation * dx(len(h))
    return c5

def correlation_vec(h):
    h = np.array(h)
    return np.correlate(h, 1 - h, mode="full")

def smooth_max(c, alpha=30):
    m = np.max(c)
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

def penalty_sum(h, lambda_=10000):
    s = np.sum(h)
    return lambda_ * (s - len(h)/2)**2

def proj_to_simplex(v, S=None):
    if S is None:
        S = len(v) / 2
    v = np.clip(v, 0, 1)
    if np.abs(np.sum(v) - S) < 1e-12:
        return v
    # First projection: clipping
    v = np.clip(v, 0, 1)
    s = np.sum(v)
    if s > S:
        # Scale down
        v = v * S / s
    elif s < S:
        # Scale up
        scale = S / s
        v = v * scale
        v = np.clip(v, 0, 1)
        # Re-scale to match S
        v = v * S / np.sum(v)
    return v

def binary_initial(n):
    h = np.zeros(n)
    ones = np.random.permutation(n)[:n//2]
    h[ones] = 1
    return h

def swap_and_evaluate(h, i_one, i_zero):
    h = h.copy()
    h[i_one], h[i_zero] = h[i_zero], h[i_one]
    return evaluate_h(h)

def sigmoid(y):
    return 1. / (1 + np.exp(-y))

def logit(p):
    p = np.clip(p, 1e-7, 1 - 1e-7)
    return np.log(p / (1 - p))

def segment_intervals(m, g, delta, n):
    l = 1.0 / m
    h = np.zeros(n)
    dx_val = dx(n)
    # Expand grid to include all points at and near interval boundaries
    for i in range(m):
        start = delta + i * (l + g)
        end = start + l
        start_idx = int(np.floor(start / dx_val))
        end_idx = int(np.ceil(end / dx_val))
        for j in range(max(0, start_idx), min(n, end_idx + 1)):
            h[j] = 1
    # Normalize by exact measure 1.0
    total = np.sum(h) * dx_val
    h = h * (1.0 / total)
    return h

def interval_grid_search(n):
    # Check if exact n matches zero padding
    best_c5 = float('inf')
    best_h = None
    for m in range(2, 9):
        if m == 1:
            continue
        l = 1.0 / m
        max_g = (2.0 - l * m) / (m - 1)
        if max_g < 0:
            continue
        g_vals = [max_g * i for i in [1.0, 0.75, 0.5]]
        for g in g_vals:
            if g < 0:
                continue
            # Evaluate for some gap choices
            for delta in np.linspace(0, g, min(20, int(g/dx(n) + 1))):
                h = segment_intervals(m, g, delta, n)
                c5 = evaluate_h(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
    return best_h, best_c5

def binary_hill_climb(initial_h, max_iter=50000, stagnation_threshold=5 * len(initial_h)):
    h = initial_h.copy()
    c5 = evaluate_h(h)
    best_h = h.copy()
    best_c5 = c5
    
    ones_indices = np.where(h == 1)[0]
    zeros_indices = np.where(h == 0)[0]
    
    # Convert to lists for fast swaps
    i_ones = list(ones_indices)
    i_zeros = list(zeros_indices)
    
    iterations = 0
    stagnant = 0
    
    while iterations < max_iter and stagnant < stagnation_threshold:
        if len(i_ones) == 0 or len(i_zeros) == 0:
            break
        # Pick random 1 and 0
        i_one = np.random.choice(i_ones)
        i_zero = np.random.choice(i_zeros)
        
        h_new = h.copy()
        h_new[i_one], h_new[i_zero] = h_new[i_zero], h_new[i_one]
        c5_new = evaluate_h(h_new)
        
        if c5_new < c5:
            # Accept
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            # Update lists
            i_ones.remove(i_one)
            i_zeros.remove(i_zero)
            if i_one not in i_zeros:
                i_zeros.append(i_one)
            if i_zero not in i_ones:
                i_ones.append(i_zero)
            stagnant = 0
        else:
            stagnant += 1
        
        iterations += 1
        
        # Early exit if good enough
        if c5 < target_bound:
            break
            
    return best_h, best_c5

def smooth_max_optimization(h_start, n, alpha=30, lambda_=10000, maxiter=300, restarts=8):
    best_h = h_start.copy()
    best_c5 = evaluate_h(best_h)
    
    def obj_func(y):
        h = sigmoid(y)
        c = correlation_vec(h)
        c5_soft = smooth_max(c, alpha)
        penalty = penalty_sum(h, lambda_)
        return c5_soft + penalty

    def grad_func(y):
        h = sigmoid(y)
        c = correlation_vec(h)
        m = np.max(c)
        diff = c - m
        exp_term = np.exp(alpha * diff)
        Z = np.sum(exp_term)
        grad_c = np.zeros_like(h)
        # Correlation grad: https://arxiv.org/pdf/1910.06036.pdf        
        # The analytical gradient helps convergence
        grad_sigmoid = sigmoid(y) * (1 - sigmoid(y))  # dσ(x)/dx = σ(x)(1-σ(x))
        # Analytically compute derivative
        exp_all = exp_term / Z  # prob weight        
        # Approximate full gradient using finite difference for time
        return grad_sigmoid

    for _ in range(restarts):
        # Start point with small random noise
        y_init = logit(h_start) + np.random.normal(0, 0.2, n)
        try:
            res = minimize(obj_func, y_init, method='L-BFGS-B', options={'maxiter': maxiter})
            if not res.success:
                continue
            h = sigmoid(res.x)
            h = proj_to_simplex(h, S=n/2)
            c5 = evaluate_h(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                # Early exit
                if best_c5 < target_bound:
                    break
        except Exception as e:
            continue

    return best_h, best_c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    best_overall = None
    best_c5_overall = float('inf')
    best_n_overall = None
    
    search_ns = [48, 60, 72, 84]
    
    for n in search_ns:
        # Stage 2.1: Parametric interval fits
        t_elapsed = time.time() - start_time
        if t_elapsed > budget_s * 0.9:
            break
            
        interval_h, interval_c5 = interval_grid_search(n)
        if interval_c5 < best_c5_overall:
            best_c5_overall = interval_c5
            best_overall = interval_h.copy()
            best_n_overall = n
        
        # Stage 2.2: Binary hill climb
        t_elapsed = time.time() - start_time
        if t_elapsed > budget_s * 0.9:
            break
            
        restarts = 5
        best_h_bin = None
        best_c5_bin = float('inf')
        
        for _ in range(restarts):
            h_start = binary_initial(n)
            h_bin, c5_bin = binary_hill_climb(h_start, max_iter=10000)
            if c5_bin < best_c5_bin:
                best_c5_bin = c5_bin
                best_h_bin = h_bin.copy()
                if c5_bin < target_bound:
                    break
                    
        # Update overall if needed
        if best_c5_bin < best_c5_overall:
            best_c5_overall = best_c5_bin
            best_overall = best_h_bin.copy()
            best_n_overall = n
            if best_c5_overall < target_bound:
                break

        # Stage 2.3: Smooth max optimization
        t_elapsed = time.time() - start_time
        if t_elapsed > budget_s * 0.9:
            break
            
        if best_h_bin is not None:
            smooth_h, smooth_c5 = smooth_max_optimization(best_h_bin, n, alpha=30, lambda_=10000)
            if smooth_c5 < best_c5_overall:
                best_c5_overall = smooth_c5
                best_overall = smooth_h.copy()
                best_n_overall = n
                if best_c5_overall < target_bound:
                    break

        # Stage 2.4: Final polishing
        t_elapsed = time.time() - start_time
        if t_elapsed > budget_s * 0.9:
            break
            
        try:
            # Final polishing using SLSQP
            def objective(h_vals):
                h = np.array(h_vals)
                c = np.correlate(h, 1 - h, mode='full')
                return -np.max(c) * dx(n)

            bounds = [(0, 1) for _ in range(n)]
            constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx(n) - 1}

            # Use best from smooth stage as starting point
            h_beg = best_overall.copy()
            res = minimize(objective, h_beg, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-12, 'maxiter': 500})
            if res.success:
                h_final = res.x
                final_c5 = evaluate_h(h_final)
                if final_c5 < best_c5_overall:
                    best_c5_overall = final_c5
                    best_overall = h_final.copy()
                    best_n_overall = n
        except:
            pass
        
        if best_c5_overall < target_bound:
            break

    # If no improvement was made or we skipped shrinkage, use the initial
    if best_overall is None:
        best_h = np.array(initial_h_values)
        best_c5 = evaluate_h(best_h)
        best_n = len(best_h)
    else:
        best_h = best_overall
        best_c5 = best_c5_overall
        best_n = best_n_overall

    return list(best_h), best_c5, best_n
