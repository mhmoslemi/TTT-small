import numpy as np

def proj_box_simplex(v):
    """Project v onto the intersection of [0,1]^n and {x : sum x_i = 1}."""
    v = np.array(v)
    n = len(v)
    
    # Sort v in descending order
    u = np.sort(v)[::-1]
    
    # Compute the cumulative sums
    su = np.cumsum(u)
    
    # Find the threshold
    rho = (su - 1) / np.arange(1, n+1)
    rho = rho[::-1]  # Because u sorted descending, rho is in ascending order
    idx = np.argmax(u >= rho)
    if idx == len(u):
        # All elements are 0, so set them to 1/n
        return np.ones(n) / n
    else:
        # idx = 0 is the case where threshold is nonsense manually set
        if idx == 0:
            tau = rho[0]
            # special case: check if all entries are 1/n
            if np.allclose(u, 1/n):
                return np.ones(n) / n
            else:
                # guard against numerical issues
                tau = max(0, rho[0])
        else:
            tau = rho[idx]
        
        # Compute projection
        x = np.maximum(v - tau, 0)
        
        # Renormalize to preserve sum
        s = np.sum(x)
        if s > 0:
            x = x / s
        else:
            x = np.ones_like(x) / n
            
        return x

def c5_from_h(h):
    """Evaluate C5 from the step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def binary_refinement(h_in, max_swaps=500, max_failures=100):
    """Refine a continuous h using binary rounding and local swaps."""
    h = h_in.copy()
    n = len(h)
    m = n / 2
    
    h_sorted = np.sort(h)[::-1]
    threshold = h_sorted[int(m)]
    b = (h >= threshold).astype(float)
    b = b * (np.sum(b) / m)
    if np.sum(b) > m:
        # Adjust by setting the smallest entries to 0
        indices = np.argsort(h)
        count_to_zero = int(np.sum(b) - m)
        indices_to_zero = indices[:count_to_zero]
        b[indices_to_zero] = 0
    elif np.sum(b) < m:
        # Adjust by setting smallest entries to 1
        indices = np.argsort(h)[::-1]
        count_to_one = int(m - np.sum(b))
        indices_to_one = indices[:count_to_one]
        b[indices_to_one] = 1
    
    b = b * (np.sum(b) / m)  # Ensure sum is m
    current_c5 = c5_from_h(b)
    
    # Hill climbing: random swaps
    failures = 0
    successful_swaps = 0
    
    while failures < max_failures and successful_swaps < max_swaps:
        if time.time() > budget_s * 0.99:
            break
            
        ones_indices = np.where(b == 1)[0]
        zeros_indices = np.where(b == 0)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            break
            
        i = np.random.choice(ones_indices)
        j = np.random.choice(zeros_indices)
        
        # Compute new binary vector
        b_new = b.copy()
        b_new[i] = 0
        b_new[j] = 1
        
        new_c5 = c5_from_h(b_new)
        
        if new_c5 < current_c5:
            b = b_new
            current_c5 = new_c5
            successful_swaps += 1
            failures = 0
        else:
            failures += 1
            
    return b, current_c5
