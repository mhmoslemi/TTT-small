import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

# Constants
n_points = 48
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def evaluate_h_binary(h_values):
    """Evaluate the C_5 bound for given binary h values."""
    h = np.array(h_values, dtype=int)
    # Compute correlation 
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def evaluate_h_binary_as_float(h_values):
    """Evaluate the C_5 bound for given binary h values as float."""
    h = np.array(h_values, dtype=float)
    # Compute correlation 
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def initialize_binary_h(n):
    """Initialize binary h with exactly n/2 ones."""
    h = np.zeros(n, dtype=int)
    h[:n//2] = 1
    np.random.shuffle(h)
    return h

def compute_correlation_differences(h, i, j):
    """Compute correlation differences for swapping h[i] and h[j] in binary case."""
    n = len(h)
    # Compute the effect on all shifts
    old_corr = np.correlate(h, h, mode="full")
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]
    new_corr = np.correlate(h_new, h_new, mode="full")
    
    # We want to minimize max over correlation, so minimize max difference
    # For our specific problem, min_corr = min over all shifts of sum_i h_i h_{i+d} 
    # C5 = 1 - dx * min_corr, so maximizing min_corr minimizes C5.
    # Just use difference in maximum correlation for now.
    delta = np.max(new_corr) - np.max(old_corr)
    return delta

def simulated_annealing_binary(start_h, max_iter=5000000, initial_temp=0.1, cooling_rate=0.9999):
    """Run simulated annealing on binary h values."""
    h_current = start_h.copy()
    c5_current = evaluate_h_binary(h_current)

    best_h = h_current.copy()
    best_c5 = c5_current

    temp = initial_temp
    iter_count = 0
    
    # Precompute indices with 1s and 0s
    ones_indices = np.where(h_current == 1)[0]
    zeros_indices = np.where(h_current == 0)[0]
    ones_list = list(ones_indices)
    zeros_list = list(zeros_indices)

    for _ in range(max_iter):
        if temp < 1e-10:
            break
            
        # Select a random 1 and 0 to swap
        if len(ones_list) > 0 and len(zeros_list) > 0:
            i = np.random.choice(ones_list)
            j = np.random.choice(zeros_list)
            
            # Compute new correlation
            h_new = h_current.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
            
            c5_new = evaluate_h_binary(h_new)
            delta = c5_new - c5_current
            
            # Accept with probability
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                h_current = h_new
                c5_current = c5_new
                
                # Update indices
                ones_list = list(np.where(h_current == 1)[0])
                zeros_list = list(np.where(h_current == 0)[0])
                
                if c5_current < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_current
            
            iter_count += 1
            # Update temperature
            temp *= cooling_rate
            
            if best_c5 < target_bound:
                break
                
    return best_h, best_c5

def run(seed=42, budget_s=600):
    """Main optimization routine."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Try multiple n values to find better solution
    n_candidates = [48, 56, 64, 80]
    best_h = None
    best_c5 = float('inf')
    
    for n in n_candidates:
        if time.time() - start_time > budget_s * 0.9 if n != n_candidates[-1] else time.time() - start_time > budget_s * 0.9:
            break
            
        # Set up time for this n
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 10:
            break
            
        # Use more time for higher n (more points = harder optimization)
        if n == 48:
            max_iter = int(500000)
        elif n == 56:
            max_iter = int(300000)
        elif n == 64:
            max_iter = int(200000)
        else:  # n == 80
            max_iter = int(150000)
        
        # Initialize binary h
        h_current = initialize_binary_h(n)
        
        # Run simulated annealing
        try:
            h_solution, c5_solution = simulated_annealing_binary(
                h_current, max_iter=max_iter, initial_temp=0.1, cooling_rate=0.9999
            )
            
            if c5_solution < best_c5:
                best_h = h_solution
                best_c5 = c5_solution
                
            if best_c5 < target_bound:
                break
                
        except Exception as e:
            # Fallback and continue to next n
            continue
    
    if best_h is None:
        n = 48
        h_current = initialize_binary_h(n)
        h_solution, c5_solution = simulated_annealing_binary(
            h_current, max_iter=100000, initial_temp=0.1, cooling_rate=0.9999
        )
        best_h = h_solution
        best_c5 = c5_solution
    
    # Return as list of float values
    h_float_list = [float(x) for x in best_h]
    return h_float_list, best_c5, len(best_h)
