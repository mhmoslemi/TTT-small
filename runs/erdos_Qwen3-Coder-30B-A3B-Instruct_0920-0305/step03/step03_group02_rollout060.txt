def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def discrete_optimize_h(h_start, max_iter=300):
    """Optimize h using differential evolution with bounds."""
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    def objective(h):
        return evaluate_h(h)
    
    result = differential_evolution(objective, bounds, maxiter=max_iter, 
                                   constraints=constraints, seed=42,
                                   popsize=20, mutation=(0.5, 1), recombination=0.7)
    return result.x if result.success else h_start

def run(seed=42, budget_s=600):
    """Main optimization routine."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h_current = np.array(initial_h_values)
    # Interpolate to desired length
    if len(h_current) != n_points:
        old_x = np.linspace(0, 2, len(h_current))
        new_x = np.linspace(0, 2, n_points)
        h_current = np.interp(new_x, old_x, h_current)
    
    c5_current = evaluate_h(h_current)
    
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # Main iterative optimization
    iter_num = 0
    max_iter = 200
    
    while time.time() - start_time < budget_s * 0.95:
        # Try global optimization
        try:
            h_opt = discrete_optimize_h(h_current, max_iter=max_iter)
            c5_opt = evaluate_h(h_opt)
            
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                if best_c5 < target_bound:
                    break
        except:
            pass
        
        # Occasionally, use a progressive refinement approach
        if iter_num % 5 == 0 and time.time() - start_time < budget_s * 0.9:
            # Local refinement
            local_h = best_h.copy()
            try:
                # SLM-based refinement
                bounds = [(0, 1) for _ in range(n_points)]
                constraints = {'type': 'eq', 'fun': constraint_eq}
                
                from scipy.optimize import minimize
                result = minimize(evaluate_h, local_h, method='SLSQP', 
                                 bounds=bounds, constraints=constraints,
                                 options={'maxiter': 50, 'ftol': 1e-12, 'gtol': 1e-10})
                if result.success:
                    local_h = result.x
                    local_c5 = evaluate_h(local_h)
                    if local_c5 < best_c5:
                        best_h = local_h.copy()
                        best_c5 = local_c5
            except:
                pass
        
        iter_num += 1
        if best_c5 < target_bound:
            break
    
    # Final normalization to satisfy integral constraint exactly
    scale = 1.0 / (np.sum(best_h) * dx)
    best_h *= scale
    final_c5 = evaluate_h(best_h)
    
    return list(best_h), final_c5, n_points
