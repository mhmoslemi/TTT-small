import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from cvxpy import *
import time
import math

def weight_normalise(x, n):
    """Rescale vector to have sum = n/2"""
    return x * (n / 2) / np.sum(x)

def max_defect(h):
    """Compute exact C5 = max_k integral h(x)(1-h(x+k)) dx"""
    corr = correlate(h, 1-h, mode="full")
    max_corr = np.max(corr)
    return max_corr, np.argmax(corr) - len(h) + 1

def softmax_objective(h, alpha):
    """Compute soft maximum and its gradient"""
    n = len(h)
    # Compute all correlation terms or just use existing one?
    # For performance, precompute indices explicitly:
    corr_vals = []
    corr_indices = []
    for shift in range(1, n):
        # corr[i] = h[i] * (1 - h[i+shift])
        val = 0.0
        for i in range(n - shift):
            val += h[i] * (1 - h[i + shift])
        corr_vals.append(val)
        corr_indices.append(shift)
    
    # Now compute log-sum-exp
    corr_vals = np.array(corr_vals)
    max_val = np.max(corr_vals)
    exp_vals = np.exp(alpha * (corr_vals - max_val))
    Z = np.sum(exp_vals)
    log_Z = max_val + np.log(Z)
    
    # Gradient numerator: sum_s w_s * (1 - h[i+s] - h[i-s] if exists)
    grad = np.zeros_like(h)
    for s_idx, s in enumerate(corr_indices):
        w_s = exp_vals[s_idx] / Z
        for i in range(n - s):
            grad[i] += w_s * (1 - h[i + s])
        for i in range(s, n):
            grad[i] -= w_s * h[i - s]
    
    return (1 / alpha) * log_Z, grad

def build_milp(n):
    # Create variables
    h = Variable(n, boolean=True)
    w = Variable((n, n-1), nonneg=True)  # w[i,s] = h[i] AND h[i+s]
    t = Variable()
    
    # Constraints
    constraints = [
        sum(h) == n / 2,
        t <= 0  # To allow maximization
    ]
    
    # AND constraints: w[i,s] = h[i] AND h[i+s]
    # w[i,s] <= h[i]
    # w[i,s] <= h[i+s]
    # w[i,s] >= h[i] + h[i+s] - 1
    for s in range(1, n):
        for i in range(n - s):
            constraints.append(w[i, s-1] <= h[i])
            constraints.append(w[i, s-1] <= h[i + s])
            constraints.append(w[i, s-1] >= h[i] + h[i + s] - 1)
            # Set up t constraint: sum over all w[i,s] >= t
            constraints.append(sum(w[:, s-1]) >= t)
    
    # Objective: maximize t
    prob = Problem(Maximize(t), constraints)
    
    # Solve with time limit
    try:
        prob.solve(solver=CBC, maxTime=180, seconds=180, ratioGap=0.0)
        if prob.status in [OPTIMAL, FEASIBLE]:
            return np.round(h.value).astype(int)
        else:
            return None
    except:
        return None

def refine_continuous(h_bin, n):
    """Use smooth max objective to refine binary solution to continuous."""
    def cb_hess(x):
        return None  # Let optimizer approximate Hessian
    
    # Determine transition points to convert to continuous formulation
    h_cont = h_bin.copy()
    
    # Fix positions where h_bin = 1 to be exactly 1.
    # Start with binary solution as base
    h0 = h_cont.astype(float)
    h0 = weight_normalise(h0, n)
    
    # Initialize smooth max optimizer
    alpha_values = [5, 15, 40]
    h_refined = h0.copy()
    
    # For close balling: identify patterns and smooth them a bit
    for alpha in alpha_values:
        def obj_func(h):
            val, grad = softmax_objective(h, alpha)
            return val, grad
        
        def callback(xk):
            pass  # For now, no custom tracing
            
        # Optimize using trust-constr with gradient information
        bounds = [(0, 1) for _ in range(n)]
        res = minimize(
            lambda h: obj_func(h)[0],
            x0=h_refined,
            method='trust-constr',
            jac=lambda h: obj_func(h)[1],
            bounds=bounds,
            tol=1e-6,
            callback=callback,
            options={'maxiter': 200, 'verbose': 0}
        )
        
        if res.success:
            h_refined = res.x.copy()
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = weight_normalise(h_refined, n)
        else:
            break  # Early exit on failure
        
    return h_refined

def anneal(h, n):
    """Swap-based stochastic refinement"""
    max_def, worst_idx = max_defect(h)
    
    # The change will be to make the score smaller, so we do not increase
    # the maximum defect: we either accept or reject the move
    corr = correlate(h, 1 - h, mode="full")
    # Find indices where the correlation with 'worst_idx' is high
    # This requires computing the list of indices first
    # Fast way: for shift k, we need h[i] * (1 - h[i+k]) to be large
    eps = 1e-8
    h_bin = (h > 0.5).astype(int)
    for _ in range(5000):
        # Identify good indices to withdraw from, and places to give to
        # Let's compute actual correlations in full
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        # Special index: argmax_k corr[k] where k = len_heuristic - 1 + k_pos
        max_defect_val = np.max(corr)
        worst_shift = np.argmax(corr) - (n - 1)
        
        # Find positions where h[i] > 0 AND h[i+worst_shift] is small
        to_transfer = []
        for i in range(n):
            # account for boundary conditions
            if i + worst_shift < 0 or i + worst_shift >= n:
                continue
            if h[i] > eps and h[i + worst_shift] < 1 - eps:
                to_transfer.append(i)
        # Similarly find good places to add
        to_pickup = []
        for i in range(n):
            if i + worst_shift < 0 or i + worst_shift >= n:
                continue
            if h[i] < 1 - eps and h[i + worst_shift] > eps:
                to_pickup.append(i)
        
        if not to_transfer or not to_pickup:
            break
        
        # Pick random
        import random
        a_i = random.choice(to_transfer)
        b_j = random.choice(to_pickup)
        
        a_k = a_i + worst_shift
        b_k = b_j + worst_shift
        
        # Compute delta in the defect
        h_old = h.copy()  # Just for symbolics now
        
        # Determine how much to move
        delta = min(h[a_i], 1 - h[b_j]) * 0.2
        if delta < 1e-8: continue # too small to matter
        
        # Apply move
        h[a_i] -= delta
        h[b_j] += delta
        h[a_k] -= delta
        h[b_k] += delta
    
        # Normalize again
        h = weight_normalise(h, n)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100  # Starts with this resolution
    dx = 2.0 / n_points
    h_best = np.zeros(n_points)
    
    # Try increasing resolution or instead decrease if already too big
    if (time.time() - start_time) > budget_s * 0.9:
        n_points = 80
        dx = 2.0 / n_points
        
    # Step 1: Solve a MILP
    h0 = build_milp(n_points)
    if h0 is None:
        # fallback to random restart on continuous path
        h0 = np.random.randint(0, 2, n_points)
        h0 = weight_normalise(h0.astype(float), n_points)
    
    # Step 2: Refine using smooth max optimization
    h_refined = refine_continuous(h0, n_points)
    
    # Step 3: Polishing search by swaps  
    h_final = anneal(h_refined, n_points)
    
    c5_bound, _ = max_defect(h_final)
    return (h_final.tolist(), c5_bound, n_points)
