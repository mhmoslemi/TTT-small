```python
import numpy as np
from scipy.optimize import minimize
import time

# Constants
target_bound = 0.3807
n_points = 48  # Even number to allow exact half ones
dx = 2.0 / n_points

def evaluate_h_binary(h_values):
    """Evaluate the C_5 bound for given binary h values."""
    h = np.array(h_values, dtype=int)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def evaluate_h_continuous(h_values):
    """Evaluate the C_5 bound for continuous h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def swap_indices(h_values, i, j):
    """Swap values at indices i and j."""
    h_new = h_values.copy()
    h_new[i] = h_values[j]
    h_new[j] = h_values[i]
    return h_new

def run(seed=42, budget_s=600):
    """Main optimization routine using Simulated Annealing."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize binary pattern with random half ones
    h_current = np.zeros(n_points, dtype=int)
    ones_indices = np.random.choice(n_points, size=n_points // 2, replace=False)
    h_current[ones_indices] = 1
    
    # Evaluate initial solution
    c5_current = evaluate_h_binary(h_current)
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # Set up simulated annealing parameters
    initial_temp = 0.1
    cooling_rate = 0.9999
    temp = initial_temp
    max_iter = 5000000  # Might be reduced by time constraint
    iter_count = 0
    accept_count = 0
    
    # Tracking improvement
    last_improvement = 0
    
    while time.time() - start_time < budget_s * 0.9:
        iter_count += 1
        # Occasionally reset temperature if no improvement for a while
        if iter_count - last_improvement > 10000:
            temp = initial_temp
        
        # Select two random indices: one with value 1 (i), one with value 0 (j)
        ones_indices = np.where(h_current == 1)[0]
        zeros_indices = np.where(h_current == 0)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            continue
        
        i = np.random.choice(ones_indices)
        j = np.random.choice(zeros_indices)
        
        # Create new solution by swapping
        h_new = swap_indices(h_current, i, j)
        c5_new = evaluate_h_binary(h_new)
        
        # Accept or reject based on simulated annealing probability
        delta = c5_new - c5_current
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h_current = h_new
            c5_current = c5_new
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                last_improvement = iter_count
                if best_c5 < target_bound:
                    break
        
        # Update temperature
        temp *= cooling_rate
        
        # Occasionally switch to local optimization
        if iter_count % 1000 == 0:
            # Try local optimization for diversification
            try:
                # Simple hill climbing on a few swaps
                for _ in range(100):
                    ones_indices = np.where(h_current == 1)[0]
                    zeros_indices = np.where(h_current == 0)[0]
                    if len(ones_indices) == 0 or len(zeros_indices) == 0:
                        break
                    i = np.random.choice(ones_indices)
                    j = np.random.choice(zeros_indices)
                    h_test = swap_indices(h_current, i, j)
                    c5_test = evaluate_h_binary(h_test)
                    if c5_test < c5_current:
                        h_current = h_test
                        c5_current = c5_test
                        if c5_current < best_c5:
                            best_h = h_current.copy()
                            best_c5 = c5_current
                            last_improvement = iter_count
            except:
                pass
    
    # Final optimization with SLSQP for continuous relaxation
    h_continuous = best_h.astype(float)
    bounds = [(0, 1) for _ in range(n_points)]
    def objective(h):
        return evaluate_h_continuous(h)
        
    # Ensure sum constraint
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        result = minimize(objective, h_continuous, method='SLSQP', 
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-10})
        if result.success:
            h_continuous = result.x
            best_c5 = evaluate_h_continuous(h_continuous)
    except:
        pass
    
    # Final adjustment to nearest binary values while satisfying integral
    # Normalize continuous solution to sum = 1
    h_final = h_continuous / np.sum(h_continuous) * (n_points / 2)
    # Round to nearest binary values while preserving sum
    # First sort and then assign based on fractional parts
    sorted_indices = np.argsort(h_final)[::-1]
    h_binary = np.zeros(n_points)
    
    # Set exactly half of the highest values to 1
    num_ones = n_points // 2
    for i in range(num_ones):
        h_binary[sorted_indices[i]] = 1
    
    # Adjust values to ensure exact sum
    current_sum = np.sum(h_binary)
    if current_sum != n_points / 2:
        diff = int((n_points / 2) - current_sum)
        if diff > 0:
            # Find zeros to turn to ones
            zero_indices = np.where(h_binary == 0)[0]
            for i in range(diff):
                if i < len(zero_indices):
                    h_binary[zero_indices[i]] = 1
        else:
            # Find ones to turn to zeros
            ones_indices = np.where(h_binary == 1)[0]
            for i in range(-diff):
                if i < len(ones_indices):
                    h_binary[ones_indices[i]] = 0
    
    final_c5 = evaluate_h_binary(h_binary)
    
    return list(h_binary), final_c5, n_points
```
