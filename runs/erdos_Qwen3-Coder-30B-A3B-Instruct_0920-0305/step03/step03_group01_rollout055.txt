def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):

            # Expand block heights to full resolution
            h_expanded = np.repeat(u, n // m)
            # Pad if needed
            h_expanded = np.pad(h_expanded, (0, n - len(h_expanded)), 'constant', constant_values=0)
            h_expanded = h_expanded[:n]
            return evaluate_c5(h_expanded)
            
        # Random search with CMA-ES
        bounds = [(0, 1) for _ in range(m-1)] + [(0, 1)]
        # Custom constraints to remove last variable
        def constraint_eq(u):
            last_u = 0.5 * m - np.sum(u[:-1])
            return last_u - u[-1]
            
        try:
            from cma import CMAEvolutionStrategy
            es = CMAEvolutionStrategy([0.5] * m, 0.2, {'seed': seed, 'popsize': min(4+m, 20)})

                if time.time() - start_time > budget_s * 0.2:

    # Stage B: Projected gradient descent on full resolution
    def stage_b():
        h = best_h.copy()
        # Set up the update rules - subgradient method for max-overlap
        maxiter = 3000
        alpha = 0.1  # Initial step size
        for t in range(maxiter):
            if time.time() - start_time > budget_s * 0.7:
                break
            # Find the worst shift
            corr = correlate(h, 1-h, mode='full')
            k_worst = np.argmax(corr)
            # The gradient is: grad_i(c_k(h)) = (1 - h_{i+k}) - h_{i-k}
            g = np.zeros_like(h)
            for i in range(len(h)):
                g[i] = (1 - h[(i + k_worst) % len(h)]) - h[(i - k_worst) % len(h)]
            h_new = h - alpha * g
            # Re-project to feasible set
            # Box constraints 
            h_new = np.clip(h_new, 0, 1)
            # Equal sum constraint using water-filling
            def project_sum(h_test):
                # First allocate to reduce sumality
                def sum_lam(lam):
                    x = np.clip(h_test - lam, 0, 1)
                    return np.sum(x) - n//2
                # Binary search on lambda
                lo, hi = -2, 3
                for _ in range(50):
                    mid = (lo + hi) / 2
                    if sum_lam(mid) > 0:
                        lo = mid
                    else:
                        hi = mid
                lam = (lo + hi) / 2
                proj_h = np.clip(h_test - lam, 0, 1)
                return proj_h
            
            h = project_sum(h_new)
            c5_candidate = evaluate_c5(h)
            if c5_candidate < best_c5:
                best_h = h.copy()
                best_c5 = c5_candidate
            alpha *= 0.999
        return best_h, best_c5

            c5 = evaluate_c5(b)

        b = best_binary.copy()

            # Adjust the sum
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            if len(ones) > n // 2:
                to_remove = len(ones) - n // 2
                for _ in range(to_remove):
                    idx = np.random.choice(ones)
                    b[idx] = 0
                    ones = np.where(b == 1)[0]

                sum_val = 0
                for i in range(n):
                    if i+k < n:

                overlaps.append(sum_val)

            max_idx = np.argmax(wks)
            # Find improving swap
            best_delta = 0

            total_pairs = n * (n-1) // 2

                    old_sum = b[i] * b[i + max_idx]  # earlier overlap, replaced b[i]

                        old_sum = b[i] * b[i + max_idx] 
                    else:
                        old_sum = 0
                    new_sum = 1 * b[j + max_idx]  # replaced by 1 
                    delta = (1 - old_sum) - (1 - new_sum) 
                    if delta < best_delta:
                        best_delta = delta
                        best_i, best_j = i, j
            if best_i >= 0 and best_delta < 0:
                # Perform swap
                b[best_i] = 0
                b[best_j] = 1
                c5 = evaluate_c5(b)
                if c5 < best_c5_binary:
