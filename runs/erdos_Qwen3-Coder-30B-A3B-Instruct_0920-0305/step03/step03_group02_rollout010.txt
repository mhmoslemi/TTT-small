```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

# Constants
n_points = 80
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def run(seed=42, budget_s=600):
    """Main optimization routine."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Use a piecewise constant approach for better exploration
    # Start with a simple shape (tent function)
    x = np.linspace(0, 2, n_points)
    h_initial = np.maximum(0, 1 - np.abs(x - 1))
    h_initial /= np.sum(h_initial) * dx  # normalize
    
    # Refine using differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}

    # Differential evolution with multiple restarts
    best_h = h_initial.copy()
    best_c5 = evaluate_h(best_h)
    
    # Different configurations to try
    configs = [
        {'method': 'best1bin', 'recombination': 0.9, 'mutation': (0.5, 1), 'maxiter': 500},
        {'method': 'rand1bin', 'recombination': 0.9, 'mutation': (0.5, 1), 'maxiter': 500},
        {'method': 'currenttobest1bin', 'recombination': 0.9, 'mutation': (0.5, 1), 'maxiter': 500},
    ]
    
    for config in configs:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        try:
            result = differential_evolution(evaluate_h, bounds, 
                                          maxiter=config['maxiter'],
                                          popsize=15, 
                                          mutation=config['mutation'],
                                          recombination=config['recombination'],
                                          seed=seed, 
                                          tol=1e-6,
                                          polish=True)
            
            if result.success:
                h_test = result.x
                c5_test = evaluate_h(h_test)
                if c5_test < best_c5:
                    best_h = h_test.copy()
                    best_c5 = c5_test
        except:
            continue
    
    # Local refinement with SLSQP using the best result so far
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_eq}
        result = minimize(evaluate_h, best_h, method='SLSQP', bounds=bounds, 
                         constraints=constraints,
                         options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-10})
        if result.success:
            best_h = result.x
    except:
        pass
    
    # Final evaluation
    final_c5 = evaluate_h(best_h)
    
    # Post-process to ensure constraint satisfaction
    final_h = best_h.copy()
    current_sum = np.sum(final_h) * dx
    if abs(current_sum - 1.0) > 1e-6:
        final_h *= (1.0 / current_sum)
    
    final_c5 = evaluate_h(final_h)
    
    return list(final_h), final_c5, n_points
```
