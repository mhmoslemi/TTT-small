def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

    def solve_milp_binary(n):
        m = n // 2
        dx = 2.0 / n
        # Use CVXPY to solve the integer program
        b = Variable(n, boolean=True)  # binary variables
        y = Variable((n, n), boolean=True)  # intermediate product variables
        t = Variable()  # lower bound on overlaps
        
        # Constraints
        constraints = [
            cp_sum(b) == m,
            t >= 0
        ]
        
        # Product linearisation: y[i,j] = b[i]*b[j]
        for i in range(n):
            for j in range(i, n):
                constraints.extend([
                    y[i, j] <= b[i],
                    y[i, j] <= b[j],
                    y[i, j] >= b[i] + b[j] - 1
                ])
        
        # Overlap constraints
        for s in range(n):
            overlap = 0
            for i in range(n - s):
                overlap += y[i, i + s]
            constraints.append(overlap >= t)
        
        # Objective
        objective = Minimize(-t)
        prob = Problem(objective, constraints)
        
        # Try solving within the remaining time
        remaining_time = budget_s * 0.3 - (time.time() - start_time)
        if remaining_time <= 0.1:
            return None, None

            prob.solve(solver=cp.CBC, verbose=False, 
                      time_limit=remaining_time * 0.9)

        # Project x onto the probability simplex
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if s == 0:
            x = np.ones_like(x) / len(x)
        else:
