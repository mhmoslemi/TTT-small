import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

# Helper functions
def proj_box_simplex(v):
    """Euclidean projection of v onto {x : 0 ≤ x_i ≤ 1, sum x_i = 1}"""
    v = np.array(v)
    n = len(v)
    # Sort v in descending order
    u = np.sort(v)[::-1]
    # Find the threshold
    css = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - css / ind > 0
    if not np.any(cond):
        # All entries are 0
        return np.zeros_like(v)
    k = np.argmax(cond)
    tau = css[k] / (k + 1)
    # Project onto simplex and then clip to [0,1]
    x = np.maximum(v - tau, 0)
    # Normalize to sum to 1
    x = x / np.sum(x)
    return x

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def spectral_search(n, budget_s, seed=42):
    """Spectral alternating projection method"""
    np.random.seed(seed)

    R = 30  # Number of restarts
    it_max = 500
    best_c5 = float('inf')
    best_h = None
    
    # Find an initial estimate for M using the initial sequence

    if len(h_start) != n:
        h_start = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    H_start = np.fft.fft(h_start)
    M_start = np.mean(np.abs(H_start[1:])**2)**0.5  # average magnitude of non-DC components
    
    for r in range(R):
        if time.time() - t0 > budget_s * 0.95:
            break
            
        # Random feasible start
        u = np.random.exponential(1, n)
        h0 = u / np.sum(u) * m
        h0 = np.clip(h0, 0, 1)
        h0 = proj_box_simplex(h0)
        
        # Set M for this run (maintain magnitude flatness)
        M = M_start
        h_current = h0.copy()
        
        last_c5 = float('inf')
        for it in range(it_max):
            if time.time() - t0 > budget_s * 0.95:
                break
            
            # FFT
            H = np.fft.fft(h_current)
            
            # Enforce constant magnitude for non-zero frequencies
            H_target = H.copy()
            H_target[1:] = M * np.exp(1j * np.angle(H[1:]))
            
            # Inverse FFT and real part
            h_tilde = np.real(np.fft.ifft(H_target))
            
            # Project onto box and simplex
            h_current = proj_box_simplex(h_tilde)
            
            # Evaluate
            c5 = evaluate_c5(h_current)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_current.copy()
            
            # Early stopping
            if abs(c5 - last_c5) < 1e-8 and it > 10:
                break
            last_c5 = c5
            
    return best_h, best_c5

def binary_refinement(h):
    """Refine the solution to a binary one using combinatorial swaps"""
    n = len(h)
    # Convert to binary by thresholding
    b = (h > 0.5).astype(float)
    current_sum = np.sum(b)
    target_sum = n / 2
    
    # Adjust the number of 1s
    if current_sum > target_sum:
        # Find indices with 1s but should be 0
        ones = np.where(b == 1)[0]
        # Sort by h values (ascending) - choose to change 1 to 0 the ones with closest to 0.5
        sorted_indices = sorted(ones, key=lambda i: h[i])
        for i in range(int(current_sum - target_sum)):
            b[sorted_indices[i]] = 0
    elif current_sum < target_sum:
        # Find indices with 0s that should be 1
        zeros = np.where(b == 0)[0]
        sorted_indices = sorted(zeros, key=lambda i: -h[i])
        for i in range(int(target_sum - current_sum)):
            b[sorted_indices[i]] = 1
    
    # Local hill-climb using swaps
    h_current = b.copy()
    max_swaps = 5 * n
    max_failures = 2 * n
    failures = 0
    accepted_swaps = 0
    
    while accepted_swaps < max_swaps and failures < max_failures:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Pick two random indices: one 0 and one 1
        ones = np.where(h_current == 1)[0]
        zeros = np.where(h_current == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        idx1 = np.random.choice(ones)
        idx2 = np.random.choice(zeros)
        
        # Perform swap
        h_temp = h_current.copy()
        h_temp[idx1] = 0
        h_temp[idx2] = 1
        
        # Test the new vector
        c5_new = evaluate_c5(h_temp)
        c5_old = evaluate_c5(h_current)
        
        if c5_new < c5_old:
            h_current = h_temp
            accepted_swaps += 1
            failures = 0  # reset failure counter
        else:
            failures += 1
            
    return h_current

        if n % 2 != 0:
            continue
            
        h_candidate, c5_candidate = spectral_search(n, budget_s, seed+10) 
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            
        if best_c5 <= 0.3807:
            break
            
    # Optional binary refinement
    if time.time() - start_time < budget_s * 0.95:
        h_refined = binary_refinement(best_h)
        c5_refined = evaluate_c5(h_refined)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    
    # Final polishing with projected gradient descent
    if time.time() - start_time < budget_s * 0.95:
        # Use the current best_h as starting point
        bounds = [(0, 1) for _ in range(len(best_h))]
        
        def loss_func(h):
            return evaluate_c5(h)
            
        def constraint_eq(h):
            return 1.0 - np.sum(h) * 2.0 / len(h)  # Normalize dx=2/n
            
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        result = minimize(loss_func, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10})
        
        if result.success:
            h_final = np.clip(result.x, 0, 1)
            # Renormalize
            h_final = h_final / np.sum(h_final) * (len(h_final) / 2)
            c5_final = evaluate_c5(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    
    return (best_h.tolist(), best_c5, len(best_h))
