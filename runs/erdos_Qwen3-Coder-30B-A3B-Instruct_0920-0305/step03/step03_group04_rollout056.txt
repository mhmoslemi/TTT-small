import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time
import cvxpy as cp
from cvxpy import Minimize, Problem, PSD, log_sum_exp

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    dx = 2.0 / len(h)
    c5 = max_correlation * dx
    return c5

def correlation_vec(h):
    """Get full correlation vector (2n-1 elements)."""
    h = np.array(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return corr

def logsumexp_smooth_max(c, alpha=30):
    """Soft maximum of correlation vector using log-sum-exp."""
    return log_sum_exp(alpha * c) / alpha

def proj_to_simplex(v, s=1.0):
    """Project vector onto the simplex with sum = s."""
    v = np.array(v)
    v = np.clip(v, 0, 1)
    if np.abs(v.sum() - s) < 1e-8:
        return v
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    rho = np.argmax(v_sorted > (s - cumsum) / np.arange(1, len(v_sorted)+1) )
    theta = (s - cumsum[rho]) / (rho + 1)
    return np.maximum(v - theta, 0)

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def logit(x):
    x = np.clip(x, 1e-6, 1 - 1e-6)
    return np.log(x / (1 - x))

# Constants
N_LIST = [36, 42, 48, 54, 60]
target_bound = 0.3807
MAX_SWAPS = 5 * 10**5
MAX_SWAPS_RESTART = 2 * 10**3
R = 30  # Random hyperplane rounding trials
EPS = 1e-6

def solve_sdp(n):
    """Solve SDP relaxation for given n."""
    dx = 2.0 / n
    # Variables
    x = cp.Variable(n)
    X = cp.Variable((n, n), PSD=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        0 <= x, x <= 1,
        cp.sum(x) == n/2,
        cp.diag(X) == x
    ]
    
    # McCormick envelopes
    for i in range(n):
        for j in range(i+1, n):
            constraints += [
                X[i,j] >= 0,
                X[i,j] <= x[i],
                X[i,j] <= x[j],
                X[i,j] >= x[i] + x[j] - 1
            ]
    
    # Overlap constraints per shift k
    shifts = range(n)
    for k in shifts:
        # Correlation term: sum of h[i] - h[i]*h[i+k] = sum of h[i] - sum of X[i,i+k]
        # We sum over i = 0 .. n-1-k
        x_sum = cp.sum(x[:n-k])
        X_sum = cp.sum(cp.diag(X, k))  # Extract diagonal with offset k
        constraints += [x_sum - X_sum <= t]
    
    # Objective
    objective = Minimize(t)
    
    # Solve
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver="SCS", max_iters=8000, eps=1e-5)
        return x.value, X.value, t.value
    except:
        return None, None, None

def round_from_sdp(x_sol, X_sol, n):
    """Random hyperplane rounding from SDP solution."""
    # Note: X_sol is not necessarily low rank. We could perform eigendecomposition
    # and keep significant components to approximate low rank, but will assume
    # it's numerically sufficient or small enough.
    # For now, use full SVD decomposition:
    try:
        _, V = np.linalg.eigh(X_sol)
        # Take most significant components
        V_truncated = V[:, -min(20, X_sol.shape[0]):]
        # Random hyperplane rounding
        g = np.random.randn(V_truncated.shape[1])
        signs = np.sign(V_truncated.T @ g)
        h_rounded = (signs >= 0).astype(float)
        # Ensure correct count
        diff = int(n/2 - np.sum(h_rounded))
        if diff != 0:
            # Find coordinates with smallest absolute values to flip
            # The row vector V_truncated.T @ g gives the projection value.
            idxs = np.argsort(np.abs(V_truncated.T @ g))
            flip_indices = idxs[:abs(diff)]
            if diff > 0:
                h_rounded[flip_indices] = 0  # Flip 1->0
            else:
                h_rounded[flip_indices] = 1  # Flip 0->1
        return h_rounded
    except:
        # Fallback: use a simple random vector
        return np.random.rand(n) > 0.5

def hill_climb_swap(h, max_swaps=MAX_SWAPS, restart_interval=MAX_SWAPS_RESTART):
    """Combinatorial hill climb using swaps."""
    h = np.array(h)
    n = len(h)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    h = h.copy()
    
    # Precompute correlation vector
    c = correlation_vec(h)
    current_c5 = np.max(c) * 2.0 / n
    
    count = 0
    no_improvement_iter = 0
    best_h = h.copy()
    best_c5 = current_c5
    hit_target = False

    while count < max_swaps and not hit_target:
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Temporarily mutate
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        ones_new = [x for x in ones if x != i] + [j]
        zeros_new = [x for x in zeros if x != j] + [i]

        # Exact evaluation
        c_new = correlation_vec(h_new)
        c5_new = np.max(c_new) * 2.0 / n
        
        count += 1
        if c5_new < current_c5:
            # Accept
            h = h_new
            ones = ones_new
            zeros = zeros_new
            current_c5 = c5_new
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                if best_c5 <= target_bound:
                    hit_target = True
            no_improvement_iter = 0
        else:
            no_improvement_iter += 1

        # Accelerated improvement
        if no_improvement_iter >= restart_interval:
            # Find worst shift
            max_idx = np.argmax(c)
            k = max_idx - (n - 1)
            # k can be negative or positive
            if k >= 0:
                # Correlation indices: [i, i+k] for i = 0..n-1-k
                mismatches = []
                for i in range(n - k):
                    if h[i] == 1 and h[i+k] == 0:
                        mismatches.append(i)
                # Try to fix a mismatch
                if mismatches:
                    for m in mismatches:
                        # Find a zero that we can swap with the mth on
                        try:
                            j = next(z for z in zeros if h[z] == 0 and 
                                   (z == m or z == m + k))  # Naive dual compliance
                            if j == m + k:
                                pass  # Not suitable due to ordering
                            else:
                                # Segment for straightforward swap check
                                h_swap = h.copy()
                                h_swap[m] = 0
                                h_swap[j] = 1
                                c_swap = correlation_vec(h_swap)
                                c5_swap = np.max(c_swap) * 2.0 / n
                                if c5_swap < current_c5:
                                    h = h_swap
                                    ones[ones.index(m)] = j  # Swap the indices
                                    zeros[zeros.index(j)] = m
                                    current_c5 = c5_swap
                                    if current_c5 < best_c5:
                                        best_h = h.copy()
                                        best_c5 = current_c5
                                        if best_c5 <= target_bound:
                                            hit_target = True
                                    no_improvement_iter = 0
                                    count += 1
                                    break  # For next main loop iteration
                        except StopIteration:
                            continue  # No suitable candidate
            # Restart with random vector after n times
            if no_improvement_iter > 5 * restart_interval:
                # too many consecutive failures: restart
                h = np.random.rand(n) > 0.5
                ones = np.where(h == 1)[0].tolist()
                zeros = np.where(h == 0)[0].tolist()
                c = correlation_vec(h)
                current_c5 = np.max(c) * 2.0 / n
                no_improvement_iter = 0
                count = 0  # Not resetting count is intentional as a measure of time elapsed

    return best_h, best_c5

def smooth_refinement(h_binary, max_iter=300, alpha=30, lam=10.0):
    """Use smooth max as surrogate over continuous logit space."""
    h = np.array(h_binary)
    n = len(h)
    # Sigmoid mapping to [0,1]
    def h_oracle(y):
        return sigmoid(y)

    # Objective (smoothed surrogate)
    def obj(y):
        h_eval = h_oracle(y)
        c = correlation_vec(h_eval)
        soft_max = logsumexp_smooth_max(c, alpha)
        penalty = lam * (np.sum(h_eval) - n/2)**2
        return soft_max + penalty

    # Gradient
    def grad(y):
        h_eval = h_oracle(y)
        # Derivative via chain rule:
        # derivative of logsumexp(α c) is (exp(α c) * α * dc)/sum(exp(α c))
        # dc is derivative of c = h * (1 - h) shifted; requires full compute.
        # approx equal to d/dy over logsumexp function via chain rule
        c = correlation_vec(h_eval)
        exp_c = np.exp(alpha * c)
        sum_exp = np.sum(exp_c)
        # Compute df/dx where x=h(y), then df/dy = df/dx * dh/dy
        # dh/dy = sigmoid'(y) = sigmoid(y) * (1 - sigmoid(y))
        dh_dy = h_eval * (1 - h_eval)
        # raw gradient d(logsumexp)/dc * d(c)/dy for each shift
        # This is far from efficient and cannot do it directly: we c manually.
        # For approx purpose, no pair gradient needed: treat as approximate sub-gradient:
        gradient = np.zeros_like(h_eval)
        # This is really <= weak sub-gradient. For fast convergence, we skip true Jacobian/
        # Numerical approximation.
        # Using any value is slightly safer, we'll do a default case  
        return np.ones_like(y)

    # Initialize starting point
    y0 = logit(h + EPS)
    # Track value over time, stops early if no improvement
    # Perform L-BFGS until convergence or max_iter
    try:
        from scipy.optimize import minimize
        # Simple approximation with very informal refine estimation.
        y_opt = y0.copy()
        h_opt = h_oracle(y_opt)
        for _ in range(max_iter):
            c = correlation_vec(h_opt)
            soft_max = logsumexp_smooth_max(c, alpha)
            penalty = lam * (np.sum(h_opt) - n/2)**2
            total = soft_max + penalty
            if total < 1e-8:
                break
            # Simple update: change of variables: change in target function is small?
            y_opt = y_opt - 0.01 * np.random.randn(len(y_opt)) * 0.005
            h_opt = h_oracle(y_opt)
            # Project to sum = n/2
            h_opt = proj_to_simplex(h_opt, n/2)
            c = correlation_vec(h_opt)
            soft_max = logsumexp_smooth_max(c, alpha)
            penalty = lam * (np.sum(h_opt) - n/2)**2
            total = soft_max + penalty
        # Final projection
        h_opt = proj_to_simplex(h_opt, n/2)
        return h_opt
    except Exception as e:
        return h_binary  # Couldn't refine, return binary.

def run(seed=42, budget_s=600):
    """Main run function with all phases."""
    np.random.seed(seed)
    start_time = time.time()

    # Which n values to try
    n_candidates = N_LIST  # [36, 42, 44, 48, 54, 60]
    best_c5 = 100000.0
    best_h = None
    best_n = None

    for n in n_candidates:
        if time.time() - start_time > 0.95 * budget_s:
            break
        dx = 2.0 / n
        print(f"Solving SDP for n = {n}")

        # Phase A - SDP relaxation
        x_sol, X_sol, t_star = solve_sdp(n)
        if x_sol is None:
            print(f"SDP for n={n} failed, skipping.")
            continue
        print(f"SDP finished for n={n}, t* = {t_star}")

        # Phase B - Rounding
        h_rounded = []
        for r in range(R):
            h_r = round_from_sdp(x_sol, X_sol, n)
            c5_r = evaluate_h(h_r)
            if c5_r < best_c5:
                best_h = h_r.copy()
                best_c5 = c5_r
                if best_c5 <= target_bound:
                    print(f"Early exit at n={n}; C5 = {best_c5}")
                    # Final polish
                    try:
                        result = minimize(evaluate_h, best_h, method='SLSQP',
                                          bounds=[(0,1) for _ in range(n)],
                                          constraints={'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0},
                                          options={'maxiter': 100, 'ftol': 1e-15, 'gtol': 1e-12})
                        best_h = result.x
                        best_c5 = evaluate_h(best_h)
                    except Exception:
                        pass
                    return list(best_h), best_c5, n
        
        # Now hill climb from SDP rounded versions:
        for _ in range(10):
            if time.time() - start_time > 0.95 * budget_s:
                break
            h_r = round_from_sdp(x_sol, X_sol, n)
            h_r, c5_r = hill_climb_swap(h_r, max_swaps=20000, restart_interval=2000)
            if c5_r < best_c5:
                best_h = h_r.copy()
                best_c5 = c5_r
                if best_c5 <= target_bound:
                    print(f"Early exit at n={n}, C5 = {best_c5}")
                    # Final polish
                    try:
                        result = minimize(evaluate_h, best_h, method='SLSQP',
                                          bounds=[(0,1) for _ in range(n)],
                                          constraints={'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0},
                                          options={'maxiter': 100, 'ftol': 1e-15, 'gtol': 1e-12})
                        best_h = result.x
                        best_c5 = evaluate_h(best_h)
                    except Exception:
                        pass
                    return list(best_h), best_c5, n

        # If still not found target, do smooth refinement
        print(f"After r={R} rounds: best is metric {best_c5}")
        if best_h is not None:
            h_refined = smooth_refinement(best_h, max_iter=300)
            c5_refined = evaluate_h(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined

    # Phase E - Final exact polishing
    if best_h is not None:
        try:
            dx = 2.0 / len(best_h)
            result = minimize(evaluate_h, best_h, method='SLSQP',
                              bounds=[(0,1) for _ in range(len(best_h))],
                              constraints={'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0},
                              options={'maxiter': 100, 'ftol': 1e-15, 'gtol': 1e-12})
            best_h = result.x
            best_c5 = evaluate_h(best_h)
        except Exception as e:
            print(f"Final polishing failed: {e}")

    return list(best_h), best_c5, n
