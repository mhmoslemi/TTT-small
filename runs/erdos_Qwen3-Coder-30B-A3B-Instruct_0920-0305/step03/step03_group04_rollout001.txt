import numpy as np
from scipy.optimize import minimize
import time

# Constants
n_points = 50
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def smooth_max(values, alpha=30.0):
    """Smooth approximation to max using log-sum-exp."""
    max_val = np.max(values)
    # For numerical stability, subtract max
    exps = np.exp(alpha * (values - max_val))
    return max_val + np.log(np.sum(exps)) / alpha

def smooth_objective(h_values):
    """Smooth objective function for optimization."""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    return smooth_max(correlation)

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

        h_current = np.clip(h_current, 0, 1)
    
    # Initialize bounds and constraints for SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    best_h = h_current.copy()
    best_c5 = evaluate_h(best_h)
