import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Stage A: Low-dim global search with block parametrization
    def stage_a():
        n_points = 200
        m = 10  # number of blocks
        dx = 2.0 / n_points

        def expand_block_params(u):
            # Input: u = [u1, u2, ..., um] where um is derived from the constraint
            if len(u) != m:
                raise ValueError("u must have length m")
            u_m = max(0, min(1, m/2 - np.sum(u[:-1])))
            u = np.append(u[:-1], u_m)
            # Create step function
            block_size = n_points // m
            h = np.repeat(u, block_size)
            # Pad or crop to get exactly n_points
            if len(h) > n_points:
                h = h[:n_points]
            elif len(h) < n_points:
                h = np.pad(h, (0, n_points - len(h)), 'constant', constant_values=0)
            return h

        def objective(u):
            if len(u) != m - 1:
                raise ValueError("u must have length m-1")
            h = expand_block_params(u)
            c5 = evaluate_c5(h)
            return c5

        # Initial point and bounds for u-vector
        u0 = np.ones(m - 1) * 0.5
        bounds = [(0, 1) for _ in range(m - 1)]

        # Use differential evolution with fixed number of evaluations
        res = differential_evolution(
            objective, 
            bounds, 
            seed=seed,
            maxiter=2000,
            popsize=5,  # smaller popsize for faster stage
            strategy='best1bin'
        )
        if res.success:
            h = expand_block_params(res.x)
            return h
        else:
            # fallback to initial h
            return np.array(initial_h_values)

    # Stage B: Full-resolution projected sub-gradient descent
    def stage_b(h0):
        n = len(h0)
        dx = 2.0 / n
        n_iter = 3000

        # Initialize with the stage A solution
        h = h0.copy()
        h_sum = np.sum(h)
        if abs(h_sum - n/2) > 1e-6:
            # normalize to correct sum
            h = h * (n/2) / h_sum

        alpha0 = 0.01
        tolerance = 1e-6
        
        def project_h(h):
            # Project onto constraint sum = n/2 and box [0,1]
            y = h.copy()
            # Bisection for lambda such that sum of clipped(y - lambda) = n/2
            # Setting up binary search on lambda
            low, high = -2.0, 3.0
            for _ in range(50):
                mid = (low + high) / 2.0
                x = np.clip(y - mid, 0, 1)
                s = np.sum(x)
                if s > n/2:
                    low = mid
                else:
                    high = mid
            x = np.clip(y - low, 0, 1)
            return x

        # Find max overlap term
        def find_worst_shift(h):
            h_rev = 1 - h
            corr = correlate(h, h_rev, mode="full")
            max_shift = np.argmax(corr)
            return max_shift

        for t in range(n_iter):
            # Find worst shift
            k_best = find_worst_shift(h)
            # Compute subgradient for the worst shift
            g = np.zeros_like(h)
            g[:-k_best] = 1 - h[k_best:]
            g[k_best:] = - h[:-k_best]
            # Step with decreasing step size
            alpha = alpha0 / np.sqrt(t + 1)
            h_new = h - alpha * g
            # Project
            h = project_h(h_new)
            # Normalize to sum = n/2
            h = h * (n/2) / np.sum(h)

        return h

    # Stage C: Binary rounding + combinatorial local search
    def stage_c(h_continuous):
        n = len(h_continuous)
        n_half = n // 2
        T = 30  # number of random rounds
        best_h = None
        best_c5 = np.inf

        # Define a simple function to extract binary representation from
        # random sample with probabilities proportional to h_continuous
        def round_to_binary(h):
            idx = np.random.permutation(n)
            h_sorted = h[idx]
            # We want exactly n_half ones, so we sample
            # Setting ones where h_sorted is among the top n_half
            sorted_idx = np.argsort(h_sorted)[::-1]
            ones = sorted_idx[:n_half]
            b = np.zeros(n, dtype=int)
            b[idx[ones]] = 1
            return b

        # Multiple random samples
        for _ in range(T):
            b = round_to_binary(h_continuous)
            # Evaluate exact overlap
            c5 = evaluate_c5(b.astype(float))
            if c5 < best_c5:
                best_h = b.copy()
                best_c5 = c5

        # Use this best binary as start for local search
        def compute_shift_contributions(b):
            # Precompute contributions to shifts
            n = len(b)
            contrib = np.zeros((n, n), dtype=int)
            for k in range(1, n):
                contrib[:, k] = b[:-k] * b[k:]  # convolution-style
            return contrib

        def update_shift_contributions(contrib, b, i, j, change):
            # update contrib[1:n, :]
            # change is a list of (i, k) where x[k] is affected by b[i] being toggled
            # For simplicity, let's compute all again for a better implementation
            n = len(b)
            for k in range(1, n):
                if i < n-k:
                    contrib[i, k] = b[i] * b[i+k] if i+k < n else 0
                if j < n-k:
                    contrib[j, k] = b[j] * b[j+k] if j+k < n else 0

        def compute_overlap_contributions(h):
            # h is binary
            n = len(h)
            # sums over shifts
            shift_sums = np.zeros(n)
            # compute ov_k = sum(h_i) - sum(w_{i,k})
            for k in range(1, n):
                shift_sums[k] = np.sum(h) - np.sum(h[:-k] * h[k:])
            # overlap term for shift k is sum(h) - sum(w_{i,k}) = 1 - shift_sums[k]
            return shift_sums

        def worst_overlap(shift_sums):
            return 1 - np.max(shift_sums)

        # Run swaps
        def apply_swap(b, i, j):
            # swap b[i] with b[j]
            b[i], b[j] = b[j], b[i]

        def get_candidate_pairs(b):
            idx_1 = np.where(b == 1)[0]
            idx_0 = np.where(b == 0)[0]
            # Form all pairs i \in 1s, j \in 0s
            pairs = []
            for i in idx_1:
                for j in idx_0:
                    if np.abs(i - j) < n_half:  # to reduce cost, limit pairs
                        pairs.append((i, j))
            return pairs

        b = best_h.copy()
        shift_sums = compute_overlap_contributions(b)
        old_c5 = worst_overlap(shift_sums)

        # Local search loop
        max_iter = 10
        for iter in range(max_iter):
            improved = False
            pairs = get_candidate_pairs(b)
            # Shuffle pairs for randomness
            np.random.shuffle(pairs)
            # Sample a subset
            sample_size = min(500, len(pairs))
            pairs = pairs[:sample_size]

            for i, j in pairs:
                # Temporarily compute functions with swapped pair
                btrial = b.copy()
                btrial[i], btrial[j] = btrial[j], btrial[i]
                new_shift_sums = compute_overlap_contributions(btrial)
                new_c5 = worst_overlap(new_shift_sums)
                delta = new_c5 - old_c5
                if delta < 0:
                    b = btrial
                    shift_sums = new_shift_sums
                    old_c5 = new_c5
                    improved = True

            if not improved:
                break

        return b, old_c5

    # Stage D: MILP refinement (optional approach)
    def stage_d(h_binary):
        n = len(h_binary)
        dx = 2.0 / n

        # Solve exact MILP if time remains
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 30:
            return h_binary.astype(float), evaluate_c5(h_binary.astype(float))

        # Define a small selection of shifts to focus on the worst ones
        # This makes it more tractable
        try:
            h = Variable(n, integer=True)
            w = Variable((n, n), integer=True)
            C = Variable()
            constraints = [h >= 0, h <= 1]
            constraints.append(sum(h) == n // 2)
            for i in range(n):
                for k in range(1, n):
                    if i + k < n:
                        # AND constraints
                        constraints += [w[i, k] <= h[i]]
                        constraints += [w[i, k] <= h[i + k]]
                        constraints += [w[i, k] >= h[i] + h[i + k] - 1]
                        # Symmetry
                        constraints += [w[i, k] >= 0]
            shift_sums = [sum(w[i, k] for i in range(n - k)) for k in range(1, n)]
            for k in range(1, n):
                constraints.append(sum(w[i, k] for i in range(n - k)) + C >= n // 2)

            # Objective
            objective = Minimize(C)
            prob = Problem(objective, constraints)
            # try solving
            # We will increase the max time for solver to see results
            prob.solve(solver=ECOS_BB, max_iters=10000)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                # Extract the binary solution
                h_sol = h.value
                # Normalize and return
                return h_sol, evaluate_c5(h_sol)
            else:
                return h_binary.astype(float), evaluate_c5(h_binary.astype(float))
        except:
            return h_binary.astype(float), evaluate_c5(h_binary.astype(float))

    # Execute stages in order
    h_stage_a = stage_a()
    if (time.time() - start_time) > 0.95 * budget_s:
        return (h_stage_a.tolist(), evaluate_c5(h_stage_a), len(h_stage_a))

    h_stage_b = stage_b(h_stage_a)
    if (time.time() - start_time) > 0.95 * budget_s:
        return (h_stage_b.tolist(), evaluate_c5(h_stage_b), len(h_stage_b))

    h_binary, c5_binary = stage_c(h_stage_b)
    if (time.time() - start_time) > 0.95 * budget_s:
        return (h_binary.tolist(), c5_binary, len(h_binary))

    h_final, c5_final = stage_d(h_binary)

    return (h_final.tolist(), c5_final, len(h_final))
