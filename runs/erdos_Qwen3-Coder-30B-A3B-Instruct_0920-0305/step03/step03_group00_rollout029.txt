import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from cvxpy import *
import time
import cvxpy as cp

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n = 100
    # Check for fallback
    if budget_s < 200:
        n = 80

    # SDP relaxation approach
    print(f"Starting SDP relaxation with n={n}")
    dx = 2.0 / n

    # Define variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.diag(X) == h,  # X[i,i] = h[i]
        X >> 0,           # X is PSD
        0 <= h, h <= 1,   # box constraints
        cp.sum(h) == n / 2  # weight constraint
    ]

    # Add McCormick envelopes
    for i in range(n):
        for j in range(i + 1, n):
            # X[i,j] <= min(h[i], h[j])
            constraints.append(X[i, j] <= h[i])
            constraints.append(X[i, j] <= h[j])
            # X[i,j] >= h[i] + h[j] - 1
            constraints.append(X[i, j] >= h[i] + h[j] - 1)

    # Overlap constraints
    # We approximate overlap: O_s ≈ sum_{i=0}^{n-1-s} X[i,i+s]
    # Loop through all valid shifts (from 1 to n-1)
    for s in range(1, n):
        O_s = cp.sum([X[i, i + s] for i in range(n - s)])
        constraints.append(O_s >= t)

    # Objective
    prob = cp.Problem(cp.Maximize(t), constraints)

    # Try solving with CVXPY + SCS
    try:
        prob.solve(solver=SCS, max_iters=3000, verbose=False, eps=1e-5)
        X_opt = X.value
        h_opt = h.value
        t_val = t.value
        print(f"SDP solved. T = {t_val:.6f}")
    except Exception as e:
        print(f"SDP solver failed with error: {e}")
        # Fall back to random restart rough optimization
        def objective(h):
            return evaluate_c5(h)
        h0 = np.array(initial_h_values)
        h0 = h0[:n]
        result = minimize(objective, h0, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n)],
                          options={'maxiter': 100})
        return (result.x.tolist(), result.fun, n)

    # Check solution quality
    if not h_opt is None:
        # Try using the relaxed h_opt directly
        h_relaxed = np.array(h_opt).copy()
        h_relaxed = (n / 2) * h_relaxed / np.sum(h_relaxed)
        c5_relaxed = evaluate_c5(h_relaxed)
        candidates = [h_relaxed]
    else:
        # Fallback to init h
        h0 = np.array(initial_h_values)
        h0 = h0[:n]
        c5_relaxed = evaluate_c5(h0)
        candidates = [h0]

    print(f"Relaxed h gives C5 = {c5_relaxed:.6f}")

    # Randomized Rounding
    rounded_candidates = []
    num_rounds = min(30, max(5, int(300 / (n / 10))))
    print(f"Performing randomized rounding with {num_rounds} trials")

    try:
        # Compute eigen-decomposition of X_opt
        V, D = np.linalg.eigh(X_opt)
        D = np.maximum(D, 0)  # Ensure non-negative
        sqrt_D = np.sqrt(D)
        Y = V @ np.diag(sqrt_D)

        for r in range(num_rounds):
            if (time.time() - start_time) > (budget_s - 100):
                break
            g = np.random.randn(Y.shape[1])  # Sample random vector
            y = Y @ g  # y = Yg

            # Binary vector
            sorted_indices = np.argsort(y)[::-1]
            h_candidate = np.zeros(n)
            h_candidate[sorted_indices[:n // 2]] = 1

            # Greedy swap improvement
            # We only do cheap local search
            def improved_candidate(h_in, max_swaps=100):
                h_mod = h_in.copy()
                swaps = 0
                for swap in range(max_swaps):
                    best_delta = 0
                    best_i, best_j = -1, -1
                    for i in range(n):
                        if h_mod[i] == 0:
                            continue
                        for j in range(n):
                            if h_mod[j] == 1:
                                continue
                            # Test swap
                            h_new = h_mod.copy()
                            h_new[i] = 0
                            h_new[j] = 1
                            c5_new = evaluate_c5(h_new)
                            c5_old = evaluate_c5(h_mod)
                            delta = c5_old - c5_new
                            if delta > best_delta:
                                best_delta = delta
                                best_i, best_j = i, j
                    if best_i >= 0:
                        h_mod[best_i] = 0
                        h_mod[best_j] = 1
                        swaps += 1
                    else:
                        break
                return h_mod

            h_candidate = improved_candidate(h_candidate)
            rounded_candidates.append(h_candidate)
    except Exception as e:
        print(f"Error in rounding: {e}")

    candidates.extend(rounded_candidates)

    # Local refinement
    print("Performing local refinement via trust-constr")

    def soft_max_objective(h, alpha=10.0):
        """ Soft max approximation of max overlap """
        dx = 2.0 / len(h)
        all_overlaps = []
        for k in range(1, len(h)):
            shift_h = np.concatenate([h[k:], np.zeros(k)])
            overlap = np.sum(h * (1 - shift_h)) * dx
            all_overlaps.append(overlap)
        overlaps = np.array(all_overlaps)
        return np.log(np.sum(np.exp(alpha * overlaps))) / alpha

    def soft_grad(h, alpha=10.0):
        """Gradient of soft max overlap""" 
        dx = 2.0 / len(h)
        n = len(h)
        all_overlaps = []
        weights = []
        for k in range(1, n):
            shift_h = np.concatenate([h[k:], np.zeros(k)])
            overlap = np.sum(h * (1 - shift_h)) * dx
            all_overlaps.append(overlap)

        overlaps = np.array(all_overlaps)
        exps = np.exp(alpha * overlaps)
        Z_inv = 1.0 / np.sum(exps)
        weights = exps * Z_inv
        # Compute gradient_terms
        grad = np.zeros(n)
        for k, weight in enumerate(weights):
            # derivative of overlap w.r.t. h:
            shift_h = np.concatenate([h[k+1:], np.zeros(k+1)])
            temp = np.zeros(n)
            temp[:-k-1] = 1 - shift_h[k+1:]
            temp[k+1:] = -(1 - h[:-k-1])
            grad += weight * temp
        return -grad * alpha

    def refine_candidate(h_start):
        # Feed it to optimizer
        def fun(x):
            return -soft_max_objective(x, alpha=20)

        def jac(x):
            return -soft_grad(x, alpha=20)

        # Constraints
        bounds = [(0, 1) for _ in range(n)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2}
        try:
            res = minimize(fun, h_start, method='trust-constr', jac=jac,
                           bounds=bounds, constraints=cons, options={
                               'maxiter': 100, 'gtol': 1e-8, 'ftol': 1e-8})
            if res.success:
                return res.x
        except:
            pass
        return h_start

    best_c5 = np.inf
    best_h = None

    # Refine candidates in order
    for i, h_in in enumerate(candidates):
        if (time.time() - start_time) > (budget_s - 50):
            break
        h_refined = refine_candidate(h_in)
        c5_refined = evaluate_c5(h_refined)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    
    # Final fallback
    if best_h is None:
        h0 = np.array(initial_h_values)
        h0 = h0[:n]
        def objective(h):
            return evaluate_c5(h)
        result = minimize(objective, h0, method='SLSQP', bounds=[(0, 1) for _ in range(n)])
        best_h = result.x
        best_c5 = result.fun

    return (best_h.tolist(), best_c5, n)
