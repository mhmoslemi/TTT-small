def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def binary_exact_optimization(n, time_limit):
    try:
        import cvxpy as cp
        from cvxpy import Variable, Minimize, Problem, sum_entries, multiply, matmul, ones, Constraint
        
        # Setup
        m = n // 2
        dx = 2.0 / n
        
        # Binary variables for step function
        b = Variable(n, boolean=True)
        
        # Product binary variables
        y = Variable((n, n), boolean=True)
        
        # Lower bound on overlaps
        t = Variable()
        
        # Constraints
        constraints = []
        
        # Weight constraint
        constraints.append(sum_entries(b) == m)
        
        # Product constraints
        for i in range(n):
            for j in range(i, n):
                constraints.append(y[i, j] <= b[i])
                constraints.append(y[i, j] <= b[j])
                constraints.append(y[i, j] >= b[i] + b[j] - 1)
        
        # Shift overlap constraints
        for s in range(n):
            # Compute overlap for shift s
            overlap = 0
            for i in range(n):
                if i + s < n:
                    overlap += y[i, i + s]
            constraints.append(overlap >= t)
        
        # Objective
        objective = Minimize(-t)
        prob = Problem(objective, constraints)
        
        # Check time limit
        prob.solve(solver=cp.CBC, timelimit=time_limit)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURATE]:
            t_val = -prob.value
            b_opt = np.array([float(b[i].value) for i in range(n)])
            
            # Only return improvement if we found a better bound
            c5_val = (m - t_val) * dx
            return b_opt, c5_val
            
    except Exception as e:
        return None
    return None


def binary_simulated_annealing(n, time_limit):
    # Stochastic search for high-resolution improvement
    max_iterations = 10000
    T0 = 1.0
    T_min = 1e-6
    alpha = 0.99
    
    start = time.time()
    if n <= 0:
        return None
    dx = 2.0 / n
    
    h_best = np.ones(n)
    h_best[n//2:] = 0
    np.random.shuffle(h_best)
    h_best = np.array((h_best > 0.5), dtype=float)
    
    # Ensure exact weight
    ones_count = np.sum(h_best)
    if ones_count > n // 2:
        # Change some 1s to 0s
        indices = np.where(h_best == 1)[0]
        to_change = int(ones_count - n // 2)
        for idx in np.random.choice(indices, to_change, replace=False):
            h_best[idx] = 0
    elif ones_count < n // 2:
        # Change some 0s to 1s
        indices = np.where(h_best == 0)[0]
        to_change = int(n // 2 - ones_count)
        for idx in np.random.choice(indices, to_change, replace=False):
            h_best[idx] = 1
    
    c5_best = evaluate_c5(h_best)
    
    T = T0
    iterations = 0
    
    while time.time() - start < time_limit and T > T_min:
        iterations += 1
        # Make a random move: swap a 0 and a 1
        ones_indices = np.where(h_best == 1)[0]
        zeros_indices = np.where(h_best == 0)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            break
            
        i = np.random.choice(ones_indices)
        j = np.random.choice(zeros_indices)
        
        h_new = h_best.copy()
        h_new[i] = 0
        h_new[j] = 1
        
        c5_new = evaluate_c5(h_new)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
        else:
            # Accept with probability exp(-(E' - E)/T)
            delta_E = c5_new - c5_best
            if np.random.rand() < np.exp(-delta_E / T):
                h_best = h_new
                c5_best = c5_new
                
        T *= alpha
        if iterations % 5000 == 0:
            if (time.time() - start) >= time_limit:
                break
    
    return (h_best, c5_best)


def continuous_refinement(h_initial, time_budget):
    start = time.time()
    n = len(h_initial)
    h_current = h_initial.copy()
    dx = 2.0 / n
    
    def compute_gradients(h):
        target_shifts = []
        c_vals = []
        for s in range(n):
            # Evaluate and track s with max correlation
            corr = correlate(h, 1 - h, mode="full")[n-1-s : 2*n-1-s]
            c_val = np.sum(corr)
            c_vals.append(c_val)
        max_val = max(c_vals)
        target_shifts = [i for i, val in enumerate(c_vals) if val == max_val]
        
        # Compute multigradient
        g = np.zeros(n)
        for s in target_shifts:
            for i in range(n):
                j = i + s
                g[i] += 1
                if 0 <= j < n:
                    g[i] -= h[j]
                j = i - s
                if 0 <= j < n:
                    g[i] -= h[j]
        return g / len(target_shifts)
    
    def project_to_simplex(h):
        # Project onto the simplex: h_i >= 0, sum(h) = 1
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s > 0:
            h = h / s
        else:
            h = np.ones_like(h) / len(h)
        return h
    
    last_c5 = evaluate_c5(h_current)
    iter_num = 0
    
    while time.time() - start < time_budget:
        iter_num += 1
        g = compute_gradients(h_current)
        
        # Update
        alpha = 1e-2 / (1 + iter_num ** 0.5)
        h_new = h_current - alpha * g
        h_new = np.clip(h_new, 0, 1)
        
        # Project to feasible set: sum(h) * dx = 1
        scale = 1.0 / (np.sum(h_new) * dx)
        h_new = h_new * scale
        
        c5_new = evaluate_c5(h_new)
        if c5_new < last_c5:
            h_current = h_new
            last_c5 = c5_new
        else:
            # Accept with a chance
            if np.random.rand() < np.exp(-(c5_new - last_c5) / 0.01):
                h_current = h_new
                last_c5 = c5_new
        if iter_num > 5000:
            break
    return h_current
