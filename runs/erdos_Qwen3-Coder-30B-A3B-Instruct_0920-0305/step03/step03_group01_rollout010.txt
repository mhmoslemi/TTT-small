import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_onto_box_and_hyperplane(y, target):
    """Project y onto the feasible region: 0≤x≤1 and sum x = target"""
    def sum_clip(λ):
        return np.clip(y - λ, 0, 1).sum()
    
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = sum_clip(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0, 1)

        term1 = np.empty(n)

        term1[n - shift:] = 1

    h_new = h - step_size * grad
    
    # Project onto feasible region (box and sum constraint)
    h_new = project_onto_box_and_hyperplane(h_new, n / 2)
    
    return h_new
