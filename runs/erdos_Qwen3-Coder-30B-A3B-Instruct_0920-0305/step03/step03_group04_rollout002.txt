```python
import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time
import math

# Constants and helper functions
n_values = [48, 60, 72, 84]
dx = lambda n: 2.0 / n
pi = math.pi
sigmoid = lambda y: 1.0 / (1.0 + np.exp(-y))
logit = lambda p: np.log(p / (1.0 - p))
proj_to_simplex = lambda v, S=1.0: np.clip(v, 0, 1) * (S / np.sum(np.clip(v, 0, 1)))

def correlation_vec(h):
    return np.correlate(h, 1 - h, mode="full")

def evaluate(h):
    c = correlation_vec(h)
    return dx(len(h)) * np.max(c)

def smooth_max(c, alpha=30.0):
    # Soft max = max + log(sum(exp(alpha*(c-max)))) / alpha
    m = np.max(c)
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

def penalty_sum(h, lambda_sum=10000):
    sum_h = np.sum(h)
    return lambda_sum * (sum_h - 0.5 * len(h))**2

def binary_initial(n):
    h = np.zeros(n)
    ones_indices = np.random.choice(n, size=n//2, replace=False)
    h[ones_indices] = 1
    return h

def swap_and_evaluate(h, i_one, i_zero):
    h_new = h.copy()
    h_new[i_one], h_new[i_zero] = h_new[i_zero], h_new[i_one]
    return evaluate(h_new)

def get_interval_vec(h, n, m, g, delta):
    """Construct a pattern from m unit intervals, gap g, offset delta"""
    dx_val = dx(n)
    # Determine start and end of interval bodies
    interval_span = 1.0 / m
    interval_start = delta
    interval_end = interval_start + interval_span
    
    h_new = np.zeros(n)
    for interval in range(m):
        interval_start = delta + interval * (interval_span + g)
        interval_end = interval_start + interval_span
        
        start_idx = int(np.floor(interval_start / dx_val))
        end_idx = int(np.ceil(interval_end / dx_val))
        
        h_new[start_idx:end_idx] = 1.0
    
    # Clip to [0,1] and ensure sum = n/2
    h_new = np.clip(h_new, 0, 1)
    h_new *= (0.5 * n) / np.sum(h_new) if np.sum(h_new) > 0 else 1
    return h_new

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def check_time():
        return time.time() - t0 > budget_s * 0.95

    # Stage 2.1 – Interval family search
    for n in n_values:
        T = budget_s * 0.15
        t_start = time.time()
        best_from_family = None
        best_c5_from_family = float('inf')
        
        # Guess interval configuration with gaps = 0
        for m in range(2, 9):
            if m > 2:
                interval_len = 1.0 / m
                gap_len = max(0, (2 - m * interval_len) / (m - 1))
                g_vals = [gap_len * x for x in [1.0, 0.75, 0.5]]
            else:
                g_vals = [0.0]
            for g in g_vals:
                if 2 < m * (1/m) + (m - 1) * g:
                    continue
                for delta in np.arange(0, g + 1e-10, dx(n) / 2):
                    try:
                        h = get_interval_vec(None, n, m, g, delta)
                        c5 = evaluate(h)
                        if c5 < best_c5_from_family:
                            best_c5_from_family = c5
                            best_from_family = h.copy()
                    except:
                        continue
                if check_time() or time.time() - t_start > T:
                    break
            if check_time() or time.time() - t_start > T:
                break
        if best_c5_from_family < best_c5:
            best_h = best_from_family.copy()
            best_c5 = best_c5_from_family
            best_n = n
        if check_time():
            break
    
    # Stage 2.2 – Binary hill climbing on multiple n
    for n in n_values:
        if check_time():
            break
        T = budget_s * 0.5 / len(n_values)
        t_start = time.time()
        
        h_best_binary = binary_initial(n)
        c5_best_binary = evaluate(h_best_binary)
        
        # Keep track of 1's and 0's indices
        ones_idx = np.where(h_best_binary == 1)[0]
        zeros_idx = np.where(h_best_binary == 0)[0]
        
        max_iter = int(T * 2000)
        iter_num = 0
        
        # Occasionally reinitialise or stop early if target met
        stagnation_counter = 0
        for _ in range(max_iter):
            if check_time() or time.time() - t_start > T:
                break
            # Pick indices to swap
            i_one = np.random.choice(ones_idx)
            i_zero = np.random.choice(zeros_idx)
            # Swap
            h_new = h_best_binary.copy()
            h_new[i_one], h_new[i_zero] = h_new[i_zero], h_new[i_one]
            c5_new = evaluate(h_new)
            
            if c5_new < c5_best_binary:
                # Accept the change
                h_best_binary = h_new
                c5_best_binary = c5_new
                # Update index lists
                ones_idx = np.where(h_best_binary == 1)[0]
                zeros_idx = np.where(h_best_binary == 0)[0]
                stagnation_counter = 0
            else:
                stagnation_counter += 1
                
            if stagnation_counter >= 5 * n:
                h_best_binary = binary_initial(n)
                c5_best_binary = evaluate(h_best_binary)
                ones_idx = np.where(h_best_binary == 1)[0]
                zeros_idx = np.where(h_best_binary == 0)[0]
                stagnation_counter = 0
                
            if c5_best_binary <= 0.3807:
                break
                
            iter_num += 1

        if c5_best_binary < best_c5:
            best_h = h_best_binary
            best_c5 = c5_best_binary
            best_n = n
    
    # Stage 2.3 – Smooth max L-BFGS optimization
    for n in n_values:
        if check_time():
            break
        h_start = best_h if best_n == n else binary_initial(n)
        
        t_start = time.time()
        T = budget_s * 0.25
        
        best_h_continuous = h_start.copy()
        best_c5_continuous = evaluate(h_start)
        
        # Use L-BFGS-B on smooth surrogate
        def smooth_objective(y):
            h = sigmoid(y)
            c = correlation_vec(h)
            return smooth_max(c) + penalty_sum(h)
            
        for _ in range(8):  # K restarts
            if check_time() or time.time() - t_start > T:
                break
            y_start = logit(h_start) + np.random.normal(0, 0.2, len(h_start))
            try:
                res = minimize(
                    smooth_objective,
                    y_start,
                    method='L-BFGS-B',
                    options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-10}
                )
                h_opt = sigmoid(res.x)
                h_opt = proj_to_simplex(h_opt, 0.5 * n)
                c5_opt = evaluate(h_opt)
                
                if c5_opt < best_c5_continuous:
                    best_c5_continuous = c5_opt
                    best_h_continuous = h_opt.copy()
            except:
                continue
                
            if best_c5_continuous <= 0.3807:
                break
                
        if best_c5_continuous < best_c5:
            best_h_continuous = proj_to_simplex(best_h_continuous, 0.5 * n)
            best_c5 = best_c5_continuous
            best_h = best_h_continuous.copy()
            best_n = n
            
    # Stage 2.4 – Final polishing with SLSQP
    if not check_time():
        t_start = time.time()
        T = budget_s * 0.1
        
        def objective(h):
            return evaluate(h)
            
        bounds = [(0, 1) for _ in range(len(best_h))]
        constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) * dx(len(h)) - 1}
        try:
            res = minimize(
                objective,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'ftol': 1e-12, 'maxiter': 500}
            )
            if res.success:
                final_h = res.x
                final_c5 = evaluate(final_h)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return list(best_h), best_c5, best_n
```
