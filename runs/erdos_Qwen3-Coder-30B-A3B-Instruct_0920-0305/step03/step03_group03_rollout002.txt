import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cvx_sum, Bool, atLeast, atMost
import cvxpy as cp
from cvxpy import ECOS_BB, GLPK_MI, CBC


def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx


def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    best_h = np.array(initial_h_values)
    best_c5 = evaluate_c5(best_h)
    n_points = len(best_h)

    # Early exit if target already met
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)

    def solve_milp(n):
        # Binary optimization using MILP solver
        m = n // 2  # Target weight
        # Use ECOS_BB for binary variables, with time limit
        time_limit = 0.5 * budget_s
        h = Variable(n, boolean=True)
        t = Variable()
        # Weight constraint
        constraints = [cp.sum(h) == m, t <= 0]
        # Compute overlaps
        for s in range(n):
            overlap = 0
            for i in range(n - s):
                overlap += h[i] * h[i + s]
            constraints.append(overlap >= t)
        prob = Problem(Maximize(t), constraints)
        try:
            prob.solve(solver=ECOS_BB, verbose=False, time_limit=time_limit)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                t_val = t.value
                c5_val = (m - t_val) * 2.0 / n
                h_val = h.value
                if c5_val < best_c5:
                    return h_val, c5_val
        except:
            pass
        return None, None

    # 1. Try exact binary optimization for small grid
    for n in [42, 44, 48, 52, 56, 60]:
        if (time.time() - start_time) > budget_s * 0.9:
            break
        h_bin, c5_bin = solve_milp(n)
        if h_bin is not None and c5_bin < best_c5:
            best_h = h_bin
            best_c5 = c5_bin
            n_points = n

    # 2. Simulated Annealing on larger grids
    def sa_search(n, n_steps):
        h = np.zeros(n)
        h[:n//2] = 1
        np.random.shuffle(h)
        current_c5 = evaluate_c5(h)
        best_h = h.copy()
        best_c5 = current_c5
        T = 1.0
        T_min = 1e-5
        alpha = 0.99
        t = 0
        max_steps = n_steps
        while t < max_steps and (time.time() - start_time) < budget_s * 0.9:
            s = np.random.randint(0, n)
            t1 = np.random.randint(0, n)
            h_new = h.copy()
            h_new[s], h_new[t1] = h_new[t1], h_new[s]  # swap
            new_c5 = evaluate_c5(h_new)
            if new_c5 < current_c5:
                current_c5 = new_c5
                h = h_new
                if new_c5 < best_c5:
                    best_h = h_new
                    best_c5 = new_c5
            else:
                delta = new_c5 - current_c5
                if np.random.rand() < np.exp(-delta / T):
                    current_c5 = new_c5
                    h = h_new
            t += 1
            T = max(T_min, T * alpha)
        return best_h, best_c5

    # Do large grid searches using SA
    large_n = [44, 48, 52, 56, 60]
    for n in large_n:
        if (time.time() - start_time) > budget_s * 0.9:
            break
        h_sa, c5_sa = sa_search(n, 10 * n * n)
        if c5_sa < best_c5:
            best_h = h_sa
            best_c5 = c5_sa
            n_points = n

    # 3. Refinement with continuous methods
    def gradient_descent(h0, max_iter=500):
        h = h0.copy()
        dx = 2.0 / len(h)
        c0 = evaluate_c5(h)
        for k in range(max_iter):
            corr = correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(corr)
            s = max_idx - (len(h) - 1)
            grad = np.zeros_like(h)
            for i in range(len(h)):
                if i + s < len(h) and i + s >= 0:
                    grad[i] = 1 - h[i + s]
            lr = 0.001 / (1 + k**0.5)
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            scale = 1.0 / (np.sum(h_new) * dx)
            h_new *= scale
            c_new = evaluate_c5(h_new)
            if c_new < c0:
                h = h_new
                c0 = c_new
            else:
                break
        return h, c0

    # Multiple restarts
    h_refined, c5_refined = gradient_descent(best_h, 2000)
    if c5_refined < best_c5:
        best_h = h_refined
        best_c5 = c5_refined

    # Fallback to original algorithm if needed
    if (time.time() - start_time) < budget_s * 0.9 and best_c5 > 0.3807:
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': lambda h: np.sum(h) * 2.0 / n_points - 1}
            de_res = differential_evolution(evaluate_c5, bounds, maxiter=200,
                                           popsize=20, mutation=(0.5, 1),
                                           recombination=0.7, strategy='rand1bin')
            if de_res.success:
                scale = 1.0 / (np.sum(de_res.x) * 2.0 / n_points)
                h_candidate = de_res.x * scale
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        except:
            pass

    return (best_h.tolist(), best_c5, n_points)
