import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from cvxpy import *
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    dx = 2.0 / len(h)
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Phase A: SDP Relaxation
    n = 100
    dx = 2.0 / n

    # Convex relaxation variables
    h = Variable(n)
    X = Variable((n, n), PSD=True)
    t = Variable()

    # Constraints
    constraints = [
        # X = h h^T (diagonal)
        [X[i, i] == h[i] for i in range(n)],
        # Bounds on h
        h >= 0,
        h <= 1,
        # Weight constraint
        sum(h) == n / 2,
        # McCormick envelopes for X[i,j] ≤ h[i], X[i,j] ≤ h[j], X[i,j] ≥ h[i]+h[j]-1
        # We consider only pairs that are relevant to shifts (for efficiency)
    ]

    # Add overlap constraints
    for s in range(1, n):
        # Overlap for shift s
        overlap_s = sum(X[i, i+s] for i in range(n - s))
        constraints.append(overlap_s >= t)

    # SDP objective
    objective = Maximize(t)

    # Create and solve problem
    prob = Problem(objective, constraints)
    
    # Solver settings
    # time limit = 150 seconds
    # and multiple rounds of solving still feasible within budget
    
    try:
        prob.solve(solver=SCS, max_iters=5000, feastol=1e-5, abstol=1e-5)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h_sdp_val = h.value
            t_opt = t.value
        else:
            # If SDP fails, fall back to a lower n
            n = 80
            dx = 2.0 / n
            h = Variable(n)
            X = Variable((n, n), PSD=True)
            t = Variable()
            constraints = [
                [X[i, i] == h[i] for i in range(n)],
                h >= 0,
                h <= 1,
                sum(h) == n / 2,
            ]

            # Overlap constraints
            for s in range(1, n):
                overlap_s = sum(X[i, i+s] for i in range(n - s))
                constraints.append(overlap_s >= t)

            objective = Maximize(t)
            prob = Problem(objective, constraints)
            
            prob.solve(solver=SCS, max_iters=5000, feastol=1e-5, abstol=1e-5)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_sdp_val = h.value
                t_opt = t.value
            else:
                # worse fallback: random restart
                best_h = np.array(initial_h_values)
                best_c5 = evaluate_c5(best_h)
                return (best_h.tolist(), best_c5, len(best_h))
    except:
        # If we cannot solve the SDP, fall back to random restart
        try:
            return run(seed, budget_s=100)  # Try again with less time
        except:
            # Final fallback
            best_h = np.array(initial_h_values)
            best_c5 = evaluate_c5(best_h)
            return (best_h.tolist(), best_c5, len(best_h))

    # Normalize h to satisfy the integral constraint (sum = n/2) 
    h_sdp_val = np.clip(h_sdp_val, 0, 1)
    h_sdp_val = h_sdp_val / np.sum(h_sdp_val) * (n / 2)

    # Phase B: Randomized rounding of h_sdp
    candidates = [h_sdp_val.copy()]
    np.random.seed(seed + 1)
    # Round to binary vectors with random vector from Cholesky decomposition
    try:
        U, s, Vh = np.linalg.svd(X.value)
        rank = np.sum(s > 1e-8)
        Y = U[:, :rank] * np.sqrt(s[:rank])
        # Try to generate roundings with differing orders of magnitude, but manageable number
        rounds = 20 if time.time() - start_time < 400 else 3
        for i in range(rounds):
            if (time.time() - start_time) > budget_s * 0.7:
                break
            # Generate a random vector g ~ N(0, I)
            g = np.random.normal(size=rank)
            y = Y @ g  # sample from Gaussian rounded to continuous
            # Turn into binary vector
            indices = np.argsort(y)[::-1]  # Get indices of sorted values (descending)
            h_bin = np.zeros(n)
            h_bin[indices[:n//2]] = 1
            # Greedy improvement
            h_bin = swap_improve(h_bin, n, dx, time.time(), start_time, budget_s)
            candidates.append(h_bin)
    except:
        pass

    # Phase C: Local refinement
    def softmax_corr(h, alpha=1.0):
        """Softmax approximation to max"""
        dx = 2.0 / len(h)
        corr_vals = []
        for k in range(1, len(h)):
            corr = np.sum(h[:-k] * (1 - h[k:]))
            corr_vals.append(corr * dx)
        corr_vals = np.array(corr_vals)
        return np.log(np.sum(np.exp(alpha * corr_vals))) / alpha

    def softmax_grad(h, alpha=1.0):
        """Gradient of the soft max approximation"""
        dx = 2.0 / len(h)
        n = len(h)
        corr_vals = []
        weights = []

        for k in range(1, len(h)):
            corr = np.sum(h[:-k] * (1 - h[k:]))
            corr_vals.append(corr * dx)
        corr_vals = np.array(corr_vals)

        exp_vals = np.exp(alpha * corr_vals)
        weights = exp_vals / np.sum(exp_vals)

        grad = np.zeros_like(h)
        for i in range(n):
            for k in range(1, n):
                if i+k < n:
                    grad[i] += weights[k-1] * (1 - h[i+k])
                if i-k >= 0:
                    grad[i] += weights[k-1] * (1 - h[i-k])
        return grad
        
    def opt_with_smoothing(h0, max_iter=200):
        dx = 2.0 / len(h0)
        def obj(h):
            # Simple smooth max
            return softmax_corr(h[:len(h0)], alpha=40)  # Final level just to make sure
        
        def jac(h):
            return softmax_grad(h[:len(h0)], alpha=40)
            
        # Project to feasible space
        bounds = [(0,1) for _ in range(len(h0))]
        res = minimize(obj, h0, method='trust-constr', jac=jac, bounds=bounds,
                       options={'maxiter': max_iter, 'gtol': 1e-8, 'ftol': 1e-8},
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n/2})
        if res.success:
            return res.x
        else:
            # fall back to old values if crack
            return h0.copy()
                
    # Now run the refinement
    best_c5 = evaluate_c5(h_sdp_val)
    best_h = h_sdp_val.copy()

    for i, h0 in enumerate(candidates):
        if (time.time() - start_time) > budget_s * 0.9:
            break
        h_refined = opt_with_smoothing(h0)
        c5_candidate = evaluate_c5(h_refined)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_refined
    # Final check
    return (best_h.tolist(), best_c5, n)

def swap_improve(h, n, dx, current_time, start_time, budget_s):
    """Try to improve the binary h by swapping 0 and 1"""
    def compute_corr(h):
        res = 0
        max_val = 0
        for i in range(1, n):
            corr = np.sum(h[:-i] * (1 - h[i:]))
            max_val = max(max_val, corr * dx)
        return max_val

    orig_val = compute_corr(h)
    idx1s = np.where(h == 1)[0]
    idx0s = np.where(h == 0)[0]
    
    # Try up to 100 swaps
    for _ in range(100):
        if (time.time() - start_time) > budget_s * 0.95:
            break
        if len(idx0s) == 0 or len(idx1s) == 0:
            break
        i = np.random.choice(idx1s)
        j = np.random.choice(idx0s)
        
        # make a temporary swap 
        h[i], h[j] = h[j], h[i]
        new_val = compute_corr(h)
        if new_val >= orig_val:
            h[i], h[j] = h[j], h[i]
        else:
            orig_val = new_val
            idx1s = np.where(h == 1)[0]
            idx0s = np.where(h == 0)[0]
    return h
