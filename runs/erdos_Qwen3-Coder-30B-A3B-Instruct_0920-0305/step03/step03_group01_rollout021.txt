import numpy as np
from scipy.signal import correlate
import time
from math import sqrt

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_onto_box_and_hyperplane(y, target):
    """Project y onto box [0,1] and hyperplane sum = target using binary search"""
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo+hi)/2
        x = np.clip(y - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo+hi)/2, 0, 1)

def compute_gradient(h, shift):
    """Compute gradient for correlation at given shift"""
    n = len(h)
    if shift >= 0:
        term1 = np.empty(n)
        term1[:n-shift] = 1 - h[shift:]
        term1[n-shift:] = 1
        term2 = np.empty(n)
        term2[shift:] = h[:n-shift]
        term2[:shift] = 0
    else:
        s = -shift
        term1 = np.empty(n)
        term1[s:] = 1 - h[:n-s]
        term1[:s] = 1
        term2 = np.empty(n)
        term2[:n-s] = h[s:]
        term2[n-s:] = 0
    return term1 - term2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 256  # Use more points to better capture shape
    dx = 2.0 / n_points
    
    # Initialize with deterministic shape
    # Start with symmetric tent-like shape
    h = np.zeros(n_points)
    center = (n_points - 1) / 2.0
    for i in range(n_points):
        dist = abs(i - center) / center
        h[i] = max(0, 1 - dist)
    
    # Scale to satisfy integral constraint
    scale = 1.0 / sum(h) * (n_points / 2)
    h = h * scale
    
    # Ensure all values are in [0, 1]
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Gradient descent parameters
    base_step = 0.05
    max_iter = 10000
    
    iter_cnt = 0
    stagnation_count = 0
    last_improvement = 0
    
    while (time.time() - start_time) < budget_s * 0.95 and iter_cnt < max_iter:
        # Compute correlation for all shifts
        corr = correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        max_ind = np.where(corr == max_corr)[0]
        shift = max_ind[0] - (n_points-1)
        
        # Compute gradient
        grad = compute_gradient(h, shift)
        
        # Update with step size
        step = base_step / sqrt(iter_cnt + 1) if iter_cnt > 0 else base_step
        h_new = h - step * grad
        
        # Project onto feasible set
        h_new = project_onto_box_and_hyperplane(h_new, n_points / 2)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new)
        
        # Keep improvement
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            last_improvement = iter_cnt
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Occasionally add noise to escape local minima
        if stagnation_count > 1000 and (time.time() - start_time) < budget_s * 0.9:
            # Add small random noise
            noise = np.random.normal(0, 0.01, n_points)
            h_new = h_new + noise
            # Project again
            h_new = project_onto_box_and_hyperplane(h_new, n_points / 2)
            # Check if the noisy version is better
            c5_noisy = evaluate_c5(h_new)
            if c5_noisy < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_noisy
                last_improvement = iter_cnt
                stagnation_count = 0
        
        h = h_new
        iter_cnt += 1
        
        # Break early if we've been stagnating for a long time
        if iter_cnt - last_improvement > 5000:
            break
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
