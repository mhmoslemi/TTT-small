```python
import numpy as np
from scipy.optimize import minimize
import time

# Constants
n_points = 50
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def smooth_max(values, alpha=20.0):
    """Smooth approximation of max using log-sum-exp."""
    max_val = np.max(values)
    # Subtract to improve numerical stability
    exps = np.exp(alpha * (values - max_val))
    return max_val + np.log(np.sum(exps)) / alpha

def smooth_objective(h_values):
    """Smooth approximation of the C_5 objective."""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Return smooth approximation of maximum correlation
    return smooth_max(correlation)

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def initialize_h():
    """Initialize with a heuristic that is known to work well."""
    # Alternating pattern with slight adjustments to allow fractional values
    h = np.zeros(n_points)
    # Insert a modified tent function
    center = n_points // 2
    for i in range(n_points):
        dist = abs(i - center)
        h[i] = max(0, 1.0 - dist / (n_points / 2.0))
    # Scale to ensure integral is 1
    h = h * (1.0 / np.sum(h) * (n_points / 2.0))
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600):
    """Main optimization routine."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with a heuristic initialization
    h_current = initialize_h()
    c5_current = evaluate_h(h_current)
    
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # Local optimization with multiple restarts
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    max_iter = 500
    restarts = 10
    
    for restart in range(restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random restart
        h_start = np.random.rand(n_points)
        h_start = h_start / np.sum(h_start) * (n_points / 2.0)
        h_start = np.clip(h_start, 0, 1)
        
        try:
            # Smooth local optimization
            result = minimize(smooth_objective, h_start, method='SLSQP', 
                             bounds=bounds, constraints=constraints,
                             options={'maxiter': max_iter, 'ftol': 1e-10, 'gtol': 1e-8})
            if result.success:
                h_test = result.x
                c5_test = evaluate_h(h_test)
                if c5_test < best_c5:
                    best_h = h_test.copy()
                    best_c5 = c5_test
        except:
            continue
            
        if best_c5 < target_bound:
            break
    
    # Final local refinement with original objective
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        result = minimize(evaluate_h, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-10})
        if result.success:
            best_h = result.x
    except:
        pass
    
    # Final evaluation
    final_c5 = evaluate_h(best_h)
    
    return list(best_h), final_c5, n_points
```
