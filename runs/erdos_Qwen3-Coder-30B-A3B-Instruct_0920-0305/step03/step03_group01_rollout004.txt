import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def phase_A_low_dim_global_search(seed, budget_s, n_points):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    m = 10  # number of blocks
    n_per_block = n_points // m
    if n_per_block * m != n_points:
        n_per_block = n_points // m
        m = n_points // n_per_block

    def objective_block(u):
        # u: vector of m block heights, last computed from others to satisfy mass constraint
        if len(u) != m - 1:
            return float('inf')
        u_m = max(0, min(1, m/2 - np.sum(u)))
        if u_m < 0 or u_m > 1:
            return float('inf')
        u_full = np.append(u, u_m)
        h_full = np.repeat(u_full, n_per_block)
        # pad or truncate if needed
        if len(h_full) > n_points:
            h_full = h_full[:n_points]
        elif len(h_full) < n_points:
            h_full = np.pad(h_full, (0, n_points - len(h_full)), 'constant', constant_values=0)
        return evaluate_c5(h_full)

    bounds = [(0, 1) for _ in range(m-1)]
    # Use a global optimization algorithm - CMA-ES-like
    # Simulate with differential evolution for simplicity
    try:
        de_res = differential_evolution(
            objective_block, 
            bounds, 
            seed=seed, 
            maxiter=1500, 
            popsize=4 + m,
            mutation=(0.7, 1),
            recombination=0.9,
            strategy='best1bin'
        )
        if de_res.success:
            u_best = de_res.x
            u_m = max(0, min(1, m/2 - np.sum(u_best)))
            u_full = np.append(u_best, u_m)
            h_best = np.repeat(u_full, n_per_block)
            if len(h_best) > n_points:
                h_best = h_best[:n_points]
            elif len(h_best) < n_points:
                h_best = np.pad(h_best, (0, n_points - len(h_best)), 'constant', constant_values=0)
            return h_best
    except:
        pass
        
    # fallback to initial
    return np.array(initial_h_values)

def phase_B_projected_gradient_descent(h_init, n_points, budget_s):
    start_time = time.time()
    
    h = h_init.copy()
    dx = 2.0 / n_points
    
    def compute_overlap_grad(h, k):
        # compute gradient of overlap for shift k
        grad = np.zeros(n_points)
        for i in range(n_points):
            # i+k out of range? - treat as 0
            if i+k >= 0 and i+k < n_points:
                grad[i] = 1 - h[i+k]
            else:
                grad[i] = 1.0
            # i-k out of range? - treat as 0
            if i-k >= 0 and i-k < n_points:
                grad[i] -= h[i-k]
            else:
                grad[i] -= 0.0
        return grad

    def project_to_feasible(h):
        # Project onto box and equality
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h > 0 else 1.0
        h = h * scale
        h = np.clip(h, 0, 1)
        return h

    alpha = 0.01
    max_iter = 5000
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for t in range(max_iter):
        if time.time() - start_time > budget_s * 0.6:
            break
            
        # find worst shift
        corr = correlate(h, 1-h, mode="full")
        k_corr = np.argmax(corr)
        # The shift corresponds to k_corr - (n_points - 1)
        shift = k_corr - (n_points - 1)
        grad_h = compute_overlap_grad(h, shift)
        
        # gradient step
        h_new = h - alpha / np.sqrt(t+1) * grad_h
        h_new = project_to_feasible(h_new)
        
        # update h and evaluate
        h = h_new  
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        
    return best_h

def phase_C_binary_rounding_and_local_search(h_full, n_points, budget_s):
    start_time = time.time()
    
    def compute_overlaps(h):
        # return list of overlaps for all shifts
        corr = correlate(h, 1-h, mode="full")
        return corr
    
    # Round to binary
    K_round = 30  # number of binary roundings to try
    best_h_binary = None
    best_c5 = float('inf')
    
    for _ in range(K_round):
        if time.time() - start_time > budget_s * 0.8:
            break
        h_rand = np.random.rand(n_points)
        # Normalize with mass constraint
        h_norm = h_rand * (n_points/2) / np.sum(h_rand)
        # Round probabilistically
        h_binary = np.zeros(n_points)
        # select top half
        idxs = np.argsort(h_norm)[::-1]
        h_binary[idxs[:n_points//2]] = 1
        c5 = evaluate_c5(h_binary)
        if c5 < best_c5:
            best_h_binary = h_binary.copy()
            best_c5 = c5
    
    if best_h_binary is None:
        # fallback to optimizing from a simple transformation
        best_h_binary = np.zeros(n_points)
        probs = h_full / np.sum(h_full) if np.sum(h_full) > 0 else np.ones(n_points)  
        idxs = np.argsort(probs)[::-1]
        best_h_binary[idxs[:n_points//2]] = 1
        best_c5 = evaluate_c5(best_h_binary)
    
    # Local search
    def update_overlap_delta(h, k, i, j):
        # update overlap for shift k due to a swap between i and j
        # h[i] goes from 1 to 0 and h[j] from 0 to 1
        delta = 0
        # calculate the change in shift k
        for idx in [i, j]:
            old_contrib = 1 - h[idx+k] if idx+k >= 0 and idx+k < n_points else 0
            new_contrib = 1 - h[idx+k] if idx+k >= 0 and idx+k < n_points and h[idx] == 0 else 1 - h[idx+k] if idx+k >= 0 and idx+k < n_points and h[idx] == 1 else 0
            if idx + k >= 0 and idx + k < n_points:
                delta += new_contrib - old_contrib
            else:
                delta += 0  # no contribution
        return delta if k >= 0 and k < n_points else 0
    
    def compute_c5_from_overlaps(overlaps):
        return np.max(overlaps) * 0.01  # dx = 0.01
    
    def compute_overlaps_from_binary(h, n):
        # precompute overlaps for easy update
        overlaps = np.zeros(2*n - 1)
        for k in range(2*n - 1):
            corr_value = 0
            shift_idx = k - (n-1)
            for i in range(n):
                j = i + shift_idx
                if 0 <= j < n:
                    corr_value += h[i] * (1 - h[j])  # h[i] * (1 - h[i+k])
            overlaps[k] = corr_value
        return overlaps
    
    # Switch to full overlap computation for binary
    overlaps = compute_overlaps_from_binary(best_h_binary, n_points)
    
    # single local loop with swaps
    max_swap_rounds = 10
    for round_idx in range(max_swap_rounds):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        min_delta = 0
        winner_pair = None
        # check a random sample of pairs
        candidates = []
        for i in range(n_points):
            if best_h_binary[i] == 1:
                for j in range(n_points):
                    if best_h_binary[j] == 0:
                        candidates.append((i,j))
        # sample a few pairs
        if len(candidates) > 1000:
            candidates = np.random.choice(candidates, 1000, replace=False)
        for i, j in candidates:
            # compute overlap change if we swap
            new_overlaps = overlaps.copy()
            # adjustment for shift k = i-j
            shift = i - j
            for k in range(2*n_points - 1):
                idx = k - (n_points - 1)
                old_term_i = 0 if i + idx < 0 or i + idx >= n_points else best_h_binary[i] * (1 - best_h_binary[i + idx]) 
                new_term_i = 0 if i + idx < 0 or i + idx >= n_points else 0 * (1 - best_h_binary[i + idx])
                old_term_j = 0 if j + idx < 0 or j + idx >= n_points else best_h_binary[j] * (1 - best_h_binary[j + idx])
                new_term_j = 0 if j + idx < 0 or j + idx >= n_points else 1 * (1 - best_h_binary[j + idx])
                new_overlaps[k] += (new_term_i - old_term_i) + (new_term_j - old_term_j)
            new_c5 = np.max(new_overlaps) * 0.01
            delta = new_c5 - best_c5
            if delta < min_delta:
                min_delta = delta
                winner_pair = (i, j)
                
        if winner_pair is not None:
            i, j = winner_pair
            # perform swap
            best_h_binary[i] = 0
            best_h_binary[j] = 1
            overlaps = compute_overlaps_from_binary(best_h_binary, n_points)
            best_c5 = np.max(overlaps) * 0.01
        else:
            break
            
    return best_h_binary

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 200  # Use 200 points for better resolution
    
    # Phase A: Low-dimensional global optimization
    h = phase_A_low_dim_global_search(seed, budget_s, n_points)
    
    # Phase B: Projected gradient decent refinement  
    h = phase_B_projected_gradient_descent(h, n_points, budget_s)
    
    # Phase C: Binary rounding and local refinement
    h = phase_C_binary_rounding_and_local_search(h, n_points, budget_s)
    
    c5 = evaluate_c5(h)
    
    return (h.tolist(), c5, n_points)
