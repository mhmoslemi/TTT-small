```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

# Constants
n_points = 60
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def run(seed=42, budget_s=600):
    """Main optimization routine."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with a symmetric tent-like shape
    x = np.linspace(0, 2, n_points)
    h_current = np.maximum(0, 1 - np.abs(x - 1))
    h_current = h_current / np.sum(h_current) * (1/dx)  # Scale to integral 1
    # Ensure bounds
    h_current = np.clip(h_current, 0, 1)
    c5_current = evaluate_h(h_current)
    
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # Objective function for optimization (constraint handling)
    def objective(h):
        return evaluate_h(h)
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints for optimization
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    # Use differential evolution for global search
    de_options = {'maxiter': 500, 'popsize': 15, 'mutation': (0.5, 1), 'recombination': 0.7}
    
    while time.time() - start_time < budget_s * 0.95:
        # Differential evolution optimization
        try:
            result_de = differential_evolution(
                objective,
                bounds,
                constraints=constraints,
                seed=seed,
                maxiter=300,
                popsize=15,
                mutation=(0.5, 1),
                recombination=0.7,
                disp=False,
                atol=1e-6
            )
            if result_de.success:
                h_candidate = result_de.x
                c5_candidate = evaluate_h(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except Exception as e:
            pass
        
        # Local refinement with SLSQP
        try:
            result_slsqp = minimize(
                objective,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            if result_slsqp.success:
                h_local = result_slsqp.x
                c5_local = evaluate_h(h_local)
                if c5_local < best_c5:
                    best_h = h_local.copy()
                    best_c5 = c5_local
        except Exception as e:
            pass
        
        # Occasionally perform another differential evolution run with different seed
        if time.time() - start_time > 10 and (time.time() - start_time) % 50 < 10:
            try:
                result_de2 = differential_evolution(
                    objective,
                    bounds,
                    constraints=constraints,
                    seed=seed + int(time.time() - start_time),
                    maxiter=200,
                    popsize=10,
                    mutation=(0.7, 1),
                    recombination=0.8,
                    disp=False
                )
                if result_de2.success:
                    h_candidate2 = result_de2.x
                    c5_candidate2 = evaluate_h(h_candidate2)
                    if c5_candidate2 < best_c5:
                        best_h = h_candidate2.copy()
                        best_c5 = c5_candidate2
            except Exception as e:
                pass

        # Break early if we meet target
        if best_c5 < target_bound:
            break

    # Final refinement with more intensively local optimization
    try:
        result_final = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-8}
        )
        if result_final.success:
            best_h = result_final.x
            best_c5 = evaluate_h(best_h)
    except Exception as e:
        pass

    # Ensure constraints are satisfied
    current_sum = np.sum(best_h) * dx
    if abs(current_sum - 1.0) > 1e-6:
        best_h *= (1.0 / current_sum)

    return list(best_h), best_c5, n_points
```
