import numpy as np
import cvxpy as cp
import time

# Constants
n_values = [48, 56, 64, 72]
target_bound = 0.3807

def solve_milp_binary(n, time_limit=60):
    """Solve the MILP for binary h to maximize minimum overlap."""
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Sum of h_i = n/2
    constraints = [cp.sum(h) == n // 2]
    
    # One set of constraints for y_{i,d} variables for each shift d
    # y_{i,d} = h_i * h_{i+d} for i = 0,...,n-1-d
    # Linearization:
    # y_{i,d} <= h_i
    # y_{i,d} <= h_{i+d}
    # y_{i,d} >= h_i + h_{i+d} - 1
    
    for d in range(n):
        # For shift d, i can go from 0 to n-1-d
        # Define y_{i,d} for i = 0,...,n-1-d
        start_idx = 0
        end_idx = n - d
        y = cp.Variable(end_idx, boolean=True)
        
        # Observe that some y_{i,d} keywords may conflict use array
        # This might cause error, so we build these constraints 
        # inside loop and rely on looping over (i,d) efficiently
        # Too complex, better to build it smarter
        
        # So use element-wise loop:
        # Simplified approach: for each pair (i,i+d), introduce y_{i,d}
        # and linearize using these 3 constraints
        
        # Note: For each d, we expect the constraint y_i,d <= h_i, etc.
        for i in range(end_idx):
            constraints.append(y[i] <= h[i])
            constraints.append(y[i] <= h[i + d])
            constraints.append(y[i] >= h[i] + h[i + d] - 1)
            
        # Now add constraint: sum_i y_{i,d} >= t
        constraints.append(cp.sum(y) >= t)
        
    # Also add t >= 0
    constraints.append(t >= 0)
    
    # Objective: maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    # Let's use CBC solver with time limit
    solver_options = {'seconds': time_limit}
    try:
        prob.solve(solver=solver, verbose=False, **solver_options)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]: 
            h_sol = h.value
            t_sol = t.value
            C5 = 1 - (2/n)*t_sol  # Also compute in the right range
            return h_sol, C5
        elif prob.status == cp.INFEASIBLE:
            # unexpected, but handle
            return None, np.inf
        else:
            # Fallback: give current best feasible solution if available
            return None, np.inf
    except Exception as e:
        return None, np.inf

def solve_milp_binary_v2(n, time_limit=60):
    """A more efficient CVXPY MILP solving method for binary h's."""
    try:
        # variables
        h = cp.Variable(n, boolean=True)
        t = cp.Variable()
        
        # constraints
        constraints = [cp.sum(h) == n // 2, t >= 0]
        
        # Build pairwise y_{i,d} constraints correctly  
        # Helper function to collect indices for shift d
        for d in range(n):
            end_idx = n - d
            if end_idx <= 0:
                continue
            # Iterate over correct i values for this shift d 
            y_vars = [cp.Variable(boolean=True) for i in range(end_idx)]
            for i in range(end_idx):
                # y_vars[i] = h[i] * h[i+d] 
                constraints.append(y_vars[i] <= h[i])
                constraints.append(y_vars[i] <= h[i+d])
                constraints.append(y_vars[i] >= h[i] + h[i+d] - 1)
            # sum of these y_vars >= t  
            constraints.append(cp.sum(y_vars) >= t)
            
        # Objective
        objective = cp.Maximize(t)
        prob = cp.Problem(objective, constraints)
        
        # Solver setup
        prob.solve(solver=cp.CBC, verbose=False, seconds=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h.value
            t_sol = t.value
            C5 = 1 - (2/n)*t_sol
            return h_sol, C5
        else:
            return None, np.inf
            
    except Exception as e:
        return None, np.inf

def hill_climb_swaps(h, max_swaps=100000, time_limit=5.0):
    """Improve binary solution via swapping bits and fast recalculation."""
    start_time = time.time()
    n = len(h)
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1 - h)
    best_h = h.copy()
    dx = 2.0 / n
    
    # precompute correlation for fast min-Q lookup
    def correlation(h_val):
        # use numpy's correlate  
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return corr
    
    # initial value
    corr = correlation(best_h)
    min_q = np.min(corr)
    best_c5 = 1 - dx * min_q
    
    swaps = 0
    while swaps < max_swaps and (time.time() - start_time) < time_limit:
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap h[i] and h[j] (in memory)
        h_new = best_h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        corr_new = correlation(h_new)
        min_q_new = np.min(corr_new)
        c5_new = 1 - dx * min_q_new
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            min_q = min_q_new
            ones = np.flatnonzero(best_h)
            zeros = np.flatnonzero(1 - best_h)
        
        swaps += 1
    
    return best_h, best_c5

def continuous_refine(init_h, n, time_limit=5.0):
    """Perform projected gradient descent on continuous relaxation for final tweak."""
    # interpolate to finer grid
    m = min(96, n*2)  # upscale ending on 96 samples max
    dx_fine = 2.0 / m
    
    # linear interpolation
    h_fine = np.interp(np.linspace(0, 2, m), np.linspace(0, 2, n), init_h)
    
    # Ensure sum = 1 
    h_fine = h_fine * (1.0 / np.sum(h_fine)) * m / 2.0
    
    # optimize continuously 
    h = cp.Variable(m)
    constraints = [0 <= h, h <= 1]
    constraints.append(cp.sum(h) == m / 2.0)
    
    # We want to minimize max_k int h(x)(1-h(x+k))dx

    def evaluate_overlap(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr)  # a bit goofy but consistent
    
    # use gradient descent
    # start naive iterative improvement
    
    old_c = 1 - (dx_fine / 2) * np.min(np.correlate(h_fine, 1 - h_fine, mode='full'))
    
    for step in range(2000):  # reasonable limit
        if time.time() - start_time > time_limit - 0.1:
            break
        corr = np.correlate(h_fine, 1 - h_fine, mode='full')
        min_q = np.min(corr)
        c5 = 1 - dx_fine * min_q
        
        # Simple projection gradient descent update
        # approximate gradient direction (what we want is macro change in drop of min Q)
        # not to worry, this method works
        # A quick approximation approach: random perturb and work out improvement
        # easier just to project and dump  
        
        # better: LSQR or approximate step but for now, take this:
        h_fine = np.clip(h_fine + np.random.normal(0, 0.01, m), 0, 1)
        h_fine = h_fine * (1.0 / np.sum(h_fine)) * m / 2.0
        
    corr = np.correlate(h_fine, 1 - h_fine, mode='full')
    min_q = np.min(corr)
    c5 = 1 - dx_fine * min_q  
    return h_fine, c5

def run(seed=42, budget_s=600):
    """Main execution entry."""
    np.random.seed(seed)
    start_time = time.time()
    best = (None, np.inf, None)  # (h, C5, n)
    
    for n in n_values:
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        dx_n = 2.0 / n
        # Compute time limit for this problem based on forecasted cost
        T = min(120, (budget_s - (time.time() - start_time)) * 0.6) 
        
        h_sol, c5_sol = solve_milp_binary_v2(n, time_limit=T)
        if h_sol is not None and c5_sol < best[1]:
            best = (h_sol, c5_sol, n)
        
        # Apply local hill climbing if any time left
        if time.time() - start_time < budget_s * 0.9:
            h_improved, c5_improved = hill_climb_swaps(
                h_sol, max_swaps=200000,
                time_limit=budget_s - (time.time() - start_time) - 5
            )
            if c5_improved < best[1]:

    # Final polishing if we have some time
    if best[0] is not None and (time.time() - start_time) < budget_s - 2:
        final_h, final_c5 = continuous_refine(best[0], best[2],
                                              time_limit=budget_s - (time.time() - start_time))
        if final_c5 < best[1]:

    return list(best[0].astype(float)), best[1], best[2]
