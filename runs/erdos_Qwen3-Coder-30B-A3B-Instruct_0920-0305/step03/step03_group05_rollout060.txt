import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

# Constants
n_points = 48  # Even number to allow exact half ones
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def evaluate_h_binary(h_values):
    """Evaluate the C_5 bound for binary h values."""
    h = np.array(h_values, dtype=int)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def local_search(h_values, max_iter=200):
    """Refine solution using local optimization."""
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    def objective(h):
        return evaluate_h(h)
    
    try:
        result = minimize(objective, h_values, method='SLSQP', 
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-10})
        if result.success:
            return result.x
    except:
        pass
    return h_values

def random_binary_pattern(n):
    """Generate a random binary pattern with exactly n/2 ones."""
    h = np.zeros(n)
    ones_indices = np.random.choice(n, size=n//2, replace=False)
    h[ones_indices] = 1
    return h

def simulated_annealing_binary(h_initial, max_iter=1000000, initial_temp=0.1, cooling_rate=0.9999):
    """Run simulated annealing with binary constraints."""
    h_current = h_initial.copy()
    c5_current = evaluate_h_binary(h_current)
    
    best_h = h_current.copy()
    best_c5 = c5_current
    
    temp = initial_temp
    
    # Precompute indices for fast swap operations
    ones_indices = np.where(h_current == 1)[0]
    zeros_indices = np.where(h_current == 0)[0]
    
    for iteration in range(max_iter):
        # Randomly select one 1 and one 0
        i = np.random.choice(ones_indices)
        j = np.random.choice(zeros_indices)
        
        # Create new solution by swapping
        h_new = h_current.copy()
        h_new[i] = 0
        h_new[j] = 1
        
        c5_new = evaluate_h_binary(h_new)
        
        # Accept or reject the move
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            # Update indices
            ones_indices = np.where(h_current == 1)[0]
            zeros_indices = np.where(h_current == 0)[0]
            
            if c5_new < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_new
        else:
            # Accept with probability
            if np.random.random() < np.exp(-(c5_new - c5_current) / temp):
                h_current = h_new
                c5_current = c5_new
                # Update indices
                ones_indices = np.where(h_current == 1)[0]
                zeros_indices = np.where(h_current == 0)[0]
        
        # Cool down
        temp *= cooling_rate
        
        # Occasionally reset temperature to avoid local minima
        if iteration % 10000 == 0 and temp < 1e-5:
            temp = initial_temp
    
    return best_h, best_c5

def run(seed=42, budget_s=600):
    """Main optimization routine using binary pattern search."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial binary pattern
    best_h = random_binary_pattern(n_points)
    best_c5 = evaluate_h_binary(best_h)
    
    # Set up time allocation
    remaining_time = budget_s * 0.95
    
    # Run multiple rounds of optimized search
    while time.time() - start_time < budget_s * 0.95:
        # Run simulated annealing for current pattern
        h_new, c5_new = simulated_annealing_binary(
            best_h, 
            max_iter=int(500000 * remaining_time / budget_s),  # Scale based on remaining time
            initial_temp=0.1,
            cooling_rate=0.99999
        )
        
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            if best_c5 < target_bound:
                break
        
        # Also try crude local optimization as fallback
        # Use SLSQP to refine the solution if possible
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': constraint_eq}
            result = minimize(evaluate_h, best_h, method='SLSQP', 
                             bounds=bounds, constraints=constraints,
                             options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-8})
            if result.success:
                h_refine = result.x
                c5_refine = evaluate_h(h_refine)
                if c5_refine < best_c5:
                    best_h = h_refine
                    best_c5 = c5_refine
        except:
            pass
        
        # Redistribute remaining time
        elapsed = time.time() - start_time
        remaining_time = budget_s - elapsed
        
        if remaining_time < 1:
            break
    
    # Final constraint enforcement
    # Ensure the integral constraint is preserved
    current_sum = np.sum(best_h) * dx
    if abs(current_sum - 1.0) > 1e-6:
        best_h *= (1.0 / current_sum)
    
    # Final optimization step using local solver
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_eq}
        result = minimize(evaluate_h, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-12})
        if result.success:
            best_h = result.x
    except:
        pass
    
    # Final evaluation
    final_c5 = evaluate_h(best_h)
    
    return list(best_h), final_c5, n_points
