import numpy as np
import cvxpy as cp
import time

# Constants
n_candidates = [48, 56, 64, 72]
dx = 2.0 / 48  # default, will be recalculated per n
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def solve_milp_binary(n, time_limit=60):
    """Solve the binary MILP for the given n."""
    dx = 2.0 / n
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == n / 2]
    # Linearization using auxiliary binary variables y[i,d]
    y_vars = {}
    # Define y_{i,d} for all valid (i,d) pairs for d=0 to n-1
    for d in range(n):
        if d == 0:
            y_vars[d] = cp.Variable(n, boolean=True)
        else:
            y_vars[d] = cp.Variable(n - d, boolean=True)
    # Add constraints for all y_{i,d}
    for d in range(n):
        if d == 0:
            y = y_vars[d]
            for i in range(n):
                constraints.append(y[i] <= h[i])
                constraints.append(y[i] <= h[i])
                constraints.append(y[i] >= h[i] + h[i] - 1)
        else:
            y = y_vars[d]
            for i in range(n - d):
                constraints.append(y[i] <= h[i])
                constraints.append(y[i] <= h[i + d])
                constraints.append(y[i] >= h[i] + h[i + d] - 1)
    # Add constraints on sums of y to compute Q_d
    for d in range(n):
        if d == 0:
            y = y_vars[d]
            constraints.append(cp.sum(y) == t)
        else:
            y = y_vars[d]
            constraints.append(cp.sum(y) >= t)
    # Maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CBC, time_limit=time_limit, verbose=False)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            h_sol = h.value.astype(int)
            t_sol = t.value
            c5 = 1 - (2/n) * t_sol
            return h_sol, c5
        elif prob.status in ['infeasible', 'unbounded']:
            return None, np.inf
        else:
            # In case of other status, return best known solution
            h_sol = h.value.astype(int)
            t_sol = t.value if t.value is not None else 0
            c5 = 1 - (2/n) * t_sol
            return h_sol, c5
    except Exception:
        return None, np.inf

def hill_climb_swaps(h_start, max_swaps=200000, time_limit=60):
    """Hill-climbing with swap moves."""
    h = h_start.copy()
    # Index lists
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1 - h)
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_h(h)
    start_time = time.time()
    swaps_left = max_swaps
    while swaps_left > 0 and (time.time() - start_time < time_limit - 0.1):
        # Randomly pick a 1 and a 0 to swap
        i1 = np.random.randint(0, len(ones))
        i0 = np.random.randint(0, len(zeros))
        idx1 = ones[i1]
        idx0 = zeros[i0]
        # Apply swap
        h[idx1] = 0
        h[idx0] = 1
        # Update lists
        ones[i1] = idx0
        zeros[i0] = idx1
        # Evaluate new bound
        c5 = evaluate_h(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        else:
            # Revert swap
            h[idx1] = 1
            h[idx0] = 0
            ones[i1] = idx1
            zeros[i0] = idx0
        swaps_left -= 1
    return best_h, best_c5

def continuous_refine(h_start, n, time_limit=60):
    """Refine with projected gradient descent on continuous variables."""
    h = h_start.copy()
    dx = 2.0 / n
    # Interpolate to finer grid by linearizing to n*2 points
    h_fine = np.interp(np.linspace(0, 2, n*2), np.linspace(0, 2, n), h)
    # Now run a simple LO (Linear Optimization) on top of this
    # We want to minimize C5 = maximum of auto-correlations with (1-h)
    # Instead, we just project the gradient onto sum==n/2 and keep in [0,1]
    # Simple as just SLSQP-like iterative refinement
    h_refined = h_fine.copy()
    bounds = [(0, 1) for _ in range(len(h_refined))]
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}
    # define inner function to avoid lambda closures
    def objective(hx):
        return evaluate_h(hx)

    try:
        from scipy.optimize import minimize
        result = minimize(objective, h_refined, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': int(time_limit * 100) if time_limit > 1 else 1000,
                                   'ftol': 1e-15, 'gtol': 1e-12})
        if result.success:
            h_refined = result.x
    except:
        pass
    c5_fine = evaluate_h(h_refined)
    return h_refined, c5_fine
