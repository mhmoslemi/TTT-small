```python
import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time

# Constants
n_list = [48, 60, 72, 84]
dx_list = [2.0 / n for n in n_list]
target_bound = 0.3807
max_iter_binary = 30000  # This is per restart, not total
max_iter_smooth = 300
max_iter_slsqp = 500
n_restart_smooth = 8
n_restart_binary = 5
alpha = 30
lambda_penalty = 10000

# Helper functions
def dx(n):
    return 2.0 / n

def evaluate(h_values):
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx(len(h))
    return c5

def correlation_vec(h_values):
    h = np.array(h_values)
    return np.correlate(h, 1 - h, mode="full")

def smooth_max(c, alpha):
    m = np.max(c)
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

def penalty_sum(h, lambda_penalty):
    return lambda_penalty * (np.sum(h) - len(h)/2)**2

def proj_to_simplex(v, S=None):
    if S is None:
        S = len(v) / 2
    v = np.clip(v, 0, 1)
    v_sum = np.sum(v)
    if abs(v_sum - S) < 1e-12:
        return v
    v_diff = S - v_sum
    # Linearly scale the vector to have sum S
    if v_sum > 0:
        v = v + v_diff * (v / v_sum)
    # Clip again and adjust to sum exactly S.
    v = np.clip(v, 0, 1)
    v_sum = np.sum(v)
    v = v * S / v_sum
    return v

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def logit(p):
    return np.log(p / (1 - p))

def binary_initial(n):
    v = np.zeros(n)
    ones = np.random.permutation(n)
    v[ones[:n//2]] = 1
    return v

def swap_and_evaluate(h, i_one, i_zero):
    h_new = h.copy()
    h_new[i_one], h_new[i_zero] = h_new[i_zero], h_new[i_one]
    return evaluate(h_new)

def compute_cost_and_grad(y, h0, lambda_penalty, alpha):
    h = sigmoid(y)
    corr = correlation_vec(h)
    # soft-max
    c_soft = smooth_max(corr, alpha)
    # add penalty
    c = c_soft + penalty_sum(h, lambda_penalty)
    # compute gradients numerically
    eps = 1e-6
    grad = np.zeros_like(y)
    for i in range(len(y)):
        y_eps = y.copy()
        y_eps[i] += eps
        h_eps = sigmoid(y_eps)
        corr_eps = correlation_vec(h_eps)
        c_up = smooth_max(corr_eps, alpha) + penalty_sum(h_eps, lambda_penalty)
        grad[i] = (c_up - c) / eps
    return c, grad

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    best_n = None

    # Stage 2.1: interval family search
    for n in n_list:
        dx_val = dx(n)
        best_candidate = None
        best_c5_candidate = float('inf')
        interval_params = [(2, 0.2), (3, 0.1), (4, 0.1), (5, 0.05)]
        
        for m, g in interval_params:
            l = 1.0 / m
            total_width = m * l + (m - 1) * g
            if total_width <= 2:
                for offset in np.arange(0, g, dx_val/2):
                    h_temp = np.zeros(n)
                    start = int(offset / dx_val)
                    for i in range(m):
                        pos_start = start + i * (int(l/dx_val) + int(g/dx_val))
                        pos_end = pos_start + int(l/dx_val)
                        for j in range(pos_start, min(pos_end, n)):
                            h_temp[j] = 1
                    h_temp = proj_to_simplex(h_temp, S=n/2)
                    c5_temp = evaluate(h_temp)
                    if c5_temp < best_c5_candidate:
                        best_c5_candidate = c5_temp
                        best_candidate = h_temp[:]
        if best_c5_candidate < best_c5:
            best_c5 = best_c5_candidate
            best_h = best_candidate[:]
            best_n = n
        if time.time() - start_time > 0.15 * budget_s:
            break

    # Stage 2.2: binary hill climbing
    for n in n_list:
        if time.time() - start_time > 0.65 * budget_s:
            break
        dx_val = dx(n)
        best_h_binary = None
        best_c5_binary = float('inf')
        
        for r in range(n_restart_binary):
            h = binary_initial(n)
            initial = evaluate(h)
            if initial < best_c5_binary:
                best_c5_binary = initial
                best_h_binary = h[:]
            # binary hill climb
            max_stagnation = 5 * n
            stagnation_counter = 0
            for i in range(max_iter_binary):
                if time.time() - start_time > 0.65 * budget_s:
                    break
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i_one = np.random.choice(ones)
                i_zero = np.random.choice(zeros)
                h_new = h.copy()
                h_new[i_one], h_new[i_zero] = h_new[i_zero], h_new[i_one]
                c5_new = evaluate(h_new)
                if c5_new < best_c5_binary:
                    h = h_new
                    best_c5_binary = c5_new
                    best_h_binary = h[:]
                    stagnation_counter = 0
                else:
                    stagnation_counter += 1
                    if stagnation_counter > max_stagnation:
                        # Restart
                        h = binary_initial(n)
                        stagnation_counter = 0
                        if evaluate(h) < best_c5_binary:
                            best_c5_binary = evaluate(h)
                            best_h_binary = h[:]
                if best_c5_binary < target_bound:
                    break
            if best_c5_binary < target_bound:
                break
        if best_h_binary is not None:
            if best_c5_binary < best_c5:
                best_c5 = best_c5_binary
                best_h = best_h_binary[:]
                best_n = n

    # Stage 2.3: smooth max refinement
    for n in n_list:
        if time.time() - start_time > 0.9 * budget_s:
            break
        if best_h is None or n != best_n:
            continue
        h_in = best_h[:]
        dx_val = dx(n)
        best_h_smooth = None
        best_c5_smooth = float('inf')
        for r in range(n_restart_smooth):
            if time.time() - start_time > 0.9 * budget_s:
                break
            h_start = h_in + np.random.normal(0, 0.2, n)
            # Use L-BFGS-B with numerically computed gradient
            try:
                res = minimize(
                    lambda y: compute_cost_and_grad(y, h_in, lambda_penalty, alpha)[0],
                    logit(h_start),
                    method='L-BFGS-B',
                    jac=lambda y: compute_cost_and_grad(y, h_in, lambda_penalty, alpha)[1],
                    options={'maxiter': max_iter_smooth, 'ftol': 1e-15}
                )
                h_final = sigmoid(res.x)
                h_final = proj_to_simplex(h_final, S=n/2)
                c5_final = evaluate(h_final)
                if c5_final < best_c5_smooth:
                    best_c5_smooth = c5_final
                    best_h_smooth = h_final[:]
            except:
                pass
            if best_c5_smooth < target_bound or time.time() - start_time > 0.9 * budget_s:
                break
        if best_h_smooth is not None and best_c5_smooth < best_c5:
            best_c5 = best_c5_smooth
            best_h = best_h_smooth[:]
            best_n = n

    # Stage 2.4: Exact polishing
    if best_h is not None:
        bounds = [(0, 1) for _ in range(len(best_h))]
        constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) * dx(best_n) - 1.0}
        try:
            res = minimize(evaluate, best_h, method='SLSQP', bounds=bounds, constraints=constraints,
                           options={'ftol': 1e-12, 'maxiter': max_iter_slsqp})
            if res.success:
                best_h = res.x
                best_c5 = evaluate(best_h)
        except:
            pass

    return list(best_h), best_c5, best_n
```
