def proj_box_simplex(v):
    """Project v onto the intersection of [0,1]^n and the simplex {x: sum x_i = 1}"""
    v = np.asarray(v)
    n = len(v)
    # Sort v in descending order
    u = np.sort(v)[::-1]
    # Compute cumsum
    cumsum = np.cumsum(u)
    # Find the threshold
    rho = (cumsum - 1) / np.arange(1, n + 1)
    # Find the largest index where rho[i] <= u[i]
    idx = np.where(rho <= u)[0]
    if len(idx) == 0:
        # All entries are too large, just project onto [0,1] and normalize
        w = np.clip(v, 0, 1)
        return w / np.sum(w)
    else:
        i = idx[-1]
        tau = rho[i]
        w = np.clip(v - tau, 0, 1)
        return w / np.sum(w)

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def spectral_search(n, budget, t0, R=30):

            H_new = H.copy()
            H_new[1:] = M * np.exp(1j * np.angle(H[1:]))
            
            # Inverse FFT
            h_tilde = np.real(np.fft.ifft(H_new))
            
            # Project back to feasible set

            h = np.clip(h, 0, 1)

            # Check convergence
            c5 = evaluate_c5(h)

                best_h = h.copy()
                best_c5 = c5

            # Early exit if not improving much
            if t > 5 and abs(c5 - evaluate_c5(h_prev)) < 1e-8:
                break
            h_prev = h.copy()
            
        if best_h is None or evaluate_c5(h) < best_c5:
            best_h = h.copy()
            best_c5 = evaluate_c5(h)
    
    return best_h, best_c5

def binary_refinement(h):

    n = len(h)
    m = n / 2
    # Convert to binary
    b = (h > 0.5).astype(float)
    if np.sum(b) != m:
        # Adjust to have exactly m ones
        idx = np.argsort(-h)  # indices of sorted in descending order
        ones_idx = idx[:int(m)]
        zeros_idx = idx[int(m):]
        b = np.zeros(n)
        b[ones_idx] = 1

    # Local hill climb with swaps
    best_b = b.copy()
    best_c5 = evaluate_c5(best_b)
    max_swaps = 5 * n
    no_improvement_count = 0
    max_no_improvement = 2 * n
    
    attempts = 0
    while attempts < max_swaps and no_improvement_count < max_no_improvement and time.time() - t0 < budget * 0.95:
        # Pick a random 1 and 0
        ones = np.where(best_b == 1)[0]
        zeros = np.where(best_b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Try a swap
        new_b = best_b.copy()
        new_b[i] = 0
        new_b[j] = 1
        
        new_c5 = evaluate_c5(new_b)
        if new_c5 < best_c5:
            best_b = new_b
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        attempts += 1
        
    return best_b, best_c5
