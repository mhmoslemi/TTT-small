import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_binary_milp(n, time_limit):
    """Solve the binary MILP to find optimal pattern"""
    m = n // 2
    # Binary variables for the pattern
    b = cp.Variable(n, boolean=True)
    # Integer variable for minimum overlap
    t = cp.Variable(integer=True)
    
    # Constraints
    constraints = [cp.sum(b) == m]
    
    # Build overlap constraints for each shift
    overlap = []
    for s in range(n):
        # Sum over i where both b_i and b_{i+s} are 1 (within bounds)
        if s == 0:
            # For shift 0, sum of all b[i] = m, but we treat it specially
            overlap.append(cp.sum(b))
        else:
            # For shift s, sum over valid indices
            overlap_s = cp.sum([cp.minimum(b[i], b[i+s]) for i in range(n - s)])
            overlap.append(overlap_s)
    
    # Add constraint that each overlap >= t
    for i in range(n):
        constraints.append(overlap[i] >= t)
    
    # Maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CBC, timeLimit=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_VERTEX]:
            solution = b.value
            optimal_t = t.value
            return solution, optimal_t
    except:
        pass
    
    return None, None

def refine_with_gradient(h0, n_points, budget_s):
    """Use gradient-based refinement to improve continuous solution"""
    import scipy.optimize as sopt
    
    dx = 2.0 / n_points
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Start with solution from binary MILP but make it continuous
    h_start = h0.copy()
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start / (np.sum(h_start) * dx)  # Normalize
    
    try:
        res = sopt.minimize(objective, h_start, method='SLSQP', bounds=bounds, constraints=cons,
                            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12})
        if res.success:
            h_refined = res.x
            h_refined = h_refined / (np.sum(h_refined) * dx)  # Re-normalize
            return h_refined
    except:
        pass
    
    return h_start
