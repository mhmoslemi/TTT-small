def weight_normalise(h, n):
    """Normalize h to satisfy sum(h) = n/2"""
    scale = (n / 2) / np.sum(h)
    return h * scale

def max_defect(h):
    """Compute exact C5 = max_k int h(x)(1 - h(x+k)) dx"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def softmax_objective(h, alpha):
    """Compute soft max approximation and its gradient for C5"""
    n = len(h)
    dx = 2.0 / n
    h_extended = np.concatenate([h, [0] * (n-1)]) 
    inv_alpha = 1.0 / alpha if alpha > 0 else 1e10
    
    # Compute f_s = sum h[i] * (1 - h[i+s]) for all shifts s
    total_f = np.zeros(n-1)
    grad = np.zeros(n)
    
    for s in range(1, n):  # shift from 1 to n-1
        f_s = 0.0
        for i in range(n-s):
            f_s += h[i] * (1.0 - h[i+s])
        total_f[s-1] = f_s
    
    # Compute softmax
    exp_f = np.exp(alpha * total_f)
    exp_sum = np.sum(exp_f)
    if exp_sum == 0.0:
        exp_sum = 1e-15
    
    soft_max = inv_alpha * np.log(exp_sum)
    weights = exp_f / exp_sum
    
    # Gradient computation
    for s in range(1, n):
        for i in range(n - s):
            grad[i] += weights[s-1] * (1.0 - (i+s < n) * h[i+s])
            if i+s < n:
                grad[i+s] -= weights[s-1] * h[i]
    
    grad *= dx
    return soft_max, grad

def build_milp(n):
    """Build and solve a binary MILP to maximize minimum overlap"""
    # Variables
    h = cp.Variable(n, boolean=True)
    w = cp.Variable((n, n-1), nonneg=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h) == n // 2,  # Weight constraint
        t <= 0.5  # Initial lower bound
    ]
    
    # AND constraints for each shift s
    for s in range(1, n):
        for i in range(n - s):
            # w[i,s] <= h[i]
            constraints.append(w[i, s-1] <= h[i])
            # w[i,s] <= h[i+s]
            constraints.append(w[i, s-1] <= h[i+s])
            # w[i,s] >= h[i] + h[i+s] - 1
            constraints.append(w[i, s-1] >= h[i] + h[i+s] - 1)
    
    # Minimum overlap constraint
    for s in range(1, n):
        constraints.append(cp.sum(w[:, s-1]) >= t)
    
    # Objective: maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        # Solver settings to avoid eating the whole budget
        prob.solve(solver=cp.CBC, verbose=False, 
                   # maxTime=180,  # This might not be supported by all environments
                   # Exact arg: supports ratios like: {"seconds": 180, "ratioGap": 0.0}
                   )
        # If solution is found
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h.value
            return h_sol
        else:
            # Binary solution exists but solver didn't complete properly
            return h.value if h.value is not None else None
    except:
        return None

def refine_continuous(h_bin, n):
    """Refine the binary solution using gradient-based continuous optimization."""
    max_iter = 200
    dx = 2.0 / n
    h0 = h_bin.copy()
    
    # Smooth max surrogate optimization with increasing alpha
    alphas = [5, 15, 40]
    
    for alpha in alphas:
        def fun_and_grad(h):
            val, grad = softmax_objective(h, alpha)
            return val, grad
        
        # Wrap into minimize-friendly form
        def obj(h):
            val, _ = fun_and_grad(h)
            return val

        def grad(h):
            _, g = fun_and_grad(h)
            return g

        # Prepare bounds and constraints
        bounds = [(0, 1) for _ in range(n)]
        eq_con = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}
        
        # Optimize
        result = minimize(
            obj, h0, method='trust-constr', 
            jac=grad, bounds=bounds, constraints=eq_con,
            options={'maxiter': max_iter, 'gtol': 1e-6}
        )
        
        if result.success:
            h0 = result.x.copy()
            h0 = np.clip(h0, 0, 1)
            h0 = weight_normalise(h0, n)
        else:
            h0 = np.clip(h0, 0, 1)
            h0 = weight_normalise(h0, n)
        
    return h0

def anneal(h, n):
    """Apply swap-style local search to further improve the solution."""
    dx = 2.0 / n
    max_iter = 5000
    gamma = 0.2
    temp_factor = 0.99
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    
    idx_max = np.argmax(corr)
    if idx_max == 0:
        s_star = 0
    else:
        s_star = idx_max - n + 1
    
    # Select indices in the "bad" set
    bad_indices = []
    for i in range(n - s_star):
        if h[i] > 0 and h[i + s_star] < 0.5:
            bad_indices.append(i)
    
    for iteration in range(max_iter):
        if not bad_indices:
            break

        # Exchange small δ amounts
        i = np.random.choice(bad_indices)
        j = np.random.choice(range(n))
        if abs(h[j] - 1.0) < 1e-6 or h[j + s_star] > 0.95:  # skip difficult exchanges
            continue
            
        δ = gamma * min(h[i], 1 - h[j])
        if δ < 1e-6:
            continue
            
        # Apply changes
        h[i] -= δ
        h[j] += δ
        h[i + s_star] += δ
        h[j + s_star] -= δ
        
        # Normalize
        h = weight_normalise(h, n)
        
        # Re-evaluate
        new_corr = correlate(h, 1 - h, mode="full")
        new_max_corr = np.max(new_corr)
        
        if new_max_corr > max_corr:
            # Revert
            h[i] += δ
            h[j] -= δ
            h[i + s_star] -= δ
            h[j + s_star] += δ
            h = weight_normalise(h, n)
        else:
            # Update max defect and indices
            max_corr = new_max_corr
            if iteration % 200 == 0:
                gamma *= temp_factor
                if gamma < 1e-5:
                    break
    
    return h
