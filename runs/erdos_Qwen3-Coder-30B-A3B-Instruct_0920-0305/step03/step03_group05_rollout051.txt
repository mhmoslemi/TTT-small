```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

# Constants
n_points = 48  # Even number for binary feasibility
dx = 2.0 / n_points
target_bound = 0.379

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def evaluate_h_binary(h_values):
    """Evaluate the C_5 bound for binary h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def local_search(h_values, max_iter=200):
    """Refine solution using local optimization."""
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    def objective(h):
        return evaluate_h(h)
    
    try:
        result = minimize(objective, h_values, method='SLSQP', 
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-10})
        if result.success:
            return result.x
    except:
        pass
    return h_values

def generate_initial_binary():
    """Generate initial binary pattern with exactly n_points/2 ones."""
    h = np.zeros(n_points)
    ones_indices = np.random.choice(n_points, size=n_points//2, replace=False)
    h[ones_indices] = 1
    return h

def get_candidates(h):
    """Get lists of indices where h=1 and h=0."""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    return ones, zeros

def evaluate_swap(h, i, j):
    """Evaluate the effect of swapping h[i] and h[j]."""
    h_new = h.copy()
    h_new[i] = 1
    h_new[j] = 0
    
    # Compute new correlation and C5
    correlation = np.correlate(h_new, 1 - h_new, mode="full")
    max_correlation = np.max(correlation)
    new_c5 = max_correlation * dx
    
    return new_c5

def local_search_binary(h_values, max_iter=1000):
    """Improve a binary solution via local swaps."""
    h = h_values.copy().astype(int)
    current_c5 = evaluate_h_binary(h)
    best_h = h.copy()
    best_c5 = current_c5
    iter_count = 0
    
    ones_list, zeros_list = get_candidates(h)
    
    while iter_count < max_iter:
        # Generate candidate swaps
        swaps = []
        for _ in range(500):
            if len(ones_list) > 0 and len(zeros_list) > 0:
                i = np.random.choice(ones_list)
                j = np.random.choice(zeros_list)
                new_c5 = evaluate_swap(h, i, j)
                swaps.append((i, j, new_c5))
        
        # Sort swaps by improvement
        swaps.sort(key=lambda x: x[2])
        
        improved = False
        for i, j, new_c5 in swaps:
            if new_c5 < best_c5:
                h[i] = 0
                h[j] = 1
                best_c5 = new_c5
                best_h = h.copy()
                if new_c5 < target_bound:
                    return best_h, best_c5
                improved = True
                # Update candidates after swap
                ones_list = np.where(h == 1)[0]
                zeros_list = np.where(h == 0)[0]
                break
        
        if not improved:
            # No improvement, try a few random swaps anyway
            for _ in range(3):
                if len(ones_list) > 0 and len(zeros_list) > 0:
                    i = np.random.choice(ones_list)
                    j = np.random.choice(zeros_list)
                    new_c5 = evaluate_swap(h, i, j)
                    if new_c5 < best_c5:
                        h[i] = 0
                        h[j] = 1
                        best_c5 = new_c5
                        best_h = h.copy()
                        if new_c5 < target_bound:
                            return best_h, best_c5
                        ones_list = np.where(h == 1)[0]
                        zeros_list = np.where(h == 0)[0]
        
        iter_count += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600):
    """Main optimization routine for binary solutions."""
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = generate_initial_binary()
    best_c5 = evaluate_h_binary(best_h)
    remaining_time = budget_s * 0.95
    
    # Try local improvements with binary approaches
    for _ in range(50):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Local search on binary pattern
        h_local, c5_local = local_search_binary(best_h, max_iter=500)
        if c5_local < best_c5:
            best_h = h_local
            best_c5 = c5_local
        if best_c5 < target_bound:
            break
    
    # Final local optimization
    if time.time() - start_time < budget_s * 0.95:
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_eq}
        result = minimize(evaluate_h, best_h, method='SLSQP', bounds=bounds, constraints=constraints,
                         options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-12})
        if result.success:
            best_h = result.x
            best_c5 = evaluate_h(best_h)
    
    return list(best_h), best_c5, n_points
```
