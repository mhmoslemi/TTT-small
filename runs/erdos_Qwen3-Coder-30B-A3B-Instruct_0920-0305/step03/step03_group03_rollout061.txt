import numpy as np
import cvxpy as cp
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

        n_test = 60
        dx_test = 2.0 / n_test
        m = n_test // 2
        
        # Create binary variables for the problem
        b = cp.Variable(n_test, boolean=True)
        t = cp.Variable(integer=True)
        
        # Constraints
        constraints = [cp.sum(b) == m]
        
        # We need to compute correlations
        # Define y variables for each pair
        # For each shift s, compute overlap_s = sum_{i} b_i * b_{i+s} 
        # Using aux variables y[i,s] representing b_i * b_{i+s}
        overlaps = []
        # Pre-create 2D variable matrix for y_{i,s}
        Y = cp.Variable((n_test, n_test), boolean=True)
        
        # Set all constraints for Y variables
        for i in range(n_test):
            for j in range(n_test):
                # Y[i,j] should represent b_i * b_j
                constraints.extend([
                    Y[i,j] <= b[i],
                    Y[i,j] <= b[j],
                    Y[i,j] >= b[i] + b[j] - 1
                ])
        
        # Calculate overlap for each shift
        for s in range(n_test):
            # Calculate sum Y[i, i+s] for valid indices
            overlap_expr = cp.sum([Y[i, i+s] for i in range(n_test-s)])
            overlaps.append(overlap_expr)
        
        # For all shift overlaps, we have overlaps[i] >= t where i from 0 to n_test-1
        for overlap in overlaps:
            constraints.append(overlap >= t)
        
        # Objective: maximize t
        objective = cp.Maximize(t)
        prob = cp.Problem(objective, constraints)

        # Solve with time limit
        time_limit = min(500, budget_s * 0.7)
        prob.solve(solver=cp.CBC, timeLimit=time_limit)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            # Retrieve solution
            b_sol = b.value
            t_sol = t.value
            
            if b_sol is not None and t_sol is not None:
                # Convert to step function
                h_new = b_sol.astype(float)
                # Normalize (should be fine with constraint)
                h_new = h_new * (1.0 / np.sum(h_new) * n_test * dx_test)
