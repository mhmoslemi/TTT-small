import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def weight_normalise(h, n):
    """Rescale h to satisfy sum(h) = n/2"""
    dx = 2.0 / n
    scale = (n / 2) / np.sum(h)
    return h * scale

def max_defect(h):
    """Compute exact C5 = max_k int h(x)(1-h(x+k)) dx"""
    corr = correlate(h, 1 - h, mode="full")
    max_val = np.max(corr)
    return max_val, np.argmax(corr) - (len(h) - 1)

def softmax_objective(h, alpha, n):
    """Soft max approximation of max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / n
    # Create correlation
    corr = correlate(h, 1 - h, mode="full")
    
    # Compute soft max
    exp_corr = np.exp(alpha * corr)
    Z = np.sum(exp_corr)
    soft_max_val = np.log(Z) / alpha
    
    # Compute gradient (approximately)
    # Use numerical differentiation for verification
    weights = exp_corr / Z
    grad = np.zeros_like(h)
    
    # compute gradient of soft max by back-propagation
    # ∂Φ/∂h_j = sum_s (w_s * (1 - h_{j+s} * 1_{j+s<n} - h_{j-s} * 1_{j-s>=0}))
    for i in range(len(h)):
        for s in range(1, len(h)):
            if i + s < len(h):
                grad[i] += weights[s] * (-h[i + s])
            if i - s >= 0:
                grad[i] += weights[s] * (-h[i - s])
        grad[i] += weights[0]  # The h[i] term from sum_s w_s * (1 - h[i+s])
    grad /= alpha
    return soft_max_val * dx, grad * dx

def build_milp(n):
    """Build binary MILP model to maximise minimum overlap"""
    # Variables
    h = cp.Variable(n, boolean=True)
    w = cp.Variable((n, n-1), nonneg=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h) == n/2,  # Weight constraint
    ]
    
    # AND constraints for each shift
    for s in range(1, n-1):
        for i in range(n-s):
            # w[i,s] <= h[i]
            constraints.append(w[i,s] <= h[i])
            # w[i,s] <= h[i+s]
            constraints.append(w[i,s] <= h[i+s])
            # w[i,s] >= h[i] + h[i+s] - 1
            constraints.append(w[i,s] >= h[i] + h[i+s] - 1)
    
    # Minimum overlap constraint: sum of w[i,s] >= t for all s
    for s in range(1, n-1):
        constraints.append(cp.sum(w[:, s]) >= t)
    
    # Objective: maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    solver_options = {"maxTime": 180, "seconds": 180, "ratioGap": 0.0}
    
    try:
        prob.solve(solver=CP.SCS, verbose=False, **solver_options)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value
        else:
            # Return best feasible solution encountered
            return [1 if x >= 0.5 else 0 for x in h.value]
    except:
        # In case of crash, return a simple pattern
        return [1 if i < n/2 else 0 for i in range(n)]

def refine_continuous(h_bin, n, alpha_schedule=[5, 15, 40], max_iter=200):
    """Refine binary solution using smooth max and trust-constr"""
    dx = 2.0 / n
    h0 = h_bin.copy().astype(float) 
    h0 = weight_normalise(h0, n)
    
    # Separate into fixed part (middle of blocks) and free part
    # For now, we'll just use trust-constr with gradient
    
    def obj_grad(h):
        val = 0
        grad = np.zeros_like(h)
        for _alpha in alpha_schedule:
            v, g = softmax_objective(h, _alpha, n)
            val += v
            grad += g
        return val, grad
    
    # Use trust-constr with analytic gradient
    bounds = [(0, 1) for _ in range(n)]
    
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}
    res = minimize(lambda x: obj_grad(x)[0], h0, method='trust-constr',
                   jac=lambda x: obj_grad(x)[1], bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'gtol': 1e-6, 'ftol': 1e-6})
    
    if res.success:
        h_refined = res.x
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = weight_normalise(h_refined, n)
        return h_refined
    else:
        h_refined = np.clip(h0, 0, 1)
        h_refined = weight_normalise(h_refined, n)
        return h_refined

def anneal(h, n, max_iter=5000, gamma_start=0.2, cooling_rate=0.99):
    """Simulated annealing style swapping procedure"""
    dx = 2.0 / n
    h = h.copy().astype(float)
    h = weight_normalise(h, n) 
    
    corr = correlate(h, 1-h, mode="full")  # precompute
    max_k = np.argmax(corr)
    max_def = corr[max_k]
    
    gamma = gamma_start
    for i in range(max_iter):
        if i > 0 and i % 100 == 0:
            gamma *= cooling_rate
            
        # find indices affected by max_k shift
        indices_for_shift = []
        for j in range(len(h)):
            if h[j] > 0 and (j + max_k < len(h)) and (h[j + max_k] < 0.5):
                indices_for_shift.append(j)
        
        if not indices_for_shift:
            continue
            
        i_idx = np.random.choice(indices_for_shift)
        j_idx = np.random.choice(np.arange(len(h)))
        
        # Check if this swap gains us anything
        h_new = h.copy()
        delta = min(h[i_idx], 1 - h[j_idx]) * gamma
        
        if delta > 1e-8:
            h_new[i_idx] -= delta
            h_new[j_idx] += delta
            
            if i_idx + max_k < len(h):
                h_new[i_idx + max_k] += delta
            if j_idx + max_k < len(h):
                h_new[j_idx + max_k] -= delta
            
            # Ensure validity
            h_new = np.clip(h_new, 0, 1)
            h_new = weight_normalise(h_new, n)
            
            new_corr = correlate(h_new, 1 - h_new, mode="full")
            new_max = np.max(new_corr)
            if new_max <= max_def:
                h = h_new
                max_def = new_max
                max_k = np.argmax(new_corr)
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n = 100
    elapsed = time.time() - start_time
    if elapsed > 500:
        n = 80
    
    # Stage 1: MILP to get a near-optimal binary pattern
    binary_sol = build_milp(n)
    h_bin = np.array([float(x) for x in binary_sol])
    
    elapsed = time.time() - start_time
    if elapsed > 180 or not h_bin.size:
        # Fallback: generate random matrices and optimize
        def try_random(n_restarts=30):
            best = np.ones(n) * 0.5
            best_c5 = np.inf
            for _ in range(n_restarts):
                h = np.random.rand(n)
                h = np.clip(h, 0, 1)
                h = weight_normalise(h, n)
                c5, _ = max_defect(h)
                if c5 < best_c5:
                    best = h
                    best_c5 = c5
            return best, best_c5
        h_initial, c5_initial = try_random()
        return (h_initial.tolist(), c5_initial, n)
    
    # Stage 2: Continuous refinement
    h_continuous = refine_continuous(h_bin, n)
    
    # Stage 3: Swap annealing
    h_final = anneal(h_continuous, n)
    
    # Final computation
    final_c5, _ = max_defect(h_final)
    
    return (h_final.tolist(), final_c5, n)
