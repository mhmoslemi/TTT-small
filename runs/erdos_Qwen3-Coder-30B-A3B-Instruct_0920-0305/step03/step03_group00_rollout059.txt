import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from cvxpy import *
import time
import cvxpy as cp

def exact_C5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 100  # Grid resolution
    dx = 2.0 / n
    max_rounding_trials = 30
    max_refinement_iterations = 200
    
    # Precompute the indices for all valid pairs
    valid_pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            if j - i < n:
                valid_pairs.append((i, j))
    
    # Initial check to see if the SDP is computationally feasible
    # If we cannot solve the SDP in less than a generous amount of time,
    # we fall back to the existing method.
    try:
        # Create the SDP problem
        h = Variable(n)
        X = Variable((n, n), symmetric=True)
        t = Variable()

        # Constraints
        constraints = []

        # Toeplitz-like structure for X to approximate h*h^T
        for i in range(n):
            constraints.append(X[i, i] == h[i])

        # McCormick envelopes
        for i in range(n):
            for j in range(i+1, n):
                if j - i < n:
                    constraints.append(X[i, j] <= h[i])
                    constraints.append(X[i, j] <= h[j])
                    constraints.append(X[i, j] >= h[i] + h[j] - 1)

        # Overlap constraints for all shifts s = 1 to n-1
        for s in range(1, n):
            o_s = 0
            for i in range(n - s):
                o_s += X[i, i + s]
            constraints.append(o_s >= t)

        # Full weight constraint
        constraints.append(sum(h) == n / 2)
        # Bound constraints
        for i in range(n):
            constraints.append(0 <= h[i])
            constraints.append(h[i] <= 1)
        # Positive semi-definiteness of X
        constraints.append(X >> 0)

        # Objective: maximize t
        objective = Maximize(t)
        prob = Problem(objective, constraints)

        # Solve with an early break once time reaches the limit
        elapsed = time.time() - start_time
        if elapsed > 100:
            raise RuntimeError("Early break: SDP setup already passed time limit")

        # Attempt to solve the problem
        prob.solve(solver=SCS, max_iters=1000, verbose=False, eps=1e-5)
        elapsed = time.time() - start_time
        if elapsed > 150:
            raise RuntimeError("SDP took too long")  # Fallback to numerical method if not fast enough

        # If we successfully solved, extract the solution
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h_opt = h.value
            t_opt = t.value
            X_opt = X.value
        else:
            # Fall back to a reasonable solution if SDP fails
            h_opt = np.ones(n) * 0.5  # Default initializer
            t_opt = 0
            X_opt = np.eye(n) * 0.5
    except Exception as e:
        # Fallback to pure random search
        print(f"Falling back to random method due to SDP error: {e}")
        return run_fallback(seed, budget_s)

    # Try various candidates and refine them
    candidates = []

    # Use the continuous solution directly
    h_relaxed = h_opt
    h_relaxed = np.clip(h_relaxed, 0, 1)
    scale = (n / 2) / np.sum(h_relaxed)
    h_relaxed = h_relaxed * scale
    candidates.append(h_relaxed)

    # Perform randomized rounding
    if np.all(np.abs(np.linalg.eigvals(X_opt)) > 1e-10):
        # Use eigen-decomposition
        try:
            lambdas, V = np.linalg.eigh(X_opt)
            r = min(20, len(lambdas))
            U = V[:, :r]
            lambdas = lambdas[:r]
            S = np.sqrt(np.maximum(lambdas, 0))
            Y = U @ np.diag(S)

            for _ in range(min(max_rounding_trials, 30)):
                if time.time() - start_time > budget_s * 0.7:
                    break
                g = np.random.randn(r)
                y = Y @ g
                h_bin = np.zeros(n)
                idxs = np.argsort(-y)[:n // 2]
                h_bin[idxs] = 1
                # Simple post-processing to ensure correct sum
                current_sum = np.sum(h_bin)
                diff = current_sum - (n // 2)
                if diff > 0:
                    # Remove some 1s
                    ones = np.where(h_bin == 1)[0]
                    drop = np.random.choice(ones, diff, replace=False)
                    h_bin[drop] = 0
                elif diff < 0:
                    # Add some 0s
                    zeros = np.where(h_bin == 0)[0]
                    add = np.random.choice(zeros, -diff, replace=False)
                    h_bin[add] = 1
                candidates.append(h_bin.copy())
        except Exception as e:
            pass

    # Local refinement using trust-constr
    def smooth_max(h, alpha):
        dx = 2.0 / n
        f_s = np.array([np.sum(h[i:] * (1 - h[:-i])) for i in range(1, n)])
        return np.log(np.sum(np.exp(alpha * f_s))) / alpha * dx

    def smooth_max_grad(h, alpha):
        dx = 2.0 / n
        f_s = np.array([np.sum(h[i:] * (1 - h[:-i])) for i in range(1, n)])
        weights = np.exp(alpha * f_s) / np.sum(np.exp(alpha * f_s))
        grad = np.zeros_like(h)
        for i in range(1, n):
            if i < n:
                grad[i] += weights[i-1]
            if i > 0:
                grad[-i] += weights[i-1]
        return grad * dx * alpha

    best_c5 = np.inf
    best_h = h_opt

    def smooth_objective(h):
        # Alpha schedule
        alpha_schedule = [5., 15., 40.]
        total_val = 0
        for a in alpha_schedule:
            total_val += smooth_max(h, a)
        return total_val

    def smooth_grad(h):
        alpha_schedule = [5., 15., 40.]
        total_grad = np.zeros_like(h)
        for a in alpha_schedule:
            total_grad += smooth_max_grad(h, a)
        return total_grad

    def refine_candidate(h0):
        # Define bounds
        bounds = [(0, 1) for _ in range(n)]
        # Linear constraint
        def constraint_eq(h):
            return np.sum(h) - (n / 2)
        cons = {'type': 'eq', 'fun': constraint_eq}

        # Approximate optimizer using trust-constr
        try:
            res = minimize(smooth_objective, h0, method='trust-constr',
                           bounds=bounds, constraints=cons,
                           jac=smooth_grad, options={'maxiter': max_refinement_iterations, 'gtol': 1e-8})
            if res.success:
                h_refined = res.x
                h_refined = np.clip(h_refined, 0, 1)
                scale = (n / 2) / np.sum(h_refined)
                h_refined = h_refined * scale
                return h_refined
        except:
            pass
        return h0

    # Refinement loop
    for h_candidate in candidates:
        if time.time() - start_time > budget_s:
            break
        h_refined = refine_candidate(h_candidate)
        try:
            c5 = exact_C5(h_refined)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_refined.copy()
        except:
            pass

    # Final fallback if anything didn't work
    if np.isnan(best_c5) or best_c5 > 1.0:
        best_h, best_c5 = run_fallback(seed, budget_s)

    return (best_h.tolist(), best_c5, n)

def run_fallback(seed, budget_s):
    np.random.seed(seed)
    n_points = 42
    h0 = np.array(initial_h_values)
    
    def objective(h):
        dx = 2.0 / n_points
        corr = correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def constraint_eq(h):
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try different optimization strategies
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Simple grid search strategy
    max_restarts = 100
    for i in range(max_restarts):
        h = best_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(h) * dx)
        h = h * scale_factor
        try:
            res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12})
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
    
    # Evaluate the final solution
    c5_final = exact_C5(best_h)
    return (best_h.tolist(), c5_final, n_points)
