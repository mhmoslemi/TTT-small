import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_milp_binary(n, time_limit=100):
    """Solve the binary MILP formulation for the Erdos constant"""
    m = n // 2
    # Binary variables for h_i
    b = cp.Variable(n, boolean=True)
    # Auxiliary variables for overlap computation
    # We'll define overlap_s for each shift s
    # Overlap for shift s = sum_{i=0}^{n-1-s} b_i * b_{i+s} 
    t = cp.Variable(integer=True)
    
    # Constraints
    constraints = [cp.sum(b) == m]
    
    # Create constraints for each shift s
    overlaps = []
    for s in range(n):
        # For each shift, we compute the sum of products b_i * b_{i+s}  
        # for valid indices i such that i+s < n
        overlap_s = cp.sum([b[i] * b[i+s] for i in range(n - s)])
        overlaps.append(overlap_s)
        constraints.append(overlap_s >= t)
    
    # Maximize t
    objective = cp.Maximize(t)
    
    prob = cp.Problem(objective, constraints)
    
    # Solve with CBC solver
    try:
        prob.solve(solver=cp.CBC, timeLimit=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_EOF]:
            b_sol = b.value
            t_sol = t.value
            return b_sol, t_sol, overlaps
        else:
            return None, None, None
    except:
        return None, None, None
