import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import Minimize, Problem
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_milp(n, time_limit):
    """Solve the MILP for binary h to maximize minimum overlap."""
    m = n // 2
    dx = 2.0 / n
    
    # Variables
    b = cp.Variable(n, boolean=True)  # binary variables
    y = cp.Variable((n, n), boolean=True)  # binary product variables
    t = cp.Variable()  # minimum overlap
    
    # Constraints
    constraints = [
        cp.sum(b) == m,
        t >= 0
    ]
    
    # ... Product linearization
    for i in range(n):
        for j in range(i, n):
            constraints += [
                y[i, j] <= b[i],
                y[i, j] <= b[j],
                y[i, j] >= b[i] + b[j] - 1
            ]
    
    # ... Overlap constraints for each shift
    for s in range(n):
        overlap = cp.sum([y[i, i+s] for i in range(n-s)])
        constraints += [overlap >= t]
    
    # Objective
    objective = Maximize(t)
    prob = Problem(objective, constraints)
    
    # Solve with time limit
    try:
        prob.solve(solver=cp.CBC, time_limit=time_limit)
        b_val = b.value
        t_val = t.value
        return b_val, (m - t_val) * dx
    except:
        # If we couldn't solve, return None
        return None, None

def projected_subgradient_descent(h0, max_iter=2000, step_size=1e-3):
    """Improve a given h using projected subgradient descent."""
    n = len(h0)
    h = h0.copy()
    dx = 2.0 / n
    
    def compute_max_shift():
        corr = correlate(h, 1 - h, mode="full")
        return np.argmax(corr)
    
    def gradient_at_shift(s):
        """Gradient of correlation at shift `s` w.r.t. h_i"""
        grad = np.zeros(n)
        for i in range(n):
            j = i + s
            if j < n:
                grad[i] += 1 - h[j]
        return grad

    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    last_c5 = best_c5

    for it in range(max_iter):
        if it % 100 == 0:
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

        # Compute the shift that currently gives max correlation
        s = compute_max_shift()
        grad = gradient_at_shift(s)
        
        # Descent step
        h_new = h - step_size * grad
        
        # Clamp to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project onto simplex
        s_sum = np.sum(h_new) * dx
        h_new = h_new * (1.0 / s_sum)
        
        h = h_new
        
        if abs(last_c5 - evaluate_c5(h)) < 1e-10:
            break
        last_c5 = evaluate_c5(h)

    if evaluate_c5(h) < best_c5:
        best_h = h

    return best_h, evaluate_c5(best_h)
