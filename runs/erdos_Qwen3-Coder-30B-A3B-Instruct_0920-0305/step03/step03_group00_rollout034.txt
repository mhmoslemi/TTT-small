import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import *

def weight_normalise(x, target_sum=1.0):
    """Normalize vector to have specified sum."""
    return x * (target_sum / np.sum(x))

def max_defect(h):
    """Compute exact C5 using correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx, np.argmax(corr) - (n - 1)

def softmax_objective(h, alpha, n_points):
    """Smooth max objective with gradient."""
    dx = 2.0 / n_points
    # Compute correlation vector for all shifts
    corr = correlate(h, 1 - h, mode='full')
    # Extract valid correlation values (non-zero)
    valid_corr = corr[n_points - 1:]  # Only positive shifts (1 to n-1)
    # Soft max
    exp_corr = np.exp(alpha * valid_corr)
    Z = np.sum(exp_corr)
    if Z == 0:
        return 0.0, np.zeros_like(h)
    
    weights = exp_corr / Z
    # Gradient with respect to h at index i
    grad = np.zeros_like(h)
    for s in range(1, n_points):  # Shifts from 1 to n-1
        if s < n_points:
            # Weight for this shift
            w = weights[s - 1]
            # Add contribution to gradient
            # d/dh[i] of f_s = h[i] * (1 - h[i+s]) if (i+s) < n
            # d/dh[i] of h[i] * (1 - h[i+s]) = 1 - h[i+s] 
            # d/dh[i+s] of h[i] * (1 - h[i+s]) = -h[i]
            for i in range(n - s):  # i such that i+s < n
                grad[i] += w * (1 - h[i+s])
                if i+s < n:
                    grad[i+s] -= w * h[i]
    return np.sum(valid_corr * weights) / alpha, grad

def build_milp(n):
    """Build MILP model to find binary solution that maximizes minimum overlap."""
    # Define variables
    h = Variable(n, boolean=True)
    w = Variable((n, n-1), nonneg=True)  # w[i,s] is h[i] AND h[i+s] for s = 1..n-1
    t = Variable()  # Minimum overlap

    # Constraints
    constraints = [
        sum(h) == n / 2,  # Weight constraint
        t <= 1,  # 't' is at most 1 (initial)
    ]

    # Linearize AND operations
    for s in range(1, n):
        for i in range(n - s):
            # w[i,s] <= h[i]  
            constraints.append(w[i,s-1] <= h[i])
            # w[i,s] <= h[i+s]  
            constraints.append(w[i,s-1] <= h[i+s])
            # w[i,s] >= h[i] + h[i+s] - 1
            constraints.append(w[i,s-1] >= h[i] + h[i+s] - 1)
    
    # Minimum overlap constraints
    for s in range(1, n):
        constraints.append(sum(w[:,s-1]) >= t)
    
    # Objective: maximize t
    objective = Maximize(t)
    problem = Problem(objective, constraints)
    
    # Solve with timeout
    try:
        problem.solve(solver=CVXOPT, verbose=False, max_iters=1000)
        # Return value
        return h.value
    except:
        return None

def refine_continuous(h0, n_points, n_iter=200):
    """Refine binary solution using smooth max."""
    dx = 2.0 / n_points
    
    # Initial guess: align to h0 but clip to [0,1] and normalize
    h = np.copy(h0)
    h = np.clip(h, 0, 1)
    h = weight_normalise(h, target_sum=1.0)
    
    # Initialize the gradient function using compute_gradient
    def fun_der(h):
        # This will be used in trust-constr or similar method
        val, grad = softmax_objective(h, alpha=5.0, n_points=n_points)
        return val, grad
    
    # Optimizer settings
    bounds = [(0, 1) for _ in range(n_points)]
    args = (n_points,)
    
    # Run with increasing alpha values
    alphas = [5.0, 15.0, 40.0]
    
    for alpha in alphas:
        # Use L-BFGS-B as a fallback, or trust-constr with custom objective
        def obj_and_grad(h_in):
            val, grad = softmax_objective(h_in, alpha, n_points)
            return val, grad
            
        res = minimize(
            fun=lambda x: softmax_objective(x, alpha, n_points)[0],
            x0=h,
            method='trust-constr',
            bounds=bounds,
            jac=lambda x: softmax_objective(x, alpha, n_points)[1],
            options={'maxiter': min(n_iter, 500), 'gtol': 1e-8, 'ftol': 1e-12},
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - 1.0}] if alpha == 40.0 else ()
        )
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            # Normalize exactly
            h = weight_normalise(h, target_sum=1.0)
        else:
            # Revert to current if not successful
            pass
    
    return h

def anneal(h, n_points, max_iter=5000, initial_gamma=0.2):
    """Simple swap-based annealing."""
    dx = 2.0 / n_points
    c = correlate(h, 1 - h, mode='full')
    max_corr = np.max(c)
    best_h = np.copy(h)
    best_c5 = max_corr * dx
    
    gamma = initial_gamma
    for i in range(max_iter):
        if i % 100 == 0:
            pass  # Optional progress update

        # Find the shift with maximum correlation
        idx = np.argmax(c)
        s_star = idx - (n_points - 1)
        # Indices where d[i] = h[i] * (1 - h[i+s_star]) is non-zero
        # i such that h[i] > 0 and h[i+s_star] < 1 (or out of bounds)
        indices = np.where(h > 0)[0]
        from_indices = indices[np.where(indices + s_star < n_points)][np.where(h[indices + s_star] < 1.0)[0]]
        to_indices = np.where(h < 1.0)[0]  # Not getting rid of the main RC value
        # Avoid indexing issues
        if len(from_indices) == 0 or len(to_indices) == 0:
            continue
        
        i_from = from_indices[np.random.randint(len(from_indices))]
        i_to = to_indices[np.random.randint(len(to_indices))]
        
        # Ensure valid positions
        if i_from + s_star >= n_points or i_to + s_star >= n_points:
            continue
            
        # Calculate change in h values
        delta = min(h[i_from], 1.0 - h[i_to]) * gamma
        if delta < 1e-10:  # Nothing to exchange
            continue
        
        # Store current values
        h_save_from = h[i_from]
        h_save_to = h[i_to]
        h_save_from_s = h[i_from + s_star]
        h_save_to_s = h[i_to + s_star]
        
        # Apply change
        h[i_from] -= delta
        h[i_to] += delta
        h[i_from + s_star] -= delta
        h[i_to + s_star] += delta

        # Keep within bounds all-around corrections to maintain exact constraint
        # Simply clip
        h = np.clip(h, 0, 1)
        # Ensure exact sum constraint
        h = weight_normalise(h, 1.0)
        
        # Recompute correlation
        new_c = correlate(h, 1 - h, mode='full')
        new_max_corr = np.max(new_c)
        new_c5 = new_max_corr * dx
        
        if new_c5 <= best_c5:
            best_c5 = new_c5
            best_h = np.copy(h)
        else:
            # Revert
            h[i_from] = h_save_from
            h[i_to] = h_save_to
            h[i_from + s_star] = h_save_from_s
            h[i_to + s_star] = h_save_to_s
            h = np.clip(h, 0, 1)
            h = weight_normalise(h, 1.0)
            
        gamma = max(0.001, gamma * 0.99)
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Choose resolution
    n_points = 100
    if budget_s < 300:
        n_points = 80
    dx = 2.0 / n_points
    
    # Phase 0: preliminary setup
    elapsed = time.time() - start_time
    if elapsed >= 0.5:
        return [], float('inf'), 0
    
    # Phase 1: Binary MILP
    print("Solving MILP...")

    h_binary = build_milp(n_points)
    if h_binary is None:
        print("MILP failed; Using random restarts.")
        best_h = None
        best_c5 = float('inf')
        for _ in range(30):
            h_test = np.random.uniform(0, 1, n_points)
            h_test = weight_normalise(h_test, 1.0)
            c5_test, _ = max_defect(h_test)
            if c5_test < best_c5:
                best_h = h_test
                best_c5 = c5_test
        return (best_h.tolist(), best_c5, n_points)
    
    h_binary = np.array(h_binary).astype(float)
    h_binary = weight_normalise(h_binary, 1.0)

    elapsed = time.time() - start_time
    if elapsed >= budget_s * 0.3:
        # Compute value directly, fallback if no line follows
        c5, _ = max_defect(h_binary)
        return (h_binary.tolist(), c5, n_points)
    
    # Phase 2: Continuous refinement using smooth max
    print("Refining solution...")
    h_refined = refine_continuous(h_binary, n_points, n_iter=min(200, int(250 - elapsed)))
    
    # Phase 3: Swapping optimization
    print("Annealing refinement...")
    final_h, final_c5 = anneal(h_refined, n_points, max_iter=2000, initial_gamma=0.1)
    
    # Final computation of exact C5
    c5_exact, _ = max_defect(final_h)

    return (final_h.tolist(), c5_exact, n_points)
