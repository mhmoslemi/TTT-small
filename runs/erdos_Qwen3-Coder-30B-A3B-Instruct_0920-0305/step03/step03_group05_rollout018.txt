import numpy as np
import cvxpy as cp
import time

def solve_milp_binary(n, time_limit=60):
    """Solve the binary MILP formulation for the Erdős problem."""
    # Variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Objective: maximize t
    objective = cp.Maximize(t)
    
    # Constraints
    constraints = [cp.sum(h) == n // 2, t >= 0]
    
    # Build auxiliary variables and constraints for each shift
    y_vars = {}
    q_vars = {}
    
    for d in range(n):
        # y_{i,d} for i = 0 to n-1-d
        start_idx = 0
        end_idx = n - d
        if end_idx > start_idx:
            y = cp.Variable(end_idx, boolean=True)
            y_vars[d] = y
            
            # Linearization constraints
            for i in range(start_idx, end_idx):
                constraints.append(y[i] <= h[i])
                constraints.append(y[i] <= h[i + d])
                constraints.append(y[i] >= h[i] + h[i + d] - 1)
            
            # Define Q_d
            q_var = cp.sum(y)
            q_vars[d] = q_var
            constraints.append(q_var >= t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CBC, verbose=False, 
                   solver_options={'seconds': time_limit})
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_sol = h.value
            t_sol = t.value
            c5 = 1 - (2/n) * t_sol
            return h_sol, c5
        else:
            # Fallback to best known solution if optimal not found
            h_sol = h.value
            if h_sol is not None:
                # Compute t based on current solution
                t_sol = 0
                min_q = float('inf')
                for d in range(n):
                    start_idx = 0
                    end_idx = n - d
                    if end_idx > start_idx:
                        q_d = np.sum(h_sol[start_idx:end_idx] * h_sol[start_idx+d:end_idx+d])
                        min_q = min(min_q, q_d)
                t_sol = min_q
                c5 = 1 - (2/n) * t_sol
                return h_sol, c5
            return None, float('inf')
    except Exception:
        return None, float('inf')


def hill_climb_swaps(h, max_swaps=200000, time_limit=10):
    """Perform swap-based hill climbing on a binary vector."""
    start = time.time()
    n = len(h)
    ones = set(np.flatnonzero(h))
    zeros = set(np.flatnonzero(1 - h))
    
    # Precompute full autocorrelation for reference
    def compute_bound():
        # This may be expensive but get rough Q at least
        _, auto_corr = np.correlate(h, h, mode='full')
        return 1 - (2.0/n) * np.min(auto_corr)
    
    best_h = h.copy()
    best_c5 = compute_bound()
    
    swaps = 0
    while swaps < max_swaps and (time.time() - start) < time_limit:
        # Pick random one and zero
        i = np.random.choice(list(ones))
        j = np.random.choice(list(zeros))
        # Swap
        new_h = h.copy()
        new_h[i] = 0
        new_h[j] = 1
        ones.remove(i)
        ones.add(j)
        zeros.add(i)
        zeros.remove(j)
        
        # Fast eval of bound
        _, auto_corr = np.correlate(new_h, new_h, mode='full')
        c5 = 1 - (2.0/n) * np.min(auto_corr)
        
        if c5 < best_c5:
            best_h = new_h
            best_c5 = c5
            h = new_h
            ones = set(np.flatnonzero(h))
            zeros = set(np.flatnonzero(1 - h))
        
        swaps += 1
        
    return best_h, best_c5


def continuous_refine(h_init, n, time_limit=2):
    """Refine a binary vector to continuous solution."""
    start = time.time()
    h = h_init.astype(float)
    dx = 2.0 / n
    
    # Do simple projected gradient on interval [0,1] and sum constraint
    def project_onto_sum(h, target_sum=1.0):
        current_sum = np.sum(h) * dx
        if abs(current_sum - target_sum) > 1e-10:
            h *= target_sum / current_sum
        return np.clip(h, 0, 1)
    
    def compute_bound(h):
        _, auto_corr = np.correlate(h, h, mode='full')
        return 1 - 0.5 * np.min(auto_corr)  # Adjust scaling
    
    # Take small gradient steps (approx., rough)
    for iter_ in range(1000):
        if time.time() - start > time_limit:
            break
        
        # Approximate gradient
        c5_current = compute_bound(h)
        # Perturb slightly randomly
        perturb = np.random.normal(0, 0.001, n)
        h_new = h + 0.005 * perturb
        h_new = project_onto_sum(h_new)
        c5_new = compute_bound(h_new)
        
        if c5_new < c5_current:
            h = h_new
    
    return h, compute_bound(h)


def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    n_values = [48, 56, 64, 72]
    used_time = 0
    
    for n in n_values:
        if used_time > 0.95*budget_s:
            break
            
        remaining_time = budget_s - used_time
        time_for_milp = min(120, 0.6 * remaining_time)
        
        h_sol, c5_sol = solve_milp_binary(n, time_limit=time_for_milp)
        
        if h_sol is not None and c5_sol < best_c5:
            best_h = h_sol
            best_c5 = c5_sol
            best_n = n
            used_time = time.time() - start_time
            
        if used_time > 0.95*budget_s:
            break
    
    # If we found something and have time for improvement
    if best_h is not None and (time.time() - start_time) < budget_s * 0.9:
        remaining = budget_s - (time.time() - start_time) - 2  # Safety
        if remaining > 0:
            # Try local search
            best_h, c5_improved = hill_climb_swaps(
                best_h, max_swaps=200000, time_limit=remaining)
            if c5_improved < best_c5:
                best_c5 = c5_improved
                
    return list(best_h.astype(float)), best_c5, best_n
