import numpy as np
import cvxpy as cp
import time

# Use a fixed number of points to ensure consistent behavior with
# problem constraints and execution time
n_points = 48
dx = 2.0 / n_points

def solve_milp_binary(n, time_limit=60):
    """Attempt to solve the MILP formulation for the binary h vector."""
    # Model variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()

    # Constraints: sum of h is n/2
    constraints = [cp.sum(h) == n // 2]

    # Add auxiliary binary variables and constraints for linearisation of h_i * h_{i+d}
    # We can create these dynamically. For each shift d from 0 to n-1,
    # we define y_{i,d} = h_i * h_{i+d} for i = 0 to n-1-d.
    # Linearization: y <= h_i, y <= h_{i+d}, y >= h_i + h_{i+d} - 1

    # Counter for number of auxiliary variables (for info)
    total_y = 0
    q_expressions = []

    for d in range(n):
        # This form has 0 <= i <= n-1-d (so we avoid index overflow)
        expr = cp.sum([cp.Variable(1, boolean=True) for _ in range(n - d)])
        # This is not straight-forward to manage in cvxpy directly due to variable reuse
        # So we proceed iteratively defining constraints across all i in valid range
        y_vars = []
        for i in range(n - d):
            y = cp.Variable(1, boolean=True)
            # The constraints y <= h[i], y <= h[i+d], y >= h[i] + h[i+d] - 1
            constraints.append(y <= h[i])
            constraints.append(y <= h[i + d])
            constraints.append(y >= h[i] + h[i + d] - 1)
            y_vars.append(y)
            total_y += 1

        # Accumulate Q_d = sum of y_{i,d} 
        q_d = cp.sum(y_vars)
        q_expressions.append(q_d)

    # Add that Q_d >= t for all d
    for q_d in q_expressions:
        constraints.append(q_d >= t)

    # Objective: maximize t
    objective = cp.Maximize(t)

    # Build problem
    prob = cp.Problem(objective, constraints)

    # Solve with CBCC
    try:
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            if h.value is not None and t.value is not None:
                h_value = np.array(h.value).astype(float)
                t_value = float(t.value)
                # Compute the actual bound
                c5_bound = 1 - (2/n) * t_value
                return h_value, c5_bound
            else:
                return None, None
        elif prob.status == cp.SOLVER_ERROR:
            # Fallback to a feasible solution if available
            if h.value is not None and t.value is not None:
                h_value = np.array(h.value).astype(float)
                t_value = float(t.value)
                c5_bound = 1 - (2/n) * t_value
                return h_value, c5_bound
            else:
                return None, None
        else:
            # Could not solve to optimality in time; but still return current best
            # Here we interpret that it's either solution truncated or infeasible
            # Since no intermediate values are accessible directly, we try a simple fallback approach
            return None, None
    except Exception:
        return None, None


def hill_climb_swaps(h_initial, max_swaps=100000, time_limit=10):
    """Use a greedy swap-based hill-climber to refine the binary h."""
    h = h_initial.copy()
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1 - h)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    for _ in range(max_swaps):
        if time.time() - start_time > time_limit:
            break
            
        # Pick a random one and a random zero
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Swap values
        h[i], h[j] = h[j], h[i]
        
        # Evaluate new C5
        new_c5 = compute_c5(h)
        
        # Accept only if it's better
        if new_c5 < best_c5:
            best_h[:] = h[:]
            best_c5 = new_c5
        else:
            # Revert
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_c5


def compute_c5(h_values):
    """Evaluate the C5 bound for a given h vector."""
    h = np.asarray(h_values)
    correlation = np.correlate(h, 1 - h, mode='full')
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5
