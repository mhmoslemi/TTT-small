def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h) 
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx = 2.0 / n
    dx = 2.0 / len(h)
    c5 = max_correlation * dx
    return c5

def solve_milp_binary(n, time_limit=60):
    """Solve MILP for binary h to maximize min overlap."""
    # Define variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Compute Q_d for each shift d (including negative by symmetry)
    # Sum_i y_{i,d} = Q_d where y_{i,d} = h_i * h_{i+d}
    # We'll build linear constraints for all valid d.
    constraints = [cp.sum(h) == n // 2]
    
    # Variable for each cross product term
    y_vars = {}
    
    for d in range(n):
        for i in range(n - d):
            y_vars[(i, d)] = cp.Variable(boolean=True)
            # Linearization constraints
            constraints.append(y_vars[(i, d)] <= h[i])
            constraints.append(y_vars[(i, d)] <= h[i + d])
            constraints.append(y_vars[(i, d)] >= h[i] + h[i + d] - 1)
        
        # Sum of y values for shift d
        Q_d = cp.sum([y_vars[(i, d)] for i in range(n - d)])
        # Enforce that this Q_d >= t
        constraints.append(Q_d >= t)
    
    # Objective: maximize t
    objective = cp.Maximize(t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Use CBC solver with time limit
    try:
        prob.solve(solver=cp.CBC, verbose=False, 
                   time_limit=time_limit,
                   seconds=time_limit)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            t_opt = t.value
            c5 = 1 - (2.0 / n) * t_opt
            return h_opt, c5
        elif prob.status in [cp.INFEASIBLE, cp.UNBOUNDED]:
            return None, None
        else:
            h_opt = h.value
            t_opt = t.value
            c5 = 1 - (2.0 / n) * t_opt
            return h_opt, c5
    except:
        return None, None

def hill_climb_swaps(h_init, max_swaps=100000, time_limit=10):
    """Improve h using swap-based hill-climbing."""
    start_time = time.time()
    h = h_init.copy()
    dx = 2.0 / len(h)
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1 - h)
    
    def evaluate_current():
        correlation = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(correlation)
        return max_corr

    current_bound = evaluate_current()
    best_h = h.copy()
    best_bound = current_bound
    
    swaps = 0
    while swaps < max_swaps and (time.time() - start_time) < time_limit:
        # Pick a random 1 and 0 to swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # Swap
        h[i], h[j] = h[j], h[i]
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        swaps += 1
        # Evaluate
        new_bound = evaluate_current()
        if new_bound < current_bound:
            current_bound = new_bound
            best_h = h.copy()
            best_bound = new_bound
        else:
            # Revert
            h[i], h[j] = h[j], h[i]
            ones = np.flatnonzero(h)
            zeros = np.flatnonzero(1 - h)
        
    return best_h, 1 - dx * best_bound

    zeros = np.flatnonzero(1 - h_new)
    
    def compute_corr(h_arr):
        corr = np.correlate(h_arr, 1 - h_arr, mode="full")
        return np.max(corr)
    
    # Projected gradient descent using SLSQP update steps
    for it in range(1000):
        if time.time() - start_time > time_limit:
            break
            
        # Compute gradient ~ by finite differencing
        current_max = compute_corr(h_new)
        # With a small epsilon
        epsilon = 1e-4
        grad = np.zeros(h_new.shape)
        for i in range(len(h_new)):
            h_hyp = h_new.copy()
            h_hyp[i] += epsilon
            h_hyp = clamp(h_hyp)
            corr_hyp = compute_corr(h_hyp)
            grad[i] = (corr_hyp - current_max) / epsilon

        h_new -= 0.01 * grad

        curr_sum = np.sum(h_new) * dx_new
        if abs(curr_sum - 1.0) > 1e-5:
            h_new *= (1.0 / curr_sum)
    
    # Final eval
    final_corr = compute_corr(h_new)
    final_bound = 1 - dx_new * final_corr
    
    return h_new, final_bound
