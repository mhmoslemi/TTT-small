import numpy as np
import cvxpy as cp
import time

# Constants
n_values = [48, 56, 64, 72]
target_bound = 0.3807
dx = 2.0 / 48  # Default dx for n=48

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def solve_milp_binary(n, time_limit=60):
    """Solve the max-min MILP for binary h to find t and compute C5 bound."""
    # Create variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()

    # Constraints
    constraints = [cp.sum(h) == n // 2, t >= 0]

    # Define y variables and linearization constraints
    for d in range(n):
        # For shift d, range of valid i values
        if d == 0:
            # When d=0, y_{i,0} = h_i * h_i = h_i
            # No extra variables needed, but we can include it for consistency
            pass
        else:
            # Only pairs where i+d < n
            range_i = n - d
            # Define auxiliary y_{i,d}
            y = cp.Variable(range_i, boolean=True)
            # These constraints ensure y[i,d] = h[i] * h[i+d]
            for i in range(range_i):
                constraints.append(y[i] <= h[i])
                constraints.append(y[i] <= h[i+d])
                constraints.append(y[i] >= h[i] + h[i+d] - 1)

            # Add the constraint Q_d >= t replaced by sum(y,:) >= t
            # The sum is the autocorrelation Q_d
            constraints.append(cp.sum(y) >= t)

    # Objective: Maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)

    # Solve
    prob.solve(solver=cp.CBC, verbose=False, time_limit=time_limit)
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_opt = h.value
        t_opt = t.value
        c5_bound = 1 - (2/n) * t_opt
        return h_opt, c5_bound
    else:
        # If infeasible or timed out, return current best if available
        h_current = h.value if h.value is not None else np.zeros(n)
        t_current = t.value if t.value is not None else 0
        c5_bound = 1 - (2/n) * t_current
        return h_current, c5_bound

def hill_climb_swaps(h, max_swaps=200_000, time_limit=60):
    """Hill-climbing with swaps to improve a binary solution."""
    start = time.time()
    h = np.array(h, dtype=float)
    
    ones = np.nonzero(h)[0]
    zeros = np.nonzero(1-h)[0]
    n = len(h)
    n2 = n // 2
    
    # Compute initial Q
    Q = np.correlate(h, h, mode='full')
    min_Q = np.min(Q)
    
    swaps = 0
    while swaps < max_swaps and time.time() - start < time_limit:
        # Shuffle to collect new (i,j) pairs
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        for i in ones[:min(10, len(ones))]:  # Keep it fast
            for j in zeros[:min(10, len(zeros))]:
                # Check if swap increases min_Q
                h_swap = h.copy()
                h_swap[i] = 1 - h_swap[i]
                h_swap[j] = 1 - h_swap[j]
                
                # Recompute Q for swapped h
                Q_swap = np.correlate(h_swap, h_swap, mode='full')
                min_Q_swap = np.min(Q_swap)
                
                if min_Q_swap > min_Q:
                    h = h_swap
                    min_Q = min_Q_swap
                    break
            else:
                continue
            break
        swaps += 1
    
    c5 = 1 - (2/n) * min_Q
    return h, c5

def run(seed=42, budget_s=600):
    """Main function to run optimization within given budget."""
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    for n in n_values:
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Time allocation
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
            
        # Use fraction of remaining time for this model
        T = min(120, 0.6 * remaining)
        
        h_opt, c5_opt = solve_milp_binary(n, time_limit=T)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n = n

        # Optional local search improvement for the present h
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        # Local improvement via hill-climbing
        h_improved, c5_improved = hill_climb_swaps(
            h_opt, max_swaps=100000, time_limit=remaining - 5
        )
        if c5_improved < best_c5:
            best_h = h_improved
            best_c5 = c5_improved
            best_n = n
            
    return list(best_h.astype(float)), best_c5, best_n
