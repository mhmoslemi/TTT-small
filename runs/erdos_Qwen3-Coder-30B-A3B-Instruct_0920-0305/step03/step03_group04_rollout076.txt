```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import expit
import time
import cvxpy as cp

# Constants
target_bound = 0.3807
n_list = [36, 42, 48, 54, 60]
max_iters_sdp = 8000
max_iters_hill = 50000
num_rounds = 30
max_time_per_phase = 600
log_alpha = 30
penalty_weight = 1e3


def evaluate_h(h):
    """Calculate the overlap integral C5."""
    h = np.array(h)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    return max_correlation * 2.0 / len(h)


def correlation_vec(h):
    h = np.array(h)
    correlation = np.correlate(h, 1 - h, mode="full")
    return correlation


def logsumexp(x, axis=None, keepdims=False):
    x_max = np.max(x, axis=axis, keepdims=True)
    return x_max + np.log(np.sum(np.exp(x - x_max), axis=axis, keepdims=keepdims))


def proj_to_simplex(v, s=0.5):
    """Project vector v onto the simplex {x : sum(x) = s, x >= 0}."""
    v = np.asarray(v)
    if np.sum(v) == s:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, len(u) + 1) > (cssv - s))
    theta = (cssv[rho] - s) / (rho + 1)
    return np.maximum(v - theta, 0)


def sigmoid(x):
    return expit(x)


def logit(x):
    eps = 1e-6
    x = np.clip(x, eps, 1 - eps)
    return np.log(x / (1 - x))


def solve_sdp(n):
    """Solve the SDP relaxation for the given n."""
    # Define variables
    x = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        0 <= x,
        x <= 1,
        cp.sum(x) == n / 2,
        cp.diag(X) == x,  # Diagonal entries of X must equal x
    ]

    # Linear constraints between X and x (McCormick envelopes)
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(X[i, j] >= 0)
            constraints.append(X[i, j] <= x[i])
            constraints.append(X[i, j] <= x[j])
            constraints.append(X[i, j] >= x[i] + x[j] - 1)

    # Overlap constraints
    for k in range(n):
        # Sum over i=0 to n-1-k
        sum_x = cp.sum(x[:n-k])
        sum_X = cp.sum(X[:n-k, k:])
        constraints.append(sum_x - sum_X <= t)

    # PSD constraint
    constraints.append(X >> 0)

    # Objective
    objective = cp.Minimize(t)

    # Build and solve the problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=max_iters_sdp, eps=1e-5)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_opt = x.value
            X_opt = X.value
            t_opt = t.value
        else:
            return None, None, None
    except Exception:
        return None, None, None

    return x_opt, X_opt, t_opt


def round_from_sdp(x_opt, X_opt, n):
    """Use random hyperplane rounding to produce a binary vector."""
    # Perform eigenvalue decompostion: X ≈ V V^T
    try:
        eigvals, eigvecs = np.linalg.eigh(X_opt)
        # Take top 10 eigenvectors (or all if fewer)
        k = min(10, len(eigvals))
        V = eigvecs[:, -k:] * np.sqrt(np.maximum(eigvals[-k:], 0))
    except Exception:
        # Fall back to 1D truncation if decomposition fails
        V = np.array([X_opt.flat[:n]]).T

    best_h = None
    best_c5 = np.inf

    for _ in range(num_rounds):
        # Generate a random Gaussian vector
        g = np.random.randn(V.shape[1])
        # Compute projection of each row of V onto g
        projected = V @ g
        # Threshold to get binary
        h_proposal = (projected >= 0).astype(float)

        # Adjust to exactly sum to n/2
        current_sum = np.sum(h_proposal)
        diff = int(current_sum - n / 2)
        if diff > 0:  # Too many ones
            # Flip the ones with smallest |projected| to zero
            indices_ones = np.where(h_proposal == 1)[0]
            flip_indices = np.argsort(np.abs(projected[indices_ones]))[:diff]
            for i in flip_indices:
                h_proposal[indices_ones[i]] = 0
        elif diff < 0:  # Need more ones
            # Find indices of zeros with smallest |projected|
            indices_zeros = np.where(h_proposal == 0)[0]
            flip_indices = np.argsort(np.abs(projected[indices_zeros]))[:abs(diff)]
            for i in flip_indices:
                h_proposal[indices_zeros[i]] = 1

        c5 = evaluate_h(h_proposal)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_proposal.copy()

    return best_h, best_c5


def hill_climb_binary(h_start, n):
    """Perform extensive swap-based hill climbing on binary vector."""
    # Initial state
    h = h_start.copy()
    C5_current = evaluate_h(h)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    best_h = h.copy()
    best_C5 = C5_current
    swaps = 0
    max_failed = 10000  # threshold to trigger fallback search
    failed_streak = 0

    while swaps < max_iters_hill:
        if failed_streak >= max_failed:
            # Fallback determination-based scan
            # Find worst shift k*
            c_vec = correlation_vec(h)
            k_star = np.argmax(c_vec)

            # Find mismatched ones at this shift
            mismatches = []
            for i in range(n - k_star):
                if h[i] == 1 and h[i + k_star] == 0:
                    mismatches.append(i)
            if len(mismatches) > 0:
                # Try swapping a mismatched one with a zero that would improve shift k*
                suggested_swaps = []
                for i in mismatches:
                    for j in zeros:
                        if i != j:
                            h_swap = h.copy()
                            h_swap[i], h_swap[j] = h_swap[j], h_swap[i]
                            c_new = correlation_vec(h_swap)
                            improved = c_new[k_star] < c_vec[k_star]
                            if improved:
                                suggested_swaps.append((i, j))
                if suggested_swaps:
                    i, j = suggested_swaps[0]  # Accept first improving one
                    h[i], h[j] = h[j], h[i]
                    ones[ones.index(i)] = j
                    zeros[zeros.index(j)] = i
                    C5_current = evaluate_h(h)
                    failed_streak = 0
                    if C5_current < best_C5:
                        best_C5 = C5_current
                        best_h = h.copy()
                    continue  # Don't count this as a swap at the end

            # No improvement possible: restart with new binary vector
            h = np.random.rand(n) < 0.5
            h = np.round(h)
            h = proj_to_simplex(h, n / 2)  # Project to exact constraint
            C5_current = evaluate_h(h)
            ones = np.where(h == 1)[0].tolist()
            zeros = np.where(h == 0)[0].tolist()
            failed_streak = 0
            swaps += 1
            continue

        # Regular swap
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        swaps += 1

        # Perform swap
        h[i], h[j] = h[j], h[i]
        ones[ones.index(i)] = j
        zeros[zeros.index(j)] = i

        new_C5 = evaluate_h(h)
        if new_C5 < C5_current:
            C5_current = new_C5
            failed_streak = 0
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h.copy()
        else:
            # Rollback swap
            h[i], h[j] = h[j], h[i]
            ones[ones.index(j)] = i
            zeros[zeros.index(i)] = j
            failed_streak += 1

    return best_h, best_C5


def smooth_optimize_h(y_init, h_init, n):
    """Optimize h using smooth surrogate of max correlation."""
    def smooth_obj(y):
        h = sigmoid(y)
        corrs = correlation_vec(h)
        soft_max = logsumexp(log_alpha * corrs) / log_alpha
        penalty = penalty_weight * (np.sum(h) - n / 2) ** 2
        return soft_max + penalty

    bounds = [(-20, 20)] * n
    res = minimize(smooth_obj, y_init, method='L-BFGSB', bounds=bounds, options={'maxiter': 300})

    y_opt = res.x
    h_opt = sigmoid(y_opt)
    h_projected = proj_to_simplex(h_opt, n / 2)
    C5_final = evaluate_h(h_projected)
    return h_projected, C5_final


def run(seed=42, budget_s=600):
    """Main function to run the optimization."""
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_C5 = np.inf
    best_n = None

    # Phase A + B: SDP solve + Rounding
    for n in n_list:
        if time.time() - t0 > budget_s * 0.9:
            break
        try:
            x_opt, X_opt, t_opt = solve_sdp(n)
            if x_opt is None:
                continue
        except Exception:
            continue

        # Keep the lowest lower bound found, even if not promising; it may still provide good path
        if t_opt < best_C5:
            best_C5 = t_opt

        # Rounding
        h_rounded, C5_rounded = round_from_sdp(x_opt, X_opt, n)
        if h_rounded is not None and C5_rounded < best_C5:
            best_h = h_rounded.copy()
            best_C5 = C5_rounded
            best_n = n

    # Phase C: Hill climbing from best rounded binary if exists or random start
    if best_h is not None:
        h_hill, C5_hill = hill_climb_binary(best_h, best_n)
        if C5_hill < best_C5:
            best_h = h_hill.copy()
            best_C5 = C5_hill
    else:
        # Fallback to random
        n = n_list[0]  # Assume n is valid
        h_rand = np.random.rand(n) < 0.5
        h_rand = proj_to_simplex(h_rand, n / 2)
        h_hill, C5_hill = hill_climb_binary(h_rand, n)
        if C5_hill < best_C5:
            best_h = h_hill.copy()
            best_C5 = C5_hill
            best_n = n

    # Phase D: Smooth optimization
    if time.time() - t0 > 0.9 * budget_s:
        pass  # Skip
    else:
        y_start = logit(best_h + 1e-6)
        h_smooth, C5_smooth = smooth_optimize_h(y_start, best_h, best_n)
        if C5_smooth < best_C5:
            best_h = h_smooth.copy()
            best_C5 = C5_smooth

    # Phase E: Final polishing with exact objective
    # For minimizing the exact max, use SLSQP with bounds and equality constraint
    bounds = [(0, 1)] * best_n
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - best_n / 2}

    def objective_exact(x):
        return evaluate_h(x)

    try:
        res = minimize(objective_exact, best_h, method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-10})
        if res.success:
            best_h = res.x
    except Exception:
        pass

    # Final evaluation
    final_C5 = evaluate_h(best_h)
    return list(best_h), final_C5, best_n
```
