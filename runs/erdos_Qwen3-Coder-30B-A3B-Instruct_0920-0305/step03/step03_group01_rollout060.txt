def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_onto_box_and_hyperplane(y, target):
    """Project onto [0,1] box and sum constraint using binary search"""
    # Simple binary search to find lambda
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0, 1)

def compute_gradient(h, shift):
    """Compute gradient for correlation at a specific shift."""
    n = len(h)
    grad = np.zeros_like(h)

    if shift >= 0:
        # comute term1: 1 - h[i+shift]
        term1 = np.empty(n)
        term1[:n-shift] = 1 - h[shift:]
        term1[n-shift:] = 1
        # compute term2: h[i-shift]  (if negative, bypass)
        term2 = np.empty(n)
        term2[shift:] = h[:n-shift]
        term2[:shift] = 0
    else:  # shift < 0
        s = -shift
        term1 = np.empty(n)
        term1[s:] = 1 - h[:n-s]
        term1[:s] = 1
        term2 = np.empty(n)
        term2[:n-s] = h[s:]
        term2[n-s:] = 0
    
    grad = term1 - term2
    return grad

    # Initialize with a decent starting set
    for i in range(max_iter):

        if time.time() - start_time > time_budget * 0.95:

        corr = correlate(h, 1 - h, mode="full")

        h = np.clip(h_proj, 0, 1)
        
        # Evaluate new C5
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
    return best_h, best_c5
