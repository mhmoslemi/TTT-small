# Constants
n_values = [36, 42, 48, 54, 60]
dx = 2.0 / 48  # Default to 48 if not specified in run
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

def correlation_vec(h_vec):
    """Get correlation vector for a given h vector (full correlation)."""
    h = np.array(h_vec)
    corr = np.correlate(h, 1 - h, mode="full")
    return corr

def sigmoid(x):
    """Numerically stable sigmoid function."""
    return 1.0 / (1.0 + np.exp(-x))

def logit(x):
    """Inverse sigmoid, with clipping."""
    x = np.clip(x, 1e-6, 1 - 1e-6)
    return np.log(x / (1.0 - x))

def proj_to_simplex(v, s=1.0):
    """Project vector v onto the simplex {x : sum(x) = s, x >= 0}."""
    v = np.array(v)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - s
    ind = np.where(u > css / np.arange(1, len(u) + 1))[0]
    if len(ind) == 0:
        theta = 0.0
    else:
        theta = css[ind[-1]] / (ind[-1] + 1)
    return np.maximum(v - theta, 0.0)

def solve_sdp(n):
    """Solve SDP relaxation for given n; return tuple of (t_min, X*, x*) or (None, None, None) if fails."""
    x = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # constraint
    constraints = [
        0 <= x, x <= 1,
        cp.sum(x) == n / 2,
        cp.diag(X) == x,  # diagonal
        X >> 0,           # positive semidefinite
        t >= 0            # simplify sum and so on
    ]

    # add McCormick constraints to make X[i,j] ≈ x[i]*x[j] when binary
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(X[i, j] >= 0)
            constraints.append(X[i, j] <= x[i])
            constraints.append(X[i, j] <= x[j])
            constraints.append(X[i, j] >= x[i] + x[j] - 1)

    # overlap constraints
    for k in range(n):
        if k == 0:
            # Total sum of h minus sum of h[i] * h[i+k] for k=0 -> end
            overlap = cp.sum(x) - cp.sum(cp.diag(X, k))
        else:
            # Start from i=0, go up to n-k to avoid indexing overflow
            overlap = cp.sum(x[:n-k]) - cp.sum(cp.diag(X, k))
        constraints.append(overlap <= t)

    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=SCS, max_iters=10000, eps=1e-5)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            return (t.value, X.value, x.value)
    except Exception as e:
        print("SDP solve failed for n=", n, "with error:", str(e))
    return (None, None, None)

def hyperplane_round(Xstar, n):
    """Use hyperplane rounding based on leading eigen-decomposition of Xstar."""
    # Eigen-decompose to get low-rank reconstruction
    try:
        # Compute eigenvalues of Xstar (real as it's psd)
        evals, evecs = np.linalg.eigh(Xstar)
        r = min(20, n)  # small rank reasonable to start
        # Only take positive eigenvalues
        mask = evals > 1e-6
        if not np.any(mask):
            # if all eigenvals near zero, just make a random binary
            h = np.random.rand(n) > 0.5
            h = h.astype(float)
            return h

        top_evals = evals[mask][-r:]   # largest r eigenvalues
        top_evecs = evecs[:, mask][:, -r:]  # corresponding vectors

        # recreate Low-rank factor
        V = np.dot(top_evecs, np.diag(np.sqrt(top_evals)))
        g = np.random.randn(r)
        # For each row of V, IEEE sign of dot product with g
        s = np.sign(np.dot(V, g))
        # map s to binary vector
        h = (s >= 0).astype(float)

        # ensure it sums to n/2 (adjust if needed)
        current_sum = np.sum(h)
        if abs(current_sum - n/2) > 0.1:
            # flip the needed bits to balance sum
            diff = int(np.round(current_sum - n/2))

            if diff > 0:
                # too many ones: flip to zero the indices with smallest abs(V[i].dot(g))
                to_flip = np.argsort(np.abs(np.dot(V, g)))[:diff]
                for i in to_flip:
                    h[i] = 0
            elif diff < 0:
                # too many zeros: flip to 1 the indices with smallest abs(V[i].dot(g))
                to_flip = np.argsort(np.abs(np.dot(V, g)))[:abs(diff)]
                for i in to_flip:
                    h[i] = 1

        return h
    except Exception:
        # fallback to random binary (roughly 50% ones)
        return np.random.rand(n) > 0.5

def hill_climb_binary(h_start, max_swaps=50000):
    """Combinatorial hill-climber: binary swap to reduce max correlation."""
    h = np.array(h_start).copy()
    nrows = len(h)
    C5_best = evaluate_h(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    time_start = time.time()

    successful_swaps = 0
    iterations = 0
    no_progress_iter = 0
    max_no_progress = 10000

    while iterations < max_swaps and (time.time() - time_start) < 200:
        if no_progress_iter > max_no_progress:
            # hardcoded improvement step: identify worst shift and attempt local fixes
            corr_full = correlation_vec(h)
            kstar = np.argmax(corr_full)
            if kstar <= nrows - 1:
                mismatches = np.where((h[:-kstar] == 1) & (h[kstar:] == 0))[0]
                if len(mismatches) > 0:
                    # evaluate impact of any swap with a zero from mismatches list
                    for m in mismatches[:min(5, len(mismatches))]:  # check first few
                        # Try to find a good zero to swap with
                        for j in zeros[:min(10, len(zeros))]: # search 10 best candidates
                            if j <= m:
                                continue  # avoid redundant swap (j further away)
                            h_new = h.copy()
                            h_new[m] = 0
                            h_new[j] = 1
                            C5_new = evaluate_h(h_new)
                            if C5_new < C5_best:
                                h = h_new
                                C5_best = C5_new
                                ones = np.where(h == 1)[0]
                                zeros = np.where(h == 0)[0]
                                successful_swaps += 1
                                no_progress_iter = 0
                                break  # Once one improved swap executed, restart greedy search

        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1

        C5_new = evaluate_h(h_new)
        if C5_new < C5_best:
            h = h_new.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            C5_best = C5_new
            successful_swaps += 1
            no_progress_iter = 0
        else:
            no_progress_iter += 1

        iterations += 1

    return h, C5_best

def smooth_max_refinement(h_init, alpha=30.0, max_iter=300):
    """Refine fractional function."""
    def obj_f(y):
        h = sigmoid(y)
        corr = correlation_vec(h)
        soft_max = logsumexp(alpha * corr) / alpha
        pen = 1e4 * (np.sum(h) - 0.5 * len(h))**2
        return soft_max + pen

    def grad_f(y):
        h = sigmoid(y)
        corr = correlation_vec(h)
        exp_term = np.exp(alpha * corr)
        sum_exp = np.sum(exp_term)
        grad_corr = np.correlate(1-h, np.ones_like(h), mode='full')
        grad_corr = grad_corr[:len(corr)]
        # Approximat the gradient of softmax term
        soft_grad = (1./alpha) * (exp_term / sum_exp) * grad_corr
        # Gradient of penalty term
        pen_grad = 2 * 1e4 * (np.sum(h) - 0.5 * len(h)) * h * (1-h)
        return soft_grad + pen_grad  # note: need more precise gradient computation

    y = logit(h_init + 1e-6)
    res = minimize(obj_f, y, method='L-BFGS-B', jac=grad_f, maxiter=max_iter)
    h_refined = sigmoid(res.x)
    h_final = proj_to_simplex(h_refined, len(h_init)/2)
    return h_final

def run(seed=42, budget_s=600):
    # Start timing
    start_time = time.time()

    best_h = None
    best_c5 = float('inf')
    best_n = None

    # Refine from a ton of n's
    for n in n_values:
        if time.time() - start_time > budget_s * 0.9:
            break

        # Solve SDP for this n
        t_min, Xstar, xstar = solve_sdp(n)
        if t_min is None:
            continue  # Skip unsolved

        print(f"SDP solved for n={n}: t_min={t_min:.5f}")

        best_rnd_h = None
        best_rnd_c5 = float('inf')

        for _ in range(30):
            h_rounded = hyperplane_round(Xstar, n)
            h_rounded = (h_rounded > 0.5).astype(float)  # convert back to binarized one
            # now ensure it sums to n/2 exactly
            current_sum = np.sum(h_rounded)
            required = n / 2
            adj = int(np.round(required - current_sum))
            if adj > 0:
                # too few ones: must turn z=0 into one
                zero_indices = np.where(h_rounded == 0)[0]
                offset = [i for i in range(n) if (h_rounded[i] == 0)]
                for i in offset[:adj]:
                    h_rounded[i] = 1
            elif adj < 0:
                ones = np.where(h_rounded == 1)[0]
                for i in ones[:abs(adj)]:
                    h_rounded[i] = 0
            else:
                pass  # already right

            # hill-climb
            h_climb, c5 = hill_climb_binary(h_rounded, max_swaps=20000)
            if c5 < best_rnd_c5:
                best_rnd_c5 = c5
                best_rnd_h = h_climb.copy()

        if best_rnd_c5 < best_c5:
            best_c5 = best_rnd_c5
            best_h = best_rnd_h
            best_n = n

        # Additional refinement if target bound is achievable
        if best_rnd_c5 < 0.4 and time.time() - start_time < budget_s - 5:
            h_opt = smooth_max_refinement(best_rnd_h, alpha=30.0, max_iter=100)
            c5_opt = evaluate_h(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n

        # Quick early exit if we hit the target
        if best_c5 <= target_bound:
            break

    # Final polishing via exact objective
    if best_h is not None:
        h_final = best_h.copy()
        # Run SLSQP to get a local minimum on the true function
        bounds = [(0, 1) for i in range(len(h_final))]
        def obj(h):
            return evaluate_h(h)  # compute exact c5
        try:
            result = minimize(obj, h_final, method='SLSQP',
                              bounds=bounds,
                              options={'maxiter': 500, 'ftol': 1e-12})
            if result.success:
                final_h = result.x
                final_c5 = evaluate_h(final_h)
                if final_c5 < best_c5:
                    best_c5 = final_c5
                    best_h = final_h
        except:
            pass

    if best_h is None:
        # Fallback to prior approach using initial_h_values and simple SLSQP
        try:
            h = np.array(initial_h_values)
            bounds = [(0, 1) for i in range(len(h))]
            result = minimize(evaluate_h, h, method='SLSQP', bounds=bounds, options={'maxiter': 500})
            return list(result.x), evaluate_h(result.x), len(h)
        except:
            return [0.5]*47, 0.4, 47  # Worst case fallback

    return list(best_h), best_c5, best_n
