import numpy as np
import cvxpy as cp
import time

# Constants
n_values = [48, 56, 64, 72]
target_bound = 0.3807
dx = 2.0 / 48  # standardize dx for initial setup


def solve_milp_binary(n, time_limit=60):
    """Solve the binary MILP formulation for given n."""
    # Variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()

    # Constraints
    constraints = [cp.sum(h) == n // 2, t >= 0]

    # Generate auxiliary variables and constraints for linearization
    # For each shift d, we define sum of h_i * h_{i+d} = Q_d
    for d in range(n):
        # y_{i,d} = h_i * h_{i+d}
        # We only consider valid indices: i+d < n, so i ranges from 0 to n-1-d
        y_vars = cp.Variable(n - d, boolean=True)
        # Add linearization constraints: y_{i,d} <= h_i; y_{i,d} <= h_{i+d}
        # y_{i,d} >= h_i + h_{i+d} - 1
        for i in range(n - d):
            constraints.append(y_vars[i] <= h[i])
            constraints.append(y_vars[i] <= h[i + d])
            constraints.append(y_vars[i] >= h[i] + h[i + d] - 1)

        # Q_d = sum of y_{i,d} over all valid i
        q_d = cp.sum(y_vars)
        # Enforce q_d >= t for all d
        constraints.append(q_d >= t)

    # Objective
    objective = cp.Maximize(t)

    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with CBC
    try:
        prob.solve(solver=cp.CBC, 
                   verbose=False,
                   maximumSeconds=time_limit)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_star = h.value
            if h_star is not None:
                t_star = t.value
                c5_bound = 1 - (2/n) * t_star
                return h_star, c5_bound
        # In case of early termination (infeasible but better bound)
        if prob.status == cp.SOLVER_ERROR or prob.status == cp.USER_LIMIT:
            if abs(t.value) > 0:
                h_star = h.value
                if h_star is not None:
                    t_star = t.value
                    c5_bound = 1 - (2/n) * t_star
                    return h_star, c5_bound
    except Exception:
        pass
    return None, None


def hill_climb_swaps(h, max_swaps=200000, time_limit=300):
    """Improves binary pattern using swap-based hill-climber."""
    start = time.time()
    h = np.array(h).copy()
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1 - h)

    # compute initial Q = correlation
    def evaluate_c5():
        Q = np.correlate(h, 1 - h, mode='full')
        return 1 - dx * np.min(Q)

    c5 = evaluate_c5()
    
    swaps = 0
    while swaps < max_swaps and (time.time() - start) < time_limit:
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # swap h[i] and h[j]
        h[i], h[j] = h[j], h[i]
        
        # update lists
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)

        new_c5 = evaluate_c5()
        if new_c5 < c5:
            c5 = new_c5
        else:
            # undo the swap if not helpful
            h[i], h[j] = h[j], h[i]
            ones = np.flatnonzero(h)
            zeros = np.flatnonzero(1 - h)
        swaps += 1

    return h, c5


def continuous_refine(h, n, time_limit=100):

    # Create finer interpolation to n points (assuming h is from smaller)
    h_fine = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h)), h)
    h_fine = np.clip(h_fine, 0, 1)

    # Project onto sum = n/2 (since dx = 2/n)
