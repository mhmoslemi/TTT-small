def weight_normalise(x, target_sum=1.0):
    """Rescales x to have sum equal to target_sum"""
    return x * (target_sum / np.sum(x))

def max_defect(h):
    """Compute exact C5 = max_k int h(x)(1-h(x+k)) dx = 1 - 2/n * min_k sum h[i] h[i+k]"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1-h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def build_milp(n):
    """Build binary MILP for max-min overlap"""
    h = Variable(n, boolean=True)
    w = Variable((n, n-1))  # w[i,s] for i=0..n-1, s=0..n-2 (shifts 1..n-1)  
    #Each element of w is w[i,s]  
    # we want w[i,s] = h[i] * h[i+s] with s index 1 to n-1  
    # Every i and s, we ensure w[i,s] is the binary AND of h[i] and h[i+s]  
    # SPARSE @INDEX APPROACH SCALES BETTER (LOGICPRIME) 

    # So our constraints are:
    #  
    # 1. Element-wise logic AND:
    # For 1 <= s <= n-1, for 0 <= i <= n-s-1:
    #           w[i,s] <= h[i]  
    #           w[i,s] <= h[i+s]  
    #           w[i,s] >= h[i] + h[i+s] - 1  
    # This is the standard linearisation  
    # Variables: h[i] (0 to n-1), w[i,s] (0 to n-s-1, 0 to n-2)  
    # Objective: 
    # 2. Min overlap constraint for each shift s:
    # Let t = minimum sum of w[i,s] over i for shift s=1..n-1
    # That means: sum_{i=0}^{n-s-1} w[i,s] >= t  
    # Though that's max min so we later 
    t = Variable(1)  # True: scalar variable = min sum over all shifts

    # weight constraint:
    constraints = [sum(h) == n/2]
    
    # Milp model
    for s in range(1, n):  # s from 1 to n-1  
        # i is from 0 to n-s-1  
        for i in range(0, n-s):
            # Add conservation constraints for w[i,s] = h[i] AND h[i+s]  
            constraints.append(w[i,s-1] <= h[i]) # s-1 for 0-based indexing  
            constraints.append(w[i,s-1] <= h[i+s])
            constraints.append(w[i,s-1] >= h[i] + h[i+s] - 1)

    # And the t constraints:
    # For each shift s:
    # sum over i of w[i,s] >= t  
    for s in range(1, n):
        idx = s - 1  # index into w  
        sum_w_s = sum(w[i, idx] for i in range(0, n-s))  
        constraints.append(sum_w_s >= t)

    # Objective: max t 
    objective = Maximize(t)
    
    # Create the problem and return optimal
    prob = Problem(objective, constraints)  
    return prob, h, w, t

def refine_continuous(h0, n):
    """Run trust-constr with smooth max and iterate α from 5 to 15 to 40."""
    dx = 2.0 / n  
    h0 = np.array(h0)
    # massage h0 to ensure weight constraint
    h0 = weight_normalise(h0, target_sum=1.0)  # So that sum h[i] dx = 1, so sum h[i] = n/2
    h_cont = h0.copy()  
    # Boundaries?
    bounds = [(0,1) for i in range(n)]

    # Remove any points that are fixed at 1 or 0
    # assume all are free until we preprocess

    # fit to feasible 1s and grid points  
    # Now smooth max following α = 5 -> 15 -> 40   # Use soft max approx based on 5, 15, 40  

    for alpha in [5, 15, 40]:
        res = minimize(
            lambda x: softmax_objective(x, alpha)[0], 
            h_cont, 
            method="trust-constr", 
            bounds=bounds, 
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2} ]
        )
        if res.success:
            h_cont = res.x  
        else:
            print(f"Warning: fail to converge with α={alpha}")
            break
    h_cont = np.clip(h_cont, 0, 1)  
    h_cont = weight_normalise(h_cont, target_sum=1.0)
    return h_cont

def anneal(h):
    """Stochastic fine tuning via swap steps."""
    dx = 2.0 / len(h)
    # Original correlation: compute D / dx
    c = correlate(h, 1-h, 'full') * dx  
    max_know = np.max(c)
    best = h[:]
    best_c = max_know
    # Precompute map of which positions matter  
    shift_max_idx = np.argmax(c)  
    shift_val = shift_max_idx - len(h) + 1 
    # shift_val from -len(h)+1 to +len(h)-1, so actually from 1 to len(h)-1 if shift is positive
    max_shift = shift_val  

    # Idea: for position j for shift x, if h[j] > 0 and [j+x] has low h. It's the most wrong  
    # Avoid double loop 

    c = c[len(h)-1:]  # Only positive indices  
    max_ind = np.argmax(c)
    max_shift = max_ind + 1
    
    max_c = c[max_ind]
    # compute c = h * (1-h) for each shift k 
    # This is already done

    print(f"Initial max shift={max_shift}, value={max_c}")

    it_cycle = 0
    gamma = 0.2
    limit = 5000
    timeout = 550.0
    start = time.time()
    # For each swap, we try:
    # 1. Find the max shift, and its right/important indices  
    # 2. For that max shift k, find indices that are positioned to most harm: h[j] > 0 and h[j+k] approximate small  
    # 3. Find a swap point where h[j] is high and h[j+k] is low, but some where h[l] is low but h[l+k] is high  
    # 4. Do a small move of gamma weight 

    while it_cycle < limit and time.time() - start < timeout:
        c = correlate(best, 1-best, 'full') * dx
        # get the max-shift index
        max_ind = np.argmax(c[len(best)//2:])  # We only look at positive shifts (index > len(h)//2 )  
        max_shift = max_ind + 1 

        # Determine the indices that are vectorwise active  
        # These are indices where h[i] * (1-h[i+k]) is large. 
        # So h[i] > 0 and also h[i+k] is small.  
        # (('h' not zeros and not ones)) most beneficial.  

        # Practically, for this shift, destabilize positions with big h[i] and small h[i+k]  
        # then swap with position such that h[l] small and 1-h[l+k] large  
        # But be careful we make balance  

        # We look at the shift shift_max  
        near_worst = []
        for i in range(len(best)):
            if i + max_shift < len(best):
                if best[i] > 0.01 and best[i+max_shift] < 0.99:
                    near_worst.append(i)
        if not near_worst:
            # no clear focal points
            break
        # pick a random such index  
        i = np.random.choice(near_worst) 
        # Pick another area for a solvent  
        j = np.random.randint(0, len(best))  
        # Use a soft selection if possible.  

        # Sel best[j] that has low h[j], but h[j+k] not too low
        choices = []
        for idx in range(len(best)):
            if idx + max_shift < len(best):
                if best[idx] < 0.3: # low h  
                    # but also h[idx+max_shift] not too low  
                    if best[idx+max_shift] > 0.1:  
                        choices.append(idx) 
        if choices:
            j = np.random.choice(choices)
        else:
            # if no such choices, just pick any
            j = np.random.randint(len(best)) 

        # calculate amount to move 
        flow = min(best[i], 1.0 - best[j], gamma)  # minimum in move amounts  
        if flow < 1e-10:
            continue  

        # Apply swap with fallback  
        h_new = best[:]
        h_new[i] -= flow
        h_new[j] += flow  
        h_new[i+max_shift] += flow if i+max_shift < len(best) else 0
        h_new[j-max_shift] -= flow if j-max_shift >= 0 else 0
        # Ensure Feasibility?  
        # Keep weight sum restrictively 'equal'  
        h_new = weight_normalise(h_new, 1.0)  # project to feasible simplex

        # compute new def  
        c_new = correlate(h_new, 1-h_new, 'full') * dx
        max_k_new = np.max(c_new) 

        if max_k_new <= best_c:  # No degradation
            best = h_new[:]  
            best_c = max_k_new
        elif np.random.rand() < 0.1:  # Accepting slightly worse with small chance  
            # Sometimes we are trapped in a very narrow energy well, so allow some very rare warming up  
            best = h_new[:]  
            best_c = max_k_new  

        # cooling
        gamma *= 0.99
        it_cycle += 1  

    return best, best_c

    print(f"Start MILP stage")  
    try:
        problem, h_var, w_var, t_var = build_milp(n)  
        # solver_opts = {"maxTime": 180, "seconds": 180, "ratioGap": 0.0}
        problem.solve(solver=CPLEX, verbose=False, maxTime=180, seconds=180, ratioGap=0.0)  # possibly timeout?  
        print(f"MILP solver result: success={problem.status == OPTIMAL or problem.status == FEASIBLE}")
        if problem.status in [OPTIMAL, FEASIBLE]:  
            h_opt = h_var.value  
            # Convert to list
            h_bin = [float(x) for x in h_opt]  
            print("Binary solution obtained from MILP: removing blocks...")
        else:
            print("MILP solver timed out/preempted")  
            # fallback to continuous
            h_bin = h_current[:]  
        
    except Exception as e:
        print(f"MILP failed with error {e}, fallback to continuous")  
        h_bin = h_current[:]  

    # Phase 2: discrete binary → continuous refinement  
    print("Continuation refining binary pattern")
    h_cont = refine_continuous(h_bin, n)  
    h_initial = h_cont[:]  

    print("Refined continuous", max_defect(h_cont)) 

    # Phase 3: Post-processing swap phase
    print("Swap refinement phase...")
    h_final, c5_final = anneal(h_cont)
    print(f"Final result: C5 = {c5_final:.6f}")

    # Return the best solution
    answer_c5 = c5_final
    answer_h = h_final[:]
    answer_n = n

    # Check time budget still remains
    print(f"Returning after {time.time()-start_time:.3f} seconds")
    return (answer_h, answer_c5, answer_n)
