from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

from itertools import combinations

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

    def project_simplex(x, s=1.0):
        """Project x onto the simplex {y: sum(y)=s, y>=0}"""
        u = np.sort(x)[::-1]
        css = np.cumsum(u)
        rho = np.where(u > (s - css) / np.arange(1, len(u) + 1))[0]
        if len(rho) == 0:
            return np.maximum(x - (np.sum(x) - s) / len(x), 0)
        else:
            theta = (s - css[rho[-1]]) / (rho[-1] + 1)
            return np.maximum(x - theta, 0)
    
    def binary_milp_optimize(n, time_limit):
        try:
            # Create variables
            b = cp.Variable(n, boolean=True)
            t = cp.Variable()
            
            # Constraints
            constraints = [cp.sum(b) == n // 2, t >= 0]
            
            # Product constraints
            for i in range(n):
                for j in range(i, n):
                    y = cp.Variable(boolean=True)
                    constraints += [
                        y <= b[i],
                        y <= b[j],
                        y >= b[i] + b[j] - 1
                    ]
                    # Now we need to express sum of y[i,i+s] for each shift
                    # This is a bit tricky, seeing as we're modeling this naively
                    # Another approach is to simulate the shifts in the objective
                    # but for constraint purposes, compute overall overlap of each shift
                
            shifts = [0]  # Start with base case
            
            # Constraint for each shift
            # Sum_{i: 0 ≤ i < n, i+s < n} b_i * b_{i+s} ≥ t
            # Which is equivalent to: sum over valid pairs (i, i+s) of b_i * b_{i+s} ≥ t
            
            for s in range(n):
                terms = []
                for i in range(n - s):
                    y = cp.Variable(boolean=True)
                    constraints += [
                        y <= b[i],
                        y <= b[i + s],
                        y >= b[i] + b[i + s] - 1
                    ]
                    terms.append(y)
                constraints.append(cp.sum(terms) >= t)
            
            # Objective
            prob = cp.Problem(cp.Maximize(t), constraints)
            
            # Solve
            prob.solve(solver=cp.CBC, max_seconds=time_limit)
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_ERR]:
                t_opt = t.value if t.value is not None else 0
                # Convert b to the floating point representation
                h_binary = np.array([float(x.value) for x in b])
                return h_binary, t_opt, True
        except:
            pass
        return None, None, False
    
    def minimize_with_gradient_descent(h0, max_iter=2000, tol=1e-8):

        best_c5 = evaluate_c5(h)

                break

            g = np.zeros_like(h)
            for s in active_shifts:
                # For shift s, the gradient term is (1 - h_{i+s}) - (1 - h_{i-s}) == h_{i-s} - h_{i+s}
                # Simplistic but we approximate:
                # At i: 1 - h_{i+s}, so gradient with respect to h_i is -1
                # However we use standard approach for derivative of correlation
                # Let C_s = sum_i h_i*(1 - h_{i+s}), so dC_s/dh_i = 1 - h_{i+s} (when i+s is valid) or 0 otherwise
                # But also dC_s/dh_i = -h_{i-s} (when i-s is valid) or 0 otherwise
                # Approximate the gradient
                s_idx = s
                for j in range(len(h)):
                    if j + s_idx < len(h):
                        g[j] += (1 - h[j + s_idx])
                    if j >= s_idx:
                        g[j] -= h[j - s_idx]
            
            # Normalize gradient
            if np.linalg.norm(g) > 1e-10:
                g = g / np.linalg.norm(g)
            
            # Step
            alpha = 1e-3 / np.sqrt(1 + i)

            # Projection
            h_new = np.clip(h_new, 0, 1)
            h_new = project_simplex(h_new, 1.0)
            
            c5_new = evaluate_c5(h_new)

                best_c5 = c5_new
                best_h = h_new.copy()
                
            h = h_new
            if abs(c5_new - best_c5) < tol:
                break
        
        return best_h, best_c5
