def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_milp_binary(n, time_limit):
    """Solve binary MILP for exact optimum on n points"""
    m = n // 2
    dx = 2.0 / n
    b = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(b) == m,
        t <= 0  # will be updated later
    ]
    
    # Compute all overlaps efficiently using matrix notation
    # For a shift s, the overlap is sum_{i=0}^{n-1-s} b[i] * b[i+s] 
    # Construct the problem
    overlaps = []
    for s in range(n):
        if s == 0:
            # When s=0, overlap is sum of b[i]*b[i] = sum of b[i]
            overlap = cp.sum(b)
        else:
            # Sum b[i]*b[i+s] for valid i
            expr = cp.sum(cp.multiply(b[:n-s], b[s:]))
            overlaps.append(expr)
    
    # All overlaps must be >= t
    for overlap in overlaps:
        constraints.append(overlap >= t)
    
    # Objective: maximize t
    prob = Problem(Maximize(t), constraints)

    # Try to solve with a time limit
    try:
        prob.solve(solver=cp.CBC, time_limit=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_USING_CVXOPT]:
            t_opt = t.value
            binary_solution = b.value
            return binary_solution, t_opt
        else:
            return None, None
    except:
        return None, None

def project_simplex(x, s=1.0):
    """Project vector x onto the simplex {y : sum(y) = s, y >= 0}"""
    # Sort the values in descending order
    sorted_x = np.sort(x)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_x)
    # Find the threshold
    rho = np.argmax(sorted_x * np.arange(1, len(sorted_x)+1) > (cumsum - s))
    if rho == 0 and sorted_x[0] * len(sorted_x) <= s:
        rho = len(sorted_x)
    theta = (cumsum[rho-1] - s) / rho if rho > 0 else sorted_x[0] - s
    # Project back
    y = np.clip(x - theta, 0, None)
    return y

def gradient_step(h, step_size=1e-3, max_iter=2000):
    """Improve h using projected gradient descent"""
    n = len(h)
    dx = 2.0 / n
    last_c5 = evaluate_c5(h)
    best_c5 = last_c5
    best_h = h.copy()

    for iter_num in range(max_iter):
        if iter_num % 100 == 0:
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if c5 <= 0.3807:
                break

        # Find active shifts
        corr = correlate(h, 1 - h, mode="full")
        s_max = np.argmax(corr)
        active_shifts = np.where(corr == np.max(corr))[0]

        # Compute subgradient
        sub_grad = np.zeros_like(h)
        for s in active_shifts:
            if s < n:
                # Forward shift
                for i in range(n-s):
                    sub_grad[i] += 1 - h[i+s]
                # Backward shift  
                for i in range(s, n):
                    sub_grad[i] += -h[i-s]
            else:
                # Backward shift
                s_back = s - n
                for i in range(n-s_back):
                    sub_grad[i+s_back] += 1 - h[i]
                for i in range(s_back, n):
                    sub_grad[i] += -h[i-s_back]

        # Normalize subgradient
        sub_grad /= len(active_shifts)

        # Evolve 
        h_new = h - step_size * sub_grad
        h_new = np.clip(h_new, 0, 1)
        h_new = project_simplex(h_new)
        
        # Check if improvement
        new_c5 = evaluate_c5(h_new)
        if new_c5 < last_c5:
            h = h_new
            last_c5 = new_c5
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        
        step_size *= 0.999  # decay learning rate
    return best_h, best_c5

def simulated_annealing(n, num_steps):
    """Simulated Annealing for binary configuration"""
    np.random.seed()
    h = np.zeros(n, dtype=int)
    h[:n//2] = 1
    np.random.shuffle(h)

    current_energy = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_energy

    # Start schedule
    T0 = 0.1
    tau = num_steps
    for t in range(num_steps):
        if t % 1000 == 0 and t > 0:
            T = T0 * (1 - t / tau)
            if T < 1e-8:
                break

        # Local move: swap 1 and 0
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        while h[i] == h[j]:
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
        
        # Perform move
        h[i], h[j] = h[j], h[i]

        # Evaluate new energy
        new_energy = evaluate_c5(h)
        
        # Metropolis criterion
        if new_energy < current_energy or np.random.rand() < np.exp(-(new_energy - current_energy) / T):
            current_energy = new_energy
        else:
            # Undo the move
            h[i], h[j] = h[j], h[i]

        if current_energy < best_c5:
            best_c5 = current_energy
            best_h = h.copy()
            if current_energy <= 0.3807:
                break

    return best_h, best_c5
