import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, log as cp_log, exp as cp_exp

def weight_normalise(h, n):
    """Rescales h to satisfy sum(h) = n/2"""
    dx = 2.0 / n
    scale = (n / 2.0) / np.sum(h)
    return h * scale

def max_defect(h):
    """Computes exact C5 = max_k int h(x)(1-h(x+k)) dx"""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr / 2.0  #dx = 2/n so we scale

def build_milp(n):
    """Builds and solves the binary MILP for maximum minimum overlap"""
    # Variables
    h = Variable(n, boolean=True)
    w = Variable((n, n - 1), nonneg=True)  # w[i,s] = h[i] * h[i+s]
    t = Variable()  # minimum overlap
    dx = 2.0 / n
    
    # Constraints
    constraints = [
        cp_sum(h) == n / 2.0  # weight constraint
    ]
    
    # AND constraints: w[i,s] = h[i] * h[i+s]
    for s in range(1, n):
        for i in range(n - s):
            # w[i,s] <= h[i]
            constraints.append(w[i, s-1] <= h[i])
            # w[i,s] <= h[i+s]
            constraints.append(w[i, s-1] <= h[i+s])
            # w[i,s] >= h[i] + h[i+s] - 1
            constraints.append(w[i, s-1] >= h[i] + h[i+s] - 1)
    
    # minimum overlap constraint
    for s in range(1, n):
        constraints.append(cp_sum(w[:, s-1]) >= t)
        
    # Objective
    objective = Minimize(-t)  # maximize t
    prob = Problem(objective, constraints)
    
    # Parameters
    solver_opts = {"maxTime": 180, "ratioGap": 0.0}
    
    # Solve
    try:
        prob.solve(solver=cp.CBC, **solver_opts)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value
        else:
            # Fallback to best known solution
            return None
    except:
        return None

def softmax_objective(h, alpha, n):
    """Log-sum-exp approximation of max over shifts"""
    dx = 2.0 / n
    # Compute correlation for all shifts (full correlation)
    corr_full = correlate(h, 1 - h, mode='full')
    # Shift k is in corr_full at index k (not including the full convolution)
    corr = corr_full[len(corr_full)//2:]  # Only positive shifts
    
    # Soft maximum
    exp_vals = np.exp(alpha * corr)
    total = np.sum(exp_vals)
    soft_max = (1.0 / alpha) * np.log(total)
    
    # Gradient
    weights = exp_vals / total
    grad = np.zeros_like(h)
    for i in range(len(h)):
        for s in range(1, len(corr)):  # shift s
            if i + s < len(h):
                grad[i] += weights[s-1] * (1 - h[i + s])
            if i - s >= 0:
                grad[i] -= weights[s-1] * h[i - s]
    grad = grad / alpha
    
    return soft_max, grad

def refine_continuous(h_bin, n):
    """Refine binary solution with smooth max to get continuous"""
    max_iter_per_stage = 200
    dx = 2.0 / n
    h_refined = h_bin.copy()
    h_refined = weight_normalise(h_refined, n)
    
    # Set up free variables near boundary
    # Locate change points
    change_points = [0]
    for i in range(1, len(h_bin)):
        if h_bin[i] != h_bin[i-1]:
            change_points.append(i)
    change_points.append(len(h_bin))
    
    # Identify boundaries as free variables with up to 4 points each
    # Represent as a local patch around change points
    boundary_indices = []
    for i, cp in enumerate(change_points):
        if i == 0:
            local_start = 0
        else:
            local_start = max(0, cp - 4)
        if i == len(change_points) - 1:
            local_end = len(h_bin)
        else:
            local_end = min(len(h_bin), cp + 4)
        
        for j in range(local_start, local_end):
            if (j == 0) or (j == len(h_bin) - 1) or (h_bin[j] != h_bin[j-1] or h_bin[j] != h_bin[j+1]):
                boundary_indices.append((j, i)) # index, change point

    #-------- Assume variables are not binary, use refined approach ---
    bs = np.linspace(0, 1, 50)
    num_bs = len(bs)
    best_h = None
    best_c5 = float('inf')
    
    # Relax the binary structure
    # Smooth max approximation to refine
    h0 = h_refined.copy()
    
    h_final = h0.copy()
    
    # Alpha schedule
    alphas = [5.0, 15.0, 40.0]  # Increased convergence order
    
    for alpha in alphas:
        # Solve with new alpha
        def obj_and_grad(h):            
            val, grad = softmax_objective(h, alpha, n)
            return val, grad
            
        # Solve numerically
        res = minimize(
            lambda h: obj_and_grad(h)[0],
            h_final,
            method='trust-constr',
            jac=lambda h: obj_and_grad(h)[1],
            bounds=[(0.0, 1.0)]*n,
            constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - n/2},
            options={'maxiter': max_iter_per_stage, 'gtol': 1e-6}
        )
        
        if res.success:
            h_final = res.x.copy()
            h_final = weight_normalise(h_final, n)
            c5 = max_defect(h_final)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_final.copy()
        else:
            # If not successful, move to next alpha but use last successful
            pass       
        
    return best_h if best_h is not None else h_final


def swap_anneal(h0, n, max_iters=5000):
    """Annealing that swaps mass between indices to reduce maximum defect"""
    h = h0.copy()
    dx = 2.0 / n
    
    # Precompute correlations for fast update
    corr_full = correlate(h, 1 - h, mode='full')
    corr = corr_full[len(corr_full) // 2:]
    
    max_defect_val = np.max(corr)
    
    # Start with id temperature and a small changing amount
    gamma = 0.2
    temp = 1.0
    
    # Track history for early stopping
    last_improvement = 0
    converged = False
    it = 0
    
    # Track operations for debugging
    start_time = time.time()
    max_time = 550  # Leave 50s for finishing
    
    while it < max_iters and (time.time() - start_time) < max_time:
        # Find the worst shift
        s_star = np.argmax(corr)
        if corr[s_star] == max_defect_val:
            # Very last attempt before stopping
            break
            
        # Find indices contributing to max shift (high h[i], low h[i+s_star] )
        good_set_indices = []
        for i in range(len(h)):
            if h[i] > 0.1 and i+s_star < len(h) and h[i+s_star] < 0.5:
                good_set_indices.append(i)
        
        if not good_set_indices:
            # If no good candidates, better to stop early
            break
            
        # Pick a random element from good set
        i_b = np.random.choice(good_set_indices)
        # Find a spot to transfer from
        candidate_j = []
        for j in range(len(h)):
            if h[j] < 0.5 and j+s_star < len(h) and h[j+s_star] > 0.2:  # Can take mass from high area
                candidate_j.append(j)
        
        if not candidate_j:
            break
            
        j_b = np.random.choice(candidate_j)
        
        # Mass flow: take delta = min(h[i_b], 1-h[j_b]) * gamma
        delta = min(h[i_b], 1.0 - h[j_b]) * gamma
        if delta < 1e-6:
            continue # Too small change
            
        # Compute effect on max defect
        old_defect_s = corr[s_star]
        
        # Do transfer
        h[i_b] -= delta
        h[i_b + s_star] -= delta  # Also update the shifted location (if that matters for max calc)
        h[j_b] += delta
        h[j_b + s_star] += delta  # Potential improvement also for shifted half (or not used)
        
        # Update the correlations for just the affected shifts
        # Manually update the contribution
        # Recompute corr row: do old max analysis
        
        # Let's recompute max correlation efficiently
        corr_new = correlate(h, 1 - h, mode='full')
        max_new = np.max(corr_new[len(corr_new) // 2:])
        
        # Accept according to Boltzmann rule
        if max_new <= old_defect_s:
            corr[s_star] = max_new  # Apply update for that specific shift
            max_defect_val = max_new
        else:
            # undo
            h[i_b] += delta
            h[j_b] -= delta
            h[i_b + s_star] += delta
            h[j_b + s_star] -= delta
            # Don't update corr because didn't accept move
            
        # Cooling schedule
        gamma *= 0.99
        it += 1
        
        # Gravity bounce to prevent getting stuck
        if it % 100 == 0 and gamma < 0.02:
            # Collate performance
            print(f"Annealing iteration {it}, C5: {np.max(corr)}, Gamma: {gamma}")
            # ... potentially abandon to avoid severely stuck
            # If if no progress, abandon
            # For now, just break if too playing 
    # Post computation of final correlation
    final_corr = correlate(h, 1 - h, mode='full')
    final_max = np.max(final_corr[len(final_corr) // 2:])
    h_final = h.copy()
    return h_final, final_max

def run(seed=42, budget_s=600, **kwargs):

    n = 100  # Target resolution

    h_best = None

    # --- Try relaxed to allow binary and proceed warm ---

    print("Solving binary MILP...", flush=True)

    print("MILP finished.")

    if h_bin is None:
        print("Falling back to random start continuous optimization...", flush=True)

        h_initial = np.random.rand(n)
        h_initial = weight_normalise(h_initial, n)
        h_best = h_initial.copy()
        c5_best = max_defect(h_best)
    else:
        print("Binary MILP successful.")
        # Refine found solution
        h_cont = refine_continuous(h_bin, n)
        h_best = h_cont.copy()
        c5_best = max_defect(h_best)
        
    # Add swap-annealing
    h_anneal, c5_anneal = swap_anneal(h_best, n)
    if c5_anneal < c5_best:
        h_best = h_anneal
        c5_best = c5_anneal
    
    # Final power benchmark to make sure we don't drop focus in later stage
    # Do one last full-exact computation here.
    c5_final = max_defect(h_best)
    
    return (h_best.tolist(), c5_final, n)
