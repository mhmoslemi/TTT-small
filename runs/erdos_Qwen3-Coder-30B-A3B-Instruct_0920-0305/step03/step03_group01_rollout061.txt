import numpy as np
from scipy.signal import correlate
import time
import math

def project_onto_box_and_hyperplane(y, target):
    # Binary search for the Lagrange multiplier lambda
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0, 1)

def compute_gradient(h, shift):
    n = len(h)
    if shift >= 0:
        term1 = np.empty(n)
        term1[:n-shift] = 1 - h[shift:]
        term1[n-shift:] = 1
        term2 = np.empty(n)
        term2[shift:] = h[:n-shift]
        term2[:shift] = 0
    else:
        s = -shift
        term1 = np.empty(n)
        term1[s:] = 1 - h[:n-s]
        term1[:s] = 1
        term2 = np.empty(n)
        term2[:n-s] = h[s:]
        term2[n-s:] = 0
    return term1 - term2

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    start_time = time.time()
    n_points = 256  # Use 256 points for better resolution
    
    # Initial guess to start, tent shape
    x = np.linspace(0, 2, n_points)
    h_initial = np.maximum(0, 1 - np.abs(x - 1) * 2)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)  # Scale for sum constraint
    
    # Optimization parameters
    alpha = 0.05
    max_iter = 5000
    stagnation_threshold = 100
    best_h = h_initial.copy()
    best_c5 = evaluate_c5(h_initial)
    
    # Gradient descent optimization
    iter_count = 0
    last_improvement = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s * 0.98:
        # Compute correlation
        corr = correlate(best_h, 1 - best_h, mode="full")
        max_corr = np.max(corr)
        max_ind = np.where(corr == max_corr)[0]
        shift = max_ind[0] - (n_points - 1)
        
        # Compute gradient for the worst shift
        g = compute_gradient(best_h, shift)
        
        # Update step
        h_new = best_h - alpha * g
        
        # Project onto feasible region
        h_new = project_onto_box_and_hyperplane(h_new, n_points / 2)
        
        # Evaluate new point
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            last_improvement = iter_count
        
        iter_count += 1
        
        # Stopping condition
        if iter_count - last_improvement > stagnation_threshold:
            break
            
        # Gradually decrease learning rate
        alpha = max(0.001, alpha * 0.999)
    
    # Final check
    final_c5 = evaluate_c5(best_h)
    
    # Return best solution found
    return (best_h.tolist(), final_c5, n_points)
