import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time
import cvxpy as cp
from cvxpy import SCS

# Constants
N_VALUES = [36, 42, 48, 54, 60]  # Even n values to try
R = 30  # Number of random rounds for SDP rounding
MAX_SWAPS = 50000  # Per hill climb
S = 2000  # Restart threshold in hill climbing
EMBED_DIM = 10  # For rounding (rank of PSD factor)
ALPHA = 30.0  # Softmax parameter
LAMBDA = 100.0  # Penalty for sum constraint
TARGET_BOUND = 0.3807
TIME_SAFETY_MARGIN = 5  # seconds

def evaluate_h(h_values):
    """Compute the exact C5 bound of h."""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * (2.0 / len(h))

def smooth_max(h_values, alpha=ALPHA):
    """Compute smooth approximation of max correlation."""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    log_sum_exp = np.log(np.sum(np.exp(alpha * correlation)))
    return log_sum_exp / alpha

def proj_to_simplex(v, s=1.0):
    """Project vector v onto the simplex with sum=s."""
    v = np.asarray(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(u) + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[rho - 1] / rho
        result = np.maximum(v - theta, 0)
        return result
    else:
        return np.zeros_like(v)

def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-x))

def logit(x):
    return np.log(x / (1 - x))

def solve_sdp(n):
    """Solve the SDP relaxation for a given n."""
    # Variables
    x = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    constraints = [
        0 <= x, x <= 1,
        cp.sum(x) == n / 2
    ]
    
    # Diagonal equality
    for i in range(n):
        constraints.append(X[i, i] == x[i])

    # McCormick constraints
    for i in range(n):
        for j in range(i + 1, n):
            constraints.extend([
                X[i, j] >= 0,
                X[i, j] <= x[i],
                X[i, j] <= x[j],
                X[i, j] >= x[i] + x[j] - 1
            ])

    # Overlap constraints
    for k in range(n):
        # sum h[i] * (1 - h[i+k]) for i = 0,..,n-1-k
        # ->
        # sum h[i] - sum h[i] h[i+k] for i = 0,..,n-1-k
        # this is <= sum h[i] - sum X_{i, i+k} 
        left_sum = cp.sum(x[:n - k])
        diag_k = cp.diag(X, k)
        right_sum = cp.sum(diag_k)
        constraints.append(left_sum - right_sum <= t)

    # PSD constraint
    constraints.append(X >> 0)

    objective = cp.Minimize(t)
    
    try:
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=SCS, max_iters=8000, eps=1e-4)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value, X.value, prob.value
    except Exception:
        pass
    return None, None, None

def round_sdp_solution(x_star, X_star, n):
    """Use random hyperplane rounding of the SDP solution."""
    # Perform SVD or eigen-decomposition on X*
    U, s, _ = np.linalg.svd(X_star)
    s = np.maximum(s, 0)  # ensure non-negative
    V = U @ np.diag(np.sqrt(s))
    
    best_h = None
    best_c5 = float('inf')

    for _ in range(R):
        g = np.random.randn(EMBED_DIM)
        v = V[:, :EMBED_DIM] @ g
        sgn = (v >= 0).astype(int)
        
        # Ensure sum = n/2
        current_sum = np.sum(sgn)
        delta = int(current_sum - n / 2)
        
        if delta > 0:
            # Too many 1s: make delta zeros
            indices_to_flip = np.argsort(np.abs(v))[-delta:]
            for i in indices_to_flip:
                if sgn[i] == 1:
                    sgn[i] = 0
        elif delta < 0:
            # Too many 0s: make -delta ones
            indices_to_flip = np.argsort(np.abs(v))[:abs(delta)]
            for i in indices_to_flip:
                if sgn[i] == 0:
                    sgn[i] = 1
        
        c5 = evaluate_h(sgn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = sgn.copy()
    return best_h, best_c5

def hill_climb_binary(h_init, n):
    """Perform hill climb on binary vectors with swaps."""
    h = np.array(h_init, dtype=int)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    best_h = h.copy()
    best_c5 = evaluate_h(best_h)
    last_improved = 0
    swaps = 0

    while swaps < MAX_SWAPS:
        if swaps - last_improved > S:
            # Find worst shift
            c = np.correlate(h, 1 - h, mode="full")
            k = np.argmax(c)
            mtch = []
            for i in range(n - k):
                if h[i] == 1 and h[i + k] == 0:
                    mtch.append(i)
            if mtch:
                # Try all swaps that could fix one mismatch
                for i in mtch:
                    for j in zeros:
                        h_new = h.copy()
                        h_new[i] = 0
                        h_new[j] = 1
                        c5_new = evaluate_h(h_new)
                        if c5_new < best_c5:
                            h = h_new.copy()
                            best_c5 = c5_new
                            ones = list(np.where(h == 1)[0])
                            zeros = list(np.where(h == 0)[0])
                            last_improved = swaps
                            break
                    else:
                        continue
                    break
                else:
                    # No change found for this mismatches, so restart
                    return best_h, best_c5

        # Normal swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        c5_new = evaluate_h(h_new)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            ones = list(np.where(h == 1)[0])
            zeros = list(np.where(h == 0)[0])
            last_improved = swaps
        swaps += 1
        if swaps % 10000 == 0 and best_c5 < TARGET_BOUND:
            break

    return best_h, best_c5

def refine_with_smooth_max(h_init, n):
    """Refine a vector using smooth max surrogate."""
    def objective(y):
        h = sigmoid(y)
        soft = smooth_max(h)
        penalty = LAMBDA * (np.sum(h) - n / 2)**2
        return soft + penalty

    y0 = logit(np.clip(h_init, 1e-6, 1 - 1e-6))
    try:
        res = minimize(objective, y0, method='L-BFGSB', bounds=[(-50, 50)] * n, options={'maxiter': 300})
        if res.success:
            h_opt = sigmoid(res.x)
            h_proj = proj_to_simplex(h_opt, n / 2)
            return h_proj, evaluate_h(h_proj)
    except:
        pass
    return h_init, evaluate_h(h_init)
