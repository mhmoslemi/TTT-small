def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute full correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx to get C5
    c5 = max_correlation * dx
    return c5


def solve_milp_binary(n, time_limit=120):
    """Solve the exact binary MILP formulation for given n."""
    dx = 2.0 / n
    # Variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(pos=True)
    # Add constraint that sum of h is n/2
    constraints = [cp.sum(h) == n / 2]

    # Precompute quantities to avoid repeated computation
    y_vars = {}
    Q_vals = []

    for d in range(n):
        # Compute the range of valid i for this d
        i_range = list(range(n - d))
        y = cp.Variable(len(i_range), boolean=True)
        y_vars[d] = y
        # Compute Q_d = sum_{i} y_{i,d}
        Q_d = cp.sum(y)
        Q_vals.append(Q_d)
        # Add MILP linearization constraints for y_{i,d}
        for i, y_i in zip(i_range, y):
            constraints.extend([
                y_i <= h[i],
                y_i <= h[i + d],
                y_i >= h[i] + h[i + d] - 1
            ])
        # Add constraint that Q_d >= t
        constraints.append(Q_d >= t)

    # Objective is to maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    # Solve with time limit
    try:
        prob.solve(solver=cp.CBC, verbose=False, time_limit=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            t_opt = t.value
            c5_bound = 1 - (2 / n) * t_opt
            return h_opt, c5_bound
        elif prob.status == cp.SOLVER_ERROR:
            # Attempt to return best known incumbent
            h_opt = h.value
            if h_opt is not None:
                # Recalculate what the current bound is
                t_opt = t.value if t.value is not None else 0.0
                c5_bound = 1 - (2 / n) * t_opt
                return h_opt, c5_bound
    except Exception:
        pass
    return None, float("inf")


def hill_climb_swaps(h, max_swaps=200000, time_limit=300):
    """Improves solution by picking random pairs and swapping."""
    start = time.time()
    h = h.copy()
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1 - h)
    n = len(h)
    dx = 2.0 / n
    evals = 0

    while evals < max_swaps and time.time() - start < time_limit:
        # Try a random move
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # Swap
        h[i], h[j] = h[j], h[i]
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        # Evaluate
        c5 = evaluate_h(h)
        if c5 < 0.3807:
            return h, c5
        evals += 1
    return h, evaluate_h(h)
