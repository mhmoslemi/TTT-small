```python
import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time

# Constants
target_bound = 0.3807
n_list = [48, 60, 72, 84]
n_points = 72
dx = 2.0 / n_points

def evaluate_h(h):
    """Compute the exact C5 value for given h."""
    h = np.array(h)
    # Compute full correlation: correlate h with (1-h)
    c = np.correlate(h, 1-h, mode='full')
    # Return max correlation times dx
    return np.max(c) * dx

def correlation_vec(h):
    """Return the full correlation vector."""
    return np.correlate(h, 1-h, mode='full')

def smooth_max(c, alpha=30):
    """Approximate max with smooth log-sum-exp."""
    m = np.max(c)
    if m == -np.inf:
        return -np.inf
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

def penalty_sum(h, lam=10000):
    """Penalize deviation from integral constraint."""
    return lam * (np.sum(h) - n_points / 2)**2

def proj_to_simplex(v, S=None):
    """Project vector v into the simplex with sum=S."""
    if S is None:
        S = n_points / 2
    v = np.clip(v, 0, 1)
    total = np.sum(v)
    if total > S:
        # Shrink
        v -= (total - S) / len(v)
    elif total < S:
        # Expand
        v += (S - total) / len(v)
    # Ensure it's in [0,1]
    v = np.clip(v, 0, 1)
    # Adjust by scaling
    if abs(total - S) > 1e-10:
        v = v * S / total
    return np.clip(v, 0, 1)

def sigmoid(y):
    """Logistic sigmoid."""
    return 1 / (1 + np.exp(-y))

def logit(p):
    """Inverse sigmoid."""
    return np.log(p / (1 - p))

def binary_initial(n):
    """Generate a random binary vector with exactly n/2 ones."""
    half = n // 2
    h = np.zeros(n)
    h[:half] = 1
    np.random.shuffle(h)
    return h

def swap_and_evaluate(h, i_one, i_zero):
    """Swap h[i_one] and h[i_zero], return new C5."""
    hnew = h.copy()
    hnew[i_one] = 0
    hnew[i_zero] = 1
    return evaluate_h(hnew)

def binary_hill_climb(h_in, max_iter=50000):
    """Binary hill-climbing."""
    h = h_in.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    for _ in range(max_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_one = ones[np.random.randint(len(ones))]
        i_zero = zeros[np.random.randint(len(zeros))]
        # Perform swap
        h[i_one] = 0
        h[i_zero] = 1
        c5 = evaluate_h(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            # Update lists
            ones.remove(i_one)
            zeros.remove(i_zero)
            ones.append(i_zero)
            zeros.append(i_one)
        else:
            # Revert
            h[i_one] = 1
            h[i_zero] = 0
    return best_h, best_c5

def parametric_intervals(n):
    """Build h from a shifted interval pattern."""
    h = np.zeros(n)
    total_len = 1.0
    m = 2
    while m <= 8:
        g = (2 - m * total_len) / (m - 1)
        if g < 0: break
        for d in [0, g/4, g/2, 3*g/4, g]:
            d = max(0, min(g, d))
            start = d
            end = start + total_len
            if end <= 2:
                i_start = int(start/dx)
                i_end = int(end/dx)
                if i_end < n:
                    h[i_start:i_end] = 1
                    yield h.copy()
            # reset
            h[:] = 0
        m += 1

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None

    # Stage 2.1: Parametric interval family search
    print("Stage 2.1: parametric intervals...")
    for n in n_list:
        dx_n = 2.0 / n
        for h_candidate in parametric_intervals(n):
            # norm to integral 1
            h_norm = h_candidate * (n / (2 * np.sum(h_candidate)))
            c5_val = evaluate_h(h_norm)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_norm.copy()
                best_n = n
            if c5_val <= target_bound:
                print(f"Early break at {c5_val}")
                return list(best_h), best_c5, best_n

    # Stage 2.2: Binary hill climbing
    print("Stage 2.2: binary hill-climb...")
    time_left = budget_s - (time.time() - start_time)
    for n in n_list:
        n_time = time_left * 0.5 / len(n_list)
        if time.time() - start_time + n_time > budget_s - 5:
            break
        t0_n = time.time()
        h_init = binary_initial(n)
        h_best, c5_best = binary_hill_climb(h_init, max_iter=30000)
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best.copy()
            best_n = n
        if c5_best <= target_bound:
            print(f"Binary early break at {c5_best} for n={n}")
            return list(best_h), best_c5, best_n
        print(f"Binary stage for n={n}, time={time.time() - t0_n:.2f}, c5={c5_best}")

    # Stage 2.3: Smooth-max refinement
    print("Stage 2.3: smooth-max refinement...")
    if best_h is not None:
        h_refined = best_h.copy()
        best_c5_ref = best_c5
        # Multiple restarts
        for r in range(8):
            y0 = logit(h_refined) + np.random.normal(0, 0.2, n_points)
            result = minimize(lambda y: smooth_max(correlation_vec(sigmoid(y)), alpha=30) + penalty_sum(sigmoid(y)), 
                             y0, method='L-BFGS-B', bounds=[(None, None)]*n_points, 
                             options={'maxiter': 300, 'ftol': 1e-12})
            h_after = sigmoid(result.x)
            h_projected = proj_to_simplex(h_after)
            c5_after = evaluate_h(h_projected)
            if c5_after < best_c5_ref:
                best_c5_ref = c5_after
                h_refined = h_projected.copy()
            if c5_after <= target_bound:
                print(f"Smooth refinement early break at {c5_after}")
                return list(h_refined), c5_after, best_n
        best_h = h_refined
        best_c5 = best_c5_ref

    # Stage 2.4: Exact polishing
    print("Stage 2.4: exact polishing (SLSQP)...")
    if best_h is not None:
        def obj_fn(h):
            return evaluate_h(h)
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h)*dx - 1}]
        bounds = [(0, 1)] * n_points
        result = minimize(obj_fn, best_h, method='SLSQP', bounds=bounds, constraints=constraints,
                         options={'ftol': 1e-12, 'maxiter': 500})
        polished_h = result.x
        polished_c5 = evaluate_h(polished_h)
        if polished_c5 < best_c5:
            best_c5 = polished_c5
            best_h = polished_h.copy()

    return list(best_h), best_c5, best_n
```
