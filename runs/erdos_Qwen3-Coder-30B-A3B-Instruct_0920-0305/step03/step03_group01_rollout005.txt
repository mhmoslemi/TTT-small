def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

            h = np.concatenate([np.repeat(u[i], block_size + (1 if i < (n_points % m) else 0)) for i in range(m)])

    def project_onto_constraint(x):

        # Special case: index out of bounds treated as 0
        subgrad = np.zeros_like(h)

                subgrad[i] += (1 - h[i + k])
            if i - k >= 0 and i - k < n_points:
                subgrad[i] -= h[i - k]
        return subgrad
