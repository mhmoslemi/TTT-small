import numpy as np
import cvxpy as cp
import time

def solve_milp_binary(n, time_limit=120):
    """Solve the binary MILP formulation for given n."""
    dx = 2.0 / n
    # Variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()

    # Constraints
    constraints = [cp.sum(h) == n // 2, t >= 0]

    # Generate all shifts d from 0 to n-1
    for d in range(n):
        # For shift d, the valid i range is 0 to n-1-d
        range_i = range(n - d)
        # Create auxiliary binary y[i,d] variables
        y = cp.Variable(len(range_i), boolean=True)
        # Connect y to h
        for i in range_i:
            constraints.append(y[i] <= h[i])
            constraints.append(y[i] <= h[i + d])
            constraints.append(y[i] >= h[i] + h[i + d] - 1)
        # q_d = sum(y[i,d])
        q_d = cp.sum(y)
        # Impose q_d >= t
        constraints.append(q_d >= t)

    # Objective
    objective = cp.Maximize(t)

    # Problem
    prob = cp.Problem(objective, constraints)

    # Solve
    try:
        prob.solve(solver=cp.CBC, verbose=False, 
                   time_limit=time_limit, 
                   maximum_seconds=time_limit)
    except Exception:
        return None, float('inf')

    if prob.status == cp.OPTIMAL or prob.status == cp.SOLVER_ERROR:
        h_val = h.value
        t_val = t.value
        if h_val is not None and t_val is not None:
            # Update constraints for future checks
            c5_bound = 1 - (2/n) * t_val
            return h_val, c5_bound
    return None, float('inf')

def hill_climb_swaps(h_current, max_swaps=100000, time_limit=10):
    """Apply hill-climbing swaps to improve the binary vector."""
    start_time = time.time()
    h = h_current.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    n = len(h)
    dx = 2.0 / n

    def compute_c5():
        # Compute correlation of h with (1 - h)
        corr = np.correlate(h, 1 - h, mode="full")
        max_correlation = np.max(corr)
        c5 = max_correlation * dx
        return c5

    best_c5 = compute_c5()
    best_h = h.copy()

    swaps = 0
    while swaps < max_swaps and (time.time() - start_time) < time_limit:
        # Pick a random one and zero
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Swap
        h[i], h[j] = h[j], h[i]
        
        c5 = compute_c5()
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        else:
            # Revert swap
            h[i], h[j] = h[j], h[i]

        swaps += 1

    return best_h, best_c5

def continuous_refine(h, n, time_limit=5):
    """Refine a binary solution to continuous space."""
    start_time = time.time()
    h = np.array(h, dtype=float)
    dx = 2.0 / n
    # All h values are initially in [0,1]
    ones = np.where(h >= 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    suggested_n = int(np.ceil(n * 1.5))  # Slightly larger grid
    h_fine = np.interp(np.linspace(0, 2, suggested_n), 
                       np.linspace(0, 2, n), h)
    h_fine = np.clip(h_fine, 0, 1)
    dx_fine = 2.0 / suggested_n

    # Refine via projected gradient descent
    # Simple projection: ensure sum = suggested_n/2
    h_fine = h_fine * (suggested_n / 2) / np.sum(h_fine)
    h_fine = np.clip(h_fine, 0, 1)

    def compute_c5_fine(h_f):
        corr = np.correlate(h_f, 1 - h_f, mode="full")
        max_correlation = np.max(corr)
        return max_correlation * dx_fine

    c5 = compute_c5_fine(h_fine)
    return h_fine, c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None

    # Candidate n values to try (even)
    candidates = [48, 56, 64, 72]
    
    for n in candidates:
        if time.time() - start_time > 0.95 * budget_s:
            break
        # Compute time slice for this model
        remaining = budget_s - (time.time() - start_time)
        T = min(120, remaining * 0.6)
        if T < 1:
            break
        
        h, c5 = solve_milp_binary(n, time_limit=T)
        if h is not None and c5 < best_c5:
            best_h = h
            best_c5 = c5
            best_n = n
        
        # Apply hill climbing
        if time.time() - start_time < 0.9 * budget_s:
            h_improved, c5_improved = hill_climb_swaps(
                h, max_swaps=200000, 
                time_limit=budget_s - (time.time() - start_time) - 5
            )
            if c5_improved < best_c5:
                best_h = h_improved
                best_c5 = c5_improved
                best_n = n
    
    # Final refinement if applicable
    if best_h is not None and (time.time() - start_time) < budget_s - 2:
        h_final, c5_final = continuous_refine(best_h, best_n, 
                                              time_limit=budget_s - (time.time() - start_time))
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    # Return as list of floats
    return list(best_h.astype(float)), best_c5, best_n
