def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h):
    # Project onto box [0,1] and sum constraint
    h = np.clip(h, 0, 1)
    dx = 2.0 / len(h)
    total = np.sum(h) * dx
    if abs(total - 1.0) > 1e-10:
        # Adjust all entries with a constant factor to make the sum equal to 1.0
        h = h * (1.0 / total)
        h = np.clip(h, 0, 1)
    return h

def coarse_optimization(n_points=200, m=10, budget_s=120):
    start_time = time.time()
    # Use CMA-ES here
    from cma import CMAEvolutionStrategy
    np.random.seed(42)
    
    def eval_func(u):
        # Expand u to full vector
        block_size = n_points // m
        h_full = np.repeat(u, block_size)
        if len(h_full) < n_points:
            h_full = np.pad(h_full, (0, n_points - len(h_full)), 'edge')
        # Ensure sum = 1.0
        h_full = project_to_feasible(h_full)
        return evaluate_c5(h_full)
        
    bounds = [[0, 1] for _ in range(m)]
    strategy = CMAEvolutionStrategy(x0=[0.5]*m, sigma0=0.2, 
                                   inopts={'popsize': 4+m, 'bounds': bounds})
    
    best_u = None
    best_value = float('inf')
    
    while not strategy.stop() and (time.time() - start_time) < budget_s:
        X = strategy.ask()
        f = []
        for u in X:
            # Add constraint that last element equals what's needed
            u_copy = u.copy()
            if len(u_copy) > 1:
                u_copy[-1] = max(0, min(1, m/2. - sum(u_copy[:-1])))
            f.append(eval_func(u_copy))
        strategy.tell(X, f)
        best_val = min(f)
        if best_val < best_value:
            best_value = best_val
            best_idx = np.argmin(f)
            best_u = X[best_idx]
    
    # Expand best to full vector
    if best_u is None:
        h_full = np.ones(n_points) * 0.5
    else:
        block_size = n_points // m
        h_full = np.repeat(best_u, block_size)
        if len(h_full) < n_points:
            h_full = np.pad(h_full, (0, n_points - len(h_full)), 'edge')
        h_full = project_to_feasible(h_full)
    
    return h_full

def projected_gradient_descent(h_init, budget_s=240):
    start_time = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    
    # Max iterations
    max_iter = 5000
    alpha0 = 0.01
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for t in range(max_iter):
        if (time.time() - start_time) >= budget_s:
            break
            
        # Find worst shift
        corr = correlate(h, 1 - h, mode="full")
        s = np.argmax(corr) - (n - 1)  # Convert back to shift index
        
        # Compute gradient
        g = np.zeros(n)
        for i in range(n):
            if i+s >= 0 and i+s < n:
                g[i] = 1 - h[i+s] - h[i-s]
            else:
                g[i] = 1 - h[i-s] if i-s >= 0 and i-s < n else 1 - h[i+s] if i+s >= 0 and i+s < n else 0
            
        # Gradient step with decaying learning rate
        alpha = alpha0 / (1.0 + math.sqrt(t))
        h_new = h - alpha * g
        h_new = project_to_feasible(h_new)
        
        # Check if new is better
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        h = h_new
        
    return best_h, best_c5

def binarize_and_improve(h_continuous, budget_s=180):
    start_time = time.time()
    n = len(h_continuous)
    dx = 2.0 / n
    best_c5 = float('inf')
    best_h = None
    
    # Generate a bunch of random binary vectors
    K = 30
    for _ in range(K):
        if (time.time() - start_time) >= budget_s:
            break
            
        b = np.zeros(n, dtype=int)
        # Sampling approach: sort indices by h value in decreasing order, take top n/2
        idxs = list(range(n))
        scores = list(enumerate(h_continuous))
        scores.sort(key=lambda x: x[1], reverse=True)
        top_idxs = [i for i, _ in scores[:n//2]]
        b[top_idxs] = 1
            
        # Project to have exact weight n/2
        count = np.sum(b)
        if count > n//2:
            idxs_to_zero = np.random.choice(np.where(b == 1)[0], count - n//2, replace=False)
            b[idxs_to_zero] = 0
        elif count < n//2:
            idxs_to_one = np.random.choice(np.where(b == 0)[0], n//2 - count, replace=False)
            b[idxs_to_one] = 1
    
        # Evaluate and keep best
        c5_val = evaluate_c5(b)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = b.copy()
    
    # If got some binarized vector, refine via swaps
    if best_h is not None:
        # Precompute contributions for efficient updates
        best_h = best_h.astype(float)
        # Start binary local search
        n_iter = 0
        while (time.time() - start_time) < budget_s and n_iter < 10:
            improved = False
            # Try swaps
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
            swap_list = list(np.ndindex(len(ones), len(zeros)))
            # Sample subset for efficiency  
            if len(swap_list) > 1000:
                swap_list = np.random.choice(swap_list, size=1000, replace=False).tolist()
                
            for i, j in swap_list [:]:
                i_idx = ones[i]
                j_idx = zeros[j]
                if i_idx != j_idx:
                    new_h = best_h.copy()
                    new_h[i_idx] = 0
                    new_h[j_idx] = 1
                    
                    c5_new = evaluate_c5(new_h)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = new_h.copy()
                        improved = True
            if not improved:
                break
            n_iter += 1
    
    return best_h, best_c5
