import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp

def weight_normalise(x, n):
    """Rescales a vector to satisfy sum(x) = n/2"""
    return x * (n / 2) / np.sum(x)

def max_defect(h):
    """Compute the maximum overlap integral C5 and shift"""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    max_shift = np.argmax(corr) - (len(h) - 1)
    return max_corr, max_shift

def softmax_objective(h, alpha):
    """Compute smooth max surrogate and its gradient"""
    n = len(h)
    # Compute all correlation terms with shift
    corr = correlate(h, 1 - h, mode='full')
    corr = corr[len(h)-1:]  # Only positive shifts
    exp_corr = np.exp(alpha * corr)
    exp_sum = np.sum(exp_corr)
    if exp_sum == 0:
        # Should not happen but be safe
        grad = np.zeros_like(h)
        return np.inf, grad

    w = exp_corr / exp_sum  # Soft weights
    # Compute the smooth max
    phi = np.log(exp_sum) / alpha
    
    # Compute gradient
    grad = np.zeros_like(h)
    for s in range(1, n):
        if s >= n:
            break
        # Compute derivative for shift s
        for i in range(n):
            if i + s < n:
                # Both terms are present
                dfi = 1.0 - h[i + s]
                dfj = -h[i]
                # Contribution to gradient
                grad[i] += w[s-1] * dfi
                if i + s < n:
                    grad[i + s] += w[s-1] * dfj
    return phi, grad

def build_milp(n):
    """Build the MILP problem for binary h"""
    # Solve a simpler version: let h be binary, maximize minimum overlap
    
    # Variables: h[i] binary and w[i,s] auxiliary
    h = cp.Variable(n, boolean=True)
    w = cp.Variable((n, n-1), nonneg=True)
    
    # Constraint: sum(h) = n/2
    weight_cons = cp.Equality(cp.sum(h), n / 2)
    
    # Linearization for w[i,s] = h[i] * h[i+s]
    constraints = [weight_cons]
    for s in range(1, n):
        for i in range(n-s):
            # w[i,s] <= h[i]
            constraints.append(w[i, s-1] <= h[i])
            # w[i,s] <= h[i+s]  
            constraints.append(w[i, s-1] <= h[i+s])
            # w[i,s] >= h[i] + h[i+s] - 1
            constraints.append(w[i, s-1] >= h[i] + h[i+s] - 1)
    
    # Min overlap constraint
    t = cp.Variable()
    for s in range(1, n):
        constraints.append(cp.sum(w[:, s-1]) >= t)
    
    # Objective max t
    prob = cp.Problem(cp.Maximize(t), constraints)
    
    # Set solver parameters
    try:
        prob.solve(solver=cp.CBC, verbose=False, 
                   time_limit=180, ratioGap=0.0)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value
        else:
            return None
    except Exception as e:
        return None

def refine_continuous(h0, n, max_iter=200):
    """Run the smooth max refinement with increasing alpha"""
    h_new = h0.copy()
    # Fix values to make a valid distribution
    h_new = weight_normalise(h_new, n)
    
    # Fix interior 1s if they are too far from boundaries
    # Place constraints to make blocks more regular  
    h_cont = h_new.copy()
    
    # Smooth max stages
    for alpha in [5, 15, 40]:
        # Define objective with gradient
        def obj_grad(x):
            value, grad = softmax_objective(x, alpha)
            return value, grad
        
        # Use trust-constr with analytic gradient
        res = minimize(obj_grad, h_cont, method='trust-constr',
                       jac=True,
                       bounds=[(0, 1) for _ in range(n)],
                       options={'maxiter': max_iter, 'gtol': 1e-6,
                               'xtol': 1e-8},
                       constraints=({'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}))
        if res success:
            h_cont = res.x.copy()
            # Ensure sum constraint strictly respected
            h_cont = weight_normalise(h_cont, n)
        else:
            # Fallback to original if not successful
            pass
        h_cont = np.clip(h_cont, 0, 1)
        h_cont = weight_normalise(h_cont, n)
        
    return h_cont

def anneal(h_orig, n, max_swaps=5000, cooling_rate=0.99, init_gamma=0.2, max_time=150):
    """Swap-based sim anneal to reduce residual max defect"""
    start_time = time.time()
    h = h_orig.copy()
    h = weight_normalise(h, n)
    c_orig, _ = max_defect(h)
    gamma = init_gamma
    
    # Precompute correlation vector
    c_full = correlate(h, 1 - h, mode='full')
    c_positive = c_full[len(h)-1:]
    
    # First compute max shift
    max_shift = np.argmax(c_positive)
    max_val = np.max(c_positive)
    
    for rep in range(max_swaps):
        if time.time() - start_time > max_time:
            break
            
        # Identify indices that cause max shift contribution
        shift_val = c_positive[max_shift]
        # Find i such that h[i] > 0 and h[i+shift] is small
        # This kind of identifies likely locations to swap
        i_cause = []
        for i in range(n):
            if i + max_shift < n:
                if h[i] > 0 and h[i + max_shift] < 0.3:
                    i_cause.append(i)
        
        if not i_cause:
            continue
            
        # Select random i
        i = np.random.choice(i_cause)
        j_candidates = []
        for j in range(n):
            if j + max_shift < n:
                if h[j] < 0.3 and h[j + max_shift] > 0.5:
                    j_candidates.append(j)
        
        if not j_candidates:
            continue
            
        j = np.random.choice(j_candidates)
        
        # Adjust j according to gamma (swap mass)
        delta = min(h[i], 1 - h[j]) * gamma
        if delta <= 1e-8:
            continue
            
        # Perform shift
        h[i] -= delta
        h[j] += delta
        h[i + max_shift] += delta
        h[j + max_shift] -= delta
        
        # Renormalize
        h = weight_normalise(h, n)
        
        # Update correlation efficiently
        corr = correlate(h, 1 - h, mode='full')
        c_positive = corr[len(h)-1:]
        new_max = np.max(c_positive)
        if new_max <= max_val:
            # Accept
            max_val = new_max
            max_shift = np.argmax(c_positive)
            # Cool down
            gamma *= cooling_rate
        else:
            # Revert
            h[i] += delta
            h[j] -= delta
            h[i + max_shift] -= delta
            h[j + max_shift] += delta
            # Don't cool, might be local minima
            # gamma = min(0.99 * gamma, 0.01)

    # At end, recompute final value
    final_c5, final_shift = max_defect(h)
    return h, final_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Decide n based on budget
    n_points = 100  # Starting with better resolution
    
    # Phase 0: Initialize and empty buffer
    best_h = None
    best_c5 = float('inf')
    
    # Phase 1: Solve MILP for binary pattern
    # Try to solve binary MILP with CBC
    milp_start = time.time()
    binary_solution = build_milp(n_points)
    
    # Check if solution exists and is valid
    if binary_solution is not None:
        h0 = binary_solution
        h0 = weight_normalise(h0, n_points)
        # Just use as warm start in continuous phase
    else:
        # Fallback to random sample
        h0 = np.random.rand(n_points)
        h0 = weight_normalise(h0, n_points)
    
    # Phase 2: Refine continuously
    h_refined = refine_continuous(h0, n_points)
    c5_refined = max_defect(h_refined)[0]
    
    # Phase 3: Swap aneal
    h_final, final_c5 = anneal(h_refined, n_points, max_time=150)
    c5_final = max_defect(h_final)[0]  # Exact evaluation
    
    # Final check
    # potentially improve with full C5 eval here too
    if c5_final < best_c5:
        best_h = h_final
        best_c5 = c5_final
    
    return (best_h.tolist(), best_c5, n_points)
