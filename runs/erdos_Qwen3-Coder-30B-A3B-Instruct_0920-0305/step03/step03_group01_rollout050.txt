import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_onto_box_and_hyperplane(y, target):
    """Project vector y onto box [0,1] and hyperplane sum = target"""
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0, 1)

def gradient_update(h, dx):
    """Compute gradient w.r.t. to max overlap at current h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    max_ind = np.where(corr == max_corr)[0]
    # Use first shift achieving max
    shift = max_ind[0] - (len(h) - 1)
    n = len(h)
    if shift >= 0:
        term1 = np.empty(n)
        term1[:n-shift] = 1 - h[shift:]
        term1[n-shift:] = 1
        term2 = np.empty(n)
        term2[shift:] = h[:n-shift]
        term2[:shift] = 0
    else:
        s = -shift
        term1 = np.empty(n)
        term1[s:] = 1 - h[:n-s]
        term1[:s] = 1
        term2 = np.empty(n)
        term2[:n-s] = h[s:]
        term2[n-s:] = 0
    grad = term1 - term2
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Initial shape: tent function
    x = np.linspace(0, 2, n_points)
    h0 = np.maximum(0, 1 - np.abs(x - 1) * 2)
    # Normalize to sum = n/2
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Store best solution found so far
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Project onto feasible set: ensure 0 <= h <= 1 and sum = n/2
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return project_onto_box_and_hyperplane(h_vec, n_points / 2)
    
    # Gradient descent parameters
    step_size = 0.01
    max_iter = 5000
    patience = 50
    patience_counter = 0
    prev_c5 = best_c5 + 1e-6
    
    # Main gradient descent loop
    for i in range(max_iter):
        if (time.time() - start_time) > (budget_s * 0.98):
            break
            
        # Evaluate gradient of max overlap
        grad = gradient_update(best_h, dx)
        
        # Take gradient step to reduce overlap
        new_h = best_h - step_size * grad
        new_h = project_h(new_h)
        
        # Compute new C5 value
        new_c5 = evaluate_c5(new_h)
        
        # Update best solution if improvement
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Adaptive step size: reduce if no improvement for some iterations
        if patience_counter > patience:
            step_size *= 0.95
            patience_counter = 0
        
        # Early stopping on small changes
        if abs(new_c5 - prev_c5) < 1e-10:
            break
        prev_c5 = new_c5
    
    # Additional random restarts to improve solution
    for restart in range(100):
        if (time.time() - start_time) > (budget_s * 0.95):
            break
            
        # Random initial guess
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = project_h(h_rand)
        
        # Local optimization from this point
        for i in range(200):
            if (time.time() - start_time) > (budget_s * 0.95):
                break
            grad = gradient_update(h_rand, dx)
            h_rand = h_rand - step_size * grad
            h_rand = project_h(h_rand)
            new_c5 = evaluate_c5(h_rand)
            if new_c5 < best_c5:
                best_h = h_rand.copy()
                best_c5 = new_c5
        
    return (best_h.tolist(), best_c5, n_points)
