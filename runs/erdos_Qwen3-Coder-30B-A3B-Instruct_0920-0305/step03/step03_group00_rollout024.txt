import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def weight_normalise(x, n):
    """Rescales x to have sum = n/2"""
    return x * (n / 2) / np.sum(x)

def max_defect(h):
    """Compute exact C5 and return the shift that achieves it"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    max_shift = np.argmax(corr)
    return max_corr, max_shift

def softmax_objective(h, alpha):
    """Return smooth max and gradient"""
    n = len(h)
    # Compute all f_s(h) = sum h[i] * (1 - h[i+s]) over all valid s
    corr = correlate(h, 1 - h, mode="full")
    # Extract valid correlations (shifts 1 to n-1)  
    f_vals = corr[n-1:]  # Only the full length part starting from shift 1

    # Softmax approximation
    exp_f = np.exp(alpha * f_vals)
    Z = np.sum(exp_f)
    Phi = np.log(Z) / alpha
    
    # Gradient computation 
    w = exp_f / Z
    grad = np.zeros_like(h)
    for s in range(1, n):
        idxs = np.arange(0, n - s)
        idxs_s = np.arange(s, n) 
        # Contribution from term s
        grad[idxs] += w[s-1] * (1 - h[idxs_s])
        grad[idxs_s] -= w[s-1] * h[idxs] 

    return Phi, grad

def build_milp(n):
    """Build and solve binary MILP to estimate binary C5"""
    # Variables
    h = cp.Variable(n, boolean=True)
    w = cp.Variable((n, n-1), nonneg=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.sum(h) == n / 2
    ]
    
    # AND constraints (w[i,s] = h[i] * h[i+s])  
    for s in range(1, n):
        for i in range(n - s):
            constraints.extend([
                w[i,s-1] <= h[i],
                w[i,s-1] <= h[i+s],
                w[i,s-1] >= h[i] + h[i+s] - 1
            ])
            
    # Min overlap constraint
    for s in range(1, n):
        constraints.append(cp.sum(w[:, s-1]) >= t)
    
    # Objective
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)

    # Solve
    try:
        # Allow up to 180 seconds
        prob.solve(solver=CPLEX, time_limit=180, verbose=False)
        return h.value
    except:
        return None

def refine_continuous(h0, n, max_iter=200):
    """Refine continuous from binary solution with soft max"""
    h = h0.copy().astype(float)
    
    # Fix interior 1s, only optimize edges (L-wider boundary) 
    # detect blocks
    h_rounded = np.round(h)
    block_start = np.concatenate(([0], np.where(np.diff(h_rounded) != 0)[0] + 1, [n]))
    edges = []
    for i in range(len(block_start)-1):
        start, end = block_start[i], block_start[i+1]
        if end - start > 1:
            # potentially two edges to approximate
            if h_rounded[start] == 1:
                edges.append((start, start+2))  # right edge
            if h_rounded[end-1] == 1: 
                edges.append((end-3, end-1))    # left edge
    
    # Define free indices (some nearby for any block boundary)
    free_indices = set()
    L = 2
    for start, end in edges:
        for i in range(max(0, start-L), min(n, start+L)):
            free_indices.add(i)
        for i in range(max(0, end-L), min(n, end+L)):
            free_indices.add(i)
    free_indices = sorted(list(free_indices))
    
    # print(f"refine_continuous ({len(free_indices)} free vars)")
    
    def obj_grad(h_free):
        # Reshape h through interpolation
        h_full = h0.copy()
        h_full[free_indices] = h_free
        
        # Re-normalize (enforce linear constraint exactly)  
        scale = (n/2) / np.sum(h_full)
        h_full = h_full * scale
        
        # Re-normalize free part to keep constraints satisfied
        h_full[free_indices] = h_free
        h_full = weight_normalise(h_full, n)
        
        phi, grad = softmax_objective(h_full, 40)
        dphi_dh_free = grad[free_indices]
        return phi, dphi_dh_free
    
    h_free = h0[free_indices].copy()
    
    # Optimize with trust-constr with analytic gradients
    res = minimize(obj_grad, h_free, method='trust-constr', 
                   jac=True, options={'maxiter': max_iter, 'gtol': 1e-6})
    
    if res.success:
        h_final = h0.copy()
        h_final[free_indices] = res.x
        h_final = weight_normalise(h_final, n)
        return h_final
    else:
        return h0

def swap_anneal(h, n):
    """Swap-style annealing to improve solution"""  
    max_iter = 5000
    h = h.copy()
    c5, s_star = max_defect(h)
    gamma = 0.2  # temperature-like parameter
    i_try = 0
    
    # Precompute correlation matrix
    corr_h = correlate(h, 1 - h, 'full')
    
    while i_try < max_iter:
        # Keep a repaired copy
        h_try = h.copy() 
        c5_prev = c5
        
        # Find indices which are "critical" for s_star
        critical_indices = []
        for i in range(n):
            if h[i] > 0 and (i + s_star < n) and h[i+s_star] < 0.5:
                critical_indices.append(i)
                
        if not critical_indices:
            break
            
        # Pick some random moves
        i = np.random.choice(critical_indices)
        j_list = []
        for j in range(n):
            if h[j] < 0.5 and (j + s_star < n) and h[j+s_star] > 0.5:
                j_list.append(j)
        if not j_list:
            j = i + s_star // 2  # fallback
        else:
            j = np.random.choice(j_list)
        
        # Apply small exchange
        delta = min(h[i], 1 - h[j]) * gamma
        h_try[i] -= delta
        h_try[j] += delta
        h_try[i+s_star] += delta
        h_try[j+s_star] -= delta
        
        # Renormalize (maintain integral constraint)
        h_try = weight_normalise(h_try, n)
        
        # Compute new values
        corr_new = correlate(h_try, 1 - h_try, 'full')
        c5_new = np.max(corr_new)
        
        # Always accept if we don't make it worse
        if c5_new <= c5 + 1e-10:
            h = h_try
            c5 = c5_new
        else:
            # Only decrease temperature gradually
            gamma *= 0.99
        
        i_try += 1
            
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    n = 100
    
    # Check if we have the exact initial solution
    try:
        x = np.array(initial_h_values)
        if len(x) == 100:
            # Use this direct solution
            h = x.copy()
            h = weight_normalise(h, n)
        else:
            # Fall back to construction via earlier attempt
            h = np.zeros(n)
            n_half = n // 2
            h[:n_half] = 1.0
            h = weight_normalise(h, n)
    except:
        # Fall back to pure random
        h = np.random.uniform(0, 1, size=n)
        h = weight_normalise(h, n)
        
    best_h = h.copy()
    best_c5, _ = max_defect(best_h)
    
    #====== Stage 1: Binary MILP ======#
    if (time.time() - start_time) < (budget_s * 0.7):
        try:
            h_bin = build_milp(n)
            if h_bin is not None:
                h_bin = np.round(h_bin).astype(int)
                h_init = weight_normalise(h_bin, n)
                c5_bin, _ = max_defect(h_init)
                if c5_bin < best_c5:
                    best_h = h_init.copy()
                    best_c5 = c5_bin
                # Refine the binary solution
                h_cont = refine_continuous(h_init, n)
                c5_cont, _ = max_defect(h_cont)
                if c5_cont < best_c5:
                    best_h = h_cont.copy()
                    best_c5 = c5_cont
        except:
            pass
    
    #====== Stage 2: Continuous refinement ======#
    if (time.time() - start_time) < (budget_s * 0.85):
        h_refined = refine_continuous(best_h, n, max_iter=200)
        c5_refined, _ = max_defect(h_refined)
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    #====== Stage 3: Swap annealing post-processing ======#
    if (time.time() - start_time) < (budget_s - 50):
        h_anneal = swap_anneal(best_h, n)
        c5_anneal, _ = max_defect(h_anneal)
        if c5_anneal < best_c5:
            best_h = h_anneal.copy()
            best_c5 = c5_anneal
            
    # Output final bound
    final_c5 = best_c5
    return (best_h.tolist(), final_c5, n)
