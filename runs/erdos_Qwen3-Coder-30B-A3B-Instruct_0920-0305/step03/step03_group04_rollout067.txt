import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time
import cvxpy as cp
from cvxpy import SCS
import random


# Constants
n_values = [36, 42, 48, 54, 60]
dx = 2.0 / 48  # default
target_bound = 0.3807
R = 30  # rounding rounds
max_swaps = 50000
max_iter_smooth = 300
max_iter_slsqp = 500
MAX_ITERS_SDP = 8000
TIME_PER_SDP = 30


def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5


def correlate_vec(h):
    """Return the full correlation vector, length 2n-1."""
    corr = np.correlate(h, 1 - h, mode="full")
    return corr


def logsumexp(x, axis=None, keepdims=False):
    """Numerically stable logsumexp."""
    x_max = np.max(x, axis=axis, keepdims=True)
    return np.log(np.sum(np.exp(x - x_max), axis=axis, keepdims=keepdims)) + x_max


def smooth_max(c, alpha=30):
    """Soft maximum of correlation vector."""
    return logsumexp(alpha * c) / alpha


def proj_to_simplex(v, s=0.5):
    """Project a vector `v` onto the simplex with sum=s."""
    v = np.array(v)
    v = np.clip(v, 0, 1)
    v_sum = np.sum(v)
    if abs(v_sum - s) < 1e-6:
        return v
    # bistable projection method
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u > cssv / np.arange(1, len(u) + 1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w


def sigmoid(x):
    return 1. / (1. + np.exp(-x))


def logit(x):
    x = np.clip(x, 1e-6, 1 - 1e-6)
    return np.log(x / (1. - x))


def sdp_formulation(n, dx):
    """Formulate and solve the SDP relaxation for n points."""
    x = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    constraints = [
        0 <= x,
        x <= 1,
        cp.sum(x) == n / 2,
        cp.diag(X) == x,
    ]

    for i in range(n):
        for j in range(i+1, n):
            constraints += [
                X[i, j] >= 0,
                X[i, j] <= x[i],
                X[i, j] <= x[j],
                X[i, j] >= x[i] + x[j] - 1
            ]

    for k in range(n):
        shifted_corr = cp.sum(x[:n-k]) - cp.sum(cp.diag(X, k))
        constraints += [shifted_corr <= t]

    constraints += [X >> 0]

    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=SCS, max_iters=MAX_ITERS_SDP, eps=1e-5)
        return prob, x.value, X.value
    except Exception as e:
        return prob, None, None


def random_rounding(X_sol, x_sol, n, dy):
    """Perform random hyperplane rounding and return binary vector with exact sum."""
    # Use eigen-decomposition to get low-rank approx
    try:
        const = 10  # look at first few components
        U, Vt = np.linalg.eigh(X_sol)
        V = U[:, -const:] * np.sqrt(Vt[-const:])
        V = V[:, ::-1]  # largest to smallest
        
        best_h = None
        best_c5 = float('inf')
        
        for _ in range(R):
            g = np.random.randn(V.shape[1])
            signs = np.sign(V @ g)
            h_raw = (signs >= 0).astype(float)
            
            # Adjust to have exactly n/2 ones
            cur_sum = np.sum(h_raw)
            delta = int(cur_sum - n/2)
            if delta > 0:
                # too many ones, flip the ones with smallest |inner_product|
                indices = np.argsort(np.abs(V @ g))[:delta]
                h_raw[indices] = 0
            elif delta < 0:
                # not enough ones, flip the zeros with smallest |inner_product|
                indices = np.argsort(np.abs(V @ g))[delta:]
                h_raw[indices] = 1
            
            if np.sum(h_raw) != n/2:
                print("After rounding, sum is not n/2:", np.sum(h_raw), n/2)
                continue
            
            c5 = evaluate_h(h_raw)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_raw.copy()
        
        return best_h, best_c5
    except Exception as e:
        print("Error in random rounding:", e)
        return None, float('inf')


def hill_climb_binary(h_initial, n):
    """Run massive binary hill climb starting from h_initial."""
    h = np.array(h_initial.copy())
    n_points = len(h)
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    
    c5_best = evaluate_h(h)
    best_h = h.copy()
    iter_no_improve = 0
    max_improvement_iter = 1000
    max_swaps = 50000
    
    for _ in range(max_swaps):
        if iter_no_improve >= max_improvement_iter:
            # Attempt deterministic improvement
            c = correlate_vec(h)
            max_k = np.argmax(c)
            if max_k < len(c):
                # Identify mismatching pairs where h[i]=1 but h[i+k]=0
                mismatch_indices = []
                for i in range(n - max_k):
                    if h[i] > 0.5 and h[i + max_k] < 0.5:
                        mismatch_indices.append(i)
                if mismatch_indices:
                    # Try swapping a mismatching '1' with a '0'
                    for i in mismatch_indices:
                        for j in zeros:
                            h_test = h.copy()
                            h_test[i] = 0
                            h_test[j] = 1
                            c5_test = evaluate_h(h_test)
                            if c5_test < c5_best:
                                h = h_test
                                c5_best = c5_test
                                # update 'ones'
                                ones = list(np.where(h > 0.5)[0])
                                zeros = list(np.where(h < 0.5)[0])
                                iter_no_improve = 0
                                break
                    else:
                        # Random restart
                        h = np.random.rand(n) > 0.5
                        ones = list(np.where(h > 0.5)[0])
                        zeros = list(np.where(h < 0.5)[0])
                        c5_best = evaluate_h(h)
                        iter_no_improve = 0
                else:
                    iter_no_improve = 0
            else:
                iter_no_improve = 0
                
        if iter_no_improve < max_improvement_iter:
            i = random.choice(ones)
            j = random.choice(zeros)
            h_trial = h.copy()
            h_trial[i] = 0
            h_trial[j] = 1
            c5_new = evaluate_h(h_trial)
            if c5_new < c5_best:
                c5_best = c5_new
                h = h_trial
                ones = list(np.where(h > 0.5)[0])
                zeros = list(np.where(h < 0.5)[0])
                iter_no_improve = 0
            else:
                iter_no_improve += 1
        else:
            # Restart if too many no-improvements
            h = np.random.rand(n) > 0.5
            ones = list(np.where(h > 0.5)[0])
            zeros = list(np.where(h < 0.5)[0])
            c5_best = evaluate_h(h)
            iter_no_improve = 0
            
        if c5_best <= target_bound:
            break
            
    return h, c5_best


def refine_with_smooth_max(h_start, n, penalty=10.0, alpha=30):
    """Refine the binary/fractional h_start with smooth surrogate."""
    h = np.clip(h_start, 1e-5, 1 - 1e-5)
    y = logit(h)
    dx = 2.0 / n
    
    def fun(y_vec):
        h_test = sigmoid(y_vec)
        c = correlate_vec(h_test)
        soft_max = smooth_max(c, alpha=alpha)
        penalty_term = penalty * (np.sum(h_test) - n/2)**2
        return soft_max + penalty_term
    
    def grad(y_vec):
        h_test = sigmoid(y_vec)
        c = correlate_vec(h_test)
        c = np.array(c)
        exp_c = np.exp(alpha * c)
        s = np.sum(exp_c)
        dsoft = (alpha * np.sum(c * exp_c)) / (s * alpha)
        
        dpenalty = penalty * 2 * (np.sum(h_test) - n/2)
        grad_h = dsoft * alpha * (h_test * (1 - h_test)) + dpenalty
        
        return grad_h * (h_test * (1 - h_test))
    
    bounds = [(-20, 20)] * n
    result = minimize(fun, y, method='L-BFGS-B', bounds=bounds, jac=grad, 
                      options={'maxiter': max_iter_smooth})
    h_opt = sigmoid(result.x)
    h_opt = proj_to_simplex(h_opt, n/2)
    return h_opt


def run(seed=42, budget_s=600):
    np.random.seed(seed)
    t0 = time.time()
    
    best_c5 = float("inf")
    best_h = None
    best_n = None
    
    # Loop over n values
    for n in n_values:
        dx = 2.0 / n
        if time.time() - t0 > budget_s * 0.9:
            break
        try:
            print(f"Solving SDP for n={n}")
            prob, x_sol, X_sol = sdp_formulation(n, dx)
            if x_sol is None:
                print("SDP failed for n =", n)
                continue
                
            print("Rounding SDP solution for n =", n)
            round_h, round_c5 = random_rounding(X_sol, x_sol, n, dx)
            if round_h is None:
                print("Rounding failed for n =", n)
                continue
                
            print(f"Rounded C5 for n={n}: {round_c5}")
            if round_c5 < best_c5:
                best_c5 = round_c5
                best_h = round_h.copy()
                best_n = n
                
            # Refine the rounded binary vector with smooth function
            print("Refining with smooth max for n =", n)
            refined_h = refine_with_smooth_max(round_h, n)
            refined_c5 = evaluate_h(refined_h)
            print(f"Refined C5 for n={n}: {refined_c5}")
            
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h.copy()
                best_n = n

            # Hill climb on that binary vector
            print("Hill climbing on binary vector for n =", n)
            hc_h, hc_c5 = hill_climb_binary(round_h, n)
            print(f"Hill-climbed C5 for n={n}: {hc_c5}")
            
            if hc_c5 < best_c5:
                best_c5 = hc_c5
                best_h = hc_h.copy()
                best_n = n
                
            # Now improve the refined with hill climb
            if refined_c5 < hc_c5:
                hc_h2, hc_c52 = hill_climb_binary(refined_h, n)
                print(f"Secondary H hill-climbed C5 for n={n}: {hc_c52}")
                if hc_c52 < best_c5:
                    best_c5 = hc_c52
                    best_h = hc_h2.copy()
                    best_n = n
                    
            if best_c5 <= target_bound:
                print("Early exit at target bound reached.")
                break

        except Exception as e:
            print("Exception in loop:", e)
            continue
            
    if best_h is None:
        print("No valid construction found, using simple approach.")
        h = np.full(48, 0.5)
        best_h = h.copy()
        best_c5 = evaluate_h(best_h)
        best_n = 48
        
    # Final polishing with SLSQP
    print("Final polishing with SLSQP.")
    
    def obj(h_v):
        return evaluate_h(h_v)
    
    bounds = [(0, 1)] * best_n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - best_n / 2.0}
    
    try:
        result = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                         constraints=cons,options={'maxiter': max_iter_slsqp,
                                                'ftol': 1e-12, 'gtol': 1e-10})
        if result.success:
            best_h = result.x
        final_c5 = evaluate_h(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
    except Exception as e:
        print("SLSQP failed:", e)
        
    return list(best_h), best_c5, best_n
