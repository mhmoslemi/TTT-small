import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.linalg import eigh
from scipy.stats import norm

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define problem size
    n = 100
    dx = 2.0 / n

    # Phase 1: SDP relaxation
    best_h_sdp = None
    best_c5_sdp = float('inf')
    try:
        # Create CVXPY variables
        h = cp.Variable(n)
        X = cp.Variable((n, n), symmetric=True)
        t = cp.Variable()

        # Constraints for SDP relaxation
        constraints = [
            cp.diag(X) == h,
            cp.sum(h) == n/2,
            0 <= h,
            h <= 1,
            X >> 0,  # Positive semidefinite
        ]

        # Add McCormick envelopes
        for i in range(n):
            for j in range(i+1, n):
                if i + 1 <= j:
                    constraints.extend([
                        X[i,j] <= h[i],
                        X[i,j] <= h[j],
                        X[i,j] >= h[i] + h[j] - 1,
                    ])

        # Add overlap constraints per shift
        for s in range(1, n):
            o_s = cp.sum([X[i, i+s] for i in range(n-s)])
            constraints.append(o_s >= t)

        # Maximize t
        objective = cp.Maximize(t)
        prob = cp.Problem(objective, constraints)

        prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-5, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sdp = h.value
            # Normalize h to satisfy integral constraint
            scale = (n/2) / cp.sum(h).value
            h_sdp = h_sdp * scale
            c5_sdp = evaluate_c5(h_sdp)
            if c5_sdp < best_c5_sdp:
                best_h_sdp = h_sdp
                best_c5_sdp = c5_sdp
    except:
        pass

    # Fallback to a simpler approach if SDP fails
    if best_c5_sdp == float('inf'):
        try:
            # Random restart smooth max optimization
            best_h, best_c5 = None, float('inf')
            for _ in range(30):
                h0 = np.random.rand(n)
                h0 = h0 / np.sum(h0) * (n/2)
                # Softmax approximation of the max overlap
                def soft_max_overlap(h):
                    dx = 2.0 / n
                    corr = correlate(h, 1-h, mode="full")
                    return -np.max(corr * dx)

                res = minimize(soft_max_overlap, h0, method='L-BFGS-B', bounds=[(0,1) for _ in range(n)])
                if res.success:
                    h_candidate = res.x
                    scale = (n/2) / np.sum(h_candidate)
                    h_candidate = h_candidate * scale
                    c5_candidate = evaluate_c5(h_candidate)
                    if c5_candidate < best_c5:
                        best_c5 = c5_candidate
                        best_h = h_candidate
            if best_h is not None:
                best_h_sdp = best_h
                best_c5_sdp = best_c5
        except:
            pass
    
    # Proceed with rounding
    candidates = []
    if best_h_sdp is not None:
        candidates.append(best_h_sdp.copy())
            
        # Randomized rounding from SDP solution
        try:
            X_sdp = np.outer(best_h_sdp, best_h_sdp)
            # For better results, use the SDP solution's leading components
            # But given we only have a heuristics, just work with X directly
            eigenvals, eigenvecs = eigh(X_sdp)
            if eigenvals[-1] > 1e-8:
                r = min(20, len(eigenvals))
                Y = eigenvecs[:, -r:] * np.sqrt(eigenvals[-r:])
                for _ in range(20):
                    if (time.time() - start_time) > budget_s * 0.75:
                        break
                    g = np.random.normal(0, 1, r)
                    y = Y @ g
                    # Sort to determine top n/2 indices
                    sorted_indices = np.argsort(-y)
                    h_candidate = np.zeros(n)
                    h_candidate[sorted_indices[:n//2]] = 1.0
                    # Normalize to desired sum
                    scale = (n/2) / np.sum(h_candidate)
                    h_candidate = h_candidate * scale
                    c5_candidate = evaluate_c5(h_candidate)
                    candidates.append(h_candidate)
        except:
            pass
            
    # Local refinement using trust-constr
    best_final_h, best_final_c5 = None, float('inf')
    
    # Define smooth max
    def smooth_max(h, alpha):
        dx = 2.0 / n
        corr = correlate(h, 1 - h, mode="full")
        corr = corr * dx
        return -np.log(np.sum(np.exp(alpha * corr))) / alpha

    # Gradient of smooth max
    def grad_smooth_max(h, alpha):
        dx = 2.0 / n
        corr = correlate(h, 1 - h, mode="full")
        corr = corr * dx
        exp_corr = np.exp(alpha * corr)
        Z = np.sum(exp_corr)
        weights = exp_corr / Z
        # Compute gradient using correlation structure
        grad = np.zeros_like(h)
        for i in range(n):
            for k in range(n):
                idx = k - i
                if idx >= 0 and idx < len(corr):
                    grad[i] += weights[idx] * (1.0 - (h[i + idx] if i + idx < n else 0.0))  # Syntax error fixed
        return grad  # This is a rough approximation: real formula involves more complex derivative chaining

    for hinit in candidates:
        if (time.time() - start_time) > budget_s * 0.9:
            break

        h_refined = hinit.copy()
        # Gradient descent steps with increasing alpha
        for alpha in [5, 15, 40]:
            if (time.time() - start_time) > budget_s * 0.95:
                break

            def objective(x):
                return smooth_max(x, alpha)
            def gradient(x):
                return grad_smooth_max(x, alpha)
            # Simulate constrained optimization with trust-constr
            bounds = [(0, 1) for _ in range(n)]
            # Create a simple closure for such access
            def smooth_max_reduced(x):
                return smooth_max(x, alpha)
                
            res = minimize(smooth_max_reduced, h_refined, method='trust-constr',
                           bounds=bounds, 
                           options={'maxiter': 200, 'gtol': 1e-8})
            if res.success:
                h_refined = res.x
                # Normalize
                scale = (n/2) / np.sum(h_refined)
                h_refined = h_refined * scale

        c5_refined = evaluate_c5(h_refined)
        if c5_refined < best_final_c5:
            best_final_c5 = c5_refined
            best_final_h = h_refined.copy()

    if best_final_h is not None:
        return (best_final_h.tolist(), best_final_c5, n)
    else:
        # Final fallback
        try:
            h_zero = np.ones(n) * 0.5
            h_zero = h_zero / np.sum(h_zero) * (n/2)
            c5_zero = evaluate_c5(h_zero)
            return (h_zero.tolist(), c5_zero, n)
        except:
            return ([0.5]*n, 0.5, n)
