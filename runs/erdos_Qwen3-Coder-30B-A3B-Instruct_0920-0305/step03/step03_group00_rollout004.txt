import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import cvxpy as cp
import time

def weight_normalise(x, n):
    """Rescales vector x to satisfy sum(x) = n/2"""
    return x * (n / 2) / np.sum(x)

def max_defect(h):
    """Compute exact C5 = max_k int h(x)(1-h(x+k)) dx"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def softmax_objective(h, alpha):
    """Return soft-max objective Φ_α and gradient ∇Φ_α"""
    n = len(h)
    dx = 2.0 / n
    # Compute all f_s(h) = sum h[i] * (1 - h[i+s])
    # No need to materialize full matrix; compute correlations incrementally

    # For speed, efficiently compute:
    # f_s = sum_i h[i] * (1 - h[i+s]) = sum_i h[i] - sum_i h[i] * h[i+s] = (1 - corr[i,s])
    # So f_s = (n - corr[s + n-1]) / dx, since corr[s + n-1] = sum_i h[i] * h[i+s]
    corr_full = correlate(h, 1 - h, mode="full")
    corr_hh = correlate(h, h, mode="full")
    shifts = np.arange(1, n)
    fs = np.array([n - corr_hh[s + n - 1] for s in shifts]) / dx

    # Soft max
    exp_fs = np.exp(alpha * fs)
    Z = np.sum(exp_fs)
    phi = np.log(Z) / alpha

    # Gradient w.r.t h_j
    grad = np.zeros_like(h)
    for s in shifts:
        # Gradient for shift s
        w_s = exp_fs[s - 1] / Z
        for i in range(n):
            if i + s < n:
                grad[i] += w_s * (1 - h[i + s])
            if i - s >= 0:
                grad[i] += w_s * (-h[i - s])
    return phi, grad

def refine_continuous(h0, n):
    """Refine the binary solution with smooth max optimization"""
    # For continuous phase, keep the pattern but ensure binary values at block boundaries
    # This is actually an aggressive optimization that assumes h0 was opt in binary
    # We'll use `trust-constr` with analytic gradient
    dx = 2.0 / n
    
    def objective(h):
        phi, _ = softmax_objective(h, 5.0)
        return phi
    
    def gradient(h):
        _, g = softmax_objective(h, 5.0)
        return g
    
    bounds = [(0, 1) for _ in range(n)]
    
    # Update alpha schedule to 5->15->40
    alphas = [5.0, 15.0, 40.0]
    h_current = h0.copy()
    
    # Optimization settings
    maxiter = 200
    for alpha in alphas:
        # Redefine objective function with new alpha
        def obj(h):
            phi, _ = softmax_objective(h, alpha)
            return phi

        def grad(h):
            _, g = softmax_objective(h, alpha)
            return g

        res = minimize(
            obj,
            h_current,
            method='trust-constr',
            jac=grad,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n/2},
            options={'maxiter': maxiter, 'gtol': 1e-6, 'xtol': 1e-6}
        )
        
        if res.success:
            h_current = res.x
            # Clip to [0,1] if anything escapes slightly
            h_current = np.clip(h_current, 0, 1)
            # Re-normalize exactly
            h_current = weight_normalise(h_current, n)
        else:
            # Fallback: still try
            h_current = np.clip(h_current, 0, 1)
            h_current = weight_normalise(h_current, n)
            
    return h_current

def anneal(h, n):
    """Stochastic swap procedure to refine further"""
    dx = 2.0 / n
    max_corr = correlate(h, 1 - h, mode="full")
    
    num_proposals = 5000
    gamma = 0.2
    temp = 1.0
    t_start = time.time()
    
    for i in range(num_proposals):
        if (time.time() - t_start) > 550:
            break
        
        # Find the shift with maximum correlation
        pos = np.argmax(max_corr)
        s_star = pos - n + 1
        if s_star <= 0:
            s_star = 1 - s_star
            
        # Index where h[i] > 0 and h[i+s_star] is small
        # too adversary - just pick a weighted random sample  
        probs = h * (1 - np.roll(h, s_star))  # for eff metric
        if np.sum(probs) < 1e-10:
            # force proceed
            idxs = np.arange(n)
        else:
            probs = probs / np.sum(probs)
            idxs = np.random.choice(np.arange(n), p=probs)
        
        # Pick two indices (i, j) where i contributes badly
        i1 = np.random.choice(idxs, p=probs[idxs] if len(idxs) > 0 else None)
        
        # Distributed choice for (j) from low, but diff region
        probs_j = np.maximum(1 - h, 0) * h
        if np.sum(probs_j) < 1e-10:
            j1 = np.random.randint(0, n)
        else:
            probs_j = probs_j / np.sum(probs_j)
            j1 = np.random.choice(np.arange(n), p=probs_j)
            
        # Small delta movement to swap
        delta = min(h[i1], 1 - h[j1]) * gamma
        h[i1] -= delta
        h[j1] += delta
        # Also lower bound in corner
        if i1 + s_star < n:
            h[i1 + s_star] += delta    
        if j1 - s_star >= 0:
            h[j1 - s_star] -= delta
        
        # At end, re-normalize and update correlation
        h = weight_normalise(h, n)
        max_corr = correlate(h, 1 - h, mode="full")
        
        gamma *= 0.99
        temp *= 0.9999
    
    return h

def build_milp(n):
    """Build binary MILP to maximize minimum aperiodic cross-correlation"""
    # Variables
    h = cp.Variable(n, boolean=True)
    w = cp.Variable((n, n-1), nonnegative=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.sum(h) == n/2
    ]

    # Linearisation of w[i,s] = h[i] * h[i+s]
    for s in range(1, n):
        for i in range(n):
            if i + s < n:
                constraints.append(w[i, s-1] <= h[i])
                constraints.append(w[i, s-1] <= h[i + s])
                constraints.append(w[i, s-1] >= h[i] + h[i + s] - 1)
    
    # Minimum overlap constraint
    for s in range(1, n):
        s_idx = s - 1
        constraints.append(cp.sum(w[:, s_idx]) >= t)

    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit (CBC)
    prob.solve(solver=cp.CBC, maxSeconds=180, seconds=180, ratioGap=0.0)
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        solution = h.value
        if solution is not None:
            return solution
    # Time limit hit or infeasible, but return what was found
    if hasattr(prob, 'solution') and prob.solution is not None:
        sol = h.value
        if sol is not None:
            return sol
    return None
