import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage A: Low-dim global optimization with CMA-ES-like approach
    n_points = 200
    dx = 2.0 / n_points
    m = 10  # num blocks
    max_evaluations = 1500
    initial_block_heights = np.array(initial_h_values)
    initial_block_heights = initial_block_heights[:m] if len(initial_block_heights) >= m else np.pad(initial_block_heights, (0, m - len(initial_block_heights)), 'constant')
    
    # Adjust to make sure sum is m/2
    current_sum = np.sum(initial_block_heights)
    if current_sum > 0:
        initial_block_heights = initial_block_heights * (m/2) / current_sum
    
    def create_h_from_blocks(u_vec):
        u = np.array(u_vec)
        if len(u) != m:
            raise ValueError("u_vec must have length m")
        # Expand to full resolution
        expanded = np.repeat(u, n_points // m)
        if len(expanded) < n_points:
            expanded = np.pad(expanded, (0, n_points - len(expanded)), 'constant', constant_values=u[-1])
        elif len(expanded) > n_points:
            expanded = expanded[:n_points]
        return expanded

    def evaluate_overlaps(u):
        h = create_h_from_blocks(u)
        # Using the exact correlation used by the checker
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx

    def constrained_u(u):
        # Ensure last block height is valid when m > 1
        if m > 1:
            u_m = m/2 - np.sum(u[:-1])
            u_m = np.clip(u_m, 0, 1)
            u[-1] = u_m
        return u

    # Projected gradient descent for full 200-dimensional problem
    def projected_gradient_descent(h0, max_iter=3000, alpha0=1e-3):
        h = h0.copy()
        best_h = h.copy()
        best_c5 = evaluate_overlaps(constrained_u(h))
        
        for t in range(max_iter):
            # Compute subgradient for the worst overlap
            corr = np.correlate(h, 1-h, mode="full")
            k_idx = np.argmax(corr)
            k_shift = k_idx - (n_points - 1)
            
            # Compute gradient for this k
            g = np.zeros(n_points)
            # Handle indices carefully
            if k_shift >= 0:
                # h[i] appears in overlap k where h[i] contributes (1 - h[i+k])
                # For index i, partial derivative w.r.t h[i] is (1 - h[i+k]) - h[i-k]
                # But h[i] = 0 if i<0 or i>=n; same for h[i-k] and h[i+k]
                for i in range(n_points):
                    if i-k_shift >= 0 and i-k_shift < n_points:
                        # contribution from h[i+k] 
                        g[i] -= h[i-k_shift]
                    if i+k_shift >= 0 and i+k_shift < n_points:
                        # contribution from h[i-k]
                        g[i] -= h[i+k_shift]
                    g[i] += 1  # from the 1 in (1-h[i+k])

            # Apply gradient step
            alpha = alpha0 / np.sqrt(t + 1)
            h_new = h - alpha * g
            
            # Project onto constraint
            # Find lambda such that clip(h_new - lambda, 0, 1) sums to n/2
            y = h_new.copy()
            def project_h(y):
                def find_lambda(lam):
                    x = np.clip(y - lam, 0, 1)
                    return np.sum(x) - n_points/2
                
                low, high = -2.0, 3.0
                for _ in range(50):
                    mid = (low + high) / 2
                    val = find_lambda(mid)
                    if val > 0:
                        low = mid
                    else:
                        high = mid
                lam = (low + high) / 2
                x = np.clip(y - lam, 0, 1)
                return x
            
            h = project_h(y)
            current_c5 = evaluate_overlaps(constrained_u(h))
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                
            if t > 200 and abs(best_c5 - current_c5) < 1e-6:
                break
                
        return best_h, best_c5

    # Stage B: Full resolution projected gradient descent
    u_initial = initial_block_heights.copy()
    h_full, c5_b = projected_gradient_descent(create_h_from_blocks(u_initial))
    
    # Keep track of best so far for stages C and D
    best_h = h_full
    best_c5 = c5_b

    # Stage C: Binary rounding and swap-based local search
    def binary_rounding(h_continuous, K=30):
        n = len(h_continuous)
        best_c5 = float('inf')
        best_binary = None
        for _ in range(K):
            # Create binary vector
            probs = h_continuous
            indices = np.argsort(probs)[::-1]
            binary = np.zeros(n, dtype=int)
            count = 0
            for i in indices:
                if count < n//2:
                    binary[i] = 1
                    count += 1
                else:
                    break
            # Ensure exactly half are 1’s
            while np.sum(binary) < n//2:
                # Pick lowest probability indices
                zero_els = np.where(binary == 0)[0]
                if len(zero_els) == 0:
                    break
                # Choose smallest probability index
                min_prob_idx = np.argmin([probs[i] for i in zero_els])
                binary[zero_els[min_prob_idx]] = 1
            while np.sum(binary) > n//2:
                zero_els = np.where(binary == 1)[0]
                if len(zero_els) == 0:
                    break
                # Choose highest probability index to set to 0
                max_prob_idx = np.argmax([probs[i] for i in zero_els])
                binary[zero_els[max_prob_idx]] = 0
            # Evaluate
            # Option: use numpy for speed if js gets unoptimized later
            corr = np.correlate(binary, 1-binary, mode="full")
            max_corr = np.max(corr) * dx
            if max_corr < best_c5:
                best_c5 = max_corr
                best_binary = binary.copy()
        return best_binary, best_c5

    # Perform binary rounding
    binary_v, c5_c = binary_rounding(h_full)
    if c5_c < best_c5:
        best_h = binary_v
        best_c5 = c5_c

    # Performs swaps and evaluates improvements from scratch
    def evaluate_binary_swap_moves(h, i, j):
        # Update the \sum h_i (1 - h_{i+k}) in the overlap sums for all k
        # Store overlap sums per shift
        n = len(h)
        h_shifted = [None] * (2*n - 1)
        for k in range(2*n - 1):
            shift = k - (n - 1)
            if shift >= 0:
                # Index i in shifted vector
                if i >= 0 and i < n and (i+shift) < n:
                    h_shifted[k] = h[i+shift]
                else:
                    h_shifted[k] = 0
            else:
                # Index i in shifted vector
                if i >= 0 and i < n and (i+shift) >= 0:
                    h_shifted[k] = h[i+shift]
                else:
                    h_shifted[k] = 0
        
        # Compute actual sums
        S = [0] * (2*n - 1)
        for i_shift, h_in_shift in enumerate(h_shifted):
            if h_in_shift != None:
                for k in range(2*n - 1):
                    shift = k - (n - 1)
                    if i_shift >= 0 and i_shift < n and (i_shift + shift) >= 0 and (i_shift + shift) < n:
                        S[k] += h[i_shift] * h[i_shift + shift]
        
        # Then do the swap
        h_copy = h.copy()
        h_copy[i], h_copy[j] = h_copy[j], h_copy[i]
        S_new = [0] * (2*n - 1)
        for i_shift in range(n):
            for k in range(2*n - 1):
                shift = k - (n - 1)
                if i_shift >= 0 and i_shift < n and (i_shift + shift) >= 0 and (i_shift + shift) < n:
                    S_new[k] += h_copy[i_shift] * h_copy[i_shift + shift]

        # Calculate deltas
        min_delta = float('inf')
        for k in range(2*n - 1):
            S_diff = S[k] - S_new[k]
            # The shifting is h_i(1-h_{i+k}) 
            new_over = n/2 - S_new[k]
            old_over = n/2 - S[k]
            delta_c5 = (1 - new_over) - (1 - old_over)  # = old_over - new_over
            if delta_c5 < min_delta:
                min_delta = delta_c5
        return -min_delta if min_delta < 0 else 0.0

    def local_search_binary(h):
        h = h.copy()
        improved = True
        rounds = 0
        while improved and rounds < 10:
            max_improvement = 0
            best_swap = None
            # Sample swaps (heuristic: sample 90% of all swaps if small, otherwise 1%)
            n = len(h)
            total_pairs = sum(i * (n-i) for i in range(n)) // 2
            max_pairs_to_check = min(10000, total_pairs // 10)
            pairs_checked = 0
            for i in range(n):
                if pairs_checked >= max_pairs_to_check:
                    break
                for j in range(i+1, n):
                    if h[i] == 1 and h[j] == 0:
                        improvement = evaluate_binary_swap_moves(h, i, j)
                        if improvement < max_improvement:
                            max_improvement = improvement
                            best_swap = (i, j)
                    pairs_checked += 1
                    if pairs_checked >= max_pairs_to_check:
                        break
                if pairs_checked >= max_pairs_to_check:
                    break
            
            if max_improvement < 0:
                i, j = best_swap
                h[i], h[j] = h[j], h[i]
                rounds += 1
            else:
                improved = False
                break
        return h

    binary_final = local_search_binary(binary_v.astype(float))
    corr_final = np.correlate(binary_final, 1 - binary_final, mode="full")
    final_c5 = np.max(corr_final) * dx
    if final_c5 < best_c5:
        best_h = binary_final
        best_c5 = final_c5

    # Stage D: Optional MILP if time permits
    if budget_s - (time.time() - start_time) > 50:
        try:
            # Initialize binary vector correctly
            b = binary_final
            n = len(b)
            h_vars = Variable(n, boolean=True)
            w_vars = [Variable(n*(n-1), boolean=True) for _ in range(n)]
            C = Variable()
            # Get effective shifts
            corr = np.correlate(b, 1 - b, mode="full")
            shift_indices = list(range(len(corr)))
            
            # Constraints
            objectives = []
            for i in range(n):
                for j in range(n):
                    k = abs(i - j)
                    if i < j:
                        # Add unique constraints only
                        w_vars[i][j] = h_vars[i] * h_vars[j]
                        # Linearization: w[i,j] ≤ h[i], w[i,j] ≤ h[j], w[i,j] ≥ h[i] + h[j] - 1  
                        objectives.append(w_vars[i][j] <= h_vars[i])
                        objectives.append(w_vars[i][j] <= h_vars[j])
                        objectives.append(w_vars[i][j] >= h_vars[i] + h_vars[j] - 1)
                
                # Constraint on binary sum
                objectives.append( sum(h_vars) == n // 2 )
                
                #Constraint on C
                for k in range(n):
                    ov_k = sum([w_vars[i][k] for i in range(n)])
                    objectives.append(ov_k <= C)
                      
            # Min max_overlap = C
            constraints = objectives
            prob = Problem(Minimize(C), constraints)
            prob.solve(solver=GUROBI, verbose=False, timeLimit=30)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                c5_milp = C.value * dx
                if c5_milp < best_c5:
                    best_c5 = c5_milp
        except:
            pass
    
    return (best_h.tolist(), best_c5, n_points)
