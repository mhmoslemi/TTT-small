def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

        t = Variable()
        
        # Constraints
        constraints = []
        
        # McCormick envelopes
        for i in range(n):
            for j in range(i+1, n):
                if i+j < n: # To conform with shifts
                    constraints.append(X[i,j] <= h[i])
                    constraints.append(X[i,j] <= h[j])
                    constraints.append(X[i,j] >= h[i] + h[j] - 1)
        
        # Diagonal
        for i in range(n):
            constraints.append(X[i,i] == h[i])
        
        # Box constraints
        for i in range(n):
            constraints.append(h[i] <= 1)
            constraints.append(h[i] >= 0)
        
        # Integral constraint
        constraints.append(sum(h) == n/2)
        
        # Overlap constraints
        for s in range(1, n):
            # overlap_sum = sum_{i=0}^{n-1-s} X[i, i+s]
            overlap = 0
            for i in range(n-s):
                overlap += X[i, i+s]
            constraints.append(overlap >= t)
        
        # Objective
        objective = Maximize(t)
        prob = Problem(objective, constraints)
        
        # Solve with SCS
        prob.solve(solver=SCS, max_iters=5000, eps=1e-5, verbose=False)

        exp_f = np.exp(alpha * f_s)
        return np.log(np.sum(exp_f)) / alpha
