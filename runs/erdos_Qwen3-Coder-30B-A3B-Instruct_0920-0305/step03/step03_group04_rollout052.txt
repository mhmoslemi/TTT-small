import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time

n_values = [48, 60, 72, 84]
dxs = [2.0 / n for n in n_values]
target_bound = 0.3807
max_iter = 500
seed = 42
np.random.seed(seed)

def evaluate_h(h):
    # Full correlation array
    c = np.correlate(h, 1 - h, mode="full")
    # Scale and return maximum
    return np.max(c) * (2.0 / len(h))

def correlation_vec(h):
    return np.correlate(h, 1 - h, mode="full")

def smooth_max(c, alpha=30.0):
    m = np.max(c)
    if np.isinf(m) or np.isnan(m):
        return m
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

def proj_to_simplex(v, S=None):
    if S is None:
        S = np.sum(v)
    if S == 0:
        return np.zeros_like(v)
    v_clipped = np.clip(v, 0, 1)
    v_sum = np.sum(v_clipped)
    if v_sum == S:
        return v_clipped
    if v_sum > S:
        diff = v_sum - S
        idxs = np.argsort(-v_clipped)
        for i in idxs:
            if v_clipped[i] >= diff:
                v_clipped[i] -= diff
                return v_clipped
            else:
                diff -= v_clipped[i]
                v_clipped[i] = 0
        return v_clipped
    else:
        diff = S - v_sum
        idxs = np.argsort(v_clipped)
        for i in idxs:
            if 1.0 - v_clipped[i] >= diff:
                v_clipped[i] += diff
                return v_clipped
            else:
                diff -= (1.0 - v_clipped[i])
                v_clipped[i] = 1.0
        return v_clipped

def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-x))

def logit(p):
    return np.log(p / (1.0 - p))

def binary_initial(n):
    h = np.zeros(n)
    indices = np.random.permutation(n)[:n // 2]
    h[indices] = 1
    return h

def swap_and_evaluate(h, i_one, i_zero):
    h_new = h.copy()
    h_new[i_one] = 0
    h_new[i_zero] = 1
    return evaluate_h(h_new)

def binary_hill_climb(h0, max_iter):
    h = h0.copy()
    best_h = h.copy()
    best_val = evaluate_h(h)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    iter_count = 0
    stagnation_count = 0
    max_stagnation = max_iter
    
    while iter_count < max_iter:
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        
        val_new = evaluate_h(h_new)
        if val_new < best_val:
            # Accept the move
            h = h_new
            best_h = h_new.copy()
            best_val = val_new
            
            # Update indices
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count > max_stagnation:
                # Reinitialize
                h = binary_initial(len(h0))
                best_h = h.copy()
                best_val = evaluate_h(h)
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                stagnation_count = 0
                
        iter_count += 1
        
    return best_h, best_val

def parametric_intvl_family(n):
    h_best = None
    c5_best = np.inf
    for m in range(2, 9):
        g_base = (2 - m * (1.0 / m)) / (m - 1)
        if g_base < 0: continue
        gs = [0.5 * g_base, 0.75 * g_base, 0.9 * g_base, g_base]
        for g in gs:
            if m * (1.0 / m) + (m - 1) * g > 2: continue
            for delta in np.arange(0, g, 0.5 * (2.0 / n)):
                h = np.zeros(n)
                dx = 2.0 / n
                pos = delta
                for i in range(m):
                    start = int(pos / dx)
                    end = int((pos + 1.0 / m) / dx)
                    if end > n: end = n
                    if start < n:
                        h[start:end] = 1.0
                    pos += (1.0 / m) + g
                h = proj_to_simplex(h, S=n/2)
                c5 = evaluate_h(h)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h.copy()
    return h_best, c5_best

def refine_with_smooth_max(h0, dx, n):
    h0 = h0.copy()
    h0 = proj_to_simplex(h0, S=n/2)
    y0 = logit(h0)
    c0 = correlation_vec(h0)
    smooth_max_val = smooth_max(c0, alpha=30.0)
    lambda_ = 10000.0
    def obj(y):
        h = sigmoid(y)
        c = correlation_vec(h)
        smooth_val = smooth_max(c, alpha=30.0)
        penalty = lambda_ * (np.sum(h) - n/2)**2
        return smooth_val + penalty

    def grad(y):
        h = sigmoid(y)
        d_h = h * (1 - h)
        c = correlation_vec(h)
        m = np.max(c)
        exps = np.exp(30 * (c - m))
        sum_exp = np.sum(exps)
        grad_c = (30 * np.exp(30 * (c - m)) / sum_exp) if sum_exp > 0 else np.zeros_like(c)
        grad_h = np.correlate(grad_c, np.ones_like(h), mode='full')[:len(h)]
        grad_y = d_h * grad_h
        return grad_y + 2 * lambda_ * (np.sum(h) - n/2) * d_h
    
    # Multiple restarts
    best_h = h0.copy()
    best_val = evaluate_h(h0)
    for _ in range(8):
        y_start = y0 + np.random.normal(0, 0.2, n)
        # Optimization with bounds
        res = minimize(obj, y_start, method='L-BFGS-B', jac=grad, options={'maxiter': 300, 'ftol': 1e-10})
        if res.success:
            h = sigmoid(res.x)
            h = proj_to_simplex(h, S=n/2)
            val = evaluate_h(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
    
    return best_h, best_val
