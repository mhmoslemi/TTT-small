import numpy as np
from scipy.optimize import minimize
import time

# Parameters
n_points = 48
dx = 2.0 / n_points

def evaluate_h_binary(h_values):
    """Evaluate the C_5 bound for given binary h values."""
    h = np.array(h_values, dtype=int)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def evaluate_h_continuous(h_values):
    """Evaluate the C_5 bound for continuous h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def local_search_binary(h_values, max_iter=1000):
    """Refine binary solution using local search."""
    h_current = np.array(h_values, dtype=int)
    c5_current = evaluate_h_binary(h_current)
    
    for _ in range(max_iter):
        # Find all indices where h=1 and h=0
        ones = np.where(h_current == 1)[0]
        zeros = np.where(h_current == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        # Try swapping a 1 and 0
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Create new h
        h_new = h_current.copy()
        h_new[i] = 0
        h_new[j] = 1
        
        c5_new = evaluate_h_binary(h_new)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
        elif np.random.rand() < 0.01:  # Accept with low probability
            h_current = h_new
            c5_current = c5_new
    
    return h_current, c5_current

def simulated_annealing_binary(h_start, max_iter=500000, initial_temp=0.1, cooling_rate=0.9999):
    """Perform simulated annealing on binary h values."""
    h_current = np.array(h_start, dtype=int)
    c5_current = evaluate_h_binary(h_current)
    
    best_h = h_current.copy()
    best_c5 = c5_current
    
    temp = initial_temp
    
    for i in range(max_iter):
        # Find all indices where h=1 and h=0
        ones = np.where(h_current == 1)[0]
        zeros = np.where(h_current == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        # Try swapping a 1 and 0
        i_swap = np.random.choice(ones)
        j_swap = np.random.choice(zeros)
        
        # Create new h
        h_new = h_current.copy()
        h_new[i_swap] = 0
        h_new[j_swap] = 1
        
        c5_new = evaluate_h_binary(h_new)
        
        # Accept or reject
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
        else:
            p_accept = np.exp(-(c5_new - c5_current) / temp)
            if np.random.rand() < p_accept:
                h_current = h_new
                c5_current = c5_new
                
        # Update best
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            
        # Cool down
        temp *= cooling_rate
        
        # Occasionally perform local search
        if i % 1000 == 0:
            h_ls, c5_ls = local_search_binary(h_current, 100)
            if c5_ls < c5_current:
                h_current = h_ls
                c5_current = c5_ls
                
    return best_h, best_c5

def initialize_binary(n_points):
    """Initialize binary pattern with exactly half ones."""
    h = np.zeros(n_points, dtype=int)
    ones_indices = np.random.choice(n_points, size=n_points//2, replace=False)
    h[ones_indices] = 1
    return h

def run(seed=42, budget_s=600):
    """Main optimization routine."""
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    
    # Single run with binary simulated annealing
    h_start = initialize_binary(n_points)
    h_result, c5_result = simulated_annealing_binary(h_start, max_iter=100000)
    
    if c5_result < best_c5:
        best_h = h_result
        best_c5 = c5_result
    
    # Try multiple restarts
    for i in range(5):
        if time.time() - start_time > budget_s * 0.95:
            break
        h_start = initialize_binary(n_points)
        h_result, c5_result = simulated_annealing_binary(h_start, max_iter=50000)
        if c5_result < best_c5:
            best_h = h_result
            best_c5 = c5_result
    
    # Final local refinement
    if best_h is not None:
        best_h, best_c5 = local_search_binary(best_h, 5000)
    
    # Convert to list
    result_h = list(best_h.astype(float))
    
    return result_h, best_c5, n_points
