import numpy as np
import cvxpy as cp
import time

def solve_milp_binary(n, time_limit=100):
    """Solve the binary MILP formulation for given n."""
    dx = 2.0 / n
    # Variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h) == n // 2, t >= 0]

    # Derived variables and constraints for linearization
    # Create auxiliary binary y_{i,d} for i = 0..n-1-d, d = 0..n-1
    for d in range(n):
        for i in range(n - d):
            y = cp.Variable(boolean=True)
            constraints.extend([
                y <= h[i],
                y <= h[i + d],
                y >= h[i] + h[i + d] - 1,
            ])
            # Accumulate for Q_d
            if d == 0:
                # Sum_{i} y_{i,0} = Q_0 = sum h_i
                pass
            else:
                pass  # actual usage later ...
                
    # This approach will dynamically create auxiliary variables. 
    # For n=48 to 72, this is manageable.
    
    # To keep compact:
    # Let's compute Q_d = sum_{i=0}^{n-1-d} h_i * h_{i+d} 
    # via introducing the products y_{i,d}
    # keep a list of y_{i,d} for each d, and define Q_d accordingly
    
    # for clarity, we'll separate variable definitions:
    y_vars = {}
    Q = []

    # Create auxiliary variables y_{i,d} and define Q_d
    for d in range(n):
        y_vars[d] = []
        for i in range(n - d):
            y = cp.Variable(boolean=True)
            y_vars[d].append(y)
            constraints.extend([
                y <= h[i],
                y <= h[i + d],
                y >= h[i] + h[i + d] - 1,
            ])
        # Q_d = sum of all y_{i,d}
        Q.append(cp.sum(y_vars[d]))
        if d > 0:  # to avoid double counting Q_0 and Q_d for d=0
            constraints.append(Q[d] >= t)
        else:
            constraints.append(Q[0] >= t)  # Q_0 = sum h_i = n/2, we don't want it to be too small

    # Objective: maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)

    # Solve with timeout
    try:
        prob.solve(solver=cp.CBC, time_limit=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_val = h.value
            t_val = t.value
            C5 = 1 - (2.0 / n) * t_val
            return h_val, C5
        else:
            # Fallback to best feasible if problem is infeasible or suboptimal
            h_val = h.value
            if h_val is not None:
                t_val = t.value
                C5 = 1 - (2.0 / n) * t_val
                return h_val, C5
            return None, np.inf
    except Exception:
        return None, np.inf

def hill_climb_swaps(h, max_swaps=200000, time_limit=120):
    """Improve a binary solution via swapping 1s and 0s."""
    start = time.time()
    h = np.array(h, dtype=float)
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1 - h)
    n = len(h)
    dx = 2.0 / n
    current = np.copy(h)
    
    def eval_C5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return 1.0 - dx * np.min(corr)
    
    best_h = np.copy(current)
    best_C5 = eval_C5(current)
    
    swaps = 0
    while swaps < max_swaps and time.time() - start < time_limit:
        swaps += 1
        # Swap a 1 with 0
        if len(ones) > 0 and len(zeros) > 0:
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            test_h = np.copy(current)
            test_h[i] = 0
            test_h[j] = 1
            
            test_C5 = eval_C5(test_h)
            if test_C5 < best_C5:
                best_C5 = test_C5
                best_h = test_h
                current = test_h
                ones = np.flatnonzero(current)
                zeros = np.flatnonzero(1 - current)
        else:
            continue
    
    return best_h, best_C5

def continuous_refine(h_binary, n_original, time_limit=10):
    """Refine the solution using continuous gradient descent."""
    start = time.time()
    h = np.array(h_binary, dtype=float)
    n = h.shape[0]
    dx = 2.0 / n
    target_sum = n / 2
    
    # Interpolate to a finer grid (e.g., double the points)
    def interpolate(h, n_out):
        h_out = np.interp(np.linspace(0, 2, n_out), np.linspace(0, 2, n), h)
        return h_out
    
    n_fine = min(n * 2, 96)  # upsample to at most 96
    h_fine = interpolate(h, n_fine)
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (target_sum / np.sum(h_fine))
    
    # Projected gradient descent on h_fine (only keep it in [0,1] and sum = target_sum)
    t = 0.01  # step size
    max_iter = 50000
    for i in range(max_iter):
        if time.time() - start > time_limit:
            break
        # Evaluate gradient
        corr = np.correlate(h_fine, 1 - h_fine, mode="full")
        C5_val = 1 - dx * np.min(corr)
        # For gradient descent on the max-min function, one computes the Jacobian.
        # Since this is complex, we get an approximate descent step using finite differences.
        # But better: we will just project onto constraints.
        grad = np.ones_like(h_fine)  # Placeholder
        # Simple gradient ascent on C5 (even better: what if we relax to minimize C5?)
        h_fine -= t * grad
        # Project h_fine to [0,1]
        h_fine = np.clip(h_fine, 0, 1)
        # Project sum
        current_sum = np.sum(h_fine)
        if abs(current_sum) > 1e-6:
            h_fine *= (target_sum / current_sum)
    return h_fine, 1 - dx * np.min(np.correlate(h_fine, 1 - h_fine, mode="full"))

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_solution = (None, float('inf'), None)

    # Candidate n values
    n_candidates = [48, 56, 64, 72]

    for n in n_candidates:
        if time.time() - start_time > 0.95 * budget_s:
            break
        
        time_remaining = budget_s - (time.time() - start_time)
        if time_remaining < 20:
            break
            
        T = min(120, time_remaining * 0.6)
        
        try:
            h_result, C5_result = solve_milp_binary(n, time_limit=T)
        except Exception:
            h_result, C5_result = None, float('inf')

        if h_result is not None and C5_result < best_solution[1]:
            best_solution = (h_result, C5_result, n)

    # Local improvement
    h_best, C5_best, n_best = best_solution
    if h_best is not None:
        time_remaining = budget_s - (time.time() - start_time)
        if time_remaining > 10:
            try:
                h_improved, C5_improved = hill_climb_swaps(
                    h_best, max_swaps=100000, time_limit=time_remaining/2
                )
                if C5_improved < C5_best:
                    h_best = h_improved
                    C5_best = C5_improved
            except:
                pass

    # Final continuous refinement  
    if h_best is not None:
        time_remaining = budget_s - (time.time() - start_time)
        if time_remaining > 2:
            try:
                h_final, C5_final = continuous_refine(h_best, n_best, time_limit=time_remaining/2)
                if C5_final < C5_best:
                    h_best = h_final
                    C5_best = C5_final
            except:
                pass

    return list(h_best), C5_best, n_best
