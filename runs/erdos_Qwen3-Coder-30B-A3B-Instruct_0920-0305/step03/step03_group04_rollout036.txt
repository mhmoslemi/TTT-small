import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import SDP

# Constants
n_candidates = [36, 42, 48, 54, 60]
threshold_C5 = 0.3807
dx = 2.0 / 48  # Default
n_points = 48  # Fixed for speed and to stay within budget

def evaluate_h(h_values):
    """Compute C5 = max_k ∫ h(x)(1 - h(x+k)) dx using numpy correlation."""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full") 
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

def correlation_vec(h):
    """Returns the full correlation vector between h and (1-h) in [0, 2n-1)."""
    h = np.array(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return corr

def smooth_max(c, alpha=30.0):
    """Smooth approximation to max using log-sum-exp. Returns scalar."""
    return np.logaddexp.reduce(alpha * c) / alpha

def proj_to_simplex(v, s=1.0):
    """Project vector v to the simplex {x: x >= 0, sum(x) = s}."""
    v = np.clip(v, 0, None)
    if np.sum(v) <= s:
        return v * s / np.sum(v) if np.sum(v) > 1e-10 else v
    
    # Use the standard simplex projection (sort and find threshold)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(u)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Case where sum(v) < s
        return v * s / np.sum(v) if np.sum(v) > 1e-10 else v

    rho = np.argmax(cond)
    theta = cssv[rho] / (rho+1)
    return np.maximum(v - theta, 0)

def logit(x, eps=1e-6):
    x = np.clip(x, eps, 1 - eps)
    return np.log(x / (1 - x))

def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-x))

def solve_sdp(n, timeout_limit=25):
    """Solve the SDP relaxation for N points."""
    t = time.time()
    x = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t_var = cp.Variable()
    constraints = [
        0 <= x,
        x <= 1,
        cp.sum(x) == n/2,
        cp.diag(X) == x,
        X >> 0,
    ]
    # Add McCormick constraints per i,j
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(X[i, j] >= 0)
            constraints.append(X[i, j] <= x[i])
            constraints.append(X[i, j] <= x[j])
            constraints.append(X[i, j] >= x[i] + x[j] - 1)
            
    # Overlap constraints for each shift k
    corr_pairs = []
    for k in range(n):
        # Σ_{i=0}^{n‑1‑k} x_i – Σ_{i=0}^{n‑1‑k} X_{i,i+k}
        if k == 0:
            l_sum = cp.sum(x)
            r_sum = cp.sum(X[:n, :n])  # full diagonal
        elif k > 0:
            l_sum = cp.sum(x[:n-k])
            r_sum = cp.sum(cp.diag(X[:n-k, k:]))
        corr_pairs.append(l_sum - r_sum <= t_var)
    constraints.extend(corr_pairs)
    objective = cp.Minimize(t_var)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver='SCS', max_iters=8000, eps=1e-5)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            sol_x = x.value
            sol_X = X.value
            t_star = t_var.value
            return sol_x, sol_X, t_star, True
        else:
            return None, None, float('inf'), False
    except:
        return None, None, float('inf'), False

def round_from_sdp(x_star, X_star, n, R=30):
    """Round from SDP solution to binary, keeping exactly n/2 ones."""
    U, svals, Vt = np.linalg.svd(X_star, full_matrices=False)
    rank = np.sum(svals > 1e-10)
    V = U[:, :rank] * np.sqrt(svals[:rank])
    
    best_h = None
    best_c5 = float('inf')
    
    for _ in range(R):
        g = np.random.randn(rank)
        s = np.sign(V @ g)
        # Roughly how many 1s?
        cnt = int(np.sum(s >= 0))
        # Adjust to n/2 by flipping a few
        delta = cnt - n//2
        if delta > 0:
            # Too many 1s: flip delta 1s to 0s
            flipped_ij = np.argsort(np.abs(V @ g))[-delta:]
            s[flipped_ij] = -1
        elif delta < 0:
            # Too few 1s: flip |delta| 0s to 1s
            flipped_ij = np.argsort(np.abs(V @ g))[:abs(delta)]
            s[flipped_ij] = 1
        h = (s >= 0).astype(float)
        c5 = evaluate_h(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def hill_climb_binary(h_init, max_swaps=500000):
    """Perform a large-scale binary hill-climb with acceleration."""
    h = h_init.copy()
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    if len(ones) == 0 or len(zeros) == 0:
        return h, evaluate_h(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h)
    current_c5 = best_c5
    
    swaps = 0
    restart_after = 20000
    last_improved = 0
    
    while swaps < max_swaps:
        if swaps - last_improved > restart_after:
            # Restart with a new random binary
            h = np.random.random(n_points) < 0.5
            h = h.astype(float)
            h_sum = np.sum(h)
            ratio = (n_points / 2) / h_sum
            h *= ratio  # normalize
            ones = np.where(h == 1)[0].tolist()
            zeros = np.where(h == 0)[0].tolist()
            current_c5 = evaluate_h(h)
            best_c5 = current_c5
            best_h = h.copy()
            last_improved = swaps
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        h[i], h[j] = h[j], h[i]  # swap
        ones.remove(i)
        zeros.remove(j)
        ones.append(j)
        zeros.append(i)
        
        new_c5 = evaluate_h(h)
        if new_c5 < current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            last_improved = swaps
        else:
            # Revert swap
            h[i], h[j] = h[j], h[i]
            ones.remove(j)
            zeros.remove(i)
            ones.append(i)
            zeros.append(j)
            
        swaps += 1
    return best_h, best_c5

def refine_smoothly(h_init, max_iter=300):
    """Refine using smooth surrogate objective with L-BFGS-B."""
    h = h_init.copy()
    y = logit(h)
    
    def smooth_obj(y_in):
        h_in = sigmoid(y_in)
        corr = correlation_vec(h_in)
        soft_max = smooth_max(corr)
        penalty = 100 * ((np.sum(h_in) - n_points/2) ** 2)
        return soft_max + penalty
    
    def grad_func(y_in):
        h_in = sigmoid(y_in)
        corr = correlation_vec(h_in)
        # Gradient of soft max term (use central diff approx)
        eps = 1e-8
        grad = np.zeros_like(y_in)
        for i in range(len(y_in)):
            y_plus = y_in.copy()
            y_plus[i] += eps
            y_minus = y_in.copy()
            y_minus[i] -= eps
            f_plus = smooth_obj(y_plus)
            f_minus = smooth_obj(y_minus)
            grad[i] = (f_plus - f_minus) / (2 * eps)
        # Penalty gradient
        grad += 200 * (np.sum(h_in) - n_points/2) * (h_in * (1 - h_in))
        return grad
    
    try:
        res = minimize(smooth_obj, y, method='L-BFGS-B', jac=grad_func,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-10})
        if res.success:
            y_opt = res.x
            h_opt = sigmoid(y_opt)
            h_opt = proj_to_simplex(h_opt, s=n_points/2)
            return h_opt
    except:
        pass
    return h_init

        if sdp_time > budget_s * 0.2:
            break
        if n > n_points:
            continue
        try:
            x_star, X_star, t_star, success = solve_sdp(n, timeout_limit=10)
            if success and x_star is not None:
                sdp_results.append((x_star, X_star, t_star, n))
                sdp_time = time.time() - t0
        except:
            sdp_time = time.time() - t0
            pass
        if sdp_time > budget_s * 0.2:
            break
    
    # Phase B: Rounding from SDP
    rounding_time = time.time() - t0
    if rounding_time < budget_s * 0.3:
        best_h_rounded = None
        best_c5_rounded = float('inf')
        for x_in, _, _, n in sdp_results:
            if rounding_time > budget_s * 0.3:
                break
            if n == n_points:
                h_rounded, c5_rounded = round_from_sdp(x_in, np.outer(x_in, x_in), n, R=20)
                if c5_rounded < best_c5_rounded:
                    best_c5_rounded = c5_rounded
                    best_h_rounded = h_rounded
        if best_h_rounded is not None:
            best_h = best_h_rounded.copy()
            best_c5 = best_c5_rounded
    
    # Phase C: hill_climb
    hill_time = time.time() - t0
    if hill_time < budget_s * 0.7:
        h_improved, c5_improved = hill_climb_binary(best_h, max_swaps=300000)
        if c5_improved < best_c5:
            best_h = h_improved.copy()
            best_c5 = c5_improved
        if best_c5 <= threshold_C5:
            # Extra polish
            polish_result = minimize(evaluate_h, best_h, method='SLSQP', 
                                   bounds=[(0, 1)] * n_points,
                                   constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2},
                                   options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-10})
            if polish_result.success:
                best_h = polish_result.x
            return list(best_h), evaluate_h(best_h), n_points

    # Phase D: smooth refinement
    refine_time = time.time() - t0
    if refine_time < budget_s * 0.9:
        h_smooth = refine_smoothly(best_h, max_iter=200)
        c5_smooth = evaluate_h(h_smooth)
        if c5_smooth < best_c5:
            best_h = h_smooth.copy()
            best_c5 = c5_smooth
    
    # Phase E: final polish
    polish_time = time.time() - t0
    if polish_time < budget_s - 2:
        polish_result = minimize(evaluate_h, best_h, method='SLSQP', 
                               bounds=[(0, 1)] * n_points,
                               constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2},
                               options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-10})
        if polish_result.success:
            best_h = polish_result.x
    
    return list(best_h), evaluate_h(best_h), n_points
