from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def smooth_max_objective(h, alpha=10.0):
    """Smooth approximation of max function for gradient-based optimization"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlation values
    corr = correlate(h, 1 - h, mode="full")
    # Take only the relevant portion (excluding zero shift which is 0)
    corr_vals = corr[n-1:]  # from 0 to n-1 shifts
    # Apply smooth max approximation
    smooth_max = np.log(np.sum(np.exp(alpha * corr_vals))) / alpha
    return -smooth_max  # negative because we minimize
