import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import SCS
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Choose resolution
    n = 100 if budget_s > 200 else 80

    # Set up the SDP
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    # Constraints to make X approximate h * h^T
    constraints = [
        cp.diag(X) == h,
        cp.sum(h) == n / 2.0,
        0 <= h,
        h <= 1,
        X >> 0  # PSD constraint
    ]
    
    # Add McCormick envelopes (X_ij <= h_i, X_ij <= h_j, X_ij >= h_i + h_j - 1)
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(X[i, j] <= h[i])
            constraints.append(X[i, j] <= h[j])
            constraints.append(X[i, j] >= h[i] + h[j] - 1)
    
    # Define overlaps for each shift s and constraint in terms of t
    shifts = list(range(1, n))
    overlaps = []
    
    for s in shifts:
        o = 0
        for i in range(n - s):
            o += X[i, i + s]
        overlaps.append(o)
    
    # Add overlapping constraints
    for s, o in enumerate(overlaps):
        constraints.append(o >= t)
    
    # Objective
    objective = cp.Maximize(t)
    
    # Create problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with maximal time limit
    try:
        prob.solve(solver=SCS, max_iters=20000, eps=1e-5, verbose=False, 
                   time_limit=150.0)
        h_opt = h.value
        X_opt = X.value
        t_opt = t.value
    except Exception as e:
        # Fallback if SDP fails
        h_opt = np.ones(n) * 0.5
        h_opt = h_opt / np.sum(h_opt) * n / 2
        t_opt = 0
        X_opt = np.outer(h_opt, h_opt)
    
    candidates = [h_opt]

    # Randomized rounding procedure with multiple trials
    rounding_trials = min(30, max(10, int((budget_s - 150.0) / 4.0)))
    if X_opt is not None: 
        # Factorize with small rank to sample generally
        try:
            # Use eigenvalue decomposition
            L, V = np.linalg.eigh(X_opt)
            V = V[:, L > 1e-8]
            L = L[L > 1e-8]
            Y = V @ np.diag(np.sqrt(L))
            
            for _ in range(rounding_trials):
                if time.time() - start_time > budget_s - 10:
                    break
                # Sample Gaussian vector
                g = np.random.standard_normal(Y.shape[1])
                y = Y @ g
            
                # Sort indices to pick top-n/2
                indices = np.argsort(-y)
                h_candidate = np.zeros(n)
                h_candidate[indices[:n//2]] = 1.0
                
                # Refinement step to make sure sum is correct
                if np.sum(h_candidate) != n / 2:
                    diff = int(n / 2 - np.sum(h_candidate))  
                    if diff < 0:
                        # Remove some 1s
                        to_zero = indices[n//2 + diff:n//2]
                        h_candidate[to_zero] = 0.0
                    elif diff > 0:
                        # Add some 0s
                        to_one = indices[n//2:n//2+diff]
                        h_candidate[to_one] = 1.0
                        
                # Optimization of candidate
                try:
                    def soft_obj(h_input):
                        f_values = []
                        dx = 2.0 / n
                        for s in range(1, n):
                            val = 0
                            for i in range(n - s):
                                val += h_input[i] * (1 - h_input[i + s])
                            f_values.append(val * dx)
                        return -np.max(f_values)  # Minimize negative max
                    
                    res = minimize(soft_obj, h_candidate, method='L-BFGS-B', bounds=[(0, 1)] * n, options={'maxiter': 20})
                    if res.success:
                        h_candidate = res.x
                        h_candidate = h_candidate / np.sum(h_candidate) * n / 2
                except:
                    pass
                
                candidates.append(h_candidate)
        except Exception as e:
            pass
    
    # Local refinement step using trust-constr
    def smooth_objective(h_vec, alpha):
        if len(np.shape(h_vec)) == 0:
            h_vec = np.array([h_vec])
        dx = 2.0 / n
        f_s = []
        for s in range(1, n):
            val = 0
            for i in range(n - s):
                val += h_vec[i] * (1 - h_vec[i + s])
            f_s.append(val * dx)
        soft_sum = np.sum(np.exp(alpha * np.array(f_s)))
        return -np.log(soft_sum) / alpha
    
    def grad_smooth(h_vec, alpha):
        if len(np.shape(h_vec)) == 0:
            h_vec = np.array([h_vec])
        dx = 2.0 / n
        f_s = []
        df_da = []
        for s in range(1, n):
            val = 0
            df = np.zeros_like(h_vec)
            for i in range(n - s):
                val += h_vec[i] * (1 - h_vec[i + s])
                df[i] += 1 - h_vec[i + s]
                if i + s < len(h_vec): 
                    df[i + s] -= h_vec[i]
            f_s.append(val * dx)
            df_da.append(df * dx)
                
        f_s = np.array(f_s)
        weights = np.exp(alpha * f_s) / np.sum(np.exp(alpha * f_s))
        grad = np.zeros_like(h_vec)
        for i, df in enumerate(df_da):
            grad += weights[i] * df
        return -grad / alpha

    # Final local refinement of candidates
    best_h = None
    best_c5 = float('inf')
    
    for h_candidate in candidates:
        if time.time() - start_time > budget_s - 10:
            break
            
        try:
            h_scaled = h_candidate / np.sum(h_candidate) * n / 2
            
            # First, try fine tuning with increasingly smooth alpha
            for alph in [5, 15, 40]:
                try:
                    def obj(h_in):
                        return smooth_objective(h_in, alph)
                    def grad(h_in):
                        return grad_smooth(h_in, alph)
                    res = minimize(obj, h_scaled, method='trust-constr', jac=grad,
                                   bounds=[(0, 1)] * n, 
                                   constraints=({'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2}), 
                                   options={'maxiter': 200, 'gtol': 1e-8})
                    if res.success:
                        h_ref = res.x
                        h_ref = h_ref / np.sum(h_ref) * n / 2
                    else:
                        h_ref = h_scaled
                except:
                    h_ref = h_scaled
                h_scaled = h_ref
            
            # Compute true C5
            c5_true = evaluate_c5(h_scaled)
            if c5_true < best_c5:
                best_c5 = c5_true
                best_h = h_scaled.copy()
                
        except Exception as e:
            continue
    
    # Use fallback strategy if needed
    if best_h is None:
        try:
            h_init = np.ones(n) * 0.5
            h_init = h_init * n / 2 / np.sum(h_init)
            
            def obj_smooth(h_vec):
                dx = 2.0 / n
                f_s = []
                for s in range(1, n):
                    val = 0
                    for i in range(n - s):
                        val += h_vec[i] * (1 - h_vec[i + s])
                    f_s.append(val * dx)
                soft_sum = np.sum(np.exp(5.0 * np.array(f_s)))
                return -np.log(soft_sum) / 5.0
            
            res = minimize(obj_smooth, h_init, method='L-BFGS-B', bounds=[(0, 1)] * n, 
                           options={'maxiter': 500})
            if res.success:
                h_final = res.x
                h_final = h_final / np.sum(h_final) * n / 2
                best_c5 = evaluate_c5(h_final)
                best_h = h_final
        except:
            pass
    
    if best_h is None:
        best_h = np.ones(n) * 0.5
        best_h = best_h * n / 2 / np.sum(best_h)
        best_c5 = evaluate_c5(best_h)

    return (best_h.tolist(), best_c5, n)
