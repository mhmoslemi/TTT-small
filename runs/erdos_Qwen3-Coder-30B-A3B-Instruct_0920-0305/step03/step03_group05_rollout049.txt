def compute_q_values(h):
    """Compute the maximum of correlated values (i.e., min of h(x)(1-h(x+k)) dx)."""
    h_np = np.array(h)
    # Compute the convolution (h conv 1-h)
    corr = np.correlate(h_np, 1 - h_np, mode='full')
    return np.max(corr)

def solve_milp_binary(n, time_limit=100):
    """Solve MILP to find optimal binary h for C5 minimization."""
    dx = 2.0 / n
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()

    # The constraints of the mixed integer program:
    constraints = [cp.sum(h) == n // 2]

    # Linearization of h_i * h_{i+d} as y_{i,d}
    y_vars = {}

    for d in range(n):
        # Index i runs from 0 to n-1-d to ensure i+d < n
        if d == 0:
            # For d=0, we overlap with itself
            for i in range(n):
                y_vars[(i, 0)] = cp.Variable(boolean=True)
                constraints.append(y_vars[(i,0)] <= h[i])
                constraints.append(y_vars[(i,0)] <= h[i])
                constraints.append(y_vars[(i,0)] >= 2*h[i] - 1)
        else:
            # For d > 0, we have index positions i such that i+d < n
            for i in range(n - d):
                y_vars[(i, d)] = cp.Variable(boolean=True)
                # y_{i,d} <= h_i
                constraints.append(y_vars[(i, d)] <= h[i])
                # y_{i,d} <= h_{i+d}
                constraints.append(y_vars[(i, d)] <= h[i + d])
                # y_{i,d} >= h_i + h_{i+d} - 1
                constraints.append(y_vars[(i, d)] >= h[i] + h[i + d] - 1)

    # Build constraints: for each shift d, Q_d = sum y_{i,d}
    # Then we will add constraint Q_d >= t for all d
    Q_values = []
    for d in range(n):
        if d == 0:
            # Sum of y for i=0 to n-1 for d=0
            Q = cp.sum([y_vars[(i, 0)] for i in range(n)])
        else:
            # Sum of y for i=0 to n-1-d for d > 0
            Q = cp.sum([y_vars[(i, d)] for i in range(n - d)])
        Q_values.append(Q)
        constraints.append(Q >= t)

    # solve the MILP
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)

    # Solve with CBC and specified time limit
    try:
        prob.solve(solver=cp.CBC, time_limit=time_limit)
        if prob.status in [cp.OPTIMAL, cp.SOLVED]:
            h_sol = h.value
            t_opt = t.value
            if h_sol is not None and t_opt is not None:
                c5_opt = 1 - 2/n * t_opt
                return h_sol, c5_opt
    except Exception:
        pass
    return None, float('inf')

def hill_climb_swaps(h_init, max_swaps=200000, time_limit=None):
    """Improve binary solution with swaps between 0 and 1."""
    start_time = time.time()
    h = np.array(h_init, dtype=float)
    dx = 2.0 / len(h)
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1 - h)

    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = 1 - dx * np.max(corr)

    count = 0
    while count < max_swaps:
        if time_limit is not None and time.time() - start_time > time_limit - 2:
            break

        # randomly pick one 1-index and one 0-index to swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # swap values
        h[i], h[j] = h[j], h[i]

        # recompute the overlap
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = 1 - dx * np.max(corr)

        if c5 < best_c5:
            best_h[:] = h[:]
            best_c5 = c5

        # revert swap
        h[i], h[j] = h[j], h[i]

        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)

        count += 1

    return best_h, best_c5

def continuous_refine(h_init, n, time_limit=5):
    """Patch the discrete step function with continuous method using projected gradient."""
    h = h_init.copy()
    dx = 2.0 / n
    # adjust to ensure integral constraint is satisfied
    current_sum = np.sum(h) * dx
    h *= (1.0 / current_sum)
    
    start_time = time.time()
    iter_count = 0
    while time.time() - start_time < time_limit:
        iter_count += 1
        # compute overlap (min of correlations)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = 1 - dx * np.max(corr)

        # project onto interval [0,1]
        h = np.clip(h, 0, 1)
        # keep only the integral constraint
        h = h * (1.0 / np.sum(h) * (n / 2))
        
        if iter_count > 1000:
            break
            
    return h, c5

def run(seed=42, budget_s=600):
    """Main function to find minimal C5 using MILP and hill climbing."""
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None

    for n in [48, 56, 64, 72]:
        if time.time() - start_time > 0.95 * budget_s:
            break

        T_remaining = budget_s - (time.time() - start_time)
        if T_remaining <= 10:
            break

        # Allocate 60% of remaining time for this run
        time_limit = min(120, int(0.6 * T_remaining))

        h_sol, c5_sol = solve_milp_binary(n, time_limit=time_limit)

        if c5_sol < best_c5:
            best_h = h_sol
            best_c5 = c5_sol
            best_n = n

    if best_h is not None:
        # improve by hill climb step
        if time.time() - start_time < budget_s * 0.9:
            h_improved, c5_improved = hill_climb_swaps(
                best_h, max_swaps=100000, time_limit=budget_s - (time.time() - start_time) - 5
            )
            if c5_improved < best_c5:
                best_h, best_c5 = h_improved, c5_improved

    return list(best_h), best_c5, best_n
