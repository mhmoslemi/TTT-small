import numpy as np
from scipy.optimize import minimize
import time

# Parameters
n_points = 80
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_to_simplex_box(y, target):
    """Project y onto the set where 0 <= x <= 1 and sum(x) = target."""
    lo = min(y) - 1.0
    hi = max(y)
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2, 0, 1)
    # Adjust rounding errors
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0, 1)
    return x

            if elapsed > budget_s * 0.95:

    # Final check
    best_h = project_to_simplex_box(best_h, n_points / 2)
    final_c5 = evaluate_h(best_h)
    
    return list(best_h), final_c5, n_points
