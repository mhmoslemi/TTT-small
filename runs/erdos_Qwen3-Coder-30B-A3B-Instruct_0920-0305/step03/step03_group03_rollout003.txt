import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Track best solution
    best_h = np.array(initial_h_values)
    best_c5 = evaluate_c5(best_h)
    n_best = len(best_h)
    
    # Function to project onto the probability simplex
    def project_simplex(v, s=1.0):
        # Project onto the simplex {x : x >= 0, sum x = s}
        # Using the algorithm from Duchi et al. (2008)
        v = np.array(v)
        n = len(v)
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0]
        if len(rho) == 0:
            theta = 0.0
        else:
            theta = (cssv[rho[-1]] - s) / (rho[-1] + 1)
        return np.maximum(v - theta, 0)

    # Exact binary optimization: Solve MILP for n = 42
    def solve_binary_milp(n, time_limit=None):
        # Variable declaration
        m = n // 2
        dx = 2.0 / n
        
        b = cp.Variable(n, boolean=True)
        y = cp.Variable((n, n), boolean=True)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(b) == m,
        ]
        
        # Product linearization for all i,j
        for i in range(n):
            for j in range(i, n):
                constraints.append(y[i, j] <= b[i])
                constraints.append(y[i, j] <= b[j])
                constraints.append(y[i, j] >= b[i] + b[j] - 1)

        # Overlap constraints
        for s in range(n):
            overlap = cp.sum([y[i, i + s] for i in range(n - s)])
            constraints.append(overlap >= t)
            
        # Objective
        objective = cp.Maximize(t)
        
        # Problem
        prob = cp.Problem(objective, constraints)
        
        # Convert to string for error handling if needed
        try:
            if time_limit:
                prob.solve(solver=cp.GLPK_MI, time_limit=time_limit)
            else:
                prob.solve(solver=cp.GLPK_MI)
            if prob.status == 'optimal':
                b_sol = np.array([float(b[i].value) for i in range(n)])
                t_val = t.value
                c5_binary = (m - t_val)*dx
                return b_sol, c5_binary
        except:
            pass
        
        return None, None

    # Binary optimization for n=42
    elapsed = time.time() - start_time
    time_left = budget_s * 0.3 - elapsed
    if time_left > 0.1:
        # Use MILP solver for n=42
        try:
            b_sol, c5_binary = solve_binary_milp(42, time_limit=time_left*0.9)
            if b_sol is not None and c5_binary < best_c5:
                best_h = b_sol
                best_c5 = c5_binary
                n_best = 42
        except:
            pass

    # Stochastic combinatorial search on larger grids
    for n in [44, 48, 52, 56, 60]:
        if time.time() - start_time > budget_s * 0.6:
            break
            
        elapsed = time.time() - start_time
        time_left = budget_s * 0.3 - elapsed
        if time_left <= 0.1:
            break
        
        # Number of iterations for SA
        max_iter = int(10*n*n)
        current_iter = 0
        t0 = 1.0
        alpha = 0.95
        T_schedule = t0 * (alpha ** np.arange(max_iter))
        
        # Initialize binary vector with weight n//2
        h = np.zeros(n)
        h[:n//2] = 1
        np.random.shuffle(h)
        
        best_h_bin = h.copy()
        best_c5_current = evaluate_c5(h)
        
        t_idx = 0
        while t_idx < max_iter and (time.time() - start_time) < budget_s * 0.95:
            # Swap random 1 and 0
            idx_1 = np.where(h == 1)[0]
            idx_0 = np.where(h == 0)[0]
            
            if len(idx_0) == 0 or len(idx_1) == 0:
                break
                
            i = np.random.choice(idx_1)
            j = np.random.choice(idx_0)
            
            h[i], h[j] = h[j], h[i]
            
            c5_new = evaluate_c5(h)
            if c5_new < best_c5_current:
                best_h_bin = h.copy()
                best_c5_current = c5_new
            
            t_idx += 1

        # Refinement step
        if best_c5_current < best_c5:
            h_cont = best_h_bin.copy()
            t_refine = 0
            while t_refine < 300 and (time.time() - start_time) < budget_s:
                corr_val = correlate(h_cont, 1 - h_cont, mode="full")
                max_i = np.argmax(corr_val)
                s_ind = max_i - (len(h_cont) - 1)
                
                # Approximate the gradient at indices of shift s_ind
                grad = np.zeros_like(h_cont)
                for i in range(len(h_cont)):
                    i_s = (i + s_ind) % len(h_cont)
                    i_s_neg = (i - s_ind) % len(h_cont)
                    grad[i] = 1 - h_cont[i_s] - h_cont[i_s_neg]
                
                # Take gradient step
                lr = 1e-2
                h_new = h_cont - lr * grad
                
                # Project to [0,1]
                h_new = np.clip(h_new, 0, 1)
                
                # Project to simplex
                h_new = project_simplex(h_new)
                
                # Evaluate new value
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5_current:
                    h_cont = h_new
                    best_c5_current = c5_new
                else:
                    h_cont = h_new
                
                t_refine += 1
            
            if best_c5_current < best_c5:
                best_h = h_cont
                best_c5 = best_c5_current
                n_best = n

    # Final polishing with gradient descent
    def project_and_refine(h_start, max_iter=1000):
        h = h_start.copy()
        for i in range(max_iter):
            if (time.time() - start_time) > budget_s * 0.95:
                break
                
            corr_val = correlate(h, 1 - h, mode="full")
            max_i = np.argmax(corr_val)
            s_ind = max_i - (len(h) - 1)
            
            # Compute gradient at shift s_ind
            grad = np.zeros_like(h)
            for i in range(len(h)):
                i_s = (i + s_ind) % len(h)
                i_s_neg = (i - s_ind) % len(h)
                grad[i] = 1 - h[i_s] - h[i_s_neg]
            
            # Gradient step
            lr = 1e-2
            h_new = h - lr * grad
            
            # Project to feasible set
            h_new = np.clip(h_new, 0, 1)
            h_new = project_simplex(h_new)
            
            # Evaluations
            c5_new = evaluate_c5(h_new)
            if c5_new < evaluate_c5(h):
                h = h_new
            else:
                break
        return h
    
    if (time.time() - start_time) < budget_s * 0.9:
        refined = project_and_refine(best_h, 500)
        refined_c5 = evaluate_c5(refined)
        if refined_c5 < best_c5:
            best_h = refined
            best_c5 = refined_c5

    # Fallback to differential evolution finalizing
    if (time.time() - start_time) < budget_s * 0.9:
        bounds = [(0, 1) for _ in range(n_best)]
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            dx = 2.0 / n_best
            return np.sum(h) * dx - 1.0
        
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            de_res = differential_evolution(objective, bounds, seed=seed+1000, maxiter=500)
            if de_res.success:
                h_candidate = de_res.x
                dx = 2.0 / n_best
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
            
    return (best_h.tolist(), best_c5, n_best)
