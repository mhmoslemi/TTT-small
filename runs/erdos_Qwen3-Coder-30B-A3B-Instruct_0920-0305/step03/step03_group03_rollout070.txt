import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_binary_milp(n, time_limit):
    """Solve the binary MILP formulation for given n"""
    m = n // 2
    # Binary variables
    b = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    
    # Constraints
    constraints = [cp.sum(b) == m]
    
    # Define all y variables for products
    Y = cp.Variable((n, n), boolean=True)
    
    # Add product constraints
    for i in range(n):
        for j in range(n):
            if i <= j:
                constraints += [
                    Y[i, j] <= b[i],
                    Y[i, j] <= b[j],
                    Y[i, j] >= b[i] + b[j] - 1
                ]
    
    # For each shift s, compute overlap_s = sum_{i} Y[i, i+s]
    overlap = []
    for s in range(n):
        if s == 0:
            overlap.append(cp.sum(b))
        else:
            # Sum over valid pairs
            overlap_s = cp.sum([Y[i, i+s] for i in range(n-s)])
            overlap.append(overlap_s)
            
    # Add constraints that each overlap is at least t
    for s in range(n):
        constraints.append(overlap[s] >= t)
    
    # Objective: maximize t (minimum overlap among all shifts)
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    try:
        prob.solve(solver=cp.CBC, time_limit=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_EOF]:
            # Extract solution
            b_sol = b.value.round().astype(int)
            t_sol = int(t.value)
            return b_sol, t_sol
        return None, None
    except:
        return None, None
