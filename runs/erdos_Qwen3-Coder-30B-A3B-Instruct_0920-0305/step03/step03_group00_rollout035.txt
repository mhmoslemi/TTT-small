import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Minimize, Variable, sum, at_most, Bool

def weight_normalise(x, n):
    """Rescale x to satisfy sum(x) = n/2"""
    return x * (n / 2) / np.sum(x)

def max_defect(h):
    """Compute C5 and corresponding shift"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    best_shift = np.argmax(corr)
    # Correct shift index from full correlation output
    shift = best_shift - (len(h) - 1)
    return max_corr, shift

def softmax_objective(h, alpha):
    """Evaluate log-sum-exp approximation to max and its gradient"""
    n = len(h)
    dx = 2.0 / n
    
    # Compute all f_s(h) values over shifts s = 1, ..., n-1
    corr = correlate(h, 1 - h, mode="full")
    corr = corr[len(h)-1:]  # Only positive shifts 1, ..., n-1

    # Softmax approximation
    exp_vals = np.exp(alpha * corr)
    Z = np.sum(exp_vals)
    if Z == 0:
        # Should not happen in practice
        Phi = np.max(corr)
        grad = np.zeros_like(h)
    else:
        weights = exp_vals / Z
        Phi = np.log(Z) / alpha

        # Gradient
        grad = np.zeros_like(h)
        # Map shifts back into h indices for gradient computation
        for s, weight in enumerate(weights):
            if s >= 1 and s <= n-1:
                # Weighted contribution from shift s
                # Note: we are computing d/dh_j of sum_s(w_s * f_s)
                # f_s = sum_i h[i] * (1 - h[i+s]) for i+i<s constraint
                # f_s = sum_{i : i+s < n} h[i] * (1 - h[i+s])
                # For each i, df_s/dh_i = weight * (1 - h[i+s] if i+s < n)
                # For each i, df_s/dh_{i+s} = weight * (-h[i] if i < n-s)
                # So accumulate contributions into dPhi/dh_j
                for i in range(n - s):
                    if i < n:
                        grad[i] += weight * (1.0 - h[i + s])
                    if i + s < n:
                        grad[i + s] -= weight * h[i]
        grad = grad / alpha  # Adjust by alpha scaling

    return Phi * dx, grad

def build_milp(n):
    """Build a mixed-integer program to find the binary h that maximizes min overlap"""
    h = Variable(n, boolean=True)
    w = Variable((n, n - 1), nonneg=True)
    t = Variable()

    # Common constraints
    constraints = [
        sum(h) == n/2,  # weight constraint
        t <= 1.0,  # bound
    ]

    # AND constraints:
    # w[i,s] <= h[i], w[i,s] <= h[i+s] for s = 1..n-1, i=0..n-1-s
    # w[i,s] >= h[i] + h[i+s] - 1
    for s in range(1, n):
        for i in range(n - s):
            constraints.append(w[i, s-1] <= h[i])
            constraints.append(w[i, s-1] <= h[i + s])
            constraints.append(w[i, s-1] >= h[i] + h[i + s] - 1)
        # Ensure unused slots are zero
        for i in range(n - s, n):
            constraints.append(w[i, s-1] == 0)
    
    # Minimum overlap constraint:
    for s in range(1, n):
        constraints.append(sum(w[:, s-1]) >= t)

    # Objective: maximize t
    problem = cp.Problem(Maximize(t), constraints)
    
    solver_opts = {"maxTime": 180, "seconds": 180, "ratioGap": 0.0}
    try:
        problem.solve(solver=solver=solver_opts)
        # Extract optimal binary solution
        h_opt = h.value
        t_opt = t.value
        return h_opt.astype(float), t_opt
    except:
        # If solution is not found, return last known feasible
        # The solution process by CBC is expensive, best to return None if too soon
        return None, 0.0

def refine_continuous(h_bin, n, alpha_schedule=[5, 15, 40]):
    """Refinement stage using smooth max approximation"""

    # Identify boundaries
    h_cont = h_bin.copy()
    h_cont = np.clip(h_cont, 0, 1)
    # Base initial h on h_bin, then relax interior 1's and boundary transitions
    h0 = np.zeros(n)
    # Only allow fractional values around the transition zones
    # Identify blocks of zeros and ones from binary h
    # For faster and safe run, simplify by starting directly:
    # repeat h_bin, then refine via smooth max
    
    h0 = h_bin.copy()
    
    # Step 1: refine the edges with a coarse-to-fine schedule
    h_opt = h0.copy()

    for alpha in alpha_schedule:
        def objective(h):
            val, grad = softmax_objective(h, alpha)
            return val

        def gradient(h):
            val, grad = softmax_objective(h, alpha)
            return grad

        # Optimize with trust-constr, assuming equality constraints are handled
        # Let's just use a simple approximation for equalities, just rescale when violated
        def top_level_optimisation():
            bounds = [(0.0, 1.0)] * n
            constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}

            # Initial guess is already h0
            h_guess = h_opt.copy()
            # Ensure it is valid, in case bounds make it invalid
            h_guess = np.clip(h_guess, 0, 1)
            # Renormalize
            h_guess = weight_normalise(h_guess, n)

            try:
                # Use the gradient
                res = minimize(objective, h_guess, method='trust-constr',
                               jac=gradient, bounds=bounds, constraints=constraints,
                               options={'gtol': 1e-6, 'maxiter': 200})
                if res.success:
                    h_opt_new = res.x
                    h_opt = h_opt_new
                else:
                    # In case it fails, fallback
                    h_opt = h_guess
                
            except Exception as e:
                h_opt = h_guess
            return h_opt

        h_opt = top_level_optimisation()
        # Final clipping after optimization
        h_opt = np.clip(h_opt, 0, 1)
        # Rescale to exactly sum to n/2
        h_opt = weight_normalise(h_opt, n)
        
    return h_opt

def anneal(h, n_points, max_iter=5000):
    """Swap anneling to refine the worst shift"""
    
    # Initialize correlation matrix for the actual max defect computation
    corr_orig = correlate(h, 1 - h, mode='full')
    max_def = np.max(corr_orig)  # This is the C5 constant after normalization
    max_shift = np.argmax(corr_orig) - (n_points - 1)
    
    h = h.copy()
    h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    def compute_max_defect():
        corr = correlate(h, 1-h, mode="full")
        return np.max(corr), np.argmax(corr) - (n_points - 1)
        
    # Sampling approach: run multiple cycles of swaps
    cooling_rate = 0.99
    gamma = 0.2
    for i in range(max_iter):
        # step 1: find the maximal defect shift
        max_def, shift = compute_max_defect()
        
        # Step 2: identify candidates contributing to the max defect
        # corr[i] = sum over all j of h[j] * (1 - h[j+shift])  
        # So it reflects f_shift(h) = sum_i h[i] * (1 - h[i+shift])
        # Those i where h[i] > 0 and h[i + shift] low contribute to the off-diagonal
        # In practice, compute from scratch or use incremental update
        corr = correlate(h, 1-h, mode="full")
        indicative_shift = shift
        if indicative_shift < 0:
            # negative shift means shift_backward
            active_index = np.arange(n_points + indicative_shift)  # <n_points+shift
            defended_from = np.arange(-indicative_shift, n_points)
        else:
            active_index = np.arange(n_points - indicative_shift)
            defended_from = np.arange(indicative_shift, n_points)
        # The i that contribute to max_defect(shift)
        contrib_i = active_index[np.where(h[active_index] > 0.5) and (h[active_index + indicative_shift] < 0.5)]
        # Most relevant elements
        contrib_i = contrib_i[:int(len(contrib_i)/2)]
        # Further select candidates, e.g. those contributing to the maximum
        
        # Let target_i be values that are maybe big (contributing to overlap)
        # Let target_j be small values and small further out
        # Find target endings
        h_shifted = np.roll(h, -indicative_shift, axis=0)
        discr_val = corr[len(h) - 1 + indicative_shift]
        # pseudo breaks at meaningful contributions
        
        # Target indices from bad regions around shift
        good_indices = np.where(h < 0.2 - 1e-3)  # low less than small
        low_indices = good_indices[0]
        # Low values which may not yet be maximally damaged (to convert)
        bad_orient = np.roll(h, -indicative_shift, axis=0)
        targets = np.where((h_shifted < 0.1) & (h < 0.9))[0]
        # If there's no good target, go easier
        if len(targets) == 0:
            targets = np.where((h_shifted < 0.5) & (h < 0.9))[0]
        # Simple pairing
        if len(contrib_i) > 0 and len(targets) > 0:
            i_idx = np.random.choice(contrib_i)
            j_idx = np.random.choice(targets)
            if i_idx + indicative_shift < len(h) and j_idx + indicative_shift < len(h):
                if h[i_idx] > h[j_idx]:
                    delta = min(h[i_idx], 1.0 - h[j_idx]) * gamma
                    h[i_idx] -= delta
                    h[j_idx] += delta
                    # Update shift effects
                    # Optionally, compensate for both shift effects
                    if h[i_idx + indicative_shift] < 1.0:
                        h[i_idx + indicative_shift] += delta
                    if h[j_idx + indicative_shift] < 1.0:
                        h[j_idx + indicative_shift] -= delta
        gamma *= cooling_rate

        if i % 20 == 0:
            # Periodic max recomputation
            max_def, shift = compute_max_defect()
            
    # renormalize
    h = weight_normalise(h, n_points)
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Step 0: Set up iters based on memory
    n = 100  # Reasonable starting size; adjust if time too short
    dx = 2.0 / n
    
    # Stage 1: Binary MILP 
    h_milp, t_opt = None, 0.0
    if (time.time() - start_time) < (budget_s * 0.3):
        try:
            h_milp, t_opt = build_milp(n)
        except Exception as e:
            h_milp = None
            
    # If MILP didn't work, fallback to random restarts
    if h_milp is None: 
        # Fallback pure continuous optimization
        h = np.random.rand(n)
        h = weight_normalise(h, n)
        h_opt = h.copy()
        # Do simple refinement using all constraints
        h_opt = refine_continuous(h_opt, n)
        # Sample random starting points for more robustness
        best_h = h_opt.copy()
        best_c5 = max_defect(h_opt)[0]
        for i in range(30):
            if (time.time() - start_time) > 0.75 * budget_s:
                break
            h = np.random.rand(n)
            h = weight_normalise(h, n)
            h_f = refine_continuous(h, n)
            val, shift = max_defect(h_f)
            if val < best_c5:
                best_c5 = val
                best_h = h_f.copy()
    else:
        # Run through the stages described
        h_bin = h_milp
        # Evaluate its value
        t_val = np.sum(h_bin) - 0.5 * np.inner(h_bin, np.roll(h_bin, 1))
        best_c5 = 1.0 - 2.0/n * t_val  # approximate C5 from t
        h_opt = refine_continuous(h_bin, n)
        # Phase 3: Final swap-annealing polish
        h_final = anneal(h_opt, n)
        best_h, best_c5 = h_final, max_defect(h_final)[0]
    
    # Final adjustment: confidence on exact max
    best_h = weight_normalise(best_h, n)
    actual_c5 = max_defect(best_h)[0] * dx  # final correct C5
    
    return (best_h.tolist(), actual_c5, n)
