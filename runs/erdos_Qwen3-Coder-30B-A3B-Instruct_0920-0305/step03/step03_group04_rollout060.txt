import numpy as np
from scipy.optimize import minimize
import time

# Constants
n_points = 70  # Even number for binary construction
dx = 2.0 / n_points
target_bound = 0.3807
max_iter = 1000

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def smooth_max(values, alpha=20.0):
    """Smooth approximation to max using log-sum-exp."""
    max_val = np.max(values)
    # subtract to avoid overflow
    exps = np.exp(alpha * (values - max_val))
    return max_val + np.log(np.sum(exps)) / alpha

def smooth_objective(h_values):
    """Smooth objective function to minimize."""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    return smooth_max(correlation)

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def run(seed=42, budget_s=600):
    """Main optimization routine."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
    else:
        # Default initialization
        h_current = np.random.uniform(0, 1, n_points)
        h_current = h_current / np.sum(h_current) * (n_points / 2)
        h_current = np.clip(h_current, 0, 1)
    
    best_h = h_current.copy()
    best_c5 = evaluate_h(h_current)
    
    # Main optimization loop with multiple restarts
    iter_num = 0
    max_iter_total = 2000
    
    while time.time() - start_time < budget_s * 0.95 and iter_num < max_iter_total:
        # Local optimization with smooth objective
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            result = minimize(smooth_objective, h_current, method='SLSQP', 
                             bounds=bounds, constraints=constraints,
                             options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-10})
            if result.success:
                h_current = result.x
                c5_current = evaluate_h(h_current)
                if c5_current < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_current
        except:
            pass
        
        # Random perturbation for exploration
        h_test = h_current.copy()
        indices = np.random.choice(n_points, size=max(1, int(0.1 * n_points)), replace=False)
        for i in indices:
            h_test[i] = np.clip(h_test[i] + np.random.uniform(-0.1, 0.1), 0, 1)
        
        # Renormalize to satisfy sum constraint (process to 1, 1 approx of all to avoid negative)
        sum_h = np.sum(h_test)
        if abs(sum_h - n_points/2) > 1e-6:
            h_test *= (n_points / 2) / sum_h
        
        c5_test = evaluate_h(h_test)
        if c5_test < best_c5:
            best_h = h_test.copy()
            best_c5 = c5_test
        
        iter_num += 1
        if best_c5 < target_bound:
            break
    
    # Final refinement using SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        result = minimize(smooth_objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-10})
        if result.success:
            best_h = result.x
    except:
        pass
    
    final_c5 = evaluate_h(best_h)
    
    # Final adjustment to ensure sum constraint
    final_h = best_h.copy()
    sum_h = np.sum(final_h)
    if abs(sum_h - n_points/2) > 1e-6:
        final_h = final_h * (n_points / 2) / sum_h
    final_c5 = evaluate_h(final_h)
    
    return list(final_h), final_c5, n_points
