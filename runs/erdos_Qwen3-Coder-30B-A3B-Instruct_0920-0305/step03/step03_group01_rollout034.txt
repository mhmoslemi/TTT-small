def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value scaled by dx
    max_corr = np.max(corr)
    return max_corr * dx

def generate_block_vector(uvec, n_blocks, n_total):
    # Generate a step function from block heights uvec
    # uvec should have length n_blocks. Each block gets n_total/n_blocks samples
    # If n_total isn't divisible by n_blocks, distribute the remainder around
    blocksize = n_total // n_blocks
    remainder = n_total % n_blocks
    h = np.empty(n_total)
    start = 0
    for i in range(n_blocks):
        # Distribute remainder to blocks
        actual_blocksize = blocksize + (1 if i < remainder else 0)
        h[start:start + actual_blocksize] = uvec[i]
        start += actual_blocksize
    return h

def cmaes_global_search(n_points=200, n_blocks=10, max_evals=1500, seed=42):
    np.random.seed(seed)
    # Parametrization: unconstrained block heights u_j ∈ ℝ  
    # Constraint: sum(u_j) = m/2, where m = n_blocks  
    # We reduce the last one manually
    def obj(uvec):
        # clamp uvec to valid bounds, may be invalid
        uvec = np.clip(uvec, 0, 1)
        # Rearrange back to full vector with correct total
        u_m = 0.5 * n_blocks - np.sum(uvec[:-1])
        uvec = np.append(uvec[:-1], np.clip(u_m, 0, 1))
        h = generate_block_vector(uvec, n_blocks, n_points)
        return evaluate_c5(h)
    
    import cma
    # Slightly simpler starting point
    initial_popsize = max(4, n_blocks + 4)
    es = cma.CMAEvolutionStrategy(
        np.full(n_blocks - 1, 0.5),
        0.2,
        {'popsize': initial_popsize, 'seed': seed}
    )
    
    best_val = float('inf')
    best_h = None
    for _ in range(max_evals // initial_popsize):
        try:
            X = es.ask()
            vals = [obj(x) for x in X]
            es.tell(X, vals)
            idx = np.argmin(vals)
            if vals[idx] < best_val:
                best_val = vals[idx]
                uvec = X[idx]

                h = generate_block_vector(uvec, n_blocks, n_points)
                best_h = h
        except Exception as e:
            print(f'Error in CMA-ES iteration: {e}')
            break
    return best_h

def compute_overlap_deltas(h, s, i, j):
    # handles boundary conditions with zero padding
    n = len(h)
    delta = 0.0
    
    def get_h(idx):
        if 0 <= idx < n:
            return h[idx]
        return 0.0

    # Compute new contributions for the terms affected by swapping i and j
    # For each shift k:
    # Before: h[i] * h[i+k], h[j] * h[j+k]  
    # After:  0 * h[i+k], h[j] * h[j+k]  
    #                      or h[i] * h[i+k], 0 * h[j+k]

    for k in range(n):  
        i_k = i + k
        j_k = j + k
        
        # Before swap contribution to overlap k
        i_old = get_h(i) * get_h(i_k)
        j_old = get_h(j) * get_h(j_k)
        
        # After swap contribution
        i_new = 0.0 * get_h(i_k)  # h[i] is now 0
        j_new = get_h(j) * get_h(j_k)  # h[j] still here
        # We have the key term from only one of i or j having become zero
        if i < n and i_k < n and i_old != 0:
            delta += (i_old - i_new)  # This should be -h[i_k] 
        if j < n and j_k < n and j_old != 0:
            delta += (j_old - j_new)  # This should be +h[j_k] 

    # Then the most powerful update:
    # Let's compute difference in 1 - sum(h[i](1-h[i+k])) which gives
    # sum((h[i](1-h[i+k])) for i,k) when replacing i by 0, j by 1
    delta = 0.0
    # i goes from 1 to 0, so it removes h[i] * h[i+k] from each shift where valid
    # j goes from 0 to 1, so it adds h[j] * h[j+k] for valid shifts where h[j]=0
    # We sum over all shifts k (0..n-1), testing if indices are in range
    
    for k in range(n):
        # Contribution for h[i] = 1 is -1 * h[i+k]
        # Contribution for h[j] = 0 is +1 * h[j+k]
        i_k = i + k
        j_k = j + k
        
        if 0 <= i_k < n and 0 <= i < n:
            delta += -h[i_k]
        if 0 <= j_k < n and 0 <= j < n:
            delta += h[j_k]

    return delta

def projected_gradient_descent(h_init, max_iter=3000, alpha0=0.01):
    h = h_init.copy()
    n = len(h)
    dx = 2.0 / n
    
    def project_onto_feasible(x):
        # Project onto [0,1]^n and then onto sum = n/2
        # Use water filling
        y = np.copy(x)
        # Bisection to find lambda
        def sum_under_lambda(lam):
            return np.sum(np.clip(y - lam, 0, 1))
        
        # Clip lower bound and simplify logic
        y = np.clip(y, 0, 1) # First step
        # Now solve for lambda such that sum = n/2
        # Use a simple iterative algorithm or billionth degree bisection
        lower = -2.0
        upper = 3.0
        
        for _ in range(50):  # A few iterations should suffice
            midpoint = (lower + upper) / 2.0
            val = sum_under_lambda(midpoint)
            if val < n/2:
                upper = midpoint
            else:
                lower = midpoint
        
        lambda_adjust = (lower + upper) / 2.0
        x_proj = np.clip(y - lambda_adjust, 0, 1)
        # Normalize to correct mean (this should be exactly n/2)
        scale = (n / 2.0) / np.sum(x_proj)
        return x_proj * scale
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Step size schedule
    def alpha(t):
        return alpha0 / np.sqrt(t + 1)
    
    for t in range(max_iter):
        # Compute overlap for all overlaps
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        # The "max_k" corresponds to shift k
        # shift as number is from 1 to n-1 
        shift = max_k - n + 1  # Convert index to shift
        
        # Compute gradient component-wise for shift shift (k_idx = shift + n - 1)
        g = np.empty_like(h)
        for i in range(n):
            # Reverse logic: if shift is positive, we add to i, check range
            idx1 = i
            idx2 = i + shift
            g[i] = 1 - (h[idx2] if 0 <= idx2 < n else 0) - (h[idx1-shift] if 0 <= idx1-shift < n else 0)
        
        h_new = h - alpha(t) * g
        h = project_onto_feasible(h_new)
        
        new_c5 = evaluate_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
        if t > 200 and (best_c5 - new_c5) < 1e-6:
            break  # Stopping criterion
            
    return best_h, best_c5
