import numpy as np

def proj_box_simplex(v):
    """Project vector v onto the intersection of [0,1]^n and {x: sum x = 1}"""
    v = np.array(v)
    n = len(v)
    # Sort v in descending order
    u = np.sort(v)[::-1]
    # Compute cumulative sums
    sv = np.cumsum(u)
    # Find the threshold
    rho = (sv - 1) / np.arange(1, n + 1)
    # Find the index where u[i] > rho[i]
    idx = np.where(u > rho)[0]
    if len(idx) == 0:
        tau = rho[-1]
    else:
        tau = rho[idx[-1]]
    # Compute the projection
    x = np.maximum(v - tau, 0)
    x = x / np.sum(x)
    return x

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def spectral_search(n, seed, budget_s):
    np.random.seed(seed)
    t0 = time.time()
    dx = 2.0 / n
    m = n / 2

    best_c5 = float('inf')

    remaining_budget = budget_s * 0.8  # Leave some margin for final polishing
    
    for restart in range(R):
        if time.time() - t0 > remaining_budget:
            break
            
        # Random start with simplex projection
        u = np.random.exponential(1, n)
        h0 = u / np.sum(u) / dx  # scale to integral = 1
        h0 = np.clip(h0, 0, 1)
        h0 = proj_box_simplex(h0)
        
        # compute target magnitude from initial vector
        H0 = np.fft.fft(h0)
        M = np.mean(np.abs(H0[1:]))
        # Safety: ensure M is not too close to zero to avoid degeneracy
        M = max(1e-6, M)
        
        h = h0.copy()
        prev_c5 = float('inf')
        for t in range(it_max):
            # FFT
            H = np.fft.fft(h)
            # Magnitude enforcement
            H_new = H.copy()
            H_new[1:] = M * np.exp(1j * np.angle(H[1:]))
            # Inverse FFT
            h_new = np.real(np.fft.ifft(H_new))
            # Project onto feasible set
            h = proj_box_simplex(h_new)
            # Convergence check (very basic)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if abs(c5 - prev_c5) < 1e-8:
                break
            prev_c5 = c5
            
            if time.time() - t0 > remaining_budget:
                break
    
    return best_h, best_c5

def binary_refinement(h):
    """Refine continuous h into binary-like pattern"""
    n = len(h)
    # Sort and get indices
    idxs = np.argsort(h)[::-1]
    # Find threshold for sum = n/2
    ones = int(n / 2)
    # This could improve with combinatorial search for exact count
    h_binary = np.zeros(n)
    h_binary[idxs[:ones]] = 1
    # polish with small swaps
    c5 = evaluate_c5(h_binary)
    if c5 > 0.3809:
        # Try swaps to improve
        for _ in range(5 * n):
            if np.random.rand() < 0.5:  # sometimes skip to avoid too many iterations
                continue
            i, j = np.random.choice(n, 2, replace=False)
            while h_binary[i] == 0:
                i = np.random.choice(n)
            while h_binary[j] == 1:
                j = np.random.choice(n)
            # swap
            h_binary[i], h_binary[j] = 1, 0
            c5_new = evaluate_c5(h_binary)
            if c5_new >= c5:
                # revert
                h_binary[i], h_binary[j] = 0, 1
            else:
                c5 = c5_new
    return h_binary
