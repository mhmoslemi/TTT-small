def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h):
    """Project h to satisfy constraints: 0<=h<=1 and sum(h)*dx=1"""
    n = len(h)
    dx = 2.0 / n

    # Apply box constraints
    h = np.clip(h, 0, 1)

    # Adjust to meet sum constraint using waterfilling
    def waterfilling(target_sum):
        # We want sum of h to be target_sum
        # h[i] = clip(y[i] - lambda, 0, 1)
        # The sum of clipped values y - lambda is sum(y) - n*lambda
        # If sum(y) - n*lambda = target_sum, then lambda = (sum(y) - target_sum) / n

        # First, get the initial sum
        h_sum = np.sum(h)

        # If already satisfied, no need to adjust
        if abs(h_sum - target_sum) < 1e-12:
            return h.copy()

        # Use bisection to find lambda
        low = -10.0  # safety
        high = 10.0
        for _ in range(100):
            mid = (low + high) / 2.0
            trial = np.clip(h - mid, 0, 1)
            trial_sum = np.sum(trial)
            if abs(trial_sum - target_sum) < 1e-10:
                return trial
            elif trial_sum > target_sum:
                low = mid
            else:
                high = mid
        return np.clip(h - (np.sum(h) - target_sum) / n, 0, 1)

    return waterfilling(1.0)
