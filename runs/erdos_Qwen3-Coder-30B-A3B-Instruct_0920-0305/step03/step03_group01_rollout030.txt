import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_onto_box_and_hyperplane(y, target):
    """Project onto box [0,1] and hyperplane sum = target using binary search"""
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0, 1)

def compute_gradient(h, shift):
    """Compute gradient of correlation at given shift"""
    n = len(h)
    if shift >= 0:
        term1 = np.empty(n)
        term1[:n-shift] = 1 - h[shift:]
        term1[n-shift:] = 1
        term2 = np.empty(n)
        term2[shift:] = h[:n-shift]
        term2[:shift] = 0
    else:
        s = -shift
        term1 = np.empty(n)
        term1[s:] = 1 - h[:n-s]
        term1[:s] = 1
        term2 = np.empty(n)
        term2[:n-s] = h[s:]
        term2[n-s:] = 0
    return term1 - term2

def gradient_descent_step(h, alpha):
    """Perform one step of gradient descent for C5 minimization"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    max_ind = np.where(corr == max_corr)[0]
    shift = max_ind[0] - (n - 1)
    
    grad = compute_gradient(h, shift)
    h_new = h - alpha * grad
    
    # Project back to feasible region
    target_sum = n / 2.0
    h_new = project_onto_box_and_hyperplane(h_new, target_sum)
    
    return h_new

def tent_shape(n):
    """Create a symmetric tent shape function"""
    x = np.linspace(0, 2, n)
    center = 1.0
    width = 1.0
    h = np.maximum(0, 1 - np.abs(x - center) / (width / 2))
    # Normalize to integral = 1
    h = h * (1.0 / np.trapz(h, x))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 200  # Using more points for better resolution
    max_iter = 5000
    base_alpha = 0.05
    
    # Initialize with tent shape
    h0 = tent_shape(n_points)
    # Ensure sum constraint (should be integral = 1)
    dx = 2.0 / n_points
    h0 = h0 * (1.0 / (np.sum(h0) * dx))  # Scale to integral = 1
    
    # Track best solution
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    iteration = 0
    last_improvement = 0
    no_improvement_count = 0
    
    # Multi restart scheme
    num_restarts = 10
    restart_interval = max_iter // num_restarts
    
    while (time.time() - start_time) < budget_s * 0.95 and iteration < max_iter:
        # Try multiple restarts
        for restart in range(num_restarts):
            if (time.time() - start_time) > budget_s * 0.95:
                break
                
            # Random restart
            if restart > 0:
                h = np.random.uniform(0, 1, n_points)
                h = h / (np.sum(h) * dx) * (n_points / 2)
            else:
                h = h0.copy()
            
            # Gradient descent restart
            for i in range(restart_interval):
                if (time.time() - start_time) > budget_s * 0.95:
                    break
                    
                # Decreasing step size schedule
                alpha = base_alpha / np.sqrt(1 + iteration)
                h_new = gradient_descent_step(h, alpha)
                
                # Check improvement
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    last_improvement = iteration
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
                    
                h = h_new
                iteration += 1
                
                # Occasionally add random noise to escape local minima
                if no_improvement_count > 50 and (time.time() - start_time) < budget_s * 0.9:
                    h = h + np.random.normal(0, 0.005, n_points)
                    h = project_onto_box_and_hyperplane(h, n_points / 2)
                    no_improvement_count = 0
        
        # Early termination check
        if iteration - last_improvement > 2000 and best_c5 < 0.3807:
            break
    
    # Convert to consistent format
    return (best_h.tolist(), best_c5, n_points)
