import numpy as np

from scipy.signal import correlate
import time

def proj_box_simplex(v):
    """
    Project vector v onto the intersection of the box [0,1]^n and the simplex {x : sum x_i = 1}
    """
    v = np.asarray(v)
    n = len(v)
    # Sort v in descending order
    vsort = np.sort(v)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(vsort)
    # Find the threshold
    k = np.argmax(vsort * np.arange(1, n+1) > (cumsum - 1))
    if k == 0:
        tau = (cumsum[0] - 1) / 1
    else:
        tau = (cumsum[k-1] - 1) / k
    # Project onto [0,1] and normalize
    proj = np.clip(v - tau, 0, 1)
    # Renormalize to sum to 1
    proj = proj / np.sum(proj)
    return proj

def evaluate_c5(h):
    """Evaluate C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

            h = proj_box_simplex(h_tilde)

            last_c5 = c5

def binary_refinement(h):

    n = len(h)
    m = 1.0  # integral constraint
    
    # Round to nearest 0.5
    b = (h > 0.5).astype(int)
    # Adjust to correct weight (binary version)
    current_weight = np.sum(b)
    diff = int(m) - current_weight
    if diff > 0:
        # Need to flip some 0s to 1s
        idx_0 = np.where(b == 0)[0]
        if len(idx_0) >= diff:
            flip_idx = np.random.choice(idx_0, diff, replace=False)
            b[flip_idx] = 1
    elif diff < 0:
        # Need to flip some 1s to 0s
        idx_1 = np.where(b == 1)[0]
        if len(idx_1) >= -diff:
            flip_idx = np.random.choice(idx_1, -diff, replace=False)
            b[flip_idx] = 0
    
    # Now local hill climb
    current_h = b.astype(float)
    current_c5 = evaluate_c5(current_h)
    improvement = True
    attempts = 0
    max_attempts = 5 * n
    
    while improvement and attempts < max_attempts:
        improvement = False
        attempts += 1
        
        # Get 1s and 0s
        ones = np.where(current_h == 1)[0]
        zeros = np.where(current_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        # Try a few random swaps
        for _ in range(10):
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Swap
            trial_h = current_h.copy()
            trial_h[i] = 0
            trial_h[j] = 1
            
            trial_c5 = evaluate_c5(trial_h)
            if trial_c5 < current_c5:
                current_h = trial_h
                current_c5 = trial_c5
                improvement = True
                break
    
    return current_h, current_c5

def fallback_optimization(n_points, budget, t0):
    """Fallback optimization with random search + local polishing"""
    best_h = None
    best_c5 = float('inf')
    
    remaining = budget - (time.time() - t0)
    if remaining < 1:
        return None, float('inf')
    
    # Generate random candidates
    num_candidates = min(200, int(remaining*5))
    for _ in range(num_candidates):
        u = np.random.exponential(1.0, n_points)
        h = u / np.sum(u)
        h = h * 1.0 / np.sum(h)  # Ensure sum = 1
        h = np.clip(h, 0, 1)
        h = proj_box_simplex(h)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Local polishing using projected gradient descent
    if best_h is not None and best_c5 < float('inf'):
        h = best_h.copy()
        c5 = evaluate_c5(h)
        
        # Simple gradient-based local refinement
        for _ in range(100):
            if (time.time() - t0) > budget * 0.95:
                break
                
            # Basic finite difference gradient
            eps = 1e-5
            grad = np.zeros_like(h)
            f0 = evaluate_c5(h)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = proj_box_simplex(h_plus)
                
                grad[i] = (evaluate_c5(h_plus) - f0) / eps
            
            # Gradient step with projection
            lr = 1e-3
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = proj_box_simplex(h_new)
            
            c5_new = evaluate_c5(h_new)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
            else:
                break
    
    return best_h, best_c5
