import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import math

# Global constants
target_bound = 0.3807
n_values_to_try = [48, 60, 72, 84]

def dx(n): return 2.0 / n

def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-x))

def logit(p):
    return np.log(p / (1.0 - p))

def proj_to_simplex(v, S=None):
    if S is None: S = v.sum()
    v = np.clip(v, 0, 1)
    s = v.sum() - S
    if abs(s) < 1e-12: return v
    if s > 0:
        # must clip some entries to 0
        for _ in range(100):
            pos = v > 0
            if not pos.any(): break
            v[pos] -= s / pos.sum()
            if v[pos].min() >= 0: break
            v[np.where(pos)[0][0]] = 0.
    elif s < 0:
        # must clip some entries to 1
        for _ in range(100):
            pos = v < 1
            if not pos.any(): break
            v[pos] += s / pos.sum()
            if v[pos].max() <= 1: break
            v[np.where(pos)[0][0]] = 1.
    return np.clip(v, 0, 1)

def correlation_vec(h):
    return correlate(h, 1-h, mode='full')

def evaluate(h):
    c = correlation_vec(h)
    return dx(len(h)) * np.max(c)

def smooth_max(c, alpha=30.0):
    m = np.max(c)
    if m == -np.inf:
        return m
    c_shifted = alpha * (c - m)
    return m + np.log(np.sum(np.exp(c_shifted))) / alpha

def penalty_sum(h, lambda_sum=10000.0):
    return lambda_sum * (np.sum(h) - len(h)/2)**2

def binary_initial(n):
    h = np.zeros(n)
    ones = np.random.permutation(n)[:n//2]
    h[ones] = 1
    return h

def swap_and_evaluate(h, i_one, i_zero):
    h_new = h.copy()
    h_new[i_one] = 0
    h_new[i_zero] = 1
    return evaluate(h_new)

def binary_hill_climb(h0, max_iter=50000, T=None):
    n = len(h0)
    h = h0.copy()
    c0 = evaluate(h)
    best_h = h.copy()
    best_c0 = c0
    one_indices = np.where(h == 1)[0]
    zero_indices = np.where(h == 0)[0]

    for i in range(max_iter):
        if T and time.time() > T:
            break
        if len(one_indices) == 0 or len(zero_indices) == 0:
            break
        i_one = np.random.choice(one_indices)
        i_zero = np.random.choice(zero_indices)
        h_new = h.copy()
        h_new[i_one] = 0
        h_new[i_zero] = 1
        c_new = evaluate(h_new)
        if c_new < best_c0:
            best_h = h_new.copy()
            best_c0 = c_new
        if c_new < c0:
            # Accept improvement
            h = h_new
            c0 = c_new
            one_indices = np.where(h == 1)[0]
            zero_indices = np.where(h == 0)[0]
        else:
            # Don't accept the swap, but update lists anyway
            one_indices = np.where(h == 1)[0]
            zero_indices = np.where(h == 0)[0]

    return best_h, best_c0

def smooth_obj(y, h0_ref, alpha=30.0, lam=10000.0):
    h = sigmoid(y)
    c = correlation_vec(h)
    return smooth_max(c, alpha) + penalty_sum(h, lam)

def smooth_grad(y, h0_ref, alpha=30.0, lam=10000.0):
    h = sigmoid(y)
    c = correlation_vec(h)
    # Gradient of smooth_max
    m = np.max(c)
    c_shifted = alpha * (c - m)
    exps = np.exp(c_shifted)
    dsoft = exps / np.sum(exps)
    # Chain rule for sigmoid
    s = sigmoid(y)
    ds = s * (1 - s)
    # Derivative of c w.r.t. h
    # For correlation c[i] = sum_{k} h[i-k] * (1 - h_k)
    # Entry d(c[i])/d(h[j]) = sign(1 or 0) depending on if i==j+k or i+k==j
    # Conceptually, hardcoding the entries is safer for small sizes
    # So we leave it as is, rely on gradient approximation via finite diffs

    # Approximate gradient
    eps = 1e-4
    grad_summ = np.zeros_like(y)
    for i in range(len(y)):
        y_plus = y.copy()
        y_plus[i] += eps
        y_minus = y.copy()
        y_minus[i] -= eps
        object_plus = smooth_obj(y_plus, h0_ref, alpha, lam)
        object_minus = smooth_obj(y_minus, h0_ref, alpha, lam)
        grad_summ[i] = (object_plus - object_minus) / (2*eps)
    return grad_summ

def smooth_optimize(h0, maxiter=300, n_restarts=8):
    n = len(h0)
    best_h = h0.copy()
    best_c0 = evaluate(h0)
    for _ in range(n_restarts):
        # Start from perturbed binary vector
        y = logit(h0.copy()) + np.random.normal(0, 0.2, n)
        sol = minimize(smooth_obj, y, args=(h0,), method='L-BFGSB',
                       options={'maxiter': maxiter, 'ftol': 1e-12},
                       bounds=[(-20, 20)]*n)
        if sol.success:
            h_opt = sigmoid(sol.x)
            h_opt = proj_to_simplex(h_opt)
            c_opt = evaluate(h_opt)
            if c_opt < best_c0:
                best_h = h_opt
                best_c0 = c_opt
        else:
            # If L-BFGS fails, we might start over - just use the current value
            pass
    return best_h, best_c0

def exact_optimize(h0, maxiter=500):
    n = len(h0)
    def obj(h):
        return evaluate(h)

    bounds = [(0, 1) for _ in range(n)]
    con = {'type': 'eq', 'fun': lambda x: np.sum(x)*dx(n) - 1.0}

    sol = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=con,
                   options={'ftol': 1e-12, 'maxiter': maxiter})

    if sol.success:
        return sol.x, evaluate(sol.x)
    return h0, evaluate(h0)

def parametric_interval_family_search(n, max_m=8):
    best_h = binary_initial(n)
    best_c = evaluate(best_h)
    local_dx = dx(n)
    for m in range(2, max_m+1):
        ell = 1.0 / m
        max_g = (2.0 - m * ell) / (m - 1) if m > 1 else 0
        for g in [0.0, 0.5*max_g, 0.75*max_g]:
            if max_g <= 0: g = 0
            num_points_per_interval = int(ell / local_dx + 0.5)
            for delta in np.arange(0, g+local_dx/2, local_dx/2):
                h = np.zeros(n)
                start = int(delta / local_dx + 0.5)
                for i in range(m):
                    left = start + i*(num_points_per_interval + int(g / local_dx + 0.5))
                    right = left + num_points_per_interval
                    left = max(0, left)
                    right = min(n, right)
                    if right > left:
                        h[left:right] = 1
                # Normalize
                h = proj_to_simplex(h, S=n/2)
                c = evaluate(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
    return best_h, best_c
