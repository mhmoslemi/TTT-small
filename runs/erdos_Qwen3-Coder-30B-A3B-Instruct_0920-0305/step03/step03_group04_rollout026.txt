import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time
import cvxpy as cp
from cvxpy import SCS

# Constants
n_vals = [36, 42, 48, 54, 60]
n_points = 48  # Default to best performing
target_bound = 0.3807
dx = 2.0 / n_points

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def correlation_vec(h_values):
    """Return the full correlation vector (length 2*n-1) for h_values."""
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    return corr

def logsumexp(x, axis=None, keepdims=False):
    """Numerically stable logsumexp function."""
    x_max = np.max(x, axis=axis, keepdims=True)
    return x_max + np.log(np.sum(np.exp(x - x_max), axis=axis, keepdims=keepdims))

def smooth_max(corr_vec, alpha=30):
    """Compute smooth approximation to max(corr_vec)."""
    return logsumexp(alpha * corr_vec) / alpha

def proj_to_simplex(v, s=1.0):
    """Project vector v onto the simplex {x : sum(x) = s, x ≥ 0}"""
    # Sort v in descending order
    v_sorted = np.sort(v)[::-1]
    # Compute cumulative sums in descending order
    cs = np.cumsum(v_sorted)
    # Find the index of the largest θ such that v_sorted[i] - θ > 0
    # This is equivalent to θ = (cs[i] - s) / (i+1) for the case that the i+1 elements are active
    # and the threshold τ = max(v_sorted[i] - (cs[i] - s) / (i+1) for i in range(len(v_sorted)))
    
    # Alternative numerical approach:
    # Start from the largest j where v_sorted[j] > (cs[j] - s) / (j+1)
    # the largest j s.t. v_sorted[j] > (cs[j] - s) / (j+1) is such that
    # for all i <= j,  v_sorted[i] >= (cs[i] - s) / (i+1)
    # so (cs[j] - s) / (j+1) <= v_sorted[j]
    # and z =  (cs[j] - s) / (j+1) for that j
    # The projection should be closest to original v in L2, so we can compute:
    # z = (sum(v) - s) / (length of v)
    # But here we use the correct body for simplex projection via clipping.
    
    # More precise:
    # Let z = (sum(v) - s) / n
    # But that's wrong, we want:
    # Let x be the reconstructed simplex
    # we want x = argmin ||z-v||^2 s.t. sum(x) = s, x_i >= 0
    # solution via auxilliary variables:
    # One can show the solution is:
    # v_i = max(0, x_i), where x_i = v_i - lambda + s/n, and lambda solves the equation below.
    
    # In fact, use the more robust way:

    n = len(v)
    v_sorted = np.sort(v)[::-1]
    cumsums = np.cumsum(v_sorted)
    # Find theta such that sum( max(0, v_sorted[i] - theta) ) = s
    theta = np.inf
    for i in range(n):
        # theta = (cumsums[i] - s) / (i+1)
        z = (cumsums[i] - s) / (i+1)
        if z >= v_sorted[i]:
            # this is not the correct z for this index, continue to i+1
            continue
        else:
            # The first one satisfying more conditions, and such that
            # v_sorted[i] - z >= 0, that is, z <= v_sorted[i]
            if i == 0:
                if (cumsums[i] - s) >= 0:
                    # all might be truncated to some value; try z=0
                    theta = 0
                else:
                    theta = z
            else:
                theta = z
            break
    if np.isinf(theta):
        theta = (np.sum(v) - s) / n
    # Now apply the threshold projection:
    result = np.maximum(v - theta, 0)
    # Adjust for sum
    norm_factor = s / np.sum(result)
    result = np.clip(result * norm_factor, 0, 1)
    return result

def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-x))

def logit(x, eps=1e-6):
    x = np.clip(x, eps, 1-eps)
    return np.log(x / (1 - x))

def sdp_relaxation(n, max_time=30):
    """Solve SDP relaxation of the binary problem."""
    # Variables
    x = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        0 <= x,
        x <= 1,
        cp.sum(x) == n/2,
        X >> 0,  # PSD constraint
        cp.diag(X) == x,
    ]

    # Add McCormick constraints for all pairs i < j
    for i in range(n):
        for j in range(i+1, n):
            constraints += [
                X[i,j] >= 0,
                X[i,j] <= x[i],
                X[i,j] <= x[j],
                X[i,j] >= x[i] + x[j] - 1,
            ]

    # Add overlap constraints for all shifts k
    for k in range(n):
        # Sum over h[i] for i=0 to n-1-k, minus sum over X_{i,i+k}
        expr = cp.sum(x[0: n-k]) - cp.sum(cp.diag(X, k))
        constraints.append(expr <= t)

    # Objective
    objective = cp.Minimize(t)

    # Form problem
    prob = cp.Problem(objective, constraints)

    # Solve using SCS
    try:
        start_time = time.time()
        prob.solve(solver=SCS, max_iters=10000, eps=1e-6)
        end_time = time.time()
        if end_time - start_time > max_time:
            return None, None
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_sol = x.value
            X_sol = X.value
            t_sol = t.value
            return x_sol, X_sol, t_sol
        else:
            print(f"SDP solve failed for n={n}, status: {prob.status}")
            return None, None, None
    except Exception as e:
        print(f"SDP solve error for n={n}: {e}")
        return None, None, None

def round_from_sdp(x_hat, X_hat, n, n_iter=30):
    """Apply random hyperplane rounding."""
    # Decompose X_hat ≈ V @ V^T
    try:
        # Use SVD to get low-rank approximation
        U,s,Vt = np.linalg.svd(X_hat)
        # Keep only significant components
        rank = min(10, n)
        # Choose r components
        V = U[:, :rank] * np.sqrt(s[:rank])
        # Now V[i,:] projects the i-th vector into low-d space
        best_h = None
        best_c5 = np.inf
        for _ in range(n_iter):
            g = np.random.randn(rank)
            s_num = np.sign(V @ g)  # s in [-1, 1]
            # Convert s back to {0,1} by mapping 1 to 1; -1 to 0.
            h(candidate) = (s_num > 0).astype(float)
            if np.sum(h) != n/2:
                # Correct sum by flipping
                diff = np.sum(h) - n/2
                if diff > 0:  # Too many 1s
                    # Flip the diff smallest absolute differences from 1 to 0
                    indices_1 = np.where(h == 1)[0]
                    diff_from_zero = np.abs(V[indices_1, :] @ g)
                    flip_indices = np.argsort(diff_from_zero)[:int(diff)]
                    h[indices_1[flip_indices]] = 0
                else:  # Too few 1s
                    # Flip the |diff| smallest from 0 to 1
                    indices_0 = np.where(h == 0)[0]
                    diff_from_zero = np.abs(V[indices_0, :] @ g)
                    flip_indices = np.argsort(diff_from_zero)[:int(-diff)]
                    h[indices_0[flip_indices]] = 1
            c5 = evaluate_h(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        return best_h, best_c5
    except:
        return None, np.inf

def hill_climb_binary(h_start, max_iter=50000):
    """Hill-climb on binary values using swaps."""
    h = h_start.copy()
    n = len(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    c5_current = evaluate_h(h)
    # Normal cooling rate
    cooling = 0.9999999  # Or flat
    for i in range(max_iter):
        # Random swap between one and zero
        if len(ones) > 0 and len(zeros) > 0:
            i_one = np.random.choice(ones)
            i_zero = np.random.choice(zeros)
            # Create new h
            h_new = h.copy()
            h_new[i_one] = 0.0
            h_new[i_zero] = 1.0
            c5_new = evaluate_h(h_new)
            if c5_new < c5_current:
                # Accept
                h = h_new
                c5_current = c5_new
                ones.remove(i_one)
                zeros.remove(i_zero)
                ones.append(i_zero)
                zeros.append(i_one)
            else:
                # rare chance to accept despite worsening
                pass  # No tolerance for acceptance
        else:
            break
    # Check if the overshoot of 1 ones is better by fixing it: 
    # It might have improved in an intermediate step.
    # But let us just accept that and return.
    return h, c5_current

def smooth_refinement(h_input, alpha=30, max_iter=300, lambda_reg=10):
    """Smooth approximation using logsumexp."""

    # Wrap the objective so it works for optimization tools
    def smooth_obj(y):
        h = sigmoid(y)
        corr = correlation_vec(h)
        soft = smooth_max(corr, alpha)
        penalty = lambda_reg * (np.sum(h) - n_points/2)**2
        return soft + penalty

    # Back to our h_input to starting point
    y = logit(h_input - 1e-6)
    # Optimization
    try:
        # Use L-BFGS-B for efficient bound-constrained optimization
        result = minimize(smooth_obj, y, method='L-BFGS-B', jac=None, options={'maxiter': max_iter})
        if result.success:
            y_opt = result.x
            h = sigmoid(y_opt)
            # Project to simplex
            h_proj = proj_to_simplex(h, s=n_points/2)
            return h_proj
        else:
            return h_input  # fallback
    except:
        return h_input

def run(seed=42, budget_s=600):
    """Comprehensive algorithm:
    1. SDP relaxation
    2. Rounding
    3. Hill-climb swap optimization
    4. Smooth approximation refinement
    5. Final polishing with SLSQP
    """
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = np.inf
    n_best = n_points

    # Keep a cache of n-values that worked
    cache = {}

    # Phase A: SDP Relaxations
    for n in n_vals:
        if time.time() - start_time > budget_s * 0.95:
            break

        print(f"Phase A: solving SDP for n={n}")
        x_sol, X_sol, t_sol = sdp_relaxation(n, max_time=30)
        if x_sol is None:
            print(f"Rejecting n={n} due to failed SDP.")
            continue
        cache[n] = {'x': x_sol, 'X': X_sol, 't': t_sol}
        print(f"SDP for n={n} yielded lower bound t={t_sol:.6f}")
    
    # Phase B: Random Hyperplane Rounding  
    if not cache:
        # If no SDP then start with random binary hill-climb
        print("No SDP results, initialize random starts.")
        
    # Collect all rounds
    rounds = []
    
    for n in sorted(cache.keys()):
        x_sol = cache[n]['x']  # Feasible fractional solution
        X_sol = cache[n]['X']
        # Round to binary
        h_rounded, c5_rounded = round_from_sdp(x_sol, X_sol, n, n_iter=40)
        if h_rounded is not None:
            rounds.append((h_rounded.copy(), c5_rounded, n))
            print(f"Rounded sample for n={n}, C5={c5_rounded:.6f}")
        else:
            print(f"Rounding failed for n={n}")

    # For the rest, do hill-climb from different random starts if not enough
    if not rounds:
        rounds = []
        n_samples = 10
        n = n_best  # keep default
        for idx in range(n_samples):
            h_rand = np.random.rand(n)
            h_rand = (h_rand > 0.5).astype(float)  # Binary
            # Force sum = n/2
            current_sum = np.sum(h_rand)
            diff = current_sum - n/2
            if abs(diff) > 1e-6:
                flipped = int(abs(diff))
                from_idx = np.where(h_rand == (1 if diff > 0 else 0))[0]
                if flipped > len(from_idx):
                    flipped = len(from_idx)
                to_flip = np.random.choice(from_idx, size=flipped, replace=False)
                h_rand[to_flip] = 1 - h_rand[to_flip]
            rounds.append((h_rand.copy(), evaluate_h(h_rand), n))

    # Phase C: Hill-climbing on best roundings 
    rounds_by_c5 = sorted(rounds, key=lambda x: x[1])
    best_h_early = rounds_by_c5[0][0].copy()  # first one
    best_c5_early = rounds_by_c5[0][1]
    n_from_round = rounds_by_c5[0][2]

    print("Phase C: hill-climb")
    h_climb, c5_climb = hill_climb_binary(best_h_early, max_iter=50000)
    if c5_climb < best_c5_early:
        best_h_early = h_climb
        best_c5_early = c5_climb
        n_from_round = n_from_round
    print(f"Hill-climbed result: C5 = {best_c5_early:.6f}")

    # Store the best so far 
    if best_c5_early < best_c5:
        best_c5 = best_c5_early
        best_h = best_h_early.copy()
        n_best = n_from_round

    if best_c5 <= target_bound:
        print("Early termination: bound met.")
    else:
        # Phase D: Smooth approximation refinement
        print("Phase D: smooth approximation refinement")
        try:
            h_refined = smooth_refinement(best_h.copy(), alpha=30, max_iter=300)
            c5_refined = evaluate_h(h_refined)
            if c5_refined < best_c5_early:
                best_h = h_refined
                best_c5 = c5_refined
                print(f"Smooth refinement improved C5 to {best_c5:.6f}")
        except Exception as e:
            print(f"Smooth refinement failed: {e}")

    # Phase E: Final polishing with SLSQP
    # Use the most suitable prior solution to restart
    print("Phase E: exact polishing")
    try:
        h_final = best_h.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}
        
        def objective(x):
            return evaluate_h(x)

        # Final umbrella step with SLSQP
        result = minimize(objective, h_final, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-10})

        if result.success:
            h_final = result.x
            c5_final = evaluate_h(h_final)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
        else:
            print("SLSQP failed.")
    except Exception as e:
        print(f"SLSQP polishing failed: {e}")

    # Ensure normalization
    np.clip(best_h, 0, 1, out=best_h)
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        norm = 1.0 / (np.sum(best_h) * dx)
        best_h *= norm

    print(f"Final best C5: {best_c5:.6f} with n={n_best}")
    return list(best_h), best_c5, n_best
