import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time

# Constants
n_values = [48, 60, 72, 84]
target_bound = 0.3807
dx_list = [2.0 / n for n in n_values]

def evaluate_h(h):
    """Evaluate the C_5 bound for given h values."""
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    return max_correlation * 2.0 / len(h)

def correlation_vec(h):
    """Return the full correlation array."""
    return np.correlate(h, 1 - h, mode="full")

def smooth_max(c, alpha=30):
    """Soft maximum using log-sum-exp."""
    m = np.max(c)
    if np.isinf(m) or np.isnan(m):
        return m
    diff = c - m
    exp_diff = np.exp(alpha * diff)
    return m + np.log(np.sum(exp_diff)) / alpha

def penalty_sum(h, lam=10000):
    """Quadratic penalty for integral constraint."""
    return lam * (np.sum(h) - len(h) / 2) ** 2

def proj_to_simplex(v, S=None):
    """Project vector onto probability simplex."""
    if S is None:
        S = len(v) / 2
    v = np.clip(v, 0, 1)
    s = np.sum(v)
    if abs(s - S) < 1e-10:
        return v
    if s > S:
        # Shrink to sum S
        w = v - (s - S) / len(v)
        w = np.clip(w, 0, 1)
        return w
    else:
        # Too small, rescale
        scale = S / s
        return v * scale

def sigmoid(x):
    """Sigmoid function."""
    return 1 / (1 + np.exp(-x))

def logit(p):
    """Logit function."""
    return np.log(p / (1 - p))

def binary_initial(n):
    """Generate initial random binary vector."""
    h = np.zeros(n)
    ones = np.random.choice(n, size=n//2, replace=False)
    h[ones] = 1
    return h

def swap_and_evaluate(h, i_one, i_zero):
    """Swap h[i_one] and h[i_zero] and evaluate new C5."""
    new_h = h.copy()
    new_h[i_one] = 0
    new_h[i_zero] = 1
    return evaluate_h(new_h)

def binary_hill_climb(n, max_iter=50000, max_stagnation=5 * n):
    """Binary hill climb with early stopping."""
    best_h = binary_initial(n)
    best_c5 = evaluate_h(best_h)
    best_c5_exact = best_c5
    one_indices = np.where(best_h == 1)[0].tolist()
    zero_indices = np.where(best_h == 0)[0].tolist()

    stagnation_count = 0
    for _ in range(max_iter):
        # Take several swaps if stagnating
        if stagnation_count > max_stagnation:
            # Reinitialize with different random binary
            new_h = binary_initial(n)
            new_c5 = evaluate_h(new_h)
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
                best_c5_exact = new_c5
                one_indices = np.where(best_h == 1)[0].tolist()
                zero_indices = np.where(best_h == 0)[0].tolist()
            stagnation_count = 0
            continue

        i_one = np.random.choice(one_indices)
        i_zero = np.random.choice(zero_indices)
        new_h = best_h.copy()
        new_h[i_one] = 0
        new_h[i_zero] = 1

        new_c5 = evaluate_h(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            best_c5_exact = new_c5
            one_indices.remove(i_one)
            one_indices.append(i_zero)
            zero_indices.remove(i_zero)
            zero_indices.append(i_one)
            stagnation_count = 0
        else:
            stagnation_count += 1

    return best_h, best_c5_exact

def smooth_max_optimization(h_init, alpha=30, lam=10000, maxiter=300, n_restarts=8):
    """Optimize using smooth max surrogate."""
    def objective(y, h0=h_init):
        h = sigmoid(y)
        c = correlation_vec(h)
        return smooth_max(c, alpha) + penalty_sum(h, lam)

    def gradient(y, h0=h_init):
        """Compute numerical gradient of objective."""
        h = sigmoid(y)
        c = correlation_vec(h)
        m = np.max(c)
        if m == -np.inf:
            return np.zeros_like(y)
        diff = c - m
        exp_diff = np.exp(alpha * diff)
        exp_sum = np.sum(exp_diff)
        if exp_sum < 1e-14:
            return np.zeros_like(y)
        grad = alpha * np.exp(alpha * diff) / exp_sum
        # Derivative of c wrt h (approximate sum fetching)
        corr_grad = np.zeros_like(h)
        for i in range(len(h)):
            corr_grad[i] = 0
            for k in range(len(h) - i):
                corr_grad[i] += (1 - h[i + k]) * (1 if k == 0 else 0)  # Simplified
            for k in range(i):
                corr_grad[i] += (1 - h[i - k]) * (1 if k == 0 else 0)  # Simplified

        # Compute gradient of the smooth max
        d_smooth = alpha * (grad - np.sum(grad)) / exp_sum
        d_sigmoid = h * (1 - h)
        return (d_smooth * d_sigmoid).sum()

    best_h = None
    best_c5 = np.inf

    y_start = logit(h_init)
    for i in range(n_restarts):
        if i > 0:
            # Add small noise to restart
            y_rnd = y_start + np.random.normal(0, 0.2, len(y_start))
            y_start = y_rnd
        try:
            sol = minimize(objective, y_start, method='L-BFGSB', options={'maxiter': maxiter})
            if sol.success:
                h = sigmoid(sol.x)
                h = proj_to_simplex(h)
                c5 = evaluate_h(h)
                if c5 < best_c5:
                    best_h = h
                    best_c5 = c5
        except Exception:
            continue

    if best_h is None:
        return h_init, evaluate_h(h_init)
    return best_h, best_c5

def structured_interval_search(n):
    """Search over a parametrized family of interval patterns."""
    dx = 2.0 / n
    best_c5 = np.inf
    best_h = None
    for m in range(2, 9):  # Number of intervals
        if 2 * m > n:
            continue
        l = 1.0 / m  # Length of each interval
        gap = (2.0 - m * l) / (m - 1.0)  # Max gap
        step = min(dx, gap * 0.5)
        for delta in np.arange(0.0, gap + step/2.0, step):
            h = np.zeros(n)
            start = delta
            for i in range(m):
                if start >= 2.0:
                    break
                end = min(start + l, 2.0)
                istart = int(np.round(start * n / 2.0))
                iend = int(np.round(end * n / 2.0))
                if iend > istart:
                    h[istart:iend] = 1
                start = end + gap
            if np.sum(h) >= 1e-5:
                h = proj_to_simplex(h, S=n / 2)
                c5 = evaluate_h(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=600):
    """Main optimization routine."""
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = np.inf
    best_n = None

    # Stage 1: Structured interval patterns
    for idx, n in enumerate(n_values):
        if time.time() - start_time > budget_s * 0.1:
            break
        h, c5 = structured_interval_search(n)
        if h is not None and c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = n
        # Early exit
        if c5 <= target_bound:
            break

    # Stage 2: Binary hill climbing with time slicing
    remaining_time = budget_s * 0.7
    for idx, n in enumerate(n_values):
        if time.time() - start_time > budget_s * 0.8 or remaining_time < 5:
            break
        remaining_time = budget_s * 0.7 - (time.time() - start_time)
        local_budget = min(remaining_time * 0.8 / len(n_values), 100)
        if local_budget < 1:
            break
        h_bit, c5_bit = binary_hill_climb(n, max_iter=10000)
        if c5_bit < best_c5:
            best_h = h_bit.copy()
            best_c5 = c5_bit
            best_n = n
        if c5_bit <= target_bound:
            break

    # Stage 3: Smooth max optimization
    for n in n_values:
        if time.time() - start_time > budget_s * 0.95:
            break
        h_smooth, c5_smooth = smooth_max_optimization(best_h.copy(), maxiter=200)
        if c5_smooth < best_c5:
            best_h = h_smooth.copy()
            best_c5 = c5_smooth
            best_n = n

    # Stage 4: Exact polishing
    # Reconstruct correlations and recalculate final
    bounds = [(0, 1) for _ in range(len(best_h))]
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) * 2.0 / len(x) - 1.0}
    try:
        result = minimize(evaluate_h, best_h, method='SLSQP', bounds=bounds,
                         constraints=constraints, options={'maxiter': 100, 'ftol': 1e-12})
        if result.success:
            final_h = result.x
            final_c5 = evaluate_h(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass

    return list(best_h), best_c5, best_n
