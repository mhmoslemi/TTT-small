```python
import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time
import cvxpy as cp
from cvxpy import SCS
import math

n_values = [36, 42, 48, 54, 60]
dxs = [2.0 / n for n in n_values]
target_c5 = 0.3807
max_iters_sdp = 8000
max_iters_hill = 50000
rounding_repetitions = 30
max_iters_smooth = 300
max_hill_cycles = 100000

def correlation_vec(h):
    """Compute the cross-correlation vector of h with (1 - h)."""
    h_len = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return corr

def evaluate_h(h):
    """Evaluate the C5 bound for given h values."""
    corr = correlation_vec(h)
    max_correlation = np.max(corr)
    return max_correlation * dx

def proj_to_simplex(v, s=1.0):
    """Project vector v onto the simplex with sum = s."""
    v = np.clip(v, 0, 1)
    v_sorted = np.sort(v)[::-1]
    rho = np.argmax(v_sorted.cumsum() > s)
    theta = (v_sorted.cumsum()[rho] - s) / (rho + 1)
    result = np.maximum(v - theta, 0)
    return result

def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-x))

def logit(x):
    x = np.clip(x, 1e-7, 1 - 1e-7)
    return np.log(x / (1 - x))

def sdp_relaxation(h_len, dx):
    """Solve the SDP relaxation and return a fractional solution."""
    # Create CVXPY variables
    x = cp.Variable(h_len)
    X = cp.Variable((h_len, h_len), PSD=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        0 <= x, x <= 1,
        cp.sum(x) == h_len / 2,
        cp.diag(X) == x
    ]

    # Add McCormick constraints
    for i in range(h_len):
        for j in range(i+1, h_len):
            constraints.extend([
                X[i, j] >= 0,
                X[i, j] <= x[i],
                X[i, j] <= x[j],
                X[i, j] >= x[i] + x[j] - 1
            ])

    # Add overlap constraints
    for k in range(h_len):
        # Sum of (h[i] * (1 - h[i + k])) for valid indices
        corr_k = cp.sum(x[:h_len - k]) - cp.sum(cp.diag(X, k))
        constraints.append(corr_k <= t)

    # Objective
    objective = cp.Minimize(t)

    # Solve the SDP
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=SCS, max_iters=max_iters_sdp, eps=1e-5)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value, X.value
    except:
        pass
    return None, None

def round_from_sdp(X, x, n_points):
    """Round the SDP solution to binary vector using randomized hyperplane rounding."""
    # Use a truncated SVD to get suitable rank for the factorization
    U, s, Vt = np.linalg.svd(X, full_matrices=False)
    # Keep only top 10 components
    V = U[:, :10] * np.sqrt(s[:10])
    best_h = None
    best_c5 = float('inf')
    
    for _ in range(rounding_repetitions):
        g = np.random.randn(V.shape[1])
        sgn = np.sign(V @ g)
        h_trial = (sgn >= 0).astype(float)
        # Fix the sum to be n/2
        current_sum = np.sum(h_trial)
        diff = int(current_sum - n_points / 2)
        if diff > 0:
            # Too many ones, flip diff smallest abs values
            idxs = np.argsort(np.abs(V @ g))
            flip_indices = idxs[:diff]
            h_trial[flip_indices] = 0
        elif diff < 0:
            # Too few ones, flip -diff smallest abs values
            idxs = np.argsort(np.abs(V @ g))
            flip_indices = idxs[-diff:]
            h_trial[flip_indices] = 1
        c5_trial = evaluate_h(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
    return best_h, best_c5

def hill_climb_binary(h, n_points, max_iters):
    """Perform a hill-climbing heuristic on the binary vector."""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    if len(ones) == 0 or len(zeros) == 0:
        return h, evaluate_h(h)

    best_h = h.copy()
    best_c5 = evaluate_h(h)

    for _ in range(max_iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h_trial = h.copy()
        h_trial[i] = 0
        h_trial[j] = 1

        c5_trial = evaluate_h(h_trial)
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]

    return best_h, best_c5

def smooth_optimize(h, n_points, dx):
    """Optimize a smooth surrogate of the objective using L-BFGS-B."""
    def soft_max_objective(y):
        h_vals = sigmoid(y)
        corr = correlation_vec(h_vals)
        # soft max with temperature 30
        temp = 30
        soft_max = np.log(np.sum(np.exp(temp * corr))) / temp
        # penalize constraint
        penalty = 1e3 * (np.sum(h_vals) - n_points / 2)**2
        return soft_max + penalty
    
    def objective_jac(y):
        h_vals = sigmoid(y)
        corr = correlation_vec(h_vals)
        temp = 30
        exp_corr = np.exp(temp * corr)
        exp_sum = np.sum(exp_corr)
        soft_grad = np.zeros_like(y)
        for i in range(len(y)):
            # derivative of sigmoid, logsumexp derivative
            sig_i = sigmoid(y[i])
            d_sig_i = sig_i * (1 - sig_i)
            d_logsum = np.sum(exp_corr * corr) / exp_sum
            soft_grad[i] = temp * d_logsum * d_sig_i + 2e3 * (np.sum(h_vals) - n_points / 2) * d_sig_i
        
        return soft_grad

    y_start = logit(h + 1e-6)
    result = minimize(soft_max_objective, y_start, method='L-BFGS-B', jac=objective_jac, options={'maxiter': max_iters_smooth})
    if result.success:
        h_opt = sigmoid(result.x)
        h_opt = proj_to_simplex(h_opt, n_points / 2)
        return h_opt, evaluate_h(h_opt)
    return h, evaluate_h(h)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None

    for n_idx, (n_point, dx) in enumerate(zip(n_values, dxs)):
        if time.time() - start_time > budget_s - 10:
            break

        x_vals, X_vals = sdp_relaxation(n_point, dx)
        if x_vals is None:
            continue

        h_binary, c5_binary = round_from_sdp(X_vals, x_vals, n_point)
        if h_binary is None:
            continue

        if c5_binary < best_c5:
            best_c5 = c5_binary
            best_h = h_binary.copy()
            best_n = n_point

        # Local search on binary solution
        h_local, c5_local = hill_climb_binary(h_binary, n_point, max_iters_hill)
        if c5_local < best_c5:
            best_c5 = c5_local
            best_h = h_local.copy()
            best_n = n_point

        # Smooth refinement
        h_smooth, c5_smooth = smooth_optimize(h_local, n_point, dx)
        if c5_smooth < best_c5:
            best_c5 = c5_smooth
            best_h = h_smooth.copy()
            best_n = n_point

        if best_c5 <= target_c5:
            break

    # Final polishing using SLSQP
    if best_h is not None:
        bounds = [(0, 1) for _ in range(best_n)]
        # Ensure the sum constraint
        def constraint_sum(h):
            return np.sum(h) - best_n / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        def obj_func(h):
            return evaluate_h(h)

        result = minimize(obj_func, best_h, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-8})

        if result.success:
            best_h = result.x
            best_c5 = evaluate_h(best_h)

    # Final normalization
    best_h = np.clip(best_h, 0, 1)
    if abs(np.sum(best_h) - best_n / 2) > 1e-6:
        best_h *= (best_n / 2) / np.sum(best_h)

    final_c5 = evaluate_h(best_h)
    return list(best_h), final_c5, best_n
```
