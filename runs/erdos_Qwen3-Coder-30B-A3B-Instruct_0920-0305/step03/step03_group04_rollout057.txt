```python
import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time
import cvxpy as cp
from cvxpy.atoms.affine.wraps import exp
import math

# Constants
target_bound = 0.3807
n_values = [36, 42, 48, 54, 60]
n_points = 48  # default, will be overwritten if SDP works
dx = 2.0 / n_points

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def correlation_vec(h_values):
    """Evaluate the full vector of correlations."""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    return correlation

def proj_to_simplex(y, S):
    """Projects vector y onto the simplex defined by sum(y) = S."""
    u = np.sort(y)[::-1]
    rho = np.where(u > (np.cumsum(u) - S) / np.arange(1, len(u) + 1))[0]
    if len(rho) == 0:
        theta = np.max(0, (np.sum(y) - S) / len(y))
        return np.clip(y - theta, 0, 1)
    else:
        i = rho[-1]
        theta = (np.sum(u[:i+1]) - S) / (i + 1)
        return np.clip(y - theta, 0, 1)

def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-x))

def logit(x):
    return np.log(x / (1.0 - x))

def smooth_max(c, alpha=30.0):
    """Log-sum-exp approximation of max."""
    return np.log(np.sum(np.exp(alpha * c))) / alpha

def sdplp_relaxation(n):
    """Solve the SDP relaxation for given n."""
    # Variables
    x = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        0 <= x,
        x <= 1,
        cp.sum(x) == n/2,
        cp.diag(X) == x,
        t >= 0
    ]
    
    # Add McCormick relaxation constraints for i<j
    idx_pairs = [(i,j) for i in range(n) for j in range(i+1,n)]
    for i, j in idx_pairs:
        constraints += [
            X[i,j] >= 0,
            X[i,j] <= x[i],
            X[i,j] <= x[j],
            X[i,j] >= x[i] + x[j] - 1
        ]
    
    # Add overlap constraints
    for k in range(n):
        if k == 0:
            sl = slice(0,n)
            X_sl = X[sl, sl]
        else:
            sl = slice(0, n-k)
            X_sl = X[sl, k:]
        constraints += [
            cp.sum(x[sl]) - cp.sum(cp.diag(X_sl)) <= t
        ]
    
    # PSD constraint
    constraints += [X >> 0]

    # Solve SDP with SCS
    try:
        problem = cp.Problem(cp.Minimize(t), constraints)
        # Limit max_iters to ensure time bound
        problem.solve(solver=cp.SCS, max_iters=5000, eps=1e-5)
        return problem.status, problem.value, x.value, X.value
    except Exception as e:
        return "infeasible", float('inf'), None, None

def rounding(X, x, R=30):
    """Rounding of SDP solution to binary vector."""
    # Mostly use eigen-value decomposition
    try:
        if np.linalg.matrix_rank(X) <= 10:
            U, s, _ = np.linalg.svd(X, full_matrices=False)
            V = U * np.sqrt(s)
        else:
            # Truncate to 10 principal components
            _, s, U = np.linalg.svd(X, full_matrices=False)
            U = U[:10]
            s = s[:10]
            V = U * np.sqrt(s)
            
        best_h = None
        best_c5 = float('inf')
        for seed in range(R):
            np.random.seed(seed)
            g = np.random.normal(0, 1, V.shape[1])
            sgn = np.sign(V @ g)
            h = np.array(sgn >= 0, dtype=float)
            # Adjust normalization
            diff = np.sum(h) - (len(h)/2)
            if diff > 0:
                # Find top diff smallest values and flip to 0
                flip_idx = np.argsort(np.abs(V @ g))[:int(diff)]
                for idx in flip_idx:
                    h[idx] = 0
            elif diff < 0:
                # Find top |diff| largest values and flip to 1
                flip_idx = np.argsort(-np.abs(V @ g))[:int(-diff)]
                for idx in flip_idx:
                    h[idx] = 1
                
            c5 = evaluate_h(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
                
        return best_h, best_c5
    except Exception as e:
        return None, float('inf')

def hill_climb_binary(h_start, max_iters=50000):
    """Perform combinatorial hill climb from binary solution."""
    h = h_start.copy()
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    c5_current = evaluate_h(h)
    num_swaps = 0
    restart_count = 0
    
    for _ in range(max_iters):
        if len(ones) < 1 or len(zeros) < 1:
            break
        # Pick random index to swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # Make swap
        h[i], h[j] = h[j], h[i]
        ones.remove(i)
        zeros.remove(j)
        ones.append(j)
        zeros.append(i)
        
        # Check if it's an improvement
        c5_new = evaluate_h(h)
        if c5_new < c5_current:
            c5_current = c5_new
        else:
            # Undo swap
            h[i], h[j] = h[j], h[i]
            ones.remove(j)
            zeros.remove(i)
            ones.append(i)
            zeros.append(j)
        
        num_swaps += 1
        if num_swaps % (20000) == 0:
            # Occasionally perform first improvement
            k_star = np.argmax(correlation_vec(h))
            if k_star > 0:
                # Get indices causing worst overlap
                mismatches = [i for i in range(len(h) - k_star) if h[i] == 1 and h[i+k_star] == 0]
                if len(mismatches) > 0:
                    for i in mismatches:
                        # Find random zero that can reduce mismatch
                        for j in zeros:
                            if j >= i and j <= i + k_star:
                                continue
                            old_h = h.copy()
                            h[i], h[j] = h[j], h[i]
                            c5_new = evaluate_h(h)
                            if c5_new < c5_current:
                                c5_current = c5_new
                                ones.remove(i)
                                zeros.remove(j)
                                ones.append(j)
                                zeros.append(i)
                                break
                            else:
                                h = old_h
            # If still not improving for a while, restart
            if num_swaps % (10000) == 0:
                restart_count += 1
                if restart_count > 3:
                    break
                
    return h, c5_current

def smooth_optimize(h_start, max_iter=300):
    """Optimize with smooth max surrogate."""
    def objective(y):
        h = sigmoid(y)
        c = correlation_vec(h)
        soft = smooth_max(c, alpha=30.0)
        # Penalty to stay on simplex
        penalty = 1000 * (np.sum(h) - n_points/2)**2
        return soft + penalty
    
    y_start = logit(h_start + 1e-6)
    result = minimize(objective, y_start, method='L-BFGSB', options={'maxiter': max_iter})
    if result.success:
        h_final = sigmoid(result.x)
        h_final = proj_to_simplex(h_final, n_points/2)
        return h_final
    else:
        return h_start

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_t = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    for n_try in n_values:
        if time.time() - start_t > budget_s * 0.9:
            break
        time_left = budget_s - (time.time() - start_t)
        if time_left < 20:
            break
            
        status, lower_bound, x_sol, X_sol = sdplp_relaxation(n_try)
        if status == 'optimal' and not np.isnan(lower_bound):
            round_h, round_c5 = rounding(X_sol, x_sol, R=30)
            if round_c5 < best_c5:
                best_c5 = round_c5
                best_h = round_h.copy()
                best_n = n_try
            # Hill climb from rounded
            if time.time() - start_t > budget_s * 0.9:
                break
            hc_h, hc_c5 = hill_climb_binary(round_h, max_iters=10000)
            if hc_c5 < best_c5:
                best_c5 = hc_c5
                best_h = hc_h.copy()
                best_n = n_try
            # Refinement if time allows
            if time.time() - start_t > budget_s * 0.8:
                break
            smooth_h = smooth_optimize(hc_h, max_iter=100)
            smooth_c5 = evaluate_h(smooth_h)
            if smooth_c5 < best_c5:
                best_c5 = smooth_c5
                best_h = smooth_h.copy()
                best_n = n_try
                
            if best_c5 <= target_bound:
                break

    if best_h is None:
        # Fallback: local search on good starting solutions
        # We'll use a random binary construction and optimize
        best_h = np.random.rand(n_points)
        best_h = ((best_h - best_h.min()) / (best_h.max() - best_h.min()))
        # Ensure sum = 1 (i.e. dx * sum(h) = 1 -> sum(h) = n/2)
        best_h *= (n_points / 2) / np.sum(best_h)
        # Clip at [0,1]
        best_h = np.clip(best_h, 0, 1)
        best_c5 = evaluate_h(best_h)
        # Systematic hill climb
        best_h, best_c5 = hill_climb_binary(best_h, max_iters=100000)
    
    # Final polishing with SLSQP
    if time.time() - start_t < budget_s - 5:
        # Constraint for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}
        
        def objective(x):
            return evaluate_h(x)
            
        try:
            result = minimize(objective, best_h, method='SLSQP', 
                             bounds=bounds, constraints=constraints,
                             options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-10})
            if result.success:
                best_h = result.x
                best_c5 = evaluate_h(best_h)
        except:
            pass
    
    return list(best_h), best_c5, best_n
```
