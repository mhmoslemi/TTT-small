import numpy as np
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def closest_binary_vector(h):
    """Round to nearest binary vector with same sum"""
    n = len(h)
    target_sum = n / 2
    # Non-negative values
    h = np.clip(h, 0, 1)
    # Mass conservation: weighted average must equal 0.5
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    else:
        # Project to correct sum
        scale = target_sum / current_sum
        h = h * scale
        return h.copy()

def gradient_descent_refinement(start_h, max_iter=2000, tol=1e-8):
    """Refine a solution with projected sub-gradient descent"""
    n = len(start_h)
    dx = 2.0 / n
    h = start_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Precompute all correlations
    def compute_all_correlations(h):
        corr = correlate(h, 1-h, mode="full")
        return corr
    
    for i in range(max_iter):
        corr = compute_all_correlations(h)
        max_corr = np.max(corr)
        active_shifts = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        # Compute sub-gradient of max correlation
        grad = np.zeros_like(h)
        for s in active_shifts:
            # Shifts for correlation
            # s corresponds to: h_i * (1 - h_{i+s})
            # Need to compute partial derivatives:
            # ∂C_s/∂h_i = 1 - h_{i+s} - h_{i-s}
            for i in range(n):
                if i + s < n:
                    grad[i] += 1 - h[i+s]
                if i - s >= 0:
                    grad[i] -= h[i-s] 
        # Normalize gradient, scale by learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad = grad / grad_norm * 1e-3
        # Gradient step
        h_new = h - grad
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Project onto simplex
        h_new = closest_binary_vector(h_new)
        # Evaluate
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new.copy()
        if abs(c5_new - best_c5) < tol:
            break
    
    return best_h, best_c5

def solve_binary_milp(n, time_limit):
    """Solve exact MILP for binary case to get integer bound"""
    m = n // 2  # exactly half the points should be 1
    
    start_time = time.time()
    
    # Variables
    b = cp.Variable(n, boolean=True)
    t = cp.Variable(pos=True)
    
    # Constraints
    constraints = [cp.sum(b) == m]
    
    # Product linearisation for y_{i,j} = b_i * b_j
    # Store linear constraints for pairs (i,j) s.t. i <= j
    y_vars = {}
    y_extra = []
    for i in range(n):
        for j in range(i, n):
            y = cp.Variable(pos=True)
            y_vars[(i,j)] = y
            y_extra.append(y)
            # Add McCormick constraints
            constraints.extend([
                y <= b[i],
                y <= b[j],
                y >= b[i] + b[j] - 1
            ])
    
    # Overlap constraints for each shift
    # Index shifts from 0 to n-1, where s = 0 means no shift
    # For shift s, overlap = sum_{i} b[i] * b[i+s]  (with wrapped indexing)
    for s in range(n):
        overlap_expr = 0
        for i in range(n):
            j = (i + s) % n
            overlap_expr += y_vars[(i,j)]
        constraints.append(overlap_expr >= t)
    
    # Objective
    objective = cp.Maximize(t)
    
    prob = cp.Problem(objective, constraints)
    
    try:
        # Use ECOS_BB for binary problem
        prob.solve(solver=cp.ECOS_BB, time_limit=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            t_val = t.value
            if t_val is not None:
                c5_val = (m - t_val) * (2.0 / n)
                b_sol = b.value
                return b_sol, c5_val, True
        return None, None, False
    except:
        return None, None, False

def run_annealing_for_n(n, budget_per_n, seed=42):
    """Stochastic binary search on large n grid"""
    best_h = None
    best_c5 = float('inf')
    np.random.seed(seed)
    
    start_time = time.time()
    max_steps = 10 * n * n
    
    # Initial random binary vector
    h = np.random.choice([0, 1], size=n, p=[0.5, 0.5])
    h = closest_binary_vector(h)
    
    c5 = evaluate_c5(h)
    current_h = h.copy()
    current_c5 = c5
    
    # Simulated annealing parameters
    T = 0.1  # initial temperature
    T_min = 1e-6
    alpha = 0.999  # cooling factor
    step = 0
    
    while step < max_steps and (time.time() - start_time) < budget_per_n:
        # Flip two random bits
        i = np.random.randint(n)
        j = np.random.randint(n)
        if current_h[i] == current_h[j]:
            continue
        # New candidate
        h_new = current_h.copy()
        h_new[i] = 1 - h_new[i]
        h_new[j] = 1 - h_new[j]
        
        # Ensure target weight
        h_new = closest_binary_vector(h_new)
        
        c5_new = evaluate_c5(h_new)
        
        delta = c5_new - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = h_new.copy()
            current_c5 = c5_new
            
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
        
        T *= alpha
        step += 1
        
        if step % 1000 == 0:
            # Periodic check of time
            if (time.time() - start_time) > budget_per_n:
                break
    
    return best_h, best_c5
