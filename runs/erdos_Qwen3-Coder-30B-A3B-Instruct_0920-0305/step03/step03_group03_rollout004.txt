import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def proj_box_simplex(v):
    """Project vector v onto the intersection of [0,1]^n and the unit simplex."""
    v = np.array(v)
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    # Find the threshold
    css = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - css / ind > 0
    if not np.any(cond):
        # All values are zero or negative
        return np.zeros_like(v)
    rho = np.argmax(cond)
    theta = css[rho] / (rho + 1)
    # Project onto the simplex and truncate to [0,1]
    x = np.clip(v - theta, 0, 1)
    # Renormalize
    s = np.sum(x)
    return x / s if s > 0 else x

def c5_from_h(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def spectral_search(n, budget_s, seed=42):
    """Spectral alternating projection with binary refinement"""
    np.random.seed(seed)
    t0 = time.time()
    dx = 2.0 / n
    m = n / 2  # target sum
    it_max = 500
    R = 30  # restarts
    best_c5 = float('inf')
    best_h = None

    # Target magnitude in frequency domain: derived from empirical value
    # In the best-known solutions, most of the power is in the non-zero components

    # Fixed M based on empirical experience
    M = 1.0  # this will be refined as we go

    # Inner loop
    for restart in range(R):
        if time.time() - t0 > budget_s * 0.98:
            break

        # Random start
        h = np.random.uniform(0, 1, n)
        h = proj_box_simplex(h)
        h = h * (1.0 / np.sum(h) * m)  # scale to match μ = m

        # Compute initial M
        if restart == 0:
            H = np.fft.fft(h)
            M = np.mean(np.abs(H[1:]) ** 2) ** 0.5

        for iteration in range(it_max):
            if time.time() - t0 > budget_s * 0.98:
                break

            # FFT
            H = np.fft.fft(h)

            # Enforce magnitude flatness in non-DC components
            H_new = H.copy()
            H_new[1:] = M * np.exp(1j * np.angle(H[1:]))
            # DC component unchanged
            H_new[0] = H[0]

            # Inverse FFT
            h_tilde = np.real(np.fft.ifft(H_new))

            # Project into feasible simplex
            h = proj_box_simplex(h_tilde)

            # Compute C5
            c5 = c5_from_h(h)

            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

            # Check for convergence
            if iteration > 10 and abs(c5 - c5_from_h(h)) < 1e-8:
                break

    return best_h, best_c5

def binary_refinement(h, n, budget_s):
    """Refine continuous solution to binary by thresholding and local swaps."""
    t0 = time.time()
    m = n / 2
    threshold = 0.5
    # Threshold to binary
    b = (h > threshold).astype(float)
    # Adjust to exactly m ones
    diff = np.sum(b) - m
    if diff > 0:
        # Too many ones
        zero_indices = np.where(b == 1)[0]
        zero_indices = np.random.choice(zero_indices, int(diff), replace=False)
        b[zero_indices] = 0
    elif diff < 0:
        # Too few ones
        one_indices = np.where(b == 0)[0]
        one_indices = np.random.choice(one_indices, int(-diff), replace=False)
        b[one_indices] = 1

    # Step up the binary solution to continuous using projected gradient descent
    h_refined = b.copy()
    best_h = h_refined.copy()
    best_c5 = c5_from_h(h_refined)

    # Local search by swapping
    attempts = 0
    successful_swaps = 0
    max_attempts = 5 * n
    max_no_improvement = 2 * n

    while attempts < max_attempts:
        if time.time() - t0 > budget_s * 0.98:
            break

        # Random swap
        zero_indices = np.where(h_refined == 0)[0]
        one_indices = np.where(h_refined == 1)[0]
        if len(zero_indices) == 0 or len(one_indices) == 0:
            break

        i0 = np.random.choice(zero_indices)
        i1 = np.random.choice(one_indices)

        # Try swap
        h_trial = h_refined.copy()
        h_trial[i0] = 1
        h_trial[i1] = 0

        c5_trial = c5_from_h(h_trial)
        if c5_trial < best_c5:
            h_refined = h_trial
            best_c5 = c5_trial
            successful_swaps += 1
        attempts += 1

        # Exit early if no improvement over a long time
        if successful_swaps == 0 and (attempts >= max_no_improvement):
            break

    return best_h, best_c5

def fallback_random_search(n, budget_s, seed=42):
    """A fallback that performs random searches with local polishing."""
    t0 = time.time()
    best_c5 = float('inf')
    best_h = None
    k = 200  # sample size
    k1 = 20  # kept best samples

    # Generate random samples
    samples = []
    for i in range(k):
        if time.time() - t0 > budget_s * 0.98:
            break
        h = np.random.uniform(0, 1, n)
        h = proj_box_simplex(h)
        h = h * (1.0 / np.sum(h) * n / 2)
        samples.append(h)
    
    # Score samples
    scores = [c5_from_h(h) for h in samples]
    best_indices = np.argsort(scores)[:min(k1, len(samples))]
    
    # Local polishing
    for idx in best_indices:
        h = samples[idx].copy()
        for _ in range(20):
            if time.time() - t0 > budget_s * 0.98:
                break
            # Simple perturbation
            h_pert = h + np.random.normal(0, 0.01, n)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = proj_box_simplex(h_pert)
            h_pert = h_pert * (1.0 / np.sum(h_pert) * n / 2)
            c5 = c5_from_h(h_pert)
            if c5 < c5_from_h(h):
                h = h_pert
        c5 = c5_from_h(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5
