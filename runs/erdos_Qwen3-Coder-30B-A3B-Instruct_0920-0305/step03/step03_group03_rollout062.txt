import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp
from scipy.optimize import minimize
from math import ceil, floor
from copy import deepcopy

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    n_points = 0
    best_n = 0

    # 1. Preparations
    initial_h = np.array(initial_h_values)
    initial_c5 = evaluate_c5(initial_h)
    best_h = initial_h.copy()
    best_c5 = initial_c5
    n_points = len(initial_h)
    best_n = n_points

    # MILP routine with cvxpy
    def solve_binary_milp(n):
        m = n // 2
        dx = 2.0 / n
        # Create binary variables
        b = cp.Variable(n, boolean=True)
        y = cp.Variable((n, n), boolean=True)
        t = cp.Variable()
        # Weight constraint
        constraints = [cp.sum(b) == m]
        # Product constraints
        for i in range(n):
            for j in range(i, n):
                constraints += [
                    y[i, j] <= b[i],
                    y[i, j] <= b[j],
                    y[i, j] >= b[i] + b[j] - 1
                ]
        # Overlap constraints for all shifts
        for s in range(n):
            if s == 0:
                overlap = cp.sum(y[0:n, 0:n])
            else:
                overlap = cp.sum(y[0:n-s, s:n])
            constraints += [overlap >= t]
        # Solve
        prob = cp.Problem(cp.Maximize(t), constraints)
        try:
            prob.solve(solver=cp.GLPK_MI, verbose=False, time_limit=0.6*budget_s)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                t_val = t.value
                if t_val is not None:
                    c5_bound = (m - t_val) * dx
                    h_opt = b.value.astype(float)
                    # Return h as a float array
                    return h_opt, c5_bound, n
        except Exception as e:
            pass
        return None, None, n

    # 2. MILP on grid of interest
    mid_time = time.time()
    n_vals = [42, 44, 48, 52, 56, 60]
    for n in n_vals:
        if time.time() - start_time > 0.3 * budget_s:
            break
        h_opt, c5_bound, n_used = solve_binary_milp(n)
        if h_opt is not None and c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
            n_points = n_used
            best_n = n_used
        # Exit early if target met
        if best_c5 <= 0.3807:
            return (best_h.tolist(), best_c5, n_points)

    # 3. Continuous refinement with gradient
    def approximate_gradient(h):
        """Compute approximate gradient for the max correlation"""
        n = len(h)
        dx = 2.0 / n
        # Find indices that achieve maximum
        corr = correlate(h, 1 - h, mode="full")
        max_val = np.max(corr)
        # compute all shift values, their indices
        shifts = np.arange(n)
        max_shifts = np.where(corr >= max_val - 1e-10)[0]
        if len(max_shifts) == 0:
            return np.zeros_like(h)
        # Average gradient over maximum shifts
        g = np.zeros_like(h)
        for s_idx in max_shifts:
            s = s_idx - (n - 1)
            # Gradient at point i
            for i in range(n):
                contrib = 1
                if i + s >= 0 and i + s < n:
                    contrib -= h[i + s]  # gradient w.r.t. to h_i
                if i - s >= 0 and i - s < n:
                    contrib -= h[i - s]
                g[i] += contrib
        g /= len(max_shifts)
        return g

    # 3.1 Projected gradient descent (from returned binary of last check)
    if best_h is not None:
        h_current = best_h.copy()
        iter_count = 0
        patience = 0
        while (time.time() - start_time) < 0.6 * budget_s and patience < 1000:
            iter_count += 1
            alpha = 0.1 / (1 + iter_count * 0.001)
            grad = approximate_gradient(h_current)
            h_new = h_current - alpha * grad
            # Clamp to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Project to L1 simplex: 
            s = np.sum(h_new) * dx
            h_new = h_new / s
            # Evaluate
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                n_points = len(h_new)
                best_n = n_points
                patience = 0
            else:
                patience += 1
            h_current = h_new
            if best_c5 <= 0.3807:
                return (best_h.tolist(), best_c5, n_points)

    # 4. Simulated Annealing on large grids
    def simulated_annealing_binary(n):
        """Simulated Annealing over binary strings"""
        dx = 2.0 / n
        m = n // 2
        # initialize with 1's in first m positions, 0's in the last (n-m)
        h = np.zeros(n)
        h[:m] = 1
        np.random.shuffle(h)
        current_c5 = evaluate_c5(h)
        best_c5 = current_c5
        best_h = h.copy()
        T0 = 0.1
        T_min = 1e-12
        tau = max(10*n*n, 50000)  # total number of iterations
        T = T0
        k = 0
        
        while k < tau and (time.time() - start_time) < 0.8 * budget_s:
            k += 1
            if T <= T_min: break
            # pick two random indices
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            if h[i] == h[j]: continue
            # Swap values
            h[i], h[j] = h[j], h[i]
            try:
                new_c5 = evaluate_c5(h)
            except:
                # revert
                h[i], h[j] = h[j], h[i]
                continue
            if new_c5 < current_c5:
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                # Accept all improvements
                T = T0 * (1 - k / tau)  # cooling schedule
            else:
                p = np.exp(-(new_c5 - current_c5) / T)
                if np.random.uniform() < p:
                    current_c5 = new_c5
                    # Accept anyway
                    T = T0 * (1 - k / tau)
                else:
                    # Revert if no acceptance
                    h[i], h[j] = h[j], h[i]
            if best_c5 <= 0.3807:
                return best_h, best_c5, n

        # Convert to continuous, refine after SA by gradient
        if k >= tau or (time.time() - start_time) >= 0.8 * budget_s:
            h_cont = best_h.copy()
            h_f = h_cont.astype(float)
            h_f = h_f * (n/2) / np.sum(h_f)  # ensure sum = (n/2)*dx = 1
            # Refine once
            for _ in range(100):
                g = approximate_gradient(h_f)
                h_f = h_f - 0.05 * g
                h_f = np.clip(h_f, 0, 1)
                h_f = h_f * (n/2) / np.sum(h_f)
                maybe_c5 = evaluate_c5(h_f)
                if maybe_c5 < best_c5:
                    best_c5 = maybe_c5
                    h_cont = h_f.copy()
            best_h = h_cont.copy()
            
        return best_h, best_c5, n

    # Run SA over various lengths
    for n in [44, 48, 52, 56, 60]:
        if (time.time() - start_time) > 0.85 * budget_s:
            break
        try:
            h_sa, c5_sa, n_used = simulated_annealing_binary(n)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                n_points = n_used
                best_n = n_used
            if best_c5 <= 0.3807:
                return (best_h.tolist(), best_c5, n_points)
        except:
            continue

    # 5. fallback if needed
    try:
        # Use the previous algorithm
        de_res = differential_evolution(lambda x: evaluate_c5(x), 
            bounds=[(0, 1)]*len(initial_h), seed=seed+1000, maxiter=500, 
            popsize=20, mutation=(0.5, 1), recombination=0.7, strategy='best1bin')
        if de_res.success:
            h_candidate = de_res.x
            dx = 2.0 / len(h_candidate)
            h_candidate = h_candidate * (1.0 / (np.sum(h_candidate) * dx))
            c5_candidate = evaluate_c5(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
                n_points = len(h_candidate)
                best_n = n_points
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
