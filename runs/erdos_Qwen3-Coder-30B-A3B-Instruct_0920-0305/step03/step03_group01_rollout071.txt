import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_onto_box_and_hyperplane(y, target):
    """Project onto box [0,1] and hyperplane sum = target using waterfilling"""
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0, 1)

def gradient_update(h, max_ind, shift):
    """Compute gradient for a specific shift"""
    n = len(h)
    if shift >= 0:
        term1 = np.empty(n)
        term1[:n-shift] = 1 - h[shift:]
        term1[n-shift:] = 1
        term2 = np.empty(n)
        term2[shift:] = h[:n-shift]
        term2[:shift] = 0
    else:
        s = -shift
        term1 = np.empty(n)
        term1[s:] = 1 - h[:n-s]
        term1[:s] = 1
        term2 = np.empty(n)
        term2[:n-s] = h[s:]
        term2[n-s:] = 0
    grad = term1 - term2
    return grad

def calculate_max_shift(corr):
    """Get the shift that achieves maximum correlation"""
    max_corr = np.max(corr)
    max_ind = np.where(corr == max_corr)[0]
    return max_ind[0] - (len(corr) - 1), max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 256
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize to tent shape
    h0 = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        h0[i] = max(0, 1 - abs(x - 1))
    
    # Scale to satisfy sum constraint
    h0 = h0 * target_sum / np.sum(h0)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Gradient descent with restarts
    max_iterations = 10000
    step_size = 0.05
    decay_factor = 0.9999
    
    restarts = 0
    max_restarts = 50
    
    while (time.time() - start_time) < budget_s * 0.95 and restarts < max_restarts:
        h = best_h.copy()
        prev_best_c5 = best_c5
        
        for i in range(max_iterations):
            if (time.time() - start_time) > budget_s * 0.95:
                break
                
            # Compute correlation and find worst shift
            corr = correlate(h, 1 - h, mode="full")
            shift, _ = calculate_max_shift(corr)
            
            # Compute gradient for this shift  
            grad = gradient_update(h, None, shift)
            
            # Update with gradient
            h_new = h - step_size * grad
            
            # Project onto feasible region
            h_new = project_onto_box_and_hyperplane(h_new, target_sum)
            
            # Compute C5
            c5_new = evaluate_c5(h_new)
            
            # Update if better
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            
            # Decay step size
            step_size *= decay_factor
            
            # Add jitter occasionally to escape local minima
            if i % 500 == 0 and np.random.random() < 0.1:
                noise = np.random.normal(0, 0.02, n_points)
                h_new = h + noise
                h_new = np.clip(h_new, 0, 1)
                h_new = project_onto_box_and_hyperplane(h_new, target_sum)
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                h = h_new.copy()
                
        restarts += 1
        # Random restart every few iterations
        if restarts % 3 == 0:
            noise = np.random.uniform(0, 0.1, n_points)
            h = best_h + noise
            h = np.clip(h, 0, 1)
            h = project_onto_box_and_hyperplane(h, target_sum)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    
    # Final refinement
    final_h = best_h.copy()
    final_c5 = evaluate_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
