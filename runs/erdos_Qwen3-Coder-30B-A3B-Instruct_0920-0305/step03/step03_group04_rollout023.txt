def dx(n):
    return 2.0 / n

def evaluate(h):
    """Evaluate the C5 bound for given h values."""
    h = np.array(h)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx(len(h))
    return c5

def correlation_vec(h):
    """Return full correlation vector."""
    return np.correlate(h, 1 - h, mode="full")

def smooth_max(c, alpha):
    """Smooth version of max."""
    m = np.max(c)
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

def penalty_sum(h, lambda_pen):
    """Penalty to enforce sum of h = n/2."""
    return lambda_pen * (np.sum(h) - len(h) / 2) ** 2

def proj_to_simplex(v, S=None):
    """Project v onto the simplex with sum = S."""
    if S is None:
        S = len(v) / 2
    v = np.clip(v, 0, 1)
    s = np.sum(v)
    if abs(s - S) < 1e-10:
        return v
    if s < S:
        # Add to lowest values
        sorted_idx = np.argsort(v)
        remaining = S - s
        v[sorted_idx] = np.maximum(v[sorted_idx], remaining / (len(v) - np.argmax(v[sorted_idx] >= 0)))
        return v
    else:
        # Scale down
        residual = s - S
        v = v - residual / len(v)
        v = np.clip(v, 0, 1)
        return v

def binary_initial(n):
    """Generate a random binary h with n/2 ones."""
    h = np.zeros(n)
    indices = np.random.choice(n, size=n//2, replace=False)
    h[indices] = 1
    return h

def swap_and_evaluate(h, i_one, i_zero):
    """Swap entries at i_one and i_zero and return new C5."""
    h_new = h.copy()
    h_new[i_one] = 0
    h_new[i_zero] = 1
    return evaluate(h_new)

def logit(p):
    """Logit function."""
    return np.log(p / (1 - p))

def sigmoid(y):
    """Sigmoid function."""
    return 1 / (1 + np.exp(-y))

def solve_binary(n, budget_s, t_start):
    """Binary hill climbing."""
    best_h = None
    best_c5 = np.inf
    best_count = np.inf
    T_remaining = budget_s - (time.time() - t_start)
    if T_remaining < 15:
        return best_h, best_c5
    
    # Generate restarts
    restarts = 5
    for r in range(restarts):
        h = binary_initial(n)
        c5 = evaluate(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_count = np.sum(1 - (h[None, :] * h[:, None]).flatten() * (1 - np.eye(n))).sum()
        
        # Hill climb
        S = 5 * n
        attempts = 0
        stuck = 0
        while attempts < S and time.time() - t_start < budget_s * 0.9:
            # Find 1s and 0s
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            if len(ones) == 0 or len(zeros) == 0:
                break
                
            # Choose one 1 and one 0 at random
            i_one = np.random.choice(ones)
            i_zero = np.random.choice(zeros)
            h_new = h.copy()
            h_new[i_one] = 0
            h_new[i_zero] = 1

            new_c5 = evaluate(h_new)
            new_count = np.sum(1 - (h_new[None, :] * h_new[:, None]).flatten() * (1 - np.eye(n))).sum()
            
            if new_c5 < c5:
                h = h_new
                c5 = new_c5
                attempts = 0
                stuck = 0
                if new_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
                    best_count = new_count
            else:
                attempts += 1
                stuck += 1
                if stuck >= S // 2:
                    # Restart
                    h = binary_initial(n)
                    attempts = 0
                    stuck = 0
    return best_h, best_c5
