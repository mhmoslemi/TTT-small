import numpy as np
import cvxpy as cp
import time

def solve_milp_binary(n, time_limit=60):
    """Solve the binary MILP for the Erdos problem with n points."""
    dx = 2.0 / n
    
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Constraints: sum of h_i = n/2
    constraints = [cp.sum(h) == n // 2]
    
    # Linearization for all shifts d = 0..n-1
    for d in range(n):
        # Variables y_{i,d} for i = 0 to n-1-d
        y = cp.Variable(n - d, boolean=True)
        # Add y_{i,d} <= h_i
        # Add y_{i,d} <= h_{i+d}
        # Add y_{i,d} >= h_i + h_{i+d} - 1
        for i in range(n - d):
            constraints.append(y[i] <= h[i])
            constraints.append(y[i] <= h[i + d])
            constraints.append(y[i] >= h[i] + h[i + d] - 1)
        # Q_d = sum_{i} y_{i,d}
        constraints.append(cp.sum(y) >= t)
        
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CBC, verbose=False, 
                   time_limit=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_vals = h.value
            t_val = t.value
            c5_bound = 1 - (2 / n) * t_val
            return h_vals, c5_bound
        elif prob.status in [cp.SOLVER_ERROR, cp.INFEASIBLE, cp.UNBOUNDED]:
            return None, float('inf')
        else:
            # Try to get the best found solution
            if h.value is not None:
                h_vals = h.value
                # Since solver may not return exact t_value in time:
                # compute t_val as minimal overlap of current solution and backtrack
                t_val = float('inf')
                # Compute Q_d = sum_{i} h_{i} * h_{i+d} for d=0 to n-1
                corr = np.correlate(h_vals, h_vals, mode='full')
                # Indices of the full convolution that match valid correlations
                offset = len(corr) - n
                Q_vals = corr[offset:offset+n]
                t_val = min(Q_vals)
                c5_bound = 1 - (2 / n) * t_val
                return h_vals, c5_bound
    except:
        return None, float('inf')
    return None, float('inf')

def hill_climb_swaps(h, max_swaps=10000, time_limit=60):
    """Hill-climb improvement via swaps (binary-hill climb)."""
    start = time.time()
    h = h.copy()
    dx = 2.0 / len(h)
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1 - h)
    n = len(h)
    sum_h = np.sum(h)
    
    best_c5 = 1 - (2 / n) * np.min(np.correlate(h, h, mode='full')[n-1:n*2-1])
    max_iterations = max_swaps
    for _ in range(max_iterations):
        if time.time() - start > time_limit:
            break
        # pick random one and zero
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # swap values
        h[i] = 0
        h[j] = 1
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        
        # evaluate new bound
        corr = np.correlate(h, h, mode='full')
        offset = len(corr) - n
        Q_vals = corr[offset:offset+n]
        new_c5 = 1 - (2 / n) * np.min(Q_vals)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            # rollback
            h[i] = 1
            h[j] = 0
            ones = np.flatnonzero(h)
            zeros = np.flatnonzero(1 - h)
    
    # compute final result (return best or final)
    corr = np.correlate(h, h, mode='full')
    offset = len(corr) - n
    Q_vals = corr[offset:offset+n]
    final_c5 = 1 - (2 / n) * np.min(Q_vals)
    return h, final_c5

    dx = 2.0 / n
    # Rescale to correct sum
    h *= (n / 2) / np.sum(h)
    
    num_iter = 0
    step = 0.05
    max_iter = 10000 # 10000 step crude but fast
        
    while num_iter < max_iter:
        if time.time() - start > time_limit:
            break
        grad = np.zeros_like(h)
        corr = np.correlate(h, h, mode='full')
        offset = len(corr) - n
        Q_vals = corr[offset:offset+n]
        min_idx = np.argmin(Q_vals)
        # Gradients for point near min_overlap
        for i in range(min_idx, min_idx + 1):
            # Compute gradient of Q_min wrt h_i
            # This is a simplification - just one gradient term for effect.
            # More precise would use chain rule on full convolution computation
            if 0 <= i < n:
                # gradient details: Q[i] = sum_{j} h_j * h_{j+shift}
                # We're interested in which *pair* contributes exactly
                # to being u min. We compute derivative of that Q
                # derivative of Q[w] wrt h[i] = sum_{l} d(h[i]*h[i+w-l])/d(h[i])
                # But only factors that are non-zero.
                # Rather than discrete now: use numerics
                # perturb to estimate gradient
                import math
                eps = 1e-3
                h_pos = h.copy()
                h_neg = h.copy()
                h_pos[i] += eps
                h_neg[i] -= eps
                q_pos = np.correlate(h_pos, h_pos, mode='full')[offset:offset+n]
                q_neg = np.correlate(h_neg, h_neg, mode='full')[offset:offset+n]
                grad[i] = (np.min(q_pos) - np.min(q_neg))/(2*eps)
        
        # Apply gradient step
        h -= step * grad
        # Project onto sum constraint
        h *= (n/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        num_iter += 1
        
        if num_iter % 100 == 0:
            Q_full = np.correlate(h, h, mode='full')
            offset = len(Q_full) - n
            Q_vals = Q_full[offset:offset+n]
            c5 = 1 - (2 / n) * np.min(Q_vals)
            if c5 < 0.379:
                break # stop early if good enough
    return h, 1 - (2 / n) * np.min(np.correlate(h, h, mode='full')[n-1:n*2-1])
