    h = np.array(h_values)

    correlation = np.correlate(h, 1 - h, mode="full")

        return evaluate_h(h)

                         options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-10})
        if result.success:
            return result.x
    except:
        pass
    return h_values
