```python
import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time

n_values = [48, 60, 72, 84]
dx_values = [2.0/n for n in n_values]
target_bound = 0.3807

def evaluate(h):
    """Exact evaluation of C5."""
    c = np.correlate(h, 1 - h, mode="full")
    return np.max(c) * (2.0 / len(h))

def correlation_vec(h):
    """Compute the full correlation vector."""
    return np.correlate(h, 1 - h, mode="full")

def smooth_max(c, alpha=30.0):
    """Compute smooth max of array c."""
    m = np.max(c)
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

def penalty_sum(h, lam=10000.0):
    """Quadratic penalty for sum deviation."""
    return lam * (np.sum(h) - len(h)/2)**2

def proj_to_simplex(v, S=None):
    """Project vector into simplex with sum S."""
    if S is None:
        S = len(v) / 2
    v = np.clip(v, 0, 1)
    s = np.sum(v)
    if abs(s - S) < 1e-10:
        return v
    v += (S - s) / len(v)
    v = np.clip(v, 0, 1)
    return v

def binary_initial(n):
    """Generate a random binary vector with exactly n/2 ones."""
    h = np.zeros(n)
    ones = np.random.permutation(n)[:n//2]
    h[ones] = 1
    return h

def swap_and_evaluate(h, i_one, i_zero):
    """Swap h[i_one] and h[i_zero] and return the new C5."""
    h_new = h.copy()
    h_new[i_one], h_new[i_zero] = h_new[i_zero], h_new[i_one]
    return evaluate(h_new)

def sigmoid(x):
    """Sigmoid function."""
    return 1.0 / (1.0 + np.exp(-x))

def logit(p):
    """Logit function."""
    return np.log(p / (1.0 - p))

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    # 2.1 - Parametric search over interval families
    def interval_family_search():
        global best_h, best_c5, best_n
        for n in [46, 48, 50, 60]:
            dx = 2.0 / n
            best_h_n = None
            best_c5_n = float('inf')
            for m in range(2, 9):
                ℓ = 1.0 / m
                max_g = (2.0 - m * ℓ) / (m - 1.0) if m > 1 else 0.0
                if max_g < 0:
                    continue
                g_list = [max_g * x for x in [1.0, 0.75, 0.5]]
                for g in g_list:
                    if g < 0:
                        continue
                    end = 2.0 - (m - 1) * g
                    if end < 0:
                        continue
                    for δ in np.arange(0, g, dx / 2):
                        if δ > g:
                            continue
                        h = np.zeros(n)
                        pos = δ
                        for i in range(m):
                            start_idx = int(pos / dx)
                            end_pos = pos + ℓ
                            end_idx = int(end_pos / dx)
                            if end_idx > n:
                                end_idx = n
                            if start_idx < n:
                                h[start_idx:end_idx] = 1.0
                            pos += ℓ + g
                        h = proj_to_simplex(h)
                        c5 = evaluate(h)
                        if c5 < best_c5_n:
                            best_h_n = h.copy()
                            best_c5_n = c5
                        if c5 < target_bound:
                            return h, c5, n
            if best_c5_n < best_c5:
                best_h = best_h_n
                best_c5 = best_c5_n
                best_n = n

    print("Starting interval family search...")
    interval_family_search()

    # 2.2 Binary Hill Climbing
    print("Starting binary hill climbing...")
    n_list = [48, 60, 72, 84]
    T_remaining = budget_s - (time.time() - t0)
    time_per_n = T_remaining * 0.5 / len(n_list)
    for n in n_list:
        if time.time() - t0 > budget_s * 0.95:
            break
        dx = 2.0 / n
        best_h_bin = None
        best_c5_bin = float('inf')
        # 5 restarts, each with 5000 iterations
        for k in range(5):
            h = binary_initial(n)
            if time.time() - t0 > budget_s * 0.95:
                break
            c5 = evaluate(h)
            max_iter = 5000
            stuck_counter = 0
            old_c5 = c5
            for i in range(max_iter):
                if time.time() - t0 > budget_s * 0.95:
                    break
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i_one = np.random.choice(ones)
                i_zero = np.random.choice(zeros)
                h_new = h.copy()
                h_new[i_one], h_new[i_zero] = h_new[i_zero], h_new[i_one]
                c5_new = evaluate(h_new)
                if c5_new < c5 - 1e-10:
                    h = h_new
                    c5 = c5_new
                    stuck_counter = 0
                else:
                    stuck_counter += 1
                if c5_new < best_c5_bin:
                    best_h_bin = h.copy()
                    best_c5_bin = c5_new
                if stuck_counter > 5 * n:
                    # Random restart
                    h = binary_initial(n)
                    c5 = evaluate(h)
                    stuck_counter = 0
                if np.abs(c5 - old_c5) < 1e-10:
                    stuck_counter += 1
                else:
                    stuck_counter = 0
                old_c5 = c5
                if c5 < target_bound:
                    break
            if best_c5_bin < best_c5:
                best_h = best_h_bin
                best_c5 = best_c5_bin
                best_n = n
        print(f"Binary hill climbing done for n={n}, best C5={best_c5_bin}")

    # 2.3 Smooth max optimization
    print("Starting smooth max refinement...")
    T_remaining = budget_s - (time.time() - t0)
    if T_remaining < 5:
        print("Not enough time for smooth max.")
        pass
    else:
        for n in n_list:
            if best_c5 < target_bound:
                break
            if time.time() - t0 > budget_s * 0.95:
                break
            dx = 2.0 / n
            h_in = best_h if best_n == n else binary_initial(n)
            if np.abs(np.sum(h_in) - n / 2) > 1e-6:
                h_in = proj_to_simplex(h_in)
    
            def smooth_obj(y):
                h = sigmoid(y)
                c = correlation_vec(h)
                return smooth_max(c, 30.0) + penalty_sum(h)
    
            h_best = h_in.copy()
            c5_best = evaluate(h_best)
            
            # 8 restarts
            for r in range(8):
                if time.time() - t0 > budget_s * 0.95:
                    break
                y0 = logit(h_best) + np.random.normal(0, 0.2, n)
                result = minimize(smooth_obj, y0, method='L-BFGSB', options={'maxiter': 300})
                if not result.success:
                    continue
                h_trial = sigmoid(result.x)
                h_trial = proj_to_simplex(h_trial)
                c5_trial = evaluate(h_trial)
                if c5_trial < c5_best:
                    h_best = h_trial
                    c5_best = c5_trial
                if c5_best < target_bound:
                    break
    
            # Final projection
            h_best = proj_to_simplex(h_best)
            c5_best = evaluate(h_best)
            if c5_best < best_c5:
                best_h = h_best
                best_c5 = c5_best
                best_n = n

    # 2.4 SLSQP polishing
    print("Starting SLSQP polishing...")
    T_remaining = budget_s - (time.time() - t0)
    if T_remaining < 5:
        print("Not enough time for polishing.")
    else:
        bounds = [(0, 1) for i in range(len(best_h))]
        constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) * (2.0 / len(best_h)) - 1.0}
        try:
            result = minimize(evaluate, best_h, method='SLSQP', bounds=bounds, constraints=constraints,
                              options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-10})
            if result.success:
                best_h = result.x
                best_c5 = evaluate(best_h)
        except:
            pass

    return list(best_h), best_c5, best_n
```
