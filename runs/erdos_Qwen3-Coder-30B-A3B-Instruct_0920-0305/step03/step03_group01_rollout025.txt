import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

            h = np.repeat(u_full, n_points // n_blocks)
            # Pad if necessary
            if len(h) < n_points:
                h = np.pad(h, (0, n_points - len(h)), mode='constant', constant_values=h[-1])
            elif len(h) > n_points:
                h = h[:n_points]
            return h

            if np.any(u_vec < 0) or np.any(u_vec > 1) or u_vec[-1] < 0 or u_vec[-1] > 1:
                return np.inf
            h = block_param_to_h(u_vec)
            return evaluate_c5(h)

            u_best = res.x
