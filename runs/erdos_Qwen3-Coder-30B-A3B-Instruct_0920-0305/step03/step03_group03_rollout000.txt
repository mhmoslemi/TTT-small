def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_milp_binary(n, time_limit):
    """Solve MILP for binary h to maximize minimum overlap"""
    m = n // 2
    # Binary variables
    b = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    
    # Constraints
    constraints = [cp.sum(b) == m]
    
    # Precompute all shift indices for efficient loop
    shift_indices = []
    for s in range(n):
        shift_indices.append(list(range(n - s)))
    
    # For each shift s, we need sum_i b_i * b_{i+s} = overlap_s
    # Define auxiliary variables y[i,s] representing product b_i * b_{i+s}
    Y = cp.Variable((n, n), boolean=True)
    
    # Add constraints for each pair
    for i in range(n):
        for j in range(n):
            # Y[i,j] should represent b[i] * b[j]
            constraints += [Y[i,j] <= b[i]]
            constraints += [Y[i,j] <= b[j]]
            constraints += [Y[i,j] >= b[i] + b[j] - 1]
    
    # For each shift s, compute overlap_s = sum_{i=0}^{n-s-1} Y[i,i+s]
    overlaps = []
    for s in range(n):
        if s == 0:
            # Overlap at shift 0: sum of all b_i = m
            overlap_s = cp.sum(b)
        else:
            # For shift s, overlapping indices from 0 to n-s-1
            overlap_s = cp.sum([Y[i, i+s] for i in range(n - s)])
        overlaps.append(overlap_s)
        # Constrain overlap_s >= t
        constraints.append(overlap_s >= t)
    
    # Objective: maximize t
    prob = cp.Problem(cp.Maximize(t), constraints)
    
    # Solve with time limit
    try:
        prob.solve(solver=cp.CBC, timeLimit=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return b.value, t.value, overlaps
    except:
        pass
    return None, None, None
