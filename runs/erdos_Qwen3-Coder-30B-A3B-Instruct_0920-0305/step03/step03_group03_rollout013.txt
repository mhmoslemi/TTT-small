import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_best = None

    # Precomputed initial solution
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    n_best = n_points

    # Function to compute minimum overlap
    def compute_min_overlap(h):
        n = len(h)
        dx = 2.0 / n
        corr = correlate(h, h, mode="full")
        # We are looking for min of (n/2 - overlap) = n/2 - max(overlap)
        # Actually, we are directly computing the maximum correlation of h(x) * (1 - h(x+k))
        corr_h_minus = correlate(h, 1 - h, mode="full")
        max_corr_h_minus = np.max(corr_h_minus)
        return max_corr_h_minus * dx

    # Compute min overlap in a more structured way for continuous refinement
    def project_simplex(arr, target_sum=1.0):
        """Project a vector onto the simplex sum(x_i)=target_sum, x_i>=0"""
        # Sort in descending order
        sorted_arr = np.sort(arr)[::-1]
        # Find the threshold
        cumsum = np.cumsum(sorted_arr)
        threshold = (cumsum - target_sum) / np.arange(1, len(arr) + 1)
        idx = np.argmax(sorted_arr > threshold)
        theta = threshold[idx]
        # Project
        projected = np.maximum(arr - theta, 0)
        # Normalize to sum to target_sum
        projected = projected * (target_sum / np.sum(projected))
        return projected

    # Simulated Annealing for finding good binary configurations
    def binary_simulated_annealing(n, max_time):
        # Start with random binary vector
        h = np.random.rand(n) > 0.5
        h = h.astype(float)
        # Set exact count
        ones_count = n // 2
        num_ones = np.sum(h)
        if num_ones > ones_count:
            # Zero out some ones
            zero_indices = np.where(h == 1)[0]
            to_zero = int(num_ones - ones_count)
            zero_mask = np.random.choice(zero_indices, to_zero, replace=False)
            h[zero_mask] = 0
        elif num_ones < ones_count:
            # Set some zeros to ones
            one_indices = np.where(h == 0)[0]
            to_one = int(ones_count - num_ones)
            one_mask = np.random.choice(one_indices, to_one, replace=False)
            h[one_mask] = 1

        current_energy = evaluate_c5(h)
        best_h = h.copy()
        best_energy = current_energy

        T0 = 1.0
        T_min = 1e-10
        tau = 10 * n ** 2
        T = T0
        step = 0

        while T > T_min and (time.time() - start_time) < max_time * 0.9:
            # Random move: swap one random 1 and one random 0
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]

            if len(ones_idx) > 0 and len(zeros_idx) > 0:
                i = np.random.choice(ones_idx)
                j = np.random.choice(zeros_idx)
                h_new = h.copy()
                h_new[i] = 0
                h_new[j] = 1

                new_energy = evaluate_c5(h_new)

                if new_energy < current_energy:
                    h = h_new.copy()
                    current_energy = new_energy
                    if new_energy < best_energy:
                        best_h = h_new.copy()
                        best_energy = new_energy
                else:
                    # Accept with probability
                    delta_E = new_energy - current_energy
                    if np.random.rand() < np.exp(-delta_E / T):
                        h = h_new.copy()
                        current_energy = new_energy

                # Cool down
                T = T0 * (1.0 - step / tau)
                step += 1

        return best_h, best_energy

    # Run continuous refinement from the initial solution
    # First try it on the current best h
    retry_count = 0
    while retry_count < 3 and (time.time() - start_time) < budget_s * 0.7:
        # Run gradient descent for a few iterations
        h_current = best_h.copy()
        h_current = h_current / np.sum(h_current) * 2.0  # normalize to integral 1
        dx = 2.0 / len(h_current)
        for _ in range(1000):
            if (time.time() - start_time) > budget_s * 0.95:
                break
            
            # Compute maximum correlation positions
            corr_h_h = correlate(h_current, h_current, mode="full")
            idx_max = np.argmax(corr_h_h)
            max_corr_val = corr_h_h[idx_max]
            
            # Compute the actual correlation that correspond to C5
            corr_h_1h = correlate(h_current, 1 - h_current, mode="full")
            idx_c5 = np.argmax(corr_h_1h)
            max_c5 = corr_h_1h[idx_c5]
            
            # Compute approximate gradient using finite differencing
            grad_approx = np.zeros_like(h_current)
            eps = 1e-6
            for i in range(len(h_current)):
                h_plus = h_current.copy()
                h_plus[i] += eps
                # project and renormalize
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus / np.sum(h_plus) * 2.0  # renormalize to integral 1

                corr_plus = correlate(h_plus, 1 - h_plus, mode="full") 
                grad_approx[i] = (corr_plus[idx_c5] - max_c5) / eps

            # Descent step
            step_size = 1e-3
            h_new = h_current - step_size * grad_approx
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new / np.sum(h_new) * 2.0  # renormalize

            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                n_best = len(h_new)

            h_current = h_new

        retry_count += 1

    # Try model with larger dimensions using annealing
    lengths_to_try = [44, 48, 52, 56, 60]
    
    for n_trial in lengths_to_try:
        if (time.time() - start_time) > budget_s * 0.9:
            break
            
        max_time_per_trial = budget_s / 5 * 0.6  # allocate 0.6 of a fifth of budget
        h_trial_best, c5_trial_best = binary_simulated_annealing(n_trial, max_time_per_trial)
        
        # Convert to floats again, normalize to l1 integral = 1
        h_trial_best = h_trial_best.astype(float)
        h_trial_best = h_trial_best / np.sum(h_trial_best) * 2.0

        # Refine with gradient approach
        h_final_refine = h_trial_best.copy()
        for iter_refine in range(200):
            if (time.time() - start_time) > budget_s * 0.95:
                break
                
            corr_h_1h = correlate(h_final_refine, 1 - h_final_refine, mode="full")
            idx_top = np.argmax(corr_h_1h)
            corr_val = corr_h_1h[idx_top]

            # Finite difference gradient
            grad = np.zeros_like(h_final_refine)
            eps = 1e-7
            for i in range(len(h_final_refine)):
                h_plus = h_final_refine.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus / np.sum(h_plus) * 2.0
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                grad[i] = (corr_plus[idx_top] - corr_val) / eps

            lr = 1e-3
            h_new = h_final_refine - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new / np.sum(h_new) * 2.0

            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                n_best = len(h_new)

            h_final_refine = h_new

    # Final code fallback if better solution not found
    if best_c5 > 0.3807:
        # Use the base algorithm from previous def
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            def objective(h):
                h = np.array(h)
                h = h / np.sum(h) * 2.0  # normalize to integral 1
                return evaluate_c5(h)

            # Use differential evolution with random start
            de_res = differential_evolution(objective, bounds, seed=seed, maxiter=1000, 
                                          popsize=30, mutation=(0.7, 1), recombination=0.9,
                                          strategy='best1bin')
            if de_res.success:
                h_candidate = de_res.x
                h_candidate = h_candidate / np.sum(h_candidate) * 2.0
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate
                    n_best = n_points
        except:
            pass

    return (best_h.tolist(), best_c5, n_best)
