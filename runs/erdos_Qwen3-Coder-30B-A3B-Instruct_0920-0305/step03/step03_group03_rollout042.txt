from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

    def solve_binary_milp(n):
        m = n // 2  # target weight
        dx = 2.0 / n
        # Variables
        b = cp.Variable(n, boolean=True)  # binary variables
        y = cp.Variable((n, n), boolean=True)  # products
        t = cp.Variable()  # minimum overlap

        # Constraints
        constraints = [
            cp.sum(b) == m,
            t >= 0
        ]

        # Product linearization
        for i in range(n):
            for j in range(i, n):
                constraints += [
                    y[i, j] <= b[i],
                    y[i, j] <= b[j],
                    y[i, j] >= b[i] + b[j] - 1,
                ]

        # Shift overlaps
        for s in range(n):
            overlap_s = cp.sum([y[i, i + s] for i in range(n - s)])
            constraints += [overlap_s >= t]

        # Objective
        objective = Minimize(-t)  # maximize t

        prob = Problem(objective, constraints)
        time_limit = min(exact_time - (time.time() - exact_start), 100)
        try:
            prob.solve(solver=cp.CBC, time_limit=time_limit)
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_EOF]:
                t_val = -prob.objective.value
                h_binary = b.value.astype(float)

    def project_simplex(h):
