# Global constants
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_c5_exact(h):
    """Evaluate the discrete C5 based on the built-in correlation"""
    # numpy correlate gives us double the correlation, so we use mode='full'
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h):
    """Project h onto feasible set: 0 <= h[i] <= 1 and sum(h) = n/2"""
    # Step 1: Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Step 2: Water-filling to achieve sum = n/2
    # We compute lambda such that sum(max(h - lambda, 0)) = n/2
    # Using bisection method for lambda
    y = h.copy()
    sum_y = np.sum(y)
    
    # We need to find lambda such that sum(max(y - lambda, 0)) = n/2
    lambda_min, lambda_max = -2.0, 3.0
    tol = 1e-12
    
    for _ in range(100):
        lambda_mid = (lambda_min + lambda_max) * 0.5
        x = np.clip(y - lambda_mid, 0, 1)
        sum_x = np.sum(x)
        if sum_x > n/2:
            lambda_min = lambda_mid
        else:
            lambda_max = lambda_mid
        if abs(sum_x - n/2) < tol:
            break
            
    h_proj = np.clip(y - lambda_mid, 0, 1)
    return h_proj

def projected_subgradient_descent(h0):
    """Run projected gradient descent minimizing the max overlap"""
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = evaluate_c5_exact(h)

    n_iter = 3000
    alpha0 = 0.01
    
    for t in range(n_iter):
        # Find worst shift index k (correlation index)
        corr = correlate(h, 1-h, mode='full')
        k_max = np.argmax(corr)
        k = k_max - n_points + 1  # reverse of how it's indexed in full mode
        
        # Compute gradient for that shift k
        g = np.zeros(n_points)
        for i in range(n_points):
            if i + k >= 0 and i + k < n_points:
                g[i] = (1 - h[i + k]) - h[i - k] if i - k >= 0 and i - k < n_points else 1 - h[i + k]
            else:
                g[i] = 1 - h[i + k] if i + k >= 0 and i + k < n_points else 1 - h[i - k] if i - k >= 0 and i - k < n_points else 1
            
        # Step
        alpha = alpha0 / sqrt(1 + t)
        h_new = h - alpha * g
        
        # Project
        h = project_to_feasible(h_new)
        
        # Update best if better
        c5 = evaluate_c5_exact(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
    return best_h, best_c5

def base_block_parametrization(u):
    """Create h from block heights u"""
    # Length of each block: blocks == n_points // m, will be integer factor of n
    # u[0: m-1] are free, u[m-1] = 0.5 - sum(u[0:m-1])
    m = len(u)
    u = np.clip(u, 0, 1)
    u_last = 0.5 - np.sum(u[:-1])
    u_last = np.clip(u_last, 0, 1)
    u = np.append(u[:-1], u_last)
    
    # Replicate each value n/m times
    blocks = n_points // m
    h = np.repeat(u, blocks)
    # Ensure length exactly n_points
    h = h[:n_points]
    return h

def binary_rounding(h, k_rounds=30):
    """Probabilistic rounding of continuous h to binary vectors"""
    n = len(h)
    best_h = None
    best_c5 = np.infty
    for _ in range(k_rounds):
        # sampling step: interpret each h[i] as probability, sample binary vector
        b = np.zeros(n)
        # Create list of indices, possibly use sorting or other strategies
        indices = np.arange(n)
        # Shuffle indices to randomize
        np.random.shuffle(indices)
        
        # Binary sampling logic using sorted indices for weight control
        weights = h[indices]
        sorted_indices = indices[np.argsort(-weights)]  # decreasing order
        
        # Sample just enough ones using sort-based approach
        num_ones = n // 2
        b[sorted_indices[:num_ones]] = 1
        
        c5 = evaluate_c5_exact(b)
        if c5 < best_c5:
            best_c5 = c5
            best_h = b.copy()
    return best_h, best_c5
