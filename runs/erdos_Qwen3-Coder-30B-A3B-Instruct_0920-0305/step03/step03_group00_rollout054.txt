def weight_normalise(x, n):
    """Rescales a vector to satisfy sum(x) = n/2"""
    return x * (n / 2) / np.sum(x)

def max_defect(h):
    """Computes exact C5 and the shift that achieves it"""
    corr = correlate(h, 1 - h, mode="full")
    max_val = np.max(corr)
    max_shift = np.argmax(corr) - len(h) + 1
    return max_val / len(h), max_shift

def build_milp(n):
    """Builds and solves the MILP model"""
    # Create variables
    h = Variable(n, boolean=True)
    w = Variable((n, n-1), nonneg=True)  # w[i, s] = h[i] AND h[i+s]  
    t = Variable()
    
    # Objective
    objective = Maximize(t)
    
    # Constraints
    constraints = []
    # Weight constraint  
    constraints.append(sum(h) == n / 2)

    # AND constraints: w[i,s] = h[i] AND h[i+s] 
    for s in range(1, n):
        for i in range(n - s):
            # w[i,s] <= h[i]
            constraints.append(w[i, s-1] <= h[i])
            # w[i,s] <= h[i+s]
            constraints.append(w[i, s-1] <= h[i + s])
            # w[i,s] >= h[i] + h[i+s] - 1
            constraints.append(w[i, s-1] >= h[i] + h[i + s] - 1)

    # Minimum overlap constraints: sum over i of w[i,s] >= t 
    for s in range(1, n):
        constraints.append(sum(w[:, s-1]) >= t)
    
    # Form objective
    prob = Problem(objective, constraints)
    # Solve
    try:
        prob.solve(solver=CPLEX, verbose=False, time_limit=180, ratioGap=0.0)
        return h.value
    except:
        # Return best feasible if solver failed
        return None

def refine_continuous(h0):
    """Refines the binary solution with continuous optimisation"""
    n = len(h0)
    h = h0.copy()
    # Re-scale to fixed weight
    h = weight_normalise(h, n)  # now sum = n/2
    # The spacing is crucial here. For refinements, we want to leave the critical blocks as they are.
    # For practicality:
    # The binary h0 should be deterministic (higher precision)  
    # 
    # Strategy:
    # Enumerate blob boundaries of each 1 or 0 or 0.5 (we use ||0 or 1 part only)
    # After tiling:
    # Each such boundary (not very strict):
    #     Let us split blocks of 1s and 0s.  If a blob has > 1 elements, we allow transition zones.
    #     In our refinement, we only slightly vary a tiny bit of the transition edge 
    #     ( = 1, or capacity or if some weird average is more precise)
    
    # Use smooth max approach with triple epoch 
    h_now = h.copy()
    for alpha in [5, 15, 40]:
        # Solve a min problem Phi_alpha(h) subject to linear constraints h[i] in [0,1], sum(h) = n/2
        # We take a converter over "h" and minimize smooth surrogate with specific weights
        # Define objective function (Φ) and its gradient  
        def obj_grad(h):
            # calling our softmax here; for gradient:
            v, g = softmax_objective(h, alpha)
            # Note: the internal routine already computes corr = correlate(h, 1-h)  
            # which means inner gradient calculator works correctly.
            return v, g
        
        # scipy.optimize.minimize with trust-constr and works with equality
        bounds = [(0, 1) for _ in range(n)]
        # We'll force constraint keeping by solution we initial (triangle)
        # We also split it into multiple parts, and make convergence obj carefully TWO passes at least.
        # Start with the bound method
        res = minimize(
            fun=lambda x: obj_grad(x)[0],
            jac=lambda x: obj_grad(x)[1],
            x0=h_now,  
            method='trust-constr',
            bounds=bounds,
            options={'disp': False, 'maxiter': 200}
        )
        if res.success:
            h_now = res.x
        else:
            # fallback to projected gradient maybe, or accept ...
            pass
        # Re-normalize to weight sum 
        h_now = weight_normalise(h_now, n)
        # Clip to [0, 1] and ensure non-negativity
        h_now = np.clip(h_now, 0, 1)
    
    return h_now 

def anneal(h):
    """Swaps to reduce the peak."},
   """  
    n = len(h)
    h = h.copy()
    max_iter = 5000
    temp = 0.2
    for i in range(max_iter):
        # precompute correlation c[2*n-1] = hidden, we need shift from 1 to n-1 for max.
        # We compute exact shift by which we have the peak. 
        c = correlate(h, 1 - h, mode="full")  
        max_s = np.argmax(c[1:]) + 1   # 1 .. n-1  
        # c is of size 2n-1. 
        # At c[1] => shift 1,  c[2] => shift 2 etc.   
        # Ashift is from 1 to n-1, so in Python, c[1] to c[n-1]  
        which_indices = np.where(h * (1 - h[max_s:]) > 0)[0]  # indices where both h and h-L are non-zero > 0.
        # Oh no, this is wrong. We check if h[i] > 0 and 1-h[i+s] is not too large.
        # Actually columns of correlation matrix should be fine.
        # check for local maximums case, let's compute current values.
        if which_indices.size == 0:
            break
        # Actually we're looking for indices that contribute to current max.
        # value at shift=1 is sum(h[i] * (1 - h[i+1]))
        # What does this mean? 
        # The current shift (max_s) has correlation sum_i h[i] * (1 - h[i-max_s]) = c[max_s]
        # We can optimize it more directly: 
        # i is such that h[i] > 0 and h[i + max_s] is small.  
        # Note: 1-h[i+max_s] is the contribution.
        # Hence, h[i] * (1 - h[i+max_s]) is part of that offset.
        # So we find regions that make positive contribution to c[max_s]  
        # for i < n:
        #    if h[i] * (1 - h[i+max_s]) > 0, we want to reduce this contribution.
        #    That means: reduce h[i] or increase h[i+max_s].
        # But also: h[i+s] should be still 0 or near 0.
        # So get:
        #   i where h[i] > 0
        #   i where h[i+max_s] is small (near 0)
        #   a_i where both are active.
        # But also need to be careful to not remove too much from the right chunk.
        # So:
        # Select such i:
        s = max_s
        # indices where h[i] = 1, h[i + s] is high? We want to reduce the case when they're both > 0.
        # Will require reformulation. Use better representation .
        # The actual update will be very useful!  
        # But three terms involved:
        # Let's just do a naive search:
        # Take i which makes contribution to c max_s.
        indices = []  
        # at c_s = sum_i h[i] * (1 - h[i+s])
        # Thus for i to have non-zero contribution to c_s:
        # h[i] > 0   and  (1 - h[i+s])  > 0 
        # That means h[i] >= 0, h[i+s] <= 1.  
        # Hyperspecific type in neural net:
        # h[i], h[i+s] are both very large (say > 0.9), then h[i] (1-h[i+s]) is small. 
        # if h[i] = 1 but h[i+s] = 1.0, contribution is 0.  
        # We look closely:
        
        # An actual better way:
        # look at h and 1-h, then, correlation c_s = inner product.  
        # When we're looking at c_s, we just compute:
        # take a loop through i to write out sum_i h[i] * (1 - h[i+s])  
        # and ofc the terms that better them up.
        # You can just loop to find positive term, 
        # but more precisely, get indices where h[i] > 0 and 1 - h[i+s] > 0.
        for i in range(n):
            if h[i] > 0.0 and (1 - h[i + s]) > 0.0:
                indices.append(i)
                
        # And also we want i+s to be in range and not already part of current high level region.  
        # Other candidates  - to take high value of 1-h[i-s] 
        # Take not so current highs:
        # j that have h[j + s] high.
        indices_s = [] 
        for i in range(n):
            if i + s < n and h[i + s] < 0.5:  # low h[i+s] 
                indices_s.append(i)
                
        if len(indices) == 0 or len(indices_s) == 0:
            continue
            
        # Now sample one index from each
        i = np.random.choice(indices)
        j = np.random.choice(indices_s)
                
        if i + s >= n or j + s >= n:
            continue
            
        # Decrease h[i], increase h[j]:  
        delta = min(h[i], 1 - h[j]) * temp 
        if delta == 0:
            continue
            
        h[i] -= delta
        h[j] += delta
        
        # And also apply similar to h[i+s] and h[j+s] (as balancing).
        h[i+s] += delta  
        h[j+s] -= delta 
        
        # Now we have not violated constraints:
        # re-normalize indices:
        h = weight_normalise(h, n)
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Cool temperature
        temp *= 0.99
    return h

def run(seed=42, budget_s=600):
    """Run the main algorithm."""
    np.random.seed(seed)
    start = time.time()
    
    # Phase 0: Determine n based on budget.
    # If n=100, schedule can fit; otherwise run n=80.
    n = 100
    if budget_s < 400:
        n = 80
        
    dx = 2.0 / n
    print(f"Using n={n}")
    
    # Phase 1: Binary MILP
    h_bin = None
    # Solve with time limit
    try:
        h_bin = build_milp(n)
        if h_bin is not None:
            h_bin = np.array(h_bin)
            print(f"MILP done, binary solution. Weight {np.sum(h_bin)}")
        else:
            print("MILP returned no valid solution.")
    except Exception as e:
        print(f"MILP problem: {e}")
        h_bin = None
    
    # Fallback if no binary solution
    if h_bin is None or np.all(h_bin == 0):
        print("Falling back to random construction.")
        h_bin = np.random.rand(n)
        h_bin = np.clip(h_bin, 0, 1) 
        h_bin = weight_normalise(h_bin, n)
    
    # Phase 2: Continuous refinement
    h_cont = refine_continuous(h_bin)
    print(f"Continuous refinement done. Before: {max_defect(h_bin)[0]}, After: {max_defect(h_cont)[0]}")
    
    # Phase 3: Swap Anneal
    h_best = anneal(h_cont)
    c5_best = max_defect(h_best)[0]
    print(f"After swap anneal: {c5_best}")
    
    return (h_best.tolist(), c5_best, n)
