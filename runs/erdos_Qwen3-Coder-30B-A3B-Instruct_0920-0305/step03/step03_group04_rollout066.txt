import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time
import cvxpy as cp
from cvxpy import logsumexp
from scipy.special import expit as sigmoid

# Problem parameters
target_bound = 0.3807
n_values = [36, 42, 48, 54, 60]  # Even, ≤ 60
R = 30  # Rounding repetitions
max_swaps = 50000  # For hill climbing
S = 1000  # Restart threshold
max_iters_scs = 8000  # Maximum iterations for SCS
eps_float = 1e-12
dx = 2.0 / 48  # Use 48 since that's given as the best trial size


def init_h():

    return np.array(initial_h_values) if 'initial_h_values' in globals() else np.ones(48) * 0.5


def evaluate_h(h_values):
    """Compute C5 = max_k ∫ h(x)(1 - h(x+k)) dx"""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    return max_correlation * dx


def logsumexp_stable(x, axis=None, keepdims=False):
    # For numerical stability: subtract max before exponentiating
    x_max = np.max(x, axis=axis, keepdims=True)
    return np.log(np.sum(np.exp(x - x_max), axis=axis, keepdims=keepdims)) + x_max


def smooth_max(h, alpha=30):
    """Use soft-max of correlations with parameter alpha"""
    corr = np.correlate(h, 1 - h, mode="full")
    return logsumexp_stable(alpha * corr) / alpha


def constraint_sum(h):
    """Constraint: sum(h) == n/2"""
    return np.sum(h) * dx - 1.0


def proj_to_simplex(y, target_sum=1.0):
    """Project y to the simplex with given sum"""
    y = np.asarray(y)
    u = np.sort(y)[::-1]
    css = np.cumsum(u) - target_sum
    rho = np.argmax(u > css / np.arange(1, len(u) + 1)) if len(u) > 0 else 0
    theta = css[rho] / (rho + 1)
    return np.clip(y - theta, 0, 1)


def solve_sdp(n, max_time_s=30):
    """Solve SDP relaxation for given n"""
    # A = np.zeros((n, n))
    # for i in range(n):
    #     for j in range(n):
    #         if abs(i - j) >= n:
    #             A[i, j] = 0
    #         else:
    #             A[i, j] = 1.0

    x = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    constraints = [
        0 <= x,
        x <= 1,
        cp.sum(x) == n / 2,
        cp.diag(X) == x,
        X >> 0  # PSD
    ]

    # Add McCormick bounds
    for i in range(n):
        for j in range(i + 1, n):
            constraints.extend([
                X[i, j] >= 0,
                X[i, j] <= x[i],
                X[i, j] <= x[j],
                X[i, j] >= x[i] + x[j] - 1
            ])

    # Add overlap constraints
    for k in range(n):
        # The actual partial sums for shift k
        left = x[:n-k]
        diag_k = cp.diag(X, k)
        overlap = cp.sum(left) - cp.sum(diag_k)
        constraints.append(overlap <= t)

    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    try:
        start = time.time()
        prob.solve(solver=cp.SCS, max_iters=max_iters_scs, eps=1e-5)
        elapsed = time.time() - start
        if elapsed > max_time_s or prob.status != cp.OPTIMAL:
            return None, None

        x_sol = x.value
        X_sol = X.value
        t_val = prob.value
        return x_sol, X_sol, t_val
    except Exception as e:
        return None, None, None


def round_from_sdp(X_sol, x_sol, n):
    """Round fractional SDP solution to integer solution"""
    # Do the eigen-decomposition: X ≈ V V^T
    try:
        U, s, Vt = np.linalg.svd(X_sol, full_matrices=False)
        V = U @ np.diag(np.sqrt(np.maximum(s, 0)))
        V = V[:, :min(10, V.shape[1])]  # Use top components
    
        best_h = None
        best_c5 = np.inf
    
        for r_idx in range(R):
            g = np.random.randn(V.shape[1])
            s = np.sign(V @ g)  # shape n
            h = (s > 0).astype(float)
        
            # adjust to have sum = n/2 exactly
            diff = int(np.sum(h) - n/2)
            if diff != 0:
                if diff > 0:
                    # too many ones
                    idxs = np.argsort(np.abs(V @ g))
                    for i in range(diff):
                        h[idxs[i]] = 0
                else:
                    # too few ones
                    idxs = np.argsort(np.abs(V @ g))[::-1]
                    for i in range(-diff):
                        h[idxs[i]] = 1

            assert abs(np.sum(h) - n/2) < 1e-6, "Sum mismatch"
            c5 = evaluate_h(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        return best_h, best_c5
    except Exception as e:
        return None, np.inf


def hill_climb_binary(h, n, max_swaps, restart_threshold=S):
    """
    Hill climb with swapping bits to minimize C5
    h binary, non-zero local search
    """
    # Convert to lists of one and zero indices
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    valid_pairs = [(i, j) for i in ones for j in zeros]
    if len(valid_pairs) == 0:
        return h, evaluate_h(h)
        
    c5_current = evaluate_h(h)
    c5_best = c5_current
    h_best = h.copy()
    iter_count = 0
    restart = 0
    while iter_count < max_swaps:
        if restart >= restart_threshold and restart_threshold > 0:
            h = np.zeros_like(h)
            h[ones] = 1
            h_best = h.copy()
            c5_best = evaluate_h(h)
            iter_count = 0
            restart = 0
        
        # Pick random swap
        idx = np.random.choice(len(valid_pairs))
        i, j = valid_pairs[idx]
    
        # Try the swap
        h_new = h.copy()
        h_new[i] = 1 - h_new[i]
        h_new[j] = 1 - h_new[j]
        
        c5_new = evaluate_h(h_new)
        if c5_new < c5_current:
            h = h_new
            c5_current = c5_new
            # Update indices if necessary
            if h[i] == 1:
                ones.append(i)
                zeros.remove(i)
            else:
                ones.remove(i)
                zeros.append(i)
            if h[j] == 1:
                ones.append(j)
                zeros.remove(j)
            else:
                ones.remove(j)
                zeros.append(j)
            if c5_new < c5_best:
                h_best = h.copy()
                c5_best = c5_new
            restart = 0
        else:
            restart += 1
            if restart > restart_threshold:
                restart = 0
                # Continue or maybe do many swaps to avoid getting stuck
        iter_count += 1
        if c5_best <= target_bound:
            break

    return h_best, c5_best


def refine_continuous(h, alpha=30, lambda_reg=1e3, max_iter=300):
    """Refine h using smooth surrogate objective"""
    # Transform to unconstrained space
    h_clipped = np.clip(h, 1e-6, 1 - 1e-6)
    y = np.log(h_clipped / (1 - h_clipped))  # logit

    def fun(y):
        h_test = sigmoid(y)
        corr = np.correlate(h_test, 1 - h_test, mode="full")
        soft = logsumexp_stable(alpha * corr) / alpha
        penalty = lambda_reg * (np.sum(h_test) * dx - 1.0)**2
        return soft + penalty

    def grad(y):
        h_test = sigmoid(y)
        corr = np.correlate(h_test, 1 - h_test, mode="full")
        # Derivative of logsumexp_alpha(c) = (exp(alpha*c) * alpha) / (sum(exp(alpha*c)))
        expal = np.exp(alpha * corr)
        Z = np.sum(expal)
        dsoft = (alpha / Z) * (expal @ np.ones_like(corr))  # Simplified for now
        # Check structure:
        # It's actually ∂/∂y_i of sum_j c_j * exp(α c_j) / Z
        # Unfortunately we can't unroll this easily in vectorized fashion
        dp = 2 * lambda_reg * (np.sum(h_test) * dx - 1.0) * dx
        # Hessian is discarded in L-BFGS-B (but can be added if needed)
        return dsoft

    result = minimize(fun, y, method="L-BFGS-B", options={"maxiter": max_iter})
    y_final = result.x
    h_final = sigmoid(y_final)
    h_proj = proj_to_simplex(h_final, 1.0)
    return h_proj


def run(seed=42, budget_s=600):
    np.random.seed(seed)
    t0 = time.time()
    best_h = init_h()
    best_c5 = evaluate_h(best_h)
    best_n = len(best_h)
    remaining_time = lambda: budget_s - (time.time() - t0)
    
    # Phase A: SDP
    sdp_solutions = {}
    if remaining_time() > 50:
        for n in n_values:
            if remaining_time() < 40:
                break
            x_sol, X_sol, t_val = solve_sdp(n, max_time_s=35)
            if x_sol is not None:
                sdp_solutions[n] = (x_sol, X_sol, t_val)
    
    # Phase B: rounding from SDP
    roundings = []
    if remaining_time() > 50:
        for n, (x, X, t) in sdp_solutions.items():
            if remaining_time() < 10:
                break
            h_round, c5 = round_from_sdp(X, x, n)
            if h_round is not None and c5 < np.inf:
                roundings.append((h_round, c5, n))
    
    # Phase C: hill climbing on rounding result
    if remaining_time() > 100:
        for h, c5, n in roundings:
            if remaining_time() < 10:
                break
            h_improved, c5_improved = hill_climb_binary(h, n, max_swaps)
            if c5_improved < best_c5:
                best_h, best_c5, best_n = h_improved, c5_improved, n
        
        # Now try a few random starts in parallel (lower shadow)
        for n in n_values[:1 if len(n_values) > 1 else len(n_values)]:
            if remaining_time() < 20:
                break
            for _ in range(10):
                h_rand = np.random.rand(n)
                h_rand = h_rand / np.sum(h_rand) * (n / 2)
                h_rand = np.clip(h_rand, 1e-6, 1-1e-6)
                h_improved, c5_improved = hill_climb_binary(h_rand, n, max_swaps)
                if c5_improved < best_c5:
                    best_h, best_c5, best_n = h_improved, c5_improved, n
                
    # Phase D: continuous refinement
    if remaining_time() > 30:
        try:
            h_refined = refine_continuous(best_h)
            c5_refined = evaluate_h(h_refined)
            if c5_refined < best_c5:
                best_h, best_c5 = h_refined, c5_refined
        except:
            pass  # Skip if no further refinement works

    # Phase E: Final polishing
    if remaining_time() > 5:
        try:
            bounds = [(0, 1) for _ in range(best_n)]
            constraints = {'type': 'eq', 'fun': constraint_sum}
            result = minimize(evaluate_h, best_h, method='SLSQP', bounds=bounds, constraints=constraints,
                             options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-10})
            if result.success:
                best_h = result.x
        except:
            pass  # Fallback to best_h
        
    final_c5 = evaluate_h(best_h)
    return list(best_h), final_c5, best_n
