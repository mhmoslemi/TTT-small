import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def proj_box_simplex(v):
    """Project vector v onto the intersection of box [0,1]^n and simplex {x: sum x_i = 1}"""
    v = np.array(v)
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    # Find threshold
    tmpsum = 0.0
    for i in range(n):
        tmpsum += u[i]
        if tmpsum > 1.0:
            break
    if i == n - 1:
        # All elements are 0
        return np.zeros(n)
    # i is the index of last element that satisfies the constraint
    rho = (1.0 - tmpsum + u[i]) / (i + 1.0)
    # Compute projection
    x = np.maximum(v - rho, 0.0)
    # Renormalize
    x = x / np.sum(x)
    return x

def c5_from_h(h):
    """Compute C5 value for given step function h"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def random_feasible_vector(n):
    """Generate a random feasible vector with sum=1 and entries in [0,1]"""
    u = np.random.uniform(0, 1, n)
    h = u / np.sum(u)
    h = h * (2.0 / n)  # scale to have integral 1
    h = proj_box_simplex(h)
    return h

def spectral_search(n, budget_s, seed=42):

    best_c5 = float('inf')

    dx = 2.0 / n
    h0 = h0 / dx  # Since sum = 1, we scale by dx to have sum = 1/dx = n/2
    H0 = np.fft.fft(h0)
    
    # Target magnitude for non-zero frequencies
    M = np.mean(np.abs(H0[1:]))
    
    # If M is 0 or NaN, just return the original vector
    if np.isnan(M) or M <= 0:
        M = 0.5
    
    # Restart loop
    for r in range(R):
        if time.time() - t_start > budget_s * 0.99:
            break
            
        h = random_feasible_vector(n)
        h = h / dx  # Rely on projection to keep the sum constraint
        
        # Inner iterations
        for t in range(500):
            if time.time() - t_start > budget_s * 0.99:
                break
                
            # FFT
            H = np.fft.fft(h)
            
            # Modify spectrum to have constant magnitude
            H_new = H.copy()
            H_new[1:] = M * np.exp(1j * np.angle(H[1:]))
            
            # Inverse FFT
            h_tilde = np.real(np.fft.ifft(H_new))
            
            # Project onto feasible set (box and simplex)
            h = proj_box_simplex(h_tilde)
            
            # Update best if improved
            c5 = c5_from_h(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Early stopping if no significant improvement in last 10 steps
            if t > 10 and abs(c5 - best_c5) < 1e-8:
                break
                
    # Return best found vector
    if best_h is not None:
        return best_h, best_c5
    else:
        return h0, c5_from_h(h0)

def binary_refinement(h_start, n, budget_s):
    """Refine a continuous vector by making it 'binary-like' and doing local swaps"""
    t0 = time.time()
    
    # Convert to binary
    h = h_start.copy()
    sorted_indices = np.argsort(h)[::-1]
    m = n // 2  # expected number of 1s for sum=1 with dx=2/n
    b = np.zeros(n)
    b[sorted_indices[:m]] = 1.0
    
    # Start with current best
    h_current = b.copy()
    c5_current = c5_from_h(h_current)
    
    # Local combinatorial hill-climb
    # Equivalent to maximizing the negative of C5
    attempts = 0
    success = 0
    max_attempts = 5 * n
    max_no_success = 2 * n
    
    while attempts < max_attempts and success < max_no_success:
        if time.time() - t0 > budget_s * 0.99:
            break
            
        # Random swap
        idx_0 = np.random.choice(n)
        idx_1 = np.random.choice(n)
        if idx_0 == idx_1:
            continue
            
        # Only swap if they differ
        if h_current[idx_0] == h_current[idx_1]:
            continue
            
        # Try swap
        h_candidate = h_current.copy()
        h_candidate[idx_0], h_candidate[idx_1] = h_candidate[idx_1], h_candidate[idx_0]
        
        c5_candidate = c5_from_h(h_candidate)
        # Accept if better (smaller C5)
        if c5_candidate < c5_current:
            h_current = h_candidate.copy()
            c5_current = c5_candidate
            success = 0
        else:
            success += 1
            
        attempts += 1
        
    # Final polishing with projected gradient descent

        # Approximate with logsumexp over all shifts, smoothed version
        dx = 2.0 / n
        corr = correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx, 0
    
    # Reuse original h_current
    best_h = h_current.copy()
    best_c5 = c5_current
    
    return best_h, best_c5
