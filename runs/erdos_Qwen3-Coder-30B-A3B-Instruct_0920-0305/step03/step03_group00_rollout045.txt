import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from cvxpy import *
import time

def weight_normalise(x, n):
    """Rescales a vector to satisfy sum(x) = n/2"""
    return x * (n / 2) / np.sum(x)

def max_defect(h):
    """Computes exact C5 and the responsible shift(s)"""
    corr = correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    max_shifts = np.where(corr == max_val)[0]
    # Adjust for zero-based indexing in full correlation
    actual_shifts = max_shifts - (len(h) - 1)
    return max_val, actual_shifts

def softmax_objective(h, alpha):
    """Returns Φ_α(h) and its gradient"""
    n = len(h)
    # Compute all f_s = sum h[i] * (1 - h[i+s])
    # For numerical stability, we only compute non-zero shifts
    fs = np.array([np.sum(h[i] * (1 - h[i+s])) for s in range(1, n) if i+s < n for i in range(n-s)])
    fs = np.array([np.sum(h[i] * (1 - h[i+s])) for s in range(1, n) for i in range(n-s)])
    fs = fs.reshape(n-1, n-s)
    # Flatten f_s values for log-sum-exp
    fs_flat = fs.flatten()
    # Compute softmax
    exp_fs = np.exp(alpha * fs_flat)
    Z = np.sum(exp_fs)
    phi_alpha = np.log(Z) / alpha
    # Gradient computation
    weights = exp_fs / Z
    grad = np.zeros(n)
    # For each shift s and index i
    for s in range(1, n):
        for i in range(n-s):
            grad[i] += weights[(s-1)*(n-s) + i] * (1 - h[i+s])
            if i+s < n:
                grad[i+s] -= weights[(s-1)*(n-s) + i] * h[i]
    grad /= alpha
    return phi_alpha, grad

def build_milp(n, time_limit=180):
    """Builds and solves the binary MILP."""
    # Variables 
    h = Bool(n)
    w = Variable((n, n-1), boolean=True)
    t = Variable()
    
    # Constraints
    constraints = [
        sum(h) == n/2,
        t <= sum(w[:, s]) for s in range(n-1)]
    
    # AND constraints
    for s in range(1, n-1):
        for i in range(n-s):
            constraints.extend([
                w[i, s] <= h[i],
                w[i, s] <= h[i+s],
                w[i, s] >= h[i] + h[i+s] - 1
            ])
    
    # Objective
    prob = Problem(Maximize(t), constraints)
    
    # Solve
    prob.solve(solver=CPLEX, time_limit=time_limit)
    
    # Extract solution
    h_sol = h.value
    if h_sol is not None:
        return np.round(h_sol).astype(int)
    else:
        return None

def refine_continuous(h_binary, n):
    """Refines binary solution to a continuous one"""
    h = h_binary.astype(float)
    # Keep 1s at block interiors, let boundaries vary with less variables
    L = 4
    # Find block boundaries
    gaps = np.diff(np.concatenate([[0], np.where(np.diff(h) != 0)[0]+1, [n]]))
    blocks = np.split(h_binary, np.where(np.diff(h) != 0)[0] + 1)
    # Replace transition blocks with free variables
    free_vars = []
    free_indices = []
    for i, b in enumerate(blocks):
        if len(b) > 0 and b[0] == 1:  # A block of 1s
            # For each boundary, save transition point index if it's a boundary
            if i > 0:
                idx = np.where(np.diff(h_binary) != 0)[0][i-1] + 1
                if idx < L:
                    free_vars.append(0.5)
                    free_indices.append(idx)
                else:
                    for j in range(min(L, idx)):
                        free_vars.append(0.5)
                        free_indices.append(idx - j)
        # Same for end of block
        if i < len(blocks) - 1:
            idx = np.where(np.diff(h_binary) != 0)[0][i] + 1
            if idx >= n - L:
                free_vars.append(0.5)
                free_indices.append(idx)
            else:
                for j in range(min(L, n - idx)):
                    free_vars.append(0.5)
                    free_indices.append(idx + j)
    
    # Simplified version: only put free variables at spots that are at least 2 away from ends
    # Try to find real transitions
    candidates = np.where(np.diff(h_binary) != 0)[0]
    free_vars = []
    free_indices = []
    for idx in candidates:
        if idx > 1 and idx < n-2:
            free_vars.append(0.5)
            free_indices.append(idx)
        
    # Now begin refining from the initial pattern, applying smooth max with smoothness increasing
    h_initial = h_binary.astype(float)
    free_start = len(free_vars)  
    for alpha in [5, 15, 40]:
        def closure_obj(x):
            # reconstruct h from x
            h_re = h_initial.copy()
            for i, (idx, val) in enumerate(zip(free_indices, x)):
                h_re[idx] = val
            # Compute objective
            max_val, _ = max_defect(h_re)
            return max_val
            
        def closure_grad(x):
            # compute gradient from smooth surrogate
            h_re = h_initial.copy()
            for i, (idx, val) in enumerate(zip(free_indices, x)):
                h_re[idx] = val
            _, g = softmax_objective(h_re, alpha)
            # gradient for variable x associated with fixed indices
            grad = np.zeros_like(x)
            for i, idx in enumerate(free_indices):
                grad[i] = g[idx]
            return grad
        
        # Optimization with fixed box bounds
        bounds = [(0, 1) for _ in range(len(free_vars))]
        # Using trust-constr with analytic gradient
        res = minimize(closure_obj, np.array(free_vars), method='trust-constr', 
                       jac=closure_grad, bounds=bounds,
                       options={'maxiter': 200, 'xtol': 1e-6, 'gtol': 1e-6})
        if res.success:
            for i, idx in enumerate(free_indices):
                h_initial[idx] = res.x[i]
        else:
            break
    return h_initial

def anneal(h_initial, n):
    """Swap annealing to polish final solution"""
    h = h_initial.copy()
    # Let's get an upper bound estimate via exact max
    corr = correlate(h, 1 - h, mode='full')
    corr_full = np.copy(corr) 
    max_def, shift_indices = np.max(corr), np.where(corr == np.max(corr))[0] - (n - 1)
    max_shift = shift_indices[0]
    
    # Slow cooling schedule
    gamma = 0.2  # intensity of change
    iter_count = 0
    max_iter = 5000
    last_best = corr_full[:2*n-1]
    
    while iter_count < max_iter:
        iter_count += 1
        if iter_count > 500 and gamma < 0.02:
            break

        # Find indices that contribute to current max shift
        contrib_indices = np.where((h > 0) & (np.roll(h, max_shift) < 0.5))[0]
        # Try to find a target index that’s low near this shift but high away from it
        target_indices = np.where((h < 0.5) & (np.roll(h, -max_shift) > 0.5))[0]
        
        if len(contrib_indices) > 0 and len(target_indices) > 0:
            i = np.random.choice(contrib_indices)
            j = np.random.choice(target_indices)
            
            # Compute the delta
            delta = min(h[i], 1 - h[j]) * gamma
            # Make the move
            h[i] -= delta
            h[j] += delta
            h[np.clip(i + max_shift, 0, n-1)] -= delta
            h[np.clip(j - max_shift, 0, n-1)] += delta
            # Normalize
            h = weight_normalise(h, n)
            # Recompute the correlator
            corr = correlate(h, 1 - h, mode='full')
            new_max = np.max(corr[:2*n-1])
            if new_max <= max_def:
                max_def = new_max
                max_shift = np.where(corr[:2*n-1] == max_def)[0][0] - (n - 1)
                last_best = corr.full
        gamma *= 0.99
        
    # Return final result and exact bound
    # Final computation of the max defect across all possible shifts
    corr = correlate(h, 1 - h, mode='full')
    final_max_defect = np.max(corr)
    return h, final_max_defect

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 100  # Grid size
    dx = 2.0 / n
    
    # Phase 0: Allocate memory
    time.sleep(0.1)
    if (time.time() - start_time) > 0.5:
        n = 80  # Fallback to smaller grid if needed
        dx = 2.0/n
        # Even smaller allocation might be needed depending on computation time.
    
    # Phase 1: MILP binary solution 
    h_binary = None
    try:
        h_binary = build_milp(n, time_limit=180)
    except Exception:
        pass
    
    if h_binary is None or np.sum(h_binary) != n/2:
        # Fallback to random initialization
        h0 = np.random.random(n)
        h0 = weight_normalise(h0, n)
        h_initial = h0
    else:
        h_initial = h_binary.astype(float)
        
    # Phase 2: refined smooth optimization
    try:
        h_continuous = refine_continuous(h_initial, n)
    except Exception:
        h_continuous = h_initial
    
    # Phase 3: swap-annealing final polishing
    try:
        h_best, c5_final = anneal(h_continuous, n)
    except Exception:
        h_best = h_continuous
        c5_final, _ = max_defect(h_continuous)
        
    return (h_best.tolist(), c5_final, n)
