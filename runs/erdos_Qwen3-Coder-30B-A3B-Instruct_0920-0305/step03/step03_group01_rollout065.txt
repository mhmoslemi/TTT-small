import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
from scipy.optimize import minimize

import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):

    best_c5 = float('inf')
    
    # Stage A: Coarse global search with block heights
    def stage_a():
        n = 200  # Full resolution
        m = 10   # Number of blocks
        
        def block_to_full(u):
            # u is of length m
            # Replicate each element n/m times
            u = np.clip(u, 0, 1)
            u_m = 0.5 * m - np.sum(u[:-1]) if len(u) > 1 else 0.5 * m
            u_m = np.clip(u_m, 0, 1)
            u = np.append(u[:-1], u_m) if len(u) > 1 else np.array([u_m])
            if len(u) == m:
                block_size = n // m
                h_full = np.repeat(u, block_size)
                if len(h_full) < n:
                    h_full = np.append(h_full, np.full(n - len(h_full), u[-1]))
                elif len(h_full) > n:
                    h_full = h_full[:n]
                return h_full
            else:
                return np.ones(n) * 0.5
        
        def objective(u):
            if len(u) != m:
                return float('inf')
            h_full = block_to_full(u)
            c5_val = evaluate_c5(h_full)
            return c5_val
        
        # CMA-ES parameters
        bounds = [(0, 1) for _ in range(m-1)] + [(0, 1)]
        init = np.ones(m) * 0.5
        init[-1] = 0.5 - np.sum(init[:-1])
        
        # Use differential evolution to get better starting point
        # But we want low dimension space for faster evolution
        res = differential_evolution(
            objective, 
            bounds,
            seed=seed,
            maxiter=1500,
            popsize=4 + m,
            mutation=(0.5, 1),
            recombination=0.7,
            strategy='best1bin'
        )
        
        if res.success:
            u_star = res.x
            h_star = block_to_full(u_star)
            return h_star
        else:
            # fallback to straight line
            h = np.ones(n) * 0.5
            return h

    def stage_b(h0, n=200):
        dx = 2.0 / n
        num_iter = 3000
        alpha0 = 0.01
        h = h0.copy()
        best_h = h.copy()
        best_c = evaluate_c5(h)
        
        for t in range(num_iter):
            # Compute overlap for all k
            corr = correlate(h, 1 - h, mode="full")
            k_max = np.argmax(corr)
            k_best = k_max - n + 1  # Convert to actual shift value
            
            # Compute sub-gradient w.r.t. h for shift k_best
            g = np.zeros(n)
            for i in range(n):
                # compute derivative of overlap with shift k_best
                # contribution of h_i = h_i * (1 - h[i + k_best])
                if i + k_best >= 0 and i + k_best < n:
                    g[i] = (1 - h[i + k_best])
                if i - k_best >= 0 and i - k_best < n:
                    g[i] -= h[i - k_best]
            
            # Step and maintain boundedness
            alpha = alpha0 / np.sqrt(t + 1)
            h = h - alpha * g
            
            # Water-filling projection to the feasible set
            y = h.copy()
            target_sum = n / 2.0
            
            def objective_lambda(lam):
                x = np.clip(y - lam, 0, 1)
                return np.sum(x) - target_sum
            
            # Bisection for lambda
            lb, ub = -2, 3
            for _ in range(50):
                lam = (lb + ub) / 2
                val = objective_lambda(lam)
                if val < 0:
                    ub = lam
                else:
                    lb = lam
            lam = (lb + ub) / 2
            h = np.clip(y - lam, 0, 1)
            
            # Update best solution
            c = evaluate_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            
            if t > 200 and (best_c - evaluate_c5(h)) < 1e-6:
                # Fallback cleanup
                c = evaluate_c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                break
        
        return best_h, best_c
    
    # Stage C: Binary rounding and local search
    def stage_c(h_best):
        n = len(h_best)
        K = 30  # number of random rounds
        best_round = None
        best_round_c5 = float('inf')
        
        # Random rounding attempts
        for _ in range(K):
            # Probabilistic rounding that respects the sum constraint
            indices = np.argsort(-h_best)
            b = np.zeros(n, dtype=int)
            count = 0
            
            for i in indices:
                if count >= n // 2:
                    break
                if np.random.rand() < h_best[i]:
                    b[i] = 1
                    count += 1
            
            # Fix exact count if needed
            if count < n // 2:
                # indices not set to 1
                remaining = np.where(b == 0)[0][:n//2-count]
                b[remaining] = 1
            elif count > n // 2:
                # Remove extra 1s
                ones = np.where(b == 1)[0]
                to_remove = count - n // 2
                remove_indices = np.random.choice(ones, to_remove, replace=False)
                b[remove_indices] = 0
                
            c5_round = evaluate_c5(b.astype(float))
            if c5_round < best_round_c5:
                best_round_c5 = c5_round
                best_round = b.copy()
        
        if best_round is None:
            return h_best.copy(), float('inf')
        
        # Local search on this binary candidate
        best_b = best_round
        b = best_round.copy()
        b_prev = b.copy()
        h = b.astype(float)
        
        # Precompute all contributions (O(n^2) but done once per b)
        def compute_overlaps(bv):
            # use numpy approaches
            corr = correlate(bv, 1 - bv, mode="full")
            # return maximum overlapping term
            return np.max(corr)
        
        # rounding-off criteria (precision)
        max_iter = 10
        for _ in range(max_iter):
            improved = False
            
            # Sample pairs (i,j) where b_i=1, b_j=0
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            # limited pairs
            pairs = []
            num_pairs_sample = min(10000, len(ones) * len(zeros) // 10)
            for _ in range(num_pairs_sample):
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                pairs.append((i, j))
            
            best_delta = 0
            best_pair = None
            
            for (i, j) in pairs:
                # compute change in overlap
                delta_shifts = np.zeros(2 * n - 1)
                
                # change in each shift k
                old_val = h[i] * (1 - h[i + 1]) if i + 1 < n else 0.0
                new_val = h[j] * (1 - h[j + 1]) if j + 1 < n else 0.0
                delta_shifts[1 + 1] = new_val - old_val
                
                # Skip if other indices don't affect the shift k properly
                old_val = h[i] * (1 - h[i - 1]) if i - 1 >= 0 else 0.0
                new_val = h[j] * (1 - h[j - 1]) if j - 1 >= 0 else 0.0
                delta_shifts[n - 1 + (1 - 1)] = new_val - old_val
                
                # For now, just focus on overlaps between directly adjacent or symmetric around 0
                # Pourhose simulation
                    
                # You can better simulate each overlap change by computing two invertible corrs
                # But that's too expensive, temporal gaps in small improvement
                # Realtime updates are also good, but even those are inexact
                old_overlap = compute_overlaps(b)
                b[i], b[j] = 0, 1  # Swap in binary representation
                new_overlap = compute_overlaps(b)
                b[i], b[j] = 1, 0  # revert
                
                if old_overlap - new_overlap > best_delta:
                    best_delta = old_overlap - new_overlap
                    best_pair = (i, j)
            
            if best_pair is not None and best_delta > 1e-10:
                i, j = best_pair
                b[i], b[j] = 0, 1
                h = b.astype(float)
                improved = True
            else:
                break
        
        final_h = h.copy()
        final_c5 = evaluate_c5(final_h)
        
        return final_h, final_c5
    
    # Stage D: MILP refinement
    def stage_d(h_binary, n=200, remaining_time=600):
        # Exact MILP formulation
        if remaining_time < 30 or n < 180:

        dx = 2.0 / n
        
        # define continuous h as binary variables
        h_vars = Variable(n, boolean=True)
        
        # constraint sum = n/2
        constraints = [sum(h_vars) == n // 2]
        
        # variables for product w_{i,k} = h_i * h_{i+k}
        rows = []
        shifted_vars = []
        for i in range(n):
            row = []
            for k in range(n):
                # w[i, k] = h_i * h_{i + k} (with zero padding)
                # we define this only for valid pairs:
                if i + k < n:
                    w = Variable(boolean=True)
                    row.append(w)
                    # enforce w = h_i AND h_{i+k}
                    constraints.extend([
                        w >= h_vars[i] + h_vars[i + k] - 1,
                        w <= h_vars[i],
                        w <= h_vars[i + k]
                    ])
                else:
                    row.append(None)
            rows.append(row)
            
        # Compute max overlap over all k
        max_overlaps = []
        for k in range(1, n):  # Skip shift 0
            # S_k = Σ_i h_i h_{i+k}
            S_k = 0
            for i in range(n):
                if i + k < n:
                    S_k += rows[i][k]
            # overlap = n/2 - S_k
            max_overlaps.append(S_k)
        
        overlaps_sum = sum(max_overlaps)
        half = n / 2.0
        
        # If not enough info, just cap it to avoid errors
        # can be omitable in clean versions, but this avoids bound issues
        
        # Define the optimization model
        C = Variable()
        for i in range(len(max_overlaps)):
            constraints.append(max_overlaps[i] + C >= half)
        
        objective = Minimize(C)
        prob = Problem(objective, constraints)

            prob.solve(solver=CVXOPT, verbose=False, 
                       solver_kwargs={'maximumSeconds': remaining_time - 5})
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                C_val = C.value
                # Convert to continuous version for return
                if C_val is not None:
                    # Computing overlap given a binary vector
                    h_final = [float(x.value) for x in h_vars]
                    final_c5 = C_val * dx
                    return h_final, final_c5

    if (time.time() - start_time) < 0.8 * budget_s:
        try:
            h_coarse = stage_a()
            # refine with Projected Subgradient.
            h_stage_b, c5_stage_b = stage_b(h_coarse, 200)
            if c5_stage_b < best_c5:
                best_h = h_stage_b
                best_c5 = c5_stage_b
        except:
            pass
    
    # Phase B - Local Search
    if (time.time() - start_time) < 0.9 * budget_s:
        try:
            h_refined, c5_refined = stage_c(best_h if best_h is not None else np.ones(200) * 0.5)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
        except:
            pass
    
    # Phase C - MILP solvers
    if best_h is not None:
        try:
            remaining = budget_s - (time.time() - start_time)
            if remaining > 10:
                h_milp, c5_milp = stage_d(best_h, n=200, remaining_time=remaining)
                if c5_milp < best_c5:
                    best_h = h_milp
                    best_c5 = c5_milp
        except Exception as e:
