def proj_box_simplex(v):
    """Project v onto the intersection of [0,1]^n and {x: sum(x) = 1}"""
    # Sort in descending order
    u = np.sort(v)[::-1]
    # Compute cumulative sums
    cu = np.cumsum(u)
    # Find the threshold index
    ind = np.arange(1, len(u) + 1)
    mask = u > (1.0 - cu) / ind
    idx = np.sum(mask)
    # Compute threshold
    if idx == 0:
        t = 1.0 / len(v)
    else:
        t = (1.0 - cu[idx - 1]) / idx
    # Project to [0,1] and enforce sum = 1
    x = np.maximum(v - t, 0)
    x = (x / np.sum(x)) * 1.0
    return x

def c5_from_h(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx
