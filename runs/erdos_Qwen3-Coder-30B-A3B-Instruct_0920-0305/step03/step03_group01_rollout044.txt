import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage A: Low-dimensional global optimization
    print("Stage A: Low-dimensional global optimization")
    n_points = 200
    dx = 2.0 / n_points
    m = 10  # number of blocks
    bounds = [(0, 1) for _ in range(m - 1)]  # last one will be constrained
    popsize = 4 + m

    def block_to_full(u_vals):
        # Clip u_m to [0, 1] and compute u_m
        u_m = max(0, min(1, m / 2 - sum(u_vals)))
        u_vals = np.append(u_vals, u_m)
        
        # Fill full vector using block pattern
        points_per_block = n_points // m
        h_full = np.repeat(u_vals, points_per_block)
        h_full = np.pad(h_full, (0, n_points - len(h_full)), mode="constant", constant_values=0)
        return h_full[:n_points]

    def objective_block(u_vals):
        h_full = block_to_full(u_vals)
        # Normalize to satisfy integral constraint
        scale = 1.0 / (np.sum(h_full) * dx)
        h_full = h_full * scale
        # Evaluate C5
        return evaluate_c5(h_full)

    best_c5 = float('inf')
    best_h = None

    try:
        from cma import CMAEvolutionStrategy
        es = CMAEvolutionStrategy([0.5] * (m - 1), 0.2, {'popsize': popsize, 'verbose': -1})
        max_evals = 1500
        evals = 0
        while evals < max_evals and (time.time() - start_time) < budget_s * 0.2:
            X = es.ask()
            F = []
            for x in X:
                try:
                    val = objective_block(x)
                    F.append(val)
                except:
                    F.append(float('inf'))
            es.tell(X, F)
            evals += len(X)
            if np.min(F) < best_c5:
                idx = np.argmin(F)
                best_c5 = F[idx]
                best_h = block_to_full(X[idx])
    except:
        # fallback to simpler optimization if cma not available
        pass

    if best_h is None:
        # fallback to a simple random initialization
        h0 = np.array(initial_h_values, dtype=float)
        # rescale to ensure the integral is 1:
        dx = 2 / len(h0)
        scale = 1.0 / (np.sum(h0) * dx)
        best_h = h0 * scale
        best_c5 = evaluate_c5(best_h)
    else:
        # normalize best_h to have integral = 1
        dx = 2 / n_points
        scale = 1.0 / (np.sum(best_h) * dx)
        best_h = best_h * scale
        best_c5 = evaluate_c5(best_h)

    print("Stage A completed with C5 = ", best_c5)
    
    # Stage B: Full-resolution projected gradient descent
    print("Stage B: Full-resolution projected gradient descent")
    def project_onto_feasible_space(x):
        # project x onto simplex {x : sum(x)=n/2, 0<=x<=1}
        # uses iterative waterfilling method
        # start from y = x
        y = np.copy(x)
        # Solve for λ such that sum of clip(yᵢ - λ, 0, 1) = n/2
        def sum_of_clipped(lam):
            return np.sum(np.clip(y - lam, 0, 1))
        low, high = -2, 3
        for _ in range(50):
            mid = (low + high) / 2
            val = sum_of_clipped(mid)
            if val > n_points / 2:
                low = mid
            else:
                high = mid
        lam = (low + high) / 2
        x_proj = np.clip(y - lam, 0, 1)
        return x_proj

    h_best = best_h.copy()
    current_c5 = best_c5
    max_iter = 3000
    alpha0 = 0.1  # initial step size
    improvement = 1e-6
    last_best_c5 = current_c5 + 100
    for it in range(max_iter):
        if (time.time() - start_time) > budget_s * 0.6:
            break
            
        # find worst shift
        corr = correlate(h_best, 1 - h_best, mode="full")
        max_shift_idx = np.argmax(corr)
        k = max_shift_idx - (n_points - 1)
        
        # compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k >= 0 and i + k < n_points:
                grad[i] += 1 - h_best[i + k]
            if i - k >= 0 and i - k < n_points:
                grad[i] -= h_best[i - k]
                
        # gradient step
        alpha = alpha0 / np.sqrt(it + 1)
        h_new = h_best - alpha * grad
        h_new = project_onto_feasible_space(h_new)
        
        # update current iterate
        h_best = h_new
        current_c5 = evaluate_c5(h_best)
        
        # check for convergence
        if abs(current_c5 - last_best_c5) < improvement:
            break
        if current_c5 < last_best_c5:
            last_best_c5 = current_c5
    
    print("Stage B completed with C5 = ", current_c5)
    best_c5 = current_c5
    best_h = h_best
    
    # Stage C: Binary rounding and local search
    print("Stage C: Binary rounding and local search")
    
    def get_binary_rounding(h, k_rounds=30):
        # round probabilistically to binary vector
        best_h_bin = None
        best_c5_val = float('inf')
        
        for _ in range(k_rounds):
            # probabilistic rounding
            order = np.random.permutation(n_points)
            count = 0
            hBin = np.zeros(n_points)
            for idx in order:
                if count >= n_points / 2:
                    break
                if np.random.random() < h[idx]:
                    hBin[idx] = 1
                    count += 1
            
            # fill remaining positions with 0s if needed
            remaining = n_points // 2 - count
            if remaining > 0:
                for i in range(n_points):
                    if count >= n_points / 2:
                        break
                    if hBin[i] == 0:
                        hBin[i] = 1
                        count += 1
            
            c5_new = evaluate_c5(hBin)
            if c5_new < best_c5_val:
                best_c5_val = c5_new
                best_h_bin = hBin.copy()
                
        return best_h_bin, best_c5_val
    
    def update_overlaps_for_swap(h, S_k, i_idx, j_idx, k_idx):
        # assume out-of-range = 0
        n = len(h)
        # check which shifts these indices affect:
        for k in range(1, n):
            # for k-th shift, these are the indices that affect it:
            # i and i+k
            # j and j+k
            # i and i-k
            # j and j-k
            # for backwards shifts (k < i and k < j) we need 1-(i-k) +=, 1-(j-k) +=,
            # etc but lets just compute all line by line based on existing contributions
            # increment/decrement overlap entries when we do the swap
            # (i swapped) affects input: h[i] * h[i+k]
            # and h[i] * h[i-k]
            if i_idx + k < n and k_idx > 0:
                old_val_i = h[i_idx] * h[i_idx + k]
                new_val_i = (1 - h[i_idx]) * h[i_idx + k]
                delta_S = (new_val_i - old_val_i)  # since it's a 1 - term in overlap
                S_k[k] += delta_S
            
            if i_idx >= k and k_idx > 0:
                old_val_i = h[i_idx] * h[i_idx - k]
                new_val_i = (1 - h[i_idx]) * h[i_idx - k]
                delta_S = (new_val_i - old_val_i)
                S_k[k] += delta_S
                
            if j_idx + k < n and k_idx > 0:
                old_val_j = h[j_idx] * h[j_idx + k]
                new_val_j = h[j_idx] * (1 - h[j_idx + k])
                delta_S = (new_val_j - old_val_j)
                S_k[k] += delta_S
                
            if j_idx >= k and k_idx > 0:
                old_val_j = h[j_idx] * h[j_idx - k]
                new_val_j = h[j_idx] * (1 - h[j_idx - k])
                delta_S = (new_val_j - old_val_j)
                S_k[k] += delta_S
    
    def evaluate_binary_swap(h, S_k, i_idx, j_idx):
        # compute the change in overlap
        old_c5 = 1 - S_k
        old_max = np.max(old_c5)
        # effect of swap of i with j:
        # h[i] becomes 0, h[j] becomes 1
        # compute new S_k values in place
        S_new = S_k.copy()
        k_start = 1
        k_end = n_points - 1
        # this is an approximate but efficient way to estimate the new values for all k
        # (this does not cover all indices but should be quite close for most cases)
        # for simplicity we probably use dumb zero padding for boundary if needed
        for k in range(k_start, k_end):
            if i_idx + k < n_points and j_idx >= k:
                # i and i+k, j and j+k are now j and j+k, i and i+k respectively
                S_new[k] = S_k[k] - h[i_idx] * h[i_idx + k] + (1 - h[i_idx]) * h[i_idx + k] - h[j_idx] * h[j_idx - k] + h[j_idx] * h[j_idx - k]
            elif i_idx >= k and j_idx + k < n_points:
                # j and j-k, i and i-k are now i and i-k, j and j-k respectively
                S_new[k] = S_k[k] - h[i_idx] * h[i_idx - k] + h[i_idx] * h[i_idx - k] - h[j_idx] * h[j_idx + k] + h[j_idx] * h[j_idx + k]
        new_c5 = 1 - S_new
        new_max = np.max(new_c5)
        delta = new_max - old_max
        return delta
    
    # get binary rounding
    h_bin, c5_bin = get_binary_rounding(h_best)
    if c5_bin < best_c5:
        best_h = h_bin
        best_c5 = c5_bin
        
    # local search with swaps
    h_bin = best_h.copy()
    c5_bin = best_c5
    # precompute S_k values (overlap sums)
    # from the original h_bin
    corr = correlate(h_bin, h_bin, mode="full")
    S_k = np.zeros(n_points)
    for k in range(1, n_points):
        # compute overlap for shift k...
        # note that correlation h[i] * h[i+k] only contributes to overlap ((n/2) - sum_{i=0}^{n-1} h[i] * h[i+k])
        S_k[k] = np.sum(h_bin * np.roll(h_bin, k)[::-1])
        
    # optimize with swaps
    for round_idx in range(10):  # run several rounds
        improved = True
        iteration = 0
        while improved and iteration < 1000:
            improved = False
            # sample a small fraction of all possible pairs
            h_bin = best_h.copy()
            c5_bin = best_c5
            np.random.seed(seed + round_idx * 100 + iteration)
            num_samples = min(2000, n_points * n_points // 100)
            hit_pairs = set()
            for _ in range(num_samples):
                i_idx, j_idx = np.random.randint(0, n_points, 2)
                if h_bin[i_idx] == 1 and h_bin[j_idx] == 0:
                    # check if we already sampled this pair
                    key = (min(i_idx, j_idx), max(i_idx, j_idx))
                    if key not in hit_pairs:
                        hit_pairs.add(key)
                        delta = evaluate_binary_swap(h_bin, S_k, i_idx, j_idx)
                        if delta < -1e-8:  # if improvement
                            # perform the swap
                            h_bin[i_idx] = 0
                            h_bin[j_idx] = 1
                            # update S_k for the affected shifts
                            update_overlaps_for_swap(h_bin, S_k, i_idx, j_idx, 0)
                            best_h = h_bin
                            best_c5 = 1 - np.max(S_k)  # because overlap = (n/2) - sum of products
                            improved = True
                            break
            iteration += 1
            
    # final evaluation
    final_c5 = evaluate_c5(best_h)

    print("Stage C completed with C5 = ", final_c5)
    
    # Final MILP refinement (if time permits)
    if (time.time() - start_time) > budget_s * 0.8:
        print("Skipping MILP due to time constraint")
    else:
        try:
            print("Stage D: MILP refinement")
            # define the MILP
            b_vars = cp.Variable(n_points, boolean=True)
            # constraint
            constraint1 = cp.sum(b_vars) == n_points // 2
            # define w_{i,k} = b_i * b_{i+k} for valid indices
            constraints = [constraint1]
            
            # only enumerate those that are relevant:
            n_shifts = int(0.5 * n_points) # example - use a reasonable number
            # we'll do this block at a time to make it manageable
            corr = correlate(best_h, 1 - best_h, mode="full")[:]  # ignore middle term which is 0 (or voluntary zeros)
            worst_shifts = np.argsort(corr)[-n_shifts:][::-1]  # get the worst shifts
            
            C = cp.Variable(1, pos=True)
            # for each shift k in worst_shifts:
            for k in worst_shifts:
                # note we're mapping the Correlation index k -> shift k: 
                # it corresponds to convolution, so shift k corresponds to w_{i, k} = b_i * b_{i+k}
                # thus the actual overlap value = sum_i b_i * (1 - b_{i+k})
                # arise from: [b_i * 1 = b_i], [b_i * 0 = 0], so b_i - (sum b_i * b_{i+k})
                # --> sum b_i - sum b_i * b_{i+k}
                #            == n/2 - sum b_i * b_{i+k}
                # hence the inequality becomes:  n/2 - sum b_i * b_{i+k} <= C
                # or sum b_i * b_{i+k} + C >= n/2
                # (there are really 2 terms u_{i,k} = b_i*b_{i+k} (0 or 1) and the constraint is 
                # sum b_i * b_{i+k} + C >= 100  so that sum u_{i,k} >= 100 - C)
                
                # this is a simpler version
                # Let us set up a table
                w_vars = cp.Variable((n_points,), boolean=True)
                w_and = []
                # Adding constraints: w_{i,k} = b_i and b_{i+k}
                for i in range(n_points):
                    if i + k < n_points:
                        # add w_{i,k} <= b_i
                        w_and.append(w_vars[i] <= b_vars[i])
                        # add w_{i,k} <= b_{i+k}
                        w_and.append(w_vars[i] <= b_vars[i+k])
                        # add w_{i,k} >= b_i + b_{i+k} - 1
                        w_and.append(w_vars[i] >= b_vars[i] + b_vars[i+k] - 1)
                    
                # define overlap at shift k:
                ov_k = (n_points / 2) - cp.sum(w_vars)
                # relaxation condition:
                # ov_k - C <= 0  leads to ov_k <= C
                constraints += w_and
                constraints.append(ov_k <= C)
                
            # objective
            objective = cp.Minimize(C)
            prob = cp.Problem(objective, constraints)
            # solve with small time limit  
            prob.solve(solver=CP.SCS, verbose=False)  # using solver with less memory pressure
            
            # update result if LP results in better bound
            if prob.status == 'optimal' or prob.status == 'unknown':
                new_c5 = float(C.value) * dx
                if new_c5 < final_c5:
                    print(f"MILP gives improved bound {new_c5}")
                    final_c5 = new_c5
        except Exception as e:
            print(f"MILP failed due to error: {e}")
            pass
    
    return (best_h.tolist(), final_c5, n_points)
