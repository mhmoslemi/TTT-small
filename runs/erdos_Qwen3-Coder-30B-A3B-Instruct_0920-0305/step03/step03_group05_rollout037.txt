def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

    # For large n this will be very slow to compute exactly, so we keep it simple
    
    # For simplicity of implementation, use a rough approximation using a symmetric difference
    # A better approach would be to compute the actual gradient of correlation, 
    # but for the moment we use a heuristic.
    for i in range(n):
        h_test = h.copy()
        h_test[i] += step_size
        if h_test[i] > 1.0: h_test[i] = 1.0
        if h_test[i] < 0.0: h_test[i] = 0.0
        
        # Check the effect on the correlation
        h_test2 = h.copy()

        v1 = evaluate_h(h_test)
        v2 = evaluate_h(h_test2)
        grad[i] = (v1 - v2) / (2 * step_size)
    
    # Gradient ascent
    h_new = h - step_size * grad
    h_new = np.clip(h_new, 0, 1)
    
    # Rescale to satisfy constraint
    current_sum = np.sum(h_new) * dx
    h_new *= (1.0 / current_sum)
    
    return h_new

    # Normalize
    current_sum = np.sum(h) * dx
    if abs(current_sum - 1.0) > 1e-6:
        h *= (1.0 / current_sum)
    
    # Optionally add a second peak
    center2 = 1.5
    width2 = 0.2
    for i in range(n_points):
        x = i * dx
        h[i] += max(0, 1 - abs(x - center2) / width2) * 0.5
    
    # Normalize again
    current_sum = np.sum(h) * dx
    if abs(current_sum - 1.0) > 1e-6:
        h *= (1.0 / current_sum)
    
    h = np.clip(h, 0, 1)
    return h

def local_search(h_values, max_iter=200):
    """Refine solution using local optimization."""
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    def objective(h):
        return evaluate_h(h)
    
    try:
        # Use L-BFGS-B for better convergence properties
        result = minimize(objective, h_values, method='L-BFGS-B', 
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-10})
        if result.success:
            return result.x
    except Exception as e:
        pass
    return h_values

    if abs(current_sum - 1.0) > 1e-6:
        h_new *= (1.0 / current_sum)
    return h_new
