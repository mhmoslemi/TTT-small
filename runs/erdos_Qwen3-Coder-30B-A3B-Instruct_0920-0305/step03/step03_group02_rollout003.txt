# Constants

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def local_search(h_values, max_iter=500):
    """Refine solution using local optimization."""
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    def objective(h):
        return evaluate_h(h)
    
    try:
        result = minimize(objective, h_values, method='SLSQP', 
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-10})
        if result.success:
            return result.x
    except:
        pass
    return h_values

def perturb_solution(h_values, perturbation=0.05):
    """Perturb solution slightly."""
    h_new = h_values.copy()
    indices = np.random.choice(n_points, size=max(1, int(0.1 * n_points)), replace=False)
    for i in indices:
        h_new[i] = np.clip(h_new[i] + np.random.uniform(-perturbation, perturbation), 0, 1)
    # Normalize to satisfy integral constraint
    current_sum = np.sum(h_new) * dx
    if abs(current_sum - 1.0) > 1e-6:
        h_new *= (1.0 / current_sum)
    return h_new

    gbest_value = float('inf')

    c1 = 1.5  # cognitive
    c2 = 1.5  # social

    for iter_num in range(max_iter):
        for i in range(n_particles):
            h = particles[i]
            current_value = evaluate_h(h)
            
            # Update personal best
            if current_value < gbest_value:
                gbest = h.copy()
                gbest_value = current_value
            
            # Update velocity and position
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (h - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            particles[i] += velocities[i]
            # Clip to [0,1]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize
            particles[i] *= (1.0 / (np.sum(particles[i]) * dx))
    
    return gbest, gbest_value
