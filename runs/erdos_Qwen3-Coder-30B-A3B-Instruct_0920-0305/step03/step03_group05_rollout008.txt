import numpy as np
import cvxpy as cp
import time

def solve_milp_binary(n, time_limit=60):
    """Solve the binary MILP for the Erdős problem."""
    dx = 2.0 / n
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Ensure sum equals n/2
    constraints = [cp.sum(h) == n // 2]
    
    # Build y variables and constraints for each shift d
    y_vars = {}
    for d in range(n):
        # y_{i,d} = h_i * h_{i+d} for i = 0 to n-1-d
        for i in range(n - d):
            y = cp.Variable(boolean=True)
            y_vars[(i, d)] = y
            constraints += [
                y <= h[i],
                y <= h[i + d],
                y >= h[i] + h[i + d] - 1
            ]
    
    # Add constraints that Q_d >= t for all d
    for d in range(n):
        if d == 0:
            # Q_0 = sum h_i
            Q_d = cp.sum(h)
        else:
            # Q_d = sum of y_{i,d}
            Q_d = cp.sum([y_vars[(i, d)] for i in range(n - d)])
        constraints.append(Q_d >= t)
    
    # Objective: maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    try:
        prob.solve(solver=cp.CBC, warm_start=False, 
                   verbose=False, 
                   eps_abs=1e-6, 
                   eps_rel=1e-6,
                   seconds=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h.value.astype(float)
            t_val = t.value
            c5_bound = 1 - (2/n) * t_val
            return h_sol, c5_bound
        elif prob.status in [cp.INFEASIBLE, cp.UNBOUNDED]:
            return None, float('inf')
        else:
            # Try to get a feasible solution from the solver if available
            try:
                if h.value is not None:
                    h_sol = h.value.astype(float)
                    t_val = t.value if t.value is not None else 0
                    c5_bound = 1 - (2/n) * t_val
                    return h_sol, c5_bound
            except:
                pass
            return None, float('inf')
    except Exception:
        return None, float('inf')


def hill_climb_swaps(h_values, max_swaps=100000, time_limit=30):
    """Improve a binary solution using swaps."""
    start_time = time.time()
    h = h_values.copy()
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1 - h)
    dx = 2.0 / len(h)
    
    # Compute initial overlap
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    
    swaps_done = 0
    while swaps_done < max_swaps and time.time() - start_time < time_limit:
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # Swap the values
        h[i], h[j] = h[j], h[i]
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        
        # Compute new correlation
        corr = np.correlate(h, 1 - h, mode="full")
        new_c5 = np.max(corr) * dx
        
        if new_c5 >= c5:
            h[i], h[j] = h[j], h[i]
            ones = np.flatnonzero(h)
            zeros = np.flatnonzero(1 - h)
        else:
            c5 = new_c5
        swaps_done += 1
    
    return h, c5
