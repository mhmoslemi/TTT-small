import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
from cvxpy import *
import time
import math

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial solution
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    
    best_h = initial_h.copy()
    best_C5 = evaluate_c5(initial_h)
    best_n_points = n_points
    
    # Exact binary optimization for small n
    def solve_binary_milp(n, time_limit):
        # Create variables
        b = Variable(n, boolean=True)
        y = Variable((n, n), boolean=True)
        t = Variable()
        
        # Constraints
        constraints = [
            sum(b) == n//2,
            t >= 0
        ]
        
        # Product constraints
        for i in range(n):
            for j in range(i, n):
                constraints.extend([
                    y[i, j] <= b[i],
                    y[i, j] <= b[j],
                    y[i, j] >= b[i] + b[j] - 1
                ])
        
        # For each shift s = 0 to n-1
        for s in range(n):
            overlap = 0
            for i in range(n - s):
                overlap += y[i, i + s]
            constraints.append(overlap >= t)
        
        # Objective
        objective = Maximize(t)
        prob = Problem(objective, constraints)
        
        # Solve
        prob.solve(solver=CPLEX, time_limit=time_limit)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            b_val = b.value
            t_val = t.value
            return b_val, t_val
        else:
            return None, None

    # Exact optimization for n=42
    if (time.time() - start_time) < budget_s * 0.3:
        try:
            b_opt, t_opt = solve_binary_milp(42, budget_s * 0.2)
            if b_opt is not None:
                h_binary = b_opt.astype(float)
                dx = 2.0 / 42
                C5_binary = (21 - t_opt) * dx  # 21 = 42/2
                if C5_binary < best_C5:
                    best_h = h_binary
                    best_C5 = C5_binary
                    best_n_points = 42
        except:
            pass

    # Annealing on larger lengths
    lengths = [44, 48, 52, 56, 60]
    steps_per_length = int(budget_s * 0.3 / len(lengths)) if len(lengths) > 0 else 0

    for n in lengths:
        if (time.time() - start_time) > budget_s * 0.6:
            break
        try:
            # Initial random binary with equal 0 and 1
            h_current = np.zeros(n)
            ones = np.random.choice(n, n//2, replace=False)
            h_current[ones] = 1
            
            # Simulated annealing for binary vectors
            current_energy = evaluate_c5(h_current)
            best_energy = current_energy
            best_h_binary = h_current.copy()
            
            max_steps = steps_per_length
            step = 0
            T0 = 1.0
            T = T0
            
            while step < max_steps and (time.time() - start_time) < budget_s * 0.9:
                # Pick random 1 and 0
                one_idx = np.random.choice(np.where(h_current == 1)[0])
                zero_idx = np.random.choice(np.where(h_current == 0)[0])
                
                # Swap
                h_new = h_current.copy()
                h_new[one_idx] = 0
                h_new[zero_idx] = 1
                
                new_energy = evaluate_c5(h_new)
                
                if new_energy < current_energy or np.random.rand() < np.exp(-(new_energy - current_energy) / T):
                    h_current = h_new
                    current_energy = new_energy
                
                if current_energy < best_energy:
                    best_energy = current_energy
                    best_h_binary = h_current.copy()
                
                T = T0 * (1.0 - step / max_steps)
                step += 1
            
            # Convert to float and refine
            h_float = best_h_binary.astype(float)
            dx = 2.0 / n
            h_float = h_float / (dx * np.sum(h_float))  # Normalize to sum=1
            
            # Continuous refinement via iterative gradient step
            max_iter = 1000
            lr = 1e-3
            for i in range(max_iter):
                if (time.time() - start_time) > budget_s * 0.95:
                    break
                C = evaluate_c5(h_float)
                corr = correlate(h_float, 1 - h_float, mode="full")
                max_idx = np.argmax(corr)
                s = max_idx - (len(h_float) - 1)  # Index into full convolution array
                if s < 0: s = -s  # clean this up for negative shifts

                grad = np.zeros_like(h_float)
                for j in range(len(h_float)):
                    j_shift = (j + s) % len(h_float)
                    if j_shift >= 0:
                        grad[j] += 1 - h_float[j_shift]
                    if j >= s:
                        grad[j] += -h_float[j - s]

                # Apply gradient step
                h_new = h_float - lr * grad
                h_new = np.clip(h_new, 0, 1)
                
                # Projection to simplex (sum constraint)
                h_new = h_new / (np.sum(h_new) * dx)  # Ensure integral is 1
                trial_C = evaluate_c5(h_new)
                if trial_C < C:
                    h_float = h_new
                    if trial_C < best_C5:
                        best_C5 = trial_C
                        best_h = h_float.copy()
                        best_n_points = n
                        
        except:
            pass
    
    # Final iterative refinement on the best found
    if (time.time() - start_time) < budget_s * 0.95:
        try:
            n_points = best_n_points
            dx = 2.0 / n_points
            h_current = best_h.copy()
            max_iter = 5000
            lr = 1e-3
            for iter_num in range(max_iter):
                if (time.time() - start_time) > budget_s * 0.99:
                    break
                C_prev = evaluate_c5(h_current)
                
                # Gradient of the max correlation
                corr = correlate(h_current, 1 - h_current, mode="full")
                max_idx = np.argmax(corr)
                s = max_idx - (len(h_current) - 1)
                if s < 0: s = -s
                
                grad = np.zeros_like(h_current)
                for i in range(len(h_current)):
                    i_shift = (i + s) % len(h_current)
                    if i_shift >= 0:
                        grad[i] += 1 - h_current[i_shift]
                    if i >= s:
                        grad[i] += -h_current[i - s]

                # Apply gradient step
                h_new = h_current - lr * grad
                h_new = np.clip(h_new, 0, 1)

                # Project to simplex
                h_new = h_new / (np.sum(h_new) * dx)  # ensure sum of h_i*dx = 1
                C_new = evaluate_c5(h_new)
                if C_new < C_prev:
                    h_current = h_new
                    if C_new < best_C5:
                        best_C5 = C_new
                        best_h = h_current.copy()
        except:
            pass

    return (best_h.tolist(), best_C5, best_n_points)
