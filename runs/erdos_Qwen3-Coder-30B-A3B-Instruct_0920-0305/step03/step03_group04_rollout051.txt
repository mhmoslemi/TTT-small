import numpy as np
from scipy.optimize import minimize
import time

# Constants
n_points = 50
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def smooth_max(values, alpha=20.0):
    """Soft max via log-sum-exp to approximate max."""
    max_val = np.max(values)
    # subtract max for numerical stability
    exps = np.exp(alpha * (values - max_val))
    return max_val + np.log(np.sum(exps)) / alpha

def smooth_objective(h_values):
    """Smooth approximation to the objective function."""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Use smooth max approximation
    return smooth_max(correlation, alpha=20.0) * dx

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def project_to_constraint(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    # Ensure bounds [0,1]
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600):
    """Main optimization routine."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction or uniform distribution
    if 'initial_h_values' in globals():

    else:
        # Initialize with random values and project to constraint
        h_current = np.random.rand(n_points)
        h_current = project_to_constraint(h_current, n_points / 2)
    
    # Initialize best solution
    best_h = h_current.copy()
    best_c5 = evaluate_h(best_h)
    
    # Main restart loop
    restarts = 0
    max_restarts = 50
    max_iter_per_restart = 100
    
    while time.time() - start_time < budget_s * 0.95 and restarts < max_restarts:
        # Random initial perturbation
        h_test = h_current.copy()
        # Add small Gaussian noise
        noise = np.random.normal(0, 0.02, n_points)
        h_test += noise
        # Project to constraints
        h_test = project_to_constraint(h_test, n_points / 2)
        
        # Local optimization using SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            result = minimize(smooth_objective, h_test, method='SLSQP',
                            bounds=bounds, constraints=constraints,
                            options={'maxiter': max_iter_per_restart, 'ftol': 1e-12, 'gtol': 1e-10})
            if result.success:
                h_refined = project_to_constraint(result.x, n_points / 2)
                c5_refined = evaluate_h(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
                    if best_c5 < target_bound:
                        break
        except:
            pass
        
        h_current = best_h.copy()
        restarts += 1
        
        # Occasionally run another optimization with different starting point
        if restarts % 5 == 0 and time.time() - start_time < budget_s * 0.9:
            # Start with fresh random values
            h_fresh = np.random.rand(n_points)
            h_fresh = project_to_constraint(h_fresh, n_points / 2)
            try:
                result = minimize(smooth_objective, h_fresh, method='SLSQP',
                                bounds=bounds, constraints=constraints,
                                options={'maxiter': max_iter_per_restart, 'ftol': 1e-12, 'gtol': 1e-10})
                if result.success:
                    h_refined = project_to_constraint(result.x, n_points / 2)
                    c5_refined = evaluate_h(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined.copy()
                        best_c5 = c5_refined
                        if best_c5 < target_bound:

            except:
                pass
    
    # Final refinement with exact objective
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    try:

    except:
