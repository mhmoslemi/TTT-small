import numpy as np
from scipy.optimize import minimize
import time

# Constants
n_points = 50  # Even number for binary case
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def smooth_max(values, alpha=20.0):
    """Smooth approximation of max using log-sum-exp."""
    max_val = np.max(values)
    # subtract to avoid overflow
    exps = np.exp(alpha * (values - max_val))
    return max_val + np.log(np.sum(exps)) / alpha

def smooth_objective(h_values):
    """Smoothed objective for optimization."""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    return smooth_max(correlation)

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def run(seed=42, budget_s=600):
    """Main optimization routine."""
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = np.inf
    
    # Multiple restarts with different initializations
    n_restarts = 20
    max_iter = 200
    
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Initialize h with random values and normalize to sum to 1
        h_init = np.random.uniform(0, 1, n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h_init = np.clip(h_init, 0, 1)
        
        # SLSQP optimization with smooth objective
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            result = minimize(smooth_objective, h_init, method='SLSQP', 
                            bounds=bounds, constraints=constraints,
                            options={'maxiter': max_iter, 'ftol': 1e-10, 'gtol': 1e-8})
            if result.success:
                h_opt = result.x
                # Evaluate exact C5
                c5_opt = evaluate_h(h_opt)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
        except:
            pass
            
        # Also try binary hill climbing for some restarts
        if restart % 5 == 0 and best_c5 > target_bound:
            # Attempt binary hill climbing
            h_binary = np.zeros(n_points)
            ones_indices = np.random.choice(n_points, size=n_points//2, replace=False)
            h_binary[ones_indices] = 1.0
            
            # Simple local search on binary pattern
            best_binary = h_binary.copy()
            best_c5_binary = evaluate_h(best_binary)
            
            for iter_binary in range(200):
                if time.time() - start_time > budget_s * 0.95:
                    break
                    
                # Random swap
                ones = np.where(best_binary == 1)[0]
                zeros = np.where(best_binary == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    i_one = np.random.choice(ones)
                    i_zero = np.random.choice(zeros)
                    # Try swap
                    test_binary = best_binary.copy()
                    test_binary[i_one] = 0
                    test_binary[i_zero] = 1
                    c5_test = evaluate_h(test_binary)
                    if c5_test < best_c5_binary:
                        best_binary = test_binary.copy()
                        best_c5_binary = c5_test
                        if best_c5_binary < target_bound:
                            break
            
            if best_c5_binary < best_c5:
                best_h = best_binary
                best_c5 = best_c5_binary
    
    # Final refinement
    if best_h is not None:
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            result = minimize(evaluate_h, best_h, method='SLSQP', 
                            bounds=bounds, constraints=constraints,
                            options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-10})
            if result.success:
                best_h = result.x
        except:
            pass
            
        # Final evaluation
        final_c5 = evaluate_h(best_h)
        return list(best_h), final_c5, n_points

        h_fallback = np.clip(h_fallback, 0, 1)

        constraints = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            result = minimize(evaluate_h, h_fallback, method='SLSQP', 
                            bounds=bounds, constraints=constraints,
                            options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-10})
            if result.success:
                h_fallback = result.x
        except:
            pass
            
        final_c5 = evaluate_h(h_fallback)
        return list(h_fallback), final_c5, n_points
