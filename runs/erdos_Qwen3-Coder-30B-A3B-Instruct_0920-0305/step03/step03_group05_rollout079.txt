import numpy as np
import cvxpy as cp
import time

def solve_milp_binary(n, time_limit=60):
    """Solve the MILP formulation of the Erdos overlap problem for even n."""
    dx = 2.0 / n
    # Variables: h_i binary, t continuous
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()

    # Constraints
    constraints = [cp.sum(h) == n/2, t >= 0]

    # Add variables y_{i,d} for each shift d, i valid
    # For shift d (from 0 to n-1)
    for d in range(n):
        # valid indices i for shift d (i+d < n)
        y = cp.Variable(n - d, boolean=True)
        # y_{i,d} := h_i * h_{i+d}, for i = 0,...,n-1-d
        # linearize: y <= h_i, y <= h_{i+d}, y >= h_i + h_{i+d} - 1
        for i in range(n - d):
            constraints.append(y[i] <= h[i])
            constraints.append(y[i] <= h[i + d])
            constraints.append(y[i] >= h[i] + h[i + d] - 1)

        # Q_d = sum(y_{i,d}) = sum(h_i * h_{i+d}) for all valid i
        # Add constraint Q_d >= t to enforce
        Q_d = cp.sum(y)
        constraints.append(Q_d >= t)
    
    # Objective
    objective = cp.Maximize(t)

    # Solve
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CBC, 
               verbose=False,
               maxTime=time_limit)

    if prob.status in [cp.OPTIMAL, cp.UNBOUNDED]:
        h_sol = h.value
        t_opt = t.value
        C5_bound = 1 - (2/n) * t_opt
        return h_sol, C5_bound
    else:
        # Return best known (incumbent) solution if available
        if h.value is not None and t.value is not None:
            h_sol = h.value
            t_opt = t.value
            C5_bound = 1 - (2/n) * t_opt
            return h_sol, C5_bound
        else:
            return None, np.inf

def hill_climb_swaps(h_init, max_swaps=200000, time_limit=300):
    """Apply hill-climbing swaps to improve a binary pattern."""
    n = len(h_init)
    dx = 2.0 / n
    ones = np.flatnonzero(h_init)
    zeros = np.flatnonzero(1 - h_init)

    h_current = h_init.copy()
    Q = np.correlate(h_current, h_current, mode='full')
    min_Q = np.min(Q)
    C5_current = 1 - dx * min_Q

    start = time.time()
    swap_count = 0

    while swap_count < max_swaps and (time.time() - start) < time_limit:
        # Pick a random '1' and '0'
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # Simulate swap
        h_new = h_current.copy()
        h_new[i] = 0
        h_new[j] = 1

        # Recompute overlaps
        Q_new = np.correlate(h_new, h_new, mode='full')
        min_Q_new = np.min(Q_new)
        C5_new = 1 - dx * min_Q_new

        if C5_new < C5_current:
            h_current = h_new
            min_Q = min_Q_new
            C5_current = C5_new
            ones = np.flatnonzero(h_current)
            zeros = np.flatnonzero(1 - h_current)

        swap_count += 1

    return h_current, C5_current

def continuous_refine(h_binary, n_orig, time_limit=60):
    """Refine binary solution to continuous using projected gradient descent."""
    n_fine = min(96, 2*n_orig)
    dx = 2.0 / n_fine
    h_fine = np.interp(np.linspace(0, 2, n_fine),
                       np.linspace(0, 2, n_orig),
                       h_binary)
    # Project to sum = n_fine/2
    h_fine = h_fine * (n_fine / 2) / np.sum(h_fine)

    # Compute the C5 to begin with
    dx = 2.0 / n_fine
    Q = np.correlate(h_fine, h_fine, mode='full')
    min_Q = np.min(Q)
    C5_current = 1 - dx * min_Q

    # Simple projected gradient step: slightly change h_fine
    # But since it's a very small improvement step, we'll do it for few steps
    start = time.time()
    steps = 0
    max_steps = 10000
    while steps < max_steps and (time.time() - start) < time_limit:
        # Gradient is negative to minimize functions like (1-h) in functions
        # However, for a quick local convex-approximation, it's simplest
        # Perform a clipped update using small steps
        steps += 1
        # Slight random perturbation, keep in [0,1] and project to sum
        p = np.random.normal(0, 0.001, n_fine)
        h_new = h_fine + p
        h_new = np.clip(h_new, 0, 1)
        # Re-project to get correct sum
        h_new = h_new * (n_fine / 2) / np.sum(h_new)

        Q_new = np.correlate(h_new, h_new, mode='full')
        min_Q_new = np.min(Q_new)
        C5_new = 1 - dx * min_Q_new

        if C5_new < C5_current:
            h_fine = h_new
            C5_current = C5_new

    return h_fine, C5_current

# Main entry point
def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None

    # Try increasing even n sizes
    candidate_ns = [48, 56, 64, 72]
    for n in candidate_ns:
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Solve MILP with time limit as a fraction of remaining time
        time_left = budget_s - (time.time() - start_time)
        T = min(120, time_left * 0.6)
        if T < 5: break

        h_sol, c5_bound = solve_milp_binary(n, time_limit=T)
        if h_sol is not None and c5_bound < best_c5:
            best_h = h_sol
            best_c5 = c5_bound
            best_n = n

        # Optionally run local swaps if not time limited
        if time.time() - start_time < 0.9 * budget_s and best_h is not None:
            try:
                h_improved, c5_improved = hill_climb_swaps(
                    best_h,
                    max_swaps=200000,
                    time_limit=budget_s - (time.time() - start_time) - 5
                )
                if c5_improved < best_c5:
                    best_h = h_improved
                    best_c5 = c5_improved
                    best_n = n
            except:
                pass

    # Optional final continuous refinement if any solution found
    if best_h is not None and time.time() - start_time < budget_s - 2:
        try:
            h_final, c5_final = continuous_refine(best_h, best_n,
                                                  time_limit=budget_s - (time.time() - start_time) - 2)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
        except:
            pass

    # Return final result
    return list(best_h.astype(float)), best_c5, best_n
