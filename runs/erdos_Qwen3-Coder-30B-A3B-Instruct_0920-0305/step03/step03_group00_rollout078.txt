import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Minimize, Problem, PSD
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def greedy_swap_improve(h, max_swaps=200):
    """Improve h via greedy swaps to reduce max overlap."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    max_iter = max_swaps
    for _ in range(max_iter):
        corr = correlate(h, 1 - h, mode="full")
        worst_s = np.argmax(corr)
        if worst_s <= n-1:
            # corr includes all shifts, index 0 is shift 0
            # full correlation has length 2*n - 1, shift s is at index s + (n-1)
            shift_idx = worst_s + (n - 1)
            if shift_idx >= 0 and shift_idx < len(corr):
                # Evaluate by swapping two entries where possible
                # Find positions where swapping helps
                best_diff = 0
                best_i, best_j = -1, -1
                for i in range(n):
                    j = i + worst_s
                    if j >= n: continue
                    # Try swapping 0->1 and 1->0
                    if h[i] == 0 and h[j] == 1:
                        old_diff = h[i] * (1 - h[j])
                        h[i], h[j] = 1, 0
                        # Compute new overlap at shift worst_s
                        # But only adjust a small part of the correlation
                        # we focus on efficient search
                        old_corr = corr[shift_idx]  
                        new_corr = 0
                        c = 0
                        for k in range(max(0, i - worst_s), min(n, i + 1)):
                            if k + worst_s < n:
                                new_corr += h[k] * (1 - h[k + worst_s])
                                c += 1
                        if c > 0:
                            new_corr = new_corr / c
                        diff = old_corr - new_corr
                        if diff > best_diff:
                            best_diff = diff
                            best_i, best_j = i, j
                        h[i], h[j] = 0, 1
                if best_i >= 0:
                    h[best_i], h[best_j] = h[best_j], h[best_i]
                else:
                    break
            else:
                break
        else:
            break
    return h

def trust_constr_optimize(h0, alpha_schedule=[5, 15, 40], bounds=None, eq_sum=None):
    """Optimize h using trust-constr with log-sum-exp approx of the max."""
    if bounds is None:
        bounds = [(0, 1) for _ in range(len(h0))]

    def soft_max_obj(h, alpha):
        # compute values for each shift
        n = len(h)
        dx = 2.0 / n
        # Compute also overlap per shift
        corr_vals = []
        for s in range(1, n):
            val = 0
            for i in range(n - s):
                val += h[i] * (1 - h[i + s])
            corr_vals.append(val * dx)
        # Softmax
        if alpha == 0:
            return np.max(corr_vals)
        max_val = np.max(corr_vals)
        expvals = np.exp(alpha * (np.array(corr_vals) - max_val))
        return max_val + np.log(np.sum(expvals)) / alpha

    def gradient_soft(h, alpha):
        n = len(h)
        dx = 2.0 / n
        # Compute unshifted corr_vals
        corr_vals = []
        for s in range(1, n):
            val = 0
            for i in range(n - s):
                val += h[i] * (1 - h[i + s])
            corr_vals.append(val * dx)
        if alpha == 0:
            # Return gradient of max
            mask = np.array(corr_vals) == np.max(corr_vals)
            idx = np.argmax(mask)
            g = np.zeros_like(h)
            # For s = idx+1, look at gradient of overlap in those positions
            s = idx + 1
            for i in range(n - s):
                # if s > 0: grad at i (i is one term in overlap) 
                # but for each i, influence comes from h[i] term and h[i+s] term
                if i < len(g):
                    g[i] += 1 - h[i + s]
                if i + s < len(g):
                    g[i + s] -= h[i]
            return g * dx
        # Standard gradients from logsumexp
        max_val = np.max(corr_vals)
        expvals = np.exp(alpha * (np.array(corr_vals) - max_val))
        wts = expvals / np.sum(expvals)
        g = np.zeros_like(h)
        for s, wt in enumerate(wts):
            s_idx = s + 1  # shift from 1 to n-1
            for i in range(n - s_idx):
                if i < len(g):
                    g[i] += wt * (1 - h[i + s_idx])
                if i + s_idx < len(g):
                    g[i + s_idx] -= wt * h[i]
        return g * dx

    h_opt = h0.copy()
    for alpha in alpha_schedule:
        def obj_func(h):
            return soft_max_obj(h, alpha)
        def jac(h):
            return gradient_soft(h, alpha)

        # Constraints 
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - eq_sum}

        min_res = minimize(obj_func,
                           h_opt,
                           method='trust-constr',
                           bounds=bounds,
                           constraints=cons,
                           jac=jac,
                           options={'maxiter': 200,
                                    'gtol': 1e-7,
                                    'xtol': 1e-8,
                                    'ftol': 1e-8})

        if min_res.success:
            h_opt = min_res.x
        else:
            # print("Optimization did not converge with alpha:", alpha)
            pass
    return h_opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 100  # Majority of time should be used on SDP or break early  
    # Try to keep element size manageable (X is n×n = ~10k entries), so want n ≤120
    if n > 120:
        n = 120

    best_h = None
    best_c5 = float('inf')
    # Phase A: SDP
    try:
        # Build the SDP model
        h_opt = cp.Variable(n)
        X = cp.Variable((n, n), symmetric=True)

        # Constraints
        constraints = [
            X >> 0,  # PSD
            cp.diag(X) == h_opt,  # diag(X) = h
            cp.sum(h_opt) == n / 2,  # sum(h) = n/2
            h_opt >= 0,    # h_i >= 0
            h_opt <= 1     # h_i <= 1
        ]

        # Add McCormick envelopes for products
        # For simplicity, only add a few of the product constraints if needed. 
        for i in range(n):
            for j in range(i+1, n):
                # iff h_i h_j are both between 0 to 1 (which they are, by constraints)
                constraints.append(X[i,j] <= h_opt[i])
                constraints.append(X[i,j] <= h_opt[j])
                constraints.append(X[i,j] >= h_opt[i] + h_opt[j] - 1)

        # Define overlaps per shift 
        overloads = []
        # Go through shift indices 1 to n-1
        max_shift = n - 1
        # t = cp.Variable()  # we minimize -t or maximize t, but since  
        # we can leave it linear and use the standard std distro form
        # Instead, we wait to find out max of vertices

        # Use a cheap way to build this
        # For direct use of the matrix
        t = cp.Variable()
        for s in range(1, max_shift + 1):
            # sum of X[i, i+s] for valid i
            expr = cp.sum([X[i, i+s] for i in range(n-s)])
            expr = expr  # No scaling needed for now
            constraints.append(expr >= t)

        # Objective
        objective = cp.Maximize(t)
        prob = Problem(objective, constraints)
        # try default SCS which is often fatter for large problems
        prob.solve(solver=cp.SCS, max_iters=2000, verbose=False, eps=1e-5)

        if prob.status in ['optimal', 'optimal_inaccurate']:
            h_sd = h_opt.value
            t_val = t.value
            # Rescale h_sd to match unit area
            h_sd = h_sd / np.sum(h_sd) * (n / 2)
            # Update best if not exist or is better 
            c5_base = 1 - 2 * t_val / n
            if c5_base < best_c5:
                best_c5 = c5_base
                best_h = h_sd.copy()

        # Try relaxation again with a smaller n if not successful
        if best_h is None and n > 80:
            n = 80
            h_opt = cp.Variable(n)
            X = cp.Variable((n, n), symmetric=True)
            constraints = [
                X >> 0,
                cp.diag(X) == h_opt,
                cp.sum(h_opt) == n / 2,
                h_opt >= 0,
                h_opt <= 1
            ]

            # McCormick limits
            for i in range(n):
                for j in range(i+1, n):
                    constraints.append(X[i,j] <= h_opt[i])
                    constraints.append(X[i,j] <= h_opt[j])
                    constraints.append(X[i,j] >= h_opt[i] + h_opt[j] - 1)

            # Overlaps
            t = cp.Variable()
            for s in range(1, max_shift + 1):
                expr = cp.sum([X[i, i+s] for i in range(n-s)])
                constraints.append(expr >= t)

            objective = cp.Maximize(t)
            prob = Problem(objective, constraints)
            prob.solve(solver=cp.SCS, max_iters=500, verbose=False, eps=1e-5)

            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_sd = h_opt.value
                t_val = t.value
                h_sd = h_sd / np.sum(h_sd) * (n / 2)
                c5_base = 1 - 2 * t_val / n
                if c5_base < best_c5:
                    best_c5 = c5_base
                    best_h = h_sd.copy()

    except Exception as e:
        # fallback: small heuristic or go to other method
        pass

    # If no SDP succeeded, fallback to basic trust-constr optimization
    if best_h is None:
        print("Falling back to random starts + smooth max optimization.")
        n = 80
        max_restarts = 30
        fallback_c5s = []
        for i in range(max_restarts):
            if time.time() - start_time > budget_s - 5:
                break
            # Random initialization with area constraint
            h = np.random.rand(n)
            h = h / np.sum(h) * (n / 2)
            h = np.clip(h, 0, 1)  # box constraints
                
            # Use same trust-constr on small random restarts
            h_opt = trust_constr_optimize(h, bounds=[(0,1) for _ in range(n)], eq_sum=n/2)
            if len(h_opt) < n:
                h_opt = np.pad(h_opt, (0, n - len(h_opt)), 'constant', constant_values=0)
            elif len(h_opt) > n:
                h_opt = h_opt[:n]

            c5 = evaluate_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    # Phase B: Rounding strategy
    if not best_h is None:
        h_final = best_h
        dx = 2.0 / len(h_final)
        # Fast pre-eval:
        c5_pre = evaluate_c5(h_final)
        if c5_pre < best_c5:
            best_c5 = c5_pre
            # Do the quality check and refine stage
        # Attempt refinement using randomized rounding
        # Use eigen decomposition of the relaxed solution
        # Create pseudo matrix which we split
        X_rect = np.outer(h_final, h_final)
        # Add symmetry terms if needed or stick to current true mode
        try:
            U, s, Vt = np.linalg.svd(X_rect)
            rank = np.sum(s > 1e-8)
            Y = U[:, :rank] @ np.diag(np.sqrt(s[:rank]))
            num_rounds = min(20, 30)  # At most 30 or when time runs out
            round_start = time.time()
            for i in range(num_rounds):
                if time.time() - start_time > budget_s - 5:
                    break
                g = np.random.randn(rank)
                y = Y @ g
                h_bin = np.zeros_like(h_final)
                perm = np.argsort(-y)
                h_bin[perm[:int(len(h_final)/2)]] = 1
                # Run correction via greedy swaps
                h_corr = greedy_swap_improve(h_bin, max_swaps=200)
                c5_rounded = evaluate_c5(h_corr)
                if c5_rounded < best_c5:
                    best_c5 = c5_rounded
                    best_h = h_corr.copy()
        except Exception:
            pass

    # Final refinement via trust-constr
    if best_h is not None:
        # double-check best_h format and ensure it is valid
        h_refined = trust_constr_optimize(best_h.copy(),
                                          bounds=[(0,1) for _ in range(len(best_h))],
                                          eq_sum=n / 2)
        if len(h_refined) < n:
            h_refined = np.pad(h_refined, (0, n - len(h_refined)), 'constant', constant_values=0)
        elif len(h_refined) > n:
            h_refined = h_refined[:n]

        c5_final = evaluate_c5(h_refined)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_refined

    return (best_h.tolist(), best_c5, n)
