```python
import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time
import math

# Constants and global helpers

def dx(n):
    return 2.0 / n

def evaluate(h):
    """Compute the exact C5 bound."""
    h = np.array(h)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    return max_correlation * dx(len(h))

def correlation_vec(h):
    """Return the full correlation vector."""
    return np.correlate(h, 1 - h, mode="full")

def smooth_max(c, alpha=30):
    """Soft maximum approximation."""
    m = np.max(c)
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

def penalty_sum(h, lam=10000):
    """Quadratic penalty on sum deviation from n/2."""
    return lam * (np.sum(h) - len(h)/2)**2

def proj_to_simplex(v, S=None):
    """Project vector v onto the simplex with sum = S."""
    if S is None:
        S = len(v) / 2
    v = np.clip(v, 0, 1)
    s = np.sum(v)
    if s <= S:
        return v * (S / s) if s > 0 else np.full_like(v, S / len(v))
    else:
        # Sort v in descending order and apply projection
        sorted_v = np.sort(v)[::-1]
        cumsum = np.cumsum(sorted_v)
        idx = np.argmax(sorted_v * np.arange(1, len(v)+1) > (S - cumsum + sorted_v))
        theta = (cumsum[idx] - S) / (idx + 1)
        return np.clip(v - theta, 0, 1)

def logit(p):
    return np.log(p / (1 - p))

def sigmoid(y):
    return 1 / (1 + np.exp(-y))

def binary_initial(n):
    """Generate random binary vector with exactly n/2 ones."""
    v = np.zeros(n)
    ones = np.random.permutation(n)[:n//2]
    v[ones] = 1
    return v

def swap_and_evaluate(h, i_one, i_zero):
    """Swap positions i_one (1) and i_zero (0) in h, and return new C5."""
    h_new = h.copy()
    h_new[i_one] = 0
    h_new[i_zero] = 1
    return evaluate(h_new)

# Start main execution

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None

    target_bound = 0.3807

    # Stage 1: Interval family search
    n_vals = [50, 60, 72, 84]
    intervals = [(2, 0.5), (2, 0.75), (3, 0.5), (4, 1.0)]
    for n in n_vals:
        for m, g_scale in intervals:
            if m > n:
                continue
            # Build interval pattern
            ell = 1.0 / m
            g = g_scale * (2 - m * ell) / (m - 1)
            if 2 * g + m * ell > 2:
                continue
            h = np.zeros(n)
            dx_val = dx(n)
            for i in range(m):
                start = i * (ell + g) + 0.25 * g
                end = start + ell
                start_idx = int(np.floor(start / dx_val))
                end_idx = int(np.ceil(end / dx_val))
                h[max(0, start_idx):min(n, end_idx)] = 1
            # Normalize sum to n/2
            h = h * (n / 2) / np.sum(h)
            c5 = evaluate(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n

    # Stage 2: Binary hill-climbing across several n
    # Find time reservations
    t_remaining = budget_s - (time.time() - start_time)
    if t_remaining < 10:
        return list(best_h), best_c5, best_n
    time_per_n = min(40, t_remaining * 0.5 / len(n_vals))

    for n in n_vals:
        t_left = budget_s - (time.time() - start_time)
        if t_left < 10:
            break
        t_stop = time.time() + min(time_per_n, t_left)
        h_best_bin = None
        c5_best_bin = float('inf')

        for run in range(5):  # Multiple restarts
            # Generate binary initial
            h = binary_initial(n)
            c5 = evaluate(h)
            if c5 < c5_best_bin:
                h_best_bin = h.copy()
                c5_best_bin = c5

            # For improvement attempts, do more
            count = 0
            while time.time() < t_stop and count < 10000:
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i_one = np.random.choice(ones)
                i_zero = np.random.choice(zeros)
                h_new = h.copy()
                h_new[i_one] = 0
                h_new[i_zero] = 1
                c5_new = evaluate(h_new)
                if c5_new < c5:
                    h = h_new
                    c5 = c5_new
                    if c5 < c5_best_bin:
                        h_best_bin = h.copy()
                        c5_best_bin = c5
                count += 1
            if c5_best_bin < target_bound:
                break

        if h_best_bin is not None:
            c5_eval = evaluate(h_best_bin)
            if c5_eval < best_c5:
                best_c5 = c5_eval
                best_h = h_best_bin.copy()
                best_n = n

    # Stage 3: Smooth max refinement with restarts
    if best_h is not None:
        h_best = best_h.copy()
        n = best_n
        c5_best = best_c5
        t_remaining = budget_s - (time.time() - start_time)
        if t_remaining > 10:
            h_r = h_best.copy()
            for restart in range(8):
                y0 = logit(h_r) + np.random.normal(0, 0.2, n)
                def obj(y):
                    h = sigmoid(y)
                    corr = correlation_vec(h)
                    return smooth_max(corr, 30) + penalty_sum(h, 10000)
                res = minimize(obj, y0, method='L-BFGS-B', options={'maxiter': 300})
                h_refined = sigmoid(res.x)
                h_refined = proj_to_simplex(h_refined, S=n/2)
                new_c5 = evaluate(h_refined)
                if new_c5 < c5_best:
                    h_best = h_refined
                    c5_best = new_c5
                if c5_best < target_bound:
                    break
            best_c5 = c5_best
            best_h = h_best

    # Stage 4: Final polishing
    t_remaining = budget_s - (time.time() - start_time)
    if t_remaining > 5 and best_h is not None:
        bounds = [(0., 1.)] * len(best_h)
        def obj(h):
            return evaluate(h)

        # Final SLSQP refinement
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, 
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x)*dx(len(x)) - 1.0},
                       options={'ftol': 1e-12, 'maxiter': 500})
        final_c5 = evaluate(res.x)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = res.x

    return list(best_h), best_c5, best_n
```
