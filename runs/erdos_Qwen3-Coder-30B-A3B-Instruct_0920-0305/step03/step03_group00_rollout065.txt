import numpy as np

from scipy.signal import correlate
import time
import cvxpy as cp

def weight_normalise(x, n):
    """Normalise the vector to sum to n/2"""
    dx = 2.0 / n
    scale = (n / 2) / np.sum(x)
    return x * scale

def max_defect(h):
    """Compute exact max defect C5 and the corresponding shift"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    best_shift = np.argmax(corr)
    return max_corr, best_shift

def softmax_objective(h, alpha, n):
    """Return Φ_α(h) and its gradient"""
    dx = 2.0 / n
    # Compute correlation in full mode
    corr = correlate(h, 1 - h, mode="full")
    # Compute soft max part
    exp_corr = np.exp(alpha * corr)
    Z = np.sum(exp_corr)
    # Phi_alpha = (1/alpha) * log(Z)
    phi = np.log(Z) / alpha
    
    # Compute gradient
    # grad_i = sum_s (w_s * (1 - h[i+s] * 1_{i+s<n} - h[i-s] * 1_{i-s>=0}))
    # where w_s = exp(alpha * corr_s) / Z
    
    # Instead of computing explicitly, we do it incrementally
    # Since h is already normalized (so sum h = n/2) and corr is already computed
    
    # Create full correlation vector for all shifts
    corr_full = corr  # This is already full
    
    # Compute gradients per index
    grad = np.zeros_like(h)
    for i in range(n):
        d = 0
        for s in range(1, n):
            if i + s < n:
                w_s = exp_corr[s] / Z
                d += w_s * (1 - h[i+s])
            if i - s >= 0:
                w_s = exp_corr[n-1+s] / Z  # mirrored shift
                d += w_s * (1 - h[i-s])
        grad[i] = d
    
    return phi, grad

def build_milp(n):
    """Build MILP to maximize the minimum overlap"""
    # Binary variables for h[i]
    h = Variable(n, boolean=True)
    # Continuous aux variables w[i,s] = h[i] * h[i+s] 
    w = Variable((n, n-1))  # w[i,s] is defined for s = 0..n-2, but valid only for i+s < n
    # Min overlap t
    t = Variable()
    
    # Weight constraint: sum h = n/2
    constraints = [cp.sum(h) == n/2]
    
    # Linearize the AND constraint w[i,s] = h[i] * h[i+s] 
    for s in range(n-1):
        for i in range(n):
            if i + s < n:
                # w[i,s] <= h[i]  
                constraints.append(w[i,s] <= h[i])
                # w[i,s] <= h[i+s]
                constraints.append(w[i,s] <= h[i+s])
                # w[i,s] >= h[i] + h[i+s] - 1
                constraints.append(w[i,s] >= h[i] + h[i+s] - 1)
    
    # min_overlap <= sum of w[i,s] for all valid i one shift
    for s in range(n-1):
        # Only append if it's a valid sum
        valid_indices = [i for i in range(n) if i + s < n]
        if valid_indices:
            constraints.append(cp.sum(w[valid_indices, s]) >= t)
    
    # Maximize min overlap t
    problem = Problem(Maximize(t), constraints)
    return problem, h, w, t

def refine_continuous(h0, n, maxiter=200):
    """Refine the continuous solution using smooth max approximation"""
    # Try to improve the binary solution by continuous relaxation
    
    # Keep the h0 as starting point
    h_prev = h0.copy()
    dx = 2.0 / n
    # Normalize it 
    h_prev = weight_normalise(h_prev, n)
    
    # For smooth max optimization, method='trust-constr'
    # Use alpha = 5 then 15, then 40
    alphas = [5, 15, 40]
    
    # Ensure it sums to n/2
    h_prev = weight_normalise(h_prev, n)
    
    for alpha in alphas:
        def obj(h):
            val, grad = softmax_objective(h, alpha, n)
            return val
            
        def jac(h):
            _, grad = softmax_objective(h, alpha, n)
            return grad
            
        # Build simple bounds        
        bounds = [(0, 1) for _ in range(n)]
        
        # Optimize with bounds as constraint
        res = minimize(obj, h_prev, method='trust-constr', jac=jac, 
                       bounds=bounds, constraints=({'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}),
                       options={'maxiter': maxiter, 'gtol': 1e-5, 'ftol': 1e-8})
        
        if not res.success:
            print(f"Warning: trust-constr failed for alpha={alpha}")
        
        h_prev = res.x
        h_prev = weight_normalise(h_prev, n)
        
    return h_prev

def anneal(h, n, max_iters=5000, cooling_rate=0.99, initial_gamma=0.2):
    """Simulated-annealing style swap improvements"""
    dx = 2.0 / n
    c = correlate(h, 1-h, mode='full')
    max_corr = np.max(c)
    shift_max = np.argmax(c)
    
    h = h.copy()
    h_prev = h.copy()
    gamma = initial_gamma
    
    for it in range(max_iters):
        if it % 100 == 0:
            c = correlate(h, 1-h, mode='full')
            max_corr = np.max(c)
            shift_max = np.argmax(c)
        
        # Find indices contributing to max shift
        # h[i] * (1 - h[i+shift_max]) is high
        contrib = np.where((h > 0) & (np.roll(h, -shift_max) < 1.0))[0]
        if len(contrib) == 0:
            continue
        
        # Find indices where swapping makes sense
        # Can move mass from contrib to low regions
        not_contrib = np.where((h < 1.0) & (np.roll(h, -shift_max) > 0.1))[0]
        if len(not_contrib) == 0:
            continue
            
        # Select random pair
        i = np.random.choice(contrib)
        j = np.random.choice(not_contrib)
        
        # Calculate potential swap
        if i == j:
            continue
        delta_i = np.roll(h, -shift_max)[i] 
        delta_j = np.roll(h, shift_max)[j] 
        
        if delta_i < 0.01 or delta_j < 0.01:
            continue
            
        # small amount of mass to move
        delta = min(h[i], 1-h[j], gamma * 0.5)
        if delta < 1e-6:
            continue
            
        # Perform move
        h[i] -= delta
        h[j] += delta
        h[i+shift_max] += delta
        h[j-shift_max] -= delta
        
        # Reinforce normalization to ensure sum = n/2 
        sum_diff = np.sum(h) - n/2
        h[0] += sum_diff
        
        # Compute new features
        c_new = correlate(h, 1-h, mode='full')
        max_corr_new = np.max(c_new)
        
        # Accept or reject
        if max_corr_new > max_corr:
            h = h_prev.copy()
            gamma *= cooling_rate  
            continue
            
        # Accept - update best
        h_prev = h.copy()
        max_corr = max_corr_new
        shift_max = np.argmax(c_new)
        
        gamma *= cooling_rate
        
    return h
