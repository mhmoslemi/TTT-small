import numpy as np
from scipy.optimize import differential_evolution

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Setup
    n_points = 200  # 200 samples (n=200 leads to dx=0.01)
    dx = 2.0 / n_points

    # Stage A: Low-dimensional global search with block parametrization
    print("Starting Stage A (Low-dim global search)...")
    m = min(10, n_points // 2)  # Number of blocks
    bounds_block = [(0, 1) for _ in range(m-1)]  # First (m-1) free
    
    def to_full_block(u_vals):  # u = [u1, ..., um-1], u_m = (m/2 - sum(u))
        u = np.array(u_vals)
        u_m = m / 2.0 - np.sum(u)
        u_m = np.clip(u_m, 0, 1)
        return np.concatenate([u, [u_m]])
    
    def block_to_h(u_vals):
        u = to_full_block(u_vals)
        h = np.repeat(u, n_points // m)
        # Handle remainder cases
        extra = n_points % m
        if extra > 0:
            h = np.pad(h, (0, extra), constant_values=u[-1])
        return h[:n_points]
    
    def evaluate_block_obj(u_vals):
        h = block_to_h(u_vals)
        return evaluate_c5(h)

    # Use CMA-ES to optimize the block heights, though we'll wrap with our own simple call
    # Goal is to give the best convex-like local structure
    best_c5 = float('inf')
    best_u = None
    best_h = None

    # CMA-ES simulation with random search good enough for now:
    n_evals = 0
    max_evals = 1500
    for _ in range(100 if max_evals > 1000 else 10):  # Simplified: we sample 100 times
        if n_evals >= max_evals:
            break
        # Random sample:
        u = np.random.rand(m - 1) * 1.0  # Uniform in [0,1]
        h = block_to_h(u)
        c5 = evaluate_c5(h)
        n_evals += 1
        if c5 < best_c5:
            best_c5 = c5
            best_u = u.copy()
            best_h = h.copy()
            
    print(f"Stage A best c5: {best_c5}")
    
    # Stage B: Full-resolution projected gradient descent on max overlap
    print("Starting Stage B (Projected gradient descent)...")
    
    def project_to_constraint(h):
        """Project h onto sum(h) = n/2 constraint with box bounds [0,1]"""
        # Use water filling - binary search for lambda
        tol = 1e-12
        y = np.copy(h)
        low, high = -2.0, 3.0
        for k in range(50):
            mid = (low + high) / 2.0
            x = np.clip(y - mid, 0., 1.)
            s = np.sum(x)
            if abs(s - n_points / 2.0) < tol:
                return x  # converged
            elif s > n_points / 2.0:
                low = mid
            else:
                high = mid
        # fallback
        x = np.clip(y - (low + high) / 2., 0.0, 1.0)
        return x
    
    current_h = best_h.copy()
    best_in_stage_b = evaluate_c5(current_h)
    best_h_in_stage_b = current_h.copy()

    current_c5 = evaluate_c5(current_h)
    iter_count = 0
    max_iter = 3000
    alpha_0 = 1e-2
    tolerance = 1e-6
    last_improvements = []
    
    while iter_count < max_iter and (time.time() - start_time) < 0.6 * budget_s:
        # Retrieve max shift index
        corr_full = correlate(current_h, 1 - current_h, mode="full")
        k_max = np.argmax(corr_full)
        k_idx = k_max - (n_points - 1)   # convert to actual shift index
        
        # Compute subgradient
        g = np.zeros(n_points)
        for i in range(n_points):
            if (i + k_idx) < n_points:
                g[i] += -(1 - current_h[i + k_idx])
            if (i - k_idx) >= 0:
                g[i] += -current_h[i - k_idx]
        
        # Gradient step
        alpha = alpha_0 / np.sqrt(iter_count + 1)
        h_new = current_h - alpha * g
        
        # Project
        h_proj = project_to_constraint(h_new)
        
        # Evaluate
        c5_new = evaluate_c5(h_proj)
        iter_count += 1
        
        if c5_new < best_in_stage_b:
            best_in_stage_b = c5_new
            best_h_in_stage_b = h_proj.copy()
            
        # Track improvement
        last_improvements.append(c5_new)
        if len(last_improvements) > 200:
            last_improvements.pop(0)
            avg_improvement = np.mean(np.diff(last_improvements))
            if avg_improvement < tolerance:
                break

    print(f"Stage B best c5: {best_in_stage_b}")
    current_h = best_h_in_stage_b.copy()
    
    # Stage C: Binary rounding + swap local search    
    print("Starting Stage C (Binary rounding + local search)...")

    def binary_round(h, n_to_set):
        """
        Stochastic rounding to binary:
        probs=h; take a random permutation of indices
        walk through permutation, flip ones based on prob until n_to_set are selected.
        """
        n = len(h)
        unif = np.random.rand(n)
        perm = np.argsort(unif)
        ones = set()
        for i in perm:
            if len(ones) < n_to_set:
                if h[i] > unif[i]:
                    ones.add(i)
            if len(ones) >= n_to_set:
                break
        while len(ones) < n_to_set:
            # Extra tie breaking
            for i in range(n):
                if i not in ones:
                    ones.add(i)
                    if len(ones) == n_to_set:
                        break
        b = np.zeros(n, dtype=int)
        for i in ones:
            b[i] = 1
        return b
    
    # Pattern matching and micro-optimization
    best_binary_c5 = float('inf')
    best_binary_h = None

    n_rounds = 30
    # Try several random binary vectors
    for _ in range(n_rounds):
        if (time.time() - start_time) > 0.7 * budget_s:
            break
        b = binary_round(current_h, n_points // 2)
        # Convert back to float
        h_candidate = b.astype(float)
        c5_candidate = evaluate_c5(h_candidate)
        if c5_candidate < best_binary_c5:
            best_binary_c5 = c5_candidate
            best_binary_h = h_candidate

    # Now apply combinatorial refinement via swaps
    # We only process a subset of candidate swaps per iteration
    best_h_binary = best_binary_h.copy()
    while (time.time() - start_time) < 0.9 * budget_s:
        best_delta_c5 = 0.0
        best_i = -1
        best_j = -1

        # Build overlapping contribution tables for efficient swap update
        # Not used in full code for brevity; we rely on recompute where valid
        # We just compute a couple of swaps manually here to demonstrate
        
        for i in range(0, n_points, 10):  # Sample a few pairs
            for j in range(i + 1, min(i + 30, n_points)):
                if best_h_binary[i] == 1 and best_h_binary[j] == 0:
                    # Save copy
                    temp_h = best_h_binary.copy()
                    temp_h[i] = 0
                    temp_h[j] = 1
                    
                    # Compute overlap at shift k=0 to middle k
                    c5_new = evaluate_c5(temp_h)
                    delta = c5_new - best_binary_c5
                    if delta < best_delta_c5:
                        best_delta_c5 = delta
                        best_i = i
                        best_j = j

        if best_delta_c5 < 0:
            # Perform the swap
            best_h_binary[best_i] = 0
            best_h_binary[best_j] = 1
            best_binary_c5 += best_delta_c5
        else:
            break  # No improvement

    final_h = best_h_binary
    final_c5 = evaluate_c5(final_h)
    
    # Use final_h, final_c5, n_points = 200
    return (final_h.tolist(), final_c5, n_points)
