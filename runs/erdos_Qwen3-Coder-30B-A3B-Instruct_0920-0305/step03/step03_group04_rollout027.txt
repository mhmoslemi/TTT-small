import numpy as np

n_values = [36, 42, 48, 54, 60]
dx = 2.0 / 48  # We will target 48 points for our final best solution
target_bound = 0.3807

def evaluate_h(h_values):
    """Computes C5 for a given h vector."""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

def correlation_vec(h_values):
    """Returns the full correlation vector."""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    return correlation

def logsumexp(x, axis=None, keepdims=False):
    """Numerically stable version of log(sum(exp(x)))"""
    x_max = np.max(x, axis=axis, keepdims=True)
    return x_max + np.log(np.sum(np.exp(x - x_max), axis=axis, keepdims=keepdims))

def smooth_max(x, alpha=30.0):
    """Smooth approximation to max using log-sum-exp."""
    return logsumexp(alpha * x) / alpha

def sigmoid(x):
    """Sigmoid activation function."""
    return 1.0 / (1.0 + np.exp(-x))

def logit(x):
    """Inverse of sigmoid."""
    x = np.clip(x, 1e-12, 1 - 1e-12)
    return np.log(x / (1.0 - x))

def proj_to_simplex(v, s=1.0):
    """Project a vector v onto the probability simplex with sum=s."""
    v = np.array(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.where(u > cssv / np.arange(1, len(u)+1))[0]
    if len(ind) == 0:
        theta = 0.0
    else:
        theta = cssv[ind[-1]] / (ind[-1] + 1)
    return np.maximum(v - theta, 0.0)

def solve_sdp(n):
    """Solve SDP relaxation for problem."""
    x = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    constraints = [
        0 <= x,
        x <= 1,
        cp.sum(x) == n / 2,
        cp.diag(X) == x,
    ]
    
    # McCormick envelopes for X[i,j] = x[i]*x[j]
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(X[i, j] >= 0)
            constraints.append(X[i, j] <= x[i])
            constraints.append(X[i, j] <= x[j])
            constraints.append(X[i, j] >= x[i] + x[j] - 1)
    
    # Overlap constraints for all shifts
    for k in range(n):
        if k == 0:
            overlap = cp.sum(x)
        else:
            overlap = cp.sum(x[:n-k]) - cp.sum(cp.diag(X, k))
        constraints.append(overlap <= t)
        
    constraints.append(X >> 0)  # PSD constraint
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        prob.solve(solver=SCS, max_iters=8000, eps=1e-5)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value, X.value
        else:
            return None, None
    except Exception:
        return None, None

def hyperplane_round(X_star, x_star, n, R=30):
    """Sample R times using random hyperplane rounding."""
    best_h = None
    best_c5 = np.inf
    
    # Get a low rank factorization: X = V*V.T
    try:
        U, s, Vt = np.linalg.svd(X_star, full_matrices=False)
        r = min(int(np.sqrt(n)), 20)
        V = U[:, :r] * np.sqrt(s[:r])
        for _ in range(R):
            g = np.random.randn(r)
            s_i = np.sign(V @ g)
            x_i = (s_i + 1) / 2  # map to 0/1
            # Correct for sum
            current_sum = np.sum(x_i)
            diff = current_sum - n/2
            if abs(diff) > 1e-6:
                indices = np.argsort(np.abs(V @ g))[:int(abs(diff))]
                for idx in indices:
                    x_i[idx] = 1 - x_i[idx]
            # Clip to [0,1] and normalize to sum 1
            x_i = np.clip(x_i, 0, 1)
            c5 = evaluate_h(x_i)
            if c5 < best_c5:
                best_c5 = c5
                best_h = x_i.copy()
    except Exception:
        # Fallback to random binary if SVD fails
        for _ in range(R):
            x_i = np.random.rand(n) > 0.5
            x_i = x_i.astype(float)
            current_sum = np.sum(x_i)
            if abs(current_sum - n/2) > 1e-6:
                # Flip some bits to match sum
                n_flips = int(abs(current_sum - n/2))
                flip_indices = np.argsort(np.abs(0.5 - x_i))[:n_flips]
                for idx in flip_indices:
                    x_i[idx] = 1 - x_i[idx]
            # Normalize to sum 1
            x_i = (n/2) * x_i / np.sum(x_i)
            c5 = evaluate_h(x_i)
            if c5 < best_c5:
                best_c5 = c5
                best_h = x_i.copy()
    return best_h, best_c5

def hill_climb_binary(h_start, max_swaps=50000):
    """Perform binary hill climb using swaps."""
    h = np.array(h_start)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    current_c5 = evaluate_h(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    last_improved = 0
    
    for i in range(max_swaps):
        if i - last_improved > 10000:
            # Try systematic search for worst shift
            worst_shift = np.argmax(correlation_vec(h))
            mismatches = []
            for k in range(len(h) - worst_shift):
                if h[k] == 1 and h[k + worst_shift] == 0:
                    mismatches.append(k)
            if mismatches:
                # Try swapping a mismatched 1 with a 0
                try:
                    try_idx = random.choice(mismatches)
                    not_zero_idx = random.choice(zeros)
                    h_copy = h.copy()
                    h_copy[try_idx] = 0
                    h_copy[not_zero_idx] = 1
                    nc5 = evaluate_h(h_copy)
                    if nc5 < current_c5:
                        # Accept swap
                        h = h_copy
                        ones.remove(try_idx)
                        ones.append(not_zero_idx)
                        zeros.remove(not_zero_idx)
                        zeros.append(try_idx)
                        current_c5 = nc5
                        if current_c5 < best_c5:
                            best_c5 = current_c5
                            best_h = h.copy()
                        last_improved = i
                except:
                    pass
    
        # Do regular random swap
        try:
            i_idx = random.choice(ones)
            j_idx = random.choice(zeros)
            h_copy = h.copy()
            h_copy[i_idx] = 0
            h_copy[j_idx] = 1
            nc5 = evaluate_h(h_copy)
            if nc5 < current_c5:
                # Accept
                h = h_copy
                ones.remove(i_idx)
                ones.append(j_idx)
                zeros.remove(j_idx)
                zeros.append(i_idx)
                current_c5 = nc5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                last_improved = i
        except:
            continue
            
        if current_c5 < target_bound:
            break
    
    return best_h, best_c5

def smooth_refinement(h_start):
    """Refine h using smooth surrogate loss."""
    h = np.array(h_start)
    y = logit(h)
    n = len(h)
    
    def obj_func(y_flat):
        h_trial = sigmoid(y_flat)
        c_vec = correlation_vec(h_trial)
        soft = smooth_max(c_vec, alpha=30.0)
        penalty = 10000 * (np.sum(h_trial) - n/2)**2
        return soft + penalty
    
    # Optimize with L-BFGS-B
    result = minimize(obj_func, y, method='L-BFGS-B', bounds=[(-20, 20)]*n, options={'maxiter': 300})
    
    if result.success:
        h_refined = sigmoid(result.x)
        # Project onto simplex
        h_refined = proj_to_simplex(h_refined, n/2)
        return h_refined
    return h_start
