import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import SolverError

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # --- Phase A: SDP Relaxation -------------------------------

    # Choose grid size
    n_points = 100  # If SDP fails, will fall back to n=80
    dx = 2.0 / n_points

    # Decision variables
    h = cp.Variable(n_points)
    X = cp.Variable((n_points, n_points), symmetric=True)
    t = cp.Variable()

    # Constraints for SDP relaxation
    # McCormick relaxations for products
    constraints = [cp.diag(X) == h]
    constraints.append(cp.sum(h) == n_points / 2)
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    constraints.append(t >= 0)

    # Product constraints (McCormick envelopes)
    for i in range(n_points):
        for j in range(i+1, n_points):
            if i + 1 > j:
                continue
            # X[i,j] ≤ h[i], X[i,j] ≤ h[j] 
            constraints.append(X[i,j] <= h[i])
            constraints.append(X[i,j] <= h[j])
            # X[i,j] ≥ h[i] + h[j] - 1
            constraints.append(X[i,j] >= h[i] + h[j] - 1)

    # Overlap constraints per shift s from 1 to n_points-1
    # Compute overlaps efficiently: sum_{i = 0}^{n_points - s - 1} X[i, i+s]
    for s in range(1, n_points):
        o_s = cp.sum([X[i, i+s] for i in range(n_points - s)])
        constraints.append(o_s >= t)

    # Objective: maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    best_sdp_c5 = float('inf')
    best_h_sdp = None

    try:
        # Solve with time limit
        prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-5, time_limit=150)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            t_opt = t.value
            c5_lower_bound = 1.0 - (2/n_points)*t_opt
            if c5_lower_bound < best_sdp_c5:
                best_sdp_c5 = c5_lower_bound
                best_h_sdp = h_opt.copy()
    except SolverError:
        pass
    except Exception:
        pass

    # Fallback to small n if SDP hangs or fails
    if best_h_sdp is None or np.isnan(best_h_sdp).any():
        n_points = 80
        dx = 2.0 / n_points
        try:
            h = cp.Variable(n_points)
            X = cp.Variable((n_points, n_points), symmetric=True)
            t = cp.Variable()
            
            constraints = [cp.diag(X) == h]
            constraints.append(cp.sum(h) == n_points / 2)
            constraints.append(h >= 0)
            constraints.append(h <= 1)
            constraints.append(t >= 0)

            for i in range(n_points):
                for j in range(i+1, n_points):
                    if i + 1 > j:
                        continue
                    constraints.append(X[i,j] <= h[i])
                    constraints.append(X[i,j] <= h[j])
                    constraints.append(X[i,j] >= h[i] + h[j] - 1)

            for s in range(1, n_points):
                o_s = cp.sum([X[i, i+s] for i in range(n_points - s)])
                constraints.append(o_s >= t)

            objective = cp.Maximize(t)
            prob = cp.Problem(objective, constraints)
            
            prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-5, time_limit=150)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_opt = h.value
                t_opt = t.value
                c5_lower_bound = 1.0 - (2/n_points)*t_opt
                best_sdp_c5 = c5_lower_bound
                best_h_sdp = h_opt.copy()
        except Exception:
            pass

    if best_h_sdp is None:
        # Return a fallback if nothing worked
        h_fallback = np.ones(n_points) / 2
        h_fallback[0:10] = 0.5
        h_fallback[n_points//2:n_points//2+10] = 0.5
        h_fallback /= np.sum(h_fallback)*dx
        best_h_sdp = h_fallback
        best_sdp_c5 = evaluate_c5(h_fallback)

    # --- Phase B: Rounding and Local Refinement ------------------------

    candidates = [best_h_sdp.copy()]
    best_h_final = best_h_sdp.copy()
    best_c5_final = evaluate_c5(best_h_final)

    # Rounding with random walks
    try:
        # Decompose into Y = V * sqrt(Λ) where V Λ V^T = X_opt (truncated)
        X_opt = best_h_sdp[:,None] * best_h_sdp[None,:]  # approx h h^T
        for i in range(n_points):
            for j in range(i+1, n_points):
                if i + 1 > j:
                    continue
                X_opt[i,j] = np.minimum(best_h_sdp[i], best_h_sdp[j])
                X_opt[i,j] = np.maximum(X_opt[i,j], best_h_sdp[i] + best_h_sdp[j] - 1)
                X_opt[j,i] = X_opt[i,j]

        # SVD for simulating rounding
        u, s, vh = np.linalg.svd(X_opt)
        rank = np.sum(s > 1e-8)
        r = min(10, rank)
        V = u[:, :r]
        sqrt_s = np.sqrt(s[:r])
        Y = V * sqrt_s[None,:]  # Y is r x n, \approx X^{1/2}
        
        rounds = 0
        while rounds < 20 and time.time() - start_time < budget_s - 100:
            # Sample random vector
            g = np.random.randn(r)
            y = Y.T @ g
            
            # Threshold to binary (heuristically try to include exactly n/2)
            idxs_sorted = np.argsort(-y)
            h_bin = np.zeros(n_points)
            h_bin[idxs_sorted[:n_points//2]] = 1
            
            # Refine via swaps
            def compute_overlap(h):
                corr = correlate(h, 1-h, mode="full") 
                return np.max(corr) * dx

            for _ in range(50):  # expensive swaps
                idx_0 = np.where(h_bin == 0)[0]
                idx_1 = np.where(h_bin == 1)[0]
                if len(idx_0) == 0 or len(idx_1) == 0: break
                
                i0 = np.random.choice(idx_0)
                i1 = np.random.choice(idx_1)
                new_h = h_bin.copy()
                new_h[i0], new_h[i1] = new_h[i1], new_h[i0]
                if compute_overlap(new_h) < compute_overlap(h_bin):
                    h_bin = new_h
            
            if compute_overlap(h_bin) < best_c5_final:
                best_h_final = h_bin.copy()
                best_c5_final = compute_overlap(h_bin)
            rounds += 1
            
    except Exception:
        pass

    # --- Phase C: Some final deterministic refinement ------------------

    def soft_max_grad(h):
        # compute ∇ Φ_α(h) with α schedule
        dx = 2.0 / n_points
        def f_s(h, s):
            # Length of this shift: n_points - s
            idxs = np.arange(n_points - s)
            return np.sum(h[idxs] * (1 - h[idxs + s])) * dx
        f_vals = [f_s(h, s) for s in range(1, n_points)]
        alpha_schedule = [5, 15, 40]
        
        # Compute gradient direction
        grad = np.zeros_like(h)
        for alpha in alpha_schedule:
            exp_fs = np.exp(alpha * np.array(f_vals))
            wts = exp_fs / np.sum(exp_fs)
            for i, s in enumerate(range(1, n_points)):
                grad[i] += wts[i] * (1 - (h[i+s] if i+s < n_points else 0) - (h[i-s] if i-s >= 0 else 0))
        return grad

    # Apply local search using smooth max (trust-constr)
    try:
        # Initial vector
        h0 = best_h_final.copy()
        ratio = np.sum(h0) / (n_points/2)  # normalize later
        h0 = h0 / ratio  # scale to sum(n/2)
        
        def smooth_max_objective(h):
            dx = 2.0 / n_points
            alpha = 40
            f_vals = [np.sum(h[i:] * (1 - h[i+1:])) * dx for i in range(n_points - 1)]
            exp_fs = np.exp(alpha * np.array(f_vals))
            return -np.sum(exp_fs) / alpha
        
        # Gradient (could cache and optimize)
        def smooth_max_grad(h):
            dx = 2.0 / n_points
            alpha = 40
            f_vals = [np.sum(h[i:] * (1 - h[i+1:])) * dx for i in range(n_points - 1)]
            exp_fs = np.exp(alpha * np.array(f_vals))
            sum_exp = np.sum(exp_fs)
            weights = exp_fs / sum_exp
            grad = np.zeros_like(h)
            # Forward differences
            for i in range(n_points):
                for s in range(1, n_points - i if i < n_points - 1 else 0):
                    if s >= 1:
                        grad[i] += weights[s-1] * (1 - (h[i+s] if i+s < n_points else 0))
                        grad[i+s] += weights[s-1] * (-h[i] if i < n_points else 0)
            return -grad  # scales by neg because maximize
        
        def objective(h):
            return np.max(np.correlate(h, 1 - h, mode="full")) * dx

        # Refinement with trust-constr (tighter than other methods at this point)
        for count in range(5):
            h_ref = h0.copy()
            try:
                # Project onto constraints, Solve
                res = minimize(smooth_max_objective, h_ref, method='trust-constr', 
                              bounds=[(0., 1.) for _ in range(n_points)],
                              constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2},
                              options={'maxiter': 100, 'gtol': 1e-10, 'xtol': 1e-10})
                if res.success:
                    h_ref = res.x
                else:
                    break
                # Final objective evaluation
                c5_val = evaluate_c5(h_ref)
                if c5_val < best_c5_final:
                    best_c5_final = c5_val
                    best_h_final = h_ref.copy()
                h0 = h_ref
            except Exception:
                 break
    except Exception as e:
        pass

    return (best_h_final.tolist(), best_c5_final, n_points)
