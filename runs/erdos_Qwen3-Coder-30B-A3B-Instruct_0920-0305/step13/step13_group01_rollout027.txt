import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

# Global constants
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_correlation(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5_exact(h):
    """Exact evaluation of C5 using np.correlate"""
    corr = correlate(h, 1 - h, mode='full')
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_logsumexp(h, alpha):
    """Compute smooth approximation to max using log-sum-exp"""
    corr = compute_correlation(h)
    corr = corr.real  # Ensure real values
    
    # Stabilize: subtract max to prevent overflow
    max_corr = np.max(corr)
    exp_corr = np.exp(alpha * (corr - max_corr))
    log_sum_exp = np.log(np.sum(exp_corr)) + max_corr * alpha
    return dx * log_sum_exp / alpha

def gradient_logsumexp(h, alpha):
    """Compute gradient of smooth max with respect to h"""
    corr = compute_correlation(h)
    corr = corr.real
    
    # Stabilize: subtract max to prevent overflow
    max_corr = np.max(corr)
    exp_corr = np.exp(alpha * (corr - max_corr))
    weights = exp_corr / np.sum(exp_corr)
    
    # Gradient computation
    grad = np.zeros_like(h)
    # Need to compute weights * (1 - h_shifted) for all shifts
    # But we'll do it efficiently with convolution-like operations
    
    # Reshape weights to be (2n-1,1) for broadcasting
    weights_ext = np.zeros(2 * len(h) - 1)
    weights_ext[:len(weights)] = weights
    
    # For each shift k, weight*1 and weight*shifted term
    # Using the fact that correlation is symmetric and shift operations
    
    # Efficient way using FFTs or direct sum
    # Each component of grad is sum over k of weight[k] * (1 - h[i+k] - h[i-k])
    # This is (sum weight[k]) + sum weight[k] * (-h[i+k] - h[i-k]) 
    # But not quite - careful index handling
    
    # Let's go with direct, more straightforward:
    for k in range(len(h)):
        # Compute: grad[i] += weight[k] * (1 - h[i+k] - h[i-k])  
        k_idx = k - (len(h) - 1)  # k = 0->-(n-1), 1->-(n-2), ..., n-1->0, ..., 2n-2->(n-1)
        # This is counter-intuitive, but doing proper index handling
        # We need to make sure we're taking from the right place
        # Each k maps to the shift in correlation array
        # For a full convolution of correct size, do it directly:
        warnings.filterwarnings('ignore')  # ignore invalid value warnings from nan propagation
    # Reconstruct gradient manually to avoid complications
    # Carefully loop:
    grad = np.zeros_like(h)
    L = 2 * len(h) - 1
    
    # Compute weight*1 + weight*shift terms for all valid i
    for k in range(-len(h)+1, len(h)):
        # This is weight at correlation index corresponding to shift k
        # Will map correlation index k+L//2 to k shift, e.g. index L//2 maps to shift 0
        corr_index = k + L // 2
        if 0 <= corr_index < L:
            w = weights[corr_index]
            if k >= 0:
                # h[i+k] terms, for i = 0 to len(h)-k-1 should be considered
                for i in range(len(h) - k):
                    grad[i] -= w * h[i + k]
                # h[i-k] terms for i = k to len(h)-1  
                for i in range(k, len(h)):
                    grad[i] -= w * h[i - k]
            else:  # k < 0
                # h[i+k] terms for i = -k to len(h)-1 = should be i = 0 to len(h)+k-1
                for i in range(len(h) + k):
                    grad[i] -= w * h[i - k]
                # h[i-k] for i = 0 to -k-1
                for i in range(-k):
                    grad[i] -= w * h[i + k]
    # Add the weight*1 terms
    grad += np.sum(weights)
    return grad

def create_initial_vectors():

    starts = []
    
    # 1. Replicated legacy pattern (extended from 120 to 240)
    h0 = np.array(initial_h_values)
    h0_extended = np.tile(h0, 2)
    starts.append(h0_extended)
    
    # 2. Pure sinusoid
    x = np.linspace(0, 2, n_points, endpoint=False) + dx / 2
    phase = np.random.rand() * 2 * np.pi
    h1 = 0.5 + 0.5 * np.sin(np.pi * x + phase)
    h1 = project_to_sum_box(h1, target_sum)
    starts.append(h1)
    
    # 3. Amplitude-tuned sinusoid
    a = np.random.rand() * 0.5
    h2 = 0.5 + a * np.sin(np.pi * x + phase)
    h2 = project_to_sum_box(h2, target_sum)
    starts.append(h2)
    
    # 4. Triangular wave
    h3 = np.zeros_like(x)
    mid = n_points // 2
    h3[:mid] = np.linspace(0, 1, mid)
    h3[mid:] = np.linspace(1, 0, n_points - mid)
    h3 = project_to_sum_box(h3, target_sum)
    starts.append(h3)
    
    # 5. Random uniform with projection
    h4 = np.random.rand(n_points)
    h4 = project_to_sum_box(h4, target_sum)
    starts.append(h4)
    
    # 6. Hybrid: sine + random
    h5 = 0.7 * (0.5 + 0.5 * np.sin(np.pi * x + phase)) + 0.3 * np.random.rand(n_points)
    h5 = project_to_sum_box(h5, target_sum)
    starts.append(h5)
    
    return starts

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Create initial vectors
    initial_vectors = create_initial_vectors()
    
    best_c5 = float('inf')
    best_h = None
    
    # Alpha levels for continuation
    alphas = [1.0, 5.0, 20.0, 80.0, 300.0]
    
    # Optimization settings
    method = 'SLSQP'
    
    for restart_idx, h_start in enumerate(initial_vectors):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        current_h = h_start.copy()
        current_c5 = evaluate_c5_exact(current_h)
        
        # Continuation schedule (solve series of smooth approximations)
        for alpha in alphas:
            # Append a small random perturbation to avoid stagnation
            if alpha > 10:
                current_h += np.random.normal(0, 0.01, n_points)
                current_h = np.clip(current_h, 0, 1)
                current_h = project_to_sum_box(current_h, target_sum)
            
            def obj_func(h):
                return smooth_max_logsumexp(h, alpha)
            
            def grad_func(h):
                return gradient_logsumexp(h, alpha)
                
            # Define constraints
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}] 
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Optimize (use a lower tolerance to prevent tightly converging to modest optima)
            res = minimize(obj_func, current_h, method=method, jac=grad_func, 
                          constraints=cons, bounds=bounds, 
                          options={'ftol': 1e-9, 'gtol': 1e-9, 'maxiter': 200})
                
            if res.success:
                current_h = res.x.copy()
                # Project again to ensure adherence to sum constraint
                current_h = project_to_sum_box(current_h, target_sum)
                current_c5 = evaluate_c5_exact(current_h)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                
        # End continuation now check if found better solution
   
    # Local search: improve the best solution found
    if best_h is not None:
        current_h = best_h.copy()
        current_c5 = evaluate_c5_exact(current_h)
        
        max_swaps = 3000
        no_improvement_count = 0
        threshold_count = 100
        
        for i in range(max_swaps):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Try swap move
            i1, i2 = np.random.choice(n_points, 2, replace=False)
            h_test = current_h.copy()
            h_test[i1], h_test[i2] = h_test[i2], h_test[i1]
            
            # Project to box sum constraint
            h_test = project_to_sum_box(h_test, target_sum)
            c5_test = evaluate_c5_exact(h_test)
            
            if c5_test < current_c5:
                current_h = h_test
                current_c5 = c5_test
                no_improvement_count = 0
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
            else:
                no_improvement_count += 1
            
            # Stop early if no improvement for some time
            if no_improvement_count > threshold_count:
                break
                
    # Final evaluation using exact correlation
    if best_h is None:
        # fallback: use replicated initial pattern
        best_h = np.tile(np.array(initial_h_values), 2)
        best_h = project_to_sum_box(best_h, target_sum)
        
    final_c5 = evaluate_c5_exact(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
