import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from collections import defaultdict

def max_corr(h):
    """Compute maximum correlation value"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 for the given h"""
    return 2.0/len(h) * max_corr(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    target_M = int(np.floor(0.3807 / dx))  # 22
    
    # Load initial solution
    h_cont = np.array(initial_h_values)
    h0 = (h_cont >= 0.5).astype(int)
    current_sum = np.sum(h0)
    
    # Adjust to exactly 60 ones
    if current_sum > 60:
        # Find indices with highest continuous values to flip to 0
        flip_indices = np.argsort(h_cont)[::-1][:current_sum - 60]
        h0[flip_indices] = 0
    elif current_sum < 60:
        # Find indices with lowest continuous values to flip to 1
        flip_indices = np.argsort(h_cont)[:60 - current_sum]
        h0[flip_indices] = 1
    
    # Warm-start for MILP: identify fixed positions
    FixedOne = set(np.where(h0 == 1)[0])
    FixedZero = set(np.where(h0 == 0)[0])
    FreeIdx = set(range(n_points)) - FixedOne - FixedZero
    
    # Build and solve MILP
    milp_time = min(260, budget_s - 100)
    
    if FreeIdx and milp_time > 10:
        # Create optimization variables
        if FreeIdx:
            h_vars = cp.Variable(len(FreeIdx), integer=True)
            h_binary = np.zeros(n_points)
            free_indices = sorted(FreeIdx)
            for i, idx in enumerate(free_indices):
                h_binary[idx] = h_vars[i]
            # Set fixed values
            for i in FixedOne:
                h_binary[i] = 1
            for i in FixedZero:
                h_binary[i] = 0
        else:
            h_binary = h0.astype(int)
        
        # Set up constraints
        M = cp.Variable()
        
        # Constraints for fixed values
        constraints = [cp.sum(h_binary) == 60, M >= 0]
        
        # Add the product linearization constraints
        # We only consider lags where target can be exceeded
        lags_to_consider = range(-97, 98)  # n - target_M = 120 - 22 = 98
        
        # Create auxiliary variables for cross products
        w_vars = {}
        I_vars = {}
        
        for l in lags_to_consider:
            if l >= 0:
                # Forward lags
                n_lag = n_points - l
                if n_lag > 0:
                    # Create variables w[i,l] for i = 0 to n_lag-1
                    w_l = cp.Variable(n_lag, integer=True)
                    w_vars[l] = w_l
                    I_l = cp.sum(h_binary[:n_lag] - w_l)
                    I_vars[l] = I_l
                    # Add constraints
                    for i in range(n_lag):
                        constraints.extend([
                            w_l[i] <= h_binary[i],
                            w_l[i] <= h_binary[i+l],
                            w_l[i] >= h_binary[i] + h_binary[i+l] - 1
                        ])
                    constraints.append(I_l <= M)
            else:
                # Backward lags
                l_abs = -l
                n_lag = n_points - l_abs
                if n_lag > 0:
                    w_l = cp.Variable(n_lag, integer=True)
                    w_vars[-l_abs] = w_l
                    # Indices in h_binary for forward lag l_abs
                    I_l = cp.sum(h_binary[l_abs:] - w_l)
                    I_vars[-l_abs] = I_l
                    # Add constraints
                    for i in range(n_lag):
                        constraints.extend([
                            w_l[i] <= h_binary[l_abs+i],
                            w_l[i] <= h_binary[i],
                            w_l[i] >= h_binary[l_abs+i] + h_binary[i] - 1
                        ])
                    constraints.append(I_l <= M)
        
        # Add constraints to make all I_l <= M
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        try:
            prob.solve(solver=cp.CBC, cbc_params={'maxTime': milp_time})
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                # Extract solution
                if FreeIdx and len(FreeIdx) > 0:
                    for i, idx in enumerate(free_indices):
                        h0[idx] = int(round(h_vars[i].value))
                    # Ensure sum is exactly 60
                    current_sum = np.sum(h0)
                    if current_sum == 60:
                        pass
                    elif current_sum < 60:
                        # Find zeros to flip to ones
                        zero_indices = np.where(h0 == 0)[0]
                        for i in range(60 - current_sum):
                            if i < len(zero_indices):
                                h0[zero_indices[i]] = 1
                    else:
                        # Find ones to flip to zeros
                        one_indices = np.where(h0 == 1)[0]
                        for i in range(current_sum - 60):
                            if i < len(one_indices):
                                h0[one_indices[i]] = 0
                else:
                    if np.sum(h0) != 60:
                        # If we have all fixed positions, still ensure the constraint
                        counter = 0
                        while np.sum(h0) != 60:
                            if np.sum(h0) > 60:
                                # Find ones to flip to zeros
                                one_indices = np.where(h0 == 1)[0]
                                for i in range(np.sum(h0) - 60):
                                    if i < len(one_indices) and one_indices[i] not in FixedOne:
                                        h0[one_indices[i]] = 0
                                        counter += 1
                                        if counter > 100:
                                            break
                            else:
                                # Find zeros to flip to ones
                                zero_indices = np.where(h0 == 0)[0]
                                for i in range(60 - np.sum(h0)):
                                    if i < len(zero_indices) and zero_indices[i] not in FixedZero:
                                        h0[zero_indices[i]] = 1
                                        counter += 1
                                        if counter > 100:
                                            break
        except:
            pass
    
    # If we didn't get a better solution from MILP, use original solution
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Post-processing: Critical-lag greedy descent
    if time.time() - start_time < budget_s - 10:
        h = best_h.copy()
        corr = correlate(h, 1 - h, mode='full')
        max_corr_val = np.max(corr)
        
        # Build lag_to_indices mapping
        lag_to_indices = defaultdict(list)
        for l in range(-n_points + 1, n_points):
            if l >= 0:
                for i in range(n_points - l):
                    if h[i] == 1 and h[i + l] == 0:
                        lag_to_indices[l].append(i)
            else:
                for i in range(n_points + l):
                    if h[i] == 1 and h[i - l] == 0:
                        lag_to_indices[l].append(i)
        
        # Convert lag_to_indices from dict to a more convenient structure
        lag_to_indices_list = {}
        for l in range(-n_points + 1, n_points):
            lag_to_indices_list[l] = lag_to_indices[l]
        
        # Re-evaluate correlation to make sure our data structures are correct
        corr = correlate(h, 1 - h, mode='full')
        max_corr_val = np.max(corr)
        max_lag = np.argmax(corr) - (n_points - 1)
        
        # Run critical lag improvement
        swap_count = 0
        max_swaps = 100000
        
        while time.time() - start_time < budget_s - 20 and swap_count < max_swaps and max_corr_val > target_M:
            # Find the critical lag
            max_lag = np.argmax(corr) - (n_points - 1)
            
            # If there are no mismatches for this lag, we're done
            if len(lag_to_indices_list[max_lag]) == 0:
                break
            
            # Pick a random mismatch
            i = np.random.choice(lag_to_indices_list[max_lag])
            
            # Find a suitable position to swap with
            candidates = []
            # Make sure this is during sensible range
            if max_lag >= 0:
                # Forward lag: look for 0s that don't create new mismatches
                for j in range(n_points):
                    if h[j] == 0 and (j + max_lag >= n_points or h[j + max_lag] == 1):
                        candidates.append(j)
            else:
                # Backward lag: look for 0s that don't create new mismatches
                for j in range(n_points):
                    if h[j] == 0 and (j - max_lag < 0 or h[j - max_lag] == 1):
                        candidates.append(j)
            
            if len(candidates) == 0:
                # If no good candidate, try a random 0
                zero_indices = np.where(h == 0)[0]
                if len(zero_indices) > 0:
                    j = np.random.choice(zero_indices)
                else:
                    swap_count += 1
                    continue
            else:
                j = np.random.choice(candidates)
            
            # Perform swap
            h[i], h[j] = h[j], h[i]
            swap_count += 1
            
            # Rebuild the mismatch set for this lag
            corr = correlate(h, 1 - h, mode='full')
            max_corr_val = np.max(corr)
            max_lag = np.argmax(corr) - (n_points - 1)
            
            # Update lag_to_indices for the affected lags
            for l in range(-n_points + 1, n_points):
                lag_to_indices_list[l].clear()

                    for i in range(n_points - l):
                        if h[i] == 1 and h[i + l] == 0:
                            lag_to_indices_list[l].append(i)
                else:
                    for i in range(n_points + l):

                            lag_to_indices_list[l].append(i)

        best_h = h.copy()

            ones_indices = np.random.choice(n_points, size=60, replace=False)

            h = h0.copy()
            corr = correlate(h, 1 - h, mode='full')

                if l >= 0:

                        if h[i] == 1 and h[i + l] == 0:
                            lag_to_indices[l].append(i)
                else:
                    for i in range(n_points + l):
                        if h[i] == 1 and h[i - l] == 0:
                            lag_to_indices[l].append(i)
            
            swap_count = 0
            max_swaps = 50000
            
            while swap_count < max_swaps and max_corr_val > target_M:
                # Find the critical lag
                max_lag = np.argmax(corr) - (n_points - 1)
                
                if len(lag_to_indices[max_lag]) == 0:
                    break
                
                # Pick a random mismatch
                i = np.random.choice(lag_to_indices[max_lag])
                
                # Find a suitable position to swap with
                candidates = []
                if max_lag >= 0:
                    # Forward lag: look for 0s that don't create new mismatches
                    for j in range(n_points):
                        if h[j] == 0 and (j + max_lag >= n_points or h[j + max_lag] == 1):
                            candidates.append(j)
                else:
                    # Backward lag: look for 0s that don't create new mismatches
                    for j in range(n_points):
                        if h[j] == 0 and (j - max_lag < 0 or h[j - max_lag] == 1):
                            candidates.append(j)
                
                if len(candidates) == 0:
                    # If no good candidate, try a random 0
                    zero_indices = np.where(h == 0)[0]
                    if len(zero_indices) > 0:
                        j = np.random.choice(zero_indices)
                    else:
                        break
                else:
                    j = np.random.choice(candidates)
                
                # Perform swap
                h[i], h[j] = h[j], h[i]
                swap_count += 1
                
                # Rebuild the mismatch set for this lag
                corr = correlate(h, 1 - h, mode='full')
                max_corr_val = np.max(corr)
                
                # Update lag_to_indices for the affected lags
                for l in range(-n_points + 1, n_points):
                    lag_to_indices[l] = []
                    if l >= 0:
                        for i in range(n_points - l):
                            if h[i] == 1 and h[i + l] == 0:
                                lag_to_indices[l].append(i)

        best_local_h = best_h.copy()

            if improved_c5 < best_local_c5:
                best_local_c5 = improved_c5
                best_local_h = improved_h.copy()
        
        if best_local_c5 < best_c5:
            best_h = best_local_h.copy()
            best_c5 = best_local_c5
    
    return (best_h.tolist(), float(best_c5), n_points)
