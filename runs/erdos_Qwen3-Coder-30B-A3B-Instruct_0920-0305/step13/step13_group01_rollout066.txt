import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

# Constants
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_correlation_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5_exact(h):
    """Compute exact C5 using np.correlate"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_logsumexp(corr, alpha):
    """Smooth approximation of max with log-sum-exp"""
    stabilized = corr - np.max(corr)
    exp_corr = np.exp(alpha * stabilized)
    return np.log(np.sum(exp_corr)) / alpha

def smooth_max_grad(corr, alpha, h):
    """Gradient of smooth max with respect to h"""
    stabilized = corr - np.max(corr)
    exp_corr = np.exp(alpha * stabilized)
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient: sum over k of w_k * (1 - h_{i+k} - h_{i-k})
    grad = np.zeros_like(h)
    for k in range(len(h)):
        if k == 0:
            grad += weights[k] * (1 - h)
        else:
            grad += weights[k] * (1 - np.roll(h, k) - np.roll(h, -k))
    
    return dx * grad

def compute_smooth_objective(h, alpha):
    """Compute the smooth objective for a given α"""
    corr = compute_correlation_fft(h)
    return smooth_max_logsumexp(corr, alpha)

def compute_smooth_gradient(h, alpha):
    """Compute the gradient of the smooth objective for a given α"""
    corr = compute_correlation_fft(h)
    return smooth_max_grad(corr, alpha, h)

def continuation_optimization(h0, alpha_levels=[1, 5, 20, 80, 300]):
    """Optimize using continuation method"""
    h_current = h0.copy()
    
    for alpha in alpha_levels:
        # Define objective and gradient
        def obj(h):
            return compute_smooth_objective(h, alpha)
        
        def grad(h):
            return compute_smooth_gradient(h, alpha)
        
        # Constraints
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum, 
                'jac': lambda h: np.ones_like(h)}
        
        # Bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimize
        try:
            res = minimize(obj, h_current, method='SLSQP', jac=grad, 
                          bounds=bounds, constraints=cons, 
                          options={'ftol': 1e-9, 'gtol': 1e-9, 'maxiter': 200})
            if res.success:
                h_current = res.x
            else:
                # Fallback to trust-constr if SLSQP fails
                res = minimize(obj, h_current, method='trust-constr', jac=grad,
                              bounds=bounds, constraints=cons,
                              options={'gtol': 1e-9, 'maxiter': 200})
                if res.success:
                    h_current = res.x
        except:
            pass
        
        # Project back to feasible region
        h_current = project_to_sum_box(h_current, target_sum)
    
    return h_current

    # 4. Triangular wave

    rand_uniform = np.random.uniform(0, 1, n_points)

    hybrid = 0.7 * sinusoid + 0.3 * rand_uniform
    vectors.append(project_to_sum_box(hybrid, target_sum))

def local_search(h_best):
    """Perform local discrete search for polishing"""
    h_current = h_best.copy()
    c5_current = evaluate_c5_exact(h_current)
    
    # Swap moves
    for _ in range(3000):
        i, j = np.random.choice(n_points, 2, replace=False)
        h_trial = h_current.copy()
        h_trial[i], h_trial[j] = h_trial[j], h_trial[i]
        h_trial = project_to_sum_box(h_trial, target_sum)
        
        c5_trial = evaluate_c5_exact(h_trial)
        if c5_trial < c5_current:
            h_current = h_trial
            c5_current = c5_trial
    
    return h_current

    start_time = time.time()
    
    # Create initial vectors
    initial_vectors = create_initial_vectors()
    
    best_c5 = float('inf')
    best_h = None
    
    # Run continuation optimizations
    num_starts = 0
    for i, h_start in enumerate(initial_vectors):
        if time.time() - start_time > budget_s - 10:

        h_optimized = continuation_optimization(h_start)
        c5_opt = evaluate_c5_exact(h_optimized)

            best_h = h_optimized.copy()
        
        num_starts += 1
