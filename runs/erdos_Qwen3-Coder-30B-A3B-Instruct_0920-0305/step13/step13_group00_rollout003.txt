    constraints = [
        h >= 0,
        h <= 1,

        t >= 0
    ]

    for s in range(1, n_points):
        # For shift s, we have terms h[i] * h[i+s] where i+s < n
        for i in range(n_points - s):
            y = cp.Variable()
            constraints.extend([
                y >= 0,
                y <= h[i],
                y <= h[i + s],
                y >= h[i] + h[i + s] - 1
            ])
