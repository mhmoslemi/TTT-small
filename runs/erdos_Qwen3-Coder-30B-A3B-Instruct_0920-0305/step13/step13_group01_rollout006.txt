def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5_exact(h):
    """Evaluate exact C5 using numpy correlate"""
    corr = correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_logsumexp(h, alpha):
    """Compute gradient of smooth max with log-sum-exp approximation"""
    corr = compute_corr_fft(h)
    # Stabilize: subtract max value to prevent overflow
    max_corr = np.max(corr)
    exp_corr = np.exp(alpha * (corr - max_corr))
    Z = np.sum(exp_corr)
    weights = exp_corr / Z
    
    grad = np.zeros_like(h)
    n = len(h)
    # Compute gradient as sum of weighted shifts of h
    for k in range(-n+1, n):
        if k < 0:
            shifted = np.concatenate([np.zeros(-k), h[:n+k]])
        elif k > 0:
            shifted = np.concatenate([h[k:], np.zeros(k)])
        else:
            shifted = h.copy()
        grad += weights[n-1+k] * (1 - shifted)
    return dx * grad

    opt_kwargs = {
        'method': 'SLSQP',
        'bounds': [(0, 1) for _ in range(n_points)],
        'constraints': [{'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}],
        'options': {'ftol': 1e-9, 'gtol': 1e-9, 'maxiter': 200}
    }
    
    for alpha in alpha_sequence:
        # Compute gradient
        grad = gradient_logsumexp(h, alpha)
        
        # Define objective function and gradient for scipy
        def obj(h):
            corr = compute_corr_fft(h)
            max_corr = np.max(corr)
            stabilized_corr = corr - max_corr
            exp_corr = np.exp(alpha * stabilized_corr)
            Z = np.sum(exp_corr)
            return dx * (max_corr + np.log(Z) / alpha)
        
        def grad_func(h):
            return gradient_logsumexp(h, alpha)
        
        # Optimize with gradient
        res = minimize(obj, h, method='SLSQP', jac=grad_func, bounds=[(0,1) for _ in range(n_points)],
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}],
                       options={'ftol': 1e-9, 'gtol': 1e-9, 'maxiter': 200})
        
        if res.success:
            h = res.x.copy()
        else:
            # fallback to projected gradient if optimization fails
            h = project_to_sum_box(h - 0.01 * grad, TARGET_SUM)
        
        # Ensure feasibility
        h = project_to_sum_box(h, TARGET_SUM)
    
    return h

def local_search(h):
    """Perform discrete local search to improve solution"""
    h_best = h.copy()
    c5_best = evaluate_c5_exact(h)
    
    # Try swaps and small transfers
    for _ in range(3000):
        if np.random.rand() < 0.5:
            # Try a swap
            i, j = np.random.choice(n_points, 2, replace=False)
            h_trial = h_best.copy()
            h_trial[i], h_trial[j] = h_trial[j], h_trial[i]
            h_trial = project_to_sum_box(h_trial, TARGET_SUM)
            c5_trial = evaluate_c5_exact(h_trial)
        else:
            # Try small transfer
            i, j = np.random.choice(n_points, 2, replace=False)
            h_trial = h_best.copy()
            delta = np.random.uniform(0, 0.01)
            if h_trial[i] >= delta:
                h_trial[i] -= delta
                h_trial[j] += delta
                h_trial = project_to_sum_box(h_trial, TARGET_SUM)
                c5_trial = evaluate_c5_exact(h_trial)
            else:
                continue
                
        if c5_trial < c5_best:
            h_best = h_trial
            c5_best = c5_trial
    
    return h_best, c5_best
