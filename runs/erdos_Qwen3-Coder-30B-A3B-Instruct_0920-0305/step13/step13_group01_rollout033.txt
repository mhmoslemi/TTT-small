def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    # Return maximum correlation value scaled by dx
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):

    current_sum = np.sum(x)

        return np.clip(x, 0, 1)

    if diff > 0:
        # Need to increase sum
        x = np.clip(x + diff / len(x), 0, 1)
    else:
        # Need to decrease sum
        x = np.clip(x + diff / len(x), 0, 1)
    
    # Fine adjustment with binary search if needed
    def objective(lam):
        projected = np.clip(x + lam, 0, 1)

    a, b = -1.0, 1.0

            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
