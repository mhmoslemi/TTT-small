import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def create_convex_subproblem(h_current, lags_set, n):
    """Create and solve the convex subproblem for the given lags"""
    
    # Variables
    h = cp.Variable(n)
    T = cp.Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1, cp.sum(h) == TARGET_SUM]
    
    # Add lag constraints
    for ell in lags_set:
        # Compute the convex part: sum(h) + 1/4 * sum((h_i - h_{i+ell})^2)
        # Compute the linear part: 1/2 * sum((h_i^(t) + h_{i+ell}^(t)) * (h_i + h_{i+ell}))
        if ell == 0:
            # Simple case for ell = 0
            convex_part = cp.sum(h) + 0.25 * cp.sum_squares(h - h)
            linear_part = 0.5 * cp.sum((h_current + h_current) * (h + h))
        else:
            # For other lags: h_i and h_{i+ell}
            if ell > 0:
                h_shifted = cp.hstack([cp.Variable(ell), h[:-ell]])
            else:
                h_shifted = cp.hstack([h[-ell:], cp.Variable(-ell)])
                
            # Temporary variables for convolution
            convex_term = cp.sum(h) + 0.25 * cp.sum_squares(h - cp.roll(h, ell))
            linear_term = 0.5 * cp.sum((h_current + cp.roll(h_current, ell)) * (h + cp.roll(h, ell)))
            constraints.append(convex_term - linear_term <= T)
    
    # Objective
    objective = cp.Minimize(T)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            T_opt = T.value
            return h_opt, T_opt, True
    except:
        pass
    
    return h_current, float('inf'), False

def generate_initial_lags(n):
    """Generate initial set of lags including 0 and key shifts"""
    base_lags = [0, 1, 2, 3, 4, 5, 10, 15, 20, 25, 30]
    lags = [0]
    for lag in base_lags:
        lags.extend([-lag, lag])
    lags = list(set(lags))
    lags = [l for l in lags if -n+1 <= l <= n-1]
    return lags

def find_violating_lags(h, T, all_lags, tolerance=1e-6):
    """Find lags that violate the constraint by more than tolerance"""
    I_values = compute_corr_fft(h)
    max_val = np.max(I_values)
    if max_val <= T + tolerance:
        return []
    violating = []
    for ell in all_lags:
        if ell < 0:
            corr_val = I_values[n_points - 1 + ell] 
        else:
            corr_val = I_values[ell]
        if corr_val > T + tolerance:
            violating.append(ell)
    return violating

def deterministic_rounding(h_cont, n):
    """Deterministic rounding method that maintains sum constraint"""
    # Sort indices by descending values
    sorted_idx = np.argsort(h_cont)[::-1]
    b = np.zeros(n)
    b[sorted_idx[:n//2]] = 1
    return b

def critical_lag_polish(h_binary, max_iterations=10000):
    """Polish binary solution using critical lag swaps"""
    b = h_binary.copy()
    best_solution = b.copy()
    best_c5 = evaluate_c5(b)
    
    start_time = time.time()
    for iteration in range(max_iterations):
        if time.time() - start_time > 50:  # Moderate time limit for polish
            break
            
        corr_full = compute_corr_fft(b)
        max_lag = np.argmax(corr_full)
        max_val = corr_full[max_lag]
        if max_val <= 22:  # target threshold
            break
            
        # Find indices with mismatched 1→0 pairs for worst lag
        if max_lag < n_points - 1:
            # Positive lag case
            i_indices = np.where((b[:n_points - max_lag] == 1) & (b[max_lag:] == 0))[0]
        else:
            # Negative lag case
            i_indices = np.where((b[max_lag - n_points + 1:] == 1) & (b[:n_points - max_lag + n_points - 1] == 0))[0]
        
        if len(i_indices) == 0:
            break
            
        i = np.random.choice(i_indices)
        if max_lag < n_points - 1:
            # Find compatible zeros
            z_indices = np.where((b[max_lag:] == 0) & (b[:n_points - max_lag] == 1))[0]  # this condition may need further refinement
        else:
            z_indices = np.where((b[:max_lag - n_points + 1] == 0) & (b[max_lag - n_points + 1:] == 1))[0]
        
        if len(z_indices) == 0:
            j = np.random.choice(np.where(b == 0)[0])
        else:
            j = np.random.choice(z_indices)
            
        # Perform swap
        b[i], b[j] = b[j], b[i]
        current_c5 = evaluate_c5(b)
        if current_c5 < best_c5:
            best_solution = b.copy()
            best_c5 = current_c5
    
    return best_solution, best_c5

            break
            
        prob_count += 1
        current_h = h_opt
        
        # Evaluate true correlations
        corr_full = compute_corr_fft(current_h)
        max_corr = np.max(corr_full)
        current_c5 = dx * max_corr
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Find violating lags
        violating_lags = find_violating_lags(current_h, T_opt, lags)
        if len(violating_lags) > 0:
            # Add most violated lags
            lags.extend(violating_lags[:min(5, len(violating_lags))])  # Limit additions
            lags = list(set(lags))  # Remove duplicates
            lags = [l for l in lags if -n_points+1 <= l <= n_points-1]
        else:
            # No violations, break out of loop
            break
            
    
    # 2. Deterministic rounding and polishing
    if time.time() - start_time < 0.9 * budget_s:
        # 2a. Rounding
        best_binary = deterministic_rounding(best_h, n_points)
        binary_c5 = evaluate_c5(best_binary)
        
        if binary_c5 < best_c5:
            best_h = best_binary
            best_c5 = binary_c5
            
        # 2b. Critical lag polishing
        polished_h, polished_c5 = critical_lag_polish(best_binary, 5000)
        
        if polished_c5 < best_c5:
            best_h = polished_h
            best_c5 = polished_c5
    
    # Final check 
    return (best_h.tolist(), float(best_c5), n_points)
