import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_correlation(h):
    """Compute correlation using FFT."""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5_exact(h):
    """Compute exact C5 using np.correlate."""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(60):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max(h, alpha):
    """Smooth approximation to max using log-sum-exp."""
    corr = compute_correlation(h)
    corr = corr * dx  # Scale by dx
    exp_corr = np.exp(alpha * corr)
    # Stabilization to prevent overflow
    max_corr = np.max(corr)
    exp_corr_stable = np.exp(alpha * (corr - max_corr))
    return (max_corr + np.log(np.sum(exp_corr_stable)) / alpha) * dx

def smooth_grad(h, alpha):
    """Analytic gradient of smooth max."""
    corr = compute_correlation(h)
    corr = corr * dx  # Scale by dx
    exp_corr = np.exp(alpha * corr)
    # Stabilization to prevent overflow
    max_corr = np.max(corr)
    exp_corr_stable = np.exp(alpha * (corr - max_corr))
    weights = exp_corr_stable / np.sum(exp_corr_stable)
    
    # Compute gradient
    grad = np.zeros_like(h)
    L = len(h)
    
    for k in range(-L+1, L):  # k from -(n-1) to (n-1)
        if k >= 0:
            shifted_h = np.concatenate([np.zeros(k), h[:-k]])
        else:
            shifted_h = np.concatenate([h[-k:], np.zeros(-k)])
        # Weighted sum: (1 - h_{i+k} - h_{i-k})
        if k == 0:
            grad += weights * (1 - 2 * h)
        else:
            # h_{i-k} 
            if k > 0:
                shifted_h_neg = np.concatenate([np.zeros(k), h[:-k]])
            else:
                shifted_h_neg = np.concatenate([h[-k:], np.zeros(-k)])
            grad += weights * (1 - shifted_h - shifted_h_neg)
    
    # Apply scaling
    grad *= dx
    return grad

def compute_alpha_sequence():
    """Return continuation sequence of alpha values."""
    return [1, 5, 20, 80, 300]

def solve_level(h0, alpha, max_iter=200):
    """Solve one level of the continuation."""
    def fun(h):
        return smooth_max(h, alpha)
    
    def jac(h):
        return smooth_grad(h, alpha)
    
    # Define constraints
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum, 'jac': lambda h: np.ones_like(h)}
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    res = minimize(fun, h0, method='SLSQP', jac=jac, bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-9, 'gtol': 1e-9})
    
    # Project to ensure sum constraint is exact
    h_result = project_to_sum_box(res.x, target_sum)
    return h_result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Generate diverse initializations
    initial_families = []
    
    # 1. Repeated legacy (from initial_h_values, stretched to 240)
    h_legacy = np.tile(initial_h_values, n_points // len(initial_h_values))
    h_legacy = project_to_sum_box(h_legacy, target_sum)
    initial_families.append(h_legacy.copy())
    
    # 2. Pure sinusoid
    x = np.linspace(0, 2, n_points)
    phase = np.random.rand() * 2 * np.pi
    h_sin = 0.5 + 0.5 * np.sin(np.pi * x + phase)
    h_sin = project_to_sum_box(h_sin, target_sum)
    initial_families.append(h_sin.copy())
    
    # 3. Amplitude-tuned sinusoid
    amp = np.random.rand() * 0.5
    h_sin_amp = 0.5 + amp * np.sin(np.pi * x)
    h_sin_amp = project_to_sum_box(h_sin_amp, target_sum)
    initial_families.append(h_sin_amp.copy())
    
    # 4. Triangular wave
    x = np.linspace(0, 2, n_points)
    h_tri = np.where(x <= 1, 2 * x, 2 - 2 * x)
    h_tri = project_to_sum_box(h_tri, target_sum)
    initial_families.append(h_tri.copy())
    
    # 5. Random uniform (projected)
    h_rand = np.random.rand(n_points)
    h_rand = project_to_sum_box(h_rand, target_sum)
    initial_families.append(h_rand.copy())
    
    # 6. Hybrid (sinusoid + random)
    h_hybrid = 0.7 * h_sin + 0.3 * h_rand
    h_hybrid = project_to_sum_box(h_hybrid, target_sum)
    initial_families.append(h_hybrid.copy())
    
    best_c5 = float('inf')
    best_h = None
    
    alpha_sequence = compute_alpha_sequence()
    
    # Iterate through initial starts
    for i, h0 in enumerate(initial_families):
        if time.time() - start_time > budget_s - 10:
            break
            
        current_h = h0.copy()
        for alpha in alpha_sequence:
            if time.time() - start_time > budget_s - 10:
                break
            current_h = solve_level(current_h, alpha)
            
            # Occasionally perturb solution for better exploration
            if alpha == alpha_sequence[2]:
                noise = np.random.normal(0, 0.01, n_points)
                current_h = project_to_sum_box(current_h + noise, target_sum)
        
        c5 = evaluate_c5_exact(current_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = current_h.copy()
    
    # Final polishing with discrete search 
    if best_h is not None:
        current_h = best_h.copy()
        current_c5 = evaluate_c5_exact(current_h)
        max_swaps = 3000
        no_improvement_count = 0
        
        for _ in range(max_swaps):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Try a random swap
            i, j = np.random.choice(n_points, 2, replace=False)
            temp_h = current_h.copy()
            temp_h[i], temp_h[j] = temp_h[j], temp_h[i]
            temp_h = project_to_sum_box(temp_h, target_sum)
            
            temp_c5 = evaluate_c5_exact(temp_h)
            if temp_c5 < current_c5:
                current_h = temp_h
                current_c5 = temp_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                if no_improvement_count > 200:
                    break
    
    # Final evaluation
    final_c5 = evaluate_c5_exact(current_h)
    return (current_h.tolist(), float(final_c5), n_points)
