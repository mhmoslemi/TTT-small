def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""

    b = 1 - h
    L = 2 * len(h) - 1

    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):

    L = 2 * len(h) - 1

    return corr * dx

            shift = shift_idx - (N - 1)

                g[i] -= 2 * weight * h[i]

                # h[i+shift] contribution (i from 0 to N-shift-1)
                if i + shift < N:
                    g[i] -= weight * h[i + shift]

                # h[i-shift] = h[i+t] contribution (i from 0 to N-t-1)
                if i + t < N:
                    g[i] -= weight * h[i + t]
    
    return g
