def max_corr(h):
    """Compute maximum correlation value"""
    corr = correlate(h, 1-h, mode="full")
    return np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 value"""
    corr = correlate(h, 1-h, mode="full")
    return 2.0/len(h) * np.max(corr)
