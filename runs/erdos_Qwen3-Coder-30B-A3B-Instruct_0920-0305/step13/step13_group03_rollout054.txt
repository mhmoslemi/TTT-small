import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cvxsum, norm
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    target_C5 = 0.3807
    target_M = int(target_C5 / dx)  # target max overlap in integer units
    
    # Initial vector (from provided values)
    h0 = np.array(initial_h_values)
    h0 = np.clip(h0, 0, 1)
    # Project to correct sum
    current_sum = np.sum(h0)
    h0 = h0 * (n_points / 2) / current_sum
    
    # Compute initial correlation to get baseline
    corr0 = correlate(h0, 1 - h0, mode='full')
    initial_max = np.max(corr0)
    
    # Convex-Concave Procedure with cutting plane
    def compute_convex_surrogate(h, lags):
        """Compute convex surrogate for all given lags at point h"""
        n = len(h)
        result = []
        for ell in lags:
            if ell == 0:
                # Direct sum term + convex part for lag 0
                term = np.sum(h) + 0.25 * np.sum((h - np.roll(h, ell))**2)
            else:
                term = np.sum(h) + 0.25 * np.sum((h - np.roll(h, ell))**2) - 0.5 * np.sum((h + np.roll(h, ell)) * (h + np.roll(h, ell)))
            result.append(term)
        return np.array(result)
    
    def solve_qp_with_constraints(h_current, active_lags, max_iter=100):
        """Solve the convex QP with cutting plane approach"""
        n = len(h_current)
        # Variables
        h = Variable(n)
        T = Variable()
        
        # Constraints
        constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2, T >= 0]
        
        # Add convex constraints for active lags (this is simplified)
        for ell in active_lags:
            if ell == 0:
                # Simplified surrogate for lag 0 which is sum(h)
                constraint_expr = cp.sum(h) + 0.25 * cp.sum((h - cp.roll(h, ell))**2)
            else:
                # More full formulation
                constraint_expr = (cp.sum(h) + 
                                 0.25 * cp.sum((h - cp.roll(h, ell))**2) - 
                                 0.5 * cp.sum((h + cp.roll(h, ell)) * (h + cp.roll(h, ell))))
            constraints.append(constraint_expr <= T)
            
        # Objective
        objective = Minimize(T)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=max_iter)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                return h.value, T.value
        except:
            pass
        return h_current, np.max(compute_convex_surrogate(h_current, active_lags))
    
    # Initialize active lags set (starting with key lags)
    initial_lags = [0, -24, -48, -72, -96, 24, 48, 72, 96] # n=120, so 1/5 = 24 etc.

    # Start with the initial solution
    h = h0.copy()
    best_h = h0.copy()
    best_C5 = initial_max * dx
    
    # Main CCP loop
    max_iter = 1000
    t1 = start_time + 0.4 * budget_s
    iter_count = 0
    active_lags = initial_lags.copy()
    
    while time.time() < t1 and iter_count < max_iter:
        iter_count += 1
        # Solve current relaxed problem with active lags
        h_new, T_new = solve_qp_with_constraints(h, active_lags)
        if T_new < 0:
            # numerical issue handling
            h_new = h.copy()
        else:
            h = h_new.copy()
        
        # Evaluate true correlations using correlation
        corr = correlate(h, 1 - h, mode='full')
        max_correlation = np.max(corr)
        
        # Find violating lags
        violators = []
        for ell in range(-n_points+1, n_points):
            if ell in active_lags:
                continue  # already in active set
            # compute the true correlation at this lag
            try:
                l = ell if ell >= 0 else n_points + ell
                r = ell if ell <= 0 else 0
                val = np.sum(h[r:r+n_points] * (1-h[l:l+n_points]))
                if val > T_new + 1e-6:  # use a tolerance
                    violators.append(ell)
            except:
                continue
        
        if len(violators) == 0:
            # Check if we are good enough
            if max_correlation <= target_M:
                break
                
        # Add worst violator to active set
        if len(violators) > 0:
            # Select the most violated
            # In practice, we add the max violator
            worst_violator = violators[0]
            active_lags.append(worst_violator)
            # remove inactive max in long lists
            if len(active_lags) > 100:
                active_lags = [0, 24, 48, 72, 96, -24, -48, -72, -96] + violators[:5]
        
        # Evaluate actual C5
        current_C5 = max_correlation * dx
        if current_C5 < best_C5:
            best_h = h.copy()
            best_C5 = current_C5
    
    # Now rounding and polishing stage
    t2 = time.time() + 0.3 * budget_s
    best_h_rounded = None
    best_rounded_C5 = float('inf')
    
    # Try multiple random rounds
    for _ in range(30):
        if time.time() >= t2:
            break
        # deterministic rounding (take top n/2 indices)
        indices_sorted = np.argsort(h)[::-1]
        b = np.zeros(n_points)
        b[indices_sorted[:n_points//2]] = 1
        # Project to correct sum if needed
        actual_sum = np.sum(b)
        if actual_sum != n_points / 2:
            # if we are not exactly correct, adjust
            to_flip_indices = []
            if actual_sum < n_points/2:
                # flip 0s to 1s
                for i in range(n_points):
                    if b[i] == 0 and i not in indices_sorted[:n_points//2]:
                        to_flip_indices.append(i)
                        if len(to_flip_indices) == (n_points//2 - actual_sum):
                            break
            else:
                # flip 1s to 0s
                for i in range(n_points):
                    if b[i] == 1 and i in indices_sorted[:n_points//2]:
                        to_flip_indices.append(i)
                        if len(to_flip_indices) == (actual_sum - n_points/2):
                            break
            for i in to_flip_indices:
                b[i] = 1 if b[i] == 0 else 0
        
        # evaluate
        corr = correlate(b, 1 - b, mode='full')
        max_overlap = np.max(corr)
        C5 = max_overlap * dx
        
        if C5 < best_rounded_C5:
            best_rounded_C5 = C5
            best_h_rounded = b.copy()
    
    # Critical lag polishing using BFS
    h_polished = best_h_rounded.copy() if best_h_rounded is not None else best_h.copy()
    corr = correlate(h_polished, 1 - h_polished, mode='full')
    best_polished_C5 = np.max(corr) * dx
    best_polished_h = h_polished.copy()
    
    t3 = time.time() + 0.2 * budget_s
    while time.time() < t3 and best_polished_C5 > target_C5:
        max_ell = np.argmax(corr)
        max_val = corr[max_ell]
        if max_val <= target_M:
            break
        
        # Try efficient local search
        # getting shift corresponding to max_ell index
        if max_ell < n_points:
            ell = max_ell  # for positive lag
        else:
            ell = max_ell - n_points  # getting negative lag
            
        # Find mismatched 1,0 pairs for this lag
        B_indices = []
        for i in range(n_points):
            if (ell > 0 and i + ell < n_points and h_polished[i] == 1 and h_polished[i + ell] == 0) or \
               (ell < 0 and i + ell >= 0 and h_polished[i] == 1 and h_polished[i + ell] == 0):
                B_indices.append(i)
        
        if not B_indices:
            break
            
        # pick random 1 to flip to 0 to reduce this alignment
        i_flip = np.random.choice(B_indices)
        # find good 0's to flip to 1
        Z_indices = []
        for j in range(n_points):
            if h_polished[j] == 0:
                check_ell = ell
                if check_ell > 0 and j + check_ell < n_points:
                    if h_polished[j + check_ell] == 1:
                        Z_indices.append(j)
                elif check_ell < 0 and j + check_ell >= 0:
                    if h_polished[j + check_ell] == 1:
                        Z_indices.append(j)
                        
        if not Z_indices:
            # no good candidate, just find any 0, and flip to 1
            flip_j = np.random.choice([i for i in range(n_points) if h_polished[i] == 0])
        else:
            flip_j = np.random.choice(Z_indices)
            
        # Actually flip 
        h_polished[i_flip], h_polished[flip_j] = h_polished[flip_j], h_polished[i_flip]
        
        corr = correlate(h_polished, 1 - h_polished, mode='full')
        new_max = np.max(corr)
        new_C5 = new_max * dx
        
        if new_C5 < best_polished_C5:
            best_polished_C5 = new_C5
            best_polished_h = h_polished.copy()
            # Early exit?
            if best_polished_C5 <= target_C5:
                break
            
    # Final result using best solution found
    final_h = best_polished_h if best_polished_C5 <= target_C5 else best_h_rounded if best_h_rounded is not None else best_h
    final_corr = correlate(final_h, 1 - final_h, mode='full')
    final_C5 = np.max(final_corr) * dx
    
    return (final_h.tolist(), float(final_C5), n_points)
