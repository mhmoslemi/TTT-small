# Global variables
n_points = 120
dx = 2.0 / n_points
target_M = int(np.floor(0.3807 / dx))  # = 22

def max_corr(h):
    """Compute maximum overlap for given h"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    return dx * max_corr(h)

def critical_lag_search(h, max_swaps=30000, time_limit=None):
    """Greedy local search targeting the worst lag"""
    start_time = time.time()
    h = h.copy()
    I = correlate(h, 1 - h, mode='full')
    M = np.max(I)
    
    # Map lags to indices where h[i]=1 and h[i+lag]=0
    lag_to_indices = {}
    n = len(h)
    
    def update_lag_map():
        """Update the lag-to-indices mapping"""
        lag_to_indices.clear()
        for lag in range(-n+1, n):
            indices = []
            if lag >= 0:
                # h[i] = 1, h[i+lag] = 0
                for i in range(n - lag):
                    if h[i] == 1 and h[i + lag] == 0:
                        indices.append(i)
            else:
                # h[i] = 1, h[i+lag] = 0 where lag < 0 means i+lag >= 0
                for i in range(-lag):
                    if h[i] == 1 and h[i + lag] == 0:
                        indices.append(i)
            lag_to_indices[lag] = indices
    
    update_lag_map()
    
    swap_count = 0
    while M > target_M and swap_count < max_swaps:
        if time_limit and time.time() > time_limit:
            break
            
        # Find worst lag
        worst_lag = np.argmax(I)
        mismatches = lag_to_indices[worst_lag]
        
        if not mismatches:
            break
            
        # Pick a random mismatch position
        i = np.random.choice(mismatches)
        
        # Find suitable j to swap with
        candidates = []
        for j in range(n):
            if h[j] == 0:
                # Make sure swapping i and j doesn't create new mismatches
                if worst_lag >= 0:
                    # Check j + worst_lag
                    if j + worst_lag >= n or h[j + worst_lag] == 1:
                        candidates.append(j)
                else:
                    # Check i + worst_lag
                    if i + worst_lag < 0 or h[i + worst_lag] == 1:
                        candidates.append(j)
        
        if candidates:
            j = np.random.choice(candidates)
        else:
            # Just pick a random zero
            zero_indices = np.where(h == 0)[0]
            if len(zero_indices) > 0:
                j = np.random.choice(zero_indices)
            else:
                break
        
        # Perform swap
        h[i], h[j] = h[j], h[i]
        
        # Update I and M more efficiently
        # This is a simplified update for demonstration
        I = correlate(h, 1 - h, mode='full')
        M = np.max(I)
        
        # Update lag map
        update_lag_map()
        
        swap_count += 1
    
    return h, M

def round_to_binary(h_cont, target_sum=60):
    """Round continuous vector to binary with target sum"""
    # Round to binary
    h = (h_cont >= 0.5).astype(int)
    
    current_sum = np.sum(h)
    
    if current_sum == target_sum:
        return h
    
    if current_sum < target_sum:
        # Need to flip more zeros to ones
        diff = target_sum - current_sum
        # Flip the ones that are closest to 0.5
        diffs = np.abs(h_cont - 0.5)
        flip_indices = np.argsort(diffs)[:diff]
        for idx in flip_indices:
            if h[idx] == 0:
                h[idx] = 1
    else:
        # Need to flip some ones to zeros
        diff = current_sum - target_sum
        # Flip the ones that are closest to 0.5
        diffs = np.abs(h_cont - 0.5)
        flip_indices = np.argsort(diffs)[-diff:]
        for idx in flip_indices:
            if h[idx] == 1:
                h[idx] = 0
    
    return h
