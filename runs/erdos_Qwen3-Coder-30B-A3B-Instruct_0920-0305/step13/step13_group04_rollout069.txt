import numpy as np
from scipy.optimize import differential_evolution
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_interval_overlap(a1, b1, a2, b2):
    """Compute overlap length between two intervals [a1, b1] and [a2, b2]"""
    return max(0, min(b1, b2) - max(a1, a2))

def compute_worst_overlap(intervals):
    """Compute the worst-case intersection measure for given intervals"""
    if not intervals:
        return 0
    
    # Generate all candidate shift values
    endpoints = []
    for a, b in intervals:
        endpoints.extend([a, b])
    
    # All differences between endpoints
    candidates = set()
    for i in range(len(endpoints)):
        for j in range(len(endpoints)):
            candidates.add(endpoints[i] - endpoints[j])
    
    # Include boundaries
    candidates.add(-2.0)
    candidates.add(2.0)
    
    min_overlap = float('inf')
    
    for k in candidates:
        overlap = 0.0
        for a1, b1 in intervals:
            # For interval [a1, b1], shifted by k it becomes [a1+k, b1+k]
            for a2, b2 in intervals:
                # Compute overlap between [a1, b1] and [a2+k, b2+k]
                overlap += compute_interval_overlap(a1, b1, a2 + k, b2 + k)
        min_overlap = min(min_overlap, overlap)
    
    return min_overlap

def intervals_to_h(intervals, n_points, dx):
    """Convert intervals to discrete step function h"""
    h = np.zeros(n_points)
    for i in range(n_points):
        cell_start = i * dx
        cell_end = (i + 1) * dx
        coverage = 0.0
        for a, b in intervals:
            coverage += compute_interval_overlap(a, b, cell_start, cell_end)
        h[i] = coverage / dx
    
    # Ensure sum constraint
    total = np.sum(h) * dx
    if abs(total - 1.0) > 1e-10:
        h = h * (1.0 / total)
    
    return h

def objective(v):
    """Objective function to minimize"""
    K = len(v) // 2
    raw_lengths = v[:K]
    raw_gaps = v[K:]
    
    # Normalize
    lengths = raw_lengths / np.sum(raw_lengths)
    gaps = raw_gaps / np.sum(raw_gaps)
    
    # Compute interval positions
    intervals = []
    pos = 0
    for i in range(K):
        a = pos
        b = a + lengths[i]
        intervals.append((a, b))
        pos = b + gaps[i]
    
    # Compute worst overlap
    min_overlap = compute_worst_overlap(intervals)
    c5 = 1.0 - min_overlap
    
    return c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # For K = 5 intervals
    K = 5
    n_vars = 2 * K
    
    # Define bounds for each variable (positive raw parameters)
    bounds = [(1e-6, 1.0) for _ in range(n_vars)]
    
    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,
        maxiter=int(1000 * (budget_s / 600)),  # Scale iteration count by budget
        popsize=15,
        seed=seed,
        disp=False,
        tol=1e-8
    )
    
    # Extract optimal parameters
    optimal_params = result.x
    
    # Reconstruct intervals and h
    raw_lengths = optimal_params[:K]
    raw_gaps = optimal_params[K:]
    
    # Normalize
    lengths = raw_lengths / np.sum(raw_lengths)
    gaps = raw_gaps / np.sum(raw_gaps)
    
    # Compute interval positions
    intervals = []
    pos = 0
    for i in range(K):
        a = pos
        b = a + lengths[i]
        intervals.append((a, b))
        pos = b + gaps[i]
    
    # Convert to h
    h = intervals_to_h(intervals, n_points, dx)
    
    # Compute final C5
    c5_bound = objective(optimal_params)
    
    # If we've run out of time, return what we have
    return (h.tolist(), float(c5_bound), n_points)
