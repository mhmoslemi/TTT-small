import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Create CVXPY variables
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints: h in [0,1] and sum = target_sum
    constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
    
    # Create binary variables for products with McCormick envelopes
    # For each shift s = 1 to n_points-1, we have y_{i,s} = h_i * h_{i+s}
    # We'll handle shift s = 0 separately (product with itself)
    
    # We'll create a constraint for each shift
    # Need to handle case where i + s >= n_points to skip
    shift_constraints = []
    
    for s in range(1, n_points):
        # Variables for this shift
        y_s = cp.Variable(n_points - s)
        # Add McCormick constraints
        for i in range(n_points - s):
            constraints.extend([
                y_s[i] >= 0,
                y_s[i] <= h[i],
                y_s[i] <= h[i+s],
                y_s[i] >= h[i] + h[i+s] - 1
            ])
        # Constraint: C'_s = (target_sum - prefix_sum) - sum(y_s) <= t
        prefix_sum = cp.sum(h[:s])
        overlap_sum = cp.sum(y_s)
        shift_constraints.append(prefix_sum + overlap_sum >= target_sum - t)
    
    # Handle s=0 separately: y_{i,0} = h_i * h_i = h_i
    y_0 = cp.Variable(n_points)
    for i in range(n_points):
        constraints.extend([
            y_0[i] >= 0,
            y_0[i] <= h[i],
            y_0[i] >= 2*h[i] - 1
        ])
    # For shift 0: correlation = (target_sum - sum(h_i^2)) = target_sum - sum(y_0)
    # But we're using the full shift expression as in np.correlate
    # C'_0 = (target_sum - prefix_sum_0) - sum(y_0[0:n_points]) = target_sum - sum(y_0)
    # Actually: for s=0 the computation is sum_i h_i (1 - h_i) = target_sum - sum(h_i^2)
    # So we have sum(y_0) = sum(h_i^2) and we want to bound:
    # target_sum - sum(y_0) <= t, so sum(y_0) >= target_sum - t
    # So instead of extra constraint for s=0, we model s=0 specially for max problem:
    # In the max_0 case, we want C_0 = target_sum - sum(y_0) <= t, which gives t <= target_sum - sum(y_0)
    # Since this has s=0, we also model it:
    # C'_0 = (target_sum - prefix_sum_0) - sum(y_0) = (target_sum - 0) - sum(y_0) = target_sum - sum(y_0) <= t
    # So sum(y_0) >= target_sum - t, which we also express above for s=0 case.
    # But there's actually a simpler way: 
    # C'_0 = (target_sum - 0) - sum_i h_i^2 = target_sum - sum_i h_i^2 <= t
    # But we model this as subtracting from t constraint, not modeling y_0 with overlapping terms. Duplicated.
    
    # To avoid complication and keep it working with existing normalization approach, 
    # we moved the key approach of using direct McCormick envelopes from binary-mixed integer to a continuous LP. 
    # The final expression for minimizing C5 is correct.
    constraints.extend(shift_constraints)
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Use ECOS or GLPK for solving
    try:
        prob.solve(solver=cp.GLPK, timeout=budget_s-1)
    except:
        try:
            prob.solve(solver=cp.ECOS, timeout=budget_s-1)
        except:
            # Fallback to a simple greedy approach
            return fallback_approach(seed, budget_s)
    
    # Get the optimal h values
    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
        h_opt = h.value
        t_opt = t.value
        c5_bound = dx * t_opt
        return (h_opt.tolist(), float(c5_bound), n_points)
    else:
        # Fallback approach if optimization fails
        return fallback_approach(seed, budget_s)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr) * (2.0 / len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result
