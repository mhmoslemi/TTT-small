import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_and_grad(corr, alpha):
    """Compute smooth max and gradient"""
    # Stabilize log-sum-exp
    max_corr = np.max(corr)
    shifted_corr = corr - max_corr
    exp_corr = np.exp(alpha * shifted_corr)
    sum_exp = np.sum(exp_corr)
    
    if sum_exp == 0:
        weights = np.zeros_like(corr)
    else:
        weights = exp_corr / sum_exp
    
    return np.log(sum_exp) / alpha, weights

def smooth_objective_and_grad(h, alpha, dx):
    """Compute smooth objective and its gradient for the continuous problem"""
    corr = compute_corr_fft(h)
    obj, weights = smooth_max_and_grad(corr, alpha)
    
    # Gradient computation
    grad = np.zeros_like(h)
    n = len(h)
    for k in range(-n+1, n):
        if k >= 0:
            shift_forward = h[k:]
            shift_backward = np.concatenate([np.zeros(k), h[:n-k]])
        else:
            shift_forward = np.concatenate([np.zeros(-k), h[:-k]])
            shift_backward = h[:-k]
        
        # Weighted average of (1 - h_{k} - h_{-k})
        weighted_grad = weights[k+n-1] * (1 - shift_forward - shift_backward)
        if k >= 0:
            grad[k:] += weighted_grad[:-k] if k > 0 else weighted_grad[:]
        else:
            grad[:n+k] += weighted_grad[-k:] if k < 0 else weighted_grad[:]
            
    return dx * obj, dx * grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 240
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate initial guesses
    initial_guesses = []
    
    # 1. Replicated initial_h_values

    if len(h_replicated) < n_points:
        h_replicated = np.pad(h_replicated, (0, n_points - len(h_replicated)), 'constant')
    h_replicated = project_to_sum_box(h_replicated, target_sum)
    initial_guesses.append(h_replicated.copy())
    
    # 2. Sinusoidal patterns
    x = np.linspace(0, 2, n_points, endpoint=False) + dx/2
    for _ in range(3):
        phase = np.random.rand() * 2 * np.pi
        amplitude = np.random.rand() * 0.5
        h_sin = 0.5 + amplitude * np.sin(np.pi * x + phase)
        h_sin = project_to_sum_box(h_sin, target_sum)
        initial_guesses.append(h_sin)
    
    # 3. Triangular wave
    h_tri = np.zeros(n_points)
    midpoint = n_points // 2
    h_tri[:midpoint] = np.linspace(0, 1, midpoint)
    h_tri[midpoint:] = np.linspace(1, 0, n_points - midpoint)
    h_tri = project_to_sum_box(h_tri, target_sum)
    initial_guesses.append(h_tri)
    
    # 4. Random uniform
    h_rand = np.random.rand(n_points)
    h_rand = project_to_sum_box(h_rand, target_sum)
    initial_guesses.append(h_rand)
    
    # 5. Hybrid (combination of sinusoid and random)
    h_hybrid = 0.7 * (0.5 + 0.3 * np.sin(np.pi * x + np.random.rand() * 2 * np.pi)) + \
               0.3 * np.random.rand(n_points)
    h_hybrid = project_to_sum_box(h_hybrid, target_sum)
    initial_guesses.append(h_hybrid)
    
    best_c5 = float('inf')
    best_h = None
    
    # Compute all correlation levels
    alpha_levels = [1, 5, 20, 80, 300]
    
    # Try each initial guess
    for i, h0 in enumerate(initial_guesses):
        current_h = h0.copy()
        current_c5 = c5_from_corr(compute_corr_fft(current_h), n_points)
        
        for alpha in alpha_levels:
            # Optimization with trust-constr
            def obj_and_grad(h):
                return smooth_objective_and_grad(h, alpha, dx)
            
            # Constraints
            def sum_constraint(h):
                return np.sum(h) - target_sum
            
            cons = {'type': 'eq', 'fun': sum_constraint, 'jac': lambda h: np.ones_like(h)}
            
            # Bounds
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Optimize
            try:
                res = minimize(obj_and_grad, current_h, method='trust-constr', 
                              bounds=bounds, constraints=cons, jac=True, 
                              options={'gtol': 1e-9, 'maxiter': 200})
                if res.success:
                    current_h = res.x
                    # Project to ensure sum constraint is satisfied exactly
                    current_h = project_to_sum_box(current_h, target_sum)
                else:
                    # Use the previous solution if optimization fails
                    pass
            except:
                # Use previous solution if optimization fails
                pass
            
            # Evaluate
            corr = compute_corr_fft(current_h)
            current_c5 = c5_from_corr(corr, n_points)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        
        # Early termination check
        if time.time() - start_time > 0.9 * budget_s:
            break

    # Local search refinement
    if best_h is not None:
        current_h = best_h.copy()
        current_c5 = c5_from_corr(compute_corr_fft(current_h), n_points)
        
        max_swaps = 3000
        for iter_num in range(max_swaps):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Try a random swap
            i, j = np.random.choice(n_points, 2, replace=False)
            temp_h = current_h.copy()
            temp_h[i], temp_h[j] = temp_h[j], temp_h[i]
            temp_h = project_to_sum_box(temp_h, target_sum)
            
            temp_corr = compute_corr_fft(temp_h)
            temp_c5 = c5_from_corr(temp_corr, n_points)
            
            if temp_c5 <= current_c5:
                current_h = temp_h
                current_c5 = temp_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
        
        # Final exact evaluation
        final_corr = compute_corr_fft(best_h)
        final_c5 = c5_from_corr(final_corr, n_points)
        
        return (best_h.tolist(), float(final_c5), n_points)
    
    # Fallback: return initial solution
    h0 = np.array(initial_h_values)
    h0 = project_to_sum_box(h0, target_sum)
    corr = compute_corr_fft(h0)
    c5 = c5_from_corr(corr, n_points)
    
    return (h0.tolist(), float(c5), n_points)
