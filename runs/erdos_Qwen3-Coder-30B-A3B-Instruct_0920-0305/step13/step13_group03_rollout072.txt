import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

# Global variables
n_points = 120
dx = 2.0 / n_points
target_M = int(np.floor(0.3807 / dx))  # 22
ℓ_max = n_points - target_M  # 98

def max_corr(h):
    """Compute maximum correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr)

def compute_lag_indices(h):
    """Compute lag indices for critical lag search"""
    I = correlate(h, 1 - h, mode='full')
    max_lag = len(h) - 1
    lag_to_indices = {}
    for lag in range(-max_lag, max_lag + 1):
        if lag >= 0:
            indices = np.where((h[:-lag] == 1) & (h[lag:] == 0))[0]
        else:
            indices = np.where((h[-lag:] == 1) & (h[:-lag] == 0))[0]
        lag_to_indices[lag] = indices
    return I, lag_to_indices

def greedy_critical_lag_search(h, max_iter=50000):
    """Greedy local search on critical lag"""
    h = h.copy()
    current_I, lag_to_indices = compute_lag_indices(h)
    current_M = np.max(current_I)
    
    iter_count = 0
    while iter_count < max_iter and current_M > target_M:
        # Find worst lag
        worst_lag = np.argmax(current_I)
        mismatches = lag_to_indices[worst_lag - (n_points - 1)]
        
        if len(mismatches) == 0:
            break
            
        # Pick a random mismatch
        i = np.random.choice(mismatches)
        
        # Find candidate positions to swap with
        candidates = []
        if worst_lag >= 0:
            # Check positions that don't create new mismatches
            for j in range(n_points):
                if h[j] == 0 and (j + worst_lag >= n_points or h[j + worst_lag] == 1):
                    candidates.append(j)
        else:
            # Check positions that don't create new mismatches
            for j in range(n_points):
                if h[j] == 0 and (j - worst_lag < 0 or h[j - worst_lag] == 1):
                    candidates.append(j)
        
        if len(candidates) > 0:
            j = np.random.choice(candidates)
        else:
            # Fallback: find any zero position
            zero_indices = np.where(h == 0)[0]
            if len(zero_indices) > 0:
                j = np.random.choice(zero_indices)
            else:
                break
                
        # Perform the swap
        h[i], h[j] = h[j], h[i]
        
        # Recompute
        current_I, lag_to_indices = compute_lag_indices(h)
        current_M = np.max(current_I)
        iter_count += 1
        
    return h, current_M

    for lag in range(-ℓ_max, ℓ_max + 1):

            for i in range(n_points + lag):

                    total += h_vars[i] - w_vars[(i, lag)]
                else:
                    total += h_fixed[i] - w_vars[(i, lag)]

    prob = cp.Problem(cp.Minimize(M), constraints)

            return h_opt, prob.value

        pass
    
    return None, None
