import numpy as np
from scipy.optimize import differential_evolution
import time

def compute_worst_overlap(intervals):
    """
    Compute the worst-case overlap I(k) for given intervals.
    intervals: list of (start, end) tuples
    """
    if not intervals:
        return 0.0
    
    # Generate all candidate k values from interval endpoint differences
    endpoints = []
    for start, end in intervals:
        endpoints.extend([start, end])
    
    # Include boundaries
    candidates = set([0.0])
    
    # Add all differences between endpoints
    for i in range(len(endpoints)):
        for j in range(len(endpoints)):
            diff = endpoints[i] - endpoints[j]
            if -2 <= diff <= 2:
                candidates.add(diff)
    
    candidates = sorted(list(candidates))
    
    min_overlap = float('inf')
    
    # For each candidate k, compute overlap I(k)
    for k in candidates:
        overlap = 0.0
        for start1, end1 in intervals:
            for start2, end2 in intervals:
                # Compute overlap of [start1, end1] and [start2+k, end2+k]
                overlap_start = max(start1, start2 + k)
                overlap_end = min(end1, end2 + k)
                if overlap_start < overlap_end:
                    overlap += (overlap_end - overlap_start)
        min_overlap = min(min_overlap, overlap)
    
    return min_overlap

def interval_objective(params, K, n_points, dx):
    """
    Objective function to minimize maximum overlap.
    params: flattened array of [lengths, gaps]
    """
    # Extract lengths and gaps
    lengths = params[:K]
    gaps = params[K:]
    
    # Normalize to ensure sum = 1
    lengths = lengths / np.sum(lengths) * 1.0
    gaps = gaps / np.sum(gaps) * 1.0
    
    # Compute interval positions
    intervals = []
    pos = gaps[0]  # Start position after initial gap
    
    for i in range(K):
        start = pos
        end = start + lengths[i]
        intervals.append((start, end))
        pos = end + gaps[i+1]  # Gap after this interval
    
    # Compute worst overlap
    min_overlap = compute_worst_overlap(intervals)
    
    # C5 = 1 - min_overlap (since integral = 1 - overlap measure)
    return 1.0 - min_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set parameters
    K = 6  # Number of intervals
    n_points = 256
    dx = 2.0 / n_points
    total_vars = 2 * K + 1  # lengths + gaps (including gap before and after)
    
    # Define bounds for differential evolution
    bounds = [(1e-6, 1.0) for _ in range(total_vars)]
    
    # Define objective function
    def objective(params):
        return interval_objective(params, K, n_points, dx)
    
    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,
        maxiter=int(1000 * budget_s / 600),  # Scale iterations with budget
        popsize=max(15, K * 2),
        seed=seed,
        strategy='best1bin',
        tol=1e-8,
        recombination=0.7,
        disp=False
    )
    
    # Extract best solution
    best_params = result.x
    best_value = result.fun
    
    # Convert back to intervals
    lengths = best_params[:K]
    gaps = best_params[K:]
    lengths = lengths / np.sum(lengths) * 1.0
    gaps = gaps / np.sum(gaps) * 1.0
    
    # Compute interval positions
    intervals = []
    pos = gaps[0]
    for i in range(K):
        start = pos
        end = start + lengths[i]
        intervals.append((start, end))
        pos = end + gaps[i+1]
    
    # Convert intervals to discrete h values
    h = np.zeros(n_points)
    for i in range(n_points):
        cell_start = i * dx
        cell_end = (i + 1) * dx
        covered_length = 0.0
        for start, end in intervals:
            overlap_start = max(start, cell_start)
            overlap_end = min(end, cell_end)
            if overlap_start < overlap_end:
                covered_length += (overlap_end - overlap_start)
        h[i] = covered_length / dx
    
    # Ensure sum constraint
    current_sum = np.sum(h) * dx
    if abs(current_sum - 1.0) > 1e-10:
        # Adjust slightly to meet sum constraint
        scale_factor = 1.0 / current_sum
        h *= scale_factor
    
    # Final evaluation
    # Use the eval function to compute actual C5 bound
    # Here we just return the computed bound
    final_bound = best_value
    
    return (h.tolist(), float(final_bound), n_points)
