import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def max_corr(h):
    """Compute maximum correlation value"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 value"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def greedy_critical_lag_improvement(h, target_M, max_iter=10000):
    """Improve solution using greedy critical-lag swaps"""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    
    # Precompute bad positions for each lag
    def get_bad_positions(ℓ):
        if ℓ >= 0:
            return np.where((h[:n-ℓ] == 1) & (h[ℓ:] == 0))[0]
        else:
            return np.where((h[-ℓ:] == 1) & (h[:n+ℓ] == 0))[0]
    
    # Precompute all bad positions
    bad_positions = {}
    for ℓ in range(-n+1, n):
        bad_positions[ℓ] = get_bad_positions(ℓ)
    
    corr = correlate(h, 1 - h, mode='full')
    M = int(np.max(corr))
    
    iter_count = 0
    while iter_count < max_iter and M > target_M:
        # Find worst lag
        max_idx = np.argmax(corr)
        # Convert from full correlation index to actual lag
        ℓ = max_idx - (n - 1)
        
        # Get bad positions for this lag
        bad = bad_positions[ℓ]
        if len(bad) == 0:
            break
            
        # Pick random bad position
        i = np.random.choice(bad)
        
        # Find good swap position
        candidates = []
        if ℓ >= 0:
            # Positive lag
            for j in range(n):
                if h[j] == 0:
                    # Check if this swap doesn't create new mismatch at lag ℓ
                    if j + ℓ >= n or h[j + ℓ] == 1:
                        candidates.append(j)
        else:
            # Negative lag
            for j in range(n):
                if h[j] == 0:
                    # Check if this swap doesn't create new mismatch at lag ℓ
                    if j + ℓ < 0 or h[j + ℓ] == 1:
                        candidates.append(j)
        
        if not candidates:
            # No good candidates, pick any 0 position
            zero_indices = np.where(h == 0)[0]
            if len(zero_indices) == 0:
                break
            j = np.random.choice(zero_indices)
        else:
            j = np.random.choice(candidates)
        
        # Perform swap
        h[i], h[j] = h[j], h[i]
        
        # Rebuild bad positions for affected lags
        for l in range(-n+1, n):
            bad_positions[l] = get_bad_positions(l)
        
        # Update correlation
        corr = correlate(h, 1 - h, mode='full')
        M = int(np.max(corr))
        iter_count += 1
    
    return h, M
