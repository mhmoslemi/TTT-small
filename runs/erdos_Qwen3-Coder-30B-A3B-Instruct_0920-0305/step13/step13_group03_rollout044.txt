import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
from math import floor

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_convex_subproblem(h_current, lags, n_points):
    """Solve the convex subproblem using CVXPY"""
    dx = 2.0 / n_points
    h = cp.Variable(n_points)
    T = cp.Variable()
    
    # Objective
    objective = cp.Minimize(T)
    
    # Constraints
    constraints = []
    
    # Box constraints
    constraints.extend([h >= 0, h <= 1])
    
    # Sum constraint
    constraints.append(cp.sum(h) == n_points / 2)
    
    # Convex constraints for each lag in the active set
    for ell in lags:
        # Compute indices for lag ell
        if ell > 0:
            start_idx = ell
            end_idx = n_points
            h_shifted = h[start_idx:end_idx]
            h_current_shifted = h_current[start_idx:end_idx]
            h_base = h[:n_points-ell]
            h_current_base = h_current[:n_points-ell]
        else:
            start_idx = 0
            end_idx = n_points + ell
            h_shifted = h[(-ell):]
            h_current_shifted = h_current[(-ell):]
            h_base = h[:n_points+ell]
            h_current_base = h_current[:n_points+ell]
            
        # Convex part: sum_i h_i + 0.25 * sum_i (h_i - h_{i+ell})^2
        convex_part = cp.sum(h) + 0.25 * cp.sum((h_base - h_shifted)**2)
        
        # Linear part: -0.5 * sum_i (h_current_i + h_current_{i+ell}) * (h_i + h_{i+ell})
        linear_part = -0.5 * cp.sum((h_current_base + h_current_shifted) * (h_base + h_shifted))
        
        # Add constraint
        constraints.append(convex_part + linear_part <= T)
    
    # Form and solve the problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value, T.value
    except:
        pass
    
    # Fallback to previous values if optimization fails
    return h_current, float('inf')

def find_violated_lags(h_opt, T_opt, n_points, eps=1e-6):
    """Find violated lags and return the most violated ones"""
    corr = compute_corr_fft(h_opt)
    max_corr = np.max(corr)
    if max_corr <= T_opt + eps:
        return []
    
    # Find all lags with violation
    violations = []
    for i in range(len(corr)):
        if corr[i] > T_opt + eps:
            violations.append(i - n_points + 1)
    
    # Return the most violated
    return violations[:5] # Top 5 most violated lags

def round_to_binary(h_cont, n_points):
    """Round continuous vector to binary while maintaining sum constraint"""
    # Sort indices by decreasing h_cont values
    sorted_indices = np.argsort(h_cont)[::-1]
    
    # Create binary vector
    h_binary = np.zeros(n_points)
    h_binary[sorted_indices[:n_points//2]] = 1
    
    # Project back to correct sum
    h_binary = project_to_sum_box(h_binary, n_points / 2)
    
    return h_binary

def polish_with_critical_lag(h_binary, n_points, target_M, max_time):
    """Polish binary solution using critical lag swaps"""
    start_time = time.time()
    best_h = h_binary.copy()
    best_c5 = evaluate_c5(h_binary)
    
    while time.time() - start_time < max_time and best_c5 > target_M:
        corr = compute_corr_fft(best_h)
        worst_lag = np.argmax(corr) - n_points + 1
        max_corr = corr[np.argmax(corr)]
        
        if max_corr <= target_M:
            break
            
        # Find mismatched 1->0 pairs
        if worst_lag > 0:
            # Forward lag
            B = np.where((best_h[:-worst_lag] == 1) & (best_h[worst_lag:] == 0))[0]
        else:
            # Backward lag
            B = np.where((best_h[-worst_lag:] == 1) & (best_h[:-worst_lag] == 0))[0]
            worst_lag = abs(worst_lag)
        
        if len(B) == 0:
            break
            
        # Pick a random 1 from B
        i = np.random.choice(B)
        
        # Find candidate zeros for swapping
        if worst_lag > 0:
            Z = np.where((best_h == 0) & ((i + worst_lag >= n_points) | (best_h[i + worst_lag] == 1)))[0]
        else:
            Z = np.where((best_h == 0) & ((i + worst_lag >= n_points) | (best_h[i + worst_lag] == 1)))[0]
        
        if len(Z) == 0:
            j = np.random.choice(np.where(best_h == 0)[0])
        else:
            j = np.random.choice(Z)
        
        # Perform swap
        new_h = best_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        
        # Check if this is an improvement
        new_corr = compute_corr_fft(new_h)
        new_max_corr = np.max(new_corr)
        
        if new_max_corr < best_c5:
            best_h = new_h.copy()
            best_c5 = new_max_corr
            if best_c5 <= target_M:
                break
        else:
            # Accept with small probability (simulated annealing)
            import random
            if random.random() < 0.01:
                best_h = new_h.copy()
                best_c5 = new_max_corr
    
    return best_h
