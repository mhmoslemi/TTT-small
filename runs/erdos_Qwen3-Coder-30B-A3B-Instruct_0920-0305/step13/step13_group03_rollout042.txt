def max_corr(h):
    """Compute maximum overlap correlation"""
    corr = correlate(h, 1-h, mode="full")
    return np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 value"""
    return dx * max_corr(h)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def round_to_binary(h_cont, target_sum=60):
    """Round continuous vector to binary with exact sum"""
    h_bin = (h_cont >= 0.5).astype(int)
    current_sum = np.sum(h_bin)
    
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        diff = current_sum - target_sum
        # Flip positions with smallest values
        flip_indices = np.argsort(h_cont)[:diff]
        h_bin[flip_indices] = 0
    elif current_sum < target_sum:
        # Need to flip some 0s to 1s
        diff = target_sum - current_sum
        # Flip positions with largest values
        flip_indices = np.argsort(h_cont)[-diff:]
        h_bin[flip_indices] = 1
    
    return h_bin

def greedy_critical_lag_search(h_start, max_iter=10000, time_limit=300):

            j_vals = np.arange(-ell, n_points)

            i_vals = np.arange(0, n_points - ell)
            j_vals = np.arange(ell, n_points)
        else:
            return []
        return list(zip(i_vals, j_vals))
    
    # Compute initial correlation
    corr = correlate(h, 1-h, mode="full")
    M = np.max(corr)
    
    iter_count = 0
    while time.time() - start_time < time_limit and iter_count < max_iter and M > target_M:
        iter_count += 1
        # Find worst lag
        worst_lag_idx = np.argmax(corr)
        # Translate full correlation index back to lag
        worst_lag = worst_lag_idx - n_points + 1
        
        # Find mismatched positions at this lag
        mismatched_indices = []

            # h[i] = 1, h[i+worst_lag] = 0
            for i in range(n_points - worst_lag):
                if h[i] == 1 and h[i + worst_lag] == 0:
                    mismatched_indices.append(i)
        elif worst_lag < 0:
            # h[i] = 1, h[i+worst_lag] = 0 where worst_lag < 0
            for i in range(-worst_lag, n_points):

                    mismatched_indices.append(i)
