import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

# Global constants
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_corr_raw(h):
    """Compute raw (unscaled) correlation using FFT."""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_c5_exact(h):
    """Compute exact C5 using np.correlate."""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(60):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_smooth_max_grad(h, alpha):
    """Compute gradient of smooth max approximation."""
    corr_raw = compute_corr_raw(h)
    # Log-sum-exp stabilization: subtract the maximum to prevent overflow
    max_corr = np.max(corr_raw)
    exp_corr = np.exp(alpha * (corr_raw - max_corr))
    weights = exp_corr / np.sum(exp_corr)
    
    # Gradient computation
    grad = np.zeros_like(h)
    for k in range(len(h)):
        # Shift h by k positions to the right (so h[i+k] is accessed as h[i])
        if k == 0:
            shifted = h.copy()
        else:
            shifted = np.zeros_like(h)
            if k > 0:
                shifted[k:] = h[:-k]
            else:
                shifted[:k] = h[-k:]
        
        # Reverse shift for left shift
        if k == 0:
            rev_shifted = h.copy()
        else:
            rev_shifted = np.zeros_like(h)
            if k > 0:
                rev_shifted[:-k] = h[k:]
            else:
                rev_shifted[-k:] = h[:k]
        
        # Add to gradient
        grad += weights[k] * (1 - shifted - rev_shifted)
    
    return dx * grad

def solve_continuation(h0, alpha_levels):
    """Solve the optimization using continuation on alpha."""
    h = h0.copy()
    h = project_to_sum_box(h, target_sum)
    
    for alpha in alpha_levels:
        # Define objective and gradient for current alpha
        def obj(h):
            corr_raw = compute_corr_raw(h)
            max_corr = np.max(corr_raw)
            exp_corr = np.exp(alpha * (corr_raw - max_corr))
            return np.log(np.sum(exp_corr)) / alpha
        
        def grad(h):
            return compute_smooth_max_grad(h, alpha)
        
        # Callback for convergence checks
        def callback(x):
            pass
        
        # Use trust-constr with analytic gradient
        res = minimize(
            obj,
            h,
            method='trust-constr',
            jac=grad,
            bounds=[(0, 1)] * len(h),
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                        'jac': lambda x: np.ones_like(x)},
            options={'maxiter': 200, 'disp': False},
            callback=callback
        )
        
        if res.success:
            h = res.x
        else:
            # Fallback to SLSQP if trust-constr fails
            res = minimize(
                obj,
                h,
                method='SLSQP',
                jac=grad,
                bounds=[(0, 1)] * len(h),
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                options={'maxiter': 200, 'disp': False}
            )
            if res.success:
                h = res.x
            else:
                # If both fail, keep current h and proceed
                pass
        
        # Project after optimization step
        h = project_to_sum_box(h, target_sum)
    
    return h
