import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

# Global constants
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_correlation(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_exact(h):
    """Compute exact C5 using np.correlate"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_logsumexp(h, alpha):
    """Compute smooth approximation to max with parameter alpha"""
    corr = compute_correlation(h)
    # Stable computation: subtract max to avoid overflow
    max_corr = np.max(corr)
    exp_corr = np.exp(alpha * (corr - max_corr))
    return dx * (max_corr + np.log(np.sum(exp_corr)) / alpha)

def gradient_smooth_max(h, alpha):
    """Compute gradient of smooth max with respect to h"""
    corr = compute_correlation(h)
    max_corr = np.max(corr)
    exp_corr = np.exp(alpha * (corr - max_corr))
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient: sum of weights * (1 - shift_plus(h) - shift_minus(h))
    grad = np.zeros_like(h)
    for k in range(-len(h)+1, len(h)):
        if k > 0:
            # shift plus: h[i+k]
            shifted = np.concatenate([np.zeros(k), h[:-k]])
        elif k < 0:
            # shift minus: h[i-k] 
            shifted = np.concatenate([h[-k:], np.zeros(-k)])
        else:
            shifted = h.copy()
        grad -= alpha * weights * shifted
    
    # Add constant term - 1 for each index
    grad -= alpha * weights
    
    return dx * grad

def continuation_optimization(h0, alpha_sequence=[1, 5, 20, 80, 300], max_iter=200, method='SLSQP'):
    """Optimize using continuation on alpha"""
    h = h0.copy()
    for i, alpha in enumerate(alpha_sequence):
        # Use trust-constr for better behavior with large alpha
        def objective(h):
            return smooth_max_logsumexp(h, alpha)
        def gradient(h):
            return gradient_smooth_max(h, alpha)
        
        # Constraints
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                        'jac': lambda x: np.ones_like(x)}]
        bounds = [(0, 1) for _ in range(len(h))]
        
        # Optimize with constraints
        result = minimize(objective, h, method=method, jac=gradient, 
                          bounds=bounds, constraints=constraints, 
                          options={'maxiter': max_iter, 'ftol': 1e-9, 'gtol': 1e-9})
        
        h = result.x
        
        # Project to ensure sum constraint is exact
        h = project_to_sum_box(h, target_sum)
        
        # Early stop if gradient norm is small (indicating stagnation)
        grad = gradient_smooth_max(h, alpha)
        if np.linalg.norm(grad) < 1e-6:
            break
    
    return h

def local_search(h, max_iter=3000):
    """Perform discrete local search"""
    best_h = h.copy()
    best_c5 = c5_exact(best_h)
    
    for _ in range(max_iter):
        # Try a swap
        i, j = np.random.choice(n_points, 2, replace=False)
        h_test = best_h.copy()
        h_test[i], h_test[j] = h_test[j], h_test[i]
        h_test = project_to_sum_box(h_test, target_sum)
        
        new_c5 = c5_exact(h_test)
        if new_c5 < best_c5:
            best_h = h_test
            best_c5 = new_c5
        
        # Try small transfer
        if np.random.rand() < 0.5:
            i, j = np.random.choice(n_points, 2, replace=False)
            delta = np.random.uniform(0.01, 0.05)
            if best_h[i] >= delta and best_h[j] <= 1 - delta:
                h_test = best_h.copy()
                h_test[i] -= delta
                h_test[j] += delta
                h_test = project_to_sum_box(h_test, target_sum)
                
                new_c5 = c5_exact(h_test)
                if new_c5 < best_c5:
                    best_h = h_test
                    best_c5 = new_c5
    
    return best_h, best_c5
