import numpy as np
import time

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Evaluate C5 using FFT correlation"""
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    end_time = start_time + budget_s
    
    # Use n=256 for better resolution
    n_points = 256
    dx = 2.0 / n_points
    target_sum = n_points // 2
    
    best_h = None
    best_c5 = float('inf')
    
    # Multiple restarts
    restarts = 0
    while time.time() < end_time:
        restarts += 1
        # Initialize random binary pattern with exactly target_sum ones
        h = np.zeros(n_points)
        ones_indices = np.random.choice(n_points, target_sum, replace=False)
        h[ones_indices] = 1.0
        
        # Evaluate initial solution
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Simulated Annealing
        T = 0.02
        T_min = 1e-6
        alpha = 0.9997
        max_iter = 200000
        
        iter_count = 0
        while time.time() < end_time and iter_count < max_iter:
            # Make a random swap
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            
            if len(ones_idx) > 0 and len(zeros_idx) > 0:
                i = np.random.choice(ones_idx)
                j = np.random.choice(zeros_idx)
                
                # Create new solution
                h_new = h.copy()
                h_new[i] = 0
                h_new[j] = 1
                
                # Evaluate new solution
                new_c5 = evaluate_c5(h_new)
                delta = new_c5 - current_c5
                
                # Accept or reject
                if delta < 0 or np.random.rand() < np.exp(-delta / T):
                    h = h_new
                    current_c5 = new_c5
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()
            
            # Update temperature
            T = max(T_min, T * alpha)
            iter_count += 1
            
            # Early stopping if no improvement for a while
            if iter_count % 10000 == 0 and iter_count > 0:
                if time.time() > end_time - 10:  # Leave 10 seconds for final tuning
                    break
        
        # Greedy improvement after SA
        while time.time() < end_time:
            corr = compute_corr_fft(h)
            max_idx = np.argmax(corr)
            s = max_idx - (n_points - 1)
            
            # Compute contribution array for shift s
            if s >= 0:
                h_shift = np.concatenate([h[s:], np.zeros(s)])
            else:
                h_shift = np.concatenate([np.zeros(-s), h[:n_points + s]])
            contrib = 1.0 - h_shift
            
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            if len(ones) == 0 or len(zeros) == 0:
                break
                
            contrib_ones = contrib[ones]
            contrib_zeros = contrib[zeros]
            
            i_max = ones[np.argmax(contrib_ones)]
            j_min = zeros[np.argmin(contrib_zeros)]
            
            if contrib[i_max] > contrib[j_min] + 1e-12:
                h[i_max] = 0
                h[j_min] = 1
                current_c5 = evaluate_c5(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            else:
                break
    
    return (best_h.tolist(), float(best_c5), n_points)
