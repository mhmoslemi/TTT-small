import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT correlation"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    bounds = [(0, 1) for _ in range(n_points)]

                h_candidate = project_to_sum_box(result.x, TARGET_SUM)
                c5_candidate = evaluate_c5(h_candidate)
                de_results.append((h_candidate, c5_candidate))
        except:

    for h_candidate, c5_candidate in de_results:
        if c5_candidate < best_c5:

        def objective_lbfgs(x):
            x_proj = project_to_sum_box(x, TARGET_SUM)

        h_current = best_h.copy()

            improved = False

                old_val = h_current[j]
