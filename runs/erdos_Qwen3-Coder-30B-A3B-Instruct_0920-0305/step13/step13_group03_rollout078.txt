import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def sa_strategy(h0, c5_0, start_time, budget_s):
    """Simulated Annealing strategy"""
    np.random.seed(int(time.time()) % 1000000)
    
    h_current = h0.copy()
    c5_current = c5_0
    T = 0.1
    T_min = 1e-6
    alpha = 0.995
    max_iter = 50000
    patience = 500
    patience_count = 0
    max_perturb = 0.1
    
    t1 = start_time + 0.3 * budget_s
    iter_count = 0
    
    while time.time() < t1 and iter_count < max_iter:
        iter_count += 1
        h_new = h_current.copy()
        # Perturb a few indices
        num_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for i in indices:
            perturb = np.random.uniform(-max_perturb, max_perturb)
            h_new[i] += perturb
            
        # Project back to feasible set
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
            patience_count = 0
        else:
            if np.random.rand() < np.exp(-(c5_new - c5_current) / T):
                h_current = h_new.copy()
                c5_current = c5_new
            patience_count += 1
        
        # Cooling schedule
        T = max(T_min, T * alpha)
        
        if patience_count > patience:
            break
            
    return h_current, c5_current

def gradient_strategy(h0, c5_0, start_time, budget_s):
    """Gradient-based strategy with better numerical differentiation"""
    try:
        from scipy.optimize import minimize
        np.random.seed(int(time.time()) % 1000000)
        
        def objective(h):
            corr = compute_corr_fft(h)
            return c5_from_corr(corr, n_points)
        
        # Try to refine with gradient information
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Start with current best
        h_start = h0.copy()
        h_start = project_to_sum_box(h_start, TARGET_SUM)
        
        # Use L-BFGS-B with better settings
        result = minimize(objective, h_start, method='L-BFGS-B', 
                         bounds=bounds, options={'maxiter': 1000, 'gtol': 1e-9})
        
        if result.success:
            h_final = project_to_sum_box(result.x, TARGET_SUM)
            c5_final = evaluate_c5(h_final)
            return h_final, c5_final
        else:
            return h0.copy(), c5_0
            
    except:
        return h0.copy(), c5_0

def evolutionary_strategy(h0, c5_0, start_time, budget_s):
    """Evolutionary strategy with mutation operators"""
    np.random.seed(int(time.time()) % 1000000)
    
    population_size = 20
    generations = 500
    mutation_rate = 0.1
    
    # Create initial population
    population = []
    for _ in range(population_size):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_c5(h) for h in population]
    
    t1 = start_time + 0.2 * budget_s
    gen_count = 0
    
    while time.time() < t1 and gen_count < generations:
        # Tournament selection
        idx1, idx2 = np.random.choice(population_size, 2, replace=False)
        parent1 = population[idx1] if fitness[idx1] < fitness[idx2] else population[idx2]
        parent2 = population[idx1] if fitness[idx1] >= fitness[idx2] else population[idx2]
        
        # Crossover
        crossover_point = np.random.randint(1, n_points-1)
        child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        
        # Mutation
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                child[i] += np.random.normal(0, 0.02)
        
        # Clip and project
        child = np.clip(child, 0, 1)
        child = project_to_sum_box(child, TARGET_SUM)
        
        # Replace worst individual
        worst_idx = np.argmax(fitness)
        population[worst_idx] = child
        fitness[worst_idx] = evaluate_c5(child)
        
        gen_count += 1
    
    # Return best individual
    best_idx = np.argmin(fitness)
    return population[best_idx], fitness[best_idx]
