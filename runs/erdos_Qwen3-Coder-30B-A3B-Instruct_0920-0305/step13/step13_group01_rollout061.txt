```python
import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT - returns raw correlation values"""
    corr = compute_corr_fft(h)
    return np.max(corr)

def evaluate_c5_precise(h):
    """Precise evaluation using np.correlate"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x - lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = np.min(x) - 1.0, np.max(x) + 1.0
    for _ in range(60):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x - (a + b)/2, 0, 1)
    return result

def compute_subgradient(h, corr):
    """Compute subgradient for the max correlation"""
    n = len(h)
    max_val = np.max(corr)
    eps = 1e-12
    active_idx = np.where(corr >= max_val - eps)[0]
    
    # Initialize gradient with constant term for each active lag
    g = np.full(n, len(active_idx), dtype=float)
    
    for idx in active_idx:
        k = idx - (n - 1)
        if k >= 0:
            # Subtract h_{j + k} and h_{j - k}
            g[:n - k] -= h[k:]
            g[k:] -= h[:n - k]
        else:
            # k < 0, equivalent to k = -d
            d = -k
            g[d:] -= h[:n - d]
            g[:n - d] -= h[d:]
    
    return g, max_val

def generate_sin_start():
    """Generate sinusoidal starting point"""
    x = (np.arange(n_points) + 0.5) * dx
    # Random amplitude between 0 and 0.5 to ensure h in [0,1]
    a = np.random.rand() * 0.5
    phase = np.random.rand() * 2 * np.pi
    h = 0.5 + a * np.sin(np.pi * x + phase)
    # Ensure box constraints
    h = np.clip(h, 0, 1)
    # Project to correct sum
    h = project_to_sum_box(h, target_sum)
    return h

def generate_random_start():
    """Generate random starting point"""
    h = np.random.rand(n_points)
    h = project_to_sum_box(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    global n_points, dx, target_sum
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    best_c5 = float('inf')
    best_h = None
    
    # Generate initial solutions
    initial_solutions = []
    
    # Use original initial values if available
    if 'initial_h_values' in globals() and len(initial_h_values) == 120:
        h0 = np.repeat(initial_h_values, 2)
        h0 = project_to_sum_box(h0, target_sum)
        initial_solutions.append(('initial', h0))
    
    # Add sinusoidal starts
    for i in range(5):
        h = generate_sin_start()
        initial_solutions.append((f'sin_{i}', h))
    
    # Add random starts
    for i in range(5):
        h = generate_random_start()
        initial_solutions.append((f'rand_{i}', h))
    
    # Run optimization on each initial solution
    max_iters_per_start = 8000
    remaining_time = budget_s - (time.time() - start_time)
    
    for name, h0 in initial_solutions:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Initialize
        h = h0.copy()
        current_c5 = evaluate_c5_fast(h)
        best_local_c5 = current_c5
        
        # Subgradient descent
        base_lr = 0.4
        for iter_num in range(max_iters_per_start):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute correlation
            corr = compute_corr_fft(h)
            
            # Compute subgradient
            g, max_corr = compute_subgradient(h, corr)
            
            # Compute step size
            step = base_lr / np.sqrt(iter_num + 1)
            
            # Update
            h_new = h - step * g
            
            # Project
            h_new = project_to_sum_box(h_new, target_sum)
            
            # Evaluate
            new_c5_raw = np.max(compute_corr_fft(h_new))
            
            # Accept if improvement
            if new_c5_raw <= max_corr:
                h = h_new
                max_corr = new_c5_raw
                if max_corr * dx < best_local_c5:
                    best_local_c5 = max_corr * dx
                    if max_corr * dx < best_c5:
                        best_c5 = max_corr * dx
                        best_h = h.copy()
            else:
                # No improvement, but still update to explore
                h = h_new
            
            # Occasionally reduce learning rate
            if iter_num > 0 and iter_num % 1000 == 0 and base_lr > 0.01:
                base_lr *= 0.95
    
    # Final refinement with better evaluation
    if best_h is not None:
        final_c5 = evaluate_c5_precise(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    # If no improvement found, use the initial solution
    if best_h is None:
        if 'initial_h_values' in globals() and len(initial_h_values) == 120:
            h0 = np.repeat(initial_h_values, 2)
            h0 = project_to_sum_box(h0, target_sum)
            best_h = h0
            best_c5 = evaluate_c5_precise(h0)
        else:
            best_h = generate_random_start()
            best_c5 = evaluate_c5_precise(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
```
