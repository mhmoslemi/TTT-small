```python
import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1 << (L - 1).bit_length()
    corr = np.fft.ifft(np.fft.fft(h_rev, fft_size) * np.fft.fft(b, fft_size)).real
    return corr[:L]

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return np.max(corr)

def evaluate_c5_precise(h):
    """Precise evaluation of C5 using numpy correlate"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x - lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = np.min(x) - 1.0, np.max(x) + 1.0
    for _ in range(60):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x - (a + b)/2, 0, 1)
    return result

def compute_subgradient(h, corr):
    """Compute subgradient of max correlation"""
    n = len(h)
    max_val = np.max(corr)
    eps = 1e-12
    active_idx = np.where(corr >= max_val - eps)[0]
    
    # Initialize gradient with constant term for each active lag
    g = np.full(n, len(active_idx), dtype=float)
    
    for idx in active_idx:
        k = idx - (n - 1)
        if k >= 0:
            # For positive lags: subtract h_{j+k} and h_{j-k}
            g[:n - k] -= h[k:]
            g[k:] -= h[:n - k]
        else:
            # For negative lags: same logic
            d = -k
            g[d:] -= h[:n - d]
            g[:n - d] -= h[d:]
    
    return g, max_val

def generate_sin_initial():
    """Generate sinusoidal initial solution"""
    x = (np.arange(n_points) + 0.5) * dx
    # Random amplitude (0 to 0.5) and phase
    a = np.random.rand() * 0.5
    phase = np.random.rand() * 2 * np.pi
    h = 0.5 + a * np.sin(np.pi * x + phase)
    return project_to_sum_box(h, target_sum)

def generate_uniform_initial():
    """Generate uniform random initial solution"""
    h = np.random.rand(n_points)
    return project_to_sum_box(h, target_sum)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with repeated initial solution if available
    if 'initial_h_values' in globals() and len(initial_h_values) == 120:
        h0 = np.repeat(initial_h_values, 2)
    else:
        h0 = np.ones(n_points) * 0.5
    
    # Ensure h0 is valid
    h0 = project_to_sum_box(h0, target_sum)
    best_c5 = evaluate_c5_precise(h0)
    best_h = h0.copy()
    
    # Generate multiple initial solutions
    initial_solutions = [h0]
    for _ in range(4):
        initial_solutions.append(generate_sin_initial())
    for _ in range(4):
        initial_solutions.append(generate_uniform_initial())
    
    # Subgradient descent with multiple restarts
    max_iter_per_restart = 10000
    base_lr = 0.3
    
    for restart_idx, h_start in enumerate(initial_solutions):
        if time.time() - start_time > budget_s - 10:
            break
            
        h = h_start.copy()
        current_c5 = evaluate_c5_fast(h)
        
        for iter_num in range(max_iter_per_restart):
            if time.time() - start_time > budget_s - 5:
                break
                
            corr = compute_corr_fft(h)
            g, max_corr = compute_subgradient(h, corr)
            
            # Adaptive step size
            step = base_lr / np.sqrt(iter_num + 1)
            
            # Apply step
            h_new = h - step * g
            
            # Project back to feasible region
            h_new = project_to_sum_box(h_new, target_sum)
            
            # Evaluate new solution
            new_corr = compute_corr_fft(h_new)
            new_c5_raw = np.max(new_corr)
            
            if new_c5_raw < max_corr:
                h = h_new
                current_c5 = new_c5_raw * dx
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
    
    # Final refinement with precise evaluation
    best_c5 = evaluate_c5_precise(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
```
