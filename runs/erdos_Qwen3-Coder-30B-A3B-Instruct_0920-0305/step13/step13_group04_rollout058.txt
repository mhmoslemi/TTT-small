import numpy as np
from scipy.optimize import differential_evolution
import time

def compute_worst_overlap(intervals):
    """Compute worst-case overlap I(k) for a list of intervals"""
    if not intervals:
        return 0.0
    
    # Generate all candidate k values where overlap changes
    endpoints = []
    for a, b in intervals:
        endpoints.extend([a, b])
    
    # All differences between endpoints (including zero difference)
    candidates = set([0.0])
    for i in range(len(endpoints)):
        for j in range(len(endpoints)):
            candidates.add(endpoints[i] - endpoints[j])
    
    # Filter candidates to reasonable range [-2, 2] 
    candidates = [k for k in candidates if -2 <= k <= 2]
    
    min_overlap = float('inf')
    
    # For each candidate k, compute intersection measure
    for k in candidates:
        total_overlap = 0.0
        for a1, b1 in intervals:
            # Compute overlap with shifted intervals (a1+k, b1+k)
            for a2, b2 in intervals:
                # Intersection of [a1,b1] and [a2+k, b2+k]
                left = max(a1, a2 + k)
                right = min(b1, b2 + k)
                if right > left:
                    total_overlap += (right - left)
        min_overlap = min(min_overlap, total_overlap)
        
    return min_overlap

def interval_to_h(intervals, n_points=256):
    """Convert intervals to step function h"""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    for i in range(n_points):
        cell_start = i * dx
        cell_end = (i + 1) * dx
        
        # Compute overlap of cell with all intervals
        overlap = 0.0
        for a, b in intervals:
            # Intersection of [cell_start, cell_end] with [a, b]
            left = max(cell_start, a)
            right = min(cell_end, b)
            if right > left:
                overlap += (right - left)
        
        h[i] = overlap / dx
    
    return h

def objective_function(raw_params):
    """Objective function for optimization"""
    K = len(raw_params) // 2
    raw_lengths = raw_params[:K]
    raw_gaps = raw_params[K:]
    
    # Normalize lengths  
    lengths = np.array(raw_lengths) / np.sum(raw_lengths)
    
    # Normalize gaps
    gaps = np.array(raw_gaps) / np.sum(raw_gaps)
    
    # Compute interval positions
    intervals = []
    pos = gaps[0]
    for i in range(K):
        a = pos
        b = a + lengths[i]
        intervals.append((a, b))
        pos = b + gaps[i+1]
    
    # Compute worst overlap
    min_overlap = compute_worst_overlap(intervals)
    
    # Objective is 1 - min_overlap (we want to maximize min overlap)
    return 1 - min_overlap
