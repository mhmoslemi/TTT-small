import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = correlate(h, 1-h, mode="full")
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_convex_subproblem(h_prev, lags, n_points, dx):
    """Solve the convex subproblem using CVXPY"""
    h = Variable(n_points)
    T = Variable()
    
    # Define constraints
    constraints = [0 <= h, h <= 1]
    constraints.append(sum(h) == n_points / 2)
    
    # Add convex surrogate constraints for each lag
    for ell in lags:
        # Calculate the quadratic part
        quad_term = 0
        linear_term = 0
        for i in range(n_points):
            if 0 <= i + ell < n_points:
                quad_term += (h[i] - h[i + ell])**2
                linear_term += (h_prev[i] + h_prev[i + ell]) * (h[i] + h[i + ell])
        
        # Add constraint (quadratic + linear terms <= T)
        constraints.append(quad_term / 4 - linear_term / 2 <= T)
    
    # Objective function
    objective = Minimize(T)
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status == 'optimal':
            return h.value, T.value
    except:
        pass
        
    return h_prev, float('inf')

def generate_initial_lags(n):
    """Generate initial set of lags"""
    lags = [0]
    for i in range(1, 5):
        lags.extend([i * n // 5, -i * n // 5])
    return list(set(lags))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    target_M = int(0.3807 / dx)  # target C5 bound in terms of overlap count
    
    # Initialize
    best_c5 = float('inf')
    best_h = None
    
    # Start with initial solution
    h0 = np.array(initial_h_values)
    h0 = project_to_sum_box(h0, n_points / 2)
    
    current_h = h0.copy()
    current_c5 = evaluate_c5(h0)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
    
    # Generate initial lags
    lags = generate_initial_lags(n_points)
    
    # Convex-concave procedure with cutting plane
    max_iter = 1000
    iter_count = 0
    patience = 0
    patience_limit = 10
    
    while time.time() - start_time < 0.4 * budget_s and iter_count < max_iter:
        iter_count += 1
        prev_h = current_h.copy()
        
        # Solve convex subproblem
        new_h, t_value = solve_convex_subproblem(current_h, lags, n_points, dx)
        
        if t_value == float('inf'):
            break
            
        current_h = new_h
        current_c5 = evaluate_c5(current_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            patience = 0
        else:
            patience += 1
            
        if patience > patience_limit:
            break
            
        # Evaluate all lags to find violations
        corr = correlate(current_h, 1-current_h, mode="full")
        max_corr = np.max(corr)
        
        # If we're close to target, start being more exact
        if max_corr <= target_M + 5:  # tolerance
            break
            
        # Find violating lags (those that exceed current T by a margin)
        violations = []
        for i, corr_val in enumerate(corr):
            if corr_val > t_value + 1e-6 and i not in lags:
                violations.append(i - n_points + 1)  # convert index to lag
                
        if violations:
            # Add most violated lags
            lags.extend(violations[:min(5, len(violations))])
            # Keep only recent lags to avoid explosion
            if len(lags) > 80:
                lags = lags[-40:]
    
    # Rounding phase
    if best_h is not None:
        best_h_process = best_h.copy()
        
        # 30 random rounding attempts
        best_binary = None
        best_binary_c5 = float('inf')
        
        for _ in range(30):
            # Sort by value and set first half to 1
            sort_indices = np.argsort(best_h_process)[::-1]
            binary_h = np.zeros(n_points)
            binary_h[sort_indices[:n_points//2]] = 1
            
            # Ensure sum constraint
            binary_h = project_to_sum_box(binary_h, n_points / 2)
            
            # Evaluate
            binary_c5 = evaluate_c5(binary_h)
            if binary_c5 < best_binary_c5:
                best_binary_c5 = binary_c5
                best_binary = binary_h.copy()
        
        if best_binary is not None:
            best_h = best_binary
            best_c5 = best_binary_c5
    
    # Local improvement with critical-lag swaps
    if best_h is not None:
        current_h = best_h.copy()
        current_c5 = evaluate_c5(current_h)
        
        # Run critical-lag polishing
        corr = correlate(current_h, 1-current_h, mode="full")
        max_corr = np.max(corr)
        
        # Do critical-lag swaps until no improvement or we run out of time
        while time.time() - start_time < 0.8 * budget_s and max_corr > target_M:
            # Find worst lag
            worst_lag_idx = np.argmax(corr)
            worst_lag = worst_lag_idx - n_points + 1
            
            # Find mismatched 1->0 pairs for this lag
            mismatched_ones = []
            for i in range(n_points):
                if i + worst_lag < n_points and i + worst_lag >= 0:
                    if current_h[i] == 1 and current_h[i + worst_lag] == 0:
                        mismatched_ones.append(i)
            
            if not mismatched_ones:
                break
            
            # Pick a random mismatched 1
            i = np.random.choice(mismatched_ones)
            
            # Find candidate positions for 0 that don't create new mismatch
            candidate_zeros = []
            for j in range(n_points):
                # Check if j is a valid zero that won't create new mismatch at this lag
                if current_h[j] == 0:
                    if j + worst_lag >= n_points or current_h[j + worst_lag] == 1:
                        candidate_zeros.append(j)
            
            if not candidate_zeros:
                j = np.random.randint(n_points)
            else:
                j = np.random.choice(candidate_zeros)
            
            # Perform swap
            current_h[i], current_h[j] = current_h[j], current_h[i]
            
            # Re-evaluate
            corr = correlate(current_h, 1-current_h, mode="full")
            max_corr = np.max(corr)
            
            if max_corr < current_c5:
                current_c5 = max_corr
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
    
    # Final evaluation
    if best_h is None:
        best_h = h0.copy()
    
    final_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), float(final_c5), n_points)
