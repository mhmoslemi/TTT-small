import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def grad_shift(h, shift):
    """Compute gradient for a specific shift"""
    N = len(h)
    g = np.zeros(N)
    
    if shift >= 0:
        # Forward shift: h[i] * (1 - h[i+shift])
        if shift < N:
            g[:N-shift] += 1 - h[shift:]
            g[shift:] += -h[:N-shift]
    else:
        # Backward shift: h[i] * (1 - h[i+shift]) where shift is negative
        t = -shift
        if t < N:
            g[t:] += 1 - h[:N-t]
            g[:N-t] += -h[t:]
    
    return g

def compute_softmax_gradient(h, corr, gamma):
    """Compute gradient using softmax approximation"""
    # Compute weights with numerical stability
    max_corr = np.max(corr)
    w = np.exp(gamma * (corr - max_corr))
    w /= np.sum(w)
    
    # Compute weighted gradient
    grad = np.zeros_like(h)
    shifts = np.arange(-n_points + 1, n_points)
    
    for i, (shift, weight) in enumerate(zip(shifts, w)):
        if weight > 1e-12:  # Only consider significant weights
            g_shift = grad_shift(h, shift)
            grad += weight * g_shift
    
    return grad * dx
