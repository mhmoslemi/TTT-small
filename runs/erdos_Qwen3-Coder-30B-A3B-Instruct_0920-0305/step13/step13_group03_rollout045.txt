import numpy as np

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def round_binary(h_cont, target_sum):
    """Round continuous solution to binary vector"""
    n = len(h_cont)
    # Sort indices by decreasing values
    sorted_idx = np.argsort(-h_cont)
    b = np.zeros(n)
    b[sorted_idx[:int(target_sum)]] = 1
    
    # Ensure exact sum
    ones_count = int(np.sum(b))
    if ones_count > target_sum:
        # Flip some 1s to 0s
        flip_indices = sorted_idx[int(target_sum):ones_count]
        for idx in flip_indices:
            b[idx] = 0
    elif ones_count < target_sum:
        # Flip some 0s to 1s
        zero_indices = np.where(b == 0)[0]
        for idx in zero_indices[:int(target_sum - ones_count)]:
            b[idx] = 1
    
    return b

def critical_lag_polish(b, max_iter=10000):
    """Perform critical-lag greedy local search"""
    n = len(b)
    max_I = c5_from_corr(compute_corr_fft(b), n)
    
    # Precompute full correlation
    I = compute_corr_fft(b)
    
    iter_count = 0
    while iter_count < max_iter and max_I > 0:
        # Find worst lag
        max_idx = np.argmax(I)
        # Convert to actual lag
        lag = max_idx - (n - 1)
        
        # Get mismatched 1→0 pairs that hurt this lag
        B = []
        for i in range(n):
            if b[i] == 1 and i + lag < n and b[i + lag] == 0:
                B.append(i)
        
        if not B:
            break
            
        # Pick a random mismatched 1
        i = np.random.choice(B)
        
        # Find candidate zeros that won't create new mismatches
        Z = []
        for j in range(n):
            if b[j] == 0:
                # Check if swapping won't create new mismatch at this lag
                if j + lag >= n or b[j + lag] == 1:
                    Z.append(j)
        
        # If no good zeros, pick any zero
        if not Z:
            j = np.random.choice(np.where(b == 0)[0])
        else:
            j = np.random.choice(Z)
        
        # Perform swap
        b[i], b[j] = b[j], b[i]
        
        # Recompute correlation
        I = compute_corr_fft(b)
        max_I = c5_from_corr(I, n)
        iter_count += 1
    
    return b
