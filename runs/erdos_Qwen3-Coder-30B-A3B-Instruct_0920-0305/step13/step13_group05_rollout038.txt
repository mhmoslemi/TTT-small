# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smoothed_max(x, gamma=100.0):
    """Smooth approximation of max using softmax"""
    # Use log-sum-exp trick for numerical stability
    return np.log(np.sum(np.exp(gamma * x))) / gamma

def smooth_objective(h, gamma=100.0):
    """Smoothed version of the objective function"""
    corr = compute_corr_fft(h)
    # Use smooth max instead of exact max
    return dx * smoothed_max(corr, gamma)

def smooth_gradient(h, gamma=100.0):
    """Gradient of smoothed objective"""
    corr = compute_corr_fft(h)
    max_idx = np.argmax(corr)
    
    # Compute gradient of the smoothed max
    exp_corr = np.exp(gamma * corr)
    exp_sum = np.sum(exp_corr)
    
    # Gradient of smooth max
    dmax = exp_corr / exp_sum
    
    # Gradient of correlation with respect to h
    # We only need to compute gradient at max_idx
    grad = np.zeros_like(h)
    
    # For simplicity, since correlation is conv, we compute 
    # approximate gradients using finite differences
    eps = 1e-6
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_minus = h.copy()
        h_minus[i] -= eps
        
        corr_plus = compute_corr_fft(h_plus)
        corr_minus = compute_corr_fft(h_minus)
        
        grad[i] = (corr_plus[max_idx] - corr_minus[max_idx]) / (2 * eps)
    
    return dx * gamma * (dmax[max_idx] - np.mean(dmax)) * grad

    # Ensure we don't return None
    if best_h is None:
        best_h = h0.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
