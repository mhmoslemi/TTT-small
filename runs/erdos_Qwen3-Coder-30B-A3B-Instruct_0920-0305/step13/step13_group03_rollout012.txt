import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def max_corr(h):
    """Compute the maximum overlap integral"""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using the given discrete step function"""
    dx = 2.0 / len(h)
    return dx * max_corr(h)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    if current_sum != n_points / 2:
        diff = int(current_sum - n_points / 2)
        if diff > 0:  # Too many ones
            # Find indices with highest continuous values and flip them to zero
            flip_indices = np.argsort(h_cont)[-diff:]
            h_init[flip_indices] = 0
        else:  # Too few ones
            # Find indices with lowest continuous values and flip them to one
            flip_indices = np.argsort(h_cont)[:abs(diff)]
            h_init[flip_indices] = 1
    
    # Warm start for MILP: identify fixed variables
    FixedOne = np.where(h_init == 1)[0]
    FixedZero = np.where(h_init == 0)[0]
    FreeIdx = np.setdiff1d(np.arange(n_points), np.concatenate([FixedOne, FixedZero]))
    
    # Build and solve MILP
    milp_time = min(260, budget_s - 100)
    h_milp = h_init.copy()
    M_opt = max_corr(h_milp)
    
    if M_opt <= target_M:
        # Already good enough
        pass
    else:
        # Formulate and solve MILP
        try:
            # Create decision variables
            h_vars = cp.Variable(n_points, boolean=True)
            
            # Fix already known variables
            for i in FixedOne:
                h_vars[i].value = 1.0
            for i in FixedZero:
                h_vars[i].value = 0.0
            
            # Create auxiliary variables for products h[i] * h[i+delta]
            w_vars = {}
            lags = []
            for delta in range(-97, 98):  # Only consider lags that might be critical
                lags.append(delta)
                for i in range(n_points - abs(delta)):
                    if (delta >= 0):
                        w_vars[(i, delta)] = cp.Variable(boolean=True)
                    else:
                        w_vars[(i, delta)] = cp.Variable(boolean=True)
            
            # Constraints for product linearization
            constraints = []
            
            # Sum constraint
            constraints.append(cp.sum(h_vars) == n_points/2)
            
            # Product constraints
            for delta in lags:
                for i in range(n_points - abs(delta)):
                    if delta >= 0:
                        # h[i] * h[i+delta] = w[i,delta]
                        constraints.append(w_vars[(i, delta)] <= h_vars[i])
                        constraints.append(w_vars[(i, delta)] <= h_vars[i + delta])
                        constraints.append(w_vars[(i, delta)] >= h_vars[i] + h_vars[i + delta] - 1)
                    else:
                        # h[i] * h[i+delta] = w[i,delta] where delta < 0
                        constraints.append(w_vars[(i, delta)] <= h_vars[i])
                        constraints.append(w_vars[(i, delta)] <= h_vars[i - delta])
                        constraints.append(w_vars[(i, delta)] >= h_vars[i] + h_vars[i - delta] - 1)

                else:
                    I_vars[delta] = cp.sum(h_vars[-delta:] - 
                                          [w_vars[(i, delta)] for i in range(n_points+delta)])

            M = cp.Variable()
            for delta in lags:
                constraints.append(I_vars[delta] <= M)
            
            # Objective
            prob = cp.Problem(cp.Minimize(M), constraints)

    corr = correlate(h_current, 1 - h_current, mode="full")

    timeout = start_time + budget_s - 10
    max_iter = 100000

            break
            
        corr = correlate(h_current, 1 - h_current, mode="full")
        max_lag_idx = np.argmax(corr)

        if corr[max_lag_idx] <= target_M * dx:

        bad_positions = []
        if best_lag >= 0:
            # h[i] = 1, h[i+best_lag] = 0
            for i in range(len(h_current) - best_lag):
                if h_current[i] == 1 and h_current[i + best_lag] == 0:
                    bad_positions.append(i)
        else:
            # h[i] = 1, h[i+best_lag] = 0 where best_lag < 0 means i+|best_lag| >= n
