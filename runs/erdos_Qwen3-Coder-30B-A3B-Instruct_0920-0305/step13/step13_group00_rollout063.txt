import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Create CVXPY variables
    h = cp.Variable(n_points, name="h")
    t = cp.Variable(name="t")
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum
    ]
    
    # Create y variables for each shift
    # y[i,s] represents h[i] * h[i+s] for valid shifts
    shifts = range(1, n_points)  # s from 1 to n-1

    # The overlap term O_s = sum_{i=0}^{n-1-s} h[i]*h[i+s] for s=0 to n-1
    # For s=0, overlap should be sum h[i]^2 
    # We'll use a combined approach with all shifts handled
    
    # Let's reformulate: for each shift s in 0..n-1
    # C'_s = (target_sum - prefix_sum[s]) - sum_{i=0}^{n-1-s} h[i] * h[i+s]
    # prefix_sum[s] = sum_{i=0}^{s-1} h[i]
    
    # We'll handle all shifts with constraints
    for s in range(n_points):  # s=0 to n-1
        # For shift s, define overlap sum
        if s == 0:
            # O_0 = sum h[i]^2 ≈ sum y[i,0] where y[i,0] = h[i]^2 (not special case for marking)
            # Since we linearize later with McCormick for h[i]*h[i+s], this is fine 
            continue  # will handle specially
        else:
            # For shift s: O_s = sum_{i=0}^{n-1-s} h[i] * h[i+s]
            # We linearize using McCormick envelopes
            
            # We create variables for y_{i,s} for valid i: 0 to n-1-s
            y_s_vars = []
            for i in range(n_points - s):
                if i + s < n_points:
                    y_var = cp.Variable(name=f"y_{i}_{s}")
                    y_s_vars.append(y_var)
                    
                    # Add McCormick constraints for y_var = h[i] * h[i+s]
                    constraints.extend([
                        y_var >= 0,
                        y_var <= h[i],
                        y_var <= h[i+s],
                        y_var >= h[i] + h[i+s] - 1
                    ])

            # Which means: O_s + prefix_sum[s] >= target_sum - t

    # For all s, define prefixes explicitly

    # Introduce binary constraints for better computation:
    # Actually calculate all shift-based overlaps with constraints

    total_constraints = []
    all_y_vars = []
    
    # We need to track the y variables properly for all valid (i,s) pairs 
    y_vars_dict = {}
    
    for s in range(n_points):  # shifts 0 to n-1  
        # For shift s: overlap_s = sum_{i=0}^{n-1-s} h[i] * h[i+s]
        # Index ranges that exist: 0 <= i <= n-1-s
        
        if s == 0:
            # This is special case, sum h[i]^2 = sum_{i=0}^{n-1} h[i]^2  
            # We'll later compute this as sum y[i,0] for i from 0 to n-1
            pass
        else:
            # For s > 0: we create y[i,s] for i = 0..n-1-s
            for i in range(n_points - s):
                y_var = cp.Variable(name=f"y_{i}_{s}")
                y_vars_dict[(i,s)] = y_var
                all_y_vars.append(y_var)
                
                # Add McCormick constraints for y[i,s] = h[i]*h[i+s]
                constraints.extend([
                    y_var >= 0,
                    y_var <= h[i],
                    y_var <= h[i+s],
                    y_var >= h[i] + h[i+s] - 1
                ])
    
    # Now add the objective functions for all s:
    # For all shifts, C'_s = (target_sum - prefix_sum[s]) - O_s   <= t
    # Define prefix sums carefully to distinguish by shift s:
    # We must compute prefix sum as variable, not just constraint value.
    # O_s = sum_{i=0}^{n-1-s} y[i,s]
    
    # Defining P_s explicitly: P_s = sum_{i=0}^{s-1} h[i]
    P_vars = [cp.Variable(name=f"P_{s}") for s in range(n_points)]
    
    for s in range(n_points):
        # Define P_s = sum_{i=0}^{s-1} h[i]
        if s == 0:
            constraints.append(P_vars[s] == 0)
        else:
            constraints.append(P_vars[s] == P_vars[s-1] + h[s-1])
    
    # Now for shift s, C'_s = (target_sum - P_s) - O_s <= t
    # Need: O_s = sum_{i=0}^{n-1-s} y[i,s]
    # So: (target_sum - P_s) - sum y[i,s] <= t
    # Which means: sum y[i,s] >= (target_sum - P_s) - t
    # Or equivalently: sum y[i,s] + P_s >= target_sum - t
    
    # So we add constraints for valid shifts:
    for s in range(n_points):
        if s == 0:
            # For s=0: sum = sum h[i]^2, which will be computed via y[i,0]

            total_y_0 = cp.sum([y_vars_dict[(i,0)] for i in range(n_points)])
            constraints.append(total_y_0 + 0 >= target_sum - t)  # this handles s=0 case
        else:
            # For s >=1:
            # sum_{i=0}^{n-1-s} y[i,s] can be handled
            valid_indices = list(range(n_points - s))  # i from 0 to n-1-s
            sum_y_s = cp.sum([y_vars_dict[(i,s)] for i in valid_indices])
            constraints.append(sum_y_s + P_vars[s] >= target_sum - t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with solver
    try:
        prob.solve(solver=cp.GLPK, verbose=False, time_limit=budget_s*0.9, max_iters=10000)
    except:
        # If GLPK fails, fall back to ECOS or note issue

        except:
            # If that fails, use default parameters or iterate through options
            pass
