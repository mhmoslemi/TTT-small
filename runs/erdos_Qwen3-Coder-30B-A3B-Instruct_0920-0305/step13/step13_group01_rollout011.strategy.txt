def compute_subgradient(h, corr):
    n = len(h)
    max_val = np.max(corr)
    # tolerance
    eps = 1e-12
    active_idx = np.where(corr >= max_val - eps)[0]
    # count active
    m = len(active_idx)
    # initialize gradient with constant term: 1 per active lag per component
    g = np.full(n, m, dtype=float)
    for idx in active_idx:
        k = idx - (n - 1)
        if k >= 0:
            k_abs = k
            # subtract h_{j + k}
            # For j from 0 to n - k -1
            g[:n - k_abs] -= h[k_abs:]
            # subtract h_{j - k} for j >= k
            g[k_abs:] -= h[:n - k_abs]
        else:
            d = -k
            # Equivalent to k negative, treat as above
            g[d:] -= h[:n - d]
            g[:n - d] -= h[d:]
    return g, max_val
