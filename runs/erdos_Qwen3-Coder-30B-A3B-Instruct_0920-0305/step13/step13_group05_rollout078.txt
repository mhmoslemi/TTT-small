    """Smooth approximation to max function using softmax"""
    x = np.array(x)

    exp_x = np.exp(beta * (x - max_x))
    return max_x + np.log(np.sum(exp_x)) / beta

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Simple projection: clip and rescale
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x
    
    # Scale to target sum
    scale = target_sum / current_sum
    x = x * scale
    return np.clip(x, 0, 1)
