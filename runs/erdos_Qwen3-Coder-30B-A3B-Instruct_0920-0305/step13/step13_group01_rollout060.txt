import time

# Global variables
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1 << (L - 1).bit_length()
    corr = np.fft.ifft(np.fft.fft(h_rev, fft_size) * np.fft.fft(b, fft_size)).real[:L]
    return corr

def evaluate_c5_precise(h):
    """Evaluate C5 using np.correlate for exact computation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def evaluate_c5_fast(h):
    """Fast evaluation using FFT"""
    corr = compute_corr_fft(h)
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x - lam, 0, 1)
        return np.sum(projected) - target_sum

    # Find initial interval for lambda
    lo = np.min(x) - 1.0
    hi = np.max(x) + 1.0
    
    # Ensure correct interval
    while f(lo) * f(hi) > 0:
        if f(lo) > 0:
            lo -= 1.0
        else:
            hi += 1.0
    
    # Binary search
    for _ in range(60):
        lam = (lo + hi) / 2
        if abs(f(lam)) < 1e-12:
            break
        if f(lam) > 0:
            lo = lam
        else:
            hi = lam
    
    result = np.clip(x - (lo + hi) / 2, 0, 1)
    return result

def compute_subgradient(h, corr):
    """Compute subgradient of max correlation"""
    n = len(h)
    max_val = np.max(corr)
    eps = 1e-12
    active_idx = np.where(corr >= max_val - eps)[0]
    
    # Initialize gradient with constant term for each active lag
    g = np.full(n, len(active_idx), dtype=float)
    
    for idx in active_idx:
        k = idx - (n - 1)
        if k >= 0:
            # Subtract h_{j + k} for j from 0 to n - k - 1
            g[:n - k] -= h[k:]
            # Subtract h_{j - k} for j from k to n - 1
            g[k:] -= h[:n - k]
        else:
            # k < 0, equivalent to k = -d where d > 0
            d = -k
            g[d:] -= h[:n - d]
            g[:n - d] -= h[d:]
    
    return g, max_val
