import numpy as np
from scipy.signal import correlate
import time

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def round_to_binary(h_cont, target_sum):
    """Round continuous solution to binary with correct sum"""
    # Sort indices by decreasing h_cont values
    sorted_indices = np.argsort(h_cont)[::-1]
    
    # Create binary vector with first target_sum indices set to 1
    binary_h = np.zeros_like(h_cont)
    binary_h[sorted_indices[:int(target_sum)]] = 1
    
    # Ensure exact sum
    ones_count = np.sum(binary_h)
    if ones_count > target_sum:
        # Need to flip some 1s to 0s
        flip_indices = sorted_indices[int(target_sum):ones_count]
        np.random.shuffle(flip_indices)
        binary_h[flip_indices[:int(ones_count - target_sum)]] = 0
    elif ones_count < target_sum:
        # Need to flip some 0s to 1s
        zero_indices = np.where(binary_h == 0)[0]
        np.random.shuffle(zero_indices)
        binary_h[zero_indices[:int(target_sum - ones_count)]] = 1
    
    return binary_h

def critical_lag_polish(h_binary, max_iter=10000):
    """Apply critical lag greedy polishing"""
    # Compute all correlations
    corr = compute_corr_fft(h_binary)
    max_corr = np.max(corr)
    
    for _ in range(max_iter):
        # Find worst lag
        max_idx = np.argmax(corr)
        # Convert linear index to lag
        lag = max_idx - (len(h_binary) - 1)
        
        # Find mismatched pairs that hurt this lag
        B = []
        for i in range(len(h_binary)):
            if i + lag < len(h_binary):
                if h_binary[i] == 1 and h_binary[i + lag] == 0:
                    B.append(i)
        
        if not B:
            break
            
        # Pick random element from B
        i = np.random.choice(B)
        
        # Find candidate zeros that do not create new mismatches
        Z = []
        for j in range(len(h_binary)):
            if h_binary[j] == 0:
                if j + lag >= len(h_binary) or h_binary[j + lag] == 1:
                    Z.append(j)
        
        if not Z:
            # If no good zero found, pick any zero
            zero_indices = np.where(h_binary == 0)[0]
            if len(zero_indices) > 0:
                j = np.random.choice(zero_indices)
            else:
                break
        else:
            j = np.random.choice(Z)
        
        # Swap
        h_binary[i], h_binary[j] = h_binary[j], h_binary[i]
        
        # Recompute correlation
        corr = compute_corr_fft(h_binary)
        new_max = np.max(corr)
        if new_max >= max_corr:
            # Undo if no improvement
            h_binary[i], h_binary[j] = h_binary[j], h_binary[i]
            break
        max_corr = new_max
        
        # Check if we've reached target
        if max_corr <= 22:  # target_M = floor(0.3807/dx) where dx = 2.0/120
            break
    
    return h_binary

def run(seed=42, budget_s=600, **kwargs):

    target_M = int(0.3807 / dx)  # target value

    # Project to feasible set
    h0 = project_to_sum_box(h0, target_sum)
    
    # Convex-concave procedure with cutting plane
    h_cont = h0.copy()
    max_lags = set([0, -24, -48, -72, -96, 24, 48, 72, 96])  # Initial lag set
    
    # Generate all lags to check
    all_lags = set(range(-n_points+1, n_points))
    
    # Iterative cutting plane approach
    iteration = 0
    max_iter = 100
    
    while time.time() - start_time < 0.7 * budget_s and iteration < max_iter:
        # Build the convex subproblem
        h = cp.Variable(n_points)
        T = cp.Variable()
        
        # Initialize constraints
        constraints = [0 <= h, h <= 1, cp.sum(h) == target_sum]
        
        # Add convex constraints for each active lag
        for ell in max_lags:
            # We rewrite I_ell(h) = sum(h_i) + 1/4*sum((h_i - h_{i+ell})^2) - 1/4*sum((h_i + h_{i+ell})^2)
            # For convexity, we use the concave part approximation:
            # -1/4 * sum((h_i + h_{i+ell})^2) <= -1/2 * sum((h_i^(t) + h_{i+ell}^(t)) * (h_i + h_{i+ell})) + const
            # Hence: I_ell(h) >= sum(h_i) + 1/4*sum((h_i - h_{i+ell})^2) - 1/2*sum((h_i^(t) + h_{i+ell}^(t)) * (h_i + h_{i+ell})) 
            # Linearization point: h_cont
            if ell >= 0:
                extra_term = cp.sum(cp.multiply(h_cont[:-ell] + h_cont[ell:], h[:-ell] + h[ell:]))
            else:
                extra_term = cp.sum(cp.multiply(h_cont[-ell:] + h_cont[:ell], h[-ell:] + h[:ell]))
            
            # Convex part
            convex_term = cp.sum(h) + 0.25 * cp.sum((h - cp.roll(h, ell))**2)  
            # Approximated concave part
            concave_term = 0.5 * extra_term
            
            # The constraint is that the convex surrogate is less than T
            constraints.append(convex_term - concave_term <= T)
        
        # Objective
        objective = cp.Minimize(T)
        prob = cp.Problem(objective, constraints)
        
        # Solve the problem
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_cont = h.value
                optimal_T = T.value
            else:
                break
        except:
            break
        
        # Evaluate true overlaps for all lags
        corr_full = compute_corr_fft(h_cont)
        actual_lags = np.argmax(np.abs(corr_full)) - (n_points - 1)
        max_I = np.max(corr_full)
        
        # Check for violations
        violated_lags = []
        for ell in all_lags:
            shifted_h = np.roll(h_cont, ell) if ell > 0 else np.roll(h_cont, ell)
            I_ell = np.sum(h_cont * (1 - shifted_h))
            if I_ell > optimal_T + 1e-6:
                violated_lags.append(ell)
        
        if not violated_lags:
            break
            
        # Add most violated lags
        for ell in violated_lags[:5]:  # Add a few at a time
            max_lags.add(ell)
        
        iteration += 1
    
    # Round to binary and polish
    h_binary = round_to_binary(h_cont, target_sum)
    
    # Run critical lag polishing
    h_binary = critical_lag_polish(h_binary)
    
    # Final evaluation
    corr_final = compute_corr_fft(h_binary)
    c5_bound = dx * np.max(corr_final)
    
    # If still not good enough, do additional polishing
    if c5_bound > 0.3807:
        for _ in range(5000):  # Add a bit more optimization
            corr = compute_corr_fft(h_binary)
            max_idx = np.argmax(corr)
            lag = max_idx - (n_points - 1)
            
            # Find mismatched pairs
            B = []
            for i in range(len(h_binary)):
                if i + lag < len(h_binary):
                    if h_binary[i] == 1 and h_binary[i + lag] == 0:
                        B.append(i)
            
            if not B:
                break
                
            i = np.random.choice(B)
            
            # Find zero for swapping
            Z = []
            for j in range(len(h_binary)):
                if h_binary[j] == 0:
                    if j + lag >= len(h_binary) or h_binary[j + lag] == 1:
                        Z.append(j)
            
            if Z:
                j = np.random.choice(Z)
                h_binary[i], h_binary[j] = h_binary[j], h_binary[i]
            else:
                break
                
            corr = compute_corr_fft(h_binary)
            if np.max(corr) <= 22:
                break
    
    # Final evaluation
    corr_final = compute_corr_fft(h_binary)
    final_c5 = dx * np.max(corr_final)
    
    return (h_binary.tolist(), float(final_c5), n_points)
