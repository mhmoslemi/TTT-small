import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def max_corr(h):
    """Compute maximum overlap correlation"""
    corr = correlate(h, 1-h, mode="full")
    return np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 value"""
    dx = 2.0 / len(h)
    return dx * max_corr(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    target_M = int(np.floor(0.3807 / dx))  # Should be 22

    current_sum = np.sum(h0)
    target_sum = n_points // 2
    
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        diff = current_sum - target_sum
        # Flip positions closest to 0.5
        flip_indices = np.argsort(np.abs(h_cont - 0.5))[:diff]
        h0[flip_indices] = 0
    elif current_sum < target_sum:
        # Need to flip some 0s to 1s
        diff = target_sum - current_sum
        # Flip positions closest to 0.5
        flip_indices = np.argsort(np.abs(h_cont - 0.5))[:diff]
        h0[flip_indices] = 1
    
    # Ensure exact sum
    current_sum = np.sum(h0)
    if current_sum != target_sum:
        # Adjust by flipping elements that don't change the sum constraint much
        flip_indices = np.where(h0 == 0)[0]
        if len(flip_indices) >= target_sum - current_sum:
            indices_to_flip = np.random.choice(flip_indices, target_sum - current_sum, replace=False)
            h0[indices_to_flip] = 1
        else:
            flip_indices = np.where(h0 == 1)[0]
            indices_to_flip = np.random.choice(flip_indices, current_sum - target_sum, replace=False)
            h0[indices_to_flip] = 0
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    
    # MILP approach
    print("Starting MILP optimization...")
    milp_time = min(260, budget_s * 0.4)  # Reserve time for local refinement

    fixed_ones = np.where(h0 == 1)[0]
    fixed_zeros = np.where(h0 == 0)[0]
    
    # Create variables for free positions only
    free_indices = np.setdiff1d(np.arange(n_points), np.union1d(fixed_ones, fixed_zeros))
    n_free = len(free_indices)
    
    # If almost all positions are already fixed, just use the greedy method
    if n_free < 5:
        print("Using greedy local search approach")
        # Direct greedy search on the current solution
        h = h0.copy()
        max_iter = int(1e6)

            corr = correlate(h, 1-h, mode="full")
            max_lag = np.argmax(corr)
            max_val = corr[max_lag]
            
            if max_val <= target_M:
                break
                
            # Find mismatched positions
            lag_idx = max_lag - (n_points - 1)
            mismatches = []
            if lag_idx >= 0:
                for i in range(n_points - lag_idx):
                    if h[i] == 1 and h[i + lag_idx] == 0:
                        mismatches.append(i)
            else:
                for i in range(-lag_idx, n_points):
                    if h[i] == 1 and h[i + lag_idx] == 0:
                        mismatches.append(i)

                break

            i = np.random.choice(mismatches)
            # Find candidates to swap with
            candidates = []
            for j in range(n_points):
                if h[j] == 0:
                    # Check if swapping would avoid creating new mismatches at current lag
                    if lag_idx >= 0:

                            candidates.append(j)
                    else:
                        if (i + lag_idx < 0) or (h[i + lag_idx] == 1):
                            candidates.append(j)

                j = np.random.choice(candidates)

                # Just pick a random zero

            h[i], h[j] = h[j], h[i]
            iter_count += 1

            new_c5 = evaluate_c5(h)

    h = h0.copy()

    iter_count = 0
    patience = 0

    while time.time() - start_time < budget_s and iter_count < max_iter:

            lag_idx = max_lag - (n_points - 1)

                # Positive lag case
                for i in range(n_points - lag_idx):
                    if h[i] == 1 and h[i + lag_idx] == 0:
                        mismatches.append(i)
            else:
                # Negative lag case
                for i in range(-lag_idx, n_points):
                    if h[i] == 1 and h[i + lag_idx] == 0:
                        mismatches.append(i)
            
            if len(mismatches) == 0:
                patience += 1
                if patience > max_patience:

            i = np.random.choice(mismatches)
            # Find candidates to swap with
            candidates = []
            for j in range(n_points):
                if h[j] == 0:
                    # Check if swapping would maintain the sum constraint and not increase the maximum overlap
                    # Create temporary new array
                    temp_h = h.copy()
                    temp_h[i], temp_h[j] = temp_h[j], temp_h[i]
                    
                    # Compute new correlation
                    new_corr = correlate(temp_h, 1-temp_h, mode="full")
                    new_max = np.max(new_corr)
                    
                    # Check if it doesn't make things worse
                    if new_max <= max_val:
                        candidates.append(j)
            
            if len(candidates) > 0:
                j = np.random.choice(candidates)

                    j = np.random.choice(zero_indices)
                else:
                    patience += 1
                    if patience > max_patience:
                        break

                    continue

            h[i], h[j] = h[j], h[i]
            iter_count += 1
            patience = 0
            
            # Update best if improvement
            new_c5 = evaluate_c5(h)
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                patience = 0
                
        except Exception as e:
            break

    return (best_h.tolist(), float(best_c5), n_points)
