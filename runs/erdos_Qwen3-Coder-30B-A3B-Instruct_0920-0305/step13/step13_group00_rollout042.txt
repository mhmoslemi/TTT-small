import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 120

    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints for h
    constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
    
    # Create y variables for each shift
    # For shift s, we have n_points - s - 1 variables
    y_vars = {}
    
    # Compute the constraints for each shift
    for s in range(n_points):
        if s == 0:
            # For shift 0, y[i,0] = h[i] * h[i] = h[i]
            y_vars[s] = cp.Variable(n_points)
            constraints.extend([
                y_vars[s] >= 0,
                y_vars[s] <= h,
                y_vars[s] >= 2*h - 1,
                y_vars[s] == h  # Since y[i,0] = h[i]^2 = h[i] for h[i] in [0,1]
            ])
        else:
            # For shift s, we have n_points - s variables
            y_vars[s] = cp.Variable(n_points - s)
            # Add McCormick constraints for each pair
            for i in range(n_points - s):
                constraints.extend([
                    y_vars[s][i] >= 0,
                    y_vars[s][i] <= h[i],
                    y_vars[s][i] <= h[i+s],
                    y_vars[s][i] >= h[i] + h[i+s] - 1
                ])
    
    # Add correlation constraints
    # C'_s = (target_sum - P_s) - sum(y_{i,s})
    # We want to minimize t such that C'_s <= t for all s
    for s in range(n_points):
        if s == 0:
            # For s = 0, overlap sum is sum of h[i] 
            overlap_s = cp.sum(y_vars[s])
        else:
            # For s > 0, overlap sum is sum of y_{i,s}
            overlap_s = cp.sum(y_vars[s])
        
        # Prefix sum: P_s = sum_{i=0}^{s-1} h[i]
        if s == 0:
            prefix_sum = 0
        else:
            prefix_sum = cp.sum(h[:s])

        correlation_s = (target_sum - prefix_sum) - overlap_s

        constraints.append(correlation_s <= t)

    objective = cp.Minimize(t)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - np.array(h)
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    dx = 2.0 / len(h)
    return dx * np.max(corr)

def local_refinement(h_init, target_sum, dx, start_time, budget_s):
    """Refine solution with gradient-based optimization"""
    h = h_init.copy()
    
    # Simple gradient-based refinement
    max_iter = 1000
    for i in range(max_iter):
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] = min(h[j] + eps, 1.0)
            h_plus = project_to_sum_box(h_plus, target_sum)
            
            h_minus = h.copy()
            h_minus[j] = max(h[j] - eps, 0.0)
            h_minus = project_to_sum_box(h_minus, target_sum)
            
            c5_plus = evaluate_c5(h_plus)
            c5_minus = evaluate_c5(h_minus)
            
            grad[j] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update with gradient
        step_size = 0.01
        h_new = h - step_size * grad
        h_new = project_to_sum_box(h_new, target_sum)
        
        c5_new = evaluate_c5(h_new)
        if c5_new < evaluate_c5(h):
            h = h_new
        else:
            break
    
    return h
