import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    size = 1 << (L - 1).bit_length()
    A = np.fft.fft(h_rev, size)
    B = np.fft.fft(b, size)
    corr = np.fft.ifft(A * B).real[:L]
    return corr

def evaluate_c5_fast(h):
    """Fast evaluation using FFT"""
    corr = compute_corr_fft(h)
    return np.max(corr)

def evaluate_c5_precise(h):
    """Precise evaluation using np.correlate"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x - lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = np.min(x) - 1.0, np.max(x) + 1.0
    tol = 1e-12
    
    for _ in range(60):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x - (a + b)/2, 0, 1)
    return result

def compute_subgradient(h, corr):
    """Compute subgradient for the max correlation"""
    n = len(h)
    max_val = np.max(corr)
    eps = 1e-12
    active_idx = np.where(corr >= max_val - eps)[0]
    
    # Initialize gradient with constant term: 1 per active lag per component
    g = np.full(n, len(active_idx), dtype=float)
    
    for idx in active_idx:
        k = idx - (n - 1)
        if k >= 0:
            # Subtract h_{j + k} for j from 0 to n - k - 1
            g[:n - k] -= h[k:]
            # Subtract h_{j - k} for j from k to n - 1
            g[k:] -= h[:n - k]
        else:
            d = -k
            # For negative k, equivalent to positive shifts
            g[d:] -= h[:n - d]
            g[:n - d] -= h[d:]
    
    return g, max_val

def generate_sin_initial():
    """Generate sinusoidal initial solution"""
    x = (np.arange(n_points) + 0.5) * dx
    a = np.random.rand() * 0.5
    phase = np.random.rand() * 2 * np.pi
    h = 0.5 + a * np.sin(np.pi * x + phase)
    h = project_to_sum_box(h, target_sum)
    return h

    candidates = [h0]

        candidates.append(generate_sin_initial())
    
    # Random starts
    for _ in range(5):
        h_random = np.random.rand(n_points)
        h_random = project_to_sum_box(h_random, target_sum)
        candidates.append(h_random)
    
    # Subgradient descent for each candidate
    for i, h_start in enumerate(candidates):
        if time.time() - start_time > budget_s - 10:
            break
            
        h = h_start.copy()
        current_c5 = evaluate_c5_fast(h)
        max_corr = current_c5
        
        # Run optimization
        max_iters = 10000
        base_lr = 0.3
        
        for iter_num in range(max_iters):
            if time.time() - start_time > budget_s - 5:
                break
                
            corr = compute_corr_fft(h)
            g, max_corr = compute_subgradient(h, corr)
            
            # Compute step size
            step = base_lr / np.sqrt(iter_num + 1)
            
            # Compute direction
            norm_g = np.linalg.norm(g)
            if norm_g > 0:
                direction = g / norm_g
            else:
                direction = np.zeros_like(g)
            
            # Update
            h_new = h - step * direction
            h_new = project_to_sum_box(h_new, target_sum)

            new_corr = np.max(compute_corr_fft(h_new))
            new_c5 = new_corr * dx
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
            
            # Accept improvement
            if new_corr < max_corr:
                h = h_new
                max_corr = new_corr
    
    # Final local search with swaps
    h = best_h.copy()
    current_c5 = evaluate_c5_precise(h)
    
    for _ in range(2000):
        if time.time() - start_time > budget_s - 2:
            break

        i, j = np.random.randint(0, n_points, 2)
        h_temp = h.copy()

                best_h = h.copy()
