# Global variables
TARGET_SUM = None

dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Simple projection: adjust values to meet sum constraint
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x
    
    # Linear adjustment
    adjustment = (target_sum - current_sum) / len(x)
    result = x + adjustment
    
    # Clamp to [0,1]
    result = np.clip(result, 0, 1)
    
    # If still not exact, use more precise method
    if abs(np.sum(result) - target_sum) > 1e-8:
        # Binary search approach
        def f(lam):
            projected = np.clip(x + lam, 0, 1)
            return np.sum(projected) - target_sum

        a, b = -1.0, 1.0
        while f(a) * f(b) > 0:
            a -= 0.1
            b += 0.1
        
        tol = 1e-10
        for _ in range(50):
            m = (a + b) / 2
            fm = f(m)
            if abs(fm) < tol:
                break
            if fm * f(a) < 0:
                b = m
            else:
                a = m
        
        result = np.clip(x + (a + b)/2, 0, 1)
    
    return result
