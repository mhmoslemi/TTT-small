import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
from math import floor

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_convex_subproblem(h_current, lags, n_points, dx):
    """Solve the convex subproblem using CVXPY"""
    h = cp.Variable(n_points)
    T = cp.Variable()
    
    # Objective: minimize T
    objective = cp.Minimize(T)
    
    # Constraints
    constraints = []
    
    # Box constraints and sum constraint
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    constraints.append(cp.sum(h) == n_points / 2)
    
    # Convex constraints
    for ell in lags:
        # I_ell^t(h) = sum(h_i) + 1/4 * sum(h_i - h_{i+ell})^2 - 1/2 * sum(h_i^t + h_{i+ell}^t) * (h_i + h_{i+ell})
        # = sum(h_i) + 1/4 * sum((h_i - h_{i+ell})^2) - 1/2 * sum((h_i^t + h_{i+ell}^t) * (h_i + h_{i+ell}))
        # = sum(h_i) + 1/4 * sum(h_i^2 - 2*h_i*h_{i+ell} + h_{i+ell}^2) - 1/2 * sum(h_i^t*h_i + h_i^t*h_{i+ell} + h_{i+ell}^t*h_i + h_{i+ell}^t*h_{i+ell})
        # = sum(h_i) + 1/4 * sum((h_i - h_{i+ell})^2) - 1/2 * sum((h_i^t + h_{i+ell}^t) * (h_i + h_{i+ell}))
        # = sum(h_i) + 1/4 * sum(h_i^2 + h_{i+ell}^2 - 2*h_i*h_{i+ell}) 
        # - 1/2 * sum(h_i^t*h_i + h_i^t*h_{i+ell} + h_{i+ell}^t*h_i + h_{i+ell}^t*h_{i+ell}) 
        # = sum(h_i) + 1/4 * sum(h_i*h_{i+ell} - h_i*h_{i+ell}) 
        # - 1/2 * sum((h_i^t + h_{i+ell}^t) * (h_i + h_{i+ell}))
        # = sum(h_i) - 1/2 * sum((h_i^t + h_{i+ell}^t) * (h_i + h_{i+ell})) + 1/4 * sum(h_i^2 + h_{i+ell}^2)
        # = sum(h_i) - 1/2 * (sum(h_i^t*h_i) + sum(h_i^t*h_{i+ell}) + sum(h_{i+ell}^t*h_i) + sum(h_{i+ell}^t*h_{i+ell})) + 1/4*sum(h_i^2 + h_{i+ell}^2)
        # = sum(h_i) - 1/2 * (sum(h_i^t*h_i) + sum(h_i^t*h_{i+ell}) + sum(h_{i+ell}^t*h_i) + sum(h_{i+ell}^t*h_{i+ell})) + 1/4*sum(h_i^2 + h_{i+ell}^2)
        # The constraint is I_ell^t(h) <= T
        
        left_sum = cp.sum(h)
        concave_term = 0
        for i in range(n_points):
            j = i + ell
            if 0 <= j < n_points:
                concave_term += (h_current[i] + h_current[j]) * (h[i] + h[j])
        
        # Quadratic part
        quad_term = 0
        for i in range(n_points):
            j = i + ell
            if 0 <= j < n_points:
                quad_term += (h[i] - h[j])**2
        
        # Full expression 
        # = sum(h_i) + quad_term/4 - concave_term/2
        # <= T
        # So: sum(h_i) + quad_term/4 - concave_term/2 <= T
        
        constraints.append(left_sum + quad_term/4 - concave_term/2 <= T)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    try:
        # Use ECOS solver which is fast for small QPs
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value, T.value
    except:
        pass
    
    # Fallback to a relaxed approach if CVXPY fails
    h_sol = h_current.copy()
    t_sol = evaluate_c5(h_sol)
    return h_sol, t_sol

def get_initial_lags(n):
    """Get initial set of lags for cutting plane"""
    lags = [0]
    for k in range(1, 5):
        v = k * n // 5
        lags.extend([-v, v])
    return lags

def is_violated(h, T, ell, threshold=1e-6):
    """Check if a lag is actually violated"""
    corr = compute_corr_fft(h)
    max_corr = np.max(corr)
    idx = ell + (len(corr) - 1) // 2
    return corr[idx] > T + threshold

def get_violating_lags(h, T, all_lags, threshold=1e-6):
    """Get lags that are violated"""
    corr = compute_corr_fft(h)
    violating = []
    for ell in all_lags:
        idx = ell + (len(corr) - 1) // 2
        if corr[idx] > T + threshold and ell not in violating:
            violating.append(ell)
    return violating

def rounding_phase(h_cont, n_points, dx):
    """Perform rounding to get binary solution"""
    best_h = None
    best_c5 = float('inf')
    
    # Multiple draws
    for _ in range(30):
        # Sort indices by decreasing h_cont values
        indices = np.argsort(h_cont)[::-1]
        h = np.zeros(n_points)
        h[indices[:n_points//2]] = 1
        
        # Apply small random flips to diversify
        num_flips = min(4, n_points//10)
        flip_indices = np.random.choice(n_points, num_flips, replace=False)
        for i in flip_indices:
            h[i] = 1 - h[i]
            
        # Project to sum constraint
        h = project_to_sum_box(h, n_points / 2)
        
        # Evaluate
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    if best_h is None:
        return h_cont.copy(), evaluate_c5(h_cont)
    
    return best_h, best_c5

def local_search_polish(h_binary, dx, max_time=200):
    """Critical-lag greedy polishing"""
    start_time = time.time()
    best_h = h_binary.copy()
    best_c5 = evaluate_c5(best_h)
    
    target_M = int(0.3807 / dx)
    
    while time.time() - start_time < max_time and best_c5 > target_M:
        corr = compute_corr_fft(best_h)
        max_ell = np.argmax(corr)
        max_val = corr[max_ell]
        
        if max_val <= target_M:
            break
            
        # Get the actual lag from index
        actual_ell = max_ell - (len(corr) - 1) // 2
        
        # Find mismatched pairs that hurt this lag
        B = []
        for i in range(len(best_h)):
            j = i + actual_ell
            if 0 <= j < len(best_h):
                if best_h[i] == 1 and best_h[j] == 0:
                    B.append(i)
        
        if not B:
            break
            
        # Choose a random element from B
        i = np.random.choice(B)
        
        # Find eligible positions to swap to
        Z = []
        for j in range(len(best_h)):
            j_shift = j + actual_ell
            if 0 <= j_shift < len(best_h):
                if best_h[j] == 0 and best_h[j_shift] == 1:
                    Z.append(j)
        
        if Z:
            j = np.random.choice(Z)
        else:
            # If no valid swap, just choose a zero
            zero_indices = np.where(best_h == 0)[0]
            if len(zero_indices) > 0:
                j = np.random.choice(zero_indices)
            else:
                break
                
        # Perform swap
        best_h[i], best_h[j] = best_h[j], best_h[i]
        
        # Recompute corr
        corr = compute_corr_fft(best_h)
        max_val = np.max(corr)
        best_c5 = max_val
        
    return best_h, best_c5
