import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT correlation"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

            corr = np.correlate(h, 1-h, mode='full')
            
            # Find worst shift
            max_corr_idx = np.argmax(corr)

                if i + worst_shift >= 0 and i + worst_shift < len(h):
                    # In the current shift k (using worst_shift), h_i contributes to correlation

                    grad[i] = -(1 - h[i + worst_shift])
                else:

            # Move in gradient direction (but avoid boundary issues)
            h_new = project_to_sum_box(h - step_size * grad, TARGET_SUM)
            c5_new = evaluate_c5(h_new)
            
            if c5_new < evaluate_c5(h):
                h = h_new
                if c5_new < best_c5:

                    best_h = h.copy()

        c5_restart = evaluate_c5(h_restart)
        
        if c5_restart < best_c5:
            best_c5 = c5_restart
            best_h = h_restart.copy()

    if best_h is None:
        best_h = h0.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
