import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, Binary, sum as cp_sum, maximum as cp_maximum

def max_corr(h):
    """Compute maximum correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 120
    dx = 2.0 / n_points
    target_M = int(np.floor(0.3807 / dx))  # 22

    # Round to binary and adjust sum to 60
    h0 = (h_cont >= 0.5).astype(int)
    current_sum = np.sum(h0)
    
    if current_sum > 60:
        # Need to flip some 1s to 0s
        diff = current_sum - 60
        # Get indices of 1s, sorted by how close they are to 0.5
        ones_indices = np.where(h0 == 1)[0]
        # Sort by distance from 0.5 (ascending)
        distances = np.abs(h_cont[ones_indices] - 0.5)
        sorted_indices = np.argsort(distances)
        flip_indices = ones_indices[sorted_indices[:diff]]
        h0[flip_indices] = 0
    elif current_sum < 60:
        # Need to flip some 0s to 1s
        diff = 60 - current_sum
        # Get indices of 0s, sorted by how close they are to 0.5
        zeros_indices = np.where(h0 == 0)[0]
        # Sort by distance from 0.5 (ascending)
        distances = np.abs(h_cont[zeros_indices] - 0.5)
        sorted_indices = np.argsort(distances)
        flip_indices = zeros_indices[sorted_indices[:diff]]
        h0[flip_indices] = 1
    
    # Create warm-start binary vector
    h_fixed = h0.copy()
    
    # Identify fixed positions
    fixed_ones = np.where(h_fixed == 1)[0]
    fixed_zeros = np.where(h_fixed == 0)[0]
    free_indices = np.setdiff1d(np.arange(n_points), np.concatenate([fixed_ones, fixed_zeros]))
    
    # Check if we already have a good solution
    if max_corr(h_fixed) <= target_M:
        return (h_fixed.tolist(), float(dx * max_corr(h_fixed)), n_points)
    
    # Build MILP problem
    # Only consider lags where we might exceed the target
    l_max = n_points - target_M  # 98
    valid_lags = list(range(-l_max, l_max + 1))
    
    # Define binary variables for free positions
    h_vars = Variable(n_points, boolean=True)
    
    # Set fixed values
    for i in fixed_ones:
        h_vars[i].value = 1
    for i in fixed_zeros:
        h_vars[i].value = 0
    
    # Create auxiliary variables for products
    # For each valid lag, create a binary variable w[i,ℓ] = h[i] * h[i+ℓ]
    w_vars = {}
    constraints = []
    
    # Sum constraint
    constraints.append(cp_sum(h_vars) == 60)
    
    # Add constraints for the product terms
    for ell in valid_lags:
        if ell >= 0:
            # For positive lags: indices go from 0 to n-ell-1
            for i in range(n_points - ell):
                if i in free_indices or (i + ell) in free_indices:
                    w_var = Variable(boolean=True)
                    w_vars[(i, ell)] = w_var
                    # Add constraints for the product
                    constraints.append(w_var <= h_vars[i])
                    constraints.append(w_var <= h_vars[i + ell])
                    constraints.append(w_var >= h_vars[i] + h_vars[i + ell] - 1)
        else:
            # For negative lags: indices go from -ell to n-1
            for i in range(-ell, n_points):
                if i in free_indices or (i + ell) in free_indices:
                    w_var = Variable(boolean=True)
                    w_vars[(i, ell)] = w_var
                    # Add constraints for the product
                    constraints.append(w_var <= h_vars[i])
                    constraints.append(w_var <= h_vars[i + ell])
                    constraints.append(w_var >= h_vars[i] + h_vars[i + ell] - 1)
    
    # Create M variable and constraints
    M = Variable()
    
    # Add constraints for each lag
    for ell in valid_lags:
        if ell >= 0:
            # Sum h[i] - w[i,ell] for i=0,...,n-ell-1
            expr = cp_sum(h_vars[i] for i in range(n_points - ell)) - \
                   cp_sum(w_vars[(i, ell)] for i in range(n_points - ell))
        else:
            # Sum h[i] - w[i,ell] for i=-ell,...,n-1
            expr = cp_sum(h_vars[i] for i in range(-ell, n_points)) - \
                   cp_sum(w_vars[(i, ell)] for i in range(-ell, n_points))
        constraints.append(expr <= M)
    
    # Optimize
    prob = Problem(Minimize(M), constraints)
    
    # We have ~260 seconds for the MILP
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:  # Leave some margin
        milp_time = min(260, remaining_time - 10)  # Set lower bound and free up time for refinement
        try:
            prob.solve(solver=cp.CBC, cbc_params={'maxTime': milp_time})
            if prob.status in ['optimal', 'optimal_inaccurate']:
                # Get the solution
                sol_vals = np.array([float(h_vars[i].value) for i in range(n_points)])
                h_milp = np.round(sol_vals).astype(int)
                current_M = int(np.ceil(max_corr(h_milp)))
                if current_M <= target_M:
                    return (h_milp.tolist(), float(dx * current_M), n_points)
            else:
                # Use heuristic solution as fallback
                h_milp = h_fixed
                current_M = int(np.ceil(max_corr(h_milp)))
        except:
            # Fallback to the heuristic
            h_milp = h_fixed
            current_M = int(np.ceil(max_corr(h_milp)))
    else:
        h_milp = h_fixed
        current_M = int(np.ceil(max_corr(h_milp)))
    
    # Post-processing local search - critical lag climbing algorithm
    h_current = h_milp.copy()
    current_M = int(np.ceil(max_corr(h_current)))
    
    def get_mismatch_positions(ell):
        """Get indices where h[i]=1 but h[i+ell]=0"""
        if ell >= 0:
            mismatch = np.where((h_current[:n_points-ell] == 1) & (h_current[ell:] == 0))[0]
        else:
            mismatch = np.where((h_current[-ell:] == 1) & (h_current[:-ell] == 0))[0]
        return mismatch
    
    # Remaining time for local search
    remaining_time = budget_s - (time.time() - start_time)
    max_swaps = min(10000, max(1000, int(remaining_time * 100)))
    
    # Function to compute correlation
    def compute_corr_fft(h):
        h_rev = h[::-1]
        b = 1 - h
        L = 2 * len(h) - 1
        corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
        return corr
    
    # Refine using critical lag local search
    for swap_count in range(max_swaps):
        if time.time() - start_time > budget_s - 2:
            break
            
        if current_M <= target_M:
            break
            
        # Find worst lag
        corr = compute_corr_fft(h_current)
        ell_star = np.argmax(corr) - (n_points - 1)
        
        # Get mismatched positions for this lag
        mismatch = get_mismatch_positions(ell_star)
        
        if len(mismatch) == 0:
            continue
            
        # Choose one of the mismatches
        i = np.random.choice(mismatch)
        
        # Find suitable swap position j
        candidates = []
        if ell_star >= 0:
            # Look for positions where h[j+ell_star] = 1 or j+ell_star >= n_points
            for j in range(n_points):
                if h_current[j] == 0:
                    if j + ell_star >= n_points or h_current[j + ell_star] == 1:
                        candidates.append(j)
        else:
            # Look for positions where h[j+ell_star] = 1 or j+ell_star < 0
            for j in range(n_points):
                if h_current[j] == 0:
                    if j + ell_star < 0 or h_current[j + ell_star] == 1:
                        candidates.append(j)
        
        if not candidates:
            # If no suitable candidates, pick a random zero
            zeros_positions = np.where(h_current == 0)[0]
            if len(zeros_positions) > 0:
                j = np.random.choice(zeros_positions)
                # Accept even if it doesn't improve (small chance of escape)
                h_current[i], h_current[j] = h_current[j], h_current[i]
                current_M = int(np.ceil(max_corr(h_current)))
                if current_M <= target_M:
                    break
        else:
            j = np.random.choice(candidates)
            # Perform the swap
            h_current[i], h_current[j] = h_current[j], h_current[i]
            
            # Recompute
            current_M = int(np.ceil(max_corr(h_current)))
            
            if current_M <= target_M:
                break
    
    # Final evaluation
    final_M = int(np.ceil(max_corr(h_current)))
    c5_value = dx * final_M
    
    return (h_current.tolist(), float(c5_value), n_points)
