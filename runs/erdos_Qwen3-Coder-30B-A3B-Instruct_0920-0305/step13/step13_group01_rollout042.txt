import numpy as np

# Global variables
TARGET_SUM = None

dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_descent(h, learning_rate, max_iter, min_lr, start_time, budget_s):
    for i in range(max_iter):
        if time.time() >= start_time + budget_s:
            break
            
        # Compute gradient via finite differences
        eps = 1e-4
        base_c5 = evaluate_c5(h)
        grad = np.zeros_like(h)
        
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_to_sum_box(h_plus, TARGET_SUM)
            c5_plus = evaluate_c5(h_plus)
            grad[j] = (c5_plus - base_c5) / eps
            
        # Gradient step
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Update learning rate
        if evaluate_c5(h_new) < evaluate_c5(h):
            h = h_new
        else:
            learning_rate *= 0.99
        
        if learning_rate < min_lr:
            break
            
    return h, evaluate_c5(h)

def simulated_annealing(h, temp, min_temp, alpha, max_iter, start_time, budget_s):
    current_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        if time.time() >= start_time + budget_s:
            break
            
        # Create neighbor by perturbing a few points
        neighbor = h.copy()
        num_changes = max(1, np.random.randint(1, 20))
        indices = np.random.choice(n_points, num_changes, replace=False)
        
        for idx in indices:
            neighbor[idx] += np.random.normal(0, 0.05)
            neighbor[idx] = np.clip(neighbor[idx], 0, 1)
        
        neighbor = project_to_sum_box(neighbor, TARGET_SUM)
        new_c5 = evaluate_c5(neighbor)
        
        # Accept or reject based on Metropolis criterion
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / temp):
            h = neighbor
            current_c5 = new_c5
            
        # Cool down
        temp *= alpha
        if temp < min_temp:
            temp = min_temp
            
    return h, current_c5

def local_search_swaps(h, max_swaps, start_time, budget_s):
    current_c5 = evaluate_c5(h)
    
    for i in range(max_swaps):
        if time.time() >= start_time + budget_s:
            break
            
        # Try a random swap
        i1, i2 = np.random.choice(n_points, 2, replace=False)
        temp_h = h.copy()
        temp_h[i1], temp_h[i2] = temp_h[i2], temp_h[i1]
        temp_h = project_to_sum_box(temp_h, TARGET_SUM)
        
        temp_c5 = evaluate_c5(temp_h)
        if temp_c5 < current_c5:
            h = temp_h
            current_c5 = temp_c5
            
    return h, current_c5

def differential_evolution(h, popsize, max_gen, start_time, budget_s):
    # Create initial population
    population = [project_to_sum_box(np.random.rand(n_points), TARGET_SUM) 
                  for _ in range(popsize)]
    population[0] = h.copy()  # Ensure the best solution is in the population
    
    for gen in range(max_gen):
        if time.time() >= start_time + budget_s:
            break
            
        for i in range(popsize):
            # Mutation and crossover
            r1, r2, r3 = np.random.choice(popsize, 3, replace=False)
            mutant = population[r1] + 0.8 * (population[r2] - population[r3])
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            child = population[i].copy()
            jrand = np.random.randint(n_points)
            for j in range(n_points):
                if np.random.rand() < 0.7 or j == jrand:
                    child[j] = mutant[j]
            
            # Repair
            child = project_to_sum_box(child, TARGET_SUM)
            
            # Selection
            child_c5 = evaluate_c5(child)
            if child_c5 < evaluate_c5(population[i]):
                population[i] = child
    
    # Return best solution from population
    best_idx = np.argmin([evaluate_c5(p) for p in population])
    return population[best_idx], evaluate_c5(population[best_idx])
