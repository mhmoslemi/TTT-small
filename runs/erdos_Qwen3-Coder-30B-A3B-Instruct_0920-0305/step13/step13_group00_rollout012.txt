import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Create CVXPY variables
    h = Variable(n_points)
    t = Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == target_sum,
        t >= 0
    ]
    
    # For each shift s, create y variables representing h_i * h_{i+s}
    # And add McCormick constraints
    for s in range(1, n_points):
        # Create y variables for this shift
        y = Variable(n_points - s)
        
        # Add McCormick constraints for each y
        for i in range(n_points - s):
            constraints.extend([
                y[i] >= 0,
                y[i] <= h[i],
                y[i] <= h[i + s],
                y[i] >= h[i] + h[i + s] - 1
            ])
        
        # Add constraint that this shift's overlap is bounded by the expression
        # C'_s = (S - P_s) - sum(y)
        # where P_s = sum of first s elements of h
        prefix_sum = sum(h[:s])
        overlap = sum(y)
        constraints.append(overlap + prefix_sum >= target_sum - t)
    
    # Handle shift 0 separately
    y0 = Variable(n_points)
    for i in range(n_points):
        constraints.extend([
            y0[i] >= 0,
            y0[i] <= h[i],
            y0[i] >= 2 * h[i] - 1
        ])
    
    # For shift 0, we have correlation = sum(h) - sum(h_i^2) = target_sum - sum(y0)
    # We want to minimize max correlation, so we bound 
    # C'_0 = target_sum - sum(y0) <= t
    # Which gives sum(y0) >= target_sum - t
    constraints.append(sum(y0) >= target_sum - t)
    
    # Objective
    objective = Minimize(t)
    
    # Create and solve problem
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=ECOS, verbose=False, max_iters=1000)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h_opt = h.value
            t_opt = t.value
            c5_bound = dx * t_opt
            return (h_opt.tolist(), float(c5_bound), n_points)
    except:
        pass
    
    # Fallback to previous solution if CVXPY fails
    h0 = np.array(initial_h_values)
    h0 = h0 / np.sum(h0) * target_sum
    h0 = np.clip(h0, 0, 1)
    
    # Evaluate C5 for the initial solution
    corr = np.correlate(h0, 1-h0, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (h0.tolist(), float(c5_bound), n_points)
