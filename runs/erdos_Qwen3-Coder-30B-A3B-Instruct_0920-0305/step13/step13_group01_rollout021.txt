import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    # Use next power of 2 for FFT
    fft_size = 1 << (L - 1).bit_length()
    corr = np.fft.ifft(np.fft.fft(h_rev, fft_size) * np.fft.fft(b, fft_size)).real
    return corr[:L]

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT, returns raw correlation max"""
    corr = compute_corr_fft(h)
    return np.max(corr)

def evaluate_c5_precise(h):
    """Precise evaluation using np.correlate for final validation"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Initial bounds
    a, b = np.min(x) - 1.0, np.max(x) + 1.0
    # Handle edge cases where bounds might be too extreme
    if a > b:
        a, b = b, a
    
    # Binary search
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_subgradient(h, corr):
    """Compute subgradient of max correlation"""
    n = len(h)
    max_val = np.max(corr)
    eps = 1e-12
    active_idx = np.where(corr >= max_val - eps)[0]
    
    # Initialize gradient with constant term for each active lag
    m = len(active_idx)
    g = np.full(n, m, dtype=float)
    
    for idx in active_idx:
        k = idx - (n - 1)
        if k >= 0:
            # Subtract h_{j + k} for j from 0 to n - k - 1
            g[:n - k] -= h[k:]
            # Subtract h_{j - k} for j from k to n - 1
            g[k:] -= h[:n - k]
        else:
            # k < 0
            d = -k
            g[d:] -= h[:n - d]
            g[:n - d] -= h[d:]
    
    return g, max_val

def generate_sin_initial():
    """Generate sinusoidal initial solution"""
    x = (np.arange(n_points) + 0.5) * dx
    # Random amplitude between 0 and 0.5 to ensure values in [0,1]
    amplitude = np.random.rand() * 0.5
    phase = np.random.rand() * 2 * np.pi
    h = 0.5 + amplitude * np.sin(np.pi * x + phase)
    return project_to_sum_box(h, target_sum)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with repeated initial values if available
    if 'initial_h_values' in globals() and len(initial_h_values) == 120:
        h0 = np.repeat(initial_h_values, 2)
    else:
        h0 = np.ones(n_points) * 0.5
    
    h0 = project_to_sum_box(h0, target_sum)
    best_c5 = evaluate_c5_precise(h0)
    best_h = h0.copy()
    
    # Generate various initial solutions
    initial_solutions = [h0]
    for _ in range(4):
        initial_solutions.append(generate_sin_initial())
    
    # Subgradient descent with random restarts
    max_iter_per_restart = 15000
    restarts = 0
    
    for i, h_start in enumerate(initial_solutions):
        if time.time() - start_time > budget_s - 5:
            break
            
        h = h_start.copy()
        current_c5 = evaluate_c5_fast(h)
        current_max_corr = np.max(compute_corr_fft(h))
        
        for iter_num in range(max_iter_per_restart):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute correlation and subgradient
            corr = compute_corr_fft(h)
            g, max_corr = compute_subgradient(h, corr)
            
            # Compute adaptive step size
            step_size = 0.3 / np.sqrt(iter_num + 1)
            
            # Apply step
            h_new = h - step_size * g
            
            # Project onto feasible set
            h_new = project_to_sum_box(h_new, target_sum)
            
            # Evaluate new solution
            new_corr = np.max(compute_corr_fft(h_new))
            new_c5 = new_corr * dx
            
            if new_c5 <= current_c5:
                h = h_new
                current_c5 = new_c5
                current_max_corr = new_corr
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
            else:
                # Move with probability based on temperature
                if np.random.rand() < 0.01:  # Small probability
                    h = h_new
                else:
                    h = h  # Keep current solution
            
            restarts += 1
    
    # Perform final local search with swaps to escape plateaus
    current_h = best_h.copy()
    current_c5 = evaluate_c5_precise(current_h)
    
    for _ in range(5000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try a random swap
        i, j = np.random.randint(0, n_points, 2)
        temp_h = current_h.copy()
        temp_h[i], temp_h[j] = temp_h[j], temp_h[i]
        temp_h = project_to_sum_box(temp_h, target_sum)
        
        temp_c5 = evaluate_c5_precise(temp_h)
        if temp_c5 < current_c5:
            current_h = temp_h
            current_c5 = temp_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        elif np.random.rand() < 0.001:
            # Small probability of accepting worse solutions
            current_h = temp_h
            current_c5 = temp_c5
    
    return (best_h.tolist(), float(best_c5), n_points)
