import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_convex_subproblem(h_current, lags_used):
    """Solve the convex subproblem for given lags"""
    h = cp.Variable(n_points)
    T = cp.Variable()
    
    # Constraints
    constraints = [0 <= h, h <= 1, cp.sum(h) == TARGET_SUM]
    
    # Add convex surrogate constraints
    for ell in lags_used:
        # Compute the index ranges
        if ell >= 0:
            left_idx = slice(0, n_points - ell)
            right_idx = slice(ell, n_points)
        else:
            left_idx = slice(-ell, n_points)
            right_idx = slice(0, n_points + ell)
        
        # Convex quadratic part
        convex_term = cp.sum(h) + 0.25 * cp.sum((h[left_idx] - h[right_idx])**2)
        
        # Linear part from concave approximation
        linear_term = 0.5 * cp.sum((h_current[left_idx] + h_current[right_idx]) * (h[left_idx] + h[right_idx]))
        
        # Add constraint
        constraints.append(convex_term - linear_term <= T)
    
    # Objective
    objective = cp.Minimize(T)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value, T.value
    except:
        pass
    
    return None, None

def find_violating_lags(h_opt, t_opt, all_lags, tolerance=1e-6):
    """Find all lags that violate the surrogate"""
    corr = compute_corr_fft(h_opt)
    prob = np.max(corr) > t_opt + tolerance
    violating = []
    if prob:
        for i, val in enumerate(corr):
            if val > t_opt + tolerance:
                # Convert linear index to lag
                lag = i - (n_points - 1)
                violating.append(lag)
    return violating

def rounding_phase(h_cont):
    """Multiple random draws and selection"""
    best_h = None
    best_c5 = float('inf')
    
    for _ in range(30):
        # Sort indices by h_cont values
        sorted_indices = np.argsort(-h_cont)
        # Set first TARGET_SUM to 1, rest to 0
        h_binary = np.zeros(n_points)
        h_binary[sorted_indices[:int(TARGET_SUM)]] = 1
        
        # Project back to correct sum
        h_binary = project_to_sum_box(h_binary, TARGET_SUM)
        
        # Evaluate
        c5 = c5_from_corr(compute_corr_fft(h_binary))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_binary.copy()
    
    return best_h, best_c5

def critical_lag_polish(h_binary, target_c5):
    """Critical lag greedy local search"""
    corr = compute_corr_fft(h_binary)
    max_lag = np.argmax(corr)
    max_val = corr[max_lag]
    
    start_time = time.time()
    timeout = 200  # seconds
    
    while max_val > target_c5 and time.time() - start_time < timeout:
        # Find the worst lag
        worst_lag = np.argmax(corr)
        # Find mismatched pairs: 1 at i, 0 at i+worst_lag
        mismatches = []
        for i in range(n_points):
            if h_binary[i] == 1 and i + worst_lag < n_points and h_binary[i + worst_lag] == 0:
                mismatches.append(i)
        
        if not mismatches:
            break
            
        # Pick a random mismatch
        i = np.random.choice(mismatches)
        # Find candidate zero positions with no new mismatches
        candidates = []
        for j in range(n_points):
            if h_binary[j] == 0:
                # Check if swapping creates new mismatch at worst_lag
                if j + worst_lag >= n_points or h_binary[j + worst_lag] == 1:
                    candidates.append(j)
        if candidates:
            j = np.random.choice(candidates)
        else:
            # Fallback: pick random zero
            zero_indices = np.where(h_binary == 0)[0]
            j = np.random.choice(zero_indices)
        
        # Perform the swap
        h_binary[i], h_binary[j] = h_binary[j], h_binary[i]
        
        # Recompute
        corr = compute_corr_fft(h_binary)
        max_lag = np.argmax(corr)
        max_val = corr[max_lag]
    
    return h_binary, c5_from_corr(corr)

    # Convex-concave procedure with cutting plane
    h_current = h0.copy()
    t_current = float('inf')
    
    max_iter = 50
    for iteration in range(max_iter):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        h_opt, t_opt = solve_convex_subproblem(h_current, lags_used)
        
        if h_opt is None:
            break
            
        # Evaluate all lags
        corr_all = compute_corr_fft(h_opt)
        max_all = np.max(corr_all)
        
        if max_all <= t_opt + 1e-6:
            # No violations, we're done
            h_current = h_opt
            t_current = t_opt
            break
            
        # Find violating lags
        violating = find_violating_lags(h_opt, t_opt, range(-n_points+1, n_points))
        if not violating:
            h_current = h_opt
            t_current = t_opt
            break
            
        # Add most violated lags to the constraint set
        lags_used.extend(violating[:min(3, len(violating))])
        h_current = h_opt
        t_current = t_opt
    
    # Rounding phase
    h_binary, c5_rounded = rounding_phase(h_current)
    
    # Polish the binary solution with critical-lag method
    target_bound = 0.3807
    h_polished, c5_final = critical_lag_polish(h_binary, target_bound)
    
    # If not good enough, try exhaustive 2-opt search
    if c5_final > target_bound and time.time() - start_time < budget_s - 10:
        # Exhaustive 2-opt search
        corr_best = compute_corr_fft(h_polished)
        max_corr = np.max(corr_best)
        
        for i in range(60):
            if time.time() - start_time > budget_s - 5:
                break
            for j in range(i+1, 60):
                # Try swap
                h_test = h_polished.copy()
                h_test[i], h_test[j] = h_test[j], h_test[i]
                # Project to sum constraint
                h_test = project_to_sum_box(h_test, TARGET_SUM)
                corr_test = compute_corr_fft(h_test)
                max_new = np.max(corr_test)
                
                if max_new < max_corr:
                    h_polished = h_test
                    max_corr = max_new
                    c5_final = dx * max_corr
    
    return (h_polished.tolist(), float(c5_final), n_points)
