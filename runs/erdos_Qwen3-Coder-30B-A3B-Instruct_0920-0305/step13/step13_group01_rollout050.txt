import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT - returns raw correlation max"""
    corr = compute_corr_fft(h)
    return np.max(corr)

def evaluate_c5_precise(h):
    """Precise evaluation using np.correlate as in original"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x - lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = np.min(x) - 1.0, np.max(x) + 1.0
    for _ in range(60):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x - (a + b)/2, 0, 1)
    return result

def compute_subgradient(h, corr):
    """Compute subgradient for the max correlation"""
    n = len(h)
    max_val = np.max(corr)
    eps = 1e-12
    active_idx = np.where(corr >= max_val - eps)[0]
    m = len(active_idx)
    
    g = np.full(n, m, dtype=float)
    for idx in active_idx:
        k = idx - (n - 1)
        if k >= 0:
            k_abs = k
            g[:n - k_abs] -= h[k_abs:]
            g[k_abs:] -= h[:n - k_abs]
        else:
            d = -k
            g[d:] -= h[:n - d]
            g[:n - d] -= h[d:]
    return g, max_val

def generate_sin_initial():
    """Generate sinusoidal initial solution"""
    x = (np.arange(n_points) + 0.5) * dx
    a = np.random.rand() * 0.5
    phase = np.random.rand() * 2 * np.pi
    h = 0.5 + a * np.sin(np.pi * x + phase)
    h = np.clip(h, 0, 1)
    h = project_to_sum_box(h, target_sum)
    return h

    candidates = [h0]

    for _ in range(3):
        h = np.random.rand(n_points)

    max_iters_per_start = 10000

        h = h_start.copy()

        best_local_c5 = current_c5

            h_new = h - step * g

    return (best_h.tolist(), float(best_c5), n_points)
