from scipy.signal import correlate
import time
from scipy.optimize import dual_annealing

# Global variables
TARGET_SUM = None

def evaluate_c5(h):
    """Fast evaluation of C5 using linear correlation"""
    # Compute correlation of h with (1-h) using linear correlation (full mode)
    corr = correlate(h, 1 - h, mode='full')
    # Return maximum correlation scaled by dx
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result
