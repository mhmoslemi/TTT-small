def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Use a more robust projection method
    x = np.array(x)
    # Sort values to find the threshold
    sorted_x = np.sort(x)[::-1]
    
    # Find the threshold using the cumulative sum approach
    cumsum = np.cumsum(sorted_x)
    threshold_indices = np.where(sorted_x * np.arange(1, len(sorted_x) + 1) - cumsum + target_sum > 0)[0]
    
    if len(threshold_indices) == 0:
        # All elements should be 0 (impossible if target_sum > 0)
        return np.clip(x, 0, 1)
    else:
        # Get the last valid index
        j = threshold_indices[-1]
        threshold = (cumsum[j] - target_sum) / (j + 1)
        
        # Project
        projected = np.clip(x - threshold, 0, 1)
        return projected

def softmax_smooth_max(x, gamma=100.0):
    """Compute smooth approximation of max using softmax"""
    x = np.array(x)
    x_max = np.max(x) 
    exp_x = np.exp(gamma * (x - x_max))
    return x_max + np.log(np.sum(exp_x)) / gamma

def compute_smooth_max_gradient(x, gamma=100.0):
    """Compute gradient of smooth max approximation"""

    x_max = np.max(x)
    exp_x = np.exp(gamma * (x - x_max))
    exp_sum = np.sum(exp_x)
    weights = exp_x / exp_sum
    return weights
