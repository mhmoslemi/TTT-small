import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Create CVXPY variables
    h = Variable(n_points)
    t = Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == target_sum,
        t >= 0
    ]
    
    # For each shift s, create variables and constraints
    for s in range(n_points):
        # Create y variables for this shift
        y_vars = [Variable() for _ in range(n_points - s)]
        
        # Add McCormick constraints for each product term
        for i in range(n_points - s):
            y = y_vars[i]
            constraints.extend([
                y >= 0,
                y <= h[i],
                y <= h[i + s],
                y >= h[i] + h[i + s] - 1
            ])
        
        # Compute overlap for this shift
        overlap = sum(y_vars)
        
        # Add constraint that correlation <= t
        # Correlation = (target_sum - prefix_sum) - overlap
        # prefix_sum = sum of h_i for i=0 to s-1
        prefix_sum = sum(h[:s]) if s > 0 else 0
        correlation = (target_sum - prefix_sum) - overlap
        constraints.append(correlation <= t)
    
    # Objective: minimize t
    objective = Minimize(t)
    
    # Create and solve the problem
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=ECOS, verbose=False, max_iters=1000)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h_opt = h.value
            t_opt = t.value
            c5_bound = dx * t_opt
        else:
            # Fallback to initial solution if optimization fails
            h_opt = np.array(initial_h_values)
            c5_bound = evaluate_erdos_solution(h_opt)
    except:
        # Fallback to initial solution if optimization fails
        h_opt = np.array(initial_h_values)
        c5_bound = evaluate_erdos_solution(h_opt)
    
    # Final refinement with local search if time allows
    if time.time() - start_time < 0.9 * budget_s:
        # Simple gradient-based refinement
        step_size = 0.01
        patience = 100
        patience_count = 0
        prev_c5 = c5_bound
        
        for _ in range(1000):
            if time.time() - start_time > 0.95 * budget_s:
                break
                
            # Compute gradient using finite differences
            grad = np.zeros(n_points)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h_opt.copy()
                h_plus[i] = min(h_plus[i] + eps, 1.0)
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (target_sum / np.sum(h_plus))  # Maintain sum constraint
                
                h_minus = h_opt.copy()
                h_minus[i] = max(h_minus[i] - eps, 0.0)
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (target_sum / np.sum(h_minus))  # Maintain sum constraint
                
                c5_plus = evaluate_erdos_solution(h_plus)
                c5_minus = evaluate_erdos_solution(h_minus)
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new = h_opt - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (target_sum / np.sum(h_new))  # Maintain sum constraint
            
            c5_new = evaluate_erdos_solution(h_new)
            
            if c5_new < c5_bound:
                h_opt = h_new
                c5_bound = c5_new
                patience_count = 0
            else:
                patience_count += 1
                if patience_count > patience:
                    break
    
    return (h_opt.tolist(), float(c5_bound), n_points)
