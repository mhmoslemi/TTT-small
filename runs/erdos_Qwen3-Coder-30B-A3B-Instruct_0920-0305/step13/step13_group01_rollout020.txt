```python
import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1 << (L - 1).bit_length()
    A = np.fft.fft(h_rev, fft_size)
    B = np.fft.fft(b, fft_size)
    corr = np.fft.ifft(A * B).real[:L]
    return corr

def evaluate_c5_precise(h):
    """Evaluate C5 using numpy correlate for precision"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_subgradient(h, corr):
    """Compute subgradient for the max correlation"""
    n = len(h)
    max_val = np.max(corr)
    # tolerance
    eps = 1e-12
    active_idx = np.where(corr >= max_val - eps)[0]
    
    # initialize gradient with constant term: 1 per active lag per component
    g = np.full(n, len(active_idx), dtype=float)
    
    for idx in active_idx:
        k = idx - (n - 1)
        if k >= 0:
            # subtract h_{j + k}
            g[:n - k] -= h[k:]
            # subtract h_{j - k} for j >= k
            g[k:] -= h[:n - k]
        else:
            d = -k
            # Equivalent to k negative, treat as above
            g[d:] -= h[:n - d]
            g[:n - d] -= h[d:]
    
    return g, max_val

def generate_sin_initial():
    """Generate sinusoidal initial solution"""
    x = (np.arange(n_points) + 0.5) * dx
    # Random amplitude [0, 0.5] and phase
    amp = np.random.rand() * 0.5
    phase = np.random.rand() * 2 * np.pi
    h = 0.5 + amp * np.sin(np.pi * x + phase)
    # Ensure box constraints
    h = np.clip(h, 0, 1)
    # Project to sum constraint
    h = project_to_sum_box(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    global n_points, dx, target_sum
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from provided solution if possible
    if 'initial_h_values' in globals() and len(initial_h_values) == 120:
        h0 = np.repeat(initial_h_values, 2)
    else:
        h0 = np.full(n_points, 0.5)
    
    # Ensure h0 satisfies constraints
    h0 = project_to_sum_box(h0, target_sum)
    
    best_c5 = evaluate_c5_precise(h0)
    best_h = h0.copy()
    
    # Generate multiple initial candidates
    candidates = [h0]
    
    # Add sinusoidal initial shapes
    for _ in range(5):
        candidates.append(generate_sin_initial())
    
    # Add random uniform initializations
    for _ in range(3):
        h_random = np.random.rand(n_points)
        h_random = project_to_sum_box(h_random, target_sum)
        candidates.append(h_random)
    
    max_iters_per_start = 10000
    base_lr = 0.3
    
    for i, h_start in enumerate(candidates):
        if time.time() - start_time > budget_s - 10:
            break
            
        h = h_start.copy()
        current_c5 = evaluate_c5_fast(h)
        
        for iter_num in range(max_iters_per_start):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute correlation
            corr = compute_corr_fft(h)
            
            # Compute subgradient
            g, max_corr = compute_subgradient(h, corr)
            
            # Step size
            step = base_lr / np.sqrt(iter_num + 1)
            
            # Update
            h_new = h - step * g
            
            # Project
            h_new = project_to_sum_box(h_new, target_sum)
            
            # Evaluate new C5
            new_corr = compute_corr_fft(h_new)
            new_c5 = np.max(new_corr) * dx
            
            if new_c5 < current_c5:
                h = h_new
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
    
    # Final local search with swaps
    for _ in range(2000):
        if time.time() - start_time > budget_s - 2:
            break
            
        i, j = np.random.randint(0, n_points, 2)
        temp_h = best_h.copy()
        temp_h[i], temp_h[j] = temp_h[j], temp_h[i]
        temp_h = project_to_sum_box(temp_h, target_sum)
        
        temp_c5 = evaluate_c5_precise(temp_h)
        if temp_c5 < best_c5:
            best_c5 = temp_c5
            best_h = temp_h.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
```
