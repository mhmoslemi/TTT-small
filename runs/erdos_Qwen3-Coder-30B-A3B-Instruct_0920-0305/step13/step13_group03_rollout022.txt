import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

# Global variables
n_points = 120
dx = 2.0 / n_points
target_M = int(np.floor(0.3807 / dx))  # 22
ℓ_max = n_points - target_M  # 98

def max_corr(h):
    """Compute maximum correlation value"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 value"""
    return dx * max_corr(h)

def build_lag_to_indices(h):
    """Build mapping from lag to mismatched indices"""
    n = len(h)
    lag_to_indices = {}
    for ℓ in range(-ℓ_max, ℓ_max + 1):
        if ℓ >= 0:
            indices = [i for i in range(n - ℓ) if h[i] == 1 and h[i + ℓ] == 0]
        else:
            indices = [i for i in range(-ℓ) if h[i] == 1 and h[i - ℓ] == 0]
        lag_to_indices[ℓ] = indices
    return lag_to_indices

def greedy_swap(h, lag_to_indices, M):
    """Perform greedy swap to improve worst lag"""
    n = len(h)
    while M > target_M:
        # Find worst lag
        worst_ℓ = max(range(-ℓ_max, ℓ_max + 1), key=lambda ℓ: max_corr(h)[ℓ + n - 1])
        
        # Get bad indices for this lag
        B = lag_to_indices[worst_ℓ]
        if not B:
            break
            
        # Pick a random bad index
        i = np.random.choice(B)
        
        # Find candidates for swap
        candidates = []
        for j in range(n):
            if h[j] == 0:
                # Check if swapping won't create new mismatch at ℓ
                if worst_ℓ >= 0:
                    if j + worst_ℓ >= n or h[j + worst_ℓ] == 1:
                        candidates.append(j)
                else:
                    if j - worst_ℓ >= n or h[j - worst_ℓ] == 1:
                        candidates.append(j)
        
        if candidates:
            j = np.random.choice(candidates)
        else:
            # If no good candidate, pick random 0 and accept if doesn't increase M
            zero_indices = [i for i in range(n) if h[i] == 0]
            if zero_indices:
                j = np.random.choice(zero_indices)
            else:
                break
                
        # Perform swap
        h[i], h[j] = h[j], h[i]
        
        # Recompute lag_to_indices
        lag_to_indices = build_lag_to_indices(h)
        M = max_corr(h)
        
    return h, lag_to_indices, M
