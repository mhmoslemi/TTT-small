import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5_precise(h):
    """Evaluate C5 using np.correlate for precision"""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_subgradient(h, corr):
    """Compute subgradient for the max correlation objective"""
    n = len(h)
    max_val = np.max(corr)
    eps = 1e-12
    active_idx = np.where(corr >= max_val - eps)[0]
    
    if len(active_idx) == 0:
        return np.zeros(n), max_val
    
    # Initialize gradient with constant term for each active lag
    g = np.full(n, len(active_idx), dtype=float)
    
    for idx in active_idx:
        k = idx - (n - 1)
        if k >= 0:
            # Subtract h_{j+k} for j from 0 to n-k-1
            g[:n-k] -= h[k:]
            # Subtract h_{j-k} for j from k to n-1
            g[k:] -= h[:n-k]
        else:
            # For k < 0, similar operation
            d = -k
            g[d:] -= h[:n-d]
            g[:n-d] -= h[d:]
    
    return g, max_val

def generate_sinusoidal_initial():
    """Generate sinusoidal initial solution"""
    x = (np.arange(n_points) + 0.5) * dx
    # Random amplitude and phase
    amplitude = np.random.rand() * 0.5
    phase = np.random.rand() * 2 * np.pi
    h = 0.5 + amplitude * np.sin(np.pi * x + phase)
    # Ensure values are within [0,1]
    h = np.clip(h, 0, 1)
    # Project to ensure exact sum constraint
    h = project_to_sum_box(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from repeated initial solution if available
    if 'initial_h_values' in globals() and len(initial_h_values) == 120:
        h0 = np.repeat(initial_h_values, 2)
    else:
        h0 = np.random.rand(n_points)
        h0 = project_to_sum_box(h0, target_sum)
    
    # Evaluate initial solution
    best_c5 = evaluate_c5_precise(h0)
    best_h = h0.copy()
    
    # Generate multiple initial candidates
    candidates = [h0]
    
    # Add sinusoidal initial solutions
    for _ in range(5):
        candidates.append(generate_sinusoidal_initial())
    
    # Add uniform random initial solutions
    for _ in range(3):
        h_rand = np.random.rand(n_points)
        h_rand = project_to_sum_box(h_rand, target_sum)
        candidates.append(h_rand)
    
    # Run subgradient descent for each candidate
    max_iters_per_restart = 10000
    base_lr = 0.3
    
    for i, h_start in enumerate(candidates):
        if time.time() - start_time > budget_s - 10:
            break
            
        h = h_start.copy()
        current_c5 = evaluate_c5_fast(h)
        best_local_c5 = current_c5
        
        for iter_num in range(max_iters_per_restart):
            if time.time() - start_time > budget_s - 5:
                break
                
            corr = compute_corr_fft(h)
            g, max_corr = compute_subgradient(h, corr)
            
            # Step size with decay
            step = base_lr / np.sqrt(iter_num + 1)
            
            # Update step
            h_new = h - step * g
            
            # Project to feasible region
            h_new = project_to_sum_box(h_new, target_sum)
            
            # Evaluate new solution
            new_corr = compute_corr_fft(h_new)
            new_c5 = np.max(new_corr)
            
            if new_c5 < max_corr:
                h = h_new
                max_corr = new_c5
                if new_c5 * dx < best_c5:
                    best_c5 = new_c5 * dx
                    best_h = h_new.copy()
            else:
                h = h_new
        
        # Local search improvement
        for _ in range(2000):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Swap two random indices
            i, j = np.random.choice(n_points, 2, replace=False)
            h_swap = h_new.copy()
            h_swap[i], h_swap[j] = h_swap[j], h_swap[i]
            
            # Project to constraint
            h_swap = project_to_sum_box(h_swap, target_sum)
            
            # Evaluate
            new_c5_precise = evaluate_c5_precise(h_swap)
            if new_c5_precise < best_c5:
                best_c5 = new_c5_precise
                best_h = h_swap.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
