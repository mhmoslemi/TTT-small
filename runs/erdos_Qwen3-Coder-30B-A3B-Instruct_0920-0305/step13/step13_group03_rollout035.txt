import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_convex_subproblem(h_current, lags, n_points):
    """Solve the convex subproblem using CVXPY"""
    # Create variables
    h = Variable(n_points)
    T = Variable(1)
    
    # Constraints
    constraints = [h >= 0, h <= 1]
    constraints.append(sum(h) == n_points / 2)
    
    # Add convex constraints for each lag in the current set
    for ell in lags:
        if ell >= 0:
            # For positive lags
            expr = 0
            for i in range(n_points):
                if i + ell < n_points:
                    expr += h[i] + (h[i] - h[i + ell])**2 / 4
                else:
                    expr += h[i]
            expr -= sum((h_current[i] + h_current[i + ell]) * (h[i] + h[i + ell]) for i in range(n_points - ell))
            constraints.append(expr <= T)
        else:
            # For negative lags
            expr = 0
            for i in range(n_points):
                if i - ell < n_points:
                    expr += h[i] + (h[i] - h[i - ell])**2 / 4
                else:
                    expr += h[i]
            expr -= sum((h_current[i] + h_current[i - ell]) * (h[i] + h[i - ell]) for i in range(n_points + ell))
            constraints.append(expr <= T)
    
    # Objective
    objective = Minimize(T)
    
    # Solve
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status == 'optimal':
            return h.value, T.value
    except:
        pass
    
    # Fallback to simple projection
    return h_current.copy(), np.inf

def find_violating_lags(h_opt, T_opt, h_current, n_points, epsilon=1e-6):
    """Find lags that violate the constraint"""
    all_lags = list(range(-n_points + 1, n_points))
    violating = []
    corr = compute_corr_fft(h_opt)
    for i, ell in enumerate(all_lags):
        if ell >= 0:
            # Positive lags
            val = 0
            for j in range(n_points):
                if j + ell < n_points:
                    val += h_opt[j] * (1 - h_opt[j + ell])
            if val > T_opt + epsilon:
                violating.append(ell)
        else:
            # Negative lags  
            val = 0
            for j in range(n_points):
                if j - ell < n_points:
                    val += h_opt[j] * (1 - h_opt[j - ell])
            if val > T_opt + epsilon:
                violating.append(ell)
    return violating

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    target_M = int(0.3807 / dx)  # Target maximum overlap value
    
    # Initial solution
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_current = np.array(initial_h_values)
    else:
        h_current = np.full(n_points, 0.5)
    
    h_current = project_to_sum_box(h_current, n_points / 2)
    
    # Initial lags set
    initial_lags = [0, -24, -48, -72, -96, 24, 48, 72, 96]  # ±⌊n/5⌋, ±⌊2n/5⌋, ±⌊3n/5⌋, ±⌊4n/5⌋
    lags = initial_lags.copy()
    
    best_c5 = float('inf')
    best_h = None
    
    # Convex-concave procedure with cutting plane
    max_iterations = 100
    iteration = 0
    
    while time.time() < start_time + 0.4 * budget_s and iteration < max_iterations:
        iteration += 1
        
        # Solve convex subproblem
        h_opt, T_opt = solve_convex_subproblem(h_current, lags, n_points)
        
        if T_opt == np.inf:
            break
            
        # Evaluate all correlations
        corr = compute_corr_fft(h_opt)
        current_c5 = c5_from_corr(corr, n_points)
        
        # Check if current solution is optimal
        if current_c5 <= target_M:
            h_current = h_opt
            break
            
        # Find violating lags
        violating = find_violating_lags(h_opt, T_opt, h_current, n_points)
        
        if not violating:
            # No violations, we're done
            h_current = h_opt
            break
            
        # Add most violating lag
        lags.extend(violating[:min(5, len(violating))])
        
        # Keep the best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
        
        h_current = h_opt
        
        # Prevent infinite loops
        if len(lags) > 100:
            break
    
    # If we haven't found a good solution yet, proceed to rounding
    if best_h is None:
        best_h = h_current
        best_c5 = c5_from_corr(compute_corr_fft(h_current), n_points)
    
    # Randomized rounding
    best_binary = None
    best_binary_c5 = float('inf')
    
    num_rounds = 30
    for _ in range(num_rounds):
        if time.time() > start_time + 0.6 * budget_s:
            break
            
        # Round based on sorted values
        sorted_indices = np.argsort(best_h)[::-1]
        binary_h = np.zeros(n_points)
        binary_h[sorted_indices[:n_points//2]] = 1
        binary_h = project_to_sum_box(binary_h, n_points / 2)
        
        # Calculate C5 for this binary
        corr = compute_corr_fft(binary_h)
        binary_c5 = c5_from_corr(corr, n_points)
        
        if binary_c5 < best_binary_c5:
            best_binary_c5 = binary_c5
            best_binary = binary_h.copy()
    
    if best_binary is None:
        best_binary = best_h
        best_binary_c5 = c5_from_corr(compute_corr_fft(best_h), n_points)
    
    # Critical-lag greedy local search
    def polish_with_critical_lag(h, max_time):
        start_polish = time.time()
        h = h.copy()
        while time.time() < start_polish + max_time:
            corr = compute_corr_fft(h)
            max_shift = np.argmax(corr)
            # If we've already achieved our target, stop
            if corr[max_shift] <= target_M:
                break
                
            # Find worst shift
            worst_shift = max_shift - (n_points - 1)  # Convert to actual shift
            # Find all '1' elements that are mismatched for this shift
            mismatched_ones = []
            for i in range(n_points):
                if worst_shift >= 0:
                    if i + worst_shift < n_points:
                        if h[i] == 1 and h[i + worst_shift] == 0:
                            mismatched_ones.append(i)
                elif worst_shift < 0:
                    if i - worst_shift < n_points:
                        if h[i] == 1 and h[i - worst_shift] == 0:
                            mismatched_ones.append(i)
            
            if not mismatched_ones:
                break
                
            i_idx = np.random.choice(mismatched_ones)
            # Find zeros that could be swapped without creating new mismatches
            zero_candidates = []
            for j in range(n_points):
                if worst_shift >= 0:
                    if j + worst_shift < n_points:
                        if h[j] == 0 and h[j + worst_shift] == 1:
                            zero_candidates.append(j)
                elif worst_shift < 0:
                    if j - worst_shift < n_points:
                        if h[j] == 0 and h[j - worst_shift] == 1:
                            zero_candidates.append(j)
            
            if not zero_candidates:
                j_idx = np.random.randint(n_points)
            else:
                j_idx = np.random.choice(zero_candidates)
            
            # Perform swap
            h[i_idx], h[j_idx] = h[j_idx], h[i_idx]
            
            # Check if we improved
            corr = compute_corr_fft(h)
            if corr[max_shift] <= target_M:
                break
        
        return h, c5_from_corr(compute_corr_fft(h), n_points)
    
    # Polishing with more time allocated
    final_h, final_c5 = polish_with_critical_lag(best_binary, 0.2 * budget_s)
    
    # Return the best solution found
    return (final_h.tolist(), float(final_c5 * dx), n_points)
