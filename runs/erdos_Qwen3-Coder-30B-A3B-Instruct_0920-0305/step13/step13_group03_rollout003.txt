def max_corr(h):
    """Compute maximum correlation of h with 1-h"""
    corr = correlate(h, 1-h, mode="full")
    return np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 bound for given h"""
    return 2.0/len(h) * max_corr(h)
