import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

# Global constants
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_correlation(h):
    """Compute correlation using FFT for efficiency"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5_exact(h):
    """Compute exact C5 using np.correlate"""
    corr = correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(60):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_gradient(h, alpha):
    """Compute gradient of smooth max with respect to h for fixed alpha"""
    corr = compute_correlation(h)
    
    # Stabilize log-sum-exp: subtract max
    max_corr = np.max(corr)
    exp_corr = np.exp(alpha * (corr - max_corr))
    sum_exp = np.sum(exp_corr)
    weights = exp_corr / sum_exp
    
    # Gradient computation
    grad = np.zeros_like(h)
    # For each shift k
    L = len(h)
    for k in range(-L+1, L):
        if k >= 0:
            # h[i] contributes to S_{i+k} if i+k < L, and to S_{i-k} if i-k >= 0
            if k == 0:
                # S_i = h_i * (1 - h_i) 
                grad += weights[k+L-1] * (1 - 2 * h)
            else:
                # S_{i+k} = h_i * (1 - h_{i+k})
                # grad_{i} = weights_{i+k} * (1 - h_{i+k}) + weights_{i-k} * (-h_{i-k})
                shift_plus = np.roll(h, -k)
                shift_minus = np.roll(h, k)
                # Ensure boundary conditions
                shift_plus[:k] = 0
                shift_minus[-k:] = 0
                grad += weights[k+L-1] * (1 - shift_plus) 
                grad += weights[-k+L-1] * (-shift_minus)
    
    return dx * grad

def continuation_optimization(h0, alpha_sequence, method='trust-constr'):
    """Solve the C5 minimization problem using continuation on alpha"""
    h = h0.copy()
    
    # Define the constraint function and its Jacobian
    def constraint_fun(h):
        return np.sum(h) - target_sum
    
    def constraint_jac(h):
        return np.ones_like(h)
    
    constraints = {'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}
    bounds = [(0, 1) for _ in range(n_points)]
    
    for alpha in alpha_sequence:
        # Objective function
        def objective(h):
            corr = compute_correlation(h)
            max_corr = np.max(corr)
            exp_corr = np.exp(alpha * (corr - max_corr))
            sum_exp = np.sum(exp_corr)
            return dx * (max_corr + np.log(sum_exp) / alpha)
            
        # Gradient function
        def grad(h):
            return smooth_max_gradient(h, alpha)
        
        # Optimize with constraints
        if method == 'trust-constr':
            res = minimize(objective, h, method='trust-constr', jac=grad, bounds=bounds,
                           constraints=constraints, options={'gtol': 1e-9, 'xtol': 1e-9})
        else:  # SLSQP
            res = minimize(objective, h, method='SLSQP', jac=grad, bounds=bounds,
                           constraints=constraints, options={'ftol': 1e-9})
        
        h = res.x
        # Project to ensure exact sum constraint
        h = project_to_sum_box(h, target_sum)
        
        # Early stopping if converged
        if res.fun < 1e-10:
            break
    
    return h

    extended = np.tile(legacy_pattern, n_points // len(legacy_pattern) + 1)
    extended = extended[:n_points]
    initials.append(project_to_sum_box(extended, target_sum))
    
    # 2. Sinusoidal pattern
    x = np.linspace(0, 2, n_points)
    phase = np.random.uniform(0, 2 * np.pi)
    amplitude = np.random.uniform(0, 0.5)
    sin_wave = 0.5 + amplitude * np.sin(np.pi * x + phase)
    initials.append(project_to_sum_box(sin_wave, target_sum))
    
    # 3. Amplitude-tuned sinusoid
    amplitude2 = np.random.uniform(0, 0.5)
    sin_wave2 = 0.5 + amplitude2 * np.sin(np.pi * x)
    initials.append(project_to_sum_box(sin_wave2, target_sum))
    
    # 4. Triangular wave
    triangular = np.zeros(n_points)
    mid = n_points // 2
    triangular[:mid] = np.linspace(0, 1, mid)
    triangular[mid:] = np.linspace(1, 0, n_points - mid)
    initials.append(project_to_sum_box(triangular, target_sum))
    
    # 5. Random uniform
    random_uniform = np.random.uniform(0, 1, n_points)
    initials.append(project_to_sum_box(random_uniform, target_sum))
    
    # 6. Convex combination of two patterns
    combo = 0.7 * initials[0] + 0.3 * initials[1]
    initials.append(project_to_sum_box(combo, target_sum))
    
    return initials

def local_search(h):
    """Perform greedy local search to polish the solution"""
    best_h = h.copy()
    best_c5 = evaluate_c5_exact(best_h)
    patience = 0
    max_patience = 1000
    
    for _ in range(3000):
        if patience > max_patience:
            break
            
        # Try a swap
        i, j = np.random.choice(n_points, 2, replace=False)
        h_try = best_h.copy()
        h_try[i], h_try[j] = h_try[j], h_try[i]
        h_try = project_to_sum_box(h_try, target_sum)
        c5_try = evaluate_c5_exact(h_try)
        
        if c5_try <= best_c5:
            best_h = h_try
            best_c5 = c5_try
            patience = 0
        else:
            patience += 1
            
        # Try a transfer
        if np.random.rand() < 0.5:
            i, j = np.random.choice(n_points, 2, replace=False)
            delta = np.random.uniform(0, 0.01)
            if best_h[i] >= delta and best_h[j] <= 1 - delta:
                h_try = best_h.copy()
                h_try[i] -= delta
                h_try[j] += delta
                h_try = project_to_sum_box(h_try, target_sum)
                c5_try = evaluate_c5_exact(h_try)
                
                if c5_try <= best_c5:
                    best_h = h_try
                    best_c5 = c5_try
                    patience = 0
                else:
                    patience += 1
    
    return best_h, best_c5

    for i, h0 in enumerate(initials):

            h_opt = continuation_optimization(h0, alpha_sequence, method='trust-constr')
