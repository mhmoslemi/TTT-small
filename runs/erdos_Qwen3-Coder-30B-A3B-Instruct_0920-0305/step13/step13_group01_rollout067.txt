import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

# Global configuration
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT with proper scaling"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5_exact(h):
    """Exact evaluation of C5 using np.correlate"""
    corr = correlate(h, 1 - h, mode='full')
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_alpha(corr, alpha):
    """Compute smooth approximation to max with parameter alpha"""
    # Stabilize by subtracting the maximum
    max_corr = np.max(corr)
    stable_corr = corr - max_corr
    exp_corr = np.exp(alpha * stable_corr)
    return max_corr + np.log(np.sum(exp_corr)) / alpha

def smooth_grad_alpha(corr, alpha, h):
    """Compute gradient of smooth max with respect to h"""
    # Stabilize by subtracting max
    max_corr = np.max(corr)
    stable_corr = corr - max_corr
    exp_corr = np.exp(alpha * stable_corr)
    weights = exp_corr / np.sum(exp_corr)
    
    # The weights are for each shift k, but we need gradients per h_i
    # Weighted sum of differences (h_{i+k} + h_{i-k} - 1) 
    grad = np.zeros_like(h)
    L = len(h)
    
    # For each shift k (from -L+1 to L-1)
    for k in range(-L+1, L):
        if k > 0:
            # h[i+k] contributes to shift k
            if k < L:
                grad[:-k] += weights[k + L - 1] * h[k:] 
            if k >= L:
                continue
        elif k < 0:
            # h[i-k] contributes to shift k  
            if -k < L:
                grad[-k:] += weights[k + L - 1] * h[:-k] 
            if -k >= L:
                continue
        else:
            # k=0, no shifting
            grad += weights[L-1] * (1.0 - h)
    
    # Normalize by scale factor and dx
    return dx * grad

def setup_initials():
    """Create various initial constructions"""
    initials = []
    
    # 1. Repeated initial construction
    repeats = n_points // len(initial_h_values)
    h_repeated = np.tile(initial_h_values, repeats)[:n_points]
    initials.append(project_to_sum_box(h_repeated, target_sum))
    
    # 2. Pure sinusoid
    x = np.linspace(0, 2, n_points, endpoint=False) + dx/2
    phi = np.random.rand() * 2 * np.pi
    h_sine = 0.5 + 0.5 * np.sin(np.pi * x + phi)
    initials.append(project_to_sum_box(h_sine, target_sum))
    
    # 3. Sinusoid with random amplitude
    a = np.random.rand() * 0.5
    h_sine_a = 0.5 + a * np.sin(np.pi * x)
    initials.append(project_to_sum_box(h_sine_a, target_sum))
    
    # 4. Triangular wave
    h_tri = np.zeros(n_points)
    mid = n_points // 2
    h_tri[:mid] = np.linspace(0, 1, mid)
    h_tri[mid:] = np.linspace(1, 0, n_points - mid)
    h_tri = h_tri * (2.0 / np.sum(h_tri))  # Scale to match area = 1
    initials.append(project_to_sum_box(h_tri, target_sum))
    
    # 5. Random uniform with projection
    h_rand = np.random.rand(n_points)
    initials.append(project_to_sum_box(h_rand, target_sum))
    
    # 6. Hybrid construction (sinusoid + noise)
    h_hybrid = 0.7 * (0.5 + 0.5 * np.sin(np.pi * x)) + 0.3 * np.random.rand(n_points)
    initials.append(project_to_sum_box(h_hybrid, target_sum))
    
    return initials

def run_continuation(h0, alpha_levels, method='SLSQP'):
    """Run continuation on alpha levels"""
    h = h0.copy()
    best_c5 = float('inf')
    best_h = h0.copy()
    
    for alpha in alpha_levels:
        # Objective function
        def obj(h):
            corr = compute_corr_fft(h)
            return smooth_max_alpha(corr, alpha)
        
        # Gradient function
        def grad(h):
            corr = compute_corr_fft(h)
            return smooth_grad_alpha(corr, alpha, h)
        
        # Constraints
        def constraint_sum(h):
            return np.sum(h) - target_sum
        cons = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}
        
        # Bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimize
        try:
            result = minimize(obj, h, method=method, jac=grad, constraints=cons, bounds=bounds,
                             options={'ftol': 1e-9, 'gtol': 1e-9, 'maxiter': 200})
            if result.success:
                h = result.x.copy()
                # Project to ensure sum constraint is precisely met
                h = project_to_sum_box(h, target_sum)
                c5 = evaluate_c5_exact(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            else:
                # Use current h as fallback if optimization failed
                c5 = evaluate_c5_exact(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        except:
            # Fallback to exact eval if anything fails
            c5 = evaluate_c5_exact(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5

def local_search(h, max_iter=3000):
    """Perform local search using swap and transfer moves"""
    h_best = h.copy()
    c5_best = evaluate_c5_exact(h_best)
    
    iter_no_improve = 0
    for i in range(max_iter):
        if iter_no_improve > 1000:
            break
            
        # Try a swap move
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        h_trial = h_best.copy()
        h_trial[idx1], h_trial[idx2] = h_trial[idx2], h_trial[idx1]
        h_trial = project_to_sum_box(h_trial, target_sum)
        c5_trial = evaluate_c5_exact(h_trial)
        
        if c5_trial < c5_best:
            h_best = h_trial.copy()
            c5_best = c5_trial
            iter_no_improve = 0
        else:
            iter_no_improve += 1
            
        # Try small transfer move
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        delta = np.random.rand() * 0.02
        if h_best[idx1] >= delta and h_best[idx2] + delta <= 1:
            h_trial = h_best.copy()
            h_trial[idx1] -= delta
            h_trial[idx2] += delta
            h_trial = project_to_sum_box(h_trial, target_sum)
            c5_trial = evaluate_c5_exact(h_trial)
            
            if c5_trial < c5_best:
                h_best = h_trial.copy()
                c5_best = c5_trial
                iter_no_improve = 0
            else:
                iter_no_improve += 1
    
    return h_best, c5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Setup
    initials = setup_initials()
    alpha_levels = [1.0, 5.0, 20.0, 80.0, 300.0]
    best_c5 = float('inf')
    best_h = None
    
    # Run continuation for each initial
    num_starts = 0
    while time.time() - start_time < budget_s - 5 and num_starts < 10:
        h0 = initials[num_starts % len(initials)]
        h_final, c5_final = run_continuation(h0, alpha_levels)
        
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
        
        num_starts += 1
    
    # Local search polishing
    if best_h is not None:
        best_h, best_c5 = local_search(best_h)
    
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_c5_exact(best_h)
        return (best_h.tolist(), float(final_c5), n_points)
    
    # Fallback to initial solution
    h0 = np.array(initial_h_values)
    h0 = project_to_sum_box(h0, target_sum)
    final_c5 = evaluate_c5_exact(h0)
    return (h0.tolist(), float(final_c5), n_points)
