# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort x in descending order
    sorted_indices = np.argsort(x)[::-1]
    sorted_x = x[sorted_indices]
    
    # Use lemma-based projection for simplex constraint
    # Using the fact that we want sum = target_sum
    # Project onto the feasible set: 0 <= x_i <= 1 and sum x_i = target_sum
    cumsum = np.cumsum(sorted_x)
    n = len(sorted_x)
    
    # Find the threshold index
    threshold = -1
    for i in range(n):
        if cumsum[i] <= target_sum - (n - i - 1):
            threshold = i
        else:
            break
    
    if threshold == -1:
        # All elements can be set to their maximum value
        lambda_val = target_sum / n
        if lambda_val > 1:
            lambda_val = 1
        result = np.full(n, lambda_val)
    else:
        lambda_val = (target_sum - cumsum[threshold]) / (n - threshold - 1)
        result = np.zeros(n)
        for i in range(n):
            if i <= threshold:
                result[i] = sorted_x[i]
            else:
                result[i] = max(0, min(1, lambda_val))
    
    # Reorder back
    final_result = np.empty_like(result)
    final_result[sorted_indices] = result
    return final_result

def smooth_max(x, alpha=100):
    """Softmax approximation of max"""
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def smooth_max_grad(x, alpha=100):
    """Gradient of smooth max"""
    exp_x = np.exp(alpha * x)
    return exp_x / (np.sum(exp_x) * alpha)

def smooth_objective(h, alpha=100):
    """Smooth approximation of the objective"""
    corr = compute_corr_fft(h)
    return smooth_max(corr, alpha)

def smooth_gradient(h, alpha=100):
    """Gradient of smooth objective using finite differences"""
    eps = 1e-6
    grad = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_minus = h.copy()
        h_minus[i] -= eps
        corr_plus = compute_corr_fft(h_plus)
        corr_minus = compute_corr_fft(h_minus)
        grad[i] = (smooth_max(corr_plus, alpha) - smooth_max(corr_minus, alpha)) / (2 * eps)
    return grad
