import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 120
    dx = 2.0 / n_points
    target_M = int(0.3807 / dx)  # Target for max correlation
    
    # Initial setup
    h0 = np.array(initial_h_values) if 'initial_h_values' in globals() else np.ones(n_points) * 0.5
    h0 = project_to_sum_box(h0, n_points / 2)
    
    # CCP with cutting plane approach
    L = {0, -24, -48, -72, -96, 24, 48, 72, 96}  # Initial lags
    h_cont = h0.copy()
    
    # Convex-concave procedure with cutting plane
    max_iter = 1000
    epsilon = 1e-6
    tolerance = 1e-3
    t1 = start_time + 0.4 * budget_s
    iter_count = 0
    
    while time.time() < t1 and iter_count < max_iter:
        iter_count += 1
        # Build quadratic program
        h = Variable(n_points)
        T = Variable()
        
        # Constraints
        constraints = [0 <= h, h <= 1, sum(h) == n_points / 2, T >= 0]
        
        # Build convex surrogates for each lag in L
        for l in L:
            # Compute shifted indices with zero-padding
            if l >= 0:
                h_shift = h[l:] + np.zeros(l)  # Pad with zeros
                h_shift = h_shift[:n_points-l] + h[:l]
            else:  # l < 0
                h_shift = h[:l] + np.zeros(-l)  # Pad with zeros
                h_shift = h_shift[n_points+l:] + h[n_points+l:]
            
            # Convex part: sum(h_i) + 0.25 * sum((h_i - h_{i+l})^2)
            # Linear part: -0.5 * sum((h_i^(t) + h_{i+l}^(t)) * (h_i + h_{i+l}))
            convex_part = sum(h) + 0.25 * sum((h[i] - h_shift[i])**2 for i in range(len(h_shift)))
            
            # Linear part using current h_cont
            linear_part = -0.5 * sum((h_cont[i] + h_cont[(i+l) % n_points]) * (h[i] + h_shift[i]) 
                                   for i in range(len(h_shift)))
            
            constraints.append(convex_part + linear_part <= T)
        
        # Solve QP
        objective = Minimize(T)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status == 'optimal':
                h_val = np.array(h.value).flatten()
                h_cont = project_to_sum_box(h_val, n_points / 2)
                T_val = T.value
            else:
                break
        except:
            break
            
        # Evaluate actual overlaps for all lags
        corr_all = compute_corr_fft(h_cont)
        max_I = np.max(corr_all)
        
        # Check if we've found the optimal solution
        if max_I <= T_val + epsilon:
            break
            
        # Add most violated lags
        violations = [i - n_points + 1 for i in range(2*n_points - 1) 
                      if corr_all[i] > T_val + epsilon]
        
        # Add top violations to constraint set
        if violations:
            # Add lags that are among the most violated ones
            L.update(violations[:min(5, len(violations))])
            
        # Prevent too many lags from accumulating
        if len(L) > 100:
            # Keep only the most critical lags based on current correlation
            abs_corr = np.abs(corr_all)
            critical_lags = np.argsort(-abs_corr)[1:20]  # Top 20 lags
            L = set(critical_lags)
            L.add(0)
    
    # Rounding to binary
    best_binary = None
    best_c5 = float('inf')
    r_count = 30
    
    for r in range(r_count):
        # Sort indices by h_cont values and set first half to 1
        idx = np.argsort(-h_cont)
        binary_h = np.zeros(n_points)
        binary_h[idx[:n_points//2]] = 1
        
        # Keep the sum constraint and project back
        binary_h = project_to_sum_box(binary_h, n_points / 2)
        
        # Evaluate with FFT
        corr = compute_corr_fft(binary_h)
        c5 = np.max(corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_binary = binary_h.copy()
    
    # Critical lag local improvement
    if best_binary is not None:
        h_current = best_binary.copy()
        current_c5 = best_c5
        
        # Do critical lag polishing
        t2 = start_time + 0.8 * budget_s
        while time.time() < t2:
            corr = compute_corr_fft(h_current)
            worst_lag = np.argmax(corr) - n_points + 1
            
            # Find mismatched pairs
            B_indices = []
            for i in range(n_points):
                j = (i + worst_lag) % n_points
                if h_current[i] == 1 and (j >= n_points or h_current[j] == 0):
                    B_indices.append(i)
            
            if not B_indices:
                break
                
            # Pick a random mismatch
            i_to_flip = np.random.choice(B_indices)
            j_candidates = []
            
            # Find suitable zero indices that won't create new mismatches
            for j in range(n_points):
                if h_current[j] == 0:
                    # Check if this would create a mismatch at worst_lag
                    if j + worst_lag >= n_points or h_current[j + worst_lag] == 1:
                        j_candidates.append(j)
            
            if not j_candidates:
                # Fallback: just use any zero
                zero_indices = np.where(h_current == 0)[0]
                if len(zero_indices) > 0:
                    j_to_flip = np.random.choice(zero_indices)
                else:
                    break
            else:
                j_to_flip = np.random.choice(j_candidates)
            
            # Perform the swap
            h_current[i_to_flip], h_current[j_to_flip] = h_current[j_to_flip], h_current[i_to_flip]
            
            # Reevaluate
            corr_updated = compute_corr_fft(h_current)
            new_c5 = np.max(corr_updated)
            
            if new_c5 < current_c5:
                current_c5 = new_c5
                # Check if we've hit target
                if current_c5 <= target_M:
                    break
            # Safety: break if we've got a reasonable solution already
            if current_c5 <= target_M + 2:
                break
    
    final_h = h_current if 'h_current' in locals() else best_binary
    final_c5 = current_c5 if 'current_c5' in locals() else best_c5
    
    return (final_h.tolist(), float(final_c5 * dx), n_points)
