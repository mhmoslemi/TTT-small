from scipy.signal import correlate

# Global constants
n_points = 240

def compute_correlation(h):
    """Compute correlation using FFT for faster correlation"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_exact(h):
    """Compute the exact C5 using np.correlate"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(y, target_sum):
    """Project vector y onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(y + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(y + (a + b)/2, 0, 1)
    return result

def smooth_max_gradient(h, alpha):
    """Compute gradient of smooth max with parameter alpha"""
    corr = compute_correlation(h)
    
    # Subtract max for numerical stability
    max_corr = np.max(corr)
    exp_corr = np.exp(alpha * (corr - max_corr))
    weights = exp_corr / np.sum(exp_corr)
    
    grad = np.zeros_like(h)
    L = len(h)
    
    for k in range(-L+1, L):
        if k >= 0:
            shift_forward = h[k:]
            shift_backward = h[:L-k]
        else:
            shift_forward = h[:k]
            shift_backward = h[-k:]
            
        # Pad with zeros
        if k >= 0:
            shifted_backward = np.pad(shift_backward, (k, 0), constant_values=0)
            shifted_forward = np.pad(shift_forward, (0, k), constant_values=0)
        else:
            shifted_backward = np.pad(shift_backward, (-k, 0), constant_values=0)
            shifted_forward = np.pad(shift_forward, (0, -k), constant_values=0)
        
        grad += weights[k + L - 1] * (1 - shifted_forward - shifted_backward)
    
    return dx * grad

def optimize_with_continuation(h0, alpha_values, method='trust-constr'):
    """Optimize with continuation on alpha values"""
    current_h = h0.copy()
    best_h = current_h.copy()
    best_c5 = c5_exact(current_h)
    
    for alpha in alpha_values:
        # Define objective and gradient
        def objective(h):
            corr = compute_correlation(h)
            max_corr = np.max(corr)
            exp_corr = np.exp(alpha * (corr - max_corr))
            return np.log(np.sum(exp_corr)) / alpha
        
        def gradient(h):
            return smooth_max_gradient(h, alpha)
        
        # Constraints
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum, 'jac': lambda h: np.ones_like(h)}
        bounds = [(0, 1)] * len(h0)
        
        # Optimize
        res = minimize(objective, current_h, method=method, jac=gradient, bounds=bounds, constraints=cons, 
                       options={'ftol': 1e-9, 'gtol': 1e-9, 'maxiter': 200})
        
        if res.success:
            current_h = res.x
            # Project to ensure sum constraint is exactly satisfied
            current_h = project_to_sum_box(current_h, target_sum)
            current_c5 = c5_exact(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        
    return best_h, best_c5
