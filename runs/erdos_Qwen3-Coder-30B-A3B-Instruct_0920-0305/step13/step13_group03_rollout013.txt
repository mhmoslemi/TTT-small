def max_corr(h):
    """Compute the maximum overlap for the given h"""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 for the given h"""
    dx = 2.0 / len(h)
    return dx * max_corr(h)
