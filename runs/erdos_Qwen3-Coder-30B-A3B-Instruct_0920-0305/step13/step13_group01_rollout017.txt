import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

# Global constants
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT, returns raw (unscaled) correlation vector."""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value."""
    return dx * np.max(corr)

def evaluate_c5_exact(h):
    """Exact evaluation of C5 using np.correlate."""
    corr = correlate(h, 1 - h, mode='full')
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_and_grad(h, alpha):
    """Compute smooth max and its gradient using log-sum-exp."""
    corr = compute_corr_fft(h)
    # Stabilize the exponentials
    corr_max = np.max(corr)
    exp_corr = np.exp(alpha * (corr - corr_max))
    Z = np.sum(exp_corr)
    weights = exp_corr / Z
    
    # Compute gradient: ∇Φ = dx * Σ_k w_k * (1 - shift(h,k) - shift(h,-k))
    grad = np.zeros_like(h)
    for k in range(len(h)):
        if k == 0:
            grad += weights[k] * (1 - h)
        else:
            # shift h to the right (positive k) and left (negative k)
            shift_right = np.concatenate([h[k:], np.zeros(k)])
            shift_left = np.concatenate([np.zeros(k), h[:-k]])
            grad += weights[k] * (1 - shift_right - shift_left)
    
    return np.sum(weights * corr) / alpha + corr_max, dx * grad

    # 3. Amplitude-tuned sinusoid
    amp = np.random.uniform(0, 0.5)
    sin_amp_h = 0.5 + amp * np.sin(np.pi * x)
    initial_families.append(project_to_sum_box(sin_amp_h, TARGET_SUM))
    
    # 4. Triangular wave
    tri_x = np.linspace(0, 2, n_points)
    tri_h = np.where(tri_x < 1, 2*tri_x, -2*(tri_x-2))
    initial_families.append(project_to_sum_box(tri_h, TARGET_SUM))
    
    # 5. Random uniform
    rand_h = np.random.rand(n_points)
    initial_families.append(project_to_sum_box(rand_h, TARGET_SUM))
    
    # 6. Hybrid
    hybrid_h = 0.7 * sin_h + 0.3 * rand_h
    initial_families.append(project_to_sum_box(hybrid_h, TARGET_SUM))
    
    best_c5 = float('inf')
    best_h = None
    
    # Alpha levels for continuation
    alphas = [1, 5, 20, 80, 300]
    
    # Each start runs the continuation
    num_starts = 0
    for i, h0 in enumerate(initial_families):
        if time.time() - start_time > budget_s - 10:
            break
        num_starts += 1
        
        current_h = h0.copy()
        current_c5 = evaluate_c5_exact(current_h)
        
        # Continuation over alpha levels
        for alpha in alphas:

                break
            
            # Use trust-constr with gradient
            def obj_and_grad(h):
                loss, grad = smooth_max_and_grad(h, alpha)
                return loss, grad
            
            # Define bounds and constraints
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - TARGET_SUM}
            
            res = minimize(obj_and_grad, current_h,
                          method='trust-constr',
                          jac=True,
                          bounds=bounds,
                          constraints=cons,
                          options={'gtol': 1e-9, 'maxiter': 200})
            
            if res.success:
                current_h = res.x.copy()

                current_h = project_to_sum_box(current_h, TARGET_SUM)
                current_c5 = evaluate_c5_exact(current_h)

                    best_c5 = current_c5
                    best_h = current_h.copy()

                res = minimize(obj_and_grad, current_h,
                              method='SLSQP',

                              bounds=bounds,
                              constraints=cons,

                    current_h = project_to_sum_box(current_h, TARGET_SUM)
                    current_c5 = evaluate_c5_exact(current_h)
                    
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = current_h.copy()
    
    # Local search polishing
    if best_h is not None and time.time() - start_time < budget_s - 5:
        current_h = best_h.copy()
        current_c5 = evaluate_c5_exact(current_h)
        
        # Swap moves
        for _ in range(3000):
            if time.time() - start_time > budget_s - 5:
                break
            i, j = np.random.choice(n_points, 2, replace=False)
            temp_h = current_h.copy()
            temp_h[i], temp_h[j] = temp_h[j], temp_h[i]
            temp_h = project_to_sum_box(temp_h, TARGET_SUM)
            temp_c5 = evaluate_c5_exact(temp_h)
            
            if temp_c5 < current_c5:
                current_h = temp_h
                current_c5 = temp_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
    
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_c5_exact(best_h)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        # Return a basic solution if nothing worked
        return ([0.5] * n_points, 0.5, n_points)
