import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time
from scipy.optimize import minimize

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_convex_subproblem(h_current, lags, n_points, dx):
    """Solve the convex subproblem using CVXPY"""
    # Create variables
    h = Variable(n_points)
    T = Variable()
    
    # Constraints
    constraints = [0 <= h, h <= 1]
    constraints.append(sum(h) == n_points / 2)
    
    # Add the lag constraints
    for ell in lags:
        # Compute the linear and quadratic parts
        # I_ell = sum(h_i) + 0.25 * sum((h_i - h_{i+ell})^2) - 0.5 * sum((h_i^t + h_{i+ell}^t) * (h_i + h_{i+ell}))
        linear_part = sum(h)
        
        # Quadratic part: 0.25 * sum((h_i - h_{i+ell})^2)
        quad_part = 0
        if ell >= 0:
            # For ell >= 0, h_{i+ell} is h[i+ell] when i+ell < n
            quad_part = 0.25 * sum((h[i] - h[i + ell])**2 for i in range(n_points - ell))
        else:
            # For ell < 0, h_{i+ell} is h[i+ell] which is h[i+ell] when i+ell >= 0
            quad_part = 0.25 * sum((h[i] - h[i - ell])**2 for i in range(-ell, n_points))
        
        # Linear part: -0.5 * sum((h_i^t + h_{i+ell}^t) * (h_i + h_{i+ell}))
        linear_term = 0
        if ell >= 0:
            for i in range(n_points - ell):
                linear_term += 0.5 * (h_current[i] + h_current[i + ell]) * (h[i] + h[i + ell])
        else:
            for i in range(-ell, n_points):
                linear_term += 0.5 * (h_current[i] + h_current[i - ell]) * (h[i] + h[i - ell])
        
        # Constraint: linear_part + quad_part - linear_term <= T
        constraints.append(linear_part + quad_part - linear_term <= T)
    
    # Objective
    objective = Minimize(T)
    
    # Solve the problem
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            return h.value, T.value
    except:
        pass
    
    return h_current, np.inf

def find_violating_lags(h, n_points, dx, T_val, epsilon=1e-6):
    """Find all lags that violate the constraint"""
    corr = compute_corr_fft(h)
    max_corr = c5_from_corr(corr, n_points)
    
    # Convert to actual overlaps
    actual_overlaps = corr * dx
    
    # Find violations
    violating_lags = []
    for i, overlap in enumerate(actual_overlaps):
        if overlap > T_val * dx + epsilon:
            # Convert index back to lag
            lag = i - (n_points - 1)
            violating_lags.append(lag)
            
    return violating_lags

def round_solution(h_cont, n_points):
    """Round to a binary solution with correct sum"""
    # deterministic rounding: take the top n/2 entries
    indices_sorted = np.argsort(h_cont)[::-1]
    h_binary = np.zeros(n_points)
    h_binary[indices_sorted[:n_points//2]] = 1
    
    # ensure it's exactly the right sum
    ones_count = np.sum(h_binary)
    if ones_count > n_points//2:
        # turn some 1s into 0s
        zeros_indices = np.where(h_binary == 0)[0]
        num_zeros_needed = ones_count - n_points//2
        if len(zeros_indices) >= num_zeros_needed:
            flip_indices = np.random.choice(zeros_indices, num_zeros_needed, replace=False)
            h_binary[flip_indices] = 1
    elif ones_count < n_points//2:
        # turn some 0s into 1s
        ones_indices = np.where(h_binary == 1)[0]
        num_ones_needed = n_points//2 - ones_count
        if len(ones_indices) >= num_ones_needed:
            flip_indices = np.random.choice(ones_indices, num_ones_needed, replace=False)
            h_binary[flip_indices] = 0
    
    # project to feasible set
    h_binary = project_to_sum_box(h_binary, n_points / 2)
    return h_binary

def polish_solution(h_binary, n_points, target_c5, max_iter=1000):
    """Apply critical-lag local search to polished solution"""
    best_h = h_binary.copy()
    corr = compute_corr_fft(best_h)
    current_max = np.max(corr)
    
    for iteration in range(max_iter):
        if current_max <= target_c5:
            break
            
        # Find the worst lag
        worst_lag_idx = np.argmax(corr)
        worst_lag = worst_lag_idx - (n_points - 1)
        
        # Find indices where we have a 1 that creates a mismatch
        mismatch_ones = []
        for i in range(n_points):
            if best_h[i] == 1:
                if worst_lag >= 0:
                    if i + worst_lag < n_points and best_h[i + worst_lag] == 0:
                        mismatch_ones.append(i)
                else:
                    if i - worst_lag >= 0 and best_h[i - worst_lag] == 0:
                        mismatch_ones.append(i)
        
        if not mismatch_ones:
            # No more mismatches, stop
            break
            
        # Pick a random mismatched 1
        i = np.random.choice(mismatch_ones)
        
        # Find candidate zeros that won't create new mismatches
        candidate_zeros = []
        for j in range(n_points):
            if best_h[j] == 0:
                # Check if this would create a new mismatch with the worst lag
                valid = True
                if worst_lag >= 0:
                    if j + worst_lag < n_points and best_h[j + worst_lag] == 1:
                        valid = False
                else:
                    if j - worst_lag >= 0 and best_h[j - worst_lag] == 1:
                        valid = False
                        
                if valid:
                    candidate_zeros.append(j)
                    
        if candidate_zeros:
            j = np.random.choice(candidate_zeros)
        else:
            # If no good zeros, just pick any zero
            zero_indices = np.where(best_h == 0)[0]
            j = np.random.choice(zero_indices)
            
        # Perform swap
        best_h[i] = 0
        best_h[j] = 1
        
        # Recompute correlation
        corr = compute_corr_fft(best_h)
        current_max = np.max(corr)
        
    return best_h, current_max
