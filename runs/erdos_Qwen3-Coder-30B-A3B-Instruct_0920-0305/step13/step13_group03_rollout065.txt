import numpy as np
from scipy.signal import correlate
import time

def compute_overlap(h, n_points):
    """Compute overlap for all lags using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n_points - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_convex_subproblem(h_prev, lags, n_points, target_sum):
    """Solve convex subproblem with cutting plane method"""
    # Define variables
    h = Variable(n_points)
    T = Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1, sum(h) == target_sum]
    
    # Add lag constraints
    for ell in lags:
        # Construct the convex surrogate for lag ell
        # I_ell(h) <= T where I_ell(h) = sum(h_i) + 0.25*sum(h_i-h_{i+ell})^2 - 0.5*sum(h_i^{(t)}+h_{i+ell}^{(t)})*(h_i+h_{i+ell})
        
        # Create the surrogate: quadratic + linear + constant
        if ell < 0:
            # h_i in h, h_{i+ell} in h_prev
            h_shifted = h_prev[-ell:] + np.concatenate([np.zeros(-ell), h_prev[:-ell]])
            h_current = h[:-ell] + np.concatenate([np.zeros(-ell), h])
            h_shifted_prev = h_prev[int(ell):] + np.concatenate([np.zeros(-ell), h_prev[:int(ell)]])
            h_current_prev = h[int(ell):] + np.concatenate([np.zeros(-ell), h])
        else:
            # h_i in h, h_{i+ell} in h_prev
            h_shifted = np.concatenate([h_prev[ell:], np.zeros(ell)]) + h
            h_current = np.concatenate([h[ell:], np.zeros(ell)]) + h
            h_shifted_prev = np.concatenate([h_prev[ell:], np.zeros(ell)]) + h_prev
            h_current_prev = np.concatenate([h[ell:], np.zeros(ell)]) + h_prev
            
        # Create the surrogate: convex part + linear part
        convex_part = sum(h) + 0.25 * sum((h - h_shifted)**2)
        linear_part = -0.5 * sum((h_prev + h_shifted_prev) * (h + h_current))
        surrogate = convex_part + linear_part
        
        constraints.append(surrogate <= T)
    
    # Objective
    objective = Minimize(T)
    
    # Solve the problem
    prob = Problem(objective, constraints)
    
    try:
        # Use ECOS solver
        prob.solve(solver=ECOS, verbose=False)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            return h.value, T.value
    except:
        pass
    
    return None, None

def rounding_method(h_cont, n_points, target_sum):
    """Randomized rounding approach"""
    best_h = None
    best_c5 = float('inf')
    
    # Multiple random draws
    for _ in range(30):
        # Create binary vector by sorting and taking top half
        sorted_indices = np.argsort(h_cont)[::-1]
        temp_h = np.zeros(n_points)
        temp_h[sorted_indices[:int(target_sum)]] = 1
        
        # Ensure exact sum constraint
        temp_h = project_to_sum_box(temp_h, target_sum)
        
        # Evaluate true overlap
        corr = compute_overlap(temp_h, n_points)
        c5_val = c5_from_corr(corr, n_points)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = temp_h.copy()
    
    return best_h, best_c5

def critical_lag_polish(h_binary, n_points, target_c5):
    """Critical lag greedy polishing"""
    start_time = time.time()
    
    # Compute initial correlations
    corr = compute_overlap(h_binary, n_points)
    max_lag_idx = np.argmax(corr)
    max_corr_val = corr[max_lag_idx]
    
    max_lag = max_lag_idx - (n_points - 1)  # Convert index to lag value
    
    while max_corr_val > target_c5 and time.time() - start_time < 180:  # 3 minutes max
        # Find indices where we have 1 in h_binary but 0 in shifted version for this lag
        if max_lag > 0:
            positions = np.where((h_binary == 1) & (np.concatenate([np.zeros(max_lag), h_binary[:-max_lag]]) == 0))[0]
        else:
            positions = np.where((h_binary == 1) & (np.concatenate([h_binary[-max_lag:], np.zeros(-max_lag)]) == 0))[0]
        
        if len(positions) == 0:
            break
            
        i = np.random.choice(positions)
        
        # Find valid positions to swap with
        if max_lag > 0:
            valid_positions = np.where((h_binary == 0) & (np.concatenate([np.zeros(max_lag), h_binary[:-max_lag]]) == 1))[0]
        else:
            valid_positions = np.where((h_binary == 0) & (np.concatenate([h_binary[-max_lag:], np.zeros(-max_lag)]) == 1))[0]
            
        if len(valid_positions) > 0:
            j = np.random.choice(valid_positions)
        else:
            # Fall back to random zero
            valid_zeros = np.where(h_binary == 0)[0]
            if len(valid_zeros) > 0:
                j = np.random.choice(valid_zeros)
            else:
                break
                
        # Perform swap
        h_binary[i], h_binary[j] = h_binary[j], h_binary[i]
        
        # Recompute correlation
        corr = compute_overlap(h_binary, n_points)
        max_lag_idx = np.argmax(corr)
        max_corr_val = corr[max_lag_idx]
        max_lag = max_lag_idx - (n_points - 1)
    
    return h_binary, max_corr_val

        best_c5 = current_c5
        best_h = h_current.copy()

    # Convex-Concave Procedure with cutting planes
    lags = [0, -24, 24, -48, 48, -72, 72, -96, 96]  # Key lags, periodic structure
    
    max_iter = 100
    iter_count = 0
    
    t1 = start_time + 0.4 * budget_s  # Leave time for rounding and polishing
    
    while time.time() < t1 and iter_count < max_iter:
        iter_count += 1
        h_sol, t_sol = solve_convex_subproblem(h_current, lags, n_points, target_sum)
        
        if h_sol is None:
            break
        
        h_current = h_sol.copy()
        h_current = project_to_sum_box(h_current, target_sum)
        
        # Evaluate all overlaps
        corr_full = compute_overlap(h_current, n_points)
        all_overlaps = corr_full
        
        # Check if we have new constraints that violate the bound
        violations = np.where(all_overlaps > t_sol + 1e-6)[0]
        
        # Add most violated lags
        if len(violations) > 0:
            # Add the top violation
            # Use the indices of most violated lags
            max_violation_idx = np.argmax(all_overlaps)
            new_lag = max_violation_idx - (n_points - 1)
            
            if new_lag not in lags:
                lags.append(new_lag)
                if len(lags) > 80:  # Avoid too many lags
                    lags = lags[:40]  # Keep only top 40 lags

            best_c5 = current_c5

    if best_h is None:
        last_h = h0.copy()
        last_h = project_to_sum_box(last_h, target_sum)
        corr = compute_overlap(last_h, n_points)
        last_c5 = c5_from_corr(corr, n_points)

            best_c5 = c5_rounded
            best_h = h_rounded.copy()
        
        # Polish
        if best_h is not None:
            polished_h, polished_c5 = critical_lag_polish(best_h.copy(), n_points, target_c5)

    if best_c5 > target_c5:

                    best_h = h_test.copy()
