import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1 << (L - 1).bit_length()
    corr = np.fft.ifft(np.fft.fft(h_rev, fft_size) * np.fft.fft(b, fft_size)).real
    return corr[:L]

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return np.max(corr)

def evaluate_c5_precise(h):
    """Precise evaluation of C5 using numpy correlate"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x - lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = np.min(x) - 1.0, np.max(x) + 1.0
    tol = 1e-12
    for _ in range(60):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x - (a + b)/2, 0, 1)
    return result

def compute_subgradient(h, corr):
    """Compute subgradient of max correlation with respect to h"""
    n = len(h)
    max_val = np.max(corr)
    eps = 1e-12
    active_idx = np.where(corr >= max_val - eps)[0]
    
    m = len(active_idx)
    g = np.full(n, m, dtype=float)
    
    for idx in active_idx:
        k = idx - (n - 1)
        if k >= 0:
            # Subtract h_{j+k} for j from 0 to n-k-1
            g[:n - k] -= h[k:]
            # Subtract h_{j-k} for j from k to n-1
            g[k:] -= h[:n - k]
        else:
            d = -k
            # For k < 0, use equivalent logic
            g[d:] -= h[:n - d]
            g[:n - d] -= h[d:]
    
    return g, max_val

def generate_sine_initial():
    """Generate initial solution based on sine wave"""
    x = (np.arange(n_points) + 0.5) * dx
    # Generate random amplitude and phase
    amplitude = np.random.rand() * 0.5
    phase = np.random.rand() * 2 * np.pi
    h = 0.5 + amplitude * np.sin(np.pi * x + phase)
    # Ensure values are within [0,1]
    h = np.clip(h, 0, 1)
    # Project to correct sum
    h = project_to_sum_box(h, target_sum)
    return h

        initial_solutions.append(h_sine.copy())

        h_rand = np.random.rand(n_points)

        # Subgradient descent with Adam-style momentum
        base_lr = 0.3
        m = np.zeros(n_points)
        v = np.zeros(n_points)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8

            v_hat = v / (1 - beta2**(iter_num + 1))

            h_new = project_to_sum_box(h_new, target_sum)
