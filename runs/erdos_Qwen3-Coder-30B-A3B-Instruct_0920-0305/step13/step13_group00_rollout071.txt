def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Fast correlation computation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_gradient(h, corr, alpha=10.0):
    """Compute gradient using softmax approximation of max"""
    # Compute weights for softmax approximation
    max_corr = np.max(corr)
    if max_corr > 0:
        w = np.exp(alpha * (corr - max_corr))
    else:
        w = np.exp(alpha * corr)
    
    w = w / np.sum(w)
    
    # Initialize gradient
    g = np.ones(len(h))
    
    # Compute weighted sum of shifted values for each shift
    for i in range(len(h)):
        # For each shift in the full correlation
        for j in range(len(corr)):
            shift = j - (len(h) - 1)
            weight = w[j]
            
            if shift == 0:
                g[i] -= 2 * weight * h[i]
            elif shift > 0:
                # h[i+shift] contribution from left to right shift
                if i + shift < len(h):
                    g[i] -= weight * h[i + shift]
                # h[i-shift] contribution from right to left shift  
                if i - shift >= 0:
                    g[i] -= weight * h[i - shift]
            else:
                # shift < 0, so shift is negative
                t = -shift
                # h[i+t] contribution from right to left shift
                if i + t < len(h):
                    g[i] -= weight * h[i + t]
                # h[i-t] contribution from left to right shift
                if i - t >= 0:
                    g[i] -= weight * h[i - t]
    
    return g
