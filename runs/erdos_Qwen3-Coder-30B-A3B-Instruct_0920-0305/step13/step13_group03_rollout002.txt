import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.fft import fft, ifft

# Global variables
n_points = 120
dx = 2.0 / n_points
target_M = int(np.floor(0.3807 / dx))  # 22
ℓ_max = n_points - target_M  # 98

def max_corr(h):
    """Compute max correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(ifft(fft(h_rev, L) * fft(b, L)))
    return np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    return dx * max_corr(h)

def round_to_binary(h_cont):
    """Round continuous solution to binary with exact sum constraint"""
    h_binary = (h_cont >= 0.5).astype(float)
    current_sum = np.sum(h_binary)
    target_sum = n_points // 2
    
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        diff = int(current_sum - target_sum)
        # Find indices with values closest to 0.5 to flip
        distances = np.abs(h_cont - 0.5)
        flip_indices = np.argpartition(distances, diff)[:diff]
        h_binary[flip_indices] = 0
    elif current_sum < target_sum:
        # Need to flip some 0s to 1s
        diff = int(target_sum - current_sum)
        # Find indices with values closest to 0.5 to flip
        distances = np.abs(h_cont - 0.5)
        flip_indices = np.argpartition(distances, diff)[:diff]
        h_binary[flip_indices] = 1
    
    return h_binary

def compute_lag_indices(h, ℓ):
    """Compute the indices where h[i]=1 and h[i+ℓ]=0"""
    if ℓ >= 0:
        # Positive lag: h[i] and h[i+ℓ]
        valid_range = range(0, len(h) - ℓ)
        return [i for i in valid_range if h[i] == 1 and h[i + ℓ] == 0]
    else:
        # Negative lag: h[i+ℓ] and h[i] 
        valid_range = range(-ℓ, len(h))
        return [i for i in valid_range if h[i + ℓ] == 1 and h[i] == 0]

def greedy_critical_lag_search(h, max_iter=10000):
    """Greedy local search for critical lag"""
    current_h = h.copy()
    corr = correlate(current_h, 1 - current_h, mode="full")
    current_M = np.max(corr)
    
    # Build lag to indices mapping
    lag_to_indices = {}
    for ℓ in range(-ℓ_max, ℓ_max + 1):
        lag_to_indices[ℓ] = compute_lag_indices(current_h, ℓ)
    
    iter_count = 0
    while iter_count < max_iter and current_M > target_M:
        # Find worst lag
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(current_h) - 1)
        
        # Get mismatched positions for worst lag
        mismatched = lag_to_indices[max_lag]
        if not mismatched:
            break
            
        # Pick a random mismatched position
        i = np.random.choice(mismatched)
        
        # Find candidate positions to swap with
        candidates = []
        if max_lag >= 0:
            # Positive lag: i and i+lag
            for j in range(len(current_h)):
                if current_h[j] == 0 and (j + max_lag >= len(current_h) or current_h[j + max_lag] == 1):
                    candidates.append(j)
        else:
            # Negative lag: i+lag and i
            for j in range(len(current_h)):
                if current_h[j] == 0 and (j >= -max_lag and current_h[j - max_lag] == 1):
                    candidates.append(j)
        
        if candidates:
            j = np.random.choice(candidates)
        else:
            # Fallback: pick a random zero
            zero_indices = np.where(current_h == 0)[0]
            if len(zero_indices) > 0:
                j = np.random.choice(zero_indices)
            else:
                break
            
        # Perform swap
        current_h[i], current_h[j] = current_h[j], current_h[i]
        
        # Update lag indices efficiently
        for lag in range(-ℓ_max, ℓ_max + 1):
            if lag >= 0:
                if i < len(current_h) - lag and current_h[i] == 1 and current_h[i + lag] == 0:
                    lag_to_indices[lag].append(i)
                elif i < len(current_h) - lag and current_h[i] == 0 and current_h[i + lag] == 1:
                    try:
                        lag_to_indices[lag].remove(i)
                    except:
                        pass
                if j < len(current_h) - lag and current_h[j] == 1 and current_h[j + lag] == 0:
                    lag_to_indices[lag].append(j)
                elif j < len(current_h) - lag and current_h[j] == 0 and current_h[j + lag] == 1:
                    try:
                        lag_to_indices[lag].remove(j)
                    except:
                        pass
            else:
                if i + lag >= 0 and current_h[i + lag] == 1 and current_h[i] == 0:
                    lag_to_indices[lag].append(i + lag)
                elif i + lag >= 0 and current_h[i + lag] == 0 and current_h[i] == 1:
                    try:
                        lag_to_indices[lag].remove(i + lag)
                    except:
                        pass
                if j + lag >= 0 and current_h[j + lag] == 1 and current_h[j] == 0:
                    lag_to_indices[lag].append(j + lag)
                elif j + lag >= 0 and current_h[j + lag] == 0 and current_h[j] == 1:
                    try:
                        lag_to_indices[lag].remove(j + lag)
                    except:
                        pass
        
        # Recompute correlation
        corr = correlate(current_h, 1 - current_h, mode="full")
        current_M = np.max(corr)
        iter_count += 1
        
    return current_h, current_M

                    constraints.extend([
                        W[(i, ℓ)] <= h_vars[i],
                        W[(i, ℓ)] <= h_vars[i + ℓ],
                        W[(i, ℓ)] >= h_vars[i] + h_vars[i + ℓ] - 1
                    ])

                    W[(i, ℓ)] = cp.Variable(boolean=True)

                        W[(i, ℓ)] >= h_vars[i] + h_vars[i + ℓ] - 1
                    ])

            if ℓ >= 0:
                expr = cp.sum(h_vars[:n_points-ℓ] - [W[(i, ℓ)] for i in range(n_points-ℓ)])
            else:
                expr = cp.sum(h_vars[-ℓ:] - [W[(i, ℓ)] for i in range(-ℓ, n_points)])

                        best_h = h_opt
                        best_c5 = c5
                current_h = best_h
                current_c5 = best_c5
    
    return (current_h.tolist(), float(current_c5), n_points)
