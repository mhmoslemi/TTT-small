import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Create CVXPY variables
    h = Variable(n_points)
    t = Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == target_sum,
        t >= 0
    ]
    
    # Add McCormick constraints for each shift
    for s in range(1, n_points):
        # For shift s, we have indices i in [0, n_points-1-s]
        # Create auxiliary variables y_{i,s} for product h_i * h_{i+s}
        y_vars = []
        
        for i in range(n_points - s):
            y = Variable()
            y_vars.append(y)
            
            # McCormick constraints
            constraints.extend([
                y >= 0,
                y <= h[i],
                y <= h[i + s],
                y >= h[i] + h[i + s] - 1
            ])
        
        # Add constraint that sum of y variables is at least (target_sum - t) - prefix_sum
        # This represents: sum of products >= (target_sum - t) - sum_{i=0}^{s-1} h_i
        prefix_sum = sum(h[:s])  # This is a linear expression
        sum_y = sum(y_vars)
        constraints.append(sum_y >= target_sum - t - prefix_sum)
    
    # Special case for s=0 (shift 0)
    # Here we have products h_i * h_i = h_i
    y_0_vars = []
    for i in range(n_points):
        y = Variable()
        y_0_vars.append(y)
        constraints.extend([
            y >= 0,
            y <= h[i],
            y <= h[i],
            y >= 2*h[i] - 1
        ])
    
    # For shift s=0, the overlap is just the sum of h_i, so the correlation is 
    # (target_sum - sum of h_i) = 0. But let's handle it properly
    # Actually for shift s=0, the correlation is sum of h_i * (1-h_i), 
    # which we can approximate with target_sum - sum of y_0_vars
    sum_y_0 = sum(y_0_vars)
    constraints.append(target_sum - sum_y_0 <= t)
    
    # Objective: minimize t
    objective = Minimize(t)
    
    # Create and solve the problem
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h_opt = h.value
            t_opt = t.value
            c5_bound = dx * t_opt
        else:
            # Revert to the initial solution
            h_opt = np.array(initial_h_values)
            c5_bound = evaluate_erdos_solution(h_opt)
    except:
        # Revert to the initial solution
        h_opt = np.array(initial_h_values)
        c5_bound = evaluate_erdos_solution(h_opt)
        
    return (h_opt.tolist(), float(c5_bound), n_points)
