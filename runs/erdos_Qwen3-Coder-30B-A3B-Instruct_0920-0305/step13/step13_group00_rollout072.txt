import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Create CVXPY variables
    h = Variable(n_points)
    t = Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == target_sum,
        t >= 0
    ]
    
    # Add McCormick constraints for each product term
    # We'll use a more efficient approach by directly referencing the correlation terms
    # For each shift s from 0 to n_points-1:
    # C'_s = (target_sum - prefix_sum[s]) - sum_over_i h_i * h_{i+s}
    
    # Compute prefix sums
    prefix_sum = [sum(h[:i]) for i in range(n_points)]
    
    # Instead of creating explicit variables for each y_{i,s}, 
    # we'll add constraints using McCormick envelopes directly
    # For each shift s from 0 to n_points-1
    for s in range(n_points):
        # For shift s, we compute sum_{i=0}^{n_points-1-s} h_i * h_{i+s}
        # This is the overlap term for shift s
        if s == 0:
            # Handle shift 0  
            overlap_s = sum(h[i] * h[i] for i in range(n_points))  # This just counts sum of squares
        else:
            # For shift s, we have overlapping terms for i from 0 to n_points-1-s
            overlap_s = sum(h[i] * h[i+s] for i in range(n_points-s))
            
        # Constraint for this shift: C'_s = (target_sum - P_s) - overlap_s <= t
        # But we want to minimize max(C'_s) = max((target_sum - P_s) - overlap_s)
        # Equivalently, we minimize t such that (target_sum - P_s) - overlap_s <= t
        # i.e., overlap_s >= (target_sum - P_s) - t
        
        # Compute prefix sum P_s = sum_{i=0}^{s-1} h_i
        if s == 0:
            prefix = 0
        else:
            prefix = sum(h[i] for i in range(s))
            
        # Apply constraint: overlap_s >= (target_sum - prefix) - t
        constraints.append(overlap_s >= (target_sum - prefix) - t)
    
    # Objective: minimize t
    objective = Minimize(t)
    
    # Create and solve the problem
    prob = Problem(objective, constraints)
    
    # Solve with ECOS (efficient for this structure)
    try:
        prob.solve(solver=ECOS, verbose=False, max_iters=5000)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h_opt = h.value
            t_opt = t.value
            c5_bound = dx * t_opt
        else:
            # Fallback to the previous solution
            h_opt = np.array(initial_h_values)
            c5_bound = evaluate_erdos_solution(h_opt)
    except:
        # Fallback to previous approach if LP fails
        h_opt = np.array(initial_h_values)
        c5_bound = evaluate_erdos_solution(h_opt)
    
    # Refine with local optimization to improve solution quality
    if time.time() - start_time < budget_s * 0.9:
        # Try gradient-like approach for small improvements
        h_refined = h_opt.copy()
        max_iter = 1000
        for iter_num in range(max_iter):
            if time.time() - start_time >= budget_s * 0.95:
                break
                
            # Simple gradient approximation
            grad = np.zeros_like(h_refined)
            eps = 1e-4
            
            for i in range(n_points):
                # Forward difference
                h_plus = h_refined.copy()
                h_plus[i] = min(h_plus[i] + eps, 1.0)
                h_plus = np.clip(h_plus, 0, 1)
                
                # Project to sum constraint
                sum_diff = np.sum(h_plus) - target_sum
                if abs(sum_diff) > 1e-6:
                    delta = sum_diff / n_points
                    h_plus -= delta
                
                c5_plus = evaluate_erdos_solution(h_plus)
                
                # Backward difference
                h_minus = h_refined.copy()
                h_minus[i] = max(h_minus[i] - eps, 0.0)
                h_minus = np.clip(h_minus, 0, 1)
                
                # Project to sum constraint
                sum_diff = np.sum(h_minus) - target_sum
                if abs(sum_diff) > 1e-6:
                    delta = sum_diff / n_points
                    h_minus -= delta
                
                c5_minus = evaluate_erdos_solution(h_minus)
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            step_size = 0.01
            h_new = h_refined - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            
            # Project to sum constraint
            sum_diff = np.sum(h_new) - target_sum
            if abs(sum_diff) > 1e-6:
                delta = sum_diff / n_points
                h_new -= delta
            
            c5_new = evaluate_erdos_solution(h_new)
            
            if c5_new < evaluate_erdos_solution(h_refined):
                h_refined = h_new.copy()
                c5_bound = c5_new
            else:
                # Reduce step size
                step_size *= 0.9
                
    return (h_refined.tolist(), float(c5_bound), n_points)
