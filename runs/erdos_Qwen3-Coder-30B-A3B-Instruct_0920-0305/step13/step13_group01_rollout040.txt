import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1 << (L - 1).bit_length()
    corr = np.fft.ifft(np.fft.fft(h_rev, fft_size) * np.fft.fft(b, fft_size)).real[:L]
    return corr

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT - returns raw correlation max"""
    corr = compute_corr_fft(h)
    return np.max(corr)

def evaluate_c5_precise(h):
    """Precise evaluation using np.correlate"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x - lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = np.min(x) - 1.0, np.max(x) + 1.0
    tol = 1e-12
    
    for _ in range(60):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x - (a + b)/2, 0, 1)
    return result

def compute_subgradient(h, corr):
    """Compute subgradient for the max correlation"""
    n = len(h)
    max_val = np.max(corr)
    eps = 1e-12
    active_idx = np.where(corr >= max_val - eps)[0]
    
    # Initialize gradient with constant term
    g = np.full(n, len(active_idx), dtype=float)
    
    for idx in active_idx:
        k = idx - (n - 1)
        if k >= 0:
            # Subtract h_{j + k} for j from 0 to n - k -1
            g[:n - k] -= h[k:]
            # Subtract h_{j - k} for j >= k
            g[k:] -= h[:n - k]
        else:
            # For k < 0, equivalent to positive shift but reversed
            d = -k
            g[d:] -= h[:n - d]
            g[:n - d] -= h[d:]
    
    return g, max_val

def generate_sin_initial():
    """Generate sinusoidal initial solution"""
    x = (np.arange(n_points) + 0.5) * dx
    # Random amplitude and phase
    amplitude = np.random.rand() * 0.5
    phase = np.random.rand() * 2 * np.pi
    h = 0.5 + amplitude * np.sin(np.pi * x + phase)
    # Ensure within [0,1] bounds
    h = np.clip(h, 0, 1)
    # Project to exact sum constraint
    h = project_to_sum_box(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from repeated initial solution if available
    if 'initial_h_values' in globals() and len(initial_h_values) == 120:
        h0 = np.repeat(initial_h_values, 2)
    else:
        h0 = np.random.rand(n_points)
        h0 = project_to_sum_box(h0, target_sum)
    
    # Evaluate initial solution
    best_c5 = evaluate_c5_precise(h0)
    best_h = h0.copy()
    
    # Generate multiple initial solutions for restarts
    initial_solutions = [h0.copy()]
    
    # Add sinusoidal initial solutions
    for _ in range(4):
        initial_solutions.append(generate_sin_initial())
    
    # Add random uniform solutions
    for _ in range(2):
        h_rand = np.random.rand(n_points)
        h_rand = project_to_sum_box(h_rand, target_sum)
        initial_solutions.append(h_rand)
    
    # Run subgradient descent from each initial solution
    max_iters_per_restart = 10000
    base_lr = 0.3
    
    for i, h_start in enumerate(initial_solutions):
        if time.time() - start_time > budget_s - 10:
            break
            
        h = h_start.copy()
        current_c5 = evaluate_c5_fast(h)
        best_local_c5 = current_c5
        
        for iter_count in range(max_iters_per_restart):
            if time.time() - start_time > budget_s - 5:
                break
                
            corr = compute_corr_fft(h)
            g, max_corr = compute_subgradient(h, corr)
            
            # Step size schedule
            step = base_lr / np.sqrt(iter_count + 1)
            
            # Apply step
            h_new = h - step * g
            
            # Project to constraints
            h_new = project_to_sum_box(h_new, target_sum)
            
            # Evaluate new solution
            new_corr = compute_corr_fft(h_new)
            new_c5_raw = np.max(new_corr)
            
            if new_c5_raw < max_corr:
                h = h_new
                max_corr = new_c5_raw
                if new_c5_raw * dx < best_c5:
                    best_c5 = new_c5_raw * dx
                    best_h = h_new.copy()
            else:
                h = h_new
                
            if new_c5_raw < best_local_c5:
                best_local_c5 = new_c5_raw
    
    # Final local search with neighbor swaps
    if time.time() - start_time < budget_s - 5:
        h = best_h.copy()
        current_c5 = evaluate_c5_precise(h)
        for _ in range(5000):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Random swap
            i, j = np.random.randint(0, n_points, 2)
            h_swap = h.copy()
            h_swap[i], h_swap[j] = h_swap[j], h_swap[i]
            
            # Project to constraints
            h_swap = project_to_sum_box(h_swap, target_sum)
            
            # Evaluate
            new_c5 = evaluate_c5_precise(h_swap)
            if new_c5 < current_c5:
                h = h_swap
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
