def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_gradient(h, corr, weights):
    """Compute weighted gradient"""
    g = np.ones(len(h))
    
    # For each shift, add weighted contribution
    for i in range(len(corr)):
        s = i - (len(h) - 1)  # shift value
        w = weights[i]
        
        if s == 0:
            g -= 2 * w * h
        elif s > 0:
            # h_{i+s} contribution to g[0:N-s]
            g[:len(h)-s] -= w * h[s:]
            # h_{i-s} contribution to g[s:N]
            g[s:] -= w * h[:len(h)-s]
        else:
            # s < 0
            t = -s
            # h_{i+s} = h_{i-t} contribution to g[t:N]  
            g[t:] -= w * h[:len(h)-t]
            # h_{i-s} = h_{i+t} contribution to g[0:N-t]
            g[:len(h)-t] -= w * h[t:]
    
    return g
