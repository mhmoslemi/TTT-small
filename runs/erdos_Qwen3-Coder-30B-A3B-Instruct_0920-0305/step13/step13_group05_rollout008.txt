def softmax_smooth_max(arr, beta=100.0):
    """Compute smooth approximation of maximum using softmax"""
    if len(arr) == 0:
        return 0.0
    # Add a small epsilon to avoid numerical issues
    arr = np.array(arr) + 1e-10
    exp_arr = np.exp(beta * arr)
    return np.sum(arr * exp_arr) / np.sum(exp_arr)

def softmax_smooth_max_gradient(arr, beta=100.0):
    """Compute gradient of smooth maximum"""
    if len(arr) == 0:
        return np.array([])
    arr = np.array(arr) + 1e-10
    exp_arr = np.exp(beta * arr)
    sum_exp = np.sum(exp_arr)
    sum_exp_arr = np.sum(arr * exp_arr)
    
    grad = exp_arr * (beta * arr - sum_exp_arr / sum_exp) / sum_exp
    return grad

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

            # Need to decrease some values

                    result[sorted_indices[i]] -= reduce_by
                    decrease_amount -= reduce_by

                        break

            for i in range(threshold_idx - 1, -1, -1):
                if result[sorted_indices[i]] < 1:
                    increase_by = min(increase_amount, 1 - result[sorted_indices[i]])
                    result[sorted_indices[i]] += increase_by
                    increase_amount -= increase_by
                    if increase_amount <= 0:
                        break
    
    return result

def compute_smooth_c5(h, beta=100.0):
    """Compute smooth approximation of C5 objective"""
    corr = compute_corr_fft(h)
    return dx * softmax_smooth_max(corr, beta)

    corr = compute_corr_fft(h)
    max_index = np.argmax(corr)
    
    # Get the gradient with respect to h
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr_fft = np.fft.fft(h_rev, L) * np.fft.fft(b, L)
    corr_grad = np.fft.ifft(np.conj(np.fft.fft(h_rev, L)) * np.fft.fft(1 - h, L)).real
    
    # Approximate gradient of max using softmax gradient 
    grad = np.zeros_like(h)

        h_minus[i] -= 1e-6

        # Estimate derivative at point i
        grad[i] = (corr_plus[max_index] - corr_minus[max_index]) / (2e-6)
    
    return grad * dx
