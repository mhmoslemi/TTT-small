import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

# Global parameters
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT, returning raw (unscaled) values"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5_exact(h):
    """Compute exact C5 using np.correlate"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_objective(h, alpha=1.0):
    """Smooth approximation to max via log-sum-exp"""
    corr = compute_corr_fft(h)
    # Stable compute log-sum-exp with stabilization
    max_corr = np.max(corr)
    exp_corr = np.exp(alpha * (corr - max_corr))
    log_sum_exp = max_corr + np.log(np.sum(exp_corr))
    return dx * log_sum_exp / alpha

def smooth_max_gradient(h, alpha=1.0):
    """Analytic gradient of smooth max objective"""
    corr = compute_corr_fft(h)
    max_corr = np.max(corr)
    exp_corr = np.exp(alpha * (corr - max_corr))
    weights = exp_corr / np.sum(exp_corr)
    
    # Gradient computation for S_k = sum h_i * (1 - h_{i+k})
    # ∂S_k/∂h_i = 1 - h_{i+k} - h_{i-k}
    grad = np.zeros_like(h)
    L = len(h)
    for k in range(-L+1, L):
        if k >= 0:
            grad += weights[k+L-1] * (1 - h[k:] - np.pad(h[:L-k], (0, k), 'constant'))
        else:
            grad += weights[k+L-1] * (1 - np.pad(h[-k:], (0, -k), 'constant') - h[:L+k])
    
    return dx * grad

    # 3. Amplitude-tuned sinusoid

    h2 = 0.5 + a * np.sin(np.pi * x)
    initial_solutions.append(project_to_sum_box(h2, target_sum))

    h3 = np.zeros(n_points)

    h3 = h3 * (2 / np.sum(h3))  # Scale to area 1

    h4 = np.random.rand(n_points)

                    options={'maxiter': 200}
                )

                # Use trust-constr for better precision at high alpha
                res = minimize(
                    smooth_max_objective,
                    current_h,
                    args=(alpha,),

    # Local search polishing

        current_h = best_h.copy()
        current_c5 = evaluate_c5_exact(current_h)
        
        max_swaps = 3000
        no_improvement_count = 0
        
        for _ in range(max_swaps):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Try random swap
            i, j = np.random.choice(n_points, 2, replace=False)
            temp_h = current_h.copy()
            temp_h[i], temp_h[j] = temp_h[j], temp_h[i]
            temp_h = project_to_sum_box(temp_h, target_sum)
            
            temp_c5 = evaluate_c5_exact(temp_h)
            
            if temp_c5 < current_c5:
                current_h = temp_h
                current_c5 = temp_c5
                no_improvement_count = 0
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
            else:
                no_improvement_count += 1
                if no_improvement_count >= 500:
                    break

    # Final evaluation of the true C5 bound
    final_c5 = evaluate_c5_exact(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
