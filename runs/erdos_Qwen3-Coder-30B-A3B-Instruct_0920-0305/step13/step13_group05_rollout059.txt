TARGET_SUM = None

dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort values to find the right threshold
    sorted_x = np.sort(x)[::-1]
    cumsum = np.cumsum(sorted_x)
    
    # Find threshold
    threshold_idx = -1
    for i in range(len(sorted_x)):
        if cumsum[i] <= target_sum:
            threshold_idx = i
        else:
            break
    
    if threshold_idx == -1:
        # All values should be 0 or 1, adjust based on target
        threshold = max(0, target_sum / len(x))
        projected = np.clip(x, 0, threshold)
        current_sum = np.sum(projected)
        if current_sum > target_sum:
            # Need to decrease some values
            diff = current_sum - target_sum
            indices = np.argsort(projected)[::-1][:int(diff/dx)]
            projected[indices] = 0
        elif current_sum < target_sum:
            # Need to increase some values
            diff = target_sum - current_sum
            indices = np.argsort(projected)[:int(diff/dx)]
            projected[indices] = 1
    else:
        # Normal projection
        threshold = (cumsum[threshold_idx] - target_sum) / (len(x) - threshold_idx - 1)
        projected = np.clip(x - threshold, 0, 1)
    
    return projected

    exp_x = np.exp(beta * x)

def projected_gradient_descent(h_start, max_iter=2000, step_size=1e-3, beta=100, tolerance=1e-6):
    """Projected gradient descent with backtracking line search"""
    h = h_start.copy()
    prev_obj = smooth_objective(h, beta)
    
    for i in range(max_iter):
        # Compute gradient
        grad = smooth_objective_gradient(h, beta)
        
        # Line search for step size
        alpha = step_size
        h_candidate = h - alpha * grad
        h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
        
        # Backtracking line search
        while True:
            new_obj = smooth_objective(h_candidate, beta)
            if new_obj <= prev_obj - alpha * 0.5 * np.sum(grad * grad):
                h = h_candidate
                break
            else:
                alpha *= 0.5
                h_candidate = h - alpha * grad
                h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
        
        obj = smooth_objective(h, beta)
        if abs(prev_obj - obj) < tolerance:
            break
        prev_obj = obj
        
        # Occasionally try perturbation to escape local minima

            if np.random.rand() < 0.3:
                h += np.random.normal(0, 0.01, len(h))
                h = project_to_sum_box(h, TARGET_SUM)
    
    return h
