```python
import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1 << (L - 1).bit_length()
    corr = np.fft.ifft(np.fft.fft(h_rev, fft_size) * np.fft.fft(b, fft_size)).real
    return corr[:L]

def evaluate_c5_precise(h):
    """Evaluate C5 using np.correlate for precise result"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def evaluate_c5_fast(corr):
    """Evaluate C5 from correlation values (already multiplied by dx)"""
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda to satisfy sum constraint
    lo = np.min(x) - 1.0
    hi = np.max(x) + 1.0
    
    for _ in range(60):
        lam = (lo + hi) / 2
        proj = np.clip(x - lam, 0, 1)
        s = np.sum(proj)
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    
    return np.clip(x - (lo + hi) / 2, 0, 1)

def compute_subgradient(h, corr):
    """Compute subgradient of max correlation with respect to h"""
    n = len(h)
    max_val = np.max(corr)
    eps = 1e-12
    active_idx = np.where(corr >= max_val - eps)[0]
    
    # Count active indices
    m = len(active_idx)
    
    # Initialize gradient with constant term: 1 per active lag per component
    g = np.full(n, m, dtype=float)
    
    for idx in active_idx:
        k = idx - (n - 1)
        if k >= 0:
            # Subtract h_{j + k} for j from 0 to n - k - 1
            g[:n - k] -= h[k:]
            # Subtract h_{j - k} for j from k to n - 1
            g[k:] -= h[:n - k]
        else:
            # For negative k, equivalent to positive k but swap roles
            d = -k
            g[d:] -= h[:n - d]
            g[:n - d] -= h[d:]
    
    return g

def generate_sinusoidal_start():
    """Generate sinusoidal initial solution"""
    x = (np.arange(n_points) + 0.5) * dx
    # Random amplitude between 0 and 0.5 to ensure values stay in [0,1]
    amplitude = np.random.rand() * 0.5
    phase = np.random.rand() * 2 * np.pi
    h = 0.5 + amplitude * np.sin(np.pi * x + phase)
    # Project to ensure sum constraint
    h = project_to_sum_box(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with repeated initial values if available
    if 'initial_h_values' in globals() and len(initial_h_values) == 120:
        h0 = np.repeat(initial_h_values, 2)
    else:
        # Generate random initial solution
        h0 = np.random.rand(n_points)
        h0 = project_to_sum_box(h0, target_sum)
    
    # Evaluate initial solution
    best_c5 = evaluate_c5_precise(h0)
    best_h = h0.copy()
    
    # Generate additional start candidates
    candidates = [h0]
    
    # Add sinusoidal starts
    for _ in range(3):
        candidates.append(generate_sinusoidal_start())
    
    # Add uniform random starts
    for _ in range(2):
        h_random = np.random.rand(n_points)
        h_random = project_to_sum_box(h_random, target_sum)
        candidates.append(h_random)
    
    # Run subgradient descent on each candidate
    max_iters_per_start = 20000
    base_lr = 0.5
    
    for candidate in candidates:
        if time.time() - start_time > budget_s - 10:
            break
            
        h = candidate.copy()
        current_c5 = evaluate_c5_precise(h)
        
        for iteration in range(max_iters_per_start):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute correlation
            corr = compute_corr_fft(h)
            
            # Compute subgradient
            g = compute_subgradient(h, corr)
            
            # Compute step size
            step = base_lr / np.sqrt(iteration + 1)
            
            # Update
            h_new = h - step * g
            
            # Project
            h_new = project_to_sum_box(h_new, target_sum)
            
            # Evaluate
            new_c5 = evaluate_c5_precise(h_new)
            
            if new_c5 < current_c5:
                h = h_new
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
    
    # Final local search with swaps
    if time.time() - start_time < budget_s - 5:
        h = best_h.copy()
        current_c5 = evaluate_c5_precise(h)
        
        for _ in range(10000):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Try a random swap
            i, j = np.random.randint(0, n_points, 2)
            temp_h = h.copy()
            temp_h[i], temp_h[j] = temp_h[j], temp_h[i]
            temp_h = project_to_sum_box(temp_h, target_sum)
            
            temp_c5 = evaluate_c5_precise(temp_h)
            if temp_c5 < current_c5:
                h = temp_h
                current_c5 = temp_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
```
