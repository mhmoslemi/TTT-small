def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute full correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_gradient(h, corr, weights):
    """Compute weighted gradient for given correlation and weights"""
    g = np.ones(len(h))
    
    # For each shift, accumulate contribution
    for j in range(len(corr)):
        w = weights[j]
        s = j - (len(h) - 1)
        
        if s == 0:
            g -= 2 * w * h
        elif s > 0:
            # For shift s > 0:
            # term -h_{i+s} contributes to g[0:N-s] 
            # term -h_{i-s} contributes to g[s:N]
            g[:len(h)-s] -= w * h[s:]
            g[s:] -= w * h[:len(h)-s]
        else:  # s < 0
            # For shift s < 0, let t = -s > 0:
            # term -h_{i+s} = -h_{i-t} contributes to g[t:]
            # term -h_{i-s} = -h_{i+t} contributes to g[:len(h)-t]
            t = -s
            g[t:] -= w * h[:len(h)-t]
            g[:len(h)-t] -= w * h[t:]
    
    return g
