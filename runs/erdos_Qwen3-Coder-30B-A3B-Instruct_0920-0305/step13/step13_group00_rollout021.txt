def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_gradient(h, corr, alpha=10.0):
    """Compute gradient using softmax approximation"""
    # Compute weights
    max_corr = np.max(corr)
    weights = np.exp(alpha * (corr - max_corr))
    weights = weights / np.sum(weights)
    
    # Compute gradient
    grad = np.ones(len(h))
    
    # For each shift s (in correlation index)
    for i in range(len(corr)):
        s = i - (len(h) - 1)  # shift value corresponding to index
        w = weights[i]
        
        if s == 0:
            grad -= 2 * w * h
        else:
            # Positive shift: h[i] contributes to h[i+s] and h[i-s]
            if s > 0:
                # h[i+s] contribution
                if len(h) - s > 0:
                    grad[:len(h)-s] -= w * h[s:]
                # h[i-s] contribution  
                if s < len(h):
                    grad[s:] -= w * h[:len(h)-s]
            else:
                # Negative shift: treat as positive but in reverse
                t = -s
                if t < len(h):
                    grad[t:] -= w * h[:len(h)-t]
                if len(h) - t > 0:
                    grad[:len(h)-t] -= w * h[t:]
    
    return grad
