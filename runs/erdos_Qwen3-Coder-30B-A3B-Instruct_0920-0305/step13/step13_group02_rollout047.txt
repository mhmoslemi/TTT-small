import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import dual_annealing
import random

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT correlation"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        h = project_to_sum_box(h, TARGET_SUM)
        strategies.append(h)
    
    # Strategy 2: Piecewise constant functions
    for _ in range(5):

            h[start:end] = random.random()
        h = project_to_sum_box(h, TARGET_SUM)
        strategies.append(h)

    for _ in range(5):
        h = np.zeros(n_points)
        center = random.randint(0, n_points - 1)

        for i in range(n_points):
            dist = abs(i - center)
            h[i] = max(0, 1 - dist / width)
        h = project_to_sum_box(h, TARGET_SUM)
        strategies.append(h)

        pert = np.random.normal(0, 0.05, n_points)

            h_new = project_to_sum_box(h + pert, TARGET_SUM)
            c5_new = evaluate_c5(h_new)
            
            if c5_new < evaluate_c5(h):
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
    
    # Ensure we return something
    if best_h is None:
        best_h = h0.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
