import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1 << (L - 1).bit_length()
    corr = np.fft.ifft(np.fft.fft(h_rev, fft_size) * np.fft.fft(b, fft_size)).real[:L]
    return corr

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT (returns unscaled correlation max)"""
    corr = compute_corr_fft(h)
    return np.max(corr)

def evaluate_c5_precise(h):
    """Precise evaluation of C5 using np.correlate"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x - lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = np.min(x) - 1, np.max(x)
    tol = 1e-12
    for _ in range(60):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x - (a + b)/2, 0, 1)
    return result

def compute_subgradient(h, corr):
    """Compute subgradient of max correlation"""
    n = len(h)
    max_val = np.max(corr)
    eps = 1e-12
    active_idx = np.where(corr >= max_val - eps)[0]
    
    # Initialize gradient with constant term
    g = np.full(n, len(active_idx), dtype=float)
    
    for idx in active_idx:
        k = idx - (n - 1)
        if k >= 0:
            # Subtract h_{j+k} for j < n-k and h_{j-k} for j >= k
            g[:n-k] -= h[k:]
            g[k:] -= h[:n-k]
        else:
            # k < 0 case
            d = -k
            g[d:] -= h[:n-d]
            g[:n-d] -= h[d:]
    
    return g, max_val

def generate_sin_initial():
    """Generate sinusoidal initial solution"""
    x = (np.arange(n_points) + 0.5) * dx
    # Random amplitude in [0, 0.5] to keep values in [0,1]
    a = np.random.rand() * 0.5
    phase = np.random.rand() * 2 * np.pi
    h = 0.5 + a * np.sin(np.pi * x + phase)
    return project_to_sum_box(h, target_sum)

            h_new = h - step * g
