import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem setup
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Create CVXPY variables
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum
    ]
    
    # For each shift s, we need to define the correlation constraint
    # Correlation for shift s is: sum_{i=0}^{n-1-s} h[i+s] * (1 - h[i])
    # This equals: sum_{i=0}^{n-1-s} h[i+s] - sum_{i=0}^{n-1-s} h[i+s] * h[i]
    # Which is: (sum of h from s to n-1) - sum of h[i+s] * h[i]
    
    # We'll use y[i,s] = h[i+s] * h[i] to represent the product terms
    # And compute prefix sums P_s = sum_{i=0}^{s-1} h[i]
    
    # Define variables for product terms and prefix sums
    # Note: We'll store all y terms in a list of arrays for each shift
    
    # For the LP formulation:
    # We have: C'_s = (target_sum - P_s) - O_s
    # where O_s = sum_{i=0}^{n-1-s} h[i+s] * h[i] = sum of y[i,s]
    # We want to minimize max_s C'_s = max_s (target_sum - P_s - O_s)
    
    # So the constraint is: target_sum - P_s - O_s <= t for all s
    # This is equivalent to: O_s >= target_sum - P_s - t
    
    # Explicitly write out constraints for each shift s
    for s in range(n_points):
        if s == 0:
            # O_0 = sum_{i=0}^{n-1} y[i,0] = sum_{i=0}^{n-1} h[i]^2
            # Since y[i,0] = h[i] * h[i] = h[i]^2 (McCormick shows y[i,0] = h[i]^2)
            # We'll set up constraints using McCormick envelopes for each pair
            pass
        
        # Create variable for shift s correlation constraint
        if s > 0:
            # Get the valid indices for this shift
            valid_indices = list(range(n_points - s))
            O_s = cp.sum([h[i+s] * h[i] for i in valid_indices])
            # Prefix sum for shift s = sum_{i=0}^{s-1} h[i]
            P_s = cp.sum(h[:s])
            # Constraint: O_s >= (target_sum - P_s) - t
            constraints.append(O_s >= (target_sum - P_s) - t)
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Try to solve with ECOS first
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            t_opt = t.value
            c5_bound = dx * t_opt
        else:
            # Fallback to a gradient-based approach if LP fails
            return run_gradient_fallback(seed, budget_s, n_points, dx, target_sum)
    except:
        # Fallback to a gradient-based approach if LP fails
        return run_gradient_fallback(seed, budget_s, n_points, dx, target_sum)
    
    # Ensure the result is within bounds
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = project_to_sum_box(h_opt, target_sum)
    c5_bound = dx * t_opt
    
    return (h_opt.tolist(), float(c5_bound), n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Simple projection
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x
    
    # Use a simple iterative approach
    tol = 1e-8
    max_iter = 1000
    for _ in range(max_iter):
        diff = target_sum - current_sum
        if abs(diff) < tol:
            break
        # Distribute difference to elements that can still increase/decrease
        update = diff / len(x)
        x += update
        x = np.clip(x, 0, 1)  
        current_sum = np.sum(x)
    
    return x

        h_arr = np.array(h_vals)
        corr = np.correlate(h_arr, 1 - h_arr, mode="full")
        return np.max(corr) * dx

    best_c5 = evaluate_c5(h)

        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(h[i] + eps, 1.0)
            h_plus = project_to_sum_box(h_plus, target_sum)
            
            h_minus = h.copy()

            h_minus = project_to_sum_box(h_minus, target_sum)

            c5_minus = evaluate_c5(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)

        step_size = 0.01
        h_new = h - step_size * grad
        h_new = project_to_sum_box(h_new, target_sum)
        
        c5_new = evaluate_c5(h_new)

            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
