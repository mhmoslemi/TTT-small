def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmax_smooth_max(h, gamma=1000.0):
    """Compute smooth approximation of max using softmax"""
    corr = compute_corr_fft(h)
    # Use softmax to approximate the maximum
    exp_corr = np.exp(gamma * corr)
    smooth_max = np.log(np.sum(exp_corr)) / gamma
    return smooth_max

def smooth_gradient(h, gamma=1000.0):
    """Compute gradient of smooth max approximation"""
    corr = compute_corr_fft(h)
    max_idx = np.argmax(corr)
    
    # Compute gradient of smooth max approximation
    exp_corr = np.exp(gamma * corr)
    exp_sum = np.sum(exp_corr)
    # Gradient of log(sum(exp(gamma * corr))) w.r.t. h
    # We need gradient of the correlation with respect to h
    # Approximate: computationally expensive to do full gradient, 
    # so we use finite differences with different epsilon
    
    eps = 1e-4
    grad = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_minus = h.copy()
        h_minus[i] -= eps
        
        corr_plus = compute_corr_fft(h_plus)
        corr_minus = compute_corr_fft(h_minus)
        
        grad[i] = (corr_plus[max_idx] - corr_minus[max_idx]) / (2 * eps)
    
    return grad * dx * gamma * np.exp(gamma * corr[max_idx]) / exp_sum
