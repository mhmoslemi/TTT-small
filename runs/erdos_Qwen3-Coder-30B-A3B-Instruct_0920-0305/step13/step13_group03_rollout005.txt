import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_c5(h, dx):
    """Evaluate the C5 bound using FFT"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_convex_subproblem(h_current, lags, n, dx):
    """Solve the convex subproblem using CVXPY"""
    h = Variable(n)
    T = Variable()
    
    # Convex constraints
    constraints = [h >= 0, h <= 1, sum(h) == n/2, T >= 0]
    
    # Add lag constraints
    for ell in lags:
        # Compute the convex part of I_ell(h)
        # I_ell(h) = sum(h_i) + 0.25 * sum((h_i - h_{i+ell})^2) - 0.25 * sum((h_i + h_{i+ell})^2)
        # For the concave part we use first order approximation at h_current
        if ell >= 0:
            concave_term = 0.25 * sum((h_current[ell:] + h_current[:n-ell]) * (h[ell:] + h[:n-ell]))
        else:
            ell_abs = -ell
            concave_term = 0.25 * sum((h_current[:n-ell_abs] + h_current[ell_abs:]) * (h[:n-ell_abs] + h[ell_abs:]))
        
        # Linear part of concave approximation
        if ell >= 0:
            linear_approx = 0.5 * sum((h_current[ell:] + h_current[:n-ell]) * (h[ell:] + h[:n-ell]))
        else:
            ell_abs = -ell
            linear_approx = 0.5 * sum((h_current[:n-ell_abs] + h_current[ell_abs:]) * (h[:n-ell_abs] + h[ell_abs:]))
        
        # Convex part of I_ell
        convex_part = sum(h) + 0.25 * sum((h[i] - h[i+ell])**2 for i in range(n-abs(ell)))
        # Simplify for our finite sum case
        if ell >= 0:
            convex_part = sum(h) + 0.25 * sum((h[i] - h[i+ell])**2 for i in range(n-ell))
        else:
            ell_abs = -ell
            convex_part = sum(h) + 0.25 * sum((h[i] - h[i+ell])**2 for i in range(n-ell_abs))
        
        # Full constraint
        constraints.append(convex_part - linear_approx <= T)
    
    # Objective
    objective = Minimize(T)
    
    # Solve
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=ECOS, verbose=False)
        return h.value, T.value
    except:
        return h_current, np.inf

def get_violating_lags(h, lags, T_current, eps=1e-6):
    """Get the lags that violate the current bound"""
    corr = correlate(h, 1 - h, mode='full')
    # Get all lags
    all_lags = np.arange(-len(h)+1, len(h))
    violating = []
    for ell in all_lags:
        idx = ell + len(h) - 1
        if corr[idx] > T_current + eps:
            violating.append(ell)
    return violating

def round_to_binary(h_cont, n):
    """Random round to binary with deterministic polishing"""
    # Monte Carlo rounding with top-k
    best_h = None
    best_c5 = float('inf')
    
    for _ in range(30):  # 30 random draws
        # Sort by decreasing values
        sorted_indices = np.argsort(h_cont)[::-1]
        h_binary = np.zeros(n)
        h_binary[sorted_indices[:n//2]] = 1
        
        # Minor randomness by flipping some bits
        flip_count = min(4, n//10)  # Flip up to 4 bits
        if flip_count > 0:
            flip_indices = np.random.choice(n//2, flip_count, replace=False)
            for idx in flip_indices:
                h_binary[sorted_indices[idx]] = 0
            # Add back some 1s
            non_ones = np.where(h_binary == 0)[0]
            if len(non_ones) >= flip_count:
                new_ones = np.random.choice(non_ones, flip_count, replace=False)
                h_binary[new_ones] = 1
        
        # Project to sum constraint (ensure exactly n/2 ones)
        ones_count = np.sum(h_binary)
        if ones_count > n//2:
            # Turn some 1s to 0s
            ones_indices = np.where(h_binary == 1)[0]
            flip_count = int(ones_count - n//2)
            if flip_count > 0:
                flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
                h_binary[flip_indices] = 0
        elif ones_count < n//2:
            # Turn some 0s to 1s
            zeros_indices = np.where(h_binary == 0)[0]
            flip_count = int(n//2 - ones_count)
            if flip_count > 0:
                flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
                h_binary[flip_indices] = 1
                
        h_binary = project_to_sum_box(h_binary, n/2)
        
        c5 = evaluate_c5(h_binary, 2.0/n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_binary.copy()
    
    if best_h is None:
        # Fallback to deterministic rounding
        sorted_indices = np.argsort(h_cont)[::-1]
        h_binary = np.zeros(n)
        h_binary[sorted_indices[:n//2]] = 1
        best_h = h_binary.copy()
    
    return best_h, best_c5

def polish_with_critical_lag(h, max_iter=5000):
    """Polish with critical lag greedy swaps"""
    start_time = time.time()
    
    # Evaluate initial correlations
    corr = correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    max_lag = max_idx - len(h) + 1
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < 180 and max_lag != 0:
        # Get worst lag and improving swaps
        corr = correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_lag = max_idx - len(h) + 1
        
        if max_lag == 0:
            break
            
        # Find 1s that are 1 in original but 0 in shifted position
        if max_lag >= 0:
            mismatches = np.where((h[:-max_lag] == 1) & (h[max_lag:] == 0))[0]
        else:
            mismatches = np.where((h[-max_lag:] == 1) & (h[:-max_lag] == 0))[0]
            
        if len(mismatches) == 0:
            break
            
        # Ideal swap: pick worst mismatch and swap with a good zero
        i = mismatches[np.random.randint(len(mismatches))]
        
        # Find valid zero candidates
        if max_lag >= 0:
            valid_zeros = np.where((h == 0) & ((i + max_lag >= len(h)) | (h[i + max_lag] == 1)))[0]
        else:
            valid_zeros = np.where((h == 0) & ((i - max_lag >= len(h)) | (h[i - max_lag] == 1)))[0]
            
        if len(valid_zeros) == 0:
            # If no valid candidate, pick any zero
            zeros = np.where(h == 0)[0]
            if len(zeros) > 0:
                j = zeros[np.random.randint(len(zeros))]
            else:
                break
        else:
            j = valid_zeros[np.random.randint(len(valid_zeros))]
            
        # Make the swap
        h[i], h[j] = h[j], h[i]
        iter_count += 1
        
    return h, evaluate_c5(h, 2.0/len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 120
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        h0 = np.ones(n_points) * 0.5
        
    h0 = project_to_sum_box(h0, n_points / 2)
    
    # Initial lags
    initial_lags = [0, -24, -48, -72, -96, 24, 48, 72, 96]
    
    # Parameters for CCP
    max_ccp_iter = 100
    tol = 1e-6
    
    h_current = h0.copy()
    lags = initial_lags.copy()
    
    # Iterative CCP with cutting plane
    for _ in range(max_ccp_iter):
        if time.time() - start_time > 0.4 * budget_s:
            break
            
        h_opt, T_opt = solve_convex_subproblem(h_current, lags, n_points, dx)
        
        if np.isinf(T_opt):
            break
            
        # Check if we have enough bounds
        # Evaluate true overlaps for all lags
        corr = correlate(h_opt, 1 - h_opt, mode='full')
        all_lags = np.arange(-n_points + 1, n_points)
        max_corr = np.max(corr)
        if max_corr <= T_opt + tol:
            h_current = h_opt.copy()
            break
            
        # Find violating lags
        violating_lags = []
        for ell in all_lags:
            idx = ell + n_points - 1
            if corr[idx] > T_opt + tol:
                violating_lags.append(ell)
                
        if not violating_lags:
            h_current = h_opt.copy()
            break
            
        # Add most violated lag
        max_violation = 0
        best_violation = None
        for ell in violating_lags:
            idx = ell + n_points - 1
            violation = corr[idx] - T_opt
            if violation > max_violation:
                max_violation = violation
                best_violation = ell
                
        if best_violation is not None:
            lags.append(best_violation)
            
        h_current = h_opt.copy()
    
    # Round to binary
    h_binary, c5_binary = round_to_binary(h_current, n_points)
    
    # Polish with critical lag swaps
    h_final, c5_final = polish_with_critical_lag(h_binary)
    
    # Ensure we have a good solution
    best_h = h_final
    best_c5 = c5_final
    
    # Fallback 2-opt if needed
    if best_c5 > 0.3807:
        corr = correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        curr_max_lag = max_idx - len(best_h) + 1
        max_iter = 1000
        
        if time.time() - start_time < 0.9 * budget_s:
            start_time_opt = time.time()
            # Try exhaustive 2-opt in small region
            best_h_local = best_h.copy()
            best_c5_local = best_c5
            
            # Find best 2-opt swap from current
            for i in range(len(best_h) // 2):  # Reduce search space
                for j in range(i+1, len(best_h) // 2):
                    if time.time() - start_time_opt > 3:
                        break
                    h_test = best_h_local.copy()
                    h_test[i], h_test[j] = h_test[j], h_test[i]
                    c5_test = evaluate_c5(h_test, dx)
                    if c5_test < best_c5_local:
                        best_c5_local = c5_test
                        best_h_local = h_test.copy()
                        
            if best_c5_local < best_c5:
                best_h = best_h_local
                best_c5 = best_c5_local
    
    return (best_h.tolist(), float(best_c5), n_points)
