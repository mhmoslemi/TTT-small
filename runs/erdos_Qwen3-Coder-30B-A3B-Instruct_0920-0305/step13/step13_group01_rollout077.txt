import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

# Global parameters
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_correlation_fft(h):
    """Compute correlation using FFT: returns (2n-1) element array"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_exact(h):
    """Exact evaluation using np.correlate"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(60):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_alpha(h, alpha=1.0):
    """Smooth approximation to max using log-sum-exp"""
    corr = compute_correlation_fft(h)
    corr = corr * dx  # Scale by dx
    # Subtract max for numerical stability
    max_corr = np.max(corr)
    stable_corr = corr - max_corr
    exp_corr = np.exp(alpha * stable_corr)
    log_sum_exp = np.log(np.sum(exp_corr))
    return (log_sum_exp + max_corr * alpha) / alpha

def grad_smooth_max_alpha(h, alpha=1.0):
    """Gradient of smooth max w.r.t h"""
    corr = compute_correlation_fft(h)
    corr = corr * dx

    max_corr = np.max(corr)
    stable_corr = corr - max_corr
    exp_corr = np.exp(alpha * stable_corr)
    weights = exp_corr / np.sum(exp_corr)

    # Compute gradient using the formula
    grad = np.zeros_like(h)
    n = len(h)
    L = 2 * n - 1
    for k in range(-n + 1, n):
        if k >= 0:
            shift = np.concatenate([np.zeros(k), h[:-k]])
        else:
            shift = np.concatenate([h[-k:], np.zeros(-k)])
        grad += weights[k + n - 1] * (1 - shift - np.roll(h, -k))

    return -grad  # Negative because we minimize

def solve_continuation(h_start, alpha_schedule=[1, 5, 20, 80, 300]):
    """Use continuation on alpha to solve the optimization problem"""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = c5_exact(h)
    
    for alpha in alpha_schedule:
        # Define objective and gradient
        def obj(h):
            return smooth_max_alpha(h, alpha)
        
        def jac(h):
            return grad_smooth_max_alpha(h, alpha)
        
        # Solve with constrained optimization
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
        bounds = [(0, 1)] * len(h)
        
        # Use trust-constr as it's more robust for constrained problems
        try:
            result = minimize(obj, h, method='trust-constr', jac=jac, 
                             bounds=bounds, constraints=constraints,
                             options={'gtol': 1e-9, 'ftol': 1e-9, 'maxiter': 200})
            
            if result.success:
                h = result.x
                # Re-project to ensure constraints
                h = project_to_sum_box(h, TARGET_SUM)
                c5_here = c5_exact(h)
                if c5_here < best_c5:
                    best_c5 = c5_here
                    best_h = h.copy()
        except Exception as e:
            pass  # If optimization fails, continue with previous best

    return best_h, best_c5

def local_search(h, max_iter=3000):
    """Perform local search to improve the solution"""
    h = h.copy()
    c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = c5

    for _ in range(max_iter):
        # Try swap move
        i, j = np.random.choice(n_points, 2, replace=False)
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = c5_exact(h_new)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        # Try small transfer
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(0.001, 0.01)
        if h[i] >= delta and 1 - h[j] >= delta:
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            c5_new = c5_exact(h_new)
            if c5_new < c5:
                h, c5 = h_new, c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None

    # Generate diverse initial guesses
    initial_guesses = []
    
    # 1. Repeated pattern from initial_h_values (120 points)
    base_pattern = np.array(initial_h_values)
    extended_pattern = np.tile(base_pattern, n_points // len(base_pattern))
    extended_pattern = extended_pattern[:n_points]
    initial_guesses.append(extended_pattern)
    
    # 2. Sinusoidal
    x = np.arange(n_points) * dx + dx/2
    phase = np.random.uniform(0, 2*np.pi)
    amplitude = np.random.uniform(0.1, 0.5)
    h_sine = 0.5 + amplitude * np.sin(np.pi * x + phase)
    h_sine = project_to_sum_box(h_sine, TARGET_SUM)
    initial_guesses.append(h_sine)
    
    # 3. Amplitude-tuned sinusoid
    amplitude = np.random.uniform(0.1, 0.5)
    h_sine2 = 0.5 + amplitude * np.sin(np.pi * x)
    h_sine2 = project_to_sum_box(h_sine2, TARGET_SUM)
    initial_guesses.append(h_sine2)
    
    # 4. Triangular wave
    h_tri = np.zeros(n_points)
    mid = n_points // 2
    h_tri[:mid] = np.linspace(0, 1, mid)
    h_tri[mid:] = np.linspace(1, 0, n_points - mid)
    h_tri = project_to_sum_box(h_tri, TARGET_SUM)
    initial_guesses.append(h_tri)
    
    # 5. Random uniform with projection
    h_rand = np.random.rand(n_points)
    h_rand = project_to_sum_box(h_rand, TARGET_SUM)
    initial_guesses.append(h_rand)
    
    # 6. Hybrid (sinusoid + random)
    h_hybrid = 0.7 * h_sine + 0.3 * np.random.rand(n_points)
    h_hybrid = project_to_sum_box(h_hybrid, TARGET_SUM)
    initial_guesses.append(h_hybrid)

    # Run optimization for each initial guess
    for i, h0 in enumerate(initial_guesses):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        h_opt, c5_opt = solve_continuation(h0)
        h_opt, c5_opt = local_search(h_opt)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final exact evaluation on best solution
    if best_h is not None:
        final_c5 = c5_exact(best_h)
        return (best_h.tolist(), float(final_c5), n_points)

    # Fallback to original solution if nothing worked
    h0 = np.array(initial_h_values)
    h0 = project_to_sum_box(h0, TARGET_SUM)
    final_c5 = c5_exact(h0)
    return (h0.tolist(), float(final_c5), n_points)
