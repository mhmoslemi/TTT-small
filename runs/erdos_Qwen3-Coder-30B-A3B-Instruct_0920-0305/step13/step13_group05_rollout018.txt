# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Use a more robust projection algorithm
    x = np.array(x)
    # Sort the values
    sorted_x = np.sort(x)[::-1]
    
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_x)
    
    # Find the threshold
    threshold_idx = -1
    for i in range(len(sorted_x)):
        if sorted_x[i] > (target_sum - cumsum[i]) / (i + 1):
            threshold_idx = i
        else:
            break
    
    if threshold_idx == -1:
        # All elements are above the threshold
        threshold = target_sum / len(x)
        return np.clip(x, 0, threshold)
    else:
        # Apply the projection
        threshold = (target_sum - cumsum[threshold_idx]) / (len(x) - threshold_idx)
        projected = np.clip(x, 0, threshold)
        # Adjust sum to target
        current_sum = np.sum(projected)
        if current_sum < target_sum:
            # Add the difference to elements that were clipped
            diff = target_sum - current_sum
            mask = projected == threshold
            if np.sum(mask) > 0:
                to_add = diff / np.sum(mask)
                projected = np.clip(projected + to_add, 0, 1)
        return projected

def softmax_smooth_max(x, gamma=100.0):
    """Approximate the maximum using softmax smoothing"""
    x = np.array(x)
    # Apply softmax to approximate the maximum function
    exp_x = np.exp(gamma * x)
    sum_exp = np.sum(exp_x)
    if sum_exp == 0:
        return 0.0
    return np.sum(x * exp_x) / sum_exp

def compute_smooth_max_gradient(x, gamma=100.0):
    """Compute the gradient of softmax smoothed maximum"""
    exp_x = np.exp(gamma * x)
    sum_exp = np.sum(exp_x)
    if sum_exp == 0:
        return np.zeros_like(x)
    # Compute derivative of softmax approximation
    d_softmax = exp_x / sum_exp
    d_sum = np.sum(d_softmax)
    gradient = (exp_x * (gamma * x - np.sum(gamma * x * exp_x / sum_exp)) + 
                d_softmax * (1 - d_sum)) / sum_exp
    return gradient

def compute_c5_smooth(h, gamma=100.0):
    """Compute smoothed C5 using softmax"""
    corr = compute_corr_fft(h)
    # Use softmax smooth approximation for maximum
    return dx * softmax_smooth_max(corr, gamma)

def compute_gradient_smooth(h, gamma=100.0):
    """Compute gradient of smooth C5 approximation"""
    corr = compute_corr_fft(h)
    max_idx = np.argmax(corr)
    # Compute gradient with respect to corr values  
    grad_corr = np.zeros_like(corr)
    grad_corr[max_idx] = 1.0
    # Apply inverse FFT to get gradient w.r.t original h
    L = len(h) * 2 - 1
    grad_h = np.fft.ifft(np.fft.fft(grad_corr, L) * np.fft.fft(1 - h[::-1], L)).real
    # Interleave the gradients
    grad = np.zeros_like(h)
    for i in range(len(h)):
        grad[i] = grad_h[len(h)-1+i]
    return dx * grad

def gradient_descent_with_projection(h0, max_iter=10000, step_size=1e-3, 
                                   gamma=100.0, tol=1e-6):
    """Projected gradient descent with smoothing"""
    h = h0.copy()
    current_c5 = compute_c5_smooth(h, gamma)
    
    for i in range(max_iter):
        if i % 1000 == 0:
            new_c5 = evaluate_c5(h)
            if new_c5 < current_c5:
                current_c5 = new_c5
        
        # Compute gradient of smoothed objective
        grad = compute_gradient_smooth(h, gamma)
        
        # Step in negative gradient direction
        h_new = h - step_size * grad
        
        # Project onto feasible set
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Check for convergence
        if np.linalg.norm(h_new - h) < tol:
            break
            
        h = h_new
    
    return h, evaluate_c5(h)
