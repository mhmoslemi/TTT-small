    """Fast evaluation of C5 using FFT"""

    b = 1 - h
    L = 2 * len(h) - 1

    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_gradient(h, corr, alpha=10.0):

    max_corr = np.max(corr)
    exp_corr = np.exp(alpha * (corr - max_corr))
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient: 1 - sum_s w_s * (h_{i+s} + h_{i-s})
    grad = np.ones_like(h)
    
    # Handle shift indices
    n = len(h)
    for i in range(n):
        for s in range(-(n-1), n):
            if s == 0:
                grad[i] -= 2 * weights[n-1] * h[i]

                if i + s < n:
                    grad[i] -= weights[s + n - 1] * h[i + s]
                if i - s >= 0:
                    grad[i] -= weights[-s + n - 1] * h[i - s]
            else:  # s < 0
                t = -s
                if i + t < n:
                    grad[i] -= weights[t + n - 1] * h[i + t]
                if i - t >= 0:
                    grad[i] -= weights[-t + n - 1] * h[i - t]
    
    return grad
