TARGET_SUM = None

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h, corr, alpha):
    """Compute gradient using softmax approximation of max"""
    # compute weights
    max_corr = np.max(corr)
    w = np.exp(alpha * (corr - max_corr))
    w = w / np.sum(w)
    
    # compute gradient using weighted sum of individual gradients
    grad = np.ones(len(h))
    
    # For each shift, we compute contribution to gradient
    # For shift s, gradient contribution is - (h_{i+s} + h_{i-s})
    for i in range(len(h)):
        for s in range(-len(h)+1, len(h)):
            if s == 0:
                grad[i] -= 2 * w[len(h)-1] * h[i]
            elif s > 0:
                if i + s < len(h):
                    grad[i] -= w[len(h)-1+s] * h[i+s]
                if i - s >= 0:
                    grad[i] -= w[len(h)-1+s] * h[i-s]
            else:  # s < 0
                t = -s
                if i + t < len(h):
                    grad[i] -= w[len(h)-1+s] * h[i+t]
                if i - t >= 0:
                    grad[i] -= w[len(h)-1+s] * h[i-t]
    
    return grad
