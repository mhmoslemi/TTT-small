from scipy.optimize import differential_evolution
import time

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Simple projection approach using sorting
    sorted_indices = np.argsort(x)[::-1]
    current_sum = np.sum(x)
    
    if current_sum > target_sum:
        # Need to decrease values
        for i in sorted_indices:
            if current_sum <= target_sum:
                break
            reduction = min(x[i], current_sum - target_sum)
            x[i] -= reduction
            current_sum -= reduction
    else:
        # Need to increase values
        for i in sorted_indices:
            if current_sum >= target_sum:
                break
            increase = min(1 - x[i], target_sum - current_sum)
            x[i] += increase
            current_sum += increase
    
    return np.clip(x, 0, 1)
