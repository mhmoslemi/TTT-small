import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using linear correlation"""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Simple projection algorithm
    x = np.array(x)
    x = np.clip(x, 0, 1)
    
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x
    
    # Sort values to find which to adjust
    sorted_indices = np.argsort(x)
    diff = target_sum - current_sum
    
    if diff > 0:  # Need to increase sum
        # Increase smaller values first
        for i in sorted_indices:
            if diff <= 0:
                break
            can_increase = 1.0 - x[i]
            increase = min(can_increase, diff)
            x[i] += increase
            diff -= increase
    else:  # Need to decrease sum
        # Decrease larger values first
        for i in reversed(sorted_indices):
            if diff >= 0:
                break
            can_decrease = x[i]
            decrease = min(can_decrease, -diff)
            x[i] -= decrease
            diff += decrease
    
    return x
