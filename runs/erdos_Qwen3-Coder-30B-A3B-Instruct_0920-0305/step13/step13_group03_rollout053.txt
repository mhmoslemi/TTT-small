import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def max_corr(h):
    """Compute the maximum overlap correlation for given h"""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    return dx * max_corr(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    target_M = int(np.floor(0.3807 / dx))  # This corresponds to 22
    
    # Load initial solution
    initial_h = np.array(initial_h_values)
    
    # Convert to binary with proper sum
    h_rounded = (initial_h >= 0.5).astype(float)
    current_sum = np.sum(h_rounded)
    
    # Adjust to exactly 60 ones
    if current_sum > 60:
        # Need to flip some 1s to 0s
        diff = int(current_sum - 60)
        # Sort by distance from 0.5 and flip the closest to 0.5
        distances = np.abs(initial_h - 0.5)
        flip_indices = np.argpartition(distances, diff)[:diff]
        h_rounded[flip_indices] = 0
    elif current_sum < 60:
        # Need to flip some 0s to 1s
        diff = int(60 - current_sum)
        # Sort by distance from 0.5 and flip the closest to 0.5
        distances = np.abs(initial_h - 0.5)
        flip_indices = np.argpartition(distances, diff)[:diff]
        h_rounded[flip_indices] = 1
    
    # Ensure sum is exactly 60
    current_sum = np.sum(h_rounded)
    if current_sum != 60:
        # If we have more or fewer than 60 ones, adjust carefully
        diff = int(60 - current_sum)
        if diff > 0:
            # Need to set some 0s to 1s
            zero_indices = np.where(h_rounded == 0)[0]
            if len(zero_indices) >= diff:
                h_rounded[zero_indices[:diff]] = 1
        elif diff < 0:
            # Need to set some 1s to 0s
            one_indices = np.where(h_rounded == 1)[0]
            if len(one_indices) >= -diff:
                h_rounded[one_indices[:(-diff)]] = 0
    
    # Create warm start for MILP
    h_initial = h_rounded.copy()
    current_c5 = evaluate_c5(h_initial)
    
    best_h = h_initial.copy()
    best_c5 = current_c5
    
    # MILP approach for exact solution
    try:
        # MILP time allocation
        milp_time = min(260, budget_s * 0.4)
        
        # Define the problem - only free variables
        h_vars = cp.Variable(n_points, boolean=True)
        
        # Define wet/lag sets for only problematic lags that could exceed target
        # In our case, we only need to consider lags that could possibly reach 23 or more
        # We keep lags from -98 to +98 (since n = 120 and max_index = 119)
        lags_considered = list(range(-98, 99))  # 197 lags
    
        # Constraints for sum = 60
        constraints = [cp.sum(h_vars) == 60]
        
        # For each lag, define helper variables and constraints for the overlap calculation
        # I_ell = sum(h_i - h_i * h_{i+ell}) = sum(h_i) - sum(h_i * h_{i+ell}) 
        max_I = cp.Variable()
        constraints.append(max_I >= 0)
        
        # We will add specific constraints for the lags that are most critical
        for ell in lags_considered:
            if ell >= 0:
                # For positive lags, we go from 0 to n-ell-1
                indices = list(range(n_points - ell))
                # We need to compute sum(h_i) - sum(h_i * h_{i+ell})
                # We introduce w_{i,ell} = h_i * h_{i+ell}
                w_vars = cp.Variable(len(indices), boolean=True)
                # Constraint: w[i,ell] <= h[i], w[i,ell] <= h[i+ell], w[i,ell] >= h[i] + h[i+ell] - 1
                for i, idx in enumerate(indices):
                    constraints.extend([
                        w_vars[i] <= h_vars[idx],
                        w_vars[i] <= h_vars[idx + ell],
                        w_vars[i] >= h_vars[idx] + h_vars[idx + ell] - 1
                    ])
                # Constraint: I_ell = sum(h_i - w[i,ell]) <= max_I
                I_ell = cp.sum(h_vars[indices]) - cp.sum(w_vars)
                constraints.append(I_ell <= max_I)
            else:
                # For negative lags, we go from -ell to n-1
                indices = list(range(-ell, n_points))
                # We introduce w_{i,ell} = h_i * h_{i+ell} where i+ell >= 0
                w_vars = cp.Variable(len(indices), boolean=True)
                # Constraint: w[i,ell] <= h[i], w[i,ell] <= h[i+ell], w[i,ell] >= h[i] + h[i+ell] - 1
                for i, idx in enumerate(indices):
                    constraints.extend([
                        w_vars[i] <= h_vars[idx],
                        w_vars[i] <= h_vars[idx + ell],
                        w_vars[i] >= h_vars[idx] + h_vars[idx + ell] - 1
                    ])
                # Constraint: I_ell = sum(h_i - w[i,ell]) <= max_I
                I_ell = cp.sum(h_vars[indices]) - cp.sum(w_vars)
                constraints.append(I_ell <= max_I)
        
        # Minimize max_I
        prob = cp.Problem(cp.Minimize(max_I), constraints)
        prob.solve(solver=cp.CBC, cbc_params={'maxTime': milp_time})
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            solution = h_vars.value
            sol_h = np.round(solution).astype(int)
            sol_c5 = evaluate_c5(sol_h)
            if sol_c5 < best_c5:
                best_c5 = sol_c5
                best_h = sol_h.copy()
        
    except:
        # If MILP fails or doesn't exist, we'll continue with heuristics
        pass

    # Critical lag refinement procedure
    h_current = best_h.copy()

    max_iter = 100000
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:

        corr = correlate(h_current, 1 - h_current, mode="full")
        max_corr_idx = np.argmax(corr)
        max_val = corr[max_corr_idx]
        if max_val <= target_M:
            break

        lag = max_corr_idx - (len(h_current) - 1)  # -98 to +98
        
        # Find a list of mismatched positions for this lag
        if lag >= 0:
            mismatch_positions = np.where((h_current[:len(h_current)-lag] == 1) &

        else:
            mismatch_positions = np.where((h_current[-lag:] == 1) &

        if len(mismatch_positions) == 0:
            break

        i = np.random.choice(mismatch_positions)
        
        # Find a position j that when swapped reduces the mismatch for this lag
        if lag >= 0:

        if len(candidates) > 0:
            j = np.random.choice(candidates)
        else:
            # Try any zero position
            zero_positions = np.where(h_current == 0)[0]
            j = np.random.choice(zero_positions)

        h_current[i], h_current[j] = h_current[j], h_current[i]

            best_h = h_current.copy()

    # Final fallback if no improvement yet
    if best_c5 >= 0.3807:

        num_restarts = min(10, max(1, int((budget_s - time.time() + start_time) / 5)))
        for _ in range(num_restarts):

            h_refined = h_rand.copy()

                    best_c5 = current_max
                    best_h = h_refined.copy()
