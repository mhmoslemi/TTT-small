import cvxpy as cp
import time

    np.random.seed(seed)

    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2

    h = cp.Variable(n_points)
    t = cp.Variable()

    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum,
        t >= 0
    ]

    for s in range(n_points):
        # For shift s, we consider indices where both h[i] and h[i+s] exist
        if s == 0:
            # For shift 0, we want to maximize sum of h[i]^2
            # This is handled by the McCormick envelopes
            pass
        else:
            # Create variable for products h[i] * h[i+s] 
            # This is for indices i where i+s < n_points
            valid_indices = range(n_points - s)
            y = cp.Variable(len(valid_indices))
            
            # Add McCormick constraints for each product
            for idx, i in enumerate(valid_indices):
                constraints.extend([
                    y[idx] >= 0,
                    y[idx] <= h[i],
                    y[idx] <= h[i + s],
                    y[idx] >= h[i] + h[i + s] - 1
                ])
            
            # Add constraint that sum of products is bounded by (target_sum - C_s)
            # When shift s is considered, C_s = (target_sum - prefix_sum) - sum of products
            # So we want sum of products to be as large as possible
            # For bounds on C_s, we use: C_s = (target_sum - prefix_sum) - sum(y)  
            # We want to minimize max(C_s) so we set t >= C_s for all s
            
            prefix_sum = cp.sum(h[:s]) if s < n_points else 0

            # compute the sum of products sum(y) and ensure that s doesn't exceed t
            # Since we're trying to minimize max(C_s) = max( (target_sum - prefix_sum) - sum(y) )
            # We minimize t with constraint: (target_sum - prefix_sum) - sum(y) <= t 
            constraints.append(cp.sum(y) + prefix_sum >= target_sum - t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Formulate and solve
    prob = cp.Problem(objective, constraints)
