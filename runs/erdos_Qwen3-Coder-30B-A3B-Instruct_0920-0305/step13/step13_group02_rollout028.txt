import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT correlation"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def create_initial_solution():
    """Create a better initial solution based on mathematical insight"""
    # Try a construction that places more mass in the middle regions
    h = np.zeros(n_points)
    
    # Create a piecewise constant function with some structure
    # Middle region: higher values, outer regions: lower values
    mid_start = n_points // 3
    mid_end = 2 * n_points // 3
    
    # Set middle region to have higher values
    h[mid_start:mid_end] = 0.8
    
    # Normalize to ensure sum constraint
    current_sum = np.sum(h)
    scale_factor = TARGET_SUM / current_sum
    h = h * scale_factor
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    # Project to exact sum
    h = project_to_sum_box(h, TARGET_SUM)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Use better initial solution
    h0 = create_initial_solution()
    current_h = h0.copy()
    current_c5 = evaluate_c5(h0)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()

    # Simulated Annealing approach with adaptive cooling
    h = best_h.copy() if best_h is not None else h0.copy()
    current_c5 = best_c5 if best_h is not None else current_c5
    
    # Parameters for simulated annealing
    temp = 1.0
    min_temp = 1e-8
    alpha = 0.98
    max_iter = 10000
    
    t_end = start_time + budget_s
    
    iter_num = 0
    while time.time() < t_end and iter_num < max_iter:
        iter_num += 1
        
        # Create neighbor solution
        h_new = h.copy()
        
        # Randomly modify some elements
        num_changes = max(1, n_points // 20)
        for _ in range(num_changes):
            idx = random.randint(0, n_points - 1)
            # Small random perturbation
            delta = random.uniform(-0.1, 0.1)
            h_new[idx] += delta
            
        # Project to constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Calculate new C5
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject based on Metropolis criterion
        if c5_new < current_c5 or random.random() < np.exp((current_c5 - c5_new) / temp):
            h = h_new
            current_c5 = c5_new
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Cool down
        temp *= alpha
        if temp < min_temp:
            temp = min_temp

    # Additional refinement: Local hill climbing
    if best_h is not None:
        h = best_h.copy()
        current_c5 = best_c5
        
        # Try to locally improve by adjusting individual points
        local_search_iter = 5000
        for _ in range(local_search_iter):
            if time.time() >= t_end:
                break
                
            # Try to improve by small adjustments to individual points
            h_new = h.copy()
            idx = random.randint(0, n_points - 1)
            # Make a small change
            delta = random.uniform(-0.05, 0.05)
            h_new[idx] += delta
            
            # Project to constraints
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            c5_new = evaluate_c5(h_new)
            
            if c5_new < current_c5:
                h = h_new
                current_c5 = c5_new
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()

    # Try some specific constructions
    if best_h is not None:
        # Try a symmetric construction with different parameters
        h_sym = np.zeros(n_points)
        middle_start = n_points // 4
        middle_end = 3 * n_points // 4
        
        # Create a smooth transition
        h_sym[middle_start:middle_end] = 1.0
        
        # Ensure correct sum
        h_sym = project_to_sum_box(h_sym, TARGET_SUM)
        
        c5_sym = evaluate_c5(h_sym)
        if c5_sym < best_c5:
            best_c5 = c5_sym
            best_h = h_sym.copy()

    if best_h is None:
        best_h = h0.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
