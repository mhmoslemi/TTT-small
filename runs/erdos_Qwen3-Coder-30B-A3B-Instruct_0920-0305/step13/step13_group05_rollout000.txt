import numpy as np

import time

# Global variables
TARGET_SUM = None
n_points = 120

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmax_subgradient_descent(h0, max_iter=50000, gamma_init=0.5, gamma_factor=1.02, step_size=0.5):
    """Perform softmax subgradient descent to minimize C5"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    gamma = gamma_init
    
    for i in range(max_iter):
        # Compute correlation
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        
        # Compute softmax weights
        corr_shifted = corr - max_corr
        exp_corr = np.exp(gamma * corr_shifted)
        weights = exp_corr / np.sum(exp_corr)
        
        # Compute weighted gradient
        grad = np.zeros(n_points)
        shifts = np.arange(-n_points + 1, n_points)
        
        for s_idx, s in enumerate(shifts):
            weight = weights[s_idx]
            if s >= 0:
                # Forward contribution
                if s < n_points:
                    grad[:n_points-s] += weight * (1 - h[s:])
                # Backward contribution  
                if s > 0:
                    grad[s:] += weight * (-h[:n_points-s])
            else:
                # Negative shift
                t = -s
                if t < n_points:
                    grad[t:] += weight * (1 - h[:n_points-t])
                if t > 0:
                    grad[:n_points-t] += weight * (-h[t:])
        
        # Scale gradient
        grad = dx * grad
        
        # Update step
        step = step_size / np.sqrt(i + 1)
        h = h - step * grad
        
        # Project
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Evaluate
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Update gamma
        if i % 1000 == 0:
            gamma *= gamma_factor
    
    return best_h, best_c5
