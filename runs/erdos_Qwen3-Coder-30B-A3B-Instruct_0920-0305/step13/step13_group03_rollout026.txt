import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import differential_evolution, minimize

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    for i, h_start in enumerate(initializations):

    # Final cleanup with binary rounding
    if best_h is not None:
        # Convert to binary approximation
        binary_h = best_h.copy()
        # Apply thresholding based on median
        threshold = np.median(binary_h)
        binary_h = (binary_h >= threshold).astype(float)
        
        # Adjust to correct sum
        ones_count = np.sum(binary_h)
        if ones_count > TARGET_SUM:
            flip_indices = np.where(binary_h == 1)[0]
            flip_count = int(ones_count - TARGET_SUM)
            if flip_count > 0:
                flip_indices = np.random.choice(flip_indices, flip_count, replace=False)
                binary_h[flip_indices] = 0
        elif ones_count < TARGET_SUM:
            flip_indices = np.where(binary_h == 0)[0]
            flip_count = int(TARGET_SUM - ones_count)

                flip_indices = np.random.choice(flip_indices, flip_count, replace=False)
                binary_h[flip_indices] = 1

        final_c5 = evaluate_c5(binary_h)

            best_c5 = final_c5
            best_h = binary_h.copy()

    if best_h is None:
        best_h = h0.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
