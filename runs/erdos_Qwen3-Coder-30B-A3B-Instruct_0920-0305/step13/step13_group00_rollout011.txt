def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_gradient(h, corr, alpha):
    """Compute gradient using weighted sum of correlation gradients"""
    N = len(h)
    # Get weights from softmax approximation
    max_corr = np.max(corr)
    weights = np.exp(alpha * (corr - max_corr))
    weights = weights / np.sum(weights)
    
    # Compute gradient for each position
    g = np.ones(N)
    
    # For each shift, add contribution to gradient
    for i in range(2 * N - 1):
        s = i - (N - 1)  # shift from -N+1 to N-1
        w = weights[i]
        
        if s == 0:
            # For shift 0: derivative of sum h_i*h_{i+s} = sum h_i^2
            # We compute h_i*(1-h_{i+s}) = h_i - h_i*h_{i+s} for s=0
            # The derivative w.r.t. h_i is 1 - 2*h_i
            g -= 2 * w * h
        else:
            # For shift s != 0:
            # The derivative is - h_{i+s} - h_{i-s}
            if s > 0:
                # Both h_{i+s} and h_{i-s} contribute
                g[:N-s] -= w * h[s:]
                g[s:] -= w * h[:N-s]
            else:
                # s < 0
                t = -s
                g[t:] -= w * h[:N-t]
                g[:N-t] -= w * h[t:]
    
    return g
