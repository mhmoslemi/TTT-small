def evaluate_c5(h, dx):
    # Use numpy correlate with full mode to compute overlap
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto [0,1] box with sum = target_sum"""
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x
    
    # Use simple approach: adjust values to reach target sum
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase sum, make values more uniform
        # If all values are less than 1, add diff while respecting bounds
        sorted_indices = np.argsort(x)
        for i in range(len(x)):
            idx = sorted_indices[i]
            add_amount = min(diff, 1.0 - x[idx])
            x[idx] += add_amount
            diff -= add_amount
            if diff <= 1e-10:
                break
    else:
        # Need to decrease sum
        # Reduce values accordingly to decrease sum
        diff = -diff
        sorted_indices = np.argsort(x)[::-1]  # Descending order
        for i in range(len(x)):
            idx = sorted_indices[i]
            subtract_amount = min(diff, x[idx])
            x[idx] -= subtract_amount
            diff -= subtract_amount
            if diff <= 1e-10:
                break
                
    return x
