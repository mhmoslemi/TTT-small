import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

# Global constants
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5_exact(h):
    """Exact evaluation of C5 using np.correlate"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(60):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_logsumexp(corr, alpha):
    """Smooth approximation to max with parameter alpha"""
    # Stabilize the computation
    max_corr = np.max(corr)
    exp_corr = np.exp(alpha * (corr - max_corr))
    return max_corr + np.log(np.sum(exp_corr)) / alpha

    h_sine = 0.5 + 0.5 * np.sin(np.pi * x + phi)
    h_sine = project_to_sum_box(h_sine, target_sum)
    
    # 3. Amplitude-tuned sinusoid
    amp = np.random.uniform(0, 0.5)
    h_sine_amplitude = 0.5 + amp * np.sin(np.pi * x)
    h_sine_amplitude = project_to_sum_box(h_sine_amplitude, target_sum)
    
    # 4. Triangular wave
    h_triangle = np.zeros(n_points)
    middle = n_points // 2
    h_triangle[:middle] = np.linspace(0, 1, middle)
    h_triangle[middle:] = np.linspace(1, 0, n_points - middle)
    h_triangle = project_to_sum_box(h_triangle, target_sum)
    
    # 5. Random
    h_random = np.random.rand(n_points)
    h_random = project_to_sum_box(h_random, target_sum)
    
    # 6. Hybrid
    h_hybrid = 0.7 * h_sine + 0.3 * h_random
    h_hybrid = project_to_sum_box(h_hybrid, target_sum)
    
    initial_solutions = [
        initial_h,
        h_sine,
        h_sine_amplitude,
        h_triangle,
        h_random,
        h_hybrid
    ]
    
    best_c5 = float('inf')
    best_h = None
    
    # Run continuation across multiple initializations
    continuation_levels = [1, 5, 20, 80, 300]
    max_iter_per_level = 200
    
    for idx, h_start in enumerate(initial_solutions):
        if time.time() > start_time + 0.9 * budget_s:
            break
        current_h = h_start.copy()
        current_c5 = evaluate_c5_exact(current_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
        
        # Continuation approach
        prev_h = None
        for alpha in continuation_levels:
            # Define objective function for this alpha
            def obj(h):
                corr = compute_corr_fft(h)
                value = smooth_max_logsumexp(corr, alpha)
                return value
                
            def grad(h):
                corr = compute_corr_fft(h)
                g = smooth_grad_logsumexp(corr, alpha, h)
                return g
                
            # Define constraints
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum,
                    'jac': lambda h: np.ones_like(h)}]
            
            # Optimize using SLSQP with gradient
            result = minimize(obj, current_h, method='SLSQP', jac=grad,
                              bounds=[(0, 1) for _ in range(n_points)],
                              constraints=cons, 
                              options={'maxiter': max_iter_per_level, 'ftol': 1e-9,
                                       'gtol': 1e-9})
            
            now_h = result.x.copy()
            now_h = project_to_sum_box(now_h, target_sum)  # ensure sum constraint
            now_c5 = evaluate_c5_exact(now_h)

                best_c5 = now_c5
                best_h = now_h.copy()
                
            current_h = now_h.copy()

                change = np.linalg.norm(current_h - prev_h)

                    # Add noise to escape local min
                    noise = np.random.normal(0, 0.01, n_points)
                    current_h = project_to_sum_box(current_h + noise, target_sum)
            prev_h = current_h.copy()

        stall_count = 0

                break
                
            # Try multiple types of moves
            move_type = np.random.choice(['swap', 'transfer'])
            
            if move_type == 'swap':
                i, j = np.random.choice(n_points, 2, replace=False)
                temp_h = current_h.copy()
                temp_h[i], temp_h[j] = temp_h[j], temp_h[i]
                temp_h = project_to_sum_box(temp_h, target_sum)
                temp_c5 = evaluate_c5_exact(temp_h)
                if temp_c5 <= current_c5:
                    current_h = temp_h
                    current_c5 = temp_c5
                    stall_count = 0
                else:
                    stall_count += 1
                
            elif move_type == 'transfer':
                i, j = np.random.choice(n_points, 2, replace=False)
                delta = np.random.uniform(0.005, 0.05)
                if current_h[i] >= delta and current_h[j] <= 1 - delta:
                    temp_h = current_h.copy()
                    temp_h[i] -= delta
                    temp_h[j] += delta
                    temp_h = project_to_sum_box(temp_h, target_sum)
                    temp_c5 = evaluate_c5_exact(temp_h)
                    if temp_c5 <= current_c5:
                        current_h = temp_h
                        current_c5 = temp_c5
                        stall_count = 0
                    else:
                        stall_count += 1
            
            if stall_count > max_stall:
                break

    if best_h is None:
        best_h = initial_h
        
    final_c5 = evaluate_c5_exact(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
