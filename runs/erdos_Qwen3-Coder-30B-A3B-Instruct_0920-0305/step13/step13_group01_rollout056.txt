```python
import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

# Problem parameters
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_correlation(h):
    """Compute correlation using FFT, same as before."""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5_exact(h):
    """Exact C5 using numpy correlate."""
    corr = compute_correlation(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target):
    """Project vector x onto box [0,1] with sum = target."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def logsumexp_stable(x, axis=None):
    """Stable log-sum-exp computation."""
    x_max = np.max(x, axis=axis, keepdims=True)
    return x_max + np.log(np.sum(np.exp(x - x_max), axis=axis, keepdims=True))

def smooth_max_alpha(h, alpha):
    """Compute smooth max of correlations."""
    corr = compute_correlation(h)
    corr = corr * alpha  # Scale by alpha
    log_sum = logsumexp_stable(corr)
    return dx * log_sum / alpha

def grad_smooth_max_alpha(h, alpha):
    """Gradient of smooth max w.r.t. h."""
    corr = compute_correlation(h)
    corr = corr * alpha
    exp_corr = np.exp(corr - np.max(corr, axis=None))
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient: for each shift k, weight k contributes to grad
    grad = np.zeros_like(h)
    n = len(h)
    for k in range(-n + 1, n):
        if k > 0:
            grad[:-k] += weights[k + n - 1] * (1 - h[k:])
        elif k < 0:
            grad[-k:] += weights[k + n - 1] * (1 - h[:k])
        else:
            grad += weights[n - 1] * (1 - 2 * h)
    return dx * grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Create diversified initializations
    initial_h_list = []

    # 1. Repeat initial pattern
    repeat_pattern = np.array(initial_h_values)
    repeated = np.tile(repeat_pattern, n_points // len(repeat_pattern) + 1)[:n_points]
    initial_h_list.append(project_to_sum_box(repeated, target_sum))

    # 2. Sinusoidal with random phase
    x_vals = (np.arange(n_points) + 0.5) * dx
    phase = np.random.rand() * 2 * np.pi
    h_sin = 0.5 + 0.5 * np.sin(np.pi * x_vals + phase)
    initial_h_list.append(project_to_sum_box(h_sin, target_sum))

    # 3. Amplitude-tuned sinusoid
    amp = np.random.rand() * 0.5
    h_sin_amp = 0.5 + amp * np.sin(np.pi * x_vals)
    initial_h_list.append(project_to_sum_box(h_sin_amp, target_sum))

    # 4. Triangular wave
    # Make a triangular wave: rise-fall over [0,2]
    x = np.linspace(0, 2, n_points)
    h_tri = np.where(x < 1, 2 * x, 2 - 2 * (x - 1))
    # Renormalize to get target sum
    h_tri = h_tri * target_sum / np.sum(h_tri)
    initial_h_list.append(h_tri)

    # 5. Random uniform, projected
    random_uniform = np.random.rand(n_points)
    initial_h_list.append(project_to_sum_box(random_uniform, target_sum))

    # 6. Hybrid: sinusoid + random
    hybrid = 0.7 * h_sin + 0.3 * np.random.rand(n_points)
    initial_h_list.append(project_to_sum_box(hybrid, target_sum))
    
    best_c5 = float('inf')
    best_h = None

    # Continuation levels
    alpha_levels = [1, 5, 20, 80, 300]
    
    # For tracking how many starts we can do in time
    start_count = 0
    max_starts = 10  # reasonable limit for 600 seconds
    
    for h_start in initial_h_list:
        if start_time + budget_s - time.time() < 10:
            break
            
        h_current = h_start.copy()
        
        # High-level optimization loop using continuation
        for alpha in alpha_levels:
            if time.time() >= start_time + 0.95 * budget_s:
                break
                
            # Define objective and gradient
            def obj(h):
                return smooth_max_alpha(h, alpha)
            
            def grad(h):
                return grad_smooth_max_alpha(h, alpha)
            
            # Constraints: box constraints + sum constraint
            bounds = [(0, 1)] * n_points
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                    'jac': lambda x: np.ones_like(x)}
            
            # Optimization with trust-constr
            res = minimize(obj, h_current, method='trust-constr', jac=grad, 
                          bounds=bounds, constraints=cons,
                          options={'gtol': 1e-9, 'ftol': 1e-9, 'maxiter': 200})
            
            if res.success:
                h_current = res.x
                h_current = project_to_sum_box(h_current, target_sum)
            else:
                # Perturb slightly if optimization failed
                h_current += np.random.normal(0, 0.005, n_points)
                h_current = project_to_sum_box(h_current, target_sum)
        
        # Final evaluation
        current_c5 = evaluate_c5_exact(h_current)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
    
    # Local search polishing
    if best_h is not None:
        h_best = best_h.copy()
        current_c5 = evaluate_c5_exact(h_best)
        swapped = False
        
        for _ in range(3000):
            if time.time() >= start_time + budget_s:
                break
            
            i, j = np.random.choice(n_points, 2, replace=False)
            h_trial = h_best.copy()
            h_trial[i], h_trial[j] = h_trial[j], h_trial[i]
            h_trial = project_to_sum_box(h_trial, target_sum)
            
            new_c5 = evaluate_c5_exact(h_trial)
            if new_c5 <= current_c5:
                h_best = h_trial
                current_c5 = new_c5
                swapped = True
            elif not swapped and np.random.rand() < 0.01:
                # Accept occasional worse solutions to escape local minima
                h_best = h_trial
                current_c5 = new_c5
            
            # Early stopping if improvement is too slow
            if not swapped and _ > 200:
                break
            elif swapped:
                swapped = False
        
        best_c5 = evaluate_c5_exact(h_best)
        best_h = h_best
    
    if best_h is None:
        # Fallback to initial pattern
        h0 = np.array(initial_h_values)
        h0 = project_to_sum_box(h0, target_sum)
        best_h = h0.copy()
        best_c5 = evaluate_c5_exact(h0)
    
    return (best_h.tolist(), float(best_c5), n_points)
```
