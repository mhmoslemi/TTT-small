def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Use a simple iterative approach to project onto the sum constraint
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    
    if abs(current_sum - target_sum) < 1e-10:
        return x

    tol = 1e-12
    max_iter = 1000
    for _ in range(max_iter):
        diff = target_sum - current_sum

            break
        # Adjust by moving towards target sum
        adjustment = diff / len(x)
        x = np.clip(x + adjustment, 0, 1)
        current_sum = np.sum(x)
    
    return x
