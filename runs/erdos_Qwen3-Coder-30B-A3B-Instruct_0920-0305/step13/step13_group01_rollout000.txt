import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1 << (L - 1).bit_length()
    corr = np.fft.ifft(np.fft.fft(h_rev, fft_size) * np.fft.fft(b, fft_size)).real[:L]
    return corr

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT (returns raw correlation max)"""
    corr = compute_corr_fft(h)
    return np.max(corr)

def evaluate_c5_precise(h):
    """Precise evaluation using np.correlate"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Find appropriate bounds for lambda
    lo = np.min(x) - 1.0
    hi = np.max(x) + 1.0
    
    # Binary search
    tol = 1e-12
    for _ in range(60):
        m = (lo + hi) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(lo) < 0:
            hi = m
        else:
            lo = m
    
    result = np.clip(x + (lo + hi)/2, 0, 1)
    return result

def compute_subgradient(h, corr):
    """Compute subgradient for the max correlation objective"""
    n = len(h)
    max_val = np.max(corr)
    # tolerance for finding active indices
    eps = 1e-12
    active_idx = np.where(corr >= max_val - eps)[0]
    
    # Initialize gradient with constant term: 1 per active lag per component
    g = np.full(n, len(active_idx), dtype=float)
    
    for idx in active_idx:
        k = idx - (n - 1)
        if k >= 0:
            # subtract h_{j + k} for j from 0 to n - k -1
            g[:n - k] -= h[k:]
            # subtract h_{j - k} for j from k to n-1
            g[k:] -= h[:n - k]
        else:
            d = -k
            # Equivalent to k negative case
            g[d:] -= h[:n - d]
            g[:n - d] -= h[d:]
    
    return g, max_val

def generate_sin_initial():
    """Generate sinusoidal initial solution"""
    x = (np.arange(n_points) + 0.5) * dx
    # Random amplitude and phase
    a = np.random.rand() * 0.5
    phase = np.random.rand() * 2 * np.pi
    h = 0.5 + a * np.sin(np.pi * x + phase)
    # Ensure values are in [0,1]
    h = np.clip(h, 0, 1)
    # Project to exact sum
    h = project_to_sum_box(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    global n_points, dx, target_sum
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with repeated initial solution if available
    if 'initial_h_values' in globals() and len(initial_h_values) == 120:
        h0 = np.repeat(initial_h_values, 2)
    else:
        h0 = np.full(n_points, 0.5)  # Default uniform distribution
    
    # Ensure h0 satisfies constraints
    h0 = project_to_sum_box(h0, target_sum)
    
    # Evaluate initial solution
    best_c5 = evaluate_c5_precise(h0)
    best_h = h0.copy()
    
    # Generate multiple initial candidates
    candidates = [h0]
    for _ in range(4):
        candidates.append(generate_sin_initial())
    
    # Add a few random starts
    for _ in range(3):
        h_random = np.random.rand(n_points)
        h_random = project_to_sum_box(h_random, target_sum)
        candidates.append(h_random)
    
    # Perform subgradient descent on each candidate
    max_iters_per_start = 10000
    base_lr = 0.3
    
    for i, h_start in enumerate(candidates):
        h = h_start.copy()
        current_c5 = evaluate_c5_fast(h)
        
        for iter_num in range(max_iters_per_start):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute correlation and subgradient
            corr = compute_corr_fft(h)
            g, max_corr = compute_subgradient(h, corr)
            
            # Compute step size
            step = base_lr / np.sqrt(iter_num + 1)
            
            # Update with gradient descent step
            h_new = h - step * g
            
            # Project to feasible region
            h_new = project_to_sum_box(h_new, target_sum)
            
            # Evaluate new solution
            new_corr = compute_corr_fft(h_new)
            new_max_corr = np.max(new_corr)
            new_c5 = new_max_corr * dx
            
            # Accept improvement or keep current
            if new_c5 < current_c5:
                h = h_new
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
    
    # Final local search with swaps
    for _ in range(2000):
        if time.time() - start_time > budget_s - 2:
            break
        
        # Random swap of two indices
        i, j = np.random.randint(0, n_points, 2)
        h_temp = best_h.copy()
        h_temp[i], h_temp[j] = h_temp[j], h_temp[i]
        
        # Project to constraints
        h_temp = project_to_sum_box(h_temp, target_sum)
        
        # Evaluate
        temp_c5 = evaluate_c5_precise(h_temp)
        if temp_c5 < best_c5:
            best_c5 = temp_c5
            best_h = h_temp.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
