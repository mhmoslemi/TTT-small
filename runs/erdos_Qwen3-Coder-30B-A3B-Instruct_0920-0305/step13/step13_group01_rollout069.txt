import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        # Evaluate fitness (lower C5 is better)

            fitness.append(1.0 / (1e-10 + c5))  # Avoid division by zero
            
        # Selection (tournament selection)
        new_population = []
        for _ in range(pop_size):
            # Tournament selection
            idx1, idx2 = np.random.choice(pop_size, 2, replace=False)
            if fitness[idx1] > fitness[idx2]:
                selected = population[idx1]
            else:
                selected = population[idx2]
            new_population.append(selected.copy())

        next_generation = []
        for i in range(0, pop_size, 2):
            parent1 = new_population[i]
            parent2 = new_population[i+1] if i+1 < pop_size else new_population[0]

                child1 = np.where(mask, parent1, parent2)

                    best_c5 = current_c5
                    best_h = current_h.copy()
            
            # Cooling schedule
            temp *= alpha
            if temp < min_temp:
                temp = min_temp

    if best_h is None:
        best_h = h0.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
