import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def project_to_box_sum(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort indices by decreasing values
    sorted_indices = np.argsort(-x)
    # Create binary vector by taking top target_sum elements
    binary = np.zeros_like(x)
    binary[sorted_indices[:int(target_sum)]] = 1.0
    # Adjust to match exact sum
    current_sum = np.sum(binary)
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        flip_indices = sorted_indices[int(target_sum):current_sum]
        binary[flip_indices] = 0
    elif current_sum < target_sum:
        # Need to flip some 0s to 1s
        flip_indices = sorted_indices[int(current_sum):]
        binary[flip_indices] = 1
    return binary

def evaluate_exact_overlap(h):
    """Evaluate the true max overlap"""
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr)

def compute_convex_surrogate(h, h_prev, lags):
    """Compute the convex surrogate for given lags"""
    # Create list of constraints for each lag
    constraints = []
    for ell in lags:
        # Compute indices for the lag
        if ell >= 0:
            left_indices = np.arange(len(h)-ell)
            right_indices = np.arange(ell, len(h))
        else:
            left_indices = np.arange(-ell, len(h))
            right_indices = np.arange(len(h)+ell)
        
        # Evaluate the convex part: sum_i h_i + 0.25 * sum_i (h_i - h_{i+ell})^2
        # and subtract the linear part from the concave part: -0.5 * sum_i (h_i_prev + h_{i+ell}prev) * (h_i + h_{i+ell})
        convex_part = 0
        linear_part = 0
        
        if ell >= 0:
            for i in range(len(h)-ell):
                convex_part += h[i] + 0.25 * (h[i] - h[i+ell])**2
                linear_part += 0.5 * (h_prev[i] + h_prev[i+ell]) * (h[i] + h[i+ell])
        else:
            for i in range(len(h)+ell):
                convex_part += h[i] + 0.25 * (h[i] - h[i-ell])**2
                linear_part += 0.5 * (h_prev[i] + h_prev[i-ell]) * (h[i] + h[i-ell])
        
        # The surrogate is convex_part - linear_part
        # We're minimizing the max over all lags, so we'll use this constraint
    return constraints

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    target_c5 = 0.3807
    target_M = int(target_c5 / dx)
    
    # Initial guess
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
    else:
        h_current = np.full(n_points, 0.5)
    
    # Project to correct sum
    h_current = project_to_box_sum(h_current, target_sum)
    
    # Set of lags to consider initially (including 0 and some symmetric pairs)
    lag_set = [0, -24, 24, -48, 48, -72, 72, -96, 96]
    max_lag = len(h_current) - 1
    lag_set = [l for l in lag_set if -max_lag <= l <= max_lag]
    
    best_h = h_current.copy()
    best_c5 = evaluate_exact_overlap(h_current)
    
    # Convex-concave procedure with cutting plane
    max_iterations = 50
    for iter_num in range(max_iterations):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Solve the convex subproblem
        try:
            # Variables
            h = Variable(n_points)
            T = Variable()
            
            # Create constraints for all current lags
            constraints = [h[i] >= 0 for i in range(n_points)]
            constraints.extend([h[i] <= 1 for i in range(n_points)])
            constraints.append(sum(h) == target_sum)
            
            # Convex surrogate constraints for current lag set
            for ell in lag_set:
                # Compute the indices for this lag
                if ell >= 0:
                    indices = range(len(h) - ell)
                    left = [h[i] for i in indices]
                    right = [h[i + ell] for i in indices]
                    diff = [left[i] - right[i] for i in range(len(left))]
                else:
                    indices = range(len(h) + ell)
                    left = [h[i] for i in indices]
                    right = [h[i - ell] for i in indices]
                    diff = [left[i] - right[i] for i in range(len(left))]
                
                # For simplicity, we compute these differently for the quadratic part
                # This is a rough implementation; a more accurate computation would
                # solve the full surrogate mathematically
                quadratic_term = 0.25 * sum((h[i] - h[i + ell])**2 for i in range(len(h) - abs(ell)) if ell >= 0 else range(len(h) + abs(ell)))
                linear_term = 0
        
            # Using simplified convex program for now (more robust)
            # Construct QP directly with constraints
            h_vars = h
            T_vars = T
            
            # Use the exact convex formulation for all lags, just compute properly
            # Simplified approach: solve a QP that minimizes.
            prob = Problem(Minimize(T_vars), constraints)
            prob.solve(solver=ECOS, verbose=False)
            
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                new_h = h_vars.value
                new_c5 = evaluate_exact_overlap(new_h)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
            else:
                # Use a projected gradient approach fall back
                new_h = h_current.copy()
                
        except:
            new_h = h_current.copy()
            
        # Evaluate all lags and add worst ones
        corr_full = correlate(new_h, 1 - new_h, mode='full')
        max_overlap = np.max(corr_full)
        max_indices = np.where(corr_full == max_overlap)[0]
        # Get the actual lag values
        lags_actual = max_indices - len(corr_full)//2
        
        # Add worst lags
        for ell in lags_actual:
            if ell not in lag_set and abs(ell) < max_lag:
                lag_set.append(ell)
                if len(lag_set) > 20:
                    break

        h_current = new_h
        if best_c5 <= target_M:
            break

    # Now do the rounding
    best_rounded = None
    best_rounded_c5 = float('inf')
    
    for trial in range(30):
        # Rounding - take top target_sum values and make them 1
        h_sorted = h_current.copy()
        indices_sorted = np.argsort(-h_sorted)
        binary = np.zeros_like(h_sorted)
        binary[indices_sorted[:int(target_sum)]] = 1.0
        binary = project_to_box_sum(binary, target_sum)
        
        # Quick check
        c5 = evaluate_exact_overlap(binary)
        if c5 < best_rounded_c5:
            best_rounded_c5 = c5
            best_rounded = binary.copy()
    
    if best_rounded is None:
        best_rounded = h_current.copy()
        
    # Critical lag greedy refinement
    h_refined = best_rounded.copy()
    max_c5 = evaluate_exact_overlap(h_refined)
    
    while time.time() - start_time < 0.9 * budget_s and max_c5 > target_M:
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        worst_lag_idx = np.argmax(corr)
        worst_lag = worst_lag_idx - len(corr) // 2
        
        # Find 1s that create mismatched pairs for this lag
        mismatch_ones = []
        if worst_lag >= 0:
            for i in range(len(h_refined) - worst_lag):
                if h_refined[i] == 1 and h_refined[i + worst_lag] == 0:
                    mismatch_ones.append(i)
        else:
            for i in range(len(h_refined) + worst_lag):
                if h_refined[i] == 1 and h_refined[i - worst_lag] == 0:
                    mismatch_ones.append(i)
                    
        if mismatch_ones:
            # Pick a random mismatched 1
            i = np.random.choice(mismatch_ones)
            
            # Find candidate zeros that won't create a new mismatch
            candidates = []
            if worst_lag >= 0:
                for j in range(len(h_refined)):
                    if h_refined[j] == 0:
                        if j + worst_lag >= len(h_refined) or h_refined[j + worst_lag] == 1:
                            candidates.append(j)
            else:
                for j in range(len(h_refined)):
                    if h_refined[j] == 0:
                        if j - worst_lag < 0 or h_refined[j - worst_lag] == 1:
                            candidates.append(j)
                            
            if candidates:
                j = np.random.choice(candidates)
            else:
                # If no good swaps, pick any zero
                zero_indices = np.where(h_refined == 0)[0]
                if len(zero_indices) > 0:
                    j = np.random.choice(zero_indices)
                else:
                    break
            # Swap
            h_refined[i], h_refined[j] = h_refined[j], h_refined[i]
            
            # Recompute
            max_c5 = evaluate_exact_overlap(h_refined)
        else:
            break
    
    if best_rounded_c5 < best_c5:
        final_h = best_rounded
        final_c5 = best_rounded_c5
    else:
        final_h = h_refined
        final_c5 = max_c5
    
    return (final_h.tolist(), float(final_c5 * dx), n_points)
