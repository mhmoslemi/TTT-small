import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def adaptive_perturbation(h, target_sum, perturb_rate=0.1):
    """Generate perturbed version with adaptive perturbation size"""
    h_new = h.copy()
    num_perturb = max(1, int(n_points * perturb_rate))
    indices = np.random.choice(n_points, num_perturb, replace=False)
    
    # Use adaptive perturbation size based on current values
    for i in indices:
        # Perturb more aggressively in regions that are far from 0.5
        perturb_mag = 0.05 + 0.05 * (0.5 - abs(h_new[i] - 0.5))
        perturb = np.random.uniform(-perturb_mag, perturb_mag)
        h_new[i] += perturb
    
    return project_to_sum_box(h_new, target_sum)
