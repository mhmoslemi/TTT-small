def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    h = np.array(h_values)
    dx = 2.0 / len(h)
    
    # Compute correlation using np.correlate with full mode
    corr = np.correlate(h, 1 - h, mode="full")
    
    # The result of np.correlate with mode "full" has length 2*n-1
    # We want to consider only the main overlapping regions
    # The maximum value corresponds to the best overlap
    max_corr = np.max(corr) * dx
    
    return max_corr
