import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1 << (L - 1).bit_length()
    A = np.fft.fft(h_rev, fft_size)
    B = np.fft.fft(b, fft_size)
    corr = np.fft.ifft(A * B).real[:L]
    return corr

def evaluate_c5_precise(h):
    """Evaluate C5 using np.correlate for exact computation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def evaluate_c5_fast(corr):
    """Evaluate C5 from correlation (already multiplied by dx)"""
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = np.min(x) - 1.0, np.max(x) + 1.0
    tol = 1e-10
    for _ in range(60):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_subgradient(h, corr):
    """Compute subgradient of max correlation"""
    n = len(h)
    max_val = np.max(corr)
    eps = 1e-12
    active_idx = np.where(corr >= max_val - eps)[0]
    
    # Initialize gradient with constant term: 1 per active lag per component
    g = np.full(n, len(active_idx), dtype=float)
    
    for idx in active_idx:
        k = idx - (n - 1)
        if k >= 0:
            # Subtract h_{j + k} for j from 0 to n - k -1
            g[:n - k] -= h[k:]
            # Subtract h_{j - k} for j from k to n-1
            g[k:] -= h[:n - k]
        else:
            d = -k
            # Subtract h_{j + d} for j from d to n-1
            g[d:] -= h[:n - d]
            # Subtract h_{j - d} for j from 0 to n-d-1
            g[:n - d] -= h[d:]
    
    return g, max_val

def generate_sin_initial():
    """Generate sinusoidal initial solution"""
    x = (np.arange(n_points) + 0.5) * dx
    # Random amplitude and phase
    a = np.random.rand() * 0.5
    phase = np.random.rand() * 2 * np.pi
    h = 0.5 + a * np.sin(np.pi * x + phase)
    # Ensure values in [0,1]
    h = np.clip(h, 0, 1)
    # Project to sum constraint
    h = project_to_sum_box(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    global n_points, dx, target_sum
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from repeated initial_h_values if available and correct length
    if 'initial_h_values' in globals() and len(initial_h_values) == 120:
        h0 = np.repeat(initial_h_values, 2)
    else:
        # Fallback to random uniform
        h0 = np.random.rand(n_points)
        h0 = project_to_sum_box(h0, target_sum)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5_precise(best_h)
    
    # Generate additional starting points
    start_candidates = [h0]
    for _ in range(5):
        start_candidates.append(generate_sin_initial())
    
    # Run subgradient descent with multiple restarts
    max_iters_per_start = 10000
    restarts = 0
    
    for i, h_start in enumerate(start_candidates):
        if time.time() - start_time > budget_s - 10:
            break
            
        h = h_start.copy()
        current_c5 = evaluate_c5_precise(h)
        
        # Run subgradient descent from this start
        for iter_num in range(max_iters_per_start):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute correlation
            corr = compute_corr_fft(h)
            
            # Compute subgradient
            g, max_corr = compute_subgradient(h, corr)
            
            # Step size with decaying learning rate
            step = 0.3 / np.sqrt(iter_num + 1)
            
            # Update step direction
            norm_g = np.linalg.norm(g)
            if norm_g > 0:
                direction = g / norm_g
            else:
                direction = np.zeros_like(g)
            
            h_new = h - step * direction
            
            # Project onto box and sum constraints
            h_new = project_to_sum_box(h_new, target_sum)
            
            # Evaluate new solution
            new_corr = compute_corr_fft(h_new)
            new_c5_raw = np.max(new_corr)
            new_c5 = new_c5_raw * dx
            
            if new_c5 < current_c5:
                h = h_new
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
        
        restarts += 1
    
    # Final local search with neighbor swaps
    for _ in range(2000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Choose two random indices to swap
        i, j = np.random.randint(0, n_points, 2)
        h_swap = best_h.copy()
        h_swap[i], h_swap[j] = h_swap[j], h_swap[i]
        
        # Project to constraints
        h_swap = project_to_sum_box(h_swap, target_sum)
        
        # Evaluate
        new_c5 = evaluate_c5_precise(h_swap)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_swap.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
