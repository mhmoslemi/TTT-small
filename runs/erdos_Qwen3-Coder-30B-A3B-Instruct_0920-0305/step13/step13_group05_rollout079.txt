    x = np.array(x)

    exp_x = np.exp(beta * (x - max_x))
    return max_x + np.log(np.sum(exp_x)) / beta

def compute_corr_fft(h):
    """Compute correlation using FFT"""

        a -= 0.1
        b += 0.1

    tol = 1e-10
