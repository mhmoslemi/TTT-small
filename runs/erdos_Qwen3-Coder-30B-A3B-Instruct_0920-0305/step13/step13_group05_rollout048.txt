def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort values in descending order
    sorted_indices = np.argsort(-x)
    sorted_x = x[sorted_indices]
    
    # Cumulative sum
    cumsum = np.cumsum(sorted_x)
    
    # Find the index where we need to truncate
    # We want sum of first k elements to be <= target_sum
    # But we want the largest such k
    k = 0
    while k < len(sorted_x) and cumsum[k] > target_sum:
        k += 1
    
    # If we have to truncate to k=0, we set everything to 0
    if k == 0:
        result = np.zeros_like(x)
    else:
        # Equally distribute the remaining sum among first k elements
        remaining = target_sum - cumsum[k-1] if k > 0 else target_sum
        if k < len(sorted_x):
            # Distribute among first k elements
            avg_val = remaining / k
            # All first k elements get avg_val, others get clipped to 0
            result = np.zeros_like(x)
            result[sorted_indices[:k]] = avg_val
        else:
            # All elements are selected, just make them sum to target
            result = sorted_x.copy()
            if k > 0:
                result[:k] = target_sum / k
                
    return np.clip(result, 0, 1)

def smooth_max(x, gamma=100):
    """Smooth approximation of max using softmax"""
    # Use log-sum-exp trick for numerical stability
    return np.log(np.sum(np.exp(gamma * x))) / gamma

def smooth_max_gradient(x, gamma=100):
    """Gradient of smooth max"""
    exp_x = np.exp(gamma * x)
    return exp_x / (np.sum(exp_x))
