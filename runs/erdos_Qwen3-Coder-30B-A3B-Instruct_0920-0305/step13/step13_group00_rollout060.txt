def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""

    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_gradient(h, corr, alpha=10.0):
    """Compute gradient using softmax approximation"""
    # Compute weights
    max_corr = np.max(corr)
    if max_corr == 0:
        max_corr = 1e-10
    w = np.exp(alpha * (corr - max_corr))
    w = w / np.sum(w)
    
    # Compute gradient: sum over shifts of w_s * (1 - h_{i+s} - h_{i-s})
    g = np.ones(len(h))
    
    # For each shift, compute contribution
    for i in range(len(h)):
        # For shift s = j - (n_points-1)
        for j in range(2 * len(h) - 1):
            s = j - (len(h) - 1)
            if s == 0:
                g[i] -= 2 * w[j] * h[i]
            elif s > 0:
                # h_{i+s} term
                if i + s < len(h):
                    g[i] -= w[j] * h[i + s]
                # h_{i-s} term
                if i - s >= 0:
                    g[i] -= w[j] * h[i - s]
            else:  # s < 0
                # h_{i+s} term (which is h_{i-abs(s)})
                abs_s = -s
                if i + abs_s < len(h):
                    g[i] -= w[j] * h[i + abs_s]
                # h_{i-s} term (which is h_{i+abs(s)})
                if i - abs_s >= 0:
                    g[i] -= w[j] * h[i - abs_s]
    
    return g
