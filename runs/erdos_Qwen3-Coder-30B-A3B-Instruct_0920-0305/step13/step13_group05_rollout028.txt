def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmax_smooth_max(x, gamma=100.0):
    """Smooth approximation of max using softmax"""
    # Stable computation of softmax
    x_shifted = x - np.max(x)
    exp_x = np.exp(gamma * x_shifted)
    return np.sum(x * exp_x) / np.sum(exp_x)

def softmax_smooth_max_gradient(x, gamma=100.0):
    """Gradient of softmax smooth max"""
    x_shifted = x - np.max(x)
    exp_x = np.exp(gamma * x_shifted)
    sum_exp = np.sum(exp_x)
    softmax = exp_x / sum_exp
    return softmax

    # Use softmax smoothing of the maximum
    max_val = softmax_smooth_max(corr, gamma)

    eps = 1e-6
    grad = np.zeros_like(h)
    
    # For each component, estimate gradient by perturbing slightly
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        corr_plus = compute_corr_fft(h_plus)
        h_minus = h.copy()
        h_minus[i] -= eps
        corr_minus = compute_corr_fft(h_minus)
        
        # Smooth max gradient estimation
        grad[i] = (softmax_smooth_max(corr_plus, gamma) - softmax_smooth_max(corr_minus, gamma)) / (2 * eps)
    
    return max_val * dx, grad * dx

def accelerated_projected_gradient_descent(h0, max_iter=10000, gamma=100.0, lr_schedule=None):
    """Accelerated projected gradient descent with momentum"""
    h = h0.copy()
    y = h.copy()
    t_old = 1
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    if lr_schedule is None:
        # Use decreasing learning rates
        def lr(k):
            return 1.0 / (1.0 + 0.01 * k)
    else:
        lr = lr_schedule
    
    for k in range(max_iter):
        # Compute smooth objective and gradient
        f, g = compute_smooth_objective_and_grad(y, gamma)
        
        # Gradient step
        h_new = y - lr(k) * g
        
        # Project onto constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Accelerated update
        t_new = (1 + np.sqrt(1 + 4 * t_old * t_old)) / 2
        y_new = h_new + ((t_old - 1) / t_new) * (h_new - h)
        
        h = h_new
        y = y_new
        t_old = t_new
        
        # Update best solution
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early stopping if improvement is small
        if k > 100 and abs(f) < 1e-8:
            break
            
    return best_h, best_c5
