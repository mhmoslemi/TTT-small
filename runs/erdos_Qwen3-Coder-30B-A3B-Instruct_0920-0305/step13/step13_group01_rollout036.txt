import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

# Global constants
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5_exact(h):
    """Exact evaluation of C5 using np.correlate"""
    corr = correlate(h, 1 - h, mode='full')
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_logsumexp(corr, alpha):
    """Compute smooth approximation to max using log-sum-exp"""
    corr_shifted = corr - np.max(corr)  # stabilize
    exp_corr = np.exp(alpha * corr_shifted)
    return np.log(np.sum(exp_corr)) / alpha

def smooth_max_grad(corr, alpha, h):
    """Compute gradient of smooth max with respect to h"""
    corr_shifted = corr - np.max(corr)
    exp_corr = np.exp(alpha * corr_shifted)
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient = dx * sum_k w_k * (1 - shift_+k(h) - shift_-k(h))
    grad = np.zeros_like(h)
    n = len(h)
    
    for k in range(-n+1, n):
        if k >= 0:
            shifted_h = np.concatenate([np.zeros(k), h[:-k]])
        else:
            shifted_h = np.concatenate([h[-k:], np.zeros(-k)])
        grad += weights[k+n-1] * (1 - shifted_h)
    
    return dx * grad

    initial_guesses = []
    # 1. Repeated legacy pattern
    h0_repeated = np.tile(initial_h_values, n_points // len(initial_h_values))
    initial_guesses.append(project_to_sum_box(h0_repeated, TARGET_SUM))
    
    # 2. Pure sinusoid
    x = np.arange(n_points) * dx + dx / 2
    phi = np.random.rand() * 2 * np.pi
    h1 = 0.5 + 0.5 * np.sin(np.pi * x + phi)
    initial_guesses.append(project_to_sum_box(h1, TARGET_SUM))
    
    # 3. Amplitude-tuned sinusoid
    a = np.random.rand() * 0.5
    h2 = 0.5 + a * np.sin(np.pi * x)
    initial_guesses.append(project_to_sum_box(h2, TARGET_SUM))
    
    # 4. Triangular wave
    t = np.linspace(0, 2, n_points)
    h3 = np.zeros_like(t)
    mask1 = t <= 1
    mask2 = t > 1
    h3[mask1] = 2 * t[mask1]
    h3[mask2] = 2 - 2 * t[mask2]
    h3 = h3 * (1 / np.trapz(h3, t))
    initial_guesses.append(project_to_sum_box(h3, TARGET_SUM))
    
    # 5. Random uniform
    h4 = np.random.rand(n_points)
    initial_guesses.append(project_to_sum_box(h4, TARGET_SUM))
    
    # 6. Hybrid - sinusoid + random
    a = np.random.rand() * 0.5
    h5 = 0.5 + a * np.sin(np.pi * x) + 0.3 * np.random.rand(n_points)
    initial_guesses.append(project_to_sum_box(h5, TARGET_SUM))
    
    return initial_guesses

def continuation_optimize(h_start, alpha_levels=[1, 5, 20, 80, 300]):
    """Run optimization with continuation on alpha"""
    h = h_start.copy()
    
    for alpha in alpha_levels:
        # Define objective and gradient
        def obj(h):
            corr = compute_corr_fft(h)
            return smooth_max_logsumexp(corr, alpha)
        
        def grad(h):
            corr = compute_corr_fft(h)
            return smooth_max_grad(corr, alpha, h)
        
        # Constraints for SLSQP
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - TARGET_SUM,
                       'jac': lambda h: np.ones_like(h)}]
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimize with SLSQP
        try:
            res = minimize(obj, h, method='SLSQP', jac=grad, bounds=bounds, 
                          constraints=constraints, tol=1e-9, options={'maxiter': 200})
            if res.success:
                h = res.x
            else:
                # If failed, try trust-constr as backup
                res = minimize(obj, h, method='trust-constr', jac=grad, bounds=bounds,
                              constraints=constraints, tol=1e-9, options={'maxiter': 200})
                if res.success:
                    h = res.x
        except:
            pass
        
        # Re-project to ensure sum constraint and box constraints
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Add small noise to avoid stagnation
        if alpha > 10:
            noise = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + noise, 0, 1)
            h = project_to_sum_box(h, TARGET_SUM)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial guesses
    initial_guesses = get_initial_guesses()
    
    best_c5 = float('inf')
    best_h = None
    
    # Run continuation optimization for each initial guess
    for i, h0 in enumerate(initial_guesses):
        if time.time() - start_time > budget_s - 10:
            break
            
        try:
            h_opt = continuation_optimize(h0)
            c5_val = evaluate_c5_exact(h_opt)
            
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                
        except Exception as e:
            continue
    
    # Local greedy search
    if best_h is not None:
        current_h = best_h.copy()
        current_c5 = evaluate_c5_exact(current_h)
        
        # Swap moves
        for i in range(3000):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Try swap
            i1, i2 = np.random.choice(n_points, 2, replace=False)
            temp_h = current_h.copy()
            temp_h[i1], temp_h[i2] = temp_h[i2], temp_h[i1]
            
            # Project to feasible set
            temp_h = project_to_sum_box(temp_h, TARGET_SUM)
            temp_c5 = evaluate_c5_exact(temp_h)
            
            if temp_c5 < current_c5:
                current_h = temp_h
                current_c5 = temp_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
        
        # Transfer moves
        for i in range(3000):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Try small transfer
            i1, i2 = np.random.choice(n_points, 2, replace=False)
            delta = np.random.rand() * 0.01
            if current_h[i1] >= delta:
                temp_h = current_h.copy()
                temp_h[i1] -= delta
                temp_h[i2] += delta
                
                # Project to feasible set
                temp_h = project_to_sum_box(temp_h, TARGET_SUM)
                temp_c5 = evaluate_c5_exact(temp_h)
                
                if temp_c5 < current_c5:
                    current_h = temp_h
                    current_c5 = temp_c5
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = current_h.copy()
    
    # Final evaluation with exact correlation
    if best_h is None:
        # Fallback to initial guess if nothing found
        h0 = np.tile(initial_h_values, n_points // len(initial_h_values))
        best_h = project_to_sum_box(h0, TARGET_SUM)
        best_c5 = evaluate_c5_exact(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
