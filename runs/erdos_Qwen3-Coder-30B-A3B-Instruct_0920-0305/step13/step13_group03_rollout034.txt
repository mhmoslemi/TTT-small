import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time
import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
TARGET_C5 = 0.3807

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_convex_subproblem(h_current, lag_set):
    """Solve the convex subproblem using CVXPY"""
    n = len(h_current)
    h = Variable(n)
    T = Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1, sum(h) == TARGET_SUM]
    
    # Add Lag constraints
    for ell in lag_set:
        # Compute the quadratic term and linear term for the lag ell
        quad_term = 0
        lin_term = 0
        
        for i in range(n):
            j = i + ell
            if 0 <= j < n:
                quad_term += (h[i] - h[j])**2
                lin_term += (h_current[i] + h_current[j]) * (h[i] + h[j])
        
        # The constraint is a convex quadratic
        constraints.append(T >= 1 + quad_term/4 - lin_term/2)
    
    # Objective
    objective = cp.Minimize(T)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value, T.value
    except:
        pass
    
    # If CVXPY fails, fallback to a simpler approach
    h_sol = h_current.copy()
    return h_sol, 1e10

def get_lag_set_initial(n):
    """Initial lag set"""
    half_n = n // 2
    return [0, -half_n//2, half_n//2, -half_n, half_n, -3*half_n//4, 3*half_n//4, -half_n//4, half_n//4]

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    
    # Start with initial solution
    h0 = np.array(initial_h_values)
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    best_h = h0.copy()
    best_c5 = c5_from_corr(compute_corr_fft(h0))
    
    # Initial lag set
    lag_set = get_lag_set_initial(n_points)
    
    # CCP with cutting plane
    max_iter = 500
    epsilon = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.7:
            break
            
        # Solve convex subproblem
        h_sol, T_sol = solve_convex_subproblem(best_h, lag_set)
        
        if T_sol >= best_c5:
            break
            
        # Evaluate all lags
        corr_full = compute_corr_fft(h_sol)
        max_corr = np.max(corr_full)
        
        # Check for violations
        violations = []
        for ell in range(-n_points + 1, n_points):
            if ell >= 0:
                corr_val = corr_full[n_points-1+ell]
            else:
                corr_val = corr_full[ell]
            if corr_val > T_sol + epsilon:
                violations.append(ell)
        
        if len(violations) == 0:
            best_h = h_sol.copy()
            best_c5 = c5_from_corr(compute_corr_fft(best_h))
            break
        
        # Add most violated lags
        for v in violations[:min(3, len(violations))]:  # Add up to 3 most violated
            if v not in lag_set:
                lag_set.append(v)
    
    # Rounding and local search
    best_binary = None
    best_binary_c5 = float('inf')
    
    # Multiple random draws
    for _ in range(30):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Sort and take top half
        indices = np.argsort(best_h)[::-1]
        binary_h = np.zeros(n_points)
        binary_h[indices[:int(TARGET_SUM)]] = 1
        
        # Project back to sum
        binary_h = project_to_sum_box(binary_h, TARGET_SUM)
        
        # Evaluate
        corr = compute_corr_fft(binary_h)
        c5_val = c5_from_corr(corr)
        
        if c5_val < best_binary_c5:
            best_binary = binary_h.copy()
            best_binary_c5 = c5_val
    
    # Critical lag polishing
    if best_binary is not None:
        h_current = best_binary.copy()
        corr = compute_corr_fft(h_current)
        max_corr = np.max(corr)
        
        while max_corr > TARGET_C5 / dx and time.time() - start_time < budget_s * 0.99:
            worst_lag = np.argmax(corr)
            if worst_lag >= n_points - 1:
                worst_lag = worst_lag - (2 * n_points - 2)
            
            # Find mismatched pairs
            mismatched_ones = []
            for i in range(n_points):
                j = i + worst_lag
                if 0 <= j < n_points:
                    if h_current[i] == 1 and h_current[j] == 0:
                        mismatched_ones.append(i)
            
            if len(mismatched_ones) == 0:
                break
                
            # Pick a random mismatched 1
            i = np.random.choice(mismatched_ones)
            # Find candidate zeros that don't create new mismatches at this lag
            candidate_zeros = []
            for j in range(n_points):
                if h_current[j] == 0:
                    # Check that swapping doesn't create a new mismatch
                    j_shifted = j + worst_lag
                    if not (0 <= j_shifted < n_points and h_current[j_shifted] == 1):
                        candidate_zeros.append(j)
            
            if len(candidate_zeros) > 0:
                j = np.random.choice(candidate_zeros)
                h_current[i], h_current[j] = h_current[j], h_current[i]
            else:
                # If no good zero, just find any zero
                zeros = np.where(h_current == 0)[0]
                if len(zeros) > 0:
                    j = np.random.choice(zeros)
                    h_current[i], h_current[j] = h_current[j], h_current[i]
            
            corr = compute_corr_fft(h_current)
            max_corr = np.max(corr)
            
            current_c5 = c5_from_corr(corr)
            if current_c5 < best_binary_c5:
                best_binary_c5 = current_c5
                best_binary = h_current.copy()
    
    if best_binary is None:
        best_binary = best_h.copy()
        
    return (best_binary.tolist(), float(best_binary_c5), n_points)
