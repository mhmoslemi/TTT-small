import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h, shifts):
    """Compute subgradient for given shifts"""
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # For positive shift, h[i] contributes to correlation with (1-h[i+shift])
            # and h[i+shift] contributes to correlation with (1-h[i])
            if shift < len(h):
                # Forward contribution: h[i] * (1-h[i+shift])
                grad[:len(h)-shift] += 1 - h[shift:]
                # Backward contribution: h[i+shift] * (1-h[i])  
                grad[shift:] -= h[:-shift]
        else:
            # For negative shift, h[i] contributes to correlation with (1-h[i-shift])
            # and h[i-shift] contributes to correlation with (1-h[i])
            shift_abs = -shift
            if shift_abs < len(h):
                # Forward contribution: h[i] * (1-h[i-shift])
                grad[shift_abs:] += 1 - h[:-shift_abs]
                # Backward contribution: h[i-shift] * (1-h[i])  
                grad[:-shift_abs] -= h[shift_abs:]
    return grad
