import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_simplex(v, target_sum):
    """Projection onto the capped simplex"""
    # v is the vector to project
    # target_sum is the desired sum (n/2)
    n = len(v)
    # Binary search for the correct lambda
    lmbd_low = -1.0
    lmbd_high = 1.0
    for _ in range(60):
        lmbd = (lmbd_low + lmbd_high) / 2
        p = np.clip(v + lmbd, 0, 1)
        s = np.sum(p)
        if s > target_sum:
            lmbd_high = lmbd
        else:
            lmbd_low = lmbd
    lmbd = (lmbd_low + lmbd_high) / 2
    return np.clip(v + lmbd, 0, 1)

def compute_gradient(h, active_shifts):
    """Compute sub-gradient for current active shifts"""
    # active_shifts: list of shift indices that achieve current max overlap
    n = len(h)
    dx = 2.0 / n
    
    # Create kernel
    w = np.zeros(n)
    for k in active_shifts:
        if k >= 0:
            w[k] = 1
        else:
            w[n + k] = 1
    
    # Compute convolutions
    w_sum = np.sum(w)  # number of active shifts
    s = np.convolve(np.ones(n), w, mode='full')[:n]  # Frequency weight
    # The effective gradient: dx·(s - (w⋆h) - (w⋆h_rev))
    h_rev = h[::-1]
    conv_h = correlate(h, w, mode='full')[:n]
    conv_h_rev = correlate(h_rev, w, mode='full')[:n]
    grad = dx * (s - conv_h - conv_h_rev)
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Parameters
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2  # Sum constraint

    # Phase 1: Binary MILP
    # Formulate this as an MILP
    h_b = Variable(n_points, boolean=True)
    tau = Variable()  # epigraph variable
    
    # Define auxiliary binary variables for product h[i]*h[i+k]
    # We'll create constraints for all valid pairs (i, i+k) for each k
    
    # Since the full shift range is from -n+1 to n-1, we generate all valid pairs
    constraints = []
    y_vars = {}  # Dictionary to store y_{i,k} for each valid pair
    pairs_per_shift = {}

    # Prepare all pairs and build constraints
    for k in range(-n_points + 1, n_points):  # shifts
        pairs = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                pairs.append((i, j))
                y = Variable(boolean=True)
                y_vars[(i, j, k)] = y
                # McCormick constraints:
                # y <= h[i], y <= h[j], y >= h[i] + h[j] - 1
                constraints.append(y <= h_b[i])
                constraints.append(y <= h_b[j])
                constraints.append(y >= h_b[i] + h_b[j] - 1)
        pairs_per_shift[k] = pairs

    # Define overlap for each shift
    O = []
    for k in range(-n_points + 1, n_points):
        pairs = pairs_per_shift.get(k, [])
        if not pairs:
            continue
        # O_k = sum(h) - sum(y_{i,k})
        expr = sum(h_b[i] for i, j in pairs) - sum(y_vars[(i, j, k)] for i, j in pairs)
        O.append(expr)
        constraints.append(expr <= tau)  # for epigraph formulation

    # Area constraint
    constraints.append(sum(h_b) == target_sum)

    # Objective
    objective = Minimize(tau)

    # Create the problem
    prob = Problem(objective, constraints)

    # Solve
    c5_milp_best = float('inf')
    h_milp_best = None
    
    # Solver might take too long for large n. We use a time limit and get 
    # best feasible solution so far. Note: CBC with time limit is tricky in cvxpy.
    # Our approach is adaptive: first solve with reasonable time limit,
    # fallback to random or approximation if needed.
    try:
        prob.solve(solver='CBC', verbose=False, max_secs=100)  # Time limit in seconds
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h_milp_float = h_b.value.astype(float)
            c5_milp = evaluate_c5(h_milp_float)
            if c5_milp < c5_milp_best:
                c5_milp_best = c5_milp
                h_milp_best = h_milp_float.copy()
                print("MILP returned feasible solution with C5 =", c5_milp)
    except:
        pass

    # Fallback to using the existing initial solution if MILP fails or times out
    if h_milp_best is None:
        h_initial = np.array(initial_h_values)
        # Normalize
        scale = target_sum / np.sum(h_initial)
        h_initial *= scale
        h_milp_best = h_initial.copy()
        c5_milp_best = evaluate_c5(h_milp_best)
    
    # Phase 2: Binary local search
    # Start from best milp result, convert to integer and improve
    h_binary = np.round(h_milp_best).astype(int)
    
    # Ensure that sum is correct
    diff = np.sum(h_binary) - target_sum
    if abs(diff) > 1e-6:
        # Adjust to get the right sum
        if diff > 0:
            # Find 1s and change them to 0s
            onesidx = np.where(h_binary == 1)[0]
            for i in onesidx[:int(diff)]:
                h_binary[i] = 0
        elif diff < 0:
            # Find 0s and change them to 1s
            zerosidx = np.where(h_binary == 0)[0]
            for i in zerosidx[:int(-diff)]:
                h_binary[i] = 1

    h_binary = h_binary.astype(float)
    c5_best = evaluate_c5(h_binary)

    # Hill-climbing swap heuristic
    nsweeps = 0
    improvement = True
    while improvement and (time.time() - start_time) < 0.7 * budget_s:
        improvement = False
        # Try all swaps from i (1) to j (0)
        for i in range(n_points):
            if h_binary[i] == 1:
                for j in range(n_points):
                    if h_binary[j] == 0:
                        # Simulate the swap
                        h_temp = h_binary.copy()
                        h_temp[i] = 0
                        h_temp[j] = 1
                        c5_temp = evaluate_c5(h_temp)
                        if c5_temp < c5_best - 1e-10:
                            h_binary = h_temp
                            c5_best = c5_temp
                            improvement = True
        nsweeps += 1
        if nsweeps % 10 == 0:
            print("Binary hill-climbing sweep", nsweeps, "C5 =", c5_best)

    # Phase 3: Continuous refinement
    h_cont = h_binary.copy()
    for t in range(500):  # Number of gradient descent steps
        if (time.time() - start_time) > 0.9 * budget_s:
            break

        corr = correlate(h_cont, 1 - h_cont, mode="full")
        max_corr = np.max(corr)
        max_positions = np.where(np.abs(corr - max_corr) < 1e-10)[0]
        # Convert indices back to shift indices
        active_shifts = []
        for pos in max_positions:
            if pos <= n_points - 1:
                active_shifts.append(pos)
            else:
                active_shifts.append(pos - n_points)

        grad = compute_gradient(h_cont, active_shifts)
        eta = 0.5 * dx
        h_new = h_cont - eta * grad
        h_new = project_to_simplex(h_new, target_sum)
        
        # Symmetrize
        h_sym = (h_new + h_new[::-1]) / 2.0
        h_sym = project_to_simplex(h_sym, target_sum)
        
        c5_new = evaluate_c5(h_sym)
        h_cont = h_sym
        if c5_new < c5_best - 1e-12:
            c5_best = c5_new
            h_binary = h_sym.copy()
        else:
            # a bit of cooling
            eta *= 0.999
        if t % 50 == 0:
            print("Continuous descent step", t, "C5 =", c5_best)

    # Optionally upsample if time remains
    if (time.time() - start_time) < 0.95 * budget_s:
        n_up = n_points * 2
        dx_up = 2.0 / n_up
        h_up = np.interp(np.linspace(0, 2, n_up), np.linspace(0, 2, n_points), h_binary)
        h_up = h_up * (target_sum / np.sum(h_up))
        
        h_up_cont = h_up.copy()
        for t in range(200):  # Few iterations
            if (time.time() - start_time) > 0.99 * budget_s:
                break
            corr = correlate(h_up_cont, 1 - h_up_cont, mode="full")
            max_corr = np.max(corr)
            max_positions = np.where(np.abs(corr - max_corr) < 1e-10)[0]
            active_shifts = []
            for pos in max_positions:
                if pos <= n_up - 1:
                    active_shifts.append(pos)
                else:
                    active_shifts.append(pos - n_up)

            grad = compute_gradient(h_up_cont, active_shifts)
            eta = 0.25 * dx_up
            h_new = h_up_cont - eta * grad
            h_new = project_to_simplex(h_new, target_sum)
            
            h_sym = (h_new + h_new[::-1]) / 2.0
            h_sym = project_to_simplex(h_sym, target_sum)
            
            c5_new = evaluate_c5(h_sym)
            h_up_cont = h_sym
            if c5_new < c5_best - 1e-12:
                c5_best = c5_new
                h_binary = h_sym.copy()

    return (h_binary.tolist(), c5_best, n_points)
