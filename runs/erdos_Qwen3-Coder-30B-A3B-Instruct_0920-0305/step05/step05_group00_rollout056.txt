import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h):
    """Compute gradient for each position"""
    grad = np.zeros_like(h)
    for i in range(len(h)):
        if i > 0:
            grad[i] += h[i-1]
        if i < len(h) - 1:
            grad[i] += h[i+1]
    return grad

def guided_move(h, target_lag=None):
    """Make a guided move based on gradient"""
    if target_lag is None:
        # Find worst lag
        corr = correlate(h, 1 - h, mode='full')
        target_lag = np.argmax(corr)
    
    # Compute local gradient to determine where to transfer mass
    grad = compute_gradient(h)
    
    # Find indices of max and min gradient in range
    # We want to move from min to max ones to reduce the maximum correlation
    i_min = np.argmin(grad)
    i_max = np.argmax(grad)
    
    if i_min == i_max:
        return None
    
    # Determine how much we can move
    if target_lag > 0:  # Moving in positive direction
        if i_min + target_lag >= len(h):
            return None
        delta = min(h[i_min], 1 - h[i_min + target_lag])
    else:  # Moving in negative direction
        if i_min - target_lag < 0:
            return None
        delta = min(h[i_min], 1 - h[i_min - target_lag])
    
    if delta <= 0:
        return None
    
    return (i_min, i_min + target_lag if target_lag > 0 else i_min - target_lag, delta)
