# Constants
TARGET_SUM = None

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def smooth_max_objective(h, beta):
    """Computes smooth approximation to max of correlations"""
    corr = compute_corr_fft(h)
    corr = corr * dx  # Scale by dx
    max_corr = np.max(corr)
    exp_corr = np.exp(beta * (corr - max_corr))
    w = exp_corr / np.sum(exp_corr)
    return np.sum(w * corr)

def smooth_max_gradient(h, beta):
    """Gradient of smooth max objective"""
    corr = compute_corr_fft(h)
    corr = corr * dx
    max_corr = np.max(corr)
    exp_corr = np.exp(beta * (corr - max_corr))
    w = exp_corr / np.sum(exp_corr)
    
    # Compute ∇C_ℓ = 1 - shift_plus - shift_minus for each lag
    grad_sum = np.zeros_like(h)
    
    # Do it efficiently using FFT:
    # For each ℓ, ∇C_ℓ = 1 - h_{i+ℓ} - h_{i-ℓ} = 1 - shift_right(h,ℓ) - shift_left(h,ℓ)
    for i in range(1, len(h)):
        # shift_right: h_{i+ℓ}
        shifted_plus = np.roll(h, i, axis=0)
        # shift_left: h_{i-ℓ}
        shifted_minus = np.roll(h, -i, axis=0)
        grad = 1.0 - shifted_plus - shifted_minus
        grad_sum += w[i] * grad
    
    # Add constant vector (1) for third term
    grad_sum += w[0]  # At lag 0, shifting 0 gives the same vector
    
    return grad_sum

def project_to_box_and_sum(x, target_sum):
    """Project x onto box [0,1] and sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def projected_gradient_descent_smooth_max(h0, beta_schedule, step_size_schedule, max_iter=2000):
    """Run projected gradient descent on smooth max objective"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = np.max(compute_corr_fft(h)) * dx
    
    for k, (beta, alpha) in enumerate(zip(beta_schedule, step_size_schedule)):
        if k >= max_iter:
            break
            
        grad = smooth_max_gradient(h, beta)
        h_new = h - alpha * grad
        
        # Project back
        h_new = project_to_box_and_sum(h_new, TARGET_SUM)
        
        # Check for improvement
        c5_candidate = np.max(compute_corr_fft(h_new)) * dx
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_new.copy()
            
        h = h_new.copy()
    
    return best_h, best_c5
