import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def project_to_simplex(h, target_sum):
    """Projects h onto the bounded simplex with given target sum using Duchi's method."""
    h = np.asarray(h)
    n = len(h)
    theta = np.sort(h)[::-1]
    v = (np.cumsum(theta) - target_sum) / np.arange(1, n + 1)
    rho = np.argmax(theta > v)
    if rho == 0 and theta[0] <= v[0]:
        # All entries are 0
        return np.zeros_like(h)
    else:
        lambda_ = v[rho - 1] if rho > 0 else v[0]
        return np.clip(h - lambda_, 0, 1)

def compute_correlations(h):
    """Compute correlation vector c_k = sum h_i h_{i+k} for k=1..n-1."""
    n = len(h)
    corr = correlate(h, h, mode="full")
    return corr[n//2 + 1:]

def compute_soft_min_gradient(h, c, beta):
    """Compute gradient of soft-min surrogate using FFT-based convolution."""
    n = len(h)
    # weights: softmax of -beta * c
    c_max = np.max(c)
    exps = np.exp(-beta * (c - c_max))
    w = exps / np.sum(exps)
    w_rev = w[::-1]
    # Gradient via convolution:
    # g_i = sum_k w_k * (h_{i-k} + h_{i+k})
    # Zero-pad for linear convolution
    h_pad = np.pad(h, (n-1, n-1), mode='constant')
    w_pad = np.pad(w_rev, (n-1, n-1), mode='constant')
    conv_backward_1 = np.convolve(h_pad, w_pad, mode='valid')
    conv_backward = conv_backward_1[n-1:-n+1]
    # Now forward
    w_pad = np.pad(w, (n-1, n-1), mode='constant')
    conv_forward_1 = np.convolve(h_pad, w_pad, mode='valid')
    conv_forward = conv_forward_1[n-1:-n+1]
    # Final gradient (unsigned)
    g = -(conv_forward + conv_backward)
    return g

def evaluate_c5_exact(h):
    """Evaluate the true C5 constant with exact max."""
    n = len(h)
    dx = 2.0 / n
    corr_f = correlate(h, 1-h, mode="full")
    # Convert from full correlation to shiftless version
    max_corr = np.max(corr_f[n//2:]) 
    return max_corr * dx
