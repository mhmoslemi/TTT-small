import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def project_to_simplex(h, target_sum):
    """Project h to the bounded simplex: 0<=h<=1, sum(h)=target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    
    # Use the algorithm from Duchi et al. (2008)
    # Project onto box [0,1] first, then onto the sum constraint
    
    # Sort the values in descending order, exclude zeros, and find the projection
    h_sorted = np.sort(h)[::-1]
    
    # Find the threshold where the projection occurs
    tmpsum = 0.0
    for i in range(len(h_sorted)):
        tmpsum += h_sorted[i]
        if tmpsum > target_sum:
            break
    
    # This is the point where we need to adjust the threshold
    theta = (tmpsum - target_sum) / i if i > 0 else 0
    h_proj = np.maximum(h - theta, 0)
    h_proj = h_proj * (target_sum / np.sum(h_proj))
    return h_proj

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # Only consider non-zero shifts
    max_corr = np.max(corr[n//2+1:])  # Exclude the shift 0 and negative shifts
    return max_corr * dx

def compute_correlations(h):
    """Compute correlations using FFT for efficiency"""
    n = len(h)
    h_padded = np.pad(h, (0, n-1), 'constant')
    h_conj = np.pad(h[::-1], (0, n-1), 'constant')
    corr_fft = np.fft.irfft(np.fft.rfft(h_padded) * np.fft.rfft(h_conj))
    corr = corr_fft[:n]  # Take only the relevant part
    return corr

def compute_gradient(h, correlations, beta):
    """Compute gradient of the soft minimum with respect to h"""
    n = len(h)
    c_max = np.max(correlations[1:])
    exp_terms = np.exp(-beta * (correlations[1:] - c_max))
    w = exp_terms / np.sum(exp_terms)  # weights
    w_padded = np.pad(w, (0, n-1), 'constant')
    # Compute convolution:
    # Forward shift terms (h[i] * h[i+s])
    conv_forward = np.fft.irfft(np.fft.rfft(h) * np.fft.rfft(w_padded))[:n]
    # Backward shift terms (h[i] * h[i-s])
    w_reverse = w[::-1]
    w_reverse_padded = np.pad(w_reverse, (0, n-1), 'constant')
    conv_backward = np.fft.irfft(np.fft.rfft(h) * np.fft.rfft(w_reverse_padded))[:n]
    grad = -(conv_forward + conv_backward)
    return grad

def line_search(h, grad, step_size, beta, target_sum):
    """Backtracking line search"""
    alpha = 1.0  # initial step size
    tau = 0.5  # step size reduction factor
    epsilon = 1e-12  # small threshold
    initial_obj = 1.0 + (1.0 / beta) * np.log(np.sum(np.exp(-beta * compute_correlations(h)[1:])))
    
    while alpha > epsilon:
        h_new = project_to_simplex(h - alpha * grad, target_sum)
        new_correlations = compute_correlations(h_new)
        new_obj = 1.0 + (1.0 / beta) * np.log(np.sum(np.exp(-beta * new_correlations[1:])))
        if new_obj <= initial_obj:  # Armijo condition
            return alpha, h_new
        alpha *= tau
    return alpha, h

            # If line search fails, try a small perturbation

        # Update beta (continuation)
