import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = None

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def evaluate_c5(h):
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_grad_softmax(h, beta):
    """Compute gradient of log-sum-exp of correlations"""
    corr = compute_corr_fft(h)
    corr = corr[:len(h)]  # Only consider first n elements
    
    # Numerical stability: subtract max
    max_corr = np.max(corr)
    exp_corr = np.exp(beta * (corr - max_corr))
    sum_exp = np.sum(exp_corr)
    weights = exp_corr / sum_exp
    
    # Compute weighted gradient
    grad = np.zeros_like(h)
    for i in range(len(h)):
        # Gradient for lag i
        lag = i
        if lag > 0:
            # We efficiently handle the shifts using vectorized operations
            shift_plus = np.roll(h, -lag)
            shift_minus = np.roll(h, lag)
            shift_plus[0:lag] = 0
            shift_minus[-lag:] = 0
            grad += weights[i] * (1 - shift_plus - shift_minus)
        else:
            grad += weights[i] * (1 - h - np.roll(h, lag))
    
    # Handle all lags efficiently using matrix operations
    possible_lags = range(len(h))
    shifted_h = np.zeros((len(h), len(h)))
    for lag in possible_lags:
        shifted_h[:, lag] = np.roll(h, -lag)
        if lag > 0:
            shifted_h[:lag, lag] = 0
        else:
            shifted_h[lag:, lag] = 0
    grad_full = np.sum(weights.reshape(-1, 1) * (1 - shifted_h - shifted_h.T), axis=0)
    
    return grad_full

def smooth_objective(h, beta):
    """Compute the log-sum-exp of correlations"""
    corr = compute_corr_fft(h)
    corr = corr[:len(h)]
    max_corr = np.max(corr)
    log_sum_exp = np.log(np.sum(np.exp(beta * (corr - max_corr)))) + max_corr
    return (1 / beta) * log_sum_exp

def smoothed_gradient_descent(h0, max_iter=5000, beta_max=30, alpha0=0.4, gamma=1e-4, beta_schedule=2):
    """Perform gradient descent with logarithmically smoothed max"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    beta = 0.5
    t = 0
    
    while beta < beta_max and t < max_iter:
        # Do a bunch of iterations at current beta
        iter_per_beta = 400
        for _ in range(iter_per_beta):
            if t >= max_iter:
                break
            
            grad = compute_grad_softmax(h, beta)
            
            # Adaptive step size
            alpha = alpha0 / (1 + gamma * t)
            if alpha < 1e-8:
                break
            
            h_candidate = h - alpha * grad
            h = project_to_sum_box(h_candidate, TARGET_SUM)
            
            # Check if it's better
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
            
            # Line search if it doesn't improve
            for _ in range(20):
                if t + 1 >= max_iter:
                    break
                
                h_candidate = h - alpha * grad
                h_test = project_to_sum_box(h_candidate, TARGET_SUM)
                c5_test = evaluate_c5(h_test)
                if c5_test < c5:
                    h = h_test
                    c5 = c5_test
                    if c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = c5
                    break
                else:
                    alpha *= 0.5
            
            t += 1
            if t >= max_iter:
                break
        
        # Increase beta for next phase
        beta *= beta_schedule
            
    return best_h, best_c5

def subgradient_refinement(h, max_iters=200):
    """Refine with exact max using subgradients"""
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iters):
        corr = compute_corr_fft(h)
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        
        # Gradient of correlation at max lag
        grad = np.zeros_like(h)
        shift_plus = np.roll(h, -max_idx)
        shift_minus = np.roll(h, max_idx)
        shift_plus[0:max_idx] = 0
        shift_minus[-max_idx:] = 0
        grad = 1 - shift_plus - shift_minus
        
        # Very small step size for subgradient descent
        alpha = 1e-3 / np.sqrt(i + 1)
        h_candidate = h - alpha * grad
        h = project_to_sum_box(h_candidate, TARGET_SUM)
        
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()

    # Create initial solutions
    initial_solutions = []
    
    # Interpolate from initial_h_values
    try:
        from initial_h_values import initial_h_values
        h0 = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h0))
        x_new = np.linspace(0, 2, n_points)
        h_interpolated = np.interp(x_new, x_orig, h0)
        h_interpolated = project_to_sum_box(h_interpolated, n_points/2)
        initial_solutions.append(h_interpolated.copy())
    except:
        # Fallback if failed
        h_uniform = np.ones(n_points) * 0.5
        h_uniform = project_to_sum_box(h_uniform, n_points/2)
        initial_solutions.append(h_uniform)
    
    # Random binary pattern with correct sum
    h_bin = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
    h_bin = project_to_sum_box(h_bin, n_points/2)
    initial_solutions.append(h_bin)
    
    # Uniform random with correct sum
    h_rand = np.random.uniform(0, 1, n_points)
    h_rand = project_to_sum_box(h_rand, n_points/2)
    initial_solutions.append(h_rand)
    
    TARGET_SUM = n_points / 2
    best_h = None
    best_c5 = np.inf
    
    # Phase A - Global optimization with smooth max
    time_phase_a = 0.7 * budget_s
    phase_a_end = start_time + time_phase_a
    
    for i, h_start in enumerate(initial_solutions):
        if time.time() >= phase_a_end:
            break
        h_candidate, c5_candidate = smoothed_gradient_descent(h_start.copy(), max_iter=2000, beta_max=30)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
    
    # Phase B - Coarse-to-fine optimization
    time_phase_b = 0.2 * budget_s
    phase_b_end = start_time + time_phase_a + time_phase_b
    
    # Increase resolution
    n_fine = 2 * n_points
    dx_fine = 2.0 / n_fine
    h_fine = np.interp(np.linspace(0, 2, n_fine), np.linspace(0, 2, n_points), best_h)
    h_fine = project_to_sum_box(h_fine, n_fine/2)
    
    for i in range(20):
        if time.time() >= phase_b_end:
            break
        h_candidate, c5_candidate = smoothed_gradient_descent(h_fine.copy(), max_iter=2000, beta_max=50, alpha0=0.2)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
            h_fine = h_candidate.copy()
    
    # Phase C - Final polishing
    time_phase_c = 0.1 * budget_s
    phase_c_end = start_time + budget_s
    
    # Subgradient refinement
    h_final, c5_final = subgradient_refinement(best_h, max_iters=100)
    if c5_final < best_c5:
        best_h = h_final.copy()
        best_c5 = c5_final
    
    # Add jitter and re-refine if time allows
    if time.time() < phase_c_end - 5:
        h_jitter = best_h + np.random.normal(0, 0.01, n_points)
        h_jitter = project_to_sum_box(h_jitter, TARGET_SUM)
        h_pure, c5_pure = subgradient_refinement(h_jitter, max_iters=50)
        if c5_pure < best_c5:
            best_h = h_pure.copy()
            best_c5 = c5_pure
    
    return (best_h.tolist(), float(best_c5), n_points)
