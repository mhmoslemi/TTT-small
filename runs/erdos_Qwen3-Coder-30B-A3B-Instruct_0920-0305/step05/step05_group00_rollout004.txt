def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_search_hill_climb(h0, max_iter=1000):
    """Hill climb local search with projected updates"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for _ in range(max_iter):
        # Try small random perturbations
        idx = np.random.randint(0, len(h))
        delta = np.random.normal(0, 0.01)
        
        # Ensure we don't exceed bounds
        if h[idx] + delta >= 0 and h[idx] + delta <= 1:
            h[idx] += delta
            h = project_to_sum_box(h, TARGET_SUM)

            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            else:
                # Revert
                h[idx] -= delta
                h = project_to_sum_box(h, TARGET_SUM)
    
    return best_h

def local_search_perturb(h, prob, max_iter):
    """Locally improve solution by random perturbations"""
    h = h.copy()
    
    for _ in range(max_iter):
        if np.random.random() < prob:
            idx1, idx2 = np.random.randint(0, len(h), 2)
            delta = np.random.uniform(-0.05, 0.05)
            
            if (0 <= idx1 < len(h) and 0 <= idx2 < len(h) and

                h[idx1] += delta
                h[idx2] -= delta
                h = project_to_sum_box(h, TARGET_SUM)
    
    return h

def perturb_solution(h, magnitude):

    h_perturbed = h + np.random.normal(0, magnitude, len(h))
    h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
