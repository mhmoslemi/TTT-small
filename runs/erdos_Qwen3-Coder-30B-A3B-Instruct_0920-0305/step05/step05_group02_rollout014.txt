    def greedy_construction(n):
        # Greedy algorithm for initial binary pattern
        x = np.zeros(n, dtype=int)
        ones = set()
        
        # Simple greedy to place ones to keep the minimum correlation high
        # Start with simple 1/2 periodic pattern
        count = 0
        while count < n // 2:
            # Choose based on adding the position that seems to help the worst lag
            # For simplicity, just place in a staggered way
            i = count * 2
            if i < n:
                x[i] = 1
                ones.add(i)
                count += 1
            else:
                # Fill gaps
                for j in range(n):
                    if j not in ones:
                        x[j] = 1
                        ones.add(j)
                        count += 1
                        break
        
        return x
    
    # Function to compute pair counts for given x
    def compute_pair_counts(x, n):
        pair_counts = []
        for k in range(1, n):
            count = 0
            for i in range(n - k):
                count += x[i] * x[i + k]
            pair_counts.append(count)
        return pair_counts

            break

            x_start = greedy_construction(n)

                ones_idx = np.where(x_start == 1)[0]
                zeros_idx = np.where(x_start == 0)[0]
                needed = n // 2 - len(ones_idx)
                # Flip some zeros to ones randomly
                if needed > 0:
                    flip_idx = np.random.choice(zeros_idx, size=needed, replace=False)
                    x_start[flip_idx] = 1

                best_n = n

                # Use numpy correlation for evaluation
                corr = correlate(h, 1 - h, mode="full")
                return -np.max(corr) * dx_n  # Minimize negative for maximization

            res = minimize(objective, x_refined, method='L-BFGSB', 
                           bounds=[(0, 1) for _ in range(n)], 
                           options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12})
            if res.success:

                if norm != 0:
                    h_result = h_result * (n / 2) / norm
                corr = correlate(h_result, 1 - h_result, mode="full")
                c5_new = -np.max(corr) * dx_n
                
                if c5_new < best_c5:
                    best_h = h_result
                    best_c5 = c5_new
                    best_n = n
            # Additional handling if within last few seconds
            if (time.time() - start_time) > budget_s * 0.99:
                break
