import numpy as np

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def greedy_initialization(n):
    """Greedy initialization to get a good binary seed"""
    x = np.zeros(n, dtype=int)
    ones_positions = set()
    
    while len(ones_positions) < n // 2:
        best_i = -1
        best_min_pk = -1
        
        for i in range(n):
            if i in ones_positions:
                continue
                
            # Simulate adding i to ones_positions
            new_positions = ones_positions | {i}
            
            # Compute the new minimum p_k for all k
            min_pk = float('inf')
            for k in range(1, n):
                pk = 0
                for j in range(n - k):
                    if (j in new_positions) and (j + k in new_positions):
                        pk += 1
                min_pk = min(min_pk, pk)
                
            if min_pk > best_min_pk:
                best_min_pk = min_pk
                best_i = i
                
        if best_i != -1:
            ones_positions.add(best_i)
            x[best_i] = 1
            
    return x

def solve_milp(n, time_limit=30):
    """Solve the MILP formulation for the given n"""
    dx = 2.0 / n
    
    # Create binary variables for x_i
    x = cp.Variable(n, boolean=True)
    
    # Add symmetry breaking constraints
    x[0].value = 1
    
    # Helper to make constraints
    constraints = [cp.sum(x) == n // 2]
    
    # Compute the minimum p_k for each k
    L = cp.Variable(1, integer=True)
    
    y = {}
    p = {}
    
    # Create auxiliary variables and constraints for y_{i,k}
    for k in range(1, n):
        for i in range(n - k):
            y[i, k] = cp.Variable(boolean=True)
            constraints.append(y[i, k] <= x[i])
            constraints.append(y[i, k] <= x[i + k])
            constraints.append(y[i, k] >= x[i] + x[i + k] - 1)
    
    # Define p_k values
    for k in range(1, n):
        p[k] = cp.sum([y[i, k] for i in range(n - k)])
        constraints.append(p[k] >= L)
    
    # Objective: maximize L
    objective = cp.Maximize(L)
    
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CBC, time_limit=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            # Recover solution
            x_vals = [int(round(x[j].value)) for j in range(n)]
            # Check the L value
            min_pk = float('inf')
            for k in range(1, n):
                # Count p_k for current solution
                pk = sum(x_vals[i] * x_vals[i + k] for i in range(n - k))
                min_pk = min(min_pk, pk)
            
            return x_vals, min_pk
    except Exception as e:
        pass
        
    return None, -1

def polish_continuous(h, n_points):
    """Polish the discrete solution to continuous values"""
    # Convert to continuous version
    h = np.array(h, dtype=float)
    scaling = (n_points / 2) / np.sum(h)
    h = h * scaling
    
    def objective(h):
        # Compute max of all correlations
        h_padded = np.pad(h, (0, n_points), mode='constant')
        corrs = []
        for k in range(1, 2*n_points):
            corr = np.sum(h * h_padded[k:k+n_points])
            corrs.append(1 - corr)  # This is 1 - R(k)
        return max(corrs)
        
    def gradient(h):
        # Simplified gradient estimation
        grad = np.zeros_like(h)
        for i in range(n_points):
            for k in range(1, n_points):
                if i + k < n_points:
                    grad[i] -= (1 - h[i+k])
                if i >= k:
                    grad[i] -= (1 - h[i-k])
        return grad
        
    # Use projected gradient descent
    old_c5 = objective(h)
    for _ in range(200):
        grad = gradient(h)
        h = h - 0.05 * grad
        h = np.clip(h, 0, 1)
        # Renormalize
        scaling = (n_points / 2) / np.sum(h)
        h = h * scaling
        new_c5 = objective(h)
        if abs(new_c5 - old_c5) < 1e-5:
            break
        old_c5 = new_c5
        
    return h
