import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

def project_to_box_sum(h, target_sum):
    """Project vector h onto box [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(h + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def compute_correlations(h):
    """Compute correlations for all lags using FFT for efficiency"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def compute_smooth_objective(h, beta=1.0):
    """Compute the smoothed log-sum-exp objective"""
    corr = compute_correlations(h)
    # Shift to handle zero padding correctly
    corr = corr[:len(h)]  # Take only valid correlations
    max_corr = np.max(corr)
    if max_corr == -np.inf:
        return 0.0
    exp_corr = np.exp(beta * (corr - max_corr))
    sum_exp = np.sum(exp_corr)
    if sum_exp == 0.0:
        return 0.0
    return (1.0 / beta) * (np.log(sum_exp) + max_corr)

def compute_smooth_gradient(h, beta=1.0):
    """Compute gradient of smoothed objective"""
    corr = compute_correlations(h)
    corr = corr[:len(h)]  # Valid correlations
    max_corr = np.max(corr)
    exp_corr = np.exp(beta * (corr - max_corr))
    sum_exp = np.sum(exp_corr)
    if sum_exp == 0.0:
        return np.zeros_like(h)
    
    # Compute weights
    weights = exp_corr / sum_exp
    
    # Gradient computation: ∇C_ℓ = 1 - h_{i+ℓ} - h_{i-ℓ}
    grad = np.zeros_like(h)
    n = len(h)
    
    for i in range(n):
        # Right shift contribution
        if i + 1 < n:
            grad[i] += weights[i+1] * (-h[i+1])
        # Left shift contribution
        if i - 1 >= 0:
            grad[i] += weights[i-1] * (-h[i-1])
    
    # Constant term
    grad += weights @ np.ones_like(h)
    
    return grad

def compute_c5_value(h):
    """Return C5 using the actual max"""
    corr = compute_correlations(h)
    return np.max(corr)

def gradient_step(h, grad, alpha):
    """Perform gradient step on h with given alpha"""
    h_new = h - alpha * grad
    return project_to_box_sum(h_new, len(h) / 2)

def optimize_smoothed_objective(h0, beta_schedule, alpha0=0.4, gamma=1e-4, max_iter=2000):
    """Optimize the smoothed objective using projected gradient descent"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5_value(h)
    t = 0
    
    for beta in beta_schedule:
        beta_iters = 0
        while beta_iters < max_iter:
            if beta_iters % 100 == 0:
                c5 = compute_c5_value(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            
            grad = compute_smooth_gradient(h, beta)
            alpha = alpha0 / (1 + gamma * t)
            h_new = gradient_step(h, grad, alpha)
            
            # Check for convergence by monitoring improvement
            if np.allclose(h_new, h, rtol=1e-10):
                break
            
            # Use backtracking line search if improvement not substantial
            c5_new = compute_c5_value(h_new)
            if c5_new >= compute_c5_value(h):
                # Backtrack
                factor = 0.5
                alpha *= factor
                while alpha > 1e-8:
                    h_test = gradient_step(h, grad, alpha)
                    c5_test = compute_c5_value(h_test)
                    if c5_test < c5_new:
                        h_new = h_test
                        c5_new = c5_test
                        break
                    alpha *= factor
            
            h = h_new
            t += 1
            beta_iters += 1
            
            # Early stopping if loop is stagnating
            if t > 5000:  # Arbitrary cutoff
                break
    
    return best_h, best_c5
