import numpy as np
import time

def project_capped_simplex(v, T):
    """Project v onto the capped simplex: 0 ≤ x ≤ 1, sum(x) = T."""
    v = np.asarray(v)
    low, high = 0.0, 1.0
    v_min, v_max = v.min(), v.max()
    if T <= v_min:
        return np.full_like(v, low)
    if T >= v_max:
        return np.full_like(v, high)
    
    # Binary search on lambda
    def sum_proj(lambda_v):
        return np.clip(v - lambda_v, low, high).sum()
    
    a, b = v_min - 1.0, v_max + 1.0
    for _ in range(40):
        mid = (a + b) / 2.0
        s = sum_proj(mid)
        if s > T:
            a = mid
        else:
            b = mid
    lambda_v = (a + b) / 2.0
    return np.clip(v - lambda_v, low, high)

def compute_O_and_grad(h, alpha):
    """Returns (O, grad) where O_k = dx * sum h[i] * (1 - h[i+k])"""
    n = len(h)
    dx = 2.0 / n
    # Correlations: for each k, we compute O_k = dx * sum_{i=0}^{n-1} h[i] * (1 - h[i+k])
    # This is the same as dx * (sum h[i] - sum h[i]*h[i+k])
    # Inputs: h (n,) array in [0,1]
    # Output:
    # O: vector of length 2*n-1, O_k = dx * sum_{i=0}^{n-1} h[i] * (1 - h[i+k]) 
    # grad: gradient of F_alpha = (1/alpha) * log(sum_k exp(alpha * O_k))
    
    # First compute the 'full' correlation vector O using np.correlate. 
    # This should match the direct definition of O_k.
    O = dx * np.correlate(h, 1-h, mode='full')
    # O is of length 2*n-1; O[:n] is O_k for k=-n+1..0 (negative shifts)
    # O[n:] is O_k for k=1..n-1 (positive shifts)
    # The max is over all shifts from -n+1 to n-1
    
    # We want gradient with respect to h
    # dO_k / dh_i = dx * (1 - h[i+k]) - h[i-k] where out of bounds h is 0
    # So ∂O_k/∂h_i = dx * (1 - h[i+k]) - h[i-k] 
    # We will compute this as a matrix-vector multiply using sparse-like patterns
    
    # Compute sum over k of exp(alpha * O_k) * dO_k/dh_i (will use softmax)
    # O should have length 2*n-1
    assert len(O) == 2*n - 1
    # Apply softmax to alpha*O with numerical stable version:
    m = O.max()
    exp_alphaO = np.exp(alpha * (O - m))
    weights = exp_alphaO / exp_alphaO.sum()  # shape: 2*n-1
    # Build the gradient: 
    # grad_i = dx * sum_k weights_k * ( (1-h[i+k]) - h[i-k] )
    # In vector form, use shifts (very slightly slower than loops, but ok)
    # The code below handles the 2D array processing without explicit loops
    # Using the convolution logic: a more efficient vector version:
    
    # For each i, sum over k of weights_k * (1 - h[i+k]) - h[i-k]
    # We want to build convolutions of h1 = 1 - h, h2 = h against weights.
    # Let's extract the parts:
    # indexer = [k for k in range(-n+1, n)] => indices 0..2n-2
    # Let's compute this explicitly for better control
    
    # Start with constructing the shifted parts
    grad = np.zeros_like(h)
    # Do this more efficiently, computing: 
    # For all valid indices (that do not go out of bounds):
    # we add to grad[i]: weights_k * ( (1 - h[i + k]) - h[i - k] )
    # We must loop over k from -n+1 to n-1, but shortcuts:
    # small computation: 
    # Let's compute full correlation with h and 1-h with mode='full' and just slice:
    #    corr = np.correlate(h, 1 - h, mode='full') looks correct here
    # But we want multipliers dependent on k: weight_k * approach
    # Full setup: 
    # We want:
    # dO_k/dh_i = dx * ( (1 - h[i+k]) - h[i-k] ) 
    # So pour over all valid k and i  
    # Most efficient: go through all k and construct the two shifts:
    # Let's neglect loop overhead, build h_shifted_pos_k = [h[i+k] for i=0..n-1]
    # This is effectively done by circular shift using np.roll etc.
    
    # Use a sparse-style shift of h to extract all relevant pieces:
    # F = np.zeros(n)
    # For each k, we compute: 
    # (1 - h[i+k]) - h[i-k] for valid i 
    # Keep in matrix form - methodically collect contributions
    # We build vector of biases and reusable parts:
    
    # Build: [1 - h_k] for k = 0..n-1 as a vector (this is 1-h)
    ones_minus_h = 1 - h
    
    # For each k, the contribution to the gradient is:
    #   for i in range(n): 
    #     权重_k * (1 - h[i+k]) - h[i-k]   # index out-of-bounds are 0
    # We compute all terms: on shift by k: h→h_shifted_pos_k  
    # sliding window across all overlapping terms.
    # Debug formula would be full correlation using vstack/conv:
    # We just do for each k separately, which is a bit slower, not quite optimal,  
    # but we break it into two parts due to trickiness with non-zero boundary.
    # Essentially, for each k, we compute:
    # g = weights[k] * h_shifted_pos_k - weights[n-1+k] * h_shifted_neg_k
    # Let's define multiple helper operations:
    # Initialize result vector
    grad = np.zeros_like(h)
    # loop over all k from -n+1 to n-1:
    for k in range(-n+1, n):
        # k is index in full correlation array
        # Iterate over valid i such that h[i+k], h[i-k] are valid indices
        # The function will reflect how we’d index the full correlation vector
        idx_k = k + n - 1  # maps k in [-n+1, n-1] to [0,2n-2]
        if idx_k < 0 or idx_k >= 2 * n - 1:
            continue
        weight_k = weights[idx_k]
        
        # full k-indexed version:
        # h[i+k] when i goes from 0 to n-1, line = h[i+k] for i=0..n-1
        # h[i-k] when i goes from 0 to n-1: line = h[i-k] 
        # use proper indexing with np.roll
        shift_pos = k
        shift_neg = -k
        if k >= 0:
            shifted1 = np.roll(ones_minus_h, shift_pos)
            shifted2 = np.roll(h, shift_neg)  
        else:
            shifted1 = np.roll(ones_minus_h, shift_pos)
            shifted2 = np.roll(h, shift_neg)  
        # Add to gradient (scaled by weight):  
        # (1 - h[i+k]) - h[i-k] = shifted1[i] - shifted2[i]
        # Not the full off-center due to shift... edge effects matter
        # For np.roll with shift_pos, we need to compute bit carefully for exact indexing.
        # Better approach:
        if k == 0:
            # simply: w0 * (1 - h[i])
            grad += weight_k * ones_minus_h
        else:
            shifted1 = np.roll(ones_minus_h, -k)
            shifted2 = np.roll(h, k)  
            grad += weight_k * (shifted1 - shifted2)  
    grad *= dx
    return O, grad

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_val = np.inf
    best_h = None
    n = 64
    dx = 2.0 / n
    T = n / 2.0  # desired sum
    
    def evaluate_c5(h):
        # Use the actual max of the correlation for estimate  
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    max_restarts = 5000
    for use_seed in range(max_restarts):
        # Early stopping
        if time.time() - start_time > 0.9 * budget_s:
            break

        # choose seed type
        seed_type = use_seed % 5
        if seed_type == 0:
            # uniform random
            h = np.random.rand(n)
        elif seed_type == 1:
            # sine
            x = np.linspace(0, 2, n)
            h = 0.5 + 0.4 * np.sin(np.pi * x)
        elif seed_type == 2:
            # triangular
            x = np.linspace(0, 2, n)
            h = 1 - np.abs(x - 1) * 2
        elif seed_type == 3:
            # Gaussian bump centered
            x = np.linspace(0, 2, n)
            sigma = 0.2
            h = np.exp(-0.5 * ((x - 1.0) / sigma) ** 2)
        else:
            # binary pattern
            h = np.zeros(n)
            mid = n // 2
            width = n // 4
            h[max(0, mid - width // 2):min(n, mid + width // 2)] = 1.0  
        
        h = project_capped_simplex(h, T)
        # Now run optimizer
        alpha_start = 5
        alpha_end = 30
        lr_0 = 0.5
        max_iter = 4000
        subgrad_iters = 500
        c5_record = np.inf
        h_best_now = h.copy()
        for i in range(max_iter):
            if time.time() - start_time > 0.9 * budget_s:
                break
                
            alpha = alpha_start + (alpha_end - alpha_start) * i / max_iter
            O, grad = compute_O_and_grad(h, alpha)
            # gradient step
            lr = lr_0 / np.sqrt(i + 1)
            h = h - lr * grad
            
            # project back
            h = project_capped_simplex(h, T)
            # optional symmetric averaging?
            if i % 10 == 0:
                h = (h + h[::-1]) / 2.0
                h = project_capped_simplex(h, T)
            
            c5_now = np.max(O) * dx
            if c5_now < c5_record:
                c5_record = c5_now
                h_best_now = h.copy()
                
        # Sub-gradient polishing after smooth phase
        final_O = np.correlate(h_best_now, 1-h_best_now, mode='full') * dx
        active_k = np.argmax(final_O)
        # Get gradients of the active shifts:
        # For k = active_k, compute ∂O_k/∂h
        activ_shift = active_k - n + 1  # from 0..2n-2 to -n+1..n-1
        # Compute exact gradient of that specific O_k:
        # ∂O_k/∂h_i = dx * ( 1 - h[i+k] - h[i-k] )
        # using indices
        exact_grad = np.zeros_like(h_best_now)
        if activ_shift >= 0:
            shift_pos = activ_shift
            shift_neg = -activ_shift
            h_shifted_pos = np.roll(h_best_now, -shift_pos)
            h_shifted_neg = np.roll(h_best_now, shift_neg)
        else:
            shift_pos = -activ_shift
            shift_neg = activ_shift
            h_shifted_pos = np.roll(h_best_now, shift_pos)
            h_shifted_neg = np.roll(h_best_now, -shift_neg)
        # For just one shift, the gradient decision is simplified (but not used in formulae above)
        # But not straightforward recompute precise gradient without recomputing full shifted sums
        # So manually evaluate gradient of active shift exact as:
        # ∂O_k/∂h_i = dx * (1 - h[i+k] - h[i-k])
        exact_grad = dx * (1 - h_shifted_pos - h_shifted_neg)
        # it's a 1-step optimization:
        lr_final = 0.02 / np.sqrt(subgrad_iters)
        for i in range(subgrad_iters):
            h_best_now = h_best_now - lr_final * exact_grad
            h_best_now = project_capped_simplex(h_best_now, T)
            # optionally symmetrize?
            if i % 10 == 0:
                h_best_now = (h_best_now + h_best_now[::-1]) / 2.0
                h_best_now = project_capped_simplex(h_best_now, T)
