import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Simple projection using clipping and rescaling
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x
    
    # Just adjust values to make sum correct
    if current_sum > target_sum:
        # Need to decrease some values
        sort_idx = np.argsort(x)[::-1]  # Descending order
        remaining = target_sum
        for i in sort_idx:
            if remaining <= 0:
                x[i] = 0
            else:
                if x[i] <= remaining:
                    remaining -= x[i]
                else:
                    x[i] = remaining
                    remaining = 0
    else:
        # Need to increase some values
        sort_idx = np.argsort(x)  # Ascending order
        remaining = target_sum - current_sum
        for i in sort_idx:
            if remaining <= 0:
                break
            max_incr = 1 - x[i]
            incr = min(max_incr, remaining)
            x[i] += incr
            remaining -= incr
    
    return x

    def objective(x):
        return evaluate_c5(x)
    
    def constraint_eq(x):
        return np.sum(x) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        # Try with different starting points
        for _ in range(3):
            perturbed = h + np.random.normal(0, 0.01, n_points)
            perturbed = project_to_sum_box(perturbed, TARGET_SUM)
            
            res = minimize(
                objective, 
                perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10}
            )
            if res.success:
                opt_h = project_to_sum_box(res.x, TARGET_SUM)
                opt_c5 = evaluate_c5(opt_h)
                if opt_c5 < evaluate_c5(h):
                    h = opt_h
    except:
        pass
    
    return h
