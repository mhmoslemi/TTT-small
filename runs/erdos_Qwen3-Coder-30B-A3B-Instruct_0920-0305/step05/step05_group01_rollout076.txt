import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def random_pair_move(h):
    """Generate a pairwise mass transfer move, ensuring feasibility"""
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    delta_max = min(h[i], 1.0 - h[j])
    delta_min = max(-h[j], h[i] - 1.0)
    if delta_min >= delta_max:
        return i, j, 0.0
    delta = np.random.uniform(delta_min, delta_max)
    return i, j, delta

def sample_moves(h, selection_method='random'):
    """Generate a batch of moves using different strategies"""
    moves = []
    n = len(h)
    if selection_method == 'random':
        for _ in range(50):
            i, j, delta = random_pair_move(h)
            if delta != 0:
                moves.append((i, j, delta))
    elif selection_method == 'gradient':
        # Use gradient information from partial derivatives of correlation
        corr_full = compute_corr_fft(h)
        # Find the two highest correlations and propose moves to
        # decrease them by some small amount
        worst_indices = np.argsort(corr_full)[-10:][::-1]
        for lag in worst_indices[:5]:  # Top 5 lags
            # Manually select indices that affect this lag
            # This is a simplification
            i = np.random.randint(0, n)
            j = (i + lag) % n
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max > 0:
                delta = np.random.uniform(0.01, min(0.05, delta_max))
                moves.append((i, j, delta))
    return moves

                        best_c5 = current_c5

                    new_c5 = evaluate_c5(h_new)

                        current_c5 = new_c5
                        if current_c5 < best_c5:
                            best_h = h.copy()
                            best_c5 = current_c5
        
        iteration += 1
    
    return best_h, best_c5

def local_refine(h):
    """Refine solution using SLSQP optimizer"""
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h))]
    
    h_refined = h.copy()
    
    try:
        res = minimize(
            objective,
            h_refined,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
    except:
        pass
    return h, evaluate_c5(h)
