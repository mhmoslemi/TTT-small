def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

            sum_constraint = cp.sum(x) == n//2

            y_vars = {}
            constraints = []

            for k in range(1, n):
                for i in range(n - k):
                    y_var = cp.Variable(boolean=True)
                    y_vars[(i,k)] = y_var
                    total_pairs += 1
                    
                    # Add constraints for the product variable
                    constraints.append(y_var <= x[i])
                    constraints.append(y_var <= x[i+k])
                    constraints.append(y_var >= x[i] + x[i+k] - 1)

            pairs_per_lag = {}

                for i in range(n - k):
                    pair_key = (i,k)
                    y_pair = y_vars[pair_key]
                    lag_sum.append(y_pair)
                    # Add constraint that max pair count is at least L
                    constraints.append(y_pair <= x[i])
                    constraints.append(y_pair <= x[i + k])
                    constraints.append(y_pair >= x[i] + x[i + k] - 1)
                pairs_per_lag[k] = lag_sum
