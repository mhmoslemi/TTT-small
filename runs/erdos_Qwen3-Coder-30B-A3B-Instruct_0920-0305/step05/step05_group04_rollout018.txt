import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def conv_fft(a, b):
    """Compute linear convolution using FFT with zero-padding."""
    n = len(a)
    m = len(b)
    pad_len = n + m - 1
    a_pad = np.pad(a, (0, pad_len - n), mode='constant')
    b_pad = np.pad(b, (0, pad_len - m), mode='constant')
    A = np.fft.fft(a_pad)
    B = np.fft.fft(b_pad)
    conv_result = np.fft.ifft(A * B).real[:pad_len]
    return conv_result

def project_onto_bounded_simplex(x):
    """Project a vector onto the set {x : 0 ≤ x_i ≤ 1, sum x_i = 1}"""
    # Clip to [0,1]
    x = np.clip(x, 0, 1)
    # Total sum
    s = np.sum(x)
    # If sum is already correct
    if abs(s - 1.0) < 1e-12:
        return x
    # Adjust using bisection search for lambda
    # We need to find lambda such that sum_i clip(x_i + lambda, 0, 1) = 1.
    # This function is piecewise linear in lambda.  
    # Direct solver approach:
    # Very fast bisection to find the unique lambda
    a = 0.0
    b = 1.0
    # Ensure we sample enough for control
    for _ in range(35):
        mid = (a + b) / 2.
        x_test = np.clip(x + mid, 0, 1)
        s_test = np.sum(x_test)
        if s_test < 1.0:
            a = mid
        else:
            b = mid
    lambda_val = b
    proj = np.clip(x + lambda_val, 0, 1)
    # Double-check the sum
    if not abs(np.sum(proj) - 1.0) < 1e-6:
        proj = project_onto_bounded_simplex(x)
    return proj

def soft_weights(c, beta):
    """Compute soft weights w_k = exp(-beta * c_k) / sum(exp(-beta*c_k))"""
    c_max = np.max(c)
    exp_c = np.exp(-beta * (c - c_max))
    w = exp_c / np.sum(exp_c)
    return w

def smooth_obj_and_grad(h, beta):
    """Return smooth objective value for Φ_β(h) and its gradient."""
    n = len(h)
    # Convolve h with itself to get all correlations (shifts 0 to n-1)
    c_all = conv_fft(h, h)
    # Extract the correlations corresponding to k=1,...,n-1
    c = c_all[1:n]
    
    # Compute weight vector
    w = soft_weights(c, beta)
    # Gradient of Φ_β is - sum w_k ∇c_k = -(conv(h, w) + conv(h, reverse(w)))
    # Using FFT: h * w (convolution) -> ifft(fft(h)*fft(w))
    # But w is a row vector applied as column vector on left
    
    w_rev = w[::-1]
    g1 = conv_fft(h, w)
    g2 = conv_fft(h, w_rev)
    g = -(g1 + g2)
    
    # Value of smooth objective
    val = np.log(np.sum(np.exp(-beta * c))) / beta
    return val, g

def gradient_descent_step(h, beta, alpha, beta_max_gamma=2.0, beta_max=200):
    """Performs one step of projected gradient descent for surrogate objective."""
    # Compute smooth objective and its gradient
    obj_val, g = smooth_obj_and_grad(h, beta)
    # Step direction
    h_new = h - alpha * g

    # Project onto bounded simplex
    h_proj = project_onto_bounded_simplex(h_new)
    
    # Try different step sizes (backtracking line search)
    max_iter = 100
    for i in range(max_iter):
        h_trial = h - alpha * g
        h_proj_trial = project_onto_bounded_simplex(h_trial)
        obj_trial, _ = smooth_obj_and_grad(h_proj_trial, beta)
        if obj_trial <= obj_val - 0.1 * alpha * np.sum(g**2):
            h = h_proj_trial
            break
        else:
            alpha *= 0.5
            if alpha < 1e-12:
                break
    
    # Anneal beta 
    if beta < beta_max:
        beta = min(beta * beta_max_gamma, beta_max)
    
    return h, beta, alpha

def evaluate_c5_exact(h):
    """Evaluate exact C5 as max correlation of h with 1-h"""
    corr = correlate(h, 1 - h, mode="full")
    c5 = np.max(corr)
    return c5
