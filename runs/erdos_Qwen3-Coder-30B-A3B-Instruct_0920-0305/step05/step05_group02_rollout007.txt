def compute_binary_autocorr(x):
    """Helper to compute all p_k values for binary x"""
    n = len(x)
    p = np.zeros(n)
    for k in range(1, n):
        # For each shift k, count overlap
        p[k] = sum(x[i]*x[i+k] for i in range(n-k))
    return p

def greedy_initial_binary(n):
    """A greedy heuristic to find an initial binary solution"""
    x = np.zeros(n, dtype=int)
    selected = set()
    while len(selected) < n//2:
        best_i = -1
        best_min_p = -1
        for i in range(n):
            if i in selected: continue
            # Compute tentative p after adding i
            temp_p = np.zeros(n)
            for k in range(1,n):
                temp_p[k] = sum(x[j]*x[j+k] for j in range(n-k)) if j != i and j+k < n else 0
            # consider the effect of including i
            for k in range(1, n):
                d = abs(i-k)
                if d < n and i + k < n:
                    temp_p[k] += x[i]*x[i+k]
            min_p = min(temp_p[1:], default=n)
            if min_p > best_min_p:
                best_min_p = min_p
                best_i = i
        if best_i >= 0:
            x[best_i] = 1
            selected.add(best_i)
    return x
