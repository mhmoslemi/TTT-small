import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_simplex(v, target_sum):
    """Project vector v onto the capped simplex: sum = target_sum, 0 <= v_i <= 1"""
    # Sort v in descending order 
    v_sorted = np.sort(v)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(v_sorted)
    # Find the threshold index
    idx = np.where(v_sorted > (target_sum - cumsum + v_sorted) / np.arange(1, len(v_sorted) + 1))[0]
    if len(idx) == 0:
        theta = (target_sum - cumsum[-1]) / len(v_sorted)
        return np.clip(v_sorted + theta, 0, 1)
    else:
        theta = (target_sum - cumsum[idx[-1]]) / (idx[-1] + 1)
        v_proj = np.clip(v_sorted + theta, 0, 1)
        return v_proj

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial setup
    n = 44  # Even number so n/2 is integer
    dx = 2.0 / n
    target_sum = n / 2.0  # sum constraint

    # Initial shift set
    K = {0}
    # Variables
    h = Variable(n)
    tau = Variable(1)
    
    # Best solution
    best_h = None
    best_c5 = float('inf')

    # Keep best feasible solution encountered
    best_feasible_h = None
    best_feasible_c5 = float('inf')

    # Inital feasible h
    h0 = np.array(initial_h_values)
    h0 = h0 / np.sum(h0) * target_sum
    h0 = np.clip(h0, 0, 1)
    # Normalize
    scale = target_sum / np.sum(h0)
    h0 = h0 * scale
    # Enforce symmetry
    h0 = (h0 + h0[::-1]) / 2.0
    best_feasible_h = h0
    best_feasible_c5 = evaluate_c5(h0)

    # Column generation loop
    while time.time() - start_time < budget_s * 0.95:
        # Build problem with current active K
        # Set to fixed values
        constraints = []
        constraints += [0 <= h[i] for i in range(n)]
        constraints += [h[i] <= 1 for i in range(n)]
        constraints += [sum(h) == target_sum]

        # Symmetry constraint
        for i in range(n // 2):
            constraints.append(h[i] == h[n - 1 - i])

        # y variables for active shifts
        y_vars = {}
        active_constr = []

        # Generate full correlation (for exact evaluation later)
        # Add constraints for each active shift in K
        for k in K:
            if k not in y_vars:
                y_var = Variable(n - abs(k) if k != 0 else n)
                y_vars[k] = y_var
            # McCormick envelope for shift k (only if not already added)
            current_y = y_vars[k]
            # Generate indices for this shift
            if k >= 0:
                # i ranges from 0 to n-1-k, so j = i+k
                y_indices = list(range(n-k))
                # Note: h_{i+k} i.e. j on the right
                for i in range(n - abs(k)):
                    idx = i
                    idx2 = i + abs(k)
                    # y[i] gives h[i+k] * h[i] 
                    constraints.append(current_y[i] >= 0)
                    constraints.append(current_y[i] <= h[idx])
                    constraints.append(current_y[i] <= h[idx2])
                    constraints.append(current_y[i] >= h[idx] + h[idx2] - 1)
            else:
                # k < 0, so i+k >= 0 means i >= -k.
                # i ranges from -k to n-1, so i ranges from 0 to n+ k -1
                # j = i+k, so j ranges from k to n-1+k
                # so j ranges from max(k,0) to n-1
                # i ranges from max(-k, 0) to n-1
                y_indices = list(range(n + k))  # start from -k to n-k-1
                for i in range(n + k):
                    idx = i
                    idx2 = i - k  # i - k = j  
                    constraints.append(current_y[i] >= 0)
                    constraints.append(current_y[i] <= h[idx])
                    constraints.append(current_y[i] <= h[idx2])
                    constraints.append(current_y[i] >= h[idx] + h[idx2] - 1)
            
            # Constraint from the McCormick lines
            # Compute the total over the valid indices for shift k
            corr_sum = sum(current_y)
            constraints.append(tau >= target_sum - corr_sum)
            
        # Set objective: minimize tau
        objective = Minimize(tau)

        # Form problem
        prob = Problem(objective, constraints)

        # Try solving with GLOP or CBC
        try:
            # Solve the LP
            prob.solve(solver=CPLEX, verbose=False)
            # proxsol = prob.solve(solver=SCIP, verbose=False)
            # if proxsol is not None and not np.isnan(proxsol) and prob.status == 'optimal':
            #     pass  # good
            # else:
            #     # fallback
            #     prob.solve(solver=ECOS, verbose=False)

            h_val = [h[i].value for i in range(n)] 
            h_val = np.array(h_val)
            
            # Time-check before expensive operations
            if time.time() - start_time > budget_s * 0.9:
                break

            # Project to simplex (ensure the sum matches perfectly)
            h_proj = project_to_simplex(h_val, target_sum)

            # Symmetrise again to be safe
            h_proj = (h_proj + h_proj[::-1]) / 2.0

            # Update best feasible solution
            c5_val = evaluate_c5(h_proj)
            if c5_val < best_feasible_c5:
                best_feasible_h = h_proj
                best_feasible_c5 = c5_val

            # Now compute exact O_k values
            # Note: This computation is approximate, but we can still use approximate correlation
            # Here, we'll only be optimistic about the new shifted constraint
            corr_full = correlate(h_proj, 1 - h_proj, mode="full")
            # Find max O_k (excluding self)
            max_idx = np.argmax(corr_full)
            max_k = max_idx - (len(corr_full) - 1) // 2
            max_val = corr_full[max_idx]

            # Should also include the k=0 case
            # We can safely include more than one
            # But for now, just check if needed
            
            # Add the shift that maximised the overlap
            if max_k not in K:
                K.add(max_k)
                
            # Update best
            if c5_val < best_c5:
                best_h = h_proj.copy()
                best_c5 = c5_val

        except Exception as e:
            print("Solve failed with error: {}".format(e))
            # Fallback to simple gradient
            pass
        finally:
            # Try to get out early if enum not complete
            if time.time() - start_time > budget_s * 0.95:
                break

    # Final polishing stage: projected gradient descent on true C5
    if best_feasible_h is not None:
        h_polished = best_feasible_h.copy()
        best_polished_h = h_polished
        best_polished_c5 = evaluate_c5(h_polished)
        
        # Gradient descent
        max_iter = 2000
        eta = 0.5 * dx  # Initial step size
        max_tolerance = 1e-6
        patience = 0 
        patience_limit = 10

        # Some variables to track
        prev_c5 = best_polished_c5

        for i in range(max_iter):
            # Time check
            if time.time() - start_time > budget_s * 0.99:
                break

            # Compute C5 and gradients
            # The max overlap integral
            # approximates the derivative of max_k of h(x) (1 - h(x + k)) dx
            # We are working with discrete
            # Method: use discrete mixture of gradients

            # Obtain current max set
            corr_full = correlate(h_polished, 1 - h_polished, mode="full")
            max_i = np.argmax(corr_full)
            max_shift = max_i - (len(corr_full) - 1) // 2
            max_val = corr_full[max_i]
            # All shifts where max overlap is observed
            abs_diff = np.abs(corr_full - max_val)
            active_shifts = []
            for i in range(len(corr_full)):
                if abs_diff[i] < 1e-10:
                    active_shifts.append(i - (len(corr_full) - 1) // 2)

            # Compute the subgradient using all active shifts
            grad = np.zeros_like(h_polished)
            for k in active_shifts:
                # Compute partial derivative ∂/∂h_i of O_k = Σ_i h_i - Σ_i h_i h_{i+k}
                # ∂O_k/∂h_i = 1 - ( h_{i+k} + h_{i-k} ) (or handle out-of-bounds gracefully)
                # But since we have indices, carefully construct the shift
                # For shift k in index space (in [-n+1, n-1]):
                # O_k = Σ_i h_i - Σ_i h_i * h_{i+k}
                # ∂O_k/∂h_i = 1 - h_{i+k} (when i+k >= 0)
                # But when k < 0 then h_{i+k} is the earlier index
                # So ∂O_k/∂h_i = 1 - h_{i+k} if valid (i+k in domain)
                # Another way:
                # h_{i+k} = h[i+k] if Ok
                # Also check for symmetric boundary conditions always (expand outward)
                # We compute by convolution  
                # Let's evaluate the true derivative
                # That is done via correlation 

                # This computes:
                # d/dh_i Omega_k(h) = 1 - (h_{i+k} + h_{i-k}) (if also i-k >= 0)
                # But this is the "jump" term. What we really compute:
                # Since O_k(h) = sum h_i - sum h_i * h_{i+k} 
                # For i, we have:
                # ∂O_k/∂h_i = 1 - h_{i+k}     (since only i contributes to term h_i * h_{i+k}
                # For j = i+k, we have:
                # ∂O_k/∂h_j = - h_i

                dj = np.zeros_like(h_polished)
                for j in range(n):
                    i = j - k 
                    if i >= 0 and i < n:
                        dj[i] += 1
                        dj[j] -= 1

                dj = dj.reshape(-1)
                grad += dj
            grad = grad / len(active_shifts) # average

            # Step
            h_new = h_polished - eta * grad
            # Project
            h_new = project_to_simplex(h_new, target_sum)
            h_new = (h_new + h_new[::-1]) / 2.0

            c5_new = evaluate_c5(h_new)
            if c5_new < best_polished_c5:
                best_polished_h = h_new.copy()
                best_polished_c5 = c5_new

            # Update step size
            if c5_new < prev_c5:
                prev_c5 = c5_new
                patience = 0
            else:
                patience += 1
                if patience >= patience_limit:
                    eta *= 0.5
                    patience = 0
                    if eta < 1e-10:
                        break

            h_polished = h_new

        # At the end, final binary search refinement
        h_final = best_polished_h
        c5_final = best_polished_c5

        # Optional: convert to binary mask locally around the max
        h_bin = np.zeros_like(h_final)
        indices_sorted = np.argsort(h_final)[::-1]
        for i in range(int(target_sum)):
            h_bin[indices_sorted[i]] = 1.0

        # Local search on swapped pairs to improve
        corr_old = evaluate_c5(h_bin)
        changed = True
        while changed and time.time() - start_time < budget_s * 0.99:
            changed = False
            for i in range(n):
                for j in range(i+1, n):
                    if h_bin[i] == 1 and h_bin[j] == 0:
                        # Try swapping
                        h_test = h_bin.copy()
                        h_test[i] = 0
                        h_test[j] = 1
                        c5_test = evaluate_c5(h_test)
                        if c5_test < corr_old:
                            h_bin = h_test
                            corr_old = c5_test
                            changed = True
        # Final comparison
        c5_bin = evaluate_c5(h_bin)
        if c5_bin < c5_final:
            h_final = h_bin
            c5_final = c5_bin
        
        if best_feasible_c5 < c5_final:
            return (best_feasible_h.tolist(), best_feasible_c5, n)
        else:
            return (h_final.tolist(), c5_final, n)
    else:
        return (best_h.tolist() if best_h is not None else [0.5]*n, best_c5, n)
