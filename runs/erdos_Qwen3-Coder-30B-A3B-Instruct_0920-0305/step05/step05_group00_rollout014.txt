from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        h_plus = project_to_sum_box(h_plus, TARGET_SUM)

        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_to_sum_box(h_minus, TARGET_SUM)
        c5_minus = evaluate_c5(h_minus)
        
        return (c5_plus - c5_minus) / (2 * eps)
    
    t_start = time.time()
    
    while time.time() - t_start < run_seconds:
        # Select random indices
        indices = rng.permutation(n_points)
        
        # Use adaptive learning rate based on gradient magnitude
        for i in indices:
            grad = estimate_gradient(i, h)
            step_size = 0.05 / (1 + abs(grad)**2)
            if abs(grad) > 1e-12:
                h[i] -= step_size * grad
                
        h = project_to_sum_box(h, TARGET_SUM)
        c5 = evaluate_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Occasionally refine with local optimization
        if rng.random() < 0.05:
            h_candidate = h.copy()
            bounds = [(0, 1) for _ in range(n_points)]
            
            def objective(h):
                return evaluate_c5(h)
            
            def constraint_eq(h):
                return np.sum(h) - TARGET_SUM
            
            cons = {'type': 'eq', 'fun': constraint_eq}
            
            try:
                res = minimize(
                    objective, 
                    h_candidate, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=cons,
                    options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10}
                )
                if res.success:
                    h_candidate = res.x
                    h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
                    c5_candidate = evaluate_c5(h_candidate)
                    if c5_candidate < best_c5:
                        best_c5 = c5_candidate
                        best_h = h_candidate.copy()
                        h = h_candidate.copy()
            except:
                pass
    
    return best_h, best_c5
