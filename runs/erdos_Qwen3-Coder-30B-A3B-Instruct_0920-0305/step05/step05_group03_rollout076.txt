import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use a hybrid optimization approach
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Phase 1: Global optimization using differential evolution
    try:
        res = differential_evolution(
            objective,
            bounds,
            maxiter=1000,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            atol=1e-10,
            tol=1e-10
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Phase 2: Local refinement with trust-constr
    try:
        res = minimize(
            objective, 
            best_h, 
            method='trust-constr', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 10000, 'gtol': 1e-14, 'ftol': 1e-14}
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Phase 3: Random restart refinement
    local_restart_bound = 1000
    
    for restart in range(local_restart_bound):
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
        # Start with small random perturbations around previous best
        noise = np.random.normal(0, 0.02, n_points)
        new_h = best_h + noise
        new_h = np.clip(new_h, 0, 1)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-15}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue

    # Phase 4: Structured pattern optimization
    if (time.time() - start_time) < budget_s * 0.9:
        try:
            # Try a pattern with a central peak and decay on sides
            x = np.linspace(0, 2, n_points)
            # A hybrid of Gaussian and tent pattern
            gaussian = np.exp(-((x - 1)**2) / 0.15)
            tent = 1 - np.abs(x - 1)
            hybrid = 0.7 * gaussian + 0.3 * tent
            
            # Normalize
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(hybrid) * dx)
            hybrid = hybrid * scale_factor
            
            res = minimize(
                objective, 
                hybrid, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-16, 'gtol': 1e-16}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass

    # Phase 5: Perturb best solution and retry optimization
    if (time.time() - start_time) < budget_s * 0.9:
        try:
            # Small perturbations to escape local minima
            perturbed_h = best_h + np.random.normal(0, 0.005, n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(perturbed_h) * dx)
            perturbed_h = perturbed_h * scale_factor
            
            res = minimize(
                objective, 
                perturbed_h, 
                method='trust-constr', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 2000, 'gtol': 1e-15, 'ftol': 1e-15}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass

    return (best_h.tolist(), best_c5, n_points)
