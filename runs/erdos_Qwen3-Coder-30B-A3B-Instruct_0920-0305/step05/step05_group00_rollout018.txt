# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_smooth_max_grad(h, beta):
    """Compute gradient of smooth max objective"""
    corr = compute_corr_fft(h)
    lags = len(corr)
    # Compute C_ell for all lags  
    C_ell = corr * dx
    # Numerically stable log-sum-exp
    max_C = np.max(C_ell)
    exp_C = np.exp(beta * (C_ell - max_C))
    w = exp_C / np.sum(exp_C)
    
    # Compute gradient components for each lag
    grad = np.zeros_like(h)
    for l in range(lags):
        # Interpreted as: for i in 0..n-1, the term C_l is h[i]*(1-h[i+l]) 
        # so grad is 1 - h[i+l] - h[i-l] where negative indices default to 0
        if l == 0:
            grad += w[l] * (1 - h)  # Only the h terms
        else:
            # The expression is just the gradient of -∑ h_i (1 - h_{i+l})
            # So the derivative of h_i (1 - h_{i+l}) wrt h_i is (1 - h_{i+l})
            # and wrt h_{i+l} is -h_i
            # Total is: 1 - h_{i+l} for terms where i is in [0,n-1] and i+l < n
            # and -h_{i-l} for terms where i-l >= 0
            shift_plus = np.zeros_like(h)
            shift_plus[l:] = h[:-l]
            shift_minus = np.zeros_like(h)
            shift_minus[:-l] = h[l:]
            grad += w[l] * (1 - shift_plus - shift_minus)

    return grad

def smooth_max_descent(h0, beta_schedule, step_size_schedule, max_iter):
    """Apply smoothed max gradient descent with beta continuation"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    t = 0
    for beta in beta_schedule:
        # Adjust step size schedule for this beta
        step_size = step_size_schedule[t] if t < len(step_size_schedule) else step_size_schedule[-1]
        
        for _ in range(max_iter):
            grad = compute_smooth_max_grad(h, beta)
            h_new = h - step_size * grad
            
            # Project to feasible set
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                
            h = h_new.copy()
            t += 1
            
        if t >= len(step_size_schedule):
            break
    
    return best_h, best_c5
