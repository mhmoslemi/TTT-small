import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time
from math import log

def project(v, S):
    """Project onto the capped simplex: 0 <= v[i] <= 1, sum v[i] = S"""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2
        p = np.clip(v + lam, 0.0, 1.0)
        s = p.sum()
        if s > S:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2, 0.0, 1.0)

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def compute_O_w_grad(h, alpha):
    """Compute the vector of overlaps O_r, weights w and gradient of smooth surrogate"""
    n = len(h)
    dx = 2.0 / n
    # correlation with ones and with itself
    c1 = correlate(h, np.ones(n), mode='full')
    c2 = correlate(h, h, mode='full')
    O = (c1 - c2) * dx  # overlaps for all lags
    O_max = np.max(O)
    w = np.exp(alpha * (O - O_max))
    w = w / w.sum()
    # compute gradient components efficiently
    s = np.convolve(w, np.ones(n), mode='valid')
    g2 = correlate(w, h, mode='full')[n-1:2*n-1]
    g3 = correlate(w, h[::-1], mode='full')[n-1:2*n-1][::-1]
    grad = dx * (s - g2 - g3)
    return O, w, grad
