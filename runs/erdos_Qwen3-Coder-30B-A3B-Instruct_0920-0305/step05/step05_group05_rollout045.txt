import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
from scipy.fft import rfft, irfft
import time
from math import log, ceil

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    n = len(h)
    dx = 2.0 / n
    # Using FFT for correlation
    H = rfft(h)
    G = rfft(1 - h)
    corr = irfft(H * np.conj(G))
    # Only the valid part of correlation
    corr = corr[:n]
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_simplex(h, target_sum):
    """Project h onto the simplex with sum=target_sum using binary search"""
    # Ensure h is within [0,1]
    h = np.clip(h, 0, 1)
    # Binary search on Lagrange multiplier
    low, high = -100, 100
    for _ in range(50):
        lamb = (low + high) / 2
        z = np.clip(h - lamb, 0, 1)
        s = np.sum(z)
        if s > target_sum:
            low = lamb
        else:
            high = lamb
    z = np.clip(h - (low + high) / 2, 0, 1)
    return z

def interpolate_control_points(c, K, n):
    """Interpolate control points c to full grid of size n"""
    x_control = np.linspace(0, 2, K)
    x_full = np.linspace(0, 2, n)
    h_full = np.interp(x_full, x_control, c, left=0, right=0)
    return np.clip(h_full, 0, 1)

def objective(c, K, n, alpha=1.0):
    """Objective function with soft-max approximation"""
    h = interpolate_control_points(c, K, n)
    # Repair step to enforce constraints
    h = project_to_simplex(h, n/2)
    # Compute correlations using FFT for speed
    dx = 2.0 / n
    H = rfft(h)
    G = rfft(1 - h)
    corr = irfft(H * np.conj(G))
    corr = corr[:n]
    corr = corr * dx
    # Soft max approximation
    if alpha > 0:
        max_val = np.max(corr)
        if max_val > 100:
            corr = corr - max_val
        exp_vals = np.exp(alpha * corr)
        return np.log(np.sum(exp_vals)) / alpha
    else:
        return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 128
    K = 12  # Number of control points
    target_sum = n / 2
    dx = 2.0 / n
    
    # CMA-ES Configuration
    lambda_ = 4 + 3 * log(K)
    lambda_ = max(4, int(lambda_))
    mu = lambda_ // 2
    
    best_c5 = float('inf')
    best_h = None
    
    # Keep track of time
    remaining_time = budget_s * 0.95
    
    # Multi-restart strategy
    restarts = 0
    while remaining_time > 1.0:
        restarts += 1
        # Initialize control parameters
        c0_mean = np.random.uniform(0, 1, K)
        # Initialize CMA-ES parameters
        mean = np.copy(c0_mean)
        sigma = 0.2
        C = np.eye(K)
        weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
        weights = weights / np.sum(weights)
        mu_eff = 1.0 / np.sum(weights**2)
        
        # Annealing schedule for soft max
        alpha_start = 5.0
        alpha_end = 30.0
        max_generations = min(2000, int(remaining_time / 0.5))
        gen = 0
        last_improvement = 0
        
        # Evaluate initial point
        current_f = objective(mean, K, n, alpha_start)
        best_val = current_f
        best_mean = np.copy(mean)
        
        # CMA-ES main loop
        while gen < max_generations:
            gen += 1
            # Generate offspring
            offspring = []
            for _ in range(lambda_):
                # Sample a new solution
                z = np.random.standard_normal(K)
                x = mean + sigma * C @ z
                offspring.append(x)
            
            # Evaluate offspring
            fitnesses = []
            for sol in offspring:
                # Repair and evaluate
                h_temp = interpolate_control_points(sol, K, n)
                h_repaired = project_to_simplex(h_temp, target_sum)
                # Soft max for this generation
                alpha = alpha_start + (alpha_end - alpha_start) * (gen/max_generations)
                corr = correlate(h_repaired, 1 - h_repaired, mode='full')
                corr = corr[:len(corr)//2+1] * dx
                if alpha > 0:
                    max_val = np.max(corr)
                    if max_val > 100:
                        corr = corr - max_val
                    exp_vals = np.exp(alpha * corr)
                    f = np.log(np.sum(exp_vals)) / alpha
                else:
                    f = np.max(corr)
                fitnesses.append(f)
            
            # Sort by fitness
            indices = np.argsort(fitnesses)
            best_indices = indices[:mu]
            
            # Update mean
            z_w = np.sum([weights[i] * (offspring[best_indices[i]] - mean) for i in range(mu)], axis=0)
            mean += z_w
            
            # Update covariance matrix
            # Simplified rank-1 update (naive implementation)
            C = 0.9 * C + 0.04 * np.outer(z_w, z_w)
            
            # Update sigma
            sigma *= np.exp(0.3 * (np.linalg.norm(z_w) / np.sqrt(K) - 1))
            
            # Check improvement
            new_val = fitnesses[best_indices[0]]
            if new_val < best_val:
                best_val = new_val
                best_mean = np.copy(mean)
                last_improvement = 0
            else:
                last_improvement += 1
                if last_improvement > 50:
                    break
                
            # Check remaining time
            elapsed = time.time() - start_time
            remaining_time = budget_s * 0.95 - elapsed
            if remaining_time < 1.0:
                break
                
        # Final polishing step with exact objective
        h = interpolate_control_points(best_mean, K, n)
        h = project_to_simplex(h, target_sum)
        final_c5 = evaluate_c5(h)
        
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h.copy()
            
        # Break if we're running out of time
        if remaining_time < 1.0:
            break
    
    # One final.local refinement step on the best solution
    if best_h is not None:
        h_final = best_h.copy()
        for t in range(200):
            # Compute exact shifts
            corr = correlate(h_final, 1 - h_final, mode='full')
            corr = corr[:len(corr)//2+1]
            max_shift = np.argmax(corr)
            max_val = corr[max_shift]
            if max_val < 1e-7:
                break
            # Gradient of O_k is from the correlation which equals the difference in h's
            grad = np.zeros_like(h_final)
            for i in range(len(h_final)):
                if i + max_shift < len(h_final):
                    grad[i] += (1 - h_final[i + max_shift])
                if i - max_shift >= 0:
                    grad[i] += h_final[i - max_shift]
            
            h_final -= 0.02 * grad
            h_final = project_to_simplex(h_final, target_sum)
            corr = correlate(h_final, 1 - h_final, mode='full')
            corr = corr[:len(corr)//2+1] * dx
            new_max = np.max(corr)
            if new_max < best_c5 or best_c5 == 0:
                best_c5 = new_max
                best_h = h_final.copy()
            else:
                # Early stop if no improvement
                if t > 100 and abs(best_c5 - new_max) < 1e-8:
                    break
        
        # Final evaluation
        corr = correlate(best_h, 1 - best_h, mode='full')
        corr = corr[:len(corr)//2+1] * dx
        final_c5 = np.max(corr)
        return (best_h.tolist(), float(final_c5), n)
    
    return (best_h.tolist(), best_c5, n)
