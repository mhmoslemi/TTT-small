```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_simplex(v, target_sum):
    """Project vector v onto the simplex with sum = target_sum"""
    # Sort in descending order
    sorted_v = np.sort(v)[::-1]
    # Compute cumulative sums
    cumsum_v = np.cumsum(sorted_v)
    # Find the threshold index
    idx = np.arange(len(sorted_v)) + 1
    threshold = (cumsum_v - target_sum) / idx
    # Find the largest k such that threshold[k-1] >= sorted_v[k-1]
    # This is guaranteed to exist since target_sum >= 0 and v >= 0
    k = np.where(sorted_v > threshold)[0]
    if len(k) == 0:
        theta = threshold[-1]
    else:
        theta = threshold[k[-1]]
    # Apply projection
    result = np.maximum(v - theta, 0)
    # Scale to target sum
    result = result * target_sum / np.sum(result)
    return result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 44  # Even number for easier symmetry
    dx = 2.0 / n_points
    target_sum = n_points / 2
    # Prime the loop with an initial solution
    best_h = np.array(initial_h_values)
    if len(best_h) > n_points:
        best_h = best_h[:n_points]
    elif len(best_h) < n_points:
        best_h = np.pad(best_h, (0, n_points - len(best_h)), 'constant', constant_values=0)
    # Normalize
    scale = target_sum / np.sum(best_h)
    best_h = best_h * scale
    # Evaluate initial C5
    best_c5 = evaluate_c5(best_h)
    
    # Active set of shifts; start with {0}
    K = [0]
    
    # Keep track of best feasible solution found
    h_best = best_h.copy()
    c5_best = best_c5
    
    while (time.time() - start_time) < budget_s * 0.9:
        # Build the LP model in CVXPY
        h = Variable(n_points)
        tau = Variable()
        # Keep track of y variables
        y_vars = {}
        constraints = []
        
        # Box constraints
        constraints.extend([h >= 0, h <= 1])
        # Sum constraint
        constraints.append(sum(h) == target_sum)
        # Symmetry constraint h[i] = h[n-1-i]
        for i in range(n_points // 2):
            constraints.append(h[i] == h[n_points - 1 - i])
        
        # Generate y variables and McCormick constraints
        for k in K:
            y = Variable(n_points)
            y_vars[k] = y
            # For valid range of i+k
            for i in range(n_points):
                if 0 <= i + k < n_points:
                    # McCormick envelopes for y[i] >= h[i]*h[i+k] (and other constraints)
                    constraints.append(y[i] >= 0)
                    constraints.append(y[i] <= h[i])
                    constraints.append(y[i] <= h[i+k])
                    constraints.append(y[i] >= h[i] + h[i+k] - 1)
                else:
                    # Out of bounds indices: y[i] should be 0
                    constraints.append(y[i] == 0)
            # Constraint: tau >= sum(h) - sum(y)
            constraints.append(tau >= sum(h) - sum(y))
        
        # Objective
        objective = Minimize(tau)
        
        # Create and solve the problem
        prob = cp.Problem(objective, constraints)
        try:
            # Use GLOP if available; fallback to CBC
            prob.solve(solver=cp.GLOP, verbose=False)
        except:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except:
                # If both fail, fall back to a local solve with SCS (convex)
                try:
                    prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
                except:
                    break  # If all fail, we're done
        if prob.status != 'optimal':
            break
        h_sol = h.value
        tau_sol = tau.value
        
        # Ensure symmetry
        h_sol = (h_sol + h_sol[::-1]) / 2.0
        # Project onto simplex if not already exact
        h_sol = project_to_simplex(h_sol, target_sum)
        
        # Evaluate the true C5 with this h
        c5_sol = evaluate_c5(h_sol)
        
        if c5_sol < c5_best:
            h_best = h_sol.copy()
            c5_best = c5_sol

        # Evaluate true overlaps for current h_sol
        corr = correlate(h_sol, 1 - h_sol, mode="full")
        max_idx = np.argmax(corr)
        max_corr_val = corr[max_idx]
        k_max = max_idx - (len(corr) - 1) // 2  # Convert to shift index
        # Check if this is the current best
        if max_corr_val * dx <= tau_sol + 1e-6:
            # Feasible solution found!
            break
        else:
            # Add worst shift to active set
            if k_max not in K:
                K.append(k_max)
        
        # Check time
        if (time.time() - start_time) > budget_s * 0.95:
            break
    
    # Final polishing with projected gradient descent on true C5
    h_current = h_best.copy()
    c5_current = c5_best
    step_size = 0.5 * dx
    max_iter = 2000
    for iter in range(max_iter):
        if (time.time() - start_time) > budget_s * 0.98:
            break
        corr = correlate(h_current, 1 - h_current, mode="full")
        max_idx = np.argmax(corr)
        k_max = max_idx - (len(corr) - 1) // 2
        # Compute subgradient ∇O_k_max
        # O_k_max = sum(h) - sum(h_i * h_{i+k_max})
        # ∂O_k_max / ∂h_i = 1 - (h_{i+k_max} if exists) - (h_{i-k_max} if exists)
        g = np.ones(n_points)
        if k_max > 0:
            g[:n_points-k_max] -= h_current[k_max:]
        if k_max < 0:
            g[-k_max:] -= h_current[:n_points+k_max]
        # Take average over all active shifts (usually just one but handle multiple)
        active_shifts = np.where(np.abs(corr - corr[max_idx]) < 1e-8)[0]
        k_active = active_shifts - (len(corr) - 1) // 2 if len(active_shifts) > 0 else np.array([k_max])
        g = np.zeros(n_points)
        for k in k_active:
            temp_g = np.ones(n_points)
            if k > 0:
                temp_g[:n_points-k] -= h_current[k:]
            if k < 0:
                temp_g[-k:] -= h_current[:n_points+k]
            g += temp_g
        g /= len(k_active)
        # Descend
        h_new = h_current - step_size * g
        # Project onto simplex
        h_new = project_to_simplex(h_new, target_sum)
        # Symmetrize
        h_new = (h_new + h_new[::-1]) / 2.0
        # Evaluate
        c5_new = evaluate_c5(h_new)
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
        else:
            step_size /= 2
        if step_size < 1e-7:
            break

    # Optional binary rounding
    h_bin = h_best.copy()
    # Create binary mask
    indices = np.argsort(h_bin)[::-1][:int(target_sum)]
    h_binary = np.zeros(n_points)
    h_binary[indices] = 1.0
    # Fixing any discrepancy in the number of ones via swaps
    current_count = np.sum(h_binary)
    if current_count < target_sum:
        diff = int(target_sum - current_count)
        for _ in range(diff):
            # Find zero element that if swapped to one would reduce max overlap
            min_gain = np.inf
            idx_to_flip = -1
            for i in range(n_points):
                if h_binary[i] == 0:
                    # Check if changing to 1 improves C5
                    h_test = h_binary.copy()
                    h_test[i] = 1.0
                    c5_test = evaluate_c5(h_test)
                    if c5_test < min_gain:
                        min_gain = c5_test
                        idx_to_flip = i
            if idx_to_flip != -1:
                h_binary[idx_to_flip] = 1.0
    elif current_count > target_sum:
        diff = int(current_count - target_sum)
        for _ in range(diff):
            # Find one element that if swapped to zero would reduce max overlap
            min_gain = np.inf
            idx_to_flip = -1
            for i in range(n_points):
                if h_binary[i] == 1:
                    # Check if changing to 0 improves C5
                    h_test = h_binary.copy()
                    h_test[i] = 0.0
                    c5_test = evaluate_c5(h_test)
                    if c5_test < min_gain:
                        min_gain = c5_test
                        idx_to_flip = i
            if idx_to_flip != -1:
                h_binary[idx_to_flip] = 0.0
    # Evaluate binary version
    try:
        c5_bin = evaluate_c5(h_binary)
        if c5_bin < c5_best:
            h_best = h_binary.copy()
            c5_best = c5_bin
    except:
        pass
    
    return (h_best.tolist(), c5_best, n_points)
```
