import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
from scipy.fft import rfft, irfft
import time
import math

def repair_h(candidate, n_points, target_sum):
    """Repair candidate to ensure h ∈ [0,1] and sum = target_sum"""
    # Clip to [0,1]
    h = np.clip(candidate, 0, 1)
    # Project onto simplex: sum = target_sum
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-12:
        # Use binary search for Lagrange multiplier
        # Solve: sum_i max(0, h_i - lambda) = target_sum
        def sum_with_lambda(lam):
            return np.sum(np.maximum(0, h - lam))
        
        lower, upper = -1.0, 1.0
        for _ in range(50):
            mid = (lower + upper) / 2
            s = sum_with_lambda(mid)
            if s > target_sum:
                lower = mid
            else:
                upper = mid
        lamb = (lower + upper) / 2
        h = np.maximum(0, h - lamb)
        # Renormalize to ensure exact sum
        scale = target_sum / np.sum(h)
        h = h * scale
    return h

def evaluate_c5_fast(h, dx=1.0):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation via FFT
    H = rfft(h)
    G = rfft(1 - h)
    corr = irfft(H * np.conj(G)) 
    # Extract only the valid correlations (full mode)
    full_corr = np.zeros(len(corr) * 2 - 1)
    full_corr[:len(corr)] = corr
    full_corr[-(len(corr)-1):] = corr[1:][::-1]
    # Return max
    return np.max(full_corr) * dx

def soft_max_overlaps(O, alpha):
    """Compute log-sum-exp approximation to max"""
    max_O = np.max(O)
    # Center for numerical stability
    centered = O - max_O
    exp_O = np.exp(alpha * centered)
    log_sum = np.log(np.sum(exp_O))
    return max_O + log_sum / alpha

def cma_es_optimization(target_sum, n_points, K, budget_s):
    """CMA-ES optimization with soft max objective and repair"""
    # Control points (K locations, K values)
    np.random.seed(42)
    
    # Initialize search space: uniform distribution over [0,1] for control values
    x0 = np.random.uniform(0, 1, K)
    mean = x0.copy()
    sigma = 0.2
    lambda_pop = max(4, int(3 + 3 * np.log(K)))
    
    # For symmetric operations
    def symmetrize_h(h):
        return (h + h[::-1]) / 2
    
    # Main loop
    best_h = None
    best_value = np.inf
    start_time = time.time()
    
    # Evolutionary generations
    max_generations = 1500
    gen = 0
    
    # Anneal alpha from 5 to 30
    alpha_schedule = np.linspace(5, 30, int(max_generations * 0.7))
    alpha_schedule = np.concatenate([alpha_schedule, np.full(max_generations - len(alpha_schedule), 30)])
    
    while gen < max_generations and (time.time() - start_time) < budget_s * 0.99:
        alpha = alpha_schedule[gen]
        
        # Generate population
        offspring = []
        for _ in range(lambda_pop):
            # Sample from multivariate normal
            z = np.random.randn(K)
            new_candidate = mean + sigma * z
            offspring.append(new_candidate)
            
        # Evaluate each offspring
        values = []
        for candidate in offspring:
            # Repair to box + simplex constraints
            h_full = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, K), 
                                candidate)
            
            if np.random.rand() < 0.2 and best_h is not None:
                h_full = symmetrize_h(h_full)
            
            h_clipped = repair_h(h_full, n_points, target_sum)
            
            # Fast correlation evaluation
            O = np.correlate(h_clipped, 1-h_clipped, mode="full")
            dx = 2.0 / n_points
            max_O = np.max(O) * dx
            
            if np.isnan(max_O) or np.isinf(max_O):
                values.append(np.inf)
            else:
                # Use soft max instead of direct max early on
                soft_max_val = soft_max_overlaps(O * dx, alpha)
                # If late in optimization, use actual max  
                if gen >= int(max_generations * 0.7):
                    soft_max_val = max_O
                values.append(soft_max_val)
        
        # Selection - pick best
        idxs = np.argsort(values)[:lambda_pop//2]
        selected = [offspring[i] for i in idxs]
        selected_values = [values[i] for i in idxs]
        
        # Update mean and covariance (simplified single-step CMA-ES)
        if len(selected) > 0:
            new_mean = np.mean(selected, axis=0)
            # Update sigma with small damping
            sigma = sigma * (0.99 + 0.01 * (np.random.rand() > 0.5))
            
            # Keep track of best
            min_val_idx = np.argmin(selected_values)
            if selected_values[min_val_idx] < best_value:
                best_value = selected_values[min_val_idx]
                # Reconstruct best h for later polishing
                best_candidate = selected[min_val_idx]
                h_full = np.interp(np.linspace(0, 2, n_points), 
                                    np.linspace(0, 2, K), 
                                    best_candidate)
                if np.random.rand() < 0.2:
                    h_full = symmetrize_h(h_full)
                best_h = repair_h(h_full, n_points, target_sum)
                
            mean = new_mean
        
        gen += 1
    
    # Return best found
    if best_h is None:
        return np.ones(n_points) * 0.5, np.inf, n_points
    
    return best_h, best_value, n_points

def project_gradient_refinement(h_orig, n_points, max_iterations=500, eta=0.05):
    """Simple projected gradient refinement"""
    h = h_orig.copy()
    target_sum = n_points / 2
    dx = 2.0 / n_points
    prev_C5 = np.inf
    
    # Check convergence
    for _ in range(max_iterations):
        # Compute overlaps
        O = np.correlate(h, 1-h, mode="full")
        max_idx = np.argmax(O) - (len(O)-1)//2
        max_O = np.max(O)
        current_C5 = max_O * dx
        
        # Early exit if improvement small
        if abs(current_C5 - prev_C5) < 1e-8:
            break
        prev_C5 = current_C5
        
        # Compute sub-gradient for max overlap
        # One active shift
        active_shift = max_idx
        
        # Approximate for the gradient
        # ∂O_k/∂h_i = (1-h_{i+k}) - h_{i-k} 
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # Contributions from shift k = active_shift
            if i + active_shift >= 0 and i + active_shift < len(h):
                grad[i] += (1 - h[i + active_shift])
            if i - active_shift >= 0 and i - active_shift < len(h):
                grad[i] -= h[i - active_shift]
                
        # Update step with projected gradient descent
        h_new = h - eta * grad
        
        # Project to [0,1] and realize sum constraint
        h_proj = repair_h(h_new, n_points, target_sum)
        h = h_proj.copy()
        
        if abs(current_C5 - prev_C5) < 1e-8:
            break
    
    # Final computation
    final_O = np.correlate(h, 1-h, mode="full")
    final_C5 = np.max(final_O) * dx
    return h, final_C5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Run parameters
    n_points = 128  # Grid points for final evaluation
    K = 12  # Control points
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    best_h = None
    best_C5 = np.inf
    
    # Run multiple restarts
    restarts = 3
    
    for restart in range(restarts):
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
        # Change seed for each restart
        np.random.seed(seed + restart)
        
        # Run CMA-ES optimization
        h_candidate, c5_value, _ = cma_es_optimization(
            target_sum, n_points, K, budget_s * 0.95 / restarts
        )
        
        # Refine with gradient method
        h_refined, c5_refined = project_gradient_refinement(
            h_candidate, n_points, 500, 0.05
        )
        
        if c5_refined < best_C5:
            best_C5 = c5_refined
            best_h = h_refined.copy()
            
    # Final check using exact numpy correlation (for golden bound)
    if best_h is not None:
        corr = correlate(best_h, 1 - best_h, mode="full")
        final_C5 = np.max(corr) * dx
        return best_h.tolist(), final_C5, n_points
    else:
        # Fallback
        h_fallback = np.ones(n_points) * 0.5
        corr = correlate(h_fallback, 1 - h_fallback, mode="full")
        return h_fallback.tolist(), np.max(corr) * dx, n_points
