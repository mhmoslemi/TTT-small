def conv_fft(a, b):
    """Compute linear convolution using FFT with zero padding"""
    n = len(a)
    m = len(b)
    size = n + m - 1
    # pad to next power of 2 for efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.fft(a, fft_size)
    B = np.fft.fft(b, fft_size)
    C = A * B
    c = np.fft.ifft(C).real[:size]
    return c

def project_onto_bounded_simplex(x):
    """Project vector x onto the bounded simplex: sum x_i = 1, 0 <= x_i <= 1"""
    x = np.clip(x, 0, 1)
    s = np.sum(x)
    if abs(s - 1.0) < 1e-10:
        return x
    
    # Find lambda such that sum(clamp(x + lambda, 0, 1)) = 1
    # This is piecewise linear in lambda, so use bisection
    left, right = -1.0, 1.0
    for _ in range(50):
        mid = (left + right) / 2
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if sum_y > 1:
            right = mid
        else:
            left = mid
    lambda_ = (left + right) / 2
    return np.clip(x + lambda_, 0, 1)

def soft_weights(c, beta):
    """Compute softmax weights for soft minimum"""
    c_max = np.max(c)
    exp_c = np.exp(-beta * (c - c_max))
    return exp_c / np.sum(exp_c)

def smooth_obj_and_grad(h, beta):
    """Compute smooth surrogate value and its gradient"""
    n = len(h)
    # Compute correlations c_k = sum h_i * h_{i+k}
    c = conv_fft(h, h)
    c = c[1:]  # keep only shifts k=1, ..., n-1
    
    w = soft_weights(c, beta)
    
    # Gradient calculation using FFT
    # Define shifts
    forward_shift = np.zeros(n)
    forward_shift[1:] = w[:-1]  # w_1, ..., w_{n-2}
    backward_shift = np.zeros(n)
    backward_shift[-1] = 0  # set last element (which is w_0) to 0
    backward_shift[:-1] = w[1:]  # w_2, ..., w_{n-1} 
    
    grad = -(conv_fft(h, forward_shift) + conv_fft(h, backward_shift))
    
    Phi_beta = (1.0 / beta) * np.log(np.sum(np.exp(-beta * c)))
    
    return Phi_beta, -grad
