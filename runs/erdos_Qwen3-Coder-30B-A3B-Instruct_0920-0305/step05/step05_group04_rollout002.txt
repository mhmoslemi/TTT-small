import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def project_to_simplex(h, target_sum):
    """Project h onto the simplex {x : 0≤x≤1, sum x = target_sum}"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    
    # Bisection to solve for lambda such that sum(clamp(h+lambda, 0, 1)) = target_sum
    # The function f(lambda) = sum(clamp(h+lambda, 0, 1)) is monotonic in lambda
    def f(lam):
        return np.sum(np.clip(h + lam, 0, 1)) - target_sum

    # bracket the root
    lam_lo = -np.max(h) - 1
    lam_hi = 1 - np.min(h)
    while f(lam_lo) > 0:
        lam_lo -= 1
    while f(lam_hi) < 0:
        lam_hi += 1

    # bisection
    for _ in range(50):
        lam = (lam_lo + lam_hi) / 2
        val = f(lam)
        if abs(val) < 1e-12:
            break
        if val < 0:
            lam_lo = lam
        else:
            lam_hi = lam
    lam = (lam_lo + lam_hi) / 2
    h = np.clip(h + lam, 0, 1)
    return h

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value, considering only non-zero shifts
    max_corr = np.max(corr[n//2+1:])  # exclude shift 0
    return max_corr * dx

def compute_correlations(h):
    """Compute all correlations c_k = sum h[i] h[i+k] for k=1..n-1"""
    n = len(h)
    # use linear convolution, but we only care about lag 1..n-1
    corr = correlate(h, h, mode='full')
    return corr[n//2+1:]

def soft_min_gradient(h, c, beta):
    """Compute gradient of soft min surrogate (1/β) log sum exp(-β c_k)"""
    n = len(h)
    # Weights: exp(-beta * c_k) / sum(exp(-beta * c_k))
    # add small epsilon to stabilize
    c_max = np.max(c)
    exp_neg_c = np.exp(-beta * (c - c_max))
    w = exp_neg_c / np.sum(exp_neg_c)
    
    # Gradient per component = - sum_k w_k * grad c_k
    # grad c_k[i] = h[i-k] + h[i+k] (with boundary handling)
    # Use fft-based convolution for efficiency: conv(h, w_rev) + conv(h, w) 
    # where w_rev is w reversed
    w_rev = w[::-1]
    
    # The gradient is: - (conv(h, w_rev) + conv(h, w)) 
    # But it's 2 * (h * (w * h) convolved, which is:
    # conv(h, w_rev) + conv(h, w) - (for non-periodic) but let's just use proper conv
    # Actually, it's simpler:
    # d/dh_i [sum_k w_k c_k] = sum_k w_k d/dh_i c_k = sum_k w_k (h_{i+k} + h_{i-k})
    # But we're doing - ∑_k w_k * ∇c_k, so:
    # ∇_i [φ_β(h)] = - ∑_k w_k ( h_{i+k} + h_{i-k} )
    # Which is conv(h, rev(w)) + conv(h, w) - 2*w[0]*h[i] for i=0..n-1
    # Let's compute that as two separate convolutions:
    # First, pad h and scatter w for correct indexing
    # Actually, simpler, compute by correlation: 
    # the gradient w.r.t. h[i] is sum_k w[k] * [h[i-k]+h[i+k]]
    # So make w length n by padding with zeros (if needed) and do forward/backward conv.
    # One can treat it as a two-step convolution from h using vectors v1, v2 that 
    # have v1[i] = w[0] for coefficient of h[i] on right (for shift +1), v2[i] = w[0]
    # Actually, best to just use standard convolution, adjusting for these periodic drops
    # Let's take it simpler:
    # Take the circular correlation by FFT, but want a linear one, so zero-padding:
    
    # First, compute conv(h, w_rev) where w_rev means shift weights from the end
    # for convenience and let's just do it naively via FFT because of indexing issues
    
    # For i in 0..n-1, 
    # grad_i = - sum_{k=1}^{n-1} w_k [h_{i+k} + h_{i-k}] 
    # (with wraparound for index)
    # i.e., form vector of weight w[\cdot], treat as a shift operator
    # Let’s do the indirect convolution: 
    #     [[w_1, w_2, ..., w_{n-1}, w_{n-2}, ..., w_1]] * h
    # More involved. Better to define shifted vectors.
    
    # Recognize the work is right.
    # I'll reconstruct the full SVD or compute via spectral methods/transforms:
    
    # Hamming approx: try convolution via numpy for small shifts and use required 
    # zeros as boundary handling. But the code below works.
    # Perturb slightly so we don't create zero weight:
    w = np.array(w)
    w += 1e-16
    
    # Compute ∇ φ_β(h) using efficient convolution:
    # What is the correct convolution? Let's do it step by step: 
    # Forward shift convolution (shift +1, +2 ...) = shifted h vector
    # Backward shift = reverse shift
    
    # Shield inequalities by padding:
    h_padded = np.concatenate([h, [0] * (n-1)])
    w_rev = w[::-1]
    w_rev = np.concatenate([w_rev, [0] * (n-1)])
    
    # compute conv(h,w_rev) [backward shifts] 
    h_w_rev = np.convolve(h_padded, w_rev, mode='valid')
    # compute conv(h,w) [forward shifts]
    w = np.concatenate([w, [0] * (n-1)])
    h_w = np.convolve(h_padded, w, mode='valid')
    
    grad = -h_w_rev[:n] - h_w[:n]
    
    # Manually handle small weights:
    return grad

    h = h0.copy()
    beta = beta0
    step = step0
    best_c5 = float('inf')
    best_h = h.copy()
    last_grad_norm = float('inf')
    
    for i in range(max_iter):
        if i % 100 == 0:
            c = compute_correlations(h)
            c5_val = 1 + (1/beta)*np.log(np.sum(np.exp(-beta*c)))
            eval_c5 = evaluate_c5(h)
            if eval_c5 < best_c5:
                best_c5 = eval_c5
                best_h = h.copy()
        
        c = compute_correlations(h)
        # Compute soft-min with β
        c_max = np.max(c)
        exp_neg_c = np.exp(-beta * (c - c_max))
        w = exp_neg_c / np.sum(exp_neg_c)
        phi_beta = (1/beta)*np.log(np.sum(exp_neg_c)) + 1  # phi = 1 + (1/β) log ∑ exp(-β c_k)
        
        # Compute gradient 
        #         ∇ φ_β = -∑ w_k ∇ c_k (i.e. ∇ c_k = h_{i-k} + h_{i+k})
        # Exploit that ∇ c_k(i) = h[i-k] + h[i+k] 
        # Stats: it is almost identical to inner product of h with modulated w. clever:
        
        h_padded = np.concatenate([h, [0] * (n-1)])
        w_rev = w[::-1]
        w = w[:n]  # Ensure same length
        w_rev = w_rev[:n]
        w_rev = np.concatenate([w_rev, [0] * (n-1)])  # padded
        
        # we compute:
        #    conv(h, w_rev) and conv(h, w) and add
        #    and negate
        h_w_rev = np.convolve(h_padded, w_rev[::-1], mode='valid')
        h_w = np.convolve(h_padded, w, mode='valid')
        
        # At most 1 shifted entries
        grad = -h_w_rev[:n] - h_w[:n]
        
        # Update with backtracking line search
        alpha = 1.0
        while alpha > 1e-10:
            h_trial = h - alpha * grad
            h_trial = project_to_simplex(h_trial, n//2)
            
            c_trial = compute_correlations(h_trial)
            c5_approx_trial = 1 + (1/beta)*np.log(np.sum(np.exp(-beta * c_trial)))
            c5_exact_trial = evaluate_c5(h_trial)
            
            if c5_exact_trial <= evaluate_c5(h) or np.abs(alpha) < 1e-12:
                break
            alpha *= 0.5
        if alpha <= 1e-12:
            h_trial = project_to_simplex(h - 1e-6 * grad, n//2)
        
        h = h_trial
        c_current = compute_correlations(h)
        c5_current = evaluate_c5(h)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy() 
    
        # Adaptive adjustment of β and step size
        if i % 50 == 0:
            beta *= gamma
            beta = min(beta, beta_max)
        if i % 10 == 0:
            last_grad_norm = np.linalg.norm(grad)
            if last_grad_norm < 1e-8:
                step *= 0.9  # Decay step size slowly
    
    return best_h, best_c5
