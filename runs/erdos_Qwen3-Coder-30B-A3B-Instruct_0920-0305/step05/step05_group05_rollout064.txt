import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def repair_h(c, K, n):
    """Repair function to ensure h is in [0,1] and sum to n/2"""
    # Interpolate control points to full grid
    h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, K), c)
    h = np.clip(h, 0, 1)
    
    # Project onto simplex (sum = n/2)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Use binary search for projection onto simplex
    # We solve min_z ||z-h||^2 s.t. 0 <= z <= 1, sum z = T
    def project_simplex(z):
        # This is exact for box + simplex constraints
        # Sort z in descending order to compute the projection efficiently
        # Using the method described in: 
        # "Projection Onto A Simplex" by Duchi, Shalev-Shwartz, Singer, and Tu
        # https://stanford.edu/~jduchi/projects/DuchiSiShTu08.pdf
        z_sorted = np.sort(z)[::-1]
        rho = 0.0
        for i in range(len(z_sorted)):
            theta = (np.sum(z_sorted[:i+1]) - target_sum) / (i + 1)
            if theta > z_sorted[i]:
                rho = i + 1
            else:
                break
        if rho == 0:
            # All elements need to be zero or handling case
            return np.clip(z, 0, 1)
            
        # Lambda is the threshold
        lambda_val = (np.sum(z_sorted[:rho]) - target_sum) / rho
        # Projection: z_i = max(z_i - lambda, 0)
        projected = np.maximum(z - lambda_val, 0)
        return projected
    
    h = project_simplex(h)
    return h

def soft_max_overlaps(O, alpha):
    """Compute smooth approximation of max of overlaps"""
    max_val = np.max(O)
    # Subtract max to avoid numerical overflow
    exp_vals = np.exp(alpha * (O - max_val))
    return max_val + np.log(np.sum(exp_vals)) / alpha

def cmaes_optimize(n, K, initial_seed=42, max_iter=1500):
    np.random.seed(initial_seed)
    # Control points with K values, placed evenly over [0,2]
    x_control = np.linspace(0, 2, K)
    # Initial guess is a simple shape
    c0 = np.full(K, 0.5)  # Start with flat
    # Or use structured shape to get better start
    x = np.linspace(0, 2, K)
    # Initial shape
    c0 = 0.5 + 0.4 * np.sin(np.pi * x)
    c0 = np.clip(c0, 0, 1)
    
    # CMA-ES paramaters
    lambda_ = max(4, 3 + int(3 * np.log(K)))
    mu = lambda_ // 2
    weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    weights = weights / np.sum(weights)
    cc = 4 / (K + 4)
    cs = (mu + 2) / (K + mu + 5)
    c1 = 2 / ((K + 1.3) ** 2)
    cmu = min(1 - c1, 2 * (mu - 2 + 1 / mu) / ((K + 2) ** 2))
    sigma = 0.3
    mean = c0.copy()
    C = np.eye(K)
    chi = np.sqrt(K) * (1 - 1 / (4 * K) + 1 / (21 * K ** 2))
    
    # For soft max annealing
    alpha_start = 5.0
    alpha_end = 30.0
    alpha_schedule = np.linspace(alpha_start, alpha_end, int(max_iter * 0.7))
    
    # Main CMA-ES loop
    best_fitness = float('inf')
    best_c = None
    
    for it in range(max_iter):
        # Anneal soft max parameter
        if it < len(alpha_schedule):
            alpha = alpha_schedule[it]
        else:
            alpha = alpha_end
        
        # Generate offspring
        offspring = []
        fitnesses = []
        for _ in range(lambda_):
            # Sample direction from multivariate normal
            try:
                z = np.random.randn(K)
                x = mean + sigma * (C @ z)
                # Repair by clipping to [0,1]
                x = np.clip(x, 0, 1)
                offspring.append(x)
                
                # Project to simplex and evaluate
                h = repair_h(x, K, n)
                # Compute correlations using FFT
                dx = 2.0 / n
                corr = correlate(h, 1 - h, mode="full")
                O = corr * dx
                # Use soft max approx to get a smooth landscape
                fitness = soft_max_overlaps(O, alpha)
                fitnesses.append(fitness)
                
                if fitness < best_fitness:
                    best_fitness = fitness
                    best_c = x.copy()
            except np.linalg.LinAlgError:
                # If matrix is singular, we fallback to the current mean
                continue
                
        # Select best mu parents for update
        indices = np.argsort(fitnesses)[:mu]
        selected = [offspring[i] for i in indices]
        z_k = np.stack([(x - mean) / sigma for x in selected])
        # Update mean
        mean = mean + cs * (C @ (np.sum(weights * z_k.T, axis=1)))
        # Update evolution paths
        ps = (1 - cs) * ps + cs * (C @ (np.sum(weights * z_k.T, axis=1)))
        # Update covariance matrix
        C = (1 - c1 - cmu) * C + c1 * (ps @ ps.T) + cmu * z_k.T @ np.diag(weights) @ z_k
        # Update sigma
        sigma *= np.exp(cs / (2 * chi) * (np.linalg.norm(ps) / chi - 1))
        
        # Optionally symmetrize if convergence seems to be stuck
        # Could be added later.
        
        # Early convergence test
        if it > 200 and all(fitnesses[i] - fitnesses[i-1] < 1e-6 for i in range(1, min(20, len(fitnesses)))):
            break
    
    if best_c is None:
        best_c = c0.copy()  # Fallback to initial
    return best_c

def projected_subgradient_update(h, n, max_iter=500, eta0=0.05):
    """Apply a few iterations of projected subgradient descent"""
    dx = 2.0 / n
    # Gradient descent to refine the exact max
    for t in range(max_iter):
        # Find the k(s) that achieve the max
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        # Compute gradient for this shift (partial derivative of h)        
        grad = np.zeros(n)
        # Gradient: for shift k, we get contributions from h_i -> h_{i+k}
        # This is a difference of h's and 1-h's
        # |      h_{i+k}  |
        # |  1 - h_i      |
        # So gradient of O_k = sum_i h_i * (1 - h_{i+k})
        # => grad_i = (1 - h_{i+k}) for active terms 
        # => but also the reverse
        # So we consider: O_k(h) = sum_i h_i (1 - h_{i+k})
        # Gradient dO_k / dh_i = (1 - h_{i+k})
        # Now, this is much simpler when we use the form:
        # O_k = sum_i h_i (1 - h_{i+k}) = sum_i h_i - sum_i h_i h_{i+k} 
        #   = sum_i h_i - sum_{i+k} h_{i+k} h_i = sum_i h_i - sum_i h_i h_{i-k}
        # => dO_k/dh_i = (1 - h_{i-k}) for k>0, (1 - h_{i+k}) for k<0
        # Also consider the full form O_k = sum h_i (1 - h_{i+k})
        # So, dO_k/dh_i = (1 - h_{i+k}) if i+k exists, or 0 if not valid
        
        grad = np.zeros(n)
        for i in range(n):
            # Accumulate for shift k
            g = 0.0
            j = i - max_k  # i corresponds to i - k = j
            if 0 <= j < n:
                # We're the derivative of term h_i (1 - h_j)
                g += (1 - h[j])
            grad[i] = g
        
        # Now the update: h_{t+1} = h_t - eta_t * g
        # eta_t = eta0 / sqrt(t+1)
        eta = eta0 / np.sqrt(t + 1)
        h_new = h - eta * grad
        
        # Project back onto simplex and box
        h_new = np.clip(h_new, 0, 1)
        target_sum = n / 2.0
        current_sum = np.sum(h_new)
        scale = target_sum / current_sum if current_sum > 0 else 1
        h_new = h_new * scale
        h = h_new
        
    return h
