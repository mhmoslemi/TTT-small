import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_c5_binary(h):
    """Evaluate the C5 constant for a binary step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def log_sum_exp(x, alpha=30.0):
    """Smooth approximation to max"""
    return (1.0 / alpha) * np.log(np.sum(np.exp(alpha * x)))

def smooth_max_overlap(h, alpha=30.0):
    """Smooth approximation to the maximum overlap"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    # Extract the relevant part (negative shifts to positive shifts)
    corr_full = corr[len(corr)//2:]
    return log_sum_exp(corr_full, alpha) * dx

def project_to_constraint(x, target_sum):
    """Project x to satisfy sum(x) = target_sum"""
    current_sum = np.sum(x)
    if current_sum == 0:
        return x
    scale = target_sum / current_sum
    return x * scale

def generate_legendre_pattern(n):
    """Generate a Legendre-based binary pattern"""
    p = n // 2
    if p < 2:
        return None
    # Find the largest prime less than p
    def is_prime(n):
        if n < 2:
            return False
        for i in range(2, int(math.sqrt(n)) + 1):
            if n % i == 0:
                return False
        return True
    while not is_prime(p):
        p -= 1
        if p < 2:
            return None
    
    # Generate quadratic residues
    residues = set()
    for a in range(1, p):
        residues.add((a * a) % p)
    
    b = np.zeros(p, dtype=np.int8)
    for i in residues:
        b[i] = 1
    
    # Duplicate to fill n
    h = np.tile(b, 2)
    if len(h) > n:
        h = h[:n]
    elif len(h) < n:
        h = np.pad(h, (0, n - len(h)), mode='constant')
    
    # Adjust sum to exactly n/2
    current_sum = np.sum(h)
    needed = n // 2 - current_sum
    if needed > 0:
        # Find zeros and turn them into ones
        zeros = np.where(h == 0)[0]
        for i in range(needed):
            if i < len(zeros):
                h[zeros[i]] = 1
    elif needed < 0:
        # Find ones and turn them into zeros
        ones = np.where(h == 1)[0]
        for i in range(-needed):
            if i < len(ones):
                h[ones[i]] = 0
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define problem sizes
    n_values = [58, 62, 66, 70, 74, 78]
    
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    # Early stopping threshold
    target_c5 = 0.3807
    
    for n in n_values:
        if (time.time() - start_time) > 0.95 * budget_s:
            break
            
        # Generate deterministic starts
        starts = []
        
        # Legendre pattern
        legendre = generate_legendre_pattern(n)
        if legendre is not None:
            starts.append(legendre)
        
        # Random balanced patterns
        for i in range(3):
            perm = np.random.permutation(n)
            r = np.zeros(n, dtype=np.int8)
            r[perm[:n//2]] = 1
            starts.append(r)
        
        for h_start in starts:
            # Get initial correlation values
            dx = 2.0 / n
            corr = correlate(h_start, 1 - h_start, mode="full")
            current_c5 = np.max(corr) * dx
            worst_shifts = np.where(np.abs(corr - np.max(corr)) < 1e-10)[0]
            worst_shifts = worst_shifts - (len(corr) // 2)  # Convert to shift indices
            
            h = h_start.copy()
            
            # Annealing parameters
            T0 = 0.05
            Tf = 1e-5
            N_iter = 2*10**6
            iter_count = 0
            
            # Simulated annealing
            while iter_count < N_iter and (time.time() - start_time) < 0.95 * budget_s:
                if (time.time() - start_time) > 0.85 * budget_s:
                    break
                    
                # Update temperature
                elapsed = time.time() - start_time
                if elapsed >= 0.95 * budget_s:
                    break
                
                # Pick a worst shift
                k = np.random.choice(worst_shifts)
                k_idx = k + len(corr) // 2
                
                # Find bad ones for this shift
                bad_ones = np.where((h == 1) & (np.roll(h, -k) == 0))[0]
                if len(bad_ones) == 0:
                    iter_count += 1
                    continue
                
                # Pick a bad one
                i = np.random.choice(bad_ones)
                
                # Find good zeros
                good_zeros = np.where((h == 0) & (np.roll(h, -k) == 1))[0]
                if len(good_zeros) == 0:
                    good_zeros = np.where(h == 0)[0]
                if len(good_zeros) == 0:
                    iter_count += 1
                    continue
                    
                # Pick a good zero
                j = np.random.choice(good_zeros)
                
                # Compute change in correlation
                old_corr = corr.copy()
                new_corr = old_corr.copy()
                
                # Update for the two positions
                for m in range(len(corr)):
                    dm = m - len(corr) // 2
                    if abs(dm) > n:
                        continue
                        
                    if i <= n - 1 - abs(dm):
                        # i contributes to m
                        if dm >= 0:
                            new_corr[m] += 1 - 2 * h[i] * h[i + dm]
                        else:
                            new_corr[m] += 1 - 2 * h[i - dm] * h[i]
                    if j <= n - 1 - abs(dm):
                        # j contributes to m
                        if dm >= 0:
                            new_corr[m] += 1 - 2 * h[j] * h[j + dm]
                        else:
                            new_corr[m] += 1 - 2 * h[j - dm] * h[j]
                    
                    # Swap
                    if dm >= 0:
                        if i == j + dm:
                            new_corr[m] += (h[j] * h[j + dm] - 2 * h[i] * h[i + dm])
                        elif j == i + dm:
                            new_corr[m] += (h[i] * h[i + dm] - 2 * h[j] * h[j + dm])
                    else:
                        if j == i - dm:
                            new_corr[m] += (h[i] * h[i - dm] - 2 * h[j] * h[j - dm])
                        elif i == j - dm:
                            new_corr[m] += (h[j] * h[j - dm] - 2 * h[i] * h[i - dm])
                
                new_c5 = np.max(new_corr) * dx
                old_c5 = current_c5
                
                accept = False
                if new_c5 < old_c5:
                    accept = True
                else:
                    T = T0 * (Tf / T0) ** (iter_count / N_iter)
                    if T > 0 and np.random.rand() < np.exp(-(new_c5 - old_c5) / T):
                        accept = True
                
                if accept:
                    h[i], h[j] = h[j], h[i]
                    corr = new_corr.copy()
                    current_c5 = new_c5
                    worst_shifts = np.where(np.abs(corr - np.max(corr)) < 1e-10)[0]
                    worst_shifts = worst_shifts - (len(corr) // 2)
                    
                    if current_c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = current_c5
                        best_n = n
                        
                        if best_c5 < target_c5:
                            break
                
                iter_count += 1
                
                # Periodic restart
                if iter_count % 20000 == 0:
                    # Try a new random balanced mask
                    perm = np.random.permutation(n)
                    h = np.zeros(n, dtype=np.int8)
                    h[perm[:n//2]] = 1
                    corr = correlate(h, 1 - h, mode="full")
                    current_c5 = np.max(corr) * dx
                    worst_shifts = np.where(np.abs(corr - np.max(corr)) < 1e-10)[0]
                    worst_shifts = worst_shifts - (len(corr) // 2)
            
            # If we've found a good enough solution, break early
            if best_c5 < target_c5:
                break
        
        # Early stop if target achieved
        if best_c5 < target_c5:
            break
    
    # Fine-tune with continuous optimization if we have a solution
    if best_h is not None and (time.time() - start_time) < 0.98 * budget_s:
        h_continuous = best_h.astype(float)
        h_continuous = project_to_constraint(h_continuous, len(h_continuous) / 2)
        
        def objective(x):
            # Smooth approximation to maximum overlap
            return smooth_max_overlap(x, alpha=30.0)
        
        bounds = [(0, 1)] * len(h_continuous)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - len(h_continuous) / 2}
        
        try:
            res = minimize(
                objective, 
                h_continuous, 
                method='L-BFGS-B',
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-10, 'gtol': 1e-10}
            )
            if res.success:
                h_final = res.x.copy()
                h_final = project_to_constraint(h_final, len(h_final) / 2)
                h_final = np.clip(h_final, 0, 1)
                final_c5 = evaluate_c5(h_final)
                
                if final_c5 < best_c5:
                    best_c5 = final_c5
                    best_h = h_final.tolist()
                    best_n = len(h_final)
        except:
            pass
    
    return (best_h.tolist(), best_c5, best_n)
