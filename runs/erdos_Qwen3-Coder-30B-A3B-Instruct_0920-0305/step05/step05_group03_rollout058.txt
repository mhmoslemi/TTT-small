```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 44  # Even number so n/2 is integer
    dx = 2.0 / n
    # Active set of shifts - start with small set
    K = set([0])
    max_iterations = 200  # to prevent runaway
    best_h = None
    best_c5 = float('inf')
    max_k = 0
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    if len(h0) != n:
        h0 = np.linspace(0, 1, n)  # fallback
    
    # If time is too short, just return the initial
    if (time.time() - start_time) > 0.9 * budget_s:
        return (h0.tolist(), evaluate_c5(h0), n)

    # Begin cutting-plane approach
    try:
        iter_count = 0
        while iter_count < max_iterations:
            elapsed = time.time() - start_time
            if elapsed > 0.95 * budget_s:
                break
            
            # Build LP model
            # Variables
            h = Variable(n)
            tau = Variable()
            y_vars = {}  # map (k, i) to y_{i,k}

            # Constraints
            constraints = [
                h >= 0,
                h <= 1,
                sum(h) == n / 2,
            ]

            # Symmetry
            for i in range(n // 2):
                constraints.append(h[i] == h[n - 1 - i])

            # Add variables and constraints for current active shifts
            for k in K:
                y_vars[k] = Variable(n)
                for i in range(n):
                    if 0 <= i + k < n:
                        constraints.extend([
                            y_vars[k][i] >= 0,
                            y_vars[k][i] <= h[i],
                            y_vars[k][i] <= h[i + k],
                            y_vars[k][i] >= h[i] + h[i + k] - 1
                        ])
                # The constraint: tau >= sum(h) - sum(y_{i,k})
                constraints.append(tau >= sum(h) - sum(y_vars[k]))

            # Objective: minimize tau
            prob = cp.Problem(cp.Minimize(tau), constraints)

            # Solve using GLOP
            try:
                prob.solve(solver=cp.GLOP, verbose=False)
            except Exception:
                try:
                    prob.solve(solver=cp.CBC, verbose=False)
                except Exception:
                    break  # give up

            if prob.status != cp.OPTIMAL:
                break

            # Extract solution
            h_val = h.value
            tau_val = tau.value

            # Symmetrize if needed
            if n % 2 == 0:
                h_val = (h_val + h_val[::-1]) / 2
            
            # Now check actual overlap values
            corr = correlate(h_val, 1 - h_val, mode="full")
            max_corr_idx = np.argmax(corr)
            max_corr_value = corr[max_corr_idx]
            
            # Shift k is offset by (n-1) in numpy's correlation indexing
            k_max = max_corr_idx - (n - 1)
            
            # Evaluate true C5
            actual_c5 = max_corr_value * dx

            if actual_c5 < best_c5:
                best_h = h_val.copy()
                best_c5 = actual_c5

            # Check termination criteria
            if max_corr_value <= tau_val + 1e-6:
                # we are good
                break

            # Add new shift to active set
            K.add(k_max)
            iter_count += 1

        if best_h is None:
            best_h = h0
            best_c5 = evaluate_c5(best_h)

    except Exception as e:
        # If LP solver fails, fallback to SLSQP
        h_final = h0.copy()  
        def objective(h):
            return evaluate_c5(h)
        bounds = [(0, 1) for _ in range(n)]
        def constraint_eq(h):
            return np.sum(h) * dx - 1.0
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(objective, h_final, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 300, 'ftol': 1e-12})
            if res.success:
                h_final = res.x.copy()
                h_final = h_final * (n/2) / np.sum(h_final)  # normalize
                best_c5 = evaluate_c5(h_final)
                best_h = h_final
        except:
            pass

    # Polish with projected gradient descent
    if best_h is None:
        best_h = h0.copy()
        best_c5 = evaluate_c5(best_h)
    
    # If still low on time, do some local polish
    final_h = best_h.copy()
    final_c5 = best_c5
    
    # Try projected gradient descent
    try:
        h_current = final_h.copy()

        def compute_subgradient(h, active_shifts):  # active_shifts comes from current h
            g = np.zeros(n)
            o_values = []
            for k in active_shifts:
                o_k = np.correlate(h, 1 - h, mode="full")
                o_values.append(o_k)
            
            # Find which shifts are active at h
            max_val = max(o_values)
            active_k = [k for k, o_val in enumerate(o_values) if abs(o_val - max_val) < 1e-10]
            
            temp_g = np.zeros(n)
            for k in active_k:
                # Compute gradient of correlation for shift k
                # ∂C_k/∂h_i = 1 - (h_{i+|k|} + h_{i-|k|}) if not out of bounds, else 0.
                # Step through elements of correlation to compute gradient contributions
                for i in range(n):
                    # This handles cuts vs full correlation cleanly
                    if k >= 0:
                        contrib = 1.0 - h[i+k] if i+k < n else 0.0
                    else:
                        contrib = 1.0 - h[i+k] if i+k >= 0 else 0.0
                    temp_g[i] += contrib
            g = temp_g / len(active_k) if active_k else np.zeros(n)
            return g

        # Start with a small step-size and reduce when needed
        eta = 0.5 * dx
        max_iters = 2000
        for it in range(max_iters):
            elapsed = time.time() - start_time
            if elapsed > 0.98 * budget_s:
                break
            
            # Compute sub-gradient (need actual max shifts)
            o_vals = correlate(h_current, 1 - h_current, mode="full")
            max_o_index = np.argmax(o_vals)
            max_o_shift = max_o_index - (n - 1)
            active_k = [k for k in range(-n+1, n) if abs(np.correlate(h_current, 1-h_current, mode="full")[k+n-1]) == np.max(o_vals)]
            if not active_k:
                active_k = [max_o_shift]
            g = np.zeros(n)
            for k in active_k:
                # derivative w.r.t. h[i]: 1 - h_{i+k} - h_{i-k} (some may be 0 by z-p)
                grad_k = np.zeros(n)
                for i in range(n):
                    if i + k >= 0 and i + k < n:
                        grad_k[i] += -h_current[i + k]
                for i in range(n):
                    if i - k >= 0 and i - k < n:
                        grad_k[i] += -h_current[i - k]
                grad_k += 1
                g += grad_k / len(active_k)

            # Update h
            h_new = h_current - eta * g
            # Project to [0,1]^n
            h_new = np.clip(h_new, 0, 1)
            # Re-project to the simplex
            h_new = project_simplex(h_new, n//2)
            # Symmetrize
            h_new = (h_new + h_new[::-1]) / 2
            # clip again after symmetry 
            h_new = np.clip(h_new, 0, 1)

            # Check if we improved
            new_c5 = evaluate_c5(h_new)
            if new_c5 < final_c5:
                final_c5 = new_c5
                final_h = h_new.copy()
                # damp the step-size if there was a big jump
                if abs(new_c5 - final_c5) > 0.01:
                    eta = min(eta * 0.8, 0.01)
                else:
                    eta = min(eta * 1.1, 0.1)
            else:
                eta *= 0.7  # reduce when no progress
            
            if eta < 1e-7:
                break

            h_current = h_new.copy()
            
    except Exception as e:
        pass

    # Optionally try binary rounding to see if we can do better
    try:
        h_binary = final_h.copy()
        sorted_indices = np.argsort(h_binary)[::-1]  # descending
        h_binary[:] = 0
        h_binary[sorted_indices[:n//2]] = 1
        final_c5_binary = evaluate_c5(h_binary)
        if final_c5_binary < final_c5:
            final_c5 = final_c5_binary
            final_h = h_binary.copy()
    except:
        pass

    return (final_h.tolist(), final_c5, n)

def project_simplex(v, target_sum):
    """Project vector v onto capped simplex with sum=target_sum"""
    n = len(v)
    # waterfilling
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    idx = np.arange(1, n + 1)
    feasible = v_sorted - (target_sum - cumsum) / idx > 0
    if not np.any(feasible):
        return np.full(n, 0.0)
    k = np.argmax(feasible)
    threshold = (target_sum - cumsum[k-1]) / k if k > 0 else target_sum / n
    proj = np.maximum(v - threshold, 0)
    # ensure correct sum
    correction = (target_sum - np.sum(proj)) / n
    proj += correction
    
    return proj
```
