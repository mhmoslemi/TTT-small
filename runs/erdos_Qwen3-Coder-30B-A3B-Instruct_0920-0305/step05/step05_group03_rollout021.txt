import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project(v, S):
    """Project vector v onto the capped simplex with sum=S"""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2
        p = np.clip(v + lam, 0.0, 1.0)
        s = p.sum()
        if s > S:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2, 0.0, 1.0)

def compute_O_w_grad(h, alpha):
    """Compute the overlap vector and gradient for smooth surrogate"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations
    c1 = correlate(h, np.ones(n), mode='full')
    c2 = correlate(h, h, mode='full')
    O = (c1 - c2) * dx  # Should be length 2*n-1
    # Smoothed version
    O_max = np.max(O)
    w = np.exp(alpha * (O - O_max))
    w = w / w.sum()
    # Compute gradient
    # s = sum over r of w_r * 1_{k-r in [0,n-1]}
    s = np.convolve(w, np.ones(n), mode='valid')
    # g2 = sum over r of w_r * h_{k+r}
    g2 = correlate(w, h, mode='full')[n-1:2*n-1]
    # g3 = sum over r of w_r * h_{k-r}
    g3 = correlate(w, h[::-1], mode='full')[n-1:2*n-1][::-1]
    grad = dx * (s - g2 - g3)
    return O, w, grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Upsample to 84 points
    if n_points < 84:
        # Linear interpolation
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, 84)
        h0 = np.interp(x_new, x_orig, h0)
        n_points = 84
    
    # Initial projection onto the capped simplex (sum = n/2)
    S = n_points / 2.0
    h0 = project(h0, S)
    
    # Best solution so far
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Global search with DE
    if (time.time() - start_time) < budget_s * 0.1:
        def de_objective(h):
            # Make sure it is in valid range
            h = np.clip(h, 0, 1)
            # Project to sum = S
            h = project(h, S)
            return evaluate_c5(h)
        
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = differential_evolution(
                de_objective,
                bounds,
                popsize=12,
                maxiter=30,
                mutation=(0.5, 1.0),
                recombination=0.7,
                seed=seed,
                disp=False
            )
            if res.success:
                h_candidate = np.clip(res.x, 0, 1)
                h_candidate = project(h_candidate, S)
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
    
    # Local improvement using projected gradient on smooth surrogate
    h = best_h.copy()
    alpha = 5.0  # Smoothness parameter
    eta = 0.5 * (2.0 / n_points)  # Learning rate
    max_iter = 5000
    for iter in range(max_iter):
        # Check time
        if (time.time() - start_time) > 0.97 * budget_s:
            break

        # Compute smooth objective and gradient
        O, w, grad = compute_O_w_grad(h, alpha)

        # Perform line search
        improved = False
        for factor in [1.0, 0.5, 0.25]:
            h_trial = project(h - factor * eta * grad, S)
            # Enforce symmetry
            h_trial = (h_trial + h_trial[::-1]) / 2
            h_trial = project(h_trial, S)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial
                h = h_trial.copy()
                improved = True
                break

        if not improved:
            eta *= 0.5  # Reduce step size

        # Anneal alpha
        if iter % 200 == 0:
            alpha = min(alpha * 1.5, 50.0)

        # Occasionally perturb
        if iter % 500 == 0 and not improved:
            h = project(h + 0.02 * np.random.randn(n_points), S)
            h = (h + h[::-1]) / 2
            h = project(h, S)

    # Final evaluation
    return (best_h.tolist(), best_c5, n_points)
