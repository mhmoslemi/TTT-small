import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def local_search_move(h, corr_full):
    """Try to move mass to decrease max correlation"""
    n = len(h)
    max_lag = np.argmax(corr_full)
    
    # Look ahead at the structure
    # Try to find positions that contribute to the worst lag
    g = np.zeros(n)
    for i in range(n):
        left_idx = i - max_lag
        right_idx = i + max_lag
        if left_idx >= 0:
            g[i] += h[left_idx]
        if right_idx < n:
            g[i] += h[right_idx]
    
    # Try different moves
    best_move = None
    best_delta = 0.0
    best_improvement = 0.0
    
    # Simple heuristic: try moving from minimum to maximum of g
    i_min = np.argmin(g)
    i_max = np.argmax(g)
    
    delta_max = min(h[i_min], 1.0 - h[i_max])
    if delta_max > 0:
        delta = min(delta_max, 0.01)
        # Temporarily perform this move and compute the new max correlation
        h_candidate = h.copy()
        h_candidate[i_min] -= delta
        h_candidate[i_max] += delta
        h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
        corr_new = compute_corr_fft(h_candidate)
        new_c5 = c5_from_corr(corr_new, n)
        improvement = evaluate_c5(h) - new_c5
        if improvement > best_improvement:
            best_improvement = improvement
            best_delta = delta
            best_move = (i_min, i_max, delta)
    
    return best_move
