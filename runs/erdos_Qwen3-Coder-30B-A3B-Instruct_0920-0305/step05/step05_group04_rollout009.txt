import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import math

def conv_fft(a, b):
    """Compute linear convolution using FFT with zero padding"""
    n = len(a)
    m = len(b)
    padded_len = n + m - 1
    # zero-pad to the next power of 2 for efficient FFT
    fft_len = 1 << (int(math.ceil(math.log2(padded_len))) if padded_len > 0 else 0)
    a_padded = np.pad(a, (0, fft_len - n), 'constant')
    b_padded = np.pad(b, (0, fft_len - m), 'constant')
    A = np.fft.fft(a_padded)
    B = np.fft.fft(b_padded)
    result = np.fft.ifft(A * B)
    return np.real(result[:padded_len])

def project_onto_bounded_simplex(x):
    """Project x onto the bounded simplex with sum=1, 0≤x_i≤1"""
    # Clip to [0,1]
    x = np.clip(x, 0, 1)
    n = len(x)
    # Target sum
    target = n / 2.0
    
    # Compute current sum
    current_sum = np.sum(x)
    
    # If already satisfied, return
    if abs(current_sum - target) < 1e-12:
        return x.copy()
    
    # Use bisection to find lambda
    l, r = -1.0, 1.0
    for _ in range(50):
        mid = (l + r) / 2.0
        candidate = np.clip(x + mid, 0, 1)
        new_sum = np.sum(candidate)
        
        if new_sum < target:
            l = mid
        else:
            r = mid
    
    lambda_val = (l + r) / 2.0
    return np.clip(x + lambda_val, 0, 1)

def soft_weights(c, beta):
    """Compute softmax weights for the soft minimum"""
    c_max = np.max(c)
    exp_term = np.exp(-beta * (c - c_max))
    return exp_term / np.sum(exp_term)

def smooth_obj_and_grad(h, beta):
    """Compute Φ_β(h) and its gradient"""
    n = len(h)
    # Compute self convolution
    c = conv_fft(h, h)[:n - 1]  # Take first n-1 elements
    if len(c) == 0:
        return 0.0, np.zeros_like(h)
    
    # Compute soft weights
    w = soft_weights(c, beta)
    
    # Compute gradient as -(conv(h, w) + conv(h, reverse(w)))
    # Since correlation is symmetric, convolution direction doesn't matter
    g = -conv_fft(h, w)[:n - 1]
    g_rev = -conv_fft(h, w[::-1])[:n - 1]
    gradient = np.zeros(n)
    # combine with appropriate indexing
    for k in range(1, n):
        if k < len(g):
            gradient[k] += g[k-1]
        if (n - k) < len(g_rev):
            gradient[n - k - 1] += g_rev[n - k - 1]
    
    # Central part
    for i in range(n):
        for k in range(1, n):
            if i + k < n:
                gradient[i] += w[k-1] * h[i + k]
            if i - k >= 0:
                gradient[i] += w[k-1] * h[i - k]
    
    # correct for double counting in convolution:
    # we only wanted to sum (i+k) < n and (i-k) >=0
    # and the terms where k is even are added twice exactly
    # anyway let's be very careful here
    
    # Fix with new direct form
    # g = conv(h, w) later comes with a mismatch of indices for two directions
    # a more direct way to get the gradient is:
    # at i, terms are:
    # w_k * h_{i+k} for k < n-i
    # w_k * h_{i-k} for k <= i
    g = np.zeros(n)
    for i in range(n):
        for k in range(1, n - i):  # only account when i+k < n
            g[i] += w[k - 1] * h[i + k]
        for k in range(1, i + 1):  # only account when i-k >= 0
            g[i] += w[k - 1] * h[i - k]
    
    return -np.sum(w * c), -g

def armijo_line_search(h, grad, beta, alpha=0.5):
    """Perform Armijo line search"""
    tau = 1e-1  # initial step size
    min_alpha = 1e-12
    while tau > min_alpha:
        h_new = project_onto_bounded_simplex(h - tau * grad)
        if len(h_new) != len(h):
            tau = tau / 2.0
            continue
        # attempt computation from actual smooth function
        obj, _ = smooth_obj_and_grad(h_new, beta)
        grad_norm_sq = np.dot(grad, grad)
        if obj <= smooth_obj_and_grad(h, beta)[0] - 0.1 * tau * grad_norm_sq:
            return h_new, tau
        tau = tau / 2.0
    return h, tau

def parallel_descent_step(h, beta, alpha):
    """Perform one projected gradient descent step"""
    obj, grad = smooth_obj_and_grad(h, beta)
    h_new, alpha_eff = armijo_line_search(h, grad, beta, alpha)
    return h_new, beta, alpha_eff
