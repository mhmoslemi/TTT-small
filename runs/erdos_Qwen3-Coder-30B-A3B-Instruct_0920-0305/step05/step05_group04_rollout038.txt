import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def conv_fft(a, b):
    """Compute linear convolution using FFT with zero padding"""
    n = len(a)
    m = len(b)
    pad_len = n + m - 1
    A = np.fft.fft(a, pad_len)
    B = np.fft.fft(b, pad_len)
    conv_result = np.fft.ifft(A * B).real[:pad_len]
    return conv_result

def project_onto_bounded_simplex(x, target_sum=0.5):
    """Project x onto the simplex with sum = target_sum and bounds [0,1]"""
    x = np.array(x)
    n = len(x)
    # Clip to [0,1]
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-12:
        return x
    
    # Bisection for lambda
    def sum_with_shift(lam):
        return np.sum(np.clip(x + lam, 0, 1))
    
    # Find appropriate lambda via bisection
    low, high = -1.0, 1.0
    while abs(sum_with_shift(high) - target_sum) > 1e-12:
        high *= 2.0
    if sum_with_shift(high) < target_sum:
        return x
    while abs(high - low) > 1e-12:
        mid = (low + high) / 2.0
        if sum_with_shift(mid) > target_sum:
            high = mid
        else:
            low = mid
    
    lam = (low + high) / 2.0
    return np.clip(x + lam, 0, 1)

def soft_weights(c, beta):
    """Compute softmax weights for soft minimum with temperature beta"""
    c_max = np.max(c)
    exp_vals = np.exp(-beta * (c - c_max))
    return exp_vals / np.sum(exp_vals)

def smooth_obj_and_grad(h, beta):
    """Return Φβ(h) and ∇Φβ(h)"""
    n = len(h)
    # Compute convolutions (equivalent to c = conv(h,h))
    c = conv_fft(h, h)
    c = c[1:n]  # Keep only non-zero shifts 1 to n-1
    
    # Compute weights
    w = soft_weights(c, beta)
    # Gradient of Φβ = - sum w_k * ∇c_k
    # ∇c_k[i] = h[i+k] + h[i-k] for shift k
    # Use FFT trick to compute u = conv(h,w), v = conv(h,rev(w))
    w_rev = w[::-1]
    u = conv_fft(h, w)
    v = conv_fft(h, w_rev)
    grad = -(u + v)
    grad = grad[:n]  # Trim to length
    
    # Φβ = (1/β)*log(sum(exp(-β*c_k)))
    # But we already computed normalized weights w
    min_c = np.min(c)
    Phi = (1.0 / beta) * np.log(np.sum(np.exp(-beta * (c - min_c)))) + min_c
    
    return Phi, grad

def line_search(h, g, beta, max_iter=20):
    """Armijo backtracking line search"""
    alpha = 0.5
    t = 0.25
    Phi_prev, _ = smooth_obj_and_grad(h, beta)
    for _ in range(max_iter):
        h_new = project_onto_bounded_simplex(h - alpha * g)
        Phi_new, _ = smooth_obj_and_grad(h_new, beta)
        if Phi_new <= Phi_prev - 0.1 * alpha * np.dot(g, g):
            return h_new, alpha
        alpha *= t
    return h, alpha

def gradient_descent_with_beta_continuation(h0, beta0=1.0, beta_max=200.0, alpha0=0.5, gamma=2.0, delta=0.9, max_iter=500):
    """Main optimization loop"""
    h = h0.copy()
    beta = beta0
    alpha = alpha0
    prev_Phi = float('inf')
    
    for iter_count in range(max_iter):
        # compute Φβ and its gradient
        Phi, grad = smooth_obj_and_grad(h, beta)
        
        # termination criterion based on improvement
        if abs(Phi - prev_Phi) < 1e-10 and np.linalg.norm(grad) < 1e-9:
            break
            
        # line search
        h_new, alpha = line_search(h, grad, beta)
        h = h_new
        
        # update Beta
        if beta < beta_max:
            beta = min(beta * gamma, beta_max)
            alpha = alpha * delta  # reduce step size slowly
            
        # check progress
        prev_Phi = Phi
        if iter_count % 50 == 0:  # occasionally check how progess is
            pass
    
    return h
