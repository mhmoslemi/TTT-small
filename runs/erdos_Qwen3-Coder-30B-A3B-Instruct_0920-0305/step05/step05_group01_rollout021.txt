import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60
DX = 2.0 / 120  # dx for integration

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    low, high = -2.0, 2.0
    for _ in range(50):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target_sum) < 1e-12:
            return y
        if s > target_sum:
            high = mid
        else:
            low = mid
    y = np.clip(x + low, 0.0, 1.0)
    return y

def evaluate_c5(h):
    """Evaluate C5 using correlation"""
    corr = correlate(h, 1.0 - h, mode='full')
    return corr.max() * DX

def gradient_smoothing(h, alpha):
    """Compute gradient of smoothed max correlation"""
    n = len(h)
    # correlation of h with (1-h)
    cov_full = correlate(h, 1.0 - h, mode='full')
    I = cov_full  # shape 2n-1
    I_max = I.max()
    # Compute weights for softmax
    exp_vals = np.exp(alpha * (I - I_max))
    w = exp_vals / exp_vals.sum()
    # Compute gradient as weighted sum of per-lag grads
    grad = np.zeros(n)
    # Pre-compute arrays
    h_arr = h
    one_minus_h = 1.0 - h_arr
    
    # Note: Matrix indexing where i is at index i. Lag = shift of indices
    for idx, p_val in enumerate(w):
        lag = idx - (n - 1)  # ℓ from -(n-1) to n-1
        if lag >= 0:
            # contribution (1 - h[i+lag]) for i=0..n-1-lag
            if n - lag > 0:
                grad[:n-lag] += p_val * one_minus_h[lag:]
            # edge contribution 1 for i >= n-lag (since out-of-range h = 0)
            if lag > 0:
                grad[n-lag:] += p_val * 1.0
            # contribution - h[i-lag] for i=lag..n-1
            if lag > 0:
                grad[lag:] -= p_val * h_arr[:n-lag]
            else:
                # lag==0 part 
                grad -= p_val * h_arr
        else:
            d = -lag
            # (1 - h[i-d]) for i=d..n-1
            if n - d > 0:
                grad[d:] += p_val * one_minus_h[:n-d]
            # edge contributions 1 for i=0..d-1 (since left-out of range h[i+ℓ] = 0)
            grad[:d] += p_val * 1.0
            # - h[i+d] for i=0..n-d-1
            grad[:n-d] -= p_val * h_arr[d:]
    
    return grad

def descent_step(h, eta, alpha):
    """Perform one descent step"""
    grad = gradient_smoothing(h, alpha)
    h_new = project_to_sum_box(h - eta * grad, TARGET_SUM)
    c5_new = evaluate_c5(h_new)
    return h_new, c5_new

def gradient_descent(h_start, time_limit, seed):
    """Run gradient descent with smoothing"""
    np.random.seed(seed)
    h = h_start.copy()
    c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5
    t0 = time.time()
    
    # Schedule parameters
    alpha_start = 0.5
    alpha_end = 50.0
    eta_start = 0.1
    
    it = 0
    max_iters = 10000
    while time.time() - t0 < time_limit and it < max_iters:
        progress = min(it / 10000, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        eta = eta_start * (0.995 ** it)
        
        h_new, c5_new = descent_step(h, eta, alpha)
        if c5_new <= c5:
            h = h_new
            c5 = c5_new
        else:
            # Diversity reset if no progress for 100 steps
            if it % 100 == 0:
                h = h_start + np.random.normal(0, 0.01, len(h))
                h = project_to_sum_box(h, TARGET_SUM)
                c5 = evaluate_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        it += 1
    return best_h, best_c5

def local_refine(h):
    """Refine solution using SLSQP optimizer"""
    def objective(h):
        return evaluate_c5(h)
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h))]
    
    try:
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
    except:
        pass
    return h, evaluate_c5(h)
