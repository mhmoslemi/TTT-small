import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_box_sum(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max(h, beta):
    """Smooth approximation of max using log-sum-exp"""
    corr = compute_corr_fft(h)
    c5_vals = dx * corr  # This includes all lags (mode='full')
    c5_vals = c5_vals[0:len(h)]  # Use only lags 0 to n-1
    
    c5_vals = np.array(c5_vals)
    # Numerical stability: subtract max before exponentiation
    c5_max = np.max(c5_vals)
    exp_vals = np.exp(beta * (c5_vals - c5_max))
    return (c5_max + np.log(np.sum(exp_vals)) / beta)

def smooth_gradient(h, beta):
    """Gradient of smooth max (weighted sum of individual gradients)"""
    corr = compute_corr_fft(h)
    c5_vals = dx * corr  # Full correlation (all lags)
    # Only take relevant lags (0 to 119), corresponding to index 0 to 119 in full corr
    c5_vals_relevant = c5_vals[0:len(h)]
    
    # Compute weights (numerically stable)
    c5_vals_relevant = np.array(c5_vals_relevant)
    max_c5 = np.max(c5_vals_relevant)
    exp_vals = np.exp(beta * (c5_vals_relevant - max_c5))
    weights = exp_vals / np.sum(exp_vals)
    
    # Gradient computation: D(C5) = 1 - shift_plus - shift_minus
    grad = np.zeros_like(h)
    for i in range(len(h)):
        # For each lag l (0 <= l < len(h)), compute gradient contribution
        # Weights compensate for various lags
        for l in range(len(h)):
            weight = weights[l]
            if i + l < len(h):
                grad[i] += weight * (-h[i + l])
            if i - l >= 0:
                grad[i] += weight * (-h[i - l])
        grad[i] += weight  # constant term (1)
    
    return grad

def projected_gradient_descent(h_start, beta_schedule, max_iter=2000, step_size=0.4, gamma=1e-4):
    """Projected gradient descent for smooth max with agreement steps"""
    h = h_start.copy()
    n = len(h)
    
    # Convert beta_schedule to list
    betas = list(beta_schedule)
    
    iter_count = 0
    beta = betas[0]
    
    for beta_idx, beta in enumerate(betas):
        if iter_count >= max_iter:
            break
        step = step_size / (1 + gamma * iter_count)
        
        for _ in range(100):  # Optimization per beta
            if iter_count >= max_iter:
                break
            grad = smooth_gradient(h, beta)
            h_new = h - step * grad
            
            # Project onto box and sum constraints
            h_new = project_to_box_sum(h_new, TARGET_SUM)
            
            # Check if to continue or test step
            if abs(smooth_max(h_new, beta) - smooth_max(h, beta)) < 1e-12:
                step *= 0.5 # Reduce step size to avoid infinite loop
                if step < 1e-8:
                    break
            else:
                h = h_new
                iter_count += 1
    
    return h

def first_stage_optimization(h_initial, beta_schedule, max_iter=3000):
    """Optimization at coarse resolution"""
    h_opt = projected_gradient_descent(h_initial, beta_schedule, max_iter)
    return h_opt
