import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    # Parameters
    L = 12   # Truncation order for Fourier series
    M = 4 * L  # Number of sampling points for box constraints
    
    # Generate sampling points
    x_sample = np.linspace(0, 2, M, endpoint=False)
    
    # Create Fourier basis matrices
    cos_matrix = np.array([[np.cos(np.pi * l * x) for x in x_sample] for l in range(1, L+1)])
    sin_matrix = np.array([[np.sin(np.pi * l * x) for x in x_sample] for l in range(1, L+1)])
    
    # Decision variables
    a = cp.Variable(L)
    b = cp.Variable(L)
    s = cp.Variable(L, name='s')
    t = cp.Variable(1, name='t')
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    
    # 1. Box constraints: 0 <= h(x) <= 1 for sampled points
    h_vals = 0.5 + cos_matrix.T @ a + sin_matrix.T @ b
    box_constraints = [h_vals >= 0, h_vals <= 1]
    
    # 2. s >= a^2 + b^2 (SOC constraints)
    soc_constraints = []
    for i in range(L):
        soc_constraints.append(cp.norm(cp.hstack([2*a[i], 2*b[i], s[i] - 1])) <= s[i] + 1)
    
    # 3. Toeplitz PSD constraint for Chebyshev bound
    # Construct Toeplitz matrix T(t, s)
    # T[i,j] = t if i==j, -s[|i-j|] otherwise
    Toeplitz_rows = [t] + [-s[i] for i in range(L)]
    Toeplitz = cp.bmat([[Toeplitz_rows[j-i] if i <= j else Toeplitz_rows[i-j] 
                       for j in range(L+1)] for i in range(L+1)])
    
    psd_constraint = [Toeplitz >> 0]  # PSD constraint
    
    # Combine all constraints
    constraints = box_constraints + soc_constraints + psd_constraint
    
    # Formulate and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
