import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Phase 1: Binary MILP optimization
    best_h_bin = None
    best_c5_bin = float('inf')
    print(f"Starting MILP phase with n={n_points}")
    
    try:
        # Create binary variables
        h_bin = cp.Variable(n_points, boolean=True)
        
        # Create auxiliary variables for product terms
        # y[i,k] represents h[i] * h[i+k] when defined
        y_vars = {}
        all_pairs = {}
        shift_ranges = {}
        
        # Precompute shift ranges
        for k in range(-(n_points-1), n_points):
            pairs = []
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    pairs.append((i, j))
            if pairs:
                all_pairs[k] = pairs
                y_vars[k] = cp.Variable(len(pairs), boolean=True)
        
        # Constraints: product linearization
        constraints = []
        for k, pairs in all_pairs.items():
            for idx, (i, j) in enumerate(pairs):
                y = y_vars[k][idx]
                h_i = h_bin[i]
                h_j = h_bin[j]
                
                # McCormick constraints
                constraints += [
                    y <= h_i,
                    y <= h_j,
                    y >= h_i + h_j - 1
                ]
        
        # Area constraint
        constraints.append(cp.sum(h_bin) == n_points / 2)
        
        # Epigraph variable
        tau = cp.Variable()
        
        # Overlap constraints
        for k, pairs in all_pairs.items():
            y_sum = cp.sum(y_vars[k])
            overlap = cp.sum(h_bin) - y_sum
            constraints.append(overlap <= tau)
        
        # Objective
        objective = cp.Minimize(tau)
        prob = cp.Problem(objective, constraints)
        
        # Solve with time limit
        prob.solve(solver=cp.CBC, max_seconds=budget_s * 0.25)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_bin_val = h_bin.value
            if h_bin_val is not None:
                h_bin_val = np.array(h_bin_val)
                # Convert to float for continuous phase
                h_cont = h_bin_val.astype(float)
                c5_val = evaluate_c5(h_cont)
                best_h_bin = h_cont.copy()
                best_c5_bin = c5_val
                print(f"MILP found C5 = {c5_val}")
        else:
            print(f"MILP status: {prob.status}")
    except Exception as e:
        print(f"MILP failed with exception: {e}")
        pass

    # If MILP failed, use initial construction
    if best_h_bin is None:
        h_cont = np.array(initial_h_values).astype(float)
        scale_factor = (n_points / 2) / np.sum(h_cont)
        h_cont *= scale_factor
        best_h_bin = h_cont.copy()
        best_c5_bin = evaluate_c5(h_cont)
        print(f"Using initial construction C5 = {best_c5_bin}")

    # Phase 2: Binary hill-climbing
    print("Starting binary hill-climbing phase")
    
    def evaluate_shifts(h):
        corr = correlate(h, 1 - h, mode="full")
        half_n = n_points - 1
        return np.max(corr[half_n:2*half_n+1])
    
    current_h = best_h_bin.copy()
    current_c5 = best_c5_bin
    
    # Simulate the binary hill climbing
    # Create a binary version of current_h for efficient swapping
    binary_h = (current_h > 0.5).astype(int)
    
    best_binary_h = binary_h.copy()
    best_binary_c5 = current_c5
    
    swapped = True
    max_iterations = 5000
    iteration_counter = 0
    
    while swapped and (time.time() - start_time) < budget_s * 0.6 and iteration_counter < max_iterations:
        swapped = False
        iteration_counter += 1
        
        # Try all swaps of 1 and 0
        for i in range(n_points):
            if binary_h[i] == 1:
                # Find all 0s to swap with
                for j in range(n_points):
                    if binary_h[j] == 0:
                        # Simulate the swap
                        test_h = binary_h.copy()
                        test_h[i] = 0
                        test_h[j] = 1
                        
                        # Evaluate new C5
                        test_c5 = evaluate_shifts(test_h.astype(float))
                        
                        # If better, accept the swap
                        if test_c5 < best_binary_c5:
                            binary_h = test_h.copy()
                            best_binary_c5 = test_c5
                            swapped = True
                            break
            if swapped:
                break
    
    # Convert back to continuous
    current_h = binary_h.astype(float)
    # Re-normalize to correct area
    current_h *= (n_points / 2) / np.sum(current_h)
    current_c5 = evaluate_shifts(current_h)
    
    if current_c5 < best_binary_c5:
        best_binary_c5 = current_c5
        best_binary_h = current_h.copy()
    
    print(f"Binary hill-climbing improves C5 to {best_binary_c5}")
    
    # Phase 3: Continuous refinement
    print("Starting continuous refinement phase")
    
    # Initialize with the best binary solution
    h_best = best_binary_h.copy()
    c5_best = best_binary_c5
    
    # Helper function for projection onto capped simplex
    def project_simplex(v, target_sum=1.0):
        # Ensure values are in [0,1] and sum to target_sum
        v = np.clip(v, 0, 1)
        v_sum = np.sum(v)
        if abs(v_sum - target_sum) < 1e-10:
            return v
        
        # Water-filling approach
        v_sorted = np.sort(v)[::-1]
        cumsum = np.cumsum(v_sorted)
        n = len(v_sorted)
        # Find the threshold
        threshold_idx = -1
        for i in range(n):
            if v_sorted[i] * (i + 1) - cumsum[i] > target_sum:
                threshold_idx = i
                break
        if threshold_idx == -1:
            # All values can be increased
            threshold = (target_sum - cumsum[n-1]) / n
            result = np.clip(v_sorted + threshold, 0, 1)
        else:
            # Find the exact threshold
            threshold = (target_sum - cumsum[threshold_idx]) / (threshold_idx + 1)
            result = np.clip(v_sorted - threshold, 0, 1)
        # Replaced values based on original indices
        result_sorted = np.sort(result)[::-1]
        result_ordered = np.zeros_like(v)
        indices = np.argsort(v)[::-1]
        for i, idx in enumerate(indices):
            if i < len(result_sorted):
                result_ordered[idx] = result_sorted[i]
            else:
                result_ordered[idx] = 0
        return result_ordered

    # Sub-gradient based continuous refinement
    h = h_best.copy()
    c5 = c5_best
    eta = 0.5 * dx
    num_iterations = 0
    last_improvement = 0
    
    while (time.time() - start_time) < budget_s * 0.95 and num_iterations < 10000:
        num_iterations += 1
        # Compute overlap for all shifts
        corr = correlate(h, 1 - h, mode="full")
        half_n = n_points - 1
        overlaps = corr[half_n:2*half_n+1]
        max_overlap = np.max(overlaps)
        active_shifts = np.nonzero(overlaps == max_overlap)[0]
        active_shifts = active_shifts - half_n  # Convert back to shift indices
        
        # Compute weights for the sub-gradient
        w = np.zeros(n_points)
        for k in active_shifts:
            if k >= 0:
                # Sum up weights over valid overlaps
                # w[0:n-k] += 1
                for i in range(n_points - k):
                    w[i] += 1
            else:
                # k < 0
                # w[-k:n] += 1
                for i in range(n_points + k):
                    w[i - k] += 1
        
        # Compute sub-gradient: g = dx * (1− w⋆h−w⋆h_rev)
        # Efficient convolution
        w_rev = w[::-1]
        conv1 = correlate(w, h, mode='full')[n_points-1:2*n_points-1]
        conv2 = correlate(w_rev, h, mode='full')[n_points-1:2*n_points-1]
        gradient = dx * (1 - conv1 - conv2)
        # Update step
        h_new = h - eta * gradient
        # Project onto capped simplex
        h_new = project_simplex(h_new, n_points / 2)
        # Compute new overlap 
        test_c5 = evaluate_c5(h_new)
        
        if test_c5 < c5:
            h = h_new.copy()
            c5 = test_c5
            last_improvement = 0
            if c5 < c5_best:
                c5_best = c5
                h_best = h.copy()
            eta = min(0.5 * dx, eta)  # Reduce step size
        else:
            last_improvement += 1
            if last_improvement > 100:
                eta = max(1e-8, eta * 0.5)  # Further reduce step size
                
        if eta < 1e-8:
            break

    # Phase 4: Symmetrisation
    h_sym = (h_best + h_best[::-1]) / 2.0
    h_sym = project_simplex(h_sym, n_points / 2)
    c5_sym = evaluate_c5(h_sym)
    if c5_sym < c5_best:
        h_best = h_sym
        c5_best = c5_sym
    
    print(f"Final C5 = {c5_best}")
    return (h_best.tolist(), c5_best, n_points)
