TARGET_SUM = 60.0

def evaluate_c5(h):
    """Evaluate C5 using correlation"""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation using scipy.signal.correlate
    corr = correlate(h, 1 - h, mode="full")
    # The correlation is indexed from -n+1 to n-1
    # We want the maximum of the positive lags [0, n-1]
    max_corr = np.max(corr[n-1:])  # Take the last n elements (non-negative lags)
    return dx * max_corr

def project_to_sum_box(x, target_sum):
    """Project to sum=target_sum with box constraints [0,1]"""
    # Simple approach: clip and rescale
    y = np.clip(x, 0, 1)
    current_sum = np.sum(y)
    if abs(current_sum - target_sum) < 1e-10:
        return y
    # Simple rescaling approach
    if current_sum > 0:
        scale = target_sum / current_sum
        y = np.clip(y * scale, 0, 1)
    return y

    # Update
