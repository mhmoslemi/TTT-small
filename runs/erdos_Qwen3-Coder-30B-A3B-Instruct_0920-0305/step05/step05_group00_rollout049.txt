def compute_corr_fft(h):
    """Computed correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_objective(h, beta=1.0):
    """Smoothes the max with log-sum-exp"""
    corr = compute_corr_fft(h)
    # Numerically stable log-sum-exp computation
    max_corr = np.max(corr)
    exp_corr = np.exp(beta * (corr - max_corr))
    log_sum = max_corr + np.log(np.sum(exp_corr)) / beta
    return log_sum

def smooth_max_gradient(h, beta=1.0):
    """Compute gradient of log-sum-exp of correlations"""
    corr = compute_corr_fft(h)
    max_corr = np.max(corr)
    exp_corr = np.exp(beta * (corr - max_corr))
    weights = exp_corr / np.sum(exp_corr)
    # Construct gradient vector ∇Sβ = ∑ w_l ∇C_l
    grad = np.zeros_like(h)
    for j, w in enumerate(weights):
        # Compute C_l gradient efficiently from the equivalent convolution
        # ∇C_l = 1 - shift_plus - shift_minus
        shift_plus = np.roll(h, -j) if j > 0 else np.zeros_like(h)
        shift_minus = np.roll(h, j) if j > 0 else np.zeros_like(h)
        grad += w * (1 - shift_plus - shift_minus)
    return grad

def projected_gradient_descent(h0, beta_schedule, max_iter_per_step, alpha0=0.4, gamma=1e-4):
    """Projected gradient descent with beta continuation"""
    h = h0.copy()

        for _ in range(num_iter):
            if i == 0:
                # For low β, use explicit batch computation
                grad = smooth_max_gradient(h, beta)
            else:
                grad = smooth_max_gradient(h, beta)
            
            # Adaptive stepsize
            alpha = alpha0 / (1 + gamma * _)
            h_new = h - alpha * grad
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            # Simple backtracking if not improving
            c_old = smooth_max_objective(h, beta)
            c_new = smooth_max_objective(h_new, beta)
            if c_new >= c_old:
                alpha = alpha / 2.0
                h_new = h - alpha * grad
                h_new = project_to_sum_box(h_new, TARGET_SUM)
            h = h_new
    return h
