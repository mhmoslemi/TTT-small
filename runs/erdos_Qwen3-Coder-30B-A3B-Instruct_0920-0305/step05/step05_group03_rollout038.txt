import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Begin with initial construction (it's a list at global scope)
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Ensure we have an even number of points for symmetry reduction
    if n_points % 2 != 0:
        n_points = n_points - 1
        h0 = h0[:n_points]
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Start with k=0 only
    active_k = [0]
    K = set(active_k)
    
    # Need to modify values to ensure the sum is exactly n/2
    scale_factor = 1.0 / (np.sum(h0) * dx)
    h0 = h0 * scale_factor
    # Verify that we have the right sum
    h0_sum = np.sum(h0) * dx
    if abs(h0_sum - 1.0) > 1e-10:
        h0 = h0 * (1.0 / h0_sum)
    
    h_current = h0.copy()
    
    # Store the solution that yielded the least C5 so far
    best_feasible_h = h_current.copy()
    best_feasible_c5 = evaluate_c5(h_current)
    
    loop_count = 0
    while (time.time() - start_time) < budget_s * 0.93:
        loop_count += 1
        # Limit number of iterations
        if loop_count > 250:
            break
            
        # We will solve the LP using CVXPY with GLOP solver
            
        # Initial CVXPY variables
        h = Variable(n_points)
        tau = Variable(1)
        
        # Prepare auxiliary variables for bilinear products
        # y[i][k] will approximate h[i] * h[i+k]
        y_vars = {}
        constraints = []
        
        # Box constraints
        constraints += [0 <= h, h <= 1]
        # Sum constraint
        constraints += [sum(h) == n_points / 2]
        # Symmetry: h[i] == h[n-1-i]
        for i in range(n_points // 2):
            constraints += [h[i] == h[n_points - 1 - i]]
        
        # Add constraints for active shifts
        for k in active_k:
            if k not in y_vars:
                y_vars[k] = Variable(n_points)
            
            # Add McCormick envelopes for y[i][k]
            # y[i][k] >= 0
            # y[i][k] <= h[i]
            # y[i][k] <= h[i+k] (will pad to zero if not in range)
            # y[i][k] >= h[i] + h[i+k] - 1
            
            # We consider the valid indices for this shift k:
            # we don't need to go too far for the padding case
            y_vars[k] = Variable(n_points)
            
            for i in range(n_points):
                # For i+k we must check that it's a valid index
                j = i + k
                if 0 <= j < n_points:
                    constraints += [y_vars[k][i] >= 0]
                    constraints += [y_vars[k][i] <= h[i]]
                    constraints += [y_vars[k][i] <= h[j]]
                    constraints += [y_vars[k][i] >= h[i] + h[j] - 1]
                else:
                    # Padding: would be out of bounds
                    # Force y = 0 (as h out-of-bounds is 0)
                    constraints += [y_vars[k][i] == 0]
            
            # Add constraint on tau: tau >= sum(h) - sum(y[k])
            # The sum over i h[i] is a constant (n/2), AND so is sum y[i,k]
            sum_h = np.sum(h0) * dx  # We know h0 is properly scaled already, used in a hack way
            # Actually, just use the sum of h determined in our model, because it might be updated
            sum_y = sum(y_vars[k])
            constraint = tau >= sum_h - sum_y
            constraints.append(constraint)
        
        # Objective
        objective = Minimize(tau)
        
        # Build the problem
        prob = Problem(objective, constraints)
        
        # Try to solve with GLOP if available, fallback to CBC
        try:
            prob.solve(solver=GUROBI, verbose=False)
            # Sometimes CVXPY fails with a debug-only solver, we catch.
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_val = h.value
                tau_val = tau.value
            else:
                # If we cannot solve directly, fallback to CBC with prob options
                prob.solve(solver=CBC, verbose=False,  max_nodes=500)
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    h_val = h.value
                    tau_val = tau.value
                else:
                    break  # Failed to solve even with fallback
        except:
            # Use default fallback to CBC
            try:
                prob.solve(solver=CBC, verbose=False)
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    h_val = h.value
                    tau_val = tau.value
                else:
                    break
            except:
                break  # Still exhausted attempts
        
        if not h_val.shape or len(h_val) != n_points:
            break
            
        # Post-processing or cleanup of floating point values
        h_val = np.asarray(h_val).flatten()
        h_val = np.clip(h_val, 0, 1)
        # Re-normalize to sum to n/2
        h_sum = np.sum(h_val) * dx
        if abs(h_sum - 1.0) > 1e-10:
            h_val = h_val * (1.0 / h_sum)
        
        h_current = h_val.copy()
        
        # Compute the true C5 using the actual correlation
        actual_c5 = evaluate_c5(h_current)
        
        # Check if it's an improvement
        if actual_c5 < best_feasible_c5:
            best_feasible_h = h_current.copy()
            best_feasible_c5 = actual_c5
        
        # Determine the maximum overlap using full correlation
        corr = correlate(h_current, 1 - h_current, mode="full")
        max_k = np.argmax(corr)
        max_o = corr[max_k]
        
        k_max = max_k - (n_points - 1)  # Convert from 0-based full indexing to shifted k
        if k_max not in K:
            K.add(k_max)  # Add new k to active set
            active_k = sorted(list(K))  # Keep sorted

    # Final polishing with projected gradient descent
    h_final = best_feasible_h.copy()
    current_c5 = best_feasible_c5
    
    max_steps = 2000
    eta = 0.5 * dx
    
    for step in range(max_steps):
        if (time.time() - start_time) > budget_s * 0.99:
            break
            
        # Compute sub-gradient for C5
        # Since C5 = max_k O_k(h), we compute all O_k and find active ones
        all_overs = []
        for k in range(-(n_points - 1), n_points):
            # Reconstruct correlation via convolution (as used in evaluate)
            shift = k + (n_points - 1)
            t = 0.0
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    t += h_final[i] * h_final[j]
            corr = 1 - t / dx
            all_overs.append(corr)
        
        all_overs = np.array(all_overs)
        max_over = np.max(all_overs)
        active_ks = np.where(np.abs(all_overs - max_over) < 1e-10)[0]
        
        # Find which shift in active set we need to compute
        g = np.zeros(n_points)
        if len(active_ks) == 0:
            active_ks = [np.argmax(all_overs)]
        
        # Gradient of O_k w.r.t. h
        for k in active_ks:
            k_real = k - (n_points - 1)
            # Compute gradient
            g += 1.0  # Partial O_k_partial(h_i) = 1 for all i
            # Plus contribution from the h_{i+k} term
            for i in range(n_points):
                j = i + k_real
                if 0 <= j < n_points:
                    g[i] -= 1.0  # Because of coefficient -h_{i+k} from the term 
                else:
                    # This term is zero due to padding        
                    pass
        
        # Average the subgradient over active sets
        g = g / len(active_ks)
        
        # Gradient step (projected)
        h_new = h_final - eta * g
        
        # Projection onto simplex: [0,1], sum = 1/2 * n
        # Soft projection (water-filling approach)
        # Project first to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        #Then use the sum constraint: h_new = h_new + lambda
        #Project it to sum to n/2 (dx = 2/n and sum h * dx = 1 => sum h = n/2)
        sum_h = np.sum(h_new)
        diff = (n_points // 2) - sum_h
        # Approximate λ from difference, 'water fill'
        # In this simple case just spread the diff均匀
        add = diff / n_points
        h_new = h_new + add
        
        # Possibly enforce symmetry again (if beneficial for gradient convergence)
        h_symmetric = (h_new + h_new[::-1]) / 2
        
        # Ensure the negative doesn't go below 0 or above 1  
        h_new = np.clip(h_symmetric, 0, 1)        

        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Evaluate the new value of C5
        c5_new = evaluate_c5(h_new)
        
        if c5_new < current_c5:
            h_final = h_new.copy()
            current_c5 = c5_new
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_final.copy()
        else:
            # Reduce step size if no improvement
            eta /= 2.0
            if eta < 1e-8:
                break
    
    # Optional: binary rounding
    h_bin = np.zeros(n_points)
    h_sort = np.argsort(h_final)[::-1]
    h_bin[h_sort[:n_points // 2]] = 1.0
    # Now improve with local swaps
    c5_bin = evaluate_c5(h_bin)
    if c5_bin < current_c5:
        current_c5 = c5_bin
        best_h = h_bin.copy()
    
    return (best_h.tolist(), current_c5, n_points)
