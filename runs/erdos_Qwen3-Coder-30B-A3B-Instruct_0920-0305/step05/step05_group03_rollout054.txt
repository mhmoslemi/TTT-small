import numpy as np
from cvxpy import *
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # === 1. MILP Binary Optimization ===
    # Variables
    h = Variable(n_points, boolean=True)
    tau = Variable()

    # Constraints
    constraints = [sum(h) == n_points / 2]
    constraints.append(tau >= 0)

    # For each shift k, compute overlap
    # We'll enumerate valid pairs (i,j) where j = i + k and both indices are in [0,n)
    shifts = list(range(-n_points + 1, n_points))
    y_vars = {}  # y_vars[(i,k)] -> binary variable for h[i] * h[i+k]

    # Compute all valid (i, i+k) pairs per shift k
    shift_pairs = {}
    for k in shifts:
        pairs = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                pairs.append((i, j))
        shift_pairs[k] = pairs

    # Binary constraints for y variables
    for k in shifts:
        for i, j in shift_pairs[k]:
            y = Variable(boolean=True)
            y_vars[(i, k)] = y
            # McCormick constraints for y = h[i] * h[j]
            constraints.append(y <= h[i])
            constraints.append(y <= h[j])
            constraints.append(y >= h[i] + h[j] - 1)

    # Build overlap expressions and constraints
    for k in shifts:
        pairs = shift_pairs[k]
        if pairs:
            overlap = sum(h[i] for i, j in pairs) - sum(y_vars[(i, k)] for i, j in pairs)
            constraints.append(overlap <= tau)

    # Objective
    objective = Minimize(tau)
    prob = Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver='CBC', time_limit=budget_s * 0.25, verbose=False)
        if h.value is not None:
            best_h_bin = np.array([float(x.value) for x in h]).round(6)
        else:
            # Fallback to random feasible binary mask
            best_h_bin = np.random.choice([0., 1.], size=n_points, p=[0.5, 0.5])
            while np.abs(np.sum(best_h_bin) * dx - 1.0) > 1e-10:
                best_h_bin = np.random.choice([0., 1.], size=n_points, p=[0.5, 0.5])
            best_h_bin = np.array(best_h_bin)
    except:
        # Fallback to random feasible binary mask
        best_h_bin = np.random.choice([0., 1.], size=n_points, p=[0.5, 0.5])
        while np.abs(np.sum(best_h_bin) * dx - 1.0) > 1e-10:
            best_h_bin = np.random.choice([0., 1.], size=n_points, p=[0.5, 0.5])
        best_h_bin = np.array(best_h_bin)

    c5_bin = evaluate_c5(best_h_bin)
    best_h = best_h_bin.copy()
    best_c5 = c5_bin

    # === 2. Hill Climbing on Binary Patterns ===
    def get_hill_climb_candidates(h0):
        candidates = []
        ones = np.where(h0 == 1)[0]
        zeros = np.where(h0 == 0)[0]
        for i in ones:
            for j in zeros:
                h_new = h0.copy()
                h_new[i] = 0
                h_new[j] = 1
                candidates.append(h_new)
        return candidates

    def hill_climb(h_start, budget_frac=0.2):
        h_curr = h_start.copy()
        c5_curr = evaluate_c5(h_curr)
        best_h = h_curr.copy()
        best_c5 = c5_curr
        iter_remaining = int(10000 * budget_frac)  # adjust based on remaining time
        while iter_remaining > 0 and (time.time() - start_time) < (budget_s * 0.9):
            candidates = get_hill_climb_candidates(h_curr)
            if not candidates:
                break
            best_candidate = h_curr.copy()
            best_cand_c5 = c5_curr
            for h_cand in candidates:
                cand_c5 = evaluate_c5(h_cand)
                if cand_c5 < best_cand_c5:
                    best_cand_c5 = cand_c5
                    best_candidate = h_cand
            if best_cand_c5 < c5_curr:
                h_curr = best_candidate
                c5_curr = best_cand_c5
                if c5_curr < best_c5:
                    best_c5 = c5_curr
                    best_h = h_curr.copy()
            else:
                # no improving swaps; stop
                break
            iter_remaining -= 1
        return best_h, best_c5

    try:
        h_hill, c5_hill = hill_climb(best_h, 0.1)
        if c5_hill < best_c5:
            best_h = h_hill.copy()
            best_c5 = c5_hill
    except:
        pass

    # === 3. Continuous refinement ===
    def project_simplex(v, target_sum):
        # Project v onto the capped simplex with sum = target_sum
        low = 0.0
        high = 1.0
        while high - low > 1e-15:
            mid = (low + high) / 2
            projected = np.clip(v + mid, 0, 1)
            if np.sum(projected) > target_sum:
                high = mid
            else:
                low = mid
        return np.clip(v + low, 0, 1)

    # Initialize continuous h from best binary pattern
    h_cont = best_h.astype(float)

    max_iter = int(2000 * 0.3)
    eta = 0.5 * dx
    no_improve_count = 0
    max_no_improve = 200

    for it in range(max_iter):
        if (time.time() - start_time) > budget_s * 0.95:
            break
        
        # Compute overlaps O_k
        corr_full = correlate(h_cont, 1 - h_cont, mode='full')
        max_idx = np.argmax(corr_full)
        max_val = corr_full[max_idx]
        active_shifts = np.where(np.abs(corr_full - max_val) < 1e-8)[0]

        # Compute weight vector w_k for active shifts
        w = np.zeros(n_points)
        for k in active_shifts:
            if k < n_points:
                w[k] = 1
            else:
                w[k - n_points] = 1

        # Gradient
        convolved = np.convolve(w, np.ones_like(h_cont), mode='full')
        if len(convolved) > n_points:
            convolved = convolved[:n_points]
        grad = dx * (np.ones_like(h_cont) - convolved - convolved[::-1])

        # Gradient step
        h_new = h_cont - eta * grad
        h_new = project_simplex(h_new, n_points / 2)

        # Symmetrization
        h_symm = (h_new + h_new[::-1]) / 2
        h_symm = project_simplex(h_symm, n_points / 2)

        c5_new = evaluate_c5(h_symm)
        if c5_new < best_c5:
            h_cont = h_symm.copy()
            best_c5 = c5_new
            best_h = h_symm.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > max_no_improve:
                # Reduce step size
                eta /= 2.0
                no_improve_count = 0
                if eta < 1e-9 * dx:
                    break

    # === 4. Optional up-sampling ===
    if (time.time() - start_time) < budget_s * 0.95 and n_points <= 48:
        try:
            n_double = 2 * n_points
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, n_double)
            h_up = np.interp(x_new, x_orig, best_h)
            h_up = h_up / np.sum(h_up) * n_double / 2
            # Refine the upsampled h
            h_cont_up = h_up.copy()
            max_iter = 500
            eta = 0.5 * (2.0 / n_double)
            for it in range(max_iter):
                if (time.time() - start_time) > budget_s * 0.95:
                    break
                corr_full = correlate(h_cont_up, 1 - h_cont_up, mode='full')
                max_idx = np.argmax(corr_full)
                max_val = corr_full[max_idx]
                active_shifts = np.where(np.abs(corr_full - max_val) < 1e-8)[0]
                w = np.zeros(n_double)
                for k in active_shifts:
                    if k < n_double:
                        w[k] = 1
                    else:
                        w[k - n_double] = 1
                convolved = np.convolve(w, np.ones_like(h_cont_up), mode='full')
                if len(convolved) > n_double:
                    convolved = convolved[:n_double]
                grad = (2.0 / n_double) * (
                    np.ones_like(h_cont_up) - convolved - convolved[::-1]
                )
                h_new = h_cont_up - eta * grad
                h_new = project_simplex(h_new, n_double / 2)
                h_symm = (h_new + h_new[::-1]) / 2
                h_symm = project_simplex(h_symm, n_double / 2)
                c5_new = evaluate_c5(h_symm)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_symm.copy()
                    h_cont_up = h_symm.copy()
                else:
                    eta *= 0.95  # decrease step size on stagnation
        except:
            pass

    return (best_h.tolist(), best_c5, n_points)
