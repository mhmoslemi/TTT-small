import numpy as np
import time

def project_to_box_simplex(v, target_sum):
    """Project vector v onto box [0,1] and simplex with sum = target_sum"""
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0, 1)

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value scaled by dx
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, alpha):
    """Compute gradient of smooth approximation to max correlation"""
    n = len(h)
    dx = 2.0 / n
    
    # Compute correlation values
    corr = np.correlate(h, 1 - h, mode="full")
    O = corr * dx
    O_max = np.max(O)
    
    # Compute stable weights
    exp_term = np.exp(alpha * (O - O_max))
    w = exp_term / np.sum(exp_term)
    
    # Compute gradient
    grad = np.zeros_like(h)
    
    # For each shift, compute contributions
    for idx, shift in enumerate(range(-n+1, n)):
        w_k = w[idx]
        if w_k == 0:
            continue
            
        # compute shift_right and shift_left vectors
        if shift >= 0:
            # shift_right: ones for first shift entries, then 1 - h[shift:]
            sr = np.empty(n)
            if shift > 0:
                sr[:shift] = 1.0
            sr[shift:] = 1.0 - h[shift:]
            # shift_left: zeros for first shift entries, then h[:n-shift]
            sl = np.empty(n)
            if shift > 0:
                sl[:shift] = 0.0
            sl[shift:] = h[:n-shift]
        else:
            s = -shift
            # shift_right: ones for first s entries, then 1 - h[:n-s]
            sr = np.empty(n)
            if s > 0:
                sr[:s] = 1.0
            sr[s:] = 1.0 - h[:n-s]
            # shift_left: zeros for last s entries, then h[s:]
            sl = np.empty(n)
            if n - s > 0:
                sl[:n-s] = h[s:]
            sl[n-s:] = 0.0
            
        grad += w_k * (sr - sl)
    
    # Apply dx factor from O transformation
    grad = grad * dx
    return grad

def optimize_h_random(n, max_iters=4000, alpha_start=5, alpha_end=30, lr0=0.5):
    """Optimize h with gradient descent using smooth approximation"""
    # Initialize h with random values
    h = np.random.rand(n)
    h = project_to_box_simplex(h, n / 2)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for it in range(max_iters):
        # Update alpha
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        
        # Compute gradient
        grad = compute_gradient(h, alpha)
        
        # Step size with decreasing schedule
        lr = lr0 / np.sqrt(it + 1)
        
        # Update h

        # Project onto box and simplex

        if best_c5 < 0.379:
            break
