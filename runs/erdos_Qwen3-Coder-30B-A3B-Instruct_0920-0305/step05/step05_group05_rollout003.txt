import numpy as np
import time

def project_capped_simplex(v, target_sum):
    """
    Project vector v onto the capped simplex: 0 <= x <= 1, sum(x) = target_sum
    Uses binary search on the Lagrange multiplier lambda.
    """
    v = np.asarray(v)
    low, high = np.min(v - 1), np.max(v)
    tol = 1e-10
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        projected = np.clip(v - mid, 0, 1)
        sum_projected = np.sum(projected)
        if sum_projected > target_sum:
            low = mid
        else:
            high = mid
        if abs(sum_projected - target_sum) < tol:
            break
    return np.clip(v - (low + high) / 2.0, 0, 1)

def compute_O_and_grad(h, alpha):
    """Compute the vector of overlap values and the gradient of the smoothed max"""
    n = len(h)
    dx = 2.0 / n
    # Compute all O_k values: O_k = dx * sum h[i] * (1 - h[i+k])
    O = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        if k < n:
            # Left shift: h[i] * (1 - h[i + k])
            o = np.sum(h[:n-k] * (1 - h[k:]))
        else:
            # Right shift: h[i] * (1 - h[i - k])
            o = np.sum(h[k-n+1:] * (1 - h[:n-k+1]))
        O[k] = o * dx
    
    # Compute softmax weights (numerically stable)
    max_O = np.max(O)
    exp_O = np.exp(alpha * (O - max_O))
    w = exp_O / np.sum(exp_O)
    
    # Compute gradient of smoothed max: sum_k w_k * (1 - h_{i+k} - h_{i-k})
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if k < n:
            # Left shift: -h_{i+k} + 1 * (1 - h_{i+k}) - h_{i-k}
            h_shifted = np.concatenate([np.zeros(k), h[:n-k]])
            grad += w[k] * (1 - h_shifted)
        else:
            # Right shift: -h_{i-k} + 1 * (1 - h_{i-k})
            h_shifted = np.concatenate([h[k-n+1:], np.zeros(n-k+1)])
            grad += w[k] * (1 - h_shifted)
    grad *= dx
    
    return O, grad

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 64
    dx = 2.0 / n
    target_sum = n / 2.0
    max_iters_per_restart = 4000
    subgrad_iters = 500
    alpha_start = 5.0
    alpha_end = 30.0
    lr_0 = 0.5
    
    best_h = None
    best_val = float('inf')
    
    # Euclidean projection onto the capped simplex
    def project(h):
        return project_capped_simplex(h, target_sum)
        
    def get_initial_h(type_str):
        if type_str == "random":
            h = np.random.uniform(0, 1, n)
        elif type_str == "sine":
            x = np.linspace(0, 2, n)
            h = 0.5 + 0.4 * np.sin(np.pi * x)
        elif type_str == "gaussian":
            x = np.linspace(0, 2, n)
            mu = 1.0
            sigma = 0.2
            h = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
        elif type_str == "triangular":
            x = np.linspace(0, 2, n)
            h = np.maximum(0, 1 - np.abs(x - 1))
        elif type_str == "binary":
            h = np.zeros(n)
            h[:n//2] = 1.0
        else:
            h = np.random.uniform(0, 1, n)
        return project(h)
    
    restart = 0
    while (time.time() - start_time) < budget_s * 0.95:
        # Start with different kinds of initializers
        init_types = ["random", "sine", "gaussian", "triangular", "binary"]
        init_type = init_types[restart % len(init_types)]
        h = get_initial_h(init_type)
        
        # Smoothed gradient descent phase
        for t in range(max_iters_per_restart):
            if (time.time() - start_time) > budget_s * 0.95:
                break
            # Anneal alpha
            alpha = alpha_start + (alpha_end - alpha_start) * t / max_iters_per_restart
            O, grad = compute_O_and_grad(h, alpha)
            lr = lr_0 / np.sqrt(t + 1)
            h = h - lr * grad
            h = project(h)
            
            # Optional symmetry enforcement every few steps
            if t % 10 == 0:
                h = (h + h[::-1]) / 2.0
                h = project(h)
                
            current_val = np.max(O)
            if current_val < best_val:
                best_val = current_val
                best_h = h.copy()
        
        # Sub-gradient polishing phase using exact max gradient
        if best_h is not None:
            h = best_h.copy()
            for t in range(subgrad_iters):
                if (time.time() - start_time) > budget_s * 0.95:
                    break
                # Find max shift
                O = np.correlate(h, 1 - h, mode="full") * dx
                k_max = np.argmax(O)
                # Compute exact gradient for that k
                grad_exact = np.zeros(n)
                if k_max < n:
                    # Left shift case
                    grad_exact[:n-k_max] -= h[k_max:]
                    grad_exact[k_max:] += (1 - h[:n-k_max])
                else:
                    # Right shift case
                    k = k_max - n + 1
                    grad_exact[k:] -= h[:n-k]
                    grad_exact[:k] += (1 - h[k:])
                lr = 0.02 / np.sqrt(t + 1)
                h = h - lr * grad_exact * dx
                h = project(h)
                current_val = np.max(O)
                if current_val < best_val:
                    best_val = current_val
                    best_h = h.copy()
        
        restart += 1
        if (time.time() - start_time) > budget_s * 0.95:
            break
    
    if best_h is None:
        # Fallback to area-balanced solution
        h = np.ones(n) * 0.5
        best_val = evaluate_c5(h)
    
    return (best_h.tolist(), float(best_val), n)
