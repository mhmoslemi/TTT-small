import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
n_fine = 240
dx_fine = 2.0 / n_fine

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_gradient(h, β):
    """Compute gradient of log-sum-exp smooth approximation of max correlation"""
    # Compute all correlations
    corr = compute_corr_fft(h)
    # For numerical stability, subtract maximum
    max_corr = np.max(corr)
    exp_corr = np.exp(β * (corr - max_corr))
    w = exp_corr / np.sum(exp_corr)
    
    # Compute gradient: Σ w_ℓ ∇C_ℓ(h)
    # ∇C_ℓ(h) = 1 - h_{i+ℓ} - h_{i-ℓ} 
    grad = np.zeros_like(h)
    for i in range(len(h)):
        # Current and shifted versions (zero-padded)
        for ℓ in range(len(h)):
            if i + ℓ < len(h):
                grad[i] += w[ℓ] * (1 - h[i + ℓ])
            if i - ℓ >= 0:
                grad[i] += w[ℓ] * (1 - h[i - ℓ])
            # When i - ℓ < 0, h[i - ℓ] = h[-(ℓ - i)] = 0 (zero padded)
    
    # We expect to subtract h_{i+ℓ} and h_{i-ℓ}, but we already have h_{i-ℓ} in the other loop
    # Recompute carefully:
    grad = np.zeros_like(h)
    # We need shifts for each lag ℓ, so let's compute separately for more clarity:
    for ℓ in range(len(h)):
        if ℓ == 0:
            continue # skip shift by 0
        # negative shift (left shift)
        h_shift_neg = np.roll(h, ℓ)
        h_shift_neg[:ℓ] = 0
        # positive shift (right shift)
        h_shift_pos = np.roll(h, -ℓ)
        h_shift_pos[-ℓ:] = 0
        # gradient corresponds to the coefficient of 1 - h_{i+ℓ} - h_{i-ℓ} 
        grad += w[ℓ] * (1 - h_shift_pos - h_shift_neg)
        
    # Add gradient from ℓ=0 to handle h_{i+0}
    # Wait, this is a conceptual mistake.
    # Let's compute it in another way.
    # We have C_ℓ = dx * sum(h_i * (1 - h_{i+ℓ})) = dx * (target - sum h_i * h_{i+ℓ})
    # and we want gradient ∇C_ℓ:
    # for each i: ∂C_ℓ/∂h_i = dx * (0 - h_{i+ℓ} - h_{i-ℓ} + 0) = -dx (h_{i+ℓ} + h_{i-ℓ})
    # However, for the formulation C_ℓ = dx * sum h_i (1-h_{i+ℓ}):  
    # ∂/∂h_i = dx (1 - h_{i+ℓ}) + dx h_i (-1) = dx (1 - h_{i+ℓ} - h_i)
    # But actually, more precisely:
    # For term term h_i * (1 - h_{i+ℓ}), we consider:
    # 1) ∂/∂h_i(h_i * (1 - h_{i+ℓ})) = (1 - h_{i+ℓ})    (this is independent of h_i)  
    # 2) ∂/∂h_{i+ℓ}(h_i * ( 1 - h_{i+ℓ})) = -h_i
    # In the full sum, we have:
    # ∇C_ℓ = dx * (vector of (1 - h_{i+ℓ}) for each i) - dx * (shifted h_i)  
    # ∇C_ℓ = dx (1 - h_{i+ℓ}) - dx h_{i-ℓ}  (shifted appropriately)
    
    # So this is actually the correct vectors:
    # We compute for each ℓ:
    # shift_l = [0] + [0]*ℓ + h[0:n-ℓ]  
    # shift_r = h[ℓ:] + [0]*ℓ
    # ∇(C_ℓ) = dx * (1 - shift_l - shift_r) 
    # But need to build the matrix-vector multiplication efficiently:
    # We note again that
    # C_ℓ = dx * sum_i h_i*(1 - h_{i+ℓ})
    # ∂C_ℓ/∂h_i = dx*(1 - h_{i+ℓ})
    # Then there is also term from h_i * h_{i+ℓ} :
    # Let’s be precise now:
    # Let H be vector h_0, ..., h_{n-1}
    # Let S_ℓ be shifted version of h where back-shift of ℓ positions (i->i+ℓ) => S_ℓ[i] = h[i-ℓ] for i>=ℓ else 0
    # C_ℓ = dx * sum_i h_i * (1 - h_{i+ℓ}) = dx * sum_i h_i - sum_i h_i h_{i+ℓ}
    # ∂C_ℓ/∂h_i = dx * (1 - DEL_{i+ℓ,i}) where DEL_{i+ℓ,i} = 1 if i+ℓ < n and h_{i+ℓ} otherwise 0 (and gives prob if h_{i+ℓ} is in range)
    # ∂C_ℓ/∂h_i = - dx * (eigenvector of h_{i+ℓ} where it exists) if not including boundary, since term is h_i * h_{i+ℓ}
    # So ∂/∂h_i sum h_i h_{i+ℓ}: 
    # ∂/(∂h_i) h_i h_{i+ℓ} = h_{i+ℓ} (derivative of h_{i+ℓ} w.r.t h_i = 0) + h_i * (1 if i+ℓ<n and 0 otherwise) (derivative of h_i w.r.t h_i)
    # Therefore ∂/∂h_i (sum_i h_i h_{i+ℓ} ) = shift by ℓ inverse to left
    # More directly for all gradients:
    # So ∇C_ℓ = dx * (vector of (1 - h_{i+ℓ})) - dx * shift_backward(h, ℓ) := - dx * (h_{i+ℓ} + shift_backward(h, ℓ) + vector of 1) = - dx * v_{i,ℓ} + dx * 1  
    # Wrong handling. Let's start from basic FT terms for all lags. 
    # Take h = [h_0, h_1, ..., h_{n-1}]
    # C_ℓ = dx * sum_{i=0}^{n-1} h_i * (1-h_{i+ℓ})
    # ∂C_ℓ / ∂h_i = dx * (1 - h_{i+ℓ})  when h_{i+ℓ} is in range. 
    # But when considered as a vector gradient, we are summing for all i and using the term ∂/∂h_i term.  
    # So for each ℓ, let:
    # grad_ℓ = [0]*n
    # For i in range(n):
    #     if (i+ℓ) < n:
    #         grad_ℓ[i] += -dx h_{i+ℓ}  
    #     grad_ℓ[i] += dx  # The 1 part
    # This is also equal to:
    # vec = [dx] * n
    # vec_new = [0]*n   for each ℓ, shift h by ℓ to the left to get h_{i+ℓ}, but ensuring valid indices.
    # This is already handled perfectly by FFT.
    # So more recovery:
    # Let’s revisit the handcalculated term in view of our FFT approach:
    # C_ℓ(h) = dx * Σ_i h_i (1 - h_{i+ℓ}) 
    # ∂C_ℓ/∂h_i = dx (1 - h_{i+ℓ}) + dx (-δ_{i+ℓ, i} * h_i) but since h_{i+ℓ} does not depend on h_i, we will not do the δ term.
    # That does not seem right either.
    # Reminder: 
    # In Variational Derivative see:  
    # ∂ C_ℓ(h) / ∂h_i = dx (1 - h_{i+ℓ})   
    # Since I take term h_i*(1-h_{i+ℓ}) and vary h_i (neglecting influence of h_{i+ℓ} in the derivative with respect to h_i)  
    # i.e., h_{i+ℓ} is evaluated constant.
    # The right way is:
    # ∂/∂h_i (h_i (1 - h_{i+ℓ})) = (1 - h_{i+ℓ}) if i+ℓ < n else 0
    # OR rather ∂/∂h_i h_i h_{i+ℓ} = h_{i+ℓ} 
    # So the full D of C_ℓ would be as:
    # ∇C_ℓ = [dx*(1 - h_{0+ℓ}) + ..., dx*(1 - h_{1+ℓ}) + ..., dx*(1 - h_{n-1+ℓ}) + ...] 
    # But for h_{i+ℓ} "out of bounds", we let h_{i+ℓ} = 0. Thus
    # ∇C_ℓ = -dx * [ h_{0+ℓ}, h_{1+ℓ}, ..., h_{n-1+ℓ} ]  since we assume shaped_REGS are 0
    # or in terms of offsetting:
    # ∇(C_ℓ) = dx * [1,1,1,...] -  dx * shift_forward(h, ℓ) = dx * [1]*n - dx * shift_forward(h, ℓ)
    # Now
    # ∇S_β(h) = Σ w_ℓ ∇C_ℓ = Σ w_ℓ (dx [1] - dx shift_forward(h, ℓ)) 
    # = dx *  Σ w_ℓ * (1) - dx * Σ w_ℓ shift_forward(h, ℓ) 
    # = dx * (1) * sum(w_ℓ) - dx * V ( where V = shiftforward(func, ℓ, w_ℓ) sum with w weights)  
    # This approach is tricky to compute, so we keep with basis explanation:
    # The idea is that ∂C_ℓ(h)/∂h_i = dx (1 - h_{i+ℓ})
    # The total gradient becomes:
    # ∇S_β = Σ w_ℓ [dx (1 - h_{i+ℓ}) for all i]
    # But shift operation by ℓ reverses index: h_{i+ℓ} positions become h_{i+ℓ} 
    # The only correct and efficient way is:
    # For β = 1, compute all correlations use FFT, 
    # Exponentiate and weight
    # Then apply derivative. Get:
    # Σ w_ℓ (normal vector of 1) - Σ w_ℓ h_{i+ℓ} where for h_{i+ℓ} we used left shift of h (with proper indices)

    # Compute for all ℓ from 0 to n-1:
    grad = np.zeros_like(h)
    # We quickly use compute_corr_fft(h)
    # corr = ∑_{i} h_i (1 - h_{i+ℓ})
    # So ∇corr_ℓ = -h_{i+ℓ} for all i -> 1 - h_{i+ℓ}  
    # Actually bad: the actual C_ℓ = dx * ∑_i h_i ( 1 - h_{i+ℓ} )  
    # So ∂C_ℓ/∂h_i = dx (1 - h_{i+ℓ}) 
    # But it is elegant to do it as vector shift:
    # For i = 0, ℓ,
    # ∂C_ℓ / ∂h_i = dx*(1 - h_{i+ℓ})
    # Overall gradient:
    # ∇Sβ = - dx * Σ w_ℓ [shift_by_ℓ(h)] + dx * Σ w_ℓ [1]
    # This is due to linearity. The term ∇C_ℓ = dx ([1] - shift_forward(h, ℓ))  
    # Let’s be precise and code the gradient in a mental loop for ℓ-loop:
    # U = Σ w_ℓ h_{i+ℓ}  weight multiplies shifted h by ℓ
    # see all int indices: d = −1; add a compensation just in case
    # Each actual shift amounts to left-shifted h = h_low(h[i+ℓ])
    # So, “suppose input h = [h0, h1, h2, ..., h_{n-1}]”
    # ∂C_ℓ / ∂h_i = dx*(1 - h_{i+ℓ})
    # So ∇C_ℓ = = dx * [1] − dx*[h_{0+ℓ}, h_{1+ℓ}, ..., h_{n-1+ℓ}]
    # But we must append zeros to h_{i+ℓ} prefix with zero on right:
    # h_shift = [0,0,...] + h[0:n-ℓ] -> the shift is from left  
    # Using np.roll: use forward shift
    # That is: roll(h, -ℓ) gives h_{i+ℓ} for i in indices  
    # e.g., for h = [1,2,3,4]
    # h_{1+1} = h_2 = 3; roll(h, -1) = [2,3,4,0]  
    # h_{1+1} = h_1 = 2, but now in position 2!
    # So roll(h, -1) effectively accesses h_{i+1} in position i, so
    # we do:
    # v = dx * [1] - dx * roll(h, -ℓ)
    # But it's on zero-padded
    # So it’s safe.

    grad = np.zeros_like(h)
    for ℓ in range(len(h)):
        if ℓ == 0:
            continue
        shift_h = np.roll(h, -ℓ)
        shift_h[-ℓ:] = 0
        grad += w[ℓ] * (dx * (1 - shift_h)) 
    # And apply sum( w * 1) part:
    grad += w[0] * (dx * (1 - np.zeros_like(h)))  # We're a bit sketchy on handling zero

    # Correct: let's loop more properly:
    # We'll do it correctly:
    # For entire vector: 
    # Want ∇C_ℓ = dx*(1 - h_{i+ℓ}) for each i.
    # ∇C_ℓ = dx * ( ones_vector - (h_{i+ℓ} long) )
    # i-th element of gradient is: dx *( 1 - h_{i+ℓ} )
    # And we put in the correct position h_{i+ℓ} from shifted version.
    # So 
    # Σℓ w_ℓ ∇C_ℓ = Σ w_ℓ [ dx*(1 − h_{i+ℓ}) for all i]  
    # This equals:  
    # dx * ( Σ w_ℓ 1_{i} ) - dx * ( Σ w_ℓ h_{i+ℓ} )  
    # Here the optimization is that for fixed index i:
    # Σ w_ℓ ( h_{i+ℓ} ) is a weighted average over all the lags.  
    # So we can shuffle efficiently via FFT and output.
    
    # NOW TAKE GOOD APPROACH, compute again:
    # Final approach:
    # For all ℓ, ∇(C_ℓ) = dx*( ones - shift_{forward}(h, ℓ) ) 
    # The ∑ = 1 by linearity
    # The weighting on each term is added later.
    # ∇ (sum_ w_ℓ C_ℓ) = Σ w_ℓ ∇C_ℓ  
    # ∇S_β = ∑ w_ℓ * ( dx * ones - dx * shift_forward(h, ℓ) ) 
    #      = dx * ( 1*∑ w_ℓ ) - dx * sum(w_ℓ * shift_forward(h, ℓ))  
    # This is valid.
    # So do: buffer of zeros
    # Compute shift h by forward ℓ for all ℓ, build up weighted average 
    # Do (weighted spatial average shift of h) = Σ w_ℓ shift_{+}(h, ℓ)

    # And realize it as vectorized operations:
    grad = np.zeros_like(h)
    for ℓ in range(1, len(h)):  # skip zero lag
        shift = np.roll(h, -ℓ)
        shift[-ℓ:] = 0
        grad += w[ℓ] * shift
    
    grad = dx * (np.ones_like(h) - grad)
    return grad

def projected_gradient_descent(h0, β_schedule, step_schedule, n_steps_per_beta, tol=1e-6):
    """
    Run projected gradient descent with given β schedule and step size schedule.
    """ 
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_from_corr(compute_corr_fft(h), len(h))
    
    for β_idx, β in enumerate(β_schedule):
        # Restart step size for new β
        for t in range(n_steps_per_beta):
            # Compute gradient with β
            grad = smooth_max_gradient(h, β)
            # Step size schedule
            alpha = step_schedule(t)
            # Update
            h_new = h - alpha * grad
            # Project
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            # Evaluate
            c5_new = c5_from_corr(compute_corr_fft(h_new), len(h_new))
            
            # Update best
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            
            h = h_new.copy()
            
            if abs((c5_new - best_c5) / best_c5) < tol:
                break
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx, n_fine, dx_fine
    np.random.seed(seed)
    start_time = time.time()

    # Initialize with multiple strategies
    initial_starts = []
    # Start 1: Interpolated from initial_h_values
    h0_orig = np.array(initial_h_values)
    x_orig = np.linspace(0, 2, len(h0_orig))
    x_new = np.linspace(0, 2, n_points)
    h_interp = np.interp(x_new, x_orig, h0_orig)
    h_interp = project_to_sum_box(h_interp, n_points/2)
    initial_starts.append(h_interp)

    # Start 2: Random uniform with sum 
    h2 = np.random.uniform(0, 1, n_points)
    h2 = project_to_sum_box(h2, n_points/2)
    initial_starts.append(h2)

    # Start 3: Random binary with sum
    h3 = np.random.choice([0, 1], size=n_points)
    h3 = project_to_sum_box(h3, n_points/2)
    initial_starts.append(h3)

    # Hook 
    TARGET_SUM = n_points / 2
    
    # Parameters  
    # β schedule from low to high (smooth to sharp)
    β_schedule = [0.5]
    β_current = 0.5
    # Let β increase slowly to avoid fluctuating too much before stabilization
    while β_current < 30.0:
        β_current *= 2.0
        β_schedule.append(β_current)
    
    # Step size schedule for gradient descent 
    step_size_schedule = lambda t: 0.4 / (1.0 + 0.0001 * t)

    best_h = None
    best_c5 = np.inf

    # Phase A: Coarse-scale optimization with multiple restarts
    phase_a_t0 = time.time()
    remaining_time = budget_s * 0.7  # 70% for coarse search

    num_restarts = 0
    while time.time() - phase_a_t0 < remaining_time:
        h_start = initial_starts[num_restarts % len(initial_starts)]
        h_final, c5_final = projected_gradient_descent(
                                h_start.copy(), 
                                β_schedule, 
                                step_size_schedule,
                                400  # iterations per β
                           )
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

        num_restarts += 1
        if time.time() - phase_a_t0 > remaining_time:
            break

    # Phase B: Fine resolution refinement using the coarse h
    if best_h is not None:
        # Interpolate to fine grid (n_fine)
        x_coarse = np.linspace(0, 2, n_points)
        x_fine = np.linspace(0, 2, n_fine)
        h_fine = np.interp(x_fine, x_coarse, best_h)
        h_fine = project_to_sum_box(h_fine, n_fine/2)
        
        # Small β range for fine grid  
        β_fine = [1.0, 3.0, 5.0, 10.0, 20.0, 30.0]
        
        h_final, c5_final = projected_gradient_descent(
            h_fine.copy(), 
            β_fine, 
            step_size_schedule,
            200  # fewer iterations for fine grid
        )
        
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase C: Final polishing and sub-gradient refinement
    phase_c_t0 = time.time()
    final_budget = budget_s - (time.time() - start_time) 
    remaining_time = final_budget * 0.1  # 10% for polishing 

    while time.time() - phase_c_t0 < remaining_time:
        # Run subgradient method onto best_h
        corr = compute_corr_fft(best_h)
        worst_lag = np.argmax(corr)
        # Subgradient: ∇C_{worst_lag} = dx * [1] - shift_forward(h, worst_lag) 
        subgrad = np.zeros_like(best_h)
        shift = np.roll(best_h, -worst_lag)
        shift[-worst_lag:] = 0
        subgrad = dx * (np.ones_like(best_h) - shift) 

        # Small step size
        alpha = 1e-3 / np.sqrt(1000)
        h_new = best_h - alpha * subgrad
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        c5_new = c5_from_corr(compute_corr_fft(h_new), len(h_new))
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Add jitter occasionally
        if np.random.rand() < 0.1:
            h_jitter = best_h + np.random.normal(0, 0.01, len(best_h))
            h_jitter = project_to_sum_box(h_jitter, TARGET_SUM)
            c5_test = c5_from_corr(compute_corr_fft(h_jitter), len(h_jitter))
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_jitter.copy()

    return (best_h.tolist(), float(best_c5), n_fine)
