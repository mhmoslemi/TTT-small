def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def gradient_descent_step(h, learning_rate=0.01):
    """Single gradient descent step to minimize the maximum correlation"""
    n = len(h)
    # Compute current correlations
    corr = compute_corr_fft(h)
    max_corr_idx = np.argmax(corr)
    
    # Compute gradient for this specific lag
    g = np.zeros(n)
    for i in range(n):
        left_idx = i - max_corr_idx
        right_idx = i + max_corr_idx
        if left_idx >= 0:
            g[i] += h[left_idx]
        if right_idx < n:
            g[i] += h[right_idx]
    
    # Update step (gradient ascent to reduce violation)
    h_new = h - learning_rate * g
    return project_to_sum_box(h_new, TARGET_SUM)

    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    for i in range(n_steps):
        h_new = gradient_descent_step(h, learning_rate)
        new_c5 = evaluate_c5(h_new)
        
        # Accept if improvement, otherwise try smaller step
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        else:
            # Try smaller learning rate
            learning_rate *= 0.99
            if learning_rate < 1e-6:
                break
    
    return best_h, best_c5

    h = h0.copy()
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    n = len(h)
    
    for i in range(n_iter):

        h_test = h.copy()
        h_test[idx1], h_test[idx2] = h_test[idx2], h_test[idx1]

            current_c5 = test_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    return best_h, best_c5
