import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Phase 1: Use global optimization with differential evolution to escape local minima
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    try:
        # Use differential evolution for global search
        res = differential_evolution(
            objective, 
            bounds, 
            maxiter=1000, 
            popsize=15, 
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            atol=1e-12,
            rtol=1e-12
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Phase 2: Local refinement with trust-constr
    for restart in range(2000):  # More iterations
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
        # Random perturbation around best solution
        new_h = best_h + np.random.normal(0, 0.01, n_points) * 0.5
        new_h = np.clip(new_h, 0, 1)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='trust-constr', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'gtol': 1e-13, 'ftol': 1e-13}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue

    # Phase 3: Try improvements using optimization with specific patterns
    if (time.time() - start_time) < budget_s * 0.9:
        try:
            # Variation: Try to construct a piecewise constant function
            # Start with a refined version of the best result
            new_h = best_h.copy()
            
            # Try to find a pattern that distributes the mass more efficiently
            # Use random sampling + gradient-based optimization
            for _ in range(200):
                if (time.time() - start_time) > budget_s * 0.95:
                    break
                pert_h = new_h + np.random.normal(0, 0.005, n_points)
                pert_h = np.clip(pert_h, 0, 1)
                dx = 2.0 / n_points
                scale_factor = 1.0 / (np.sum(pert_h) * dx)
                pert_h = pert_h * scale_factor
                try:
                    res = minimize(
                        objective, 
                        pert_h, 
                        method='SLSQP', 
                        bounds=bounds, 
                        constraints=cons,
                        options={'maxiter': 1000, 'ftol': 1e-14, 'gtol': 1e-14}
                    )
                    if res.success:
                        h_candidate = res.x
                        scale = 1.0 / (np.sum(h_candidate) * dx)
                        h_candidate = h_candidate * scale
                        c5_candidate = objective(h_candidate)
                        if c5_candidate < best_c5:
                            best_h = h_candidate.copy()
                            best_c5 = c5_candidate
                except:
                    pass

        except:
            pass

    # Final refinement to optimize the precision
    try:
        # Refine once more with very precise settings
        new_h = best_h.copy()
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        res = minimize(
            objective, 
            new_h, 
            method='trust-constr', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 2000, 'gtol': 1e-15, 'ftol': 1e-15}
        )
        if res.success:
            h_candidate = res.x
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
