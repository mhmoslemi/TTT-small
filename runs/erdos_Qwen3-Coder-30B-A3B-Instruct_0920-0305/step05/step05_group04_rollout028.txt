import numpy as np

def conv_fft(a, b):
    """Compute linear convolution using FFT with zero padding."""
    n = len(a)
    m = len(b)
    L = n + m - 1
    A = np.fft.fft(a, L)
    B = np.fft.fft(b, L)
    c = np.fft.ifft(A * B).real
    return c[:L]

def project_onto_bounded_simplex(x, target_sum=0.5):
    """Project vector x onto the simplex with sum = target_sum and bounds [0,1]."""
    # Clip to [0, 1]
    x = np.clip(x, 0, 1)
    x_sum = np.sum(x)
    
    if abs(x_sum - target_sum) < 1e-12:
        return x
    
    # Use bisection to find lambda such that sum of clipped values equals target_sum
    def sum_at_lambda(lam):
        return np.sum(np.clip(x + lam, 0, 1))
    
    # Bracket lambda
    lower, upper = -1.0, 1.0
    while sum_at_lambda(lower) > target_sum:
        lower *= 2
    while sum_at_lambda(upper) < target_sum:
        upper *= 2
    
    # Bisection
    for _ in range(30):
        mid = (lower + upper) / 2
        s = sum_at_lambda(mid)
        if s > target_sum:
            upper = mid
        else:
            lower = mid
    
    lambda_opt = (lower + upper) / 2
    return np.clip(x + lambda_opt, 0, 1)

def soft_weights(c, beta):
    """Compute soft weights for log-sum-exp smoothing."""
    c_max = np.max(c)
    exp_c = np.exp(-beta * (c - c_max))
    w = exp_c / np.sum(exp_c)
    return w

def smooth_obj_and_grad(h, beta):
    """Evaluate smooth surrogate and its gradient."""
    n = len(h)
    c = conv_fft(h, h)  # Linear convolution
    c = c[1:n]  # Only non-zero shifts k = 1, ..., n-1

    # Compute soft weights
    w = soft_weights(c, beta)

    # Compute gradient
    u = conv_fft(h, w)
    v = conv_fft(h, np.flip(w))
    grad = -(u + v)

    # Objective is negative of soft-minimum
    phi = -np.log(np.sum(np.exp(-beta * c))) / beta
    
    return phi, grad

def gradient_descent_with_continuation(h0, beta0=1.0, beta_max=200.0, alpha0=0.5, gamma=2.0, delta=0.9, max_iter=2000, tol=1e-12):
    h = h0.copy()
    beta = beta0
    alpha = alpha0
    best_h = h.copy()
    best_C5 = np.max(correlate(h, 1 - h, mode="full")) * 0.5

    for it in range(max_iter):
        if it % 100 == 0:
            # Evaluate exact C5 occasionally
            c_full = correlate(h, 1 - h, mode="full")
            current_C5 = np.max(c_full) * 0.5
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()

        # Compute smooth objective and gradient
        obj_val, grad = smooth_obj_and_grad(h, beta)
        
        # Line search (Armijo condition)
        tau = alpha
        while True:
            h_candidate = project_onto_bounded_simplex(h - tau * grad)
            obj_candidate, _ = smooth_obj_and_grad(h_candidate, beta)
            if obj_candidate <= obj_val - 0.1 * tau * np.dot(grad, grad):
                break
            tau /= 2
            if tau < 1e-12:
                break

        h = h_candidate

        # Check for convergence
        if np.linalg.norm(tau * grad) < 1e-9 or abs(obj_val - smooth_obj_and_grad(h, beta)[0]) < tol:
            # Increase β (homotopy)
            beta = min(beta * gamma, beta_max)
            alpha = alpha * delta

    # Final evaluation
    c_full = correlate(h, 1 - h, mode="full")
    final_C5 = np.max(c_full) * 0.5
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = h.copy()

    return best_h, best_C5
