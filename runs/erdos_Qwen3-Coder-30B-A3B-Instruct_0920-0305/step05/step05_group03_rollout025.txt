def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

                print(f"Symmetrized up-sampled C5 = {c5_current:.6f}")
    
    return (h_best.tolist(), c5_best, len(h_best))
