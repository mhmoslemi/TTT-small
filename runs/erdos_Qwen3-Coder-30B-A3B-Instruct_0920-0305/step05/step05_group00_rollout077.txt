# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def random_move(h):
    """Generate a random feasible move"""
    i, j = np.random.choice(n_points, 2, replace=False)
    hi, hj = h[i], h[j]
    delta_min = max(-hj, hi - 1)
    delta_max = min(hi, 1 - hj)
    if delta_min >= delta_max:
        return None
    delta = np.random.uniform(delta_min, delta_max)
    return (i, j, delta)

def descent_move(h):
    """Perform a gradient-based descent update"""
    # Compute gradient direction
    g = np.zeros(n_points)
    corr = compute_corr_fft(h)
    worst_lag = np.argmax(corr)
    
    # Update gradient based on the worst lag
    for i in range(n_points):
        if i - worst_lag >= 0:
            g[i] += h[i - worst_lag]
        if i + worst_lag < n_points:
            g[i] += h[i + worst_lag]
    
    # Find best improvement direction
    idx_min = np.argmin(g)
    idx_max = np.argmax(g)
    
    if idx_min == idx_max:
        return None
    
    # Transfer minimal amount that respects constraints
    delta = min(h[idx_min], 1 - h[idx_max])
    if delta <= 0:
        return None
    
    return (idx_min, idx_max, delta)
