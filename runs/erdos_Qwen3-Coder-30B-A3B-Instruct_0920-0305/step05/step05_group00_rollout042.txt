import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Compute C5 value efficiently using correlation"""
    corr = correlate(h, 1-h, mode="full")
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_based_move(h):
    """Compute gradient and suggest moves"""
    # Compute gradient for each position
    gradient = np.zeros(len(h))
    for i in range(len(h)):
        # For each possible lag, compute contribution to gradient
        for k in range(len(h)):
            if i+k < len(h):
                gradient[i] += h[i+k]
            if i-k >= 0:
                gradient[i] += h[i-k]
    
    # Find good indices to adjust
    idx_min = np.argmin(gradient)
    idx_max = np.argmax(gradient)
    
    if idx_min == idx_max:
        return None
        
    # Compute maximum transferable amount
    delta = min(h[idx_min], 1 - h[idx_max])
    if delta <= 0:
        return None
    
    return (idx_min, idx_max, delta)

    Tf = 1e-6
    t_start = time.time()
    
    # Reduced acceptance probability
    prob = 0.9
    
    while time.time() - t_start < run_seconds:
        T = T0 * (1 - (time.time() - t_start)/run_seconds) + Tf * (time.time() - t_start)/run_seconds
        
        # Prefer more structured moves
        if rng.random() < prob:  # Gradient-guided moves
            move = gradient_based_move(h)
        else:  # Random pairwise moves
            i, j = rng.choice(n_points, 2, replace=False)
            hi, hj = h[i], h[j]
            delta_min = max(-hj, hi - 1)
            delta_max = min(hi, 1 - hj)
            if delta_min >= delta_max:
                continue
            delta = rng.uniform(delta_min, delta_max)
            move = (i, j, delta)
        
        if move is None:
            continue
            
        i, j, delta = move
        
        # Store old state
        old_h = h.copy()
        old_c5 = evaluate_c5(h)
        
        # Apply move
        h[i] -= delta
        h[j] += delta
        
        # Project to constraints
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Evaluate
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Accept based on temperature
        if c5 >= old_c5 and rng.random() > np.exp(-(c5 - old_c5) / T):
            # Revert
            h = old_h.copy()
    
    return best_h, best_c5

def improved_local_refine(h):
    """Improved local refinement with better constraint handling"""  
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, TARGET_SUM)
            return h
    except:
        pass
    return h
