            best_min_overlap = -1

                if i in positions:
                    continue

                # Estimate overlap counts
                overlaps = [0] * (n - 1)
                for j in range(n):
                    for k in range(1, n):
                        if j in cand_positions and (j + k) in cand_positions:
                            overlaps[k - 1] += 1
                min_over = min(overlaps) if overlaps else 0
                if min_over > best_min_overlap:
                    best_min_overlap = min_over
                    best_pos = i
            positions.add(best_pos)
        # Create binary vector
        for i in positions:
            h_greedy[i] = 1
        # Renormalize to correct sum
        h_greedy *= (n / 2) / np.sum(h_greedy)

        # Formulate MILP model 
        # Variables to hold the indicator values
        x = Variable(n, boolean=True)
        L = Variable(integer=True)

        # Objective: maximize L
        objective = Maximize(L)

        # Constraints
        constraints = [sum(x) == n // 2]

        # Fix first element to 1
        constraints += [x[0] == 1]

        constraints += [sum(x[:n//2]) >= sum(x[n//2:])]

            # Sum of y gives p_k
            p_k = sum(y)
            # Constraint: p_k >= L for all valid shifts
            constraints += [p_k >= L]

        # We want to solve using cvxpy and numeric solver
        problem = Problem(objective, constraints)

                    return np.max(corr) * dx
