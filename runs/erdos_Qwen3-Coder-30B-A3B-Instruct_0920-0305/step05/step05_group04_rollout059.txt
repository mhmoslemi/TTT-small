import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def conv_fft(a, b):
    """Compute linear convolution using FFT with zero padding"""
    n = len(a)
    m = len(b)
    # Zero-pad to length at least n+m-1
    L = 2 * n  # This is sufficient for linear convolution if we pad appropriately
    # use n+m-1 for completeness but keep it a power of two for efficiency
    L = max(L, n + m - 1)
    # Now let's zero-pad both
    A = np.fft.fft(a, L)
    B = np.fft.fft(b, L)
    C = A * B
    c = np.fft.ifft(C).real
    return c[:n + m - 1]

def project_onto_bounded_simplex(x):
    """Projected gradient step onto feasible set {h : sum(h) = n/2, 0 <= h_i <= 1}"""
    n = len(x)
    # temporarily clip between 0 and 1
    x = np.clip(x, 0, 1)
    
    # calculate sum and find scalar adjustment λ to enforce sum = n/2
    current_sum = np.sum(x)
    target_sum = n / 2.0
    
    # Use bisection to find the correct λ (it is piecewise linear)
    if abs(current_sum - target_sum) < 1e-15:
        return x
    
    # function: f(λ) = sum(clip(x_i + λ, 0, 1)) - target_sum
    def f(λ):
        y = x + λ
        y = np.clip(y, 0, 1)
        return np.sum(y) - target_sum

    # bisection over λ interval
    a = 0.0
    b = 1e6  # large upper bound
    steps = 0
    # Since we know at least one element will be 1, a bound is achievable
    while steps < 50:
        if abs(f(a)) < 1e-15:
            return np.clip(x + a, 0, 1)
        if abs(f(b)) < 1e-15:
            return np.clip(x + b, 0, 1)

        mid = (a + b) / 2.0
        f_mid = f(mid)
        if abs(f_mid) < 1e-15:
            return np.clip(x + mid, 0, 1)
        
        if f(0) > 0:  # if x already satisfies constraint
            sign_a = np.sign(f(a))
            sign_mid = np.sign(f_mid)
            if sign_a * sign_mid <= 0:  # change of sign
                b = mid
            else:
                a = mid
        else:  # x sum is less than target
            sign_a = np.sign(f(a))
            sign_mid = np.sign(f_mid)
            if sign_a * sign_mid <= 0:  # change of sign
                a = mid
            else:
                b = mid
        steps += 1

    # fallback to approximate case
    λ = (target_sum - current_sum) / n
    y = np.clip(x + λ, 0, 1)
    return y

def soft_weights(c, beta):
    """Compute softmax weights for soft-minimization, with numerical stabilization"""
    # subtract max to prevent overflow
    c_max = np.max(c)
    # use stable computation
    exps = np.exp(-beta * (c - c_max))
    w = exps / np.sum(exps)
    return w

def smooth_obj_and_grad(h, beta):
    """Evaluate smooth surrogate of -C5 and return gradient"""
    n = len(h)
    # compute the self correlation for shift k=1 to n-1
    c = conv_fft(h, h)
    # We are computing conv(h,h), but to get c_k we take the correlation from shift k
    # The first n elements are not necessary, we skip the first shift
    c = c[1:n]  # shift 1 to n-1 correlators
    # c_k = sum_i h_i h_{i+k}, index 0 in c corresponds to k=1
    
    # compute weights for soft-min
    w = soft_weights(c, beta)
    # soft-weighted gradient
    v1 = conv_fft(h, w)
    v2 = conv_fft(h, w[::-1])   # reverse w
    g = -(v1 + v2)[:n]  # gradient of Φ_β
    # value of Φ_β
    log_sum_exp_val = np.log(np.sum(np.exp(-beta * c))) + beta * c_max
    value = -log_sum_exp_val / beta  # Phi_beta
    
    return value, g

def gradient_descent_with_beta_continuation(h_start, beta_0=1.0, beta_max=200.0, alpha_0=0.5,
                                            gamma=2.0, delta=0.9, max_iter=2000,
                                            armijo_constant=0.1):
    """Run projected gradient descent with beta continuation on softened C5"""
    h = h_start.copy()
    n = len(h)
    beta = beta_0
    alpha = alpha_0
    iteration = 0
    
    while iteration < max_iter:
        iteration += 1
        # compute c_k (shifted correlations) using FFT
        c = conv_fft(h, h)[1:n]  # exclude zero shift
        # objective
        if beta > 1000:
            current_obj = -np.min(c)
        else:
            w = soft_weights(c, beta)
            current_obj = -np.sum(w * c) / beta  #


            # compute gradient
            v1 = conv_fft(h, w)
            v2 = conv_fft(h, w[::-1])   # reverse w
            g = -(v1 + v2)[:n]
            
            # line search with Armijo condition
            tau = alpha
            h_prev = h.copy()
            obj_prev = current_obj
            
            while tau > 1e-15:
                h_trial = h - tau * g
                h_trial = project_onto_bounded_simplex(h_trial)
                
                # compute the trial objective using original C5 definition
                c_trial = conv_fft(h_trial, h_trial)[1:n]
                obj_trial = -np.min(c_trial)  # actually compute real objective in line search
                
                # check Armijo condition
                if obj_trial <= obj_prev - armijo_constant * tau * np.linalg.norm(g)**2:
                    h = h_trial
                    alpha = alpha * delta if iteration % 10 == 0 else alpha  # adaptive step size
                    break
                else:
                    tau = tau / 2.0
            else:
                # Line search failed, proceed to next iteration anyway
                h = project_onto_bounded_simplex(h - alpha * g)
                
        # convergence check
        if np.linalg.norm(alpha * g) < 1e-9:
            break
        if iteration % 50 == 0 and beta < beta_max:
            beta = min(beta * gamma, beta_max)
            # reset step size for new beta
            alpha = alpha_0
            
        if beta > beta_max and iteration % 50 == 0:
            # final smoothing phase - reduce the update size (small step)
            # but this logic may not be necessary
            pass
    # compute final value properly using exact C5
    c_exact = conv_fft(h, h)[1:n]
    final_obj = -np.min(c_exact)
    return h, final_obj, iteration

def evaluate_c5_exact(h):
    """Compute the exact value of C5 (same as used in the test script)"""
    n_points = len(h)
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode="full")
    # max of correlation between h and (1-h)
    return np.max(corr) * dx
