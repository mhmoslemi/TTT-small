import numpy as np
from cvxpy import*
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def greedy_initialization(n):
    """Initial construction using greedy algorithm"""
    x = np.zeros(n, dtype=int)
    selected = set()
    while len(selected) < n // 2:
        best_i = -1
        best_min_p = -1
        for i in range(n):
            if i in selected:
                continue
            # Temporarily add i to simulate what happens
            temp_selected = selected | {i}
            # Compute min_p for all valid shifts
            min_p = float('inf')
            for k in range(1, n):
                count = 0
                for j in range(n - k):
                    if j in temp_selected and (j + k) in temp_selected:
                        count += 1
                min_p = min(min_p, count)
            if min_p > best_min_p:
                best_min_p = min_p
                best_i = i
        if best_i != -1:
            selected.add(best_i)
    for i in selected:
        x[i] = 1
    return x

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial guess from the initial_h_values
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Try successively smaller n values starting from 64
    ns = [64, 60, 56, 52, 48, 44, 40, 36]
    best_solution = (h0, evaluate_c5(h0), n_points)
    used_n = n_points

    for n in ns:
        if time.time() - start_time > budget_s * 0.95:
            break
        # Check if we can construct a solution (try to avoid large MILP)
        try:
            # Start with a greedy warm start
            x_start = greedy_initialization(n)
            # Greedy approach (fast) to ensure feasibility
            h_past = np.zeros(n)
            for i in range(n):
                if x_start[i]:
                    h_past[i] = 1
            dx = 2.0 / n
            scale = 1.0 / (np.sum(h_past) * dx)
            h_best = h_past * scale
            c5_best = evaluate_c5(h_best)
            
            # Even better if we proceed with small MILP
            # For n=40 take a more refined greedy approach
            h_last = h_best
            c5_last = c5_best
            for _ in range(200):
                if time.time() - start_time > budget_s * 0.97:
                    break
                # Do small perturbations
                for i in range(n):
                    r = np.random.random()
                    if r < 0.05:
                        h_last[i] = min(1.0, max(0.0, h_last[i] + (np.random.random()-0.5)*0.01))
                # Apply constraints
                dx = 2.0 / n
                scale = 1.0 / (np.sum(h_last) * dx)
                h_last = h_last * scale
                c5_new = evaluate_c5(h_last)
                if c5_new < c5_last:
                    h_best = h_last.copy()
                    c5_best = c5_new
            if c5_best < best_solution[1]:
                best_solution = (h_best.tolist(), c5_best, n)
        except Exception as e:
            continue
    
    # Final polishing step using smooth iterative refinement if time allows
    if time.time() - start_time < budget_s * 0.99:
        try:
            h_final = best_solution[0]
            n_final = best_solution[2]
            # Try a small optimization based on gradient descent
            def obj(h):
                return evaluate_c5(h)

            bounds = [(0, 1) for _ in range(n_final)]
            
            # Simple local search with bounds
            res = minimize(obj, h_final, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
            if res.success:
                h_opt = res.x
                dx = 2.0 / n_final
                scale = 1.0 / (np.sum(h_opt) * dx)
                h_opt = h_opt * scale
                c5_opt = obj(h_opt)
                if c5_opt < best_solution[1]:
                    best_solution = (h_opt.tolist(), c5_opt, n_final)
        except Exception as e:
            pass

    return best_solution
