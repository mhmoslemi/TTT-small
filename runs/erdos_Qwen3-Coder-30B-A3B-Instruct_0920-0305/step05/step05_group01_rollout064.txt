def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def guided_move(h, worst_lag):
    """Guided deterministic move for the worst lag to minimize the correlation"""
    n = len(h)
    # Compute subgradient g_i = h_{i-\ell} + h_{i+\ell}
    g = np.zeros(n)
    for i in range(n):
        left_idx = i - worst_lag
        right_idx = i + worst_lag
        if left_idx >= 0:
            g[i] += h[left_idx]
        if right_idx < n:
            g[i] += h[right_idx]
    
    # Find indices with min and max g
    i_min = np.argmin(g)
    i_max = np.argmax(g)
    
    if i_min == i_max:
        return None  # No improvement possible
    
    # Move a small amount from i_min to i_max
    delta_max = min(h[i_min], 1.0 - h[i_max])
    if delta_max <= 0:
        return None
    
    delta = min(delta_max, 0.005)  # Reduced step size
    return i_min, i_max, delta

def derivative_based_move(h):
    """Attempt derivative-based guided moves"""
    n = len(h)
    # Compute h * (1-h) one step ahead
    h_shifted = np.concatenate([h[1:], [0]])
    d = h - h_shifted
    
    # Adjust h where gradient suggests improvement
    move_incr = np.zeros(n)
    
    # When gradient is positive (h_{i+1} < h_i), move mass left to right
    positive_indices = np.where(d > 0.01)[0]
    for idx in positive_indices:
        if idx + 1 < n:
            move_incr[idx] -= 0.005
            move_incr[idx + 1] += 0.005
    
    # Check final positions for feasibility
    h_new = h + move_incr
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    
    return h_new
