import numpy as np

from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def local_improve(h, max_iter=500):

    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h))]
    
    try:
        # First optimize with bounds and constraint
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
    except:
        pass
    return h, evaluate_c5(h)

def random_step(h, step_size=0.1):
    """Apply random small perturbation"""
    h_new = h + np.random.normal(0, step_size, len(h))
    return project_to_sum_box(h_new, TARGET_SUM)

def grid_search_hill_climb(h0, num_iters=5000):

    h = h0.copy()
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Track the history for adaptive step size
    improvement_window = []
    step_factor = 0.05
    
    for i in range(num_iters):
        # Try small random steps
        h_new = random_step(h, step_factor)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < current_c5:
            h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
            improvement_window.append(True)
        else:
            improvement_window.append(False)
        
        # Adaptive step size based on recent improvement rate
        if len(improvement_window) > 50:
            recent_improvements = sum(improvement_window[-50:])
            if recent_improvements < 10:
                step_factor *= 0.95
            elif recent_improvements > 40:
                step_factor *= 1.02
            improvement_window = improvement_window[-50:]
    
    return best_h, best_c5
