import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0

def evaluate_c5(h):
    """Evaluate C5 using correlation"""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum correlation scaled by dx
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project to sum constraint with box constraints [0,1]"""
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    y = np.clip(x + low, 0, 1)
    return y

def perturb_solution(h, step_size=0.01):
    """Perturb solution by moving mass between indices"""
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    delta_max = min(h[i], 1.0 - h[j])
    if delta_max <= 0:
        return h.copy()
    delta = np.random.uniform(0, min(delta_max, step_size))
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

        if c5_trial < best_c5:

            best_c5 = c5_trial
