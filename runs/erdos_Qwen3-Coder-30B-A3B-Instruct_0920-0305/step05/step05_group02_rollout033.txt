import numpy as np
from scipy.signal import correlate
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_c5_exact(h):
    """Evaluate C5 exactly using discrete correlation"""
    n = len(h)
    dx = 2.0 / n
    # Calculate all correlation values
    corr = np.zeros(2*n - 1)
    for k in range(-n+1, n):
        if k >= 0:
            corr[k + n - 1] = sum(h[i] * (1 - h[i+k]) for i in range(n-k))
        else:
            corr[k + n - 1] = sum(h[i+k] * (1 - h[i]) for i in range(-k, n))
    return np.max(corr) * dx

def generate_legendre_pattern(n):
    """Generate a quadratic residue pattern"""
    p = n // 2
    if p < 2:
        return None
        
    # Check if p is prime
    is_prime = True
    for i in range(2, int(math.sqrt(p)) + 1):
        if p % i == 0:
            is_prime = False
            break
            
    if is_prime:
        residues = set()
        for a in range(1, p):
            residues.add((a * a) % p)
        b = np.zeros(p, dtype=int)
        for r in residues:
            if r < p:
                b[r] = 1
        # Duplicate to get length n
        h = np.concatenate([b, b])
        # Adjust if needed to make sum exactly n/2
        current_sum = np.sum(h)
        if current_sum < p:
            # Add one more 1
            idx = np.where(h == 0)[0][0]  # Any zero position
            h[idx] = 1
        elif current_sum > p:
            # Remove one 1
            idx = np.where(h == 1)[0][0]  # Any one position
            h[idx] = 0
        return h[:n]
    else:
        # Fall back to random pattern
        return None

def generate_random_pattern(n):
    """Generate a random balanced pattern"""
    indices = np.arange(n)
    np.random.shuffle(indices)
    h = np.zeros(n, dtype=int)
    h[indices[:n//2]] = 1
    return h

def compute_correlation(h):
    """Compute the discrete correlation"""
    n = len(h)
    corr = np.zeros(2*n - 1)
    for k in range(-n+1, n):
        if k >= 0:
            corr[k + n - 1] = sum(h[i] * h[i+k] for i in range(n-k))
        else:
            corr[k + n - 1] = sum(h[i+k] * h[i] for i in range(-k, n))
    return corr

def compute_worst_shifts(corr, tolerance=1e-10):
    """Find all indices where correlation attains maximum value"""
    max_val = np.max(corr)
    return np.where(np.abs(corr - max_val) < tolerance)[0]

def update_correlation_incremental(corr, h, old_i, old_j, new_i, new_j):
    """Update correlation after swapping positions i and j"""
    n = len(h)
    # Compute changes for each lag
    delta = np.zeros(2*n - 1)
    
    # For lag m, where i appears in the first sum (i, i+m)
    # and where j appears as first sum index (j, j+m)
    # and we need to be careful that we count them properly
    
    # For positions where correlation terms change due to i, j
    # i contributes to lags from i to i + n - 1
    # j contributes to lags from j to j + n - 1 

    # contribution of i to exact lags
    for m in range(-n+1, n):
        # i contributes to c_m where i in first and i+m in second
        if 0 <= m < n and 0 <= i+m < n:
            if h[m+i] == 0 and h[i] == 1:
                delta[m + n - 1] -= 1  # i switches from 1 to 0
            elif h[m+i] == 1 and h[i] == 0:
                delta[m + n - 1] += 1  # i switches from 0 to 1
        
        # Similarly for symmetric terms (from j in second position)
        if 0 <= i < n and 0 <= i-m < n:
            if h[i] == 0 and h[i-m] == 1:
                delta[m + n - 1] -= 1  # i switches from 1 to 0, impacts corr with lag -m
            elif h[i] == 1 and h[i-m] == 0:
                delta[m + n - 1] += 1  # i switches from 0 to 1
    
    # Same for j
    for m in range(-n+1, n):
        if 0 <= m < n and 0 <= j+m < n:
            if h[m+j] == 0 and h[j] == 1:
                delta[m + n - 1] -= 1  # j switches from 1 to 0
            elif h[m+j] == 1 and h[j] == 0:
                delta[m + n - 1] += 1  # j switches from 0 to 1
                
        if 0 <= j < n and 0 <= j-m < n:
            if h[j] == 0 and h[j-m] == 1:
                delta[m + n - 1] -= 1  # j switches from 1 to 0
            elif h[j] == 1 and h[j-m] == 0:
                delta[m + n - 1] += 1  # j switches from 0 to 1

    return corr + delta
