def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_c5_binary(h):
    """Evaluate C5 for a binary h using full correlation"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def generate_legendre_pattern(n):
    """Generate a Legendre-based binary pattern"""
    p = n // 2
    if p < 2 or not is_prime(p):
        # Fall back to random pattern
        h = np.zeros(n, dtype=int)
        h[:p] = 1
        np.random.shuffle(h)
        return h
    
    # Generate quadratic residues mod p
    residues = set()
    for a in range(1, p):
        r = (a * a) % p
        residues.add(r)
    
    # Create binary pattern
    b = np.zeros(p, dtype=int)
    for i in range(p):
        if i in residues:
            b[i] = 1
    
    h = np.concatenate([b, b])
    
    # Ensure exactly p ones
    ones_count = np.sum(h)
    if ones_count != p:
        # Adjust by flipping one zero to one
        if ones_count < p:
            zero_positions = np.where(h == 0)[0]
            if len(zero_positions) > 0:
                h[zero_positions[0]] = 1
        elif ones_count > p:
            one_positions = np.where(h == 1)[0]
            if len(one_positions) > 0:
                h[one_positions[0]] = 0
    
    return h

def is_prime(n):
    if n < 2:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False
    for i in range(3, int(np.sqrt(n)) + 1, 2):
        if n % i == 0:
            return False
    return True

def generate_random_pattern(n):
    """Generate a random balanced binary pattern"""
    h = np.zeros(n, dtype=int)
    h[:n//2] = 1
    np.random.shuffle(h)
    return h

def binary_to_continuous(h_binary, n):
    """Convert binary to continuous representation"""
    dx = 2.0 / n
    h_cont = h_binary.astype(float)
    h_cont = h_cont * (n/2) / np.sum(h_cont)
    return np.clip(h_cont, 0, 1)
