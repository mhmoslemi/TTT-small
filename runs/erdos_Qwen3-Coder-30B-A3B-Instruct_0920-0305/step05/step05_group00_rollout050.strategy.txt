def subgradient_descent(h0, max_iter, step0, time_limit):
    h = h0.copy()
    best_h = h.copy()
    best_c = evaluate_c5(h)
    start = time.time()
    for it in range(max_iter):
        now = time.time()
        if now - start > time_limit:
            break
        corr = compute_corr_fft(h)
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        lag = max_idx - (len(h)-1)
        # compute direction for worst lag and maybe second worst
        # Let's use top M = 3 lags
        M = 3
        # get top M indices
        top_idxs = np.argpartition(corr, -M)[-M:]
        # compute direction sum
        direction = np.zeros_like(h)
        for idx in top_idxs:
            l = idx - (len(h)-1)
            shift_plus = shift_zero(h, -l)
            shift_minus = shift_zero(h, l)
            direction += (shift_plus + shift_minus - 1)  # d = -grad
        direction /= M
        # step size schedule
        step = step0 / (np.sqrt(it+1))
        # update
        h_new = project_to_sum_box(h + step * direction, target_sum)
        # evaluate new
        new_corr = compute_corr_fft(h_new)
        new_max_corr = np.max(new_corr)
        # acceptance: if better update best
        if new_max_corr < best_c:
            best_c = new_max_corr
            best_h = h_new.copy()
        # keep new if reduces C or else maybe accept with probability? We'll use deterministic
        if new_max_corr <= max_corr:
            h = h_new
        else:
            # maybe reject or keep old
            # For robustness we can still accept some with probability
            if np.random.rand() < 0.01:
                h = h_new
