import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def simple_optimize(h0, run_seconds):
    """Simple gradient-based optimization with constraints"""
    start_time = time.time()
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h0))]
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Try different optimizers
    optimizers = ['SLSQP', 'trust-constr']
    for opt_method in optimizers:
        if time.time() - start_time > run_seconds:
            break
            
        try:
            res = minimize(
                objective,
                best_h,
                method=opt_method,
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_new = res.x
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
        except:
            continue
    
    return best_h, best_c5

def local_search(h0, run_seconds):
    """Local search with guided moves"""
    start_time = time.time()
    h = h0.copy()
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    n = len(h)
    
    # Move generator function
    def generate_move():
        # Try three different strategies
        strategy = np.random.choice(3)
        if strategy == 0:
            # Random single point change
            i = np.random.randint(0, n)
            delta = np.random.uniform(-0.1, 0.1)
            return i, delta, 'single'
        elif strategy == 1:
            # Pairwise transfer
            i, j = np.random.choice(n, 2, replace=False)
            delta = np.random.uniform(-0.05, 0.05)
            return i, delta, 'pair'
        else:
            # Local neighborhood search
            i = np.random.randint(0, n)
            delta = np.random.uniform(-0.05, 0.05)
            return i, delta, 'neighbor'

    while time.time() - start_time < run_seconds:
        # Generate a move
        i, delta, move_type = generate_move()
        
        h_new = h.copy()
        
        if move_type == 'single':
            h_new[i] += delta
        elif move_type == 'pair':
            # Move from i to j
            j = (i + np.random.randint(1, n)) % n
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max > 0:
                delta = np.clip(delta, -delta_max, delta_max)
                h_new[i] -= delta
                h_new[j] += delta
        elif move_type == 'neighbor':
            # Move to nearby points
            for k in range(max(0, i-2), min(n, i+3)):
                if k != i:
                    h_new[k] = h[i] + (h_new[k] - h[i]) * (1 - abs(i-k)/3.0) 
                    
        # Project to sum and constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        new_c5 = evaluate_c5(h_new)
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        elif np.random.rand() < 0.02:  # Metropolis test to escape local optima
            h = h_new
            current_c5 = new_c5

    return best_h, best_c5

def random_restart_search(h0, run_seconds):
    """Multi-start approach"""
    start_time = time.time()
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Simple restarts
    n_restarts = 0
    while time.time() - start_time < run_seconds * 0.8 and n_restarts < 50:
        # Random perturbation around h0 (not too large)
        h_perturbed = h0 + np.random.normal(0, 0.03, len(h0))
        h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
        
        # Local optimization of this start
        h_local, c5_local = local_search(h_perturbed, run_seconds * 0.2)
        
        if c5_local < best_c5:
            best_h, best_c5 = h_local, c5_local
            
        n_restarts += 1

    return best_h, best_c5
