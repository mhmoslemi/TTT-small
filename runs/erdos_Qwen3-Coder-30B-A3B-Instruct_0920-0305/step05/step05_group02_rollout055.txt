import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_c5_discrete(h):
    """Evaluate discretized C5 using dot products."""
    n = len(h)
    dx = 2.0 / n
    h = np.array(h)
    max_corr = 0.0
    for k in range(1, n):
        # Compute correlation term for shift k
        corr_k = 0.0
        for i in range(n - k):
            corr_k += h[i] * (1 - h[i + k])
        max_corr = max(max_corr, corr_k * dx)
    return max_corr

def solve_milp_binary(n):
    """Solve the binary integer program to maximize the minimum overlap."""
    # Use cvxpy for mixed integer linear programming
    # Variables are x_i (0/1), y_{i,k} (0/1) for pairs, L (integer)
    x = cp.Variable(n, boolean=True)
    L = cp.Variable(1, integer=True)
    y = {}
    constraints = []

    # Sum constraint: sum(x) = n/2
    constraints.append(cp.sum(x) == n // 2)

    # Symmetry breaking: fix first element and enforce ordering
    constraints.append(x[0] == 1)
    # Left-heavy ordering: sum of first half >= sum of second half
    constraints.append(cp.sum(x[:n//2]) >= cp.sum(x[n//2:]))

    # y_{i,k} = x_i AND x_{i+k}
    # Assume increasing k from 1 to n-1
    for k in range(1, n):
        for i in range(n - k):
            y[(i, k)] = cp.Variable(boolean=True)
            # McCormick envelopes for y_{i,k} = x_i * x_{i+k}
            constraints.append(y[(i, k)] <= x[i])
            constraints.append(y[(i, k)] <= x[i + k])
            constraints.append(y[(i, k)] >= x[i] + x[i + k] - 1)
    
    # For each k, define p_k = sum of y_{i,k}
    # Add constraint that all p_k >= L
    for k in range(1, n):
        p_k = cp.sum([y[(i, k)] for i in range(n - k)])
        constraints.append(p_k >= L)
    
    # Objective: maximize L
    prob = cp.Problem(cp.Maximize(L), constraints)
    
    try:
        prob.solve(solver=cp.CBC, verbose=False, maximum_seconds=30)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            x_vals = [int(round(xi.value)) for xi in x]
            L_val = int(round(L.value))
            return x_vals, L_val
    except Exception:
        return None, None
    return None, None
