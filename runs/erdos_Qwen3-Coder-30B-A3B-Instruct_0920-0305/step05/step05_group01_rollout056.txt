def evaluate_c5(h):
    """Compute C5 using full correlation"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using numpy's correlate
    corr = np.correlate(h, 1 - h, mode="full")
    # Take maximum (excluding the zero-lag term which is not relevant to the problem)
    max_corr = np.max(corr)
    return dx * max_corr

def project_to_sum_box(x, target_sum):
    """Project to sum = target_sum with box constraints [0,1]"""
    # Clip first
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x
    
    # Use a binary search approach to adjust
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    
    # Return the best approximation
    y = np.clip(x + low, 0, 1)
    return y
