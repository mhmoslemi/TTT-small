import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def conv_fft(a, b):
    """Compute linear convolution using FFT with zero padding"""
    n = len(a)
    m = len(b)
    pad_len = n + m - 1
    # Zero-pad to appropriate length
    a_pad = np.pad(a, (0, pad_len - n), 'constant')
    b_pad = np.pad(b, (0, pad_len - m), 'constant')
    # Compute FFT and convolve
    A = np.fft.fft(a_pad)
    B = np.fft.fft(b_pad)
    conv_ab = np.fft.ifft(A * np.conj(B)).real
    # Return only the linear part (not circular)
    return conv_ab[:pad_len]

def project_onto_bounded_simplex(x):
    """Project a vector onto the bounded simplex with ∑x_i = n/2 and 0 ≤ x_i ≤ 1"""
    n = len(x)
    # Clip to bounds
    x = np.clip(x, 0, 1)
    # Compute current sum
    s = np.sum(x)
    # Compute exact shift to meet constraint
    if abs(s - n / 2) < 1e-12:
        return x
        
    # The function sum_i clip(x_i + λ, 0, 1) is monotone piecewise linear in λ.
    # We use bisection to find λ such that the sum equals n/2.
    # If x_i = 0, then we want to increase it to λ > 0 (x_i + λ = λ).
    # If x_i = 1, then we want to decrease it to λ < 0 (x_i + λ = 1 + λ).
    # We need the sum of all clippings to equal n/2.
    def sum_clipped(λ):
        return np.sum(np.clip(x + λ, 0, 1))
        
    # Find bounds for lambda
    λ_lo = -np.min(x) - 1
    λ_hi = 1 - np.max(x) + 1
    
    # Bisection iteration until convergence
    for _ in range(30):
        λ = (λ_lo + λ_hi) / 2
        s = sum_clipped(λ)
        if s > n / 2:
            λ_lo = λ
        else:
            λ_hi = λ
    λ = (λ_lo + λ_hi) / 2
    
    return np.clip(x + λ, 0, 1)

def soft_weights(c, β):
    """Compute soft weights for average of correlation terms"""
    # Subtract maximum to prevent overflow
    c_max = np.max(c)
    exp_terms = np.exp(-β * (c - c_max))
    return exp_terms / np.sum(exp_terms)

def smooth_obj_and_grad(h, β):
    """Compute the smooth surrogate Φ_β(h) and its gradient"""
    # Compute self-correlation via FFT
    c = conv_fft(h, h)
    # Keep shifts k=1,...,n-1
    c = c[1:len(c)//2]  # take only necessary elements
    # Compute weights for soft-sum-exp
    w = soft_weights(c, β)
    # Compute gradient via convolutions
    # First: convolve with λ such that kernel is w and reverse
    # For gradient of sum_k w_k * c_k where c_k = sum_i h_i h_{i+k}
    # For forward shift w[i] * h[i+1] + w[i] * h[i+2] + ...
    # This corresponds to the convolution of h with w shifted right
    # Match indices
    expand_w = np.zeros(len(h))
    expand_w[:len(w)] = w
    g = -conv_fft(h, expand_w)
    g = g[:len(h)]
    # Mirror for backward shift contribution
    expand_w_rev = np.zeros(len(h))
    expand_w_rev[:len(w)] = w[::-1]
    g = g - conv_fft(h, expand_w_rev)
    g = g[:len(h)]
    # Actual gradient for negative sum of weighted correlations
    return - np.sum(-β * c * w), -g

        dx = 2.0 / n
        corr = correlate(h, 1 - h, mode="full")
        # Only consider non-zero lags
        max_corr = np.max(corr[len(corr)//2:1 + len(corr)//2])

    best_c5 = evaluate_c5(best_h)
    
    # Functions with upward scaling of β
    def objective_surrogate(h, β):
        return smooth_surrogate(h, β)
    
    # Restart-safe implementation
    β = 1.0
    α = 0.1
    β_max = 200.0  # max temperature to go near true max
    γ = 2.0        # β multiplication factor
    δ = 0.9        # decrease α
    β_steps = 0    # track how many β changes we have
    
    restarts = 15
    for restart in range(restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
        # Create a new initial vector through small random perturbations
        h_new = project_onto_bounded_simplex(h0_upsampled + np.random.normal(0, 0.01, n_points))
        h = h_new.copy()
        
        # Projected gradient descent within β scale
        for _ in range(1500):
            if time.time() - start_time > budget_s * 0.99:
                break
            # Compute current objective value using smooth surrogate
            objective_val, grad = smooth_obj_and_grad(h, β)
            # Line search
            τ = α
            while τ > 1e-12:
                h_trial = project_onto_bounded_simplex(h - τ * grad)
                trial_val = smooth_surrogate(h_trial, β)
                if trial_val <= objective_val - 0.1 * τ * (np.linalg.norm(grad)**2):
                    break
                else:
                    τ /= 2
            if τ == 0:
                break
            h = h_trial
            
            # Convergence test
            if τ * np.linalg.norm(grad) < 1e-9 or abs(objective_val - smooth_surrogate(h, β)) < 1e-12:
                # Increase β to sharpen the approximation
                β = min(β * γ, β_max)
                α *= δ
                β_steps += 1
                if β >= β_max or β_steps > 20:
                    break

        c5_eval = evaluate_c5(h)
        if c5_eval < best_c5:
            best_h = h.copy()
            best_c5 = c5_eval

    def final_obj(h):
        return evaluate_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) * (2.0 / n_points) - 1
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    res = minimize(
        final_obj,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-15, 'gtol': 1e-15, 'maxiter': 100}
    )
    
    if res.success:
        final_h = res.x
        final_h = project_onto_bounded_simplex(final_h)
        final_c5 = evaluate_c5(final_h)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
        
    return (best_h.tolist(), best_c5, n_points)
