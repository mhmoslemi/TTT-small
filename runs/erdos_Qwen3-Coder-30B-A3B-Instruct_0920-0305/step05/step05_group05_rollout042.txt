import numpy as np
import time

def project_capped_simplex(v, T):
    """Project vector v onto the capped simplex: 0 <= x <= 1, sum(x) = T"""
    v = np.asarray(v)
    lmin, lmax = 0.0, 1.0
    while lmin < lmax - 1e-10:
        l = (lmin + lmax) / 2.0
        x = np.clip(v - l, 0, 1)
        if x.sum() > T:
            lmin = l
        else:
            lmax = l
    return np.clip(v - lmin, 0, 1)

def compute_O_and_grad(h, alpha):
    """Compute overlap values O_k and their smoothed gradient."""
    dx = 2.0 / len(h)
    n = len(h)
    # compute all O_k = dx * sum h[i] * (1 - h[i+k]) over all k
    h1 = 1.0 - h
    corr = np.correlate(h, h1, mode='full')
    O = corr * dx
    # get the soft-max weights
    O_max = O.max()
    # for numerical stability, subtract max
    exp_O = np.exp(alpha * (O - O_max))
    w = exp_O / exp_O.sum()
    # compute gradient: gradient w.r.t h
    grad = np.zeros_like(h)
    for k in range(len(O)):
        k_offset = k - (n - 1)
        # slice indices in [-n+1, n-1]
        if k_offset >= 0:
            # shift h back by k_offset: h[i] becomes h[i+k_offset]
            #  h[i+k_offset] * (1 - h[i+k_offset])
            #  same for 1 - h[i], becomes 1 - h[i-k_offset]
            # we compute: (1 - h_{i+k_offset}) - h_{i-k_offset}
            h_shift = np.roll(h1, -k_offset)
            h_shift[:k_offset] = 0.0
            h_shift = h_shift[:n]
            # substrate h_{i-k_offset} = np.roll(h, k_offset)
            h_rev = np.roll(h, k_offset)
            h_rev[n-k_offset:] = 0.0
            h_rev = h_rev[:n]
            grad += w[k] * (h_shift - h_rev)
        else:
            # inverse case
            k_offset = -k_offset
            h_shift = np.roll(h1, k_offset)
            h_shift[-k_offset:] = 0.0
            h_shift = h_shift[:n]
            h_rev = np.roll(h, -k_offset)
            h_rev[:k_offset] = 0.0
            h_rev = h_rev[:n]
            grad += w[k] * (h_shift - h_rev)
    return O, grad

    best_h = None
    best_C5 = float('inf')
    
    num_restarts = 0
    while (time.time() - start_time) < budget_s * 0.95:
        num_restarts += 1
        # Try different initialization strategies
        init_type = num_restarts % 5
        if init_type == 0:
            # Uniform random
            h = np.random.rand(n)
        elif init_type == 1:
            # Sine shape
            x = np.linspace(0, 2, n)
            h = 0.5 + 0.4 * np.sin(np.pi * x)
        elif init_type == 2:
            # Gaussian shape
            x = np.linspace(0, 2, n)
            h = np.exp(-0.5 * ((x - 1.0) / 0.2) ** 2)
        elif init_type == 3:
            # Triangular
            x = np.linspace(0, 2, n)
            h = 1.0 - np.abs(x - 1.0)
            h = h / h.sum() * T  # normalize to sum to T
        else:  # init_type == 4
            # Binary mask
            h = np.zeros(n)
            mid = n // 2
            width = n // 4
            start = max(0, mid - width // 2)
            end = min(n, mid + width // 2)
            h[start:end] = 1.0
        
        # Project to feasible simplex
        h = project_capped_simplex(h, T)
        # optionally enforce symmetry (optional for efficiency)
        if num_restarts % 3 == 0:
            h = (h + h[::-1]) / 2.0
            
        # smooth gradient descent
        max_iters = 3000
        alpha_start = 5.0
        alpha_end = 30.0
        lr_0 = 0.5
        
        for t in range(max_iters):
            if (time.time() - start_time) > budget_s * 0.99:
                break
            # Anneal alpha
            alpha = alpha_start + (alpha_end - alpha_start) * t / max_iters
            O, grad = compute_O_and_grad(h, alpha)
            lr = lr_0 / np.sqrt(t + 1)
            h = h - lr * grad
            h = project_capped_simplex(h, T)

                    best_h = h.copy()
        
        # Subgradient polishing
        max_sub_iter = 500
        # sort overlap activations by value
        O = np.correlate(h, 1 - h, mode='full') * dx
        active_shifts = np.argsort(-O)[:3]  # top 3 shifts
        sub_lr = 0.02 / np.sqrt(1)
        
        for t in range(max_sub_iter):
            if (time.time() - start_time) > budget_s * 0.99:
                break
            # compute true gradient w.r.t active shift
            grad = np.zeros_like(h)
            for k in active_shifts:
                offset = k - (n - 1)
                if offset >= 0:
                    # h_rev = h[-offset:] + 0 for missing elements → Zero padding at end
                    h_rev = np.roll(h, offset)
                    h_rev[:offset] = 0.0
                    h_rev = h_rev[:n]
                    # h_shift = h[:-offset] + 0 for missing → zero padding at front  
                    h_shift = np.roll(h, -offset)
                    h_shift[-offset:] = 0.0
                    h_shift = h_shift[:n]
                    # gradient wrt overlap O_k = dx * sum h[i] * (1 - h[i+k])
                    # its partial derivative wrt h[i] = dx * (1 - h[i+k] - h[i-k])
                    # realize that h[i+k] and h[i-k] will be shifted versions of h
                    grad += (dx * (h_shift - h_rev))
                else:
                    offset = -offset
                    h_rev = np.roll(h, -offset)
                    h_rev[:offset] = 0.0
                    h_rev = h_rev[:n]
                    h_shift = np.roll(h, offset)
                    h_shift[-offset:] = 0.0
                    h_shift = h_shift[:n]
                    # gradient wrt overlap data
                    grad += (dx * (h_shift - h_rev))
                    
            h = h - sub_lr * grad
            h = project_capped_simplex(h, T)
            
            c5_now = np.max(np.correlate(h, 1 - h, mode='full') * dx)
            if c5_now < best_C5:
                best_C5 = c5_now
                best_h = h.copy()
                
    # Final refinement using SLSQP if time allows
    if best_h is not None and (time.time() - start_time) < budget_s * 0.99:

                return np.max(np.correlate(h, 1 - h, mode='full') * dx)

                return np.sum(h) * dx - 1.0

                           bounds=[(0, 1)] * len(best_h),
                           constraints={'type': 'eq', 'fun': cons},
                           options={'maxiter': 200})

                c5_candidate = obj(res.x)
