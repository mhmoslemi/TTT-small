import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    low, high = -2.0, 2.0
    for _ in range(50):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target_sum) < 1e-12:
            return y
        if s > target_sum:
            high = mid
        else:
            low = mid
    y = np.clip(x + low, 0.0, 1.0)
    return y

def evaluate_c5(h):
    """Evaluate C5 using correlation"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def gradient_smoothing(h, alpha):
    """Compute gradient of log-sum-exp smoothed objective"""
    n = len(h)
    dx = 2.0 / n
    
    # Compute correlation I = ∫ h(x)(1 - h(x+k))dx for all shifts k
    cov_full = correlate(h, 1.0 - h, mode='full')
    I = cov_full  # shape 2n-1
    I_max = I.max()
    
    # Compute softmax weights
    exp_vals = np.exp(alpha * (I - I_max))
    w = exp_vals / exp_vals.sum()
    p = w / w.sum()
    
    # Compute gradient as weighted sum of per-lag gradients
    grad = np.zeros(n)
    
    for idx, p_val in enumerate(p):
        lag = idx - (n - 1)  # lag from - (n-1) to n-1
        
        if lag >= 0:
            # (1 - h[i+lag]) contribution: for i in [0, n-1-lag]
            if n - lag > 0:
                grad[:n-lag] += p_val * (1.0 - h[lag:])
            # edge contribution: for i in [n-lag, n-1] where h[i+lag] = 0
            if lag > 0:
                grad[n-lag:] += p_val * 1.0
            # -h[i-lag] contribution: for i in [lag, n-1]
            if lag > 0:
                grad[lag:] -= p_val * h[:n-lag]
            else:
                # lag == 0 case: grad -= p_val * h
                grad -= p_val * h
        else:
            # lag < 0: let d = -lag
            d = -lag
            # (1 - h[i-d]) contribution: for i in [d, n-1]
            if n - d > 0:
                grad[d:] += p_val * (1.0 - h[:n-d])
            # edge contribution: for i in [0, d-1] where h[i+lag] = h[i-d] = 0
            grad[:d] += p_val * 1.0
            # -h[i+d] contribution: for i in [0, n-d-1]
            grad[:n-d] -= p_val * h[d:]
    
    # Apply dx to match scaling (since correlation I is scaled by dx)
    return grad * dx

def descent_step(h, eta, alpha, target_sum):
    """Perform one descent step with line search"""
    grad = gradient_smoothing(h, alpha)
    
    # Try different step sizes
    for _ in range(5):
        h_new = project_to_sum_box(h - eta * grad, target_sum)
        c5_new = evaluate_c5(h_new)
        c5_old = evaluate_c5(h)
        if c5_new <= c5_old:
            return h_new, c5_new
        eta *= 0.5
    return h, evaluate_c5(h)

def gradient_descent_run(h_start, time_limit, target_sum):
    """Run gradient descent for a limited time"""
    h = h_start.copy()
    c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    start_time = time.time()
    it = 0
    alpha = 0.5
    eta = 0.5
    
    while time.time() - start_time < time_limit and it < 5000:
        # Schedule parameters
        progress = min(it / 2000, 1.0)
        alpha = 0.5 * (50.0 ** progress)  # Gradually increase alpha
        eta = 0.5 * (0.995 ** it)         # Decreasing step size
        
        h, c5 = descent_step(h, eta, alpha, target_sum)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        it += 1
    
    return best_h, best_c5

def local_refine(h, target_sum):
    """Refine solution using SLSQP optimizer"""
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h))]
    
    try:
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, target_sum)
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
    except:
        pass
    return h, evaluate_c5(h)
