import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def project_onto_simplex(h, target_sum):
    """Project a vector h onto the bounded simplex: 0 <= h_i <= 1, sum(h) = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    # bisection on lambda to find the right adjustment
    # We want sum_i clip(h_i + lambda, 0, 1) = target_sum
    # Let f(lambda) = sum_i clip(h_i + lambda, 0, 1) - target_sum
    # Compute the derivative df/dlambda = sum_i indicator(h_i + lambda in (0,1))

    def f(lam):
        return np.sum(np.clip(h + lam, 0, 1)) - target_sum

    a, b = -h.min() - 1.0, 1.0 - h.max()
    for _ in range(50):
        if abs(f(a)) < 1e-12: break
        if abs(f(b)) < 1e-12: break
        mid = (a + b) / 2.0
        if f(mid) < 0: a = mid
        else: b = mid

    lam = (a + b) / 2.0
    h = np.clip(h + lam, 0, 1)
    return h

def compute_correlations(h):
    """Compute all correlations c_k = sum_i h[i] * h[i+k] for nonzero shifts k."""
    # Use FFT for linear convolution
    n = len(h)
    padding = 2 * n - 1
    h_padded = np.pad(h, (0, n-1), mode='constant', constant_values=0)
    corr_padded = np.convolve(h_padded, h_padded, mode='full')
    # Extract only the non-zero shifts from -n+1 to n-1, 
    # specifically from 1 to n-1 (removing shift==0 since h[i]*h[i] part is ignored)
    return corr_padded[n:]

def compute_softmin_gradient(h, c, beta):
    """Gradient of softmin (1/beta * log sum exp(-beta * c_k)) w.r.t h."""
    n = len(h)
    exp_term = np.exp(-beta * (c - np.max(c)))  # subtract max for numerical stability
    Z = np.sum(exp_term)
    w = exp_term / Z
    # Gradient: - sum_k w_k * grad c_k 
    # Since c_k = sum_i h_i * h_{i+k} = sum_{i+k < n} h_i * h_{i+k}, 
    # and assuming periodic boundary: grad c_k = h_{k}, h_{-k} 
    # Capping for out of bounds
    h_padded = np.pad(h, (0, n-1), mode='constant', constant_values=0)

    w_forward = np.zeros(2*n-1)
    w_backward = np.zeros(2*n-1)
    # compute weights as vectors
    for k in range(1,n):
        if k < n:
            w_forward[k] = w[k-1]
        if k < n:
            w_backward[k] = w[k-1]

    # Expand to match full length with padding
    c_forward = np.zeros(2*n - 1)
    c_backward = np.zeros(2*n - 1)
    
    # For forward convolution (shifted h)
    c_forward = np.convolve(h_padded, w_forward, mode='full')
    # For backward convolution (computed via reversed w)  
    c_backward = np.convolve(h_padded, w_backward[::-1], mode='full')
    
    # Take appropriate portion
    g = -0.5 * (c_forward[n-1:n-1+n] + c_backward[n-1:n-1+n])
    return g

def compute_softmin_objective(h, c, beta):
    """Compute objective value: 1 + (1/beta)* log sum exp(-beta * c_k)"""
    exp_term = np.exp(-beta * (c - np.max(c)))
    Z = np.sum(exp_term)
    log_sum_exp = np.log(Z) + np.max(c)
    return 1.0 + (1.0 / beta) * log_sum_exp

def evaluate_erdos_solution(h):
    """Return the actual C5 value via a full correlation computation."""
    # h[x] for x in [0,2]; discretized as n steps
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    # max_k  integral h(x)(1-h(x+k)) dx = max_k corr[k]
    # remove the shift k=0 which is not included in the objective
    max_corr = np.max(corr[1 : 1 + n])  # also ignore negative indices, i.e., extra.
    return max_corr * dx
