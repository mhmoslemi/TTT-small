import numpy as np

def generate_legendre_mask(n):
    """Generate a Legendre-based binary mask if possible, otherwise fallback to random."""
    p = n // 2
    if not is_prime(p):
        # Find the largest prime less than p
        for q in range(p - 1, 1, -1):
            if is_prime(q):
                p = q
                break
        else:
            # Fallback to random mask if no prime found
            return np.random.permutation([1] * (n // 2) + [0] * (n // 2))
    
    # Generate quadratic residues
    residues = set()
    for a in range(1, p):
        residues.add((a * a) % p)
    
    # Create binary vector
    b = np.zeros(p, dtype=int)
    for r in residues:
        if r < p:
            b[r] = 1
    b = b.astype(int)
    
    # Check if sum is p-1, correct it
    if np.sum(b) == p - 1:
        idx = np.where(b == 0)[0]
        if len(idx) > 0:
            b[idx[0]] = 1
    
    # Repeat to get full vector
    h = np.tile(b, 2)
    h = h[:n]
    
    return h

def is_prime(n):
    if n < 2:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False
    for i in range(3, int(math.sqrt(n)) + 1, 2):
        if n % i == 0:
            return False
    return True

def evaluate_c5_discrete(h):
    """Approximate C5 using discrete correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_c5_exact(h):
    """Evaluate the true C5 constant using full correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def compute_corr(h):
    """Compute the full correlation array efficiently."""
    n = len(h)
    corr = np.zeros(2*n - 1)
    # Front part (correlations for negative shifts) 
    for k in range(1, n):
        corr[n-1-k] = np.dot(h[:n-k], h[k:])
    # Middle part (zero lag)
    corr[n-1] = np.dot(h, h)
    # Back part (positive shifts)
    for k in range(1, n):
        corr[n-1+k] = np.dot(h[k:], h[:n-k])
    return corr

def get_worst_shifts(corr, threshold=0.001):
    """"Return indices of worst shifts."""
    max_val = np.max(corr)
    return np.where(np.abs(corr - max_val) < threshold)[0] - (len(corr) // 2)

def swap_elements(h, i, j, corr, shifts):
    """Swap 1 and 0 at positions i and j in h, update corr."""
    if h[i] == h[j]:
        return False

        i, j = j, i
    old_1 = h[i]
    old_0 = h[j]
    h[i], h[j] = old_0, old_1
    for k in range(len(shifts)):
        m = shifts[k]  # lag
        old_corr = corr[k]
        s = (len(corr) // 2)
        if m < 0:  # positive lag
            if i >= -m and i + m < len(h):
                # contribution of i (changes)  
                corr[k] -= (h[i] * h[i+m] - old_1 * old_0)
            if j >= -m and j + m < len(h):
                # contribution of j (changes)  
                corr[k] -= (h[j] * h[j+m] - old_0 * old_1)
        else:  # nonnegative lag
            # update it
            corr[k] = 0
            # heavily optimized for dense updates (inner loop here)
            # instead of fully recomputing the entire correlation element
            for l in range(len(h) - abs(m)):
                corr[k] += h[l] * (1 - h[l + m]) if m >= 0 else h[l+m] * (1 - h[l])
    return True

def smooth_max(h_list, alpha=30.0):
    """Smooth approximation to the max element of h_list using log-sum-exp."""
    return 1.0 / alpha * np.log(np.sum(np.exp(alpha * np.array(h_list))))

def polish_continuous(h_start, n_points, dx):
    """Polish the solution with continuous method."""
    def objective(x):
        # Compute max of the sum h[i]*(1-h[i+k])
        sample_size = len(x)
        vals = []
        for k in range(sample_size):
            total = 0
            for i in range(sample_size):
                if i + k < sample_size:
                    total += x[i] * (1 - x[i+k])
            vals.append(total)
        return smooth_max(vals)
    
    def constraint_function(x):
        return np.sum(x) - sample_size / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_function}
    
    try:
        result = minimize(objective, h_start, method='SLSQP', bounds=bounds, 
                         constraints=constraints, options={'ftol': 1e-10, 'gtol': 1e-10})
        if result.success:
            x_opt = result.x
            # Scale to meet constraint exactly
            scale = (n_points / 2) / np.sum(x_opt)
            x_opt = np.clip(x_opt * scale, 0, 1)
        else:
            x_opt = h_start.copy()
    except Exception:
        x_opt = h_start.copy()
    
    return x_opt
