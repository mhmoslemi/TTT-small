def project_simplex(v, sum_value):
    """Project vector v onto the capped simplex: 0 <= v[i] <= 1, sum(v) = sum_value"""
    if np.sum(v) == sum_value:
        return v
    
    # Binary search for the Lagrange multiplier
    v = np.array(v)
    u = np.sort(v)[::-1]
    rho = 0
    for i in range(len(u)):
        if u[i] - (np.sum(u[:i+1]) - sum_value) / (i+1) > 0:
            rho = i
            break
    else:
        # All elements too small
        rho = len(u) - 1
    
    lambda_ = (np.sum(u[:rho+1]) - sum_value) / (rho+1)
    return np.clip(v - lambda_, 0, 1)

def compute_correlations(h):
    """Fast computation of all correlations using FFT"""
    n = len(h)
    # Correlation using FFT: full convolution of h and (1-h)
    H = rfft(h)
    G = rfft(1 - h)
    corr = irfft(H * np.conj(G))
    # Take only the relevant part (full mode)
    return corr[:2*n-1]

def evaluate_c5(h, dx):
    """Evaluate C5 using fast FFT-based correlation"""
    n = len(h)
    corr = compute_correlations(h)
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def interpolate_control_points(control_values, n_points, n_control):
    """Interpolate control points to full grid"""
    # Control points on [0, 2] 
    x_controls = np.linspace(0, 2, n_control)
    # Full grid
    x_full = np.linspace(0, 2, n_points)
    
    # Linearly interpolate
    h_full = np.interp(x_full, x_controls, control_values)
    return h_full


def repair_candidate(control_values, n_points, n_control, target_sum):
    """Repair and return full h vector"""

    control_values = np.clip(control_values, 0, 1)

    h = interpolate_control_points(control_values, n_points, n_control)
    
    # Project onto capped simplex (sum = target_sum)
    h = project_simplex(h, target_sum)
    
    return h

def soft_max_objective(control_values, n_points, n_control, target_sum, alpha, dx):
    """Soft max surrogate for the max overlap"""
    h = repair_candidate(control_values, n_points, n_control, target_sum)
    corr = compute_correlations(h)
    max_corr_val = np.max(corr)
    
    # Soft max for smoothness
    # For numerical stability, subtract max before exponentiating
    O = corr * dx
    max_O = np.max(O)
    exp_terms = np.exp(alpha * (O - max_O))
    soft_max_val = (1/alpha) * (np.log(np.sum(exp_terms)) + max_O)
    return soft_max_val
