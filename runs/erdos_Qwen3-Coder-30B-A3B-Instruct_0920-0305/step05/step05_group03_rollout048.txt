```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction with n=44 to be even, sum = 22
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    # If sum not exactly n/2, adjust
    current_sum = np.sum(initial_h)
    if abs(current_sum - n_points / 2) > 1e-10:
        scale_factor = (n_points / 2) / current_sum
        initial_h = initial_h * scale_factor

    # Store best solution
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(initial_h)

    # Active set of shifts to consider
    K = [0]
    
    # Simple integer variable bound handling
    max_iterations = 500
    iteration = 0

    while (time.time() - start_time) < budget_s * 0.95 and iteration < max_iterations:
        iteration += 1
        
        # Build variables
        h_vars = Variable(n_points)
        tau = Variable()
        
        # Create list to collect constraints
        constraints = [0 <= h_vars, h_vars <= 1, sum(h_vars) == n_points / 2]
        # Ensure symmetry: h[i] = h[n-1-i] -- special handling 
        for i in range(n_points // 2):
            constraints.append(h_vars[i] == h_vars[n_points - 1 - i])

        # Set up y variables for active shifts
        y_vars = {}
        for k in K:
            y_vars[k] = Variable(n_points)
            # Add McCormick constraints
            # y_{i,k} >= 0
            constraints.extend([y_vars[k] >= 0])
            # y_{i,k} <= h_i
            constraints.extend([y_vars[k][i] <= h_vars[i] for i in range(n_points)])
            # y_{i,k} <= h_{i+k}
            # but index may be out-of-bounds
            if k >= 0:
                # OOB is 0 -> y_i,k <= h_i+k -> but if i+k >=n then h_{i+k} = 0
                for i in range(n_points - k):
                    constraints.append(y_vars[k][i] <= h_vars[i+k])
                for i in range(n_points - k, n_points):
                    constraints.append(y_vars[k][i] <= 0)
            else:
                # k < 0
                for i in range(-k):
                    constraints.append(y_vars[k][i] <= 0)
                for i in range(-k, n_points):
                    constraints.append(y_vars[k][i] <= h_vars[i+k])
            # y_{i,k} >= h_i + h_{i+k} - 1
            if k >= 0:
                for i in range(n_points - k):
                    constraints.append(y_vars[k][i] >= h_vars[i] + h_vars[i+k] - 1)
                for i in range(n_points - k, n_points):
                    constraints.append(y_vars[k][i] >= h_vars[i] - 1)
            else:
                for i in range(-k):
                    constraints.append(y_vars[k][i] >= h_vars[i] + h_vars[i+k] - 1)
                for i in range(-k, n_points):
                    constraints.append(y_vars[k][i] >= h_vars[i+k] - 1)
        
        # Add the main constraints for each k in K:
        for k in K:
            # Tau >= sum(h) - sum(y_{i,k})
            # Note: sum(h) is a constant (n/2) in current formulation
            # So we use: tau >= (n/2) - ∑ y_{i,k}
            constraints.append(tau >= n_points / 2 - sum(y_vars[k]))

        # Objective: minimize tau
        objective = Minimize(tau)

        # Define and solve the problem
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
        except:
            # Fallback to CBC if GLOP fails or not available
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except:
                # If both fail, break and try a local refinement
                break
                
        # Extract values
        if prob.status == 'optimal':
            h_candidate = np.array(h_vars.value).flatten()
            tau_value = tau.value
        else:
            break
            
        # Redo the constraint that helps in maintaining symmetry explicitly just in case
        h_symm = h_candidate.copy()
        if n_points % 2 == 0:
            for i in range(n_points // 2):
                avg_val = (h_symm[i] + h_symm[n_points - 1 - i]) / 2
                h_symm[i] = avg_val
                h_symm[n_points - 1 - i] = avg_val
                
        # Evaluate true overlaps using np.correlate (since we want actual C5)
        # Compute all correlations in the full mode
        corr_all = correlate(h_symm, 1 - h_symm, mode='full')
        # Correct for indexing: shift from [0, 2n-2] to [-n+1, n-1]
        # Figure out size of correlation
        total_len = len(corr_all)
        half_len = (total_len - 1) // 2
        corr_shifted = corr_all[half_len : half_len + n_points]
        
        # The maximum we seek is still in full version:
        c5_candidate = np.max(corr_all) * dx
        
        # Also compute with the constraint shift to make sure our procedure gives valid estimate:
        max_o_k = np.max(corr_all)
        
        # Check convergence with tolerances using actual max correlation calculated
        if not (np.isnan(c5_candidate) or np.isinf(c5_candidate)):
            if c5_candidate < best_c5:
                best_h = h_symm.copy()
                best_c5 = c5_candidate

        # Find worst shift from full correlation data
        idx_max = np.argmax(corr_all)
        max_shift = idx_max - half_len  # Convert from full index to shift index in [-n+1, n-1]
        
        # Terminate if current solution is close to optimal
        # due to lower-bound estimated from LP being equal to actual computed C5 within tolerance
        if abs(c5_candidate - max_o_k * dx) < 1e-6:
            # We can consider this to be optimal w.r.t. the relaxed constraints,
            # but we have true max height already
            # Although this is not needed since QP will be dual, but we can still let it finish its loop
            pass
            
        # Add the most violated shift to active set to refine
        if max_shift not in K:
            # If max_shift is too extreme (outside the region of k-values we want to keep), be careful here
            if -1*(n_points - 1) <= max_shift <= n_points - 1:
                K.append(max_shift)
        else:
            # Already in the set, might be a better machine locals that we're repeating
            pass

        # Limit iterations further based on runtime
        if (time.time() - start_time) > budget_s * 0.9:
            break

    # Final polishing step using projected sub-gradient descent
    h_final = best_h.copy()
    c5_final = best_c5

    #Compute best overlap as initial
    def compute_gradient(h, k):
        # compute the gradient of O_k(h) = sum(h_i) - sum(h_i * h_{i+k})
        # g_i = 1 - [h_{i+k} + h_{i-k}] (with zero-padding outside)
        # This corresponds to derivative of - (h_i * h_{i+k})
        # Note that indices might go beyond
        # if k > 0, then h_{i-k} = 0 for i < k -> so just h_{i+k}
        # if k < 0, then h_{i+k} = 0 for i+k < 0 -> so just h_{i-k} 
        g = np.ones_like(h)
        h_padded = np.pad(h, max(0, abs(k)), 'constant', constant_values=0)
        if k > 0:
            # shift h to the left by k positions
            h_k = h_padded[k:-k] if k > 0 and k < n_points else h_padded[k:] if k > 0 else h_padded[:n_points - k]
        elif k < 0:
            # shift h to the right by abs(k)
            h_k = h_padded[-k:] if -k > 0 and -k < n_points else h_padded[:-k] if -k > 0 else h_padded[:n_points - (-k)]
        else:
            h_k = h
        
        if k > 0:
            h_k = h_padded[k: n_points]
        elif k < 0:
            h_k = h_padded[0:-k]
        else:
            h_k = h_padded[0:n_points]

        g -= h_k  # element-wise subtraction
        return g

    def project_simplex(v, target_sum):
        # Project vector v onto { x | 0 <= x <= 1, sum(x) = target_sum }
        v_sorted = np.sort(v)[::-1]
        rho = 0
        for j in range(len(v_sorted)):
            if v_sorted[j] - (np.sum(v_sorted[:j+1]) - target_sum) / (j+1) > 0:
                rho = j
        theta = (np.sum(v_sorted[:rho+1]) - target_sum) / (rho+1) if rho >= 0 else 0
        return np.clip(v - theta, 0, 1)

    max_iter = 2000
    eta = 0.5 * dx  # step size
    for i in range(max_iter):
        if (time.time() - start_time) > budget_s * 0.99:
            break
            
        # Evaluate all overlaps for the best solution
        corr_full = correlate(h_final, 1 - h_final, mode='full')
        max_overlap_idx = np.argmax(corr_full)
        max_shift_idx = max_overlap_idx - (len(corr_full) - 1) // 2

        # Find the set of maximising shifts
        max_val = np.max(corr_full)
        active_shifts = []
        for k_i in range(-(n_points - 1), n_points):
            idx = k_i + (n_points - 1)
            if idx >= 0 and idx < len(corr_full) and abs(corr_full[idx] - max_val) < 1e-8:
                active_shifts.append(k_i)
                
        # compute sub-gradient from all active shifts
        sub_g = np.zeros(n_points)
        for k in active_shifts:
            g_k = compute_gradient(h_final, k)
            sub_g += g_k

        if len(active_shifts) > 0:
            sub_g /= len(active_shifts)

        # update step
        h_new = h_final - eta * sub_g
        h_new = project_simplex(h_new, n_points / 2)
        
        # Apply symmetry
        h_new_symm = h_new.copy()
        for i in range(n_points // 2):
            avg_val = (h_new_symm[i] + h_new_symm[n_points - 1 - i]) / 2
            h_new_symm[i] = avg_val
            h_new_symm[n_points - 1 - i] = avg_val
            
        h_new = h_new_symm

        # compute new C5
        new_c5 = evaluate_c5(h_new)
        
        # accept or reject
        if new_c5 < c5_final:
            h_final = h_new.copy()
            c5_final = new_c5
        else:
            eta *= 0.5
            if eta < 1e-7:
                break
                
    # Possibly binary rounding step
    # Option to make binary _improvement by swapping and using more discrete search (低位忍让)
    # Threshold to binary by taking largest n/2 values
    h_binary = np.zeros(n_points)
    sorted_idx = np.argsort(h_final)[::-1]  # descending sort
    indices_of_largest = sorted_idx[:n_points // 2]
    h_binary[indices_of_largest] = 1
    h_binary = project_simplex(h_binary, n_points / 2)
    
    c5_binary = evaluate_c5(h_binary)
    
    if c5_binary < c5_final:
        h_final = h_binary
        c5_final = c5_binary

    return (h_final.tolist(), c5_final, n_points)
```
