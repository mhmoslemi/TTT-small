import numpy as np
import time
from scipy.optimize import minimize, differential_evolution, dual_annealing

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    # compute cross correlation
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def normalize_h(h):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    total = np.sum(h)
    if total == 0:
        # set uniform distribution
        h = np.full(n, target / n)
        return h
    # scale to target sum while respecting bounds
    # iterative projection to bounded simplex
    h = h * (target / total)
    # iterate to enforce bounds exactly
    for _ in range(10):
        # clip again
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        diff = target - total
        if abs(diff) < 1e-12:
            break
        # free indices
        free = (h > 0) & (h < 1)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            # cannot adjust further, just scale all
            h = h * (target / total)
            continue
        delta = diff / n_free
        h[free] += delta
    # final clip
    h = np.clip(h, 0.0, 1.0)
    # final scaling to fix any tiny leftover error
    total = np.sum(h)
    if abs(total - target) > 1e-9:
        h = h * (target / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_correlations(h):
    # returns array of correlations for shifts 0..n-1 (non-negative)
    corr_full = np.correlate(h, h, mode='full')
    n = len(h)
    # shift s corresponds to index n-1 + s
    corr = corr_full[n-1:]  # s=0..n-1
    return corr

def gradient_correl_shift(h, shift):
    n = len(h)
    # compute gradient of corr for given shift
    grad = np.zeros_like(h)
    if shift == 0:
        grad = 2 * h  # derivative of sum_i h[i]^2 = 2h[i]; but we only need for shift>0
    else:
        # terms: h[i]h[i+shift]
        # for i from 0 to n-1-shift: derivative w.r.t h[i] includes h[i+shift]; w.r.t h[i+shift] includes h[i]
        # So grad[i] gets h[i+shift] (if i+shift < n)
        # and also gets h[i-shift] (if i-shift >=0)
        grad[:n-shift] += h[shift:]
        grad[shift:] += h[:n-shift]
        # This double count? Actually both contributions are correct.
    return grad

def gradient_ascent_worst_shift(h, max_iters=2000, step0=0.02, tol=1e-6):
    n = len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    for it in range(max_iters):
        corr = compute_correlations(h)
        # exclude shift 0; include shift up to n-1
        # compute min correlation among s>0
        # shift0 correlation is sum_i h[i]^2; always >=0; but to get min among >0
        # Let's restrict to s in 1..n-1
        min_corr = np.min(corr[1:])  # ignoring shift 0
        s_min = np.argmin(corr[1:]) + 1
        # compute gradient for increasing correlation at s_min
        grad = gradient_correl_shift(h, s_min)
        # step size may be scaled by (1 - min_corr) = current C5
        step = step0 * (1 - min_corr)  # or use other heuristics
        if step < 1e-12:
            break
        # make ascent step (increase correlation)
        h_new = h + step * grad
        # ensure bounds and sum
        h_new = normalize_h(h_new)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
        # accept new h anyway to continue gradient descent, else stay?
        # Option: if new C5 not improved, keep previous h
        if c5_new < best_c5:
            h = h_new
        else:
            # maybe reduce step?
            # simple fallback: keep h unchanged
            pass
        # early stop if improvement < tol
        if abs(best_c5 - c5_new) < tol:
            break
    return best_h, best_c5

        else:
            extra = n - len(h0)
            h0 = np.concatenate([h0, np.full(extra, 0.0)])
    h0 = normalize_h(h0)
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Stage 1: Gradient ascent from best_h
    h_g, c5_g = gradient_ascent_worst_shift(best_h, max_iters=3000, step0=0.02)
    if c5_g < best_c5:
        best_h, best_c5 = h_g, c5_g
    
    # Stage 2: Random gradient restarts
    n_restarts = 200
    max_iters_per = 1500
    while time.time() - start < budget_s * 0.6 and n_restarts > 0:
        # random start
        rand = np.random.rand(n)
        h_rand = normalize_h(rand)
        h_opt, c5_opt = gradient_ascent_worst_shift(h_rand, max_iters=max_iters_per, step0=0.015)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        n_restarts -= 1
    
    # Stage 3: Local SLSQP refinement
    bounds = [(0.0,1.0)] * n
    def obj(x):
        return evaluate_c5(x)
    def eq_cons(x):
        return np.sum(x) * (2.0/n) - 1.0
    cons = {'type':'eq','fun':eq_cons}
    # Use best_h as start
    try:
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'ftol':1e-12,'maxiter':500})
        if res.success:
            h_res = normalize_h(res.x)  # ensure exact sum
            c5_res = evaluate_c5(h_res)
            if c5_res < best_c5:
                best_h, best_c5 = h_res, c5_res
    except Exception:
        pass
    
    # Stage 4: Differential evolution for fine global
    # Use population initialised around best_h
    # Build init list: best_h + small random perturbations
    pop_init = [best_h]
    for _ in range(9):
        pert = best_h + np.random.normal(scale=0.05, size=n)
        pert = np.clip(pert,0,1)
        pert = normalize_h(pert)
        pop_init.append(pert)
    bounds = [(0.0,1.0)] * n
    def de_obj(x):
        # evaluate normalized
        h = normalize_h(x)
        return evaluate_c5(h)
    remaining = budget_s - (time.time() - start)
    # allocate limited iterations
    maxiter_de = max(10, int(remaining / 2.0))  # approximate remaining time
    try:
        de_res = differential_evolution(de_obj, bounds, init=pop_init, maxiter=maxiter_de,
                                        popsize=15, strategy='best1bin',
                                        seed=seed, polish=False)
        if de_res.success:
            h_de = normalize_h(de_res.x)
            c5_de = evaluate_c5(h_de)
            if c5_de < best_c5:
                best_h, best_c5 = h_de, c5_de
    except Exception:
        pass
    
    #   final attempt: run a short dual annealing
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        try:
            da_res = dual_annealing(de_obj, bounds=bounds, maxiter=1000, seed=seed)
            if da_res.success:
                h_da = normalize_h(da_res.x)
                c5_da = evaluate_c5(h_da)
                if c5_da < best_c5:
                    best_h, best_c5 = h_da, c5_da
        except Exception:
            pass
    
    # Ensure final best obey constraints
    best_h = normalize_h(best_h)
    best_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), float(best_c5), n)
