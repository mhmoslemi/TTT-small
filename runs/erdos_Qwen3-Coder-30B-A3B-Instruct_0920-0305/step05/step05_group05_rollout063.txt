import numpy as np
import time

def project_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex: 0 <= x <= 1, sum(x) = target_sum"""
    v = np.asarray(v)
    low, high = 0.0, 1.0
    # Binary search for the correct lambda
    lbd = 0.0
    u = 1.0
    # We need to find lambda such that sum(clamp(v-lambda, 0, 1)) = target_sum
    for _ in range(50):
        lamb = (lbd + u) / 2.0
        proj = np.clip(v - lamb, low, high)
        s = np.sum(proj)
        if s > target_sum:
            lbd = lamb
        else:
            u = lamb
    lamb = (lbd + u) / 2.0
    result = np.clip(v - lamb, low, high)
    # Scale to match sum exactly (ensure numerical accuracy)
    actual_sum = np.sum(result)
    if abs(actual_sum - target_sum) > 1e-10:
        # Simple rescaling for minor deviations
        scale = target_sum / actual_sum
        result *= scale
    return result

def compute_O_and_grad(h, alpha):
    """Compute O_k = dx * sum h[i] * (1 - h[i+k]) for all k, and its gradient."""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlations O_k for all shifts k
    corr = np.correlate(h, 1 - h, mode='full')  # length = 2*n - 1
    O = corr * dx  # ∫ h(x)(1 - h(x+k)) dx
    
    # Compute the smooth max gradient with reweighted sum of gradients
    # For each shift k, the contribution to gradient of h[i] is:
    # dx * ( 1 - h[i+k] - h[i-k] )
    # But we must be careful with the indexing, since we are treating very
    # long arrays from np.correlate with offset indexing
    
    # Use softmax weights of α*O to compute weighted gradient
    m = np.max(O)
    w = np.exp(alpha * (O - m))
    w /= np.sum(w)
    
    # Compute "gradient" dF/dh where F = log(sum(exp(α * O_k))) 
    # and use vectorized form of the integral-like function
    # For h[i], the gradient contribution from all k is:
    # (1 - h[i+k]) - h[i-k]  for valid k
    
    grad = np.zeros(n)
    for k in range(len(O)):
        # Convert k-index to full correlation index
        # When computing correlations via np.correlate with 'full',
        # the shift k means index k in output when k >= 0.
        # Index zero is lag 0.
        # Assume the O bulk (in lin alg.) is still [0 ... n-1 ... k, ... 2n-2] where k is shift
        # C_k = Σ_i h[i] (1 - h[i+k])  
        # Hence for derivative at h[i] we get term (1 - h[i+k]) - h[i-k]
        
        if k < n:
            # Corresponds to shift that is base offset from h index
            # For O[k] (the k th shift/convolution)
            shifted_h = np.roll(h, -k)  # Right: h[i+k]
            # 1 - h[i+k] is (1 - shifted_h[i])
            # h[i-k]  is current h[i] shift using rolling
            shifted_neg_d = np.roll(h, k)  # Left shift: h[i-k] 
            # i-th element:
            # (1 - shifted_h[i]) + shifted_neg_d[i] = 1 - shifted_h[i] + shifted_neg_d[i]
            # this can be rewritten for i fixed:
            # gradients are 1 - h[i+k] - h[i-k]
            contrib = (1 - shifted_h) - shifted_neg_d
            grad += w[k] * contrib
        else:
            # In reflection; when k >= n, it corresponds to negative shifts
            # For index k >= n, it means k' = k - (n-1) which is the shift amount from
            # the actual overlap integral, e.g., O_{k}, 0 <= k <= 2n−2.
            # In terms of indexing for convolution, if O contains:
            #        0, 1, 2, ... n-1, ... 2n-2
            #      shift 0, 1, 2, ... up to n-1, ..., 2n-2  
            # k = 0 → h[0]*(1 - h[n-1]),    signal indexing 0
            # k = 1 → h[0]*(1 - h[n-2]) + h[1]*(1 - h[n-1]) ...
            # We model with actual numpy shift Operaotions
            
            # Correct way to compute this is to again remember O is full across 2n-1 shifts
            # For j = 0, ..., 2n−2, O[j] is the overlap when shift = j − (n−1)    
            # So for each fixed i, calculate 1 - h[i+j] - h[i-(j-n+1)] for all valid j
            # this is a bit tricky with slices -- easier to loop for now or build
            # it with a concatenation trick
            
            # Actually: just use:
            # numpy version: O[k] corresponds to shift k − (n−1) = k − n + 1.
            # So new shift is (k - n + 1), which can be negative. If so special handling needed.
            offset = k - (n - 1)
            # now we construct 1 - h[i + offset] - h[i - offset]
            if offset >= 0:
                shifted_h = np.roll(h, -offset)
                shifted_neg_d = np.roll(h, offset)
            else:
                shifted_h = np.roll(h, offset)
                shifted_neg_d = np.roll(h, -offset)
            
            # Note: the flattened output of np.correlate has indices corresponding to
            # the total length including negative shifts, from 0 to 2n-2
            # In the loop representation, it looks like:
            # O[k] = dx * Σ_i h[i] (1 - h[i + k - (n-1)])
            # Let k' = k - (n - 1), so k = k' + n - 1
            # Extend compute carefully.
            # it may be more practical to construct O computation carefully
            
            # Rather than over-engineering, simple manual assignment
            contrib = np.zeros(n)
            for i in range(n):
                i_k = i + offset
                i_nk = i - offset
                contrib[i] = 0
                # check bounds if needed...
                if 0 <= i_k < n:
                    contrib[i] += (1 - h[i_k])
                if 0 <= i_nk < n:
                    contrib[i] -= h[i_nk]
            grad += w[k] * contrib
    
    grad *= dx
    return O, grad

# Helper function to compute overlap integral (used when calling eval outside)
def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - np.array(h), mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx
