import numpy as np
import time

def project_to_box_simplex(v, target_sum):
    """Project vector v onto box [0,1] and simplex with sum = target_sum"""
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0, 1)

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, alpha):
    """Compute gradient of smoothed max of correlation values"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation values for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    O = corr * dx
    
    # Compute stable softmax weights
    O_max = np.max(O)
    exp_term = np.exp(alpha * (O - O_max))
    w = exp_term / np.sum(exp_term)
    
    # Compute gradient
    grad = np.zeros_like(h)
    
    # Loop over all shifts (corr indices)
    for idx in range(2 * n - 1):
        w_k = w[idx]
        if w_k == 0:
            continue
        
        shift = idx - (n - 1)
        
        # Compute shift_right and shift_left vectors
        if shift >= 0:
            # shift_right: ones for first shift entries, then 1 - h[shift:]
            sr = np.empty(n)
            sr[:shift] = 1.0
            sr[shift:] = 1.0 - h[shift:]
            # shift_left: zeros for first shift entries, then h[:n-shift]
            sl = np.empty(n)
            sl[:shift] = 0.0
            sl[shift:] = h[:n - shift]
        else:
            s = -shift
            # shift_right: ones for first s entries, then 1 - h[:n-s]
            sr = np.empty(n)
            sr[:s] = 1.0
            sr[s:] = 1.0 - h[:n - s]
            # shift_left: zeros for last s entries, then h[s:]
            sl = np.empty(n)
            sl[:n - s] = h[s:]
            sl[n - s:] = 0.0
        
        grad += w_k * (sr - sl)
    
    return grad * dx

def optimize_h(n, max_iters, alpha_start, alpha_end, init_lr):
    """Optimize h for given parameters"""
    # Initialize with random uniform values
    h = np.random.rand(n)
    # Project onto the feasible set: sum = n/2, bounds [0,1]
    h = project_to_box_simplex(h, n / 2)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for it in range(max_iters):
        # Compute smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        
        # Compute gradient
        grad = compute_gradient(h, alpha)
        
        # Adaptive learning rate
        lr = init_lr / np.sqrt(it + 1)
        
        # Update step
        h = h - lr * grad
        
        # Project onto feasible set
        h = project_to_box_simplex(h, n / 2)
        
        # Evaluate current solution
        cur_c5 = evaluate_c5(h)
        if cur_c5 < best_c5:
            best_h = h.copy()
            best_c5 = cur_c5
            
        # Early stopping if already good enough
        if best_c5 < 0.379:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use reasonable number of points
    n = 64
    
    # Initialize best solution
    best_h = None
    best_c5 = np.inf
    
    # Parameters for optimization
    max_iters_per_restart = 3000
    alpha_start = 5.0
    alpha_end = 30.0
    init_lr = 0.5
    
    restart_count = 0
    
    # Probability of choosing each initialization type
    init_types = ['random', 'sine', 'gaussian', 'step'] 
    
    while (time.time() - start_time) < budget_s * 0.95:
        # Choose initialization method
        init_type = np.random.choice(init_types)
        
        # Generate initial h based on type
        if init_type == 'random':
            h0 = np.random.rand(n)
        elif init_type == 'sine':
            x = np.linspace(0, 2, n)
            h0 = 0.5 + 0.4 * np.sin(np.pi * x)
            h0 = np.clip(h0, 0, 1)
        elif init_type == 'gaussian':
            x = np.linspace(0, 2, n)
            mu = 1.0
            sigma = 0.2
            h0 = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
            h0 = np.clip(h0, 0, 1)
        else:  # step
            h0 = np.zeros(n)
            mid = n // 2
            width = n // 4
            start = max(0, mid - width // 2)
            end = min(n, mid + width // 2)
            h0[start:end] = 1.0
        
        # Project initial h to sum constraint
        h0 = project_to_box_simplex(h0, n / 2)
        
        # Optimize with the chosen initial h
        h, c5 = optimize_h(n, max_iters_per_restart, alpha_start, alpha_end, init_lr)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        restart_count += 1
        
        # Check if time left is enough for another restart
        if (time.time() - start_time) > (budget_s * 0.9):
            break
            
    # Final refinement with subgradient descent if we have a good solution
    if best_h is not None:
        final_h = best_h.copy()
        final_c5 = best_c5
        # Hyperparameters for final refinement
        max_final_iters = 1000
        final_lr = 0.1
        
        for it in range(max_final_iters):
            # Compute correlation and find the index of the maximum O_k
            corr = np.correlate(final_h, 1 - final_h, mode="full")
            O = corr * (2.0 / n)
            
            max_idx = np.argmax(O)
            max_shift = max_idx - (n - 1)
            
            # Apply gradient for the max shift only (subgradient decent for max)
            # This is safe because the gradient approximates the subgradient of max
            if max_shift >= 0:
                # shift_right: ones for first shift entries, then 1 - h[shift:]
                sr = np.empty(n)
                sr[:max_shift] = 1.0
                sr[max_shift:] = 1.0 - final_h[max_shift:]
                # shift_left: zeros for first shift entries, then h[:n-shift]
                sl = np.empty(n)
                sl[:max_shift] = 0.0
                sl[max_shift:] = final_h[:n - max_shift]
            else:
                s = -max_shift
                # shift_right: ones for first s entries, then 1 - h[:n-s]
                sr = np.empty(n)
                sr[:s] = 1.0
                sr[s:] = 1.0 - final_h[:n - s]
                # shift_left: zeros for last s entries, then h[s:]
                sl = np.empty(n)
                sl[:n - s] = final_h[s:]
                sl[n - s:] = 0.0
            
            # Gradient step
            grad = (sr - sl) * (2.0 / n)  # dx factor is included
            final_h = final_h - final_lr * grad
            
            # Project to feasible set
            final_h = project_to_box_simplex(final_h, n / 2)
            
            # Evaluate
            cur_c5 = evaluate_c5(final_h)
            if cur_c5 < final_c5:
                final_c5 = cur_c5
                final_h = final_h.copy()  # Keep the best version
                
            # Early stopping
            if final_c5 < 0.379:
                break
                
        # Return refined solution if better
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
            
    return (best_h.tolist(), float(best_c5), n)
