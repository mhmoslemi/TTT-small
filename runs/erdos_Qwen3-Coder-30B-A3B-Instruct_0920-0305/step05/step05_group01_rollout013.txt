import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def evaluate_c5(h):
    """Evaluate C5 using correlation"""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation of h with (1-h)
    corr = correlate(h, 1-h, mode="full")
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project to sum = target_sum with box constraints [0,1]"""
    # Binary search for lambda
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # Return approximation if not converged
    y = np.clip(x + low, 0, 1)
    return y

def generate_initial_solution(n):
    """Generate good initial solution using PDE-inspired approach"""
    # Start with example construction
    h = np.array(initial_h_values)
    
    # Interpolate to target n
    if len(h) != n:
        x_orig = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n)
        h = np.interp(x_new, x_orig, h)
    
    # Project to correct sum
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def adaptive_local_search(h0, budget_s):
    """Run adaptive local search with multiple strategies"""
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    start_time = time.time()
    
    # Strategy 1: Gradient-based descent (L-BFGS)
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    max_iter = 0
    while (time.time() - start_time) < budget_s and max_iter < 5000:
        try:
            cons = {'type': 'eq', 'fun': constraint_eq}
            bounds = [(0, 1) for _ in range(len(h0))]
            res = minimize(
                objective,
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_new = project_to_sum_box(res.x, TARGET_SUM)
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
        except:
            pass
            
        # Strategy 2: Random local moves
        if (time.time() - start_time) < budget_s:
            h_trial = best_h.copy()
            # Random small perturbations
            idx = np.random.randint(0, len(h_trial))
            perturb = np.random.normal(0, 0.01)
            h_trial[idx] = np.clip(h_trial[idx] + perturb, 0, 1)
            h_trial = project_to_sum_box(h_trial, TARGET_SUM)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial
                
        max_iter += 1
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with good starting point
    n_points = 120
    h0 = generate_initial_solution(n_points)
    
    # Do gradient-based optimization with additional local search
    h_final, c5_final = adaptive_local_search(h0, budget_s)
    
    return (h_final.tolist(), float(c5_final), n_points)
