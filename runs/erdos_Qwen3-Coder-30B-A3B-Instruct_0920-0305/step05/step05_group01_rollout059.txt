import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def evaluate_c5(h):
    """Fast evaluation using FFT-based correlation"""
    n = len(h)
    dx = 2.0 / n
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum with [0,1] bounds"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def local_refine(h, max_iter=1000):
    """Use scipy.optimize for local refinement (SLSQP)"""
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h))]
    
    try:
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-15}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
    except:
        pass
    return h, evaluate_c5(h)
