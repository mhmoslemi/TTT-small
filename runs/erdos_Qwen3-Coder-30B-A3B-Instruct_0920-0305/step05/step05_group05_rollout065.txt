def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h using FFT for efficiency"""
    n = len(h)
    dx = 2.0 / n
    # Use FFT-based correlation for speed
    H = rfft(h)
    G = rfft(1 - h)
    corr = irfft(H * np.conj(G))
    # Extract the valid part of correlation (full mode)
    corr_full = np.zeros(2*n - 1)
    corr_full[:n] = corr[:n]
    corr_full[n:] = corr[n-1:0:-1]
    max_corr = np.max(corr_full)
    return max_corr * dx

def project_to_simplex(x, target_sum):
    """Project vector x onto the capped simplex: 0 <= x_i <= 1, sum x_i = target_sum"""
    # Clip to [0,1]
    x = np.clip(x, 0, 1)
    # Check the sum
    current_sum = np.sum(x)
    if np.abs(current_sum - target_sum) < 1e-12:
        return x
    # Binary search for the correct lambda such that sum = target_sum
    def sum_with_lambda(lam):
        return np.sum(np.clip(x - lam, 0, 1))
    
    # We already know x is feasible (clipped), so we bisect between xmin and xmax
    # Where xmin is such that all entries are 0 when clipped, and xmax that all 1
    # We solve for the root of g(lambda) = sum_with_lambda(lambda) - target_sum
    # We know g(0) > target_sum and g(max(x)) < 0, so (and g is non-increasing)
    # bisect between 0 and max(x)
    low, high = 0.0, np.max(x)
    max_iter = 30
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        s = sum_with_lambda(mid)
        if s > target_sum:
            low = mid
        else:
            high = mid
        if np.abs(s - target_sum) < 1e-12:
            break
    
    return np.clip(x - low, 0, 1)

def repair_h(c, n, T):
    """Repair function to obtain full h vector from control points"""
    K = len(c)
    # Ensure we have at least 2 points for valid interpolation
    if K < 2:
        X = np.linspace(0, 2, n)
        h = np.full(n, 0.5)
        return project_to_simplex(h, T)
    
    # Linearly interpolate to full grid
    x_knots = np.linspace(0, 2, K)
    h_interp = np.interp(np.linspace(0, 2, n), x_knots, c)
    return project_to_simplex(h_interp, T)

def solve_simplex_projection(v):
    """Solve projection onto the unit simplex using binary search for lambda"""
    eps = 1e-12
    # Simplified approach for vector v to find z with sum = 1, z_i = clamp(v_i - lambda, 0, 1)
    # f(lambda) = sum(clamp(v_i - lambda, 0, 1)) - 1
    def f_lambda(lam):
        return np.sum(np.clip(v - lam, 0, 1)) - 1.0
    
    # Find where f_lambda crosses 0 via binary search
    low, high = np.min(v), np.max(v) + 1.0
    for _ in range(40):  # reasonable number of bisections
        mid = (low + high) / 2
        val = f_lambda(mid)
        if val > eps:
            low = mid
        else:
            high = mid
        if np.abs(val) < eps:
            break
    # Final projected vector
    z = np.clip(v - low, 0, 1)
    return z

def objective_with_soft_max(c, n, T, alpha=5.0, k=None):
    """Compute soft-max of correlation values for a given control vector c"""
    h = repair_h(c, n, T)
    dx = 2.0 / n
    # Compute correlation using FFT
    H = rfft(h)
    G = rfft(1 - h)
    corr = irfft(H * np.conj(G))
    # Expand correlation to full size like 'full' mode
    corr_full = np.zeros(2*n - 1)
    corr_full[:n] = corr[:n]
    corr_full[n:] = corr[n-1:0:-1]
    corr_full = corr_full * dx  # Apply dx scaling
    
    # Compute softmax of the correlation values
    max_o = np.max(corr_full)
    if np.isinf(max_o) or np.isnan(max_o):
        # if the correlation itself is invalid, penalize heavily
        return 1e10
    # Avoid numeric issues
    centered = corr_full - max_o
    exps = np.exp(alpha * centered)
    # Sum and log
    log_sum_exp = max_o + np.log(np.sum(exps))
    return log_sum_exp / alpha

def compute_max_O(h):
    """Compute the true max overlapping integral manually"""
    n = len(h)
    dx = 2.0 / n
    H = rfft(h)
    G = rfft(1 - h)
    corr = irfft(H * np.conj(G))
    corr_full = np.zeros(2*n - 1)
    corr_full[:n] = corr[:n]
    corr_full[n:] = corr[n-1:0:-1]
    return np.max(corr_full) * dx
