import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Phase 1: Use global optimization with different algorithms
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Try different optimization methods
    try:
        # Use differential evolution first (global method)
        res = differential_evolution(
            objective, 
            bounds, 
            maxiter=1000, 
            popsize=30,
            seed=seed,
            atol=1e-14,
            rtol=1e-14
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Phase 2: Local refinement with multiple restarts
    local_restarts = 200
    for restart in range(local_restarts):
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
        # Random restart with more diverse starting points
        new_h = np.random.uniform(0, 1, n_points)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-15, 'gtol': 1e-15}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue

    # Phase 3: Try pattern-based construction to further improve
    if (time.time() - start_time) < budget_s * 0.8:
        try:
            # Pattern: try a sawtooth-like pattern
            # Create some structured patterns to guide the search
            x = np.linspace(0, 2, n_points)
            
            # Try a truncated exponential shape on both sides
            # This mimics a pattern that decenters the mass
            pattern = np.zeros(n_points)
            # Left side
            left_idx = int(n_points * 0.3)
            pattern[:left_idx] = np.exp(-x[:left_idx] * 5)
            # Right side
            right_idx = int(n_points * 0.7)
            pattern[right_idx:] = np.exp(-(2 - x[right_idx:]) * 5)
            
            # Normalize
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(pattern) * dx)
            pattern = pattern * scale
            
            # Mix with best solution
            mixed_h = 0.6 * best_h + 0.4 * pattern
            
            res = minimize(
                objective, 
                mixed_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-15}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass

    # Final optimization to ensure we're truly at a local minimum
    if (time.time() - start_time) < budget_s * 0.95:
        try:
            res = minimize(
                objective, 
                best_h, 
                method='L-BFGS-B', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-15}
            )
            if res.success:
                h_candidate = res.x
                dx = 2.0 / n_points
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass

    return (best_h.tolist(), best_c5, n_points)
