import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Phase 1: Global optimization with differential evolution for initial search
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    try:
        # Use differential evolution for global search
        res = differential_evolution(
            objective,
            bounds,
            maxiter=500,
            popsize=30,
            strategy='best1bin',
            seed=seed,
            tol=1e-12
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Phase 2: Local refinement with trust-constr using existing solution
    try:
        res = minimize(
            objective, 
            best_h, 
            method='trust-constr', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 2000, 'gtol': 1e-13, 'ftol': 1e-13}
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Phase 3: Adaptive local search with multiple restarts
    max_restarts = 2000
    
    for restart in range(max_restarts):
        if (time.time() - start_time) > budget_s * 0.95:  # Leave time for final steps
            break
            
        # Create new starting point using perturbations
        new_h = best_h.copy()
        # Randomly perturb some elements
        mask = np.random.random(n_points) < 0.4
        if np.any(mask):
            perturbation = np.random.normal(0, 0.02, np.sum(mask))
            new_h[mask] += perturbation
            new_h = np.clip(new_h, 0, 1)
            
        # Normalize
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            # Try with multiple local optimizers
            for method in ['SLSQP', 'L-BFGS-B']:
                if (time.time() - start_time) > budget_s * 0.95: break
                    
                res = minimize(
                    objective, 
                    new_h, 
                    method=method, 
                    bounds=bounds, 
                    constraints=cons,
                    options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-14}
                )
                if res.success:
                    h_candidate = res.x
                    scale = 1.0 / (np.sum(h_candidate) * dx)
                    h_candidate = h_candidate * scale
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
                        break  # Take the first successful improvement
        except:
            continue
    
    # Phase 4: Exploit potential symmetry or structure in the optimal solution
    if (time.time() - start_time) < budget_s * 0.9:
        # Try a simple pattern-based approach to see if it can improve
        try:
            # Try a two-peak pattern from the best solution
            peak_positions = [0.5, 1.5]
            new_h = np.zeros(n_points)
            x = np.linspace(0, 2, n_points)
            # Create peaks at the two positions
            for pos in peak_positions:
                # Gaussian peaks
                new_h += 0.5 * np.exp(-((x - pos)**2) / 0.1)
            
            # Normalize
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(new_h) * dx)
            new_h = new_h * scale_factor
            
            # Ensure no negative values
            new_h = np.clip(new_h, 0, 1)
            
            # Normalize again after clipping
            scale_factor = 1.0 / (np.sum(new_h) * dx)
            new_h = new_h * scale_factor
            
            # Optimize this starting point
            res = minimize(
                objective, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
