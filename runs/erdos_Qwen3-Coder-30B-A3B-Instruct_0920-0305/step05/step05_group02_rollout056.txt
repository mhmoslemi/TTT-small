import numpy as np
from cvxpy import *
import time
import math
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Decide sample counts to try
    n_list = [36, 40, 44, 48, 52, 56, 60, 64]
    
    # Helper function to generate a feasible binary vector using greedy heuristic
    def greedy_initialization(n_points):
        x = np.zeros(n_points, dtype=int)
        active_set = set()
        # First insert a greedy pick
        best_i = 0
        best_min_p = -1
        for i in range(n_points):
            # Simulate adding i
            sim_p = [0] * (n_points - 1)
            for j in active_set:
                k = abs(i-j)
                if k > 0:
                    sim_p[k-1] += 1
            new_min = min(sim_p) if sim_p else 0
            if new_min > best_min_p:
                best_min_p = new_min
                best_i = i
        x[best_i] = 1
        active_set.add(best_i)
        
        # Now iterate to fill up half of n_points
        while len(active_set) < n_points // 2:
            best_i = -1
            best_min_p = -1
            for i in range(n_points):
                if i in active_set:
                    continue
                # Try adding i and compute the resulting p_k list
                sim_p = [0] * (n_points - 1)
                for j in active_set:
                    k = abs(i - j)
                    if k > 0:
                        sim_p[k-1] += 1
                new_min = min(sim_p) if sim_p else 0
                if new_min > best_min_p:
                    best_min_p = new_min
                    best_i = i
            if best_i >= 0:
                active_set.add(best_i)
                x[best_i] = 1
            else:
                break
                
        return x

    # Evaluate the overlap integral for a given discrete h
    def evaluate_c5_discrete(h_int):
        n = len(h_int)
        dx = 2.0 / n
        corr = np.correlate(h_int, 1 - np.array(h_int), mode="full")
        max_corr = np.max(corr)
        return max_corr * dx

    best_c5 = float('inf')
    best_h = []
    best_n = None

    for n in n_list:
        if (time.time() - start_time) > budget_s - 20:
            break

        # Use already precomputed initial_h_values for a warm start if possible
        try:
            warm_start_h = np.array(initial_h_values)
            if len(warm_start_h) == n:
                warm_h = warm_start_h.round().astype(int)
                while np.sum(warm_h) != n // 2:
                    i = np.random.randint(n)
                    if warm_h[i] == 0 and np.sum(warm_h) < n // 2:
                        warm_h[i] = 1
                    elif warm_h[i] == 1 and np.sum(warm_h) > n // 2:
                        warm_h[i] = 0
                warm_start = warm_h
            else:
                warm_start = None
        except:
            warm_start = None
            
        # Use a greedy method as default warm start if not available
        if warm_start is None:
            warm_start = greedy_initialization(n)

        # Create the MILP model manually to maximize L
        # But since NO cvxpy is allowed, run a simpler version

        # Alternate: Try a hybrid approach with upper bound estimation using global search
        
        # Let's try to find a binary solution via simpler method in a fast way
        try:
            # Generate a random solution and check for its validity
            x_sol = warm_start.copy()
            valid_solutions = []
            for attempt in range(512):
                if (time.time() - start_time) > budget_s * 0.95:
                    break
                # Random slight fluctuation to explore
                h_trial = x_sol.copy().astype(float)
                # Possibly flip a few bits and normalize
                flips = np.random.randint(0, n, size=10)
                for i in flips:
                    h_trial[i] = 1 - h_trial[i]
                # Normalize
                sum_val = np.sum(h_trial)
                if abs(sum_val - n/2) < 1e-6:
                    valid_solutions.append(h_trial)
                elif sum_val < n/2:
                    # add more 1's
                    missing = int(n/2 - sum_val)
                    candidates = np.where(h_trial==0)[0]
                    for idx in np.random.choice(candidates, size=min(missing, len(candidates)), replace=False):
                        h_trial[idx] = 1
                elif sum_val > n/2:
                    # reduce 1's
                    excess = int(sum_val - n/2)
                    candidates = np.where(h_trial==1)[0]
                    for idx in np.random.choice(candidates, size=min(excess, len(candidates)), replace=False):
                        h_trial[idx] = 0
                if abs(np.sum(h_trial) - n/2) < 1e-6:
                    valid_solutions.append(h_trial)
                if len(valid_solutions) > 32:
                    break

            best_h_current = None
            best_c5_current = float('inf')
            for h_test in valid_solutions:
                c5 = evaluate_c5_discrete(h_test)
                if c5 < best_c5_current:
                    best_c5_current = c5
                    best_h_current = h_test.copy()
                    
            if best_c5_current < best_c5:
                best_c5 = best_c5_current
                best_h = best_h_current
                best_n = n

        except Exception as e:
            continue

    # If no better solution found, fall back to use the initial value (presumably non-binary but sampled)
    if len(best_h) == 0:
        try:
            h_original = np.array(initial_h_values)
            n_original = len(h_original)
            # Ensure integral constraint
            correct_sum = n_original / 2
            current_sum = np.sum(h_original)
            scale_factor = correct_sum / current_sum
            h_original = h_original * scale_factor
            best_c5 = evaluate_c5_discrete(np.round(h_original).clip(0, 1))
            best_h = h_original
            best_n = n_original
        except:
            pass
    else:
        try:
            # Re-normalize to ensure integral constraint
            dx = 2.0 / best_n
            current_sum = np.sum(best_h)
            scale_factor = 1.0 / (current_sum * dx)
            best_h = (best_h * scale_factor).clip(0,1)
            best_c5 = evaluate_c5_discrete(np.round(best_h).astype(int))
        except:
            pass
            
    # Final polishing using continuous solution if needed
    # This is an additional conservative refinement step
    try:
        def compute_objective(h_continuous):
            h_int = np.round(h_continuous).clip(0,1)
            return evaluate_c5_discrete(h_int)
            
        bounds = [(0, 1) for _ in range(best_n)]
        initial_guess = np.array(best_h)
        res = minimize(compute_objective, initial_guess, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
        h_final = res.x
        final_c5 = compute_objective(h_final)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_final
    except:
        pass

    return (best_h.tolist(), best_c5, best_n)
