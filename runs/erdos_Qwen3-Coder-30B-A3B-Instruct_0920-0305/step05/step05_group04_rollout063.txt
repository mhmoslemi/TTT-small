import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def project_simplex(h, target_sum):
    """Project h onto the simplex {x : sum(x) = target_sum, 0 <= x <= 1}"""
    # Clip to bounds [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust the sum
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    
    # Use bisection on λ to find the right adjustment
    # Find λ such that sum(clip(h + λ, 0, 1)) = target_sum
    def sum_clipped(λ):
        return np.sum(np.clip(h + λ, 0, 1))
    
    # We know the function is monotonic in λ
    a, b = -1.0, 1.0
    while sum_clipped(b) < target_sum:
        b *= 2.0
    while sum_clipped(a) > target_sum:
        a *= 2.0
    
    for _ in range(30):  # Bisection iterations
        mid = (a + b) / 2.0
        s = sum_clipped(mid)
        if abs(s - target_sum) < 1e-12:
            break
        elif s > target_sum:
            b = mid
        else:
            a = mid
    
    result = np.clip(h + (a + b) / 2.0, 0, 1)
    return result

def correlate_smooth(h):
    """Compute correlation vector with zero padding."""
    # Do linear convolution, len = 2*n - 1
    c = np.convolve(h, h, mode='full')
    # Keep only shifts from 1 to n-1 (excluding the 0-shift)
    return c[len(h)-1 : len(h)-1 + len(h)-1]

def compute_gradient(h, c, beta):
    """Compute gradient of log sum exp of negatives of correlations."""
    n = len(h)
    # compute w = softmax([ -beta * c_k ])
    c_max = np.max(c)
    w = np.exp(-beta * (c - c_max))
    w = w / np.sum(w)
    
    # gradient: - sum_k w_k * (h_{i-k} + h_{i+k})
    # For computational efficiency, use FFT/convolution trick.
    # Will compute: g_i = - sum_k w_k * (h_{i-k} + h_{i+k})
    # which equals - (conv(w, h)[i] + conv(w_rev, h)[i])
    
    # Note: w is of length n-1, we need to extend it to length 2n-1 for full convolution
    w_extended = np.zeros(2 * n - 1)
    w_extended[:n-1] = w
    h_extended = np.concatenate([h, np.zeros(n-1)])
    
    # The convolution with w reversed on the right
    # Since w_reversed = w[::-1] for indices 0 to n-2
    w_rev = w[::-1]
    w_rev_extended = np.zeros(2 * n - 1)
    w_rev_extended[n-1:] = w_rev
    
    # The actual computation uses a trick:
    # We can compute conv(h, w_rev_reverse) + conv(h, w) at once,
    # This is direct convolution with a symmetric kernel.
    
    # Actually compute: g = - (conv(h, w) + conv(h_reverse, w_rev))
    # But for symmetry reasons (reverse w = w reversal due to index alignment):
    # When doing linear convolution, we compute what is needed directly.
    
    # For simplicity: just use two standard one-way convolutions, but make index handling right
    # Let's refactored for correctness:
    # Gradient at i: - sum_{k=1}^{n-1} w_k * (h_{i-k} + h_{i+k})
    
    # Reconstruct by convolution:
    w_padded = np.zeros(2*len(h)-1)
    w_padded[:len(w)] = w
    
    # c shifted forward: 1 ... n-1; reversed: n-1 ... 1
    c_forward = np.convolve(h, w_padded, mode='valid')
    w_reversed = w[::-1]
    w_reversed_padded = np.zeros(2*len(h)-1)
    w_reversed_padded[-len(w):] = w_reversed
    c_backward = np.convolve(h, w_reversed_padded, mode='valid')
    
    # shift and extend for proper indexing
    g = - (c_forward + c_backward)
    return g

def compute_objective(h, beta, c):
    n = len(h)
    c_max = np.max(c)
    # Soft minimum: log sum exp of negative correlations
    exp_terms = np.exp(-beta * (c - c_max))
    log_sum = np.log(np.sum(exp_terms)) + c_max * beta
    phi_beta = log_sum / beta
    return 1.0 + phi_beta

def evaluate_c5_exact(h):
    """Evaluate the exact C5 norm with max of overlap integrals."""
    # This matches the sizing you describe: n_points = len(h)
    dx = 2.0 / len(h)
    
    # Correlation using numpy
    corr = correlate(h, 1 - h, mode="full")
    
    # Leave out shift zero in max
    max_val = np.max(corr[1:len(h)])  # avoid index 0 which is zero shift (not in max)
    return max_val * dx
