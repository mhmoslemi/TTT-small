import numpy as np

from scipy.signal import correlate
import time

def legendre_mask(p):
    """Create a Legendre-based binary mask of length p"""
    if p <= 1 or not is_prime(p):
        # Find the largest prime less than p
        for q in range(p-1, 1, -1):
            if is_prime(q):
                p = q
                break
        else:
            return None
    
    R = set()
    for a in range(1, p):
        r = (a * a) % p
        if r != 0:
            R.add(r)

    b = np.zeros(p, dtype=np.int8)
    for i in R:
        b[i] = 1
    
    # Adjust if we have too few or too many ones
    ones_in_b = np.sum(b)
    if ones_in_b < p // 2:
        # Need to add some ones
        zeros = np.where(b == 0)[0]
        to_add = (p // 2) - ones_in_b
        for i in range(to_add):
            b[zeros[i]] = 1
    elif ones_in_b > p // 2:
        # Need to remove some ones
        ones = np.where(b == 1)[0]
        to_remove = ones_in_b - (p // 2)
        for i in range(to_remove):
            b[ones[i]] = 0

    return b

def is_prime(n):
    """Check if n is prime"""
    if n < 2:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False
    for i in range(3, int(math.sqrt(n)) + 1, 2):
        if n % i == 0:
            return False
    return True

def create_random_binary(n):
    """Create a random balanced binary vector of length n with exactly n/2 ones"""
    h = np.zeros(n, dtype=np.int8)
    ones_indices = np.random.permutation(n)[:n//2]
    h[ones_indices] = 1
    return h

def compute_correlations(h):
    """Compute full autocorrelations for the binary pattern h"""
    n = len(h)
    corr = np.zeros(2 * n - 1, dtype=np.int64)
    for k in range(-(n-1), n):
        if k >= 0:
            corr[n-1+k] = np.sum(h[:-k] * h[k:])
        else:
            corr[n-1+k] = np.sum(h[-k:] * h[:k])
    return corr

def find_worst_shifts(corr, eps=1e-10):
    """Find indices where correlation attains the maximum value"""
    n = (len(corr) + 1) // 2
    Cmax = np.max(corr)
    # Account for small numerical errors
    worst_shifts = []
    for k in range(len(corr)):
        if abs(corr[k] - Cmax) < eps:
            worst_shifts.append(k - (n-1))
    return worst_shifts

def generate_swap(h, corr, worst_shifts, n, dx):
    """Generate a candidate swap from the worst shifts"""
    if not worst_shifts:
        return None, None, None

    # Pick a random worst shift
    k = np.random.choice(worst_shifts)
    
    # Find indices where h[i] = 1 and h[i+k] = 0
    BadOnes = []
    if k >= 0:
        for i in range(n - k):
            if h[i] == 1 and h[i+k] == 0:
                BadOnes.append(i)
    else:
        k = -k
        for i in range(n - k):
            if h[i+k] == 1 and h[i] == 0:
                BadOnes.append(i)
    
    if not BadOnes:
        return None, None, None

    # Pick a random element from BadOnes
    i = np.random.choice(BadOnes)
    
    # Find indices where h[j] = 0
    zeros = []
    for j in range(n):
        if h[j] == 0:
            zeros.append(j)
    
    # Try to find a valid j that is not in BadOnes(k)
    candidates = []
    if k >= 0:
        # Avoid positions that would make a new bad one
        good_positions = set(range(n)) - set(range(i, i + k + 1)) if k < n - 1 else set()
        for j in zeros:
            if j not in good_positions or j < i or j >= i + k:
                candidates.append(j)
    else:
        k = -k
        good_positions = set(range(n)) - set(range(i + k, min(n, i + 2*k))) if k < n - 1 else set()
        for j in zeros:
            if j not in good_positions or j >= i or j < i - k:
                candidates.append(j)
    
    # If no good candidates, just use a random zero
    if not candidates:
        j = np.random.choice(zeros)
    else:
        j = np.random.choice(candidates)
    
    return i, j, k

def update_correlation_incrementally(corr, h, i, j, k, n):
    """Incrementally update correlation array to reflect the swapped positions"""
    new_corr = corr.copy()
    
    # For i-th element's contribution
    # Scan all shifts m that involve i
    for m in range(-(n-1), n):
        # Exact updates for positions affected by i and j
        if m >= 0:
            # Check if h[i] contributes to correlation m
            if i + m < n:
                old_val = h[i] * h[i+m]
                new_val = h[j] * h[j+m]
                new_corr[n-1+m] += new_val - old_val
        
        if i >= m and m >= -(n-1):
            # Check if h[i-m] contributes to correlation m  
            if i-m >= 0:
                old_val = h[i-m] * h[i] 
                new_val = h[i-m] * h[j] 
                new_corr[n-1+m] += new_val - old_val

    # For j also
    for m in range(-(n-1), n):
        if m >= 0:
            # Check if h[j] contributes to correlation m
            if j + m < n:
                old_val = h[j] * h[j+m] 
                new_val = h[i] * h[i+m]
                new_corr[n-1+m] += new_val - old_val
        if j >= m and m >= -(n-1):
            # Check if h[j-m] contributes to correlation m  
            if j-m >= 0:
                old_val = h[j-m] * h[j] 
                new_val = h[j-m] * h[i] 
                new_corr[n-1+m] += new_val - old_val

    return new_corr

def project_to_feasible(h, n):
    """Project h to the constraint sum(h) = n/2"""
    target_sum = n // 2
    current_sum = np.sum(h)
    if current_sum == target_sum:
        return h
    
    # Adjust if needed
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        ones = np.where(h == 1)[0]
        n_flip = int(current_sum - target_sum)
        for i in np.random.choice(ones, n_flip, replace=False):
            h[i] = 0
    elif current_sum < target_sum:
        # Need to flip some 0s to 1s  
        zeros = np.where(h == 0)[0]
        n_flip = int(target_sum - current_sum)
        for i in np.random.choice(zeros, n_flip, replace=False):
            h[i] = 1
    
    return h

def evaluate_h_final(h, dx):
    """Evaluate final C5 value with exact integral"""
    n = len(h)
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx
