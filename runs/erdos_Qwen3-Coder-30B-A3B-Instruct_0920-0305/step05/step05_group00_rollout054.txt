def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Use binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_descent_step(h, learning_rate=1e-3):
    """Perform a gradient step to improve the solution"""
    # Compute gradient using finite differences
    h_plus = h + 1e-6
    h_minus = h - 1e-6

    h_minus = project_to_sum_box(h_minus, TARGET_SUM)

    c5_minus = evaluate_c5(h_minus)
    
    # Estimate gradient
    grad = (c5_plus - c5_minus) / (2e-6)
    
    # Update with gradient
    h_new = h - learning_rate * grad
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    
    return h_new
