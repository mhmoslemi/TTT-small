def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def random_pair_move(h):

    delta_max = min(hi, 1 - hj)
    if delta_min >= delta_max:
        return None
    delta = np.random.uniform(delta_min, delta_max)
    return (i, j, delta)

def get_worst_lag(h):
    """Get the lag that gives maximum correlation"""
    corr = compute_corr_fft(h)
    return np.argmax(corr)

def guided_move_with_direction(h):
    """Guided deterministic move based on gradient"""
    worst_lag = get_worst_lag(h)
    
    # Compute gradient for current worst lag
    g = np.zeros(n_points)
    for i in range(n_points):
        if i - worst_lag >= 0:
            g[i] += h[i - worst_lag]
        if i + worst_lag < n_points:
            g[i] += h[i + worst_lag]
    
    # Avoid moving elements that are at extreme values
    # Find index with min and max gradients that can be perturbed
    # Min gradient points (best to decrease)
    grad_min_idx = np.argmin(g)
    # Max gradient points (best to increase)  
    grad_max_idx = np.argmax(g)
    
    if grad_min_idx == grad_max_idx:
        return None

    # Transfer minimal amount that respects constraints
    delta = min(h[grad_min_idx], 1 - h[grad_max_idx])
    if delta <= 0:
        return None
    
    return (grad_min_idx, grad_max_idx, delta)
    
def highly_local_search(h, num_iterations=1000):
    """Perform highly local gradient-based search"""
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for _ in range(num_iterations):
        move = guided_move_with_direction(h)
        if move is None:
            break
            
        i, j, delta = move
        h[i] -= delta
        h[j] += delta
        
        # Project to constraint
        h = project_to_sum_box(h, TARGET_SUM)
        
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
