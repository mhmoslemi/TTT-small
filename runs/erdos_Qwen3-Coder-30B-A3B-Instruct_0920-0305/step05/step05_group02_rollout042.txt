import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def generate_legendre_pattern(n):
    """Generate a binary pattern based on quadratic residues"""
    p = n // 2
    if p < 2:
        return None
        
    # Find the largest prime <= p
    def is_prime(num):
        if num < 2:
            return False
        for i in range(2, int(np.sqrt(num)) + 1):
            if num % i == 0:
                return False
        return True
    
    while not is_prime(p):
        p -= 1
        
    if p < 2:
        return None
    
    # Generate quadratic residues mod p
    residues = set()
    for a in range(1, p):
        residues.add((a * a) % p)
    
    # Create binary mask
    b = np.zeros(p, dtype=int)
    for r in residues:
        if r < p:
            b[r] = 1
            
    # Duplicate pattern to length n
    h = np.tile(b, 2)
    
    # Adjust sum to exact value
    current_sum = np.sum(h)
    target_sum = n // 2
    if current_sum > target_sum:
        # Find a 1 and flip to 0
        idx = np.where(h == 1)[0]
        if len(idx) > 0:
            h[idx[0]] = 0
    elif current_sum < target_sum:
        # Find a 0 and flip to 1
        idx = np.where(h == 0)[0]
        if len(idx) > 0:
            h[idx[0]] = 1
    
    return h

def generate_random_balanced(n):
    """Generate random balanced binary pattern"""
    h = np.zeros(n, dtype=int)
    ones = np.random.permutation(n)[:n//2]
    h[ones] = 1
    return h

def compute_correlation(h, dx=1.0):
    """Compute full correlation (only for binary vectors)"""
    corr = correlate(h, h, mode='full')
    # Only keep the needed lags [0, n-1] and [-n+1, -1]
    # We just compute overlap as sum of h[i] * h[i+k] for valid indices
    n = len(h)
    full_corr = np.zeros(2 * n - 1)
    for i in range(n):
        for j in range(n):
            if i + j <= 2 * n - 2:
                full_corr[i + j] += h[i] * h[j]
    
    return full_corr

def compute_c5(h, dx=1.0):
    """Compute the C5 constant via correlation (binary)"""
    n = len(h)
    full_corr = compute_correlation(h, dx)
    # We only look at positive lags (0 to n-1) since negative lags are symmetric
    # But corruptions are what we're computing - overlap values
    corr = full_corr[n-1:]  # Only positive lags
    # For overlap: h[i] * (1 - h[i+k])
    # i.e., correlation with (1 - h)
    
    # Compute the overlap integral directly
    max_val = 0.0
    for k in range(n):  # k from 0 to n-1
        overlap = 0.0
        for i in range(n - k):
            overlap += h[i] * (1 - h[i+k])
        max_val = max(max_val, overlap)
    
    return max_val * dx

def compute_overlaps(h, dx=1.0):
    """Return all overlap values for each lag"""
    n = len(h)
    overlaps = np.zeros(n)
    for k in range(n):
        overlap = 0.0
        for i in range(n - k):
            overlap += h[i] * (1 - h[i+k])
        overlaps[k] = overlap
    return overlaps

def binary_to_continuous(h):
    """Convert binary to continuous representation"""
    return h.astype(float)

def continuous_to_binary(x, threshold=0.5):
    """Convert continuous to binary representation"""
    return (x >= threshold).astype(int)

def incrementally_update_correlation(corr, h, i, j, n):
    """Update correlation after swapping h[i] and h[j]"""
    delta_corr = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        # contribution from i
        if i < n and i + m < 2 * n - 1:
            delta_corr[m] += -h[i] * h[i + m] + h[j] * h[i + m]
        if i >= m and i < n:
            delta_corr[m] += -h[i - m] * h[i] + h[i - m] * h[j]
        
        # contribution from j
        if j < n and j + m < 2 * n - 1:
            delta_corr[m] += -h[j] * h[j + m] + h[i] * h[j + m]
        if j >= m and j < n:
            delta_corr[m] += -h[j - m] * h[j] + h[j - m] * h[i]
    
    return corr + delta_corr

def find_worst_shifts(corr, eps=1e-10):
    """Find indices k of worst shifts (where corr[k] is maximum)"""
    # corr has size (2*n-1), we want indices from 0 to n-1
    # Due to symmetry, just take the first n indices in the full correlation
    max_val = np.max(corr)
    indices = np.where(np.abs(corr - max_val) < eps)[0]
    return indices

def get_bad_ones(h, k):
    """Get indices where h[i]=1 and h[i+k]=0"""
    n = len(h)
    bad_ones = []
    for i in range(n - k):
        if h[i] == 1 and h[i + k] == 0:
            bad_ones.append(i)
    return bad_ones

def get_harmless_zeros(h, k):
    """Get indices where h[i]=0 and i+k is in valid range"""
    n = len(h)
    harm_ones = []
    for i in range(n):
        if h[i] == 0:
            harm_ones.append(i)
    return harm_ones

            h = h_start.copy()

            overlaps = compute_overlaps(h, dx=dx)

            T0 = 0.05

                    break

                max_corr = np.max(corr)

                            harmless_zeros.append(j)
                        else:
                            check_condition = True
                            if i+k >= 0 and i+k < n:
                                if h[i+k] == 0:
                                    check_condition = False
                            if check_condition:
                                harmless_zeros.append(j)
                
                # Find those that are not making k-affected terms zero
                # Just pick any zero
                if len(harmless_zeros) == 0:
                    # Take any zero
                    harmless_zeros = []
                    for j in range(n):
                        if h[j] == 0:
                            harmless_zeros.append(j)

                # Pick a random zero
                j = np.random.choice(harmless_zeros)
                
                # Perform the swap
                h[i], h[j] = h[j], h[i]
                
                # Update correlation efficiently
                old_corr = corr.copy()

                for l in range(n):
                    for m in range(n):

                overlaps = compute_overlaps(h, dx=dx)

                t = T0 * (Tf / T0) ** (iter_no / N_iter)

    # Do final continuous polish on the best binary solution
    if best_h is not None:

        # Method to compute objective (smooth approximation using logsumexp)
        def smooth_c5(x):
            # Use linear relaxation
            n = len(x)
            dx = 2.0 / n
            # Compute correlations
            corr = np.zeros(n)

                corr[k] = 0.0

        # Smooth approximation
        def smooth_max_overlap(x, alpha=30.0):
            # Compute overlaps
            n = len(x)
            dx = 2.0 / n
            overlaps = []
            for k in range(n):
                overlap = 0.0

        from scipy.optimize import minimize

        # Constraint: Σ x = n/2

                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = x_final
        except:
            # If optimization fails, return current best
            pass

    return (best_h.tolist(), best_c5, best_n)
