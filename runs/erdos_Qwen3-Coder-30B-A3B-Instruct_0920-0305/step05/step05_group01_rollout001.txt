import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60
DX = 2.0 / 120.0

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    low, high = -2.0, 2.0
    for _ in range(50):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target_sum) < 1e-12:
            return y
        if s > target_sum:
            high = mid
        else:
            low = mid
    y = np.clip(x + low, 0.0, 1.0)
    return y

def evaluate_c5(h):
    """Evaluate C5 using correlation"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * DX

def gradient_smoothing(h, alpha):
    """Compute smoothed gradient using log-sum-exp approximation"""
    n = len(h)
    # correlation of h with (1-h)
    cov_full = np.correlate(h, 1.0 - h, mode='full')
    I = cov_full  # shape 2n-1
    I_max = I.max()
    # compute weights for softmax
    exp_vals = np.exp(alpha * (I - I_max))
    w = exp_vals / exp_vals.sum()
    # compute gradient as weighted sum of per-lag grads
    grad = np.zeros(n)
    # Pre-compute arrays
    one_minus_h = 1.0 - h
    for idx, p_val in enumerate(w):
        lag = idx - (n - 1)
        if lag >= 0:
            # contribution from (1 - h[i+lag]) for i=0..n-1-lag
            if n - lag > 0:
                grad[:n-lag] += p_val * one_minus_h[lag:]
            # edge contribution from h[i] since (1 - h[i+lag]) = 1 for out of range
            if lag > 0:
                grad[n-lag:] += p_val * 1.0
            # contribution from - h[i-lag] for i=lag..n-1
            if lag > 0:
                grad[lag:] -= p_val * h[:n-lag]
            else:
                # lag == 0
                grad -= p_val * h
        else:
            d = -lag
            # contribution from (1 - h[i-d]) for i=d..n-1
            if n - d > 0:
                grad[d:] += p_val * one_minus_h[:n-d]
            # edge contribution from h[i] since (1 - h[i+lag]) = 1 for out of range
            grad[:d] += p_val * 1.0
            # contribution from - h[i+d] for i=0..n-d-1
            grad[:n-d] -= p_val * h[d:]
    return grad * DX

def descent_step(h, eta, alpha):
    """Perform one descent step with line search"""
    grad = gradient_smoothing(h, alpha)
    # Try different step sizes
    for _ in range(5):
        h_new = project_to_sum_box(h - eta * grad, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        if c5_new <= evaluate_c5(h):
            return h_new, c5_new
        eta *= 0.5
    return h, evaluate_c5(h)

def gradient_descent(initial_h, time_limit):
    """Run gradient descent with smoothing"""
    h = initial_h.copy()
    c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5
    t0 = time.time()
    
    alpha_start = 0.5
    alpha_end = 30.0
    eta_start = 0.1
    
    it = 0
    max_iters = 5000
    while time.time() - t0 < time_limit and it < max_iters:
        progress = it / max_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        eta = eta_start * (0.995 ** it)
        
        h, c5 = descent_step(h, eta, alpha)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        it += 1
    return best_h, best_c5

def local_refine(h):
    """Refine solution using SLSQP optimizer"""
    def objective(h):
        return evaluate_c5(h)
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h))]
    
    try:
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
    except:
        pass
    return h, evaluate_c5(h)

    target_sum = n_points / 2.0
    
    # Interpolate to n_points

    x_new = np.linspace(0, 2, n_points)
    h_interp = np.interp(x_new, x_orig, h0)
    h0 = project_to_sum_box(h_interp, target_sum)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Run multiple restarts
    restarts = 0
    while (time.time() - start_time) < budget_s * 0.9:
        # Create new random start
        h_start = np.random.uniform(0, 1, n_points)
        h_start = project_to_sum_box(h_start, target_sum)
        
        # Add small random perturbation to previous best
        if restarts > 0:
            perturb = np.random.normal(0, 0.01, n_points)
            h_start = project_to_sum_box(h0 + perturb, target_sum)
            
        # Run gradient descent
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 10:
            break
            
        descent_time = min(remaining_time * 0.8, 120)
        h_trial, c5_trial = gradient_descent(h_start, descent_time)
        
        # Local refinement
        h_refined, c5_refined = local_refine(h_trial)
        
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
            
        restarts += 1
    
    return (best_h.tolist(), float(best_c5), n_points)
