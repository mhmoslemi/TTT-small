import time
import numpy as np

def project_to_box_simplex(v, target_sum):
    """Project vector v onto box [0,1] and simplex with sum = target_sum"""
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0, 1)

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, alpha):
    """Compute gradient of smoothed max overlap."""
    n = len(h)
    dx = 2.0 / n
    # correlation
    corr = np.correlate(h, 1 - h, mode="full")
    O = corr * dx
    O_max = np.max(O)
    # compute stable weights
    exp_term = np.exp(alpha * (O - O_max))
    w = exp_term / np.sum(exp_term)  # size 2n-1

    # compute gradient
    grad = np.zeros_like(h)
    for idx, shift in enumerate(range(-n+1, n)):
        w_k = w[idx]
        if w_k == 0:
            continue
        # compute shift_right and shift_left
        if shift >= 0:
            shift_right = np.empty(n)
            shift_right[:shift] = 1.0
            shift_right[shift:] = 1.0 - h[shift:]
            shift_left = np.empty(n)
            shift_left[:shift] = 0.0
            shift_left[shift:] = h[:n-shift]
        else:
            s = -shift
            shift_right = np.empty(n)
            shift_right[:s] = 1.0
            shift_right[s:] = 1.0 - h[:n-s]
            shift_left = np.empty(n)
            shift_left[:n-s] = h[s:]
            shift_left[n-s:] = 0.0
        grad += w_k * (shift_right - shift_left)
    return grad * dx

def optimize_h_gradient(n, max_iters=2000, alpha_start=5, alpha_end=30, init_lr=0.5):
    """Optimize h using gradient descent with adaptive alpha and learning rate."""
    # random uniform start
    h = np.random.rand(n)
    # project to correct sum
    target_sum = n / 2.0
    h = project_to_box_simplex(h, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for it in range(max_iters):
        # current objective: smoothed max
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        
        # gradient update
        grad = compute_gradient(h, alpha)
        lr = init_lr / np.sqrt(it + 1)
        h = h - lr * grad
        # project onto feasible region
        h = project_to_box_simplex(h, target_sum)
        
        # evaluate and update best
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # early stopping
        if best_c5 < 0.379:
            break
            
    return best_h, best_c5
