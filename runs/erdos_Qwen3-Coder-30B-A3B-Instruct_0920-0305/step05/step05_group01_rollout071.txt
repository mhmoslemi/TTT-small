import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60
EPS = 1e-12

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    low, high = -2.0, 2.0
    for _ in range(50):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target_sum) < EPS:
            return y
        if s > target_sum:
            high = mid
        else:
            low = mid
    y = np.clip(x + low, 0.0, 1.0)
    return y

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)
    return corr.max() * dx

def gradient_smoothing(h, alpha):
    """Compute gradient of log-sum-exp smoothing of max correlation"""
    n = len(h)
    # correlation of h with (1-h)
    cov_full = correlate(h, 1.0 - h, mode='full')
    I = cov_full  # shape 2n-1
    I_max = I.max()
    # compute weights for softmax
    exp_vals = np.exp(alpha * (I - I_max))
    w = exp_vals / exp_vals.sum()
    # compute gradient as weighted sum of per-lag grads
    grad = np.zeros(n)
    
    # Pre-compute arrays
    one_minus_h = 1.0 - h
    
    for idx, p_val in enumerate(w):
        lag = idx - (n - 1)  # ℓ positive means shift left (i+ℓ)
        if lag >= 0:
            # term (1 - h[i+lag]) for i=0..n-1-lag
            if n - lag > 0:
                grad[:n-lag] += p_val * one_minus_h[lag:]
            # edge: i where i+lag >= n => i in [n-lag, n-1]
            if lag > 0:
                grad[n-lag:] += p_val * 1.0
            # term - h[i-lag] for i=lag..n-1
            if lag > 0:
                grad[lag:] -= p_val * h[:n-lag]
            else:
                # lag==0: subtract entire h
                grad -= p_val * h
        else:
            # lag negative: let d = -lag
            d = -lag
            # term (1 - h[i-d]) for i=d..n-1
            if n - d > 0:
                grad[d:] += p_val * one_minus_h[:n-d]
            # edge: i where i+lag < 0 (i < d)
            grad[:d] += p_val * 1.0
            # term - h[i+d] for i=0..n-d-1
            grad[:n-d] -= p_val * h[d:]
    
    return grad

def descent_step(h, eta, alpha, target_sum):
    """Perform one descent step"""
    grad = gradient_smoothing(h, alpha)
    h_new = project_to_sum_box(h - eta * grad, target_sum)
    c5_new = evaluate_c5(h_new)
    return h_new, c5_new

def gradient_descent_run(h_start, time_limit, target_sum):
    """Run gradient descent with log-sum-exp smoothing"""
    h = h_start.copy()
    c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    t0 = time.time()
    it = 0
    alpha_start = 0.5
    alpha_end = 50.0
    eta_start = 0.2
    
    while time.time() - t0 < time_limit:
        progress = min(it / 2000, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        eta = eta_start * (0.995 ** it)
        
        h_new, c5_new = descent_step(h, eta, alpha, target_sum)
        
        if c5_new <= c5:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        it += 1
    
    return best_h, best_c5

def local_refine(h, target_sum):
    """Refine solution using SLSQP optimizer"""
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h))]
    
    try:
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, target_sum)
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
    except:
        pass
    return h, evaluate_c5(h)
