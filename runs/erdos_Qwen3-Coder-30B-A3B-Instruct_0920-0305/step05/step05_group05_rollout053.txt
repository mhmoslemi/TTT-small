import numpy as np
import time

def project_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex: 0 <= x <= 1, sum(x) = target_sum"""
    v = np.asarray(v)
    high = np.ones_like(v)
    low = np.zeros_like(v)
    
    # Binary search for the Lagrange multiplier lambda
    # We want sum(max(0, min(1, v - lambda))) = target_sum
    
    def sum_proj(lam):
        return np.sum(np.clip(v - lam, low, high))
    
    # Bisection search for lambda
    l, u = np.min(v - 1), np.max(v)
    for _ in range(50):  # Safe iterations
        mid = (l + u) / 2
        s = sum_proj(mid)
        if s > target_sum:
            l = mid
        else:
            u = mid

    lambda_star = (l + u) / 2
    result = np.clip(v - lambda_star, 0, 1)
    return result

def compute_O_and_grad(h, alpha):
    """Compute overlap vector O and its gradient"""
    n = len(h)
    dx = 2.0 / n
    
    # Compute all overlaps O_k = dx * sum(h[i] * (1 - h[i+k]))
    O = np.zeros(2 * n - 1)
    for k in range(-n + 1, n):
        # Indices of h[i]
        i_left = max(0, -k)
        i_right = min(n, n - k)
        if i_left < i_right:
            O[k + n - 1] = dx * np.sum(h[i_left:i_right] * (1.0 - h[i_left + k:i_right + k]))
    
    # Softmax weights
    max_O = np.max(O)
    exp_O = np.exp(alpha * (O - max_O))
    w = exp_O / np.sum(exp_O)
    
    # Gradient: sum_k w_k * ( d/dh_i O_k )
    grad = np.zeros_like(h)
    for k in range(-n + 1, n):
        if k < 0:
            # h[i] * (1 - h[i-k])
            # d/dh[i] O_k = (1 - h[i-k])
            # d/dh[i] O_k = - h[i+k]
            slice1 = slice(max(0, -k), min(n, n - k))  # i where contribution happens from h[i]
            slice2 = slice(max(0, k), min(n, n + k))   # i where 1-h[i+k] applies
            if slice1.start < slice1.stop and slice2.start < slice2.stop:
                grad[slice1.start:slice1.stop] += w[k + n - 1] * (1.0 - h[slice2.start:slice2.stop])
                # Reverse the effect for the term with - h[i+k]
                # When k < 0, i+k moves forward, so d/dh[i] h[i+k] = 1 pulls from higher indices
                if k + 1 <= 0:
                    grad[slice1.start:slice1.stop] -= w[k + n - 1] * h[slice1.start + k:slice1.stop + k]
        elif k > 0:
            # j = i + k, so d/dh[i] h[i+k] = 1, 
            # and d/dh[i] h[i] = -1
            slice1 = slice(max(0, k), min(n, n))  # i
            slice2 = slice(0, min(n, n - k))      # i+k
            if slice1.start < slice1.stop:
                # Contribution of d/dh[i] (h[i] * (1 - h[i+k])) = - h[i+k] + (1 - h[i+k])
                # The derivative is only on the h[i+k] term, and requires placing -1 weights
                grad[slice1.start:slice1.stop] += w[k + n - 1] * (-h[slice2.start:slice2.stop])  # - h[i+k]
                # and then we count the term h[i] (1 - h[i+k]) terms where d/dh[i] = 1 (at index i)
                # index i is computed from in the d/dh[i] line above
                # Use the & part that had computing the term d/dh[i] (1 - h[i+k]) which is the negative
                grad[slice1.start:slice1.stop] += w[k + n - 1] * (1.0 - h[slice2.start:slice2.stop])  # + (1-h[i+k])
        else:
            # k = 0
            # O_k = dx * sum(h[i] * (1 - h[i]))
            slice1 = slice(0, n)
            grad[slice1.start:slice1.stop] += w[k + n - 1] * (1.0 - 2.0 * h[slice1.start:slice1.stop])

    return O, grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 64
    dx = 2.0 / n
    T = n / 2
    best_h = None
    best_C5 = float('inf')
    
    max_iter = 4000  # iterations per restart setup
    subgrad_iters = 500
    alpha_start = 5
    alpha_end = 30
    lr_0 = 0.5
    
    # Predefined shape seeds
    seeds = [
        "uniform",  # uniform initial guess in [0,1]
        "triangular",  # triangular shape
        "sine",  # sine-based shape
        "gaussian", # Gaussian bump
        "binary_1",  # binary pattern
        "binary_2",  # another binary pattern
        "mixed_1",  # mixture of settings
    ]
    
    def generate_seed(key):
        if key == "uniform":
            return np.random.uniform(0, 1, n)
        elif key == "triangular":
            x = np.linspace(0, 2, n)
            h = 1.0 - np.abs(x - 1.0)
            return np.clip(h, 0, 1)
        elif key == "sine":
            x = np.linspace(0, 2, n)
            h = 0.5 + 0.4 * np.sin(np.pi * x)
            return np.clip(h, 0, 1)
        elif key == "gaussian":
            x = np.linspace(0, 2, n)
            mu = 1.0
            sigma = 0.2
            h = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
            return np.clip(h, 0, 1)
        elif key == "binary_1":
            h = np.zeros(n)
            half = n // 2
            h[:half] = 1.0
            return h
        elif key == "binary_2":
            h = np.zeros(n)
            h[n//4:3*n//4] = 1.0
            return h
        elif key == "mixed_1":
            x = np.linspace(0, 2, n)
            h = 0.5 + 0.3 * np.sin(np.pi * x) + 0.2 * np.cos(2 * np.pi * x)
            return np.clip(h, 0, 1)
        else:
            return np.random.uniform(0, 1, n)
    
    restarts = 0
    last_update = time.time()
    while (time.time() - start_time) < (budget_s * 0.95):
        restarts += 1
        if restarts > 1000 and (time.time() - last_update) > 10:
            break  # early stop mechanism
            
        key = np.random.choice(seeds)
        h = generate_seed(key)
        
        # Project onto the feasible set
        h = project_capped_simplex(h, T)
        h_prev = h.copy()
        
        # Smoothed gradient descent
        for t in range(max_iter):
            if (time.time() - start_time) > budget_s * 0.95:
                break
                
            alpha = alpha_start + (alpha_end - alpha_start) * t / max_iter
            O, grad = compute_O_and_grad(h, alpha)
            
            # Update step
            step = lr_0 / np.sqrt(t + 1)
            h = h - step * grad
            
            # Project back
            h = project_capped_simplex(h, T)
                
            # Optional symmetry enforcing
            if t % 10 == 0:
                h = (h + h[::-1]) / 2
                
            # Compute and store best
            c5_current = np.max(O) * dx
            if c5_current < best_C5:
                best_C5 = c5_current
                best_h = h.copy()
                
        # Sub-gradient polishing
        if not best_h is None:
            # Recompute O
            O_prev = np.zeros(2 * n - 1)
            for k in range(-n + 1, n):
                i_left = max(0, -k)
                i_right = min(n, n - k)
                if i_left < i_right:
                    O_prev[k + n - 1] = dx * np.sum(best_h[i_left:i_right] * (1.0 - best_h[i_left + k:i_right + k]))
            max_k = np.argmax(O_prev)
            max_val = np.max(O_prev)
            
            # Compute exact gradient for max_k each iteration
            h = best_h.copy()
            for t in range(subgrad_iters):
                if (time.time() - start_time) > budget_s * 0.95:
                    break
                O, grad = compute_O_and_grad(h, float('inf'))
                k_max = np.argmax(O)
                # In actual practice, we compute exact gradient w.r.t. max_k
                # Approximate it by making the contribution of that shift dominant
                # or focus on OD (working on indexes i, and simulate shift k)
                # let's just go back to reasoning and update with only max shift weight
                
                grad_full = np.zeros_like(h)
                # Only active shift contributes to gradient
                k = k_max - n + 1
                i_left = max(0, -k)
                i_right = min(n, n - k)
                if i_left < i_right:
                    # d/dh[i] = (1 - h[i+k]) - h[i-k]
                    # i = [i_left, i_right)
                    grad_full[i_left:i_right] += (k >= 0) * (1.0 - best_h[i_left + k:i_right + k])
                    grad_full[i_left:i_right] -= (k <= 0) * best_h[i_left - k:i_right - k]

                lr = 0.02 / np.sqrt(t + 1)
                h = h - lr * grad_full
                
                # Project to simplex
                h = project_capped_simplex(h, T)
                
                # Compute and store best
                O = np.zeros(2 * n - 1)
                for k in range(-n + 1, n):
                    i_left = max(0, -k)
                    i_right = min(n, n - k)
                    if i_left < i_right:
                        O[k + n - 1] = dx * np.sum(h[i_left:i_right] * (1.0 - h[i_left + k:i_right + k]))
                c5_current = np.max(O) * dx
                if c5_current < best_C5:
                    best_C5 = c5_current
                    best_h = h.copy()
                    
        last_update = time.time()
    
    return (best_h.tolist(), float(best_C5), n)
