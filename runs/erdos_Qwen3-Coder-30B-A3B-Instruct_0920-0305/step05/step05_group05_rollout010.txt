import numpy as np
import time

def project_to_box_simplex(v, target_sum):
    """Project vector v onto box [0,1] and simplex with sum = target_sum"""
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(50):
        lam = (lo+hi)/2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, 0, 1)

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, alpha):
    """Compute gradient of smoothed maximum overlap"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations with all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    O = corr * dx
    O_max = np.max(O)
    
    # Stable computation of exponential weights
    exp_term = np.exp(alpha * (O - O_max))
    w = exp_term / np.sum(exp_term)  # size 2n-1
    
    # Compute gradient
    grad = np.zeros_like(h)
    
    # Loop over all shifts
    for idx, shift in enumerate(range(-n+1, n)):
        w_k = w[idx]
        if w_k == 0:
            continue
            
        # Compute shift_right and shift_left vectors
        if shift >= 0:
            # shift_right[i] = 1 for i < shift else 1 - h[i-shift]
            # shift_left[i] = 0 for i < shift else h[i-shift]
            shift_right = np.empty(n)
            shift_right[:shift] = 1.0
            shift_right[shift:] = 1.0 - h[shift:]
            
            shift_left = np.empty(n)
            shift_left[:shift] = 0.0
            shift_left[shift:] = h[:n-shift]
        else:
            s = -shift
            # shift_right[i] = 1 for i < s else 1 - h[i-s]
            # shift_left[i] = 0 for i < s else h[i+s]
            shift_right = np.empty(n)
            shift_right[:s] = 1.0
            shift_right[s:] = 1.0 - h[:n-s]
            
            shift_left = np.empty(n)
            shift_left[:n-s] = h[s:]
            shift_left[n-s:] = 0.0
            
        grad += w_k * (shift_right - shift_left)
    
    grad = grad * dx  # Incorporate dx factor from O = sum * dx
    return grad

def optimize_h(n, max_iters=4000, alpha_start=5, alpha_end=30, lr0=0.5):
    """Optimize h with gradient descent and smoothing"""
    # Initialize with random vector
    h = np.random.rand(n)
    h = project_to_box_simplex(h, target_sum=n/2)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Gradient descent loop
    for it in range(max_iters):
        # Smoothed objective parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        
        # Compute gradient
        grad = compute_gradient(h, alpha)
        
        # Adaptive learning rate
        lr = lr0 / np.sqrt(it + 1)
        
        # Update step
        h = h - lr * grad
        
        # Project back onto feasible set
        h = project_to_box_simplex(h, target_sum=n/2)
        
        # Evaluate current solution
        cur_c5 = evaluate_c5(h)
        if cur_c5 < best_c5:
            best_h = h.copy()
            best_c5 = cur_c5
            
        if best_c5 < 0.379:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Using 64 points for better resolution
    n_points = 64
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    best_h = None
    best_c5 = np.inf
    
    attempts = 0
    max_attempts = 50  # Limit restarts to fit within time budget
    
    while attempts < max_attempts and (time.time() - start_time) < budget_s * 0.95:
        # Run optimization with random initialization
        h, c5 = optimize_h(n_points, max_iters=3000)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        attempts += 1
        
        # Early stopping if we found a good solution
        if best_c5 < 0.379:
            break
            
        # Quick update on final attempts
        if attempts > max_attempts - 5 and (time.time() - start_time) > budget_s * 0.9:
            break
    
    # Final optimization on the best found solution (local refinement)
    if best_h is not None:
        h_refined, c5_refined = optimize_h(n_points, max_iters=1000, alpha_start=30, alpha_end=50, lr0=0.1)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    
    # Ensure we return the final solution
    if best_h is None:
        # Fallback to linear initialization
        h = np.ones(n_points) * 0.5
        h = project_to_box_simplex(h, target_sum=target_sum)
        best_h = h
        best_c5 = evaluate_c5(h)
    
    return (best_h.tolist(), float(best_c5), n_points)
