import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    low, high = -2.0, 2.0
    for _ in range(50):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target_sum) < 1e-12:
            return y
        if s > target_sum:
            high = mid
        else:
            low = mid
    y = np.clip(x + low, 0.0, 1.0)
    return y

def evaluate_c5(h, dx):
    """Evaluate C5 using correlation with scaling"""
    corr = correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def gradient_smoothing(h, alpha, dx):
    """Compute gradient of log-sum-exp smoothing of correlations"""
    n = len(h)
    # Compute full correlation (unscaled)
    cov_full = correlate(h, 1.0 - h, mode='full')
    
    # Scale for numerical stability
    I = cov_full * dx
    I_max = I.max()
    
    # Compute softmax weights
    exp_vals = np.exp(alpha * (I - I_max))
    w = exp_vals / exp_vals.sum()
    
    # Compute gradient as weighted sum of per-lag gradients
    grad = np.zeros(n)
    
    for idx, p_val in enumerate(w):
        lag = idx - (n - 1)  # lag from -n+1 to n-1
        
        if lag >= 0:
            # (1 - h[i+lag]) term for indices where i+lag < n
            if n - lag > 0:
                grad[:n-lag] += p_val * (1.0 - h[lag:])
            
            # Edge contributions where i+lag >= n
            if lag > 0:
                grad[n-lag:] += p_val * 1.0
                
            # Subtract h[i-lag] term for indices where i-lag >= 0
            if lag > 0:
                grad[lag:] -= p_val * h[:n-lag]
            else:
                # lag == 0 case: subtract h[i] for all positions
                grad -= p_val * h
                
        else:
            # lag < 0
            d = -lag
            # (1 - h[i-d]) term for indices where i >= d
            if n - d > 0:
                grad[d:] += p_val * (1.0 - h[:n-d])
                
            # Edge contributions where i+lag < 0 (i < d)
            grad[:d] += p_val * 1.0
            
            # Subtract h[i+d] term for indices where i+d < n
            grad[:n-d] -= p_val * h[d:]
    
    return grad

def descent_step(h, eta, alpha, dx, target_sum):
    """Perform one descent step"""
    grad = gradient_smoothing(h, alpha, dx)
    h_new = project_to_sum_box(h - eta * grad, target_sum)
    return h_new

def gradient_descent(h_start, time_limit, dx, target_sum):
    """Run gradient-based descent"""
    h = h_start.copy()
    c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = c5
    
    t0 = time.time()
    it = 0
    
    # Schedule parameters
    alpha_start = 0.5
    alpha_end = 50.0
    eta_start = 0.1
    
    while time.time() - t0 < time_limit:
        progress = min(it / 2000.0, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        eta = eta_start * (0.998 ** it)
        
        h_new = descent_step(h, eta, alpha, dx, target_sum)
        c5_new = evaluate_c5(h_new, dx)
        
        if c5_new <= c5:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # Slight perturbation if no improvement
            h = project_to_sum_box(h + np.random.normal(0, 0.001, len(h)), target_sum)
            c5 = evaluate_c5(h, dx)
            
        it += 1
        if it > 2000:
            break
            
    return best_h, best_c5

def local_refine(h, target_sum, dx):
    """Refine solution using SLSQP optimizer"""
    def objective(h):
        return evaluate_c5(h, dx)
    
    def constraint_eq(h):
        return np.sum(h) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h))]
    
    try:
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, target_sum)
            c5_refined = evaluate_c5(h_refined, dx)
            return h_refined, c5_refined
    except:
        pass
    return h, evaluate_c5(h, dx)
