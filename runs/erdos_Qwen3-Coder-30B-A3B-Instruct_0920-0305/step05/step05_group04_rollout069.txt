import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def conv_fft(a, b):
    """Compute linear convolution using FFT with zero-padding."""
    n = len(a)
    m = len(b)
    N = n + m - 1
    A = np.fft.fft(a, N)
    B = np.fft.fft(b, N)
    return np.fft.ifft(A * B).real[:N]

def project_onto_bounded_simplex(x):
    """Project x onto the set {x: 0≤x_i≤1, sum x_i = n/2}."""
    n = len(x)
    x = np.clip(x, 0, 1)
    s = np.sum(x)
    target = n / 2
    
    # Bisection on lambda to find shift that makes sum == target
    if s > target:
        low, high = -1.0, 0.0
        while high - low > 1e-12:
            mid = (low + high) / 2
            y = np.clip(x + mid, 0, 1)
            if np.sum(y) < target:
                high = mid
            else:
                low = mid
        lam = low
    elif s < target:
        low, high = 0.0, 1.0
        while high - low > 1e-12:
            mid = (low + high) / 2
            y = np.clip(x + mid, 0, 1)
            if np.sum(y) > target:
                high = mid
            else:
                low = mid
        lam = high
    else:
        lam = 0.0
    
    return np.clip(x + lam, 0, 1)

def soft_weights(c, beta):
    """Compute softmax weights for soft minimum."""
    c_max = np.max(c)
    exp_term = np.exp(-beta * (c - c_max))
    return exp_term / np.sum(exp_term)

def smooth_obj_and_grad(h, beta):
    """Compute Φ_β and its gradient with FFT."""
    n = len(h)
    # Compute conv(h, h) = c_k
    c = conv_fft(h, h)
    c = c[1:n]  # Keep shifts 1,...,n-1

    # Compute soft weights
    w = soft_weights(c, beta)
    # Gradients related to shifts: 
    # g_k[i] = h[i+k] + h[i-k] if valid
    g = np.zeros_like(h)
    for k in range(1, len(c) + 1):
        if k < n:
            g[k:] += w[k-1] * h[:n-k]
        if k <= n:
            g[:n-k] += w[k-1] * h[k:]
    return -np.sum(w * c), -g

def gradient_descent_with_beta_continuation(h0, beta_start=1.0, beta_max=200.0, step_start=0.5, gamma=2.0, delta=0.95):
    """
    Run projected gradient descent on a surrogate with increasing β.
    A safe fallback logic is inside, designed to avoid overflows and saturations.
    """
    h = h0.copy()
    beta = beta_start
    step = step_start
    n = len(h)
    
    # Total iterations (budget check handled explicitly)
    total_iter = 0
    max_iter_per_restart = 1000
    converged = False
    
    while not converged and total_iter < max_iter_per_restart:
        c = conv_fft(h, h)[1:n]
        if np.isnan(c).any():
            # Possibly due to instability; reset
            return h, beta, step
        
        # Gradient descent step
        prev_h = h.copy()
        obj, grad = smooth_obj_and_grad(h, beta)
        
        # Armijo condition
        tau = step
        while tau > 1e-12:
            h_trial = project_onto_bounded_simplex(h - tau * grad)
            if np.any(np.isnan(h_trial)):
                tau /= 2
                continue
            obj_trial, _ = smooth_obj_and_grad(h_trial, beta)
            if obj_trial <= obj - 0.1 * tau * np.dot(grad, grad):
                break
            tau /= 2
        if tau < 1e-12:
            # Failed to find suitable step: penalty or restart strategy
            pass
        
        h = h_trial
        total_iter += 1
        
        # Convergence check: change in objective and norm of gradient
        obj_new, _ = smooth_obj_and_grad(h, beta)
        if abs(obj - obj_new) < 1e-12 and np.linalg.norm(grad) < 1e-9:
            if beta >= beta_max:
                break
            beta = min(beta * gamma, beta_max)
            step = step * delta

        # Early exit check if we're already quite close to a good solution
        max_shift_overlap = np.max(correlate(h, 1-h, mode='full')[n:])
        if max_shift_overlap < 0.3807:
            break

    return h, beta, step

def evaluate_c5(h):
    """Compute exact C₅ as max convolution with (1-h)."""
    corr = correlate(h, 1 - h, mode="full")
    n = len(h)
    return np.max(corr[n:])
