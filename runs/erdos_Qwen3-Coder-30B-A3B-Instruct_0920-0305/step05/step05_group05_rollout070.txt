import numpy as np
import time

def project_to_box_simplex(v, target_sum):
    """Project vector v onto box [0,1] and simplex with sum = target_sum"""
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0, 1)

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # use np.correlate instead of convolve for correct correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # Correct the scale by dx for overlap integral
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, alpha):
    """Compute gradient of smoothed max of overlap values"""
    n = len(h)
    dx = 2.0 / n
    # correlation
    corr = np.correlate(h, 1 - h, mode="full")
    O = corr * dx
    O_max = np.max(O)
    # compute stable exponential weights
    exp_term = np.exp(alpha * (O - O_max))
    w = exp_term / np.sum(exp_term)
    
    grad = np.zeros_like(h)
    
    # For each shift, compute shift_right and shift_left 
    for idx, shift in enumerate(range(-n + 1, n)):
        w_k = w[idx]
        if w_k == 0:
            continue
        
        if shift >= 0:
            # shift_right: 1 for entries where i+shift out of range, else 1 - h[i+shift]
            shift_right = np.empty(n)
            shift_right[:shift] = 1.0
            shift_right[shift:] = 1.0 - h[shift:]
            # shift_left: h[i-shift] for i>=shift, 0 else
            shift_left = np.empty(n)
            shift_left[:shift] = 0.0
            shift_left[shift:] = h[:n-shift]
        else:
            # shift negative
            s = -shift
            # shift_right: 1 for i<s, else 1 - h[i-s]
            shift_right = np.empty(n)
            shift_right[:s] = 1.0
            shift_right[s:] = 1.0 - h[:n-s]
            # shift_left: h[i+s] for i>=0, 0 otherwise (but i+s < n, so we compute)
            shift_left = np.empty(n)
            shift_left[:n-s] = h[s:]
            shift_left[n-s:] = 0.0
        
        grad += w_k * (shift_right - shift_left)
    
    return grad * dx

def optimize_h(n, max_iters, alpha_start, alpha_end, initial_h=None):
    """Optimize h vector using gradient descent with smoothed max"""
    if initial_h is not None:
        h = initial_h.copy()
    else:
        # Initialize random vector, then project to feasible region
        h = np.random.rand(n)
        h = project_to_box_simplex(h, n/2)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Gradient descent
    for it in range(max_iters):
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        
        grad = compute_gradient(h, alpha)
        
        # Step size based on iteration to ensure convergence  
        lr = 0.5 / np.sqrt(it + 1)
        h = h - lr * grad
        h = project_to_box_simplex(h, n/2)
        
        # Evaluate the current solution
        cur_c5 = evaluate_c5(h)
        if cur_c5 < best_c5:
            best_h = h.copy()
            best_c5 = cur_c5
    
    return best_h, best_c5
