import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60
DX = 2.0 / 120

def evaluate_c5(h):
    """Evaluate C5 using correlation"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * DX

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -2.0, 2.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def gradient_smoothing(h, alpha):
    """Compute gradient of log-sum-exp smoothed maximum correlation"""
    n = len(h)
    # correlation of h with (1-h)
    cov_full = np.correlate(h, 1.0 - h, mode='full')
    I = cov_full  # shape 2n-1
    I_max = I.max()
    # compute weights for softmax
    exp_vals = np.exp(alpha * (I - I_max))
    w = exp_vals / exp_vals.sum()
    # compute gradient as weighted sum of per-lag grads
    grad = np.zeros(n)
    # Pre-compute arrays
    h_arr = h
    one_minus_h = 1.0 - h_arr
    
    for idx, p_val in enumerate(w):
        lag = idx - (n - 1)  # ℓ positive means shift left (i+ℓ)
        if lag >= 0:
            # term (1 - h[i+lag]) for i=0..n-1-lag
            if n - lag > 0:
                grad[:n-lag] += p_val * one_minus_h[lag:]
            # edge: i where i+lag >= n => i in [n-lag, n-1]
            if lag > 0:
                grad[n-lag:] += p_val * 1.0
            # term - h[i-lag] for i=lag..n-1
            if lag > 0:
                grad[lag:] -= p_val * h_arr[:n-lag]
            else:
                # lag==0, subtract h[i] from all positions
                grad -= p_val * h_arr
        else:
            # lag negative -> shift right: define d = -lag
            d = -lag
            # term (1 - h[i - d]) -> (1 - h[i-d]) for i=d..n-1
            if n - d > 0:
                grad[d:] += p_val * one_minus_h[:n-d]
            # edge: i where i+lag < 0 (i < d)
            grad[:d] += p_val * 1.0
            # term - h[i+d] for i=0..n-d-1 -> -h[i+d]
            if n - d > 0:
                grad[:n-d] -= p_val * h_arr[d:]
    
    # Scale gradient by dx to match objective scaling
    return grad * DX

def descent_step(h, eta, alpha):
    """Perform one descent step"""
    grad = gradient_smoothing(h, alpha)
    # compute candidate using projected gradient step
    h_new = h - eta * grad
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    return h_new

def gradient_descent(h_start, time_limit):
    """Perform gradient descent optimization"""
    h = h_start.copy()
    c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5
    t0 = time.time()
    it = 0
    
    # schedule parameters
    alpha_start = 0.5
    alpha_end = 50.0
    eta_start = 0.1
    
    # Set maximum iterations and track progress
    max_iter = 10000
    while time.time() - t0 < time_limit and it < max_iter:
        progress = it / max_iter
        alpha = alpha_start * (alpha_end / alpha_start) ** progress  # exponential
        eta = eta_start * (0.998 ** it)  # exponential decay
        
        h_new = descent_step(h, eta, alpha)
        c5_new = evaluate_c5(h_new)
        
        if c5_new <= c5:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        it += 1
        
        # Occasionally add noise to avoid local minima
        if it % 500 == 0:
            noise = np.random.normal(0, 0.02, len(h))
            h_noisy = project_to_sum_box(h + noise, TARGET_SUM)
            c5_noisy = evaluate_c5(h_noisy)
            if c5_noisy < c5:
                h = h_noisy
                c5 = c5_noisy
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    
    return best_h, best_c5

def local_refine(h):
    """Refine solution using SLSQP optimizer"""
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h))]
    
    try:
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
    except:
        pass
    return h, evaluate_c5(h)
