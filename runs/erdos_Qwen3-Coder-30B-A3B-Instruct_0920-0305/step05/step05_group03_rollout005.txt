import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from cvxpy import *
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_simplex(v, target_sum):
    """Project vector v onto the capped simplex with sum = target_sum."""
    # Ensure v is a numpy array
    v = np.asarray(v)
    n = len(v)
    
    # Sort v in descending order to find the threshold
    sorted_v = np.sort(v)[::-1]
    
    # Find the threshold index where sum of elements above it is greater than target_sum
    cumsum = np.cumsum(sorted_v)
    threshold_index = np.argmax(cumsum > target_sum)
    
    # If all elements are included, we can simply scale
    if threshold_index == 0:
        if cumsum[0] > target_sum:
            # Need to scale everything to target_sum
            scale = target_sum / cumsum[0]
            return np.clip(v * scale, 0, 1)
        else:
            # Already valid
            return np.clip(v, 0, 1)
    
    # Threshold is at threshold_index
    theta = (target_sum - cumsum[threshold_index - 1]) / (threshold_index)
    
    # Apply projection
    projected = np.clip(v - theta, 0, 1)
    return projected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Precompute all the pairs for each shift (to build MILP constraints)
    all_pairs = {}
    for k in range(-(n_points-1), n_points):
        pairs = []
        for i in range(n_points):
            j = i + k
            if 0 <= i < n_points and 0 <= j < n_points:
                pairs.append((i, j))
        all_pairs[k] = pairs
    
    # MILP part: define variables
    h = Variable(n_points, boolean=True)
    y = {}
    for k in all_pairs:
        for (i, j) in all_pairs[k]:
            if (i, k) not in y:
                y[(i, k)] = Variable(boolean=True)
    
    # Add constraints for product linearization
    constraints = []
    for k in all_pairs:
        for (i, j) in all_pairs[k]:
            constraints += [
                y[(i, k)] <= h[i],
                y[(i, k)] <= h[j],
                y[(i, k)] >= h[i] + h[j] - 1
            ]

    # Constraint on area (sum of h must be n/2)
    constraints.append( sum(h) == n_points / 2 )

    # Define objective: tau
    tau = Variable()

    # Add overlap constraints for each shift
    for k in all_pairs:
        overlap = sum(h[i] for i in range(n_points)) - sum(y[(i, k)] for (i, j) in all_pairs[k])
        constraints.append(overlap <= tau)
    
    # Build and solve MILP
    objective = Minimize(tau)
    problem = Problem(objective, constraints)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Try solving MILP
    try:
        problem.solve(solver='CBC', time_limit=budget_s*0.25)
        if problem.status in [OPTIMAL, FEASIBLE]:
            h_sol = h.value
            if h_sol is not None:
                h_sol = np.round(h_sol)
                c5_sol = evaluate_c5(h_sol)
                if c5_sol < best_c5:
                    best_h = h_sol.copy()
                    best_c5 = c5_sol
    except:
        pass
    
    # Binary hill climbing
    h_current = best_h.copy()
    h_current = np.round(h_current)

    def compute_max_overlap(h):
        corr = correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Hill climbing
    best_binary = h_current.copy()
    best_binary_c5 = compute_max_overlap(h_current)
    
    steps = 0  # Track steps for termination condition
    max_steps = 5000  # Avoid infinite loop
    
    improved = True
    while improved and steps < max_steps and (time.time() - start_time) < budget_s * 0.5:
        improved = False
        new_h = h_current.copy()
        old_c5 = compute_max_overlap(h_current)
        for i in range(n_points):
            for j in range(i+1, n_points):
                if h_current[i] == 1 and h_current[j] == 0:
                    new_h[i] = 0
                    new_h[j] = 1
                    new_c5 = compute_max_overlap(new_h)
                    if new_c5 < best_binary_c5:
                        best_binary = new_h.copy()
                        best_binary_c5 = new_c5
                        improved = True
                        h_current = new_h.copy()
                    new_h[i] = 1
                    new_h[j] = 0
                elif h_current[i] == 0 and h_current[j] == 1:
                    new_h[i] = 1
                    new_h[j] = 0
                    new_c5 = compute_max_overlap(new_h)
                    if new_c5 < best_binary_c5:
                        best_binary = new_h.copy()
                        best_binary_c5 = new_c5
                        improved = True
                        h_current = new_h.copy()
                    new_h[i] = 0
                    new_h[j] = 1
        steps += 1

    # Continuous refinement
    h_final = best_binary.astype(float)
    h_final = project_simplex(h_final, n_points/2)

    def compute_subgradient(h):
        """Compute subgradient of max overlap for current h."""
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        # Get active shifts (some may be more than one, but we focus on highest)
        active_shifts = []
        for k in range(-(n_points-1), n_points):
            if np.isclose(corr[k + n_points - 1], corr[max_k]):
                active_shifts.append(k)
        # Build weight vector
        w = np.zeros(n_points)
        for k in active_shifts:
            start = max(0, -k)
            end = min(n_points, n_points - k)
            indices = np.arange(start, end)
            w[indices] += 1
        # Normalize if not zero
        if np.sum(w) > 0:
            w /= np.sum(w)
        return w

    # Refinement loop
    for iter in range(1000):
        if (time.time() - start_time) > budget_s * 0.9:
            break
        # Get subgradient
        w = compute_subgradient(h_final)
        # Compute gradient
        ones = np.ones(n_points)
        # Convolve w with all ones (most efficient)
        w_conv = np.convolve(w, ones, mode="full")[n_points-1:2*n_points-1]
        # Convolve w with h for the terms
        w_h = correlate(w, h_final, mode="full")
        # Convolve w with 1-h
        w_1h = correlate(w, 1 - h_final, mode="full")
        # Gradient
        # This needs some work because it was a bit off before
        g = dx * (w_conv - w_h - w_1h)
        # Take step
        eta = 0.5 * dx
        h_new = h_final - eta * g
        # Project
        h_new = project_simplex(h_new, n_points/2)
        # Symmetrize
        h_new = (h_new + h_new[::-1]) / 2
        h_new = project_simplex(h_new, n_points/2)
        # Evaluate
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_final = h_new

    # Final evaluation
    if (time.time() - start_time) < budget_s * 0.95:
        # Optional upsample if enough time
        n_upsample = n_points * 2
        h_up = np.interp(np.linspace(0, 2, n_upsample), np.linspace(0, 2, n_points), best_h)
        h_up = project_simplex(h_up, n_upsample/2)
        # refinement for upsampled
        for iter in range(200):
            if (time.time() - start_time) > budget_s:
                break
            w = compute_subgradient(h_up)
            ones = np.ones(n_upsample)
            w_conv = np.convolve(w, ones, mode="full")[n_upsample-1:2*n_upsample-1]
            w_h = correlate(w, h_up, mode="full")
            w_1h = correlate(w, 1 - h_up, mode="full")
            g = dx * (w_conv - w_h - w_1h)
            eta = 0.2 * (2.0/n_upsample)
            h_new = h_up - eta * g
            h_new = project_simplex(h_new, n_upsample/2)
            h_new = (h_new + h_new[::-1]) / 2
            h_new = project_simplex(h_new, n_upsample/2)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_up = h_new

    return (best_h.tolist(), best_c5, len(best_h))
