import numpy as np
from scipy.signal import correlate
from scipy.fft import rfft, irfft
import time
import math

def project_simplex(y, target_sum):
    """Project y onto the capped simplex: 0 <= z <= 1, sum(z) = target_sum"""
    # Binary search on lambda to find the correct projection
    lo, hi = -100.0, 100.0
    for _ in range(50):
        lamb = (lo + hi) / 2
        z = np.clip(y - lamb, 0, 1)
        s = np.sum(z)
        if s > target_sum:
            lo = lamb
        else:
            hi = lamb
    z = np.clip(y - lo, 0, 1)
    # Rescale to ensure sum = target_sum
    z = z * target_sum / np.sum(z)
    return z

def evaluate_c5_fft(h):
    """Fast evaluation of C5 using FFT"""
    n = len(h)
    dx = 2.0 / n
    # h * (1-h) correlation
    H = rfft(h)
    G = rfft(1 - h)
    corr = irfft(H * np.conj(G))
    # Remove padding
    corr = corr[:n]
    # Compute the maximum value at any shift
    max_corr = np.max(corr)
    return max_corr * dx

def repair_h(c, x_knots, n, target_sum):
    """Reconstruct full h from control points c, then repair to feasible set"""
    # Interpolate to full grid
    h = np.interp(np.linspace(0, 2, n), x_knots, c)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Project onto simplex (sum = target_sum)
    h = project_simplex(h, target_sum)
    return h

def objective(c, x_knots, n, target_sum, alpha):
    """Objective function: soft-max of all overlaps, with smoothing parameter alpha"""
    h = repair_h(c, x_knots, n, target_sum)
    # Evaluate using fast FFT
    dx = 2.0 / n
    H = rfft(h)
    G = rfft(1 - h)
    corr = irfft(H * np.conj(G))
    corr = corr[:n] * dx
    # Soft maximum: log(sum(exp(alpha * x))) / alpha
    max_corr = np.max(corr)
    if alpha < 1e10:
        # Soft max
        shifted_corr = corr - max_corr
        soft_max = max_corr + np.log(np.sum(np.exp(alpha * shifted_corr))) / alpha
        return soft_max
    else:
        # True max
        return np.max(corr)

    x_knots = np.linspace(0, 2, K)

    lambda_ = max(4, int(4 + 3 * np.log(K)))  # Population size
    mu = lambda_ // 2  # Number of parents

    ps = np.zeros(K)  # Evolution path for step size
    c1 = 2 / (K + 2)  # Learning rate for covariance matrix
    cmu = 2 * (mu - 2) / (K + 2)  # Learning rate for covariance matrix
    damps = 1 + 2 * max(0, np.sqrt((mu - 1) / (K + 1)) - 1) + c1  # Damping factor
    
    best_c = mean.copy()
    best_val = float('inf')

    # Adaptive smoothing parameter alpha
    alpha_start = 5.0
    alpha_target = 30.0
    alpha_schedule_steps = max(100, int(0.7 * budget_s / 2))
    alpha = alpha_start
    alpha_delta = (alpha_target - alpha_start) / alpha_schedule_steps
    
    num_generations = 0
    time_per_generation = 0

    num_restarts = 0
    max_restarts = 10

    restart_count = 0

            # Sample offspring
            offspring = []
            current_time = time.time()
            
            # Determine number of new generations based on time remaining

                    gens_to_do = max(1, int(time_left / time_per_generation))

                    gens_to_do = 1
            else:
                gens_to_do = 1

            for i in range(lambda_):
                # Sample new candidate
                z = np.random.multivariate_normal(np.zeros(K), C)
                x = mean + sigma * z
                x = np.clip(x, 0, 1)
                offspring.append(x)

            # Evaluate and select best
            values = []
            for c in offspring:
                v = objective(c, x_knots, n, target_sum, alpha)
                values.append(v)
            
            # Sort and select best parents
            idx = np.argsort(values)
            best_indices = idx[:mu]
            parents = [offspring[i] for i in best_indices]
            parent_values = [values[i] for i in best_indices]

            # Update best
            for i in range(mu):
                if parent_values[i] < best_val:
                    best_val = parent_values[i]
                    best_c = parents[i].copy()

            # Update mean and covariance
            weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
            weights = weights / np.sum(weights)
            mean_new = np.zeros(K)
            for i in range(mu):
                mean_new += weights[i] * parents[i]
            mean = mean_new

            # Update paths

            mean_old = mean
            zmean = np.dot(np.linalg.cholesky(C), mean - mean_old) / sigma
            pc = (1 - cmu) * pc + np.sqrt(cmu * (2 - cmu)) * zmean
            
            # Update covariance
            C = (1 - c1 - cmu) * C + c1 * (np.outer(pc, pc) + (1 - c1) * C) + cmu * np.sum([weights[i] * np.outer(parents[i] - mean, parents[i] - mean) for i in range(mu)], axis=0)
            
            # Update step size
            sigma = sigma * np.exp((np.linalg.norm(ps) - np.sqrt(2)) / damps)

            # Adaptive scheduling of alpha
            alpha = min(alpha + alpha_delta, alpha_target)
            
            # Symmetrise for symmetry encouragement
            h = repair_h(best_c, x_knots, n, target_sum)
            # Symmetrise
            h_rev = np.flip(h)
            h_sym = (h + h_rev) / 2
            # Get new candidate from symmetrized version
            h_new = h_sym.copy()
            # Quantize back to control points (interpolation)
            for i in range(K):
                best_c[i] = np.interp(x_knots[i], np.linspace(0, 2, n), h_new)
            
            # Periodic check
            total_generations += 1
            if total_generations % 20 == 0:
                if (time.time() - start_time) > budget_s * 0.95:
                    break
        
        # Exit if enough time passed
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
        restart_count += 1

    # Final polishing with projected gradient on true max
    h_final = repair_h(best_c, x_knots, n, target_sum)
    best_c_final = best_c.copy()

    # Use the best feasible solution found
    final_h = h_final
    
    # Final exact evaluation of C5
