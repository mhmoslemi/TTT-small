import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def initial_h_values():
    """Produce the initial step function from the problem specification"""
    # This is a standard construction that denotes a piecewise constant function
    # over [0,2] with n = 42 points, giving integral equal to 1.
    n = 42
    dx = 2.0 / n
    h = np.zeros(n)
    # Place two blocks of high density near 0.2 and 1.2
    # First block: from 0.1 to 0.3 = 0.2 width, so 0.2/dx = 0.2/(2/42) = 4.2 points => 4 points
    # Second block: from 1.1 to 1.3 = 0.2 width, so 4 points
    start1 = int(0.1 / dx)
    end1 = int(0.3 / dx)
    start2 = int(1.1 / dx)
    end2 = int(1.3 / dx)
    h[start1:end1] = 1
    h[start2:end2] = 1
    # Normalize to get integral = 1 (sum * dx = 1)
    integral = np.sum(h) * dx
    h = h / integral * 1  # Scale
    return h


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Start with initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    
    # Use the same optimization approach but include a better variant
    # Phase 1: Start with evolved solution if any (a minimal pattern which is really fast)
    h_best = h0.copy()
    c5_best = evaluate_c5(h0)
    
    # Additional reasonably fast refinement with SQP + accurate constraints handling
    from scipy.optimize import minimize
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(x):
        return evaluate_c5(x)
    
    def constraint_eq(x):
        dx = 2.0 / n_points
        return np.sum(x) * dx - 1.0

    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Try a few more round of local optimization with different random restarts
    for i in range(20):
        if time.time() - start_time > budget_s * 0.95:
            break
        try:
            # Small random perturbation
            x0 = h_best + np.random.uniform(-0.05, 0.05, n_points)
            x0 = np.clip(x0, 0, 1)
            # Normalize to respect integral constraint (exact)
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(x0) * dx)
            x0 = x0 * scale
            
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-9})
            if res.success:
                x_candidate = res.x
                scale = 1.0 / (np.sum(x_candidate) * dx)
                x_candidate = x_candidate * scale
                c5_candidate = objective(x_candidate)
                if c5_candidate < c5_best:
                    h_best = x_candidate.copy()
                    c5_best = c5_candidate
        except:
            continue

    # Final element-wise smoothing and botched k-means with max-min spread
    if time.time() - start_time < budget_s * 0.98:
        # Apply a hybrid tweak: do a kind of convolutional smoothing with hard threshold
        v = h_best.copy()
        dx = 2.0 / n_points
        for _ in range(5):
            kernel = np.ones(3) / 3
            v_smoothed = np.convolve(v, kernel, mode='same')
            # Hard thresholding back to feasible region
            v = np.clip(v_smoothed, 0, 1)
            # Re-normalize
            scale = 1.0 / (np.sum(v) * dx)
            v = v * scale
        final_c5 = evaluate_c5(v)
        if final_c5 < c5_best:
            h_best = v
            c5_best = final_c5

    return (h_best.tolist(), c5_best, n_points)
