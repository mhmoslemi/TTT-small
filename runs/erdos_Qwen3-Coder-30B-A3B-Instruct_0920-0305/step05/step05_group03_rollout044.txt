```python
import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Start with initial construction
    h0 = np.array(initial_h_values)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Phase 1: Binary MILP for optimal discrete solution
    print("Starting Binary MILP phase...")
    try:
        # Create binary variables
        h = cp.Variable(n_points, boolean=True)
        tau = cp.Variable()

        # Constraints for area
        constraints = [cp.sum(h) == n_points / 2]
        
        # For each shift compute the overlap
        max_shift = n_points - 1
        # Precompute the indices for each shift
        shift_indices = {}
        for k in range(-max_shift, max_shift + 1):
            indices = []
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    indices.append((i, j))
            shift_indices[k] = indices
        
        # For each shift, create auxiliary variables
        y_vars = {}
        overlaps = []
        for k in range(-max_shift, max_shift + 1):
            pairs = shift_indices[k]
            y_k = cp.Variable(len(pairs), boolean=True)
            y_vars[k] = y_k
            
            # Connect y_k to h via McCormick constraints
            for idx, (i, j) in enumerate(pairs):
                constraints += [
                    y_k[idx] <= h[i],
                    y_k[idx] <= h[j],
                    y_k[idx] >= h[i] + h[j] - 1
                ]
            
            # Compute overlap for this shift
            overlap = cp.sum(h) - cp.sum(y_k)
            overlaps.append(overlap)
            constraints.append(overlap <= tau)
        
        # Objective
        objective = cp.Minimize(tau)
        prob = cp.Problem(objective, constraints)
        
        # Solve with time limit
        prob.solve(solver=cp.CBC, verbose=False, time_limit=budget_s*0.24)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h.value
            c5_sol = evaluate_c5(h_sol)
            if c5_sol < best_c5:
                best_h = h_sol
                best_c5 = c5_sol
        print(f"MILP found C5 = {best_c5}")
    except Exception as e:
        print(f"MILP error: {e}")

    # Phase 2: Binary hill climb
    print("Starting Binary Hill Climbing...")
    def compute_overlap(h_vec, k):
        # Compute h * (1 - h_{k})
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return corr[len(corr)//2 + k] * dx

    def get_all_overlaps(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.array([corr[len(corr)//2 + k] * dx for k in range(-n_points + 1, n_points)])

    max_iter = 5000
    for _ in range(max_iter):
        if (time.time() - start_time) > budget_s * 0.5:
            break
        best_swap = None
        best_improvement = 0
        for i in range(n_points):
            if best_h[i] == 1:
                for j in range(n_points):
                    if best_h[j] == 0:
                        # Compute the potential gain of swapping
                        new_h = best_h.copy()
                        new_h[i] = 0
                        new_h[j] = 1
                        
                        # Re-evaluate all overlaps for the new h
                        new_overlaps = get_all_overlaps(new_h)
                        old_overlaps = get_all_overlaps(best_h)
                        
                        # Compute current max overlaps
                        old_max = np.max(old_overlaps)
                        new_max = np.max(new_overlaps)
                        
                        improvement = old_max - new_max
                        
                        if improvement > best_improvement:
                            best_improvement = improvement
                            best_swap = (i, j)
        
        if best_swap:
            i, j = best_swap
            best_h[i] = 0
            best_h[j] = 1
            current_c5 = evaluate_c5(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
            else:
                # Revert if no improvement
                best_h[i] = 1
                best_h[j] = 0
        else:
            # No improvement found in this iteration
            break

    print(f"After HC, C5 = {best_c5}")

    # Phase 3: Continuous refinement
    print("Starting Continuous Refinement...")
    
    def project_simplex(v, target_sum):
        """Project v onto the capped simplex with sum = target_sum"""
        # Simple waterfilling algorithm
        v_sorted = np.sort(v)[::-1]
        cumsum = np.cumsum(v_sorted)
        n = len(v)
        threshold = (cumsum - target_sum) / np.arange(1, n + 1)

        idx = -1
        for i in range(len(threshold)):
            if threshold[i] < v_sorted[i]:
                idx = i
        if idx == -1:
            # All values are below threshold
            return np.maximum(v, 0)  # just clip to [0,1]
        
        theta = threshold[idx]
        projected = np.maximum(v - theta, 0)
        # Ensure sum is correct
        current_sum = np.sum(projected)
        scale = target_sum / current_sum
        projected = projected * scale
        return projected
    
    h_cont = best_h.astype(float)
    initial_c5 = evaluate_c5(h_cont)
    print(f"Initial continuous C5 = {initial_c5}")
    
    # Perform gradient descent loop
    max_iter = 10000
    eta = 0.5 * dx
    patience = 0
    max_patience = 500
    
    for i in range(max_iter):
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
        if patience >= max_patience:
            break
            
        # Compute all overlaps (to find active k)
        corr_full = correlate(h_cont, 1 - h_cont, mode="full")
        overlaps = np.array([corr_full[len(corr_full)//2 + k] * dx for k in range(-n_points + 1, n_points)])
        
        # Find the active shifts (max overlaps)
        max_val = np.max(overlaps)
        active_k = np.where(overlaps == max_val)[0]
        # Represent as shifts
        active_shifts = [k for k in range(-n_points + 1, n_points) if k in active_k - (len(overlaps)//2)]
        
        # Construct weight vector: sum over active shifts
        w = np.zeros(n_points)
        for k in active_shifts:
            if k >= 0:
                w[k:] += 1
            else:
                w[:k] += 1
                
        # Compute sub-gradient of max overlap
        conv_w_h = correlate(w, h_cont, mode="full")
        conv_w_h = conv_w_h[len(conv_w_h)//2:]  # Left part
        conv_w_h_rev = correlate(w, h_cont[::-1], mode="full")
        conv_w_h_rev = conv_w_h_rev[len(conv_w_h_rev)//2:]  # Left part
        
        # Update using empirical sub-gradient
        gradient = dx * (np.ones(n_points) - conv_w_h - conv_w_h_rev)
        
        # Take step
        h_new = h_cont - eta * gradient
        
        # Project with integral constraint
        target_sum = n_points / 2
        h_new = project_simplex(h_new, target_sum)
        
        # Symmetrise
        h_sym = (h_new + h_new[::-1]) / 2
        h_sym = project_simplex(h_sym, target_sum)
        
        # Evaluate C5 on symmetrized version
        c5_new = evaluate_c5(h_sym)
        
        # If improvement, accept
        if c5_new < best_c5:
            best_h = h_sym.copy()
            best_c5 = c5_new
            patience = 0
        else:
            patience += 1
            # Reduce learning rate
            eta = max(1e-6 * dx, eta * 0.995)
            if eta < 1e-6 * dx:
                break
                
    print(f"Final refined C5 = {best_c5}")
    
    # Final verification
    print("Final step function evaluated...")
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
```
