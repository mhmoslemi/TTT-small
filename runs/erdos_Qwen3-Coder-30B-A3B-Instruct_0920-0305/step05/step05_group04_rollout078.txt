import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def conv_fft(a, b):
    """Compute linear convolution using FFT with zero padding"""
    n = len(a)
    m = len(b)
    L = n + m - 1
    # Zero-pad to length L
    a_pad = np.pad(a, (0, L - n), 'constant')
    b_pad = np.pad(b, (0, L - m), 'constant')
    # Compute convolution in frequency domain
    A = np.fft.fft(a_pad)
    B = np.fft.fft(b_pad)
    C = A * B
    c = np.fft.ifft(C).real
    return c[:L]

def project_onto_bounded_simplex(h):
    """Project h onto the bounded simplex: sum h_i = n/2, 0 <= h_i <= 1"""
    n = len(h)
    h_clipped = np.clip(h, 0, 1)
    s = np.sum(h_clipped)
    if abs(s - n / 2) < 1e-12:
        return h_clipped
    if s < n / 2:
        # Need to increase some values; set all caps to 1
        h_proj = np.minimum(h_clipped, 1.0)
        diff = n / 2 - np.sum(h_proj)
        # Only increase values that are < 1
        indices = np.where(h_proj < 1)[0]
        # Distribute diff among indices, prefer larger h values
        diffs = 1 - h_proj[indices]
        if np.sum(diffs) > 0:
            weights = diffs / np.sum(diffs)
            h_proj[indices] += diff * weights
        else:
            # All at cap; add smoothly
            h_proj = np.ones(n) * (1 - diff / n)
            h_proj[np.argmax(h_clipped)] += diff
        h_proj = np.clip(h_proj, 0, 1)
    else:
        # Need to reduce some values; set all floors to 0
        h_proj = np.maximum(h_clipped, 0.0)
        diff = np.sum(h_proj) - n / 2
        # Reduce values that are > 0
        indices = np.where(h_proj > 0)[0]
        # Distribute diff among indices
        diffs = h_proj[indices]
        if np.sum(diffs) > 0:
            weights = diffs / np.sum(diffs)
            h_proj[indices] -= diff * weights
        else:
            # All at floor; reduce smoothly
            h_proj = np.ones(n) * (diff / n)
            h_proj[np.argmin(h_clipped)] -= diff
        h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def soft_weights(c, beta):
    """Compute softmax weights for the soft minimum"""
    c_max = np.max(c)
    # Subtract c_max to avoid overflow
    exp_c = np.exp(-beta * (c - c_max))
    return exp_c / np.sum(exp_c)

def smooth_obj_and_grad(h, beta):
    """Compute the smooth objective and its gradient"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations using FFT
    c = conv_fft(h, h)
    # c_k = sum_{i=0}^{n-k-1} h_i * h_{i+k} for k=1,...,n-1
    # This gives us c[1:n] as c_1,...,c_{n-1}
    c_vals = c[1:n]

    if len(c_vals) == 0:
        return 0.0, np.zeros(n)

    # Compute soft weights
    weights = soft_weights(c_vals, beta)

    # Gradient of sum_k w_k * c_k is -sum_k w_k * (shifted_h)
    # Build shifted versions
    u = conv_fft(h, weights)
    v = conv_fft(h, np.flip(weights))
    grad = -(u + v)
    # Only keep first n components
    grad = grad[:n]

    # Phi_beta is -1/beta * log(sum of exp(-beta c_k))
    log_sum_exp = np.log(np.sum(np.exp(-beta * c_vals))) + beta * c_max
    Phi = -log_sum_exp / beta
    return Phi, grad

def line_search(h, g, beta, alpha=0.5):
    """Backtracking line search using Armijo condition"""
    tau = alpha
    c_prev = smooth_obj_and_grad(h, beta)[0]
    counter = 0
    while counter < 30:
        h_candidate = project_onto_bounded_simplex(h - tau * g)
        c_candidate = smooth_obj_and_grad(h_candidate, beta)[0]
        if c_candidate <= c_prev - 0.1 * tau * np.dot(g, g):
            return h_candidate, tau
        tau = tau / 2
        counter += 1
    return h, tau

def inner_loop(h, beta_start=1, beta_max=200, alpha_start=0.1, gamma=2, delta=0.9):
    """Run projected gradient descent with β continuation"""
    beta = beta_start
    alpha = alpha_start
    h = project_onto_bounded_simplex(h)
    dx = 2.0 / len(h)
    
    iter_count = 0
    while beta < beta_max and iter_count < 2000:
        # Compute objective and gradient
        Phi, grad = smooth_obj_and_grad(h, beta)
        # Line search
        h_new, tau = line_search(h, grad, beta, alpha)
        # Check convergence
        if np.linalg.norm(tau * grad) < 1e-9:
            break
        # Update
        h = h_new
        if abs(Phi - smooth_obj_and_grad(h, beta)[0]) < 1e-12:
            beta = min(beta * gamma, beta_max)
            alpha = alpha * delta
        iter_count += 1
    return h

def evaluate_c5(h):
    """Evaluate the C5 constant"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx
