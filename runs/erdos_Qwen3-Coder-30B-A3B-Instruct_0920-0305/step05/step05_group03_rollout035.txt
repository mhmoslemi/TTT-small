import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time
from scipy.optimize import minimize
from itertools import combinations

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_simplex(v, target_sum):
    """Project vector v onto the capped simplex with sum = target_sum"""
    v = np.asarray(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(u)
    # Find the threshold
    rho = np.where(u > (cumsum - target_sum) / np.arange(1, len(u) + 1))[0]
    if len(rho) == 0:
        # All values should be zero
        return np.zeros_like(v)
    rho = rho[-1] + 1
    theta = (cumsum[rho-1] - target_sum) / rho
    # Apply projection
    return np.clip(v - theta, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Phase 1: Binary MILP (if time allows)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Solve MILP
    try:
        # Assume the best binary is found, use a modified version of the code to create model
        h_bin = Variable(n_points, boolean=True)
        s = sum(h_bin)
        # The sum constraint
        constraints = [s == n_points / 2]
        
        # Create a variable to hold maximum overlap
        tau = Variable()
        
        for k in range(-n_points+1, n_points):
            # Define the pairs for shift k
            P_k = []
            for i in range(max(0, -k), min(n_points, n_points-k)):
                P_k.append((i, i+k))
            
            y_vars = []
            for i, j in P_k:
                y = Variable(boolean=True)
                constraints.extend([
                    y <= h_bin[i],
                    y <= h_bin[j],
                    y >= h_bin[i] + h_bin[j] - 1
                ])
                y_vars.append(y)
            
            # Overlap constraint for shift k
            overlap = sum(h_bin) - sum(y_vars)
            constraints.append(overlap <= tau)
        
        objective = Minimize(tau)
        prob = Problem(objective, constraints)
        
        # Try solving
        solve_time = time.time() - start_time
        if solve_time < budget_s * 0.3:
            prob.solve(solver=CBC, time_limit=budget_s * 0.25)
            
            h_bin_val = h_bin.value
            if h_bin_val is not None:
                h_bin_val = np.round(h_bin_val).astype(float)
                c5_val = evaluate_c5(h_bin_val)
                if c5_val < best_c5:
                    best_h = h_bin_val
                    best_c5 = c5_val
    except Exception as _:
        pass
    
    # Phase 2: Hill-climbing to improve binary solutions
    h_current = best_h.copy()
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current / (np.sum(h_current) * dx) * (n_points / 2)
    current_c5 = evaluate_c5(h_current)
    
    if current_c5 < best_c5:
        best_h = h_current
        best_c5 = current_c5
        
    # Simple hill climbing on binary
    h_binary = np.round(h_current).astype(int)
    if np.sum(h_binary) != n_points / 2:
        # Adjust binary by swapping
        ones_count = np.sum(h_binary)
        diff = ones_count - n_points / 2
        if diff > 0:
            # Find positions where 1 can be switched to 0
            indices = np.where(h_binary == 1)[0]
            for i in range(int(diff)):
                if len(indices) > 0:
                    idx = indices[i] 
                    h_binary[idx] = 0
        else:
            # Find positions where 0 can be switched to 1
            indices = np.where(h_binary == 0)[0]
            for i in range(int(-diff)):
                if len(indices) > 0:
                    idx = indices[i]
                    h_binary[idx] = 1
    
    # Evaluate binary solution
    h_binary = h_binary.astype(float)
    h_binary = h_binary / (np.sum(h_binary) * dx) * (n_points / 2)
    binary_c5 = evaluate_c5(h_binary)
    
    if binary_c5 < best_c5:
        best_h = h_binary
        best_c5 = binary_c5
        
    h_current = h_binary
    
    # Phase 3: Local refinement with swaps
    iter_count = 0
    max_iter = 2000
    while iter_count < max_iter and time.time() - start_time < budget_s * 0.7:
        improved = False
        for i in range(n_points):
            if h_current[i] <= 0.1:
                continue
            for j in range(n_points):
                if h_current[j] >= 0.9:
                    continue
                
                # Simulate swap i and j
                h_temp = h_current.copy()
                h_temp[i] -= dx
                h_temp[j] += dx
                
                # Ensure bounds
                if np.any(h_temp < 0) or np.any(h_temp > 1):
                    continue
                
                # Renormalize to area = 1
                h_temp = h_temp / (np.sum(h_temp) * dx) * (n_points / 2)
                
                temp_c5 = evaluate_c5(h_temp)
                if temp_c5 < best_c5:
                    h_current = h_temp
                    best_c5 = temp_c5
                    improved = True
                    
        if not improved:
            break
        iter_count += 1

    # Phase 4: Continuous improvement
    h_continuous = h_current.copy()
    
    # Sub-gradient with active shifts
    def get_active_shifts(h):
        n = len(h)
        dx = 2.0 / n
        corr = correlate(h, 1 - h, mode="full")
        max_val = np.max(corr)
        active_k = np.where(np.abs(corr - max_val) < 1e-10)[0]
        return active_k
    
    # Iterative descent
    eta = 0.5 * dx
    patience = 0
    max_patience = 200
    for i in range(5000):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        active_k = get_active_shifts(h_continuous)
        
        # Approximate the sub-gradient
        # Compute weights for active shifts
        w = np.zeros(n_points)
        for k in active_k:
            if k < 0:
                w[-k:] += 1
            else:
                w[:n_points-k] += 1
        
        # Compute the corrected gradient
        h_rev = h_continuous[::-1]
        w_conv = np.convolve(w, np.ones(n_points), mode='full')
        w_conv = w_conv[n_points-1:n_points-1+n_points]  # center part
        
        s = np.convolve(np.ones(n_points), w_conv, mode='full')
        s = s[n_points-1:n_points-1+n_points]  # center part
        
        grad = dx * (s - np.convolve(w, h_continuous, mode='full')[n_points-1:n_points-1+n_points] 
                     - np.convolve(w, h_rev, mode='full')[n_points-1:n_points-1+n_points])
        
        h_new = h_continuous - eta * grad
        
        # Project
        h_new = project_simplex(h_new, n_points / 2)
        
        # Symmetrize
        h_sym = (h_new + h_new[::-1]) / 2
        h_sym = project_simplex(h_sym, n_points / 2)
        
        c5_new = evaluate_c5(h_sym)
        if c5_new < best_c5:
            best_h = h_sym.copy()
            best_c5 = c5_new
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                eta *= 0.5
                patience = 0
                if eta < 1e-5 * dx:
                    break
    
    return (best_h.tolist(), best_c5, n_points)
