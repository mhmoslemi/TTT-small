import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
n_fine = 240
dx_fine = 2.0 / n_fine

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h, corr_vals, beta):
    """Compute gradient for the smooth max objective"""
    # Max to prevent overflow
    max_corr = np.max(corr_vals)
    exp_corr = np.exp(beta * (corr_vals - max_corr))
    w = exp_corr / np.sum(exp_corr)
    
    # Gradient: sum of weighted gradients
    # ∇C_ℓ = 1 - h_{i+ℓ} - h_{i-ℓ}, with zero padding
    grad = np.zeros_like(h)
    for i in range(len(h)):
        # h_{i+ℓ} contributions (left shift)
        for l in range(len(corr_vals)):
            if i + l < len(h):
                grad[i] += w[l] * -h[i + l]
            if i - l >= 0:
                grad[i] += w[l] * -h[i - l]
        # Add constant; handles sum constraint
        grad[i] += w.sum()
    
    return grad 

def smooth_max_optimization(h0, beta_schedule, step_schedule, max_iter, beta_max):
    """Optimize using projected gradient descent on smooth max loss"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for beta in beta_schedule:
        t_iter = 0
        while t_iter < max_iter and beta < beta_max:
            # Compute actual correlations for the current h
            corr_vals = compute_corr_fft(h)
            
            # Compute the smooth max gradient
            grad = compute_gradient(h, corr_vals, beta)
            
            # Update with projected gradient descent
            # lr = step_size * (1 / (1 + gamma * t_iter))
            lr = step_schedule(t_iter)
            if lr < 1e-10:
                break  # Converged
                
            h_new = h - lr * grad
            h = project_to_sum_box(h_new, TARGET_SUM)
            
            # Re-compute to see if this improved things
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            t_iter += 1
            
    return best_h, best_c5
