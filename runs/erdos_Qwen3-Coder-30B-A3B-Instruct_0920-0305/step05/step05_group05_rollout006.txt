import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

        u = cp.Variable(2*L)  # [a1, ..., aL, b1, ..., bL]
        s = cp.Variable(L)   # [s1, ..., sL]
        t = cp.Variable(1)   # the maximum overlap bound
        
        # Sample points for the box constraints
        M = 4*L
        x_points = np.linspace(0, 2, M)
        
        # Form the linear constraints for box bounds
        A_box = np.zeros((M, 2*L))

                A_box[i, L+j] = np.sin(np.pi * (j+1) * x)
        
        # The constraint is: 0.5 + A_box * u >= 0 and 0.5 + A_box * u <= 1
        # which means -0.5 <= A_box * u <= 0.5
        box_constraints = [
            A_box @ u >= -0.5,
            A_box @ u <= 0.5
        ]
        
        # Define SOC constraints: s_l >= a_l^2 + b_l^2
        soc_constraints = []
        for l in range(L):
            a_l = u[l]
            b_l = u[L + l]
            soc_constraints.append(cp.norm(cp.hstack([2*a_l, 2*b_l, s[l] - 1])) <= s[l] + 1)
        
        # The Toeplitz matrix T(t,s)
        # T_{ij} = t if i=j, -s[|i-j|] otherwise
        T_rows = []
        for i in range(L+1):
            row = [t if j == i else -s[abs(i-j)] if abs(i-j) < L else 0 for j in range(L+1)]
            T_rows.append(row)
        T = cp.vstack([cp.hstack(row) for row in T_rows])
        
        # PSD constraint
        psd_constraint = [T >> 0]
        
        # Objective
        objective = cp.Minimize(t)
        
        # Construct full problem
        constraints = box_constraints + soc_constraints + psd_constraint
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False, time_limit=budget_s*0.2)
