import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def conv_fft(a, b):
    """Compute linear convolution using FFT with zero padding."""
    n = len(a)
    m = len(b)
    L = n + m - 1
    # Zero-pad to length L
    a_pad = np.pad(a, (0, L - n), mode='constant')
    b_pad = np.pad(b, (0, L - m), mode='constant')
    # FFT
    A = np.fft.fft(a_pad)
    B = np.fft.fft(b_pad)
    # Convolution
    C = A * B
    c = np.fft.ifft(C).real[:L]
    return c

def project_onto_bounded_simplex(x):
    """Projection of x onto {z : 0 ≤ z_i ≤ 1, sum z_i = 1}"""
    # Clip to [0,1] first
    x = np.clip(x, 0, 1)
    s = np.sum(x)
    if np.isclose(s, 1.0):
        return x
    # solve lambda s.t. sum_i clip(x_i + lambda, 0, 1) = 1
    # This is a piecewise linear function of lambda
    # Binary search for the right lambda
    # Let g(lambda) = sum clip(x_i + lambda, 0, 1)
    # We want to solve g(lambda) = 1
    # Derivative is - (number of x_i + lambda = 0) + (number of x_i + lambda = 1)
    # Start with bounds
    low, high = -1.0, 1.0
    for _ in range(100):
        mid = (low + high)/2
        # compute g(mid)
        y = np.clip(x + mid, 0, 1)
        g = np.sum(y)
        if g < 1: 
            low = mid
        else:
            high = mid
        if abs(g - 1) < 1e-12:
            break
    lambda_ = (low + high)/2
    # final result
    x = np.clip(x + lambda_, 0, 1)
    return x

def soft_weights(c, beta):
    """Compute weights for smooth min"""
    c_max = np.max(c)
    exp_c = np.exp(-beta * (c - c_max))
    w = exp_c / np.sum(exp_c)
    return w

def smooth_obj_and_grad(h, beta):
    """Compute smooth objective and its gradient"""
    n = len(h)
    # Compute correlations c_k = sum h_i * h_{i+k}
    c_full = conv_fft(h, h)
    # Extract c_k for k=1..n-1
    c = c_full[1:n]
    # Compute soft weights
    w = soft_weights(c, beta)
    # Gradient of Phi_beta = - sum_k w_k * c_k
    # c_k is convolution: from i, look at h_{i+k} and h_{i-k}
    # g[i] = - sum_k w_k * (h_{i+k} + h_{i-k})
    # Use FFT to compute convolution
    # Compute u = conv(h, np.roll(w, k)) for all k
    # But simpler, we use first reversing w to compute backshift
    w_rev = w[::-1]
    # Note: if w has length n-1 then it corresponds to k = 1..n-1
    # so if we reverse it, then it corresponds to k=1..n-1 left shift
    # e.g., (index i) --> index i-k
    u = conv_fft(h, w)
    v = conv_fft(h, w_rev)
    # Gradient -u -v
    g = -(u + v)
    # Take only the valid part [0:n]
    g = g[:n]
    # Objective
    log_sum = -np.log(np.sum(np.exp(-beta * c))) / beta
    obj = 1.0 + log_sum
    return obj, g

def optimize_single_run(h0, beta0=1.0, beta_max=200.0, alpha0=0.5, gamma=2.0, delta=0.9, max_iter=1000):
    """Single run of projected gradient descent"""
    n = len(h0)
    h = h0.copy()
    beta = beta0
    alpha = alpha0
    dx = 2.0 / n
    
    best_h = h.copy()
    best_c5 = None
    
    for i in range(max_iter):
        # Compute smooth objective and gradient
        obj, g = smooth_obj_and_grad(h, beta)
        
        # Line search with Armijo condition
        tau = alpha
        h_new = h - tau * g
        h_new = project_onto_bounded_simplex(h_new)
        obj_new, _ = smooth_obj_and_grad(h_new, beta)
        
        # Armijo condition check
        # Try different step size (in interval [0.1, 1.0]) 
        #         until acceptance (not exactly strictly decreasing)
        # The fractional decrease is how much we reduced objective
        # compare to the model: obj_new < obj - 0.1 * tau * ||g||^2 
        # Compute ||g||^2
        g_norm2 = np.sum(g ** 2)
        
        while tau > 1e-12:
            h_trial = h - tau * g
            h_trial = project_onto_bounded_simplex(h_trial)
            
            # evaluate exactly for stopping criterion  
            obj_trial, _ = smooth_obj_and_grad(h_trial, beta)
            if obj_trial <= obj - 0.1 * tau * g_norm2:
                break
            tau = tau / 2.0

        if tau < 1e-12:
            tau = 1e-6
        h = h - tau * g

            grad_norm = np.linalg.norm(g)
            if diff < 1e-10 and grad_norm < 1e-9:
                # increase beta
                beta = min(beta * gamma, beta_max)
                alpha = alpha * delta
        old_obj = obj
        
        # Compute exact C5 for comparison
        corr = correlate(h, 1 - h, mode='full')
        exact_obj = np.max(corr) * dx  # This is the exact C5
        if best_c5 is None or exact_obj < best_c5:
            best_c5 = exact_obj

        # Early stopping with 1000 iterations limit, but some slack

            break
            
    return exact_obj, best_h

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx
