import numpy as np
from scipy.optimize import minimize
import time

def conv_fft(a, b):
    """Compute linear convolution using FFT with zero-padding"""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fa = np.fft.fft(a, size)
    fb = np.fft.fft(b, size)
    fconv = fa * fb
    conv = np.fft.ifft(fconv).real
    return conv[:size]

def project_onto_bounded_simplex(x):
    """Project x onto the set {y : sum(y)=n/2, 0≤y≤1}"""
    x = np.clip(x, 0, 1)
    n = len(x)
    target_sum = n / 2
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-12:
        return x
    # Use bisection to find lambda
    lbd, ubd = -1.0, 1.0
    while True:
        if lbd >= ubd:
            break
        mid = (lbd + ubd) / 2
        y = np.clip(x + mid, 0, 1)
        new_sum = np.sum(y)
        if new_sum < target_sum:
            ubd = mid
        else:
            lbd = mid
    # In practice use more direct method
    # = solve for shift lambda such that sum(clip(x + lambda, 0, 1)) = n/2
    def f(lam):
        return np.sum(np.clip(x + lam, 0, 1)) - target_sum
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a *= 2
        b *= 2
    # Bisection
    for _ in range(100):
        mid = (a + b) / 2
        if (b - a) < 1e-15:
            break
        if f(mid) == 0:
            break
        if f(mid) > 0:
            b = mid
        else:
            a = mid
    lam = (a + b) / 2
    x = np.clip(x + lam, 0, 1)
    return x

def soft_weights(c, beta):
    """Compute weights for soft min,
    with numerical stability by subtracting c_max
    """
    c_max = np.max(c)
    exp_vals = np.exp(-beta * (c - c_max))
    Z = np.sum(exp_vals)
    if Z == 0:
        Z = 1e-15
    w = exp_vals / Z
    return w

def smooth_obj_and_grad(h, beta):
    """Compute the smooth surrogate and its gradient"""
    # Compute c = conv(h,h) for shifts 1 to n-1
    n = len(h)
    corr = conv_fft(h, h)
    c = corr[1:n]  # Remove first element and pad to length n-1
    if len(c) != n - 1:
        c = c[:n-1]
    w = soft_weights(c, beta)
    # Compute gradient:
    # \nabla \Phi = -\sum_k w_k \nabla c_k
    # Each \nabla c_k are conv(h,shifted_w) + conv(h, reversed_shifted_w)
    # But only for valid shifts, i.e. no zero padding
    # Use reflected weights
    rev_w = w[::-1]
    with_shift = conv_fft(h, w)
    with_rev_shift = conv_fft(h, rev_w)
    # Use only the non-zero parts
    g = -(with_shift[1:n] + with_rev_shift[:n-1])
    if len(g) != n:
        # Fill up
        g = np.pad(g, (0, n - len(g)), 'constant')
    # Compute Phi
    phi = np.log(np.sum(np.exp(-beta * c))) / beta
    if np.isnan(phi) or np.isinf(phi):
        phi = 0.0
    return -phi, g

def gradient_descent_with_continuation(h0, beta0=1.0, beta_max=200.0, alpha0=0.2,
                                       gamma=2.0, delta=0.9, max_iter=5000):
    """Projected gradient descent with β continuation"""
    h = h0.copy()
    beta = beta0
    alpha = alpha0
    prev_phi = np.inf
    for i in range(max_iter):
        if i % 500 == 0:
            # Check termination
            if abs(prev_phi) < 1e-12:
                break
        # Compute phi and gradient
        phi, g = smooth_obj_and_grad(h, beta)
        # Armijo line search
        tau = alpha
        h_trial = h - tau * g
        h_trial = project_onto_bounded_simplex(h_trial)
        phi_trial, _ = smooth_obj_and_grad(h_trial, beta)
        while (phi_trial > phi - 0.1 * tau * np.dot(g, g) and tau >= 1e-12):
            tau /= 2
            if tau < 1e-12:
                break
            h_trial = h - tau * g
            h_trial = project_onto_bounded_simplex(h_trial)
            phi_trial, _ = smooth_obj_and_grad(h_trial, beta)
        h = h_trial.copy()
        # Update beta if convergence
        if abs(phi - prev_phi) < 1e-10 and tau < 1e-4:
            beta = min(beta * gamma, beta_max)
            if beta >= beta_max:
                break
            alpha *= delta
        prev_phi = phi
    return h

def compute_c5_val(h):
    """Compute the exact C5 constant"""
    n = len(h)
    dx = 2.0 / n
    corr_full = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr_full)
    return max_corr * dx
