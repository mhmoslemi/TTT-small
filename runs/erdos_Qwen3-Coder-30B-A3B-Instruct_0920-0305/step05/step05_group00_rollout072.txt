# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def sub_gradient_step(h, eps=1e-4):
    """Perform gradient step with subgradients to reduce overlap"""
    # Compute correlations with 1-h for each lag
    corr = compute_corr_fft(h)
    # Get the worst lag
    worst_lag_idx = np.argmax(corr)
    
    # Create gradient for this specific lag by looking at how h[i] affects correlation[i]
    n = len(h)
    grad = np.zeros(n)
    
    # Compute the subgradient for worst lag (normalized)
    for i in range(n):
        if i - worst_lag_idx >= 0:
            grad[i] += h[i - worst_lag_idx]
        if i + worst_lag_idx < n:
            grad[i] += h[i + worst_lag_idx]
    
    # Normalize gradient
    grad_norm = np.linalg.norm(grad)
    if grad_norm > 0:
        grad /= grad_norm
    
    # Simple gradient ascent to decrease overlap
    h_new = h - eps * grad
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    return h_new
