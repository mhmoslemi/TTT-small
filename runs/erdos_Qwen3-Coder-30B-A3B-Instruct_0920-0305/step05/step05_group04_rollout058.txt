import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import minimize
import time

def project_onto_bounded_simplex(x):
    """
    Project x onto the set {y : 0 <= y_i <= 1, sum(y) = 1}
    using the Duchi et al. (2008) algorithm with O(n) time complexity.
    """
    # Clip to [0, 1]
    x = np.clip(x, 0, 1)
    # Compute lambda such that sum of clip(x + lambda, 0, 1) == 1
    # This is a univariate root finding problem
    # Use bisection on lambda
    left, right = -1.0, 1.0
    for _ in range(100):
        lambda_val = (left + right) / 2
        y = np.clip(x + lambda_val, 0, 1)
        sum_y = np.sum(y)
        if sum_y < 1.0:
            left = lambda_val
        else:
            right = lambda_val
    # Special case handling in case of floating point issues
    lambda_val = (left + right) / 2
    y = np.clip(x + lambda_val, 0, 1)
    return y

def soft_weights(c, beta):
    """
    Compute the softmax weights w_k = exp(-beta * c_k) / sum(exp(-beta * c_t))
    with numerical stability.
    """ 
    c_max = np.max(c)
    c_shifted = c - c_max
    exp_vals = np.exp(-beta * c_shifted)
    w = exp_vals / np.sum(exp_vals)
    return w

def smooth_obj_and_grad(h, beta):
    """
    Compute smooth surrogate value and its gradient.
    Returns (Phi_beta, gradient)
    """
    # Compute linear convolution of h with itself
    # Length (2n-1) for full convolution, but we only need shifts 1 to n-1
    n = len(h)
    c = fftconvolve(h, h, mode='full')
    c = c[n-1:2*n-2]  # Shift indices to get 1,...,n-1

    # Avoid numerical instability with large negative arguments to exp
    w = soft_weights(c, beta)

    # Compute gradient using linear convolution (part for each side)
    # Forward shift and backward shift weights used in gradient expression
    # In practice we compute (conv(h,w) + conv(h,reverse(w))) = conv(h, w + reverse(w))  
    w_rev = w[::-1]
    grad = -(fftconvolve(h, w + w_rev, mode='full')[n-1:2*n-2])

    # PHI = 1/β * log(sum(exp(-beta * c_k)))  
    # but we computed manually to avoid numerical issues
    # Phi = -log(sum(exp(-beta * c))) / beta
    # Not strictly correct if we care about numerical precision here
    # So we directly compute logsumexp for Phi:  
    # Phi = 1/β * log(sum(exp(-β * c_k)))
    # Use logsumexp for numerical accuracy
    # Let's compute: log(sum_i exp(-beta * c_i))
    # Compute sum(exp(-beta * (c_i - c_max))) + c_max
    c_max = np.max(c)
    log_arg = np.sum(np.exp(-beta * (c - c_max))) 
    Phi_val = (np.log(log_arg) + c_max) / beta  # Negative of what we want?
    # We actually define Phi as in homotopy: log_sum_exp over negative beta* c_k
    # across k = 1 to n-1, so the final value is -min(c_k) which equals -AMP of -c_k
    # Correctly, if we are computing negative soft min (which is increasing in beta)
    # Then with softmax we return negative value of logsumexp.  
    # Let's think of it as a way to minimize -min(c_k), i.e., minimizing  
    # Phi = 1/β * log(sum(exp(-beta * c_k)))  
    # Total value: C5 = 1 - min(c_k) => mapped to 1 - (-Phi) = 1 + Phi 
    
    return -Phi_val, grad 

def gradient_descent_phase(h0, beta0=1.0, beta_max=200, alpha0=0.2, gamma=2.0, delta=0.9, max_iter=5000, eps=1e-9):
    """
    Perform one run of projected gradient descent with β continuation.
    """
    h = h0.copy()
    beta = beta0
    alpha = alpha0
    for i in range(max_iter):
        # compute c from convolution
        n = len(h)
        c = fftconvolve(h, h, mode='full')[n-1:2*n-2]

        # Compute weights and gradient  
        w = soft_weights(c, beta)
        # Gradient: - (conv(h,w) + conv(h, reverse(w)))
        w_rev = w[::-1]
        g = -(fftconvolve(h, w + w_rev, mode='full')[n-1:2*n-2])

        # Armijo line search
        tau = alpha
        while True:
            h_candidate = project_onto_bounded_simplex(h - tau * g)
            # Compute actual Phi_beta for h_candidate
            c_candidate = fftconvolve(h_candidate, h_candidate, mode='full')[n-1:2*n-2]
            c_max = np.max(c_candidate)
            exp_vals = np.exp(-beta * (c_candidate - c_max)) 
            Phi_candidate = (np.log(np.sum(exp_vals)) + c_max) / beta
            Phi_current = (np.log(np.sum(np.exp(-beta * (c - c_max)))) + c_max) / beta

            # Check descent
            if Phi_candidate <= Phi_current - 0.1 * tau * np.linalg.norm(g)**2:
                break
            tau *= 0.5
            if tau < 1e-12:
                break

        h = h_candidate
        # Update beta and alpha
        if np.linalg.norm(tau * g) < eps:
            beta = min(beta * gamma, beta_max)
            alpha *= delta

        # Convergence criterion
        if np.linalg.norm(tau * g) < eps:
            if i >= 1000 and beta > 100:
                break
    return h

def evaluate_C5(h):
    """Evaluate C5 using numpy correlate correct."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
