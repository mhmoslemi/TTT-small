import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use a hybrid optimization approach - start with differential evolution
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Phase 1: Global optimization using differential evolution
    try:
        res = differential_evolution(
            objective,
            bounds,
            seed=seed,
            maxiter=2000,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            atol=1e-10,
            rtol=1e-10
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Phase 2: Local refinement using trust-constr
    local_restart_bound = 2000
    try:
        for restart in range(local_restart_bound):
            if (time.time() - start_time) > budget_s * 0.95:
                break
                
            # Small random perturbation
            new_h = best_h * (1 + np.random.normal(0, 0.005, n_points))
            new_h = np.clip(new_h, 0, 1)
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(new_h) * dx)
            new_h = new_h * scale_factor
            
            res = minimize(
                objective, 
                new_h, 
                method='trust-constr', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'gtol': 1e-12, 'ftol': 1e-12}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
                    
    except:
        pass

    # Phase 3: Pattern-based initial search to prevent local optima
    if (time.time() - start_time) < budget_s * 0.8:
        try:
            # Try different structured patterns
            patterns = []
            
            # Pattern 1: Double peak
            x = np.linspace(0, 2, n_points)
            peak1 = np.exp(-((x - 0.5)**2) / 0.05)
            peak2 = np.exp(-((x - 1.5)**2) / 0.05)
            pattern1 = peak1 + peak2
            patterns.append(pattern1)
            
            # Pattern 2: Triangular
            pattern2 = 1 - np.abs(x - 1)
            patterns.append(pattern2)
            
            # Pattern 3: Step-like
            pattern3 = np.zeros(n_points)
            pattern3[n_points//4:3*n_points//4] = 1
            patterns.append(pattern3)
            
            # Try pattern decoding using scipy
            for pattern in patterns:
                p_norm = pattern / np.sum(pattern) * n_points / 2
                p_norm = np.clip(p_norm, 0, 1)
                dx = 2.0 / n_points
                scale_factor = 1.0 / (np.sum(p_norm) * dx)
                p_norm = p_norm * scale_factor
                c5_candidate = objective(p_norm)
                if c5_candidate < best_c5:
                    best_h = p_norm.copy()
                    best_c5 = c5_candidate
                    
        except:
            pass

    # Phase 4: Gradient-free evolutionary search with adaptation
    if (time.time() - start_time) < budget_s * 0.9:
        try:
            # Adaptable population search
            pop_size = min(50, max(10, n_points))
            population = []
            fitness = []
            
            for i in range(pop_size):
                h = np.random.rand(n_points)
                h = np.clip(h, 0, 1)
                dx = 2.0 / n_points
                scale_factor = 1.0 / (np.sum(h) * dx)
                h = h * scale_factor
                fit = objective(h)
                population.append(h)
                fitness.append(fit)
                
            # Sort and keep best
            sorted_indices = np.argsort(fitness)
            population = [population[i] for i in sorted_indices]
            fitness = [fitness[i] for i in sorted_indices]
            
            # Evolve from best
            num_generations = 500
            mutation_rate = 0.1
            for gen in range(num_generations):
                if (time.time() - start_time) > budget_s * 0.95:
                    break
                    
                new_population = [population[0].copy()]  # Elitism
                
                for i in range(1, pop_size):
                    # Crossover and mutation
                    parent1 = population[np.random.randint(0, pop_size//2)]
                    parent2 = population[np.random.randint(0, pop_size//2)]
                    child = (parent1 + parent2) / 2
                    
                    # Mutation
                    if np.random.rand() < mutation_rate:
                        mutation_idx = np.random.randint(0, n_points)
                        child[mutation_idx] += np.random.normal(0, 0.02)
                        child[mutation_idx] = np.clip(child[mutation_idx], 0, 1)
                    
                    # Renormalize
                    dx = 2.0 / n_points
                    scale_factor = 1.0 / (np.sum(child) * dx)
                    child = child * scale_factor
                    
                    new_population.append(child)
                    fitness.append(objective(child))
                    
                # Update population
                combined = list(zip(new_population, fitness))
                combined.sort(key=lambda x: x[1])
                population = [x[0] for x in combined[:pop_size]]
                fitness = [x[1] for x in combined[:pop_size]]
                
                if fitness[0] < best_c5:
                    best_h = population[0].copy()
                    best_c5 = fitness[0]
                    
        except:
            pass

    return (best_h.tolist(), best_c5, n_points)
