import numpy as np
from scipy.optimize import minimize
import time

def conv_fft(a, b):
    """Compute linear convolution using FFT with zero padding."""
    n = len(a)
    m = len(b)
    L = n + m - 1
    # zero-pad to length L
    a_pad = np.pad(a, (0, L - n), 'constant')
    b_pad = np.pad(b, (0, L - m), 'constant')
    # use FFT
    A = np.fft.fft(a_pad)
    B = np.fft.fft(b_pad)
    C = A * B
    c = np.fft.ifft(C).real[:L]
    return c

def project_onto_bounded_simplex(x):
    """Project x onto {x : sum(x) = n/2, 0 <= x_i <= 1}."""
    # clip to [0,1]
    x = np.clip(x, 0, 1)
    # now adjust sum to be n/2
    n = len(x)
    target_sum = n / 2
    current_sum = np.sum(x)
    
    # find the λ that will make the projection satisfy sum= target_sum
    if current_sum == target_sum:
        return x
    
    # Create a function f(λ) = sum of clipped components
    def f(λ):
        clipped = np.clip(x + λ, 0, 1)
        return np.sum(clipped)
    
    # solve for λ using bisection method
    if current_sum > target_sum:
        # we need to decrease, so λ < 0
        a, b = -1.0, 0.0
        while f(b) < target_sum:
            a, b = b, 2 * b
        # Now f(a) > target_sum, f(b) < target_sum
        tolerance = 1e-12
        while abs(f(b) - target_sum) > tolerance:
            mid = (a + b) / 2
            if f(mid) >= target_sum:
                a = mid
            else:
                b = mid
        λ = (a + b) / 2
    else:
        # we need to increase, so λ > 0
        a, b = 0.0, 1.0
        while f(b) > target_sum:
            a, b = b, 2 * b
        # Now f(a) < target_sum, f(b) > target_sum
        tolerance = 1e-12
        while abs(f(b) - target_sum) > tolerance:
            mid = (a + b) / 2
            if f(mid) <= target_sum:
                a = mid
            else:
                b = mid
        λ = (a + b) / 2

    x_new = np.clip(x + λ, 0, 1)
    return x_new

def soft_weights(c, beta):
    """Compute soft weights from c and beta."""
    c_max = np.max(c)
    exp_c = np.exp(-beta * (c - c_max))
    w = exp_c / np.sum(exp_c)
    return w

def smooth_obj_and_grad(h, beta):
    """Compute phi_beta and its gradient."""
    n = len(h)
    # Compute correlation c_k = sum_{i} h_i h_{i + k}
    c = conv_fft(h, h)[:n-1]  # For lags 1..n-1
    
    # Soft weights
    w = soft_weights(c, beta)
    # Convert w to full vector with zeros for k=0
    w_ext = np.zeros(2*n - 1)
    w_ext[1:n] = w
    
    # Gradient computation using FFT.
    # conv(h, w) + conv(h, reverse(w))
    # Reverse w (but note that w[0] = 0 due to our extension)
    w_rev = w_ext[1::-1]  # reverse with padding
    # Actually better just to do the linear convolution manually on the valid indices
    g1 = conv_fft(h, w)[1:n]  # Only first n-1 elements (k=1 to n-1)
    g2 = conv_fft(h, w_rev[:n])[n-1:0:-1]  # Reflect the origin to obtain non-shifted part
    
    # This approach computes convolution correctly because w_ext_index = [0, w, 0]
    # And we want the part where it shifts 1 step both directions.
    # Simpler and clearer computation:
    # g_sign = np.zeros_like(h)
    # for i in range(n):
    #     for k in range(1, n):
    #         kg = k - 1
    #         if i + k < n:
    #             g_sign[i] += w[kg] * h[i + k]
    #         if i >= k:
    #             g_sign[i] += w[kg] * h[i - k]
    # But since we already computed weight vector w, we use fast methods:
    
    # Convolution-based direct gradient computation based on c_k w.r.t h_i:
    # (g_j)_i = sum_k w_k D_i(c_k) = sum_k w_k (h_{i+k} + h_{i-k})
    #   where h either outside bounds is treated as 0.
    # Method: compute a = conv(h, w) and b = conv(h, reverse(w))
    # Take sum of a and b for the gradient.
    # We need extra zero-padding for correct behavior of the convolution.
    a = conv_fft(h, np.concatenate([w, [0]]))[:n]  # h * w
    b = conv_fft(h, np.concatenate([[0], w[::-1]]))[:n]  # h * reverse(w)
    g = -(a + b)
    # Finally, the value of Φ_β = (1/β) * log(... sum of exp(-β c_k))
    log_sum_exp = np.log(np.sum(np.exp(-beta * c))) + beta * c_max
    obj_val = (1 / beta) * log_sum_exp
    
    return obj_val, g

def line_search_armijo(h, g, beta, obj_func, alpha=0.5, tau=0.5, c=1e-4):
    """Perform backtracking line search."""
    while True:
        h_new = project_onto_bounded_simplex(h - alpha * g)
        val_new, _ = obj_func(h_new, beta)
        val_old, _ = obj_func(h, beta)
        if val_new <= val_old - c * alpha * np.dot(g, g):
            return h_new, alpha
        alpha *= tau
        if alpha < 1e-12:
            return h, alpha

def gradient_descent_step(h, beta, alpha0=0.5, max_iter=1000, tol=1e-9, delta=0.9, beta_max=200, gamma=2.0):
    """Perform one pass of gradient descent with β continuation."""
    h = project_onto_bounded_simplex(h)
    alpha = alpha0
    beta_curr = beta
    
    for i in range(max_iter):
        val, grad = smooth_obj_and_grad(h, beta_curr)
        g_norm = np.linalg.norm(grad)
        if g_norm < tol:
            break
        h_new, step_size = line_search_armijo(h, grad, beta_curr, smooth_obj_and_grad, alpha=alpha)
        if step_size < 1e-12:
            break
        h = h_new
        # Check convergence on small changes in value of smoother surrogate
        if abs(val) < 1e-10:
            break
        # β-continuation
        if beta_curr < beta_max:
            old_beta = beta_curr
            beta_curr = min(old_beta * gamma, beta_max)
            alpha = alpha * delta  # Reduce step-size
        else:
            # Fixed beta regime
            if i > 30:
                break  # Stop early for fixed beta
    return h
