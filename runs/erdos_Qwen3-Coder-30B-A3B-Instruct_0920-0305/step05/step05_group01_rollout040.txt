import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60
DX = 2.0 / 120.0

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    low, high = -2.0, 2.0
    for _ in range(50):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target_sum) < 1e-12:
            return y
        if s > target_sum:
            high = mid
        else:
            low = mid
    y = np.clip(x + low, 0.0, 1.0)
    return y

def evaluate_c5(h):
    """Evaluate C5 using correlation"""
    corr = correlate(h, 1.0 - h, mode='full')
    return corr.max() * DX

def gradient_smoothing(h, alpha):
    """Compute gradient of smooth approximation to max correlation"""
    n = len(h)
    # Compute correlation I_vals = ∫h(x)(1-h(x+k))dx for all lags k
    cov_full = correlate(h, 1.0 - h, mode='full')
    I = cov_full  # unscaled for weight computation
    I_max = I.max()
    
    # Compute weights for softmax
    exp_vals = np.exp(alpha * (I - I_max))
    w = exp_vals / exp_vals.sum()
    # weights should sum to 1, double check
    if abs(w.sum() - 1.0) > 1e-10:
        w /= w.sum()
    
    # Compute gradient as weighted sum of per-lag gradients
    grad = np.zeros(n)
    h_arr = h
    one_minus_h = 1.0 - h_arr
    
    for idx, p_val in enumerate(w):
        lag = idx - (n - 1)  # lag in range [-n+1, n-1]
        if lag >= 0:
            # contribution (1 - h[i+lag]) for valid indices
            # i from [0, n-lag-1] -> 0 to n-1-lag
            if n - lag > 0:
                grad[:n-lag] += p_val * one_minus_h[lag:]
            # edge contribution: add 1 for i where i+lag >= n,
            # this happens when i >= n-lag. Hence indices n-lag to n-1
            if lag > 0:
                grad[n-lag:] += p_val * 1.0
            # subtract h[i-lag] for i=lag to n-1
            if lag > 0:
                grad[lag:] -= p_val * h_arr[:n-lag]
            else:
                # lag == 0: all terms subtract h[i]
                grad -= p_val * h_arr
        else:
            d = -lag  # lag < 0 implies d > 0
            # contribution (1 - h[i-d]) for i >= d
            if n - d > 0:
                grad[d:] += p_val * one_minus_h[:n-d]
            # edge contribution: add 1 for i < d (since i+lag < 0)
            grad[:d] += p_val * 1.0
            # subtract h[i+d] for i < n-d
            grad[:n-d] -= p_val * h_arr[d:]
    
    return grad * DX  # scale by dx

def descent_step(h, eta, alpha):
    """Performs one descent step with line search"""
    grad = gradient_smoothing(h, alpha)
    # Backtracking line search
    for _ in range(10):
        h_new = project_to_sum_box(h - eta * grad, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        if c5_new <= evaluate_c5(h):
            return h_new, c5_new
        eta *= 0.5
    return h, evaluate_c5(h)

def gradient_descent(initial_h, time_limit):
    """Run gradient descent for given time limit"""
    h = initial_h.copy()
    c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    t0 = time.time()
    max_iter = 5000
    alpha_start = 0.5
    alpha_end = 50.0
    eta_start = 0.1
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
            
        progress = min(it / max_iter, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        eta = eta_start * (0.995 ** it)
        
        h, c5 = descent_step(h, eta, alpha)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

    start_time = time.time()
    
    # Initialize with interpolated version of initial_h_values
    h0 = np.array(initial_h_values)
    n_points = 120
    target_sum = n_points / 2.0
    
    # Interpolate to n_points
    x_orig = np.linspace(0, 2, len(h0))
    x_new = np.linspace(0, 2, n_points)
    h_interp = np.interp(x_new, x_orig, h0)
    h0 = project_to_sum_box(h_interp, target_sum)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Multiple restarts
    restarts = 0
    while time.time() - start_time < 0.9 * budget_s:
        restarts += 1
        # Random initialization percentage (increase randomness)
        if restarts % 3 == 0:
            # Perturb previous best slightly
            h_init = best_h + np.random.normal(0, 0.02, n_points)
        elif restarts % 3 == 1:
            # Original interpolated
            h_init = h0.copy()
        else:
            # Fresh random restart
            h_init = np.random.uniform(0, 1, n_points)
            
        h_init = project_to_sum_box(h_init, target_sum)
        
        # Run gradient descent for limited time
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 20:
            break
            
        descent_time = min(remaining_time * 0.3, 150)
        h_trial, c5_trial = gradient_descent(h_init, descent_time)
        
        if c5_trial < best_c5:
            best_h = h_trial
            best_c5 = c5_trial
    
    # Final refinement with SLSQP if time allows
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_c5(h)
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
            
        cons = {'type': 'eq', 'fun': constraint_eq}
        bounds = [(0, 1) for _ in range(len(best_h))]
        
        try:
            res = minimize(
                objective,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_refined = res.x
                h_refined = project_to_sum_box(h_refined, TARGET_SUM)
                c5_refined = evaluate_c5(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
