import numpy as np
import time

def project_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex: 0 ≤ x ≤ 1, sum(x) = target_sum"""
    v = np.asarray(v)
    low, high = 0.0, 1.0
    # Binary search on lambda to find correct projection
    l, u = np.min(v - 1), np.max(v)  # bounds for lambda
    for _ in range(50):
        lam = (l + u) / 2.0
        proj = np.clip(v - lam, low, high)
        sum_proj = np.sum(proj)
        if sum_proj > target_sum:
            l = lam
        else:
            u = lam
    lam = (l + u) / 2.0
    return np.clip(v - lam, low, high)

def compute_O_and_grad(h, alpha):
    """Compute the vector of overlaps O_k and the gradient of log-sum-exp of O"""
    n = len(h)
    dx = 2.0 / n
    # Compute all O_k = dx * sum h[i] * (1 - h[i+k])
    O = np.zeros(2*n - 1)  # full correlation length
    for k in range(2*n - 1):
        shift = k - (n - 1)
        if shift >= 0:
            # h[i] * (1 - h[i+shift]) for i in [0, ..., n-1-shift]
            O[k] = dx * np.sum(h[:-shift] * (1 - h[shift:]))
        else:
            # h[i+shift] * (1 - h[i]) for i in [0, ..., n+shift-1]
            O[k] = dx * np.sum(h[-shift:] * (1 - h[:n+shift]))
    
    # compute weights for softmax
    max_O = np.max(O)
    O_shifted = O - max_O
    exp_O = np.exp(alpha * O_shifted)
    w = exp_O / np.sum(exp_O)
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(2*n - 1):
        shift = k - (n - 1)
        if shift >= 0:
            # update grad[i] += w[k] * (1 - h[i+shift]) - w[k] * h[i-shift]
            grad[:n-shift] += w[k] * (1 - h[shift:])
            grad[shift:] -= w[k] * h[:-shift]  # -w[k] * h[i-shift], i from shift to n-1
        else:
            # shift < 0
            # i from 0 to n+shift-1
            grad[shift:] += w[k] * (1 - h[:n+shift])  # += w[k] * (1 - h[i-shift])
            grad[:n+shift] -= w[k] * h[-shift:]       # -= w[k] * h[i-shift], i from 0 to n+shift-1
    return O, grad

        strategy = restart_count % 5

            # uniform random
            h = np.random.uniform(0, 1, n)
        elif strategy == 1:
            # sine wave
            x = np.linspace(0, 2, n)
            h = 0.5 + 0.4 * np.sin(np.pi * x)
        elif strategy == 2:
            # gaussian peak
            x = np.linspace(0, 2, n)
            mu = 1.0
            sigma = 0.2
            h = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
        elif strategy == 3:
            # triangle
            x = np.linspace(0, 2, n)
            h = np.minimum(x, 2 - x)

                h -= step_sg * grad_sg * dx

                actual_C5 = np.max(corr) * dx

    if best_h is None:
        # fallback to initial if nothing improved
        corr = np.correlate(initial_h, 1 - initial_h, mode='full')
        best_C5 = np.max(corr) * dx
        best_h = initial_h.copy()
    
    return (best_h.tolist(), float(best_C5), n)
