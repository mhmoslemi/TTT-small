import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def greedy_initialization(n):
    """Initialize a binary vector via greedy algorithm"""
    x = np.zeros(n, dtype=int)
    ones_positions = set()
    
    for _ in range(n // 2):
        best_i = -1
        best_min_p = -1
        
        for i in range(n):
            if i in ones_positions:
                continue
                
            proposed_positions = ones_positions | {i}
            # Compute minimum p_k for the new set
            min_p = float('inf')
            for k in range(1, n):
                p_k = 0
                for j in range(n - k):
                    if j in proposed_positions and (j + k) in proposed_positions:
                        p_k += 1
                min_p = min(min_p, p_k)
                
            if min_p > best_min_p:
                best_min_p = min_p
                best_i = i
                
        if best_i >= 0:
            x[best_i] = 1
            ones_positions.add(best_i)
    
    return x

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    N = [36, 40, 44, 48, 52, 56, 60, 64]
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    for n in N:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n
        x_greedy = greedy_initialization(n)
        
        # Convert to continuous for initial internal representation
        h_cont = x_greedy.astype(float)
        h_cont = h_cont * (1.0 / np.sum(h_cont)) * (n / 2.0)  # Normalize 
        
        # Now run MILP to maximize L
        # Variables
        x = cp.Variable(n, boolean=True)
        y = cp.Variable((n, n), boolean=True)
        L = cp.Variable(integer=True)
        
        # Constraints
        constraints = [cp.sum(x) == n // 2]
        
        # Symmetry breaking
        constraints.append(x[0] == 1)
        # Left-heavy constraint (simplified)
        left_sum = cp.sum(x[:n//2])
        right_sum = cp.sum(x[n//2:])
        constraints.append(left_sum >= right_sum)
        
        # Product constraints
        for k in range(1, n):
            for i in range(n - k):
                # y[i,k] is x[i] AND x[i+k]
                constraints.append(y[i,k] <= x[i])
                constraints.append(y[i,k] <= x[i+k])
                constraints.append(y[i,k] >= x[i] + x[i+k] - 1)
                
                # Sum of y[i,k] over i
                constraints.append(cp.sum(y[i,k] for i in range(n - k)) >= L)
        
        # Objective
        objective = cp.Maximize(L)
        
        prob = cp.Problem(objective, constraints)
        
        # Solve with CBC
        try:
            prob.solve(solver=CPLEX, time_limit=min(30, budget_s - (time.time() - start_time)))
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURACY]:
                x_val = x.value
                L_val = L.value
                
                # Extract binary vector
                h_sol = x_val.round().astype(int) * dx / (n/2)

                # Compute final C5
                c5_sol = evaluate_c5(h_sol)
                
                if c5_sol < best_c5:
                    best_c5 = c5_sol
                    best_h = h_sol.copy()
                    best_n = n
            else:
                # Try a simple heuristics as fallback
                h_sol = h_cont.copy()
                c5_sol = evaluate_c5(h_sol)
                
                if c5_sol < best_c5:
                    best_c5 = c5_sol
                    best_h = h_sol.copy()
                    best_n = n
        except:
            # Fallback to heuristics if MILP fails
            h_sol = h_cont.copy()
            c5_sol = evaluate_c5(h_sol)
            
            if c5_sol < best_c5:
                best_c5 = c5_sol
                best_h = h_sol.copy()
                best_n = n
                
    return (best_h.tolist(), best_c5, best_n)
