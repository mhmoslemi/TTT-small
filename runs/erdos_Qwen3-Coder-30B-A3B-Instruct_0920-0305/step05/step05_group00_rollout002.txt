# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation directly"""
    h_rev = h[::-1]
    b = 1 - h
    corr = correlate(h_rev, b, mode="full")
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5"""
    corr = compute_corr(h)
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_search_step(h):
    """Try to improve h by moving mass to reduce maximum correlation"""
    corr = compute_corr(h)
    max_lag = np.argmax(corr)
    
    # We want to reduce the value at lag max_lag
    # Look for pairs that would help decrease this value
    best_improvement = 0
    best_move = None
    
    for i in range(n_points):
        for j in range(n_points):
            if i == j:
                continue
            
            # Check if we can move from i to j
            delta_min = max(-h[j], h[i] - 1)
            delta_max = min(h[i], 1 - h[j])
            
            if delta_min >= delta_max:
                continue
            
            # Try a small positive delta
            delta = (delta_min + delta_max) / 2
            
            # Test if this improves the correlation at max_lag
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            new_corr = compute_corr(h_new)
            new_max = new_corr[max_lag]
            
            improvement = corr[max_lag] - new_max
            if improvement > best_improvement:
                best_improvement = improvement
                best_move = (i, j, delta)
    
    if best_move is not None:
        i, j, delta = best_move
        h[i] -= delta
        h[j] += delta
        return project_to_sum_box(h, TARGET_SUM)
    return h
