import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def project(v, S):
    """Project vector v onto the capped simplex: 0 <= h_i <= 1, sum h_i = S"""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2
        p = np.clip(v + lam, 0.0, 1.0)
        s = p.sum()
        if s > S:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2, 0.0, 1.0)

def compute_O_w_grad(h, alpha, dx):
    """Compute overlaps O_r, weights w, and gradient of smooth objective"""
    n = len(h)
    c1 = correlate(h, np.ones(n), mode='full')
    c2 = correlate(h, h, mode='full')
    O = (c1 - c2) * dx
    max_O = np.max(O)
    w = np.exp(alpha * (O - max_O))
    w = w / w.sum()
    s = np.convolve(w, np.ones(n), mode='valid')
    g2 = correlate(w, h, mode='full')[n-1:2*n-1]
    g3 = correlate(w, h[::-1], mode='full')[n-1:2*n-1][::-1]
    g = dx * (s - g2 - g3)
    return O, w, g

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction and upsample to 84 points
    h0 = np.array(initial_h_values)
    n = 84
    # Upsample by interpolation for higher resolution
    x_orig = np.linspace(0, 2, len(h0))
    x_new = np.linspace(0, 2, n)
    h0 = np.interp(x_new, x_orig, h0)
    # Project to ensure sum == n/2 and bounds [0,1]
    S = n / 2.0
    h0 = project(h0, S)
    
    best_h = h0.copy()
    best_val = evaluate_c5(h0)
    
    # Phase 1: Global search with DE
    if budget_s > 100:
        def de_objective(h):
            # Normalize (might not fit in the bounds)
            h = project(h, S)
            return evaluate_c5(h)
        
        bounds = [(0, 1) for _ in range(n)]
        try:
            res = differential_evolution(
                de_objective,
                bounds,
                maxiter=30,
                popsize=12,
                mutation=0.8,
                recombination=0.7,
                seed=seed
            )
            if res.success:
                h_candidate = project(res.x, S)
                val_candidate = evaluate_c5(h_candidate)
                if val_candidate < best_val:
                    best_h = h_candidate.copy()
                    best_val = val_candidate
        except:
            pass

    # Phase 2: Local refinement with smooth gradient descent
    h = best_h.copy()
    alpha = 5.0
    eta = 0.5 * (2.0 / n)  # scaled step size
    max_iter = 5000
    last_improvement = 0
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > 0.97 * budget_s:
            break
            
        # Compute smooth objective and gradient
        O, w, g = compute_O_w_grad(h, alpha, 2.0/n)
        
        # Line search along descent direction
        improved = False
        for factor in [1.0, 0.5, 0.25]:
            h_trial = project(h - factor * eta * g, S)
            h_trial = (h_trial + h_trial[::-1]) / 2
            h_trial = project(h_trial, S)
            val_trial = evaluate_c5(h_trial)
            if val_trial < best_val:
                h = h_trial
                best_h = h_trial
                best_val = val_trial
                improved = True
                last_improvement = iteration
                break
        
        if not improved:
            eta *= 0.5  # Reduce step size if no improvement
        
        # Anneal alpha
        if iteration % 200 == 0 and alpha < 50.0:
            alpha *= 1.5
        
        # Occasionally perturb
        if iteration % 500 == 0 and not improved:
            h = project(h + 0.02 * np.random.randn(n), S)
            h = (h + h[::-1]) / 2
            h = project(h, S)
    
    return (best_h.tolist(), best_val, n)
