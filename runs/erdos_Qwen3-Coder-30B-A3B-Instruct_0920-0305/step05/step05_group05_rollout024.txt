def project_to_simplex(v, target_sum):
    """Project vector v onto the capped simplex: 0 <= v_i <= 1, sum v_i = target_sum"""
    # Binary search on the Lagrange multiplier
    v = np.asarray(v)
    low, high = -1000, 1000
    tol = 1e-12
    max_iter = 30
    for _ in range(max_iter):
        lam = (low + high) / 2
        w = np.clip(v - lam, 0, 1)
        s = np.sum(w)
        if s > target_sum:
            low = lam
        else:
            high = lam
        if abs(s - target_sum) < tol:
            return w
    return np.clip(v - (low + high)/2, 0, 1)

def evaluate_c5(h, dx=1.0):
    """Compute C5 = max_k int h(x)(1-h(x+k))dx"""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def interpolate_control_points(c, K, n):
    """Interpolate control values c to n points using linear interpolation"""
    x_knots = np.linspace(0, 2, K)
    x_full = np.linspace(0, 2, n)
    h_full = np.interp(x_full, x_knots, c)
    return h_full

def soft_max_overlaps(O, alpha):
    """Compute soft-max of all overlaps"""
    # Subtract max for numerical stability
    max_O = np.max(O)
    if max_O == -np.inf:
        return 0
    exp_O = np.exp(alpha * (O - max_O))
    return (max_O + np.log(np.sum(exp_O)) / alpha)

def repair_candidate(c, K, n, target_sum):
    """Repair candidate control vector to maintain feasibility"""
    c = np.clip(c, 0, 1)
    h_full = interpolate_control_points(c, K, n)
    h_full = project_to_simplex(h_full, target_sum)
    return h_full

        k_max = np.argmax(O)
        max_val = O[k_max]
        
        # Simple sub-gradient: gradient of shift k_max
        # gradient at index i of O_k_max is (1-h_{i+k_max} - h_{i-k_max})
        g = np.zeros(n)
        if k_max < n:
            g[:n-k_max] = 1 - h[k_max:] - h[:n-k_max]
        else:
            # Handle right shifts where h indices wrap
            for i in range(n):
                i_plus_k = (i + k_max - (n-1)) % n  
                i_minus_k = (i - (k_max - (n-1))) % n
                if i_plus_k < n:
                    g[i] += 1 - h[i_plus_k]
                if i_minus_k < n:
                    g[i] -= h[i_minus_k]
        
        # Update
        h = h - eta * g
        
        # Project
        h = project_to_simplex(h, n/2)
        
        # Early stopping
        if t > 100 and np.max(correlate(h, 1-h, mode="full")) * dx < 0.3808:
            break
        eta *= 0.9995  # Decreasing stepsize
    return h
