import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Phase 1: Use global optimization with restarts
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Run differential evolution with different seeds to escape local minima
    for i in range(3):
        if (time.time() - start_time) > budget_s * 0.8:
            break
        try:
            de_bounds = [(0, 1) for _ in range(n_points)]
            result = differential_evolution(
                objective,
                de_bounds,
                maxiter=100 + i * 50,
                popsize=15,
                seed=seed + i,
                mutation=(0.5, 1),
                recombination=0.7,
                disp=False
            )
            if result.success:
                h_candidate = result.x
                dx = 2.0 / n_points
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue
    
    # Phase 2: Local refinement with multiple starting points
    local_restart_bound = 200
    for restart in range(local_restart_bound):
        if (time.time() - start_time) > budget_s * 0.95:  # Leave time for final steps
            break
            
        # Try different initializations
        if restart % 3 == 0:
            # Perturb current best
            new_h = best_h + np.random.normal(0, 0.02, n_points)
        elif restart % 3 == 1:
            # Try a specific pattern
            x = np.linspace(0, 2, n_points)
            new_h = 0.5 + 0.3 * np.sin(2 * np.pi * x) 
        else:
            # Random small perturbation with bounds
            new_h = np.clip(best_h + np.random.normal(0, 0.01, n_points), 0, 1)
            
        # Normalize
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue
            
    # Phase 3: Additional local search using different methods
    if (time.time() - start_time) < budget_s * 0.9:
        # Try L-BFGS-B with different starting points
        try:
            # Perturb best solution
            x0 = best_h + np.random.normal(0, 0.005, n_points)
            x0 = np.clip(x0, 0, 1)
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(x0) * dx)
            x0 = x0 * scale_factor
            
            # L-BFGS-B is often effective for this type of problem
            res = minimize(
                objective, 
                x0, 
                method='L-BFGS-B', 
                bounds=bounds,
                options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-15}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
            
    return (best_h.tolist(), best_c5, n_points)
