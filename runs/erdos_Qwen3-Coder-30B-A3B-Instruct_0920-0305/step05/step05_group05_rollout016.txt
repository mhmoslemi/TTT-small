import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    L = 12      # Truncation order for Fourier series
    M = 4*L     # Number of sampling points for box constraints
    n_points = 64  # Final discretization points for evaluation
    
    # Create sampling points
    x_sample = np.linspace(0, 2, M, endpoint=False)
    
    # Build the Fourier basis matrices
    cos_basis = np.cos(np.pi * np.arange(1, L+1).reshape(1, -1) * x_sample.reshape(-1, 1))
    sin_basis = np.sin(np.pi * np.arange(1, L+1).reshape(1, -1) * x_sample.reshape(-1, 1))
    
    # CVXPY variables
    u = cp.Variable(2*L)  # [a_1, ..., a_L, b_1, ..., b_L]
    s = cp.Variable(L)    # [s_1, ..., s_L] where s_l = a_l^2 + b_l^2
    t = cp.Variable()     # upper bound on max overlap
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = []
    
    # Box constraints
    h_sample = 0.5 + cos_basis @ u[:L] + sin_basis @ u[L:]
    constraints.append(cp.constraints.Bounded(h_sample, 0, 1))
    
    # SOC constraints: s_l >= a_l^2 + b_l^2
    for i in range(L):
        a = u[i]
        b = u[L+i]
        s_i = s[i]
        # ||(2a, 2b, s_i-1)||_2 <= s_i+1
        constraints.append(cp.norm(cp.hstack([2*a, 2*b, s_i-1]), 2) <= s_i+1)
    
    # Toeplitz PSD constraint
    # Matrix T(t, s) = [t -s_1 -s_2 ... -s_L;
    #                   -s_1  t -s_1 ... -s_{L-1};
    #                    ...  ...  ... ...  ...;
    #                   -s_L -s_{L-1} ... t]
    T_rows = []
    for i in range(L+1):
        row = [t] + [-s[j] for j in range(i)]
        row += [0]*(L - i)
        T_rows.append(row)
    
    for i in range(L+1):
        for j in range(i+1, L+1):
            T_rows[i][j] = T_rows[j][i]
    
    T_mat = cp.bmat([[T_rows[i][j] for j in range(L+1)] for i in range(L+1)])
    constraints.append(T_mat >> 0)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=10000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
