import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import math

def legendre_pattern(n):
    """Generate a Legendre-based binary pattern if possible"""
    p = n // 2
    if p < 2:
        return None
    # Try to find a prime close to p
    candidate_p = p
    while True:
        is_prime = True
        if candidate_p < 2:
            is_prime = False
        else:
            for i in range(2, int(math.sqrt(candidate_p)) + 1):
                if candidate_p % i == 0:
                    is_prime = False
                    break
        if is_prime:
            p = candidate_p
            break
        candidate_p -= 1
        if candidate_p < 2:
            return None
    
    # Generate quadratic residues
    residues = set()
    for a in range(1, p):
        r = (a * a) % p
        if r != 0:
            residues.add(r)
    
    # Create binary pattern
    b = np.zeros(p, dtype=np.int8)
    for i in range(p):
        if i in residues:
            b[i] = 1
    
    # For the pattern of size n, repeat it
    h = np.tile(b, 2)
    if np.sum(h) == p - 1:
        # Find a zero to flip to one
        idx = np.where(h == 0)[0][0]
        h[idx] = 1
    elif np.sum(h) == p + 1:
        # Find a one to flip to zero
        idx = np.where(h == 1)[0][0]
        h[idx] = 0
    return h

def generate_random_balanced(n):
    """Generate random balanced binary vector"""
    indices = np.arange(n)
    np.random.shuffle(indices)
    h = np.zeros(n, dtype=np.int8)
    h[indices[:n//2]] = 1
    return h

def compute_correlation(h):
    """Compute full correlation array"""
    n = len(h)
    corr_full = correlate(h, h, mode="full")
    # Extract the relevant part (from -n+1 to n-1)
    start_idx = n - 1
    corr = corr_full[start_idx:start_idx + 2*n - 1]
    return corr

def evaluate_c5(h, dx=2.0):
    """Evaluate C5 for a given step function h"""
    n = len(h)
    if n == 0:
        return float('inf')
    corr = compute_correlation(h)
    # Compute overlap integral for all shifts
    # Since we want min_k R(k) where R(k) = \int h(x) h(x+k) dx
    # C5 = 1 - min_k R(k) * dx
    # In the discrete setting, we consider both h and 1-h
    corr_h_minus_hk = []
    for i in range(len(h)):
        if i == 0:
            h_minus_hk = h
        else:
            # h(x+k) = h[i:] for k=i (when indexing from 0)
            h_minus_hk = 1 - np.array(h[i:] + [0]*i)
            # Mirror it properly for all n points: 
            # we're simulating integral over [0,2] with dx = 2/n points
            h_minus_hk = np.pad(h_minus_hk, (0, n - len(h_minus_hk)), mode='constant', constant_values=0)
        corr_h_minus_hk.append(np.sum(h * h_minus_hk))
    
    # The overlap integral for each k as a correlation with 1-h
    corr_with_1_minus_h = []
    for i in range(n):
        if i == 0:
            h_reversed = 1 - h
        else:
            h_reversed = np.append(1-h[i:], [0]*i)
            h_reversed = np.pad(h_reversed, (0, n - len(h_reversed)), mode='constant', constant_values=0)
        corr_with_1_minus_h.append(np.sum(h * h_reversed))
    
    # Take maximum
    max_corr = np.max(corr_with_1_minus_h) * dx
    return max_corr

def compute_c5_from_corr(corr, dx):
    """Compute C5 from correlation array"""
    n = len(corr)
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_c5_fast(h, dx=2.0):
    """Faster numerical greenlight for the objective"""
    n = len(h)
    if n == 0:
        return float('inf')
    corr = compute_correlation(h)
    max_corr = np.max(corr) 
    return max_corr * dx

def evaluate_continuous_c5(h_float, dx=2.0):
    """Compute C5 for continuous version"""
    n = len(h_float)
    # Compute from the full correlation
    corr = correlate(h_float, 1 - h_float, mode="full")
    return np.max(corr) * dx

def get_worst_shifts(corr, tol=1e-10):
    """Get indices where max is achieved"""
    max_val = np.max(corr)
    return np.where(np.abs(corr - max_val) < tol)[0]

def init_corr_from_h(h):
    """Initialize correlation from binary x"""
    n = len(h)
    corr_full = correlate(h, h, mode="full")
    start_idx = n - 1
    return corr_full[start_idx:start_idx + 2*n - 1]

def bad_ones(h, shift):
    """Get set of indices where h[i]=1 and h[i+shift]=0"""
    n = len(h)
    bad = set()
    if shift >= 0:
        for i in range(n - shift):
            if h[i] == 1 and h[i + shift] == 0:
                bad.add(i)
    else:
        shift = -shift
        for i in range(shift, n):
            if h[i] == 1 and h[i - shift] == 0:
                bad.add(i)
    return bad

def apply_move(h, i, j, corr):
    """Apply a swap and update correlation"""
    # Swap h[i] and h[j]
    old_i, old_j = h[i], h[j]
    h[i], h[j] = h[j], h[i]
    
    n = len(h)
    # Update corr incrementally
    corr =corr.copy() # defend before mutating
    # We need to update all correlations that include indices i or j
    for m in range(-n+1, n):
        if m == 0:
            # Real part
            if i < n - 1:
                corr[m+n-1] -= (old_i - old_i * old_i)
                corr[m+n-1] += (h[i] - h[i] * h[i])
            if j < n - 1:
                corr[m+n-1] -= (old_j - old_j * old_j)
                corr[m+n-1] += (h[j] - h[j] * h[j])
        else:
            # m > 0
            if m >= 0:
                if i <= n - m - 1:
                    old = old_i * old_i if m == 0 else old_i * old_i + old_i * old_i
                    new = h[i] * h[i+m] if m == 0 else h[i] * h[i+m] + h[i] * h[i+m]
                    corr[m+n-1]  -= old
                    corr[m+n-1]  += new
                if i >= m:
                    old = old_i * old_i if m == 0 else old_i * old_i + old_i * old_i
                    new = h[i] * h[i-m] if m == 0 else h[i] * h[i-m] + h[i] * h[i-m]
                    corr[m+n-1] -= old
                    corr[m+n-1] += new
            else:  # m < 0
                m = -m
                if i <= n - m - 1:
                    old = old_i * old_i if m == 0 else old_i * old_i + old_i * old_i
                    new = h[i] * h[i+m] if m == 0 else h[i] * h[i+m] + h[i] * h[i+m]
                    corr[m+n-1] -= old
                    corr[m+n-1] += new
                if i >= m:
                    old = old_i * old_i if m == 0 else old_i * old_i + old_i * old_i
                    new = h[i] * h[i-m] if m == 0 else h[i] * h[i-m] + h[i] * h[i-m]
                    corr[m+n-1] -= old
                    corr[m+n-1] += new

    return corr

def generate_new_h(h):
    """Generate a new binary vector by swapping a 1 and a 0"""
    n = len(h)
    ones_pos = np.argwhere(h == 1).flatten()
    zeros_pos = np.argwhere(h == 0).flatten()
    if len(ones_pos) == 0 or len(zeros_pos) == 0:
        return h.copy()
    i = np.random.choice(ones_pos)
    j = np.random.choice(zeros_pos)
    new_h = h.copy()
    new_h[i], new_h[j] = new_h[j], new_h[i]
    return new_h

            # If the time is almost up, break early

                break
                
            init_h = h_init.copy()
            # Initialize full correlation
            dx = 2.0 / n
            corr = init_corr_from_h(init_h)
            max_corr = np.max(corr)
            current_c5 = max_corr * dx
            
            # Tracking
            best_current = current_c5
            best_current_vec = init_h.copy()
            iter_count = 0
            iter_max = int(2e6)
            T0 = 0.05
            Tf = 1e-5
            restart_threshold = 50000
            
            temperature = T0
            
            while iter_count < iter_max and time_remaining() > 0.5:
                # Sample worst shifts, possibly multiple times
                worst_shift_indices = get_worst_shifts(corr, tol=1e-9)
                if len(worst_shift_indices) == 0:
                    break
                k_index = np.random.choice(worst_shift_indices)
                # Convert to integer
                k = k_index - (n - 1)
                
                # Get BadOnes set
                ones_pos = np.argwhere(init_h == 1).flatten()
                bad_ones_set = set()
                if k > 0:
                    for i in ones_pos:
                        if i + k < n and init_h[i+k] == 0:
                            bad_ones_set.add(i)
                else:
                    k = abs(k)
                    for i in ones_pos:
                        if i >= k and init_h[i-k] == 0:
                            bad_ones_set.add(i)
                
                if len(bad_ones_set) == 0:
                    iter_count += 1
                    continue
                i = np.random.choice(list(bad_ones_set))

                j_candidates = np.setdiff1d(np.arange(n), list(bad_ones_set), assume_unique=True)

                temp_ratio = iter_count / iter_max
                temperature = T0 * (Tf / T0) ** temp_ratio

                    if start_idx + 1 < len(all_starts):
                        init_h = all_starts[start_idx + 1].copy()
                    else:
                        init_h = generate_random_balanced(n)
                    corr = init_corr_from_h(init_h)

                    best_current = current_c5

        if best_c5 < 0.3807:
            break
            
    # Final polishing of best binary solution with continuous optimization
    if time_remaining() > 0.1: 
        try:
            num_points = len(best_h)
            dx = 2.0 / num_points
            # Convert best_h to continuous form
            h_float = best_h.astype(float)
            # Ensure the sum is n/2
            h_float *= num_points / (2 * np.sum(h_float))
            # Optimize with L-BFGS-B
            bounds = [(0.0, 1.0)] * num_points
            def obj(x):
                corr = correlate(x, 1-x, mode="full")
                return np.max(corr) * dx
            # Define equality constraint
            def constraint(x):
                return np.sum(x) - num_points / 2
            cons = {'type': 'eq', 'fun': constraint}
            # Warm-start
            x_init = h_float.copy()
            # Minimize using SLSQP
            res = minimize(obj, x_init, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 300})
            # Ensure sum is exactly right
            x_final = res.x.copy()
            x_final = x_final * (num_points / (2 * np.sum(x_final)))
            # Constrain to [0,1]
            x_final = np.clip(x_final, 0, 1)
            refined_c5 = evaluate_continuous_c5(x_final, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = x_final
                
        except Exception as e:
            pass
    
    return (best_h.tolist(), best_c5, len(best_h))
