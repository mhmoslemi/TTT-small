import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time
import math

def project_simplex(v, sum_value):
    """
    Projects vector v onto the capped simplex: sum(v) = sum_value, 0 <= v <= 1.
    Uses binary search on Lagrange multiplier.
    """
    v = np.asarray(v)
    sum_v = np.sum(v)
    if sum_v == sum_value:
        return v.copy()
    
    # Binary search for the correct lambda
    tol = 1e-12
    low, high = -100, 100
    max_iter = 30
    
    for _ in range(max_iter):
        mid = (low + high) / 2
        projected = np.clip(v - mid, 0, 1)
        sum_projected = np.sum(projected)
        
        if abs(sum_projected - sum_value) < tol:
            return projected
        elif sum_projected > sum_value:
            low = mid
        else:
            high = mid
    
    # Fallback
    projected = np.clip(v - (low + high)/2, 0, 1)
    return projected

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def interpolate_control_points(c, x_knots, x_grid):
    """Linearly interpolate control points to full resolution grid"""
    n_grid = len(x_grid)
    h = np.interp(x_grid, x_knots, c)
    return np.clip(h, 0, 1)

def repair_h(c, K, n):
    """Apply repair steps to control points c to obtain feasible h"""
    x_knots = np.linspace(0, 2, K)
    x_grid = np.linspace(0, 2, n)
    h_full = interpolate_control_points(c, x_knots, x_grid)
    h_projected = project_simplex(h_full, n / 2)
    return h_projected

def objective_soft_max(c, K, n, alpha):
    """Objective function with soft-max smoothing"""
    h = repair_h(c, K, n)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    O = corr * dx
    # Numerically stable soft-max
    max_O = np.max(O)
    if max_O == 0:
        return 0.0
    exp_O = np.exp(alpha * (O - max_O))
    soft_max = (1 / alpha) * (np.log(np.sum(exp_O)) + max_O)
    return soft_max

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 128  # Full grid points
    K = 12   # Control points
    T = n / 2  # Target sum
    dx = 2.0 / n
    x_knots = np.linspace(0, 2, K)
    x_grid = np.linspace(0, 2, n)
    
    # Initial random control points that are feasible
    best_c = np.random.uniform(0, 1, K)
    best_c = project_simplex(best_c, T / K)  # Project to correct sum
    best_h = repair_h(best_c, K, n)
    best_C5 = evaluate_c5(best_h)
    
    # Try several restarts with slightly different starting points
    restarts = 5
    for restart_idx in range(restarts):
        if (time.time() - start_time) > budget_s * 0.9:
            break
            
        # Start from a plausible structure
        if restart_idx == 0:
            # Sine wave-like shape
            c_start = 0.5 + 0.4 * np.sin(np.pi * np.linspace(0, 2, K))
        elif restart_idx == 1:
            # Triangular shape centered
            c_start = np.clip(1 - np.abs(np.linspace(0, 2, K) - 1), 0, 1)
        elif restart_idx == 2:
            # Step function
            c_start = np.zeros(K)
            mid = K // 2
            c_start[max(0, mid - K//4):min(K, mid + K//4)] = 1
        else:  # restart_idx == 3, 4
            # Random with systematic symmetry
            c_start = np.random.uniform(0, 1, K)
        
        # Normalize to sum constraint (can be off due to interpolation)
        c_start = project_simplex(c_start, T / K)
        
        # Simulated Annealing-like approach with soft-max annealing
        current_c = c_start.copy()
        current_h = repair_h(current_c, K, n)
        current_C5 = evaluate_c5(current_h)
        
        # Soft max temperature schedule
        alpha_start = 5.0
        alpha_end = 30.0
        max_steps = 1000
        alpha = alpha_start
        step = (alpha_end - alpha_start) / (max_steps * 0.7)
        
        for t in range(max_steps):
            if (time.time() - start_time) > budget_s * 0.95:
                break
                
            alpha += step
            if alpha > alpha_end:
                alpha = alpha_end
            
            # Create new candidate by small mutation
            new_c = current_c.copy() + np.random.normal(0, 0.02, K)
            # Keep within [0, 1] (controls before soft-max)
            new_c = np.clip(new_c, 0, 1)
            
            # Evaluate with soft max
            cost_new = objective_soft_max(new_c, K, n, alpha)
            
            if cost_new < current_C5:
                current_c = new_c
                current_h = repair_h(current_c, K, n)
                current_C5 = evaluate_c5(current_h)
                
                if current_C5 < best_C5:
                    best_c = current_c.copy()
                    best_h = current_h.copy()
                    best_C5 = current_C5
                    
        # More aggressive local refinement (gradient).
        # Gradient ascend on max expected shift.
        local_h = best_h.copy()
        best_C5 = evaluate_c5(local_h)
        
        # First find the worst shift
        corr = correlate(local_h, 1 - local_h, mode="full")
        max_shift_idx = np.argmax(corr)
        max_corr = np.max(corr)
        shift = max_shift_idx - (n - 1)
        
        # Update using gradient
        for _ in range(100):
            if (time.time() - start_time) > budget_s * 0.95:
                break
            
            # Estimate subgradient from exact correlation maximum
            g = np.zeros(n)
            # Generate the active set
            if shift >= 0:
                g[shift:] = 1 - local_h[:-shift] - local_h[shift:]
            else:
                g[:shift] = 1 - local_h[-shift:] - local_h[:shift]
            
            # Simplified update with fixed step size
            eta = 0.02
            new_h = local_h - eta * g
            new_h = np.clip(new_h, 0, 1)
            new_h = project_simplex(new_h, T)  # Project to correct sum
            new_C5 = evaluate_c5(new_h)
            
            # Accept the better solution
            if new_C5 < best_C5:
                best_C5 = new_C5
                local_h = new_h.copy()
            else:
                break
                
    return (local_h.tolist(), best_C5, n)
