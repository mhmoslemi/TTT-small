```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Discretization parameters
    n_points = 44  # even number to ensure sum constraint is integer
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize the best solution
    best_h = None
    best_c5 = float('inf')
    
    # Initial active set (compact support of the initial construction)
    K = [0, -1, 1]
    
    # Define variables for this problem
    h = cp.Variable(n_points)
    tau = cp.Variable()
    
    # Keep track of y_* variables for all active shifts
    y_vars = {}
    
    # Start with the initial construction
    h0 = np.array(initial_h_values)
    original_n = len(h0)
    if original_n != n_points:
        # Interpolate to the target size
        x_orig = np.linspace(0, 2, original_n)
        x_new = np.linspace(0, 2, n_points)
        h0 = np.interp(x_new, x_orig, h0)
    
    # Normalize the sum to target_sum
    scale = target_sum / np.sum(h0)
    h0 = h0 * scale
    
    # Build initial model
    while True:
        if (time.time() - start_time) > budget_s * 0.9:
            break

        # Create the constraints list
        constraints = []

        # Box constraints
        constraints += [h >= 0, h <= 1]

        # Sum constraint with symmetry (this is the key, helps improve bounds)
        constraints += [cp.sum(h) == target_sum]

        # Symmetry constraint: h[i] = h[n-1-i]
        for i in range(n_points // 2):
            constraints += [h[i] == h[n_points - 1 - i]]

        # Add McCormick constraints for active shifts
        # Build two mappings: one for positive shifts, one for negative shifts
        K_sorted = sorted(set(K), reverse=True)  # potentially ordered

        new_y_vars = {}
        
        # Update y_vars dictionary - add missing keys
        for k in K_sorted:
            if k not in y_vars:
                y_vars[k] = cp.Variable(n_points - abs(k) if k != 0 else n_points)
                new_y_vars[k] = y_vars[k]

        # Add McCormick constraints for each active k
        for k in K:
            if k == 0:
                # No cross terms if k=0, this is the simple case
                continue
            # Replace h[i]*h[i+k] with y_{i,k}
            n_k = n_points - abs(k)
            if k > 0:
                # Forward
                y = y_vars[k]
                constraints += [y >= 0]
                constraints += [y <= h[:-k]]
                constraints += [y <= h[k:]]
                constraints += [y >= h[:-k] + h[k:] - 1]
            else:
                # Backward
                y = y_vars[k]
                constraints += [y >= 0]
                constraints += [y <= h[-k:]]
                constraints += [y <= h[:-k]]
                constraints += [y >= h[-k:] + h[:-k] - 1]
                
            # Add the constraint tau >= sum(h) - sum(y)
            # Note: When k = 0, there is no multiplication term;  
            # 	   when k != 0, we use fewer terms to account for boundary. 
            if k == 0:
                constraints += [tau >= cp.sum(h)]
            else:
                constraints += [tau >= cp.sum(h) - cp.sum(y_vars[k])]

        # Objective: minimize tau
        prob = cp.Problem(cp.Minimize(tau), constraints)

        # Try multiple solvers for robustness
        solvers = [cp.GLOP, cp.CBC, cp.GLPK]
        try:
            # Try GLOP first (efficient for LP)
            prob.solve(solver=cp.GLOP, verbose=False, max_iters=1000)
        except Exception:
            try:
                prob.solve(solver=cp.CBC, verbose=False, max_iters=2000)
            except Exception:
                try:
                    prob.solve(solver=cp.GLPK, verbose=False, max_iters=2000)
                except Exception:
                    break  # Failed to solve this iteration; exit and return best

        # Check solution status
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_val = h.value
            tau_val = tau.value

            if h_val is None:
                break

            # Evaluate the true C5 for this h (with actual correlations)
            h_candidate = np.array(h_val)
            # Symmetrize to make sure
            h_sym = (h_candidate + h_candidate[::-1]) / 2.0
            if np.sum(h_sym) > 0:
                scaled_h = h_sym / np.sum(h_sym) * target_sum
            else:
                scaled_h = h_sym
            c5_eval = evaluate_c5(scaled_h)

            # Update best
            if c5_eval < best_c5:
                best_h = scaled_h.copy()
                best_c5 = c5_eval

            # Compute actual overlaps to identify worst case
            corr = correlate(scaled_h, 1 - scaled_h, mode="full")
            max_k = np.argmax(corr)
            max_o = corr[max_k]

            # If we're within tolerance, we can terminate early
            # But use some slack for final polishing handoff
            if max_o <= tau_val + 1e-6:
                # Possible we're done; continue to next phase
                pass
            else:
                # Add the maximum shift as new constraint
                actual_k = max_k - (len(corr) - 1) // 2
                if actual_k not in K:
                    K.append(actual_k)
        else:
            # Fail to solve, fall through to next iteration
            break

    if best_h is None:
        # Fall back to h0 if no better had been found
        best_h = h0
        best_c5 = evaluate_c5(best_h)

    # Polishing phase with projected subgradient
    # Use the best-found h as starting point
    h_polish = best_h.copy()
    eta = 0.5 * dx
    for _ in range(2000):
        if (time.time() - start_time) > budget_s * 0.95:
            break

        # Compute all O_k(h_polish) 
        O_vals = []
        for k in range(-n_points + 1, n_points):
            result = np.correlate(h_polish, 1 - h_polish, mode='full')
            # idx corresponds to shift k, get result at the offset
            offset = k + n_points - 1  # shift positive/negative k to match full output
            if 0 <= offset < len(result):
                O_vals.append(result[offset])
            else:
                O_vals.append(0)

        # Find maximal shifts
        k_active = np.argmax(O_vals)
        max_val = O_vals[k_active]

        # Gradient of O_k* = 1 - (h_{i} h_{i+k}) for shift k*
        # Evaluate ∂O_k/∂h_i = 1 - (h_{i+k} + h_{i-k}) 
        g = np.zeros_like(h_polish)
        # This is for the k-th shift; going to use the result directly in full
        g_full = 0
        if 0 <= k_active + n_points - 1 < len(correlate(h_polish, np.ones_like(h_polish), mode="full")):
            # This is the direct computation in the frequency domain via convolution
            # Or just compute it directly from the correlation
            # Similar to the full list used in the correlator on the full list
            # but we construct a shifted correlation that can be done via convolution rules.
            # For huge performance, skip to getting a sparse eval
            
            # K seems to be between -n and n but normalise that to full full length
            # We can reindex. Clarify what this is...
            # More practical approach:
            # directly compute O_k = sum h_i - sum h_i h_{i+k}
            # partial derivative: d/dh_i O_k = 1 - ( h_{i+k} + h_{i-k} )
            # needed under the assumption that h_{i-k} = h_{i-k} (can be out of bound)
            # by convention, treat out-of-bounds as zero
            
            # Simple Numpy version:
            shift_k = k_active

            # Compute dO_k/dh_i = 1 - (h_{i+k} + h_{i-k})
            num = len(h_polish)
            # loop for now
            g = np.zeros_like(h_polish)
            for i in range(num):
                h_i_plus_k = h_polish[(i + shift_k) % num] if (i + shift_k) >= 0 and (i + shift_k) < num else 0
                h_i_minus_k = h_polish[(i - shift_k) % num] if (i - shift_k) >= 0 and (i - shift_k) < num else 0
                g[i] = 1 - (h_i_plus_k + h_i_minus_k)

        # Gradient of max_k O_k (some components): exploit active set
        # For now, simple average across all maximizing indices. 
        # Actually, estimate the overall subgradient – if we are close to a multiple,
        # we can be smarter.
        if np.allclose(h_polish, h_polish, rtol=1e-10):
            g = np.ones_like(h_polish) / n_points
        else:
            g = 1 - (correlate(h_polish, np.ones_like(h_polish), mode='full')[k_active + n_points - 1] if k_active - (n_points - 1) >= 0 else 0)

        # Take gradient step
        h_new = h_polish - eta * g
        
        # Project to [0,1] and sum==target_sum
        # Water filling step to handle sum constraint
        def project_to_simplex(v, target_sum):
            # This is water filling:
            # Given v, find lam such that sum(clamp(v + lam, 0, 1)) == target_sum
            v = np.copy(v)
            v = np.clip(v, 0, 1)
            sum_v = np.sum(v)
            if abs(sum_v - target_sum) < 1e-8:
                return v
            # A simplified approach: scale the values back
            if sum_v > 0:
                scale_factor = target_sum / sum_v
                v = v * scale_factor
            # Clip again to ensure bounds
            v = np.clip(v, 0, 1)
            return v
        
        # Apply symmetrize and project
        h_new = 0.5 * (h_new + h_new[::-1])  # symmetrize
        h_new = project_to_simplex(h_new, target_sum)
        
        h_polish = h_new
        
        # Evaluate C5
        c5_current = evaluate_c5(h_polish)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_polish.copy()

        # Adaptive step size
        if eta > 1e-8:
            eta *= 0.99
        else:
            break

    # Optional binary refinement
    if (time.time() - start_time) < budget_s * 0.99:
        # Sample binary mask: choose top half of the best_h
        h_bin = best_h.copy()
        sorted_idx = np.argsort(h_bin)[::-1]
        h_mask = np.zeros_like(h_bin)
        for i in range(int(target_sum)):
            h_mask[sorted_idx[i]] = 1

        # Local hill climbing on swaps
        iter_count = 0
        while True:
            improved = False
            iter_count += 1
            if iter_count > 100:
                break
            for i in range(n_points):
                if h_mask[i] == 1:
                    for j in range(n_points):
                        if h_mask[j] == 0:
                            # Simulate exchange
                            old_c5 = evaluate_c5(h_mask)
                            h_mask[i] = 0
                            h_mask[j] = 1
                            new_c5 = evaluate_c5(h_mask)
                            if new_c5 < old_c5:
                                improved = True
                            else:
                                # Undo reversal
                                h_mask[i] = 1
                                h_mask[j] = 0
            if not improved:
                break

        if np.sum(h_mask) != target_sum:
            # Fix any double-counting or missing entries via selection by rank-val
            sorted_idx = np.argsort(h_mask)[::-1]
            h_mask = np.zeros_like(h_mask)
            for i in range(int(target_sum)):
                h_mask[sorted_idx[i]] = 1

        c5_binary = evaluate_c5(h_mask)
        if c5_binary < best_c5:
            best_c5 = c5_binary
            best_h = h_mask

    return (best_h.tolist(), best_c5, n_points)
```
