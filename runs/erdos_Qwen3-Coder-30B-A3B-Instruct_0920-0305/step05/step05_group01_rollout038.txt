def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def random_pair_move(h):
    """Generate a pairwise mass transfer move, ensuring feasibility"""
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)

    delta_min = max(-h[j], h[i] - 1.0)
    if delta_min >= delta_max:
        return i, j, 0.0
    delta = np.random.uniform(delta_min, delta_max)
    return i, j, delta

def guided_move(h, corr_full):
    """Guided deterministic move for the worst lag to minimize the correlation"""
    n = len(h)
    # Find the worst lag
    worst_lag = np.argmax(corr_full)
    
    # Create gradient-like score for each position
    score = np.zeros(n)
    for i in range(n):
        if i - worst_lag >= 0:
            score[i] += h[i - worst_lag]
        if i + worst_lag < n:
            score[i] += h[i + worst_lag]
    
    # Find the best indices to move mass from and to
    i_min = np.argmin(score)
    i_max = np.argmax(score)
    
    if i_min == i_max:
        return None
    
    # Move a small amount from i_min to i_max
    delta_max = min(h[i_min], 1.0 - h[i_max])
    if delta_max <= 0:
        return None
    
    delta = min(delta_max, 0.005)  # Smaller step for fine tuning
    return i_min, i_max, delta

def local_refine(h):
    """Refine solution using SLSQP optimizer"""
    def objective(h):
        return evaluate_c5(h)
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h))]
    
    try:
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
    except:
        pass
    return h, evaluate_c5(h)
