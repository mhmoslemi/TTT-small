import numpy as np
import time

def project_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex: 0 <= x <= 1, sum(x) = target_sum"""
    v = np.asarray(v)
    low, high = 0.0, 1.0
    v_min, v_max = v.min(), v.max()
    
    # Binary search for lambda
    def sum_with_lambda(l):
        return np.clip(v - l, low, high).sum()
    
    # Bracket the root
    l_low = v_min - high
    l_high = v_max - low
    
    while sum_with_lambda(l_low) < target_sum:
        l_low *= 2
    while sum_with_lambda(l_high) > target_sum:
        l_high *= 2
    
    # Bisection
    for _ in range(40):
        l = (l_low + l_high) / 2
        s = sum_with_lambda(l)
        if s > target_sum:
            l_low = l
        else:
            l_high = l
    
    lambda_opt = (l_low + l_high) / 2
    return np.clip(v - lambda_opt, low, high)

def compute_O_and_grad(h, alpha):
    """Compute overlap values O_k and smoothed gradient"""
    n = len(h)
    dx = 2.0 / n
    O = np.zeros(2 * n - 1)
    
    # Compute all O_k
    for k in range(2 * n - 1):
        shift = k - (n - 1)
        if shift <= 0:
            h1 = h[:n + shift]
            h2 = h[-shift:] if -shift < n else np.zeros(-shift)
        else:
            h1 = h[shift:]
            h2 = h[:n - shift] if n - shift > 0 else np.zeros(n - shift)
        O[k] = np.dot(h1, 1 - h2)
    
    O *= dx
    
    # Gradient computation
    grad = np.zeros(n)
    max_O = O.max()
    exp_O = np.exp(alpha * (O - max_O))
    exp_O /= exp_O.sum()
    
    for k in range(2 * n - 1):
        shift = k - (n - 1)
        w = exp_O[k]
        if shift <= 0:
            h1 = np.pad(h[:n + shift], (0, -shift), constant_values=0)
            h2 = np.pad(h[-shift:], (n - shift, 0), constant_values=0)
        else:
            h1 = np.pad(h[shift:], (0, shift), constant_values=0)
            h2 = np.pad(h[:n - shift], (shift, 0), constant_values=0)
        grad += w * (1 - h1 - h2)
    
    grad *= dx
    return O, grad
