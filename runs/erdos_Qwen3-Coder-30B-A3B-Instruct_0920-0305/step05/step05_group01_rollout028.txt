    """Evaluate C5 using correlation"""
    n = len(h)

    # Compute full correlation
    corr = correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project to sum = target_sum and box constraints [0,1]"""
    # Binary search for lambda
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    
    y = np.clip(x + low, 0, 1)
    return y
