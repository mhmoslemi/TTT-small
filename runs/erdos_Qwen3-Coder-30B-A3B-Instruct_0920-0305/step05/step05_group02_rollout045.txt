import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp

def initial_h_values(n_points=42):
    # Create a piecewise linear function that's a spike at the center with small support
    h = np.zeros(n_points)
    x = np.linspace(0, 2, n_points)
    # Normalize to total integral = 1
    h[(x >= 0.3) & (x <= 0.7)] = 1.0
    h[(x > 0.7) & (x <= 1.3)] = 1.0 - (x[(x > 0.7) & (x <= 1.3)] - 0.7) / 0.6
    h[(x >= 1.3) & (x < 1.7)] = 1.0 - (1.7 - x[(x >= 1.3) & (x < 1.7)]) / 0.4
    # Normalize so integral = 1
    dx = 2.0 / n_points
    h = h * (1.0 / np.sum(h) / dx)
    return h.tolist()

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - np.array(h), mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different discretizations
    possible_n = [36, 40, 44, 48, 52, 56, 60, 64]
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Greedy initialization to obtain a high-quality start
    def greedy_construction(n_points):
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Find good initial distribution
        dx = 2.0 / n_points
        # Start with local high-value regions
        h[10:20] = 1.0  # Unfilled gaps will be filled via optimization
        # Normalize
        h = h * (1.0 / np.sum(h) / dx)
        return h

    # Test each size with MILP optimization
    n_optimal = None
    for n in possible_n:
        if time.time() - start_time > budget_s * 0.9:
            break
            
        dx = 2.0 / n
        try:
            # Create a better initial h for this n
            h_initial = np.array(initial_h_values(n))
            # Integrate to get the current integral
            integral_current = np.sum(h_initial) * dx
            # Adjust to get exactly 1.0
            h_initial = h_initial * (1.0 / integral_current)
            
            # Try a modified continuous heuristic before MILP
            # This is to ensure a high-quality warm-start
            h_current = h_initial.copy()
            # Evaluate current C5
            c5_current = evaluate_c5(h_current)
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                best_n = n

        except Exception as e:
            print(f"Error with n={n}: {e}")
            continue

    # Also use a final polishing step to perhaps refine the best solution
    if best_h is not None:
        try:
            # Use gradient-based optimization to refine
            # Set up optimization problem for continuous h
            def objective(h):
                return evaluate_c5(h)
                
            # Project to [0,1] and renormalize
            def project_and_normalize(h):
                h = np.clip(h, 0, 1)
                dx = 2.0 / best_n
                h = h * (1.0 / np.sum(h) / dx)
                return h
            
            # Use BFGS to refine the best solution
            x0 = best_h.copy()
            # Maximize min-overlap by minimizing 1-C5
            res = minimize(
                lambda x: -objective(x),
                x0,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(best_n)],
                options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10}
            )
            if res.success:
                refined_h = res.x.copy()
                # Re-normalize after optimization
                dx = 2.0 / best_n
                refined_h = refined_h * (1.0 / np.sum(refined_h) / dx)
                refined_c5 = evaluate_c5(refined_h)
                if refined_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
        except:
            pass

    if best_h is None:
        # Fall back to a simple solution
        h_fallback = np.array(initial_h_values(42))
        dx = 2.0 / 42
        h_fallback = h_fallback * (1.0 / np.sum(h_fallback) / dx)
        best_c5 = evaluate_c5(h_fallback)
        best_h = h_fallback
        best_n = 42

    return (best_h.tolist(), best_c5, best_n)
