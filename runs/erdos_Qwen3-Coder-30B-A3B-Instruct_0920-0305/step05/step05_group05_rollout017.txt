import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

    a = cp.Variable(L)
    b = cp.Variable(L)

    objective = cp.Minimize(t)
    
    # Constraints
    constraints = []
    
    # Box constraints: 0 <= h(x) <= 1 for sampled points
    # h(x) = 0.5 + sum(a_l * cos(pi*l*x) + b_l * sin(pi*l*x))
    h_values = 0.5 + cos_basis.T @ a + sin_basis.T @ b
    constraints.extend([h_values >= 0])
    constraints.extend([h_values <= 1])
    
    # s_l >= a_l^2 + b_l^2
    for i in range(L):
        constraints.append(cp.SOC(s[i] + 1, [2*a[i], 2*b[i], s[i] - 1]))

    # Matrix T(t,s) = t*I - S where S is Toeplitz with entries -s
    S_rows = []
    for i in range(L):
        row = [0]*L
        for j in range(L):
            if abs(i-j) == 0:
                row[j] = t
            else:
                row[j] = -s[abs(i-j)-1]
        S_rows.append(row)
    
    S_matrix = np.array(S_rows)
    constraints.append(cp.PSD(S_matrix))

    prob = cp.Problem(objective, constraints)
