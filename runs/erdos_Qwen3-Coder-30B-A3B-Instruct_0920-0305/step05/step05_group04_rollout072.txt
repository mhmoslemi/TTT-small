import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def project_simplex(h, target_sum):
    """Project h onto the simplex: sum(h) = target_sum, 0 <= h_i <= 1"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    
    if abs(h_sum - target_sum) < 1e-12:
        return h
    
    # Use Duchi et al. 2008 algorithm for projection
    # Find lambda such that sum(clip(h + lambda, 0, 1)) = target_sum
    # We'll compute it via bisection
    l, u = -1.0, 1.0
    for _ in range(50):  # iterations are enough
        mid = (l + u) / 2.0
        h_trial = np.clip(h + mid, 0, 1)
        s_trial = np.sum(h_trial)
        if s_trial > target_sum:
            u = mid
        else:
            l = mid
    
    # apply final adjustment
    final_lambda = (l + u) / 2.0
    h_final = np.clip(h + final_lambda, 0, 1)
    # correct sum
    scale_factor = target_sum / np.sum(h_final)
    return h_final * scale_factor

def compute_correlations(h):
    """Compute all correlations"""
    n = len(h)
    # zero-padded linear convolution
    c_full = correlate(h, h, mode='full')
    # extract the k=1 to k=n-1 part
    c = c_full[n-1:n-1+n-1]
    return c

def compute_softmin_gradient(h, c, beta):
    """Compute gradient of soft-min = (1/beta)*log(sum exp(-beta*c_k))"""
    exp_neg_beta_c = np.exp(-beta * c)
    Z = np.sum(exp_neg_beta_c)
    if Z == 0:
        # Avoid division by zero
        w = np.zeros_like(c)
    else:
        w = exp_neg_beta_c / Z
    
    # Gradient w.r.t h: for each h_i, we compute variable contribution
    # This can be computed via convolution (forward + backward)
    # First we use convolution to get the derivative of each correlation term w.r.t. h_i
    # A correlation c_k at index i is h_i * h_{i+k} + h_{i-k} * h_i 
    # So the gradient of c_k w.r.t h_i is h_{i+k} + h_{i-k} (handling boundary)
    # After all weighted sum, we get our final grad as:
    # - sum_k w_k (h_{i-k} + h_{i+k})
    # vector convolution way:
    # conv(h, w_rev) for backward and conv(h, w) for forward
    # but need careful handling of zero padding.
    # Use NumPy's fft-based fftconvolve for Fast computation
    
    # Recompute c and use more robust approach for numerical stability
    # Just make sure to be careful about boundary handling and weighting
    # Let's repeat using different implementation
    
    n = len(h)
    w = np.concatenate([w, np.zeros(n)])  # Pad
    h_pad = np.concatenate([h, np.zeros(n)])
    
    # Forward convolution: h * w (right-aligned)
    # But we really want weighted sum over shifted h
    # For gradient: ∇c_k = h * (δ_{k} + δ_{-k}) in signal sense, weight by w_k
    
    # Use FFT-based method from Signal Processing library:
    # ∇c_k = Re[h_{-k} + h_{k}] for shift k.
    # Gradient of the weighted sum is:
    # - sum_k w_k (h_{i-k} + h_{i+k})
    # Let's compute this directly using NumPy convolutions
    # w_shifted = reverse(w)
    w_rev = np.flip(w)
    try:
        backward = np.convolve(h_pad, w_rev, mode='valid')[:n]
        forward  = np.convolve(h_pad, w, mode='valid')[:n]
        grad = -(backward + forward)
    except Exception:
        # fallback subtractive approach
        grad = np.zeros_like(h)
        for i in range(n):
            for k in range(1, n):
                if i + k < n:
                    grad[i] -= w[k-1] * h[i + k]
                if i - k >= 0:
                    grad[i] -= w[k-1] * h[i - k]

    return grad

def objective_and_gradient(h, beta, n, dx):
    """Objective C5 via softmin approximation"""
    # Get correlations (shifts 1 to n-1)
    c = compute_correlations(h)
    # Soft minimum
    c_max = np.max(c)
    exp_terms = np.exp(-beta * (c - c_max))
    Z = np.sum(exp_terms)
    if Z == 0:
        soft_min = c_max
    else:
        soft_min = c_max - np.log(Z) / beta
    approx_c5 = 1 + soft_min
    
    # Compute gradient
    if beta < 1e10:  # if not too large, compute gradient
        try:
            grad = compute_softmin_gradient(h, c, beta)
        except Exception:
            grad = np.gradient(approx_c5, h)  # fallback
    else:
        grad = np.zeros_like(h)
    return approx_c5, grad
