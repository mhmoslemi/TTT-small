def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def random_pair_move(h):
    """Generate a pairwise mass transfer move, ensuring feasibility"""
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    delta_max = min(h[i], 1.0 - h[j])
    delta_min = max(-h[j], h[i] - 1.0)
    if delta_min >= delta_max:
        return i, j, 0.0
    delta = np.random.uniform(delta_min, delta_max)
    return i, j, delta

def guided_move(h, corr_full):
    """Guided deterministic move for the worst lag to minimize the correlation"""
    n = len(h)
    worst_lag = np.argmax(corr_full)
    
    # Compute subgradient g_i = h_{i-\ell} + h_{i+\ell}
    g = np.zeros(n)
    for i in range(n):
        left_idx = i - worst_lag
        right_idx = i + worst_lag
        if left_idx >= 0:
            g[i] += h[left_idx]
        if right_idx < n:
            g[i] += h[right_idx]
    
    # Find indices with min and max g
    i_min = np.argmin(g)
    i_max = np.argmax(g)
    
    if i_min == i_max:
        return None  # No improvement possible
    
    # Move a small amount from i_min to i_max
    delta_max = min(h[i_min], 1.0 - h[i_max])
    if delta_max <= 0:
        return None
    
    delta = min(delta_max, 0.01)  # Small fixed step
    return i_min, i_max, delta

def gen_h_from_boxes(positions, heights, n_points=120):
    """Generate h from box positions and heights."""
    h = np.zeros(n_points)
    x = np.linspace(0, 2, n_points)
    for i in range(len(positions) - 1):
        start = positions[i]
        end = positions[i+1]
        height = heights[i]
        mask = (x >= start) & (x < end)
        h[mask] = height
    # Handle last box
    mask = x >= positions[-1]
    h[mask] = heights[-1]
    return h
