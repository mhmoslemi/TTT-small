import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.fft import rfft, irfft
import time

def project_onto_simplex(h, target_sum):
    """Project vector h onto capped simplex: 0 <= h[i] <= 1, sum h = target_sum"""
    # Handle edge case
    if np.isclose(np.sum(h), target_sum):
        return h.copy()
    # Binary search for the Lagrange multiplier
    max_iter = 30
    l, u = -10.0, 10.0
    
    for _ in range(max_iter):
        lambda_val = (l + u) / 2
        z = np.clip(h - lambda_val, 0, 1)
        s = np.sum(z)
        if s > target_sum:
            l = lambda_val
        else:
            u = lambda_val
    # Final projection
    lambda_val = (l + u) / 2
    z = np.clip(h - lambda_val, 0, 1)
    # Scale to match target sum
    scale = target_sum / np.sum(z)
    return z * scale

def evaluate_c5_fft(h):
    """Fast evaluation of C5 using FFT, on a full resolution vector h"""
    n = len(h)
    dx = 2.0 / n
    # Use rfft for efficiency (h is real)
    H = rfft(h)
    G = rfft(1 - h)
    corr = irfft(H * np.conj(G))
    corr = corr[:n]  # Take only the first n values
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_c5(h):
    """Evaluate C5 using correlation (slower but more accurate for debugging)"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def eval_soft_max(h, alpha):
    """Evaluate the log-sum-exp (soft max) of correlation values"""
    n = len(h)
    dx = 2.0 / n
    if n > 1000:
        corr = correlate(h, 1 - h, mode="full")
    else:
        H = rfft(h)
        G = rfft(1 - h)
        corr = irfft(H * np.conj(G))
        corr = corr[:n]
    o_vals = dx * corr
    if len(o_vals) == 0:
        return 0.0
    max_o = np.max(o_vals)
    if np.isinf(max_o) or np.isnan(max_o):
        return np.inf
    exp_vals = np.exp(alpha * (o_vals - max_o))
    sum_exp = np.sum(exp_vals)
    if sum_exp <= 0:
        return max_o
    return max_o + np.log(sum_exp) / alpha

def repair_h(cand, knots, n_full):
    """Repair candidate vector to satisfy [0,1] and sum = n/2 constraints"""
    # Interpolate to full resolution
    x_knots = np.linspace(0, 2, len(knots))
    x_full = np.linspace(0, 2, n_full)
    h_full = np.interp(x_full, x_knots, cand)
    # Clip to [0, 1]
    h_full = np.clip(h_full, 0, 1)
    # Project onto simplex
    target_sum = n_full / 2
    h_full = project_onto_simplex(h_full, target_sum)
    return h_full

def optimize_with_cmaes(knots, n_full, alpha_schedule, budget_s):
    best_h = None
    best_c5 = np.inf
    start_time = time.time()
    
    # CMA-ES parameters
    lambda_ = int(4 + 3 * np.log(len(knots)))
    mu = lambda_ // 2
    sigma = 0.2
    mean = np.ones(len(knots)) * 0.5
    C = np.eye(len(knots))
    D = np.eye(len(knots))  # Diagonal of C
    cov = np.eye(len(knots))
    chi_n = np.sqrt(len(knots)) * (1 - 1/(4*len(knots)) + 1/(21*len(knots)**2))

    for gen in range(int(1e4)):  # safe upper limit
        if (time.time() - start_time) > budget_s * 0.95:
            break
        # Generate offspring
        offspring = []
        for i in range(lambda_):
            z = np.random.randn(len(knots))
            x = mean + sigma * (D @ z)
            offspring.append(x)
        # Evaluate
        offspring_evals = []
        for cand in offspring:
            h_full = repair_h(cand, cand, n_full)
            c5 = evaluate_c5_fft(h_full)
            offspring_evals.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_full.copy()
        # Select best mu
        indices = np.argsort(offspring_evals)[:mu]
        best_offspring = [offspring[i] for i in indices]
        # Update parameters
        mu_eff = mu / 2
        cs = (4 + mu_eff / len(knots)) / (len(knots) + 4 + 2*mu_eff)
        cc = 4 / (len(knots) + 4)
        ccov1 = 2 / (np.power(len(knots) + 1.3, 2) + mu_eff)
        ccov2 = 2 / (np.power(len(knots) + 1.3, 2) + mu_eff)
        weights = np.log(mu + 1/2) - np.log(np.arange(1, mu + 1))
        weights = weights / np.sum(weights)
        # Update mean
        z_w = np.zeros(len(knots))
        for i in range(mu):
            z_w += weights[i] * (best_offspring[i] - mean)
        mean = mean + cs * z_w
        # Update covariance matrix (simplified rank-1 update)
        pc = (1 - cc) * pc + np.sqrt(cc * (2 - cc) * mu_eff) * z_w
        C = (1 - ccov1 - ccov2) * C + ccov1 * np.outer(pc, pc) + ccov2 * np.outer(z_w, z_w)
        # Update sigma
        sigma = sigma * np.exp(cs / chi_n * np.linalg.norm(z_w) / np.sqrt(len(knots)))
        # Anneal alpha
        if gen < len(alpha_schedule):
            alpha = alpha_schedule[gen]
        else:
            alpha = alpha_schedule[-1]
    return best_h, best_c5
