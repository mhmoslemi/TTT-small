# Helper function to generate greedy warm-start
def greedy_h(n_points):
    """Generate feasible binary solution via greedy construction."""
    x = np.zeros(n_points, dtype=int)
    selected = set()
    for _ in range(n_points // 2):
        best_i = -1
        best_min_p = -1
        # Check for each unused i
        for i in range(n_points):
            if i in selected:
                continue
            # Compute penalty if we add i
            new_p = np.zeros(n_points - 1, dtype=int)
            for j in selected:
                k = abs(i - j)
                if k > 0:
                    new_p[k-1] += 1
            current_min_p = min(new_p) if len(new_p) > 0 else float('inf')
            if current_min_p > best_min_p:
                best_min_p = current_min_p
                best_i = i
        if best_i >= 0:
            selected.add(best_i)
            x[best_i] = 1
    return x

def solve_milp(n_points, timeout=30):
    """
    Solve exact MILP for binary pattern of length n_points.
    Return best integer solution x, minimal correlation count L, gap.
    """
    import cvxpy as cp
    import numpy as np
    from cvxpy import Minimize, Problem, Variable, Binary, Sum, index
    import time

    start_time = time.time()
    # Declare integral variables
    x = Variable(n_points, boolean=True)
    y = Variable((n_points, n_points - 1), boolean=True)  # y[i,k] for 0 <= i <= n-k-1, 1 <= k <= n-1
    L = Variable(integer=True)

    # The constraints and objective
    # sum of x = n/2
    constraints = [cp.sum(x) == n_points / 2]

    # bind y to x:
    for k in range(1, n_points):
        for i in range(n_points - k):
            # y[i,k] <= x[i]
            constraints.append(y[i, k-1] <= x[i])
            # y[i,k] <= x[i+k]
            constraints.append(y[i, k-1] <= x[i + k])
            # y[i,k] >= x[i] + x[i+k] - 1
            constraints.append(y[i, k-1] >= x[i] + x[i + k] - 1)

    # enforce min p_k >= L for k=1,...,n-1
    for k in range(1, n_points):
        p_k = cp.sum(y[:, k-1])
        constraints.append(p_k >= L)

    # objective: maximize L
    objective = cp.Maximize(L)

    # symmetry breaking:
    constraints.append(x[0] == 1)
    sum_left = cp.sum(x[:n_points//2])
    sum_right = cp.sum(x[n_points//2:])
    constraints.append(sum_left >= sum_right)

    prob = cp.Problem(objective, constraints)
    
    # Solve with CBC solver (or use SCIP if available)
    try:
        # Set time limit
        prob.solve(solver=cp.CBC, verbose=False, maximumSeconds=timeout)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURACY]:
            x_sol = [int(v.value >= 0.5) for v in x]
            L_val = int(L.value)
            # Compute actual C5 from binary solution
            dx = 2.0 / n_points
            h = np.array(x_sol)
            corr = correlate(h, 1 - h, mode="full")
            max_corr = np.max(corr)
            C5 = max_corr * dx
            return x_sol, C5, L_val
        else:
            return None, float('inf'), -1
    except Exception as e:
        return None, float('inf'), -1


def evaluate_c5(h):
    """Evaluate C5 for given h."""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx
