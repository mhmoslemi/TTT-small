import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
n_points_fine = 240
dx_fine = 2.0 / n_points_fine

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_objective(h, β):
    """Computes the smooth approximation to max as log-sum-exp"""
    corr = compute_corr_fft(h)
    corr_scaled = corr * dx  # scale by dx
    # subtract max to avoid numerical overflow
    max_corr = np.max(corr_scaled)
    exp_terms = np.exp(β * (corr_scaled - max_corr))
    log_sum_exp = max_corr + np.log(np.sum(exp_terms)) / β
    return log_sum_exp

def smooth_max_gradient(h, β):
    """Computes the gradient of smooth max"""
    corr = compute_corr_fft(h)
    corr_scaled = corr * dx
    max_corr = np.max(corr_scaled)
    exp_terms = np.exp(β * (corr_scaled - max_corr))
    weights = exp_terms / np.sum(exp_terms)
    
    # compute gradient of each C_ℓ
    grad_sum = np.zeros_like(h)
    for i in range(len(h)):
        # we are computing ∇C_ℓ(h) = 1 - h_{i+ℓ} - h_{i-ℓ}
        # but we do it weights × ∇C_ℓ for all ℓ
        # this becomes total sum over shifts in both directions
        
        # Only compute this for non-zero weights
        # We will compute it efficiently by shifting the array
        shift_plus = np.roll(h, -1)  # forward shift
        shift_minus = np.roll(h, 1)  # backward shift
        
        grad_term = 1 - shift_plus - shift_minus
        grad_term[0] = 1 - shift_plus[0]  # handle boundary
        grad_term[-1] = 1 - shift_minus[-1]
        
        grad_sum += weights[i] * grad_term
    
    return grad_sum

def smooth_max_gradient_direct(h, β, corr_vals=None):
    """Computes the gradient more directly, useful during optimization"""
    if corr_vals is None:
        corr = compute_corr_fft(h)
        corr_scaled = corr * dx
        max_corr = np.max(corr_scaled)
        exp_terms = np.exp(β * (corr_scaled - max_corr))
        weights = exp_terms / np.sum(exp_terms)
    else:
        max_corr = np.max(corr_vals)
        exp_terms = np.exp(β * (corr_vals - max_corr))
        weights = exp_terms / np.sum(exp_terms)
    
    # we compute ∇C_ℓ = 1 - shift_plus - shift_minus
    # we aggregate as Σ weights_ℓ ∇C_ℓ(h)
    L = len(h)
    grad_comp = np.zeros(L)
    
    # At each position i, we get 1 contribution
    for i in range(L):
        # Simplification: we'll do vectorized computation for shifts
        # and apply all contributions with weights
        grad_comp[i] += weights[0]  # first one

    # More accurately compute via vectorized shifts
    # Construct each gradient part and weight it
    grad_total = np.zeros(L)
    
    # See: for each shift ℓ, we compute ∇C_ℓ = 1 - h_{i+ℓ} - h_{i-ℓ}  
    # Let's do this with fft and inverse fft
    shift_plus = np.zeros(L)
    shift_minus = np.zeros(L)
    
    # So effectively, ∇C_ℓ = ∇ (h_i*(1-h_{i+ℓ})) = 1 - h_{i+ℓ} - h_{i-ℓ}
    
    # Building each shift manually:
    grad_total = np.ones(L)
    shift_plus = np.roll(h, -1)
    shift_minus = np.roll(h, 1)
    # handle boundaries explicitly:
    shift_plus[-1] = 0
    shift_minus[0] = 0
    
    # Now, for each y such that y = ∇C_ℓ for some ℓ, we compute weighted sum
    # But we should compute all ∇C_ℓ as shift based gradients, then average
    
    # We know from ft that ∇C_ℓ computes properly via inverse fft of multipliers
    total_grad = np.zeros(L)
    for l in range(len(weights)):  # l runs in [0, L-1]
        if l == 0:
            grad_l = np.ones(L)
        else:
            # Apply a phase vector to get the shift effect
            # wij represents that when we compute correlation of h with 1-h
            # h_{i+ℓ} part is shift by ℓ forward, we compute it using shift matrix
            # We treat nan or out-of-bound this way
            
            # So ∇C_ℓ at x at each i is:
            # 1 for i ∈ [0,N-1] - h_{i+ℓ} - h_{i-ℓ} 
            # allow for l = 0 meaning no shift
            # Make it generalize for now:
            shift_plus = np.zeros(L)
            shift_plus[l:] = h[:-l]
            shift_plus[:l] = 0
            shift_minus = np.zeros(L)
            shift_minus[:-l] = h[l:]
            shift_minus[-l:] = 0
            
            grad_l = 1 - shift_plus - shift_minus
            
        total_grad += weights[l] * grad_l

    return total_grad

def smooth_max_gradient_efficient(h, β, corr_vals=None):
    """An optimized version of gradient computation using shifts"""
    corr = compute_corr_fft(h)
    corr_scaled = corr * dx
    max_corr = np.max(corr_scaled)
    exp_terms = np.exp(β * (corr_scaled - max_corr))
    weights = exp_terms / np.sum(exp_terms)

    # At each position compute linear term + shift contributions
    L = len(h)
    grad = np.ones(L)
    
    # We compute the vectorized shift operation of the h with opposite sign
    # gradient for C_l = sum over all i: 1 - h_{i+l} - h_{i-l}
    # So for each shift ℓ, we get contributions from h shift
    grad_total = np.zeros(L) 
    
    # For each weight, compute the ∇C_ℓ
    for ℓ in range(L):
        if ℓ == 0:
            # ∇C_0 = 1 - h - h = 1 - 2*h
            grad_all_ℓ = 1 - 2 * h
        else:
            shift_p = np.roll(h, -ℓ)  # shifted forward: h_{i+ℓ}
            shift_m = np.roll(h, ℓ)   # shifted backward: h_{i-ℓ}
            # zero pad: out of range should be zero
            shift_p[-ℓ:] = 0
            shift_m[:ℓ] = 0
            grad_all_ℓ = 1 - shift_p - shift_m
            # also must zero pad in reverse directions
            grad_all_ℓ[:ℓ] = 0
            grad_all_ℓ[-ℓ:] = 0
            
        grad_total += weights[ℓ] * grad_all_ℓ
    return grad_total

def gradient_step(h, step_size, gradient):
    """Apply one gradient step and project"""
    h_new = h - step_size * gradient
    return project_to_sum_box(h_new, TARGET_SUM)

def smooth_max_optimize(h_in, β_schedule, step_schedule, max_iter):
    """Optimize using the smooth max"""
    h = h_in.copy()
    best_h = h.copy()
    best_val = smooth_max_objective(h, β_schedule[-1])  # at final β
    
    for iter_count in range(max_iter):
        β = β_schedule[min(len(β_schedule)-1, iter_count // 50)]
        
        if β < 1e-10:
            # Use a direct gradient
            grad = smooth_max_gradient_efficient(h, β)
        else:
            grad = smooth_max_gradient_efficient(h, β)
            
        step_size = step_schedule[iter_count]
        
        new_h = gradient_step(h, step_size, grad)
        new_val = smooth_max_objective(new_h, β)
        
        if new_val < best_val:
            best_val = new_val
            best_h = new_h.copy()
            
        h = new_h.copy()
        
    return best_h, best_val
