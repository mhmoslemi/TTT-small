import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def evaluate_c5(h):
    """Evaluate C5 using FFT correlation"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using scipy.signal.correlate for full mode
    corr = correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project x to sum = target_sum with box constraints [0,1]"""
    # Use binary search to find proper shift
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    
    y = np.clip(x + low, 0, 1)
    return y

def perturb_solution(h, strength=0.05):
    """Generate a perturbed solution"""
    h_pert = h + np.random.normal(0, strength, len(h))
    return project_to_sum_box(h_pert, TARGET_SUM)

def adaptive_local_search(h, max_iter=1000, step_size=0.01):
    """Adaptive local search using gradient estimates"""
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Try random perturbations
        h_trial = perturb_solution(best_h, 0.02)
        c5_trial = evaluate_c5(h_trial)
        
        if c5_trial < best_c5:
            best_h = h_trial
            best_c5 = c5_trial
            
        # Occasionally try guided move
        if i % 50 == 0 and np.random.random() < 0.3:
            # Try to find an improvement by shifting mass
            idx = np.random.randint(len(h))
            move_amount = np.random.uniform(-step_size, step_size)
            h_trial = best_h.copy()
            h_trial[idx] = np.clip(h_trial[idx] + move_amount, 0, 1)
            h_trial = project_to_sum_box(h_trial, TARGET_SUM)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial
                
    return best_h, best_c5
