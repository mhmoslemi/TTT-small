import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60
EPS = 1e-12

def project_to_sum_box(x, target_sum):
    """Project x onto box [0,1] and sum = target_sum using binary search"""
    low, high = -2.0, 2.0
    for _ in range(50):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target_sum) < EPS:
            return y
        if s > target_sum:
            high = mid
        else:
            low = mid
    y = np.clip(x + low, 0.0, 1.0)
    return y

def evaluate_c5(h, dx):
    """Evaluate C5 using correlation and dx scaling"""
    corr = correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def gradient_smoothing(h, alpha, dx):
    """Compute gradient of log-sum-exp approximation of max correlation"""
    n = len(h)
    # Compute full correlation (unscaled)
    cov_full = correlate(h, 1.0 - h, mode='full')
    I = cov_full  # shape 2n-1
    I_max = I.max()
    
    # Compute weights for softmax
    exp_vals = np.exp(alpha * (I - I_max))
    w = exp_vals / exp_vals.sum()
    p = w / w.sum()  # Normalized weights
    
    # Compute gradient as weighted sum of per-lag gradients
    grad = np.zeros(n)
    
    for idx, p_val in enumerate(p):
        lag = idx - (n - 1)  # ℓ ranges from -(n-1) to n-1
        if lag >= 0:
            # (1 - h[i+lag]) contribution
            if n - lag > 0:  
                grad[:n-lag] += p_val * (1.0 - h[lag:])
            # Edge contribution when i+lag >= n
            if lag > 0:
                grad[n-lag:] += p_val * 1.0
            # -h[i-lag] contribution
            if lag > 0:
                grad[lag:] -= p_val * h[:n-lag]
        else:
            # lag negative: let d = -lag
            d = -lag
            # (1 - h[i-d]) contribution
            if n - d > 0:
                grad[d:] += p_val * (1.0 - h[:n-d])
            # Edge contribution when i+lag < 0 (i < d) 
            grad[:d] += p_val * 1.0
            # -h[i+d] contribution
            grad[:n-d] -= p_val * h[d:]
    
    # Apply dx scaling to gradient for proper units
    grad_scaled = grad * dx
    return grad_scaled

def line_search_descent(h_start, eta, alpha, dx, max_backtrack=5):
    """Perform line search to find good descent step"""
    h = h_start.copy()
    c5 = evaluate_c5(h, dx)
    
    for _ in range(max_backtrack):
        h_candidate = project_to_sum_box(h - eta * gradient_smoothing(h, alpha, dx), TARGET_SUM)
        c5_candidate = evaluate_c5(h_candidate, dx)
        if c5_candidate <= c5:
            return h_candidate, c5_candidate
        eta *= 0.5
    return h, c5

def gradient_descent_run(h_start, time_limit):
    """Run gradient descent with log-sum-exp smoothing"""
    h = h_start.copy()
    dx = 2.0 / len(h_start)
    c5 = evaluate_c5(h, dx)
    
    best_h = h.copy()
    best_c5 = c5
    
    t0 = time.time()
    alpha = 0.5
    eta = 0.1
    it = 0
    max_iter = 5000
    
    while time.time() - t0 < time_limit and it < max_iter:
        # Schedule: increase α slowly, decrease η slowly 
        progress = min(1.0, it / 1000.0)
        alpha = 0.5 + 0.5 * (1.0 - np.exp(-3.0 * progress))  # Increase α gradually
        eta = 0.1 * (0.99 ** it)  # Decrease η exponentially
        
        h_new, c5_new = line_search_descent(h, eta, alpha, dx)
        
        h = h_new
        c5 = c5_new
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        it += 1
    
    return best_h, best_c5

def local_refine(h):
    """Refine solution using SLSQP optimizer"""
    def objective(h):
        return evaluate_c5(h, 2.0/len(h))
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h))]
    
    try:
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            c5_refined = evaluate_c5(h_refined, 2.0/len(h_refined))
            return h_refined, c5_refined
        else:
            return h, evaluate_c5(h, 2.0/len(h))
    except:
        return h, evaluate_c5(h, 2.0/len(h))
