import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.fft import rfft, irfft
import time
import math

def repair_and_interpolate(control_values, n_points, K):
    """Repair control values and interpolate to full grid"""
    # Clip control values to [0, 1]
    c = np.clip(control_values, 0, 1)
    
    # Create equi-spaced x coordinates for control points
    x_control = np.linspace(0, 2, K)
    
    # Interpolate to full resolution grid
    x_full = np.linspace(0, 2, n_points)
    h_full = np.interp(x_full, x_control, c)
    
    # Project onto capped simplex (sum = n_points/2)
    target_sum = n_points / 2.0
    current_sum = np.sum(h_full)
    
    # Binary search for lambda to project onto simplex
    if abs(current_sum - target_sum) < 1e-12:
        return h_full
    
    # Simple iterative approach when binary search fails
    max_iter = 30
    for _ in range(max_iter):
        if current_sum == target_sum:
            break
        # Use scaling to adjust
        scale = target_sum / current_sum
        h_full *= scale
        current_sum = np.sum(h_full)
        if abs(current_sum - target_sum) < 1e-12:
            break
    
    # Ensure constraint with final clipping
    h_full = np.clip(h_full, 0, 1)
    current_sum = np.sum(h_full)
    if abs(current_sum - target_sum) > 1e-10:
        # Final adjustment by rescaling
        if current_sum > 0:
            h_full = h_full * target_sum / current_sum
            h_full = np.clip(h_full, 0, 1)
    
    return h_full

def compute_soft_max_overlaps(h, alpha):
    """Compute soft-max of all overlap values"""
    n = len(h)
    dx = 2.0 / n
    # Direct correlation (better for small n)
    corr = correlate(h, 1 - h, mode="full")
    corr = corr * dx
    max_corr = np.max(corr)
    return max_corr

def compute_exact_c5(h):
    """Compute exact C5 constant"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def evaluate_objective(control_values, K, n_points, alpha, include_symmetry=True):
    """Evaluate soft-max objective with repair and interpolation"""
    # Repair and interpolate once
    h = repair_and_interpolate(control_values, n_points, K)
    
    # Optional symmetry enforcement
    if include_symmetry:
        h = (h + h[::-1]) / 2.0
    
    # Compute soft max
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    corr = corr * dx
    
    # Softmax computation
    max_corr = np.max(corr)
    if alpha > 0:
        max_corr = np.log(np.sum(np.exp(alpha * corr))) / alpha
    return max_corr

def generate_initial_control_points(K, seed=42):
    """Generate structured initial points"""
    np.random.seed(seed)
    # Start as sine wave, symmetric
    base = 0.5 + 0.3 * np.sin(np.linspace(0, 2*np.pi, K))
    base = np.clip(base, 0, 1)
    return base

def projected_gradient_descent(best_h, n_points, max_iter=500):
    """Run projected gradient descent on the exact max"""
    h = best_h.copy()
    dx = 2.0 / n_points
    eta = 0.05
    
    for i in range(max_iter):
        # Compute overlaps directly
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        
        # Compute subgradient
        g = np.zeros(n_points)
        # Gradient at k = max_idx
        # g[i] = (1 - h[i + max_idx]) - h[max_idx - i] (relatively simplified)
        # but easier to compute it explicitly using arrays
        for k in [-n_points + 1, 0, n_points - 1]:
            if k >= 0:
                g[k:] += (1 - h[:-k]) if k > 0 else (1 - h[:])
            else:
                g[:k] += h[-k:] if k < 0 else h[:]
                
        # Subtract h's shift
        if max_idx > 0:
            g[max_idx:] -= h[:-max_idx]
        else:
            g[:max_idx] -= h[-max_idx:]
        # Correct for bounds:
        g = (-1) * g * dx
        # Update with step size
        h_new = h - eta * g
        
        # Project onto simplex
        target_sum = n_points / 2.0
        current_sum = np.sum(h_new)
        
        # Simple projection via rescaling
        if current_sum > 0:
            h_new *= target_sum / current_sum

        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new *= target_sum / current_sum
        
        h = h_new
        # Compute new max for early stopping
        new_corr = correlate(h, 1 - h, mode="full")
        new_max = np.max(new_corr) * dx
        if i > 10 and abs(new_max - max_corr) < 1e-7:
            break
            
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set parameters
    n_points = 128
    K = 12 # control points
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Time trackers
    time_per_iter = 0.01
    remaining_budget = budget_s * 0.95
            
    best_c5 = float('inf')
    best_h = None
    
    # Multi-start approach
    num_restarts = 5
    best_of_restarts = None
    best_c5_of_restarts = float('inf')
    
    for restart in range(num_restarts):
        # Generate initial control points
        init_c = generate_initial_control_points(K, seed + restart)
        
        # CMA-ES-like procedure manually done
        alpha = 5.0
        max_alpha = 30.0
        alpha_incr = (max_alpha - alpha) / 1000 # gradually increase
        
        # Small modification per restart
        new_c = init_c.copy()
        
        # Simulated simple evolutionary optimization with multiple steps
        for step in range(1000):
            # Generate new candidate with small random perturbations
            new_c = init_c.copy()
            # Add small noise
            for i in range(len(new_c)):
                if np.random.rand() < 0.5:
                    delta = np.random.normal(0, 0.1)
                    new_c[i] += delta
            # Repair
            h = repair_and_interpolate(new_c, n_points, K)
            # Symmetry enforcement for stability
            h = (h + h[::-1]) / 2.0
            
            # Compute evaluation with soft max
            corr = correlate(h, 1 - h, mode="full")
            corr = corr * dx
            
            if alpha > 0:
                soft_max = np.log(np.sum(np.exp(alpha * corr))) / alpha
            else:
                soft_max = np.max(corr)
                
            if soft_max < best_c5:
                best_c5 = soft_max
                best_h = h.copy()
            
            # Increasing alpha gently
            alpha = min(alpha + alpha_incr, max_alpha)
            
            if (time.time() - start_time) > remaining_budget:
                break
                
        # Final polishing
        if best_h is not None:
            polished_h = projected_gradient_descent(best_h, n_points, max_iter=200)
            polished_c5 = compute_exact_c5(polished_h)
            if polished_c5 < best_c5_of_restarts:
                best_c5_of_restarts = polished_c5
                best_of_restarts = polished_h.copy()
                
    # Return final best solution
    if best_of_restarts is not None:
        final_c5 = compute_exact_c5(best_of_restarts)
        return (best_of_restarts.tolist(), final_c5, n_points)
    else:
        # Fallback
        init_h = generate_initial_control_points(K, seed)
        init_h = repair_and_interpolate(init_h, n_points, K)
        final_c5 = compute_exact_c5(init_h)
        return (init_h.tolist(), final_c5, n_points)
