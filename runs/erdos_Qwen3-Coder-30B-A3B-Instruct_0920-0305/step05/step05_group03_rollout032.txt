```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Best solution found so far
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Phase 1: Differential evolution for global search
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            args=(),
            maxiter=500,
            popsize=15,
            tol=1e-12,
            seed=seed,
            callback=lambda x, convergence: None
        )
        if de_result.success:
            h_candidate = de_result.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass
    
    # Phase 2: Local optimization with multiple restarts
    restarts = 300
    for i in range(restarts):
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
        # Random perturbation around the best solution
        new_h = best_h + np.random.normal(0, 0.02, n_points)
        new_h = np.clip(new_h, 0, 1)
        
        # Normalize
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='L-BFGS-B', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue
            
    # Phase 3: Try specialized patterns
    if (time.time() - start_time) < budget_s * 0.9:
        # Pattern 1: Two spikes
        try:
            x = np.linspace(0, 2, n_points)
            # Two peaks at 0.5 and 1.5
            pattern1 = np.exp(-((x - 0.5)**2) / 0.05) + np.exp(-((x - 1.5)**2) / 0.05)
            pattern1 = pattern1 / np.max(pattern1)  # Normalize to 1
            # Blend with best solution
            blended = 0.7 * best_h + 0.3 * pattern1
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(blended) * dx)
            blended = blended * scale_factor
            
            res = minimize(
                objective, 
                blended, 
                method='L-BFGS-B', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
        
        # Pattern 2: Single spike in center
        try:
            x = np.linspace(0, 2, n_points)
            # Central spike
            pattern2 = np.exp(-((x - 1.0)**2) / 0.1)
            pattern2 = pattern2 / np.max(pattern2)
            # Blend with best solution
            blended = 0.7 * best_h + 0.3 * pattern2
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(blended) * dx)
            blended = blended * scale_factor
            
            res = minimize(
                objective, 
                blended, 
                method='L-BFGS-B', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass

    # Ensure final constraint satisfaction
    dx = 2.0 / n_points
    scale = 1.0 / (np.sum(best_h) * dx)
    best_h = best_h * scale
    
    return (best_h.tolist(), best_c5, n_points)
```
