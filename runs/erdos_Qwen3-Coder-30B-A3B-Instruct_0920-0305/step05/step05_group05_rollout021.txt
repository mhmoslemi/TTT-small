import numpy as np
import time

def project_to_box_simplex(v, target_sum):
    """Project vector v onto the simplex with sum = target_sum and box constraints [0,1]"""
    # Binary search for the correct lambda
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0, 1)

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx
