import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = len(h0)
    
    # Parameters for Fourier approach
    L = 10  # Fourier truncation order
    M = 4 * L  # Number of sampling points for box constraints
    
    # Create sampling points
    x_sample = np.linspace(0, 2, M, endpoint=False)
    
    # Create the FFT matrix for box constraints
    # Each row is [1, cos(pi*l*x_sample), sin(pi*l*x_sample)] for l=1..L
    A_box = np.ones((M, 1 + 2*L))
    for l in range(1, L+1):
        A_box[:, 1+l-1] = np.cos(np.pi * l * x_sample)
        A_box[:, 1+L+l-1] = np.sin(np.pi * l * x_sample)
    
    # Define variables
    u = cp.Variable(2*L)  # [a_1, ..., a_L, b_1, ..., b_L]
    s = cp.Variable(L)    # [s_1, ..., s_L] where s_l = a_l^2 + b_l^2
    t = cp.Variable()     # upper bound on max overlap
    
    # Objective
    objective = cp.Minimize(t)
    
    # Box constraints
    h_values = A_box @ cp.hstack([0.5, u])  # h(x) values
    box_constraints = [0 <= h_values, h_values <= 1]
    
    # SOC constraints: s_l >= a_l^2 + b_l^2
    soc_constraints = []
    for l in range(L):
        a_l = u[l]
        b_l = u[L+l]
        soc_constraints.append(cp.norm(cp.vstack([2*a_l, 2*b_l, s[l]-1]), 2) <= s[l]+1)
    
    # Toeplitz constraint for Chebyshev bound
    # T = [[t, -s[0], ..., -s[L-1]],
    #      [-s[0], t, ..., -s[L-2]],
    #      ...
    #      [-s[L-1], ..., t]]
    T = cp.Variable((L+1, L+1))
    
    # Build Toeplitz matrix
    T_rows = [t]
    T_rows.extend(-s)
    T_rows = np.array(T_rows)
    
    # Fill the Toeplitz matrix
    for i in range(L+1):
        row = np.zeros(L+1)
        for j in range(L+1):
            row[j] = T_rows[abs(i-j)]
        T[i,:] = row
    
    # PSD constraint
    psd_constraint = [T >> 0]
    
    # The constraint that T represents the correct Toeplitz structure
    # We can directly impose the constraint that the_overlap polynomial
    # is bounded by t using the Toeplitz structure
    # This is equivalent to „the cosine polynomial don't exceed t“
    # In practice, directly checking Toeplitz PSD is preferred
    
    # We compute Toeplitz more carefully:
    # We form the Toeplitz matrix T with entries:
    # T[0,0]=t, T[0,j]=-s[j-1], T[i,0]=-s[i-1], T[i,j]=-s[|i-j|]
    # This forms an LH matrix:
    T = cp.Variable((L+1, L+1))
    T[0,0] = t
    for i in range(1, L+1):
        T[0,i] = -s[i-1]
        T[i,0] = -s[i-1]
        for j in range(1, L+1):
            T[i,j] = -s[abs(i-j)]
    
    # Ensure Toeplitz structure: T[i,j] = T[abs(i-j), 0]
    T_constraint = [T >> 0]
    
    # Add all constraints
    constraints = box_constraints + soc_constraints + T_constraint
    
    # Construct problem
    problem = cp.Problem(objective, constraints)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

            )
            h_reconstructed = np.clip(h_reconstructed, 0, 1)
            h_reconstructed = h_reconstructed * (1.0 / np.sum(h_reconstructed) * 2.0 / n_points)
            c5_new = evaluate_c5(h_reconstructed)
            
            if c5_new < best_c5:
                best_h = h_reconstructed.copy()
                best_c5 = c5_new
    except:
        pass
    
    # If could not improve with the large-scale problem, try a local search
    # only when there's time
    max_restarts = 300
    restart_count = 0
    
    while (time.time() - start_time) < budget_s * 0.95 and restart_count < max_restarts:
        # Try local search from current best
        try:
            h_temp = best_h.copy()
            # Add random perturbation
            indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
            for i in indices:
                h_temp[i] += np.random.normal(0, 0.02)
            h_temp = np.clip(h_temp, 0, 1)
            # Normalize
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(h_temp) * dx)
            h_temp = h_temp * scale_factor
            
            c5_temp = evaluate_c5(h_temp)
            if c5_temp < best_c5:
                best_h = h_temp.copy()
                best_c5 = c5_temp
                
        except:
            pass
        
        restart_count += 1
    
    return (best_h.tolist(), best_c5, n_points)
