def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.irfft(np.fft.rfft(h_rev, 2*n - 1) * np.fft.rfft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def perturb_move(h, rng_state):
    """Apply a stochastic perturbation"""
    np.random.set_state(rng_state)
    h_new = h.copy()
    # Perturb 10% of the points
    n_perturb = max(1, len(h) // 10)
    indices = np.random.choice(len(h), size=n_perturb, replace=False)
    for i in indices:
        # Add small Gaussian noise, clip to [0,1]
        h_new[i] = np.clip(h_new[i] + np.random.normal(0, 0.01), 0, 1)
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    return h_new

    # Simple greedy local search

    best_change = 0.0
    best_idx = -1
    best_delta = 0.0
    
    # Try moving mass between adjacent points
    for i in range(n):
        for j in range(max(0, i-5), min(n, i+6)):
            if i == j:
                continue
            # Try moving small amount from i to j
            if h[i] > 0 and h[j] < 1:
                delta = min(0.01, h[i], 1 - h[j])
                if delta > 1e-10:
                    h_trial = h_new.copy()
                    h_trial[i] -= delta
                    h_trial[j] += delta
                    h_trial = project_to_sum_box(h_trial, TARGET_SUM)
                    c5_trial = evaluate_c5(h_trial)
                    if c5_trial < evaluate_c5(h_new):
                        change = evaluate_c5(h_new) - c5_trial
                        if change > best_change:
                            best_change = change
                            best_idx = i
                            best_delta = delta
    
    if best_idx >= 0 and best_change > 1e-10:
        h_new[best_idx] -= best_delta
        h_new[best_idx + (best_idx < best_idx + 1)] += best_delta
        h_new = project_to_sum_box(h_new, TARGET_SUM)
    
    return h_new
