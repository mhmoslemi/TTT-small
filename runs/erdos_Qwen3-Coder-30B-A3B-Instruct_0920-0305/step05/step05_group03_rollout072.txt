import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Use a hybrid optimization approach with better strategy
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Strategy 1: Differential evolution for global search
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Global optimization with differential evolution
    try:
        res = differential_evolution(
            objective,
            bounds,
            constraints=cons,
            maxiter=500,
            popsize=15,
            seed=seed,
            polish=True,
            atol=1e-12,
            rtol=1e-12
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Strategy 2: Refine with multiple local searches from different starting points
    local_restart_bound = 1000
    
    for restart in range(local_restart_bound):
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
        # Start with best solution + small random perturbations
        new_h = best_h.copy()
        # Add isotropic noise
        noise = np.random.normal(0, 0.01, n_points)
        new_h = new_h + noise
        new_h = np.clip(new_h, 0, 1)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            # Try different optimization methods
            methods = ['L-BFGS-B', 'TNC', 'SLSQP']
            for method in methods:
                if (time.time() - start_time) > budget_s * 0.95:
                    break
                res = minimize(
                    objective, 
                    new_h, 
                    method=method, 
                    bounds=bounds, 
                    constraints=cons,
                    options={'maxiter': 1000, 'ftol': 1e-13, 'gtol': 1e-13}
                )
                if res.success:
                    h_candidate = res.x
                    scale = 1.0 / (np.sum(h_candidate) * dx)
                    h_candidate = h_candidate * scale
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
                        break  # Found improvement, move to next restart
        except:
            continue

    # Strategy 3: Adversarial refinement
    if (time.time() - start_time) < budget_s * 0.95:
        try:
            # Create a pattern that balances width and height
            # Start with an asymmetric distribution to escape local minima
            x = np.linspace(0, 2, n_points)
            # Mix of Gaussian and uniform pattern
            gauss_pattern = np.exp(-((x - 0.5)**2) / 0.1) + np.exp(-((x - 1.5)**2) / 0.1)
            # Normalize and adjust
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(gauss_pattern) * dx)
            gauss_pattern = gauss_pattern * scale_factor
            
            # Mix with best current solution
            mixed_h = 0.6 * best_h + 0.4 * gauss_pattern
            
            # Normalize
            scale_factor = 1.0 / (np.sum(mixed_h) * dx)
            mixed_h = mixed_h * scale_factor
            
            res = minimize(
                objective, 
                mixed_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass

    return (best_h.tolist(), best_c5, n_points)
