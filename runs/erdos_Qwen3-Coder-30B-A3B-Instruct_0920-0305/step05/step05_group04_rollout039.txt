import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def conv_fft(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    pad_len = n + m - 1
    # zero-pad to avoid circular convolution effects
    a_pad = np.pad(a, (0, pad_len - n), 'constant')
    b_pad = np.pad(b, (0, pad_len - m), 'constant')
    return np.real(np.fft.irfft(np.fft.rfft(a_pad) * np.fft.rfft(b_pad)))

def project_onto_bounded_simplex(x):
    """Project vector x onto {x : 0 <= x_i <= 1, sum x_i = 1}."""
    # Clip to [0, 1]
    x = np.clip(x, 0, 1)
    # Now adjust by shifting to make sum = 1
    s = np.sum(x)
    if abs(s - 1.0) < 1e-12:
        return x.copy()
    # Solve for lambda such that sum of clip(x+lambda, 0, 1) = 1
    # This is a piecewise linear function of lambda
    # For numerical stability, we know that the lambda such that we
    # get sum = 1 must be between -max(x) and 1-min(x).
    def sum_clipped_lambda(lambda_val):
        return np.sum(np.clip(x + lambda_val, 0, 1)) - 1.0
    
    # Binary search
    lo, hi = -np.max(x), 1.0 - np.min(x)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        val = sum_clipped_lambda(mid)
        if val > 0:
            hi = mid
        else:
            lo = mid
    lambda_val = (lo + hi) / 2.0
    return np.clip(x + lambda_val, 0, 1)

def soft_weights(c, beta):
    """Compute softmax of -beta * c, handling potential overflow."""
    c_max = np.max(c)
    exp_vals = np.exp(-beta * (c - c_max))
    return exp_vals / np.sum(exp_vals)

def smooth_obj_and_grad(h, beta):
    """
    Given h and beta, return (Φβ(h), gradient of Φβ(h))
    where Φβ(h) = (1/β) * log(sum(exp(-beta * c_k)))
    """
    # Compute correlations for shifts k=1 to n-1
    # Using FFT for efficiency
    c = conv_fft(h, h)
    # c is length 2*n-1. We only want c_1, ..., c_{n-1}
    c = c[1:n]

    # Compute soft weights
    w = soft_weights(c, beta)
    # dΦ/d h = - sum_k w_k * (h_{k+1} + h_{k-1} where applicable)
    # or in terms of convolutions:
    # let w_rev be reversed weights, then gradient is -(conv(h, w) + conv(h, w_rev))
    # Use fast convolution
    kernel_l = np.concatenate([w, np.zeros_like(w)])
    # Convolve h with w (forward shift part)
    conv_l = conv_fft(h, w)
    # Convolve h with reverse(w) (backward shift part)
    conv_r = conv_fft(h, w[::-1])
    # Now the gradient is -(conv_l + conv_r) where only the first n values are valid.
    # Note: conv_l is of length 2n-1, and conv_r also
    g = -(conv_l[:len(h)] + conv_r[:len(h)])

    # Phiβ = (1/β) * log(sum(exp(-beta * c)))
    Phi = (1.0 / beta) * np.log(np.sum(np.exp(-beta * c)))
    return Phi, g

def optimize_with_beta_continuation(h0, beta0=1.0, beta_max=200.0, alpha0=0.1, gamma=1.5, delta=0.95):
    h = h0.copy()
    beta = beta0
    alpha = alpha0
    prev_obj = np.inf
    converged_iters = 0
    max_converged = 5

    for i in range(2000):  # maximum iterations
        try:
            Phi, grad = smooth_obj_and_grad(h, beta)
            # Armijo line search
            tau = alpha
            while True:
                h_new = project_onto_bounded_simplex(h - tau * grad)
                Phi_new, _ = smooth_obj_and_grad(h_new, beta)
                if Phi_new <= Phi - 0.1 * tau * np.linalg.norm(grad)**2 or tau < 1e-12:
                    break
                tau *= 0.5

            if np.linalg.norm(tau * grad) < 1e-9:
                converged_iters += 1
            else:
                converged_iters = 0

            h = h_new
            if abs(Phi - prev_obj) < 1e-12:
                converged_iters += 1
            else:
                converged_iters = 0

            prev_obj = Phi

            if converged_iters >= max_converged:
                # Increase beta
                beta = min(beta * gamma, beta_max)
                alpha *= delta
                if abs(beta - beta_max) < 1e-2:
                    # Final phase, might reduce step size more
                    pass
                converged_iters = 0
            if i > 500 and beta > beta_max * 0.5:
                break
        except Exception as e:
            break

    return h

def evaluate_c5(h):
    """Evaluate the C5 constant."""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx
