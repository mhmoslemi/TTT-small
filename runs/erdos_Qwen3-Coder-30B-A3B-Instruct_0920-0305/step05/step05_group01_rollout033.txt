def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

    # Find the worst lag and make a move in the direction of improvement
    i = np.random.randint(0, n)
    if i + worst_lag < n:
        # Try increasing h[i] and decreasing h[i + worst_lag]
        if h[i] < 1 and h[i + worst_lag] > 0:
            delta = min(max_delta, 1 - h[i], h[i + worst_lag])

                h_new = h.copy()
                h_new[i] += delta

        # Try increasing h[i] and decreasing h[i - worst_lag]

            delta = min(max_delta, 1 - h[i], h[i - worst_lag])

                h_new = h.copy()
                h_new[i] += delta
                h_new[i - worst_lag] -= delta
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                return h_new

    i, j = np.random.choice(n, 2, replace=False)
    delta = np.random.uniform(0, max_delta)

        h_new[i] -= delta
        h_new[j] += delta
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        return h_new
    
    return h
