# Global variables
TARGET_SUM = None

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    # The maximum correlation corresponds to the maximum overlap integral
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def adaptive_local_search(h0, run_seconds, rng):

    h = h0.copy()

    step_size = 0.01
    max_iterations = int(1e6)
    patience = 1000
    
    for i in range(max_iterations):
        if time.time() - t_start > run_seconds:
            break
            
        # Try different update strategies
        strategy = rng.choice(['single', 'pair', 'gradient'])
        
        if strategy == 'single':
            # Move one point
            idx = rng.randint(0, n_points)
            delta = rng.uniform(-step_size, step_size)
            h[idx] += delta
            
        elif strategy == 'pair':
            # Move two points
            i1, i2 = rng.choice(n_points, 2, replace=False)
            delta = rng.uniform(-step_size, step_size)
            h[i1] += delta
            h[i2] -= delta
            
        else:  # gradient
            # Simple gradient direction approximation
            grad = np.zeros(n_points)
            for k in range(1, n_points):
                if k < n_points:
                    grad[:-k] += h[k:]
                if k < n_points:
                    grad[k:] += h[:-k]
            # Move in promising direction
            idx = np.argmax(grad)
            delta = step_size * rng.choice([-1, 1])
            h[idx] += delta
        
        # Project to constraints
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Evaluate
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 1000  # Reset patience
        else:
            patience -= 1
            if patience <= 0:
                # Reduce step size if no improvement
                step_size *= 0.99
                patience = 1000
    
    return best_h, best_c5
