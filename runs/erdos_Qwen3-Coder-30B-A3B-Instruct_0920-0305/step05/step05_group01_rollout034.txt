import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate C5 using correlation"""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation
    corr = correlate(h, 1 - h, mode='full')
    # Return maximum correlation scaled by dx
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project to sum constraint with box constraints [0,1]"""
    # Use bisection to find lambda
    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < 1e-12:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    return np.clip(x + low, 0, 1)
