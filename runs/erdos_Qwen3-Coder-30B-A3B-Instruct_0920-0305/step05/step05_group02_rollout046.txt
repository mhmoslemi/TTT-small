def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def greedy_initialization(n):
    """Construct a greedy warm-start binary solution"""
    x = [0] * n
    selected = set()
    while len(selected) < n // 2:
        best_i = -1
        best_min_p = -1
        for i in range(n):
            if i in selected:
                continue
            # compute hypothetical p_k values after adding i
            new_p = [0] * (n - 1)
            for j in selected:
                k = abs(i - j)
                if k > 0:
                    new_p[k-1] += 1
            min_p = min(new_p) if new_p else 0
            if min_p > best_min_p:
                best_min_p = min_p
                best_i = i
        if best_i != -1:
            x[best_i] = 1
            selected.add(best_i)
    return np.array(x)

def construct_milp_model(n, binary_x, warm_start=None):
    """Construct and return a MILP model"""
    # Variables
    x = [Variable(boolean=True) for i in range(n)]
    y = [[Variable(boolean=True) for i in range(n - k)] for k in range(1, n)]

    # Objective: maximize L (the minimum p_k)
    L = Variable(integer=True)

    # Constraints
    # 1. Sum of x[i] = n/2
    constraints = [sum(x) == n // 2]

    # 2. Symmetry breaking
    constraints.append(x[0] == 1)
    constraints.append(sum(x[:n//2]) >= sum(x[n//2:]))

    # 3. link x and y variables
    for k in range(1, n):  # k from 1 to n-1
        for i in range(n - k):
            # y_{i,k} <= x[i]
            constraints.append(y[k-1][i] <= x[i])
            # y_{i,k} <= x[i+k]
            constraints.append(y[k-1][i] <= x[i + k])
            # y_{i,k} >= x[i] + x[i+k] - 1
            constraints.append(y[k-1][i] >= x[i] + x[i + k] - 1)

    # 4. link p_k and y
    for k in range(1, n):
        p_k = sum(y[k-1])
        constraints.append(p_k >= L)

    # Set warm start if available
    if warm_start is not None:
        for i in range(n):
            x[i].value = warm_start[i]
        for k in range(1, n):
            for i in range(n - k):
                # This is approximate as we don't define y values with warm start directly
                pass

    # Build model
    objective = Maximize(L)
    prob = Problem(objective, constraints)
    return prob, x, y, L

def solve_milp(prob, x, y, L, timeout=10):
    """Solve the MILP problem with optional timeout"""
    try:
        # Set solver parameters for better performance
        prob.solve(solver=ECOS_BB, verbose=False, max_iters=1000, abstol=1e-6, reltol=1e-6, feastol=1e-6)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            return True, L.value if 'L' in locals() else None, [var.value for var in x]
    except Exception as e:
        pass
    return False, None, None

def polish_solution(h_values, n_points):
    """Use gradient-based polishing to slightly improve a binary solution"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    
    def objective(h):
        corr = np.correlate(h, 1-h, mode="full")
        return -np.max(corr) * dx  # negative for minimization
    
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    bounds = [(0, 1) for i in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}

    res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol':1e-12, 'gtol':1e-12, 'maxiter':500})
    if res.success:
        h = res.x
        h = np.clip(h, 0, 1)
        # Renormalize
        h = h * (1.0 / (np.sum(h) * dx))
        return h
    return h
