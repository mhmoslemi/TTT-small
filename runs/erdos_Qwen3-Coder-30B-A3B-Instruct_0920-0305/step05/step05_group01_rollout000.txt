import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60
EPS = 1e-12

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    low, high = -2.0, 2.0
    for _ in range(50):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target_sum) < EPS:
            return y
        if s > target_sum:
            high = mid
        else:
            low = mid
    y = np.clip(x + low, 0.0, 1.0)
    return y

def evaluate_c5(h, dx):
    """Evaluate C5 using correlation"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def gradient_smoothing(h, alpha):
    """Compute gradient of log-sum-exp approximation of max correlation"""
    n = len(h)
    dx = 2.0 / n
    
    # Compute correlation (unscaled)  
    cov_full = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute weights using softmax
    I_max = cov_full.max()
    exp_vals = np.exp(alpha * (cov_full - I_max))
    w = exp_vals / exp_vals.sum()
    
    # Compute gradient as weighted sum of per-lag gradients  
    grad = np.zeros(n)
    
    for idx, p_val in enumerate(w):
        lag = idx - (n - 1)  # lag from -n+1 to n-1
        
        if lag >= 0:
            # contribution from (1 - h[i+lag])
            if n - lag > 0:
                grad[:n-lag] += p_val * (1.0 - h[lag:])
            
            # edge contribution for i where i+lag >= n (i >= n-lag)
            if lag > 0:
                grad[n-lag:] += p_val * 1.0
            
            # contribution from -h[i-lag] (only if i-lag >= 0)
            if lag > 0:
                grad[lag:] -= p_val * h[:n-lag]
            else:  # lag == 0
                grad -= p_val * h
                
        else:
            # lag < 0, let d = -lag so d > 0
            d = -lag
            
            # contribution from (1 - h[i-d])
            if n - d > 0:
                grad[d:] += p_val * (1.0 - h[:n-d])
            
            # edge contribution for i where i+lag = i-d < 0 (i < d)
            grad[:d] += p_val * 1.0
            
            # contribution from -h[i+d]
            grad[:n-d] -= p_val * h[d:]
    
    # Scale gradient by dx since objective is scaled
    return grad * dx

def descent_step(h, eta, alpha, target_sum):
    """Perform one descent step with line search"""
    grad = gradient_smoothing(h, alpha)
    # Try step
    h_new = project_to_sum_box(h - eta * grad, target_sum)
    return h_new

def gradient_descent_run(initial_h, time_limit, target_sum):
    """Run gradient descent optimization for given time limit"""
    h = initial_h.copy()
    dx = 2.0 / len(h)
    c5 = evaluate_c5(h, dx)
    
    best_h = h.copy()
    best_c5 = c5
    
    t0 = time.time()
    it = 0
    
    # Adaptive scheduling
    alpha_start = 0.5
    alpha_end = 50.0
    eta_start = 0.1
    
    while time.time() - t0 < time_limit and it < 10000:
        fraction = min(1.0, it / 2000.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** fraction
        eta = eta_start * (0.99 ** it)
        
        h_new = descent_step(h, eta, alpha, target_sum)
        c5_new = evaluate_c5(h_new, dx)
        
        if c5_new <= c5:
            h = h_new
            c5 = c5_new
        else:
            # If didn't improve, try smaller step
            eta_small = eta * 0.5
            h_test = descent_step(h, eta_small, alpha, target_sum)
            c5_test = evaluate_c5(h_test, dx)
            if c5_test <= c5:
                h = h_test
                c5 = c5_test
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        it += 1
        
    return best_h, best_c5

def local_refine(h, target_sum):
    """Refine solution using SLSQP optimizer"""
    def objective(h):
        dx = 2.0 / len(h)
        return evaluate_c5(h, dx)
    
    def constraint_eq(h):
        return np.sum(h) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h))]
    
    try:
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, target_sum)
            dx = 2.0 / len(h_refined)
            c5_refined = evaluate_c5(h_refined, dx)
            return h_refined, c5_refined
    except:
        pass
    return h, evaluate_c5(h, 2.0 / len(h))

        h_refined, c5_refined = local_refine(h_opt, target_sum)
