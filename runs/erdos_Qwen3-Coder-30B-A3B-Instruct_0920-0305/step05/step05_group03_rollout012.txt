import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Keep track of the best solution
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Try multiple strategies
    strategies = []
    
    # Strategy 1: Direct optimization
    try:
        res = minimize(
            objective, 
            h0, 
            method='trust-constr', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 5000, 'gtol': 1e-12, 'ftol': 1e-12, 'verbose': 0}
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Strategy 2: Differential evolution on a larger search space
    try:
        de_bounds = [(0, 1) for _ in range(n_points)]
        de_res = differential_evolution(
            objective,
            de_bounds,
            maxiter=500,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            callback=lambda x, convergence: None,
            atol=1e-12,
            rtol=1e-12
        )
        if de_res.success:
            h_candidate = de_res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Strategy 3: Local search starting from various random points
    local_searches = min(100, int(budget_s * 0.3))
    for _ in range(local_searches):
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
        # Generate random starting point with correct integral
        new_h = np.random.rand(n_points)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-13, 'gtol': 1e-13, 'disp': False}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue

    # Strategy 4: Gradient-based optimization with multiple restarts
    # This time try a more targeted gradient approach
    for _ in range(50):
        if (time.time() - start_time) > budget_s * 0.9:
            break
            
        # Try a structured approach
        # Use a piecewise constant function that decreases near the middle
        # and tries to use more space early on and late on
        new_h = np.array([1.0 if abs(i - n_points//2) < n_points//4 else 0.2 
                         for i in range(n_points)])
        new_h = np.clip(new_h, 0, 1)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor

        try:
            res = minimize(
                objective, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-14, 'disp': False}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue

    # Strategy 5: Refinement using L-BFGS-B
    if (time.time() - start_time) < budget_s * 0.9:
        try:
            # Try a few more refined restarts
            h_group = [best_h] + [np.random.rand(n_points) * 0.8 for _ in range(3)]
            for h in h_group:
                dx = 2.0 / n_points
                scale_factor = 1.0 / (np.sum(h) * dx)
                h = h * scale_factor
                res = minimize(
                    objective, 
                    h, 
                    method='L-BFGS-B', 
                    bounds=bounds,
                    options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-15}
                )
                if res.success:
                    h_candidate = res.x
                    scale = 1.0 / (np.sum(h_candidate) * dx)
                    h_candidate = h_candidate * scale
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
        except:
            pass

    # Final check and rounding
    final_c5 = evaluate_c5(best_h)
    
    # Ensure integral constraint
    dx = 2.0 / n_points
    final_h = best_h * (1.0 / (np.sum(best_h) * dx)) 
    
    return (final_h.tolist(), final_c5, n_points)
