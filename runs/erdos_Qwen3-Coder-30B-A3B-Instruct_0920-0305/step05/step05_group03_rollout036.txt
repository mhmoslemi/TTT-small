import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Phase 1: Differential evolution for global search
    try:
        # Using higher popsize and more generations for better global search
        res = differential_evolution(
            objective,
            bounds,
            maxiter=500,
            popsize=15,
            tol=1e-12,
            recombination=0.8,
            seed=seed,
            mutation=(0.5, 1.0)
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Phase 2: Local optimization with multiple strategies
    local_restart_bound = 300
    
    for restart in range(local_restart_bound):
        if (time.time() - start_time) > budget_s * 0.95:  # Leave time for final steps
            break
            
        # Strategy 1: Random perturb around best
        new_h = best_h + np.random.normal(0, 0.01, n_points)  # Reduced perturbation
        new_h = np.clip(new_h, 0, 1)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue
            
        # Strategy 2: Gradient-based with different init
        # Use a more structured perturbation
        new_h = np.copy(best_h)
        # Create a smooth pattern to avoid local minima
        x = np.linspace(0, 2, n_points)
        # Add a sinusoidal component 
        sin_component = 0.1 * np.sin(2 * np.pi * x) 
        new_h = new_h + sin_component
        new_h = np.clip(new_h, 0, 1)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='L-BFGS-B', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue

    # Phase 3: Tent-like adjustment with a focus on peak reduction
    if (time.time() - start_time) < budget_s * 0.9:
        try:
            # Try to make a geometry-aware adjustment that reduces max correlation
            # Start with a K-shaped pattern
            x = np.linspace(0, 2, n_points)
            # Create shape with two peaks at ends and a valley in center
            k_shape = np.zeros_like(x)
            k_shape[x < 0.5] = 1 - 2*x[x < 0.5]
            k_shape[x > 1.5] = 1 - 2*(2-x[x > 1.5])
            k_shape[(x >= 0.5) & (x <= 1.5)] = 0
            
            # Make sure it's positive and normalize
            k_shape = np.maximum(k_shape, 0)
            
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(k_shape) * dx)
            k_shape = k_shape * scale_factor
            
            # Mix with current best (but bias towards best)
            mixed_h = 0.7 * best_h + 0.3 * k_shape
            
            # Normalize
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(mixed_h) * dx)
            mixed_h = mixed_h * scale_factor
            
            res = minimize(
                objective, 
                mixed_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass

    return (best_h.tolist(), best_c5, n_points)
