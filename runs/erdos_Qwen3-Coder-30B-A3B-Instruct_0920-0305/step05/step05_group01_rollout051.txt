import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60
EPS = 1e-12

def project_to_sum_box(x, target_sum):
    """Project onto box constraints [0,1] and sum constraint"""
    low, high = -2.0, 2.0
    for _ in range(50):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target_sum) < EPS:
            return y
        if s > target_sum:
            high = mid
        else:
            low = mid
    y = np.clip(x + low, 0.0, 1.0)
    return y

def evaluate_c5(h, dx):
    """Compute C5 using correlation"""
    corr = correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def gradient_smoothing(h, alpha, dx):
    """Compute gradient of log-sum-exp smoothing of correlations"""
    n = len(h)
    # compute correlation without scaling
    cov_full = correlate(h, 1.0 - h, mode='full')
    # for numerical stability, subtract max
    I_max = cov_full.max()
    exp_vals = np.exp(alpha * (cov_full - I_max))
    w = exp_vals / exp_vals.sum()
    # compute the weighted gradient contribution
    grad = np.zeros(n)
    for idx, p_val in enumerate(w):
        lag = idx - (n - 1)
        if lag >= 0:
            # term (1 - h[i+lag]) for i=0..n-1-lag
            if n - lag > 0:
                grad[:n-lag] += p_val * (1.0 - h[lag:])
            # edge contributions for i = n-ℓ..n-1
            if lag > 0:
                grad[n-lag:] += p_val * 1.0
            # subtract h[i-lag] for i=lag..n-1
            if lag > 0:
                grad[lag:] -= p_val * h[:n-lag]
            else:
                # lag==0: subtract h[i] for all i
                grad -= p_val * h
        else:
            # lag < 0
            d = -lag
            # (1 - h[i-d]) part for i >= d
            if n - d > 0:
                grad[d:] += p_val * (1.0 - h[:n-d])
            # edge: i where i + lag < 0 (i - d < 0) => i < d
            grad[:d] += p_val * 1.0
            # subtract h[i+d] part for i < n-d
            grad[:n-d] -= p_val * h[d:]
    return grad * dx

def descent_step(h, eta, alpha, dx, target_sum):
    """Perform a single descent step"""
    grad = gradient_smoothing(h, alpha, dx)
    h_new = project_to_sum_box(h - eta * grad, target_sum)
    return h_new

def gradient_descent(h_start, max_time, dx, target_sum):
    """Run gradient descent with smoothing"""
    h = h_start.copy()
    c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = c5
    t0 = time.time()
    it = 0
    max_iters = 10000
    alpha_start = 0.5
    alpha_end = 50.0
    eta_start = 0.1
    while time.time() - t0 < max_time and it < max_iters:
        # adaptive schedule
        progress = min(it / 5000.0, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        eta = eta_start * (0.999 ** it)
        h_new = descent_step(h, eta, alpha, dx, target_sum)
        c5_new = evaluate_c5(h_new, dx)
        if c5_new <= c5:
            h = h_new
            c5 = c5_new
        else:
            # attempt line search
            for i in range(3):
                eta *= 0.5
                h_new = descent_step(h, eta, alpha, dx, target_sum)
                c5_new = evaluate_c5(h_new, dx)
                if c5_new <= c5:
                    h = h_new
                    c5 = c5_new
                    break
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        it += 1
    return best_h, best_c5

def local_refine(h, target_sum):
    """Refine using SLSQP"""
    def objective(h):
        return evaluate_c5(h, dx)
    def constraint_eq(h):
        return np.sum(h) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h))]
    
    try:
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, target_sum)
            c5_refined = evaluate_c5(h_refined, dx)
            return h_refined, c5_refined
    except:
        pass
    return h, evaluate_c5(h, dx)

        h0 = np.interp(x_new, x_orig, h0)
    
    # Project to feasible set to ensure constraints
    h0 = project_to_sum_box(h0, target_sum)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)
    
    remaining_time = budget_s * 0.95  # leave buffer time
    
    # Multi-start gradient descent
    restarts = 0
    while time.time() - start_time < budget_s * 0.9 and restarts < 5:
        # Get time for this restart
        restart_time = min(remaining_time / (5 - restarts), 120)
        if restart_time < 10:
            break
            
        # Random perturbation of best solution
        perturbation = np.random.normal(0, 0.01, n_points)
        h_start = project_to_sum_box(best_h + perturbation, target_sum)
        
        # Run gradient descent
        h_local, c5_local = gradient_descent(h_start, restart_time, dx, target_sum)
        
        # Further local refinement
        h_refined, c5_refined = local_refine(h_local, target_sum)
        
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
            
        restarts += 1
        
        # Update remaining time with some margin
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 10:
            break
    
    return (best_h.tolist(), float(best_c5), n_points)
