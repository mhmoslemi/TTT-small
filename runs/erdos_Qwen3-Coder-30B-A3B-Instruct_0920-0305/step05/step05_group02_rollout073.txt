import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_correlation(h):
    """Compute full correlation array for a binary vector"""
    n = len(h)
    corr = np.zeros(2*n - 1)
    for k in range(-n+1, n):
        if k >= 0:
            corr[k+n-1] = np.sum(h[:n-k] * h[k:])
        else:
            corr[k+n-1] = np.sum(h[-k:] * h[:n+k])
    return corr

def evaluate_c5(h):
    """Evaluate C5 constant for a continuous function"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_c5_binary(h):
    """Evaluate C5 for binary h using correlation efficiently"""
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlation(h)
    max_corr = np.max(corr)
    return max_corr * dx

def generate_legendre_pattern(n):
    """Generate Legendre-based binary pattern if possible"""
    p = n // 2
    if p < 2 or not is_prime(p):
        # Not a suitable prime, fall back to random
        return None
    # Generate quadratic residues mod p
    residues = set()
    for a in range(1, p):
        r = (a * a) % p
        residues.add(r)
    b = np.zeros(p, dtype=int)
    for r in residues:
        if 0 <= r < p:
            b[r] = 1
    # Repeat to fill n
    h = np.tile(b, 2)
    if len(h) > n:
        h = h[:n]
    # Ensure sum is n/2 (p)
    ones_count = np.sum(h)
    if ones_count != p:
        # Flip a zero to one
        zero_positions = np.where(h == 0)[0]
        if len(zero_positions) > 0:
            flip_pos = zero_positions[0]
            h[flip_pos] = 1
    return h

def is_prime(n):
    """Check if number is prime"""
    if n < 2:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False
    for i in range(3, int(np.sqrt(n))+1, 2):
        if n % i == 0:
            return False
    return True

def generate_random_balanced_mask(n):
    """Generate a random balanced binary mask"""
    h = np.zeros(n, dtype=int)
    indices = np.random.permutation(n)
    h[indices[:n//2]] = 1
    return h

def find_bad_ones(h, k):
    """Find positions of 1s that would help reduce correlation at shift k"""
    n = len(h)
    bad_ones = []
    if k >= 0:
        for i in range(n-k):
            if h[i] == 1 and h[i+k] == 0:
                bad_ones.append(i)
    else:
        k = -k
        for i in range(k, n):
            if h[i] == 1 and h[i-k] == 0:
                bad_ones.append(i)
    return bad_ones

def compute_delta_corr(h, corr, i, j, k):
    """Compute change in correlation due to swapping h[i] and h[j]"""
    n = len(h)
    if k < 0:
        k = -k
    delta_corr = np.zeros(2*n - 1)
    
    # Handle contribution of i
    if i <= n-k-1:
        # i contributes to c_k
        old = h[i] - h[i] * h[i+k]
        new = 1 - 1 * h[i+k]  # h[i] becomes 1
        delta_corr[k+n-1] += new - old
    if i >= k:
        # i contributes to c_{-k}
        old = -h[i-k] * h[i]
        new = -h[i-k] * 1  # h[i] becomes 1
        delta_corr[-k+n-1] += new - old
    
    # Handle contribution of j
    if j <= n-k-1:
        # j contributes to c_k
        old = h[j] - h[j] * h[j+k]
        new = 0 - 0 * h[j+k]  # h[j] becomes 0
        delta_corr[k+n-1] += new - old
    if j >= k:
        # j contributes to c_{-k}
        old = -h[j-k] * h[j]
        new = -h[j-k] * 0  # h[j] becomes 0
        delta_corr[-k+n-1] += new - old
    return delta_corr

def swap(h, i, j, corr):
    """Swap h[i] with h[j] and update correlation"""
    old_h = h.copy()
    h[i], h[j] = h[j], h[i]
    # Recompute correlation
    corr = compute_correlation(h)
    return corr

def simulate_annealing(h0, n_points, seed, budget_s):
    np.random.seed(seed)
    start_time = time.time()
    best_h = h0.copy().astype(int)
    corr = compute_correlation(best_h)
    best_c5 = evaluate_c5_binary(best_h)
    max_iter = 2000000  # Consistent with cooldown length
    T0 = 0.05
    Tf = 1e-5
    iter = 0

    while iter < max_iter and (time.time() - start_time) < 0.95 * budget_s:
        t = (time.time() - start_time) / (0.95 * budget_s)
        if t > 1:
            t = 1
        T = T0 * (Tf / T0) ** t

        # Determine worst shifts
        c_max = np.max(corr)
        worst_shifts = np.where(np.abs(corr - c_max) < 1e-9)[0]
        
        if len(worst_shifts) == 0:
            break
            
        k_idx = np.random.choice(worst_shifts)
        k = k_idx - n_points + 1  # Convert to actual lag
        
        # Find bad ones for this shift
        bad_ones = find_bad_ones(best_h, k)
        
        if len(bad_ones) == 0:
            # Fallback - pick any 1
            one_indices = np.where(best_h == 1)[0]
            if len(one_indices) == 0:
                break
            bad_ones = [np.random.choice(one_indices)]

        i = np.random.choice(bad_ones)
        # Find valid j positions
        zero_positions = np.where(best_h == 0)[0]
        if len(zero_positions) == 0:
            iter += 1
            continue
        
        # Exclude positions that would generate negative effects
        # not too discriminative to avoid bottleneck
        valid_j = zero_positions  # simplest case
        j = np.random.choice(valid_j)
        
        # Compute new correlation after swap
        # Unfortunately, we need to recompute this for accuracy
        new_h = best_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_corr = compute_correlation(new_h)
        new_c5 = evaluate_c5_binary(new_h)
        
        if new_c5 < best_c5:
            best_h = new_h.copy()
            corr = new_corr
            best_c5 = new_c5
        else:
            # Accept with probability
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                best_h = new_h.copy()
                corr = new_corr
                best_c5 = new_c5
                
        iter += 1
        if iter % 10000 == 0 and best_c5 < 0.3807:
            break  # Early stop if target reached
            
        # Optional: restart from random mask every 100k iterations
        if iter % 100000 == 0:
            new_h = generate_random_balanced_mask(n_points)
            corr = compute_correlation(new_h)
            new_c5 = evaluate_c5_binary(new_h)
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5

    return best_h, best_c5

def polish_solution(h_bin, n_points, seed):
    """Polish the integer solution with continuous optimization"""
    np.random.seed(seed)
    h = h_bin.astype(float)
    dx = 2.0 / n_points
    
    # Project back to correct sum
    scale = (n_points / 2) / np.sum(h)
    h = h * scale
    
    # Smoothing parameter for log-sum-exp approximation
    alpha = 30.0
    
    def smooth_c5(h_values):
        corr = correlate(h_values, 1 - h_values, mode="full")
        # Smooth approximation of max
        return np.log(np.sum(np.exp(alpha * corr))) / alpha * dx
    
    bounds = [(0, 1) for _ in range(n_points)]
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint}
    
    try:
        res = minimize(
            smooth_c5,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 300}
        )
        if res.success:
            h_opt = res.x
            scale = (n_points / 2) / np.sum(h_opt)
            h_opt = h_opt * scale
            h_opt = np.clip(h_opt, 0, 1)
            return h_opt, evaluate_c5(h_opt)
    except:
        pass
    return h, evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    best_c5 = 1e10
    best_h = None
    best_n = None
    
    # Test different problem sizes in reasonable range
    for n_points in [58, 62, 66, 70, 74, 78]:
        if (time.time() - start_time) > 0.95 * budget_s:
            break
        # Generate initial deterministic patterns
        initial_patterns = []
        legendre_pattern = generate_legendre_pattern(n_points)
        if legendre_pattern is not None:
            initial_patterns.append(legendre_pattern)
        
        # Add some random patterns
        for _ in range(3):
            initial_patterns.append(generate_random_balanced_mask(n_points))
        
        # Add the best previous solution
        if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
            h_base = np.array(initial_h_values)
            initial_patterns.append(h_base.astype(int))
        
        for h0 in initial_patterns:
            if (time.time() - start_time) > 0.95 * budget_s:
                break
                
            h_annealed, c5_annealed = simulate_annealing(h0, n_points, seed, budget_s)
            
            # Polish with continuous optimization
            h_final, c5_final = polish_solution(h_annealed, n_points, seed)
            
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
                best_n = n_points
                
            if best_c5 < 0.3807:
                break
                
        if best_c5 < 0.3807:
            break
    
    # If no good solution found, use the best found
    if best_h is None:
        h_init = np.array(initial_h_values)
        if len(h_init) < 60:
            n_points = 60
            h_init = np.pad(h_init, (0, 60 - len(h_init)), 'constant', constant_values=(0, 0))
    else:
        h_init = best_h
        
    return (h_init.tolist(), best_c5, best_n if best_n is not None else 60)
