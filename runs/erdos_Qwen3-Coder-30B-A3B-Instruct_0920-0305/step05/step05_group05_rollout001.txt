import numpy as np
import time

def project_to_box_simplex(v, target_sum):
    # Binary search for lambda to satisfy sum = target_sum
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0, 1)

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts using linear correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Setup parameters
    n = 64  # Use 64 points for better resolution
    target_sum = n / 2
    dx = 2.0 / n
    max_iters_per_restart = 3000
    
    best_c5 = np.inf
    best_h = None
    
    # Utility to evaluate current C5
    def evaluate_c5_local(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def compute_gradient(h, alpha):
        """Compute gradient of smoothed max of overlap integrals"""
        # Compute correlation for all shifts
        conv = np.correlate(h, 1 - h, mode="full")
        O = conv * dx
        O_max = np.max(O)
        # Stable exponential for smoothing
        exp_term = np.exp(alpha * (O - O_max))
        w = exp_term / np.sum(exp_term)
        
        grad = np.zeros_like(h)
        for idx, shift in enumerate(range(-n+1, n)):
            w_k = w[idx]
            if w_k == 0:
                continue
                
            # Compute shift vectors efficiently
            if shift >= 0:
                sr = np.empty(n)
                sr[:shift] = 1.0
                sr[shift:] = 1.0 - h[shift:]
                sl = np.empty(n)
                sl[:shift] = 0.0
                sl[shift:] = h[:n-shift]
            else:
                s = -shift
                sr = np.empty(n)
                sr[:s] = 1.0
                sr[s:] = 1.0 - h[:n-s]
                sl = np.empty(n)
                sl[:n-s] = h[s:]
                sl[n-s:] = 0.0
            
            grad += w_k * (sr - sl)
        
        return grad * dx
    
    def optimize_h(n_points, max_iters, alpha_start, alpha_end, lr0):
        """Optimize h using gradient descent with projected updates"""
        h = np.random.rand(n_points)
        h = project_to_box_simplex(h, target_sum)
        
        for it in range(max_iters):
            alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
            grad = compute_gradient(h, alpha)
            
            lr = lr0 / np.sqrt(it + 1)
            h = h - lr * grad
            h = project_to_box_simplex(h, target_sum)
            
            if (time.time() - start_time) > budget_s * 0.95:
                break
                
        return h.copy()
    
    # Main optimization loop with multiple restarts
    remaining_time = budget_s * 0.9
    while (time.time() - start_time) < budget_s * 0.9:
        # Try multiple restart strategies
        restart_types = ['uniform', 'sine', 'gaussian', 'step']
        
        for restart_type in restart_types:
            if (time.time() - start_time) > budget_s * 0.95:
                break
                
            # Initialize h based on restart type
            if restart_type == 'uniform':
                h0 = np.random.rand(n)
            elif restart_type == 'sine':
                x = np.linspace(0, 2, n)
                h0 = 0.5 + 0.4 * np.sin(np.pi * x)
            elif restart_type == 'gaussian':
                x = np.linspace(0, 2, n)
                mu = 1.0
                sigma = 0.2
                h0 = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
            else:  # step
                h0 = np.zeros(n)
                mid = n // 2
                width = n // 4
                start = max(0, mid - width // 2)
                end = min(n, mid + width // 2)
                h0[start:end] = 1.0

            h0 = project_to_box_simplex(h0, target_sum)
            
            # Optimize with adaptive alpha
            optimized_h = optimize_h(n, max_iters_per_restart, 5, 30, 0.5)
            cur_c5 = evaluate_c5_local(optimized_h)
            
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = optimized_h.copy()
                
            # Early termination if we've found a good enough solution
            if best_c5 < 0.379:
                break
            
            if (time.time() - start_time) > budget_s * 0.95:
                break

    # Final refinement on the best solution found
    if best_h is not None:
        h_refine = best_h.copy()
        for _ in range(1000):
            # Use subgradient of max (gradient of the actual maximum value)
            O = np.correlate(h_refine, 1 - h_refine, mode="full") * dx
            max_idx = np.argmax(O)
            shift = max_idx - (n - 1)
            
            # Compute gradient of the max
            grad = np.zeros_like(h_refine)
            if shift >= 0:
                sr = np.empty(n)
                sr[:shift] = 1.0
                sr[shift:] = 1.0 - h_refine[shift:]
                sl = np.empty(n)
                sl[:shift] = 0.0
                sl[shift:] = h_refine[:n-shift]
            else:
                s = -shift
                sr = np.empty(n)
                sr[:s] = 1.0
                sr[s:] = 1.0 - h_refine[:n-s]
                sl = np.empty(n)
                sl[:n-s] = h_refine[s:]
                sl[n-s:] = 0.0
            
            grad = (sr - sl) * dx  # Gradient w.r.t. the maximum shift
            
            lr = 0.1 / np.sqrt(_ + 1)
            h_refine = h_refine - lr * grad
            h_refine = project_to_box_simplex(h_refine, target_sum)
            
            if _ % 100 == 0:
                cur_c5 = evaluate_c5_local(h_refine)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h_refine.copy()
                    
            if (time.time() - start_time) > budget_s * 0.99:
                break

    if best_h is None:
        # Fallback to uniform distribution if no better solution is found
        h_fallback = np.ones(n) * 0.5
        h_fallback = project_to_box_simplex(h_fallback, target_sum)
        best_c5 = evaluate_c5_local(h_fallback)
        best_h = h_fallback

    return (best_h.tolist(), float(best_c5), n)
