import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial binary vector from existing solution (try to use it as warm start)
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Range of discretisation sizes to test
    sizes = [36, 40, 44, 48, 52, 56, 60, 64]
    
    def binary_search_milp(n):
        dx = 2.0 / n
        x = cp.Variable(n, boolean=True)
        L = cp.Variable(integer=True)
        
        # Add symmetry breaking constraints
        constraints = [cp.sum(x) == n // 2, x[0] == 1]
        
        # Ensure ordering constraint to avoid symmetric copies
        left_sum = cp.sum(x[:n//2])
        right_sum = cp.sum(x[n//2:])
        constraints.append(left_sum >= right_sum)
        
        # Helper matrices for y_{i,k}
        y_vars = {}
        p_k = {}
        
        # Set y_{i,k} = x_i AND x_{i+k} for each k
        for k in range(1, n):
            y_vars[k] = []
            p_k[k] = 0
            for i in range(n - k):
                y = cp.Variable(boolean=True)
                y_vars[k].append(y)
                constraints.extend([
                    y <= x[i],
                    y <= x[i + k],
                    y >= x[i] + x[i + k] - 1
                ])
                p_k[k] += y
            constraints.append(p_k[k] >= L)
            
        # Objective function
        objective = cp.Maximize(L)
        prob = cp.Problem(objective, constraints)
        
        try:
            # Solve with time limit
            prob.solve(solver=cp.CBC, verbose=False, max_seconds=30)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                L_opt = L.value
                x_opt = x.value.astype(int)
                c5 = (n/2 - L_opt) * dx
                return x_opt, c5
            else:
                # Fallback to feasible point
                try:
                    x_feasible = []
                    for i in range(n):
                        x_feasible.append(int(round(x[i].value)))
                    x_feasible = np.array(x_feasible)
                    # Use greedy fallback if MILP fails
                    c5 = evaluate_erdos_solution(x_feasible)
                    return x_feasible, c5
                except:
                    # As a final fallback yield the initial values
                    return initial_h.copy(), evaluate_erdos_solution(initial_h)
        except Exception as e:
            # Fallback with greedy
            try:
                return greedy_initial_binary(n), evaluate_erdos_solution(greedy_initial_binary(n))
            except:
                return initial_h.copy(), evaluate_erdos_solution(initial_h)
    
    def greedy_initial_binary(n):
        dx = 2.0 / n
        h = np.zeros(n)
        positions = set()
        i = 0  # Iterate through possible positions
        count = 0
        weights = []
        while count < n // 2:
            # Compute how many new overlaps each position contributes
            f = np.zeros(n)
            f[i % n] = 1
            weights.append(i)
            if len(positions) == 0:
                worse_k = n//2
                count += 1
                positions.add(i)
            else:
                curr_best = None
                best_val = -1
                for j in range(n):
                    if j in positions: continue
                    f = np.zeros(n)
                    f[j] = 1
                    c = correlate(h, 1 - f, mode="full")
                    tv = np.max(c)
                    if tv > best_val:
                        best_val = tv
                        curr_best = j
                if curr_best is not None:
                    count += 1
                    positions.add(curr_best)
            i += 1
            if i > n * 2: break
        
        h = np.zeros(n)
        for p in positions:
            h[p] = 1
        return h
    
    def evaluate_erdos_solution(h):
        n = len(h)
        dx = 2.0 / n
        # Compute the overlap integral for all shifts
        corr = correlate(h, 1 - h, mode="full")
        # The maximum overlap value
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Execute across the discretisation sizes
    for sz in sizes:
        if time.time() - start_time > 0.95 * budget_s:
            break
        
        try:
            x_sol, c5_val = binary_search_milp(sz)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = x_sol.copy()
        except:
            try:
                # Fallback to greedy construction
                h_guess = greedy_initial_binary(sz)
                hguess = h_guess / (np.sum(h_guess) * (2.0/sz)) * (sz / 2)
                c5_guess = evaluate_erdos_solution(hguess)
                if c5_guess < best_c5:
                    best_c5 = c5_guess
                    best_h = hguess.copy()
            except:
                pass
    
    # Final polish step: Projected gradient descent on the continuous solution
    def project_onto_sum(h, target_sum=1.0):
        dx = 2.0 / len(h)
        actual_sum = np.sum(h) * dx
        if abs(actual_sum) < 1e-12:
            return h
        h = h * (target_sum / actual_sum)
        return h
    
    def smooth_reprojection(h, dx=2.0/len(h)):
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Re-normalize to sum = 1
        h = project_onto_sum(h, 1.0)
        return h
    
    # Run gradient refinement on best discrete solution
    h_best = best_h.copy()
    h_cont = h_best.astype(float).copy()
    dx = 2.0 / len(h_cont)
    
    min_max_corr = float('inf')
    for i in range(200):
        if time.time() - start_time > 0.99 * budget_s:
            break
        
        # Compute current c5 directly
        corr = correlate(h_cont, 1 - h_cont, mode="full")
        current_max = np.max(corr) * dx
        if current_max < min_max_corr:
            min_max_corr = current_max
            h_best = h_cont.copy()  # Keep the best version found
        else:
            break  # Convergence detected
        
        # Gradient descent step
        # Compute gradient of max auto correlation
        # for k in relevant lags, we have objective = max_k (sum (h_i * (1 - h_{i+k})))
        result_list = []
        for k in range(len(h_cont)//2 - 1):
            # Estimate gradient of c_k = sum h_i * (1 - h_{i+k})
            # d/dh_i c_k = (1 - h_{i+k}) if i is valid for this k
            grad = np.zeros_like(h_cont)
            if k > 0:
                grad[k:] += 1 - h_cont[:-k]
                grad[:-k] += 1 - h_cont[k:]
            else:
                # only diagonal term
                grad = 1 - h_cont
            result_list.append((grad, k))
        
        # Take max over all gradients by fragmenting problem (this is pseudo gradient approximation)
        k_max = np.argmax([np.sum(np.abs(g)) for g, _ in result_list])
        grad_for_max = result_list[k_max][0]
        
        # Update step (gradient ascent, we want to minimize)
        lr = 0.01
        h_cont += lr * grad_for_max
        # Project back onto constraints
        h_cont = smooth_reprojection(h_cont, dx)

    final_c5 = evaluate_erdos_solution(h_best)
    return (h_best.tolist(), final_c5, len(h_best))
