```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Phase 1: Use global optimization with DE
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    try:
        # Use differential evolution for global search
        res = differential_evolution(
            objective, 
            bounds, 
            maxiter=200, 
            popsize=15,
            tol=1e-12,
            seed=seed,
            callback=lambda x, convergence: None if (time.time() - start_time) < budget_s * 0.9 else None
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Phase 2: Local refinement with multiple restarts
    local_restart_bound = 500
    
    for restart in range(local_restart_bound):
        if (time.time() - start_time) > budget_s * 0.95:  # Leave time for final steps
            break
            
        # Start with small random perturbations around previous best
        new_h = best_h + np.random.normal(0, 0.02, n_points)  # Slightly larger perturbation
        new_h = np.clip(new_h, 0, 1)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue

    # Phase 3: Try a new structured approach if time allows
    if (time.time() - start_time) < budget_s * 0.9:
        try:
            # Generate a new pattern by modifying the best solution
            # Step-like pattern with plateau in the middle
            new_pattern = np.zeros(n_points)
            mid = n_points // 2
            width = n_points // 4
            
            # Create a smoothed step pattern
            for i in range(n_points):
                if i < mid - width//2:
                    new_pattern[i] = 0.0
                elif i < mid + width//2:
                    new_pattern[i] = 1.0
                else:
                    new_pattern[i] = 0.0
                    
            # Mix with current best
            mixing_factor = 0.3
            mix_pattern = mixing_factor * new_pattern + (1 - mixing_factor) * best_h
            
            # Normalize
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(mix_pattern) * dx)
            mix_pattern = mix_pattern * scale_factor
            
            # Refine
            try:
                res = minimize(
                    objective, 
                    mix_pattern, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=cons,
                    options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-14}
                )
                if res.success:
                    h_candidate = res.x
                    scale = 1.0 / (np.sum(h_candidate) * dx)
                    h_candidate = h_candidate * scale
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
            except:
                pass
                
        except:
            pass

    # Final refinement with trust-constr
    try:
        res = minimize(
            objective, 
            best_h, 
            method='trust-constr', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 2000, 'gtol': 1e-15, 'ftol': 1e-15}
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
```
