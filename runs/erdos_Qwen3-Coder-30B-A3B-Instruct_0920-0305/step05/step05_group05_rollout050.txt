import numpy as np
import time

def project_to_box_simplex(v, target_sum):
    """Project v onto box [0,1] and simplex with sum = target_sum"""
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0, 1)

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, alpha):
    """Compute gradient of smoothed max of overlap integrals"""
    n = len(h)
    dx = 2.0 / n
    # Compute overlaps for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    O = corr * dx
    O_max = np.max(O)
    
    # Compute stable softmax weights
    exp_term = np.exp(alpha * (O - O_max))
    w = exp_term / np.sum(exp_term)
    
    # Compute gradient
    grad = np.zeros_like(h)
    
    # Loop over all shift indices (-n+1 to n-1)
    for shift_idx, shift in enumerate(range(-n + 1, n)):
        w_k = w[shift_idx]
        if w_k == 0:
            continue
        
        # Compute shifted arrays
        if shift >= 0:
            # shift_right: (1 - h[i + shift]) for valid indices, 1 otherwise
            shift_right = np.ones(n)
            if shift < n:
                shift_right[shift:] = 1 - h[shift:]
            # shift_left: h[i - shift] for valid indices, 0 otherwise
            shift_left = np.zeros(n)
            if shift < n:
                shift_left[shift:] = h[:n - shift]
        else:
            s = -shift
            # shift_right: (1 - h[i - s]) for valid indices, 1 otherwise
            shift_right = np.ones(n)
            if s < n:
                shift_right[s:] = 1 - h[:n - s]
            # shift_left: h[i + s] for valid indices, 0 otherwise
            shift_left = np.zeros(n)
            if s < n:
                shift_left[:n - s] = h[s:]
        
        grad += w_k * (shift_right - shift_left)
    
    return grad * dx

def optimize_h(n, max_iters, init_lr=0.1, alpha_start=5, alpha_end=30):
    """Optimize h using gradient descent with smoothed max objective"""
    # Initialize h randomly and project to satisfy sum constraint
    h = np.random.rand(n)
    h = project_to_box_simplex(h, n / 2)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Optimization loop
    for it in range(max_iters):
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        
        # Compute gradient
        grad = compute_gradient(h, alpha)
        
        # Update step size with decay
        lr = init_lr / np.sqrt(it + 1)
        h = h - lr * grad
        
        # Project onto feasible region
        h = project_to_box_simplex(h, n / 2)
        
        # Evaluate current solution
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use parameters that work well for this problem
    n = 64  # number of points
    max_iters_per_restart = 2000
    num_restarts = 10  # number of different random restarts
    
    best_c5 = np.inf
    best_h = None
    
    # Outer loop for multiple restarts
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Optimize with specific random restart
        current_h, current_c5 = optimize_h(n, max_iters_per_restart)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Early stopping if we already have a good bound
        if best_c5 < 0.379:
            break
    
    # Final optimization of best solution
    if best_h is not None:
        final_h, final_c5 = optimize_h(n, 1000, init_lr=0.05, alpha_start=30, alpha_end=50)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
            
    # Evaluate final result to be sure
    if best_h is not None:
        actual_c5 = evaluate_c5(best_h)
        return (best_h.tolist(), float(actual_c5), n)
    else:
        # Fallback to provide any reasonable solution
        h_fallback = np.full(n, 0.5)
        c5_fallback = evaluate_c5(h_fallback)
        return (h_fallback.tolist(), float(c5_fallback), n)
