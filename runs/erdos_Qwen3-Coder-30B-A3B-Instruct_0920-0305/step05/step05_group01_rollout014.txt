TARGET_SUM = 60.0

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def random_move(h, rng):
    """Generate a random move"""
    n = len(h)
    i = rng.choice(n)
    # Adjust a small portion of h
    delta = rng.uniform(-0.1, 0.1)
    h_new = h.copy()
    h_new[i] += delta
    return project_to_sum_box(h_new, TARGET_SUM)

    n = len(h)
    
    # Look for correlations between points
    corr = np.correlate(h, h, mode='full')
    max_corr_lag = np.argmax(corr) - (len(h) - 1)
    
    # Try to modify elements near the most correlated positions
    if abs(max_corr_lag) > 0:
        # Try to make the correlation more negative
        i = rng.choice(n)
        j = (i + max_corr_lag) % n
        delta = rng.uniform(-0.05, 0.05)
        
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        return project_to_sum_box(h_new, TARGET_SUM)
    else:
        # Fallback to regular random move
        return random_move(h, rng)
