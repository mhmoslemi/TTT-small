def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_onto_simplex(v, target_sum):
    """Project vector v onto the capped simplex with sum = target_sum"""
    # Use binary search to find the correct lambda
    v = np.array(v)
    low, high = np.min(v - 1), np.max(v)
    tolerance = 1e-10
    max_iter = 100
    for _ in range(max_iter):
        mid = (low + high) / 2
        projected = np.clip(v - mid, 0, 1)
        current_sum = np.sum(projected)
        if abs(current_sum - target_sum) < tolerance:
            return projected
        elif current_sum > target_sum:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) / 2, 0, 1)

def compute_subgradient(h, max_shifts):
    """Compute subgradient of C5 at h using active shifts"""
    n = len(h)
    # Compute gradients for all active shifts
    # We use the formula ∂O_k/∂h_i = 1 - h_{i+k} - h_{i-k}
    gradients = np.zeros(n)
    for k in max_shifts:
        g = np.zeros(n)
        if k >= 0:
            # g[i] = 1 - h[i+k] - h[i-k] for i+k<n and i-k>=0
            g[k:] = 1.0
            if k > 0:
                g[:-k] -= h[k:]
            if k < n-1:
                g[k:] -= h[:-k]
        else:
            # k < 0: g[i] = 1 - h[i+k] - h[i-k] where we ensure bounds
            g[:-k] = 1.0
            if k < 0:
                g[-k:] -= h[:k]
            if k > -(n-1):
                g[:-k] -= h[-k:]
        gradients += g
    return gradients / len(max_shifts)
