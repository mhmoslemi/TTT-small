import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time
from scipy.optimize import minimize


def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx


def project_simplex(v, s=1.0):
    """Project vector v onto the capped simplex with sum = s."""
    v = np.array(v)
    n = len(v)
    # Sort v in descending order
    u = np.sort(v)[::-1]
    # Compute cumulative sums
    cu = np.cumsum(u)
    # Find index k such that u_k >= (sum(u) - s) / k
    k = np.where(u >= (cu - s) / np.arange(1, n + 1))[0]
    if len(k) == 0:
        k = n - 1
    else:
        k = k[-1]
    # Compute threshold
    theta = (cu[k] - s) / (k + 1)
    # Compute projection
    return np.clip(v - theta, 0, 1)


def binary_milp_optimize(n, target_area, time_limit=150):
    """Solve the binary MILP formulation for the Erdos problem."""
    # Create variables
    h = Variable(n, boolean=True)
    tau = Variable()

    # Define all possible shifts
    shifts = list(range(-n + 1, n))  # from -(n-1) to n-1
    # For each shift k, we need to compute the set of valid pairs
    y_vars = {}
    constraints = [sum(h) == target_area]

    # Add constraints for each shift
    for k in shifts:
        i_start = max(0, -k)
        i_end = min(n, n - k)
        shift_pairs = [(i, i + k) for i in range(i_start, i_end)]

        # Create auxiliary variables
        shift_y_vars_list = []
        for i, j in shift_pairs:
            y = Variable(boolean=True)
            y_vars[(i, j, k)] = y
            shift_y_vars_list.append(y)

        # Add McCormick constraints
        for i, j in shift_pairs:
            y = y_vars[(i, j, k)]
            constraints.append(y <= h[i])
            constraints.append(y <= h[j])
            constraints.append(y >= h[i] + h[j] - 1)

        # Compute the overlap O_k for shift k
        overlap_expr = sum(h) - sum(shift_y_vars_list)
        constraints.append(overlap_expr <= tau)

    constraints.append(tau >= 0)

    # Define objective - minimize tau
    objective = Minimize(tau)
    prob = Problem(objective, constraints)

    # Solve
    try:
        prob.solve(solver='CBC', time_limit=time_limit, verbose=False)
    except:
        pass

    if prob.status == 'optimal' or prob.status == 'optimal_inaccurate':
        h_sol = [float(h[i].value) for i in range(n)]
        return h_sol

        return h


def evaluate_overlaps(h):
    """Compute all overlap values O_k = sum(h_i) - sum(h_i * h_{i+k}) """
    n = len(h)
    dx = 2.0 / n
    corr_h = correlate(h, 1 - h, mode="full")
    overlaps = corr_h * dx
    return overlaps


def hill_climb_binary(h_start, max_iter=2000):
    """Perform hill-climbing on binary patterns using swaps."""
    h = np.array(h_start)
    n = len(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Precompute initial overlaps
    overlaps = evaluate_overlaps(best_h)
    max_overlap = np.max(overlaps)
    
    for iter in range(max_iter):
        better_found = False
        # Try all swaps (i,j) where h[i] = 1 and h[j] = 0
        for i in np.where(h == 1)[0]:
            for j in np.where(h == 0)[0]:
                # Swap
                h[i] = 0
                h[j] = 1
                
                # Evaluate new overlaps
                new_overlaps = evaluate_overlaps(h)
                new_max = np.max(new_overlaps)
                
                if new_max < max_overlap:
                    max_overlap = new_max
                    best_h = h.copy()
                    better_found = True

                # Swap back
                h[i] = 1
                h[j] = 0
                
                if better_found:
                    break
            if better_found:
                break
        
        if not better_found:
            break
            
    return best_h.tolist(), evaluate_c5(best_h)


def gradient_step_tight(h, eta=0.5, max_iter=200):
    """Apply gradient descent to minimize C5 (non-smoothed objective)."""
    n = len(h)
    dx = 2.0 / n
    h = np.array(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    for i in range(max_iter):
        # Compute all overlaps
        overlaps = evaluate_overlaps(h)
        max_idx = np.argmax(overlaps)
        max_val = overlaps[max_idx]
        
        # Compute sub-gradient indices
        k = max_idx - (n - 1)
        # We associate the weight vector w where:
        # w[r] = 1 if there is a pair (r, r+k) and r+k in [0,n-1]
        w = np.zeros(n)
        i1 = max(0, -k)
        i2 = min(n, n - k)
        if k >= 0:
            w[i1:i2] = 1
        else:
            w[-k:n + k] = 1
        # Normalize such that Σ w = n/2
        w = w * (n / 2) / np.sum(w)
        
        # Compute full gradient
        g = dx * (np.ones(n) - correlate(h, w, mode='full')[n - 1:] - correlate(h, w[::-1], mode='full')[n - 1:])
        # Take gradient step
        h_new = h - eta * g
        h_new = project_simplex(h_new, n / 2)
        
        # Symmetrize
        h_sym = (h_new + h_new[::-1]) / 2
        h_sym = project_simplex(h_sym, n / 2)
        
        c5_new = evaluate_c5(h_sym)
        if c5_new < best_c5:
            best_h = h_sym.copy()
            best_c5 = c5_new
            h = h_sym.copy()
        else:
            # Reduce learning rate if no progress
            eta = eta * 0.5
            if eta < 1e-10:
                break
    
    return best_h.tolist(), best_c5

            h_binary = h_refined
