```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use a hybrid optimization approach
    # Phase 1: Use global optimization with multiple strategies
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Strategy 1: Differential evolution on a coarse grid
    try:
        # Start with a coarse grid to get a good initial estimate
        res = differential_evolution(
            objective, 
            bounds, 
            maxiter=100, 
            popsize=15,
            seed=seed,
            disp=False
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Strategy 2: Local search with adaptive perturbations
    local_restart_bound = 2000
    
    for restart in range(local_restart_bound):
        if (time.time() - start_time) > budget_s * 0.95:  # Leave time for final steps
            break
            
        # Start with small random perturbations
        perturb_factor = 0.05 * (1.0 + 0.1 * np.random.random())
        new_h = best_h + np.random.normal(0, perturb_factor, n_points)
        new_h = np.clip(new_h, 0, 1)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-13, 'gtol': 1e-13}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue

    # Strategy 3: Pattern-based exploration with refactorization
    if (time.time() - start_time) < budget_s * 0.8:
        try:
            # Try a more regularized wave-like pattern
            x = np.linspace(0, 2, n_points)
            # wave pattern - make it more peaked at boundaries
            wave_pattern = np.sin(x * np.pi) * 0.5 + 0.5
            # Normalize
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(wave_pattern) * dx)
            wave_pattern = wave_pattern * scale_factor
            
            # Mix with best current solution
            mixed_h = 0.7 * best_h + 0.3 * wave_pattern
            
            # Normalize again
            scale_factor = 1.0 / (np.sum(mixed_h) * dx)
            mixed_h = mixed_h * scale_factor
            
            res = minimize(
                objective, 
                mixed_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
            
    # Strategy 4: A very focused local search with more iterations
    if (time.time() - start_time) < budget_s * 0.9:
        # Focus on the best solution found so far with more iteration
        try:
            res = minimize(
                objective, 
                best_h, 
                method='trust-constr', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 3000, 'gtol': 1e-14, 'ftol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                dx = 2.0 / n_points
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass

    return (best_h.tolist(), best_c5, n_points)
```
