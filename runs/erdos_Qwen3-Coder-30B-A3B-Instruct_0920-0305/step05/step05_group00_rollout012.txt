def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def get_worst_lag(h):
    """Find the lag that gives the maximum correlation value"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.argmax(corr)

def local_descent_move(h, worst_lag):
    """Make a small move that improves according to a local gradient estimate"""
    # Gradients for each position
    g = np.zeros(n_points)
    
    # Inefficient gradient calculation, but will be fast enough
    for i in range(n_points):
        if i - worst_lag >= 0:
            g[i] += h[i - worst_lag]
        if i + worst_lag < n_points:
            g[i] += h[i + worst_lag]
    
    # Find the best index to decrease and worst to increase
    idx_min = np.argmin(g)
    idx_max = np.argmax(g)
    
    if idx_min == idx_max:
        return None
    
    # Compute maximum possible transfer amount
    delta = min(h[idx_min], 1 - h[idx_max])
    if delta <= 0:
        return None
    
    return (idx_min, idx_max, delta)

def deterministic_move(h):
    """Make a deterministic move to improve the maximum correlation"""
    worst_lag = get_worst_lag(h)
    return local_descent_move(h, worst_lag)
