import numpy as np
import time

def project_capped_simplex(v, T):
    """Project v onto {x : 0 ≤ x ≤ 1, sum x = T}"""
    # binary search for lambda
    v = np.asarray(v)
    low, high = np.min(v - 1), np.max(v)
    tol = 1e-12
    for _ in range(100):
        if high - low < tol:
            break
        mid = (low + high) / 2
        projected = np.clip(v - mid, 0, 1)
        sum_proj = np.sum(projected)
        if sum_proj > T:
            low = mid
        else:
            high = mid
    lambda_ = (low + high) / 2
    return np.clip(v - lambda_, 0, 1)

def compute_O_and_grad(h, alpha):
    """Compute overlap values and their smoothed gradient"""
    n = len(h)
    dx = 2.0 / n
    # compute overlap values O_k for all k
    corr = np.correlate(h, 1 - h, mode='full')
    O = corr * dx  # shape (2*n - 1,)
    # compute softmax weights
    m = np.max(O)
    exp_O = np.exp(alpha * (O - m))
    w = exp_O / np.sum(exp_O)
    # compute gradient
    grad = np.zeros_like(h)
    # for each shift k, accumulate scaled gradient
    # O_k = dx * sum h[i] * (1 - h[i+k]) for valid i
    # d(O_k)/d(h[i]) = dx * (1 - h[i+k] - h[i-k])
    for k in range(len(O)):
        shift = k - (n - 1)
        if shift >= 0:
            # h[i] contributes to O_{n-1+k} (i-k >= 0)
            if k < len(O):
                grad += w[k] * dx * (1 - np.concatenate([h[shift:], np.zeros(shift)]))
        else:
            # shift < 0
            grad += w[k] * dx * (-np.concatenate([np.zeros(-shift), h[:shift]]))
    # Add the correction for (1 - h[i+k]) part
    # this leads to: h[i] -> dx * sum over k w_k * (1 - h[i+k]) term
    # and for -sum h[i-k] term: h[i] -> dx * sum over k w_k * (-h[i-k])
    # These are    O_k = dx * sum_i h[i] * (1 - h[i+k]) 
    # So the derivative d O_k / d h[i] = dx * (1 - h[i+k] - h[i-k])
    grad = np.zeros_like(h)
    # Efficient computation using correlation (or manual looping):
    for k in range(len(O)):
        shift = k - (n - 1)
        if shift >= 0:
            # shift by +shift
            h_shift = np.concatenate([h[shift:], np.zeros(shift)])
        else:
            h_shift = np.concatenate([np.zeros(-shift), h[:shift]])
        if shift >= 0:
            grad += w[k] * dx * (1 - h_shift)
        else:
            grad += w[k] * dx * (-h_shift)
    return O, grad

def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx
