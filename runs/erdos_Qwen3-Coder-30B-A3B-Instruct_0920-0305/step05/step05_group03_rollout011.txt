import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def project(v, S):
    """Project onto the capped simplex: 0 <= v_i <= 1, sum v_i = S"""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2
        p = np.clip(v + lam, 0.0, 1.0)
        s = p.sum()
        if s > S:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2, 0.0, 1.0)

def compute_O_w_grad(h, alpha, dx):
    """Compute overlaps O_r and gradient of log-sum-exp of overlaps"""
    n = len(h)
    c1 = correlate(h, np.ones(n), mode='full')
    c2 = correlate(h, h, mode='full')
    O = (c1 - c2) * dx  # Overlaps for all shifts (length 2n - 1)
    O_max = np.max(O)
    # Softmax weights
    w = np.exp(alpha * (O - O_max))
    w = w / np.sum(w)

    # Gradients
    s = np.convolve(w, np.ones(n), mode='valid')
    g2 = correlate(w, h, mode='full')[n-1:2*n-1]
    g3 = correlate(w, h[::-1], mode='full')[n-1:2*n-1][::-1]
    g = dx * (s - g2 - g3)
    return O, w, g

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

    np.random.seed(seed)
    start_time = time.time()

    n_points = 84  # use 84 points for balance of resolution and speed
    # Use initial construction (upsampled)

    h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0_raw)), h0_raw)
    S = n_points / 2.0
    h0 = project(h0, S)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Optional global search phase: Differential Evolution
    def de_objective(h):
        return evaluate_c5(h)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constrained_de_objective(h):
        h = project(h, S)
        return de_objective(h)
        
    # Lightweight DE run
    try:
        de_result = differential_evolution(
            constrained_de_objective,
            bounds,
            popsize=12,
            mutation=0.8,
            recombination=0.7,
            maxiter=30,
            seed=seed
        )
        if de_result.success:
            h_candidate = project(de_result.x, S)
            c5_candidate = evaluate_c5(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Local refinement with projected gradient descent
    # Use symmetry to reduce dimensionality
    h = best_h.copy()
    alpha = 5.0
    eta = 0.5 * (2.0 / n_points)
    iteration = 0
    max_iter = 5000
    
    while iteration < max_iter and (time.time() - start_time) < budget_s * 0.97:
        O, w, g = compute_O_w_grad(h, alpha, 2.0 / n_points)
        
        # Line search
        step_size = eta
        accept = False
        for factor in [1.0, 0.5, 0.25]:
            h_trial = project(h - factor * step_size * g, S)
            # Enforce symmetry
            h_trial = (h_trial + h_trial[::-1]) / 2
            h_trial = project(h_trial, S)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5:
                best_h, best_c5 = h_trial.copy(), c5_trial
                h = h_trial
                accept = True
                break
        if not accept:
            step_size *= 0.5
        
        # Anneal alpha every 200 iterations
        if iteration % 200 == 0:
            alpha = min(alpha * 1.5, 50.0)
        
        # Random perturbation every 500 iterations without improvement
        if iteration % 500 == 0 and not accept:
            h = project(h + 0.02 * np.random.randn(n_points), S)
            h = (h + h[::-1]) / 2
            h = project(h, S)
        
        iteration += 1
        
        if (time.time() - start_time) > budget_s * 0.95:
            break
    
    return (best_h.tolist(), best_c5, n_points)
