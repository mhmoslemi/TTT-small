    u = Variable(2 * L)  # [a1, ..., aL, b1, ..., bL]

    # Build the Fourier series: h(x) = 0.5 + sum_{l=1}^L (a_l cos(pi l x) + b_l sin(pi l x))
    # We need to build a matrix A such that A * u gives h(x) for each x_i
    A = np.zeros((M, 2*L))
    for i in range(M):
        for l in range(1, L+1):
            A[i, l-1] = np.cos(np.pi * l * x[i])  # a_l
            A[i, L+l-1] = np.sin(np.pi * l * x[i]) # b_l

    # 1. Box constraints: 0 <= h(x_j) <= 1
    h_vals = 0.5 + A @ u
    box_constraints = [h_vals >= 0, h_vals <= 1]

    soc_constraints = []

        b_l = u[L + l]
        soc_constraints.append( norm([2*a_l, 2*b_l, s[l] - 1]) <= s[l] + 1 )
