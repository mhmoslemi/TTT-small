import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def project(v, S):
    """Project v onto the capped simplex with sum = S"""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2
        p = np.clip(v + lam, 0.0, 1.0)
        s = p.sum()
        if s > S:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2, 0.0, 1.0)

def compute_O_w_grad(h, alpha):
    """Compute the vector of overlaps O_r, weights w, and gradient of F_alpha"""
    dx = 2.0 / len(h)
    c1 = correlate(h, np.ones(len(h)), mode='full')
    c2 = correlate(h, h, mode='full')
    O = (c1 - c2) * dx  # overlaps for all shifts r
    O_max = np.max(O)
    w = np.exp(alpha * (O - O_max))
    w = w / w.sum()
    s = np.convolve(w, np.ones(len(h)), mode='valid')
    g2 = correlate(w, h, mode='full')[len(h)-1:2*len(h)-1]
    g3 = correlate(w, h[::-1], mode='full')[len(h)-1:2*len(h)-1][::-1]
    grad = dx * (s - g2 - g3)
    return O, w, grad

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

        de_res = differential_evolution(

                best_h = h_trial.copy()

    except:
