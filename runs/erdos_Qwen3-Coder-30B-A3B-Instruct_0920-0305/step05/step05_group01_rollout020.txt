import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60
EPS = 1e-12

def project_to_sum_box(x, target_sum):
    """Project x onto box constraints [0,1] and sum = target_sum"""
    low, high = -2.0, 2.0
    for _ in range(50):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target_sum) < EPS:
            return y
        if s > target_sum:
            high = mid
        else:
            low = mid
    y = np.clip(x + low, 0.0, 1.0)
    return y

def evaluate_c5(h):
    """Evaluate C5 using correlation with dx scaling"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def gradient_smoothing(h, alpha):
    """Compute gradient of log-sum-exp smoothed objective"""
    n = len(h)
    dx = 2.0 / n
    
    # Compute correlation unscaled
    corr_full = np.correlate(h, 1.0 - h, mode='full')
    corr_full = corr_full * dx  # scale by dx
    
    # Compute softmax weights for smoothing
    I_max = corr_full.max()
    exp_vals = np.exp(alpha * (corr_full - I_max))
    w = exp_vals / exp_vals.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for idx, p_val in enumerate(w):
        lag = idx - (n - 1)
        if lag >= 0:
            # (1 - h[i+lag]) term
            if n - lag > 0:
                grad[:n-lag] += p_val * (1.0 - h[lag:])
            # edge contribution +1 for i where i+lag >= n
            if lag > 0:
                grad[n-lag:] += p_val * 1.0
            # -h[i-lag] term
            if lag > 0:
                grad[lag:] -= p_val * h[:n-lag]
            else:
                # lag == 0 case: -h[i] term (full subtraction)
                grad -= p_val * h
        else:
            # lag < 0: shift [-lag] right
            d = -lag
            # (1 - h[i-d]) term
            if n - d > 0:
                grad[d:] += p_val * (1.0 - h[:n-d])
            # edge contribution +1 for i where i+lag < 0
            grad[:d] += p_val * 1.0
            # -h[i+d] term
            grad[:n-d] -= p_val * h[d:]
    
    return grad

def descent_step(h, eta, alpha, target_sum):
    """Perform one descent step with line search"""
    grad = gradient_smoothing(h, alpha)
    # Try line search
    for _ in range(5):
        h_new = project_to_sum_box(h - eta * grad, target_sum)
        c5_new = evaluate_c5(h_new)
        c5_old = evaluate_c5(h)
        if c5_new <= c5_old:
            return h_new, c5_new
        eta *= 0.5
    # If line search fails, return original
    return h, evaluate_c5(h)

def gradient_descent(initial_h, time_limit, target_sum):
    """Run gradient descent optimization"""
    h = initial_h.copy()
    c5 = evaluate_c5(h)
    best_h_local = h.copy()
    best_c5_local = c5
    
    t0 = time.time()
    it = 0
    alpha_start = 0.5
    alpha_end = 50.0
    eta_start = 0.1
    
    while time.time() - t0 < time_limit and it < 5000:
        progress = min(1.0, it / 2000)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        eta = eta_start * (0.995 ** it)
        
        h, c5 = descent_step(h, eta, alpha, target_sum)
        
        if c5 < best_c5_local:
            best_c5_local = c5
            best_h_local = h.copy()
        
        it += 1
    
    return best_h_local, best_c5_local

def local_refine(h, target_sum):
    """Use SLSQP to refine solution"""
    def objective(x):
        return evaluate_c5(x)
        
    def constraint_eq(x):
        return np.sum(x) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h))]
    
    try:
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, target_sum)
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
    except:
        pass
    return h, evaluate_c5(h)

    h0 = project_to_sum_box(h0, target_sum)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    restart_count = 0
    time_remaining = budget_s * 0.95
    
    while time.time() - start_time < budget_s * 0.95 and restart_count < 8:
        time_per_restart = min(120, time_remaining / (8 - restart_count))
        if time_per_restart < 5:
            break
            
        # Create new initial point - random + scaled version of best
        h_start = h0 + np.random.normal(0, 0.02, n_points)
        h_start = project_to_sum_box(h_start, target_sum)
        
        # Run gradient descent
        h_opt, c5_opt = gradient_descent(h_start, time_per_restart, target_sum)
        
        # Refine with local optimization
        h_final, c5_final = local_refine(h_opt, target_sum)
        
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final
            
        restart_count += 1
        time_remaining = budget_s - (time.time() - start_time)
    
    # Final refinement
    h_final, c5_final = local_refine(best_h, target_sum)
    if c5_final < best_c5:
        best_h = h_final
        best_c5 = c5_final
    
    return (best_h.tolist(), float(best_c5), n_points)
