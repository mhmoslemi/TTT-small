```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Phase 1: Use differential evolution for global search
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Try different sampling strategies
    try:
        # Differential evolution with custom bounds
        res = differential_evolution(
            objective, 
            bounds, 
            maxiter=1000, 
            popsize=15,
            seed=seed,
            callback=lambda x, convergence: None,
            strategy='best1bin'
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Phase 2: Local optimization starting from current best
    max_local_iter = 1000
    local_iter = 0
    while (time.time() - start_time) < budget_s * 0.95 and local_iter < max_local_iter:
        local_iter += 1
        # Random perturbation
        new_h = best_h + np.random.normal(0, 0.02, n_points)
        new_h = np.clip(new_h, 0, 1)
        
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue

    # Phase 3: Try structured improvements  
    if (time.time() - start_time) < budget_s * 0.9:
        # Try a pattern-based improvement
        # Try to reduce peak values while maintaining integral
        try:
            # Try a "tent" shape with edge weighting
            x = np.linspace(0, 2, n_points)
            # Create a tent-like pattern (more concentrated at edges)
            tent_pattern = np.maximum(0, 1 - np.abs(x - 1) / 0.5)
            tent_pattern = tent_pattern / np.sum(tent_pattern) * n_points / 2.0  # Normalize
            
            # Mix with current best
            mixed_h = 0.7 * best_h + 0.3 * tent_pattern
            mixed_h = np.clip(mixed_h, 0, 1)
            
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(mixed_h) * dx)
            mixed_h = mixed_h * scale_factor
            
            res = minimize(
                objective, 
                mixed_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 200, 'ftol': 5e-13, 'gtol': 5e-13}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass

    # Phase 4: Final refinement
    if (time.time() - start_time) < budget_s * 0.95:
        try:
            # Use a more aggressive local search method
            res = minimize(
                objective, 
                best_h, 
                method='trust-constr', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'gtol': 1e-13, 'ftol': 1e-13}
            )
            if res.success:
                h_candidate = res.x
                dx = 2.0 / n_points
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass

    return (best_h.tolist(), best_c5, n_points)
```
