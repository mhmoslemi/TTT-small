import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project(v, S):
    """Euclidean projection onto the capped simplex"""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2
        p = np.clip(v + lam, 0.0, 1.0)
        s = p.sum()
        if s > S:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2, 0.0, 1.0)

def compute_O_w_grad(h, alpha):
    """Compute overlaps, weights, and gradient"""
    n = len(h)
    dx = 2.0 / n
    c1 = correlate(h, np.ones(n), mode='full')
    c2 = correlate(h, h, mode='full')
    O = (c1 - c2) * dx
    O_max = np.max(O)
    w = np.exp(alpha * (O - O_max))
    w = w / w.sum()
    
    s = np.convolve(w, np.ones(n), mode='valid')
    g2 = correlate(w, h, mode='full')[n-1:2*n-1]
    g3 = correlate(w, h[::-1], mode='full')[n-1:2*n-1][::-1]
    g = dx * (s - g2 - g3)
    return O, w, g

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 84  # Better resolution than original 42

    # Upsample to n_points
    x_orig = np.linspace(0, 2, len(h0))
    x_new = np.linspace(0, 2, n_points)
    h0 = np.interp(x_new, x_orig, h0)
    # Project to feasible set
    S = n_points / 2.0
    h0 = project(h0, S)
    
    # Best solution
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Global search using differential evolution
    def objective(x):
        x = project(x, S)
        return evaluate_c5(x)
    
    bounds = [(0, 1) for _ in range(n_points)]
    try:
        res = differential_evolution(objective, bounds, maxiter=30, popsize=12, 
                                    mutation=0.8, recombination=0.7, seed=seed)
        if res.success:
            h_candidate = project(res.x, S)
            c5_candidate = evaluate_c5(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Local refinement phase - projected gradient descent
    h = best_h.copy()
    alpha = 5.0
    eta = 0.5 * (2.0 / n_points)
    step_size_decay = 0.5
    iteration = 0
    
    while time.time() - start_time < 0.97 * budget_s:
        # Compute gradient
        O, w, g = compute_O_w_grad(h, alpha)
        
        # Line search
        for factor in [1.0, 0.5, 0.25]:
            h_trial = project(h - factor * eta * g, S)
            # Apply symmetry
            h_trial = (h_trial + h_trial[::-1]) / 2
            h_trial = project(h_trial, S)
            c5_val = evaluate_c5(h_trial)
            if c5_val < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_val
                h = h_trial.copy()
                break
        else:
            eta *= step_size_decay
        
        # Annealing
        if iteration % 200 == 0:
            alpha = min(alpha * 1.5, 50.0)
        
        # Random perturb
        if iteration % 500 == 0 and not (time.time() - start_time > 0.95 * budget_s):
            h = project(h + 0.02 * np.random.randn(n_points), S)
            h = (h + h[::-1]) / 2
            h = project(h, S)
        
        iteration += 1
        if eta < 1e-12:
            break
            
    return (best_h.tolist(), best_c5, n_points)
