import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def project_simplex(h, target_sum):
    """Project h onto the simplex with sum = target_sum and bounds [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    
    # Use bisection to find lambda such that sum(clip(h + lambda, 0, 1)) = target_sum
    l, u = -1.0, 1.0
    for _ in range(30):  # Max iterations
        lambda_ = (l + u) / 2.0
        h_proj = np.clip(h + lambda_, 0, 1)
        new_sum = np.sum(h_proj)
        if new_sum > target_sum:
            u = lambda_
        else:
            l = lambda_
    
    return np.clip(h + (l + u)/2.0, 0, 1)

def compute_correlations(h):
    """Compute auto-correlations (excluding zero shift) using FFT"""
    n = len(h)
    # Zero-pad for linear convolution
    h_pad = np.pad(h, (0, n-1), 'constant')
    corr_full = np.convolve(h_pad, h, mode='valid')[:n]
    return corr_full[1:]  # Exclude shift 0

def compute_gradient(h, c, beta):
    """Compute gradient of soft-min objective using FFT convolution"""
    n = len(h)
    # Compute weights
    c_max = np.max(c)
    exp_terms = np.exp(-beta * (c - c_max))
    w = exp_terms / np.sum(exp_terms)
    
    # Compute gradient using two convolutions: h * w and h * reverse(w)
    w_pad = np.pad(w, (0, n-1), 'constant')
    grad1 = np.convolve(h, w_pad[:n], mode='valid')
    grad2 = np.convolve(h, w_pad[-n:], mode='valid')
    grad = -(grad1 + grad2)
    return grad

def compute_phi_beta(c, beta):
    """Compute the smooth surrogate value"""
    c_max = np.max(c)
    exp_terms = np.exp(-beta * (c - c_max))
    log_sum = np.log(np.sum(exp_terms)) + c_max * beta
    return log_sum / beta

def compute_C5_exact(h):
    """Compute the true C5 using max of correlations"""
    corr = correlate(h, 1-h, mode="full")
    max_corr = np.max(corr[len(h)-1:])
    return max_corr * (2.0 / len(h))
