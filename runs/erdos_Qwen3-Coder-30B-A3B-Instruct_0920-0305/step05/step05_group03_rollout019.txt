def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_simplex(v, target_sum):
    """Project vector v onto the simplex with sum = target_sum"""
    # Sort v in descending order
    v_sorted = np.sort(v)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(v_sorted)
    # Find the threshold
    n = len(v)
    rho = np.argmax(v_sorted - (target_sum - cumsum) / np.arange(1, n+1) > 0)
    theta = (target_sum - cumsum[rho]) / (rho + 1)
    # Compute projection
    result = np.maximum(v + theta, 0)
    # Ensure sum constraint
    scale = target_sum / np.sum(result)
    return result * scale
