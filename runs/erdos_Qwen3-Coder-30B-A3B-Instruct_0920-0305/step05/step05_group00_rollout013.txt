TARGET_SUM = None

def compute_corr_direct(h):
    """Compute correlation directly using numpy correlate"""
    h_rev = h[::-1]
    b = 1 - h
    corr = np.correlate(h_rev, b, mode="full")
    return corr

    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):

    corr = compute_corr_direct(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):

    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result
