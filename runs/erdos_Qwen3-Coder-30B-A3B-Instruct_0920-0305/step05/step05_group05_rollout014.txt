import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import differential_evolution
import time

def project_simplex(y, T):
    """Project y onto the capped simplex with sum = T using binary search."""
    y = np.asarray(y)
    lower, upper = 0.0, 1.0
    while upper - lower > 1e-12:
        mid = (lower + upper) / 2
        z = np.clip(y - mid, 0, 1)
        if np.sum(z) > T:
            lower = mid
        else:
            upper = mid
    return np.clip(y - lower, 0, 1)

def evaluate_c5_fft(h):
    """Fast evaluation of C5 using FFT"""
    n = len(h)
    dx = 2.0 / n
    # Normalize to sum = n/2
    scale = n / (2.0 * np.sum(h))
    h = h * scale
    
    # Compute full correlation via FFT
    H = rfft(h)
    G = rfft(1 - h)
    corr = irfft(H * np.conj(G))
    # Take only the valid correlations (full mode)
    corr = corr[:n]  # Only first n values (correlations for k=0 to n-1, then wrap)
    
    # Maximum is in the middle since h is symmetric about center due to our setup
    max_corr = np.max(corr)
    
    # scale the result
    return max_corr * dx

def repair_h(c_vals, knots_x, n):
    """Interpolate control points and project onto simplex and box"""
    # Linear interpolation from control points to full grid
    h_full = np.interp(np.linspace(0, 2, n), knots_x, c_vals)
    
    # Clip to [0,1] first
    h_full = np.clip(h_full, 0, 1)
    
    # Project onto simplex (sum = n/2)
    h_full = project_simplex(h_full, n / 2)
    
    return h_full

def cmaes_on_control_points(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 128
    K = 12  # number of control points
    knots_x = np.linspace(0, 2, K)
    T = n / 2  # target sum
    
    # Objective function
    def obj_func(c_vals):
        if time.time() - start_time > budget_s * 0.95:
            return float('inf')
        
        h_full = repair_h(c_vals, knots_x, n)
        return evaluate_c5_fft(h_full)
    
    # Initial guess: flat
    initial_mean = np.full(K, 0.5)
    
    # CMA-ES parameters
    lambda_ = 4 + 3 * int(np.log(K))  # Population size
    mu = lambda_ // 2  # Number of selected parents
    sigma = 0.2
    alpha = 5.0  # initial soft-max parameter
    alpha_max = 30.0
    alpha_schedule = 0.0  # will increase linearly
    
    # Best solution tracking
    best_val = float('inf')
    best_h_full = None
    
    # Iteration tracking
    it = 0
    max_iterations = 1500
    
    # CMA-ES main loop
    while (time.time() - start_time) < budget_s * 0.95 and it < max_iterations:
        # Generate offspring using CMA-ES like method
        offspring = []
        for _ in range(lambda_):
            if time.time() - start_time > budget_s * 0.95:
                break
            # Sample from multivariate normal
            z = np.random.randn(K)
            c_offspring = initial_mean + sigma * z
            offspring.append(c_offspring)
        
        # Evaluate offspring
        fitnesses = []
        for c in offspring:
            f = obj_func(c)
            fitnesses.append(f)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        selected = [offspring[i] for i in sorted_indices[:mu]]
        fitness_selected = [fitnesses[i] for i in sorted_indices[:mu]]
        
        # Update best solution
        if np.min(fitnesses) < best_val:
            best_val = np.min(fitnesses)
            # Retrieve the best h_full
            idx = np.argmin(fitnesses)
            h_best = repair_h(offspring[idx], knots_x, n)
            best_h_full = h_best.copy()
        
        # Adapt mean and covariance
        # Change of basis using exponential map
        z_selected = [np.random.randn(K) for _ in range(mu)]
        h_selected = [repair_h(c, knots_x, n) for c in selected]
        
        # Keep the CMA-ES simple for speed, adjust step size
        if len(h_selected) > 0:
            # Estimate changes to avoid gradient-based optimization
            # Just update mean
            initial_mean = np.mean(selected, axis=0)
            
        # Update step size
        sigma *= 0.95  # decay step size gradually
        
        # Update alpha (annealing)
        alpha_schedule += (alpha_max - alpha) / (0.7 * max_iterations)
        alpha = alpha + alpha_schedule
        it += 1
    
    # Postprocessing with subgradient descent
    if best_h_full is not None:
        h_best = best_h_full.copy()
        # Quick gradient step for polishing  
        for i in range(100):  # few iterations only
            if time.time() - start_time > budget_s:
                break
            # Compute actual correlations to find max overlap
            corr = np.correlate(h_best, 1 - h_best, mode='full')
            k_max = np.argmax(corr)
            max_val = corr[k_max]
            # Gradient like step to reduce that maximum overlap
            h_grad = np.zeros_like(h_best)
            if k_max < len(h_best):
                for i in range(len(h_best)):
                    if i < len(h_best) - k_max:
                        h_grad[i] += (1 - h_best[i + k_max]) 
                    if i + k_max < len(h_best):
                        h_grad[i] -= h_best[i + k_max]
            # Update
            h_best = h_best - 0.02 * h_grad
            h_best = np.clip(h_best, 0, 1)
            h_best = project_simplex(h_best, n/2)
        
        # Final evaluation
        best_val = evaluate_c5_fft(h_best)
    
    return (best_h_full.tolist(), best_val, n)

def run(seed=42, budget_s=600, **kwargs):
    return cmaes_on_control_points(seed, budget_s)
