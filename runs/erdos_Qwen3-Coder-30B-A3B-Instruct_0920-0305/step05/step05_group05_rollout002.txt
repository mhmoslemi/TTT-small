import numpy as np
import time

def project_capped_simplex(v, T):
    """Project vector v onto capped simplex: 0 <= x <= 1, sum(x) = T"""
    v = np.asarray(v)
    lmin, lmax = 0.0, 1.0
    while lmin < lmax - 1e-10:
        l = (lmin + lmax) / 2.0
        projected = np.clip(v - l, 0.0, 1.0)
        if projected.sum() > T:
            lmin = l
        else:
            lmax = l
    return np.clip(v - lmin, 0.0, 1.0)

def compute_O_and_grad(h, alpha):
    """Compute overlap vector O_k and its gradient w.r.t. h"""
    dx = 2.0 / len(h)
    # Correlation: O_k = dx * sum(h[i] * (1 - h[i+k]))
    corr_full = np.correlate(h, 1 - h, mode='full') * dx
    O = corr_full[len(h)-1:]  # Only positive shifts
    max_O = O.max()
    # Stable softmax weights
    exp_O = np.exp(alpha * (O - max_O))
    w = exp_O / exp_O.sum()
    
    # Construct gradient from each shift
    grad = np.zeros_like(h)
    for k in range(len(O)):
        if k == 0:
            grad += w[k] * (1 - h)  # No shift needed
        else:
            # h shifted forward (k positions)
            h_forward = np.concatenate([h[k:], np.zeros(k)])
            # h shifted backward
            h_backward = np.concatenate([np.zeros(k), h[:-k]])
            grad += w[k] * (1 - h_forward - h_backward)
    grad *= dx
    return O, grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 64
    dx = 2.0 / n
    T = n / 2  # integral constraint
    max_iters = 4000
    alpha_start = 5.0
    alpha_end = 30.0
    lr_0 = 0.5
    subgrad_iters = 500

    best_val = np.inf
    best_h = None

    # Seed types: simplest for diversity
    seed_types = ['uniform', 'sine', 'gaussian', 'triangular']

    restarts = 0
    while (time.time() - start_time) < budget_s * 0.95:
        restarts += 1
        # Choose a seed type
        seed_type = np.random.choice(seed_types)
        # Initialize h:
        if seed_type == 'uniform':
            h = np.random.rand(n)
        elif seed_type == 'sine':
            x = np.linspace(0, 2, n)
            h = 0.5 + 0.3 * np.sin(np.pi * x)
            h = np.clip(h, 0, 1)
        elif seed_type == 'gaussian':
            x = np.linspace(0, 2, n)
            h = np.exp(-0.5 * ((x - 1.0) / 0.2) ** 2)
            h = np.clip(h, 0, 1)
        elif seed_type == 'triangular':
            x = np.linspace(0, 2, n)
            h = 1.0 - np.abs(x - 1.0) * 2.0
            h = np.clip(h, 0, 1)

        # Project to satisfy integral constraint
        h = project_capped_simplex(h, T)

        # Smoothed gradient descent
        for t in range(1, max_iters + 1):
            if (time.time() - start_time) > budget_s * 0.95:
                break

            alpha = alpha_start + (alpha_end - alpha_start) * t / max_iters
            O, grad = compute_O_and_grad(h, alpha)
            
            lr = lr_0 / np.sqrt(t)
            h = h - lr * grad
            
            h = project_capped_simplex(h, T)

            # Periodic symmetry enforcement
            if t % 10 == 0:
                h = (h + h[::-1]) / 2.0
                h = project_capped_simplex(h, T)

            # Eval
            val = O.max()
            if val < best_val:
                best_val = val
                best_h = h.copy()

        # Subgradient polishing
        if best_h is not None:
            h_polish = best_h.copy()
            # Find active k (max overlap shift)
            corr_full = np.correlate(h_polish, 1 - h_polish, mode='full') * dx
            O_full = corr_full[len(h_polish)-1:]
            active_k = np.argmax(O_full)
            active_shift = active_k - (len(h_polish) - 1)

            # Run a few iterations of subgradient
            for i in range(1, subgrad_iters + 1):
                if (time.time() - start_time) > budget_s * 0.95:
                    break
                # Compute exact gradient for active shift k
                grad_exact = np.zeros_like(h_polish)
                k = active_k
                if k == 0:
                    grad_exact = 1 - h_polish  # No shift
                else:
                    h_forward = np.concatenate([h_polish[k:], np.zeros(k)])
                    h_backward = np.concatenate([np.zeros(k), h_polish[:-k]])
                    grad_exact = 1 - h_forward - h_backward
                grad_exact *= dx
                
                lr_mod = 0.02 / np.sqrt(i)
                h_polish = h_polish - lr_mod * grad_exact
                h_polish = project_capped_simplex(h_polish, T)

                val = O_full[active_k]  # The true O_k we are minimizing
                if val < best_val:
                    best_val = val
                    best_h = h_polish.copy()

    # Final evaluation
    if best_h is not None:
        corr_full = np.correlate(best_h, 1 - best_h, mode='full') * dx
        final_val = corr_full.max()
        return (best_h.tolist(), float(final_val), n)
    else:
        return (np.random.rand(n).tolist(), np.inf, n)
