import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def compute_correlation(h):
    """Compute the full correlation array for a binary vector."""
    n = len(h)
    corr = np.zeros(2*n - 1)
    for k in range(-n+1, n):
        if k >= 0:
            corr[k+n-1] = np.sum(h[:n-k] * h[k:])
        else:
            corr[k+n-1] = np.sum(h[-k:] * h[:n+k])
    return corr

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def compute_c5_from_corr(corr, dx):
    """Compute C5 from precomputed correlation."""
    return np.max(corr) * dx

def is_valid_solution(h, n_points):
    """Check that h is within bounds and sums correctly."""
    dx = 2.0 / n_points
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) * dx - 1.0) < 1e-10)

def generate_legendre_pattern(n):
    """Generate Legendre-based binary pattern if applicable."""
    p = n // 2
    if p < 2 or not all(p % i != 0 for i in range(2, int(p**0.5)+1)) or not is_prime(p):
        # fallback to random
        return None
    residues = set()
    for a in range(1, p):
        r = (a * a) % p
        residues.add(r)
    mask = np.zeros(p, dtype=int)
    for r in residues:
        mask[r] = 1
    # duplicate to full length
    h = np.concatenate([mask, mask])
    # adjust to have exactly n/2 ones
    ones_count = np.sum(h)
    if ones_count < p:
        # Add a one to make exactly p
        idx = np.where(h == 0)[0][0]
        h[idx] = 1
    elif ones_count > p:
        # Remove one to make exactly p  
        idx = np.where(h == 1)[0][0]
        h[idx] = 0
    return h

def is_prime(n):
    """Check if number is prime."""
    if n < 2:
        return False
    for i in range(2, int(n**0.5)+1):
        if n % i == 0:
            return False
    return True

def generate_random_balanced(n):
    """Generate a random balanced binary vector."""
    h = np.zeros(n, dtype=int)
    indices = np.random.permutation(n)
    h[indices[:n//2]] = 1
    return h

def compute_worst_shifts(corr, tol=1e-10):
    """Return indices of worst (maximum) correlation values."""
    max_corr = np.max(corr)
    mask = np.abs(corr - max_corr) < tol
    return np.where(mask)[0] - len(corr)//2

def efficient_update_correlation(corr, h, i, j, n):
    """Update correlation after a swap at positions i and j."""
    # corr is length 2*n-1, indices in [-n+1, n-1]
    old_corr = corr.copy()
    
    # Affects only lags that involve i or j
    # For x = i, see which lags are affected
    for m in range(-n+1, n):
        if m >= 0:
            if i >= m and i < n - m:
                # both i and i+m are visible
                old_val = h[i-m] * h[i]
                new_val = h[i-m] * (1 - h[i])
                old_corr[m+n-1] += new_val - old_val
        else:
            # m < 0, i+m may not be valid
            if i >= 0 and i+m < n:
                old_val = h[i] * h[i+m]
                new_val = (1 - h[i]) * h[i+m]
                old_corr[m+n-1] += new_val - old_val

    # For x = j
    for m in range(-n+1, n):
        if m >= 0:
            if j >= m and j < n - m:
                old_val = h[j-m] * h[j]
                new_val = h[j-m] * (1 - h[j])
                old_corr[m+n-1] += new_val - old_val
        else:
            if j >= 0 and j+m < n:
                old_val = h[j] * h[j+m]
                new_val = (1 - h[j]) * h[j+m]
                old_corr[m+n-1] += new_val - old_val

    return old_corr

def binary_to_float(h):
    """Convert binary vector to balanced float vector."""
    h = h.astype(float)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters for annealing
    T0 = 0.05
    Tf = 1e-5
    N_iter = 2 * 10**6
    cooling_rate = (Tf / T0) ** (1.0 / N_iter)
    
    best_h = None
    best_c5 = float('inf')

                    valid_zero_positions = zero_positions
