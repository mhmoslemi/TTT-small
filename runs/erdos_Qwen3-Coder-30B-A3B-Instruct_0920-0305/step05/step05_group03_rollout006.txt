import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Phase 1: Use global optimization with more extensive search
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Try several optimization methods with different settings
    methods = [
        ('differential_evolution', {'maxiter': 2000, 'popsize': 15, 'tol': 1e-12}),
        ('L-BFGS-B', {'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-15}),
        ('TNC', {'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-15})
    ]
    
    for method_name, method_opts in methods:
        if (time.time() - start_time) > budget_s * 0.8:
            break
            
        try:
            if method_name == 'differential_evolution':
                res = differential_evolution(
                    objective, 
                    bounds, 
                    **method_opts,
                    seed=seed
                )
                if res.success:
                    h_candidate = res.x
                    dx = 2.0 / n_points
                    scale = 1.0 / (np.sum(h_candidate) * dx)
                    h_candidate = h_candidate * scale
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
            else:
                res = minimize(
                    objective, 
                    h0, 
                    method=method_name, 
                    bounds=bounds, 
                    constraints=cons,
                    options=method_opts
                )
                if res.success:
                    h_candidate = res.x
                    dx = 2.0 / n_points
                    scale = 1.0 / (np.sum(h_candidate) * dx)
                    h_candidate = h_candidate * scale
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
        except Exception as e:
            continue
    
    # Phase 2: Additional fine-tuning with random restarts
    local_restart_bound = 500
    for restart in range(local_restart_bound):
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
        # Start with small random perturbations around previous best
        new_h = best_h + np.random.normal(0, 0.005, n_points)  # Even smaller perturbation
        new_h = np.clip(new_h, 0, 1)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                dx = 2.0 / n_points
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue
    
    # Phase 3: Try to improve with pattern-based constructions if time allows
    if (time.time() - start_time) < budget_s * 0.8:
        try:
            # Try a combination of different patterns that have shown efficacy:
            # 1. Split into two parts with different heights 
            # 2. Add a middle spike
            # 3. Mix of bell-shaped patterns
            x = np.linspace(0, 2, n_points)
            
            # Create two peaks (longerwave pattern)
            peak1 = np.exp(-((x - 0.5)**2) / 0.05) 
            peak2 = np.exp(-((x - 1.5)**2) / 0.05)
            combined_peaks = peak1 + peak2
            
            # Normalize to integrate to 1
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(combined_peaks) * dx)
            combined_peaks = combined_peaks * scale_factor
            
            # Try mixing with best current solution
            mixed_h = 0.6 * best_h + 0.4 * combined_peaks
            
            # Normalize
            scale_factor = 1.0 / (np.sum(mixed_h) * dx)
            mixed_h = mixed_h * scale_factor
            
            res = minimize(
                objective, 
                mixed_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-15}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass

    # Final refinement with trust-constr
    if (time.time() - start_time) < budget_s * 0.95:
        try:
            res = minimize(
                objective,
                best_h,
                method='trust-constr',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 2000, 'gtol': 1e-15, 'ftol': 1e-15}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass

    return (best_h.tolist(), best_c5, n_points)
