import numpy as np
import time

def project_capped_simplex(v, T):
    """Project vector v onto the capped simplex: 0 <= v_i <= 1, sum v_i = T"""
    v = np.array(v)
    # Binary search for the Lagrange multiplier
    lmbda_min, lmbda_max = -1e5, 1e5
    for _ in range(50):  # Typically converges in 30-40 steps
        lmbda = (lmbda_min + lmbda_max) / 2
        proj = np.clip(v - lmbda, 0, 1)
        sum_proj = np.sum(proj)
        if sum_proj > T:
            lmbda_min = lmbda
        else:
            lmbda_max = lmbda
    lmbda = (lmbda_min + lmbda_max) / 2
    return np.clip(v - lmbda, 0, 1)

def compute_O_and_grad(h, alpha):
    """Compute overlap values and their smoothed gradient (w.r.t. h)"""
    n = len(h)
    dx = 2.0 / n
    # Compute overlaps for all shifts [0, 2n-2]
    corr = np.correlate(h, 1 - h, mode='full')
    # O_k values
    O = corr * dx
    
    # Softmax weights
    m = np.max(O)
    exp_O = np.exp(alpha * (O - m))
    w = exp_O / np.sum(exp_O)
    
    # Gradient of the smoothed objective
    grad = np.zeros(n)
    for k in range(len(O)):
        shift_right = max(0, k - (n - 1))
        shift_left  = max(0, (n - 1) - k)
        if k >= n - 1:
            # k >= n-1, so we are on the right shift
            grad += w[k] * (1 - np.roll(h, shift_left)) * dx
        else:
            # k < n-1, we are on the left shift
            grad -= w[k] * (np.roll(h, shift_right)) * dx
    
    return O, grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 64
    dx = 2.0 / n
    T = n / 2
    best_h = None
    best_C5 = float('inf')
    
    # Define several different initial construction strategies
    def initial_h_types():
        # Uniform random
        h1 = np.random.uniform(0, 1, n)
        h1 = project_capped_simplex(h1, T)
        
        # Sine shape
        x = np.linspace(0, 2, n)
        h2 = 0.5 + 0.4 * np.sin(np.pi * x)
        h2 = project_capped_simplex(h2, T)
        
        # Gaussian bump
        x = np.linspace(0, 2, n)
        mu = 1.0
        sigma = 0.2
        h3 = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
        h3 = project_capped_simplex(h3, T)
        
        # Triangular
        x = np.linspace(0, 2, n)
        h4 = np.maximum(0, 1.0 - np.abs(x - 1.0) * 2)
        h4 = project_capped_simplex(h4, T)
        
        # Binary checkerboard
        h5 = np.zeros(n)
        h5[::2] = 1.0
        h5 = project_capped_simplex(h5, T)
        
        return [h1, h2, h3, h4, h5]
    
    def evaluate_C5(h):
        # Compute the overlap integral for all shifts
        corr = np.correlate(h, 1 - h, mode="full")
        # The maximum overlap value
        max_corr = np.max(corr)
        return max_corr * dx
    
    max_iters = 4000
    alpha_start, alpha_end = 5.0, 30.0
    lr_0 = 0.5
    subgrad_iters = 500
    
    num_restarts = 0
    max_restarts = 50
    
    while num_restarts < max_restarts and (time.time() - start_time) < budget_s * 0.95:
        # Get random seed
        h_seeds = initial_h_types()
        h = h_seeds[np.random.randint(0, len(h_seeds))].copy()
        
        # Parametric gradient descent with smoothing parameter
        for t in range(max_iters):
            if (time.time() - start_time) > budget_s * 0.95:
                break
                
            # Annealing
            alpha = alpha_start + (alpha_end - alpha_start) * t / max_iters
            
            O, grad = compute_O_and_grad(h, alpha)
            step_size = lr_0 / np.sqrt(t + 1)
            h -= step_size * grad
            
            # Projection
            h = project_capped_simplex(h, T)
            
            # Every 10 steps, optionally symmetrize
            if t % 10 == 0:
                h = (h + h[::-1]) / 2.0
                h = project_capped_simplex(h, T)
                
            # Update best if found
            current_C5 = np.max(O) * dx
            if current_C5 < best_C5:
                best_h = h.copy()
                best_C5 = current_C5
                
        # Subgradient polish
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        max_shift = np.argmax(corr)
        O_active = corr * dx
        for t in range(subgrad_iters):
            if (time.time() - start_time) > budget_s * 0.95:
                break
                
            shift_k = max_shift
            shift_right = max(0, shift_k - (n - 1))
            shift_left = max(0, (n - 1) - shift_k)
            
            if shift_k >= n - 1:
                grad = (1 - np.roll(best_h, shift_left)) * dx
            else:
                grad = -np.roll(best_h, shift_right) * dx
                
            step_size = 0.02 / np.sqrt(t + 1)
            best_h -= step_size * grad
            best_h = project_capped_simplex(best_h, T)
            
            current_C5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx
            if current_C5 < best_C5:
                best_h = best_h.copy()
                best_C5 = current_C5
                
        num_restarts += 1
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
    if best_h is None:
        best_h = np.ones(n) * 0.5
        best_h = project_capped_simplex(best_h, T)
        best_C5 = evaluate_C5(best_h)
        
    return (best_h.tolist(), float(best_C5), n)
