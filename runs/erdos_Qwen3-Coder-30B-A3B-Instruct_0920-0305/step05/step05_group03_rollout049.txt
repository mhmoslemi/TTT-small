import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):

    K = [0, -2, -1, 1, 2]

    best_c5 = float('inf')

    solver = 'CBC'  # fallback if GLOP not available
    
    iteration = 0
    max_iterations = 1000  # prevent excessive iterations

    h0 = np.array(initial_h_values)

    while (time.time() - start_time) < budget_s * 0.95 and iteration < max_iterations:
        iteration += 1
        
        # Try to solve with the current active set of shifts

            h = Variable(n_points)

            constraints = []

            constraints += [h >= 0, h <= 1]

            constraints += [sum(h) == target_sum]
            
            # Symmetry constraints (element-wise enforcement)
            for i in range(n_points // 2):
                constraints += [h[i] == h[n_points - 1 - i]]
            
            # Initialize y_vars for this iteration
            y_vars = {}

            for k_index, k in enumerate(K):

                    shifted_indices = np.arange(k, n_points)

                    valid_indices = np.arange(-k, n_points)
                    shifted_indices = np.arange(0, n_points + k)
                    
                # Store y_vars for future reuse such that key (i, k) uniquely identifies a variable
                y = Variable(len(valid_indices))
                y_vars[(k, k_index)] = y
                    
                # Add McCormick constraints
                for idx, i in enumerate(valid_indices):
                    j = i + k
                    # y >= 0
                    constraints += [y[idx] >= 0]
                    # y <= h[i]  
                    constraints += [y[idx] <= h[i]]
                    # y <= h[j]
                    constraints += [y[idx] <= h[j]]
                    # y >= h[i] + h[j] - 1
                    constraints += [y[idx] >= h[i] + h[j] - 1]

                # Add constraint tau >= sum(h) - sum(y)
                sum_y = sum(y)
                constraints += [tau >= target_sum - sum_y]
            
            # Objective is to minimize tau
            prob = Problem(Minimize(tau), constraints)
            
            # Solve
            prob.solve(solver=solver, verbose=False)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_opt = h.value
                tau_opt = tau.value
                # Symmetrize if needed for cleanliness
                for i in range(n_points // 2):
                    h_opt[i] = (h_opt[i] + h_opt[n_points - 1 - i]) / 2
                    h_opt[n_points - 1 - i] = h_opt[i]
                h_opt = h_opt / np.sum(h_opt) * target_sum  # Normalize to sum = target_sum
                
                # Compute actual C5 value
                c5_actual = evaluate_c5(h_opt)

                        # Symmetrize if needed for cleanliness
                        for i in range(n_points // 2):
                            h_opt[i] = (h_opt[i] + h_opt[n_points - 1 - i]) / 2
                            h_opt[n_points - 1 - i] = h_opt[i]
                        h_opt = h_opt / np.sum(h_opt) * target_sum  # Normalize to sum = target_sum
                        
                        # Compute actual C5 value
                        c5_actual = evaluate_c5(h_opt)
                        if c5_actual < best_c5:
                            best_h = h_opt.copy()
                            best_c5 = c5_actual
                            if c5_actual < 0.379:
                                break
                except:
                    pass
        except Exception as e:
            pass
        
        # Evaluate true overlaps to get worst shift
        # Recompute overlaps over full range
        try:
            # Use the latest computed h_opt for eval if available
            if 'h_opt' in locals() and prob.status in ["optimal", "optimal_inaccurate"]:
                h_candidate = h_opt
                # Evaluate over all shifts to find max
                corr = correlate(h_candidate, 1 - h_candidate, mode="full")
                k_max = np.argmax(corr)
                max_val = corr[k_max] * dx
                
                # Record as best if better
                if max_val < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = max_val
                    if best_c5 < 0.379:
                        break
                    
                # Add this shift to the K
                # The actual shift in correlation is k_max - (n_points - 1), so compute correctly
                # correlate returns: index 0..2*n_points-2 = shift from -(n-1) to (n-1)
                actual_k = k_max - (n_points - 1)
                if actual_k not in K:
                    K.append(actual_k)
                    if len(K) > 50:  # Prevent explosion of K
                        K.sort(key=lambda x: abs(x))
                        K = K[:30]
                    # Prune K to reasonable size 50, using sorted abs
                    if len(K) > 50:
                        K = sorted(K, key=abs)[:50] 
                    
        except Exception as e:
            pass

    # Final polishing using projected gradient descent
    if best_h is None:
        # If no good solution, use initial h0
        best_h = h0.copy()
        best_c5 = evaluate_c5(best_h)
    
    # Handle case where best_h might not be properly symmetric
    for i in range(len(best_h) // 2):
        best_h[i] = (best_h[i] + best_h[n_points - 1 - i]) / 2
        best_h[n_points - 1 - i] = best_h[i]
    
    # Project to sum correctly
    best_h = best_h / np.sum(best_h) * target_sum
    
    # Now refine using gradient descent
    max_iter = 2000
    learning_rate = 0.5 * dx
    min_lr = 1e-7
    patience = 5

    h_refined = best_h.copy()
    current_c5 = evaluate_c5(h_refined)

        k = max_k - (n_points - 1) # Convert back to actual shift k

        # Compute gradient from the maximizer k
        ik_list = np.arange(0, n_points - abs(k))

        if k < 0:
            jk_list = np.arange(-k, n_points)
        else:
            jk_list = np.arange(0, n_points - k)
            
        # The gradient is 1 - (h[j] + h[i]), so we compute a vector
        grad = np.zeros(n_points)
        offset = 0 if k < 0 else k
        for i in range(len(ik_list)):
            j = jk_list[i] 
            grad[i] += 1 - (h_refined[j] + h_refined[i])  # In this case, we don't divide by num_max_k._)
            grad[j] += 1 - (h_refined[j] + h_refined[i])  # For i+k=j case, exact symmetry applies and we consider indexing carefully
            # Should ideally compute sum of derivative forms over the max_k

        # Simplified treatment of gradient approximation as sum over max_k string:
        grad = np.zeros(n_points)
        active_k = []
        if k >= 0:
            # Find matching pairs where i+k=j
            # Can also be: summed over pairs that gave max_k in corr_k where it contributed
            # That is more complex, but simpler version below will suffice:
            # What we compute is: for shift k, ∂/∂h[i] (1-sum h_m h_{m+k}) = sum of all terms ⇒ but it's randomized for mono shift approx, so
            # Just pick one problematic influence. 
            i_vals = np.arange(0, n_points-k)
            j_vals = np.arange(k, n_points)
            for i, j in zip(i_vals, j_vals):
                grad[i] += 1 - (h_refined[j] + h_refined[i])
                grad[j] += 1 - (h_refined[j] + h_refined[i])
        else:
            # k < 0
            i_vals = np.arange(-k, n_points)
            j_vals = np.arange(0, n_points+k)
            for i, j in zip(i_vals, j_vals):
                grad[i] += 1 - (h_refined[j] + h_refined[i])
                grad[j] += 1 - (h_refined[j] + h_refined[i])

        # descent step
        updated_h = h_refined - learning_rate * grad

        # Projection onto simplex (box + sum)
        # Cut off any values < 0 or > 1 and project onto sum = target_sum
        def project_to_simplex(v, target_sum):
            clamped = np.clip(v, 0, 1)
            total = np.sum(clamped)
            if abs(total - target_sum) < 1e-10:
                return clamped
            diff = target_sum - total
            # Smear diff evenly among smallest entries
            # Or use waterfilling
            # Simple heuristic: if diff > 0, increase low values; if < 0, decrease high values
            # Easier: set to 1,0 based on sign, use exact projection.
            if total < target_sum:
                # need more: positive
                sorted_indices = np.argsort(clamped)[::-1]
                needed = target_sum - total
                for i in sorted_indices:
                    increasable = target_sum - clamped[i]
                    to_fill = min(needed, increasable)
                    clamped[i] += to_fill
                    needed -= to_fill
                    if needed <= 0:
                        break
            elif total > target_sum:
                # need less: decrease values
                sorted_indices = np.argsort(clamped)
                needed = total - target_sum
                for i in sorted_indices:
                    decreasable = clamped[i]
                    to_reduce = min(needed, decreasable)
                    clamped[i] -= to_reduce
                    needed -= to_reduce
                    if needed <= 0:
                        break
            return clamped  

        h_refined = project_to_simplex(updated_h, target_sum)

        # Symmetrize again
        for i in range(n_points // 2):
            h_refined[i] = (h_refined[i] + h_refined[n_points - 1 - i]) / 2
            h_refined[n_points - 1 - i] = h_refined[i]
            
        h_refined = h_refined / np.sum(h_refined) * target_sum
        
        updated_c5 = evaluate_c5(h_refined)
        if updated_c5 < current_c5:
            current_c5 = updated_c5
            patience_count = 0
        else:
            patience_count += 1
            if patience_count >= patience:
                if learning_rate > min_lr:
                    learning_rate /= 2
                    patience_count = 0
                    # An attempt to make change in center of domain or center of history with local layers
                else:
                    break 

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_refined.copy()
            if best_c5 < 0.379:
                break

    # Optional binary rounding (for final check)
    h_binary = np.zeros(n_points)
    sorted_indices = np.argsort(best_h)[::-1]
    h_binary[sorted_indices[:int(target_sum)]] = 1.0
    h_binary = h_binary / np.sum(h_binary) * target_sum  # re-normalize to satisfies sum 
    c5_binary = evaluate_c5(h_binary)

    if c5_binary < best_c5:
        best_h = h_binary
        best_c5 = c5_binary

    return (best_h.tolist(), best_c5, n_points)
