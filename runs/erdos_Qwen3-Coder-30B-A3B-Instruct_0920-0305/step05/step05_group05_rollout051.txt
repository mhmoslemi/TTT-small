import numpy as np
import time

def project_to_box_simplex(v, target_sum):
    """Project vector v onto box [0,1] and simplex with sum = target_sum"""
    # Ensure we have the right sum
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0, 1)

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Use np.correlate to compute the overlap integral
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, alpha, dx):
    """Compute gradient of the smoothed maximum overlap"""
    n = len(h)
    conv = np.correlate(h, 1 - h, mode="full")
    O = conv * dx
    O_max = np.max(O)
    
    # Stable computation of weights
    exp_term = np.exp(alpha * (O - O_max))
    w = exp_term / np.sum(exp_term)
    
    grad = np.zeros_like(h)
    
    # Loop over all shifts
    for idx, shift in enumerate(range(-n+1, n)):
        w_k = w[idx]
        if w_k == 0:
            continue
        
        # Build shift_right and shift_left vectors
        if shift >= 0:
            # shift_right[i] = 1 if i+shift out of range, else 1 - h[i+shift]
            shift_right = np.ones(n)
            if shift < n:
                shift_right[shift:] = 1 - h[shift:]
            # shift_left[i] = h[i-shift] if i-shift >= 0, else 0
            shift_left = np.zeros(n)
            if shift < n:
                shift_left[shift:] = h[:n-shift]
        else:
            s = -shift
            # shift_right[i] = 1 if i-s < 0, else 1 - h[i-s]
            shift_right = np.ones(n)
            if s < n:
                shift_right[s:] = 1 - h[:n-s]
            # shift_left[i] = h[i+s] if i+s < n, else 0
            shift_left = np.zeros(n)
            if s < n:
                shift_left[:n-s] = h[s:]
        
        grad += w_k * (shift_right - shift_left)
    
    return grad * dx

def optimize_h(n, max_iters, alpha_start, alpha_end, lr0, init_kind="random"):
    """Optimize step function with gradient descent"""
    # Initialize h
    if init_kind == "random":
        h = np.random.rand(n)
    elif init_kind == "sine":
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.4 * np.sin(np.pi * x)
    elif init_kind == "step":
        h = np.zeros(n)
        mid = n // 2
        width = n // 4
        start = max(0, mid - width // 2)
        end = min(n, mid + width // 2)
        h[start:end] = 1.0
    elif init_kind == "gaussian":
        x = np.linspace(0, 2, n)
        mu = 1.0
        sigma = 0.2
        h = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    else:  # default to random- if not specified
        h = np.random.rand(n)
    
    # Project onto box and simplex
    target_sum = n / 2
    h = project_to_box_simplex(h, target_sum)
    
    best_c5 = np.inf
    best_h = h.copy()
    
    # Iterate with adaptive learning rate
    for it in range(max_iters):
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        lr = lr0 / np.sqrt(it + 1)
        
        # Compute gradient for smooth maximization
        grad = compute_gradient(h, alpha, 2.0/n)
        
        # Update step
        h = h - lr * grad
        
        # Project onto feasible set
        h = project_to_box_simplex(h, target_sum)
        
        # Evaluate objective
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 64  # Use moderate number of points
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    best_h = None
    best_c5 = np.inf
    
    # Restart loop with different strategies
    strategies = ["random", "sine", "step", "gaussian"]
    
    restarts = 0
    max_restarts = 50
    
    while restarts < max_restarts and (time.time() - start_time) < budget_s * 0.95:
        strategy = strategies[restarts % len(strategies)]
        h, c5 = optimize_h(
            n_points,
            max_iters=3000,
            alpha_start=5.0,
            alpha_end=30.0,
            lr0=0.5,
            init_kind=strategy
        )
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        restarts += 1
        
        # Early exit if we get good enough result
        if best_c5 < 0.379:
            break
        
        if (time.time() - start_time) > budget_s * 0.95:
            break
    
    return (best_h.tolist(), best_c5, n_points)
