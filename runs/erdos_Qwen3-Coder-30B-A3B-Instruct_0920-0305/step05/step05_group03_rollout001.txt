import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project(v, S):
    """Euclidean projection onto the capped simplex"""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2
        p = np.clip(v + lam, 0.0, 1.0)
        s = p.sum()
        if s > S:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2, 0.0, 1.0)

def compute_O_w_grad(h, alpha):
    """Compute the overlap vector O, weights w, and gradient g"""
    n = len(h)
    dx = 2.0 / n
    c1 = correlate(h, np.ones(n), mode="full")
    c2 = correlate(h, h, mode="full")
    O = (c1 - c2) * dx
    O_max = np.max(O)
    w = np.exp(alpha * (O - O_max))
    w = w / w.sum()
    s = np.convolve(w, np.ones(n), mode="valid")
    g2 = correlate(w, h, mode="full")[n-1:2*n-1]
    g3 = correlate(w, h[::-1], mode="full")[n-1:2*n-1][::-1]
    g = dx * (s - g2 - g3)
    return O, w, g

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup

    h0_upsampled = np.interp(np.linspace(0, 2, n_points), 
                            np.linspace(0, 2, len(h0)), h0)
    h0_projected = project(h0_upsampled, n_points / 2)
    
    best_h = h0_projected.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Global search with DE
    def de_objective(h):
        h_proj = project(h, n_points / 2)
        return evaluate_c5(h_proj)
    
    # DE parameters
    popsize = 12
    maxiter = 30
    mutation = 0.8
    recombination = 0.7
    
    # Run DE
    try:
        de_bounds = [(0, 1) for _ in range(n_points)]
        result = differential_evolution(
            de_objective,
            de_bounds,
            popsize=popsize,
            maxiter=maxiter,
            mutation=mutation,
            recombination=recombination,
            seed=seed,
            disp=False
        )
        if result.success:
            h_de = project(result.x, n_points / 2)
            c5_de = evaluate_c5(h_de)
            if c5_de < best_c5:
                best_h = h_de.copy()
                best_c5 = c5_de
    except:
        pass
    
    # Local refinement with gradient descent
    h_current = best_h.copy()
    alpha = 5.0
    eta = 0.5 * dx
    iteration = 0
    last_improvement = 0
    max_iterations = 5000
    
    while iteration < max_iterations and (time.time() - start_time) < budget_s * 0.97:
        O, w, g = compute_O_w_grad(h_current, alpha)
        improved = False
        
        # Line search
        for factor in [1.0, 0.5, 0.25]:
            h_trial = project(h_current - factor * eta * g, n_points / 2)
            # Symmetrize
            h_trial = (h_trial + h_trial[::-1]) / 2
            h_trial = project(h_trial, n_points / 2)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial
                h_current = h_trial.copy()
                improved = True
                last_improvement = iteration
                break
        
        if not improved:
            eta *= 0.5
        
        # Annealing
        if iteration % 200 == 0:
            alpha = min(alpha * 1.5, 50.0)
        
        # Random perturbation
        if iteration % 500 == 0 and iteration > last_improvement:
            perturb = 0.02 * np.random.randn(n_points)
            h_current = project(h_current + perturb, n_points / 2)
            h_current = (h_current + h_current[::-1]) / 2
            h_current = project(h_current, n_points / 2)
        
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
