import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def random_move(h, rng):
    """Generate a random feasible move"""
    i, j = rng.choice(n_points, 2, replace=False)
    hi, hj = h[i], h[j]
    delta_min = max(-hj, hi - 1)
    delta_max = min(hi, 1 - hj)
    if delta_min >= delta_max:
        return None
    delta = rng.uniform(delta_min, delta_max)
    return (i, j, delta)

def guided_move(h):
    """Find move that improves correlation for worst lag"""
    # Compute correlation with 1-h
    corr = correlate(h, 1 - h, mode='full')
    # Find worst lag (max correlation)
    worst_lag_idx = np.argmax(corr)
    # The actual lag index in array [0, 2*n_points-2]
    if worst_lag_idx >= n_points:
        # Convert to negative lag
        lag = worst_lag_idx - n_points + 1
    else:
        # Positive lag
        lag = -(n_points - 1) + worst_lag_idx
    
    # Find where to move for improvement
    gain = 0
    best_move = None
    if lag > 0:
        # Positive lag case: h[i] contributes to h[i-lag] correlation
        for i in range(lag, n_points):
            if h[i] > 0 and h[i-lag] < 1:
                new_gain = h[i] * (1 - h[i-lag]) - (h[i] - 1e-6) * (1 - (h[i-lag] + 1e-6))
                if new_gain > gain:
                    gain = new_gain
                    best_move = (i, i-lag, -1e-6)
    else:
        # Negative lag case: h[i] contributes to h[i+lag] correlation  
        lag = abs(lag)
        for i in range(0, n_points - lag):
            if h[i] > 0 and h[i+lag] < 1:
                new_gain = h[i] * (1 - h[i+lag]) - (h[i] - 1e-6) * (1 - (h[i+lag] + 1e-6))
                if new_gain > gain:
                    gain = new_gain
                    best_move = (i, i+lag, -1e-6)
    return best_move

def local_refine(h):
    """Local refinement using scipy.optimize"""
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, TARGET_SUM)
            return h
    except:
        pass
    return h
