import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.irfft(np.fft.rfft(h_rev, 2*n - 1) * np.fft.rfft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

    """Ensure h is a proper step function with n_points"""

    h_interp = np.interp(x_new, x_orig, h)
    return project_to_sum_box(h_interp, TARGET_SUM)

def advanced_optimization(h_start, run_seconds):
    """Advanced optimization using Nelder-Mead with refinement"""
    start_time = time.time()
    
    # Define objective function
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    # Use multiple optimization methods on restarts
    methods = ['Nelder-Mead', 'Powell', 'L-BFGS-B']
    best_h = h_start.copy()
    best_c5 = evaluate_c5(h_start)
    
    # Try a few restarts with different methods
    restarts = 0
    while time.time() - start_time < run_seconds and restarts < 10:
        # Mix methods
        method = methods[restarts % len(methods)]
        try:
            # Perturb slightly
            h_trial = best_h + np.random.normal(0, 0.02, len(best_h)) * 0.5
            h_trial = project_to_sum_box(h_trial, TARGET_SUM)
            
            cons = {'type': 'eq', 'fun': constraint_eq}
            bounds = [(0, 1) for _ in range(len(h_trial))]
            res = minimize(
                objective,
                h_trial,
                method=method,
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10}
            )

                    best_h = h_refined
                    best_c5 = c5_refined
