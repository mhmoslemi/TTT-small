import numpy as np

def project_to_simplex(v, target_sum):
    """Project vector v onto capped simplex with sum = target_sum"""
    v = np.asarray(v)
    n = len(v)
    # Binary search for the Lagrange multiplier
    # Since sum(z) = target_sum, and z[i] = clip(v[i] - lambda, 0, 1)
    # We want sum(clip(v[i] - lambda, 0, 1)) = target_sum
    
    # Ensure that all v values are in [0, 1]
    v = np.clip(v, 0, 1)
    
    # If the sum is already close to target
    s = np.sum(v)
    if abs(s - target_sum) < 1e-12:
        return v
    elif s <= target_sum:
        # All values can be increased to meet the sum
        v += (target_sum - s) / n
        return np.clip(v, 0, 1)
    else:
        # Need to decrease values, binary search for lambda
        # We have in decreasing order: v_sorted = np.sort(v)[::-1]
        # and reducing lambda decreases sum(z)
        # We want sum(clip(v[i] - lambda, 0, 1)) = target_sum
        # This is sum(max(0, v[i] - lambda)) = target_sum
        # This gives us a sequence of partitions depending on how many values 
        # are >= lambda. For decreasing values and a candidate lambda,
        # sum(z) = [sum(v_i - lambda) for i s.t. v_i >= lambda] = sum(v_i) - num * lambda
        
        # Sort values in descending order
        sorted_v = np.sort(v)[::-1]
        # We know we have at least one active constraining lambda
        low = sorted_v[-1] - target_sum
        high = sorted_v[0]
        
        for _ in range(50):
            lambda_val = (low + high) / 2.0
            # Compute sum(z) with this lambda
            z = np.maximum(v - lambda_val, 0)
            s_z = np.sum(z)
            if abs(s_z - target_sum) < 1e-12:
                return z
            elif s_z > target_sum:
                low = lambda_val
            else:
                high = lambda_val
        # Accept final value
        lambda_val = (low + high) / 2.0
        z = np.maximum(v - lambda_val, 0)
        return z

def interpolate_control_points(c, x_control, x_full):
    """Interpolate control points to full grid"""
    n = len(x_full)
    K = len(c)
    y = np.zeros(n)
    for i in range(n):
        x = x_full[i]
        if x <= x_control[0]:
            y[i] = c[0]
        elif x >= x_control[-1]:
            y[i] = c[-1]
        else:
            # Find interval containing x
            for j in range(1, K):
                if x_control[j-1] <= x < x_control[j]:
                    slope = (c[j] - c[j-1]) / (x_control[j] - x_control[j-1])
                    y[i] = c[j-1] + slope * (x - x_control[j-1])
                    break
    return y

def compute_correlation_fft(h):
    """Compute overlap using FFT"""
    n = len(h)
    dx = 2.0 / n
    h_fft = rfft(h)
    one_minus_h_fft = rfft(1 - h)
    corr = irfft(h_fft * np.conj(one_minus_h_fft))
    return corr[:n] * dx

def objective_with_soft_max(c, x_control, x_full, K, n, alpha, target_sum):
    """Soft max version of the objective"""
    h = interpolate_control_points(c, x_control, x_full)
    # Project onto feasible simplex
    h = project_to_simplex(h, target_sum)
    # Compute all correlations using FFT
    corr = compute_correlation_fft(h)
    max_corr = np.max(corr)
    
    # Check if all values are finite
    if not np.isfinite(max_corr):
        return 1e10 # Punish infeasible with penalty
    
    # Soft max approximation
    stable_corr = corr - np.max(corr)  # Avoid overflow
    if alpha > 0:
        soft_max_val = (1.0 / alpha) * np.log(np.sum(np.exp(alpha * stable_corr)))
        return soft_max_val
    else:
        return max_corr

def objective_exact(c, x_control, x_full, K, n, target_sum):
    """Exact objective for final polishing"""
    h = interpolate_control_points(c, x_control, x_full)
    # Project onto feasible simplex
    h = project_to_simplex(h, target_sum)
    # Compute all correlations using FFT
    corr = compute_correlation_fft(h)
    max_corr = np.max(corr)
    return max_corr

        pop = []
        for _ in range(lambda_pop):
            c_new = initial_values + np.random.normal(0, 0.05, K)
            c_new = np.clip(c_new, 0, 1)
            pop.append(c_new)

        # Annealing schedule
        alpha_schedule = np.linspace(alpha_start, alpha_end, int(0.7 * max_iter))

        # Iterations
        iter_count = 0
        last_improvement = 0

        while iter_count < max_iter and (time.time() - start_time) < time_limit:
            iter_count += 1
            
            # Compute values for all candidates 
            obj_vals = []

                obj_val = objective_with_soft_max(c, x_control, x_full, K, n, alpha_val, target_sum)
                obj_vals.append(obj_val)
            
            # Sort and select top-μ candidates
            sorted_indices = np.argsort(obj_vals)
            mu = lambda_pop // 2
            elite_indices = sorted_indices[:mu]
            elite_pop = [pop[i] for i in elite_indices]
            elite_vals = [obj_vals[i] for i in elite_indices]

            # Update population
            if len(elite_pop) > 0:

                old_mean = np.mean(elite_pop, axis=0)

                    # Pick an elite parent and perturb
                    parent = elite_pop[np.random.randint(0, len(elite_pop))]
                    # Add Gaussian noise
                    mutated = parent + np.random.normal(0, 0.01, K) 
                    mutated = np.clip(mutated, 0, 1)
                    pop.append(mutated)
            
            # Track best after some steps
            if iter_count >= 10 or iter_count % 20 == 0:
                for i, (c, val) in enumerate(zip(pop, obj_vals)):
                    c5_val = objective_exact(c, x_control, x_full, K, n, target_sum)
                    if c5_val < best_c5_val:
                        best_c5_val = c5_val
                        best_result = (c.copy(), c5_val)
                if iter_count >= 100 and best_c5_val < 0.3807:
                    break
            
            # Early stopping for stagnant improvement
            if iter_count >= 5:
                # Check if improvement is minimal
                old_impr = last_improvement
                for i in range(len(elite_vals)):
                    if elite_vals[i] < 1e-12:
                        continue
                    # Stop if no meaningful improvement
                    if abs(elite_vals[i] - 1e-10) < 1e-10:
                        last_improvement += 1
                        break
                if last_improvement > 50:
                    break
            
        if best_result and best_c5_val < best_c5_val:
            best_c5_val = best_result[1]
            
    return best_result
