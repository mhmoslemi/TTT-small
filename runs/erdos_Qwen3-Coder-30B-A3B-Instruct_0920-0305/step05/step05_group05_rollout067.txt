import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

        M = 4*L
        x = np.linspace(0, 2, M, endpoint=False)
        
        # Constraints for box bounds
        A_box = np.zeros((M, 2*L))
        for i, xi in enumerate(x):
            for j in range(L):
                A_box[i, j] = np.cos(np.pi*(j+1)*xi)
                A_box[i, L+j] = np.sin(np.pi*(j+1)*xi)
        
        # Box constraints: 0 <= 0.5 + A_box * u <= 1
        box_constraints = [
            0.5 + A_box @ u >= 0,
            0.5 + A_box @ u <= 1
        ]
        
        # SOC constraints: s_l >= a_l^2 + b_l^2
        soc_constraints = []
        for l in range(L):
            a_l = u[l]
            b_l = u[L+l]
            soc_constraints.append(
                cp.SOC(s[l] + 1, [2*a_l, 2*b_l, s[l] - 1])
            )
        
        # Toeplitz PSD constraint
        # Create Toeplitz matrix: T = t*I - S where S is the Toeplitz matrix of s
        # Construct the Toeplitz matrix for s vector 
        # [s_1, s_2, ..., s_L] -> Toeplitz with main diagonal [t, 0, ..., 0]
        # first row: [t, -s_1, -s_2, ..., -s_{L-1}, -s_L]
        # first column: [t, -s_1, -s_2, ..., -s_L]

            row = [t if j == 0 else -s[j-1] if 0 < j <= L and i == 0 else
                   -s[i-1] if i > 0 and j == 0 else 0 for j in range(L+1)]
            T_rows.append(row)
        T = cp.Parameter((L+1, L+1), PSD=True)
        T[0, 0] = t
        for i in range(L):
            T[0, i+1] = -s[i]
            T[i+1, 0] = -s[i]
            T[i+1, i+1] = t

        for i in range(1, L+1):

                    T[i,j] = t
                else:
                    T[i,j] = -s[abs(i-j)-1] if abs(i-j) <= L else 0
