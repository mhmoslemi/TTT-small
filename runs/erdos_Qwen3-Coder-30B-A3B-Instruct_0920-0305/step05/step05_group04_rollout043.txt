import numpy as np

def project_to_simplex(h, target_sum):
    """Project h onto the simplex with sum = target_sum and bounds [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    
    # Find lambda such that sum(clamp(h + lambda, 0, 1)) == target_sum
    def sum_clamped_h(lambda_val):
        return np.sum(np.clip(h + lambda_val, 0, 1)) - target_sum
    
    # Binary search for lambda
    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high) / 2
        val = sum_clamped_h(mid)
        if val > 0:
            high = mid
        else:
            low = mid
    
    lambda_opt = (low + high) / 2
    h_proj = np.clip(h + lambda_opt, 0, 1)
    return h_proj

def compute_correlations(h, n):
    """Compute auto-correlations using FFT, return correlations from shifts 1 to n-1"""
    # pad with zeros for linear convolution
    h_padded = np.pad(h, (0, n), mode='constant')
    corr = np.fft.irfft(np.fft.rfft(h_padded) * np.conj(np.fft.rfft(h_padded)))
    # extract correlations for non-zero shifts
    return corr[1:n]

def smooth_min_objective_and_gradient(h, beta, n):
    """Compute the smooth surrogate for the minimum correlation and its gradient"""
    dx = 2.0 / n
    c = compute_correlations(h, n)
    # Remove the zero shift (shift 0) since we are minimizing over non-zero shifts
    c_nonzero = c[1:] if len(c) > 1 else []
    if len(c_nonzero) == 0:
        return 1.0, np.zeros_like(h)
    
    # compute soft min with stability trick
    c_max = np.max(c_nonzero) if len(c_nonzero) > 0 else 0
    exp_vals = np.exp(-beta * (c_nonzero - c_max))
    sum_exp = np.sum(exp_vals)
    if sum_exp == 0:
        sum_exp = 1.0  # avoid division by zero
    w = exp_vals / sum_exp
    smooth_min = (1.0 / beta) * (np.log(sum_exp) + c_max)
    
    # Gradient: - sum_k w_k * grad(c_k)
    # The gradient of c_k w.r.t h_i is h_{i+k} + h_{i-k}
    # We compute weighted gradient via convolution
    # Two convolutions:
    # 1. Convolution of h with w (forward shifts)
    # 2. Convolution of h with w_rev (backward shifts)
    
    w_pad = np.pad(w, (0, n-1), mode='constant')   # zero pad to same length
    h_pad = np.pad(h, (0, n-1), mode='constant')
    # forward: h * w (shifted)
    grad_forward = np.convolve(h_pad, w_pad, mode='valid')
    # backward: h * rev(w) (reversed shift)
    w_rev = w[::-1]
    w_rev_pad = np.pad(w_rev, (0, n-1), mode='constant')
    grad_backward = np.convolve(h_pad, w_rev_pad, mode='valid')
    
    # For each i, gradient is - (grad_forward[i] + grad_backward[i])
    grad = -(grad_forward[:n] + grad_backward[:n])
    
    return 1.0 + smooth_min, grad

def line_search(h, grad, obj_and_grad_func, beta, n, max_backtracks=10):
    """Perform backtracking line search on the smooth objective"""
    alpha = 0.5
    step_size = alpha
    dx = 2.0 / n
    objective, _ = obj_and_grad_func(h, beta, n)
    
    for _ in range(max_backtracks):
        h_new = h - step_size * grad
        h_new_clipped = np.clip(h_new, 0, 1)
        h_proj = project_to_simplex(h_new_clipped, n/2)
        obj_new, _ = obj_and_grad_func(h_proj, beta, n)
        if obj_new <= objective - 1e-12 or step_size < 1e-12:
            return step_size, h_proj
        step_size *= 0.5
    return step_size, h_proj
