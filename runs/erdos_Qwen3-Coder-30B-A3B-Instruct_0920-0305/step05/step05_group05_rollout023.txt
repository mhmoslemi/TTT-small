import numpy as np
import time

def project_capped_simplex(v, target_sum):
    """Project v onto the capped simplex: 0 <= x <= 1, sum(x) = target_sum."""
    u = np.sort(v)[::-1]
    tmp = np.cumsum(u)
    rho = np.where(u > (target_sum - tmp) / np.arange(1, len(u) + 1))[0]
    if len(rho) == 0:
        # This should not happen if target_sum is valid
        lambda_star = (target_sum - tmp[-1]) / len(u)
        return np.clip(v - lambda_star, 0, 1)
    rho = rho[-1]
    lambda_star = (target_sum - tmp[rho]) / (rho + 1)
    return np.clip(v - lambda_star, 0, 1)

def compute_O_and_grad(h, alpha):
    """Return O = [O_k] and smoothed gradient of F_alpha = (1/alpha)*log(sum(exp(alpha*O)))"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations O_k = dx * sum h[i] * (1 - h[i+k])
    # Using np.correlate, 'full' mode produces 2*n - 1 entries
    corr = np.correlate(h, 1 - h, mode="full")
    corr *= dx  # scale by dx
    
    # Use log-sum-exp trick for numerical stability
    max_corr = np.max(corr)
    exp_corr = np.exp(alpha * (corr - max_corr))
    w = exp_corr / np.sum(exp_corr)  # weights from softmax
    
    # Gradient computation:
    # For each k,  ∂O_k/∂h[i] = dx * (1 - h[i+k] - h[i-k])
    # So ∂F_α/∂h[i] = dx * sum_k w_k * (1 - h[i+k] - h[i-k])   
    grad = np.zeros(n)
    for i in range(n):
        for k in range(2*n - 1):
            if k < n:
                j = i + k - (n-1)
                if 0 <= j < n:
                    grad[i] += w[k] * (1 - h[j])
            if k >= n:
                j = i - (k - (n-1))
                if 0 <= j < n:
                    grad[i] -= w[k] * h[j]
    
    grad *= dx
    return corr, grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 64  # Size chosen for reasonable computation in budget
    dx = 2.0 / n
    target_sum = n / 2.0  # Sum should be exactly n/2
    
    def evaluate_c5(h):
        # Evaluate overlap integral for all shifts
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    best_h = None
    best_c5 = float('inf')
    
    max_restarts = 1000
    restart_count = 0
    while restart_count < max_restarts and (time.time() - start_time) < 0.95 * budget_s:
        # Try different initializations
        init_type = restart_count % 5
        if init_type == 0:
            # Uniform random with projection
            new_h = np.random.uniform(0, 1, n)
        elif init_type == 1:
            # Triangular shape
            x = np.linspace(0, 2, n)
            new_h = 1 - np.abs(x - 1)
            new_h = np.clip(new_h, 0, 1)
        elif init_type == 2:
            # Sine shape
            x = np.linspace(0, 2, n)
            new_h = 0.5 + 0.4 * np.sin(np.pi * x)
            new_h = np.clip(new_h, 0, 1)
        elif init_type == 3:
            # Gaussian bump
            x = np.linspace(0, 2, n)
            mu = 1.0
            sigma = 0.3
            new_h = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
            new_h = np.clip(new_h, 0, 1)
        else:  # init_type == 4
            # Binary pattern
            new_h = np.zeros(n)
            new_h[n//4:3*n//4] = 1.0
        
        # Project onto feasible set
        new_h = project_capped_simplex(new_h, target_sum)
        
        # Gradient descent with smoothing annealing
        h = new_h.copy()
        alpha_start, alpha_end = 5.0, 30.0
        lr_0 = 0.5
        max_iters = 4000
        for t in range(1, max_iters + 1):
            if (time.time() - start_time) > 0.95 * budget_s:
                break
                
            alpha = alpha_start + (alpha_end - alpha_start) * t / max_iters
            corr, grad = compute_O_and_grad(h, alpha)
            
            step_size = lr_0 / np.sqrt(t)
            h -= step_size * grad
            
            # Project onto simplex
            h = project_capped_simplex(h, target_sum)
            
            # Optional symmetry enforcement
            if t % 10 == 0:
                h = (h + h[::-1]) / 2.0
                
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        restart_count += 1

    # Fine-tune the best solution using subgradient ascent in the max
    if best_h is not None:
        h_fine = best_h.copy()
        # Find the shift(s) that achieve the maximum
        corr = np.correlate(h_fine, 1 - h_fine, mode="full")
        max_shifts = np.where(corr == np.max(corr))[0]
        fine_lr = 0.02 / np.sqrt(1.0)
        fine_iters = 500
        for i in range(fine_iters):
            if (time.time() - start_time) > 0.95 * budget_s:
                break
            # Only update with gradient of max overlap
            grad_max = np.zeros(n)
            for k in max_shifts:
                if k < n:
                    # Gradient of O_k = dx sum h[i](1-h[i+k])
                    j = i + k - (n-1)
                    if 0 <= j < n: grad_max[i] += -dx * h_fine[j]
                if k >= n:
                    j = i - (k - (n-1))
                    if 0 <= j < n: grad_max[i] += dx * h_fine[j]
            h_fine -= fine_lr * grad_max
            h_fine = project_capped_simplex(h_fine, target_sum)
            current_c5 = np.max(np.correlate(h_fine, 1 - h_fine, mode="full")) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_fine.copy()

    return (best_h.tolist(), float(best_c5), n)
