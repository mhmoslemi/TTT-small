def project_capped_simplex(v, T):
    """Project vector v onto the capped simplex: 0 <= x <= 1, sum(x) = T."""
    v = np.asarray(v)
    lower, upper = 0.0, 1.0
    # Binary search for the Lagrange multiplier
    l = lower - v.max()
    u = upper - v.min()
    for _ in range(50):
        if l >= u:
            break
        lam = (l + u) / 2.0
        proj = np.clip(v - lam, lower, upper)
        if proj.sum() > T:
            l = lam
        else:
            u = lam
    lam = (l + u) / 2.0
    return np.clip(v - lam, lower, upper)

            h2 = h[:n-k] # h[:n-k]

            h1 = h[:-k]  # h[:-k]
            h2 = h[-k:]  # h[-k:]

    subgrad_iters = 500

        best_h = project_capped_simplex(best_h, T)
        best_val = evaluate_c5(best_h)
        
    return (best_h.tolist(), best_val, n)
