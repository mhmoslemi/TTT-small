import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use hybrid optimization approach with different algorithms
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Phase 1: Differential evolution with multiple restarts
    for restart in range(5):
        if (time.time() - start_time) > budget_s * 0.8:
            break
        try:
            res = differential_evolution(
                objective,
                bounds,
                maxiter=200,
                popsize=15,
                seed=seed + restart,
                strategy='best1bin'
            )
            if res.success and res.fun < best_c5:
                h_candidate = res.x.copy()
                dx = 2.0 / n_points
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
    
    # Phase 2: Local optimization from multiple starting points
    local_restarts = 20
    for restart in range(local_restarts):
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
        # Try different random initializations
        if restart < 10:
            # Perturb current best
            new_h = best_h + np.random.normal(0, 0.02, n_points)
        else:
            # Try a new pattern
            new_h = np.random.rand(n_points) * 0.5
            
        new_h = np.clip(new_h, 0, 1)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='L-BFGS-B', 
                bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x.copy()
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue

    # Phase 3: Try specific improved structures
    if (time.time() - start_time) < budget_s * 0.9:
        try:
            # Try truncated Gaussian pattern for better distribution
            x = np.linspace(0, 2, n_points)
            # Use a skewed bell shape to create more effective overlap reduction
            mean = 1.0
            std = 0.3
            gaussian = np.exp(-0.5 * ((x - mean) / std)**2)
            
            # Ensure integral constraint is met
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(gaussian) * dx)
            gaussian = gaussian * scale_factor
            gaussian = np.clip(gaussian, 0, 1)
            
            res = minimize(
                objective, 
                gaussian, 
                method='L-BFGS-B', 
                bounds=bounds,
                options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-15}
            )
            if res.success:
                h_candidate = res.x.copy()
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
            
    # Ensure final integral constraint is exactly met
    dx = 2.0 / n_points
    final_scale = 1.0 / (np.sum(best_h) * dx)
    best_h = best_h * final_scale
    
    # Final verification
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
