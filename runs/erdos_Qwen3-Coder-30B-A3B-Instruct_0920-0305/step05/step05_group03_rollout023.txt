```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use a hybrid optimization approach
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Phase 1: Use global optimization with DE as backup
    try:
        # Check if we have resources for DE
        if (time.time() - start_time) < budget_s * 0.2:
            res = differential_evolution(
                objective, 
                bounds, 
                maxiter=500, 
                popsize=15,
                seed=seed
            )
            if res.success:
                h_candidate = res.x
                dx = 2.0 / n_points
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
    except:
        pass
    
    # Phase 2: Local optimization starting from best solution
    try:
        # Use trust-constr which is good for constrained problems
        res = minimize(
            objective, 
            best_h, 
            method='trust-constr', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 2000, 'gtol': 1e-13, 'ftol': 1e-13, 'verbose': 0}
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Phase 3: Random restarts with better exploration
    max_restarts = 200
    restart_count = 0
    
    while (time.time() - start_time) < budget_s * 0.95 and restart_count < max_restarts:
        # Generate a different random starting point
        new_h = np.random.uniform(0, 1, n_points)
        # Make sure it satisfies the integral constraint
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            # Try local optimization with this new starting point
            res = minimize(
                objective, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
            
        restart_count += 1

    # Phase 4: Try clustering pattern (focused strategy)
    cluster_patterns = [
        # Upward slope
        np.array([0.0] * 10 + [0.1] * 10 + [0.2] * 10 + [0.3] * 10 + [0.4] * 10),
        # Downward slope
        np.array([0.4] * 10 + [0.3] * 10 + [0.2] * 10 + [0.1] * 10 + [0.0] * 10),
        # Gentle bell
        np.array([0.0] * 5 + [0.1] * 10 + [0.2] * 10 + [0.1] * 10 + [0.0] * 5),
    ]
    
    for pattern in cluster_patterns:
        if len(pattern) == n_points:
            pattern = pattern * (1.0 / np.sum(pattern) * 2.0)  # Normalize to integral = 1
            try:
                res = minimize(
                    objective, 
                    pattern, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=cons,
                    options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-14}
                )
                if res.success:
                    h_candidate = res.x
                    scale = 1.0 / (np.sum(h_candidate) * dx)
                    h_candidate = h_candidate * scale
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
            except:
                pass

    return (best_h.tolist(), best_c5, n_points)
```
