import numpy as np
import time

def project_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex: 0 ≤ x ≤ 1, sum(x) = target_sum"""
    v = np.asarray(v)
    low, high = 0.0, 1.0
    # Binary search for the Lagrange multiplier
    lmbda_low = np.min(v - high)
    lmbda_high = np.max(v - low)
    for _ in range(50):  # Usually converges in < 30 steps
        lmbda = (lmbda_low + lmbda_high) / 2.0
        x = np.clip(v - lmbda, low, high)
        current_sum = np.sum(x)
        if current_sum > target_sum:
            lmbda_low = lmbda
        else:
            lmbda_high = lmbda
    # Final projection
    lmbda = (lmbda_low + lmbda_high) / 2.0
    return np.clip(v - lmbda, low, high)

def softmax(x, axis=None):
    x = np.asarray(x)
    x_max = np.max(x, axis=axis, keepdims=True)
    exp_x = np.exp(x - x_max)
    return exp_x / np.sum(exp_x, axis=axis, keepdims=True)

def compute_O_and_grad(h, alpha):
    """Compute overlaps O_k and gradient of smoothed max over them"""
    dx = 2.0 / len(h)
    
    # Vectorized correlation of h and 1-h
    corr = np.correlate(h, 1 - h, mode='full')
    # Extract the unique overlap values: corr[0] to corr[2n-2]
    O = corr * dx  # Scale by dx
    
    # Softmax weights
    O_stable = O - np.max(O)  # For numerical stability
    w = softmax(alpha * O_stable)
    
    # Construct gradient component by component using vectorized slices
    grad = np.zeros_like(h)
    
    # Adjust indices for valid correlation shifts (from -n+1 to n-1)
    n = len(h)
    # For each shift index k in [0, 2n-2]
    for i in range(2 * n - 1):
        k = i - (n - 1)  # Convert index to shift
        if k >= 0:
            shift_backward = np.roll(h, k)  # indices i -> i + k
            shift_forward = np.zeros_like(h)
            shift_forward[k:] = h[:-k]
        else:
            # k < 0
            k_abs = -k
            shift_backward = np.zeros_like(h)
            shift_backward[:-k_abs] = h[k_abs:]
            shift_forward = np.roll(h, k)
        
        # Use the actual h slice for k-shifts (for efficiency)
        if k >= 0:
            # In case of double-zeros for elements near boundary
            # K is positive shift (same as convolution)
            grad += w[i] * (1 - shift_backward)  # Changed the sign as 1-h
            grad += w[i] * (-shift_forward)
        else:
            # k-negative shift: roll forward
            grad += w[i] * (1 - shift_forward)
            grad += w[i] * (-shift_backward)

    # Undo the -1 multiplication and scale with dx
    grad *= dx
    
    return O, grad
