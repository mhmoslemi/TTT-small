import numpy as np
from cvxpy import *
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = np.array(initial_h_values)
    best_c5 = evaluate_c5(best_h)
    n_points = len(best_h)
    
    # Try smaller n to get faster runtimes
    N = [36, 40, 44, 48, 52, 56, 60, 64]
    best_c5_bound = best_c5
    
    for n in N:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Create a simple greedy heuristic warm start
        x = [0]*n
        s = set()
        while len(s) < n // 2:
            best_i = -1
            best_min_p = -1
            for i in range(n):
                if i in s: continue
                new_s = s | {i}
                p_vals = []
                for k in range(1, n):
                    count = 0
                    for j in new_s:
                        if (j+k) < n and j in new_s:
                            count += 1
                    p_vals.append(count)
                min_p = min(p_vals) if p_vals else 0
                if min_p > best_min_p:
                    best_min_p = min_p
                    best_i = i
            if best_i >= 0:
                s.add(best_i)
                x[best_i] = 1
        h0 = np.array(x).astype(float)
        dx = 2.0 / n
        scale = 1.0 / (np.sum(h0) * dx)
        h0 = h0 * scale
        
        # Compute overlap integrals to get estimates
        c5_current = evaluate_c5(h0)
        if c5_current < best_c5_bound:
            best_h = h0.copy()
            best_c5_bound = c5_current
            best_h = h0.copy()
            best_c5_bound = c5_current

        # Run simple gradient-like refinement from warm start
        if time.time() - start_time > budget_s * 0.95:
            break
        try:
            # Simple iteration with small step sizes
            dx_step = 0.01
            max_iter = 500
            for _ in range(max_iter):
                if time.time() - start_time > budget_s * 0.95:
                    break
                c5_old = evaluate_c5(h0)
                # Estimate gradient via finite difference
                grad = np.zeros(n)
                for i in range(n):
                    h_plus = h0.copy()
                    h_plus[i] = min(1.0, h_plus[i] + dx_step)
                    h_minus = h0.copy()
                    h_minus[i] = max(0.0, h_minus[i] - dx_step)
                    c5_plus = evaluate_c5(h_plus)
                    c5_minus = evaluate_c5(h_minus)
                    grad[i] = (c5_plus - c5_minus) / (2*dx_step)
                
                # Gradient ascent with projection
                h1 = h0 - 0.1 * grad
                h1 = np.clip(h1, 0, 1)
                dx = 2.0 / n
                scale = 1.0 / (np.sum(h1) * dx)
                h1 = h1 * scale
                c5_new = evaluate_c5(h1)
                if c5_new < c5_old:
                    h0 = h1.copy()
                else:
                    break
                if c5_new < best_c5_bound:
                    best_h = h0.copy()
                    best_c5_bound = c5_new
        except Exception:
            pass

    return (best_h.tolist(), best_c5_bound, n_points)
