import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def greedy_initialization(n):
    """Greedy algorithm to initialize a binary pattern"""
    x = np.zeros(n, dtype=int)
    ones_placed = 0
    while ones_placed < n // 2:
        best_i = -1
        best_min_p = -1
        for i in range(n):
            if x[i] == 1:
                continue
            # Temporarily add i to set and compute new p_k values
            temp_x = x.copy()
            temp_x[i] = 1
            p_vals = []
            for k in range(1, n):
                p_k = 0
                for j in range(n - k):
                    p_k += temp_x[j] * temp_x[j + k]
                p_vals.append(p_k)
            min_p = min(p_vals) if p_vals else 0
            if min_p > best_min_p:
                best_min_p = min_p
                best_i = i
        if best_i >= 0:
            x[best_i] = 1
            ones_placed += 1
        else:
            # Force fill remaining
            for i in range(n):
                if x[i] == 0 and ones_placed < n // 2:
                    x[i] = 1
                    ones_placed += 1
    return x

def solve_milp_for_n(n, time_limit):
    """Solve MILP to maximize minimum ordered pair count for the given n"""
    # Define the MILP components and variables using cvxpy
    dx = 2.0 / n
    # Variable to maximize: minimum of p_k for k = 1..n-1
    L = Integer("L")
    
    # Binary variable indicating if h[i] = 1
    x = Binary("x", shape=(n,))
    
    # Ensure sum of x is n/2
    constraints = [sum(x) == n // 2]
    
    # Add symmetry breaking constraints
    constraints.append(x[0] == 1)
    left_sum = sum(x[:n//2])
    right_sum = sum(x[n//2:])
    constraints.append(left_sum >= right_sum)
    
    # For each lag k, compute p_k
    y = {}
    p_k_constraints = []
    for k in range(1, n):
        # y[i,k] = x[i] AND x[i+k]
        for i in range(n - k):
            y_key = (i, k)
            y[y_key] = Binary(f"y_{i}_{k}")
            # McCormick envelopes
            constraints.append(y[y_key] <= x[i])
            constraints.append(y[y_key] <= x[i+k])
            constraints.append(y[y_key] >= x[i] + x[i+k] - 1)
            constraints.append(y[y_key] >= 0)
            
        # p_k = sum over i of y[i,k]
        p_k = sum(y[(i,k)] for i in range(n - k))
        # Enforce p_k >= L
        p_k_constraints.append(p_k >= L)
        
    constraints.extend(p_k_constraints)
    
    # Objective is to maximize L
    objective = Maximize(L)
    
    # Create the problem
    prob = Problem(objective, constraints)
    
    try:
        # Solve the MILP problem with time limit
        prob.solve(solver=CPLEX, time_limit=time_limit)
        
        if prob.status in ["optimal", "optimal_inaccurate"]:
            # Extract the optimal x solution
            x_vals = [int(v.value) for v in x]
            return x_vals, L.value
        elif prob.status == "infeasible":
            return None, None
        else:
            # Take the best feasible solution if available
            x_vals = [int(round(v.value)) for v in x]
            # Calculate min p_k for this feasible x
            min_p = float('inf')
            p_vals = []
            for k in range(1, n):
                p_k = 0
                for i in range(n - k):
                    p_k += x_vals[i] * x_vals[i + k]
                p_vals.append(p_k)
            min_p = min(p_vals) if p_vals else 0
            return x_vals, min_p
                
    except Exception as e:
        return None, None

            # Convert to continuous h by scaling x to integral = 1.

                return evaluate_c5(h)

                # Simple derivative approximation

                grad = np.zeros_like(h)

                    h_plus[i] += eps
                    h_minus = h.copy()
                    h_minus[i] -= eps

                g = grad_c5(h)
                step_size = 0.01
                h_new = h - step_size * g
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n/2) / sum(h_new)
                h = h_new
                if abs(c5_objective(h) - best_c5) < 1e-6:
                    break
            
            refined_c5 = c5_objective(h)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = h
                best_n = n
