import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use a hybrid optimization approach
    # Phase 1: Global optimization with differential evolution
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    try:
        # Use differential evolution for global search
        res = differential_evolution(
            objective, 
            bounds, 
            maxiter=100,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            callback=lambda x, convergence: None
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass
    
    # Phase 2: Local optimization with trust-constr from multiple starting points
    local_restart_bound = 200
    for i in range(local_restart_bound):
        if (time.time() - start_time) > budget_s * 0.95:  # Leave time for final steps
            break
            
        # Start with random perturbation around best solution
        new_h = best_h + np.random.normal(0, 0.05, n_points)  # Larger perturbation
        new_h = np.clip(new_h, 0, 1)
        
        # Ensure the constraint is satisfied
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='trust-constr', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 200, 'gtol': 1e-10, 'ftol': 1e-10}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue
    
    # Phase 3: Simulated Annealing approach for further improvement
    if (time.time() - start_time) < budget_s * 0.95:
        T = 1.0
        T_min = 1e-8
        alpha = 0.99
        max_iter = 50000
        
        current_h = best_h.copy()
        current_c5 = evaluate_c5(current_h)
        
        for i in range(max_iter):
            if (time.time() - start_time) > budget_s * 0.95:
                break
                
            # Perturb solution
            new_h = current_h + np.random.normal(0, 0.02, n_points)
            new_h = np.clip(new_h, 0, 1)
            
            # Normalize
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(new_h) * dx)
            new_h = new_h * scale_factor
            
            new_c5 = evaluate_c5(new_h)
            
            if new_c5 < current_c5:
                current_h = new_h.copy()
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
            elif np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                current_h = new_h.copy()
                current_c5 = new_c5
                
            T = max(T * alpha, T_min)
    
    return (best_h.tolist(), best_c5, n_points)
