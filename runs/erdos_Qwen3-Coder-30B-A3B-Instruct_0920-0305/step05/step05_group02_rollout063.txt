def compute_correlation(h):
    """Compute the full correlation array for a binary vector h"""
    n = len(h)
    corr = np.zeros(2*n - 1)
    for k in range(-n+1, n):
        if k >= 0:
            corr[k + n - 1] = np.sum(h[:n-k] * h[k:])
        else:
            corr[k + n - 1] = np.sum(h[-k:] * h[:n+k])
    return corr

def evaluate_c5(h):
    """Evaluate C5 for a given h"""
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlation(h)
    max_corr = np.max(corr)
    return max_corr * dx

def generate_legendre_pattern(n):
    """Generate a Legendre-based binary pattern"""
    p = n // 2
    if p < 2:
        return None
    
    # Try to find a prime close to p
    def is_prime(num):
        if num < 2:
            return False
        for i in range(2, int(num**0.5)+1):
            if num % i == 0:
                return False
        return True
    
    # Find the largest prime <= p
    while not is_prime(p):
        p -= 1
        if p < 2:
            return None

    # Generate quadratic residues mod p
    residues = set()
    for a in range(1, p):
        residues.add((a * a) % p)
    
    # Create binary pattern of size p
    b = np.zeros(p, dtype=int)
    for r in residues:
        b[r] = 1
    # Create pattern of length n
    h = np.tile(b, 2)
    h = h[:n]
    
    # Adjust if we don't have exactly n/2 ones
    ones_count = np.sum(h)
    if ones_count > n//2:
        # Find a 1 and flip it to 0
        idx = np.where(h == 1)[0][0]
        h[idx] = 0
    elif ones_count < n//2:
        # Find a 0 and flip it to 1
        idx = np.where(h == 0)[0][0]
        h[idx] = 1
    
    return h

def generate_random_balanced_pattern(n):
    """Generate a random balanced binary pattern"""
    indices = np.random.permutation(n)
    h = np.zeros(n, dtype=int)
    h[indices[:n//2]] = 1
    return h

def get_bad_ones(h, k):
    """Get indices of 1s that when swapped could decrease correlation at lag k"""
    n = len(h)
    if k >= 0:
        bad_ones = np.where((h[:n-k] == 1) & (h[k:] == 0))[0]
    else:
        bad_ones = np.where((h[-k:] == 1) & (h[:-k] == 0))[0]
    if len(bad_ones) == 0:
        return np.array([])
    return bad_ones
