```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use a hybrid optimization approach
    # Phase 1: Use global optimization (differential evolution)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Try global optimization with different settings
    try:
        # First try differential evolution on small samples
        if n_points <= 100:
            res = differential_evolution(
                objective, 
                bounds, 
                seed=seed,
                maxiter=200,
                popsize=15,
                mutation=(0.5, 1.0),
                recombination=0.7,
                atol=1e-10,
                rtol=1e-10
            )
            if res.success:
                h_candidate = res.x
                dx = 2.0 / n_points
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
    except:
        pass

    # Phase 2: Local refinement with trust-constr
    try:
        res = minimize(
            objective, 
            best_h, 
            method='trust-constr', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'gtol': 1e-12, 'ftol': 1e-12}
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Phase 3: Multiple random restarts with different patterns
    max_restarts = 50
    for restart in range(max_restarts):
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
        # Generate different starting patterns
        pattern_type = restart % 5
        new_h = None
        
        if pattern_type == 0:  # Uniform perturbation
            new_h = best_h + np.random.normal(0, 0.02, n_points)
        elif pattern_type == 1:  # Sinusoidal perturbation
            x = np.linspace(0, 2, n_points)
            new_h = best_h + 0.03 * np.sin(2 * np.pi * x) + 0.01 * np.random.randn(n_points)
        elif pattern_type == 2:  # Gaussian bump
            x = np.linspace(0, 2, n_points)
            new_h = best_h + 0.04 * np.exp(-((x - 1)**2) / 0.05)
        elif pattern_type == 3:  # Step pattern
            new_h = best_h.copy()
            if n_points > 10:
                idx = n_points // 4
                new_h[idx:3*idx] = np.clip(best_h[idx:3*idx] + 0.1 * np.random.randn(2*idx), 0, 1)
        else:  # Random noise
            new_h = best_h + 0.05 * np.random.randn(n_points)
            
        new_h = np.clip(new_h, 0, 1)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-13, 'gtol': 1e-13}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue

    # Phase 4: Try quadratic programming approach with specialized constraints  
    if n_points <= 100 and (time.time() - start_time) < budget_s * 0.8:
        try:
            # Try L-BFGS-B optimization as final attempt to polish
            res = minimize(
                objective, 
                best_h, 
                method='L-BFGS-B', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                dx = 2.0 / n_points
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
    
    return (best_h.tolist(), best_c5, n_points)
```
