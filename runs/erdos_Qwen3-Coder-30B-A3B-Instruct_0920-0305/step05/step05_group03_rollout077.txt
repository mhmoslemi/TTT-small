```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Method 1: Differential evolution for global search
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Differential evolution with small population
    try:
        res = differential_evolution(
            objective,
            bounds,
            seed=seed,
            maxiter=200,
            popsize=10,
            mutation=(0.5, 1),
            recombination=0.7,
            atol=1e-12,
            tol=1e-12
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass
    
    # Method 2: Local refinement using trust-constr on best so far
    if (time.time() - start_time) < budget_s * 0.8:
        try:
            # Local search refinement
            res = minimize(
                objective, 
                best_h, 
                method='trust-constr', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'gtol': 1e-13, 'ftol': 1e-13}
            )
            if res.success:
                h_candidate = res.x
                dx = 2.0 / n_points
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
    
    # Method 3: Additional local search with adaptive restarts
    local_restarts = min(500, int(300000 / n_points))  # Fewer restarts for larger n
    
    for restart in range(local_restarts):
        if (time.time() - start_time) > budget_s * 0.95:  # Leave time for final steps
            break
            
        # Start with random perturbation around previous best
        noise_scale = 0.02 + 0.01 * (1.0 - restart / local_restarts)  # Gradually decrease noise
        new_h = best_h + np.random.normal(0, noise_scale, n_points)
        new_h = np.clip(new_h, 0, 1)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue
    
    # Method 4: Try to create a more structured pattern if time allows
    if (time.time() - start_time) < budget_s * 0.9:
        try:
            # Create a pattern that reduces overlap:
            # Try a two-peak pattern with careful balancing
            x = np.linspace(0, 2, n_points)
            # Try an asymmetric pattern
            peak1_pos = 0.4
            peak2_pos = 1.6
            peak1_height = 0.6
            peak2_height = 0.4
            
            # Create two peaks with a valley in between
            pattern1 = peak1_height * np.exp(-((x - peak1_pos)**2) / 0.1) 
            pattern2 = peak2_height * np.exp(-((x - peak2_pos)**2) / 0.1)
            mixed_h = pattern1 + pattern2
            
            # Normalize
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(mixed_h) * dx)
            mixed_h = mixed_h * scale_factor
            
            # Refine with local optimization
            res = minimize(
                objective, 
                mixed_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-15}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
    
    # Final check and return
    return (best_h.tolist(), best_c5, n_points)
```
