import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
n_fine = 240
dx_fine = 2.0 / n_fine

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_gradient(h, beta=1.0):
    """Compute gradient of smooth maximum correlation"""
    # Compute all correlations
    corr = compute_corr_fft(h)
    
    # Apply smoothing
    max_corr = np.max(corr)
    shifted_corr = corr - max_corr
    
    # Prevent overflow in exponentials
    exp_corr = np.exp(beta * shifted_corr)
    exp_sum = np.sum(exp_corr)
    
    # Weights for averaging
    weights = exp_corr / exp_sum
    
    # Compute gradient as weighted average of individual gradients
    grad = np.zeros_like(h)
    
    # For each lag, compute gradient contribution
    for i in range(len(h)):
        # For lag l, gradient term is (1 - h_{i+l} - h_{i-l})
        # But h_{i+l} and h_{i-l} can go out of bounds; we'll use zero-padding implicitly.
        # We will do this by converting to contiguous, but for the sake of slight simplicity in indexing
        # we will compute the component-wise sum manually
        
        # contribution from negative lags (shifted right)
        # 1 - h_{i+1} - h_{i-1} ...
        
        # let's compute term-by-term using the full correlations
        # g_l for each lag l
        # Actually we can compute all contributions simultaneously using convolution trick?
        # whoa. let's use the FFT to get all individual gradient contributions.
        
        # More direct: compute each finite difference and sum.
        # For lag l, gradient w.r.t h_i is 1 if l=0, 0 if l≠0 for both terms,
        # and from the correlation residue: it's 1 - h_{i+l} - h_{i-l} (plus what we need for these specific lags)
        
        # Since corr contains values for lags from -n+1 to n-1, 
        # we'll use indices in -n+1 to n-1 or equivalently 0-(n-1) to n-1
        
        for l in range(len(weights)):
            if l == 0:
                # this is actually overcounts because corr[0] is the actual C_0(h) = dx * sum(h_i*(1-h_i))
                continue
            else:
                if l < len(h):
                    grad[i] += weights[l] * (1.0 - h[i + l] - h[i - l])
        
        # this will be slightly inaccurate. 
        # instead we start with a base gradient that is the sum of all weight lags.
        # That's wrong too.
        
    # Better issue is we need more clarity. Use direct FFT trick.
    # We know that in vec form:
    # C_l(h) = dx * sum_{i} h_i (1 - h_{i+l}) = dx * ( sum h_i - sum h_i h_{i+l} ) 
    
    # So d C_l / d h_i = dx ( 1 - h_{i+l} - h_{i-l} ) (and 0 if i out of bounds)
    # which gives the gradient components for a fixed l, based on shifting h by l then subtracting
    
    # Accurate way:
    # Convert 1-h to a full correlation
    grad = np.zeros_like(h)
    
    # Use full corr array to compute gradients
    # corr will have lags from -n+1 ... n-1 => positions 0 ... 2*n-2
    # let's convert lag to pos in corr: lag_i = i - (n-1)
    
    # Now build weights full vector
    for l in range(len(corr)):
        lag = l - (len(corr) // 2)
        if l == 0:
            continue
        weight = weights[l]
        # So contribution to grad of h_i is weight*(1 - h_{i-l} - h_{i+l})
        # But indexing here is tricky, so we compute full shift for the weight
        
        shifted_h_prev = np.roll(h, lag)
        shifted_h_next = np.roll(h, -lag)
        grad += weight * (1.0 - shifted_h_prev - shifted_h_next)

    return grad

def smooth_max_objective(h, beta=1.0):
    corr = compute_corr_fft(h)
    # Convert to C5 style weights by not using *dx (we'll handle dx properly at the end)
    # equivalent to dx * C_l -> just C_l  
    corr = corr * dx
    
    max_corr = np.max(corr)
    shifted_corr = corr - max_corr
    exp_corr = np.exp(beta * shifted_corr)
    exp_sum = np.sum(exp_corr)
    return (1.0 / beta) * np.log(exp_sum)  

def projected_gradient_descent(h0, beta_schedule, step_size_schedule, max_iter=None, patience=20):
    h_current = h0.copy()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    iter_count = 0
    for beta in beta_schedule:
        for _ in range(max_iter):
            iter_count += 1
            grad = smooth_max_gradient(h_current, beta)
            
            # compute a step using step_size_schedule
            current_step = step_size_schedule(iter_count)
            
            # update h
            h_candidate = h_current - current_step * grad
            
            # project back to [0,1] and fix mean 
            h_current = project_to_sum_box(h_candidate, TARGET_SUM)
            
            # evaluate
            c5 = evaluate_c5(h_current)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_current.copy()
                
            # Check for convergence
            if iter_count > patience and iter_count % patience == 0:
                pass  # For now let it run
            
            # Early stopping or convergance may be needed but we assume pass run to completion
            if max_iter is not None and iter_count >= max_iter:
                break
    return best_h, best_c5
