import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def shift_zero(arr, k):
    """Shift array with zero padding"""
    if k > 0:
        return np.concatenate([np.zeros(k), arr[:-k]])
    elif k < 0:
        return np.concatenate([arr[-k:], np.zeros(-k)])
    else:
        return arr.copy()

def compute_direction(h, lag):
    """Compute gradient direction for given lag"""
    n = len(h)
    shift_plus = shift_zero(h, -lag)
    shift_minus = shift_zero(h, lag)
    return shift_plus + shift_minus - 1

def subgradient_descent(h0, target_sum, dx, max_iter=2000, step0=0.3, time_limit=10.0, top_k=3):
    """Perform subgradient descent with top-k lag averaging"""
    h = h0.copy()
    best_h = h.copy()
    best_c = evaluate_c5(h)
    start = time.time()
    
    for it in range(max_iter):
        now = time.time()
        if now - start > time_limit:
            break
            
        corr = compute_corr_fft(h)
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        lag = max_idx - (len(h)-1)
        
        # Compute direction for top-k worst lags
        top_idxs = np.argpartition(corr, -top_k)[-top_k:]
        direction = np.zeros_like(h)
        for idx in top_idxs:
            l = idx - (len(h)-1)
            shift_plus = shift_zero(h, -l)
            shift_minus = shift_zero(h, l)
            direction += (shift_plus + shift_minus - 1)  # Negated gradient direction 
        direction /= top_k
        
        # Step size schedule
        step = step0 / (np.sqrt(it+1) + 1)
        
        # Update
        h_new = project_to_sum_box(h - step * direction, target_sum)
        
        # Evaluate new
        new_corr = compute_corr_fft(h_new)
        new_max_corr = np.max(new_corr)
        
        # Accept improvement
        if new_max_corr < best_c:
            best_c = new_max_corr
            best_h = h_new.copy()
            
        # Accept if better or with small probability
        if new_max_corr <= max_corr or np.random.rand() < 0.01:
            h = h_new
            
        # Occasionally add noise
        if np.random.rand() < 0.02:
            h = project_to_sum_box(h + np.random.normal(0, 0.01, len(h)), target_sum)
    
    return best_h, best_c

def local_refine_slsqp(h0, target_sum):
    """Perform local refinement with SLSQP"""
    bounds = [(0, 1) for _ in range(len(h0))]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 2000, 'ftol': 1e-9}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, target_sum)
            return h
    except:
        pass
    return h0

    restarts = 0

            best_c5 = c5_result
            best_h = h_result.copy()

    if time.time() - start_time < 0.9 * budget_s:
        remaining_time = budget_s - (time.time() - start_time)
