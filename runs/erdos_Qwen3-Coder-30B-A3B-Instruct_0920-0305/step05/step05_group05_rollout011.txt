import numpy as np
import time

def project_to_box_simplex(v, target_sum):
    """Project vector v onto the box [0,1]^n and simplex with sum = target_sum"""
    # Binary search for the correct lambda
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0, 1)

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

        """Compute the gradient of the smoothed max objective"""
        # Compute correlation O_k = sum h[i] * (1 - h[i+k])
        corr = np.correlate(h, 1 - h, mode="full")
        O = corr * dx
        
        # Smoothed max with numerical stability
        O_max = np.max(O)
        exp_term = np.exp(alpha * (O - O_max))
        w = exp_term / np.sum(exp_term)  # weights
        
        # Compute gradient
        grad = np.zeros_like(h)
        for idx, shift in enumerate(range(-n_points + 1, n_points)):
            w_k = w[idx]
            if w_k == 0:
                continue
                
            # Compute shift_right and shift_left
            if shift >= 0:
                shift_right = np.empty(n_points)
                if shift > 0:
                    shift_right[:shift] = 1.0
                shift_right[shift:] = 1.0 - h[shift:]
                
                shift_left = np.empty(n_points)
                if shift > 0:
                    shift_left[:shift] = 0.0
                shift_left[shift:] = h[:n_points-shift]
            else:
                s = -shift
                shift_right = np.empty(n_points)
                if s > 0:
                    shift_right[:s] = 1.0
                shift_right[s:] = 1.0 - h[:n_points-s]
                
                shift_left = np.empty(n_points)
                if n_points - s > 0:
                    shift_left[:n_points-s] = h[s:]
                shift_left[n_points-s:] = 0.0
                
            grad += w_k * (shift_right - shift_left)
        
        return grad * dx
    
    # Restart loop
    restarts = 0
    while (time.time() - start_time) < budget_s * 0.95:
        # Generate initial h for this restart
        restart_type = restarts % 4
        if restart_type == 0:
            # Random initialization
            h = np.random.rand(n_points)

            # Sine shape
            x = np.linspace(0, 2, n_points)
