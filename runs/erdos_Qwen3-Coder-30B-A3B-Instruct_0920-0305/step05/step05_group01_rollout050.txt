import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60
DX = 2.0 / 120.0

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    low, high = -2.0, 2.0
    for _ in range(50):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target_sum) < 1e-12:
            return y
        if s > target_sum:
            high = mid
        else:
            low = mid
    y = np.clip(x + low, 0.0, 1.0)
    return y

def evaluate_c5(h):
    """Evaluate C5 using correlation"""
    corr = correlate(h, 1.0 - h, mode='full')
    return corr.max() * DX

def compute_gradient(h, alpha):
    """Compute gradient of smoothed max objective"""
    n = len(h)
    # correlation of h with (1-h)
    cov_full = correlate(h, 1.0 - h, mode='full')
    I = cov_full  # shape 2*n-1
    
    I_max = I.max()
    # compute weights for softmax
    exp_vals = np.exp(alpha * (I - I_max))
    w = exp_vals / exp_vals.sum()
    
    # compute gradient as weighted sum of per-lag grads
    grad = np.zeros(n)
    
    for idx, p_val in enumerate(w):
        lag = idx - (n - 1)
        if lag >= 0:
            # (1 - h[i+lag]) for i=0..n-1-lag
            if n - lag > 0:
                grad[:n-lag] += p_val * (1.0 - h[lag:])
            # edge contributions: i where i+lag >= n
            if lag > 0:
                grad[n-lag:] += p_val * 1.0
            # subtract h[i-lag] for i=lag..n-1
            if lag > 0:
                grad[lag:] -= p_val * h[:n-lag]
            else:
                # lag==0 case
                grad -= p_val * h
        else:
            # lag negative
            d = -lag
            # (1 - h[i-d]) for i=d..n-1
            if n - d > 0:
                grad[d:] += p_val * (1.0 - h[:n-d])
            # edge contributions: i where i+lag < 0
            grad[:d] += p_val * 1.0
            # subtract h[i+d] for i=0..n-d-1
            grad[:n-d] -= p_val * h[d:]
    
    return grad

def descent_step(h, eta, alpha):
    """Perform one descent step with line search"""
    grad = compute_gradient(h, alpha)
    h_new = project_to_sum_box(h - eta * grad, TARGET_SUM)
    return h_new

def gradient_descent_run(h_start, time_limit):
    """Run gradient descent with adaptive schedule"""
    h = h_start.copy()
    c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5
    start_t = time.time()
    
    alpha_start = 0.5
    alpha_end = 50.0
    eta_start = 0.1
    
    max_iter = 5000
    for it in range(max_iter):
        if time.time() - start_t > time_limit:
            break
            
        progress = it / max_iter
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        eta = eta_start * (0.995 ** it)
        
        # line search
        h_new = descent_step(h, eta, alpha)
        c5_new = evaluate_c5(h_new)
        
        if c5_new <= c5:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # try smaller step size
            for _ in range(5):
                eta *= 0.5
                h_new = descent_step(h, eta, alpha)
                c5_new = evaluate_c5(h_new)
                if c5_new <= c5:
                    h = h_new
                    c5 = c5_new
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                    break
    
    return best_h, best_c5

def local_refine(h):
    """Refine solution using SLSQP optimizer"""
    def objective(h):
        return evaluate_c5(h)
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h))]
    
    try:
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
    except:
        pass
    return h, evaluate_c5(h)
