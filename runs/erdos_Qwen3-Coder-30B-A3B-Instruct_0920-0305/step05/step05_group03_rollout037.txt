import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Phase 1: Use global optimization with differential evolution
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # We'll use a more robust global optimizer
    try:
        res = differential_evolution(
            objective, 
            bounds, 
            maxiter=500, 
            popsize=15,
            seed=seed,
           _tol=1e-12,
            updating='deferred',
            disp=False,
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Phase 2: Local refinement with trust-constr
    local_restart_bound = 300
    
    for restart in range(local_restart_bound):
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
        # Generate new starting point
        new_h = best_h + np.random.normal(0, 0.02, n_points)
        new_h = np.clip(new_h, 0, 1)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='trust-constr', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 200, 'gtol': 1e-13, 'ftol': 1e-13}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue

    # Phase 3: Apply active-set refinement
    if (time.time() - start_time) < budget_s * 0.9:
        try:
            # Try to identify a good pattern 
            # A helpful approach is to make the function value
            # more concentrated at the boundaries 
            # Similar to the original but with different shape
            
            # Create a more concentrated version
            x = np.linspace(0, 2, n_points)
            
            # Cluster at the ends to reduce overlap 
            # Use exponentially decaying form to rapidly decrease towards centers
            cluster_pattern = 1.0 - np.exp(-10 * np.abs(x - 1)) + 0.01
            
            # Mix with best current solution and refit
            mixed_h = 0.7 * best_h + 0.3 * cluster_pattern
            
            # Normalize
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(mixed_h) * dx)
            mixed_h = mixed_h * scale_factor
            
            res = minimize(
                objective, 
                mixed_h, 
                method='trust-constr', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 200, 'gtol': 1e-14, 'ftol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass

    return (best_h.tolist(), best_c5, n_points)
