import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def evaluate_c5(h):
    """Evaluate C5 using direct correlation"""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation
    corr = correlate(h, 1 - h, mode="full")
    # Max correlation value scaled by dx
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project to feasible region with sum=target_sum and bounds [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def generate_initial_guess(n_points, method="interpolated"):
    """Generate initial guess"""
    if method == "interpolated":
        # Interpolate from initial_h_values
        h0 = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h0))
        x_new = np.linspace(0, 2, n_points)
        h_interp = np.interp(x_new, x_orig, h0)
        return project_to_sum_box(h_interp, TARGET_SUM)
    elif method == "binary":
        # Binary construction
        h = np.zeros(n_points)
        ones_idx = np.random.choice(n_points, n_points // 2, replace=False)
        h[ones_idx] = 1.0
        return project_to_sum_box(h, TARGET_SUM)
    elif method == "uniform":
        # Uniform construction
        h = np.random.uniform(0, 1, n_points)
        return project_to_sum_box(h, TARGET_SUM)
    else:
        # Default to original interpolated
        h0 = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h0))
        x_new = np.linspace(0, 2, n_points)
        h_interp = np.interp(x_new, x_orig, h0)
        return project_to_sum_box(h_interp, TARGET_SUM)

def optimize_step(h, iteration):
    """Perform one optimization step using gradient descent approach"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations
    corr = correlate(h, 1 - h, mode="full")
    max_lag = np.argmax(corr)
    
    # Approximate gradient using finite differences
    eps = 1e-4
    grad = np.zeros(n)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_sum_box(h_plus, TARGET_SUM)
        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
        corr_plus_max = np.max(corr_plus)
        grad[i] = (corr_plus_max - np.max(corr)) / eps
    
    # Update step
    step_size = 0.001 + 0.0005 * (1 - iteration / 1000.0)
    h_new = h - step_size * grad
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    max_iter = 50000
    
    # Initialize
    best_h = generate_initial_guess(n_points)
    best_c5 = evaluate_c5(best_h)
    
    # Gradient-based local optimization
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        h_new = optimize_step(best_h, i)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Further refinement with scipy optimizer
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(best_h))]
    
    try:
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            c5_refined = evaluate_c5(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass
        
    return (best_h.tolist(), float(best_c5), n_points)
