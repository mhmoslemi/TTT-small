import numpy as np
import time

def project_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex: 0 <= x <= 1, sum(x) = target_sum"""
    v = np.asarray(v)
    lmin, lmax = 0.0, 1.0
    for _ in range(100):
        lambda_val = (lmin + lmax) / 2.0
        x = np.clip(v - lambda_val, 0, 1)
        s = np.sum(x)
        if s > target_sum:
            lmin = lambda_val
        else:
            lmax = lambda_val
        if abs(s - target_sum) < 1e-10:
            break
    return np.clip(v - lmin, 0, 1)

def compute_O_and_grad(h, alpha):
    """Compute overlap vector O and its gradient w.r.t. the smoothed max"""
    n = len(h)
    dx = 2.0 / n
    # Compute overlap for all shifts
    O = dx * np.correlate(h, 1 - h, mode='full')
    # Center the overlap vector for convenient indexing
    # The full correlation is of length 2*n - 1
    # let's only need the main index from -n+1 to n-1
    # but to make things clearer, we'll take the middle part
    # which is the "central" n elements for k = 0 to n-1
    # Actually, for the gradient, we can work with the full corr vector
    # and compute over all possible indices.
    # But to be precise, let's compute O_k = dx * sum h_i * (1 - h_{i+k})
    # Over all i, k from 0 to 2n-2

    # Softmax weights
    O_max = np.max(O)
    w = np.exp(alpha * (O - O_max))
    w = w / np.sum(w)

    # Compute gradient ∂F/∂h = sum_k w_k * ∂O_k/∂h
    # ∂O_k/∂h[i] = dx * ( (1 - h[i+k]) - h[i-k] )
    # Create all shifted versions of h
    grad = np.zeros_like(h)
    for k in range(len(O)):
        # Shift h for this k
        # correlation at shift k = sum_{i=0}^{n-1} h[i] * (1 - h[i+k]) if 0 <= i+k < n, else 0
        # But for making vectorized code easier, handle the shifts manually
        if k < n:
            # positif shift: h must be shifted backward, indexed by i+k.
            # h[i − k] for positions where i-k >= 0, else h[0] ... But we should not count zero values beyond.
            shift_backward = k
            shift_forward = 0
            for i in range(n):
                if i + shift_backward < n:
                    # h[i] * (1 - h[i + shift_backward]) contributes to O_k
                    # So in place of h[i] we get -h[i + shift_backward], from 1 we get 1
                    # Add: w_k * dx * (1 - (h[0] = 0){when i is outside range})
                    grad[i] -= w[k] * dx * h[i + shift_backward]
                    if i >= shift_backward:
                        # Contribution is + w_k * dx * h[i - shift_backward] from - h[i - shift_backward]
                        grad[i - shift_backward] -= w[k] * dx * h[i]
                else:
                    # No contribution to overlap due to slicing outside
                    pass
        else:
            shift_backward = 0
            shift_forward = k - n + 1
            for i in range(n):
                if i - shift_forward >= 0:
                    # Contribution from h[i] * (1 - h[i - shift_forward])
                    grad[i] -= w[k] * dx * h[i - shift_forward]
                    if i + shift_forward < n:
                        # Contribution from - h[i + shift_forward]
                        grad[i + shift_forward] -= w[k] * dx * h[i]
                else:
                    pass

    # But another way to be clear, work using the convolved results
    # O_k = dx * sum_{i=0}^{n-1} h[i] * (1 - h[i+k]) if i+k < n
    # This can be rewritten:
    # O_k = dx * sum h[i] - sum h[i] * h[i+k]
    # And its derivative:
    # dO_k/dh[i] = dx ( 1 - h[i+k] ) - dx * h[i-k]
    # for all i, and for i-k, i+k that are in [0, n-1]
    # This is easy to compute via this frame of reference
    
    # To compute dO_k/dh[i], we need to check the forward and backward slices
    grad = np.zeros_like(h)
    O_k = np.zeros(len(O))
    for shift in range(len(O)):
        k_idx = shift - (n - 1)
        if k_idx >= 0:
            # shift forward: h[0] to h[n-1-k_idx] 
            # use indexing as h[i] * (1 - h[i+k_idx])
            # increasing contribution to h[i] by -dx * h[i+k_idx]
            # to h[i+k_idx] by -dx * h[i]
            # Ugh, might be easier to modify both ends
            # we have corr = correlate(h, 1 - h, mode='full')
            # then we look at O = dx * corr
            # For shift k in [0, 2*n-2], we compute products h[i] * (1 - h[i+k])
            # This is equivalent so:
            # O_k = dx * [sum_{i = 0}^{n-1} h[i] * (1 - h[i+k])] if i+k < n
            # For a fixed shift k:
            # Let's process this carefully:
            # i runs from 0 to n-1
            # Also need terms where i+k >= n -> 0 contribution, or
            # For i = 0 to n-1, if i+k < n, then contribute term
            # Vice versa with i+k = i - k' with k' = k, so we sum also terms from h[i ]* h[i-k] 
            # Better approach:
            # Sum i = 0 to n-1 of [h[i] * (1 - h[i+k])] = sum over i h[i] - sum over i h[i] * h[i+k]
            # Gradient wrt h[i] = dx * [1 - sum_j (1 * for j=i+k) - (h[i+k] if that's in bounds)]
            # Actually, individually, for O_k:
            # dO_k/dh[i] =
            # For term h[i], contribution = dx * (1 - h[i+k])  
            # For term h[i+k], contribution = - h[i] * dx
            # This approach more efficient:
            # We iterate over i, and redistribute appropriately (to systems like h[i], h[i+k])
            # Over all such valid entries where i+k < n
            # That is i = 0 to n−1−k
            # We do this efficiently with numpy:
            if k_idx < n:
                if k_idx == 0:
                    grad += dx # contribution of 1 term for all i
                else:
                    s1 = np.arange(n - k_idx)
                    s2 = np.arange(k_idx, n)
                    grad[s1] += -dx * h[s2] # -h[i+k_idx] from h[i] term
                    grad[s2] -= -dx * h[s1] # -h[i-k_idx] from h[i+k_idx] term
                    # But wait, we are overcounting!
                    # To be more precise, It's much simpler. 
                    # Key thing: 
                    # dO_k/dh[i] = dx * (1 - h[i+k_idx]) - dx * h[i-k_idx] 
                    # which is valid for valid i values that don't go out of bounds
                    # Need to be extra careful to handle zero indices

    grad = np.zeros_like(h)
    for shift in range(len(O)):
        k = shift - (n - 1)
        if k >= 0:
            # For shift k, iterate i from 0 to n-1-k
            # for j = i+k  其中 i 有效, 大小为 n-k-1
            # So for valid values:
            # Their contribution: dO_k/dh[i] =  dx*(1 - h[i+k]) and dO_k/dh[i+k] = -dx h[i]
            s1 = np.arange(n - k - 1)  
            s2 = s1 + k
            grad[s1] += dx * w[shift]
            grad[s2] -= dx * w[shift] * h[s1] 
        elif k < 0:
            # shift k < 0
            # So for shift k = -t, i ranges from 0 to n-1+t, then j = i+k = i-t
            # value: dO_k/dh[i] = -dx * h[i-k], dO_k/dh[i-k] = -dx*h[i]
            k_abs = -k
            s1 = k_abs + np.arange(n - k_abs)  # valid i values
            s2 = np.arange(n - k_abs)          # indices for expressing h[i-k]
            best_forward = min(n, s2 + k_abs + 1) # where s1 is up to by s2+k_abs
            if len(s1) <= len(s2):
                # Updated bond q
                s1 = s1[:len(s2)]  # cut off s1 at length of s2
                s2 = s2[:len(s1)]
            else:
                s2 = s2[:len(s1)]
                s1 = s1[:len(s2)]
            # host-gpu:i-k
            # allocate to 下标
            grad[s1] -= dx * w[shift] * h[s2]  # this part above
            grad[s2] += dx * w[shift]          # minus (-) from 1
    # Normalisation done in update steps later, so just add
    return O, grad

            # Gaussian
            x = np.linspace(0, 2, n)

        elif which_seed == 3:
            # Triangle
            x = np.linspace(0, 2, n)
            h = 1 - np.abs(x - 1.0)
            h = np.clip(h, 0, 1)
        else:
            # Binary
            h = np.ones(n) * 0.5
            mid = n // 2
            width = n // 4
            start = max(0, mid - width // 2)
            end = min(n, mid + width // 2)
            h[start:end] = 1.0
            h = np.clip(h, 0, 1)
            
        # Project to constraint
        h = project_capped_simplex(h, target_sum)

        alpha_start = 5.0
        alpha_end = 30.0

            O, grad = compute_O_and_grad(h, alpha)

            lr = lr_0 / np.sqrt(iter + 1)
            h = h - lr * grad

            if iter % 10 == 0:
