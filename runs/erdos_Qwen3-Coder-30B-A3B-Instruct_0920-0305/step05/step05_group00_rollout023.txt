import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.irfft(np.fft.rfft(h_rev, L) * np.fft.rfft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def adaptive_random_move(h, prob):
    """Generate a random transfer move that respects constraints."""
    if np.random.random() < prob:
        # Prefer moves closer to boundaries
        idxs = np.arange(n_points)
        # Put more weight on boundary points
        weights = np.minimum(idxs, n_points - 1 - idxs) + 1
        i = np.random.choice(idxs, p=weights/np.sum(weights))
    else:
        i = np.random.randint(n_points)
    j = np.random.randint(n_points)
    if i == j:
        return None
    
    hi, hj = h[i], h[j]
    delta_min = max(-hj, hi - 1)
    delta_max = min(hi, 1 - hj)
    if delta_min >= delta_max:
        return None
    
    delta = np.random.uniform(delta_min, delta_max)
    return (i, j, delta)

def targeted_optimization(h0, run_seconds, rng):
    """Perform targeted optimization around current solution."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Different initialization strategies
    initial_modes = [
        lambda: h0.copy(),
        lambda: np.random.uniform(0, 1, n_points),
        lambda: np.random.choice([0, 1], size=n_points, p=[0.5, 0.5]),
    ]
    
    for _ in range(20):
        if time.time() > run_seconds:
            break
        mode = initial_modes[rng.choice(len(initial_modes))]
        h_start = mode()
        h_start = project_to_sum_box(h_start, TARGET_SUM)
        
        # SLSQP local optimization
        bounds = [(0, 1) for _ in range(n_points)]
        def objective(x):
            return evaluate_c5(x)
        def constraint_eq(x):
            return np.sum(x) - TARGET_SUM
        cons = {'type': 'eq', 'fun': constraint_eq}
        try:
            res = minimize(objective, h_start, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12})
            if res.success:
                h_candidate = res.x
                h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
            
        # Gradient-based guided move using larger steps
        corr = compute_corr_fft(h)
        worst_lag = np.argmax(corr)
        if worst_lag < n_points//2:  # Only consider small lags for speed
            for _ in range(10):
                i = (worst_lag + np.random.randint(n_points)) % n_points
                j = (i + worst_lag) % n_points
                hi, hj = h[i], h[j]
                delta_min = max(-hj, hi - 1)
                delta_max = min(hi, 1 - hj)
                if delta_min < delta_max:
                    delta = (delta_min + delta_max) / 4
                    if delta > 0:
                        h[i] -= delta
                        h[j] += delta
                        h = project_to_sum_box(h, TARGET_SUM)
                        c5 = evaluate_c5(h)
                        if c5 < best_c5:
                            best_h = h.copy()
                            best_c5 = c5
                        
    return best_h, best_c5
