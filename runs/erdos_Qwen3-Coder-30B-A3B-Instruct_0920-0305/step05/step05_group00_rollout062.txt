import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from copy import deepcopy

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def guided_move(h, worst_lag):
    """Guided deterministic move for the worst lag"""
    g = np.zeros(n_points)
    for i in range(n_points):
        if i-worst_lag >= 0:
            g[i] += h[i-worst_lag]
        if i+worst_lag < n_points:
            g[i] += h[i+worst_lag]
    
    idx_min = np.argmin(g)
    idx_max = np.argmax(g)
    if idx_min == idx_max:
        return None
    
    delta = min(h[idx_min], 1 - h[idx_max])
    if delta <= 0:
        return None
    
    return (idx_min, idx_max, delta)

def adaptive_anneal(h0, run_seconds, rng):
    """Improved simulated annealing with adaptive temperature"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Adaptive cooling schedule
    T0 = 0.1
    Tf = 1e-6
    t_start = time.time()
    iteration = 0
    
    while time.time() - t_start < run_seconds:
        T = T0 * (0.99 ** iteration) + Tf * (1 - 0.99 ** iteration)
        iteration += 1
        
        # Prefer guided moves with decreasing probability
        if rng.random() < 0.15:  # 15% guided
            corr = compute_corr_fft(h)
            worst_lag = np.argmax(corr)
            move = guided_move(h, worst_lag)
        else:
            # Random move with bias towards smaller changes
            i, j = rng.choice(n_points, 2, replace=False)
            hi, hj = h[i], h[j]
            delta_min = max(-hj, hi - 1)
            delta_max = min(hi, 1 - hj)
            if delta_min >= delta_max:
                continue
            delta = rng.uniform(delta_min, delta_max)
            delta *= 0.1  # Small changes
            move = (i, j, delta)
        
        if move is None:
            continue
            
        i, j, delta = move
        h[i] -= delta
        h[j] += delta
        
        # Project to constraint
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Evaluate
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Accept or reject using Metropolis criterion
        if c5 >= best_c5:
            if rng.random() > np.exp(-(c5 - best_c5) / T):
                h[i] += delta
                h[j] -= delta
                h = project_to_sum_box(h, TARGET_SUM)
    
    return best_h, best_c5

def local_refine(h, max_iter=1000):
    """Improved local refinement with multiple restarts"""
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Try multiple restarts
    for _ in range(3):
        try:
            h_start = best_h + np.random.normal(0, 0.02, n_points)
            h_start = project_to_sum_box(h_start, TARGET_SUM)
            res = minimize(
                objective, 
                h_start, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': max_iter, 'ftol': 1e-10, 'gtol': 1e-10}
            )
            if res.success:
                h_candidate = res.x
                h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate
        except:
            pass
    
    return best_h, best_c5

    ]

        elif strategy_idx % 3 == 1:  # Exponential decay pattern

        else:  # Random continuous
            h_in = np.random.uniform(0, 1, n_points)
            h_in = project_to_sum_box(h_in, TARGET_SUM)
        
        h_candidate, c5_candidate = adaptive_anneal(h_in, 8, rng)
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate

        if strategy_idx % 5 == 0:
            h_test, c5_test = local_refine(best_h.copy())
            if c5_test < best_c5:
                best_h = h_test
                best_c5 = c5_test
    
    # Phase 2: Refinement with dual optimization
    phase2_end = start_time + 0.9 * budget_s
    
    # Perform a few rounds of local refinement with different starting points

            break

        if c5_test < best_c5:
            best_h = h_test
            best_c5 = c5_test
    
    # Phase 3: Final polishing with gradient-based methods
    while time.time() < start_time + budget_s:
        # Add small perturbations and try to refine
        h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
        h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
        
        h_test, c5_test = local_refine(h_perturbed)
        if c5_test < best_c5:
            best_h = h_test
            best_c5 = c5_test
    
    return best_h, best_c5
