import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def project_simplex(h, target_sum):
    """Project h onto the simplex with sum = target_sum and bounds [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    # Use bisection to find lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):
        lam = (lb + ub) / 2.0
        h_new = np.clip(h + lam, 0, 1)
        s = np.sum(h_new)
        if s > target_sum:
            ub = lam
        else:
            lb = lam
        if abs(s - target_sum) < 1e-12:
            break
    return np.clip(h + lb, 0, 1)

def evaluate_c5(h):
    """Evaluate C5 using exact max overlap"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    # Take the maximum from the valid non-zero shifts
    max_corr = np.max(corr[n:]) # Skip the zero shift
    return max_corr * dx

def compute_corr_and_grad(h, beta):
    """Compute correlation vector and gradient via FFT"""
    n = len(h)
    # Zero-padded convolution
    h_pad = np.pad(h, n-1, mode='constant', constant_values=0)
    corr = np.convolve(h_pad, h, mode='valid')[:n]
    # corr[k] = sum_i h[i]*h[i+k] for k=0..n-1
    # we exclude k=0
    corr = corr[1:] # drop k=0
   
    # weights for the soft minimum
    corr_shifted = corr - np.max(corr) # prevent overflow
    exp_corr = np.exp(-beta * corr_shifted)
    w = exp_corr / np.sum(exp_corr)
    
    # Gradient via FFT
    # g[k] = - sum_t w[t] (h[k-t] + h[k+t]) for t=1..n-1
    h_rev = np.flip(h)
    # pad and reverse for the left shift
    h_rev_pad = np.pad(h_rev, n-1, mode='constant', constant_values=0)
    conv_backward = np.convolve(h_rev_pad, w, mode='valid')[n-1:]
    # forward shift
    h_pad = np.pad(h, n-1, mode='constant', constant_values=0)
    conv_forward = np.convolve(h_pad, w, mode='valid')[:n]
    
    g = -(conv_backward + conv_forward)
    
    # soft-min value
    phi_beta = (1 / beta) * np.log(np.sum(exp_corr))
    
    return corr, g, phi_beta

def optimize_single(h_start, beta_0=1.0, beta_max=200, gamma=2.0, niter=2000, max_time=540):
    """Single run of projected gradient descent with continuation"""
    n = len(h_start)
    h = h_start.copy()
    h = project_simplex(h, n/2)  # make sum correct
    
    beta = beta_0
    step_size = 0.5 * (2.0 / n)
    best_c5 = float('inf')
    best_h = h.copy()
    
    for i in range(niter):
        if time.time() > max_time:
            break
            
        c, g, phi_beta = compute_corr_and_grad(h, beta)
        c5_val = 1 + phi_beta  # our surrogate bound
        
        # Backtracking line search
        tau = step_size
        max_attempts = 10
        for _ in range(max_attempts):
            h_trial = h - tau * g
            h_trial = project_simplex(h_trial, n/2)
            c5_trial = evaluate_c5(h_trial)
            
            if c5_trial < c5_val or tau < 1e-12:
                h = h_trial
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h.copy()
                break
            tau *= 0.5
            
        # Continuation
        if i % 50 == 0 and beta < beta_max:
            beta *= gamma
            step_size *= 0.9
            
        if i % 100 == 0:
            if np.linalg.norm(g) < 1e-8:
                step_size *= 0.5
                if step_size < 1e-12:
                    break
    return best_h, best_c5
