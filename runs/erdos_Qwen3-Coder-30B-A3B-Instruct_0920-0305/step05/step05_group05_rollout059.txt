def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def genetic_strategy(best_h, bounds, cons, objective, n_points, start_time, budget_s):
    # Simple genetic algorithm inspired approach
    population_size = 30
    population = []
    
    # Initialize population
    for _ in range(population_size):
        h = np.random.uniform(0, 1, n_points)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(h) * dx)
        h = h * scale_factor
        population.append(h)
    
    # Evaluate population
    fitness = [objective(h) for h in population]
    
    # Evolve for a few generations
    for gen in range(20):
        if (time.time() - start_time) >= budget_s * 0.95:
            break
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:population_size // 3]
        
        # Create new generation
        new_population = [population[i] for i in top_indices]
        
        # Crossover and mutation
        for _ in range(population_size - len(top_indices)):
            parent1 = population[np.random.choice(top_indices)]
            parent2 = population[np.random.choice(top_indices)]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            mutate_indices = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
            for i in mutate_indices:
                child[i] += np.random.normal(0, 0.02)
            child = np.clip(child, 0, 1)
            
            # Normalize
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(child) * dx)
            child = child * scale_factor
            
            new_population.append(child)
        
        population = new_population
        fitness = [objective(h) for h in population]
    
    # Return best
    best_idx = np.argmin(fitness)
    return population[best_idx].copy(), fitness[best_idx]

                scale = 1.0 / (np.sum(h_candidate) * dx)

                if c5_candidate < best_result[1]:
