import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60
DX = 2.0 / 120

def evaluate_c5(h):
    """Evaluate C5 using correlation"""
    corr = correlate(h, 1.0 - h, mode='full')
    return corr.max() * DX

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -2.0, 2.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def gradient_smoothing(h, alpha):
    """Compute gradient of log-sum-exp approximation of max correlation"""
    n = len(h)
    # correlation of h with (1-h)
    cov_full = correlate(h, 1.0 - h, mode='full')
    I = cov_full  # shape 2n-1
    I_max = I.max()
    
    # compute weights for softmax (avoid numerical overflow)
    exp_vals = np.exp(alpha * (I - I_max))
    w = exp_vals / exp_vals.sum()
    
    # compute gradient as weighted sum of per-lag gradients
    grad = np.zeros(n)
    
    # Pre-compute arrays
    h_arr = h
    one_minus_h = 1.0 - h_arr
    
    for idx, p_val in enumerate(w):
        lag = idx - (n - 1)  # ℓ from -(n-1) to n-1
        
        if lag >= 0:
            # contribution (1 - h[i+lag]) for i from 0 to n-1-lag
            if n - lag > 0:
                grad[:n-lag] += p_val * one_minus_h[lag:]
            # edge contribution where i+lag >= n: (1 - 0) = 1
            if lag > 0:
                grad[n-lag:] += p_val * 1.0
            # subtract h[i-lag] for i from lag to n-1 
            if lag > 0:
                grad[lag:] -= p_val * h_arr[:n-lag]
            else:
                # lag == 0 case
                grad -= p_val * h_arr
        else:
            # lag < 0
            d = -lag
            # contribution (1 - h[i-d]) for i from d to n-1
            if n - d > 0:
                grad[d:] += p_val * one_minus_h[:n-d]
            # edge contribution where i+lag < 0: (1 - 0) = 1
            grad[:d] += p_val * 1.0
            # subtract h[i+d] for i from 0 to n-d-1
            grad[:n-d] -= p_val * h_arr[d:]
    
    return grad

def descent_step(h, eta, alpha):
    """Perform one descent step"""
    grad = gradient_smoothing(h, alpha)
    h_new = project_to_sum_box(h - eta * grad, TARGET_SUM)
    return h_new

def gradient_descent_run(h_start, time_limit):
    """Run gradient descent optimization"""
    h = h_start.copy()
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    t0 = time.time()
    it = 0
    alpha_start = 0.5
    alpha_end = 50.0
    eta_start = 0.1
    
    while time.time() - t0 < time_limit and it < 5000:
        # Schedule alpha and eta
        progress = min(it / 1000, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        eta = eta_start * (0.995 ** it)
        
        h_new = descent_step(h, eta, alpha)
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 <= current_c5:
            h = h_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        else:
            # Small random restart
            if np.random.rand() < 0.01:
                h = project_to_sum_box(h + np.random.normal(0, 0.01, len(h)), TARGET_SUM)
                current_c5 = evaluate_c5(h)
        
        it += 1
    
    return best_h, best_c5

def local_refine(h):
    """Refine solution using SLSQP optimizer"""
    def objective(h):
        return evaluate_c5(h)
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h))]
    
    try:
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
    except:
        pass
    return h, evaluate_c5(h)

        noise_level = 0.05
        h_start = project_to_sum_box(h0 + np.random.normal(0, noise_level, n_points), target_sum)
        
        # Run gradient descent from this point
        descent_time = min(remaining_time / max_restarts * 1.2, 120)

            best_c5 = c5_trial
