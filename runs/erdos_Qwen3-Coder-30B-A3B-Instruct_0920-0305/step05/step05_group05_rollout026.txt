import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction as a baseline
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Fourier-based approach
    # Choose L based on the number of points in h0
    L = min(15, n_points // 4)  # Cap at 15 for reasonable runtime
    if L < 3:
        L = 3

    # Generate dense sampling points in [0, 2] for enforcing pointwise bounds
    M = max(4 * L, 64)  # At least 4L points, but not fewer than 64
    x_sample = np.linspace(0, 2, M)

    # Define variables
    a = cp.Variable(L)      # cosine coefficients
    b = cp.Variable(L)      # sine coefficients
    s = cp.Variable(L)      # squared magnitudes
    t = cp.Variable(1)      # upper bound on max overlap

    # Build the Fourier series at sample points
    # h(x) = 0.5 + sum_{l=1}^L (a_l*cos(pi*l*x) + b_l*sin(pi*l*x))
    H_vals = 0.5 + np.sum(
        np.array([a_l * np.cos(np.pi * l * x_sample) + b_l * np.sin(np.pi * l * x_sample)
                  for a_l, b_l, l in zip(a.value, b.value, range(1, L+1))]),
        axis=0
    ) if a.value is not None else 0.5 + np.zeros(M)

    # Constraints to enforce h(x) in [0,1] at sample points
    A = np.array([[np.cos(np.pi * l * x) + np.sin(np.pi * l * x) for l in range(1, L+1)] 
                  for x in x_sample])
    # This approach does not use full matrix due to symbolic limitation. 
    # Instead, we build the constraints term by term.

    # Build linear constraints manually
    cons = []
    for x in x_sample:
        # h(x) = 0.5 + sum_{l=1}^L (a_l * cos(pi*l*x) + b_l * sin(pi*l*x))
        linear_expr = 0.5
        for l in range(1, L+1):
            linear_expr += a[l-1] * np.cos(np.pi * l * x) + b[l-1] * np.sin(np.pi * l * x)
        cons.append(linear_expr >= 0)
        cons.append(linear_expr <= 1)

    # SOCP constraints: s_l >= a_l^2 + b_l^2
    soc_constraints = [cp.SOC(s[l-1] + 1, [2*a[l-1], 2*b[l-1], s[l-1] - 1]) for l in range(1, L+1)]

    # Matrix constraint for the Toeplitz matrix to bound the max
    # T(t,s) = [ t -s_1 -s_2 ... -s_L;
    #           -s_1  t -s_1 ... -s_{L-1};
    #           ...
    #           -s_L -s_{L-1} ... t ]
    # T must be positive semidefinite: T >= 0
    T = cp.Variable((L+1, L+1), symmetric=True)
    T[0, 0] = t
    for i in range(L):
        T[0, i+1] = -s[i]
        T[i+1, 0] = -s[i]
        for j in range(L):
            if i == j:
                T[i+1, j+1] = t
            else:
                T[i+1, j+1] = -s[abs(i-j)]
    
    # Add SDP constraint
    psd_constraint = [T >> 0]

    # Objective
    objective = cp.Minimize(t)

    # Formulate complete problem
    # Note: optimization over small variables only, purposefully ignoring the divergence between symbolic and 
    #               problem schematic in the earlier text because proving full structure is not necessary for clear code.
    # Actual effective optimization of assignments is done via iterative solver.

    best_c5 = float('inf')
    best_h = None

    # Try different values of L to increase accuracy until timeout
    L_vals = [min(L, 10), min(L+1, 12), min(L+2, 15)]
    for l_val in L_vals:
        if (time.time() - start_time) > budget_s * 0.9:
            break
        try:
            # Reconstruct the problem and solve
            L = l_val
            
            # Variables: a, b, s are L-sized
            a = cp.Variable(L)      
            b = cp.Variable(L)      
            s = cp.Variable(L)      
            t = cp.Variable(1)      
            
            # Constraints for bounds [0,1] at sample points x_sample
            cons = []
            for x in x_sample:
                expr = 0.5
                for l in range(1, L+1):
                    expr += a[l-1] * np.cos(np.pi * l * x) + b[l-1] * np.sin(np.pi * l * x)
                cons.append(expr >= 0)
                cons.append(expr <= 1)
            
            # SOCP constraints
            soc_constraints = [
                cp.SOC(s[l-1] + 1, [2*a[l-1], 2*b[l-1], s[l-1] - 1]) 
                for l in range(1, L+1)
            ]
            
            # DT Toeplitz matrix
            T = cp.Variable((L+1, L+1), symmetric=True)
            T[0,0] = t  # constraint for entry (0,0)
            for i in range(L):
                T[0, i+1] = -s[i]
                T[i+1, 0] = -s[i]
                for j in range(L):
                    if i == j:
                        T[i+1, j+1] = t
                    else:
                        T[i+1, j+1] = -s[abs(i-j)]
                        
            psd_constraint = [T >> 0]
            
            prob = cp.Problem(objective, cons + soc_constraints + psd_constraint)
            prob.solve(solver=cp.SCS, verbose=False, max_iters=5000)

            if prob.status == cp.OPTIMAL:
                # Extract values
                a_val = a.value
                b_val = b.value
                s_val = s.value
                t_val = t.value
                
                # Now construct h using these coefficients over original grid
                x = np.linspace(0, 2, n_points)
                h_reconstructed = 0.5 + np.sum(
                    [a_val[l-1] * np.cos(np.pi * l * x) + b_val[l-1] * np.sin(np.pi * l * x)
                     for l in range(1, L+1)],
                    axis=0
                )
                h_reconstructed = np.clip(h_reconstructed, 0, 1)
                
                # Renormalize to sum to 1
                dx = 2.0 / n_points
                scale = 1.0 / (np.sum(h_reconstructed) * dx)
                h_reconstructed = h_reconstructed * scale
                
                # Compute actual C5
                c5 = np.max(correlate(h_reconstructed, 1-h_reconstructed, mode="full")) * dx
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_reconstructed.copy()
    
        except Exception as e:
            continue

    # If no solution found from SOCP, fallback to previous method
    if best_h is None:
        h0 = np.array(initial_h_values)
        n_points = len(h0)

        # Create bounds for optimization (h[i] in [0,1])
        bounds = [(0, 1) for _ in range(n_points)]
        
        def evaluate_c5(h):
            """Evaluate the C5 constant for a given step function h"""
            n = len(h)
            dx = 2.0 / n
            # Compute the overlap integral for all shifts
            corr = correlate(h, 1 - h, mode="full")
            # The maximum overlap value
            max_corr = np.max(corr)
            return max_corr * dx
        
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            # Integral constraint: sum(h) * dx = 1
            dx = 2.0 / n_points
            return np.sum(h) * dx - 1.0
        
        # Set up constraints
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        best_h = h0.copy()
        best_c5 = evaluate_c5(h0)
        
        # Try to improve the initial solution
        attempts = 0
        max_attempts = 300
        while attempts < max_attempts and (time.time() - start_time) < budget_s * 0.95:
            indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
            new_h = best_h.copy()
            for i in indices:
                new_h[i] += np.random.normal(0, 0.05)
            new_h = np.clip(new_h, 0, 1)
            
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(new_h) * dx)
            new_h = new_h * scale_factor
            
            try:
                from scipy.optimize import minimize
                res = minimize(
                    objective, 
                    new_h, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=cons,
                    options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-10}
                )
                if res.success:
                    h_candidate = res.x
                    scale = 1.0 / (np.sum(h_candidate) * dx)
                    h_candidate = h_candidate * scale
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
            except:
                pass
                
            attempts += 1

    return (best_h.tolist(), best_c5, n_points)
