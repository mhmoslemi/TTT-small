import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

    x_sample = np.linspace(0, 2, M, endpoint=False)
    
    # Decision variables for Fourier coefficients
    a = cp.Variable(L)
    b = cp.Variable(L)
    s = cp.Variable(L, nonneg=True)
    t = cp.Variable()
    
    # Constraint: s_l >= a_l^2 + b_l^2  (SOC constraints)
    soc_constraints = [cp.norm(cp.hstack([2*a[l], 2*b[l], s[l] - 1]), 2) <= s[l] + 1 
                       for l in range(L)]
    
    # Define the matrix T(t, s) for the Chebyshev bound
    # We construct Toeplitz matrix of size (L+1) x (L+1)
    rows = [t] + [-s[l] for l in range(L)]
    T_rows = [rows]
    for k in range(L):
        row = [rows[0]] + rows[1:][L-k:] + rows[1:][:k]
        T_rows.append(row)
    T = cp.vstack([cp.hstack(row) for row in T_rows])
    
    # PSD constraint for Toeplitz matrix
    psd_constraint = [T >> 0]
    
    # Box constraints: 0 <= h(x_j) <= 1
    box_constraints = []

        h_val = 0.5 + np.sum([a[l] * np.cos(np.pi * (l+1) * x) + 
                              b[l] * np.sin(np.pi * (l+1) * x) for l in range(L)])
        box_constraints.append(0 <= h_val)
        box_constraints.append(h_val <= 1)
