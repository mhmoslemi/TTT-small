def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def greedy_construction(n):
    """
    Greedy algorithm to construct a good binary pattern.
    """
    x = np.zeros(n, dtype=int)
    ones = 0
    target = n // 2
    positions = list(range(n))
    np.random.shuffle(positions)
    
    # Try to build set with minimal cooperation between shifts
    for i in range(n):
        if ones >= target:
            break
        idx = positions[i]
        if x[idx] == 0:
            # Check impact on pairs
            # For simplicity, just add the element
            x[idx] = 1
            ones += 1            
    
    return x.tolist()

def solve_milp(n, initial_guess, time_limit, start_time, total_budget):
    """
    Solve MILP to find optimal binary pattern for minimum overlap.
    """
    dx = 2.0 / n
    target = n // 2
    
    # Initial guess
    x_start = np.array(initial_guess)
    
    # Variables
    x = cp.Variable(n, boolean=True)
    L = cp.Variable(integer=True)
    
    # Constraints
    constraints = [cp.sum(x) == target, L >= 0]
    
    # Add pair constraints
    for k in range(1, n):
        p_k = cp.sum(
            [cp.minimum(x[i], x[i + k]) for i in range(n - k)]
        )
        constraints.append(p_k >= L)
    
    # Objective
    objective = cp.Maximize(L)
    prob = cp.Problem(objective, constraints)
    
    # Setup solve time limit
    time_left = max(1.0, time_limit - 5)
    
    try:
        prob.solve(solver=cp.CBC, time_limit=time_left, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            # Extract solution
            x_sol = np.round(x.value).astype(int)
            L_val = int(L.value)
            return (x_sol.tolist(), L_val)
        else:
            # Return best feasible in case optimizer fails
            if np.all(x_start >= 0) and np.sum(x_start) == target:
                # Just evaluate the initial guess
                # This is a rough check - not a deep MILP feasibility check
                return (x_start.tolist(), -1)  # Placeholder
            
    except Exception as e:
        pass
    
    return None

def polish_solution(h, n_points):
    """
    Apply L-BFGS-B optimizer onto continuous version of the solution.
    """
    dx = 2.0 / n_points
    
    def objective(h_vec):
        # Evaluate overlap integral for all shifts
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx
        
    def objective_smooth(h_vec, eps=1e-6):
        # Use differentiable surrogate for max
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return -np.log(np.sum(np.exp(corr / eps))) * dx * eps
        
    def gradient(h_vec):
        # Approximate gradient
        grad = np.zeros_like(h_vec)
        epsilon = 1e-4
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += epsilon
            h_minus = h_vec.copy()
            h_minus[i] -= epsilon
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * epsilon)
        return grad
        
    try:
        h_initial = np.array(h)
        res = minimize(
            objective_smooth, 
            h_initial, 
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            jac=gradient,
            options={'maxiter': 200, 'gtol': 1e-6}
        )
        if res.success:
            h_final = res.x
            # Renormalize
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            return h_final
    except:
        pass
    return h
