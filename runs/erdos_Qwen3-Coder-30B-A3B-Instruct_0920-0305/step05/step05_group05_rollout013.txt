import numpy as np
import time

def project_capped_simplex(v, T):
    """Project v onto {x : 0 <= x <= 1, sum(x) = T} using binary search"""
    v = np.asarray(v)
    low, high = 0.0, 1.0
    while high - low > 1e-10:
        mid = (low + high) / 2
        projected = np.clip(v - mid, 0, 1)
        if np.sum(projected) > T:
            low = mid
        else:
            high = mid
    return np.clip(v - low, 0, 1)

def compute_O_and_grad(h, alpha):
    """Compute the vector of overlaps O_k and its gradient w.r.t the smoothed max"""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps for shifts k = 0..2*n-2
    corr = np.correlate(h, 1 - h, mode='full')   # length is 2*n-1
    O = corr * dx  # This includes the correct dx scaling

    # Smoothed max
    m = np.max(O)
    exp_O = np.exp(alpha * (O - m))
    w = exp_O / np.sum(exp_O)

    # Gradient of smooth max
    grad = np.zeros(n)
    for i in range(n):
        for k in range(2 * n - 1):
            if k < n:
                shift = k
                offset = i + shift
                if offset < n:
                    grad[i] += w[k] * (1.0 - h[offset])
            if k >= n:
                shift = k - n + 1
                offset = i - shift
                if offset >= 0:
                    grad[i] += w[k] * (-h[offset])

    grad *= dx

    return O, grad
