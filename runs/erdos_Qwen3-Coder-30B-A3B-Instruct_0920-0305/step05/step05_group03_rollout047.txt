import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Initialize best solution
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Use adaptive optimization with multiple strategies
    
    # Strategy 1: Differential evolution for global search (faster convergence)
    try:
        res = differential_evolution(
            objective,
            bounds,
            maxiter=500,
            popsize=15,
            seed=seed,
            disp=False,
            tol=1e-10
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
                print(f"DE improved solution: {best_c5}")
    except:
        pass

    # Strategy 2: Local refinement with trust-constr
    try:
        res = minimize(
            objective,
            best_h,
            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'gtol': 1e-12, 'ftol': 1e-12}
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
                print(f"Trust-constr improved solution: {best_c5}")
    except:
        pass

    # Strategy 3: Take multiple random local searches
    local_restart_bound = 50
    for restart in range(local_restart_bound):
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
        # Random perturbation
        new_h = best_h + np.random.normal(0, 0.02, n_points)
        new_h = np.clip(new_h, 0, 1)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective,
                new_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-13, 'gtol': 1e-13}
            )
            if res.success:
                h_candidate = res.x
                dx = 2.0 / n_points
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
                    print(f"Local search improved solution: {best_c5}")
        except:
            continue

    # Strategy 4: Specialized pattern search
    if (time.time() - start_time) < budget_s * 0.8:
        # Try periodic pattern with quality control
        try:
            # Try symmetric pattern with smooth transitions
            # Construct a piecewise linear function (not step function but close)
            x = np.linspace(0, 2, n_points)
            # Target shape: a small central spike with flattish ends
            target = np.zeros(n_points)
            center = n_points // 2
            width = n_points // 5
            
            # Peaks in the center but not too high relative to other points
            # Create peak shape using a quadratic function + minimal spread
            for i in range(n_points):
                dist = abs(i - center) / (width / 2)
                # Smooth peak shape centered
                target[i] = max(0, 1.0 - dist**2)
            
            # Normalize
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(target) * dx)
            target = target * scale_factor
            
            # Refine with local optimizer
            res = minimize(
                objective,
                target,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
                    print(f"Pattern search improved solution: {best_c5}")
        except:
            pass

    return (best_h.tolist(), best_c5, n_points)
