def project_to_simplex(h, target_sum):
    """Project a vector h onto the bounded simplex with sum = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    
    # Use bisection on the shift parameter lambda
    def sum_shifted(lam):
        return np.sum(np.clip(h + lam, 0, 1))
    
    # Find lambda such that sum_shifted(lambda) == target_sum
    a, b = -1.0, 1.0
    while sum_shifted(a) > target_sum:
        a -= 1.0
    while sum_shifted(b) < target_sum:
        b += 1.0
    
    tol = 1e-12
    max_iter = 30
    for _ in range(max_iter):
        if abs(sum_shifted(a) - target_sum) < tol:
            break
        if abs(sum_shifted(b) - target_sum) < tol:
            break
        mid = (a + b) / 2
        if sum_shifted(mid) < target_sum:
            a = mid
        else:
            b = mid
    lam = (a + b) / 2
    h = np.clip(h + lam, 0, 1)
    return h

def compute_correlations(h):
    """Compute correlations of h with itself."""
    n = len(h)
    # Use full convolution for easier indexing
    c = correlate(h, h, mode='full')  # Length 2n-1
    # Extract the relevant part (shifts 1 to n-1)
    # Note: correlation at shift 0 is used in the denominator of weights
    return c[n-1:]

def evaluate_c5(h):
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    dx = 2.0 / len(h)
    return max_corr * dx

def compute_softmin_gradient(h, c, beta):
    """Compute gradient of the soft-min objective."""
    n = len(h)
    # Avoid numerical overflows by subtracting max
    c_max = np.max(c)
    exp_c = np.exp(-beta * (c - c_max))
    w = exp_c / np.sum(exp_c)
    # Gradient: - sum_{k=1}^{n-1} w_k * (h_{i+k} + h_{i-k})
    # Computed via convolution with w. 
    # For i=0 to n-1, we need sum_{k=1}^{n-1} w_k * h_{i+k} + w_k * h_{i-k}
    # We use zero-padding to make it linear convolution.
    # Convolution of h with w with negative indices is:
    w_padded = np.concatenate([w[::-1], np.zeros(n)])
    h_padded = np.concatenate([h, np.zeros(n)])
    grad_forward = np.convolve(h_padded, w_padded, mode='valid')[:n]
    # Reflect for backward
    w_backward = w[::-1][:-1]
    h_backward = np.concatenate([h[::-1], np.zeros(n-1)])
    grad_backward = np.convolve(h_backward, w_backward, mode='valid')[-n:][::-1]
    
    # For i = 0...n-1
    g = -(grad_forward + grad_backward)
    
    return g

def sqrt_line_search(h, g, beta, dx, max_iter=20):
    """Perform backtracking line search."""
    alpha = 0.5 * dx  # Initial step size
    c_old = 1.0 + (1.0 / beta) * np.log(np.sum(np.exp(-beta * (compute_correlations(h) - np.max(compute_correlations(h)))))
    for _ in range(max_iter):
        h_new = h - alpha * g
        h_new = project_to_simplex(h_new, len(h)/2)
        c_new = 1.0 + (1.0 / beta) * np.log(np.sum(np.exp(-beta * (compute_correlations(h_new) - np.max(compute_correlations(h_new)))))
        if c_new <= c_old and alpha > 1e-15:
            h = h_new
            c_old = c_new
            break
        alpha *= 0.5
    return h, alpha
