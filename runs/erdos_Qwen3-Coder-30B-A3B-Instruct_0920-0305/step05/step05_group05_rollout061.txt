import numpy as np
import time

def project_to_box_simplex(v, target_sum):
    """Project vector v onto box [0,1] and simplex with sum = target_sum"""
    # Binary search for lambda to satisfy sum = target_sum
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0, 1)

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, alpha):
    """Compute gradient of smoothed max objective"""
    n = len(h)
    dx = 2.0 / n
    
    # Compute correlation O_k for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    O = corr * dx
    
    # Smoothed max approximation
    O_max = np.max(O)
    exp_term = np.exp(alpha * (O - O_max))
    w = exp_term / np.sum(exp_term)
    
    # Compute gradient
    grad = np.zeros_like(h)
    
    # Loop over all shifts
    for idx, shift in enumerate(range(-n+1, n)):
        w_k = w[idx]
        if w_k == 0:
            continue
        
        # Compute shift_right and shift_left
        if shift >= 0:
            # shift_right: ones for first shift entries, then 1 - h[shift:]
            shift_right = np.empty(n)
            if shift > 0:
                shift_right[:shift] = 1.0
            shift_right[shift:] = 1.0 - h[shift:]
            
            # shift_left: zeros for first shift entries, then h[:n-shift]
            shift_left = np.empty(n)
            if shift > 0:
                shift_left[:shift] = 0.0
            shift_left[shift:] = h[:n-shift]
        else:
            s = -shift
            # shift_right: ones for first s entries, then 1 - h[:n-s]
            shift_right = np.empty(n)
            if s > 0:
                shift_right[:s] = 1.0
            shift_right[s:] = 1.0 - h[:n-s]
            
            # shift_left: zeros for last s entries, then h[s:]
            shift_left = np.empty(n)
            if n - s > 0:
                shift_left[:n-s] = h[s:]
            shift_left[n-s:] = 0.0
            
        grad += w_k * (shift_right - shift_left)
    
    return grad * dx

def optimize_h(n, max_iters, alpha_start, alpha_end, lr0, init_h=None):
    """Gradient descent optimization with smooth approximation"""
    if init_h is None:
        # Start with random uniform values
        h = np.random.rand(n)
    else:
        h = init_h.copy()
    
    # Project to satisfy sum = n/2
    target_sum = n / 2.0
    h = project_to_box_simplex(h, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for it in range(max_iters):
        # Compute smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        
        # Compute gradient
        grad = compute_gradient(h, alpha)
        
        # Adaptive learning rate
        lr = lr0 / np.sqrt(it + 1)
        
        # Step update
        h = h - lr * grad
        
        # Project back onto box and simplex
        h = project_to_box_simplex(h, target_sum)
        
        # Evaluate and keep track of best solution
        cur_c5 = evaluate_c5(h)
        if cur_c5 < best_c5:
            best_h = h.copy()
            best_c5 = cur_c5
            
        # Early stopping if target reached
        if best_c5 < 0.379:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set problem parameters
    n_points = 64
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    # Initialize best solution
    best_h = None
    best_c5 = np.inf
    
    # Perform multiple restarts
    max_restarts = 200
    restart_count = 0
    
    while restart_count < max_restarts and (time.time() - start_time) < budget_s * 0.95:
        # Initialize with different random patterns
        init_pattern = restart_count % 4
        if init_pattern == 0:
            # Random uniform
            h0 = np.random.rand(n_points)
        elif init_pattern == 1:
            # Sine-like shape
            x = np.linspace(0, 2, n_points)
            h0 = 0.5 + 0.4 * np.sin(np.pi * x)
            h0 = np.clip(h0, 0, 1)
        elif init_pattern == 2:
            # Step function
            h0 = np.zeros(n_points)
            mid = n_points // 2
            width = n_points // 4
            start = max(0, mid - width // 2)
            end = min(n_points, mid + width // 2)
            h0[start:end] = 1.0
        else:  # init_pattern == 3
            # Gaussian-like shape
            x = np.linspace(0, 2, n_points)
            mu = 1.0
            sigma = 0.2
            h0 = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
            h0 = np.clip(h0, 0, 1)
        
        # Project to constraint sum = n/2
        h0 = project_to_box_simplex(h0, target_sum)
        
        # Optimize with gradient descent
        h_opt, c5_opt = optimize_h(
            n_points, 
            max_iters=3000,
            alpha_start=5,
            alpha_end=30,
            lr0=0.5,
            init_h=h0
        )
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            
        restart_count += 1
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
    # Final refinement of best solution
    if best_h is not None:
        final_h, final_c5 = optimize_h(
            n_points,
            max_iters=2000,
            alpha_start=30,
            alpha_end=50,
            lr0=0.1,
            init_h=best_h
        )
        
        if final_c5 < best_c5:
            best_h = final_h.copy()
            best_c5 = final_c5
    
    return (best_h.tolist(), float(best_c5), n_points)
