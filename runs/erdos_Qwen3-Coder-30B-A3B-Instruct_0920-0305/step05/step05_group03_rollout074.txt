import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import *
import math

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    dx = 2.0 / n_points
    target_c5 = 0.3807
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    print(f"Initial C5 = {best_c5:.6f}")
    
    # MILP step to find binary solution
    print("Solving MILP...")
    
    # Create a new variable `b` for binary values
    b = cp.Variable(n_points, boolean=True)
    
    # Construct the sum constraint
    area_constraint = cp.sum(b) == n_points / 2
    
    # Create list of shift indices to avoid recomputation
    shift_ranges = []
    for k in range(-(n_points - 1), n_points):
        # Compute valid pairs for this shift k
        pairs = []
        for i in range(max(0, -k), min(n_points, n_points - k)):
            j = i + k
            pairs.append((i, j))
        shift_ranges.append((k, pairs))
    
    # Create y variables for all pairs
    y_vars = {}
    shift_y_vars = {}
    
    for k, pairs in shift_ranges:
        if len(pairs) == 0:
            continue
        y_vars[k] = cp.Variable(len(pairs), boolean=True)
        shift_y_vars[k] = y_vars[k]
    
    # Build the constraints for y variables
    constraints = [area_constraint]
    for k, pairs in shift_ranges:
        if len(pairs) == 0:
            continue
        y = y_vars[k]
        for idx, (i, j) in enumerate(pairs):
            # y[i, k] <= b[i]
            constraints.append(y[idx] <= b[i])
            # y[i, k] <= b[j]
            constraints.append(y[idx] <= b[j])
            # y[i, k] >= b[i] + b[j] - 1
            constraints.append(y[idx] >= b[i] + b[j] - 1)
    
    # Overlap constraints
    tau = cp.Variable()
    
    for k, pairs in shift_ranges:
        if len(pairs) == 0:
            continue
        y = y_vars[k]
        # Overlap O_k = sum(b_i) - sum(y_{i,k})
        total_b_sum = cp.sum(b) 
        y_sum = cp.sum(y)
        overlap_expr = total_b_sum - y_sum
        constraints.append(overlap_expr <= tau)
    
    # Objective
    objective = cp.Minimize(tau)
    
    # Solve MILP
    prob = cp.Problem(objective, constraints)
    
    # Try to solve the MILP under time limit
    try:
        prob.solve(solver=CP.ECOS_BB, time_limit=budget_s * 0.2, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            sol = b.value
            if sol is not None:
                h_binary = sol.astype(float)
                h_binary_c5 = evaluate_c5(h_binary)
                print(f"Binary MILP solution C5 = {h_binary_c5:.6f}")
                if h_binary_c5 < best_c5:
                    best_h = h_binary.copy()
                    best_c5 = h_binary_c5
    except Exception as e:
        print(f"MILP solve error: {e}")
        pass
    
    # Local hill-climbing on binary patterns if MILP failed
    local_time = time.time()
    if best_c5 >= target_c5 or (time.time() - start_time) > budget_s * 0.8:
        print("MILP either failed or time limit reached; resorting to binary hill-climbing...")
        
        # Sample a starting 0/1 binary pattern
        # We'll take the current best h, and quantize to [0,1]
        # This is a heuristic, but good enough
        h_01 = np.zeros(n_points)
        for i in range(n_points):
            h_01[i] = 1 if best_h[i] > 0.5 else 0
        if np.sum(h_01) > n_points / 2:
            # Reduce ones to make exact sum
            pos_indices = np.where(h_01 == 1)[0]
            num_to_flip = int(np.sum(h_01) - n_points / 2)
            flip_indices = np.random.choice(pos_indices, size=num_to_flip, replace=False)
            h_01[flip_indices] = 0
        elif np.sum(h_01) < n_points / 2:
            # Increase zeros to make exact sum
            neg_indices = np.where(h_01 == 0)[0]
            num_to_flip = int(n_points / 2 - np.sum(h_01))
            flip_indices = np.random.choice(neg_indices, size=num_to_flip, replace=False)
            h_01[flip_indices] = 1

        best_h_binary = h_01
        current_c5 = evaluate_c5(h_01)
        if current_c5 < best_c5:
            best_h = h_01
            best_c5 = current_c5
        print(f"Initial binary pattern from random C5 = {current_c5:.6f}")
        
        # Now perform exhaustive swaps
        swaps_performed = 0
        max_swaps = 10000
        improvement_waited = 0  # How many iterations since last update
        
        while time.time() - start_time < budget_s * 0.7 and improvement_waited < 500 and swaps_performed < max_swaps:
            # Pick all pairs (i, j) such that h_01[i] = 1 and h_01[j] = 0
            ones = np.where(h_01 == 1)[0]
            zeros = np.where(h_01 == 0)[0]
            
            improved = False
            best_delta = np.inf
            best_i, best_j = None, None
            
            # Try all possible (i,j) pairs
            for i in ones:
                for j in zeros:
                    # Hypothetically do a swap and see if it helps
                    h_test = h_01.copy()
                    h_test[i] = 0
                    h_test[j] = 1
                    
                    test_c5 = evaluate_c5(h_test)
                    delta = test_c5 - current_c5
                    if delta < best_delta:
                        best_delta = delta
                        best_i, best_j = i, j
            
            # If there is an improving swap, do it
            if best_i is not None and best_delta < -1e-10:  # Only take "-ớt" dips
                h_01[best_i] = 0
                h_01[best_j] = 1
                current_c5 += best_delta
                if current_c5 < best_c5:
                    best_h = h_01.copy()
                    best_c5 = current_c5
                improvement_waited = 0
                improved = True
            else:
                improvement_waited += 1
            
            swaps_performed += 1
            
            if not improved:
                # Occasionally re-randomize some bits if not improving
            
        print(f"After local hill-climb, C5 = {best_c5:.6f}")
    
    # Continuous refinement from best binary pattern
    h_cont = best_h.copy()
    print("Starting continuous refinement...")
    
    def project_onto_simplex(v, target_sum):
        # Solve for lambda s.t. sum(clamp(v + lambda, 0, 1)) = target_sum
        lo, hi = -np.inf, np.inf
        for _ in range(100):  # binary search
            mid = (lo + hi) * 0.5
            proj = np.clip(v + mid, 0, 1)
            s = np.sum(proj)
            if s < target_sum:
                lo = mid
            else:
                hi = mid
        return np.clip(v + lo, 0, 1)
    
    max_iter = 5000
    eta = 0.5 * dx  # Initial step size
    
    i = 0
    while i < max_iter and time.time() - start_time < budget_s * 0.95:
        # Evaluate current overlap maximizer shifts
        corr = correlate(h_cont, 1 - h_cont, mode='full')
        max_corr = np.max(corr)
        active_k = np.where(corr == max_corr)[0]
        active_k = active_k - (n_points - 1)  # Convert positions to shift indices
        
        # Compute sub-gradient w.r.t. h_i
        w = np.zeros(n_points)
        all_shifts = range(-(n_points - 1), n_points)
        for k in active_k:
            # Weight k contributes 1 at time indices where shift k).  
            # But we must shift over the correlation computation
            # w_r += 1 if r - k is in valid range [0, n_points-1]? No, for correlation,
            # w_{r} = number of errors where shift k gives contribution at r,
            # which gives for each r, count of i such that i+k=r and 0 <= i < n_points.
            # So r = i + k => i = r - k, and i in [0, n_points-1]
            # That gives: i = r-k with 0 ≤ r-k < n_points
            # So r in [k, k+n_points)
            # But we want i in [0, n_points-1]
            # Sub-standard case; Diagonal transpose
            for r in range(n_points):  # iterate over result index
                i_r = r - k
                if 0 <= i_r < n_points:
                    w[r] += 1
            
        # But this is in the context of h_dist = (w * h) + (w_rev * h_rev)            
        # Incrementally build gradient with r as output index:
        # grad_r = s_r - Σ w_{k,r} h_{r-k} - Σ w_{k,r} h_{rev_i} where rev_i = n-1-i, and (r+k) < n , rev_{r+k} in [0, n-1]
        # In fact it's simpler, again compute via actual correlation operation
        g = np.zeros(n_points)
        s = np.ones(n_points) * n_points / 2
        # convolve w with 1s
        w_ones = np.convolve(w, np.ones(n_points), mode='full')[n_points-1:2*n_points-1]  # size n_points
        # now compute total convolution
        # overlap = sum(h) - sum(h * h_rev) - sum(h * h_shifted) 
        # Overlap (w * h) is the gradient wrt h
        # Project to 0-1 boxes. When v is gradient estimate, project by optimal lambda
        # Compute gradient over shifts s - h ⋆ w - h_rev ⋆ w_rev
        h_rev = h_cont[::-1]
        g1 = np.convolve(h_cont, w[::-1], mode='full')[n_points-1:2*n_points-1]  
        g2 = np.convolve(h_rev, w[::-1], mode='full')[n_points-1:2*n_points-1]  
        g = s - g1 - g2
        
        h_candidate = h_cont - eta * g
        h_candidate = project_onto_simplex(h_candidate, n_points / 2)
        
        # Symmetrize
        h_sym = (h_candidate + h_candidate[::-1]) / 2.0
        h_sym = project_onto_simplex(h_sym, n_points / 2)
        
        new_c5 = evaluate_c5(h_sym)
        
        if new_c5 < best_c5:
            best_h = h_sym.copy()
            best_c5 = new_c5
            print(f"Improved C5 = {best_c5:.6f}")
            
        h_cont = h_sym
        
        i += 1
        if i % 100 == 0:
            print(f"Refinement iteration {i}, C5 = {best_c5:.6f}")
            if best_c5 <= target_c5:
                break
        
        if i % 500 == 0:
            # reduce learning rate if stagnating
            eta = max(eta * 0.9, 1e-6 * dx)
    
    print(f"Final C5 = {best_c5:.6f}")
    
    # Optional upsampling 
    if time.time() - start_time < budget_s * 0.95:
        # Try double resolution
        n_double = 2 * n_points
        h_upsampled = np.interp(np.linspace(0, 2, n_double), np.linspace(0, 2, n_points), best_h)
        h_upsampled = h_upsampled * (n_points / n_double)
        
        print("Starting upsampled refinement...")
        
        h_refine = h_upsampled.copy()
        max_iter_refine = 500
        
        try:
            for _ in range(max_iter_refine):
                # Compute overlap and identify max shifts
                corr = correlate(h_refine, 1 - h_refine, mode='full')
                max_corr = np.max(corr)
                active_k = np.where(corr == max_corr)[0]
                active_k = active_k - (n_double - 1)
                
                w = np.zeros(n_double)
                for k in active_k:
                    for r in range(n_double):
                        i = r - k
                        if 0 <= i < n_double:
                            w[r] += 1
                
                # Gradient
                s = np.ones(n_double) * n_double / 2
                g1 = np.convolve(h_refine, w[::-1], mode='full')[n_double-1:2*n_double-1]
                h_rev = h_refine[::-1]
                g2 = np.convolve(h_rev, w[::-1], mode='full')[n_double-1:2*n_double-1]
                g = s - g1 - g2
                
                h_candidate = h_refine - 0.5 * dx * g  # smaller step
                h_candidate = project_onto_simplex(h_candidate, n_double / 2)
                
                h_sym = (h_candidate + h_candidate[::-1]) / 2.0
                h_sym = project_onto_simplex(h_sym, n_double / 2)
                
                new_c5 = evaluate_c5(h_sym)
                if new_c5 < best_c5:
                    best_h = h_sym.copy()
                    best_c5 = new_c5
                h_refine = h_sym
        except Exception as e:
            print(f"Error in upsampled optimization: {e}")
            pass
                
    return (best_h.tolist(), best_c5, n_points)
