import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def local_search_move(h, worst_lag_idx):
    """Try a small move to improve the worst correlation"""
    n = len(h)
    
    # Find two points to move mass between
    # Most promising choice: move from position with lower value
    # to position with higher value to reduce the peak
    gain = 0
    best_move = None
    
    # Test moving small amounts from positions with high contribution to worst lag
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
                
            # Compute potential impact
            # We are checking if moving a small amount from i to j helps the worst lag
            # Move from i to j: h[i] -= delta, h[j] += delta
            # For the worst lag: for each position k, we check correlation effect
            
            current_we = 0
            if i >= worst_lag_idx:
                current_we += h[i - worst_lag_idx]
            if i + worst_lag_idx < n:
                current_we += h[i + worst_lag_idx]
            new_we = 0
            if j >= worst_lag_idx:
                new_we += h[j - worst_lag_idx]
            if j + worst_lag_idx < n:
                new_we += h[j + worst_lag_idx]
            
            delta_move = 0.01
            if h[i] >= delta_move and 1.0 - h[j] >= delta_move:
                gain_potential = (new_we - current_we) * delta_move
                if gain_potential < gain:
                    gain = gain_potential
                    best_move = (i, j, delta_move)
    
    return best_move

def get_worst_lag_idx(h):
    """Find lag index that produces maximum correlation"""
    corr = compute_corr_fft(h)
    return np.argmax(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 120
    target_sum = n_points / 2.0
    
    # Initialize
    h = generate_initial_solution()
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    start_time = time.time()
    elapsed = 0
    
    # Main optimization loop
    iter_count = 0
    while elapsed < budget_s:
        # Use a hybrid strategy
        action_type = iter_count % 10
        
        if action_type < 7:  # 70%: targeted local search
            # Find the worst lag
            worst_lag_idx = get_worst_lag_idx(h)
            
            # Try small moves to improve worst lag
            move = local_search_move(h, worst_lag_idx)
            if move is not None:
                i, j, delta = move
                h_trial = h.copy()
                h_trial[i] -= delta
                h_trial[j] += delta
                h_trial = project_to_sum_box(h_trial, target_sum)
                new_c5 = evaluate_c5(h_trial)
                if new_c5 < current_c5:
                    h = h_trial
                    current_c5 = new_c5
                    if current_c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = current_c5
                        
        elif action_type == 7:  # 10%: random global perturbation
            # Add some random noise
            h_trial = h + np.random.normal(0, 0.005, len(h))
            h_trial = project_to_sum_box(h_trial, target_sum)
            new_c5 = evaluate_c5(h_trial)
            if new_c5 < current_c5:
                h = h_trial
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                    
        elif action_type == 8:  # 10%: local optimization with trust-constr
            def obj(h):
                return evaluate_c5(h)
            def constraint_eq(h):
                return np.sum(h) - target_sum
                
            cons = {'type': 'eq', 'fun': constraint_eq}
            bounds = [(0, 1) for _ in range(len(h))]
            
            try:
                res = minimize(
                    obj,
                    h,
                    method='trust-constr',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 100, 'gtol': 1e-10}
                )
                if res.success:
                    h_trial = res.x
                    h_trial = project_to_sum_box(h_trial, target_sum)
                    new_c5 = evaluate_c5(h_trial)
                    if new_c5 < current_c5:
                        h = h_trial
                        current_c5 = new_c5
                        if current_c5 < best_c5:
                            best_h = h.copy()
                            best_c5 = current_c5
            except:
                pass
        else:  # 10%: derandomized search + local refinement
            # Simply try a few small modifications to avoid local minima
            h_trial = h.copy()
            i = np.random.randint(0, len(h))
            j = np.random.randint(0, len(h))
            delta = np.random.uniform(0.001, 0.01)
            if h_trial[i] >= delta and 1.0 - h_trial[j] >= delta:
                h_trial[i] -= delta
                h_trial[j] += delta
                h_trial = project_to_sum_box(h_trial, target_sum)
                new_c5 = evaluate_c5(h_trial)
                if new_c5 < current_c5:
                    h = h_trial
                    current_c5 = new_c5
                    if current_c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = current_c5
                        
        iter_count += 1
        elapsed = time.time() - start_time
        
    # Final local refinement if time allows
    if elapsed < budget_s - 1:
        try:
            def obj(h):
                return evaluate_c5(h)
            def constraint_eq(h):
                return np.sum(h) - target_sum
                
            cons = {'type': 'eq', 'fun': constraint_eq}
            bounds = [(0, 1) for _ in range(len(best_h))]
            res = minimize(
                obj,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500}
            )
            if res.success:
                h_final = res.x
                h_final = project_to_sum_box(h_final, target_sum)
                c5_final = evaluate_c5(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
