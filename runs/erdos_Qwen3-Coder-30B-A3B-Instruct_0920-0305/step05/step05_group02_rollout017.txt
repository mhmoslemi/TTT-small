                best_min_p = min_p
                best_i = i
        if best_i >= 0:
            x[best_i] = 1
            selected.add(best_i)
    return np.array(x)

def solve_milp_with_symmetry_breaking(initial_h, n, start_time, budget_s):
    import cvxpy as cp
    from cvxpy import Minimize, sum as cp_sum, Variable, Problem, Binary

    # Initialize variables
    x = Variable(n, boolean=True)
    p = [Variable(boolean=True) for k in range(n)]
    y = [[Variable(boolean=True) for k in range(n)] for i in range(n)]

    constraints = []

    # Constraint 1: sum(x) = n/2
    constraints.append(cp_sum(x) == n // 2)

    # Symmetry breaking: x[0] = 1
    constraints.append(x[0] == 1)

    # Symmetry breaking: major half
    half = n // 2
    constraints.append(cp_sum(x[:half]) >= cp_sum(x[half:]))

    # Constraint 2: enforce p_{k} >= L for all k
    # We introduce a helper variable representing L and max constraints
    L = Variable(boolean=True)
    # Approximate Autocorrelations p_k = sum x_i * x_{i+k}
    for k in range(1, n):
        for i in range(n - k):
            # y[i,k] = x[i] AND x[i+k]
            # We use bilinear AND approximation with auxiliary variable constraints:
            # Upper bounds:
            constraints.append(y[i][k] <= x[i])
            constraints.append(y[i][k] <= x[i + k])
            constraints.append(y[i][k] >= x[i] + x[i + k] - 1)
        
        # p_k = sum_{i < n-k} y[i,k]  
        p[k] = cp_sum([y[i][k] for i in range(n - k)])
        constraints.append(p[k] >= L)

    # Objective: maximize L
    objective = Minimize(-L)  # In CVXPY maximize min(p_k) via Minimize (-L)
    
    # To optimize:
    # Set initial values for warm start
    prob = Problem(objective, constraints)
    
    # Attach warm start to variables
    for i in range(n):
        x[i].value = initial_h[i]

    # Solve
    try:
        # Limit solving to short time
        solve_start = time.time()
        while (time.time() - solve_start) < 40 and (time.time() - start_time) < budget_s:
            try:
                prob.solve(solver=cp.CBC, max_seconds=30)
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    # Extract solution
                    x_val = np.array([x[i].value for i in range(n)]).astype(int)
                    return x_val
                elif prob.status in ["infeasible", "unbounded"]:
                    break
            except (ValueError, AttributeError, RuntimeError):
                break
        # Timeout or no solution seen
        return None
    except:
