import numpy as np

from scipy.signal import correlate
import time
from scipy.fft import fft, ifft

def project_simplex(h, target_sum):
    """Project h onto the simplex: sum(h) = target_sum, 0 <= h_i <= 1."""
    # First, clip to [0,1]
    h = np.clip(h, 0, 1)
    # Now adjust using binary search for the lagrange multiplier
    # We need lambda such that sum(clamp(h_i + lambda, 0, 1)) = target_sum
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        new_h = np.clip(h + mid, 0, 1)
        s = np.sum(new_h)
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    # Final adjustment
    lambda_ = (lo + hi) / 2
    h_final = np.clip(h + lambda_, 0, 1)
    # Ensure precision
    factor = target_sum / np.sum(h_final)
    return h_final * factor

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def compute_correlation(h):
    """Compute auto-correlation vector c_k = sum h[i] h[i+k] for k=1..n-1"""
    n = len(h)
    # Use FFT for fast convolution
    h_rev = h[::-1]
    # Zero-padding to size 2*n-1 to simulate linear convolution
    h_fft = fft(np.concatenate([h, [0] * (2*n - 1 - n)]))
    h_rev_fft = fft(np.concatenate([h_rev, [0] * (2*n - 1 - n)]))
    # Convolution h * h
    conv_hh = ifft(h_fft * np.conj(h_fft)).real[:n]
    # Auto-correlation c_k : only shifted correlations k=1..n-1
    return conv_hh[1:]

    # For simplicity, we will use the definition of derivative of correlation
    # grad c_k = h_{k+1} + h_{k-1} for each k
    # So we need to compute: sum of w_k * (h_{k+1} + h_{k-1}) where 1<=k<=n-1
    # which = sum w_k h_{k+1} + sum w_k h_{k-1}
    # Reverse vector to represent backward shifts
    w_rev = w[::-1]
    # Use circular convolution to get derivatives (efficient via FFT)
    # forward_part = conv(h, w)  of length n
    # backward_part = conv(h, w_rev) of length n
    # But let's use actual convolution style compute
    # This part is trickier to handle because it shouldn't wrap around in the computed h
    forward_vec = np.zeros(n)
    backward_vec = np.zeros(n)
    # Clip w array to size n for computation (for linear correlation)
    try:
        w_padded = np.concatenate([w[:n], [0] * (2*n - 1 - n)])
        h_padded = np.concatenate([h, [0] * (2*n - 1 - n)])
        f_conv = ifft(fft(h_padded) * fft(w_padded)).real[:n]
    except (ValueError, np.linalg.LinAlgError):
        f_conv = np.zeros(n)
    try:
        w_rev_padded = np.concatenate([w_rev[:n], [0] * (2*n - 1 - n)])
        h_padded_rev = np.concatenate([h[::-1], [0] * (2*n - 1 - n)])
        b_conv = ifft(fft(h_padded_rev) * fft(w_rev_padded)).real[:n]
    except (ValueError, np.linalg.LinAlgError):
        b_conv = np.zeros(n)
    # Derivative is:
    #      -sum w_k [ h_{k-1} + h_{k+1} ] =  - (b_conv + f_conv)
    # Only include shifts from 1...n-1. So use index range
    grad = -(f_conv + b_conv)
    return grad

def obj_and_grad(h, beta):
    """Return (objective value, gradient) for soft-min formulation"""
    n = len(h)
    c = compute_correlation(h)
    c_max = np.max(c)
    c_stable = c - c_max
    exp_term = np.exp(-beta * c_stable)
    obj_val = 1.0 + (1.0 / beta) * np.log(np.sum(exp_term))
    # Compute gradient explicitly
    grad = compute_gradient(h, c, beta)
    return obj_val, grad

def line_search(h, grad, step_size, beta):
    """Simple backtracking line search for sufficient decrease"""
    alpha = 1.0
    factor = 0.5
    delta = 1e-4  # Armijo parameter
    initial_obj, _ = obj_and_grad(h, beta)
    while alpha > 1e-10:
        h_new = project_simplex(h - alpha * grad, n / 2)
        obj_new, _ = obj_and_grad(h_new, beta)
        if obj_new <= initial_obj - delta * alpha * np.dot(grad, grad):
            return h_new, alpha
        alpha *= factor
    return h, alpha
