import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.irfft(np.fft.rfft(h_rev, 2*n - 1) * np.fft.rfft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def exploit_worst_correlation(h, num_steps=5):
    """Try to improve solution by moving mass away from the worst correlation"""
    n = len(h)
    corr_full = compute_corr_fft(h)
    worst_lag = np.argmax(corr_full)
    
    # Create a more systematic move that distributes the mass
    min_indices = []
    max_indices = []
    for i in range(n):
        left_idx = i - worst_lag
        right_idx = i + worst_lag
        if left_idx >= 0:
            if h[left_idx] > 0.9:  # near maximum, potential to give out mass
                max_indices.append(i)
        if right_idx < n:
            if h[right_idx] > 0.9:
                max_indices.append(i)
    
    # Sample from these indices
    if len(max_indices) == 0:
        return h.copy(), evaluate_c5(h)
        
    # Try to move from indices that have value near 1
    max_idx = max_indices[np.random.randint(len(max_indices))]
    
    # Find where to move mass to: try to reduce oscillation
    # Move towards areas with low value
    min_h = np.argmin(h)
    delta = 0.01
    if h[max_idx] > 0.01 and h[min_h] < 0.99:
        h_new = h.copy()
        h_new[max_idx] -= delta
        h_new[min_h] += delta
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        return h_new, evaluate_c5(h_new)
    
    return h.copy(), evaluate_c5(h)

def random_pair_move(h):
    """Generate a pairwise mass transfer move, ensuring feasibility"""
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    delta_max = min(h[i], 1.0 - h[j])
    delta_min = max(-h[j], h[i] - 1.0)
    if delta_min >= delta_max:
        return i, j, 0.0
    delta = np.random.uniform(delta_min, delta_max)
    return i, j, delta

        t_elapsed = time.time() - t_start

        # Switch between different move strategies
        if k % 10 == 0:

            worst_lag = np.argmax(corr_full)
            i, j = None, None
            # Try moving away from high values
            high_idx = np.argmax(h)
            low_idx = np.argmin(h)
            if h[high_idx] > 0.1 and h[low_idx] < 0.9:
                i, j = high_idx, low_idx
                delta = min(h[i], 1.0 - h[j], 0.02)
                if delta > 1e-6:
                    h_new = h.copy()
                    h_new[i] -= delta
                    h_new[j] += delta
                    h_new = project_to_sum_box(h_new, TARGET_SUM)
                    new_c5 = evaluate_c5(h_new)
                    if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                        h = h_new
                        current_c5 = new_c5
                        if current_c5 < best_c5:
                            best_h = h.copy()
                            best_c5 = current_c5
                    n_iter += 1
        elif k % 5 == 0:
            # Exploit recent worst correlations
            h_new, new_c5 = exploit_worst_correlation(h)
            if new_c5 < current_c5:
                h = h_new
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
            n_iter += 1
        else:
            # Random move
            i, j, delta = random_pair_move(h)
            if delta != 0.0:
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                
                # Project to sum and bounds
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                
                # Evaluate
                new_c5 = evaluate_c5(h_new)
                if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                    h = h_new
                    current_c5 = new_c5
                    if current_c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = current_c5
            n_iter += 1
        k += 1
    
    return best_h, best_c5

def local_refine(h):
    """Refine solution using SLSQP optimizer"""
    def objective(h):
        return evaluate_c5(h)
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h))]
    
    try:
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
    except:
        pass
    return h, evaluate_c5(h)
