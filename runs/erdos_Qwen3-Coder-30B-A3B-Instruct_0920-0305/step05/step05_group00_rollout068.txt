import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_smoothed_max_gradient(h, beta):
    """Compute gradient of smoothed max correlation"""
    corr = compute_corr_fft(h)
    corr = corr[:len(h)]  # Take the first len(h) elements, drop the lag part
    
    # Numerical stability: subtract max
    max_corr = np.max(corr)
    if max_corr > -np.inf:
        exp_corr = np.exp(beta * (corr - max_corr))
    else:
        exp_corr = np.ones_like(corr)
    
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute weighted sum of gradients
    grad = np.zeros_like(h)
    for i in range(len(h)):
        # gradient of h[i] in C_{i+shift} terms
        # For shift 0: nothing
        # For shift = k: d/dh_i (h[i] * (1 - h[i+k])) = 1 - h[i+k] 
        # But we compute differently: for each correlation lag
            
        for lag_idx, weight in enumerate(weights):
            if lag_idx == 0:
                continue  # skip original corr index
            lag = lag_idx - (len(h) - 1)
            if lag > 0:  # shift right
                if i + lag < len(h):
                    grad[i] += weight * (-h[i+lag])
            else: # lag < 0: shift left (lag is negative)
                lag_abs = -lag
                if i - lag_abs >= 0:
                    grad[i] += weight * (-h[i-lag_abs])
            
    # Add 1 for all positions: the derivative of h[i] in C[i] = h[i]*(1 - h[i+shift])
    grad += weights[0] * (1 - h)
    
    # Add all shifted versions 
    for lag_idx, weight in enumerate(weights):
        if lag_idx == 0:
            continue
        lag = lag_idx - (len(h) - 1)
        if lag > 0:
            shifted = np.zeros_like(h)
            shifted[lag:] = h[:-lag]
            grad += weight * shifted
        elif lag < 0:
            lag_abs = -lag
            shifted = np.zeros_like(h)
            shifted[:-lag_abs] = h[lag_abs:]
            grad += weight * shifted
    
    # numerically compute (in many cases this shouldn't be needed, but safe)
    # Force small negative values to zero or small positive to preserve solution shape 
    # if needed for numerics
    grad = np.clip(grad, -10, 10)  # clip large values to avoid extreme updates
    
    return grad

def compute_smoothed_max_objective(h, beta):
    """Compute log-sum-exp of the correlations"""
    corr = compute_corr_fft(h)
    corr = corr[:len(h)]  # Take the first len(h) elements
    max_corr = np.max(corr)
    if max_corr > -np.inf:
        exp_corr = np.exp(beta * (corr - max_corr))
    else:
        exp_corr = np.ones_like(corr)
    log_sum_exp = np.log(np.sum(exp_corr)) + max_corr
    return log_sum_exp / beta

def project_gradient_descent_smooth(h, steps=100, alpha=0.2, gamma=0.0001, beta=1.0):
    """Perform gradient descent with projection"""
    for t in range(1, steps+1):
        # compute gradient of smooth approximation
        grad = compute_smoothed_max_gradient(h, beta)
        
        # adaptive step size
        alpha_t = alpha / (1 + gamma * t)
        
        # update
        h_new = h - alpha_t * grad
        
        # project
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # check convergence based on change in h or objective
        if t % 50 == 0:
            obj = compute_smoothed_max_objective(h, beta)
            obj_new = compute_smoothed_max_objective(h_new, beta)
            if abs(obj - obj_new) < 1e-8:
                break
                
        h = h_new
        
    return h
