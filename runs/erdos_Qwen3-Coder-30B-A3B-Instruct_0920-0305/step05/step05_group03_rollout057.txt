```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Initialize best solution
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Phase 1: Differential evolution for global search
    try:
        de_res = differential_evolution(
            objective, 
            bounds, 
            maxiter=1000, 
            popsize=15,
            seed=seed,
            tol=1e-10,
            recombination=0.7,
            mutation=(0.5, 1.0)
        )
        if de_res.success:
            h_candidate = de_res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Phase 2: Local refinement with trust-constr
    try:
        res = minimize(
            objective, 
            best_h, 
            method='trust-constr', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 2000, 'gtol': 1e-12, 'ftol': 1e-12}
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Phase 3: Multiple random restarts with different patterns
    restarts = 200
    for i in range(restarts):
        if (time.time() - start_time) > budget_s * 0.90:
            break
            
        # Use different initialization strategies
        pattern_type = random.choice(['random', 'tent', 'spike', 'double_peak'])
        
        if pattern_type == 'random':
            new_h = np.random.random(n_points)
        elif pattern_type == 'tent':
            x = np.linspace(0, 2, n_points)
            new_h = 1 - np.abs(x - 1)
        elif pattern_type == 'spike':
            x = np.linspace(0, 2, n_points)
            new_h = np.exp(-((x - 1)**2) / 0.05)
        elif pattern_type == 'double_peak':
            x = np.linspace(0, 2, n_points)
            new_h = np.exp(-((x - 0.5)**2) / 0.1) + np.exp(-((x - 1.5)**2) / 0.1)
        
        # Normalize
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-13, 'gtol': 1e-13}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue

    # Phase 4: Pattern-driven search - try to create specific shapes
    if (time.time() - start_time) < budget_s * 0.8:
        # Try a construction with step functions and spikes
        x = np.linspace(0, 2, n_points)
        # Focus on a structure that balances values and keeps the integral constraint
        # Try something with a peak in the center and decline
        h_pattern = np.zeros(n_points)
        h_pattern[10:32] = 1  # A rectangular plateau
        # Adjust to meet integral constraint
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(h_pattern) * dx)
        h_pattern = h_pattern * scale_factor
        
        try:
            res = minimize(
                objective, 
                h_pattern, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass

    # Ensure the final solution meets constraints
    dx = 2.0 / n_points
    final_scale = 1.0 / (np.sum(best_h) * dx)
    best_h = best_h * final_scale

    return (best_h.tolist(), best_c5, n_points)
```
