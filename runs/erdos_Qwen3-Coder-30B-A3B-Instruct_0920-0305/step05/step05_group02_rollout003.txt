import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_c5_binary(h):
    """Evaluate C5 for binary h - more accurate for integer representations"""
    n = len(h)
    dx = 2.0 / n
    # Full correlation
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def generate_legendre_pattern(n):
    """Generate Legendre-based binary pattern"""
    p = n // 2
    if p < 2:
        return None
    # Try to make p prime
    while p > 1:
        is_prime = True
        for i in range(2, int(math.sqrt(p))+1):
            if p % i == 0:
                is_prime = False
                break
        if is_prime:
            break
        p -= 1
    if p <= 1:
        return None
    
    # Generate quadratic residues modulo p
    residues = set()
    for a in range(1, p):
        r = (a * a) % p
        residues.add(r)
    
    # Create binary vector
    b = np.zeros(p, dtype=int)
    for r in residues:
        if r < p:
            b[r] = 1
    # Extend to length n
    h = np.tile(b, 2)
    if len(h) > n:
        h = h[:n]
    elif len(h) < n:
        h = np.pad(h, (0, n - len(h)), constant_values=0)
    # Adjust to exactly n/2 ones
    ones_count = np.sum(h)
    if ones_count < n // 2:
        # Find zeros and flip some
        zero_indices = np.where(h == 0)[0]
        for i in range((n // 2) - ones_count):
            if i < len(zero_indices):
                h[zero_indices[i]] = 1
    elif ones_count > n // 2:
        # Find ones and flip some
        one_indices = np.where(h == 1)[0]
        for i in range(ones_count - (n // 2)):
            if i < len(one_indices):
                h[one_indices[i]] = 0
    return h

def generate_random_balanced(n):
    """Generate a random balanced binary pattern"""
    h = np.zeros(n, dtype=int)
    indices = np.random.permutation(n)
    h[indices[:n//2]] = 1
    return h

def generate_correlation(h):
    """Compute full correlation array"""
    n = len(h)
    corr = np.zeros(2 * n - 1)
    for k in range(-n + 1, n):
        # Compute Σ h[i] * (1 - h[i+k])
        total = 0
        for i in range(max(0, -k), min(n, n - k)):
            total += h[i] * (1 - h[i + k])
        corr[k + n - 1] = total
    return corr

def get_worst_shifts(corr, tolerance=1e-10):
    """Get indices where correlation reaches maximum"""
    max_val = np.max(corr)
    max_indices = np.where(np.abs(corr - max_val) < tolerance)[0]
    return max_indices

def compute_update(h, i, j, corr):
    """Compute change in correlation for a swap of h[i] and h[j]"""
    n = len(h)
    # We assume positions i and j are valid and swap their values (0/1)
    new_corr = corr.copy()
    
    # For each lag m, we compute the change in contribution
    for m in range(-n + 1, n):
        # Position i contributes to correlation at lags: [-n+1, ..., n-1]
        if i < n - abs(m):
            delta = 0
            if m >= 0:
                old_val = h[i] * h[i + m]
                new_val = (1 - h[i]) * (1 - h[i + m])
                delta += (new_val - old_val)
            if m <= 0:
                old_val = h[i - m] * h[i]
                new_val = (1 - h[i - m]) * (1 - h[i])
                delta += (new_val - old_val)
            new_corr[m + n - 1] += delta

        if j < n - abs(m):
            delta = 0
            if m >= 0:
                old_val = h[j] * h[j + m]
                new_val = (1 - h[j]) * (1 - h[j + m])
                delta += (new_val - old_val)
            if m <= 0:
                old_val = h[j - m] * h[j]
                new_val = (1 - h[j - m]) * (1 - h[j])
                delta += (new_val - old_val)
            new_corr[m + n - 1] += delta

    return new_corr
