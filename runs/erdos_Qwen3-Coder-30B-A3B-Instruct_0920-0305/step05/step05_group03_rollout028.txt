import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time
from copy import deepcopy

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use a prime number of points to avoid symmetry issues
    n_points = 44
    dx = 2.0 / n_points
    half_n = n_points // 2
    
    # Start with the initial construction
    h0 = np.array(initial_h_values)
    if len(h0) != n_points:
        # Extend or truncate to match n_points
        if len(h0) > n_points:
            h0 = h0[:n_points]
        else:
            h0 = np.pad(h0, (0, n_points - len(h0)), mode='constant', constant_values=0)
    
    # Compute initial C5
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Active set of shifts - start with simple case
    active_shifts = [0]
    
    # Build the problem iteratively
    iteration = 0
    max_iterations = 500  # Safety limit
    
    while time.time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        # Build LP model
        h = Variable(n_points)
        tau = Variable()
        
        # Build y variables for active shifts
        y_vars = {}
        constraints = []
        
        # Bounds
        constraints += [h >= 0, h <= 1]
        
        # Sum constraint
        constraints += [sum(h) == half_n]
        
        # Symmetry - enforce h[i] = h[n-1-i]
        for i in range(n_points // 2):
            constraints += [h[i] == h[n_points - 1 - i]]
        
        # Add constraints for active shifts
        total_y_vars = 0
        for k in active_shifts:
            # Create y_{i,k} variables where i+k is valid
            if k > 0:
                # y_{i,k} = h_i * h_{i+k}
                start_idx = 0
                end_idx = n_points - k
            elif k < 0:
                start_idx = -k
                end_idx = n_points
            else:
                start_idx = 0
                end_idx = n_points
            
            y = Variable(end_idx - start_idx)
            y_vars[(k, 'all')] = y
            total_y_vars += len(y)
            
            # McCormick envelope
            for i in range(start_idx, end_idx):
                j = i + k
                if j >= 0 and j < n_points:
                    # y_{i,k} >= 0
                    constraints += [y[i - start_idx] >= 0]
                    # y_{i,k} <= h_i
                    constraints += [y[i - start_idx] <= h[i]]
                    # y_{i,k} <= h_j
                    constraints += [y[i - start_idx] <= h[j]]
                    # y_{i,k} >= h_i + h_j - 1
                    constraints += [y[i - start_idx] >= h[i] + h[j] - 1]
            
            # Constraint: tau >= sum(h) - sum(y_{i,k})
            constraints += [tau >= sum(h) - sum(y)]
            
        # Objective
        objective = Minimize(tau)
        
        # Formulate and solve the problem
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=GLOP, verbose=False)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_sol = h.value
                tau_sol = tau.value
                # Apply symmetry to make it symmetric
                h_sym = np.zeros_like(h_sol)
                for i in range(n_points // 2):
                    h_sym[i] = (h_sol[i] + h_sol[n_points - 1 - i]) / 2
                    h_sym[n_points - 1 - i] = h_sym[i]
                h_sol = h_sym
                
                # Normalize (as the solver introduces a small error)
                scale_factor = half_n / np.sum(h_sol)
                h_sol = h_sol * scale_factor
                
                # Evaluate actual C5
                c5_eval = evaluate_c5(h_sol)
                
                if c5_eval < best_c5:
                    best_c5 = c5_eval
                    best_h = h_sol.copy()
                    
                # Check if this is already close enough to optimality
                if c5_eval <= tau_sol + 1e-6:
                    break
                    
                # Now compute true overlaps to identify worst shift
                corr_vals = correlate(h_sol, 1 - h_sol, mode="full")
                max_idx = np.argmax(corr_vals)
                k_max = max_idx - len(corr_vals) // 2 + 1  # adjust indexing
                # Ensure it's within reasonable range
                if abs(k_max) <= n_points // 2:
                    if k_max not in active_shifts:
                        active_shifts.append(k_max)
        
        except Exception as e:
            pass  # Skip if solve fails
            
        iteration += 1
        
        # Occasionally run a local optimization to refine
                
    # Polish the best solution with gradient-based optimization
    def project_to_simplex(h, target_sum):
        # Water-filling implementation
        h_sorted = np.sort(h)[::-1]
        n = len(h)
        cumsum = np.cumsum(h_sorted)
        for i in range(n):
            if h_sorted[i] - (target_sum - cumsum[i]) / (i + 1) >= 0:
                t = (target_sum - cumsum[i]) / (i + 1)
                break
        else:
            t = (target_sum - cumsum[-1]) / n
        h_projected = np.maximum(h - t, 0)
        scale = target_sum / np.sum(h_projected)
        h_projected = h_projected * scale
        return h_projected

    h_current = best_h.copy()
    c5_current = evaluate_c5(h_current)
    
    # Initial step size
    step_size = 0.5 * dx
    step_dec = 0.5
    
    for iter_refine in range(2000):
        if time.time() - start_time > budget_s * 0.995:
            break
            
        # Compute current gradient of max overlap
        # Get the active shifts
        corr_vals = correlate(h_current, 1 - h_current, mode="full")
        max_val = np.max(corr_vals)
        active_k = np.where(np.abs(corr_vals - max_val) < 1e-8)[0]
        active_k = [k - len(corr_vals) // 2 + 1 for k in active_k]
        
        # Compute subgradient of max overlap at current point
        g = np.zeros(n_points)
        for k in active_k:
            # grad O_k = 1 - h_{i+k} - h_{i-k} (accounting for indices)
            if k == 0:
                g += 1.0  # Large coefficient to balance others
            else:
                if k > 0:
                    h_shift = np.concatenate([h_current[k:], np.zeros(k)])
                else:
                    h_shift = np.concatenate([np.zeros(-k), h_current[:k]])
                g += 1.0 - (h_current + h_shift)
                
        g /= len(active_k)
        
        h_new = h_current - step_size * g
        h_new = project_to_simplex(h_new, half_n)
        
        # Symmetrise
        h_new_sym = np.zeros_like(h_new)
        for i in range(n_points // 2):
            h_new_sym[i] = (h_new[i] + h_new[n_points - 1 - i]) / 2
            h_new_sym[n_points - 1 - i] = h_new_sym[i]
            
        h_new = h_new_sym
        
        c5_new = evaluate_c5(h_new)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            step_size *= 1.2
        else:
            step_size *= step_dec
            
        # Early stopping if improvement is small
        if (c5_current < 0.3807) or (abs(c5_current - best_c5) < 1e-6):
            break
    
    # Final check
    if evaluate_c5(h_current) < best_c5:
        best_c5 = evaluate_c5(h_current)
        best_h = h_current
    
    # Binary rounding stage (optional micro-optimization)
    if time.time() - start_time < budget_s * 0.99:
        # Sort values and take the top n/2 to set to 1
        sorted_indices = np.argsort(h_current)[::-1]
        h_binary = np.zeros_like(h_current)
        h_binary[sorted_indices[:half_n]] = 1.0
        
        # Local swaps for improvement
        corr_original = correlate(h_binary, 1 - h_binary, mode="full")
        for _ in range(500):
            improved = False
            for i in range(n_points):
                for j in range(i+1, n_points):
                    if h_binary[i] == 1 and h_binary[j] == 0:
                        # Try swap
                        h_binary[i], h_binary[j] = 0, 1
                        corr_new = correlate(h_binary, 1 - h_binary, mode="full")
                        if np.max(corr_new) < np.max(corr_original):
                            corr_original = corr_new
                            improved = True
                        else:
                            h_binary[i], h_binary[j] = 1, 0  # Swap back
            if not improved:
                break
                
        c5_binary = evaluate_c5(h_binary)
        if c5_binary < best_c5:
            best_c5 = c5_binary
            best_h = h_binary
    
    return (best_h.tolist(), best_c5, n_points)
