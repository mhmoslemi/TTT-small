import time
import numpy as np

def project_to_box_simplex(v, target_sum):
    """Project vector v onto box [0,1] and simplex with sum = target_sum"""
    # Binary search for lambda such that sum(clipped(v - lambda)) = target_sum
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0, 1)

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, alpha):
    """Compute gradient of smoothed max of correlation values"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation vector
    corr = np.correlate(h, 1 - h, mode="full")
    O = corr * dx  # Scale by dx
    O_max = np.max(O)
    
    # Compute stable softmax weights
    exp_term = np.exp(alpha * (O - O_max))
    w = exp_term / np.sum(exp_term)
    
    # Initialize gradient
    grad = np.zeros_like(h)
    
    # Loop over all shifts (corresponding to correlation indices)
    for idx in range(2 * n - 1):
        w_k = w[idx]
        if w_k == 0:
            continue
        shift = idx - (n - 1)
        
        # Compute shift_right and shift_left vectors
        if shift >= 0:
            shift_right = np.empty(n)
            shift_right[:shift] = 1.0
            shift_right[shift:] = 1.0 - h[shift:]
            
            shift_left = np.empty(n)
            shift_left[:shift] = 0.0
            shift_left[shift:] = h[:n - shift]
        else:
            s = -shift
            shift_right = np.empty(n)
            shift_right[:s] = 1.0
            shift_right[s:] = 1.0 - h[:n - s]
            
            shift_left = np.empty(n)
            shift_left[:n - s] = h[s:]
            shift_left[n - s:] = 0.0
        
        grad += w_k * (shift_right - shift_left)
    
    return grad * dx

def optimize_h(n, max_iters=3000, alpha_start=5, alpha_end=30, init_lr=0.5):
    """Optimize h using gradient descent with smoothed max"""
    # Initialize to random values
    h = np.random.rand(n)
    h = project_to_box_simplex(h, target_sum=n/2)
    
    best_c5 = np.inf
    best_h = None
    
    for it in range(max_iters):
        # Compute current C5
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute smoothing parameter (anneal from alpha_start to alpha_end)
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters) if max_iters > 0 else alpha_start
        
        # Compute gradient
        grad = compute_gradient(h, alpha)
        
        # Compute learning rate
        lr = init_lr / np.sqrt(it + 1)
        
        # Update h
        h = h - lr * grad
        
        # Project onto box [0,1] and sum constraint
        h = project_to_box_simplex(h, target_sum=n/2)
        
        # Early stopping
        if best_c5 < 0.379:
            break

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the recommended dimension from previous code
    n_points = 64  # Good resolution and manageable
    target_sum = n_points / 2
    
    best_h = None
    best_c5 = np.inf
    
    # Time budget check
    while (time.time() - start_time) < budget_s * 0.95:
        # Perform optimization
        h, c5 = optimize_h(n_points, max_iters=3000)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Early exit if good enough
        if best_c5 <= 0.379:
            break
    
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_c5(best_h)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        # Fallback to initial_h_values if no improvement
        # Note: We cannot directly use initial_h_values since it's potentially unlisted
        # But we keep existing fallback for demonstration
        h0 = np.random.rand(n_points)
        h0 = project_to_box_simplex(h0, target_sum=target_sum)
        final_c5 = evaluate_c5(h0)
        return (h0.tolist(), float(final_c5), n_points)
