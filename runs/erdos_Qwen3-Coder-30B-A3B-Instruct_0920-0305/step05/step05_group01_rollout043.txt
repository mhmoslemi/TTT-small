import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

    best_c5 = current_c5
    
    start_time = time.time()
    
    # Multi-stage cooling schedule
    stage_duration = run_seconds / 3.0
    t_start = time.time()
    
    stage = 0
    while time.time() - start_time < run_seconds:
        if time.time() - t_start > stage_duration:
            stage += 1
            t_start = time.time()
            
        # Temperature cooling schedule
        elapsed = time.time() - start_time
        T = 0.1 * (1 - elapsed/run_seconds) + 1e-6
            
        # Select move type based on stage

        if stage == 0:
            # Large random moves
            i, j = np.random.choice(len(h), 2, replace=False)
            delta = np.random.uniform(-0.2, 0.2)
        elif stage == 1:
            # Medium random moves  
            i, j = np.random.choice(len(h), 2, replace=False)
            delta = np.random.uniform(-0.1, 0.1)
        else:  # stage == 2
            # Small random moves for polishing
            i, j = np.random.choice(len(h), 2, replace=False)
            delta = np.random.uniform(-0.05, 0.05)
        
        # Apply move
        h_new = h.copy()
        h_new[i] = max(0, min(1, h_new[i] - delta))
        h_new[j] = max(0, min(1, h_new[j] + delta))
        
        # Project to correct sum
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        new_c5 = evaluate_c5(h_new)
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            h = h_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                
    return best_h, best_c5

        # Create perturbed version
        h_perturbed = h + np.random.normal(0, 0.02, len(h))

        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    return best_h, best_c5

def local_refine(h):
    """Quick local refinement using SLSQP"""
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h))]
    
    try:
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
    except:
        pass
    return h, evaluate_c5(h)
