```python
import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_simplex(v, target_sum):
    """Projects vector v onto the capped simplex: 0 <= h_i <= 1, sum h_i = target_sum"""
    v = np.asarray(v)
    low = 0.0
    high = 1.0
    while high - low > 1e-12:
        mid = (low + high) / 2.0
        projected = np.clip(v + mid, 0, 1)
        total = np.sum(projected)
        if total > target_sum:
            high = mid
        else:
            low = mid
    return np.clip(v + low, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Initialize best solution
    h_best = np.array(initial_h_values)
    c5_best = evaluate_c5(h_best)
    best_found_at = start_time

    ### Phase 1: Binary MILP (1/4 of budget)
    milp_time = 0.25 * budget_s
    print(f"Starting MILP with {n_points} points...")
    try:
        # Create variables
        h = cp.Variable(n_points, boolean=True)
        tau = cp.Variable()

        # Define constraints
        constraints = [cp.sum(h) == n_points // 2, h >= 0, h <= 1]

        # Define overlap variables and constraints
        y_vars = {}
        O_vars = []

        for k in range(-n_points + 1, n_points):
            y_vars[k] = cp.Variable(n_points, boolean=True)
            # Add valid pairs
            valid_pairs = []
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    valid_pairs.append((i, j))

            # Add constraints for valid pairs
            for i, j in valid_pairs:
                constraints.append(y_vars[k][i] <= h[i])
                constraints.append(y_vars[k][i] <= h[j])
                constraints.append(y_vars[k][i] >= h[i] + h[j] - 1)

            # Compute overlap
            O_k = cp.sum(h) - cp.sum([y_vars[k][i] for i, _ in valid_pairs])
            constraints.append(O_k <= tau)
            O_vars.append(O_k)

        # Objective
        problem = cp.Problem(cp.Minimize(tau), constraints)
        problem.solve(solver=CP.SCS, time_limit=milp_time, verbose=False)

        h_bin = h.value
        if h_bin is not None:
            h_bin = np.round(h_bin).astype(np.float64)
            c5_bin = evaluate_c5(h_bin)
            if c5_bin < c5_best:
                c5_best = c5_bin
                h_best = h_bin.copy()
                best_found_at = time.time()
    except Exception as e:
        print(f"MILP failed: {e}")

    ### Phase 2: Hill climb on binary patterns
    print("Starting hill-climbing phase...")
    l = 0
    while (time.time() - start_time) < 0.5 * budget_s and l < 500:
        improved = False
        for i in range(n_points):
            if h_best[i] == 1.0:
                for j in range(n_points):
                    if h_best[j] == 0.0:
                        # Try swap
                        h_test = h_best.copy()
                        h_test[i] = 0.0
                        h_test[j] = 1.0
                        c5_test = evaluate_c5(h_test)
                        if c5_test < c5_best:
                            h_best = h_test
                            c5_best = c5_test
                            improved = True
                            best_found_at = time.time()
        if not improved:
            break
        l += 1

    ### Phase 3: Continuous refinement (shallow gradient steps)
    print("Starting continuous refinement...")
    
    if (time.time() - start_time) < 0.9 * budget_s:
        h_refined = h_best.copy()
        eta = 0.5 * dx
        eta_min = 1e-5 * dx
        patience = 200
        last_improved = 0
        max_iters = 10000
        
        for it in range(max_iters):
            if (time.time() - start_time) > 0.95 * budget_s:
                break
            # compute overlaps
            corr = correlate(h_refined, 1 - h_refined, mode="full")
            max_k = np.argmax(corr)
            max_overlap = corr[max_k]
            # active shifts
            active_shifts = []
            for i in range(len(corr)):
                if abs(corr[i] - max_overlap) < 1e-10:
                    active_shifts.append(i - n_points + 1)

            # weight vector (sum over active shifts)
            w = np.zeros(n_points)
            for k in active_shifts:
                w = np.add(w, np.roll(np.ones(n_points), -k))
            w = w / np.sum(w)  # normalized to unit L1 norm
            
            # gradient estimate
            g = np.zeros_like(h_refined)
            s = np.convolve(w, np.ones(n_points), mode="full")[:n_points]
            g = dx * (s - np.convolve(w, h_refined, mode="full")[:n_points] - 
                      np.convolve(w, h_refined[::-1], mode="full")[n_points-1::-1])[::-1]
            
            # take step
            h_next = h_refined - eta * g
            
            # project
            h_next = project_simplex(h_next, n_points / 2)
            
            # symmetrize
            h_sym = (h_next + h_next[::-1]) / 2.0
            h_sym = project_simplex(h_sym, n_points / 2)
            
            c5_test = evaluate_c5(h_sym)
            if c5_test < c5_best:
                h_best = h_sym.copy()
                c5_best = c5_test
                last_improved = it
            else:
                patience -= 1
                if patience <= 0:
                    eta *= 0.5
                    patience = 200
                    if eta < eta_min:
                        break

    ### Phase 4: Optional up-sampling
    if (time.time() - start_time) < 0.98 * budget_s:
        # upsample
        n_new = 2 * n_points
        dx_new = 2.0 / n_new
        h_up = np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, n_points), h_best)
        h_up = project_simplex(h_up, n_new / 2)
        
        # short continuous refinement on the fine grid
        eta = 0.5 * dx_new
        eta_min = 1e-5 * dx_new
        patience = 100
        last_improved = 0
        for it in range(500):
            if (time.time() - start_time) > 0.98 * budget_s:
                break
            corr = correlate(h_up, 1 - h_up, mode="full")
            max_k = np.argmax(corr)
            max_overlap = corr[max_k]
            
            active_shifts = []
            for i in range(len(corr)):
                if abs(corr[i] - max_overlap) < 1e-10:
                    active_shifts.append(i - n_new + 1)
            
            w = np.zeros(n_new)
            for k in active_shifts:
                w = np.add(w, np.roll(np.ones(n_new), -k))
            w = w / np.sum(w)

            g = np.zeros_like(h_up)
            s = np.convolve(w, np.ones(n_new), mode="full")[:n_new]
            g = dx_new * (s - np.convolve(w, h_up, mode="full")[:n_new] - 
                          np.convolve(w, h_up[::-1], mode="full")[n_new-1::-1])[::-1]
            
            h_up = h_up - eta * g
            h_up = project_simplex(h_up, n_new / 2)
            
            h_up = (h_up + h_up[::-1]) / 2.0
            h_up = project_simplex(h_up, n_new / 2)
            
            c5_test = evaluate_c5(h_up)
            if c5_test < c5_best:
                h_best = h_up.copy()
                c5_best = c5_test
                last_improved = it

    return (h_best.tolist(), c5_best, n_points)
```
