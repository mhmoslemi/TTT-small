import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Try a new hybrid approach: alternating global and local search
    max_global_iter = 200
    max_local_iter = 100
    
    global_iter = 0
    local_iter = 0
    
    # Let's start with a more aggressive differential evolution
    try:
        res = differential_evolution(
            objective,
            bounds,
            maxiter=100,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            strategy='best1bin'
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass
    
    # Alternate between global and local search
    while (time.time() - start_time) < budget_s * 0.95:
        # Global search
        if global_iter < max_global_iter:
            try:
                res = differential_evolution(
                    objective,
                    bounds,
                    maxiter=50,
                    popsize=20,
                    mutation=(0.7, 1),
                    recombination=0.8,
                    seed=seed + global_iter,
                    strategy='best1bin'
                )
                if res.success:
                    h_candidate = res.x
                    dx = 2.0 / n_points
                    scale = 1.0 / (np.sum(h_candidate) * dx)
                    h_candidate = h_candidate * scale
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
            except:
                pass
            global_iter += 1
        
        # Local search
        if local_iter < max_local_iter:
            try:
                # Perturb the current best
                new_h = best_h + np.random.normal(0, 0.02, n_points)
                new_h = np.clip(new_h, 0, 1)
                dx = 2.0 / n_points
                scale_factor = 1.0 / (np.sum(new_h) * dx)
                new_h = new_h * scale_factor
                
                res = minimize(
                    objective, 
                    new_h, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=cons,
                    options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-15}
                )
                if res.success:
                    h_candidate = res.x
                    scale = 1.0 / (np.sum(h_candidate) * dx)
                    h_candidate = h_candidate * scale
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
            except:
                pass
            local_iter += 1
            
        # Try a structured modification if still time
        if (time.time() - start_time) < budget_s * 0.8:
            # Modify current best with a structured pattern
            try:
                x = np.linspace(0, 2, n_points)
                # Create a pattern with spikes at the boundaries
                # and a dip in the center  
                spike_pattern = np.zeros(n_points)
                # scatter some mass at the edges
                spike_pattern[0] = 0.5
                spike_pattern[-1] = 0.5
                # Average the rest
                spike_pattern[1:-1] = 0.0
                # Add some mass in the center as well  
                center_idx = n_points // 2
                spike_pattern[center_idx] = 0.1
                
                # Add some structure to the best known 
                mixed_h = best_h * 0.7 + spike_pattern * 0.3
                
                # Normalize
                dx = 2.0 / n_points
                scale_factor = 1.0 / (np.sum(mixed_h) * dx)
                mixed_h = mixed_h * scale_factor
                
                res = minimize(
                    objective, 
                    mixed_h, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=cons,
                    options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-15}
                )
                if res.success:
                    h_candidate = res.x
                    scale = 1.0 / (np.sum(h_candidate) * dx)
                    h_candidate = h_candidate * scale
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
            except:
                pass
                
        if global_iter >= max_global_iter and local_iter >= max_local_iter:
            break
            
    # Final dump of best values found in a structured way
    # Try to optimize some small region around the best found
    if (time.time() - start_time) < budget_s * 0.9:
        try:
            # Try a local refinement focusing on a critical parameter region
            n_refine = min(50, n_points)  # Smaller region to focus
            
            # Select random subset to optimize
            indices = sorted(random.sample(range(n_points), min(30, n_points)))
            
            # Create a smaller problem with indices
            bounds_small = [bounds[i] for i in indices]
            h_small = best_h[indices]
            
            def objective_small(h_small):
                # Build full h from reduced version
                h_full = best_h.copy()
                for i, idx in enumerate(indices):
                    h_full[idx] = h_small[i]
                return evaluate_c5(h_full)
                
            res = minimize(
                objective_small, 
                h_small, 
                method='SLSQP',
                bounds=bounds_small,
                options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                # Update the full h vector
                h_candidate = best_h.copy()
                for i, idx in enumerate(indices):
                    h_candidate[idx] = res.x[i]
                # Renormalize
                dx = 2.0 / n_points
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
            
    return (best_h.tolist(), best_c5, n_points)
