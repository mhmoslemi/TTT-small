def greedy_warmstart(n):
    # Simple greedy to find a feasible binary solution
    x = np.zeros(n, dtype=int)
    S = set()
    while len(S) < n // 2:
        best_i = -1
        best_min_p = -1
        for i in range(n):
            if i in S:
                continue
            p = [0] * (n - 1)
            for j in S:
                k = abs(i - j)
                if k > 0:
                    p[k-1] += 1
            current_min_p = min(p) if p else 0
            if current_min_p > best_min_p:
                best_min_p = current_min_p
                best_i = i
        if best_i != -1:
            S.add(best_i)
    for i in S:
        x[i] = 1
    return x

def evaluate_c5_discrete(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx
