def project_to_box_simplex(v, target_sum):
    """Project vector v onto the intersection of box [0,1]^n and simplex {x: sum(x)=target_sum}"""
    # Binary search for the Lagrange multiplier lambda
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0, 1)

    # The maximum overlap value scaled by dx
    max_corr = np.max(corr)
    return max_corr * dx

            continue

        if shift >= 0:

            shift_left = np.zeros(n)
            shift_left[:n - s] = h[s:]
        grad += w_k * (shift_right - shift_left)
    
    return grad * dx
