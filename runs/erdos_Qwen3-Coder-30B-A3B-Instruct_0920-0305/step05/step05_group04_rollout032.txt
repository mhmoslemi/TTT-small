import numpy as np

def project_to_simplex(x, sum_value):
    """Project vector x onto the simplex with sum = sum_value and bounds [0,1]"""
    # Clip to [0,1] first
    x = np.clip(x, 0, 1)
    # Ensure we satisfy the sum constraint
    current_sum = np.sum(x)
    if abs(current_sum - sum_value) < 1e-12:
        return x
    
    # Use Duchi et al. 2008 algorithm for projection onto bounded simplex
    # Find lambda such that sum(max(0, x_i - lambda)) = sum_value
    # Let y = sorted(x) in descending order
    y = np.sort(x)[::-1]
    
    # Find the index where the condition holds
    tmp_sum = 0.0
    rho = 0
    for i in range(len(y)):
        tmp_sum += y[i]
        if tmp_sum - (i+1) * y[i] > sum_value:
            rho = i
            break
    else:
        rho = len(y)
    
    # Compute lambda
    if rho == 0:
        lambda_val = (tmp_sum - sum_value) / len(y)
    else:
        lambda_val = (tmp_sum - sum_value) / rho
    
    # Compute projection
    proj = np.maximum(x - lambda_val, 0)
    
    # Adjust to ensure perfect sum (small numerical errors)
    diff = sum_value - np.sum(proj)
    if abs(diff) > 1e-12:
        # Add small amounts to the smallest entries
        idxs = np.argsort(proj)
        add_amount = diff / len(idxs)
        for i in idxs:
            if abs(diff) < 1e-12:
                break
            add = min(-proj[i], diff)
            proj[i] += add
            diff -= add
    
    return proj

def compute_correlations(h):
    """Compute auto-correlations (excluding shift 0) using FFT."""
    n = len(h)
    # Zero-pad for linear convolution
    h_pad = np.pad(h, (0, n-1), mode='constant', constant_values=0)
    # Compute full convolution (including negative shifts)
    conv_full = correlate(h_pad, h_pad, mode='full')
    # Extract correlations up to n-1 (positive shifts only)
    correlations = conv_full[n-1:n-1+n-1]
    return correlations

def compute_gradient(h, correlations, beta):
    """
    Compute the gradient of the smooth surrogate rho_beta = (1/beta) log(sum exp(-beta c_k))
    """

    # Compute soft min weights
    c = correlations
    c_max = np.max(c)
    # Avoid overflow
    exp_term = np.exp(-beta * (c - c_max))
    w = exp_term / np.sum(exp_term)
    
    # Compute gradient with two convolutions
    # Convolution with w on the right (forward shifts)
    # Convolution with reverse(w) on the right (backward shifts)
    w_rev = w[::-1]
    
    # For efficiency, we avoid full numpy FFT and compute convolution manually
    # but still use a way that is fast and general
    g = np.zeros(n)
    for i in range(n):
        for k in range(1, n):  # skip k=0
            if i + k < n:
                g[i] -= w[k-1] * h[i + k]
            if i - k >= 0:
                g[i] -= w[k-1] * h[i - k]
    return g

def compute_smooth_objective(h, correlations, beta):
    """Compute 1 + (1/beta) * log(sum exp(-beta * c_k))"""
    c = correlations
    c_max = np.max(c)
    exp_term = np.exp(-beta * (c - c_max))
    log_sum = c_max + np.log(np.sum(exp_term))
    return 1.0 + (1.0/beta) * log_sum

def line_search(h, g, beta, c_prev):
    """Perform backtracking line search"""
    alpha = 1.0
    step_size = 0.5
    max_alpha = 1.0
    
    # Evaluate at current point
    h_new = h - alpha * g
    h_new = project_to_simplex(h_new, len(h)/2)
    corr_new = compute_correlations(h_new)
    c_new = compute_smooth_objective(h_new, corr_new, beta)
    
    # Armijo condition: c_new <= c_prev - alpha * c_grad * g
    # But since this is gradient descent, we just need c_new < c_prev
    desired_decrease = 1e-8
    while c_new >= c_prev and max_alpha > 1e-10:
        alpha *= step_size
        max_alpha *= step_size
        h_new = h - alpha * g
        h_new = project_to_simplex(h_new, len(h)/2)
        corr_new = compute_correlations(h_new)
        c_new = compute_smooth_objective(h_new, corr_new, beta)
