import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def project(v, S):
    """Project vector v onto the capped simplex: 0 ≤ h_i ≤ 1, sum h_i = S"""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2
        p = np.clip(v + lam, 0.0, 1.0)
        s = p.sum()
        if s > S:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2, 0.0, 1.0)

def compute_O_w_grad(h, alpha):
    """Compute smooth objective value, weights, and gradient for given h and alpha"""
    n = len(h)
    dx = 2.0 / n
    c1 = correlate(h, np.ones(n), mode='full')
    c2 = correlate(h, h, mode='full')
    O = (c1 - c2) * dx  # overlaps for all shifts
    O_max = np.max(O)
    w = np.exp(alpha * (O - O_max))
    w = w / np.sum(w)
    
    s = np.convolve(w, np.ones(n), mode='valid')
    g2 = correlate(w, h, mode='full')[n-1:2*n-1]
    g3 = correlate(w, h[::-1], mode='full')[n-1:2*n-1][::-1]
    g = dx * (s - g2 - g3)
    
    return O, w, g

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - np.array(h), mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

                best_c5 = c5_de

    h = best_h.copy()

    eta = 0.5 * (2.0 / n)
    iteration = 0

        # Line search
        accepted = False
