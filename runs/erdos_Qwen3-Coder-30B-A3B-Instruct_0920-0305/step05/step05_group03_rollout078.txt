import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_simplex(v, target_sum):
    """Project vector v onto the simplex with given sum using binary search"""
    v = np.asarray(v)
    n = len(v)
    # Sort in descending order
    sorted_v = np.sort(v)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_v)
    # Find the threshold
    rho = 0
    for i in range(n):
        if sorted_v[i] > (cumsum[i] - target_sum) / (i + 1):
            rho = i + 1
    # Compute the Lagrange multiplier
    theta = (cumsum[rho - 1] - target_sum) / rho if rho > 0 else 0
    # Project
    result = np.maximum(v - theta, 0)
    return result

def projected_subgradient_descent(h_start, max_iter=2000, tol=1e-7, step_size=0.5):
    """Run projected subgradient descent to minimize C5"""
    h = h_start.copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    eta = step_size * dx
    no_improve_count = 0
    for i in range(max_iter):
        if no_improve_count > 10:
            eta *= 0.5
            no_improve_count = 0
        # Compute all O_k
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        # Compute active shifts
        active = np.where(np.abs(corr - max_val) < 1e-10)[0]
        # Compute subgradient
        subgrad = np.zeros(n)
        for k in active:
            # k corresponds to shift; handle negative shifts
            shift = k - (n - 1)
            if shift > 0:
                # h_i - h_{i+shift}
                subgrad += 1 - h[shift:] - h[:-shift]
            elif shift < 0:
                # h_i - h_{i-shift}
                subgrad += 1 - h[:shift] - h[-shift:]
            else:
                # shift = 0: all terms are 1
                subgrad += np.ones(n)
        # Average over active
        subgrad = subgrad / len(active)
        # Descent step
        h_new = h - eta * subgrad
        # Project onto [0,1], with sum = n/2
        h_new = project_simplex(h_new, n / 2)
        # Apply symmetry
        h_new = (h_new + h_new[::-1]) / 2
        h_new = project_simplex(h_new, n / 2)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        h = h_new.copy()
        if eta < tol:
            break
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n = len(h0)
    dx = 2.0 / n
    
    # Initial active set
    active_K = set([0])
    max_iter = 1000  # give time for LP solving
    
    best_feasible = h0.copy()
    best_c5_feasible = evaluate_c5(h0)
    
    # keep loop time in mind
    for iter_num in range(max_iter):
        if (time.time() - start_time) > budget_s * 0.98:
            break
        # Build LP model
        h = Variable(n)
        tau = Variable()
        # Symmetry constraints
        for i in range(n//2):
            h[i] == h[n-1-i]
        y_list = []
        constraints = [
            0 <= h,
            h <= 1,
            sum(h) == n/2,
        ]
        # Initial terms to handle rotations / difference
        for k in active_K:
            y = Variable(n)
            y_list.append(y)
            for i in range(n):
                # Bounds on y[i]
                if i + k < n and i + k >= 0:
                    constraints.extend([
                        y[i] >= 0,
                        y[i] <= h[i],
                        y[i] <= h[i+k],
                        y[i] >= h[i] + h[i+k] - 1,
                    ])
                else:
                    constraints.extend([
                        y[i] >= 0,
                        y[i] <= 0,
                    ])
            # Tau constraint
            constraints.append(tau >= sum(h) - sum(y))
        
        # Optimize
        problem = cp.Problem(cp.Minimize(tau), constraints)
        try:
            # Using reasonably fast solver
            problem.solve(solver=cp.GLOP)
            if problem.status in [cp.OPTIMAL]:
                h_vals = np.array(h.value)
                h_vals = np.clip(h_vals, 0, 1)
                # enforce the sum
                scale = (n/2) / np.sum(h_vals)
                h_vals = h_vals * scale
                # symmetrize
                h_vals = (h_vals + h_vals[::-1]) / 2
                h_vals = np.clip(h_vals, 0, 1)
                scale = (n/2) / np.sum(h_vals)
                h_vals = h_vals * scale
                # check if actually feasible
                c5_candidate = evaluate_c5(h_vals)
                if c5_candidate <= best_c5_feasible + 1e-10:
                    best_c5_feasible = c5_candidate
                    best_feasible = h_vals.copy()
            else:
                pass
        except:
            pass

        # Compute correlations
        h_vals = best_feasible
        corr = correlate(h_vals, 1 - h_vals, mode="full")
        k_max = np.argmax(corr)
        max_val = corr[k_max]
        # Add k_max to active set if not already there
        if k_max - (n - 1) not in active_K:
            if len(active_K) >= 30:  # bounded expansion
                break
            active_K.add(k_max - (n - 1))
        else:
            # maybe the curvature or extremum is less in direction of redundancy
            # but we are tightening already, so just break
            break

    # Now polish the best feasible solution found
    best_h, best_c5 = projected_subgradient_descent(best_feasible)
    
    # Optional binary rounding
    h_binary = best_h.copy()
    idx_sorted = np.argsort(h_binary)[::-1]
    h_binary.fill(0)
    h_binary[idx_sorted[:n//2]] = 1
    c5_binary = evaluate_c5(h_binary)
    if c5_binary < best_c5:
        best_c5 = c5_binary
        best_h = h_binary
    
    return (best_h.tolist(), best_c5, n)
