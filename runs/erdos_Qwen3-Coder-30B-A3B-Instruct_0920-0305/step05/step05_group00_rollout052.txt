def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def random_walk_move(h):
    """Generate a random walk move"""
    i = np.random.randint(0, n_points)
    delta = np.random.uniform(-0.1, 0.1)
    # Adjust delta to respect bounds
    new_val = h[i] + delta
    if new_val < 0:
        delta = -h[i]
    elif new_val > 1:
        delta = 1 - h[i]
    if abs(delta) < 1e-10:
        return None
    return (i, delta)

def guided_move(h, prev_corr):
    """Guided move based on the correlation"""
    # Find indices that contribute most to the maximum correlation
    best_lag = np.argmax(prev_corr)
    # Calculate gradient-like update
    g = np.zeros(n_points)
    for i in range(n_points):
        if i - best_lag >= 0:
            g[i] += h[i - best_lag]
        if i + best_lag < n_points:
            g[i] += h[i + best_lag]
    
    # Find points to move
    idx_min = np.argmin(g)
    idx_max = np.argmax(g)
    
    if idx_min == idx_max:
        return None
    
    # Move small amount
    delta = min(0.05, h[idx_min], 1 - h[idx_max])
    if delta <= 0:
        return None
    
    return (idx_min, delta)

        corr = compute_corr_fft(h)

        move = guided_move(h, corr)
        if move is not None:
            i, delta = move
            h[i] += delta
            h = project_to_sum_box(h, TARGET_SUM)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            else:
                # revert
                h[i] -= delta
                h = project_to_sum_box(h, TARGET_SUM)

                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                else:
                    # revert
                    h[i] -= delta
                    h = project_to_sum_box(h, TARGET_SUM)
    
    return best_h, best_c5
