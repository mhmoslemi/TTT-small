import numpy as np
import time

def project_to_box_simplex(v, target_sum):
    """Project vector v onto box [0,1] and simplex with sum = target_sum"""
    # Binary search for lambda
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0, 1)

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, alpha):
    """Compute gradient of smoothed max of overlap integrals"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation O_k = sum_i h_i * (1 - h_{i+k})
    corr = np.correlate(h, 1 - h, mode="full")
    O = corr * dx
    O_max = np.max(O)
    
    # Stable computation of weights
    exp_term = np.exp(alpha * (O - O_max))
    w = exp_term / np.sum(exp_term)
    
    # Compute gradient
    grad = np.zeros_like(h)
    
    # For each shift (index l in correlation result corresponds to shift = l - (n-1))
    for idx, shift in enumerate(range(-n+1, n)):
        w_k = w[idx]
        if w_k == 0:
            continue
            
        # Compute shift_right and shift_left
        if shift >= 0:
            # shift_right[i] = 1 for i < shift, else 1 - h[i-shift]
            shift_right = np.ones(n)
            if shift < n:
                shift_right[shift:] = 1.0 - h[shift:]
            
            # shift_left[i] = 0 for i < shift, else h[i-shift]
            shift_left = np.zeros(n)
            if shift < n:
                shift_left[shift:] = h[:n-shift]
        else:
            # shift >= 0 case
            s = -shift
            # shift_right[i] = 1 for i < s, else 1 - h[i-s]
            shift_right = np.ones(n)
            if s < n:
                shift_right[s:] = 1.0 - h[:n-s]
            
            # shift_left[i] = 0 for i >= n-s, else h[i+s]
            shift_left = np.zeros(n)
            if s < n:
                shift_left[:n-s] = h[s:]
        
        # Accumulate gradient contribution
        grad += w_k * (shift_right - shift_left)
    
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Number of points for optimization
    n_points = 80
    
    best_h = None
    best_c5 = np.inf
    dx = 2.0 / n_points
    
    def project_h(h):
        return project_to_box_simplex(h, n_points / 2)
    
    # Perform multiple restarts with different initialization strategies
    max_restarts = 50
    restart_count = 0
    
    while (time.time() - start_time) < budget_s * 0.95 and restart_count < max_restarts:
        # Choose initialization type and strategy
        init_type = restart_count % 5
        if init_type == 0:
            # Random uniform + projection
            h = np.random.rand(n_points)
            h = project_h(h)
        elif init_type == 1:
            # Sine wave
            x = np.linspace(0, 2, n_points)
            h = 0.5 + 0.4 * np.sin(np.pi * x)
            h = project_h(h)
        elif init_type == 2:
            # Step function
            h = np.zeros(n_points)
            mid = n_points // 2
            width = n_points // 4
            start = max(0, mid - width // 2)
            end = min(n_points, mid + width // 2)
            h[start:end] = 1.0
            h = project_h(h)
        elif init_type == 3:
            # Gaussian
            x = np.linspace(0, 2, n_points)
            mu = 1.0
            sigma = 0.2
            h = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
            h = project_h(h)
        else:  # init_type == 4
            # Linear gradient
            h = np.linspace(0, 1, n_points)
            h = project_h(h)
        
        # Perform gradient descent
        max_iters = 3000
        for it in range(max_iters):
            if (time.time() - start_time) > budget_s * 0.95:
                break
            
            # Compute smoothed gradient with increasing alpha
            alpha = 5 + (30 - 5) * (it / max_iters)  # Anneal alpha
            grad = compute_gradient(h, alpha)
            
            # Step with decreasing learning rate
            lr = 0.2 / np.sqrt(it + 1)
            h = h - lr * grad
            
            # Project back to feasible region
            h = project_h(h)
            
            # Evaluate and update best
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Early stopping if improvement is small
            if best_c5 < 0.379:
                break
        
        restart_count += 1
    
    # Final local refinement of best solution
    if best_h is not None:
        # Run more iterations with increased smoothing
        n_refine = 2000
        for it in range(n_refine):
            if (time.time() - start_time) > budget_s * 0.95:
                break
            alpha = 50  # High value for sharp max
            grad = compute_gradient(best_h, alpha)
            lr = 0.1 / np.sqrt(it + 1)
            best_h = best_h - lr * grad
            best_h = project_h(best_h)
            c5 = evaluate_c5(best_h)
            if c5 < best_c5:
                best_c5 = c5
    
    # Make sure we return something
    if best_h is None:
        # Fallback to simple initialization
        h_fallback = np.ones(n_points) * 0.5  # uniform
        h_fallback = project_h(h_fallback)
        best_c5 = evaluate_c5(h_fallback)
        best_h = h_fallback.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
