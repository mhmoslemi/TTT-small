import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_objective(h, beta, corr_vals):
    # corr_vals = [C0, C1, ..., C_{2n-2}]
    # Implement: (1/beta) * log(sum(exp(beta * C_l)))
    corr_vals = np.array(corr_vals)
    max_corr = np.max(corr_vals)
    # Numerical stability
    exp_vals = np.exp(beta * (corr_vals - max_corr))
    log_sum = np.log(np.sum(exp_vals)) + max_corr
    return log_sum / beta

def smooth_max_gradient(h, beta, corr_vals):
    # Compute the weighted sum of individual gradients ∇C_ℓ
    # ∇C_ℓ(i) = 1 - h[i+ℓ] - h[i-ℓ] for each ℓ, where out-of-bound indices contribute 0
    # So ∇C_l = 1 - shift_plus - shift_minus, and we compute all terms in one operation
    
    # Pre-fill with ones
    grad = np.ones_like(h)

    # Compute negative shifts efficiently using FFT multiplication
    # Shift the list h left (this creates an operation what shifts left by ℓ positions)
    # Using zero-padding from left and right makes interpretation easier
    
    # We need: 
    # 1 - (h_{i+1} + h_{i-1}) for each i upon shifts of 1  (2 lags: ℓ=1 and ℓ=-1)
    # 1 - (h_{i+2} + h_{i-2}) for each i upon shifts of 2  
    # etc
    
    # Clean and direct approach:
    # Convolution might not directly compute this. So we manually construct for
    # each lag ℓ, and add the weighted gradients
    
    corr_vals = np.array(corr_vals)
    max_corr = np.max(corr_vals)
    exp_vals = np.exp(beta * (corr_vals - max_corr))
    total_exp = np.sum(exp_vals)
    weights = exp_vals / total_exp  # normalization
    
    # Build the linear gradient at each index
    total_grad = np.zeros_like(h)
    
    # Reconstruct the full set of lag components ∇Cl = 1 - h_{i+ℓ} - h_{i-ℓ}
    # by hand-shifts:  (we can do the shift efficiently using convolution with
    # impulse responses, e.g., shift→by 1
    for i in range(len(weights)):
        # i corresponds to lag = i - (n-1) in the corr array
        lag = i - (len(h) - 1)
        if lag == 0:
            continue  # ∇C0 = 0, not needed in weighted sum
    
        # Compute shift by +lag (when lag > 0) or -lag (when lag < 0)
        shifted = np.zeros_like(h)
        if lag > 0:
            shifted[lag:] = h[:-lag]
        elif lag < 0:
            shifted[:lag] = h[-lag:]
        # else lag == 0, handled above
    
        # Apply weigth and accumulate
        weight = weights[i]
        total_grad = total_grad - weight * shifted  # match sign
    
        # Apply shift by -lag
        shifted2 = np.zeros_like(h)
        if -lag > 0:
            shifted2[-lag:] = h[:lag]
        elif -lag < 0:
            shifted2[:-lag] = h[lag:]
        # else lag == 0
    
        total_grad = total_grad - weight * shifted2
    
    return total_grad

def projected_gradient_descent_smoothed(h0, beta_schedule, max_steps_per_beta, alpha0=0.4, gamma=1e-4):
    """Projected gradient descent minimizing the smoothed max"""
    h = h0.copy()
    
    # Compute initial correlation values
    corr = compute_corr_fft(h)
    corr_vals = corr.copy()
    
    for i_beta, beta in enumerate(beta_schedule):
        # For a fixed beta: perform gradient descent steps
        for step in range(max_steps_per_beta):
            # Evaluate objective and gradient
            obj_val = smooth_max_objective(h, beta, corr_vals)
            
            # Gradient of smoothed max wrt current h
            grad = smooth_max_gradient(h, beta, corr_vals)

            # Update with step size
            alpha_t = alpha0 / (1 + gamma * step)
            if alpha_t < 1e-8:
                break

            h_new = h - alpha_t * grad
            
            # Project the new point
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            # If step size is too small or no improvement, stop early
            obj_new = smooth_max_objective(h_new, beta, corr_vals)
            if obj_new >= obj_val - 1e-12:
                h = h_new
                obj_val = obj_new
            else:
                # backtracking line search
                factor = 0.5
                while factor >= 1e-6:
                    h_try = h - alpha_t * factor * grad
                    h_try = project_to_sum_box(h_try, TARGET_SUM)
                    obj_try = smooth_max_objective(h_try, beta, corr_vals)
                    if obj_try <= obj_val:
                        h = h_try
                        break
                    factor *= 0.5
                if factor < 1e-6:
                    h = h_new
            
            # Update corr_vals for next iteration to save recomputation
            corr = compute_corr_fft(h)
            corr_vals = corr.copy()
    
    return h
