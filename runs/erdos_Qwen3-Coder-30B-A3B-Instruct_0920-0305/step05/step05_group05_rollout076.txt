import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

        a = cp.Variable(L_try)
        b = cp.Variable(L_try)
        s = cp.Variable(L_try)
        t = cp.Variable()

        A = np.zeros((M, 2*L_try))
        for i, x in enumerate(x_sample):
            for l in range(1, L_try+1):
                A[i, l-1] = np.cos(np.pi * l * x)  # a_l
                A[i, L_try+l-1] = np.sin(np.pi * l * x)  # b_l
        
        # Box constraints: 0 <= h(x) <= 1
        h_lb = 0.5 + A @ cp.vstack([a, b]) >= 0
        h_ub = 0.5 + A @ cp.vstack([a, b]) <= 1
        
        # SOC constraints: s_l >= a_l^2 + b_l^2
        soc_constraints = [cp.norm(cp.vstack([2*a[l], 2*b[l], s[l]-1]), 2) <= s[l]+1 for l in range(L_try)]

        if L_try == 0:
