import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Phase 1: Global optimization with differential evolution
    try:
        # Use a more robust approach with multiple restarts
        best_h = h0.copy()
        best_c5 = evaluate_c5(h0)
        
        # Run multiple differential evolution runs
        for i in range(10):
            if (time.time() - start_time) > budget_s * 0.95:
                break
            try:
                res = differential_evolution(
                    objective, 
                    bounds, 
                    seed=seed + i,
                   maxiter=500,
                    popsize=15,
                    tol=1e-12,
                    mutation=(0.5, 1),
                    recombination=0.7,
                    callback=lambda x, convergence: None
                )
                if res.success:
                    h_candidate = res.x
                    dx = 2.0 / n_points
                    scale = 1.0 / (np.sum(h_candidate) * dx)
                    h_candidate = h_candidate * scale
                    if np.abs(np.sum(h_candidate) * dx - 1.0) < 1e-10:
                        c5_candidate = objective(h_candidate)
                        if c5_candidate < best_c5:
                            best_h = h_candidate.copy()
                            best_c5 = c5_candidate
            except:
                continue
        
    except Exception as e:
        best_h = h0.copy()
        best_c5 = evaluate_c5(h0)
    
    # Phase 2: Local refinement with trust-constr    
    try:
        res = minimize(
            objective, 
            best_h, 
            method='trust-constr', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 3000, 'gtol': 1e-14, 'ftol': 1e-14}
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            if np.abs(np.sum(h_candidate) * dx - 1.0) < 1e-10:
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
    except:
        pass

    # Phase 3: Additional fine-tuning with multiple local searches
    local_restart_bound = 300
    
    for restart in range(local_restart_bound):
        if (time.time() - start_time) > budget_s * 0.95:  # Leave time for final steps
            break
            
        # Start with small random perturbations around previous best
        new_h = best_h + np.random.normal(0, 0.005, n_points)  
        new_h = np.clip(new_h, 0, 1)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        if np.abs(np.sum(new_h) * dx - 1.0) > 1e-10:
            continue
            
        try:
            res = minimize(
                objective, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                if np.abs(np.sum(h_candidate) * dx - 1.0) < 1e-10:
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
        except:
            continue

    # Phase 4: Try pattern-based improvements if time allows
    if (time.time() - start_time) < budget_s * 0.9 and n_points > 10:
        try:
            # Try to improve with a "spiky" pattern - high values at edges
            x = np.linspace(0, 2, n_points)
            # Create a sharp peak in the middle and low values at edges
            spiky_pattern = np.zeros(n_points)
            # Add peak in the middle
            middle = n_points // 2
            spiky_pattern[middle] = 1.0
            # Smooth the edges
            for i in range(1, n_points//4):
                spiky_pattern[max(0, middle-i)] = max(0, spiky_pattern[max(0, middle-i)] + 0.5)
                spiky_pattern[min(n_points-1, middle+i)] = max(0, spiky_pattern[min(n_points-1, middle+i)] + 0.5)
            
            # Mix with current best solution
            mixed_h = 0.2 * best_h + 0.8 * spiky_pattern
            
            # Normalize
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(mixed_h) * dx)
            mixed_h = mixed_h * scale_factor
            
            res = minimize(
                objective, 
                mixed_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                if np.abs(np.sum(h_candidate) * dx - 1.0) < 1e-10:
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
        except:
            pass

    # Return final result
    return (best_h.tolist(), best_c5, n_points)
