def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def soft_max_overlaps(O, alpha):
    """Compute the soft max of overlaps for smoothing"""
    if len(O) == 0:
        return 0
    max_O = np.max(O)
    # Subtract max to prevent numerical overflow
    exp_terms = np.exp(alpha * (O - max_O))
    return max_O + np.log(np.sum(exp_terms)) / alpha

def project_to_simplex(v, target_sum):
    """Project v onto the capped simplex with sum = target_sum"""
    v = np.array(v)
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-12:
        return v
    
    # Binary search for the Lagrange multiplier lambda
    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        projected = np.clip(v - mid, 0, 1)
        new_sum = np.sum(projected)
        if new_sum > target_sum:
            low = mid
        else:
            high = mid
    lambda_val = (low + high) / 2.0
    result = np.clip(v - lambda_val, 0, 1)
    return result

def interpolate_control_points(control_points, n_full, x_control):
    """Linearly interpolate control points to full grid"""
    # x_control is from 0 to 2 with K points 
    x_full = np.linspace(0, 2, n_full)
    # Linear interpolation from control values to full grid
    h_full = np.interp(x_full, x_control, control_points)
    return h_full

def evaluate_objective_control(control_points, n_full, x_control, alpha=5.0):
    """Evaluate objective using soft max of overlap"""
    h = interpolate_control_points(control_points, n_full, x_control)

    dx = 2.0 / n_full
    corr = correlate(h, 1 - h, mode="full")
    O = corr * dx
    return soft_max_overlaps(O, alpha)
