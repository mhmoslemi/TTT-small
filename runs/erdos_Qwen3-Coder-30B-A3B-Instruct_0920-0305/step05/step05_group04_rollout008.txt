import numpy as np

def conv_fft(a, b):
    """Compute linear convolution using FFT with zero padding."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    # Use FFT size that's a power of 2 for efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def project_onto_bounded_simplex(x):
    """Project x onto {h : 0 ≤ h_i ≤ 1, sum h_i = n/2} using bisection."""
    n = len(x)
    target_sum = n / 2.0
    x = np.clip(x, 0, 1)
    
    def sum_with_shift(lam):
        return np.sum(np.clip(x + lam, 0, 1))
    
    low, high = -1, 1
    while sum_with_shift(low) < target_sum:
        low *= 2
    while sum_with_shift(high) > target_sum:
        high *= 2
    
    # Bisection method
    for _ in range(30):
        mid = (low + high) / 2
        if sum_with_shift(mid) < target_sum:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2
    return np.clip(x + lam, 0, 1)

def soft_weights(c, beta):
    """Compute softmax weights for given correlation values and temperature."""
    c_max = np.max(c)
    exp_term = np.exp(-beta * (c - c_max))
    return exp_term / np.sum(exp_term)

def smooth_obj_and_grad(h, beta):
    """Compute the smooth objective and its gradient."""
    n = len(h)
    # compute c = conv(h,h) for shifts 1 to n-1
    c = conv_fft(h, h)
    # Only the non-zero shifts (1 to n-1 for zero-padded convolution)
    c = c[1:n]
    
    # compute weights
    w = soft_weights(c, beta)
    
    # compute gradient: - (conv(h, w) + conv(h, rev(w)))
    grad = -(conv_fft(h, w) + conv_fft(h, w[::-1]))[:n]
    
    # smooth objective
    obj = 1 + (1 / beta) * np.log(np.sum(np.exp(-beta * c)))
    
    return obj, grad

def gradient_descent_step(h, beta, alpha, max_iter=1000):
    """Perform gradient descent with Armijo line search."""
    prev_obj = None
    for _ in range(max_iter):
        obj, grad = smooth_obj_and_grad(h, beta)
        if prev_obj is not None and abs(obj - prev_obj) < 1e-12:
            break
        prev_obj = obj
        
        # Armijo line search
        tau = alpha
        while True:
            h_new = project_onto_bounded_simplex(h - tau * grad)
            obj_new, _ = smooth_obj_and_grad(h_new, beta)
            if obj_new <= obj - 0.1 * tau * np.dot(grad, grad):
                break
            tau *= 0.5
            if tau < 1e-12:
                break
                
        h = h_new
    return h
