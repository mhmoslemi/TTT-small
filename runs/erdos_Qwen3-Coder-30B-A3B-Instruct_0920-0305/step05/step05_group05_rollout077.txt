import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

    # and solve the problem with convex optimization.

    L = 12  # number of cosine/sine components
    # Generate sampling points for enforcing box constraints
    M = 4 * L  # number of sampling points
    x_sample = np.linspace(0, 2, M, endpoint=False)
    
    # Build the Fourier basis matrix at sampling points
    # Each column is [1, cos(pi*x), sin(pi*x), cos(2*pi*x), sin(2*pi*x), ...]
    # x_samples are fixed at [0, 2/M, ..., 2*(M-1)/M]
    Phi = np.ones((M, 1 + 2*L))
    for ell in range(1, L+1):
        Phi[:, 2*ell-1] = np.cos(np.pi * ell * x_sample)
        Phi[:, 2*ell] = np.sin(np.pi * ell * x_sample)
    
    # Define convex variables: Fourier coefficients a and b
    # Also the squared magnitudes s (for SOC constraint)
    u = cp.Variable(2*L)  # [a1, ..., aL, b1, ..., bL]
    s = cp.Variable(L)   # [s1=a1^2+b1^2, ..., sL=aL^2+bL^2]
    
    # The objective variable t is the upper bound on max overlap
    t = cp.Variable(1)
    
    # Objective: minimize t (upper bound on overlap)
    objective = cp.Minimize(t)
    
    # Constraints:
    
    # 1. Box constraints: 0 <= h(x_i) <= 1 for all sampled points
    # h(x) = 0.5 + Phi @ u
    h_expr = 0.5 + Phi @ u
    box_constraints = [h_expr >= 0, h_expr <= 1]
    
    # 2. Second-order cone constraints for s_l >= a_l^2 + b_l^2
    # We use the form: ||(2*a_l, 2*b_l, s_l - 1)||_2 <= s_l + 1
    cone_constraints = []
    for i in range(L):
        a_i = u[i]
        b_i = u[i + L]
        s_i = s[i]
        # ||(2*a, 2*b, s-1)||_2 <= s+1
        cone_constraints.append(cp.norm([2*a_i, 2*b_i, s_i - 1], 2) <= s_i + 1)
    
    # 3. Toeplitz matrix constraint: T(t,s) is PSD for upper bound t on max overlap
    # The Toeplitz matrix T is (L+1)x(L+1)
    # T[i,j] = t if i=j, else -s[|i-j|] for i,j = 0,...,L
    # (We index from 0 so it's more straightforward)
    S = np.zeros((L+1, L+1))
    # Fill diagonal with t
    S[0,0] = t[0]
    # Fill off-diagonal with -s entries
    for i in range(1, L+1):
        S[i,0] = -s[i-1]
        S[0,i] = -s[i-1]
    for i in range(1, L+1):
        for j in range(1, L+1):
            S[i,j] = -s[abs(i-j)]
    
    psd_constraint = [cp.PSD(S)]
    
    # Assemble all constraints
    constraints = box_constraints + cone_constraints + psd_constraint
    
    # Create the problem
    prob = cp.Problem(objective, constraints)
    
    # Attempt to solve it
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000, eps=1e-6)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            # Retrieve the coefficients
            a_vals = u.value[:L]
            b_vals = u.value[L:]
            
            # Generate h(x) via the Fourier expansion
            h_recomputed = 0.5 + np.array([sum(a_vals[i]*np.cos(np.pi*(i+1)*x) + 
                                            b_vals[i]*np.sin(np.pi*(i+1)*x)) 
                                           for x in np.linspace(0, 2, n_points, endpoint=False)])
            
            # Enforce bounds
            h_recomputed = np.clip(h_recomputed, 0, 1)
            
            # Renormalize to satisfy integral=1
            dx = 2.0 / n_points
            integral = np.sum(h_recomputed) * dx
            h_recomputed = h_recomputed / integral
            
            # Compute true C5 value
            c5_value = evaluate_c5(h_recomputed)
            
            # Check if it's within budget
            elapsed = time.time() - start_time
            if elapsed >= budget_s * 0.95:

    except Exception as e:

        c5_value = evaluate_c5(h0)
        return (h0.tolist(), c5_value, n_points)
