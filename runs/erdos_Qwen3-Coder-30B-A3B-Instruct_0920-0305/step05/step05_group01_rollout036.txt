import numpy as np

from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.irfft(np.fft.rfft(h_rev, 2*n - 1) * np.fft.rfft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def random_pair_move(h):
    """Generate a pairwise mass transfer move, ensuring feasibility"""
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    delta_max = min(h[i], 1.0 - h[j])
    delta_min = max(-h[j], h[i] - 1.0)
    if delta_min >= delta_max:
        return i, j, 0.0
    delta = np.random.uniform(delta_min, delta_max)
    return i, j, delta

def guided_move(h, corr_full):
    """Guided deterministic move for the worst lag to minimize the correlation"""
    # Find the worst lag
    worst_lag = np.argmax(corr_full)
    n = len(h)
    
    # Compute subgradient g_i = h_{i-\ell} + h_{i+\ell}
    g = np.zeros(n)
    for i in range(n):
        left_idx = i - worst_lag
        right_idx = i + worst_lag
        if left_idx >= 0:
            g[i] += h[left_idx]
        if right_idx < n:
            g[i] += h[right_idx]
    
    # Find indices with min and max g
    i_min = np.argmin(g)
    i_max = np.argmax(g)
    
    if i_min == i_max:
        return None  # No improvement possible
    
    # Move a small amount from i_min to i_max
    delta_max = min(h[i_min], 1.0 - h[i_max])
    if delta_max <= 0:
        return None
    
    delta = min(delta_max, 0.005)  # Smaller step
    return i_min, i_max, delta

def optimize_by_gradient_descent(h0, max_iter=5000):
    """Try gradient descent approach with constraint"""
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h0))]
    
    try:
        res = minimize(
            objective,
            h0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
    except:
        pass
    return h0, evaluate_c5(h0)
