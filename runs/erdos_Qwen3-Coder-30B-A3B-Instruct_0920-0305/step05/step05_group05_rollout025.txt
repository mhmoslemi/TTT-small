import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    n = 128  # Grid points
    K = 12   # Control points
    dx = 2.0 / n
    T = n / 2.0  # Target sum
    
    # Compute knot positions (equi-spaced)
    knot_positions = np.linspace(0, 2, K)
    
    def project_to_simplex(h):
        """Projects h onto the capped simplex: sum = T, 0 <= h[i] <= 1"""
        # clip to [0,1]
        h = np.clip(h, 0, 1)
        # binary search for lambda
        sum_h = np.sum(h)
        if abs(sum_h - T) < 1e-12:
            return h
        lo, hi = -100, 100
        for _ in range(100):
            lambda_ = (lo + hi) / 2
            z = np.clip(h - lambda_, 0, 1)
            sum_z = np.sum(z)
            if sum_z > T:
                lo = lambda_
            else:
                hi = lambda_
        lambda_ = (lo + hi) / 2
        z = np.clip(h - lambda_, 0, 1)
        return z
    
    def interpolate_to_full(c):
        """Interpolate control values c to full grid."""
        h = np.interp(np.linspace(0, 2, n), knot_positions, c)
        return h
    
    def compute_c5(h):
        """Fast evaluation using FFT."""
        # Compute correlation
        H = np.fft.rfft(h)
        G = np.fft.rfft(1 - h)
        corr = np.fft.irfft(H * np.conj(G))
        # Only the first n elements are relevant due to circular convolution
        corr = corr[:n]
        # Max of linear correlatons
        return np.max(corr) * dx
    
    def smoothed_max(O, alpha):
        max_O = np.max(O)
        if max_O == 0:
            return 0
        O_shifted = O - max_O  # Avoid overflow
        exp_O = np.exp(alpha * O_shifted)
        return max_O + np.log(np.sum(exp_O)) / alpha
    
    def evaluate_with_smoothing(c, alpha):
        """Evaluate with soft-max smoothing."""
        h = interpolate_to_full(c)
        h = project_to_simplex(h)
        # Compute all correlations
        H = np.fft.rfft(h)
        G = np.fft.rfft(1 - h)
        corr = np.fft.irfft(H * np.conj(G))
        corr = corr[:n]
        return smoothed_max(corr, alpha)
    
    def evaluate_exact(c):
        """Evaluate the exact C5."""
        h = interpolate_to_full(c)
        h = project_to_simplex(h)
        return compute_c5(h)
    
    # Use memetic approach: CMA-ES with annealing and local refinement
    best_c5 = float('inf')
    best_h = None
    
    # Multi-start
    num_restarts = 3
    
    for restart_idx in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Start with a structured shape
        if restart_idx == 0:
            # Sine-like
            c0 = 0.5 + 0.4 * np.sin(np.pi * knot_positions)
        elif restart_idx == 1:
            # Triangular
            c0 = np.clip(1 - np.abs(knot_positions - 1), 0, 1)
        else:
            # Random
            c0 = np.random.rand(K)
        
        c0 = np.clip(c0, 0, 1)
        
        # Settings for CMA-ES (simple version)
        iters = 1500  # Max iterations (approximately)
        λ = 4 + 3 * int(np.log(K))  # Population size
        α_start = 5.0
        α_end = 30.0
        α_schedule = np.linspace(α_start, α_end, int(0.7 * iters))
        α_schedule = np.append(α_schedule, [α_end] * int(0.3 * iters))
        
        # Simple randomized search with repair
        current_best = c0.copy()
        current_best_c5 = evaluate_exact(current_best)
        
        # Evaluate initial
        current_best = project_to_simplex(interpolate_to_full(current_best))
        current_best_c5 = evaluate_exact(current_best)
        
        print(f"Restart {restart_idx}: Initial score = {current_best_c5}")
        
        # CMA-ES-like algorithm (avoid dependencies on external libraries)
        # Using random mutations + projection + selection (non-iterative but).
        # Very basic memetic optimizer:
        # Loop: sample candidates, evaluate, project, keep best
        
        # Large number of random samples
        num_samples = 5000
        for sample_i in range(num_samples):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Mutate
            c_new = current_best.copy()
            for i in range(K):
                if np.random.rand() < 0.5:
                    # Small Gaussian perturbation
                    c_new[i] += np.random.normal(0, 0.02)
            
            # Repair and evaluate
            c_new = np.clip(c_new, 0, 1)
            h_new = interpolate_to_full(c_new)
            h_new = project_to_simplex(h_new)
            c5_new = compute_c5(h_new)
            
            if c5_new < current_best_c5:
                current_best = c_new.copy()
                current_best_c5 = c5_new
                
        # Local refinement
        # Use projected gradient descent on the exact max
        # Simple projected sub-gradient descent
        
        h_current = interpolate_to_full(current_best)
        h_current = project_to_simplex(h_current)
        max_corr_idx = np.argmax(correlate(h_current, 1 - h_current, mode="full")[:n])
        lr = 0.05
        
        # Fixed number of refinements (slow) due to simplicity
        fine_iters = 300
        for k in range(fine_iters):
            if time.time() - start_time > budget_s * 0.95:
                break
        
            # Direction to increase max corr: use current optimal shift
            O_full = correlate(h_current, 1 - h_current, mode="full")[:n]
            max_k = np.argmax(O_full)
            # Compute the gradient for shift max_k
            # Slide h by max_k and construct the gradient contribution
            grad = np.zeros(n)
            for i in range(n):
                # contribution to O_max_k
                j = (i + max_k) % n
                grad[i] = (1 - h_current[j])  # Derivative w.r.t. h[i]
                j = (i - max_k) % n
                grad[i] -= h_current[j]       # Derivative w.r.t. h[i]
                
            # Perform projected gradient step
            h_new = h_current - lr * grad
            # Project back onto simplex
            h_new = project_to_simplex(h_new)
            # Re-evaluate C5
            c5_new = compute_c5(h_new)
            if c5_new < current_best_c5:
                current_best_c5 = c5_new
                h_current = h_new
                print(f"Refinement step {k}: C₅ = {c5_new}")
                
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            h_best = interpolate_to_full(current_best)
            h_best = project_to_simplex(h_best)
            best_h = h_best.copy()
            
        print(f"Restart {restart_idx} finished with C₅ = {current_best_c5}")
    
    final_c5 = compute_c5(best_h)
    print(f"Final C₅ = {final_c5}")
    
    return (best_h.tolist(), final_c5, n)
