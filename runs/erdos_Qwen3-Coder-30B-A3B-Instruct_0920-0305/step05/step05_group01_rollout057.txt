import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def random_move(h, target_sum):
    """Generate a random move that preserves sum to target"""
    n = len(h)
    # Randomly select two indices
    i, j = np.random.choice(n, 2, replace=False)
    # Compute max delta (how much we can move)
    delta_max = min(h[i], 1.0 - h[j])  # move from i to j
    if delta_max <= 0:
        return h, 0.0
    # Sample a random delta
    delta = np.random.uniform(0, delta_max)
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return project_to_sum_box(h_new, target_sum), delta

    best_h = h.copy()
    best_c5 = current_c5

    T0 = 0.01
    Tf = 1e-6
    t_start = time.time()
    
    # Variable exploration rate
    iter_count = 0
    max_iters = 1000000
    
    while time.time() - t_start < run_seconds and iter_count < max_iters:
        # Dynamic temperature
        t_elapsed = time.time() - t_start
        T = T0 * np.exp(-t_elapsed / (run_seconds * 0.3)) + Tf
        
        # Adapted move probabilities
        if iter_count % 10 == 0:
            # Deterministic move: try to avoid large correlations
            corr_full = compute_corr_fft(h)
            worst_lag = np.argmax(corr_full)
            # Make a small move aimed at reducing that correlation
            n = len(h)
            i = np.random.randint(0, n)
            j = (i + worst_lag) % n if worst_lag > 0 else np.random.randint(0, n)
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max > 0:
                delta = min(delta_max, 0.02)
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                new_c5 = evaluate_c5(h_new)
                if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                    h = h_new
                    current_c5 = new_c5
                    if current_c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = current_c5
        else:
            # Random move with adaptive step size
            h_new, delta = random_move(h, TARGET_SUM)
            if delta > 0:
                new_c5 = evaluate_c5(h_new)
                if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                    h = h_new
                    current_c5 = new_c5
                    if current_c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = current_c5
        
        iter_count += 1
    
    return best_h, best_c5

def fancy_local_optimize(h):
    """Use L-BFGS-B for better local convergence"""
    def objective(h):
        return evaluate_c5(h)
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h))]
    
    try:
        res = minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
    except:
        pass
    return h, evaluate_c5(h)
