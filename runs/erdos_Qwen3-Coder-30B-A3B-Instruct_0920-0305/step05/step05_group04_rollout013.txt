import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def project_simplex(x, sum_constraint):
    """Project x onto the simplex with upper bound 1 and given sum constraint"""
    # Clip to [0, 1]
    x = np.clip(x, 0, 1)
    # Adjust sum to sum_constraint
    x_sum = np.sum(x)
    if np.abs(x_sum - sum_constraint) < 1e-12:
        return x
    
    # Use bisection to find lambda
    def objective(lam):
        return np.sum(np.clip(x + lam, 0, 1)) - sum_constraint

    a, b = -x.min(), 1 - x.max()
    for _ in range(100):
        if a >= b:
            break
        c = (a + b) / 2
        fc = objective(c)
        if np.abs(fc) < 1e-12:
            break
        if fc > 0:
            b = c
        else:
            a = c

    lam = (a + b) / 2
    result = np.clip(x + lam, 0, 1)
    return result

def compute_correlations(h):
    """Compute correlations c_k = sum h_i h_{i+k}"""
    n = len(h)
    # Use full convolution to compute correlations, remove zero shift term (we don't want c_0)
    corr_full = correlate(h, h, mode='full')
    correlations = corr_full[n-1:]  # Only shift terms 0 to n-1 (we're interested in 1 to n-1)
    return correlations[1:]  # exclude c_0

def compute_softmin_gradient(h, correlations, beta):
    """Compute gradient of softmin using FFT-based correlations and weights"""
    n = len(h)
    # Compute soft weights
    c_max = np.max(correlations)
    exp_terms = np.exp(-beta * (correlations - c_max))
    w = exp_terms / np.sum(exp_terms)
    
    # FFT-based gradient computation: 
    # g_i = sum_k w_k (h_{i-k} + h_{i+k})
    # This is equivalent to convolution: conv(h, w) + conv(h, reverse(w))
    # But need to handle indices correctly with wrapping
    h_fft = np.fft.fft(h)
    w_fft = np.fft.fft(np.concatenate([w, [0]*(n-len(w))]))  # zero pad
    # Use real convolution (conjugate symmetry) to get linear convolution
    conv_result = np.real(np.fft.ifft(h_fft * w_fft))
    if len(conv_result) > n:
        conv_result = conv_result[:n]

    # The convolution we want has the form
    # g_i = sum_k w_k (h_{i+k} + h_{i-k}) = conv(h, w) + conv(h, reverse(w))
    # But numpy does circular convolution. For linear case, use proper zero padding and
    # note: corr(i) = h_i * h_{i+1} + h_{i+1}*h_{i+2} + ...
    
    # We compute: [h * w_rev] + [h * w]  
    # i.e., solve: 
    # for shift k: w_k h_{i-k} and w_k h_{i+k}
    # Use convolution: w_rev conv h denote h_{i-k}
    # Another: w conv h, but shifted
    # The correct way: compute:
    # w_rev * h --> gives sum of h_{i-k} * w_k  
    # Do same for h * w_rev -> reverse, let's do left-shift convolution correction
    # In Python, that's more fiddly. So compute directly:
    # g is w*(h with left shift)
    # plus w_rev*(h with right shift).
    
    # Alternative: use the nice fact that the gradient only cares about symmetries.
    # For each k, the gradient component corresponding to h_i is  w_k*(h_{i-k} + h_{i+k})
    
    # Better: convince ourselves:
    # For c_k = sum h_i h_{i+k},  
    #	g_i = \sum_k w_k h_{i+k} + \sum_k w_k h_{i-k}
    # which is exactly the 2 convolutions
    
    # Let's use proper padding 
    h_padded = np.pad(h, (0, n), mode='constant', constant_values=0)
    w_padded = np.pad(w, (0, n), mode='constant', constant_values=0)
    
    # For the first term
    g1 = np.convolve(h_padded, w_padded[:n], mode='valid')[:n]
    # For the second term with w_rev (indices)
    w_rev = w[::-1]
    w_pad_rev = np.pad(w_rev, (0, n), mode='constant', constant_values=0)
    g2 = np.convolve(h_padded, w_pad_rev[:n], mode='valid')[:n]
    g = -(g1 + g2)  # negative due to sign in softmin formula
    
    return g

def compute_softmin_objective(h, correlations, beta):
    """Compute softmin objective = 1 + (1/beta) log sum exp(-beta c_k)"""
    c_max = np.max(correlations)
    exp_terms = np.exp(-beta * (correlations - c_max))
    log_sum = np.log(np.sum(exp_terms)) + c_max
    return 1 + log_sum / beta

def line_search(h, grad, obj_func, args, step_size=1.0):
    """Backtracking line search to ensure objective decreases"""
    alpha = 1.0
    c1 = 1e-4  # Armijo condition
    max_iter = 10
    obj_start = obj_func(h, *args)
    
    for _ in range(max_iter):
        trial_h = project_simplex(h - alpha * grad, len(h)/2)
        obj_trial = obj_func(trial_h, *args)
        if obj_trial <= obj_start + c1 * alpha * np.dot(grad, h - trial_h):
            return trial_h, alpha
        alpha *= 0.5
    return h, alpha
