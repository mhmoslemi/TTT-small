import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def project_to_simplex(h, target_sum):
    """Projects h onto the simplex with sum = target_sum and bounds [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h

    # Using Duchi et al. (2008) algorithm
    # Sort h in descending order
    h_sorted = np.sort(h)[::-1]
    rho = 0
    temp_sum = 0
    for i, val in enumerate(h_sorted):
        temp_sum += val
        if temp_sum > target_sum:
            break
        rho = i + 1
    theta = (temp_sum - target_sum) / rho if rho > 0 else 0
    return np.clip(h - theta, 0, 1)

def compute_correlations(h):
    """Compute correlation c_k = sum h_i h_{i+k} for k=1..n-1 using FFT."""
    n = len(h)
    # Zero-pad to avoid circular convolution effects
    # Length should be at least 2*n - 1
    pad_len = 2 * n - 1
    h_padded = np.pad(h, (0, pad_len - n), 'constant')
    h_rev = np.pad(h[::-1], (0, pad_len - n), 'constant')
    corr = np.convolve(h_padded, h_rev, mode='valid')
    # Only need correlations from shift 1 to n-1
    return corr[n-1:2*(n-1)]

def compute_soft_min_gradient(h, correlations, beta):
    """Compute gradient of soft-min approximation."""
    n = len(h)
    # Compute weights w_k = exp(-beta * c_k) / sum exp(-beta * c_t)
    c_max = np.max(correlations)
    exp_neg_beta_c = np.exp(-beta * (correlations - c_max))
    w = exp_neg_beta_c / np.sum(exp_neg_beta_c)
    # Gradient: - sum_k w_k * d(c_k)/dh_i
    # d(c_k)/dh_i = h_{i+k} + h_{i-k}, but only consider valid indices
    # We can compute via convolution:
    # grad_i = - sum_k w_k * (h_{i+k} + h_{i-k}) = - (w * h)_{i+1} - (w_rev * h)_{i+1}
    # But we want actual convolution result, so return directly:
    # Note: the convolution w * h should be defined over valid range
    w_padded = np.pad(w, (0, n - len(w)), 'constant')
    w_rev = w_padded[::-1]
    
    # Two convolutions: forward and backward
    conv_forward = np.convolve(h, w_padded, mode='full')[:n]
    conv_backward = np.convolve(h, w_rev, mode='full')[:n]
    # Sum (with appropriate shift compensation): 
    grad = -(conv_forward + conv_backward)
    return grad

def smooth_objective_and_gradient(h, beta):
    """Compute the smooth objective and its gradient with respect to h."""
    n = len(h)
    dx = 2.0 / n
    correlations = compute_correlations(h)
    c_max = np.max(correlations)
    exp_neg_beta_c = np.exp(-beta * (correlations - c_max))
    sum_exp = np.sum(exp_neg_beta_c)
    obj_val = 1.0 + (1.0 / beta) * np.log(sum_exp)
    
    gradient = compute_soft_min_gradient(h, correlations, beta)
    return obj_val, gradient

def line_search(h, g, beta, max_iter=10):
    """Backtracking line search to find step size tau."""
    alpha = 0.5
    beta_ls = 0.8
    c = 1e-4
    h0 = h.copy()
    phi0 = smooth_objective_and_gradient(h, beta)[0]
    
    for i in range(max_iter):
        tau = alpha * (0.5 ** i)
        h_trial = h - tau * g
        h_trial = project_to_simplex(h_trial, len(h) / 2)
        phi_trial = smooth_objective_and_gradient(h_trial, beta)[0]
        if phi_trial <= phi0 + c * tau * np.dot(g, h_trial - h0):
            return tau, h_trial
    return 0.0, h

def compute_exact_c5(h):
    """Compute true C5 using discrete max overlap calculation."""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode='full')
    # full correlation lengths equals 2*n - 1
    # shift 0 has no overlap so we skip it
    # the max is at the center where shift = 0, but we take max over other shifts
    # The slice corresponds to shift 1 to 2*n-2
    later_corr = corr[n-1:2*n-1]
    # But we only want shifts from 1 to n-1
    max_overlap = np.max(later_corr[:n-1])
    # max_overlap is actually 1 - min correlation, so C5 is max_overlap
    return max_overlap
