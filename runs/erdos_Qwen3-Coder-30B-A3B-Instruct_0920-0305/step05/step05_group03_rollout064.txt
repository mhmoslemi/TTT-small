import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from cvxpy import *
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_simplex(v, target_sum):
    """Project vector v onto the capped simplex with sum = target_sum"""
    lmin, lmax = -1.0, 1.0
    while True:
        lamb = (lmin + lmax) / 2
        p = np.clip(v + lamb, 0, 1)
        s = np.sum(p)
        if s > target_sum:
            lmin = lamb
        else:
            lmax = lamb
        if abs(s - target_sum) < 1e-10:
            break
    return p

def compute_active_shifts(h):
    """Compute active shifts that give the current max overlap"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_val = np.max(corr)
    active_shifts = np.where(np.abs(corr - max_val) < 1e-10)[0]
    # Convert to actual shift values
    return active_shifts - (n - 1)

def subgradient_step(h, active_shifts, dx):
    n = len(h)
    # Compute weight vector
    w = np.zeros(n)
    for k in active_shifts:
        if k >= 0:
            w[k:] += 1
        else:
            w[:k] += 1
    # Compute gradient
    rev_h = h[::-1]
    grad = dx * (np.convolve(w, np.ones(n), mode="full") - 
                 np.convolve(w, h, mode="full") - 
                 np.convolve(w, rev_h, mode="full"))
    # Trim to original size
    grad = grad[n-1:2*n-1]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize variables
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    
    # Phase 1: Binary MILP
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Phase 1: Binary Optimisation (MILP)
    try:
        # Create binary variables
        h_binary = Variable(n_points, boolean=True)
        tau = Variable()
        
        # Constraints
        constraints = [sum(h_binary) == n_points / 2]
        
        # Compute overlap for each shift
        for k in range(-n_points + 1, n_points):
            # Compute pairs
            start_i = max(0, -k)
            end_i = min(n_points, n_points - k)
            if start_i >= end_i:
                continue
            
            pairs = [(i, i + k) for i in range(start_i, end_i)]
            if not pairs:
                continue
            
            # Add auxiliary variables and constraints for products
            y_vars = [Variable(boolean=True) for _ in range(len(pairs))]
            
            for i, (i1, i2) in enumerate(pairs):
                y = y_vars[i]
                constraints.extend([
                    y <= h_binary[i1],
                    y <= h_binary[i2],
                    y >= h_binary[i1] + h_binary[i2] - 1
                ])
            
            # Compute overlap
            left = sum(h_binary[i1] for i1, i2 in pairs)
            right = sum(y_vars)
            overlap = left - right
            constraints.append(overlap <= tau)
        
        # Objective
        objective = Minimize(tau)
        problem = Problem(objective, constraints)
        
        # Solve
        problem.solve(solver=CP.ECOS_BB, time_limit=budget_s * 0.25)
        
        if problem.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h_sol = h_binary.value
            if h_sol is not None:
                h_sol = np.array(h_sol)
                # Evaluate solution
                c5_val = evaluate_c5(h_sol)
                if c5_val < best_c5:
                    best_h = h_sol.copy()
                    best_c5 = c5_val
    
    except Exception as e:
        pass
    
    # Phase 2: Binary hill climbing
    h_binary = best_h.round().astype(int)
    current_h = h_binary.astype(float)
    current_c5 = evaluate_c5(current_h)
    
    # Greedy swapping
    improved = True
    max_swap_iterations = 10000
    swap_count = 0
    while improved and swap_count < max_swap_iterations and (time.time() - start_time) < budget_s * 0.7:
        improved = False
        for i in range(n_points):
            if current_h[i] <= 0.5:
                continue
            for j in range(n_points):
                if current_h[j] >= 0.5:
                    continue
                # Simulate swap
                new_h = current_h.copy()
                new_h[i] = 0
                new_h[j] = 1
                new_c5 = evaluate_c5(new_h)
                if new_c5 < current_c5:
                    current_h = new_h
                    current_c5 = new_c5
                    improved = True
                    break
            if improved:
                break
        swap_count += 1
    
    # Phase 3: Continuous refinement using subgradient descent
    def objective(h):
        return evaluate_c5(h)
    
    # Use current best as starting point
    h_continuous = current_h.copy()
    
    # Initialize projection
    h_continuous = project_simplex(h_continuous, n_points/2)
    
    # Get current max
    max_shifts = compute_active_shifts(h_continuous)
    
    # Subgradient descent
    alpha = 0.5 * dx
    max_iter = 1000
    last_improvement = 0
    
    for i in range(max_iter):
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
        # Compute active shifts for current h
        max_shifts = compute_active_shifts(h_continuous)
        
        # Subgradient step
        grad = subgradient_step(h_continuous, max_shifts, dx)
        h_new = h_continuous - alpha * grad
        
        # Project
        h_new = project_simplex(h_new, n_points/2)
        
        # Symmetrize
        h_sym = (h_new + h_new[::-1]) / 2
        h_sym = project_simplex(h_sym, n_points/2)
        
        # Evaluate
        c5_new = evaluate_c5(h_sym)
        if c5_new < current_c5:
            h_continuous = h_sym
            current_c5 = c5_new
            last_improvement = 0
        else:
            last_improvement += 1
            if last_improvement > 200:
                alpha *= 0.5
                if alpha < 1e-10:
                    break
    
    return (h_continuous.tolist(), current_c5, n_points)
