import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_c5_discrete(h):
    """Evaluate C5 on discrete values; used for fine tuning"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def construct_h_from_x(x, n_points):
    """Construct a continuous h from binary x vector"""
    h = np.array(x)
    # Normalize to satisfy integral constraint: sum(h) * dx = 1
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    N = [36, 40, 44, 48, 52, 56, 60, 64]  # Even numbers to try
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Greedy construction for warm start
    def greedy_initialization(n_points):
        x = np.zeros(n_points, dtype=int)
        placed = 0
        while placed < n_points // 2:
            # Compute how much each candidate position helps the minimum p_k
            best_pos = -1
            best_min_p = -1
            for i in range(n_points):
                if x[i] == 1:
                    continue
                # Simulate adding this position and compute new p_k values
                new_x = x.copy()
                new_x[i] = 1
                p_vals = []
                for k in range(1, n_points):
                    p = 0
                    for j in range(n_points - k):
                        p += new_x[j] * new_x[j + k]
                    p_vals.append(p)
                min_p = min(p_vals) if p_vals else 0
                if min_p > best_min_p:
                    best_min_p = min_p
                    best_pos = i
            if best_pos >= 0:
                x[best_pos] = 1
                placed += 1
            else:
                # Fallback: pick an arbitrary available index
                for i in range(n_points):
                    if x[i] == 0:
                        x[i] = 1
                        placed += 1
                        break
        return x.tolist()
    
    # Solve exact MILP
    def solve_milp(n_points, x_initial):
        dx = 2.0 / n_points
        # Variable for h_i - binary
        x = cp.Variable(n_points, boolean=True)
        
        # Variables for y_{i,k} = x_i * x_{i+k}
        y = {}
        for k in range(1, n_points):
            for i in range(n_points - k):
                y[(i, k)] = cp.Variable(boolean=True)
        
        # Constraints on sums
        constraints = [cp.sum(x) == n_points // 2]
        
        # Symmetry breaking
        constraints.append(x[0] == 1)
        
        # Tributary ordering constraint
        left_sum = cp.sum(x[:n_points // 2])
        right_sum = cp.sum(x[n_points // 2:])
        constraints.append(left_sum >= right_sum)
        
        # Product constraints (McCormick envelopes)
        for k in range(1, n_points):
            for i in range(n_points - k):
                # y[i,k] <= x[i], y[i,k] <= x[i+k]
                constraints.append(y[(i, k)] <= x[i])
                constraints.append(y[(i, k)] <= x[i + k])
                # y[i,k] >= x[i] + x[i+k] - 1
                constraints.append(y[(i, k)] >= x[i] + x[i + k] - 1)
        
        # Compute p_k values
        p = {}
        for k in range(1, n_points):
            p[k] = cp.sum([y[(i, k)] for i in range(n_points - k)])
        
        # Objective: maximize min(p_k)
        # Introduce L as integer variable, min(p_k) >= L
        L = cp.Variable(integer=True)
        
        for k in range(1, n_points):
            constraints.append(p[k] >= L)
        
        # Objective to maximize L
        objective = cp.Maximize(L)
        prob = cp.Problem(objective, constraints)
        
        # Add initial guess
        try:
            x.value = np.array(x_initial)
            # All the other variables should be initialized too; avoid joint
            # upper-bound solver to disable for now and just treat as boolean LP
        except:
            pass
        
        # Set solve settings to tune performance
        prob.solve(solver=cp.CBC, verbose=False, time_limit=min(30, budget_s*0.8))
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_opt = [int(round(x[i].value)) for i in range(n_points)]
            L_opt = int(round(L.value))
            h_opt = construct_h_from_x(x_opt, n_points)
            c5_val = evaluate_c5(h_opt)
            return h_opt, c5_val, n_points
        return None, float('inf'), n_points

    # Main loop over resolutions
    
    for n in N:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Get initial construction
        x_init = greedy_initialization(n)
        
        # Solve MILP
        h_milp, c5_milp, n_milp = solve_milp(n, x_init)
        
        if h_milp is not None and c5_milp < best_c5:
            # Optionally run fine-tuning
            best_h = h_milp
            best_c5 = c5_milp
            best_n = n
            
    return (best_h, best_c5, best_n) if best_h is not None else ([0] * 42, 0.381, 42)
