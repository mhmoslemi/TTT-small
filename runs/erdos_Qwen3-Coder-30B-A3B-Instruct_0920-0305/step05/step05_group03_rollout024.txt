import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp
from math import floor

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_milp_binary(n, time_limit_s):
    """Solve the MILP formulation for binary h"""
    # Create variables
    h = cp.Variable(n, boolean=True)
    y = {}
    shifts = []
    
    # Create shift lists and auxiliary variables
    for k in range(-n+1, n):
        pairs = []
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                pairs.append((i, j))
        if pairs:
            shifts.append(k)
            # For this shift, we create an auxiliary variable per pair
            y[k] = cp.Variable(len(pairs), boolean=True)
    
    # Constraints: product linearisation
    constraints = []
    for k in shifts:
        pairs = []
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                pairs.append((i, j))
        for idx, (i,j) in enumerate(pairs):
            constraints += [
                y[k][idx] <= h[i],
                y[k][idx] <= h[j],
                y[k][idx] >= h[i] + h[j] - 1
            ]
            
    # Area constraint
    constraints.append(cp.sum(h) == n/2)
    #Epigraph constraint for largest overlap
    tau = cp.Variable()
    for k in shifts:
        pairs = []
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                pairs.append((i, j))
        # Compute O_k
        O_k = cp.sum(h) - cp.sum(y[k]) # Corrected: only sum over actual pairs
        constraints.append(O_k <= tau)
        
    # Minimize the epigraph variable
    objective = cp.Minimize(tau)
    problem = cp.Problem(objective, constraints)
    
    # Solve with time limit
    try:
        problem.solve(solver=cp.CBC, time_limit=time_limit_s, verbose=False)
    except Exception:
        pass
    
    if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_val = h.value
        if h_val is not None:
            return h_val
    # Return the best known feasible solution if any
    return None

def binary_hill_climb(h_bin, max_iter=5000):
    """
    Exhaustive hill-climbing over binary patterns by swapping 0 and 1
    """
    n = len(h_bin)
    dx = 2.0 / n
    
    def evaluate_current_c5(h_vec):
        return evaluate_c5(h_vec)
    
    current_h = h_bin.copy()
    best_h = current_h.copy()
    c5 = evaluate_current_c5(current_h)
    best_c5 = c5
    
    improved = True
    iterations = 0
    
    while improved and iterations < max_iter:
        improved = False
        best_swap_delta = 0
        best_i = -1
        best_j = -1
        
        # Scan all pairs (i,j) with h[i]=1, h[j]=0
        ones = np.where(current_h == 1)[0]
        zeros = np.where(current_h == 0)[0]
        
        # Consider a large subset of swaps to reduce computation
        candidates = []
        for i in ones:
            for j in zeros:
                if abs(i-j) > 10:  # Skip close ones for speedup
                    candidates.append((i,j))
                    
        np.random.shuffle(candidates)
        candidates = candidates[:min(1000, len(candidates))]  # Limit to 1000 swaps
        
        for i, j in candidates:
            new_h = current_h.copy()
            new_h[i] = 0
            new_h[j] = 1
            
            new_c5 = evaluate_current_c5(new_h)
            
            delta = new_c5 - c5
            
            if delta < best_swap_delta:
                best_swap_delta = delta
                best_i = i
                best_j = j
        
        if best_i != -1:
            current_h[best_i] = 0
            current_h[best_j] = 1
            c5 += best_swap_delta
            if c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = c5
            improved = True
            
        iterations += 1
        if iterations % 500 == 0:
            print("Hill-climbing iteration:", iterations, ", C5:", c5)
    
    return best_h, best_c5

def project_simplex(v, target_sum):
    """
    Project a vector v onto the capped simplex with sum = target_sum
    Clip values to [0,1], then rescale to match target_sum
    """
    v = np.clip(v, 0, 1)
    total = np.sum(v)
    if total > target_sum:
        # subtract excess: subtract (total - target_sum) / n from each component
        v -= (total - target_sum) / len(v)
        v = np.clip(v, 0, 1)
    elif total < target_sum:
        # add deficit: add (target_sum - total) / n to each component
        v += (target_sum - total) / len(v)
        v = np.clip(v, 0, 1)
    return v

def continuous_refinement(h_start, max_steps=2000, min_step_size=1e-6):
    """
    Refine a continuous h vector using sub-gradient descent with active-shift weighting
    """
    h = h_start.copy()
    dx = 2.0 / len(h)
    target_sum = len(h) / 2
    current_c5 = evaluate_c5(h)
    
    step_size = 0.5 * dx
    no_improvement_count = 0
    
    for step in range(max_steps):
        # Evaluate all overlaps and identify active shifts
        corr = correlate(h, 1 - h, mode="full")
        max_val = np.max(corr)
        active_shifts = np.where(corr == max_val)[0]
        
        # Convert from full convolution indices to real shift indices
        shifts = active_shifts - (len(h) - 1)
        
        # Build weight vector for active shifts
        w = np.zeros_like(h)
        for k in shifts:
            if k >= 0:
                # weight at position i is 1 for those j such that j = i + k
                # i from 0 to n-1-k
                for i in range(len(h) - k):
                    w[i] += 1
            else:
                # k < 0: j = i + k, so i = j - k where j from -k to n-1
                for j in range(-k, len(h)):
                    w[j - k] += 1
        
        # Weights for symmetric version
        w_rev = w[::-1]
        gradient = dx * (np.ones_like(h) - w - w_rev)
        
        # Take a step
        h_new = h - step_size * gradient
        
        # Project onto capped simplex
        h_new = project_simplex(h_new, target_sum)
        
        # Symmetrise
        h_new = (h_new + h_new[::-1]) / 2
        h_new = project_simplex(h_new, target_sum)
        
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < current_c5:
            h = h_new.copy()
            current_c5 = new_c5
            no_improvement_count = 0
            step_size = min(0.5 * dx, step_size * 1.1)  # Adaptive step size
        else:
            no_improvement_count += 1
            if no_improvement_count > 200:
                step_size *= 0.5
                no_improvement_count = 0
                if step_size < min_step_size:
                    break
                
    return h, current_c5

            best_h = h0.copy()

    # Phase 2: Binary hill-climb starting from binary or random

    # Phase 3: Continuous refinement
    if best_h is not None:

    # Phase 4: Optional multi-resolution
    if ((time.time() - start_time) < 0.8 * budget_s and best_h is not None):
        print("Starting upsampling...")
        # Double resolution
        n_double = 2 * n_points

        # Fallback
        h = h0.copy()
