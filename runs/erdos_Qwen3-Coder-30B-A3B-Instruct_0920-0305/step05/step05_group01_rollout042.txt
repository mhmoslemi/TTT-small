def evaluate_c5(h):
    """Evaluate C5 using correlation"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using scipy
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum value (excluding the middle which is zero lag)
    max_corr = np.max(corr)
    return dx * max_corr

def project_to_sum_box(x, target_sum):
    """Project to sum = target_sum, box constraints [0,1]"""
    # Simple projection via clipping with bias adjustment
    y = np.clip(x, 0, 1)
    current_sum = np.sum(y)
    if abs(current_sum - target_sum) < 1e-10:
        return y
    
    # Compute mean adjustment
    adjustment = (target_sum - current_sum) / len(x)
    y = np.clip(y + adjustment, 0, 1)
    
    # Fine-tune using binary search if needed
    low, high = -1.0, 1.0
    tolerance = 1e-12
    for _ in range(50):
        mid = (low + high) / 2.0
        test_y = np.clip(y + mid, 0, 1)
        test_sum = np.sum(test_y)
        if abs(test_sum - target_sum) < tolerance:
            return test_y
        elif test_sum > target_sum:
            high = mid
        else:
            low = mid
    
    return np.clip(y + low, 0, 1)
