import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Phase 1: Use differential evolution to get a good starting point
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    try:
        # Use differential evolution with a larger population for better exploration
        res = differential_evolution(
            objective, 
            bounds, 
            args=(),
            maxiter=1000, 
            popsize=30, 
            tol=1e-12,
            seed=seed,
            callback=lambda x, convergence: None if (time.time() - start_time) > budget_s * 0.9 else None
        )
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Phase 2: Use local optimization with multiple restarts
    local_restart_bound = 500
    
    for restart in range(local_restart_bound):
        if (time.time() - start_time) > budget_s * 0.95:  # Leave time for final steps
            break
            
        # Random perturbation around the current best
        new_h = best_h + np.random.normal(0, 0.02, n_points)  # Larger perturbation
        new_h = np.clip(new_h, 0, 1)  # Ensure bounds
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            # Try both SLSQP and trust-constr
            methods = ['SLSQP', 'trust-constr']
            for method in methods:
                res = minimize(
                    objective, 
                    new_h, 
                    method=method, 
                    bounds=bounds, 
                    constraints=cons,
                    options={'maxiter': 500, 'ftol': 1e-13, 'gtol': 1e-13}
                )
                if res.success:
                    h_candidate = res.x
                    scale = 1.0 / (np.sum(h_candidate) * dx)
                    h_candidate = h_candidate * scale
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
                        break  # Accept this improvement
        except:
            continue

    # Phase 3: Try some structured improvement
    if (time.time() - start_time) < budget_s * 0.8:
        try:
            # Try perturbation with sinusoidal pattern
            x = np.linspace(0, 2, n_points)
            sinusoid = 0.5 + 0.3 * np.sin(2 * np.pi * x)
            mixed_h = 0.7 * best_h + 0.3 * sinusoid
            
            # Normalize
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(mixed_h) * dx)
            mixed_h = mixed_h * scale_factor
            
            res = minimize(
                objective, 
                mixed_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass

    return (best_h.tolist(), best_c5, n_points)
