import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
n_points_fine = 240
dx_fine = 2.0 / n_points_fine

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def evaluate_c5_fine(h):
    """Evaluation with fine grid"""
    corr = compute_corr_fft(h)
    return dx_fine * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h, corr_vals, beta):
    """Compute smoothed gradient using FFT-based correlation"""
    n = len(h)
    max_corr = np.max(corr_vals)
    
    # Compute exponents with numerical stability
    exp_vals = np.exp(beta * (corr_vals - max_corr))
    weights = exp_vals / np.sum(exp_vals)
    
    # Now compute weighted sum of gradients for each lag
    grad_temp = np.zeros_like(h)
    for lag_idx, weight in enumerate(weights):
        if lag_idx >= n:
            # Corresponds to negative lag, which is handled with reverse indexing
            lag = lag_idx - n + 1
            # lags 1..n-1: shift left by lag
            shifted = np.concatenate([h[lag:], np.zeros(lag)])
            # lags n+1..2n-2: shift right by (n-lag) 
            backshifted = np.concatenate([np.zeros(n-lag), h[:n-lag]])
            grad_temp += weight * (1 - shifted - backshifted)
        else:
            # Positive lags
            if lag_idx == 0:
                # No shift
                grad_temp += weight * (1 - h - h)
            else:
                shifted = np.concatenate([h[lag_idx:], np.zeros(lag_idx)])
                backshifted = np.concatenate([np.zeros(lag_idx), h[:-lag_idx]])
                grad_temp += weight * (1 - shifted - backshifted)
    
    return grad_temp

def projected_gradient_descent_smoothed(h_start, beta_schedule, max_iter, alpha0=0.4, gamma=1e-4):
    """Run projected gradient descent on smoothed max objective"""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    for iter_count, (beta, iters_at_beta) in enumerate(beta_schedule):
        # Compute current correlation values
        corr_vals = compute_corr_fft(h)
        # Compute gradient
        grad = compute_gradient(h, corr_vals, beta)
        alpha = alpha0 / (1.0 + gamma * iter_count)
        
        # Descent
        h_temp = h - alpha * grad
        
        # Project
        h = project_to_sum_box(h_temp, TARGET_SUM)
        
        # Evaluate
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    return best_h, best_c5

def smooth_max_optimize(h_start, n_steps=1000, alpha0=0.4, gamma=1e-4):
    """Optimize using smoothed max with homotopy"""
    # Schedule: slowly increase beta from small to large
    beta_schedule = [(0.5, 200), (1.0, 200), (2.0, 200), (4.0, 150), (8.0, 100)]
    beta_schedule.extend([(16.0, 100), (32.0, 100), (64.0, 100)])

    # Run with different beta values
    h = h_start.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()

    accum_iter = 0
    for beta, iters_for_beta in beta_schedule:
        # Run on current beta for given iterations
        curr_schedule = [(beta, iters_for_beta)]
        h, c5 = projected_gradient_descent_smoothed(h, curr_schedule, iters_for_beta, alpha0, gamma)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        accum_iter += iters_for_beta
    
    return best_h, best_c5

def local_refine(h):
    """Perform a local refinement with SLSQP"""
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    from scipy.optimize import minimize
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, TARGET_SUM)
            return h
    except:
        pass
    return h
