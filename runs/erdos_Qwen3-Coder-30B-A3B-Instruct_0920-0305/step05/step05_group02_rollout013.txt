import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_correlation(h):
    """Compute full correlation array for a binary vector h"""
    n = len(h)
    corr = np.zeros(2 * n - 1)
    for k in range(-n + 1, n):
        if k >= 0:
            corr[k + n - 1] = np.sum(h[:n-k] * h[k:])
        else:
            corr[k + n - 1] = np.sum(h[-k:] * h[:n+k])
    return corr

def compute_c5_from_corr(corr, dx):
    """Convert correlation array to C5 value"""
    return np.max(corr) * dx

def generate_legendre_pattern(n):
    """Generate Legendre-based binary pattern if possible"""
    p = n // 2
    if p < 2 or not is_prime(p):
        return None
    
    residues = set()
    for a in range(1, p):
        residues.add((a * a) % p)
    
    b = np.zeros(p, dtype=np.int8)
    for i in residues:
        if i < p:
            b[i] = 1
    
    h = np.concatenate([b, b])
    if np.sum(h) != p:
        # Adjust to get exactly p ones
        if np.sum(h) > p:
            # Find a 1 to flip to 0
            zeros = np.where(h == 0)[0]
            if len(zeros) > 0:
                h[zeros[0]] = 1
        else:
            # Find a 0 to flip to 1
            ones = np.where(h == 1)[0]
            if len(ones) > 0:
                h[ones[0]] = 0
    
    return h

def is_prime(n):
    """Check if a number is prime"""
    if n < 2:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False
    for i in range(3, int(np.sqrt(n)) + 1, 2):
        if n % i == 0:
            return False
    return True

def generate_random_balanced_mask(n):
    """Generate a random balanced binary mask"""
    h = np.zeros(n, dtype=np.int8)
    indices = np.random.permutation(n)
    h[indices[:n//2]] = 1
    return h

def get_bad_ones(h, k):
    """Find indices where h[i]=1 and h[i+k]=0 (beyond bounds becomes 0)"""
    n = len(h)
    bad_ones = set()
    if k >= 0:
        for i in range(n - k):
            if h[i] == 1 and h[i+k] == 0:
                bad_ones.add(i)
    else:
        for i in range(-k, n):
            if h[i] == 1 and h[i+k] == 0:
                bad_ones.add(i)
    return list(bad_ones)

def update_correlation_incremental(corr, h, i, j, n):
    """Update correlation vector after swapping h[i] and h[j]"""
    old_corr = corr.copy()
    # Buffer the new version in place
    for m in range(-n + 1, n):
        old_v = 0
        new_v = 0
        # Compute the exact change in lag m
        if m >= 0:
            if 0 <= i < n - m and 0 <= i + m < n:
                if h[i] == 1 and h[i + m] == 1:
                    old_v += 1
                if h[i] == 0 and h[i + m] == 0:
                    old_v += 0
            if 0 <= j < n - m and 0 <= j + m < n:
                if h[j] == 1 and h[j + m] == 1:
                    old_v += 1
                if h[j] == 0 and h[j + m] == 0:
                    old_v += 0
            if 0 <= i < n - m and 0 <= i + m < n:
                if h[j] == 1 and h[j + m] == 1:
                    new_v += 1
                if h[j] == 0 and h[j + m] == 0:
                    new_v += 0
            if 0 <= j < n - m and 0 <= j + m < n:
                if h[i] == 1 and h[i + m] == 1:
                    new_v += 1
                if h[i] == 0 and h[i + m] == 0:
                    new_v += 0
        else:
            if 0 <= i + (-m) < n and 0 <= i < n:
                if h[i - m] == 1 and h[i] == 1:
                    old_v += 1
            if 0 <= j + (-m) < n and 0 <= j < n:
                if h[j - m] == 1 and h[j] == 1:
                    old_v += 1
            if 0 <= i + (-m) < n and 0 <= i < n:
                if h[j - m] == 1 and h[j] == 1:
                    new_v += 1
            if 0 <= j + (-m) < n and 0 <= j < n:
                if h[i - m] == 1 and h[i] == 1:
                    new_v += 1
        corr[m + n - 1] += (new_v - old_v)
    return corr

def find_worst_shifts(corr, threshold=1e-6):
    """Find active worst shifts"""
    max_val = np.max(corr)
    return [i - len(corr) // 2 for i in range(len(corr)) if abs(corr[i] - max_val) < threshold]

def smooth_max(x, alpha=30):
    """Smooth approximation to max using log-sum-exp"""
    return (1.0 / alpha) * np.log(np.sum(np.exp(alpha * x)))

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx
