import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

    a = cp.Variable(L)
    b = cp.Variable(L)
    s = cp.Variable(L)
    t = cp.Variable()
    
    # Build the Fourier coefficient matrix for box constraints
    Phi = np.zeros((M, L))
    for i, x in enumerate(x_sample):
        for l in range(L):
            Phi[i, l] = np.cos(np.pi * (l + 1) * x)
    
    # Box constraints (0 <= h(x) <= 1)
    h_expr = 0.5 + Phi @ (a + 1j * b).real
    box_constraints = [
        cp.constraints.Zero(h_expr >= 0),
        cp.constraints.Zero(h_expr <= 1)
    ]
    
    # SOC constraints: s_l >= a_l^2 + b_l^2
    soc_constraints = []
    for l in range(L):
        soc_constraints.append(cp.constraints.SOC(1 + s[l], [2*a[l], 2*b[l], s[l] - 1]))
