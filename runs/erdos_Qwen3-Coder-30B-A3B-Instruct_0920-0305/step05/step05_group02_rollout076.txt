import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Set up constraints for binary ILP case
    best_h = None
    best_c5 = float('inf')
    best_n = n_points
    
    # Use a smaller set of n, e.g., 36 to 64
    n_values = list(range(36, 65, 4))  # Even numbers from 36 to 64
    
    # Try a smaller subset in case full optimization fails quickly
    for n in n_values:
        # Skip if out of time budget
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Create the state vector to be binary
        num_vars = n
        # Create a symmetric solution to reduce search space
        # For now, use a known quick heuristic guess to setup a binary problem of size N
        initial_bina = np.zeros(num_vars)
        # Weighted greedy approach to build solution
        used = set()
        position = 0
        count = num_vars // 2  # want sum to be n/2
        while len(used) < count:
            # try to pick next "best" index (maximally spread and balanced)
            choices = [i for i in range(num_vars) if i not in used]
            
            if not choices:  # no more available
                break
                
            # Score based on minimum resulting p_k (overlap count)
            best_i = choices[0]
            best_min_p = -1
            
            for i in choices:
                # Estimate overlap contribution with current used indices
                temp_used = set(used)
                temp_used.add(i)
                # Compute accidental overlaps near i for all used neighbors
                p_scores = [0] * (num_vars-1)
                for j in temp_used:
                    for k in range(num_vars):  # k shift
                        if 0 <= k + j < num_vars:
                            try:
                                p_scores[k] += 1
                            except IndexError:
                                pass
                
                p_scores = [p for p in p_scores if p > 0]
                min_p = min(p_scores) if p_scores else 0
                if min_p > best_min_p:
                    best_min_p = min_p
                    best_i = i
                    
            initial_bina[best_i] = 1
            used.add(best_i)
            
        # Now try MILP if practical
        x = cp.Variable(n, integer=True)
        # Binary constraints  
        constraints = [
            cp.sum(x) == n // 2,
            x >= 0,
            x <= 1
        ]
        
        # For autocorrelation minimization, define auxiliary variables
        # Define p_k = inner product at shift k
        y = dict()
        p = dict()
        for k in range(1, n):  # k = 1 to n-1
            p[k] = cp.Variable(integer=True)
            # Compute sum of x_i * x_{i+k} for valid i
            expr = 0
            for i in range(n - k):
                # Use trick to Boolean OR relations: y[i,k] = x[i]*x[i+k]
                y[i,k] = cp.Variable(boolean=True)
                constraints += [
                    y[i,k] <= x[i],
                    y[i,k] <= x[i+k],
                    y[i,k] >= x[i] + x[i+k] - 1    
                ]
                expr += y[i,k]
            constraints.append(p[k] == expr)
            
        # Introduce variable to maximize the minimum p_k
        L = cp.Variable(integer=True)
        for k in range(1, n):
            constraints.append(p[k] >= L)
            
        # Setup the MILP Objective:   maximize L
        problem = cp.Problem(cp.Maximize(L), constraints)
        
        # Try solving for the problem controlบรรยากาศ

        #if time.time() - start_time > budget_s * 0.9:
        #    break
            
        try:
            # Solve it
            problem.solve(solver=cp.CBC, verbose=False)
            # Extract result
            solution = x.value
            if solution is not None:
                solution = np.round(solution)
                # Scale
                solution = solution * 1.0
                scaled_solution = solution.copy()
                actual_sum = np.sum(scaled_solution)
                scale_factor = 1.0 / (actual_sum * dx)
                scaled_solution *= scale_factor
                c5_value = evaluate_c5(scaled_solution)
                if c5_value < best_c5:
                    best_c5 = c5_value
                    best_h = scaled_solution.copy()
                    best_n = n
        except:
            # Infeasible or timeout; fallback on initial heuristics
            pass
    
    # Optionally attempt continuous refinement
    if best_h is not None:
        # Set up a gradient descent refinement with convex constraint
        ref_constraints = []
        dx_ref = 2.0 / best_n
        
        # Between solutions, try gradient descent
        for i in range(500):
            if time.time() - start_time > budget_s * 0.99:
                break
            try:
                # Just evaluate a small change to see if C5 lowers
                perturb = np.random.normal(0, 0.01, len(best_h))
                new_h = best_h + perturb
                new_h = np.clip(new_h, 0, 1)
                scale_factor_new = 1.0 / (np.sum(new_h) * dx_ref)
                new_h = new_h * scale_factor_new
                new_c5 = evaluate_c5(new_h)
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
            except:
                pass

    # Final use of starting guess
    if best_h is None:
        h0 = np.array(initial_h_values)
        start_c5 = evaluate_c5(h0)
        best_h = h0.copy()
        best_c5 = start_c5
        best_n = len(h0)

    return (best_h.tolist(), best_c5, best_n)
