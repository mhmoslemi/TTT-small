import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time
from scipy.fft import rfft, irfft

def repair_h(control_values, n_points, target_sum):
    """
    Interpolate control points to full grid, and project onto capped simplex.
    """
    # Control points (K=12, placed at equally spaced positions in [0,2])
    K = len(control_values)
    x_knots = np.linspace(0, 2, K)
    h_full = np.interp(np.linspace(0, 2, n_points), x_knots, control_values)
    
    # Clip to [0,1]
    h_full = np.clip(h_full, 0, 1)
    
    # Project onto capped simplex sum = target_sum
    def project_simplex(h, target_sum):
        # Use binary search for the Lagrange multiplier
        lo, hi = -100, 100
        epsilon = 1e-12
        for _ in range(60):  # Max iterations
            lam = (lo + hi) / 2.0
            z = np.clip(h - lam, 0, 1)
            s = np.sum(z)
            if s < target_sum:
                hi = lam
            else:
                lo = lam
        z = np.clip(h - lo, 0, 1)
        return z
    
    # Project to the correct sum
    h_repaired = project_simplex(h_full, target_sum)
    return h_repaired

def compute_correlation(h):
    """Compute optimal correlation via FFT."""
    n = len(h)
    dx = 2.0 / n
    # Compute O_k = dx * sum_{i} h[i] * (1 - h[i+k])
    # Correlation convolves h with (1 - h)
    h1 = h.copy()
    one_minus_h = 1.0 - h1

    # Use FFT convolution
    H = rfft(h1)
    G = rfft(one_minus_h)
    corr = irfft(H * np.conj(G))
    # Take first 2*n-1 values (full convolution)
    corr = corr[:2*n-1]
    return corr * dx

def objective_with_smoothing(control_values, n_points, alpha):
    """Softmax approximation to max overlap."""
    h = repair_h(control_values, n_points, n_points / 2.0)
    corr = compute_correlation(h)
    max_corr = np.max(corr)
    
    # Compute log-sum-exp surrogate O_alpha(h) = (1/alpha) log(sum(exp(alpha * O_k)))
    if alpha <= 1e-10:
        return max_corr
    # Instead of adding the max to prevent overflow:
    O_k = corr
    max_O = np.max(O_k)
    exp_terms = np.exp(alpha * (O_k - max_O))
    log_sum_exp = max_O + (1.0 / alpha) * np.log(np.sum(exp_terms))
    return log_sum_exp

def objective_exact(control_values, n_points):
    """Exact objective for the last polishing phase."""
    h = repair_h(control_values, n_points, n_points / 2.0)
    corr = compute_correlation(h)
    max_corr = np.max(corr)
    return max_corr

def projected_gradient_descent(initial_h, n_points, max_iter=500, lr=0.05):
    """Apply projected sub-gradient descent on the exact max."""
    h = initial_h.copy()
    dx = 2.0 / n_points
    
    for t in range(max_iter):
        # Compute all correlations
        corr = compute_correlation(h)
        max_val = np.max(corr)
        max_indices = np.where(np.abs(corr - max_val) < 1e-10)[0]
        g = np.zeros(n_points)
        # Compute average gradient for all active shifts
        for k in max_indices:
            if k >= n_points:
                # wrap around
                g = g + (1 - np.roll(h, k)) - np.roll(h, -k)
            elif k <= 0:
                # k < 0, i.e., h[i+k] for i+k < 0 is h[i+k+n_points], so we shift 
                g = g + (1 - np.roll(h, k)) - np.roll(h, -k)
            else:
                g = g + (1 - np.roll(h, k)) - np.roll(h, -k)
        g = g / len(max_indices)
        # Gradient step
        h_new = h - lr * g
        
        # Project back
        h_new = np.clip(h_new, 0, 1)
        
        # Project onto sum constraint (standard simplex projection)
        def project_simplex(h, target_sum):
            lo, hi = -100, 100
            epsilon = 1e-12
            for _ in range(60):
                lam = (lo + hi) / 2.
                z = np.clip(h - lam, 0, 1)
                s = np.sum(z)
                if s < target_sum:
                    hi = lam
                else:
                    lo = lam
            z = np.clip(h - lo, 0, 1)
            return z
        h_new = project_simplex(h_new, n_points / 2)
        h = h_new
        
        # Check for convergence
        corr_new = compute_correlation(h)
        max_new = np.max(corr_new)
        if abs(max_new - max_val) < 1e-7:
            break

    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 128
    K = 12
    target_sum = n_points / 2.0
    
    # Multi-start strategy with different initializations
    restarts = 10
    best_c5 = float('inf')
    best_h = None
    
    for restart in range(restarts):
        # Generate a random initialisation
        control_init = np.random.uniform(0, 1, K)
        
        # CMA-ES-like update loop with a simple evolutionary approach
        # We simulate a few generations by running a few random search steps
        lambda_size = 4 + 3 * int(np.log(K))
        
        # For simplicity, we instead do a local search with a guided generator
        current_best_c5 = float('inf')
        current_best_control = control_init
        best_step = False  # Whether we found an improvement since last iteration
        iteration = 0
        max_iter = 1000
        
        # Simulate a few random samples, and try projections
        current_alpha = 5.0
        alpha_factor = (30 - 5) / 0.7 / (max_iter * 0.7)  # anneal to 30 over 70% of steps
        
        while iteration < max_iter and (time.time() - start_time) < budget_s * 0.99:
            # Try around 20 random points
            for _ in range(20):
                # Random perturbation
                old_control = current_best_control.copy()
                new_values = np.random.normal(old_control, 0.2)
                
                # Clip new control values to [0,1]
                new_values = np.clip(new_values, 0, 1)
                
                # Project to the simplex (this is important!)
                def project_simplex(h, target_sum):
                    lo, hi = -100, 100
                    for _ in range(60):
                        lam = (lo + hi) / 2.
                        z = np.clip(h - lam, 0, 1)
                        s = np.sum(z)
                        if s < target_sum:
                            hi = lam
                        else:
                            lo = lam
                    z = np.clip(h - lo, 0, 1)
                    return z

                # Sample a set of candidates, project, evaluate
                trial_control = new_values
                trial_h = repair_h(trial_control, n_points, target_sum)
                corr = compute_correlation(trial_h)
                corr_max = np.max(corr)
                if corr_max < current_best_c5:
                    current_best_c5 = corr_max
                    current_best_control = trial_control.copy()
                    best_step = True

            # Update the smoothness parameter if needed
            current_alpha += alpha_factor
            current_alpha = min(current_alpha, 30)
            
            iteration += 1

            # Early stopping if too long without improvement
            if not best_step:
                break
            else:
                best_step = False

        # Final polishing with projected gradient descent
        h_final = repair_h(current_best_control, n_points, target_sum)

        for _ in range(200):
            try:
                corr = compute_correlation(h_final)

                max_indices = np.where(np.abs(corr - max_val) < 1e-10)[0]

                        g = g + (1 - np.roll(h_final, k)) - np.roll(h_final, -k)

                        g = g + (1 - np.roll(h_final, k)) - np.roll(h_final, -k)

                h_new = project_simplex(h_new, target_sum)
                corr_new = compute_correlation(h_new)

            except:

        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = h_final
            
    # Convert to list before returning
    h_final = best_h
    if h_final is None:
        h_final = np.array([1] * 32 + [0] * 96)  # fallback
        h_final = h_final / np.sum(h_final) * target_sum  # cleanup
    
    return (h_final.tolist(), best_c5, n_points)
