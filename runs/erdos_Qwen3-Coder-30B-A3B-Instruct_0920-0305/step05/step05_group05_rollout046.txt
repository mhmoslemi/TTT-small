import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # FFT-based, convex relaxation approach
    def evaluate_c5_convex(h):
        """Fast evaluation of C5 after rescaling to integral=1"""
        dx = 2.0 / len(h)
        corr = correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Parameters for Fourier approach
    max_order = 15  # Start with reasonable truncation
    sample_density = 4  # M = sample_density * L
    
    # Try different orders to improve bound
    best_h = h0.copy()
    best_c5 = evaluate_c5_convex(h0)
    
    # Use standard sampling points for feasibility (better than random)
    max_eval_time = budget_s * 0.95
    current_order = min(max_order, 20)  # Limit for resource usage
    
    # Try optimization with increasing order
    attempts = 0
    while attempts < 10 and (time.time() - start_time) < max_eval_time:
        L = current_order
        M = sample_density * L
        
        # Ensure sampling grid is fine enough for box constraints
        x_sample = np.linspace(0, 2, M, endpoint=False)
        # Equation (3) of strategy: make sure enough sampling points to cover features
        
        # Optimize using convex relaxation:
        # Decision variables
        u = cp.Variable(2*L)
        s = cp.Variable(L, nonneg=True)  # Squared magnitudes
        t = cp.Variable()  # Upper bound to minimize
        
        # Linear constraints: 0 <= h(x_j) <= 1
        A_box = np.zeros((M, 2*L))
        for i, x in enumerate(x_sample):
            for l in range(1, L+1):
                A_box[i, l-1] = np.cos(np.pi * l * x)    # a_l coeffs
                A_box[i, L+l-1] = np.sin(np.pi * l * x)  # b_l coeffs
        
        # Build h(x) = 0.5 + sum a_l*cos + b_l*sin
        h_vals = 0.5 + A_box @ u  # Vector: M elements
        
        # Box constraints from the Fourier description
        box_constraints = [
            cp.constraints.Zero(h_vals - 1),  # 0 <= h <= 1 becomes h-1 <= 0, h >= 0
            cp.constraints.Zero(1 - h_vals)   # This is equivalent to h <= 1
        ]
        # Better approach for mixed bound constraints:
        # We solve element-wise:
        lower_constraints = [h_vals[i] >= 0 for i in range(M)]
        upper_constraints = [h_vals[i] <= 1 for i in range(M)]
        box_constraints = lower_constraints + upper_constraints
        
        # Quadratic constraints (SOC): s_l >= a_l^2 + b_l^2
        soc_constraints = []
        for l in range(L):
            a_l = u[l]
            b_l = u[L+l]
            # 2nd order cone
            soc_expr = cp.vstack([2*a_l, 2*b_l, s[l] - 1])
            soc_constraints.append(cp.norm(soc_expr, 2) <= s[l] + 1)
        
        # Construct Toeplitz matrix for max bound
        # T = [t, -s_1, ..., -s_L ; -s_1, t, ..., ; ...]
        # We constrain Toeplitz matrix to be PSD
        
        # Cvxpy requires building the matrix entry-wise
        # Since small matrices (L+1 by L+1), make it efficiently
        # Eq. (3) of Strategy: Loops below will be the Toeplitz construction manually
        
        # Constraint on Toeplitz matrix being PSD is tricky to express directly
        # But we use the fact that the maximum of the polynomial is exactly
        # the smallest t such that the Toeplitz matrix T(t,s) is psd
        
        # Direct approach with explicit Toeplitz matrix
        # This formulizes the arithmetic:
        #             [-s_1, ..., -s_L  ]
        # Toeplitz = [ -s_1 , -s_2 , ..., -s_{L-1} ]  
        #            [ ...           , -s_L      ]  
        # The Toeplitz as a matrix in cvxpy:
        
        # Let's build it more easily as via element-wise comprehension
        # Build Toeplitz with top row and first column
        n_toeplitz = L + 1
        T = cp.Variable((n_toeplitz, n_toeplitz), symmetric=True)
        
        # Fix top (and first) elements  
        T[0,0] = t
        for j in range(1, L+1):
            T[0, j] = -s[j-1]
            T[j, 0] = -s[j-1]
            
        # Other entries are fixed by Toeplitz structure
        for i in range(1, n_toeplitz):
            for j in range(1, n_toeplitz):
                if abs(i-j) <= L:
                    T[i, j] = -s[abs(i-j)]
                    
        # Add constraints: Toeplitz is PSD and other matrix equalities
        constraints = [
            T >> 0,                 # PSD: T >= 0
            cp.constraints.Zero(s - cp.multiply(u[:L], u[:L]) - cp.multiply(u[L:], u[L:])),  # s = a^2 + b^2
        ] + box_constraints + soc_constraints
        
        # Build objective
        prob = cp.Problem(cp.Minimize(t), constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=1000, eps=1e-6)
            if prob.status == "optimal":
                # Extract optimal solution
                u_value = u.value
                s_value = s.value
                t_value = t.value
                
                # Reconstruct h on full grid using Fourier coefficients
                # Reconstruct Impulse impulse function with same amplitudes for each point
                h_reconstructed = np.zeros(n_points)
                dx_recon = 2.0 / n_points
                
                # Here we just evaluate the optimal h at the full grid
                # To do so we need tosätze the fourier coefficients and then the fourier series reconstruction
                a = u_value[:L]
                b = u_value[L:]
                h_full = 0.5 + np.array([sum(a[j]*np.cos(np.pi*(j+1)*x) + b[j]*np.sin(np.pi*(j+1)*x)) 
                                          for x in np.linspace(0, 2, n_points, endpoint=False)])
                
                # Clip values to [0,1]
                h_full = np.clip(h_full, 0, 1)
                # Renormalize
                integral = np.sum(h_full) * dx_recon
                h_reconstructed = h_full / integral
                
                c5_final = evaluate_c5_convex(h_reconstructed)
                
                # If we find a better solution, store it
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_reconstructed.copy()
                
                # Small improvement by using local refinement around the solution
                if attempts < 5:
                    # Perturb slightly and run local minimization
                    h_perturbed = h_reconstructed + 0.01*(np.random.rand(n_points) - 0.5)
                    h_perturbed = np.clip(h_perturbed, 0, 1)
                    integral_perturbed = np.sum(h_perturbed) * dx_recon
                    h_perturbed = h_perturbed / integral_perturbed
                    
                    c5_perturbed = evaluate_c5_convex(h_perturbed)
                    if c5_perturbed < best_c5:
                        best_c5 = c5_perturbed
                        best_h = h_perturbed.copy()
            
        except Exception as e:
            pass
        
        attempts += 1
        current_order = min(current_order + 1, 20)  # Increase order step by step if early iterations
        # print(f"Attempts: {attempts}, C5: {best_c5}, Order: {current_order}")  # DEBUG
        
        # Stop early if timeout
        if (time.time() - start_time) > max_eval_time:
            break
    
    # Final refinement using simplicity (just final evaluation)
    # This is a low-cost final check
    if best_c5 > 0.3807:

    return (best_h.tolist(), best_c5, n_points)
