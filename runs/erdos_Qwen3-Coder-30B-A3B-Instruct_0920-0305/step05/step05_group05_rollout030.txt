import numpy as np
import time

def project_onto_box_simplex(v, target_sum):
    """Project vector v onto box [0,1] and simplex with sum = target_sum"""
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0, 1)

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Use correlation to compute overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, alpha):
    """Compute gradient of smoothed max of overlap integrals"""
    n = len(h)
    dx = 2.0 / n
    
    # Compute correlation (overlap integral for all shifts)
    corr = np.correlate(h, 1 - h, mode="full")
    O = corr * dx
    
    # Numerically stable computation of softmax weights
    O_max = np.max(O)
    exp_term = np.exp(alpha * (O - O_max))
    w = exp_term / np.sum(exp_term)
    
    # Compute gradient
    grad = np.zeros_like(h)
    
    # Loop over all shifts
    for idx, shift in enumerate(range(-n + 1, n)):
        w_k = w[idx]
        if w_k == 0:
            continue
            
        # Compute shift_right and shift_left vectors
        if shift >= 0:
            # shift_right: 1 for first shift entries, then 1 - h[shift:]
            shift_right = np.empty(n)
            shift_right[:shift] = 1.0
            shift_right[shift:] = 1.0 - h[shift:]
            
            # shift_left: zeros for first shift entries, then h[:n-shift]
            shift_left = np.empty(n)
            shift_left[:shift] = 0.0
            shift_left[shift:] = h[:n - shift]
        else:
            s = -shift
            # shift_right: 1 for first s entries, then 1 - h[:n-s]
            shift_right = np.empty(n)
            shift_right[:s] = 1.0
            shift_right[s:] = 1.0 - h[:n - s]
            
            # shift_left: h[s:] for first n-s entries, then 0
            shift_left = np.empty(n)
            shift_left[:n - s] = h[s:]
            shift_left[n - s:] = 0.0
        
        grad += w_k * (shift_right - shift_left)
    
    return grad * dx

def optimize_h(n, max_iters, init_lr=0.1, alpha_start=5, alpha_end=30):
    """Optimize h using gradient descent with projected updates"""
    # Initialize random h
    h = np.random.rand(n)
    # Project to satisfy constraint sum = n/2
    h = project_onto_box_simplex(h, target_sum=n/2)
    
    best_c5 = np.inf
    best_h = None
    
    for it in range(max_iters):
        # Compute current overlap values
        conv = np.correlate(h, 1 - h, mode="full")
        current_c5 = np.max(conv) * (2.0 / n)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        
        # Compute gradient
        grad = compute_gradient(h, alpha)
        
        # Update step size
        lr = init_lr / np.sqrt(it + 1)
        
        # Update h
        h = h - lr * grad
        
        # Project onto feasible region
        h = project_onto_box_simplex(h, target_sum=n/2)
        
        # Early stopping
        if best_c5 < 0.379:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up optimization parameters
    n = 64
    target_sum = n / 2
    dx = 2.0 / n
    
    # Initialize best solution
    best_c5 = np.inf
    best_h = None
    
    # Run multiple restarts
    restart_count = 0
    while (time.time() - start_time) < budget_s * 0.95 and restart_count < 100:
        # Perform optimization for this restart
        current_h, current_c5 = optimize_h(
            n=n,
            max_iters=2000,
            init_lr=0.1,
            alpha_start=5,
            alpha_end=30
        )
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
        
        restart_count += 1
        
        # Break early if we've found a very good solution
        if best_c5 < 0.379:
            break
    
    # Final check with optimized result using more iterations
    if best_h is not None:
        # Perform final local refinement
        final_h, final_c5 = optimize_h(
            n=n,
            max_iters=1500,
            init_lr=0.05,
            alpha_start=30,
            alpha_end=50
        )
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h.copy()
    
    # Final verification
    if best_h is not None:
        last_eval = evaluate_c5(best_h)
        if last_eval < best_c5:
            best_c5 = last_eval
        
        return (best_h.tolist(), float(best_c5), n)
    else:
        # Return default if no optimization was successful
        h_default = np.ones(n) * 0.5
        c5_default = evaluate_c5(h_default)
        return (h_default.tolist(), float(c5_default), n)
