import numpy as np
from cvxpy import *
import time
from scipy.signal import correlate
import math

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def build_and_solve_milp(n, time_budget_s):
    """solve integer program to compute maximum min p_k for 0-1 x vector."""
    # Used mathematica for inspection of solutions:
    # The ground truth for 42 points is listed via initial_h_values.
    # we should benchmark from naive greedy construction first.
    try:
        # use cvxpy with integer support (requires MILP)
        from cvxpy import Variable, Minimize, Problem, Bool, sum, vstack, hstack
        # variables
        x = Variable(n, boolean=True)
        y = {}
        for k in range(1,n):
            for i in range(n-k):
                y[(i,k)] = Variable(boolean=True)
        
        # constraints on x
        constraint0 = sum(x) == n/2
        # y constraints: McCormick for binary case
        constraints = [constraint0]
        # note: do not add all, just compute for all valid (i,k) pairs
        for k in range(1,n):
            for i in range(n-k):
                constraints.append(y[(i,k)] <= x[i])
                constraints.append(y[(i,k)] <= x[i+k])
                constraints.append(y[(i,k)] >= x[i]+x[i+k]-1)
        
        # constraints to define p_k: sum over valid y_{i,k}
        p = {}
        for k in range(1,n):
            p[k] = sum(y[(i,k)] for i in range(n-k))
        
        # minimize max_k of p_k, i.e. maximize min_k of p_k
        L = Variable(boolean=True)
        for k in range(1,n):
            constraints.append(p[k] >= L)
            
        # symmetry breaking constraints:
        # x[0] = 1, and L-H heavy (to prevent symmetric flips)
        constraints.append(x[0] == 1)
        constraints.append(sum(x[:n//2]) >= sum(x[n//2:]))
        
        # obj to maximize L
        prob = Problem(Maximize(L), constraints)
        
        # we want to compile solution within a fixed time budget
        start_time = time.time()
        # solve with integer support: converted to suitable solver.
        prob.solve(solver=SCIP, verbose=False, time_limit=time_budget_s*0.8)
        
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            # extract solution
            sol_x = [int(round(xi.value)) for xi in x]
            sol_L = int(round(L.value))
            return sol_x, sol_L
        else:
            return None, 0
            
    except Exception as e:
        return None, 0

def greedy_initialization(n):
    """initialize x_i in {0,1} such that ∑x_i = n/2, and min p_k is large."""
    x = np.zeros(n, dtype=int)
    s = set()
    while len(s) < n//2:
        best_i = -1
        best_min_p = -1e9
        for i in range(n):
            if i in s:
                continue
            new_s = set(s) | {i}
            new_p = np.zeros(n, dtype=int)
            for k in range(1, n):
                count = 0
                for j in range(n - k):
                    if j in new_s and j + k in new_s:
                        count += 1
                new_p[k] = count
            min_p = np.min(new_p[1:])  # don't care about p[0]
            if min_p > best_min_p:
                best_min_p = min_p
                best_i = i
        if best_i >= 0:
            s.add(best_i)
    for i in s:
        x[i] = 1
    return x.tolist()
