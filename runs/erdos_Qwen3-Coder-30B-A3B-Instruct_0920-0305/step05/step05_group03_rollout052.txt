```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Try different optimization strategies
    # Strategy 1: Differential Evolution (global search)
    try:
        de_res = differential_evolution(
            objective, 
            bounds, 
            maxiter=1000, 
            popsize=15, 
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            callback=lambda x, convergence: None
        )
        if de_res.success:
            h_candidate = de_res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass
    
    # Strategy 2: Local search from multiple starting points
    local_restart_bound = 500
    
    for restart in range(local_restart_bound):
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
        # Random perturbation around best solution
        new_h = best_h + np.random.normal(0, 0.015, n_points)
        new_h = np.clip(new_h, 0, 1)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue
    
    # Strategy 3: Try some more structured constructions
    if (time.time() - start_time) < budget_s * 0.8:
        # Try different initial patterns
        patterns = []
        
        # Pattern 1: Piecewise constant with few levels
        h_piecewise = np.ones(n_points)
        h_piecewise[:n_points//3] = 0.5
        h_piecewise[n_points//3:2*n_points//3] = 1.5
        h_piecewise[2*n_points//3:] = 0.5
        patterns.append(h_piecewise)
        
        # Pattern 2: Oscillating pattern
        x = np.linspace(0, 2, n_points)
        h_osc = 0.5 + 0.5 * np.sin(2 * np.pi * x)
        patterns.append(h_osc)
        
        # Pattern 3: Gaussian-like pattern with proper normalization
        h_gauss = np.exp(-((x-1)**2) / 0.2)
        patterns.append(h_gauss)
        
        for pattern in patterns:
            if (time.time() - start_time) > budget_s * 0.95:
                break
            try:
                # Normalize
                dx = 2.0 / n_points
                scale_factor = 1.0 / (np.sum(pattern) * dx)
                pattern_norm = pattern * scale_factor
                
                # Optimize this pattern
                res = minimize(
                    objective, 
                    pattern_norm, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=cons,
                    options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-14}
                )
                if res.success:
                    h_candidate = res.x
                    scale = 1.0 / (np.sum(h_candidate) * dx)
                    h_candidate = h_candidate * scale
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
            except:
                continue
    
    # Final refinement with trust-constr if time allows
    if (time.time() - start_time) < budget_s * 0.9:
        try:
            res = minimize(
                objective, 
                best_h, 
                method='trust-constr', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'gtol': 1e-15, 'ftol': 1e-15}
            )
            if res.success:
                h_candidate = res.x
                dx = 2.0 / n_points
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass

    return (best_h.tolist(), best_c5, n_points)
```
