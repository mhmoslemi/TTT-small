import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.irfft(np.fft.rfft(h_rev, 2*n - 1) * np.fft.rfft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def perturb_solution(h, perturb_mag=0.05):
    """Create a perturbed version of the solution"""
    h_perturbed = h + np.random.normal(0, perturb_mag, len(h))
    return project_to_sum_box(h_perturbed, TARGET_SUM)

def local_optimize(h0, max_iter=1000, tol=1e-12):
    """Refine solution using L-BFGS-B optimizer"""
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h0))]
    
    h_current = h0.copy()
    best_h = h_current.copy()
    best_c5 = evaluate_c5(h_current)
    
    for _ in range(max_iter):
        try:
            res = minimize(
                objective,
                h_current,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': tol, 'gtol': tol}
            )
            if res.success:
                h_refined = res.x
                h_refined = project_to_sum_box(h_refined, TARGET_SUM)
                c5_refined = evaluate_c5(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                h_current = h_refined
            else:
                break
        except:
            break
    
    return best_h, best_c5
