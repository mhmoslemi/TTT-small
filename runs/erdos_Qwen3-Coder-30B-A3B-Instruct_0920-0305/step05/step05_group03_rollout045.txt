```python
import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from cvxpy import *
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = len(initial_h_values)
    dx = 2.0 / n
    h0 = np.array(initial_h_values)
    
    def solve_milp(n, dx):
        h = Variable(n, boolean=True)
        tau = Variable()
        constraints = [sum(h) == n/2, h >= 0, h <= 1]
        # Build shift pairs and overlaps
        shifts = list(range(-(n-1), n))
        all_y = {}
        
        # For each shift k, get valid (i,j) pairs
        for k in shifts:
            pairs = []
            for i in range(n):
                j = i + k
                if 0 <= j < n:
                    pairs.append((i,j))
            if pairs:
                y_vars = [Variable(boolean=True) for _ in pairs]
                all_y[k] = y_vars
                for idx, (i, j) in enumerate(pairs):
                    y = y_vars[idx]
                    # McCormick constraints for y = h[i] * h[j]
                    constraints.append(y <= h[i])
                    constraints.append(y <= h[j])
                    constraints.append(y >= h[i] + h[j] - 1)
                # Overlap constraint O_k = sum h - sum y <= tau
                overlap = sum(h) - sum(y_vars)
                constraints.append(overlap <= tau)
        
        objective = Minimize(tau)
        problem = Problem(objective, constraints)
        try:
            problem.solve(solver='CBC', verbose=False, max_seconds=budget_s*0.24, #close to 150s
                         mip_gap=1e-6)
            if problem.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_sol = h.value
                return h_sol, problem.value
            else:
                # Fallback to best feasible
                return h.value, None
        except:
            return h0, None
    
    def simple_hill_climb(h_start, n, dx):
        h_current = h_start.copy()
        best_h = h_current.copy()
        best_c5 = evaluate_c5(h_current)
        max_cycles = 3000
        for cycle in range(max_cycles):
            if time.time() - start_time > budget_s * 0.45:
                break
            improved = False
            # Explore all swaps
            for i in range(n):
                if h_current[i] != 0:  # Looking for a 1 to swap out
                    for j in range(n):
                        if h_current[j] != 1:  # Looking for a 0 to swap in
                            # Make swap
                            h_new = h_current.copy()
                            h_new[i] = 0
                            h_new[j] = 1
                            c5_new = evaluate_c5(h_new)
                            if c5_new < best_c5:
                                best_h = h_new.copy()
                                best_c5 = c5_new
                                h_current = h_new.copy()
                                improved = True
            if not improved:
                break
        return best_h, best_c5
    
    def project_simplex(v, target_sum):
        """Project vector v onto the capped simplex with sum = target_sum"""
        # Argmin ||x - v||^2 subject to 0 <= x_i <= 1, sum x_i = target_sum
        
        # Simple binary search to find lambda such that
        # sum(clamp(v + lambda, 0, 1)) = target_sum
        
        def get_sum(lam):
            p = np.clip(v + lam, 0, 1)
            return np.sum(p)
            
        low, high = -100, 100
        for _ in range(60):
            mid = (low + high) / 2
            s = get_sum(mid)
            if s < target_sum:
                high = mid
            else:
                low = mid
                
        lambda_final = (low + high) / 2
        proj = np.clip(v + lambda_final, 0, 1)
        return proj

    def continuous_refinement(h_start, max_iter=500, min_eta=1e-5):
        h = h_start.copy()
        current_c5 = evaluate_c5(h)
        best_h = h.copy()
        best_c5 = current_c5
        eta = 0.5 * dx
        
        for it in range(max_iter):
            if time.time() - start_time > budget_s * 0.85:
                break
                
            # Compute overlap values
            o_all = correlate(h, 1 - h, mode='full')
            max_o = np.max(o_all)
            active_shifts = np.where(o_all == max_o)[0]
            # Map to k shifts (relative indices)
            k_shifts = active_shifts - len(h) + 1
            
            # Compute gradient (approximated with center of active shifts)
            # Use x axis weighted aggregation
            weights = np.zeros(len(h))
            for k in k_shifts:
                # Positive part over all valid indices
                if k >= 0:
                    weights[k:] += 1
                else:
                    weights[:k] += 1
                    
            # full weight vector (before convolution by kernel of all ones)
            w = weights.copy()
            w[w == 0] = 0
            ones = np.ones(len(h))
            w_convo = np.convolve(w, ones, mode='same')
            # Computed as ∂_h_i = dx * (1 - (w_convo)_i - (w_convo)_rev_i) 
            # But we do not include the convolution as it's too slow; instead approximate
            
            # Approximate: compute outer products sum (along shifting) = w's shifted sum
            # Apply conv and compute as in standard formula
            
            # Better approach: use subgradient directly
            # Simplified form: stationarity condition of max 
            # Near-sure to be at, say, shifts k=0, ±2,...
            # Here we binarize minimal changes per shift
            # Just evaluate all positions, and mark action
            
            # Rough gradient: for each i, contribution to O_k's gradient 
            # based on how many active shifts involve pattern i and i+k
            
            # Simpler approach: add to fixed weights as done for basic gradient
            # via full convolution
            
            # effectively linearize and take gradient
            
            g = np.zeros(len(h))
            
            # Building gradient manually via shift sum
            for k in k_shifts:
                if k >= 0:
                    for i in range(len(h) - k):
                        g[i] += 1 - dx  # contribution from h[i+k] term
                        g[i+k] += 1 - dx # from h[i] term
                else:
                    k_abs = -k
                    for i in range(len(h) - k_abs):
                        g[i] += 1 - dx
                        g[i+k_abs] += 1 - dx
                        
            # Do local gradient update with linear weight term(s) and projection
            h_new = h - eta * g
            h_new = project_simplex(h_new, n/2)
            
            # Try symmetrization and project
            h_sym = (h_new + h_new[::-1]) / 2.0
            h_sym = project_simplex(h_sym, n/2)
            
            c5_new = evaluate_c5(h_sym)
            if c5_new < best_c5:
                best_h = h_sym.copy()
                best_c5 = c5_new
                h = h_sym.copy()
                # Allow larger steps if something improves
                if eta < 0.1 * dx:
                    eta *= 1.1
            else:
                if eta > min_eta:
                    eta = max(et * 0.9, min_eta)
                h = h_new.copy()
        
        return best_h, best_c5

    # Phase 1: MILP (binary optimization)
    h_best = h0.copy()
    c5_best = evaluate_c5(h0)
    
    try:
        h_binary, tau = solve_milp(n, dx)
        if h_binary is not None and tau is not None:
            c5_milp = tau * dx
            h_binary = np.clip(h_binary, 0, 1)
            c5_milp = evaluate_c5(h_binary)
            if c5_milp < c5_best:
                h_best = h_binary.copy()
                c5_best = c5_milp
    except:
        pass

    # Phase 2: hill climbing on binary
    try:
        h_hill, c5_hill = simple_hill_climb(h_best, n, dx)
        if c5_hill < c5_best:
            h_best = h_hill.copy()
            c5_best = c5_hill
    except:
        pass

    # Phase 3: continuous refinement
    try:
        h_cont, c5_cont = continuous_refinement(h_best)
        if c5_cont < c5_best:
            h_best = h_cont.copy()
            c5_best = c5_cont
    except:
        pass

    # Phase 4: optional final symmetrisation
    try:
        h_sym = (h_best + h_best[::-1]) / 2.0
        h_sym = project_simplex(h_sym, n/2)
        c5_sym = evaluate_c5(h_sym)
        if c5_sym < c5_best:
            h_best = h_sym.copy()
            c5_best = c5_sym
    except:
        pass

    # Phase 5: optional upscale
    try:
        # Only if time allows
        if time.time() - start_time < budget_s * 0.9:
            n_up = 2 * n
            dx_up = 2.0 / n_up
            h_up = np.interp(np.linspace(0, 2, n_up), np.linspace(0, 2, n), h_best)
            # Normalise area
            scale = (n/2) / (np.sum(h_up) * dx_up)
            h_up *= scale
            h_up_cont, c5_up = continuous_refinement(h_up, 400)
            if c5_up < c5_best:
                h_best = h_up_cont.copy()
                c5_best = c5_up
    except:
        pass

    return (h_best.tolist(), c5_best, n)
```
