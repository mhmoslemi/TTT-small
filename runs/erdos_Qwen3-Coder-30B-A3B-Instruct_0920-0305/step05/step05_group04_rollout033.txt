import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def project_simplex(h, target_sum):
    """Project h onto the simplex with sum = target_sum and bounds [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    # Use bisection to find lambda
    l, u = -1.0, 1.0
    for _ in range(50):
        lam = (l + u) / 2
        h_trial = np.clip(h + lam, 0, 1)
        if np.sum(h_trial) > target_sum:
            u = lam
        else:
            l = lam
    return np.clip(h + l, 0, 1)

def compute_correlations(h):
    """Compute correlations c_k = sum h[i] * h[i+k] for k=1..n-1"""
    n = len(h)
    # pad with zeros to avoid wraparound
    h_padded = np.pad(h, (0, n-1), 'constant', constant_values=0)
    # use FFT-based convolution for efficiency (linear convolution)
    corr = np.convolve(h_padded, h[::-1], mode='valid')[:n-1]
    return corr

def compute_gradient(h, c, beta):
    """Compute gradient using FFT-based correlations"""
    n = len(h)
    w = np.exp(-beta * c)
    w_sum = np.sum(w)
    if w_sum == 0:
        w = np.ones_like(w)
        w_sum = float(len(w))
    w = w / w_sum
    
    # compute two convolutions
    w_padded = np.pad(w, (0, n-1), 'constant')
    h_padded = np.pad(h, (0, n-1), 'constant')
    conv1 = np.convolve(h_padded, w_padded, mode='valid')[:n]
    conv2 = np.convolve(h_padded[::-1], w_padded[::-1], mode='valid')[:n]
    g = -(conv1 + conv2)
    return g[:n]

def smooth_max(h, beta):
    """Compute smooth approximation to negative min of correlations"""
    c = compute_correlations(h)
    max_c = np.max(c)
    c_shift = c - max_c
    exp_term = np.exp(-beta * c_shift)
    log_sum = np.log(np.sum(exp_term)) + max_c * beta
    return log_sum / beta

def compute_true_c5(h):

    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")[n-1:]
    return np.max(corr) * dx
