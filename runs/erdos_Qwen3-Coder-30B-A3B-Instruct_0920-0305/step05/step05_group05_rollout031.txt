import time
import numpy as np

def project_to_box_simplex(v, target_sum):
    """Project vector v onto box [0,1] and simplex sum=target_sum"""
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0, 1)

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, alpha, dx):
    """Compute gradient of smoothed maximum overlap"""
    n = len(h)
    conv = np.correlate(h, 1 - h, mode="full")
    O = conv * dx
    O_max = np.max(O)
    
    # Stable computation of weights
    exp_term = np.exp(alpha * (O - O_max))
    w = exp_term / np.sum(exp_term)
    
    grad = np.zeros_like(h)
    
    # Loop through all shifts
    for idx, shift in enumerate(range(-n + 1, n)):
        w_k = w[idx]
        if w_k == 0:
            continue
            
        # Compute shift_right and shift_left vectors
        if shift >= 0:
            # shift_right: h_{i+shift} for valid i, else 1
            sr = np.empty(n)
            sr[:shift] = 1.0
            sr[shift:] = 1.0 - h[shift:]
            
            # shift_left: h_{i-shift} for valid i, else 0
            sl = np.empty(n)
            sl[:shift] = 0.0
            sl[shift:] = h[:n-shift]
        else:
            s = -shift
            # shift_right: h_{i-s} for valid i, else 1
            sr = np.empty(n)
            sr[:s] = 1.0
            sr[s:] = 1.0 - h[:n-s]
            
            # shift_left: h_{i+s} for valid i, else 0
            sl = np.empty(n)
            sl[:n-s] = h[s:]
            sl[n-s:] = 0.0
            
        grad += w_k * (sr - sl)
    
    return grad * dx

def optimize_h(n, max_iters=3000, alpha_start=5.0, alpha_end=30.0):
    """Optimize h using gradient descent"""
    # Random initialization
    h = np.random.rand(n)
    # Project to feasible set
    h = project_to_box_simplex(h, n / 2)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for it in range(max_iters):
        # Update alpha
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        
        # Compute gradient
        grad = compute_gradient(h, alpha, 2.0 / n)
        
        # Update step size
        lr = 0.1 / np.sqrt(it + 1)
        
        # Update h
        h = h - lr * grad
        
        # Project onto feasible set
        h = project_to_box_simplex(h, n / 2)
        
        # Evaluate and track best
        cur_c5 = evaluate_c5(h)
        if cur_c5 < best_c5:
            best_h = h.copy()
            best_c5 = cur_c5
        
        # Early stopping
        if best_c5 < 0.379:
            break
            
    return best_h, best_c5
