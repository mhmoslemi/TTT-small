import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def evaluate_c5(h):
    """Evaluate C5 using correlation"""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation: h * (1 - h_rev)
    corr = correlate(h, 1 - h[::-1], mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def perturb_solution(h, perturb_factor=0.02):
    """Create a small random perturbation of the solution"""
    h_perturbed = h + np.random.normal(0, perturb_factor, len(h))
    return project_to_sum_box(h_perturbed, TARGET_SUM)

        # Try small random perturbations and keep the best
        h_trial = perturb_solution(h, 0.01)
        c5_trial = evaluate_c5(h_trial)
        
        if c5_trial < c5_current:
            h = h_trial
            c5_current = c5_trial
            if c5_current < best_c5:
                best_h = h.copy()
                best_c5 = c5_current

            best_c5 = c5_refined
            
    return best_h, best_c5
