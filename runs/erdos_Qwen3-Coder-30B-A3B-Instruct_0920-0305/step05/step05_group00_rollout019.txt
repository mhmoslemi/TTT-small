# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_of_smooth_max(h, beta):
    """Compute gradient of smoothed max with respect to h"""
    corr = compute_corr_fft(h)
    C = corr * dx
    
    # Clip to prevent overflow
    C_shifted = C - np.max(C)
    exp_C = np.exp(beta * C_shifted)
    w = exp_C / np.sum(exp_C)
    
    # Compute all gradients efficiently via conv
    grad_h = np.zeros_like(h)
    
    # For each lag ell, gradient is 1 - h_{i+ell} - h_{i-ell}
    # We compute these shift-wise, reusing FFT from above for the shifts
    for i in range(len(h)):
        grad_h[i] = w[0]  # This corresponds to ell=0: constant 1 term
        # Accumulate for positive lags (from right shifts)
        for l in range(1, len(h)):
            if i+l < len(h):
                grad_h[i] -= w[l] * h[i+l]
            # Accumulate for negative lags (from left shifts)
            if i-l >= 0:
                grad_h[i] -= w[l] * h[i-l]
    
    return grad_h

def smooth_max_descent(h0, max_iter=2000, beta0=0.5, beta_max=30, alpha0=0.4, gamma=1e-4):
    """Optimize via projected gradient descent on smooth max"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    beta = beta0
    alpha = alpha0
    
    for t in range(max_iter):
        # Compute gradient
        grad = gradient_of_smooth_max(h, beta)
        
        # Gradient descent step
        h_new = h - alpha * grad
        
        # Project to feasible set
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        # Update h
        h = h_new.copy()
        
        # Step size schedule
        if t < max_iter - 1:  # avoid computing alpha for last iteration
            alpha = alpha0 / (1 + gamma * t)
        
        # Beta schedule
        if (t + 1) % 400 == 0 and beta < beta_max:
            beta *= 2
    
    return best_h, best_c5
