def generate_legendre_pattern(n):
    """Generate a Legendre-based binary pattern if possible, otherwise fallback to random"""
    p = n // 2
    if p < 2:
        return None
    
    # Check if p is prime
    def is_prime(num):
        if num < 2:
            return False
        for i in range(2, int(math.sqrt(num)) + 1):
            if num % i == 0:
                return False
        return True
    
    if not is_prime(p):
        # Find largest prime less than p
        for q in range(p-1, 1, -1):
            if is_prime(q):
                p = q
                break
        else:
            return None
    
    try:
        # Generate quadratic residues
        residues = set()
        for a in range(1, p):
            residues.add((a * a) % p)
        
        b = np.zeros(p, dtype=int)
        for r in residues:
            if r < p:
                b[r] = 1
                
        # Duplicate to get length n
        h = np.concatenate([b, b])
        ones_count = np.sum(h)
        
        # Adjust to make exactly p ones
        if ones_count < p:
            # Find a zero to flip
            for i in range(len(h)):
                if h[i] == 0:
                    h[i] = 1
                    ones_count += 1
                    if ones_count == p:
                        break
        elif ones_count > p:
            # Find a one to flip
            for i in range(len(h)):
                if h[i] == 1:
                    h[i] = 0
                    ones_count -= 1
                    if ones_count == p:
                        break
        
        return h
    except:
        return None

def generate_random_balanced(n):
    """Generate a random balanced binary pattern"""
    h = np.zeros(n, dtype=int)
    indices = np.random.permutation(n)
    h[indices[:n//2]] = 1
    return h

def compute_correlation(h):
    """Compute the full correlation array"""
    corr = correlate(h, h, mode='full')
    n = len(h)
    return corr[n-1:]

def compute_c5(h, corr=None):
    """Compute C5 bound"""
    if corr is None:
        corr = compute_correlation(h)
    n = len(h)
    dx = 2.0 / n
    # C5 = max_k integral h(x)(1 - h(x+k)) dx = 1 - min_k integral h(x)h(x+k) dx
    # So just compute the minimal autocorrelation
    min_corr = np.min(corr)
    return 1.0 - min_corr * dx

def compute_grad_c5(h, corr=None):
    """Differentiable surrogate for gradient-based optimization"""
    if corr is None:
        corr = compute_correlation(h)
    n = len(h)
    dx = 2.0 / n
    # Apply log-sum-exp to make it smooth
    alpha = 30.0
    # For numerical stability
    max_corr = np.max(corr)
    # Use log-sum-exp trick: log(sum(exp(...))) = max_corr + log(sum(exp(corr - max_corr)))
    log_sum_exp = max_corr + np.log(np.sum(np.exp(alpha * (corr - max_corr))))
    # This is an approximation, exact gradient would require differentiating max operation
    return log_sum_exp * dx

def get_bad_ones(h, k):
    """Get indices where h[i]=1 and h[i+k]=0"""
    n = len(h)
    bad_ones = []
    for i in range(n - k):
        if h[i] == 1 and h[i+k] == 0:
            bad_ones.append(i)
    return bad_ones
