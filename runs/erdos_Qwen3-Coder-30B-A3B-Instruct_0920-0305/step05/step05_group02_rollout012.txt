import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def generate_legendre_pattern(n):
    """Generate a Legendre-based binary pattern if possible."""
    p = n // 2
    if p < 2:
        return None
    # Find the largest prime <= p
    def is_prime(n):
        if n < 2:
            return False
        for i in range(2, int(math.sqrt(n)) + 1):
            if n % i == 0:
                return False
        return True
    
    while not is_prime(p):
        p -= 1
        if p < 2:
            return None
    
    # Generate quadratic residues modulo p
    residues = set()
    for a in range(1, p):
        r = (a * a) % p
        residues.add(r)
    
    # Create binary mask
    b = np.zeros(p, dtype=int)
    for i in range(p):
        if i in residues:
            b[i] = 1
    
    # Create full pattern by concatenating two copies
    h = np.tile(b, 2)
    
    # Adjust if necessary to exactly have n/2 ones
    current_ones = np.sum(h)
    if current_ones > p:
        # Find indices to flip
        flip_indices = np.where(h == 1)[0]
        for i in range(int(current_ones - p)):
            h[flip_indices[i]] = 0
    elif current_ones < p:
        # Find indices to flip
        flip_indices = np.where(h == 0)[0]
        for i in range(int(p - current_ones)):
            h[flip_indices[i]] = 1
    
    return h

def generate_random_balanced(n):
    """Generate a random balanced binary pattern."""
    h = np.zeros(n, dtype=int)
    indices = np.random.permutation(n)
    h[indices[:n//2]] = 1
    return h

def compute_correlation(h):
    """Compute the full correlation array."""
    n = len(h)
    corr = np.zeros(2*n - 1)
    for k in range(-n+1, n):
        if k >= 0:
            corr[k+n-1] = np.sum(h[:n-k] * h[k:])
        else:
            corr[k+n-1] = np.sum(h[-k:] * h[:n+k])
    return corr

def update_correlation(corr, h, i, j):
    """Update correlation array after swapping h[i] and h[j]."""
    delta = np.zeros(len(corr))
    n = len(h)
    
    # Get the lags where changes occur
    # For position i:
    if i < n - 1:  # i participates in some sums
        for k in range(i+1, n):
            delta[k+i-n+1] += h[j] - h[i]
        for k in range(0, i+1):
            if i - k >= 0:
                delta[k+i-n+1] += h[j] - h[i]
    
    # For position j:
    if j < n - 1:  # j participates in some sums
        for k in range(j+1, n):
            delta[k+j-n+1] += h[i] - h[j]
        for k in range(0, j+1):
            if j - k >= 0:
                delta[k+j-n+1] += h[i] - h[j]
                
    return corr + delta

def get_worst_shifts(corr, tol=1e-12):
    """Get all shifts where correlation equals maximum within tolerance."""
    max_corr = np.max(corr)
    worst_shifts = np.where(np.abs(corr - max_corr) < tol)[0]
    return worst_shifts

def generate_swap(h, corr, worst_shifts, n):
    """Generate a valid swap considering current worst shift."""
    k = np.random.choice(worst_shifts)
    # Find BadOnes(k) = {i : h[i]=1, h[i+k]=0}
    bad_ones = []
    if k >= 0:
        for i in range(n - k):
            if h[i] == 1 and h[i+k] == 0:
                bad_ones.append(i)
    else:
        k = abs(k)
        for i in range(k, n):
            if h[i] == 1 and h[i-k] == 0:
                bad_ones.append(i)
    if not bad_ones:
        return -1, -1, corr
        
    i = np.random.choice(bad_ones)
    # Pick a zero to swap with
    # First try to find one that doesn't interfere with current k
    zero_indices = np.where(h == 0)[0]
    potential_swap = []
    for j in zero_indices:
        if k >= 0:
            if (j < n - k and j >= 0) and not (i == j or i - k == j or i == j - k):
                potential_swap.append(j)
        else:
            if (j >= k and j < n) and not (i == j or j - k == i or i == j):
                potential_swap.append(j)
    if not potential_swap:
        # Just pick any zero
        j = np.random.choice(zero_indices)
    else:
        j = np.random.choice(potential_swap)
        
    return i, j, corr
