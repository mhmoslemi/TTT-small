import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def project(v, S):
    """Project vector v onto the capped simplex: 0 ≤ x_i ≤ 1, sum x_i = S"""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2
        p = np.clip(v + lam, 0.0, 1.0)
        s = p.sum()
        if s > S:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2, 0.0, 1.0)

def compute_O_w_grad(h, alpha):
    """Compute overlap vector, weights, and gradient of smooth max"""
    n = len(h)
    dx = 2.0 / n
    # correlation of h with ones
    c1 = correlate(h, np.ones(n), mode='full')
    # self-correlation
    c2 = correlate(h, h, mode='full')
    # O_r = (c1 - c2) * dx
    O = (c1 - c2) * dx
    # Softmax weights
    O_max = np.max(O)
    w = np.exp(alpha * (O - O_max))
    w = w / np.sum(w)
    # Gradient components
    s = np.convolve(w, np.ones(n), mode='valid')
    g2 = correlate(w, h, mode='full')[n-1:2*n-1]
    g3 = correlate(w, h[::-1], mode='full')[n-1:2*n-1][::-1]
    g = dx * (s - g2 - g3)
    return O, w, g
