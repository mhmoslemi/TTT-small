def compute_correlation_fft(h):
    """Compute correlation h * (1-h) using FFT for efficiency"""
    H = rfft(h)
    G = rfft(1 - h)
    corr = irfft(H * np.conj(G))
    return corr

def project_simplex(h, T):
    """
    Project h onto the capped simplex: h >= 0, sum(h) = T
    Using binary search on the Lagrange multiplier
    """
    h = np.array(h, dtype=float)
    low, high = 0.0, 1.0
    
    while high - low > 1e-12:
        mid = (low + high) / 2
        z = np.clip(h - mid, 0, 1)
        s = np.sum(z)
        if s >= T:
            low = mid
        else:
            high = mid
    
    z = np.clip(h - low, 0, 1)
    # Scale to get exactly the right sum
    s = np.sum(z)
    if abs(s) < 1e-12:
        return z
    z = z * (T / s)
    # Final clip to ensure bounds
    z = np.clip(z, 0, 1)
    return z

def repair_solution(c, K, n, T):
    """
    From control values c (K values), interpolate to full resolution n,
    then project onto capped simplex.
    """
    # Interpolate c to full resolution
    x_knots = np.linspace(0, 2, K)
    x_full = np.linspace(0, 2, n)
    h_interp = np.interp(x_full, x_knots, c)
    h = project_simplex(h_interp, T)
    return h

def evaluate_c5(h):
    """Fast evaluation using FFT"""
    corr = compute_correlation_fft(h)
    dx = 2.0 / len(h)
    return np.max(corr) * dx

def soft_max_overlaps(O, alpha):
    """Soft maximum of overlaps"""
    max_O = np.max(O)
    # Subtract max to avoid overflow
    weights = np.exp(alpha * (O - max_O))
    return (max_O + np.log(np.sum(weights)) / alpha)

def objective_cmaes(c, K, n, T, alpha, dx):
    """Objective function for CMA-ES with soft max"""
    h = repair_solution(c, K, n, T)
    corr = compute_correlation_fft(h)
    O = corr * dx
    return soft_max_overlaps(O, alpha)

def subgradient_descent(h, max_shifts, steps=500, eta=0.05):
    """Projected gradient descent using exact max gradient"""
    h = h.copy()
    n = len(h)
    dx = 2.0 / n
    
    for t in range(steps):
        # Compute exact overlaps
        corr = compute_correlation_fft(h)
        O = corr * dx
        
        # Find active shifts
        max_val = np.max(O)
        active_shifts = np.where(np.abs(O - max_val) < 1e-10)[0]
        if len(active_shifts) == 0:
            active_shifts = [np.argmax(O)]
            
        # Compute subgradient (average of gradients from active shifts)
        g = np.zeros(n)
        for k in active_shifts:
            # g_i = (1-h_{i+k}) - h_{i-k} for k-th shift
            g_temp = np.zeros(n)
            if k > 0:
                g_temp[k:] = h[:-k]
            elif k < 0:
                g_temp[:k] = h[-k:]
            else:
                g_temp[:] = h
            g += (1.0 - g_temp) - h  # Simplification of pattern

        # Normalize
        g = g / len(active_shifts)

        # Update with step-size
        h = h - eta * g
        
        # Project back to simplex
        h = project_simplex(h, n/2)

        # If improvement is small, reduce step size
        if t > 50 and t % 10 == 0:
            eta *= 0.95
            if eta < 1e-6:
                break
                
    return h
