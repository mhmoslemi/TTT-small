from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using direct method"""
    h_rev = h[::-1]
    b = 1 - h
    corr = np.correlate(h_rev, b, mode="full")
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using direct correlation"""
    corr = compute_corr(h)
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def analyze_gradient(h):
    """Analyze the gradient of the C5 function to guide optimization"""
    # Compute current maximum correlation
    corr = compute_corr(h)
    max_idx = np.argmax(corr)
    
    # Create gradient direction based on the maximum correlation
    # This is based on how h influences the correlation at max_idx
    grad = np.zeros(n_points)
    
    # Add contributions from both sides of the correlation peak
    # For each position i, determine what its contribution to correlation at max_idx is
    if max_idx < len(h):
        # Left side
        for i in range(max_idx):
            grad[i] += h[max_idx - i]
        # Right side (reverse)
        for i in range(len(h)):
            if i + max_idx < len(corr):
                grad[i] += h[len(h) - 1 - (i + max_idx - len(h) + 1)]
    else:
        # Full correlation
        for i in range(len(h)):
            corr_idx = i + max_idx - len(h) + 1
            if 0 <= corr_idx < len(corr):
                grad[i] += h[len(h) - 1 - corr_idx]
    
    return grad

def guided_move(h, samples=3):
    """Make guided moves based on local gradient analysis"""
    grad = analyze_gradient(h)
    
    # Sort indices based on gradient values
    sorted_idx = np.argsort(grad)
    
    # Try several random moves from the best directions
    for _ in range(samples):
        # Select a random gradient direction
        i = sorted_idx[np.random.randint(0, len(sorted_idx))]
        # Try to transfer a small amount
        delta = np.random.uniform(0.01, 0.05)
        
        # Move toward improving direction: maximize grad
        j = np.argmax(grad)
        if j == i:
            j = (j + 1) % len(h)
        
        # Check constraints
        if h[i] >= delta and h[j] <= 1 - delta:
            # Apply
            h[i] -= delta
            h[j] += delta
            return True
    
    return False

def perturb_at_lag(h, k):
    """Perturb in a fixed direction at lag k"""
    # Try to increase h at positions where lag k interaction is larger
    max_k = len(h) - 1
    d = np.zeros(len(h))
    
    # For each i, d[i] represents how much position i should be increased
    for i in range(len(h)):
        # Lag effect of h[i] on some other position
        i_k = i + k
        if i_k < len(h):
            d[i] += h[i_k]
        
        # Also compute some descent direction
        if i < len(h) - k:
            d[i] -= h[i + k]
    
    # Choose positions to adjust based on d
    increase_idx = np.where(d > np.percentile(d, 70))[0]
    decrease_idx = np.where(d < np.percentile(d, 30))[0]
    
    if len(increase_idx) > 0 and len(decrease_idx) > 0:
        i = increase_idx[np.random.randint(len(increase_idx))]
        j = decrease_idx[np.random.randint(len(decrease_idx))]
        delta = min(h[i], 1 - h[j], 0.03)
        if delta > 0:
            h[i] -= delta
            h[j] += delta
            return True
    
    return False

def local_optimize(h, iter_count=100):
    """Perform local optimization with gradient-enhanced moves"""
    for _ in range(iter_count):
        # Try many guided moves
        for _ in range(3):
            if guided_move(h):
                break
        
        # Try some lag perturbations
        for k in [0, 1, 2, 3, 4, 5]:
            if np.random.random() < 0.2:
                perturb_at_lag(h, k)
    
    return h
