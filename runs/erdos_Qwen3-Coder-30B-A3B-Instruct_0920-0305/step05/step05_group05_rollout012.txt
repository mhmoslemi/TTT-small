import numpy as np
import time

def project_capped_simplex(v, T):
    """Project vector v onto the capped simplex: 0 ≤ x ≤ 1, sum x = T"""
    v = np.asarray(v)
    # Binary search for lambda
    lmbda_min, lmbda_max = -1.0, 1.0
    for _ in range(40):
        lmbda = (lmbda_min + lmbda_max) / 2.0
        x = np.clip(v - lmbda, 0, 1)
        sum_x = np.sum(x)
        if sum_x < T:
            lmbda_max = lmbda
        else:
            lmbda_min = lmbda
    lmbda = (lmbda_min + lmbda_max) / 2.0
    x = np.clip(v - lmbda, 0, 1)
    return x

def compute_O_and_grad(h, alpha):
    """Compute vector of overlaps and the smoothed gradient of their max"""
    n = len(h)
    dx = 2.0 / n
    # O_k = dx * sum_i h[i] * (1 - h[i + k]) for k = -n+1 to n-1
    # K = -n+1 to n-1, so we index [0, 2n-2], with shift = k + (n-1)
    corr = np.correlate(h, 1 - h, mode="full")
    corr *= dx
    O = corr.copy()
    # Use log-sum-exp for smooth max with parameter alpha
    m = np.max(O)
    w = np.exp(alpha * (O - m))
    w /= np.sum(w)
    # ∂F_α / ∂h = sum_k w_k * ∂O_k / ∂h
    # ∂O_k / ∂h[i] = dx * ( (1 - h[i + k]) - h[i - k] ) (k is shift, out-of-range is 0)
    grad = np.zeros(n)
    for k in range(-n + 1, n):
        k_idx = k + (n - 1)
        if k < 0:
            # h[i] * (1 - h[i + k]) where h[i+k] = 0 if i+k < 0
            # and h[i-k] = 0 if i-k >= n
            # So h[i] * (1 - h[i + k]) is 1 + 0 for i < -k or i >= n,
            # which gives no contribution to the gradient due to h[0]'s support on [0,n)
            # But still,
            # If k < 0, h[i+k] exists only for i >= -k, so the term 1-h[i+k] contributes
            # to ∂O_k / ∂h[i] only when i >= -k,
            # and similarly, h[i-k] contributes only when i-k < n or i < n+k,
            # but also there, j = i - k, so for j to be valid we need i > k
            # We construct correspondences by 1-d shift and add stuff
            temp = np.zeros(n)
            temp[:-k] = 1.0  # h[i]*1 iff i >= -k
            if k == -n + 1:  # This is from shift = n-1, so each i contributes 1 because h[i] * (1 - h[i - (n-1)])
                temp = np.ones(n)  # All indices can make contribution with this shift when i + (n-1) < n, we use k = -(n-1)
            temp_shift = np.roll(temp, -k)
            # also we have the h[i - k] term only when i-k < n 
            # and h properly indexed in 1-h[i-k] for i-k >= 0
            temp_clone = np.zeros(n)
            temp_clone[k:] = 1
            # don't do that for positive shifts, can't get that easily
            # For k < 0:
            #   index i contributes to grad if i + k >=0 and i <= n-1 => i >= -k and i <= n-1
            #   so j = i, we get h[i] * (1 - h[i+k]) contributes to i-th position of gradient for i >= -k
            #   by -h[i-k] term, we get -h[i-k].this is added to i-k position
            #   but when i < -k, j = i - k then j < -k-k = -2k, means j is out of bounds. so it doesn't trap.
            #   O(k) = dx * sum_{i} h_{i} * (1 - h_{i+k} ) 
            #   So 
            #   ∂ O(k)/ ∂h[i] = dx * ( 1 - h[i+k] - h[i-k] )
            # For positive k:
            #   we have h[i] * (1 - h[i+k])  
            #   ∂ O(k)/ ∂h[i] = dx * ( 1 - h[i+k] - h[i-k] )  
            temp2 = np.roll(h, -k)  # gives a type to shift k index, h[i-k]
            temp2 = np.clip(temp2, 0, 1)
            temp1 = np.ones(n)
            temp1 = np.roll(temp1, -k)
            # final term is 1 - h[i+k]  
            temp3 = np.roll(h, k)
            temp3 = np.clip(temp3, 0, 1)
            # We form: 1 * (1-h[i+k])  = 1 - h[i+k]  
            term = temp1 * (1 - temp3) - temp2  # (1-h[i+k]) - h[i-k] # h[i] term contributes to gradient of i, because of simplification of
            grad += w[k_idx] * term * dx
        else:
            # For positive k: Same idea. 
            # h[i] * (1 - h[i+k]) => contributes to grad of i when i+k monotonically increases and we also check h[i+k]
            # Therefore, for i >= 0, [h[i] on i] contributes 1-h[i+k]  
            # and for i-k < n-1, -h[i-k] contributes to i-k term (i > k):  
            # But the main correction is the index of contribution
            temp1 = np.roll(h, k)
            temp1 = np.clip(temp1, 0, 1)
            temp2 = np.ones(n)
            temp2 = np.roll(temp2, k)  # [1]*n        column vector
            # partial of O_k w.r.t h[i] = dx * ( 1 - h[i+k] - h[i-k] ) 
            temp3 = np.roll(h, -k)  # h[i-k] 
            temp3 = np.clip(temp3, 0, 1)  # just pick appropriate bd.
            # Now it's proper way
            term = temp2 * (1 - temp1) - temp3
            grad += w[k_idx] * term * dx 

    return O, grad

def init_h(n, init_type="uniform"):
    if init_type == "uniform":
        h = np.random.uniform(0, 1, n)
    elif init_type == "sine":
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.4 * np.sin(np.pi * x)
        h = np.clip(h, 0, 1)
    elif init_type == "gaussian":
        x = np.linspace(0, 2, n)
        mu = 1.0
        sigma = 0.2
        h = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
        h = np.clip(h, 0, 1)
    elif init_type == "triangular":
        x = np.linspace(0, 2, n)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = np.clip(h, 0, 1)
    elif init_type == "binary":
        h = np.zeros(n)
        # Make one pulse at center
        mid = n // 2
        width = max(1, n // 4)
        start = max(0, mid - width // 2)
        end = min(n, mid + width // 2)
        h[start:end] = 1
    else:
        h = np.random.uniform(0, 1, n)
    # Project onto the simplex with sum = n/2
    dx = 2.0 / n
    h = project_capped_simplex(h, n / 2)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 64
    dx = 2.0 / n
    T = n / 2
    best_c5_bound = float('inf')
    best_h = None
    max_iter = 4000
    alpha_start = 5.0
    alpha_end = 30.0
    lr_0 = 0.5
    subgrad_iter = 500
    symmetrise_interval = 10
    remaining_time = budget_s * 0.95
    
    # Precompute range indices to avoid repeated work
    range_k = np.arange(-n + 1, n)
    shift_indices = range_k + (n - 1)
    
    num_starts = 100  # This will be adapted on timer
    restart_count = 0
    
    while (time.time() - start_time) < remaining_time and restart_count < num_starts:
        # Select initial type
        types = ["uniform", "triangular", "sine", "gaussian", "binary"]
        init_type = np.random.choice(types)
        h = init_h(n, init_type)
        # Symmetrize the initial value
        h = (h + h[::-1]) / 2
        
        alpha = alpha_start
        for t in range(1, max_iter + 1):
            if (time.time() - start_time) > remaining_time:
                break
            alpha = alpha_start + (alpha_end - alpha_start) * t / max_iter
            
            O, grad = compute_O_and_grad(h, alpha)
            c5_now = np.max(O)
            
            if c5_now < best_c5_bound:
                best_c5_bound = c5_now
                best_h = h.copy()
                
            # Update h with gradient step
            lr = lr_0 / np.sqrt(t + 1)
            h -= lr * grad
            
            # Project h onto feasible set
            h = project_capped_simplex(h, T)
            
            # Optional symmetrisation
            if t % symmetrise_interval == 0:
                h = (h + h[::-1]) / 2
            
        # Refine using subgradient
        O, _ = compute_O_and_grad(h, alpha_end)
        active_shifts = np.where(np.isclose(O, np.max(O)))[0]
        if len(active_shifts) > 1:
            active_shifts = active_shifts[[0]]
        best_shift_idx = active_shifts[0]
        best_shift = best_shift_idx - (n - 1)
        
        # Subgradient descent on active shift only
        for t in range(subgrad_iter):
            if (time.time() - start_time) > remaining_time:
                break
            # Get the gradient of the active shift
            # ∂ O_k / ∂h[i] = dx * ( 1 - h[i+k] - h[i-k] )
            grad = np.zeros(n)
            k = best_shift  # shift for current maximizer
            
            # Compute weighted gradient over k
            if k < 0:
                temp1 = np.roll(h, k)
                temp1 = np.clip(temp1, 0, 1)  # h[i+k]
                temp2 = np.ones(n)
                temp2 = np.roll(temp2, k)
                temp3 = np.roll(h, -k)
                temp3 = np.clip(temp3, 0, 1)  # h[i-k]
                term = temp2 * (1 - temp1) - temp3
                grad += 1 * term * dx  
            else:
                temp1 = np.roll(h, k)
                temp1 = np.clip(temp1, 0, 1)  # h[i+k]
                temp2 = np.ones(n)
                temp2 = np.roll(temp2, k)
                temp3 = np.roll(h, -k)
                temp3 = np.clip(temp3, 0, 1)  # h[i-k]
                term = temp2 * (1 - temp1) - temp3
                grad += 1 * term * dx
            
            lr = 0.02 / np.sqrt(t + 1)
            h -= lr * grad
            
            # Project
            h = project_capped_simplex(h, T)
            Onew, _ = compute_O_and_grad(h, alpha_end)
            c5_new = np.max(Onew)
            if c5_new < best_c5_bound:
                best_c5_bound = c5_new
                best_h = h.copy()
                
        restart_count += 1
        
        if (time.time() - start_time) > remaining_time:
            break
            
    # Final check on best
    if best_h is not None:
        final_O, _ = compute_O_and_grad(best_h, alpha_end)
        final_c5 = np.max(final_O)
        return (best_h.tolist(), float(final_c5), n)
    else:
        # Return worst-known solution if nothing is good
        worst_h = np.array([0.5] * n)
        worst_h = project_capped_simplex(worst_h, T)
        worst_O, _ = compute_O_and_grad(worst_h, alpha_end)
        return (worst_h.tolist(), float(np.max(worst_O)), n)
