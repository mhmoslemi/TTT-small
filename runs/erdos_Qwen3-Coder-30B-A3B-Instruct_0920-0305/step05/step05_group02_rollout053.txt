import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def compute_correlation(h):
    """Compute the full correlation array for a binary vector"""
    n = len(h)
    corr = np.zeros(2*n - 1)
    for k in range(-n+1, n):
        if k >= 0:
            corr[k+n-1] = np.sum(h[:n-k] * h[k:])
        else:
            corr[k+n-1] = np.sum(h[-k:] * h[:n+k])
    return corr

def update_correlation(corr, h, i, j):
    """Update correlation array after swapping h[i] and h[j]"""
    n = len(h)
    delta = np.zeros(2*n - 1)
    
    # For each lag m, compute change in correlation
    for m in range(-n+1, n):
        if m >= 0:
            # Check if i influences this lag
            if i < n - m:
                delta[m+n-1] += h[i] - h[i] * h[i + m] - (1 - h[i]) + (1 - h[i]) * (1 - h[i + m])
            # Check if j influences this lag
            if j < n - m:
                delta[m+n-1] += h[j] - h[j] * h[j + m] - (1 - h[j]) + (1 - h[j]) * (1 - h[j + m])
        else:
            # Check if i influences this lag
            if i >= -m:
                delta[m+n-1] += -h[i + m] * h[i] - (-h[i + m] * (1 - h[i])) 
            # Check if j influences this lag
            if j >= -m:
                delta[m+n-1] += -h[j + m] * h[j] - (-h[j + m] * (1 - h[j]))
    return corr + delta

def find_worst_shifts(corr, threshold=1e-10):
    """Find all shifts where correlation equals the maximum"""
    max_corr = np.max(corr)
    return [i for i in range(len(corr)) if abs(corr[i] - max_corr) < threshold]

def generate_legendre_pattern(n):
    """Generate a Legendre-based binary pattern if possible"""
    p = n // 2
    if p < 2:
        return None
    # Check if p is prime
    is_prime = True
    for i in range(2, int(np.sqrt(p)) + 1):
        if p % i == 0:
            is_prime = False
            break
    if not is_prime:
        return None
    
    # Generate quadratic residues
    residues = set()
    for a in range(1, p):
        residues.add((a*a) % p)
    
    # Create binary mask
    b = np.zeros(p, dtype=int)
    for r in residues:
        b[r] = 1
    
    # Double it to full length
    h = np.concatenate([b, b])
    # If we don't have exactly n/2 ones, fix it
    ones = np.sum(h)
    if ones != p:
        # Flip one zero to one (select first zero)
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] = 1
            ones += 1
        else:
            # In case of error, just return original (but should not happen)
            restore_h = h.copy()
            if ones > p:
                # Find and flip one one to zero
                ones_pos = np.where(h == 1)[0]
                h[ones_pos[0]] = 0
    return h

def random_balanced_mask(n):
    """Generate a random balanced binary mask"""
    h = np.zeros(n, dtype=int)
    ones_indices = np.random.choice(n, size=n//2, replace=False)
    h[ones_indices] = 1
    return h

                        # Revert swap
                        h[i], h[j] = h[j], h[i]
