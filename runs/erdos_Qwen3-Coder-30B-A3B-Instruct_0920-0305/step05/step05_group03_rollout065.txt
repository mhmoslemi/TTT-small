import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from cvxpy import *
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_hill_climb_swap(h, i, j):
    """Evaluate the change in max overlap if we swap h[i] and h[j]"""
    n = len(h)
    if h[i] == h[j]:
        return 0.0
    
    # Create a copy to simulate the swap
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]
    
    # Evaluate the new C5
    return evaluate_c5(h_new) - evaluate_c5(h)

def hill_climb_binary(h, max_swaps=10000):
    """Perform hill-climbing by swapping 1s and 0s to improve C5"""
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    swaps = 0
    improved = True
    while improved and swaps < max_swaps:
        improved = False
        for i in range(len(h)):
            for j in range(i+1, len(h)):
                if h[i] != h[j]:
                    delta = evaluate_hill_climb_swap(best_h, i, j)
                    if delta < 0:  # Found an improving swap
                        best_h[i], best_h[j] = best_h[j], best_h[i]
                        best_c5 += delta
                        improved = True
                        swaps += 1
                        break
            if improved:
                break
    return best_h, best_c5

def project_simplex(v, target_sum):
    """Project vector v onto the capped simplex with sum = target_sum"""
    # Sort in descending order for water filling
    sorted_v = np.sort(v)[::-1]
    n = len(v)
    # Find threshold index
    cumsum = np.cumsum(sorted_v)
    threshold_idx = np.argmax(sorted_v > (target_sum - cumsum + sorted_v) / np.arange(1, n+1))
    # If threshold is beyond the array, adjust
    if threshold_idx == 0 and sorted_v[0] < target_sum / n:
        # All elements should be clamped to target_sum/n
        return np.full_like(v, target_sum / n)
    
    # Compute the threshold value
    threshold = (target_sum - cumsum[threshold_idx-1]) / (n - threshold_idx + 1)
    # Clamp and adjust
    result = np.clip(v, 0, 1)
    # Recalculate adjusted values
    adjusted = np.maximum(v - threshold, 0)
    # Adjust all elements, not just those below threshold
    if np.sum(adjusted) > target_sum:
        # Scale back to target sum
        scale = target_sum / np.sum(adjusted)
        result = adjusted * scale
    else:
        result = adjusted
    
    # Recalculate sum for bounding
    current_sum = np.sum(result)
    if abs(current_sum - target_sum) > 1e-8:
        # Re-adjust using water filling
        diffs = np.sort(result)[::-1]
        cum_sums = np.cumsum(diffs)
        # Find where cumulative sum goes over the target
        for k in range(len(diffs)):
            if cum_sums[k] > target_sum:
                delta = cum_sums[k] - target_sum
                thresh = diffs[k] - delta / (k+1)
                # Project everything above threshold
                result = np.maximum(result, thresh)
                # Renormalize to target_sum
                result = result * target_sum / np.sum(result)
                break
        else:
            # All values equal to target_sum / n
            avg = target_sum / len(result)
            result = np.full_like(result, avg)
    
    return result

def project_on_simplex(h, target_sum):
    """More robust simplex projection"""
    while True:
        new_h = project_simplex(h, target_sum)
        # Check if we've converged within numerical tolerance
        dist = np.sum((new_h - h) ** 2)
        if dist < 1e-10:
            break
        h = new_h
    return h

def refine_continuous(h, max_iters=500, eta=0.5):
    """Refine continuous function using projected gradient descent"""
    current_h = h.copy()
    prev_c5 = evaluate_c5(current_h)
    n = len(current_h)
    dx = 2.0 / n
    for i in range(max_iters):
        # Compute current overlaps
        corr = correlate(current_h, 1 - current_h, mode="full")
        max_corr_idx = np.argmax(corr)
        max_overlap = corr[max_corr_idx]
        
        # Find active shifts
        active = []
        # Go through all possible shifts
        for k in range(-n+1, n):
            if k >= 0:
                # This is the early one
                corr_val = np.sum(current_h[:n-k] * (1 - current_h[k:]))
            else:
                corr_val = np.sum(current_h[-k:] * (1 - current_h[:n+k]))
            if abs(corr_val - max_overlap) < 1e-8:
                active.append(k)
        if len(active) == 0:
            active = [max_corr_idx - (n-1)]

        # Construct weight vector w
        w = np.zeros(n)
        for k in active:
            if k >= 0:
                w[:n-k] += 1
            else:
                w[-k:] += 1

        # Construct gradients
        g1 = np.convolve(w, np.ones(n), mode="full")[n-1:2*n-1]
        g2 = np.convolve(w, np.ones(n), mode="full")[n-1:2*n-1]
        g = dx * (g1 - current_h - g2[::-1])

        # Update
        current_h -= eta * g

        # Project onto capped simplex
        current_h = project_on_simplex(current_h, n/2)

        # Symmetrize
        current_h = (current_h + current_h[::-1]) / 2.0
        current_h = project_on_simplex(current_h, n/2)
        
        c5 = evaluate_c5(current_h)
        if c5 > prev_c5:
            # Reducing step size if no improvement
            eta *= 0.5
        else:
            prev_c5 = c5
            
        if eta < 1e-8:
            break

    return current_h, evaluate_c5(current_h)

    pairs_for_shift = {}
    for k in range(-n_points+1, n_points):
        pairs = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                pairs.append((i, j))
        if len(pairs) > 0:
            pairs_for_shift[k] = pairs

    # Define auxiliary binary variables for products
    y_vars = {}
    y_vars_by_k = {}

    # Add constraints for product linearisation
    constraints = [sum(h) == target_sum]

    for k, pairs in pairs_for_shift.items():
        # Create binary variables for each product
        y_list = []
        for i, j in pairs:
            y = Variable(boolean=True)
            y_vars[(i, j, k)] = y
            y_list.append(y)
            # Add McCormick constraints
            constraints.append(y <= h[i])
            constraints.append(y <= h[j])
            constraints.append(y >= h[i] + h[j] - 1)
        y_vars_by_k[k] = y_list

    # Add the overlap constraints
    overlaps = []
    tau = Variable()
    constraints.append(tau >= 0)

    for k, pairs in pairs_for_shift.items():
        y_list = y_vars_by_k[k]
        # Compute O_k = sum(h) - sum(y)
        O_k = sum(h) - sum(y_list)  # Use symbolic sums here
        constraints.append(O_k <= tau)
        overlaps.append(O_k)

    # Objective
    prob = Problem(Minimize(tau), constraints)
    prob.solve(solver='CBC', time_limit=0.4 * budget_s, verbose=False)

    # If we got a solution, extract the binary values
    h_opt = None
    if prob.status == 'optimal' or prob.status == 'optimal_inaccurate':
        h_opt = np.array([float(h_i.value) for h_i in h])
    else:
        # If MILP failed to solve, fallback to a random binary pattern
        h_opt = np.zeros(n_points)
        ones_indices = np.random.choice(n_points, int(target_sum), replace=False)
        h_opt[ones_indices] = 1.0

    # Phase 2: Hill climber on binary masks
    h_bin, c5_bin = hill_climb_binary(h_opt)
    if c5_bin < best_c5:
        best_h = h_bin
        best_c5 = c5_bin

    # Phase 3: Continuous refinement
    h_cont, c5_cont = refine_continuous(best_h, max_iters=300)
    if c5_cont < best_c5:
        best_h = h_cont
        best_c5 = c5_cont

    # Phase 4: Optional up-sampling
    if (time.time() - start_time) < 0.8 * budget_s:
        # Interpolate to double the resolution
        n_upsample = n_points * 2
        x = np.linspace(0, 2, n_points)
        x_upsample = np.linspace(0, 2, n_upsample)
        h_upsample = np.interp(x_upsample, x, best_h)

        h_refine_up, c5_refine_up = refine_continuous(h_upsample, max_iters=200, eta=0.25)
        if c5_refine_up < best_c5:
            best_h = h_refine_up
            best_c5 = c5_refine_up

    return (best_h.tolist(), best_c5, len(best_h))
