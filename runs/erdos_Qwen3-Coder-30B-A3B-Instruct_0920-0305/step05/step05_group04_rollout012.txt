import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def project_simplex(h, target_sum):
    """
    Projects h onto the simplex {x: sum(x)=target_sum, 0<=x<=1}.
    Uses the algorithm from Duchi et al. (2008) 'Efficient Projections onto the 
    L1-Ball for learning in high dimensions'.
    """
    # clip to bounds
    h = np.clip(h, 0, 1)
    # sort in descending order
    h_sorted = np.sort(h)[::-1]
    # compute cumulative sums
    cumsum = np.cumsum(h_sorted)
    # find the index that satisfies the simplex constraint (i+1)*h_sorted[i] > (cumsum[i] - target_sum)
    idx = np.where((np.arange(1, len(h_sorted) + 1)*h_sorted - cumsum + target_sum) > 0)[0]
    if len(idx) == 0:
        # all projected values are at their lower bound 0
        return np.zeros_like(h)
    # k is the index of the largest element satisfying the constraint
    k = idx[-1] 
    # compute lambda
    lambda_val = (cumsum[k] - target_sum) / (k + 1)
    # threshold 
    h_proj = np.maximum(h - lambda_val, 0)
    return h_proj

def compute_corr(h):
    """ Compute correlation vector c = conv(h,h) (linear) for shifts 1 to n-1"""
    # zero pad to have at least 2*n-1 length for linear convolution
    n = len(h)
    h_padded = np.concatenate([h, np.zeros(n-1)])
    h_rev = h[::-1]
    h_rev_padded = np.concatenate([h_rev, np.zeros(n-1)])
    # correlations at all lags: (full convolution)
    c_full = np.convolve(h_padded, h_rev_padded, mode='full')
    # extract shifts 1 to n-1  
    c = c_full[n-1:2*n-2]  # indices from n-1 to 2n-3
    return np.array(c)

def compute_obj_grad(h, beta, dx):
    """ Compute the smooth surrogate objective and its gradient """
    n = len(h)
    c = compute_corr(h)
    # soft-min with β scaling, handle numerical overflow
    c_max = np.max(c)
    exp_terms = np.exp(-beta * (c - c_max))
    Z = np.sum(exp_terms)
    phi_beta = (1.0 / beta) * (np.log(Z) + c_max)
    # Gradient with soft weights
    w = exp_terms / Z
    # convolutions with w and reverse(w)
    w_rev = w[::-1]
    grad = -np.convolve(h, w_rev, mode="valid") 
    # backend convolution with zero padding: need to do both directions 
    # compute two components: h shifted forward and backward
    # now convolve both w and w_rev with h
    # forward shift part 
    h_fwd = np.pad(h, (0, n-1), mode='constant')
    w_fwd = np.pad(w_rev, (0, n-1), mode='constant')
    # backshift part 
    h_bwd = np.pad(h, (n-1, 0), mode='constant') 
    w_bwd = np.pad(w, (n-1, 0), mode='constant') 
    # compute both convolutions (they will be truncated)
    g1 = np.convolve(h_fwd, w_fwd, mode='valid')[:n]
    g2 = np.convolve(h_bwd, w_bwd, mode='valid')[:n]
    grad = -(g1 + g2) 
    # so we compute gradient of sum(w_k c_k) via two convolutions
    # we want gradient of sum w_k c_k given w_k = exp(-beta c_k) / Z
    # (d/dh_i) sum w_k c_k = sum_k w_k (h_{i-k} + h_{i+k})
    # Because elements at boundary are zero we can use the O(n log n) or O(n^2) implementation with care
    return 1.0 + phi_beta, grad

def evaluate_c5(h):
    """Evaluate C5 using true max operation. Should be called on final result."""
    n = len(h)
    dx = 2.0 / n
    # this computes differing auto-correlations in the full convolution mode 
    corr = correlate(h, 1-h, mode="full")
    # Extract the relevant parts
    max_corr = np.max(corr[len(h)-1:])  # corresponds to shifts 1,...,n-1
    return max_corr * dx
