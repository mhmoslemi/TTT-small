import numpy as np
from cvxpy import *
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Helper: greedy construction for warm starting
    def greedy_initialization(n):
        x = np.zeros(n, dtype=int)
        selected = []
        for _ in range(n // 2):
            best_idx = -1
            best_min_p = -1
            for i in range(n):
                if i in selected:
                    continue
                tentative = selected + [i]
                # compute min p_k for this tentative set
                min_p = float('inf')
                for k in range(1, n):
                    p_k = 0
                    for j in range(n - k):
                        if j in tentative and (j + k) in tentative:
                            p_k += 1
                    min_p = min(min_p, p_k)
                if min_p > best_min_p:
                    best_min_p = min_p
                    best_idx = i
            selected.append(best_idx)
            x[best_idx] = 1
        return x

    # Helper: evaluate C5
    def compute_C5(h):
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx

    best_h = None
    best_c5 = float('inf')
    # Consider a small number of resolutions
    resolutions = [36, 40, 44, 48]
    
    for n_points in resolutions:
        if (time.time() - start_time) > budget_s * 0.9:
            break
            
        # Define variables
        x = np.array([Variable(1, boolean=True) for _ in range(n_points)])
        # Add auxiliary variables y[i,k] for valid pairs
        y_vars = {}
        for k in range(1, n_points):
            for i in range(n_points - k):
                y_vars[(i, k)] = Variable(1, boolean=True)
        
        # Build constraints
        constraints = []
        # sum of h = 1
        constraints.append(sum(x) == n_points // 2)
        # y variables are defined by bounds (McCormick)
        for k in range(1, n_points):
            for i in range(n_points - k):
                constraints.append(y_vars[(i, k)] <= x[i])
                constraints.append(y_vars[(i, k)] <= x[i + k])
                constraints.append(y_vars[(i, k)] >= x[i] + x[i + k] - 1)
        
        # Set symmetry breaking (fix first value to 1)
        constraints.append(x[0] == 1)
        
        # Left-heavy constraint
        left_sum = sum(x[:n_points//2])
        right_sum = sum(x[n_points//2:])
        constraints.append(left_sum >= right_sum)

        # Initial solution for warm start
        try:
            x_init = greedy_initialization(n_points)
            for i in range(n_points):
                x[i].value = x_init[i]
            # Easy check for feasibility 
            assert sum(x_init) == n_points // 2
        except:
            continue

        # Objective
        L = Variable(1, integer=True)
        # Define the p_k values (hard to couple in one form with CVXPY)
        # Simple approach: enumerate and enforce p_k >= L
        obj_constraints = []
        for k in range(1, n_points):
            p_k = sum(y_vars[(i, k)] for i in range(n_points - k))
            obj_constraints.append(p_k >= L)

        # Combined constraints
        all_cons = constraints + obj_constraints
        
        # Define objective to maximize L
        objective = Maximize(L)
        
        # Create the problem
        prob = Problem(objective, all_cons)

        try:
            # Solve with a time limit and error handling
            prob.solve(solver=SCIPY, time_limit=10, verbose=False)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                sol_f = [x[i].value for i in range(n_points)]
                sol_f = np.array(sol_f, dtype=float)
                sol_h = sol_f
                
                # Evaluate C5 with floating point step function
                c5_val = compute_C5(sol_h)
                
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = sol_h.copy()
                    
                # Break early if we've done well
                if best_c5 < 0.3805:
                    break
        except Exception as e:
            print("Error in solving", e)
            continue
    
    # Optional: polish with continuous refinement
    if best_h is not None and (time.time() - start_time) < budget_s * 0.95:
        # Strategy: slightly adjust to relax binary condition; project to feasible set
        def polish_objective(h):
            dx = 2.0 / len(h)
            corr = np.correlate(h, 1 - h, mode="full")
            max_corr = np.max(corr)
            return max_corr * dx

        def constraint_sum(h):
            dx = 2.0 / len(h)
            return np.sum(h) * dx - 1.0

        bounds = [(0, 1) for _ in range(len(best_h))]
        cons = {'type': 'eq', 'fun': constraint_sum}

        # Project to valid range, then do local gradient descent
        h_polished = np.clip(best_h, 0, 1)
        dx = 2.0 / len(best_h)
        scale = 1.0 / (np.sum(h_polished) * dx)
        h_polished = h_polished * scale

        try:
            result = minimize(polish_objective, h_polished, method='SLSQP', bounds=bounds, constraints=cons,
                            options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12})
            if result.success:
                c5_polished = polish_objective(result.x)
                if c5_polished < best_c5:
                    best_c5 = c5_polished
                    best_h = result.x.copy()
        except:
            pass

    # Return final best result
    return (best_h.tolist(), best_c5, len(best_h)) if best_h is not None else ([], float('inf'), 0)
