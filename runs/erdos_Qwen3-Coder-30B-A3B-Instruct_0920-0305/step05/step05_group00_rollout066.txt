# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.irfft(np.fft.rfft(h_rev, L) * np.fft.rfft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_approximation(h):
    """Generate a smooth approximation to a step function"""
    # Create a smoother version that's still piecewise constant-like
    h_smooth = np.zeros_like(h)
    
    # For each point, average with neighbors
    for i in range(n_points):
        # Average of 3 points centered on i
        start = max(0, i - 1)
        end = min(n_points, i + 2)
        h_smooth[i] = np.mean(h[start:end])
    
    # Project to valid range and sum
    h_smooth = project_to_sum_box(h_smooth, TARGET_SUM)
    return h_smooth

def simulated_annealing(h0, run_seconds, rng):
    """Simplified simulated annealing"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    T = 0.05
    decay = 0.95
    max_iter = 20000
    iter_count = 0
    
    start_time = time.time()
    while iter_count < max_iter and time.time() - start_time < run_seconds:
        # Generate a move
        i = rng.randint(0, n_points)
        j = rng.randint(0, n_points)
        # Add small change to avoid equal indices
        delta = rng.uniform(-0.1, 0.1)
        # Clip to valid range
        if 0 <= h[i] - delta <= 1 and 0 <= h[j] + delta <= 1:
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            c5_new = evaluate_c5(h_new)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            elif rng.random() < np.exp(-(c5_new - best_c5) / T):
                h = h_new.copy()
            
            T *= decay
            iter_count += 1
        else:
            iter_count += 1

    return best_h, best_c5

def local_refinement(h):
    """Refine solution with gradient-based optimization""" 
    try:
        # Use scipy's minimize with L-BFGS-B for better local search
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h_vals):
            return evaluate_c5(h_vals)
        
        def constraint_eq(h_vals):
            return np.sum(h_vals) - TARGET_SUM
        
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        res = minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h
