import numpy as np

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_gradient(h, beta=1.0):

    # Here's another way:
    # Gradient of correlation (C) is:

    # The gradient ∂C_l/∂h_i = 1 - h_{i+l} - h_{i-l} (fixed for the current l)

    # We compute the gradient of l-th correlation directly:
    # ∇C_l = 1*(1) - h_(i+1) - h_(i-1) = 1 - s_(l) - s_(-l) where s_(l) is shifted forward by l

    # corresponding to the previous optimization encapsulation:
    
    exp_corr = np.exp(beta * corr_raw)

def projected_gradient_descent(h0, max_iter=5000, alpha=0.1, beta_schedule=None, verbose=False):
    """Project gradient steps, support smooth max based optimization"""
    h = h0.copy()
    n = len(h)
    if beta_schedule is None:
        beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 30.0]

    # Start with first beta
    beta = beta_schedule[0]
    beta_idx = 0
    current_iter = 0
    
    prev_c = np.inf
    for iteration in range(max_iter):
        current_iter += 1

        # compute smooth gradient using original definition
        try:
            grad = smooth_max_gradient(h, beta=beta)
        except:
            # fallback if any issue
            corr = compute_corr_fft(h)
            corr = corr[:n]  # limit to natural range
            max_corr = np.max(corr)
            grad = np.zeros_like(h)
            # compute Heuristic gradient with voracious method
            for l in range(1, n):
                shifted_f = np.roll(h, -l)
                shifted_b = np.roll(h, l)
                grad += 1 - shifted_f - shifted_b
            # omit normalization for simplicity now but this is heuristic

        # Apply gradient step
        h = h - alpha * grad

        # Project onto box and sum constraint
        h = project_to_sum_box(h, TARGET_SUM)

        # adaptive alpha based on improvement
        if current_iter % 100 == 0:
            corr = compute_corr_fft(h)
            new_c = c5_from_corr(corr, n)
            if np.abs(new_c - prev_c) < 1e-8:
                alpha *= 0.5
                if alpha < 1e-6:
                    break
            prev_c = new_c

        # schedule change of beta
        if current_iter % 1000 == 0:
            if beta_idx < len(beta_schedule) - 1:
                beta_idx += 1
                beta = beta_schedule[beta_idx]
                if verbose:
                    print(f"Schenging beta to {beta}")
            else:
                # starting final phase with high-beta
                pass

    return h
