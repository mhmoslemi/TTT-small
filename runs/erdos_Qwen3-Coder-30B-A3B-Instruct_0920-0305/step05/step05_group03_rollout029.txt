```python
import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 44  # Even number for integer sum
    dx = 2.0 / n_points
    total_sum = n_points / 2
    
    # Start with an initial solution (e.g., uniform with slight perturbation)
    h0 = np.ones(n_points) * 0.5
    # Normalize initial h to satisfy sum constraint
    h0 = h0 * (total_sum / np.sum(h0))
    
    # Active set of shifts
    active_K = {0}
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Track LP solution
    lp_solution = None
    
    # Reference to the LP variables
    h = None
    tau = None
    
    def symmetrize_h(h_vec):
        """Enforce h[i] = h[n-1-i]"""
        n = len(h_vec)
        mid = n // 2
        sym = np.zeros(n)
        for i in range(mid):
            avg = (h_vec[i] + h_vec[n-1-i]) / 2
            sym[i] = avg
            sym[n-1-i] = avg
        if n % 2 == 1:
            sym[mid] = h_vec[mid]
        return sym
    
    def project_to_simplex(h_vec):
        """Project h to the capped simplex: 0 <= h[i] <= 1, sum h = total_sum"""
        # Simple projection via clipping and rescaling
        h_clipped = np.clip(h_vec, 0, 1)
        total = np.sum(h_clipped)
        scale = total_sum / total if total != 0 else 1.0
        h_proj = h_clipped * scale
        return h_proj

    def compute_gradients(h_vec, active_shifts):
        """Compute subgradient of the max overlap function at h_vec"""
        n = len(h_vec)
        # Number of active shifts
        num_active = len(active_shifts)
        if num_active == 0:
            return np.zeros(n)

        # Sum the gradients from each active shift
        grad_sum = np.zeros(n)
        for k in active_shifts:
            # Compute gradient of O_k = sum(h) - sum(h_i * h_{i+k})
            # So gradient of O_k w.r.t. h_i = 1 - (h_{i+k} + h_{i-k})
            # But use correlation trick for its convolution form
            
            # One could do just O_k = sum(h) - sum(h_i * h_{i+k})
            # Gradient would dx_i = 1 - (h_{i+k} if k > 0 else h_{i-k} if k < 0 else 0)
            # Actually: ∂O_k/∂h_i = 1 - (h_{i+k} + h_{i+k}^{rev}) where ^rev reflects
            
            # Easier: use correlate for convolution definition directly
            # For fixed k, we want to compute (h_i * h_{i+k})'_i = h_{i+k} + h_{i-k}
            # But adjust the indexing to reflect this properly.
            # Also we're computing ∂O_k/h_i = 1 - ∂(h_i h_{i+k})/∂h_i = 1 - h_{i+k}
            # But it's actually: ∂/∂h_i (h_i h_{i+k}) = h_{i+k} (taking derivative wrt h_i)
            # Similarly ∂/∂h_{i+k} = h_i, so all such partials are known
            
            # Assume out-of-bounds are zero, consistent with np.correlate
            conv_grad = np.zeros(n)
            if k > 0:
                conv_grad[k:] = h_vec[:-k]
            elif k < 0:
                conv_grad[:k] = h_vec[-k:]
            else:  # k == 0
                conv_grad[:] = 0

            # Adjust for symmetry — it will be the same...
            grad_k = 1.0 - conv_grad
            grad_sum += grad_k

        return grad_sum / num_active


    # Start the cutting-plane method
    iteration = 0
    max_iterations = 300
    time_limit = 0.95 * budget_s

    while time.time() - start_time < time_limit and iteration < max_iterations:
        iteration += 1

        # Build the LP model
        h = Variable(n_points)
        tau = Variable()

        # Constraints: bounds on h
        constraints = [
            h >= 0,
            h <= 1,
            sum(h) == total_sum,
        ]
        # Optional symmetry
        for i in range(n_points // 2):
            constraints.append(h[i] == h[n_points - 1 - i])

        # Add McCormick constraints for active shifts
        y_vars = {}

        for k in active_K:
            range_k = n_points - abs(k)
            if k >= 0:
                y_vars[k] = Variable(range_k)
                # k > 0: indices i=0..n-1-k, y_{i,k} = h[i] h[i+k]
                # y_{i,k} >= 0
                constraints.append(y_vars[k] >= 0)
                for i in range(range_k):
                    # y_{i,k} <= h[i]
                    constraints.append(y_vars[k][i] <= h[i])
                    # y_{i,k} <= h[i+k]
                    constraints.append(y_vars[k][i] <= h[i+k])
                    # y_{i,k} >= h[i] + h[i+k] - 1
                    constraints.append(y_vars[k][i] >= h[i] + h[i+k] - 1)
                    # Pair: h[i], h[i+k]

                # Add the constraint: τ >= sum(h) - sum(y)
                constraints.append(tau >= sum(h) - sum(y_vars[k]))
            else:
                # k < 0: indices i=|k|..n-1, y_{i,k} = h[i] h[i+k]
                y_vars[k] = Variable(range_k)
                # Full range
                for i in range(range_k):
                    # y_{i,k} >= 0
                    constraints.append(y_vars[k][i] >= 0)
                    # y_{i,k} <= h[i]
                    constraints.append(y_vars[k][i] <= h[i])
                    # y_{i,k} <= h[i+k]
                    constraints.append(y_vars[k][i] <= h[i+k])
                    # y_{i,k} >= h[i] + h[i+k] - 1
                    constraints.append(y_vars[k][i] >= h[i] + h[i+k] - 1)

                # Add constraint
                constraints.append(tau >= sum(h) - sum(y_vars[k]))

        # Objective
        objective = Minimize(tau)
        prob = Problem(objective, constraints)

        # Try to solve with GLOP, fallback to CBC
        solver = GLOP
        try:
            prob.solve(solver=solver, verbose=False)
        except:
            try:
                solver = CBC
                prob.solve(solver=solver, verbose=False)
            except:
                # No solver available, return current best solution
                return (best_h.tolist(), best_c5, n_points)

        if prob.status != 'optimal':
            # Not optimal yet - stop with current best
            break

        # Extract solution
        h_vec = h.value
        tau_val = tau.value

        # Symmetrize if needed
        h_sym = symmetrize_h(h_vec)
        # Project back onto simplex
        h_proj = project_to_simplex(h_sym)

        # Evaluate exact C5
        c5_exact = evaluate_c5(h_proj)

        if c5_exact < best_c5:
            best_h = h_proj.copy()
            best_c5 = c5_exact

        # Check if we're close to optimality
        if c5_exact <= tau_val + 1e-6:
            break  # Converged

        # Compute actual overlaps
        # Full correlation using numpy
        corr_full = correlate(h_proj, 1 - h_proj, mode='full')
        shift_indices = np.arange(-n_points+1, n_points)
        max_k = shift_indices[np.argmax(corr_full)]

        # Check if max_k already in active_K
        if max_k not in active_K:
            active_K.add(max_k)
            # Debug info
            # print(f"Iteration {iteration}: Added shift {max_k}")
        else:
            # Can add a new one for diversity
            pass

    # Final polishing phase
    h_solved = best_h.copy()
    
    # Projected sub-gradient descent method
    remaining_time = budget_s - (time.time() - start_time)
    max_steps = min(int(1000), int(remaining_time * 1000))

    # Compute initial active set
    corr_full = correlate(h_solved, 1 - h_solved, mode='full')
    shift_indices = np.arange(-n_points+1, n_points)
    max_k = shift_indices[np.argmax(corr_full)]
    active_shifts = {max_k}

    # Ensure active shift is in real range
    max_k_real = np.argmax(corr_full)
    real_shifts = [shift_indices[i] for i in range(len(shift_indices)) if i == max_k_real]
    if real_shifts:
        active_shifts = set(real_shifts)

    for step in range(max_steps):
        # Re-evaluate overlaps
        corr_full = correlate(h_solved, 1 - h_solved, mode='full')
        shift_indices = np.arange(-n_points+1, n_points)
        max_k = shift_indices[np.argmax(corr_full)]

        # Update active set
        active_shifts = {max_k}

        # Compute subgradient
        subgrad = compute_gradients(h_solved, active_shifts)

        # Adaptive step size
        if step == 0:
            eta = 0.5 * dx
        else:
            # Decrease step size based on gradient norm
            grad_norm = np.linalg.norm(subgrad)
            if grad_norm > 1e-10:
                eta = min(eta, 0.1 / (grad_norm + 1e-10))  # Prevent too large steps

        # Update h
        h_new = h_solved - eta * subgrad

        # Reflect to interval [0,1]
        h_new = np.clip(h_new, 0, 1)

        # Project to sum = total_sum
        h_projected = project_to_simplex(h_new)

        # Optional symmetry enforcement
        h_symmetrized = symmetrize_h(h_projected)
        h_projected = project_to_simplex(h_symmetrized)

        # Check improvement
        c5_new = evaluate_c5(h_projected)
        if c5_new < best_c5:
            best_h = h_projected.copy()
            best_c5 = c5_new

        # Check for convergence
        if step > 10 and abs(c5_new - best_c5) < 1e-7:
            break

        h_solved = h_projected.copy()

    # Optional binary rounding
    binary_h = h_solved.copy()
    binary_indices = np.argsort(binary_h)[::-1]  # descending
    binary_mask = np.zeros(n_points, dtype=int)
    for i in range(int(total_sum)):
        binary_mask[binary_indices[i]] = 1

    # Local swaps to improve
    def compute_c5_binary(h_bin):
        return evaluate_c5(h_bin)

    swapped = True
    while swapped:
        swapped = False
        for i in range(n_points):
            if binary_mask[i] == 1:
                for j in range(i+1, n_points):
                    if binary_mask[j] == 0:
                        # Try swap
                        temp_mask = binary_mask.copy()
                        temp_mask[i] = 0
                        temp_mask[j] = 1
                        c5_test = compute_c5_binary(temp_mask)
                        if c5_test < best_c5:
                            binary_mask = temp_mask.copy()
                            best_c5 = c5_test
                            swapped = True
                            break
        if not swapped:
            break

    # Final projection if needed
    if best_c5 > 0.3807:
        final_h = binary_mask.astype(float)
        # Ensure sum is correct
        final_h = final_h * (total_sum / np.sum(final_h))
        # Re-project
        final_h = project_to_simplex(final_h)
        if evaluate_c5(final_h) < best_c5:
            best_h = final_h.copy()
            best_c5 = evaluate_c5(final_h)

    return (best_h.tolist(), best_c5, n_points)
```
