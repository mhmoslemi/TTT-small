import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from cvxpy import *
import time

def generate_greedy_initial(n):
    # Greedy algorithm to generate initial binary vector
    x = np.zeros(n, dtype=int)
    ones = 0
    while ones < n // 2:
        best_i = -1
        best_min_p = -1
        for i in range(n):
            if x[i] == 0:
                # Compute potential p_k values if we add i
                temp_x = x.copy()
                temp_x[i] = 1
                temp_p = []
                for k in range(1, n):
                    p_k = sum(temp_x[j] * temp_x[j+k] for j in range(n-k))
                    temp_p.append(p_k)
                min_p = min(temp_p) if temp_p else 0
                if min_p > best_min_p:
                    best_min_p = min_p
                    best_i = i
        if best_i >= 0:
            x[best_i] = 1
            ones += 1
    return x

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Target is to find C5 < 0.3807
    best_c5 = float('inf')
    best_h = None
    best_n = None

    # Check different discretization levels
    n_values = [36, 40, 44, 48, 52, 56, 60, 64]

    for n in n_values:
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
        dx = 2.0 / n
        
        # Greedy initialization
        x_greedy = generate_greedy_initial(n)
        
        # Build MILP model
        x_vars = Variable(n, boolean=True)
        L = Variable(integer=True)
        
        # Constraints for x_vars sum to n//2
        constraints = [sum(x_vars) == n//2]
        
        # Add binarization constraints for y_{i,k}
        # This takes a trick of modeling through the triangular structure
        
        # We need constraints that link pairs P_k = sum_{i=0}^{n-k-1} x_i * x_{i+k} >= L
        # We'll try a different direct approach:
        
        # Explicitly attempt to minimize L using iterative guesses, through a refined approach:
        # For the current step, we'll directly run 2-stage optimization:
        # 1. Binary code via greedy
        # 2. Try continuous refine with analytic formula evaluation for maximum C5
        
        x_init = x_greedy.astype(float)
        x_init = x_init / np.sum(x_init) * (n/2)  # normalize
        h_candidate = x_init.tolist()
        current_c5 = evaluate_c5(h_candidate)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_candidate[:]
            best_n = n
            
        # Try to refine by gradient-based local search on a good binary seed
        # We do this locally for interpretation compatibility with run
        local_h = np.array(h_candidate)
        local_h = np.clip(local_h, 0, 1)
        
        def local_obj(h_vec):
            return evaluate_c5(h_vec)
            
        # Try L-BFGS-B with projected updates
        bounds = [(0, 1) for _ in range(n)]
        
        # Initial projection step (before optimization)
        local_sum = np.sum(local_h)
        local_h = local_h * (n/2) / local_sum
        
        # Run local refinement with steps
        for iter_refine in range(200):
            if (time.time() - start_time) > budget_s * 0.95:
                break
            # Info about gradient's computation via finite difference for the max term
            # but we'd just call correlate; for speed we'll reuse its logic
            
            # Simplified approach: one iteration of small step with gradient idea
            h_old = local_h.copy()
            
            # Compute citation pattern without recompute full correlations
            try:
                corr_vals = correlate(local_h, 1 - local_h, mode="full")
                max_corr = np.max(corr_vals)
                max_idx = np.argmax(corr_vals)
                
                # Perturb all directions slightly to try improve max
                perturbed = local_h.copy()
                step_size = 1e-3
                for i in range(n):
                    if local_h[i] + step_size <= 1 and local_h[i] - step_size >= 0:
                        perturbed[i] += step_size
                        new_c5 = evaluate_c5(perturbed)
                        if new_c5 < local_obj(local_h):
                            local_h[i] += step_size
                        else:
                            perturbed[i] -= step_size
                            if evaluate_c5(perturbed) < local_obj(local_h):
                                local_h[i] -= step_size
                        perturbed[i] = local_h[i]
                        
                # Re-normalize
                norm_factor = (n/2) / np.sum(local_h)
                local_h *= norm_factor
                
            except Exception as e:
                break
                
            if not np.allclose(local_h, h_old, rtol=1e-10):
                continue
            else:
                break
                
        refined_h = local_h.tolist()
        refined_c5 = evaluate_c5(refined_h)
        
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h[:]
            best_n = n
            
    return (best_h, best_c5, best_n)
