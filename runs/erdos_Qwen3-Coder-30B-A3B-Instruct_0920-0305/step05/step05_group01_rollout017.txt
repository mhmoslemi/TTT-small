import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def adaptive_anneal(h0, run_seconds, rng):
    """Enhanced simulated annealing with adaptive cooling and different move types"""

    current_c5 = evaluate_c5(h)

    # Track recent improvements

            if recent_mean < 1e-3:
                move_type = "random"
            elif recent_mean > 1e-2:
                move_type = "guided"
            else:
                move_type = "adaptive"
        
        # Generate move based on strategy
        if move_type == "guided":
            # Guided move based on worst lag
            corr_full = compute_corr_fft(h)
            worst_lag = np.argmax(corr_full)

            i, j = int(np.random.randint(0, len(h))), int(np.random.randint(0, len(h)))
            # Better approach: look at local regions near worst lag
            if worst_lag < len(h):
                local_indices = [max(0, worst_lag - 2), max(0, worst_lag - 1), 
                               worst_lag, min(len(h)-1, worst_lag + 1), 
                               min(len(h)-1, worst_lag + 2)]
                i = np.random.choice(local_indices)
                # Select j from a wider local support
                j_range = range(max(0, min(len(h)-1, i - 4)), 
                              min(len(h), max(0, i + 4) + 1))
                j = np.random.choice(j_range)
            else:
                # Random indices
                i, j = np.random.choice(len(h), 2, replace=False)
                
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max <= 0:
                continue
            delta = np.random.uniform(0, min(delta_max, 0.05))
            
        elif move_type == "random":
            # Pairwise moves
            i, j = np.random.choice(len(h), 2, replace=False)
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max <= 0:
                continue
            delta = np.random.uniform(0, min(delta_max, 0.02))
            
        else:  # adaptive
            # Ad-hoc move - make smaller adjustments and look for better local patterns
            i = np.random.randint(0, len(h))
            j = (i + np.random.randint(1, 4)) % len(h)
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max <= 0:
                continue
            delta = np.random.uniform(0, min(delta_max, 0.03))
            
        # Apply move
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # Project to constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        new_c5 = evaluate_c5(h_new)
        
        # Metropolis acceptance
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            h = h_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                recent_improvements.append(max(1e-10, current_c5 - best_c5))
                if len(recent_improvements) > max_improvements:
                    recent_improvements.pop(0)
        neighbor_count += 1
    
    return best_h, best_c5

def local_calibration(h, target_sum):
    """More precise local optimization for smooth fine-tuning"""
    def objective(h_vec):
        return evaluate_c5(h_vec)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0, 1) for _ in range(len(h))]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        if result.success:
            h_refined = result.x
            h_refined = project_to_sum_box(h_refined, target_sum)
            return h_refined, evaluate_c5(h_refined)
    except:
        pass
    return h, evaluate_c5(h)
