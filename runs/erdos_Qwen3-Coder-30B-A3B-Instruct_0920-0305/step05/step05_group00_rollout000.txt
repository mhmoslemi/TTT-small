import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def shift_zero(arr, k):
    """Shift array with zero padding"""
    if k > 0:
        return np.concatenate([np.zeros(k), arr[:-k]])
    elif k < 0:
        return np.concatenate([arr[-k:], np.zeros(-k)])
    else:
        return arr.copy()

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def max_corr_and_lag(h):
    """Find maximum correlation and corresponding lag"""
    corr = compute_corr_fft(h)
    max_idx = np.argmax(corr)
    max_corr = corr[max_idx]
    lag = max_idx - (len(h) - 1)
    return max_corr, lag

def evaluate_c5(h):
    """Evaluate C5 bound using dx * max_corr"""
    max_corr, _ = max_corr_and_lag(h)
    return dx * max_corr

def gradient_direction(h, lag, top_k=3):
    """Compute gradient direction for a specific lag, or average over top_k"""
    n = len(h)
    if top_k == 1:
        shift_plus = shift_zero(h, -lag)
        shift_minus = shift_zero(h, lag)
        direction = shift_plus + shift_minus - 1
    else:
        # Compute average direction over top-k worst lags
        corr = compute_corr_fft(h)
        # Get top k indices by correlation value (descending)
        top_idxs = np.argpartition(corr, -top_k)[-top_k:]
        direction = np.zeros_like(h)
        for idx in top_idxs:
            l = idx - (n - 1)
            shift_plus = shift_zero(h, -l)
            shift_minus = shift_zero(h, l)
            direction += (shift_plus + shift_minus - 1)
        direction /= top_k
    return direction

def subgradient_descent(h0, target_sum, dx, max_iter=3000, step0=0.3, time_limit=10.0, top_k=3):
    """Run projected subgradient descent"""
    h = h0.copy()
    best_h = h.copy()
    best_corr, _ = max_corr_and_lag(h)
    start = time.time()
    
    for it in range(max_iter):
        now = time.time()
        if now - start > time_limit:
            break
            
        # Compute max correlation and get corresponding lag
        corr = compute_corr_fft(h)
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        lag = max_idx - (len(h) - 1)
        
        # Compute gradient direction using top_k worst lags
        direction = gradient_direction(h, lag, top_k)
        
        # Step size schedule
        step = step0 / (np.sqrt(it + 1) + 1)
        
        # Update step
        h_new = project_to_sum_box(h + step * direction, target_sum)
        
        # Evaluate new correlation
        new_corr, _ = max_corr_and_lag(h_new)
        
        # Accept if improves
        if new_corr < best_corr:
            best_corr = new_corr
            best_h = h_new.copy()
        
        # Keep new if it reduces max_corr
        if new_corr <= max_corr:
            h = h_new
        else:
            # Occasionally accept worse solutions
            if np.random.rand() < 0.01:
                h = h_new
                
        # Occasionally add noise
        if np.random.rand() < 0.02:
            h = project_to_sum_box(h + np.random.normal(0, 0.01, len(h)), target_sum)
    
    return best_h, best_corr

def local_refine_slsqp(h0, target_sum):
    """Use SLSQP for local refinement"""
    bounds = [(0, 1) for _ in range(len(h0))]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, target_sum)
            return h
    except:
        pass
    return h0

def random_initial(n_points, target_sum, use_alternative=False):
    """Generate a random feasible initial vector"""
    if use_alternative and np.random.rand() < 0.5:
        # Binary pattern: 0s and 1s
        h = np.random.choice([0, 1], n_points)
        # Adjust to match sum target
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if diff > 0:
            # Need some 1s to be 0s
            sz = int(min(diff, np.sum(h)))
            if sz > 0:
                idxs = np.random.choice(np.where(h == 1)[0], sz, replace=False)
                h[idxs] = 0
        elif diff < 0:
            # Need some 0s to be 1s
            sz = int(min(-diff, np.sum(1-h)))
            if sz > 0:
                idxs = np.random.choice(np.where(h == 0)[0], sz, replace=False)
                h[idxs] = 1
        h = project_to_sum_box(h, target_sum)
    else:
        h = np.random.rand(n_points)
        h = project_to_sum_box(h, target_sum)
    return h
