import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_correlation(h):
    """Compute full correlation array for a binary vector"""
    n = len(h)
    corr = np.zeros(2*n - 1)
    for k in range(1-n, n):
        if k >= 0:
            corr[k+n-1] = np.sum(h[:n-k] * h[k:])
        else:
            corr[k+n-1] = np.sum(h[-k:] * h[:n+k])
    return corr

def update_correlation(corr, h, i, j, n):
    """Update correlation array after swapping h[i] and h[j]"""
    delta_corr = np.zeros(2*n - 1)
    # Compute changes for all lags
    for k in range(1-n, n):
        # Contribution of position i to lag k
        if i >= 0 and i+k < n:
            if i+k >= 0:
                delta_corr[k+n-1] += h[i] * h[i+k] - h[j] * h[j+k] if k >= 0 else h[i] * h[i+k] - h[j] * h[j+k]
            else:
                if i >= 0 and i+k >= 0:
                    delta_corr[k+n-1] += h[i] * h[i+k] - h[j] * h[j+k]
                else:
                    delta_corr[k+n-1] = 0
        elif i+k >= 0 and i < n:
            if i >= 0 and i+k < n:
                delta_corr[k+n-1] += h[i] * h[i+k] - h[j] * h[j+k]
            else:
                delta_corr[k+n-1] = 0
        # Contribution of position j to lag k
        if j >= 0 and j+k < n:
            if j+k >= 0:
                delta_corr[k+n-1] += h[j] * h[j+k] - h[i] * h[i+k] if k >= 0 else h[j] * h[j+k] - h[i] * h[i+k]
            else:
                if j >= 0 and j+k >= 0:
                    delta_corr[k+n-1] += h[j] * h[j+k] - h[i] * h[i+k]
                else:
                    delta_corr[k+n-1] = 0
        elif j+k >= 0 and j < n:
            if j >= 0 and j+k < n:
                delta_corr[k+n-1] += h[j] * h[j+k] - h[i] * h[i+k]
            else:
                delta_corr[k+n-1] = 0
    return corr + delta_corr

def find_worst_shifts(corr, tol=1e-10):
    """Return indices of worst shifts (where correlation is maximum)"""
    max_val = np.max(corr)
    return np.where(np.abs(corr - max_val) < tol)[0]

def generate_legendre_pattern(n):
    """Generate a Legendre-based binary pattern"""
    p = n // 2
    if p <= 1:
        return np.random.choice([0, 1], size=n, p=[0.5, 0.5])
    if not all(p % i for i in range(2, int(p**0.5)+1)):
        p = next(i for i in range(p, 1, -1) if all(i % j for j in range(2, int(i**0.5)+1)))
    R = set()
    for a in range(1, p):
        R.add((a*a) % p)
    h = np.zeros(n, dtype=np.int8)
    for i in range(p):
        if i in R:
            h[i] = 1
    h[p:] = h[:p]
    # Ensure exactly n/2 ones
    actual_ones = np.sum(h)
    if actual_ones > n//2:
        zeros = np.where(h == 0)[0]
        to_change = actual_ones - n//2
        h[zeros[:to_change]] = 1
    elif actual_ones < n//2:
        ones = np.where(h == 1)[0]
        to_change = n//2 - actual_ones
        h[ones[:to_change]] = 0
    return h

def generate_random_pattern(n):
    """Generate a random binary pattern"""
    h = np.zeros(n, dtype=np.int8)
    positions = np.random.permutation(n)
    h[positions[:n//2]] = 1
    return h

def evaluate_c5(h):
    """Evaluate C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx
