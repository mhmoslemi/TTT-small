import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Evaluate C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Simple projection: clip and scale
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x
    
    # Adjust by scaling the difference
    scale = target_sum / current_sum
    x = x * scale
    x = np.clip(x, 0, 1)
    return x

def random_local_move(h, step_size=0.05):
    """Make a small random local move"""
    i = np.random.randint(0, n_points)
    delta = np.random.uniform(-step_size, step_size)
    
    # Adjust h[i] and some neighbors to maintain smoothness
    h_new = h.copy()
    h_new[i] = h[i] + delta
    
    # Ensure we don't get too close to boundaries  
    if h_new[i] < 0:
        h_new[i] = 0
    elif h_new[i] > 1:
        h_new[i] = 1
        
    # Slightly adjust neighbors to maintain continuity
    for j in [i-1, i+1]:
        if 0 <= j < n_points:
            h_new[j] = h[j] + delta * 0.1
            h_new[j] = np.clip(h_new[j], 0, 1)
    
    return h_new

def gradient_based_move(h, step_size=0.01):

    grad = np.zeros(n_points)
    for i in range(n_points):

        h_plus = h.copy()
        h_plus[i] += step_size
        h_plus = project_to_sum_box(h_plus, TARGET_SUM)
        C_plus = evaluate_c5(h_plus)
        
        h_minus = h.copy()
        h_minus[i] -= step_size
        h_minus = project_to_sum_box(h_minus, TARGET_SUM)
        C_minus = evaluate_c5(h_minus)
        
        grad[i] = (C_plus - C_minus) / (2 * step_size)
    
    # Find indices with largest decrease in C5
    idx = np.argsort(grad)
    
    # Make small moves to reduce gradient
    h_new = h.copy()
    for i in range(min(5, len(idx))):
        j = idx[i]  # indices with most negative gradient
        h_new[j] = h[j] - grad[j] * 0.1
        h_new[j] = np.clip(h_new[j], 0, 1)
    
    return h_new
