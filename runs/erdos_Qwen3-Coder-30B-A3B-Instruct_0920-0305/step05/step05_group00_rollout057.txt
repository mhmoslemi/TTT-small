from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def guided_move(h, corr):
    """Guided move based on correlation"""
    # Get lag with max correlation
    max_lag = np.argmax(corr)
    
    # Find the maximum and minimum positions for transfer
    # This is simplified - in practice, we should compute the actual gradient
    # Here we use a greedy approach
    idx_min = np.argmin(h)
    idx_max = np.argmax(h)
    
    # Adjust for lag effect
    if max_lag < n_points:
        idx_min = max(0, min(n_points-1, idx_min - max_lag))
        idx_max = max(0, min(n_points-1, idx_max + max_lag))
    
    # Perform transfer
    delta = min(h[idx_min], 1 - h[idx_max])
    if delta <= 0:
        return None
        
    return (idx_min, idx_max, delta)
