def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h):
    """Compute gradient of C5 with respect to each point"""
    corr = compute_corr_fft(h)
    worst_lag = np.argmax(corr)
    
    # Gradient calculation directly from correlation
    grad = np.zeros_like(h)
    for i in range(len(h)):
        if i - worst_lag >= 0:
            grad[i] += h[i - worst_lag]
        if i + worst_lag < len(h):
            grad[i] += h[i + worst_lag]
    return grad

    """Better guided move using signed gradient"""
    grad = compute_gradient(h)
    # Find best pair to move
    # We look for i where grad[i] is minimum (trying to decrease cost by reducing h[i])
    # and j where grad[j] is maximum (trying to decrease cost by increasing h[j])
    idx_min = np.argmin(grad)
    idx_max = np.argmax(grad)
    
    if idx_min == idx_max:
        return None
    
    delta = min(h[idx_min], 1 - h[idx_max])
    if delta <= 0:
        return None
    
    return (idx_min, idx_max, delta)

    alpha = 0.01
    momentum = 0.9
    velocity = np.zeros_like(h)

        if i > 0 and (i % 100 == 0):
            # Reducing step size for finer tuning
            alpha *= 0.99
    
    return best_h, best_c5
