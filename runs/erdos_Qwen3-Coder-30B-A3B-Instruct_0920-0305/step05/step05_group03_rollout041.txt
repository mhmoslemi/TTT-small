import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project(v, S):
    """Project vector v onto the capped simplex with sum=S"""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2
        p = np.clip(v + lam, 0.0, 1.0)
        s = p.sum()
        if s > S:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2, 0.0, 1.0)

def compute_O_w_grad(h, alpha):
    """Compute the overlaps O_r, weights w, and gradient of smoothed objective"""
    n = len(h)
    dx = 2.0 / n
    c1 = correlate(h, np.ones(n), mode='full')
    c2 = correlate(h, h, mode='full')
    O = (c1 - c2) * dx
    O_max = np.max(O)
    w = np.exp(alpha * (O - O_max))
    w = w / w.sum()
    
    s = np.convolve(w, np.ones(n), mode='valid')
    g2 = correlate(w, h, mode='full')[n-1:2*n-1]
    g3 = correlate(w, h[::-1], mode='full')[n-1:2*n-1][::-1]
    g = dx * (s - g2 - g3)
    return O, w, g

    start_time = time.time()
    n = 84  # Upsampled from 42 to get more resolution

    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Global search using differential evolution
    def obj_func(h):
        return evaluate_c5(project(h, S))
    
    bounds = [(0, 1) for _ in range(n)]
    try:
        res = differential_evolution(obj_func, bounds, maxiter=30, popsize=12,
                                    mutation=0.8, recombination=0.7, seed=seed)
        if res.success:
            h_global = project(res.x, S)
            c5_global = evaluate_c5(h_global)
            if c5_global < best_c5:
                best_h = h_global.copy()
                best_c5 = c5_global
    except:
        pass

    # Local refinement with gradient-based method using smooth surrogate
    h = best_h.copy()
    alpha = 5.0
    eta = 0.5 * dx
    for iteration in range(5000):
        if (time.time() - start_time) > 0.97 * budget_s:
            break

        O, w, g = compute_O_w_grad(h, alpha)
        
        # Line search
        improved = False
        for factor in [1.0, 0.5, 0.25]:
            h_trial = project(h - factor * eta * g, S)
            h_trial = (h_trial + h_trial[::-1]) / 2.0
            h_trial = project(h_trial, S)
            val = evaluate_c5(h_trial)
            if val < best_c5:
                best_h = h_trial.copy()
                best_c5 = val
                h = h_trial.copy()
                improved = True
                break

        if not improved:

        # Anneal alpha

            h = (h + h[::-1]) / 2.0
            h = project(h, S)

    return (best_h.tolist(), best_c5, n)
