import numpy as np
from cvxpy import *
import time
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_binary_milp(n_points, time_limit_s):
    """Solve the MILP to find a binary optimal solution"""
    # Variables
    h = Variable(n_points, boolean=True)
    tau = Variable()
    
    # Constraints
    constraints = [tau >= 0, sum(h) == n_points / 2]
    
    # Precompute all pairs (i, i+k) for valid shifts k
    pairs_by_shift = {}
    for k in range(-n_points + 1, n_points):
        pairs = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                pairs.append((i, j))
        if pairs:
            pairs_by_shift[k] = pairs
    
    # Create auxiliary variables for each pair and shift
    y = {}
    for k, pair_list in pairs_by_shift.items():
        y[k] = Variable(len(pair_list), boolean=True)
        
    # Add product linearization constraints for each k
    for k, pair_list in pairs_by_shift.items():
        for idx, (i, j) in enumerate(pair_list):
            # y[i,k] <= h[i]
            constraints.append(y[k][idx] <= h[i])
            # y[i,k] <= h[j]
            constraints.append(y[k][idx] <= h[j])
            # y[i,k] >= h[i] + h[j] - 1
            constraints.append(y[k][idx] >= h[i] + h[j] - 1)
            
    # Define overlaps
    for k, pair_list in pairs_by_shift.items():
        o_k = sum(h[i] for i, _ in pair_list) - sum(y[k][i] for i in range(len(pair_list)))
        constraints.append(o_k <= tau)
        
    # Objective
    objective = Minimize(tau)
    prob = Problem(objective, constraints)
    
    # Solve MILP
    prob.solve(solver=CPLEX, time_limit=time_limit_s, verbose=False)
    
    if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
        return np.array(h.value).astype(float)
    else:
        # Fallback to feasibility check
        if prob.status in [INFEASIBLE, UNBOUNDED]:
            # Try a solution from the constraint set
            print(f"MILP failed with status {prob.status}. Fallback.")
        # Return a feasible random binary pattern that meets the area constraint
        h_feasible = np.zeros(n_points)
        indices = np.random.choice(n_points, n_points // 2, replace=False)
        h_feasible[indices] = 1.0
        return h_feasible

def hill_climb_binary(h_original):
    """Perform hill climb on binary mask"""
    h = h_original.copy()
    n = len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    # Compute all current overlaps
    def compute_all_overlaps(h_vec):
        dx = 2.0 / n
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        corr = corr * dx
        return corr

    # Compute correlations once and reuse
    corr_original = compute_all_overlaps(h)
    max_corr = np.max(corr_original)
    best_max = max_corr
    
    improves = 0

    for iteration in range(10000):  # At least 10k swaps or no improvements
        improved = False
        for i in range(n):
            if h[i] == 1:
                for j in range(n):
                    if h[j] == 0:
                        # Simulate swap: h[i] becomes 0, h[j] becomes 1
                        # Update correlation vector as efficiently as possible
                        h[i] = 0
                        h[j] = 1
                        
                        new_corr = compute_all_overlaps(h)
                        new_max = np.max(new_corr)
                        
                        if new_max < best_max:
                            best_max = new_max
                            best_h = h.copy()
                            improved = True
                            improves += 1
                        else:
                            # Revert back
                            h[i] = 1
                            h[j] = 0
                    if improved:
                        break
            if improved:
                break
        if not improved:
            break

    # Return best solution from this search
    return best_h, best_c5

def project_onto_simplex(v, target_sum):
    """Euclidean projection onto simplices (box and sum constraint)"""
    # Use iterative water-filling to project onto the capped simplex {x : 0≤x≤1, sum x = target_sum}
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    
    # Find the threshold
    cumsum = np.cumsum(v_sorted)
    threshold = (cumsum - target_sum) / np.arange(1, n + 1)
    pos = np.argmax(v_sorted > threshold)
    if pos == 0:
        pos = n  # Everything can be set to target_sum/n
    else:
        pos -= 1  # index from the first one that is not above threshold
        
    theta = threshold[pos - 1] if pos > 0 else 0
    x = np.clip(v - theta, 0, 1)
    # Adjust total to sum to exactly target_sum
    x = x * target_sum / np.sum(x)
    return x

def continuous_refinement(h_initial):
    """Perform gradient-based refinement on a continuous step function"""
    h = h_initial.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Set up a minimal optimization
    max_iterations = 1000
    eta = 0.5 * dx  # step size
    epsilon = 1e-5 * dx
    no_improve_count = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for iteration in range(max_iterations):
        # Compute correlation of h with 1 - h
        corr = correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        max_k_indices = np.where(corr == max_val)[0]
        n_k = len(max_k_indices)
        
        # Get the shift index that corresponds to the peak
        # Compute shift (in the mode='full', center is at n-1)
        shift = max_k_indices[0] - (n_points - 1)  # shift at the last entry
        
        # Convolve with 1s for the gradient
        ones = np.ones(n_points)
        w = np.zeros(n_points)
        w[shift + n_points - 1] = 1.0  # for shift k
        g = 0.0  # total gradient to compute
        
        # Handle all active shifts if there is a tie
        for k_idx in max_k_indices:
            if k_idx < len(corr):
                shift_idx = k_idx - (n_points - 1)
                # Create a weight vector w
                w = np.zeros(n_points)
                w[shift_idx + n_points - 1] = 1.0
                # Compute what the gradient should be for this mode'd fluctuation
                # It's positive direction, opposite to shift
                if shift_idx < 0:
                    h_rev = np.flip(h[:len(h)+shift_idx])
                    w_sig = np.convolve(w, np.ones(n_points), mode='full')[n_points-1:]
                    w_sig = w_sig[:n_points]
                else:
                    h_rev = np.flip(h[shift_idx:])
                    w_sig = w[:len(h_rev)]
                g = np.convolve(w_sig, np.ones(len(h_rev)))[:len(h)]
        
        # More accurate gradient: weight by overlaps
        w_total = np.zeros(n_points)
        for k in max_k_indices:
            w_total[k] = 1.0
        
        # Gradient for all active shifts using convolution
        ones = np.ones(n_points)
        w_convolved = np.convolve(w_total, ones, mode='full')[n_points - 1:]
        w_convolved = w_convolved[:n_points]
        g = dx * (w_convolved - h - np.flip(h))  # approximate sub-gradient

        # Take a step
        h = h - eta * g
        
        # Project
        h = project_onto_simplex(h, n_points / 2)
        
        # Optional: symmetry
        h_sym = (h + h[::-1]) / 2.0
        h_sym = project_onto_simplex(h_sym, n_points / 2)
        h = h_sym
        
        # Evaluate new value
        new_c5 = evaluate_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if no_improve_count > 200:
            eta /= 2.0
            no_improve_count = 0
            if eta < epsilon:
                break
    
    return best_h, best_c5

def upsample_and_refine(h_coarse):
    """Double-grid refine with linear interpolation"""
    n = len(h_coarse)
    n_fine = 2 * n
    dx_fine = 2.0 / n_fine
    h_fine = np.interp(np.linspace(0, 2, n_fine), np.linspace(0, 2, n), h_coarse)
    # Re-normalize
    h_fine = h_fine * (n / 2) / np.sum(h_fine)
    # Refine
    h_refined, c5_refined = continuous_refinement(h_fine)
    return h_refined, c5_refined

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up the base discretization
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Phase 1: Binary MILP
    time_remaining = budget_s * 0.25  # Allocate first ~1/4 for MILP
    
    # Treat as a binary problem: solve the exact MILP formulation for optimal binary solutions
    h_bin = solve_binary_milp(n_points, time_remaining)
    c5_bin = evaluate_c5(h_bin)
    
    # Store the results for the binary phase
    best_h = h_bin.copy()
    best_c5 = c5_bin
    
    # Phase 2: Hill climbing on binary masks
    if (time.time() - start_time) < budget_s * 0.5:
        h_limited, c5_limited = hill_climb_binary(h_bin)
        if c5_limited < best_c5:
            best_h = h_limited.copy()
            best_c5 = c5_limited
    
    # Phase 3: Continuous refinement on the best binary or intermediate result
    if (time.time() - start_time) < budget_s * 0.8:
        h_cont, c5_cont = continuous_refinement(best_h)
        if c5_cont < best_c5:
            best_h = h_cont.copy()
            best_c5 = c5_cont
    
    # Optional: Upsample for better resolution if allowed
    if (time.time() - start_time) < budget_s * 0.95:
        try:
            h_upsampled, c5_upsampled = upsample_and_refine(best_h)
            if c5_upsampled < best_c5:
                best_h = h_upsampled.copy()
                best_c5 = c5_upsampled
        except:
            pass

    return (best_h.tolist(), best_c5, n_points)
