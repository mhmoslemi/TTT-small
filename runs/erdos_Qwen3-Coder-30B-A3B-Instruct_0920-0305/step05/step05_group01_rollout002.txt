TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def gradient_step(h, learning_rate=0.01):
    """Simple gradient descent step using finite differences"""
    n = len(h)
    grad = np.zeros(n)
    eps = 1e-5
    
    base_c5 = evaluate_c5(h)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_sum_box(h_plus, TARGET_SUM)
        c5_plus = evaluate_c5(h_plus)
        grad[i] = (c5_plus - base_c5) / eps
    
    h_new = h - learning_rate * grad
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    return h_new

    patience = 0

    while time.time() - start_time < max_time:
        # Try a step
        h_new = gradient_step(h, lr)
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            best_h = h.copy()
            best_c5 = current_c5
            patience = 0
        else:
            patience += 1
            if patience > 5:
                # Reduce learning rate
                lr *= 0.5
                patience = 0

            c5_trial = evaluate_c5(h_trial)
            if c5_trial < current_c5:
                h = h_trial
                current_c5 = c5_trial

                break  # Convergence
