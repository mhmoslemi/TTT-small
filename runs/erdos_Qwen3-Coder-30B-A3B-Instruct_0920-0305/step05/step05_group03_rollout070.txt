import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def project(v, S):
    """Project v onto the capped simplex: 0 ≤ v_i ≤ 1, sum v_i = S"""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2
        p = np.clip(v + lam, 0.0, 1.0)
        s = p.sum()
        if s > S:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2, 0.0, 1.0)

def compute_O_w_grad(h, alpha):
    """Compute overlaps, weights, and gradient of log-sum-exp of overlaps"""
    n = len(h)
    dx = 2.0 / n
    c1 = correlate(h, np.ones(n), mode='full')
    c2 = correlate(h, h, mode='full')
    O = (c1 - c2) * dx
    max_O = np.max(O)
    w = np.exp(alpha * (O - max_O))
    w = w / w.sum()
    s = np.convolve(w, np.ones(n), mode='valid')
    g2 = correlate(w, h, mode='full')[n-1:2*n-1]
    g3 = correlate(w, h[::-1], mode='full')[n-1:2*n-1][::-1]
    g = dx * (s - g2 - g3)
    return O, w, g

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    n_points = len(h0) * 2  # upsample by 2
    h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
    h0 = project(h0, n_points / 2.0)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Global search with differential evolution
    if (time.time() - start_time) < 0.1 * budget_s:
        def obj_de(x):
            x = project(x, n_points / 2.0)
            return evaluate_c5(x)
        bounds = [(0, 1) for _ in range(n_points)]
        res = differential_evolution(obj_de, bounds, maxiter=30, popsize=12, 
                                    mutation=0.8, recombination=0.7, seed=seed)
        if res.success:
            h_de = project(res.x, n_points / 2.0)
            c5_de = evaluate_c5(h_de)
            if c5_de < best_c5:
                best_h = h_de.copy()
                best_c5 = c5_de

    # Local refinement with projected gradient descent
    h = best_h.copy()
    alpha = 5.0
    eta = 0.5 * (2.0 / n_points)
    iterations = 0
    max_iter = 5000
    
    for i in range(max_iter):
        if (time.time() - start_time) > 0.97 * budget_s:
            break
            
        O, w, g = compute_O_w_grad(h, alpha)
        h_trial = project(h - eta * g, n_points / 2.0)
        h_trial = (h_trial + h_trial[::-1]) / 2.0
        h_trial = project(h_trial, n_points / 2.0)
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
            h = h_trial.copy()
            iterations = 0
        else:
            iterations += 1
            if iterations > 100:
                eta *= 0.5
                iterations = 0
                
        if i % 200 == 0:
            alpha = min(alpha * 1.5, 50.0)
            
        if i % 500 == 0 and not (c5_trial < best_c5):
            h = project(h + 0.02 * np.random.randn(n_points), n_points / 2.0)
            h = (h + h[::-1]) / 2.0
            h = project(h, n_points / 2.0)
    
    return (best_h.tolist(), best_c5, n_points)
