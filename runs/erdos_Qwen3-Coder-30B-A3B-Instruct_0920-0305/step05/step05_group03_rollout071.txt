import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def project(v, S):
    """Project v onto the capped simplex: 0 <= v[i] <= 1, sum(v) = S"""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        p = np.clip(v + lam, 0.0, 1.0)
        s = p.sum()
        if s > S:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def compute_O_w_grad(h, alpha):
    """Compute overlap vector O, weights w, and gradient g for smooth surrogate."""
    n = len(h)
    dx = 2.0 / n
    c1 = correlate(h, np.ones(n), mode='full')
    c2 = correlate(h, h, mode='full')
    O = (c1 - c2) * dx
    max_O = np.max(O)
    w = np.exp(alpha * (O - max_O))
    w = w / w.sum()
    s = np.convolve(w, np.ones(n), mode='valid')
    g2 = correlate(w, h, mode='full')[n-1:2*n-1]
    g3 = correlate(w, h[::-1], mode='full')[n-1:2*n-1][::-1]
    g = dx * (s - g2 - g3)
    return O, w, g

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 84   # Up-sampled from 42 for better resolution
    h0 = np.array(initial_h_values)
    # Upsample to n_points
    h_up = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
    h0 = h_up
    # Project to satisfy the integral constraint (sum = n/2)
    S = n_points / 2.0
    h0 = project(h0, S)
    best_h = h0.copy()
    best_val = evaluate_c5(h0)

    # Optional: Global search with Differential Evolution
    def obj_to_minimize(x):
        x = project(x, S)
        return evaluate_c5(x)

    try:
        # DE with small population size
        de_res = differential_evolution(
            obj_to_minimize,
            [(-0.05, 1.05)] * n_points,
            maxiter=30,
            popsize=12,
            mutation=0.8,
            recombination=0.7,
            seed=seed+100,
            atol=1e-12,
            disp=False
        )
        if de_res.success:
            de_h = project(de_res.x, S)
            val = evaluate_c5(de_h)
            if val < best_val:
                best_h = de_h.copy()
                best_val = val
    except:
        pass

    # Local refinement with gradient descent on smooth surrogate
    h = best_h.copy()
    alpha = 5.0
    eta = 0.5 * (2.0 / n_points)
    iteration = 0
    max_iter = 5000
    last_improvement = 0

    for i in range(max_iter):
        if (time.time() - start_time) > 0.97 * budget_s:
            break
        iteration += 1
        O, w, g = compute_O_w_grad(h, alpha)
        # Try different step sizes
        for factor in [1.0, 0.5, 0.25]:
            h_trial = project(h - factor * eta * g, S)
            # Symmetrize if beneficial
            h_trial = (h_trial + h_trial[::-1]) / 2.0
            h_trial = project(h_trial, S)
            val = evaluate_c5(h_trial)
            if val < best_val:
                best_h = h_trial.copy()
                best_val = val
                h = h_trial.copy()
                last_improvement = iteration
                break
        else:
            # No improvement
            eta *= 0.5

        # Annealing schedule for alpha
        if iteration % 200 == 0:
            alpha = min(alpha * 1.5, 50)

        # Restart periodically with small perturbation if no improvement
        if iteration % 500 == 0 and iteration > last_improvement + 400:
            h = project(h + 0.02 * np.random.randn(n_points), S)
            h = (h + h[::-1]) / 2.0
            h = project(h, S)

    # Final projection and evaluation to ensure correctness
    h_final = project(best_h, S)
    final_val = evaluate_c5(h_final)

    return (h_final.tolist(), final_val, n_points)
