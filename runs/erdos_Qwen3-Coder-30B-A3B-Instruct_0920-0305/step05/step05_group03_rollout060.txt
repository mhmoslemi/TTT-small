import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project(v, S):
    """
    Project vector v onto the capped simplex:
    0 <= v_i <= 1, sum(v) = S
    """
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2
        p = np.clip(v + lam, 0.0, 1.0)
        s = p.sum()
        if s > S:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2, 0.0, 1.0)

def compute_O_w_grad(h, alpha):
    """Compute overlaps O_r and gradient of log-sum-exp smooth objective."""
    n = len(h)
    dx = 2.0 / n
    # Compute c1 and c2 for overlap calculation
    c1 = correlate(h, np.ones(n), mode='full')
    c2 = correlate(h, h, mode='full')
    O = (c1 - c2) * dx  # overlap for each shift
    O_max = O.max()
    # Compute weights for smooth max
    w = np.exp(alpha * (O - O_max))
    w = w / w.sum()  # normalized weights
    # Compute gradient components
    s = np.convolve(w, np.ones(n), mode='valid')
    g2 = correlate(w, h, mode='full')[n-1:2*n-1]
    g3 = correlate(w, h[::-1], mode='full')[n-1:2*n-1][::-1]
    grad = dx * (s - g2 - g3)
    return O, w, grad

    h0 = project(h0_upsampled, S)
    best_h = h0.copy()
    best_val = evaluate_c5(h0)
    
    # Global search with DE
    def de_objective(x):
        x_proj = project(x, S)
        return evaluate_c5(x_proj)
    
    # Run a short DE to find diverse candidates
    bounds = [(0, 1) for _ in range(n)]
    de_res = differential_evolution(
        de_objective,
        bounds,
        maxiter=30,
        popsize=12,
        mutation=0.8,
        recombination=0.7,
        seed=seed
    )
    if hasattr(de_res, 'success') and de_res.success:
        x_best = project(de_res.x, S)
        val = evaluate_c5(x_best)
        if val < best_val:
            best_h = x_best.copy()
            best_val = val
    
    # Refinement using projected gradient with smoothing (continuation)
    h = best_h.copy()
    alpha = 5.0
    eta = 0.5 * dx
    max_iter = 5000
    improved = False
    
    for iteration in range(max_iter):
        if time.time() - start_time > 0.97 * budget_s:
            break
            
        # Compute gradient
        O, w, grad = compute_O_w_grad(h, alpha)
        
        # Line search
        accept = False
        for factor in [1.0, 0.5, 0.25]:
            h_trial = project(h - factor * eta * grad, S)
            # Apply symmetry
            h_trial = (h_trial + h_trial[::-1]) / 2

        if not accept:

        if iteration % 500 == 0 and not improved:

            h = project(h, S)
            improved = False
