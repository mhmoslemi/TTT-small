import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from cvxpy import *
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    dx = 2.0 / n_points
    
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0)
    
    # Define discrete search range for n
    n_range = [36, 40, 44, 48, 52, 56, 60, 64]
    
    # To store best solution found across all n
    overall_best_h = best_h.copy()
    overall_best_c5 = best_c5
    overall_best_n = n_points

    # For each n (resolution)
    for n in n_range:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute dx
        dx = 2.0 / n
        
        # Build binary MILP model
        # Variables: x_i (binary), y_{i,k} (binary), L (integer)
        x = cp.Variable(n, boolean=True)
        y = cp.Variable((n, n), boolean=True)  # y[i,k] = x[i]*x[i+k]
        L = cp.Variable(integer=True)
        
        # Objective
        objective = cp.Maximize(L)
        
        constraints = []
        
        # Sum constraint (integral = 1)
        constraints.append(cp.sum(x) == n / 2)
        
        # Symmetry breaking: fix first element to 1
        constraints.append(x[0] == 1)
        
        # Left-heavy symmetry: assume left half has more or equal ones than right
        constraints.append(cp.sum(x[:n//2]) >= cp.sum(x[n//2:]))
        
        # Product constraints y[i,k] = x[i]*x[i+k]
        for k in range(1, n):
            for i in range(n - k):
                # y[i,k] <= x[i]
                constraints.append(y[i,k] <= x[i])
                # y[i,k] <= x[i+k]
                constraints.append(y[i,k] <= x[i+k])
                # y[i,k] >= x[i] + x[i+k] - 1
                constraints.append(y[i,k] >= x[i] + x[i+k] - 1)
        
        # Minimum ordered pair constraint: sum y_{i,k} >= L for all k
        for k in range(1, n):
            constraints.append(cp.sum(y[:,k]) >= L)
        
        # Create problem
        prob = cp.Problem(objective, constraints)
        
        # Solve with timeout
        solve_time = min(30, (budget_s - (time.time() - start_time)) * 0.5)
        try:
            prob.solve(solver=cp.CBC, time_limit=solve_time)
        except Exception as e:
            continue
            
        # Extract solution if successful
        if prob.status == cp.OPTIMAL:
            L_val = L.value
            x_sol = np.round(x.value).astype(int)
            # Compute C5 for this solution
            h_sol = x_sol.astype(float)
            new_c5 = evaluate_erdos_solution(h_sol)
            if new_c5 < overall_best_c5:
                overall_best_c5 = new_c5
                overall_best_h = h_sol
                overall_best_n = n
        elif prob.status == cp.INFEASIBLE:
            # suboptimal solution found
            try:
                # Try to extract heuristically feasible solution from existing warmstart
                x_sol = np.ones(n, dtype=int)
                x_sol[n//2:] = 0  # More balanced distribution
                h_sol = x_sol.astype(float)
                new_c5 = evaluate_erdos_solution(h_sol)
                if new_c5 < overall_best_c5:
                    overall_best_c5 = new_c5
                    overall_best_h = h_sol
                    overall_best_n = n
            except:
                continue
        else:
            continue
            
    
    # Final polishing step
    if time.time() - start_time < budget_s * 0.98:
        # Convert best solution to correct form for refinement
        polished_h = overall_best_h.copy()
        dx_refine = 2.0 / overall_best_n
        
        def ref_eval(h_vals):
            h_np = np.array(h_vals)
            corr = correlate(h_np, 1 - h_np, mode="full")
            return -np.max(corr) * dx_refine
        
        def constraint_eq(h_vals):
            return np.sum(h_vals) * dx_refine - 1.0
        
        bounds = [(0, 1) for _ in range(overall_best_n)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        # Local refinement
        res = minimize(
            ref_eval, 
            polished_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        
        if res.success:
            polished_h = res.x
            polished_c5 = -ref_eval(polished_h)
            if polished_c5 < overall_best_c5:
                overall_best_c5 = polished_c5
                overall_best_h = polished_h
                overall_best_n = overall_best_n
    
    # Final verification
    final_c5 = evaluate_erdos_solution(overall_best_h)
    
    return (overall_best_h.tolist(), final_c5, overall_best_n)
