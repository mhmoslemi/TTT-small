import numpy as np

def conv_fft(a, b):
    """Compute linear convolution using FFT with zero padding"""
    n = len(a)
    m = len(b)
    size = n + m - 1
    a_padded = np.pad(a, (0, size - n), mode='constant')
    b_padded = np.pad(b, (0, size - m), mode='constant')
    return np.fft.irfft(np.fft.rfft(a_padded) * np.fft.rfft(b_padded))

def project_onto_bounded_simplex(h):
    """Project h onto the set {x: 0 <= x_i <= 1, sum x_i = n/2}"""
    n = len(h)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    target = n / 2.0
    diff = target - sum_h
    
    if abs(diff) < 1e-12:
        return h.copy()
    
    # bisection for lambda
    min_lam = -1.0
    max_lam = 1.0
    for _ in range(30):
        lam = (min_lam + max_lam) / 2.0
        proposed = h + lam
        new_sum = np.sum(np.clip(proposed, 0, 1))
        if new_sum < target:
            max_lam = lam
        else:
            min_lam = lam
    lam = (min_lam + max_lam) / 2.0
    return np.clip(h + lam, 0, 1)

def soft_max_weights(c, beta):
    """Compute softmax weights w_k = exp(-beta * c_k) / sum(exp(-beta * c_t))"""
    c_max = np.max(c)
    exp_c = np.exp(-beta * (c - c_max))
    return exp_c / np.sum(exp_c)

def smooth_obj_and_grad(h, beta):
    """Compute smooth objective and its gradient"""
    # Compute correlation (convolution h with h)
    c = conv_fft(h, h)
    c = c[1:]  # exclude the zero shift
    # Weighted gradient
    w = soft_max_weights(c, beta)
    
    # Gradient computation
    g_forward = conv_fft(h, w)
    g_backward = conv_fft(h, np.flip(w))
    grad = -(g_forward + g_backward)[1:]  # One-indexing
    
    # Objective value
    c_max = np.max(c)
    obj = 1.0 + (1.0 / beta) * np.log(np.sum(np.exp(-beta * (c - c_max))))
    
    return obj, grad

def projected_gradient_descent(h0, beta0=1.0, beta_max=200.0, alpha0=0.1, gamma=2.0, delta=0.9, max_iter=1000):
    h = h0.copy()
    beta = beta0
    alpha = alpha0
    
    for i in range(max_iter):
        # Compute current objective and gradient
        obj, grad = smooth_obj_and_grad(h, beta)
        
        # Stepsize line search (Armijo condition)
        tau = alpha
        h_next = project_onto_bounded_simplex(h - tau * grad)
        obj_new, _ = smooth_obj_and_grad(h_next, beta)
        
        max_backtrack = 20
        for _ in range(max_backtrack):
            if obj_new <= obj - 0.1 * tau * np.dot(grad, grad):
                break
            tau = tau / 2.0
            h_next = project_onto_bounded_simplex(h - tau * grad)
            obj_new, _ = smooth_obj_and_grad(h_next, beta)
        
        # Update
        h = h_next
        alpha = alpha * delta if alpha > 1e-12 else 1e-12
        
        # Check convergence
        if np.linalg.norm(tau * grad) < 1e-9:
            break
        
        # Update beta (continuation)
        beta = min(beta * gamma, beta_max)
        if beta > beta_max:
            break
    
    return h

def evaluate_c5_exact(h):
    """Evaluate the exact C5 constant using full correlation"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx
