import numpy as np

def compute_gradient(h, w, dx):
    n = len(h)
    grad = np.zeros_like(h)
    offset = n - 1
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        shift = idx - offset
        if shift >= 0:
            # first term: (1 - h[j + shift])
            grad[:n - shift] += wk * (1 - h[shift:])
            # second term: -h[j - shift]
            grad[shift:] -= wk * h[:n - shift]
        else:
            # shift < 0
            s = -shift
            # first term: (1 - h[j + shift]) where shift < 0 implies j + shift >= 0
            grad[s:] += wk * (1 - h[:n - s])
            # second term: -h[j - shift] where j - shift >= 0 implies j >= s
            grad[:n - s] -= wk * h[s:]
    return grad * dx

def project_to_constraints(v, lower=0.0, upper=1.0, target_sum=None):
    if target_sum is None:
        target_sum = len(v) / 2.0
    lo = -np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, lower, upper)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo + hi) / 2.0, lower, upper)
    return x

def loss_and_grad(h, beta, dx):
    p = 1 - h
    c = np.correlate(h, p, mode='full') * dx
    scaled = beta * c
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w_sum = np.sum(w)
    w = w / w_sum
    loss = (max_scaled + np.log(w_sum)) / beta

    grad = compute_gradient(h, w, dx)
    return loss, grad
