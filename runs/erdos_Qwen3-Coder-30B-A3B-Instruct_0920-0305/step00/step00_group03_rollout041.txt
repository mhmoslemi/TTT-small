def project_onto_simplex_box(h, target_sum):
    """Project h onto box [0,1]^N and simplex {sum h_i = target_sum}"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Simple redistribution: scale and clip
    # But keep it numerically stable using a helper
    # We'll do a simple approach
    if diff > 0:
        # Need to add mass
        # Find indices where we can add (h < 1)
        mask = h < 1
        if np.sum(mask) == 0:
            return h  # All at upper bound, can't add
        # Distribute diff uniformly among free spots
        delta = diff / np.sum(mask)
        h[mask] = np.minimum(1, h[mask] + delta)
    elif diff < 0:
        # Need to remove mass
        # Find indices where we can remove (h > 0)
        mask = h > 0
        if np.sum(mask) == 0:
            return h  # All at lower bound, can't remove
        # Distribute diff uniformly among free spots
        delta = -diff / np.sum(mask)
        h[mask] = np.maximum(0, h[mask] - delta)
    
    return h
