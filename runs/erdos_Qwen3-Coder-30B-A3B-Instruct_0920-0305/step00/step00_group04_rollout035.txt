def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)

    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value of the correlation gives C5
    c5 = np.max(corr) * dx
    return c5
