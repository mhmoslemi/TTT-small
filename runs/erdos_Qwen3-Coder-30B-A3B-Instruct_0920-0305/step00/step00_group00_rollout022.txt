import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize
import time

def project_to_constraints(v, target_sum=0.5):
    # Project onto box [0,1] and sum = target_sum
    # Binary search for lambda
    lo = -np.max(v) - 1.0
    hi = 1 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    return x

def compute_gradient(h, w, dx):
    n = len(h)
    grad = np.zeros_like(h)
    offset = n - 1
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        shift = idx - offset
        if shift >= 0:
            # First term: grad[i] += wk * (1 - h[i + shift])
            # Indices from 0 to n - shift - 1
            grad[:n - shift] += wk * (1 - h[shift:])
            # Second term: grad[i] -= wk * h[i - shift]
            # Indices from shift to n - 1
            grad[shift:] -= wk * h[:n - shift]
        else:
            # shift < 0
            s = -shift
            # First term: grad[i] += wk * (1 - h[i + shift])
            # i + shift >= 0 => i >= -shift = s
            grad[s:] += wk * (1 - h[:n - s])
            # Second term: grad[i] -= wk * h[i - shift]
            # i - shift = i + s <= n - 1 => i <= n - s - 1
            grad[:n - s] -= wk * h[s:]
    return grad * dx

def loss_and_grad(h, beta, dx):
    p = 1 - h
    c = np.correlate(h, p, mode='full') * dx
    # Compute smooth approximation to max
    scaled = beta * c
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w_sum = np.sum(w)
    w = w / w_sum
    loss = (max_scaled + np.log(w_sum)) / beta

    grad = compute_gradient(h, w, dx)
    return loss, grad

def upscale_h(old_h, new_n):
    old_n = len(old_h)
    old_x = np.linspace(0, 2, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0, 2, new_n, endpoint=False) + 1.0 / new_n
    h_new = np.interp(new_x, old_x, old_h)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed
    rng = np.random.default_rng(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 140
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # sum(h) * dx = 1
    
    # Get initial solution if available
    try:
        initial_h_values
    except NameError:
        # fallback to random if not available  
        initial_h_values = None
        
    best_h = None
    best_c5 = np.inf
    
    # Initialize without any warm start to be safe
    max_restarts = 15
    for restart in range(max_restarts):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
        # Sample from uniform random
        h = rng.random(n_points)
        h = project_to_constraints(h, target_sum)
        
        # Optimize with SLSQP
        beta_schedule = [1.0, 5.0, 15.0, 30.0]
        for beta in beta_schedule:
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.95:
                break
            # Define objective
            def fun(x):
                loss, grad = loss_and_grad(x, beta, dx)
                return loss, grad
            
            # Constraints and bounds
            cons = ({'type':'eq', 'fun': lambda x: np.sum(x) - target_sum,
                     'jac': lambda x: np.ones_like(x)})
            bounds = [(0.0, 1.0)] * len(h)

            # Optimize
            try:
                result = minimize(fun, h, method='SLSQP', jac=True, bounds=bounds,
                                constraints=cons, options={'maxiter': 100, 'ftol':1e-9, 'disp':False})
                h = result.x.copy()
            except:
                pass  # If SLSQP fails, continue with gradient descent
        
        # Compute actual value for final check
        c = np.correlate(h, 1 - h, mode='full') * dx
        actual_c5 = c.max()
        if actual_c5 < best_c5:
            best_c5 = actual_c5
            best_h = h.copy()
        
        # Randomized restart part: slightly perturb solution
        if restart < max_restarts - 1:
            noise_scale = 0.05
            noise = rng.normal(0, noise_scale, n_points)
            noisy_h = h + noise
            h = project_to_constraints(noisy_h, target_sum)
    
    # Evaluate final solution with the provided function
    try:
        evaluate_erdos_solution
        final_c5 = evaluate_erdos_solution(best_h)
    except NameError:
        # Fallback if evaluate_erdos_solution not imported
        c = np.correlate(best_h, 1 - best_h, mode='full') * dx
        final_c5 = c.max()
    
    return (best_h.tolist(), final_c5, n_points)
