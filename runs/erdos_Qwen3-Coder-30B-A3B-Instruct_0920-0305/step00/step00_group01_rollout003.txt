def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given step function h."""

    conv = np.correlate(h, 1 - h, mode='full')
    # The maximum value corresponds to the C5
    max_conv = np.max(conv)
    return max_conv * dx
