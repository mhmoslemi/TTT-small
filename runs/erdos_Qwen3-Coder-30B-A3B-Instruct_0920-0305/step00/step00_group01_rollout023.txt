def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value across all shifts
    c5 = np.max(correlation) * dx
    return c5
