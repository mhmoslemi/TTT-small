def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value gives us C5
    c5 = np.max(correlation) * dx
    return c5
