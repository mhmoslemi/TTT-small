import numpy as np
from scipy.optimize import differential_evolution, minimize
import time
from math import sqrt

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h values"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Maximize over all possible shifts
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

def initial_h_values():
    """Initial construction from the problem description"""
    n = 70
    h = np.zeros(n)
    # Use a simple construction
    # We place a step function with a plateau
    h[10:30] = 1.0
    h[30:40] = 0.0
    h[40:60] = 1.0
    h[60:70] = 0.0
    # Normalize
    h = h * 2.0 / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Starting point
    h_original = initial_h_values()
    n_points = len(h_original)
    dx = 2.0 / n_points
    
    # Normalize to required sum
    sum_h = np.sum(h_original)
    h_original = h_original * (n_points / 2) / sum_h
    
    best_h = h_original.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    elapsed = 0
    
    # Try different optimization strategies
    
    # Strategy 1: Optimize with bounds and constraints directly
    def objective(x):
        h = np.copy(x)
        # clip to [0,1]
        h = np.clip(h, 0, 1)
        # enforce the constraint sum(h) * dx = 1 => sum(h) = n_points/2
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_erdos_solution(h)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Bounds for each variable in [0,1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use differential evolution to get reasonable starting point
    try:
        # Differential evolution with subset of points to avoid long computation
        de_bounds = [(0, 1) for _ in range(min(50, n_points))]
        
        result_de = differential_evolution(
            objective, 
            de_bounds,
            maxiter=200,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            disp=False
        )
        
        if elapsed + 30 < budget_s:
            # Then try local optimization with initial result
            x0 = np.clip(result_de.x, 0, 1)
            # Normalize
            x0 = x0 * (n_points / 2) / np.sum(x0)
            
            # Optimize with more iterate steps if time allows
            for i in range(200):
                if elapsed > budget_s - 10:
                    break
                # Just try to improve current points with basic gradient-free method
                # Not using sophisticated optimizers for simplicity and speed
                current_h = x0.copy()
                current_c5 = evaluate_erdos_solution(current_h)
                
                # Random perturbation approach
                idx = np.random.randint(0, n_points)
                perturbation = np.random.normal(0, 0.02)
                new_h = current_h.copy()
                new_h[idx] += perturbation
                new_h = np.clip(new_h, 0, 1)
                # Renormalize
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                new_c5 = evaluate_erdos_solution(new_h)
                
                if new_c5 < current_c5:
                    x0 = new_h
                    current_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = new_h.copy()
                
                elapsed = time.time() - start_time
                if elapsed > budget_s - 10:
                    break
    except Exception as e:
        pass
    
    # Try simple greedy local search
    
    if elapsed < budget_s - 10:  # Leave buffer
        # Try to improve further with manual local search
        current_h = best_h.copy()
        current_c5 = best_c5
        
        # Try batch local improvement
        local_iters = 1000
        for i in range(local_iters):
            if elapsed > budget_s - 5:
                break
            
            idx = np.random.randint(0, n_points)
            step = np.random.normal(0, 0.01)
            
            # Try perturbing one element at a time
            new_h = current_h.copy()
            new_h[idx] += step
            new_h = np.clip(new_h, 0, 1)
            # Renormalize
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            new_c5 = evaluate_erdos_solution(new_h)
            
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
                    
            elapsed = time.time() - start_time
            if elapsed > budget_s - 5:
                break
    
    # Backup to original if nothing better found
    if best_c5 >= evaluate_erdos_solution(initial_h_values()):
        best_h = initial_h_values()
        best_c5 = evaluate_erdos_solution(best_h)
        
    return best_h, best_c5, n_points
