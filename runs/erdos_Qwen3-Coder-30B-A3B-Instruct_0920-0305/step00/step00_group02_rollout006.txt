import numpy as np
import cvxpy as cp
import time
import random
from scipy.sparse import diags, csr_matrix
from scipy.sparse.linalg import eigs

def build_shift_matrices(n):
    """Build sparse shift matrices A_k for k = 1 to n//2"""
    A_k_list = []
    for k in range(1, n//2 + 1):
        # Create the off-diagonal entries for shift k
        row = np.arange(n - k)
        col = np.arange(k, n)
        data = np.ones(n - k)
        A_k = csr_matrix((data, (row, col)), shape=(n, n))
        A_k = A_k + A_k.T  # Symmetric matrix
        A_k_list.append(A_k)
    return A_k_list

def round_vector(X, n, r=8):
    """Rounds the psd matrix X to a binary vector using random hyperplane rounding"""
    # Perform eigen-decomposition
    try:
        V, _ = eigs(X, k=r, which='LM')
    except:
        # Fall back to SVD if eigs fails
        U, s, Vt = np.linalg.svd(X)
        V = U[:, :r]
    
    # Sample random direction
    g = np.random.randn(r)
    scores = V @ g
    
    # Find threshold to get exactly n/2 ones
    sorted_scores = np.sort(scores)[::-1]
    threshold = sorted_scores[n//2 - 1]
    h = (scores >= threshold).astype(float)
    
    # Ensure exact sum
    diff = int(np.sum(h) - n//2)
    if diff > 0:
        # Turn some 1s to 0s
        ones_indices = np.where(h == 1)[0]
        flip_indices = np.random.choice(ones_indices, diff, replace=False)
        h[flip_indices] = 0
    elif diff < 0:
        # Turn some 0s to 1s
        zeros_indices = np.where(h == 0)[0]
        flip_indices = np.random.choice(zeros_indices, -diff, replace=False)
        h[flip_indices] = 1
        
    return h

def local_improvement(h, n, A_k_list, dx, max_swaps=100000):
    """Perform local hill-climbing swap moves to improve worst overlap"""
    # Compute initial overlaps
    h_extended = np.concatenate([h, [0] * (n-1)])
    overlap_full = np.correlate(h_extended, h, mode='full')
    overlaps = overlap_full[n-1:]
    min_k = np.argmin(overlaps)
    
    swaps = 0
    while swaps < max_swaps:
        # Pick a 1 and a 0 at random
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Swap
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        
        # Recompute overlap vector
        h_new_extended = np.concatenate([h_new, [0] * (n-1)])
        overlap_new_full = np.correlate(h_new_extended, h_new, mode='full')
        overlap_new = overlap_new_full[n-1:]
        min_k_new = np.argmin(overlap_new)
        
        # Accept if it improves the minimum overlap
        if overlap_new[min_k_new] > overlaps[min_k]:
            h = h_new
            overlaps = overlap_new
            min_k = min_k_new
            swaps += 1
        else:
            swaps += 1
    return h
