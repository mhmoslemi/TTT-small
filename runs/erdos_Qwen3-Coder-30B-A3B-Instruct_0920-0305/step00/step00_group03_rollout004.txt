def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for a given step function h."""

    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value corresponds to the largest overlap
    max_overlap = np.max(correlation)
    return max_overlap * dx
