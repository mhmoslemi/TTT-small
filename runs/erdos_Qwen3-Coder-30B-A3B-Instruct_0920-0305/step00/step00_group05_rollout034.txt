from scipy.optimize import dual_annealing, minimize

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C5 bound
    c5 = np.max(correlation) * dx
    return c5
