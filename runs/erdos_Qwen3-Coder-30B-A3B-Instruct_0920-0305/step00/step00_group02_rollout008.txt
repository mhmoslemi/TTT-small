import numpy as np

def evaluate_c5(h):
    # h is assumed to be normalized so that sum(h) * dx = 1
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral using full convolution
    # We want max_k of integral of h(x)*(1-h(x+k))
    # This is equivalent to max of correlate(h, 1-h, mode='full')
    h_rev = h[::-1]
    conv = np.correlate(h, 1 - h_rev, mode='full')
    # The overlap integral for k in [0,2] we want the maximum
    # Since convolution is computed with h[i] and 1-h[j], the shift k goes from -1 to 1
    # But in our case we have k from 0 to 2, so we shift appropriately
    # We only care about negative lags (k from 0 to 2)
    conv_max = np.max(conv)
    return conv_max * dx
