import numpy as np
import time
from scipy import fft

def evaluate_solution(h, dx):
    # evaluate the actual C5 = max_k ∫ h(x)(1-h(x+k)) dx
    c = np.correlate(h, 1-h, mode='full') * dx
    return np.max(c)

def project_onto_box_and_sum(h, target_sum):
    """Project h onto [0,1]^n and adjust sum to match target_sum."""
    # clip
    h = np.clip(h, 0., 1.)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-9:
        return h
    # water filling to correct the sum
    # we want to adjust h by adding lambda to each component
    # so that sum(h + lambda*ones) = target_sum
    # i.e., sum(h) + n*lambda = target_sum -> lambda = (target_sum - sum(h)) / n
    offset = (target_sum - current_sum) / len(h)
    # Try to correct the sum, make sure it's still in boxes
    h = np.clip(h + offset, 0., 1.)
    return h

def gradient_of_surrogate(h, alpha, dx):
    """
    Approximate the gradient of a smooth min of overlaps.
    The surrogate Φ_α(h) = - (1/α) * log(sum_k exp(-α * S_k(h))).
    """
    n = len(h)
    # We compute all cross-correlations at once using FFT
    # h_conv_h = correlation of h with itself
    # Which is conv(h, h) with proper indexing = full convolution
    # h_rev = np.flip(h)
    h_rev = h[::-1]
    # full correlation
    c_full = fft.irfft(fft.rfft(h) * fft.rfft(h_rev), n * 2 - 1)
    S = c_full[n-1:] * dx  # S_k for k=0 to n-1, each S_k = integral \int h(x)h(x+k) dx

    # In the spirit of using FFT more efficiently, we actually compute
    # this whether or not we're using looped computation
    
    # But for efficiency, we recompute coef only if needed
    # Let's compute the gradients explicitly for speed
    # ∂S_k / ∂h_i = h_{i+k} + h_{i-k}
    # However, we form a vectorized form.
    # For diagonal entries (realizable at (i,i+k)), there are terms
    # Let's compute weight vector from zero-th shift
    weights = np.exp(-alpha * S)
    if np.sum(weights) == 0:  # Preventsing numerical collapse
        weights = np.ones_like(S)
    W = np.sum(weights)
    
    # Now compute ∇S_k / ∇h_i for all k
    # By definition, ∂S_k / ∂h_i = h_{i+k} + h_{i-k} = contribution from both terms
    grad_sum = np.zeros_like(h)
    for k in range(n):  # could be optimized using FFT
        if k == 0:
            # S_k = sum(h_i * h_{i+0}) = sum(h_i^2)
            grad = 2 * h  # Actually this is handled directly in this case.
        else:
            # We'll split it into two parts: h_i * h_{i+k} + h_{i-k} * h_i
            # So ∂S_k / ∂h_i = h_{i+k} for i+k < n, and h_{i-k} for i-k >= 0
            deriv = np.zeros_like(h)
            # when i >= k: add h_{i-k}
            if k < n:
                deriv[k:] += h[:-k]
            # when i+k < n: add h_{i+k}
            if k > 0 and k < n:
                deriv[:-k] += h[k:]
            grad = deriv
        grad_sum += weights[k] * grad
    # return normalized gradient vector
    if W <= 1e-16:
        return np.zeros_like(h)
    return grad_sum / W

def line_search_backtrack(h, grad, alpha, dx, max_iter=20, step_size=1.0):
    # approximate Φ_alpha value iteratively
    n = len(h)
    h_new = h + step_size * grad
    # project onto constraints
    h_projected = project_onto_box_and_sum(h_new, n/2)
    # compute the new Φα value
    c_full = fft.irfft(fft.rfft(h_projected) * fft.rfft(h_projected[::-1]), n * 2 - 1)
    S = c_full[n-1:] * dx
    weights_new = np.exp(-alpha * S)
    W_new = np.sum(weights_new)
    if W_new == 0:
        return h
    Phi_new = - (1.0 / alpha) * np.log(W_new)
    
    # now try extending with small stepsize
    for _ in range(max_iter):
        step_size *= 0.5
        h_test = h + step_size * grad
        h_projected = project_onto_box_and_sum(h_test, n/2)
        c_full = fft.irfft(fft.rfft(h_projected) * fft.rfft(h_projected[::-1]), n * 2 - 1)
        S = c_full[n-1:] * dx
        weights_test = np.exp(-alpha * S)
        W_test = np.sum(weights_test)
        if W_test == 0:
            return h
        Phi_test = - (1.0 / alpha) * np.log(W_test)
        if Phi_test >= Phi_new:
            return h_projected
    return h_projected

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Parameters
    n = 300  # Choose medium size: fast to evaluate and still flexible
    dx = 2.0 / n
    target_sum = n / 2.0
    
    # Initialize with Beatty sequence
    beta = (np.sqrt(5) - 1) / 2
    h = np.array([1 if ((i * beta) % 1) < 0.5 else 0 for i in range(n)])
    
    # Now adjust to have exactly target_sum ones
    current_ones = np.sum(h)
    to_flip = int(abs(current_ones - target_sum))
    if to_flip > 0:
        diff = current_ones - target_sum
        if diff > 0:
            # Need to flip ones to zeros (find ones and flip)
            ones_pos = np.where(h == 1)[0]
            flip_idx = ones_pos[:to_flip]
            h[flip_idx] = 0
        else:
            # Need to flip zeros to.ones (find zeros and flip)
            zeros_pos = np.where(h == 0)[0]
            flip_idx = zeros_pos[:to_flip]
            h[flip_idx] = 1
    
    # Ensure exact respecting of sum constraint
    h = project_onto_box_and_sum(h, target_sum)
    
    # Now perform gradient descent
    alpha = 1.0
    alpha_max = 10000.0
    alpha_halve_every = 200  # Doubling frequency, say
    last_alpha_change = 0
    eta = 1.0
    
    # Warm start
    best_h = h.copy()
    best_C5 = evaluate_solution(h, dx)
    
    # Split budget: 360s for gradient phase, 180s for local polishing, 60s remaining for safety
    gradient_budget = 360
    local_budget = 180
    
    while time.time() - start_time < gradient_budget:
        # Compute S_k values and gradients
        c_full = fft.irfft(fft.rfft(h) * fft.rfft(h[::-1]), n * 2 - 1)
        S = c_full[n-1:] * dx
        
        # Estimate current Φ_alpha
        weights = np.exp(-alpha * S)
        W = np.sum(weights)
        if W == 0:
            W = 1e-16  # To prevent log(0)
        Phi_alpha = - (1.0 / alpha) * np.log(W)
        
        # compute gradient, reusing the outline from strategy
        grad = gradient_of_surrogate(h, alpha, dx)
        # make positive ascent using line search
        h_new = line_search_backtrack(h, grad, alpha, dx)
        
        # Update h
        h = h_new
        
        # Increment alpha
        if time.time() - start_time - last_alpha_change > alpha_halve_every:
            if alpha < alpha_max:
                alpha *= 2
                last_alpha_change = time.time() - start_time
            
        # Update best ever found
        C5 = evaluate_solution(h, dx)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
    
    # Local polishing stage (binary rounding + hill climbing)
    # Rounding
    h_binary = h.copy()
    median_val = np.median(h_binary)
    h_binary = (h_binary > median_val).astype(float)
    h_binary = project_onto_box_and_sum(h_binary, target_sum)
    
    # Hill-climbing swap
    current_C5 = evaluate_solution(h_binary, dx)
    max_iter_hill = 10000  # Approximate for local budget
    num_flip_pairs = 0
    these_iters = 0
    
    while time.time() - start_time < gradient_budget + local_budget and these_iters < max_iter_hill:
        # pick two random indices: one 1, one 0
        ones = np.where(h_binary == 1)[0]
        zeros = np.where(h_binary == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break

        # Pick one randomly
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Try swap: i <- 0, j <- 1
        h_test = h_binary.copy()
        h_test[i] = 0
        h_test[j] = 1
        
        h_test = project_onto_box_and_sum(h_test, target_sum)
        test_C5 = evaluate_solution(h_test, dx)
        
        # Accept the move if it improves > 0, or with 1e-8 chance if no improvement (SA)
        if test_C5 < current_C5 or (np.random.rand() < 1e-8 and test_C5 <= current_C5):
            h_binary = h_test
            current_C5 = test_C5
            
            if test_C5 < best_C5:
                best_C5 = test_C5
                best_h = h_binary.copy()
                
        these_iters += 1
    
    # Final evaluation
    final_C5 = evaluate_solution(best_h, dx)
    
    return (best_h.tolist(), float(final_C5), n)
