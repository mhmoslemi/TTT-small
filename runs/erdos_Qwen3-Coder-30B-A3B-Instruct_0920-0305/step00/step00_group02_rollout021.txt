```python
import time
import numpy as np
import cvxpy as cp
import random
import math

def random_binary_vector(n):
    """Generate a random binary vector with exactly n/2 ones."""
    if n % 2 != 0:
        raise ValueError("n must be even")
    h = np.zeros(n, dtype=int)
    ones_indices = np.random.choice(n, n//2, replace=False)
    h[ones_indices] = 1
    return h

def compute_c5(h):
    """Compute the C5 bound using the full correlation."""
    dx = 2.0 / len(h)
    total = 1.0
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    C5 = np.max(corr) * dx
    return C5

def solve_milp(n, time_limit=50):
    """Solve the MILP formulation to find a good solution."""
    # Define variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h) == n // 2]
    
    # For each shift k, compute the overlap sum and add constraints
    for k in range(1, n):  # exclude k=0 to avoid trivial constraint
        # Number of valid positions for i
        num_positions = n - k
        if num_positions <= 0:
            continue
        # Define the overlap sum for this shift as sum of products
        # We'll define variables for each product, but this is a bit unmanageable
        # Instead, we can directly write the constraints for u_i = h_i * h_{i+k}
        # given valid index range for i
        # The overlap_k = sum_{i=0}^{n-k-1} h_i * h_{i+k}  
        
        # Linearize constraint: for y_i == h_i * h_{i+k}
        # We need constraints that enforce y_i = h_i * h_{i+k}
        # This is a classic product constraint:
        # y_i <= h_i
        # y_i <= h_{i+k}
        # y_i >= h_i + h_{i+k} - 1
        # y_i >= 0  # nonnegativity
        
        # In this case we have y_i for i = 0 to n-k-1
        y_vars = cp.Variable(num_positions, name=f'y_{k}')
        # Add linear constraints
        constraints += [y_vars <= h[:num_positions]]
        constraints += [y_vars <= h[k:]]
        constraints += [y_vars >= h[:num_positions] + h[k:] - 1]
        constraints += [y_vars >= 0]
        
        # Add to objective constraints
        constraints += [cp.sum(y_vars) >= t]
    
    # Add constraint on T from shift zero (which is always full overlap = n/2)
    # Only critical for limiting t <= n/2; but we already include for all k >= 1
    
    # Objective
    objective = cp.Maximize(t)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with CBC solver
    try:
        prob.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)
        h_sol = np.round(h.value).astype(int)
        # Compute actual C5 for the integer solution
        dx = 2.0 / n
        corr = np.correlate(h_sol, 1 - h_sol, mode='full')
        actual_c5 = np.max(corr) * dx
        return h_sol, actual_c5
    except Exception as e:
        # Return random fallback if MILP fails
        return random_binary_vector(n), compute_c5(random_binary_vector(n))

def local_search(h_initial, max_iters, time_limit):
    """Perform local search minimizing C5."""
    start = time.time()
    current_h = h_initial.copy()
    current_c5 = compute_c5(current_h)
    
    # Store best found
    best_h = current_h.copy()
    best_c5 = current_c5
    
    iters = 0
    while time.time() - start < time_limit and iters < max_iters:
        # Random swap
        ones_idx = np.where(current_h == 1)[0]
        zeros_idx = np.where(current_h == 0)[0]
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
            
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        
        # Swap
        current_h[i] = 0
        current_h[j] = 1
        
        new_c5 = compute_c5(current_h)
        
        # Accept improvement or keep
        if new_c5 < current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
        else:
            # Revert swap
            current_h[i] = 1
            current_h[j] = 0
            
        iters += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main function to find an improved upper bound for C5."""
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    
    # Best solution so far
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Smaller value to run fast
    
    try:
        # Solve MILP to get a quality starting point
        remaining_time = budget_s - (time.time() - start) - 5
        if remaining_time > 5:
            h_opt, c5_opt = solve_milp(n_points, min(remaining_time, 50))
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        else:
            # If not enough time, we'll fallback to random search
            h_opt = random_binary_vector(n_points)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except Exception as e:
        # Fallback to simple random search
        h_opt = random_binary_vector(n_points)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Perform local search on the starting solution
    remaining_time = budget_s - (time.time() - start) - 5
    if remaining_time > 5:
        try:
            final_h, final_c5 = local_search(best_h, int(1e6), remaining_time)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h.copy()
        except Exception as e:
            pass
    
    # Make sure we return an array (convert to list for standardization)
    if best_h is None:
        # Final fallback
        final_h = random_binary_vector(n_points)
        final_c5 = compute_c5(final_h)
        return final_h.tolist(), float(final_c5), n_points
    
    return best_h.tolist(), float(best_c5), n_points
```
