import numpy as np
import time

def project_onto_simplex_box(h, target_sum):
    """Project vector h onto the simplex {x : sum x = target_sum, 0 <= x <= 1}"""
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if np.abs(total - target_sum) < 1e-10:
        return h
    # Adjust all entries uniformly to reach target sum
    diff = target_sum - total
    h += diff / len(h)
    h = np.clip(h, 0, 1)
    return h

def evaluate(h):
    """Evaluate C5 = max(np.correlate(h, 1-h, mode="full") * dx)"""
    n = len(h)
    dx = 2.0 / n
    cross_corr = np.correlate(h, 1 - h, mode="full")
    return np.max(cross_corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Basic parameters
    n_points = 200

    best_h = h.copy()
    best_c5 = evaluate(h)
    
    # Optimization parameters
    max_iter = 100000
    lr = 0.1
    beta = 1.0

        if time.time() - start_time > budget_s - 1:

        # Compute S_raw array for all shifts: correlation of h with (1-h)
        cross_corr = np.correlate(h, 1 - h, mode="full")
        idx_max = np.argmax(cross_corr)
        max_lag = idx_max - (n_points - 1)
        
        # Compute gradient for the worst lag
        grad = np.zeros(n_points)
        if max_lag >= 0:
            # Compute grad contributions for terms with i, i + max_lag
            for i in range(n_points - max_lag):
                grad[i] += 1 - h[i + max_lag]
                grad[i + max_lag] -= h[i]
        else:
            # For negative shift: max_lag is negative, so (i + max_lag) < i
            l = -max_lag  # positive shift amount
            for i in range(n_points - l):
                grad[i + l] += 1 - h[i]
                grad[i] -= h[i + l]
        
        # Update h in gradient direction
        h = h - lr * grad
        h = project_onto_simplex_box(h, target_sum)
        
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
