def evaluate_c5(h):
    """Evaluate C5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation integral
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum value among all lags
    max_val = np.max(corr)
    return max_val * dx
