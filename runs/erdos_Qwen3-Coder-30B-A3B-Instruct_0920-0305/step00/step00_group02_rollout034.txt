import numpy as np
from scipy.optimize import differential_evolution
from cvxpy import *
import time

def decode(z):
    """Convert raw vector to interval lengths and gaps"""
    M = len(z) // 2
    z_l = z[:M]
    z_g = z[M:]
    
    # Softmax normalization
    l = np.exp(z_l) / np.sum(np.exp(z_l))
    g = np.exp(z_g) / np.sum(np.exp(z_g))
    
    return l, g

def build_intervals(ℓ, g):
    """Compute interval start and end points"""
    a = np.concatenate([[0], np.cumsum(ℓ[:-1]) + np.cumsum(g[:-1])])
    b = a + ℓ
    return a, b

def candidate_shifts(a, b):
    """Generate candidate shift values for exact C5 computation"""
    E = np.concatenate([a, b])
    T = set([0])
    for i in range(len(E)):
        for j in range(len(E)):
            diff = abs(E[i] - E[j])
            if diff <= 2:
                T.add(diff)
    return sorted(list(T))

def intersection_length(t, a, b):
    """Compute the overlap length for shift t"""
    total = 0
    for i in range(len(a)):
        left = max(a[i], a[i] - t)
        right = min(b[i], b[i] - t)
        if right > left:
            total += right - left
    return total

def objective(z):
    """Compute C5 for given interval parameters"""
    ℓ, g = decode(z)
    a, b = build_intervals(ℓ, g)
    T = candidate_shifts(a, b)
    I_vals = [intersection_length(t, a, b) for t in T]
    min_I = min(I_vals)
    return 1 - min_I

def run_DE(time_budget):
    """Run differential evolution with time limit"""
    M = 6  # number of intervals
    bounds = [(-5, 5)] * (2 * M + 1)  # one more gap than intervals
    
    best_z = None
    best_val = float('inf')
    
    # Estimate DE budget
    start_time = time.time()
    time_left = time_budget * 0.7
    
    # Run with different random seeds to avoid local minima
    for seed in range(4):
        np.random.seed(seed)
        try:
            if time.time() - start_time > time_left:
                break
            result = differential_evolution(
                objective,
                bounds=bounds,
                seed=seed,
                maxiter=2000,
                popsize=15*(2*M+1),
                mutation=(0.5, 1.0),
                recombination=0.7,
                atol=1e-6,
                rtol=1e-6,
                workers=1
            )
            if result.fun < best_val:
                best_val = result.fun
                best_z = result.x
        except:
            continue
    return best_z, best_val

def local_LP(z):
    """Refine the solution with local linear programming"""
    # Maximize min overlap using LP
    ℓ, g = decode(z)
    a, b = build_intervals(ℓ, g)
    T = candidate_shifts(a, b)
    
    # Find all the active breakpoints that realize min I(t)
    I_vals = [intersection_length(t, a, b) for t in T]
    min_I = min(I_vals)
    
    active_indices = [i for i in range(len(T)) if abs(I_vals[i] - min_I) < 1e-10]
    active_T = [T[i] for i in active_indices]
    
    # Build LP model
    num_vars = len(a) + len(b)
    
    # x keeps all a's and b's
    x = Variable(num_vars)
    
    # Constraints: 
    # - x <= a + ℓ (upper bounds for a)
    # - x >= a (lower bounds for a) 
    # - x[1::2] >= a + ℓ (the b's)
    # - x[1::2] <= a + 2*ℓ (x[1::2] upper bound)
    
    # Each interval has (start, end)
    # The i-th interval: [a[i], b[i]]
    # Let's add a trust region constraint that doesn't move too far
    # For this problem, we'll use simple box constraints
    
    # For now, just build the constraint matrices for alternate approach
    # Consider instead a proper LP with fixed variables except a few for perturbation
    # This version uses an iterative sanity check approach
    
    # Start simple: just re-evaluate the intervals directly with a small shift
    def verify_active_supports_unchange():
        # Compute minimum again and check if active set didn't change
        I_vals = [intersection_length(t, a, b) for t in T]
        min_I = min(I_vals)
        active_indices = [i for i in range(len(T)) if abs(I_vals[i] - min_I) < 1e-8]
        return active_indices, min_I
    
    # Small linear search to get close
    eps = 1e-3
    for _ in range(5):
        active_indices, min_I = verify_active_supports_unchange()
        # Stop if we have active breakpoints that are under tolerance
        if min_I < 0.38:
            break
        # Perturb slightly to reduce min
        for i in range(len(a)):
            for sign in (-1, 1):
                a_try = a.copy()
                b_try = b.copy()
                a_try[i] += sign * eps
                b_try[i] += sign * eps
                I_vals_try = [intersection_length(t, a_try, b_try) for t in T]
                min_I_try = min(I_vals_try)
                if min_I_try < min_I:
                    a = a_try
                    b = b_try
                    min_I = min_I_try
                    break
            active_indices, min_I = verify_active_supports_unchange()
    
    # Return refined parameters after small adjustment
    return build_intervals(ℓ, g)

def gridify(interval_starts, interval_ends, n):
    """Convert interval representation to binary vector"""
    h = np.zeros(n)
    dx = 2.0 / n
    for a, b in zip(interval_starts, interval_ends):
        # Convert interval to grid indices
        start_idx = int(np.floor(a / dx))
        end_idx = int(np.ceil(b / dx))
        for i in range(max(0, start_idx), min(n, end_idx)):
            h[i] = 1.0
    
    # Adjust to get sum exactly n/2
    current_sum = np.sum(h)
    target_sum = n / 2
    if current_sum > target_sum:
        # Find the largest difference between current and target
        # Convert excess 1's to 0's
        # Find indices with 1s to change (unoptimized greedy approach)
        # Remove from last index to first index to avoid changes polluting late parts
        excess_count = int(current_sum - target_sum)
        one_indices = np.where(h == 1)[0]
        # Color: later indices first to scatter change
        for i in range(excess_count, 0, -1):
            if i <= len(one_indices):
                ind = one_indices[-i]
                h[ind] = 0
    elif current_sum < target_sum:
        # Add to last indices (same order of effect to prevent big changes early)
        deficit = int(target_sum - current_sum)
        zero_indices = np.where(h == 0)[0]
        for i in range(0, min(deficit, len(zero_indices))):
            h[zero_indices[i]] = 1
    
    return h

def binary_hill_climb(h, time_left):
    """Hill climb to improve discrete solution"""
    dx = 2.0 / len(h)
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    n = len(h)
    swaps_made = 0
    
    # For efficiency, we only compare deflections in a small set of indices
    # This avoids running full correlation computation on full loop counts
    
    while time.time() - start_time < time_left and swaps_made < 5000:
        pos1 = np.random.randint(0, n)
        pos2 = np.random.randint(0, n)
        
        if pos1==pos2:
            continue
            
        # If it's not a valid swap (1 to 0 vs 0 to 1), skip
        if h[pos1] == h[pos2]:
            continue
            
        # Make swap
        h[pos1], h[pos2] = h[pos2], h[pos1]
        
        # Evaluate
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        else:
            # Revert
            h[pos1], h[pos2] = h[pos2], h[pos1]
            
        swaps_made += 1
        
    return best_h, best_c5

def evaluate_erdos_solution(h, dx):
    """Evaluate the current function using correlation"""
    # Ensure h is of float type
    h_f = np.asarray(h, dtype=np.float64)
    corr = np.correlate(h_f, 1 - h_f, mode='full')
    max_corr = np.max(corr)
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    start_time = time.time()
    
    # Step 1: Seeding
    M = 6
    n = 400  # Sample resolution
    
    # Step 2: Global DE
    z_best, c5_best = run_DE(budget_s * 0.7)
    
    # Step 3: Local LP refinement
    if z_best is not None:
        try:
            a, b = local_LP(z_best)
        except:
            # Fallback to original
            ℓ, g = decode(z_best)
            a, b = build_intervals(ℓ, g)
    
    # Step 4: Discretization
    interval_starts = a
    interval_ends = b
    h = gridify(interval_starts, interval_ends, n)
    
    # Step 5: Binary hill climb
    time_remaining = budget_s - (time.time() - start_time)
    h_best, c5_final = binary_hill_climb(h, time_remaining * 0.7)
    
    # Step 6: Final C5 evaluation
    dx = 2.0 / n
    c5_bound = evaluate_erdos_solution(h_best, dx)
    h_values = h_best
    
    return h_values, c5_bound, len(h_values)
