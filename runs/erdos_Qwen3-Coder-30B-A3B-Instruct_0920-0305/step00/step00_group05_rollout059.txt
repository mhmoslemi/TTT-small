import numpy as np
import cvxpy as cp
import time
import math

def full_overlap(x):
    return np.correlate(x, 1 - x, mode="full")

def worst_shift(x):
    overlaps = full_overlap(x)
    k_star = np.argmax(overlaps)
    return k_star, overlaps[k_star]

def build_mip(N, active_shifts):
    # Variables
    x = cp.Variable(N, integer=True)
    t = cp.Variable(integer=True)
    w = {}  # w[i, k] = x[i] AND x[i+k]
    
    # Constraints
    constraints = [
        cp.sum(x) == N // 2,
        x >= 0,
        x <= 1,
        x[0] == 1  # symmetry breaking
    ]

    # Add binary AND variables and constraints for active shifts
    for k in active_shifts:
        for i in range(N - k):
            w_key = (i, k)
            w[w_key] = cp.Variable(integer=True)
            constraints += [
                w[w_key] <= x[i],
                w[w_key] <= x[i + k],
                w[w_key] >= x[i] + x[i + k] - 1
            ]
    
    # Add overlap constraints for active shifts
    for k in active_shifts:
        left_side = cp.sum(x[:N - k]) - cp.sum([w[(i, k)] for i in range(N - k)])
        constraints += [left_side <= t]
    
    # Objective
    objective = cp.Minimize(t)
    
    # Return problem and variables
    prob = cp.Problem(objective, constraints)
    return prob, (x, t, w)

def solve_mip(prob, vars_dict, time_limit):
    x, t, w = vars_dict
    try:
        prob.solve(solver=cp.CBC, time_limit=time_limit)
        if prob.status not in ["optimal", "optimal_inaccurate"]:
            return None, None
        x_opt = np.round(x.value).astype(int)
        t_opt = int(t.value)
        return x_opt, t_opt
    except Exception:
        return None, None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    N = 140  # Try larger grid
    active_shifts = {1, 2, 3, 4, 5}
    dx = 2.0 / N
    
    # Initial solution construction using a variant of the known construction
    h_values = np.zeros(N)
    num_ones = N // 2
    for i in range(num_ones):
        h_values[i] = 1
    
    # Optimization loop
    best_solution = h_values.copy()
    best_t = float('inf')
    max_time_per_solve = 40  # seconds per solve for safety
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
            
        # Build and solve MIP
        prob, vars_dict = build_mip(N, active_shifts)
        x_opt, t_opt = solve_mip(prob, vars_dict, max_time_per_solve)
        
        if x_opt is None or t_opt is None:
            break
            
        # Evaluate full overlap
        current_solution = x_opt
        overlaps = full_overlap(current_solution)
        max_overlap = np.max(overlaps)
        
        # Update best
        if t_opt < best_t:
            best_t = t_opt
            best_solution = current_solution.copy()
            
        # If all overlaps are bounded by t_opt, then we have our solution
        if max_overlap <= t_opt:
            break
            
        # Otherwise add worst shift
        k_star, _ = worst_shift(current_solution)
        active_shifts.add(k_star)
        
        if len(active_shifts) > 50:
            break  # Safeguard
            
    # Finish with binary search refinement
    current_t = best_t
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
        feasible = True
        test_t = current_t - 1
        if test_t <= 0:
            break
            
        # Create new problem with test_t as upper bound
        prob_test, vars_dict_test = build_mip(N, active_shifts)
        x, t, w = vars_dict_test
        
        # Add constraints that every overlap is <= test_t
        for k in active_shifts:
            left_side = cp.sum(x[:N - k]) - cp.sum([w[(i, k)] for i in range(N - k)])
            prob_test.constraints += [left_side <= test_t]
        
        try:
            prob_test.solve(solver=cp.CBC, time_limit=10)
            if prob_test.status in ["optimal", "optimal_inaccurate"]:
                current_t = test_t
            else:
                break
        except Exception:
            break

    c5_bound = current_t * dx
    return (best_solution.tolist(), c5_bound, N)
