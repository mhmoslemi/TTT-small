import numpy as np
import cvxpy as cp
import time
import math

def full_overlap(x):
    return np.correlate(x, 1 - x, mode="full")

def worst_shift(x):
    overlaps = full_overlap(x)
    k_star = np.argmax(overlaps)
    return k_star, overlaps[k_star]

def build_mip(N, active_shifts):
    x = cp.Variable(N, integer=True)
    t = cp.Variable(integer=True)
    w = {}
    constraints = []

    # Half-ones constraint
    constraints.append(cp.sum(x) == N // 2)

    # Symmetry breaking
    constraints.append(x[0] == 1)

    # Binary constraints
    constraints.append(x >= 0)
    constraints.append(x <= 1)

    # Build auxiliary variables and constraints
    for k in active_shifts:
        w[k] = cp.Variable(N - k, integer=True)
        for i in range(N - k):
            constraints.append(w[k][i] <= x[i])
            constraints.append(w[k][i] <= x[i + k])
            constraints.append(w[k][i] >= x[i] + x[i + k] - 1)

    # Overlap constraints
    for k in active_shifts:
        overlap = cp.sum(x[:N - k]) - cp.sum(w[k])
        constraints.append(overlap <= t)

    # Objective
    objective = cp.Minimize(t)

    # Define problem
    prob = cp.Problem(objective, constraints)
    vars_dict = {"x": x, "t": t, "w": w}
    return prob, vars_dict

def solve_mip(prob, vars_dict, time_limit):
    try:
        prob.solve(solver=cp.CBC, max_iters=10000, TimeLimit=time_limit)
        x_sol = vars_dict["x"].value
        t_sol = vars_dict["t"].value
        return x_sol, t_sol
    except Exception:
        return None, None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Setup
    N = 120
    dx = 2.0 / N
    
    # Initial shifts
    active_shifts = set([1, 2, 3, 4, 5])
    
    # Try different values if needed
    best_h = None
    best_t = float('inf')
    best_N = N

    # Main solving loop
    while time.time() - start_time < 0.8 * budget_s:
        prob, vars_dict = build_mip(N, active_shifts)
        x_sol, t_sol = solve_mip(prob, vars_dict, 30)
        
        if x_sol is None or t_sol is None:
            break

        # Ensure x_sol is integer
        x_sol = np.round(x_sol).astype(int)
        
        # Check feasibility for all shifts
        overlaps = full_overlap(x_sol)
        max_overlap = np.max(overlaps)
        
        if max_overlap <= t_sol:
            break
        
        # Otherwise, add the worst shift
        k_star = np.argmax(overlaps)
        active_shifts.add(k_star)
        
        if len(active_shifts) > 200:  # Prevent too many shifts
            break
            
    # Binary search for tightening
    if x_sol is not None:
        # Run binary search on t
        t_search = int(t_sol)
        while t_search > 0:
            try:
                prob_test, _ = build_mip(N, active_shifts)
                # Reuse variables
                prob_test.constraints.append(vars_dict["t"] <= t_search)
                prob_test.solve(solver=cp.CBC, TimeLimit=5)
                if vars_dict["t"].value <= t_search:
                    t_sol = t_search
                else:
                    break
            except:
                break
            t_search -= 1

    # Evaluate the final solution
    if x_sol is not None:
        x_sol = np.round(x_sol).astype(int) 
        h_values = x_sol.tolist()
        final_c5 = t_sol * dx
        return (h_values, final_c5, N)

    # If no valid solution, use fallback method
    h_values = [1] * (N // 2) + [0] * (N // 2)
    h_values = np.array(h_values)
    final_c5 = worst_shift(h_values)[1] * dx
    return (h_values.tolist(), final_c5, N)
