import time
import random

def compute_c5_fast(h):
    n = len(h)
    dx = 2.0 / n
    size = 1 << ((2*n-1).bit_length())
    f_h = np.fft.fft(h, size)
    f_g = np.fft.fft(1 - h, size)
    corr = np.fft.ifft(f_h * np.conj(f_g)).real
    corr = corr[:2*n-1]
    max_val = np.max(corr) * dx
    return max_val

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr) * dx
    return max_val

    h = np.zeros(n)
    ones_indices = np.random.choice(n, n//2, replace=False)
    h[ones_indices] = 1
