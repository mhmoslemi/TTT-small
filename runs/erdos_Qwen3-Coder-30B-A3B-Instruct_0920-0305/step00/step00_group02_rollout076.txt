import numpy as np
import scipy.sparse as sp
import cvxpy as cp
import time
import random
from cvxpy import trace

def build_shift_matrices(n):
    """Build sparse shift matrices A_k for k=1..floor(n/2)."""
    A_list = []
    for k in range(1, (n // 2) + 1):
        # Build index pairs (i, i+k) and (i+k, i)
        rows = list(range(n - k))
        cols = list(range(k, n))
        data = [1.0] * len(rows)
        A_k = sp.csr_matrix((data, (rows, cols)), shape=(n, n))
        A_k = A_k + A_k.T  # Symmetric matrix
        A_list.append(A_k)
    return A_list
