def project_onto_simplex_box(h, target_sum):
    # Project onto the intersection of box constraints [0,1] and sum constraint
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple method: linear scaling adjustment
    if current_sum > target_sum:
        # Reduce values
        diff = current_sum - target_sum
        # Find the largest values and reduce them enough
        sorted_indices = np.argsort(h)[::-1]
        remaining = diff
        for idx in sorted_indices:
            if remaining <= 0:
                break
            reduce_amount = min(h[idx], remaining)
            h[idx] -= reduce_amount
            remaining -= reduce_amount
    else:
        # Increase values
        diff = target_sum - current_sum
        sorted_indices = np.argsort(h)
        remaining = diff
        for idx in sorted_indices:
            if remaining <= 0:
                break
            increase_amount = min(1-h[idx], remaining)
            h[idx] += increase_amount
            remaining -= increase_amount
    return h
