import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C₅ for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C₅ bound
    c5_bound = np.max(correlation) * dx
    return c5_bound

def initial_h_values():
    """Initial construction with 70 points"""
    n = 70
    h = np.zeros(n)
    # Simple construction: 35 points at 1.0, 35 points at 0.0
    h[:35] = 1.0
    return h

def objective_function(h):
    """Objective function to minimize (negative because we minimize)"""
    c5 = evaluate_erdos_solution(h)
    return -c5

def run(seed=42, budget_s=600):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Get initial solution
    h_current = initial_h_values()
    n_points = len(h_current)
    dx = 2.0 / n_points
    
    # Constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial evaluation
    start_time = time.time()
    c5_current = -objective_function(h_current)
    
    # Use differential evolution with bounds
    result = differential_evolution(
        objective_function,
        bounds,
        maxiter=1000,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        atol=1e-6,
        x0=h_current
    )
    
    # Check if we're done early
    elapsed = time.time() - start_time
    if elapsed >= budget_s * 0.95:
        # Return the best solution so far
        h_final = result.x
        c5_final = -result.fun
        return (h_final, c5_final, n_points)
    
    # Continue optimizing using local search or more iterations
    # Since we're optimizing a 70-dimensional problem with constraints
    # Let's try a small local search maneuver using L-BFGS after DE
    
    # We'll just return the best solution we found in DE
    h_final = result.x
    c5_final = -result.fun
    
    return (h_final, c5_final, n_points)
