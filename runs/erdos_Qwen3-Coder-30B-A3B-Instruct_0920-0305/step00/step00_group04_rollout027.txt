import numpy as np
import time
import math
from scipy import fftpack
from scipy.optimize import minimize
import cvxpy as cp

# Precomputed initial construction
initial_h_values = None

def evaluate_erdos_solution(h):
    """Wrapper to match the expected function interface"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using FFT
    h_padded = np.concatenate([h, np.zeros(n-1)])
    g_padded = np.concatenate([1-h, np.zeros(n-1)])
    H = fftpack.fft(h_padded)
    G = fftpack.fft(g_padded)
    corr = np.real(fftpack.ifft(H * np.conj(G)))
    corr = corr[:2*n-1]
    c5 = np.max(corr) * dx
    return c5

def compute_correlation_fft(h):
    """Compute all correlations using FFT"""
    n = len(h)
    h_padded = np.concatenate([h, np.zeros(n-1)])
    g_padded = np.concatenate([1-h, np.zeros(n-1)])
    H = fftpack.fft(h_padded)
    G = fftpack.fft(g_padded)
    corr = np.real(fftpack.ifft(H * np.conj(G)))
    return corr[:2*n-1]

def binary_to_float(h_binary, n_points):
    """Convert binary representation to float with exact sum constraint"""
    # Pad or truncate to match n_points
    h = h_binary.copy()
    if len(h) < n_points:
        h = np.pad(h, (0, n_points - len(h)), 'constant')
    elif len(h) > n_points:
        h = h[:n_points]
    # Convert to float and rescale
    h = h.astype(float)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def fast_corr_max(h):
    """Fast computation of max correlation for binary vector"""
    dx = 2.0 / len(h)
    corr = compute_correlation_fft(h)
    return np.max(corr) * dx

def optimize_binary_sa(n_points, max_time, seed=42):
    """Binary simulated annealing optimization"""
    np.random.seed(seed)
    # Initialize with provided initial construction if available
    if initial_h_values is not None and len(initial_h_values) >= n_points:
        h_init = initial_h_values[:n_points].copy()
        h_init = (h_init > 0.5).astype(int)
    else:
        h_init = np.random.choice([0,1], size=n_points, p=[0.5, 0.5])
    # Ensure exactly n_points/2 ones
    n_ones = np.sum(h_init)
    if n_ones != n_points // 2:
        # Adjust to get exactly n_points//2 ones
        indices_zeros = np.where(h_init == 0)[0]
        indices_ones = np.where(h_init == 1)[0]
        if n_ones < n_points // 2:
            # Add more ones
            to_add = n_points // 2 - n_ones
            if len(indices_zeros) >= to_add:
                np.random.shuffle(indices_zeros)
                h_init[indices_zeros[:to_add]] = 1
        else:
            # Remove extra ones
            to_remove = n_ones - n_points // 2
            if len(indices_ones) >= to_remove:
                np.random.shuffle(indices_ones)
                h_init[indices_ones[:to_remove]] = 0
    
    current_best = h_init.copy()
    best_c5 = fast_corr_max(current_best)
    # Timing
    t_start = time.time()
    
    def acceptance_probability(delta, temp):
        if delta <= 0:
            return 1.0
        return math.exp(-delta / temp)
    
    def do_swap_move(h):
        # Pick two indices to swap: one with value 0 and one with value 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            return h.copy()  # Impossible to swap, return copy
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        return h_new
    
    # Parameters
    T0 = 0.05
    alpha = 0.9995
    p = 2
    t_max = max_time  # whole duration for phase A
    
    while time.time() - t_start < max_time * 0.8:
        T = T0 * (1.0 - (time.time() - t_start) / t_max)**p
        if T <= 1e-6:
            break
        # Make one swap move
        h_proposed = do_swap_move(current_best)
        c5_proposed = fast_corr_max(h_proposed)
        delta = c5_proposed - best_c5
        if acceptance_probability(delta, T) > np.random.random():
            current_best = h_proposed
            best_c5 = c5_proposed
        
        # Keep track of global best
        if best_c5 < fast_corr_max(current_best):
            current_best = h_proposed
            best_c5 = c5_proposed
    return current_best, best_c5

def project_to_simplex(y, target_sum):
    """Project vector y onto the simplex with sum equal to target_sum"""
    # Initialize Lagrange multiplier bounds
    n = len(y)
    y = np.clip(y, 0, 1)
    y_sum = np.sum(y)
    
    if abs(y_sum - target_sum) < 1e-12:
        return y
    
    # Find lambda bounds
    y_sorted = np.sort(y)[::-1]
    y_sum_total = np.sum(y_sorted)
    
    lambda_low = np.min(y_sorted) - 1.0
    lambda_high = np.max(y_sorted) + 1.0
    tol = 1e-12
    
    while lambda_low <= lambda_high - tol:
        lambda_mid = (lambda_low + lambda_high) / 2.0
        # Compute projection at lambda_mid
        proj = np.clip(y - lambda_mid, 0, 1)
        proj_sum = np.sum(proj)
        # Check if sum is above or below
        diff = proj_sum - target_sum
        if diff > 0:
            lambda_low = lambda_mid
        else:
            lambda_high = lambda_mid
    
    # Final projection with computed lambda
    lambda_final = lambda_low
    final = np.clip(y - lambda_final, 0, 1)
    # Scale to meet exact sum
    final_sum = np.sum(final)
    if abs(final_sum) > 1e-12:
        final = final * target_sum / final_sum
    return final

def smooth_max_correlation(h, gamma=10.0):
    """Compute smooth max of correlations using a temperature parameter"""
    dx = 2.0 / len(h)
    corr = compute_correlation_fft(h)
    corr = corr * dx  # Scale to get actual energy-like values
    corr_exp = np.exp(gamma * corr)
    return np.log(np.sum(corr_exp)) / gamma

def gradient_of_smooth_max(h, gamma=10.0):
    """Compute the gradient of the smooth max correlation"""
    n = len(h)
    dx = 2.0 / n
    # Compute first correlation
    corr = compute_correlation_fft(h)
    corr *= dx
    corr_exp = np.exp(gamma * corr)
    
    # Compute weights for weighted average
    weights = corr_exp / np.sum(corr_exp)
    
    # Compute grad of f_l = dx * corr_l
    grad = np.zeros(n)
    for l in range(1, n):  # l = 0 is the unshifted case
        # Forward terms (h[i] * (1-h[i+l]) for valid l)
        grad[:n-l] += weights[l] * dx * (1 - h[l:])
        # Backward terms (h[i] * h[i-l] for valid l)
        grad[l:] += weights[n-l] * dx * h[:-l]
        
    return grad

def project_gradient_descent_binary(h_binary, max_time, gamma_init=10.0):
    """Refine binary vector to continuous values"""
    n = len(h_binary)
    dx = 2.0 / n
    h = h_binary.astype(float)
    # Ensure correct sum
    h = project_to_simplex(h, n/2)
    
    t_start = time.time()
    gamma = gamma_init
    prev_c5 = fast_corr_max(h)  # Use exact evaluation
    decreasing_count = 0
    
    # Check if first attempt was good enough
    if prev_c5 <= 0.3807:
        return h, prev_c5
        
    while time.time() - t_start < max_time:
        # Increase gamma smoothly
        if gamma < 200.0 and decreasing_count > 30:
            gamma *= 1.5
            decreasing_count = 0
        
        # Gradient step with adaptive step size
        grad = gradient_of_smooth_max(h, gamma)
        alpha = 0.1 * dx
        delta_h = -alpha * grad
        h_candidate = h + delta_h
        
        # Project to simplex
        h_proj = project_to_simplex(h_candidate, n/2)
        
        # Evaluate true C5
        c5_new = fast_corr_max(h_proj)
        
        if c5_new < prev_c5:
            h = h_proj
            prev_c5 = c5_new
            decreasing_count = 0
        else:
            decreasing_count += 1
            # Maybe reduce gamma or step size if stagnating
            gamma *= 0.95
            if decreasing_count > 50:
                break
                
        if prev_c5 <= 0.3807:
            break
            
        # Safety check
        if time.time() - t_start > max_time - 1:
            break
    return h, prev_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    n_points = 60
    t_start = time.time()
    
    # Phase A: Binary search
    h_opt, c5_bound = optimize_binary_sa(n_points, budget_s * 0.7, seed)
    
    # Phase B: Continuous refinement if needed
    if c5_bound > 0.3807 or time.time() - t_start > budget_s * 0.8:
        h_opt, c5_bound = project_gradient_descent_binary(h_opt, budget_s * 0.2)
    
    # Final polishing to ensure sum is correct and return expected result
    h_final = project_to_simplex(h_opt, n_points / 2)
    c5_final = evaluate_erdos_solution(h_final)
    
    return (h_final, c5_final, n_points)
