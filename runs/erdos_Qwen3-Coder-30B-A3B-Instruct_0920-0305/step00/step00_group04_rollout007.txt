def evaluate_c5(h, dx):
    """Evaluates C_5 for a given h using FFT."""
    n = len(h)
    # pad to next power of 2 for efficient FFT
    L = 1 << (n * 2 - 1).bit_length()
    H = fft.fft(h, L)
    G = fft.fft(1 - h, L)
    corr = fft.ifft(H * np.conj(G)).real[:2*n-1]
    return np.max(corr) * dx

def sa_step(b, dx, temp, n):
    """One step of simulated annealing."""
    b_new = b.copy()
    ones = np.where(b_new == 1)[0]
    zeros = np.where(b_new == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return b_new
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    b_new[i] = 0
    b_new[j] = 1
    return b_new

def evaluate_SA(h, dx, time_left, max_iter=10000):
    """Run one simulated annealing loop."""
    current_c5 = evaluate_c5(h, dx)
    best_sofar = h.copy()
    best_c5 = current_c5
    temp = 0.05
    alpha = 0.9995
    start_time = time.time()
    
    # Time schedule for cooling
    # Linear decreasing T
    for i in range(max_iter):
        if time.time() - start_time > time_left:
            break
        new_h = sa_step(h, dx, temp, len(h))
        new_c5 = evaluate_c5(new_h, dx)
        delta = new_c5 - current_c5
        if delta <= 0 or np.random.rand() < np.exp(-delta / temp):
            h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_sofar = h.copy()
                best_c5 = current_c5
        
        # Decrease temperature
        temp *= alpha
        
    return best_sofar, best_c5

def project_to_simplex(y, target_sum):
    """Project vector y onto capped simplex."""
    def s_func(lam):
        return np.sum(np.clip(y - lam, 0, 1)) - target_sum
    
    low, high = np.min(y) - 1, np.max(y) + 1
    high = max(high, low + 1e-10)
    
    for _ in range(120): # safe bisection
        mid = (low + high) / 2
        s = s_func(mid)
        if s > 0:
            low = mid
        else:
            high = mid
        if abs(s) < 1e-12:
            break
    
    lam = (low + high) / 2
    x = np.clip(y - lam, 0, 1)
    return x

def smooth_max(h, dx, gamma):
    """Soft maximum applied to correlation values."""
    n = len(h)
    L = 1 << (n * 2 - 1).bit_length()
    H = fft.fft(h, L)
    G = fft.fft(1 - h, L)
    corr = fft.ifft(H * np.conj(G)).real[:2*n-1]
    corr = corr * dx
    return np.log(np.sum(np.exp(gamma * corr))) / gamma

def gradient_soft_max(h, dx, gamma):
    """Computes gradient of smooth max using convolutions."""
    n = len(h)
    # Compute f_k values (correlations)
    L = 1 << (n * 2 - 1).bit_length()
    H = fft.fft(h, L)
    G = fft.fft(1 - h, L)
    corr = fft.ifft(H * np.conj(G)).real[:2*n-1]
    corr = corr * dx
    
    # Soft max weights
    weights = np.exp(gamma * corr)
    weights /= np.sum(weights)
    
    # Compute gradient as weighted sum of partial derivatives
    # Each entry of grad = sum_j weights[j] * grad_corr[j]
    grad = np.zeros(n)
    # Forward and backward partial derivatives via convolution:
    # d/dh_i (corr_k) = -1 if i == k and 1 if i == k + l
    for k in range(2*n - 1):
        if k < n:
            for i in range(n):
                if (i == k):
                    grad[i] -= weights[k]
                if i == k + 1:
                    grad[i] += weights[k]
    return grad

def gradient_descent_refinement(h, dx, max_iter=2000, gamma_start=5.0, gamma_inc=1.5, gamma_max=200.0, reduce_factor=0.9):
    """Refine binary vector using projected gradient descent."""
    h = h.copy()
    gamma = gamma_start
    obj_val = smooth_max(h, dx, gamma)
    cur_c5 = evaluate_c5(h, dx)
    best_c5 = cur_c5
    best_h = h.copy()
    
    for i in range(max_iter):
        if i % 10 == 0:
            obj_val = smooth_max(h, dx, gamma)
            # update gamma
            if gamma < gamma_max:
                gamma *= gamma_inc
            if gamma > gamma_max:
                gamma = gamma_max
        
        grad = gradient_soft_max(h, dx, gamma)
        # Take a step
        step_size = 0.1 * dx
        h_new = h - step_size * grad
        
        # Project onto simplex
        h_new = project_to_simplex(h_new, len(h)/2)
        
        # Evaluate true C_5
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < cur_c5:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            step_size *= reduce_factor
            if step_size < 1e-12:
                break
    
    return best_h, best_c5

    np.random.seed(seed)

    restart_time = min(15.0, time_left / 4.0)

        h_curr = best_h_binary.copy()
        h_temp, c5_temp = evaluate_SA(h_curr, dx, restart_time)

    final_h = project_to_simplex(final_h, n/2)
