import numpy as np
import cvxpy as cp
import time
from math import floor

def full_overlap(x):
    """Compute full correlation of x with 1-x."""
    return np.correlate(x, 1 - x, mode="full")

def worst_shift(x):
    """Return (k, max_overlap) for the worst shift."""
    overlaps = full_overlap(x)
    k = np.argmax(overlaps)
    max_overlap = overlaps[k]
    return k, max_overlap

def build_mip(N, active_shifts):
    """Build a MIP with variables xi, t, and wij,k for active shifts."""
    # Create decision variables
    x = cp.Variable(N, integer=True)
    t = cp.Variable(integer=True)
    # Create auxiliary variables for each active shift
    w = {}
    constraints = []
    
    # Half-ones constraint
    constraints.append(cp.sum(x) == N // 2)
    
    # Symmetry breaking
    constraints.append(x[0] == 1)
    
    # Bounds on x
    constraints.append(x >= 0)
    constraints.append(x <= 1)
    
    # Build auxiliary variables and constraints
    for k in active_shifts:
        # Number of pairs for shift k
        n_pairs = N - k
        w[k] = cp.Variable(n_pairs, integer=True)
        # AND constraints
        for i in range(n_pairs):
            constraints.append(w[k][i] <= x[i])
            constraints.append(w[k][i] <= x[i + k])
            constraints.append(w[k][i] >= x[i] + x[i + k] - 1)
        
        # Overlap constraint
        overlap_expr = cp.sum(x[:n_pairs]) - cp.sum(w[k])
        constraints.append(overlap_expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    prob = cp.Problem(objective, constraints)
    vars_dict = {
        'x': x,
        't': t,
        'w': w
    }
    return prob, vars_dict

def solve_mip(prob, vars_dict, time_limit=30, solver=cp.CBC):
    """Solve the MIP and extract solution."""
    try:
        prob.solve(solver=solver, TimeLimit=time_limit, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_val = np.round(vars_dict['x'].value).astype(int)
            t_val = int(vars_dict['t'].value)
            return x_val, t_val
    except:
        pass
    # Fallback to CBC if needed
    try:
        prob.solve(solver=cp.CBC, TimeLimit=time_limit, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_val = np.round(vars_dict['x'].value).astype(int)
            t_val = int(vars_dict['t'].value)
            return x_val, t_val
    except:
        return None, None
    return None, None

                if max_overlap <= t_val:

                    break

                    active_shifts.add(k)
