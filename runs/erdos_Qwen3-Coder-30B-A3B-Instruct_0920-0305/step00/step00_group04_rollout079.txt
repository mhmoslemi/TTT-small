import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Model parameters
    n = 60  # Start with 60 points (even)
    dx = 2.0 / n
    target_c5 = 0.3807
    target_int = int(np.floor(target_c5 * n / 2))
    
    # Variables
    b = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(b) == n // 2]
    
    # For each non-zero lag
    for ell in range(1, n):
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        # Define auxiliary variable w_ell[i] = b[i] AND b[i+ell]
        # Limit to indices where both i and i+ell are valid
        w = cp.Variable(n - ell, boolean=True)
        for i in range(n - ell):
            constraints.append(w[i] <= b[i])
            constraints.append(w[i] <= b[i + ell])
            constraints.append(w[i] >= b[i] + b[i + ell] - 1)
        
        # Define N_ell = sum b - sum w
        N_ell = cp.sum(b[:n - ell]) - cp.sum(w)
        constraints.append(t >= N_ell)

    # Enforce symmetry
    for ell in range(-n + 1, 0):
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        # Mirror the constraint using ell < 0
        w = cp.Variable(n + ell, boolean=True)
        for i in range(n + ell):
            constraints.append(w[i] <= b[i])
            constraints.append(w[i] <= b[i - ell])
            constraints.append(w[i] >= b[i] + b[i - ell] - 1)
        
        N_ell = cp.sum(b[-(n + ell):]) - cp.sum(w)
        constraints.append(t >= N_ell)

    # Objective
    objective = cp.Minimize(t)
    
    # Build problem
    prob = cp.Problem(objective, constraints)
    
    # Solver settings
    solver_time_limit = min(300, 0.8 * budget_s)  # Use 300s or half budget if smaller
    prob.solve(solver=cp.CBC, maxSeconds=solver_time_limit, verbose=False)
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        t_opt = t.value
        b_sol = b.value
    else:
        # Fallback to best solution found
        t_opt = t.value if t.value is not None else float('inf')
        b_sol = b.value if b.value is not None else np.zeros(n, dtype=int)
    
    # Convert to domain [0,1] and scale to have integral = 1
    h = b_sol.astype(float)
    h = h * (n / 2) / np.sum(h)  # Scale to integral = 1
    
    # Evaluate the overlap
    corr = np.correlate(h, 1 - h, mode="full") * dx
    c5_bound = np.max(corr)
    
    # Ensure consistent output format
    h_values = h.tolist()
    n_points = len(h_values)
    
    return h_values, c5_bound, n_points
