import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Grid size
    n_points = 60
    dx = 2.0 / n_points
    target_N = int(np.floor(0.3807 * n_points / 2))
    
    # Phase A: Binary Simulated Annealing
    best_binary_h = None
    best_C5 = float('inf')
    T_init = 0.05
    alpha = 0.9995
    p = 2.0
    t_phaseA = 0.7 * budget_s
    
    # Precompute FFT plan
    L = 2 ** int(np.ceil(np.log2(2 * n_points - 1)))
    
    # Initial point from provided construction
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = h_init[:n_points]
        # Normalize to sum to n/2
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        # Round to binary
        binary_h = (h_init > 0.5).astype(int)
        if np.sum(binary_h) != n_points // 2:
            # Rebalance
            ones = np.where(binary_h == 1)[0]
            zeros = np.where(binary_h == 0)[0]
            while np.sum(binary_h) < n_points // 2:
                i = np.random.choice(zeros)
                binary_h[i] = 1
                zeros = np.where(binary_h == 0)[0]
            while np.sum(binary_h) > n_points // 2:
                i = np.random.choice(ones)
                binary_h[i] = 0
                ones = np.where(binary_h == 1)[0]
        best_binary_h = binary_h
        best_C5 = evaluate_correlation(binary_h, dx, L)
    else:
        # Initialize random binary vector with sum n/2
        best_binary_h = np.zeros(n_points, dtype=int)
        ones = np.random.permutation(n_points)[:n_points // 2]
        best_binary_h[ones] = 1
        best_C5 = evaluate_correlation(best_binary_h, dx, L)
    
    # Helper function to get current binary upper bound
    def get_current_upper_bound():
        return best_C5
    
    # Aliases
    binary_h = best_binary_h.copy()
    
    def update_binary_h():
        return binary_h.copy()
    
    # Simulated Annealing outer loop
    phaseA_start = time.time()
    restart_count = 0
    while time.time() - phaseA_start < t_phaseA:
        restart_start = time.time()
        iterations = 0
        T = T_init
        h = binary_h.copy()
        cur_C5 = evaluate_correlation(h, dx, L)
        best_h = h.copy()
        best_C5_sa = cur_C5
        
        # Local edge case: heading toward option with zero flips.
        while time.time() - restart_start < 30 and iterations < 1000000:
            # Generate new state by randomly swapping
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            
            new_C5 = evaluate_correlation(h_new, dx, L)
            Δ = new_C5 - cur_C5
            
            if Δ <= 0 or np.exp(-Δ / T) > np.random.rand():
                h = h_new
                cur_C5 = new_C5
                
                if cur_C5 < best_C5_sa:
                    best_h = h.copy()
                    best_C5_sa = cur_C5
            
            iterations += 1
            T = T_init * (alpha ** iterations)
            
            if best_C5_sa <= 0.3807:
                break
        
        if best_C5_sa < best_C5:
            best_binary_h = best_h
            best_C5 = best_C5_sa
            
        restart_count += 1
        if time.time() - start_time > budget_s - 5:
            break
          
    if best_C5 <= 0.3807:
        # Skip continuous refinement
        final_h = best_binary_h.astype(float)
        final_h = final_h * (n_points / 2) / np.sum(final_h)
    else:
        # Phase B: Continuous Refinement with projected gradient
        t_phaseB = 0.25 * budget_s
        phaseB_start = time.time()
        # Initial continuous point (based on best binary)
        h = best_binary_h.astype(float)
        h = h * (n_points / 2) / np.sum(h)  # Ensure correct sum
        h = np.clip(h, 0, 1)
        
        # Smoothing for max
        gamma = 5.0
        step_size = 0.1 * dx
        patience = 0
        settled = False
        
        while time.time() - phaseB_start < t_phaseB:
            # Compute correlation
            corr = np.correlate(h, 1 - h, mode='full')
            corr_vals = corr * dx
            max_corr = np.max(corr_vals)
            
            # Early stop if below target
            if max_corr <= 0.3807:
                break
                
            # Compute surrogate
            exp_vals = np.exp(gamma * corr_vals)
            Z = np.sum(exp_vals)
            weights = exp_vals / Z
            
            # Gradient ∂f_ℓ/∂h_i
            grad = np.zeros(n_points)
            for ell in range(-n_points + 1, n_points):
                if ell >= 0:
                    # Forward correlation term
                    valid_indices = np.arange(n_points - ell)
                    target_indices = np.arange(ell, n_points)
                    if len(valid_indices) > 0:
                        grad[valid_indices] += weights[ell + n_points - 1] * (1 - h[target_indices])
                else:
                    # Backward term
                    valid_indices = np.arange(-ell, n_points)
                    target_indices = np.arange(0, n_points + ell)
                    if len(valid_indices) > 0:
                        grad[valid_indices] += weights[-ell + n_points - 1] * (-h[target_indices])
            grad *= dx
            # Gradient is small to save time, actually apply it and track progress

            # Update with gradient descent step
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0., 1.)
            
            # Project onto simplex
            # Solve sum(h_new) = n/2 with h_new[i] in [0,1]
            lam_low = np.min(h_new) - 1.
            lam_high = np.max(h_new) + 1.
            while lam_high - lam_low > 1e-8:
                lam = (lam_low + lam_high) / 2.
                proj = np.clip(h_new - lam, 0., 1.)
                diff = np.sum(proj) - n_points / 2
                if diff > 0:
                    lam_low = lam
                else:
                    lam_high = lam
            h = np.clip(h_new - (lam_low + lam_high)/2., 0., 1.)
            
            # Evaluate new max correlation
            corr_new = np.correlate(h, 1 - h, mode='full')
            corr_vals_new = corr_new * dx
            new_max_corr = np.max(corr_vals_new)
            
            if new_max_corr < max_corr:
                max_corr = new_max_corr
                patience = 0
            else:
                patience += 1
                if patience >= 5:
                    break
            
            if new_max_corr < 0.3807:
                break
                
            if gamma < 200:
                gamma *= 1.5
            else:
                # Keep so we still go down if possible
                if patience >= 3:
                    break  # stop training if no more impact with higher gamma
                
            # Step decay
            step_size *= 0.9
            
        final_h = h
        
    # Ongoing final evaluation and Post-processing
    final_h = np.clip(final_h, 0., 1.)
    
    # Ensure sum constraint
    final_sum = np.sum(final_h)
    if abs(final_sum - n_points / 2) > 1e-8:
        # Project onto the simplex
        def sum_proj(x, target_sum):
            s = np.sum(x)
            return x + (target_sum - s) / len(x)
        final_h = sum_proj(final_h, n_points / 2)
        
    c5_bound = evaluate_erdos_solution(final_h)
    
    # Make sure it is valid
    final_h = np.clip(final_h, 0, 1)
    
    # Output format
    return (final_h, float(c5_bound), int(n_points))

def evaluate_correlation(h, dx, L):
    # Compute auto-correlation between h and 1-h using FFT
    h_padded = np.zeros(L)
    h_padded[:len(h)] = h
    g_padded = np.zeros(L)
    g_padded[:len(h)] = 1 - h
    H = fft(h_padded)
    G = fft(g_padded)
    corr_full = ifft(H * np.conj(G)).real
    corr_full = corr_full[:len(h) * 2 - 1]
    corr_vals = corr_full * dx
    return np.max(corr_vals)
