import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value of the overlap integral
    max_overlap = np.max(correlation) * dx
    return max_overlap

def initial_h_values():
    """Return the initial construction as a starting point"""
    # This is the current best known construction (can be modified)
    n = 42
    h = np.zeros(n)
    # Simple construction that we'll optimize from
    mid = n // 2
    h[:mid] = 1.0
    h[mid:] = 0.0
    # Normalize to integral = 1
    h *= 2.0 / n
    return h

def objective(h):
    """Objective function to minimize (negative because we maximize)"""
    return -evaluate_erdos_solution(h)

def constraint_eq(h):
    """Equality constraint: integral of h should be 1"""
    n = len(h)
    dx = 2.0 / n
    return np.sum(h) * dx - 1.0

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial values
    h_initial = initial_h_values()
    n_points = len(h_initial)
    
    # Bounds for h values
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    # Optimization with multiple strategies to avoid local minima
    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try multiple random starts with differential evolution
    for i in range(10):
        if time.time() - start_time > budget_s - 1:
            break
            
        h0 = np.random.uniform(0, 1, n_points)
        h0 = h0 / (np.sum(h0) * 2.0 / n_points)  # Normalize to integral = 1
        try:
            result = differential_evolution(objective, bounds, seed=i, 
                                          maxiter=500, popsize=15, 
                                          tol=1e-6, constraints=constraints,
                                          disp=False)
            
            if not result.success:
                continue
            
            h_opt = result.x
            h_opt = h_opt / (np.sum(h_opt) * 2.0 / n_points)  # Normalize to integral = 1
            
            c5_val = evaluate_erdos_solution(h_opt)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
        except Exception as e:
            continue
    
    # Try local optimization from the best current value
    if time.time() - start_time < budget_s - 1:
        try:
            result = minimize(objective, best_h, method='SLSQP',
                              bounds=bounds,
                              constraints=constraints,
                              options={'maxiter': 1000, 'ftol': 1e-8})
            if result.success:
                h_opt = result.x
                h_opt = h_opt / (np.sum(h_opt) * 2.0 / n_points)  # Normalize
                c5_val = evaluate_erdos_solution(h_opt)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt.copy()
        except Exception as e:
            pass
    
    # Final evaluation
    best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
