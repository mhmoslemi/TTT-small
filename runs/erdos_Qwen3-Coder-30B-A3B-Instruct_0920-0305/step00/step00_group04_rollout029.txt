import cvxpy as cp
import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n = 60  # Start with reasonable size
    dx = 2.0 / n
    target_bound = 0.3807 * n / 2
    
    # Variables
    b = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(b) == n // 2]
    
    # Add auxiliary variables and constraints
    w_vars = {}
    
    for ell in range(1, n):
        # Create auxiliary variables for b[i] AND b[i+ell]
        w_ell = cp.Variable(n - ell, boolean=True)
        w_vars[ell] = w_ell
        
        # Add constraints
        for i in range(n - ell):
            constraints.append(w_ell[i] <= b[i])
            constraints.append(w_ell[i] <= b[i + ell])
            constraints.append(w_ell[i] >= b[i] + b[i + ell] - 1)
        
        # Define N_ell = sum(b) - sum(w_ell)
        N_ell = cp.sum(b[:n-ell]) - cp.sum(w_ell)
        constraints.append(t >= N_ell)
        
        # Also do for negative ell (symmetric)
        w_neg = cp.Variable(n - ell, boolean=True)
        w_vars[-ell] = w_neg
        for i in range(n - ell):
            constraints.append(w_neg[i] <= b[i + ell])
            constraints.append(w_neg[i] <= b[i])
            constraints.append(w_neg[i] >= b[i + ell] + b[i] - 1)
        
        N_neg = cp.sum(b[ell:]) - cp.sum(w_neg)
        constraints.append(t >= N_neg)
    
    # Symmetry breaking
    constraints.append(b[0] == 1)
    for i in range(n - 1):
        constraints.append(b[i] >= b[i+1])
    
    # Objective
    objective = cp.Minimize(t)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Configuration
    solver = cp.CBC
    solver_options = {'maxSeconds': budget_s * 0.9}  # leave some time for post-processing

    # Solve
    try:
        prob.solve(solver=solver, **solver_options)
        status = prob.status
        if status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURATE]:
            best_t = prob.value
            best_b = b.value
        else:
            # Fallback to solution found
            best_t = t.value
            best_b = b.value
    except:
        # In case solver fails completely, try to extract best available.
        try:
            prob.solve(solver=solver, **solver_options)
            best_t = t.value
            best_b = b.value
        except:
            # Return the initial guess
            best_b = np.array([1 if i < n//4 else 0 for i in range(n)])
            best_t = np.inf
    
    # Convert the binary solution to a float array scaled for the integral constraint
    h = best_b.astype(float)
    
    # Evaluate final bound
    # Todo: Better evaluate overlapping function max across all shifts
    # Currently we're using a simplified evaluation of N_max
    
    # Evaluate more accurately
    overlap_vals = []
    for k in range(-n+1, n):
        sum_val = 0
        if k >= 0:
            for i in range(n-k):
                sum_val += h[i] * (1 - h[i + k])
        else:
            for i in range(-k, n):
                sum_val += h[i] * (1 - h[i + k])
        overlap_vals.append(sum_val)
    
    final_bound = max(overlap_vals) * dx
    return (h, final_bound, n)
