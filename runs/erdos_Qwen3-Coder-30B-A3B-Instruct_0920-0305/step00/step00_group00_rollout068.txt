import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the correlation integral
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum value corresponds to the largest overlap
    max_overlap = np.max(corr)
    return max_overlap * dx

def initial_h_values():
    """Initial construction for h."""
    n_points = 45
    h = np.zeros(n_points)
    # Place a step function centered at 1 with height 1 over width 1
    start_idx = n_points // 4
    end_idx = 3 * n_points // 4
    h[start_idx:end_idx] = 1.0
    # Normalize to integral 1
    h *= 2.0 / np.sum(h)
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45
    dx = 2.0 / n_points
    
    # Start with initial construction
    h = initial_h_values()
    current_best = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Define bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_vec):
        h_vec = np.array(h_vec)
        return evaluate_erdos_solution(h_vec)
    
    def constraint_function(h_vec):
        return np.sum(h_vec) * dx - 1.0
    
    # Use differential evolution with custom constraints
    def optimize_step():
        nonlocal h, current_best, best_c5
        # Try multiple optimization restarts
        for _ in range(3):
            if time.time() - start_time > budget_s - 1:
                return
            # Perturb the current solution
            h_perturbed = h.copy()
            # Add a small random perturbation
            h_perturbed += np.random.normal(0, 0.02, n_points)
            # Clamp values to [0,1]
            h_perturbed = np.clip(h_perturbed, 0, 1)
            # Renormalize
            h_perturbed *= 1.0 / (np.sum(h_perturbed) * dx)
            val = evaluate_erdos_solution(h_perturbed)
            if val < best_c5:
                best_c5 = val
                current_best = h_perturbed.copy()
        
        # Then run local optimization
        try:
            # Subproblem is going to be 45-dimensional, so lightweight methods
            # will be sufficient
            res = differential_evolution(
                objective,
                bounds,
                seed=seed,
                maxiter=100,
                popsize=10,
                mutation=(0.5, 1.0),
                recombination=0.7,
                callback=lambda x, convergence=0: None
            )
            if res.success and res.fun < best_c5:
                best_c5 = res.fun
                current_best = res.x.copy()
        except:
            pass
    
    # Main optimization loop
    iterations = 0
    while time.time() - start_time < budget_s - 1:
        optimize_step()
        iterations += 1
        if iterations >= 200:  # Limit to prevent long loops
            break
    
    # Final check
    final_c5 = evaluate_erdos_solution(current_best)
    if final_c5 < best_c5:
        best_c5 = final_c5
        
    return (current_best, best_c5, n_points)
