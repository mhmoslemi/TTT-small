def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant C5 for a given step function h"""

    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C5 constant
    c5_value = np.max(correlation) * dx
    return c5_value
