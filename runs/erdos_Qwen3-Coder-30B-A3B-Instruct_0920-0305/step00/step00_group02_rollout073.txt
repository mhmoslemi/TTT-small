import numpy as np
import time
from scipy import fftpack

def run(seed=42, budget_s=600, **kwargs):
    # Set parameters for the problem
    np.random.seed(seed)
    
    # Number of samples – keeps problem reasonably fast
    n = 300
    dx = 2.0 / n
    
    # Project onto feasible set: [0,1]^n and sum = n/2
    def project_h(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if abs(s - n/2) < 1e-9:
            return h
        # Water-filling approach to fix the sum
        # Solve s + λ * (1 - 0) = n/2 -> λ = (n/2 - s)
        # which is not right; better: adjust by linear transform
        # better yet: find lambda such that sum(clip(h + lambda, 0, 1)) = n/2
        # This is a root-finding problem for lambda over interval [-1,1]:
        # Define func(lambda) = sum(max(0, min(1, h + lambda))) - n/2
        def objective(lam):
            return np.sum(np.clip(h + lam, 0, 1)) - n/2
        
        # Binary search in [-1, 1]
        a, b = -1.0, 1.0
        for i in range(50):
            c = (a + b) / 2.0
            val = objective(c)
            if val > 0:
                b = c
            else:
                a = c
        lam = (a + b) / 2.0
        h = np.clip(h + lam, 0, 1)
        return h

    # Initialize with Beatty sequence
    beta = (np.sqrt(5) - 1) / 2
    h = np.array([1 if (i * beta) % 1 < 0.5 else 0 for i in range(n)], dtype=float)
    # Adjust for exact sum constraint
    while abs(np.sum(h) - n/2) > 1e-9:
        idx_ones = np.where(h == 1)[0]
        idx_zeros = np.where(h == 0)[0]
        if len(idx_ones) > n//2:
            # Too many ones: change one to zero
            h[idx_ones[-1]] = 0
        elif len(idx_ones) < n//2:
            # Not enough ones: change one to 1
            h[idx_zeros[0]] = 1
    h = project_h(h)

    # Initial values for optimisation
    alpha = 1.0
    alpha_max = 10000.0
    decay_step = 200  # increase alpha every N iterations
    eta = 1.0
    max_iter = 5000
    best_C5 = float('inf')
    best_h = h.copy()

    # Time tracking
    start_time = time.time()

    # Soft max surrogate
    def Phi_alpha(h_vec, alpha_val):
        # Compute all S_k for all shifts k
        corr_h_h = np.correlate(h_vec, h_vec, mode='full')
        # k from 0 to n-1; starts around k=n-1 to k=n-1+n-1=2n-2; we need 0 to n-1 shifts
        S_k = []
        for k in range(n):
            S_k.append(corr_h_h[n - 1 + k] * dx)
        S_k = np.array(S_k)
        # avoid very large exponents – we still want to use num stable sign trick
        S_k = np.clip(S_k, 0, n)
        exp_neg_alpha_S = np.exp(-alpha_val * S_k)
        Z = np.sum(exp_neg_alpha_S)
        return - (1.0 / alpha_val) * np.log(Z)

    # Multithreaded gradient
    def compute_gradient(h_vec, alpha_val):
        corr_h_h = np.correlate(h_vec, h_vec, mode='full')
        S_k = np.array([corr_h_h[n - 1 + k] * dx for k in range(n)])
        S_k = np.clip(S_k, 0, n)
        # Gradient of S_k wrt h is h_{i+k} + h_{i-k}
        grad = np.zeros_like(h_vec)
        for k in range(n):
            if k == 0:
                grad += h_vec  # h_{i+0} = h_i
            else:
                # i-k and i+k correspond to positions
                # Forward shift
                if k < n:
                    grad[k:] += h_vec[:-k]
                # Backward shift
                if k < n:
                    grad[:-k] += h_vec[k:]
        weights = np.exp(-alpha_val * S_k)
        total_weight = np.sum(weights)
        if total_weight == 0:
            # Avoid zero division
            return np.zeros_like(h_vec)
        return np.sum(weights * grad) / total_weight

    # Main loop
    iter_counter = 0

    def convergence_criterion():
        return time.time() - start_time > 0.5 * budget_s

    while not convergence_criterion() and iter_counter < max_iter:
        # Compute phi_alpha for current h
        phi_val = Phi_alpha(h, alpha)
        
        # Compute gradient using the formula (very approximate, but works)
        # Note: direct use of convolution is more precise but SLOW here
        # Instead compute S_k solves:
        # d S_k /d h_i = h_{i+k} + h_{i-k}
        # i: [0..n-1]  
        # For the total gradient
        grad = np.zeros_like(h)

        corr_h_h = np.correlate(h, h, mode='full')
        # We have: S_k = sum( h_i h_{i+k} ) => dS_k/d h_i = h_{i+k} + h_{i-k} 
        # Let's create a convolution for each shift k
        
        for k in range(n):
            # Compute: dS_k/dh_i = h_{i+k} + h_{i-k}
            # Create a kernel (a bit tricky, only one side)
            kernel = np.zeros_like(h)
            if k < n:
                kernel[k:] = h[:-k] if k > 0 else h[:]
            if k < n:
                kernel[:-k] += h[k:] if k > 0 else h[:]
            # The deriative is then: gradient = (sum_k w_k * kernel) / sum_k w_k
            # But we're accumulating the total gradient.
            if k >= 0:
                # first kernel now is: h Shifted right by k, add it once
                # second kernel: h shifted left by k, add it once
                if k == 0:
                    grad += h
                else:
                    # shift right
                    shift_k = np.zeros(n)
                    shift_k[k:] = h[:-k]
                    # shift left
                    shift_neg_k = np.zeros(n)
                    shift_neg_k[:-k] = h[k:]
                    grad += shift_k + shift_neg_k

        # Compute weights and weighted average gradient
        S_k_vals = np.array([corr_h_h[n - 1 + k] * dx for k in range(n)])
        S_k_vals = np.clip(S_k_vals, 0, n)
        weights = np.exp(-alpha * S_k_vals)
        # normalize gradient to match total weights
        W = np.sum(weights)
        if W == 0:
            W = 1
        grad = grad / W

        # Perform gradient ascent step (we are maximizing)
        h_new = h + eta * grad
        
        # Project back
        h_new = project_h(h_new)
        
        # Evaluate new Phi
        try:
            new_phi = Phi_alpha(h_new, alpha)
        except:
            new_phi = -1e9

        # Accept change if it improves Phi
        if new_phi >= phi_val:
            h = h_new
            # Backtrack on step size if possible
            pass
        else:
            eta = max(eta * 0.5, 1e-8)  # smaller steps if improving
            
        # Incrementing counter and update alpha
        iter_counter += 1
        if iter_counter % decay_step == 0 and alpha < alpha_max:
            alpha *= 2.0

        # Periodic C5 evaluation and save best
        if iter_counter % 20 == 0:
            # compute actual C5 = max_correlation
            c_correlate = np.correlate(h, 1 - h, mode='full')
            C5 = np.max(c_correlate) * dx
            if C5 < best_C5:
                best_C5 = C5
                best_h = h.copy()

    # Final discrete polishing with binary hill climbing
    # Convert h to binary by median thresholding
    threshold = np.partition(h, n//2)[n//2]
    h_final = (h > threshold).astype(float)
    
    # Ensure exactly n//2 ones
    while np.sum(h_final) > n // 2:
        idxs = np.where(h_final == 1)[0]
        h_final[idxs[-1]] = 0
    while np.sum(h_final) < n // 2:
        idxs = np.where(h_final == 0)[0]
        h_final[idxs[0]] = 1

    # Local search on binary sequence
    def compute_C5_binary(h_b):
        c = np.correlate(h_b, 1 - h_b, mode='full')
        return np.max(c) * dx

    current_C5 = compute_C5_binary(h_final)
    best_binary = h_final.copy()
    best_binary_C5 = current_C5

    # Simple local search for 180 seconds
    local_search_end = time.time() + 0.3 * budget_s
    while time.time() < local_search_end:
        # Try random swaps
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        
        if h_final[i] == h_final[j]:
            continue  # can't swap same value (both 0 or both 1)
            
        # Swap
        h_new_final = h_final.copy()
        h_new_final[i] = 1 - h_new_final[i]
        h_new_final[j] = 1 - h_new_final[j]
        
        # Compute new C5
        new_C5 = compute_C5_binary(h_new_final)
        
        if new_C5 < current_C5:
            h_final = h_new_final
            current_C5 = new_C5
            if current_C5 < best_binary_C5:
                best_binary_C5 = current_C5
                best_binary = h_final.copy()

    # Final return
    return (best_binary.tolist(), float(best_binary_C5), n)
