from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C₅ for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum value over all shifts
    max_corr = np.max(conv)
    return max_corr * dx
