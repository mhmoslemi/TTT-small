import numpy as np
import time
from scipy.special import softmax
from scipy.signal import correlate

def project_h(h, target_sum):
    """
    Project vector h onto the set {x : 0 <= x <= 1, sum(x) = target_sum}
    Using binary search on Lagrange multiplier.
    """
    h = np.array(h)
    lb, ub = 0.0, 1.0
    sum_h = np.sum(h)
    
    if np.abs(sum_h - target_sum) < 1e-10:
        return np.clip(h, lb, ub)
    
    # Binary search for the right lambda
    def eval_sum(lam):
        x = np.clip(h + lam, lb, ub)
        return np.sum(x) - target_sum
    
    # Find bounds for lambda
    lam_min, lam_max = -100.0, 100.0
    while eval_sum(lam_min) > 0:
        lam_min *= 2.0
    while eval_sum(lam_max) < 0:
        lam_max *= 2.0

    # Bisection
    for _ in range(60):
        lam = (lam_min + lam_max) / 2.0
        val = eval_sum(lam)
        if val > 0:
            lam_min = lam
        else:
            lam_max = lam
        if abs(val) < 1e-12:
            break
    lam = (lam_min + lam_max) / 2.0
    x = np.clip(h + lam, lb, ub)
    
    # Ensure exact sum
    diff = target_sum - np.sum(x)
    # Add diff to components that aren't already at upper bound, 
    # subtract from those at lower bound
    # This ensures small changes that don't change bounds
    non_zero = np.where(x > lb)[0]
    non_one = np.where(x < ub)[0]
    to_adjust = min(len(non_zero), len(non_one))
    if to_adjust > 0:
        adjust_idx = np.argsort(np.abs(x[non_zero] - ub))[:to_adjust]
        for i in adjust_idx:
            idx = non_zero[i]
            x[idx] = min(ub, x[idx] + diff/to_adjust)
            if x[idx] > ub:
                x[idx] = ub
    return np.clip(x, lb, ub)

def grad_c_k(h, k, dx):
    """
    Get gradient of c_k(h) = dx * sum(h[i] * (1 - h[i+k]))
    For shifting, use slice operations.
    """
    g = np.zeros_like(h)
    if k >= 0:
        g[k:] += -dx * h[:-k]
        g[:-k] += dx * (1 - h[k:])
    else:
        k = -k
        g[k:] += dx * (1 - h[:-k])
        g[:-k] += -dx * h[k:]
    return g

def evaluate_erdos_solution(h_values):
    """
    Summarized function to evaluate C5, using py version as
    the official checker.
    This is a simple implementation of integrals.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute all correlations
    corr_full = correlate(h_values, 1 - h_values, mode='full')
    # Take the max
    max_corr = np.max(corr_full) * dx
    return max_corr
