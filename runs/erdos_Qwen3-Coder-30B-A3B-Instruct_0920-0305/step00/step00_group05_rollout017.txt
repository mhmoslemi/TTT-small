from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given h"""
    # Compute the full correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # The maximum value corresponds to the maximum overlap
    max_corr = np.max(corr)
    return max_corr * dx
