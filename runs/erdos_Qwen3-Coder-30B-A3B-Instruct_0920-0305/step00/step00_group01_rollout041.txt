        corr = np.correlate(h, 1 - h, mode='full')
        # Multiply by dx
        corr *= dx
        # Return maximum overlap value
        return np.max(corr)

        """Project h to satisfy sum(h) = n_points / 2 and 0 <= h <= 1."""

        if abs(sum_h - target_sum) < 1e-10:
            return h
            
        # Adjust values to maintain sum; use simple method
        diff = target_sum - sum_h
        h += diff / len(h)
        h = np.clip(h, 0, 1)
        
        return h
