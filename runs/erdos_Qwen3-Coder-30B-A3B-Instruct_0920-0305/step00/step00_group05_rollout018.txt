import numpy as np
import cvxpy as cp
import time
import math

# Pre-loaded functions
def evaluate_erdos_solution(h_values, n_points):
    # Dummy implementation - actual one is pre-loaded
    return np.max(np.correlate(h_values, 1 - np.array(h_values), mode='full')) * (2.0 / n_points)

def initial_h_values():
    # Dummy implementation - actual one is pre-loaded
    n = 70
    h = np.ones(n)
    h[:n//2] = 0
    return h.tolist()

def full_overlap(x):
    """Compute overlap for all shifts using np.correlate"""
    return np.correlate(x, 1 - np.array(x), mode='full')

def worst_shift(x):
    """Find the shift k that maximizes overlap"""
    overlaps = full_overlap(x)
    max_k = np.argmax(overlaps)
    max_overlap = overlaps[max_k]
    return max_k, max_overlap

def build_mip(N, active_shifts):
    """Build MIP model for given active shifts"""
    x = cp.Variable(N, integer=True)
    t = cp.Variable(integer=True)
    
    # Binary variables for products
    w = {}
    constraints = []
    
    # Constraints
    constraints.append(cp.sum(x) == N // 2)
    constraints.append(x[0] == 1)  # Symmetry breaking
    constraints.append(x >= 0)
    constraints.append(x <= 1)

    # Add auxiliary variables for active shifts
    for k in active_shifts:
        w[k] = cp.Variable((N - k), integer=True)
        # Enforce w[i,k] = x[i] AND x[i+k]
        for i in range(N - k):
            constraints.append(w[k][i] <= x[i])
            constraints.append(w[k][i] <= x[i + k])
            constraints.append(w[k][i] >= x[i] + x[i + k] - 1)

    # Add overlap constraints for active shifts
    for k in active_shifts:
        # Overlap = sum(x[i]) - sum(w[i,k])
        # This less equal t, but for indexing consider it as matrix constraint
        #  sum(x[i]) - sum(w[i,k]) <= t
        left_side = cp.sum(x[:N - k]) - cp.sum(w[k])
        constraints.append(left_side <= t)

    # Objective to minimize t
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    return prob, {'x': x, 't': t, 'w': w}

def solve_mip(prob, vars_dict, time_limit=30):
    """Solve MIP problem"""
    try:
        prob.solve(solver=cp.CBC, max_iters=10000, TimeLimit=time_limit)
        sol_x = vars_dict['x'].value.astype(int)
        sol_t = int(vars_dict['t'].value)
        
        # Clip solution to integers
        sol_x = np.clip(sol_x, 0, 1)
        
        if sol_x is None:
            return None, None
            
        return sol_x, sol_t
    
    except Exception as e:
        return None, None

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    N = 140  # Adjusted from record for better chance of improvement
    active_shifts = {1, 2, 3, 4, 5}
    
    best_bound = float('inf')
    best_h = None
    
    print(f"Starting optimization with N = {N}")
    
    while True:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Build problem
        prob, vars_dict = build_mip(N, active_shifts)
        
        # Solve with limited time
        try:
            x_sol, t_sol = solve_mip(prob, vars_dict, time_limit=25)
        except:
            x_sol, t_sol = None, None
            
        if x_sol is None or t_sol is None:
            print("Solver failed, fallback to local search")
            # Fallback local improvement
            if best_h is not None:
                h = np.array(best_h)
                for _ in range(500):
                    if time.time() - start_time > budget_s - 5:
                        break
                    i, j = np.random.randint(0, N, 2)
                    if i != j:
                        h[i], h[j] = h[j], h[i]
                        # Evaluate
                        c5 = evaluate_erdos_solution(h.tolist(), N)
                        if c5 < best_bound:
                            best_bound = c5
                            best_h = h.tolist()
            break
            
        # Evaluate full overlaps
        overlaps = full_overlap(x_sol)
        max_overlap_full = np.max(overlaps)
        
        if max_overlap_full <= t_sol:
            print(f"Optimal solution found with t = {t_sol}")
            possible_improvement = False
            # Binary search to tighten bound
            t_upper = t_sol
            t_lower = 0
            # Try reducing t by feasibility check
            for i in range(int(np.log2(t_sol)) + 1):
                if time.time() - start_time > budget_s - 5:
                    break
                guess_t = (t_upper + t_lower) // 2
                if guess_t == t_upper:
                    break
                
                # Construct new problem with updated bound
                try:
                    prob_new, vars_dict_new = build_mip(N, active_shifts)
                    prob_new.constraints[-1].rhs = guess_t  # Update constraint
                    # Re-solve
                    x_sol_new, _ = solve_mip(prob_new, vars_dict_new, time_limit=10)
                    if x_sol_new is not None:
                        # Verify solution satisfies constraints exactly
                        overlaps_new = full_overlap(x_sol_new)
                        max_over_new = np.max(overlaps_new)
                        if max_over_new <= guess_t:
                            t_upper = guess_t
                            # Update existing sol
                            x_sol = x_sol_new
                except:
                    pass
                t_lower = t_upper
                
            final_t = t_upper
            final_c5 = final_t * (2.0 / N)
            print(f"Final C5 = {final_c5}")
            if final_c5 < best_bound:
                best_bound = final_c5
                best_h = x_sol.tolist()
            break
            
        # Add the worst shift
        worst_k, _ = worst_shift(x_sol)
        if worst_k not in active_shifts:
            active_shifts.add(worst_k)
            print(f"Added shift {worst_k} to active set")
            
        # Check if we've made enough progress
        overlaps_current = full_overlap(x_sol)
        current_max = np.max(overlaps_current)
        if active_shifts > {1, 2, 3, 4, 5} and current_max < best_bound:
            best_bound = current_max * (2.0 / N)
            best_h = x_sol.tolist()
            
    # Final check if best bound was never updated
    if best_h is None:
        initial_h = np.array(initial_h_values())
        best_bound = evaluate_erdos_solution(initial_h.tolist(), len(initial_h))
        best_h = initial_h.tolist()
    
    return (best_h, best_bound, N)
