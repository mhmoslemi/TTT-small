import numpy as np
import time
import random
from scipy.optimize import minimize
from math import floor

# Import the required functions
from your_module import evaluate_erdos_solution, initial_h_values

def project_to_feasible(v, target_sum):
    """Project vector v to feasible set with sum = target_sum and bounds [0,1]"""
    # Clip to [0,1] first
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-10:
        return v
    elif current_sum > target_sum:
        # Need to decrease some values
        # Sort indices by descending value
        indices = np.argsort(v)[::-1]
        remaining_sum = target_sum
        for i in indices:
            if remaining_sum <= 0:
                v[i] = 0
            else:
                max_decrease = v[i] - 0
                actual_decrease = min(max_decrease, v[i] - remaining_sum)
                if actual_decrease > 0:
                    v[i] -= actual_decrease
                    remaining_sum -= actual_decrease
    else:
        # Need to increase some values
        # Sort indices by ascending value
        indices = np.argsort(v)
        remaining_sum = target_sum - current_sum
        for i in indices:
            if remaining_sum <= 0:
                break
            max_increase = 1 - v[i]
            actual_increase = min(max_increase, remaining_sum)
            if actual_increase > 0:
                v[i] += actual_increase
                remaining_sum -= actual_increase
    return v

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    n_points = len(initial_h_values)
    h_current = initial_h_values.copy()
    h_current = project_to_feasible(h_current, n_points / 2)
    
    # For faster computation, reduce sample count if needed
    # But we'll use the provided n_points that is possibly 47
    # It's good to keep a moderate number, but let's try a slightly larger one for better resolution
    # Let us use value near 50 
    n_points = min(n_points, 50)
    dx = 2.0 / n_points
    
    # Adjust h to new length
    h_current = h_current[:n_points]
    h_current = project_to_feasible(h_current, n_points / 2)
    
    # Evaluate the initial solution
    current_cost = evaluate_erdos_solution(h_current)
    best_cost = current_cost
    best_h = h_current.copy()
    
    # Parameters for simulated annealing
    max_iter = int(5e6)
    temp_start = 1.0
    temp_end = 1e-5
    cooling_rate = np.exp(np.log(temp_end / temp_start) / (max_iter * 0.7))
    
    # Main optimization loop
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.1:
            break
            
        # Generate random move
        i, j = random.sample(range(n_points), 2)
        
        # Compute bounds on delta
        max_delta = min(1 - h_current[i], h_current[j])
        min_delta = max(-h_current[i], 0 - h_current[j])
        
        # Generate a delta with tighter bound to ensure bounds are not violated
        delta = random.uniform(min_delta * 0.1, max_delta * 0.1)  # small random change

        # Propose the change
        h_new = h_current.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Project h_new to feasible region
        h_new = project_to_feasible(h_new, n_points / 2)
        
        # Evaluate new solution
        new_cost = evaluate_erdos_solution(h_new)
        
        # Accept with probability
        if new_cost < current_cost:
            h_current = h_new
            current_cost = new_cost
            if new_cost < best_cost:
                best_cost = new_cost
                best_h = h_new.copy()
        else:
            # Simulated annealing acceptance probability
            temp = temp_start * (cooling_rate ** iteration)
            if temp > 0 and random.random() < np.exp(-(new_cost - current_cost) / temp):
                h_current = h_new
                current_cost = new_cost
                
    return (best_h, best_cost, n_points)
