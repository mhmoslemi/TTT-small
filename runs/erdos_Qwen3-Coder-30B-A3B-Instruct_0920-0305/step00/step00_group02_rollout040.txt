```python
import time
import numpy as np
import cvxpy as cp
import random
import math

def random_binary_vector(n):
    """Generate a random binary vector with exactly n/2 ones."""
    if n % 2 != 0:
        raise ValueError("n must be even")
    h = np.zeros(n, dtype=int)
    ones_indices = np.random.choice(n, n//2, replace=False)
    h[ones_indices] = 1
    return h

def compute_c5(h):
    """Compute C5 using numpy correlation."""
    dx = 2.0 / len(h)
    total = 1.0
    corr = np.correlate(h, 1-h, mode='full')
    # The maximum over all shifts (k from -(n-1) to n-1)
    # We are interested in shifts k from 0 to n-1 (positive)
    # The full correlation above has length 2n-1, with index n-1 corresponding to k=0
    # We want the maximum over positive shifts (including k=0).
    # The value at index n-1+k for k>=0 is correlation for shift k
    c5_values = total - corr[n-1:] * dx
    return np.max(c5_values)

def solve_milp(n):
    """Solve MILP to find optimal binary h and compute C5 bound."""
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    constraints = [cp.sum(h) == n//2]
    
    # For each shift k, we compute. Linearization
    for k in range(1, n):  # exclude k=0 since we're interested in minimization over k
        # number of valid positions for i: n - k
        if n - k == 0:
            continue
        # Variables representing the products h_i * h_{i+k}
        y = cp.Variable(n - k, continuous=True)
        # Linearization constraints
        for i in range(n - k):
            constraints += [y[i] <= h[i]]
            constraints += [y[i] <= h[i + k]]
            constraints += [y[i] >= h[i] + h[i + k] - 1]
            constraints += [y[i] >= 0]
        
        # The sum of these products is the overlap for shift k
        constraints += [cp.sum(y) >= t]
    
    # Bounds on t
    constraints += [t <= n//2]
    
    prob = cp.Problem(cp.Maximize(t), constraints)
    
    # Try to solve
    try:
        prob.solve(solver=cp.CBC, verbose=False)
        # Round the solution to binary
        h_sol = np.round(h.value).astype(int)
        # If solution is not binary due to solver tolerance, recompute valid version
        if not np.all(np.isin(h_sol, [0, 1])):
            h_sol = np.where(h.value >= 0.5, 1, 0)
        h_sol = h_sol.astype(int)
        
        # Compute actual C5 using the computed h
        dx = 2.0 / n
        total = 1.0
        corr = np.correlate(h_sol, 1-h_sol, mode='full')
        c5_values = total - corr[n-1:] * dx
        actual_c5 = np.max(c5_values)
        return h_sol, actual_c5
    except Exception as e:
        # Fallback to heuristic
        raise Exception("MILP solver error", e)

def local_search(h_initial, max_iters, time_limit):
    """Perform local search using random swaps."""
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    cur = h_initial.copy()
    cur_c5 = best_c5
    start = time.time()
    
    # Use an appropriate temperature schedule
    T = 0.1  # Initial temperature
    decay = 0.9999  # Schedule decay
    
    for iteration in range(max_iters):
        if time.time() - start > time_limit:
            break
            
        # Generate a random swap
        ones_idx = np.where(cur == 1)[0]
        zeros_idx = np.where(cur == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            continue
            
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        
        # Perform swap
        cur[i] = 0
        cur[j] = 1
        
        # Compute new C5
        new_c5 = compute_c5(cur)
        
        # Accept or reject
        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur.copy()
        else:
            # Revert swap
            cur[i] = 1
            cur[j] = 0
            
        T *= decay  # Decrement temperature

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    
    # Try MILP with n=100
    n = 100
    try:
        h_opt, c5_opt = solve_milp(n)
        elapsed = time.time() - start
        remaining_time = budget_s - elapsed - 5  # Leave 5s for cleanup and final steps
        
        # Run local search to improve further if time remains
        if remaining_time > 5:
            best_h, best_c5 = local_search(h_opt, max_iters=100000, time_limit=remaining_time)
        else:
            best_h, best_c5 = h_opt, c5_opt
            
    except Exception as e:
        # Fallback to heuristic method
        n = 100
        h_opt = random_binary_vector(n)
        c5_opt = compute_c5(h_opt)
        elapsed = time.time() - start
        remaining_time = budget_s - elapsed - 5
        if remaining_time > 5:
            best_h, best_c5 = local_search(h_opt, max_iters=100000, time_limit=remaining_time)
        else:
            best_h, best_c5 = h_opt, c5_opt
    
    # Ensure return types are standard Python
    return best_h.tolist(), float(best_c5), int(n)
```
