import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    # Compute the correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value of the correlation
    max_overlap = np.max(corr) * dx
    return max_overlap
