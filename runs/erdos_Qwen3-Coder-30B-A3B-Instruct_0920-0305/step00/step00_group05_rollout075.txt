import numpy as np

from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    # Compute the overlap integral for each shift k
    # h is a numpy array of values in [0,1]
    # dx = 2.0 / len(h)
    # The overlap is max_k of int h(x)(1-h(x+k)) dx
    # This is equivalent to max_k of sum h[i]*(1-h[i+k]) over all valid k
    # We can compute this using convolution
    conv = correlate(h, 1-h, mode='full')
    # We want the maximum of all these values, scaled by dx
    return np.max(conv) * dx
