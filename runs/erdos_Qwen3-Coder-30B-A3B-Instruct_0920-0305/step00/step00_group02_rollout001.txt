import time
import numpy as np
import cvxpy as cp
import random
import math

def random_binary_vector(n):
    """Generate a random binary vector with exactly n/2 ones."""
    if n % 2 != 0:
        raise ValueError("n must be even")
    h = np.zeros(n)
    ones_indices = np.random.choice(n, n//2, replace=False)
    h[ones_indices] = 1
    return h.astype(int)

def compute_c5(h):
    """Compute C5 bound for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations with (1 - h) for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is for shifts from -(n-1) to (n-1)
    # But we want the maximum of the integral over [0,2] which is scaled by dx
    # So essentially we want the max of the correlations, but we're penalizing max correlation
    # because C5 = total - integral of h*(1-h_shift)
    # total = sum(h) * dx = 1.0 (by constraint)
    # so C5_k = 1 - sum(h_i * h_{i+k}) * dx
    # We want to minimize max of C5_k = max_k (1 - sum(h_i * h_{i+k}) * dx)
    # Equivalent to maximizing min_k sum(h_i * h_{i+k}) = min_k (1 - C5_k)/dx
    # or Maximizing min_k (sum(h_i * h_{i+k})) where k goes from 0 to n-1
    # But C5 = max_k of (1 - sum * dx)
    # Compute all correlation values and take maximum
    correlations = corr[n-1:]  # Only consider shifts 0 to n-1 of full correlation
    # The integral is: sum over all i, h_i * (1 - h_{i+k}) * dx = [sum h_i - sum h_i h_{i+k}] * dx
    # so C5 = (1 - sum h_i h_{i+k}) * dx = 1 - sum h_i h_{i+k} * dx
    # Need sum h_i h_{i+k} for k = 0 to n-1
    # Take max over all k of C5_k = 1 - integral_k = 1 - (sum h_i h_{i+k} * dx)
    # So min sum h_k h_{k+j} = max of 1 - C5 = max_k(1 - C5_k) = 1 - min_k C5_k.
    # Therefore, we want to maximize min_k (sum h_i h_{i+k}) across all valid shifts k
    # That's what maximizes the smallest overlap (a robust optimization)
    # C5 = max_k 1 - (sum h_i h_{i+k}) * dx
    # Minimize max_k (1 - (sum h_i h_{i+k}) * dx) with respect to h
    # = minimize max_k 1 - g_k where g_k = sum h_i h_{i+k}
    # minimize max_k (1 - g_k * dx)  since we assume g_k >= 0
    # = minimize (1 - min_k g_k * dx)
    # So we want to minimize 1 - min_k (g_k * dx) = min_k (g_k * dx) + 1
    # No. Let's look at constraints correctly
    # C5_k = 1 - sum h_i * h_{i+k} * dx
    # Sum h_i * h_{i+k} depends on shift k from 0 to n-1:
    # k=0: sum h_i * h_i = sum h_i^2 = sum h_i (since h_i 0 or 1)
    # k=1: sum h_i * h_{i+1} etc.
    # sum h_i * h_{i+k} = overlap of h with shift k
    # C5_k = 1 - overlap_k * dx for that shift
    # We have overlap_k = sum_{i=0}^{n-1-k} h_i * h_{i+k}
    # Thus, we compute max_k [1 - overlap_k * dx] = max_k (1 - overlap_k*dx)
    # = max_k 1 - max_k (overlap_k*dx)
    # Since overlap_k*dx = overlap_k * dx, minimize max_k (overlaps * dx)
    # That is min_k overlaps * dx. The constraint is min_k overlap_k >= some t
    # and we want to maximize t
    # So maximize t such that for all k: overlap_k >= t
    # we are computing: max_k C5_k where C5_k = 1 - overlap_k * dx
    # Since dx > 0, we have C5_k = 1 - overlap_k * dx, max_C5_k = 1 - min_k overlap_k * dx
    # So minimizing max C5_k = minimize (1 - min_k overlap_k * dx)
    # = minimize max_k (1 - overlap_k * dx) = (1 - min_k (overlap_k)) * dx
    # Argmin max = minimize 1 - min overlap * dx
    # Minimize 1 - min overlap * dx, that is minimize min overlap * dx
    # Since dx fixed positive, minimize min overlap.
    # Find all overlaps for all k from 0 to n-1
    # Evaluate that in the function
    # The following is equivalent computation:
    min_overlap = correlations.min() # correlations are sum of products
    c5 = 1.0 - min_overlap * dx
    return c5

def solve_milp(n, time_limit=300):
    """Solve MINLP formulation for given n."""
    # Variables
    h = cp.Variable(n, boolean=True)
    # Variable for minimum overlap
    t = cp.Variable()
    
    # Constraints: sum h_i = n/2
    constraints = [cp.sum(h) == n // 2]
    
    # For each shift k, define overlap_k = sum h_i * h_{i+k}
    # We'll use linearization approach directly on sum h_i * h_{i+k}
    # Use auxiliary variables w_{i,k} = h_i * h_{i+k} for each i=k..n-1 and the same k.
    # Introduce for each shift k:
    # sum_{i} w_{i,k} with constraint that w_{i,k} = h_i * h_{i+k}
    w = {}
    # For shift k, valid i range is [0, n-k-1]
    for k in range(1, n):  # Note: skip k=0 for shift as it's always large
        if n - k == 0:
            continue
        y = cp.Variable(n - k, boolean=False)  # This should be binary: see
        # The three cliques are: 
        # y <= h[i],   y <= h[i+k]  
        # y >= h[i] + h[i+k] - 1  
        # We also implicitly enforce y >= 0 (since h is [0,1] and linear inequalities make it yet larger)
        w[k] = y
        constraints += [y <= h[:n-k]]
        constraints += [y <= h[k:]]
        constraints += [y >= h[:n-k] + h[k:] - 1]
        # Constrain sum of overlaps >= t
        constraints += [cp.sum(y) >= t]
        
    constraints += [t <= n//2]  # upper bound
    
    # We don't enforce binary explicitly on w because inequalities ensure integer at
    # extreme points defined by h binary. But let's enforce for consistency.
    # Actually, let's define all as binary:
    # Each y >= 0, y <= h, y <= h', y >= h+h'-1
    # Using bounds: 0 <= y <= min(h_i, h_{i+k}) = min(h_i, h_{i+k})
    
    # Minimize max = 1 - min overlap * dx  by maximizing min overlap
    # Equivalent: maximize t subject to t <= overlaps_k for all k
    # So objective is maximize t
    
    prob = cp.Problem(cp.Maximize(t), constraints)
    prob.solve(solver=cp.CBC, verbose=False, max_time=time_limit)
    
    if prob.status != cp.OPTIMAL:
        raise ValueError(f"MILP solver could not find optimal solution at n={n}. Status={prob.status}")
        
    h_sol = np.round(h.value).astype(float)
    dx = 2.0 / n
    min_overlap = np.min([np.sum(h_sol[:n-k] * h_sol[k:]) for k in range(1, n)])
    c5 = 1.0 - min_overlap * dx
    return h_sol, c5

def local_search(h_init, time_limit=200):
    """Perform local search to improve solution."""
    n = len(h_init)
    dx = 2.0 / n
    cur_h = h_init.copy()
    cur_c5 = compute_c5(cur_h)
    
    start = time.time()
    
    while time.time() - start < time_limit:
        ones_idx = np.where(cur_h == 1)[0]
        zeros_idx = np.where(cur_h == 0)[0]
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
            
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        
        # Try swap and evaluate
        cur_h[i] = 0
        cur_h[j] = 1
        new_c5 = compute_c5(cur_h)
        
        if new_c5 < cur_c5:
            # Accept improvement
            cur_c5 = new_c5
        else:
            # Revert
            cur_h[i] = 1
            cur_h[j] = 0
    
    # Final evaluation and return best
    final_c5 = compute_c5(cur_h)
    return cur_h, final_c5

def run(seed=42, budget_s=600, **kwargs):
    start = time.time()
    
    # set random seed
    np.random.seed(seed)
    random.seed(seed)
    
    # Decide parameters
    n = 100  # This gives dx = 0.02
    # For C5 <= 0.3807, need min_overlap >= (1 - 0.3807) / dx = 0.6193 / 0.02 = 30.965 => 31
    # For n=100, dx=0.02, C5 = min k (1 - overlap_k * 0.02) so need min_overlap at least 31
    
    try:
        # Solve MILP
        h_opt, c5_opt = solve_milp(n, time_limit=min(budget_s - (time.time() - start) - 10, 100))
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        
        if remaining > 5:
            # Local search for any remaining time
            best_h, best_c5 = local_search(h_opt, time_limit=remaining)
        else:
            best_h, best_c5 = h_opt, c5_opt
            
    except Exception as e:
        # Fallback: random binary vector
        h_opt = random_binary_vector(n)
        c5_opt = compute_c5(h_opt)
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        
        if remaining > 5:
            best_h, best_c5 = local_search(h_opt, time_limit=remaining)
        else:
            best_h, best_c5 = h_opt, c5_opt
    
    return best_h.tolist(), float(best_c5), n
