def decode(z):
    """Decode raw vector z into normalized interval lengths and gaps."""
    M = len(z) // 2
    z_l = z[:M]
    z_g = z[M:]
    l = np.exp(z_l)
    g = np.exp(z_g)
    l = l / np.sum(l)
    g = g / np.sum(g)
    return l, g

def build_intervals(l, g):
    """Build intervals [a_i, a_i + l_i] from lengths and gaps."""
    a = np.zeros(len(l))
    a[0] = g[0]
    for i in range(1, len(l)):
        a[i] = a[i - 1] + l[i - 1] + g[i]
    b = a + l
    return a, b

def candidate_shifts(a, b):
    """Generate candidate shifts for overlap computation."""
    E = list(a) + list(b)
    E = sorted(set(E))
    T = [0.0]
    for i in range(len(E)):
        for j in range(i, len(E)):
            diff = abs(E[i] - E[j])
            if diff <= 2:
                T.append(diff)
    return sorted(set(T))

def intersection_length(t, a, b):
    """Compute intersection length with shift t for intervals [a_i, b_i]."""
    I = 0.0
    for i in range(len(a)):
        left = max(a[i], a[i] - t)
        right = min(b[i], b[i] - t)
        if right > left:
            I += right - left
    return I

def objective(z):
    """Evaluate the C5 upper bound for given interval configuration."""
    l, g = decode(z)
    a, b = build_intervals(l, g)
    T = candidate_shifts(a, b)
    min_I = float('inf')
    for t in T:
        I_t = intersection_length(t, a, b)
        if I_t < min_I:
            min_I = I_t
    return 1.0 - min_I

def gridify(intervals, n):
    """Convert intervals into a discretized step function."""
    h = np.zeros(n)
    dx = 2.0 / n
    for i in range(n):
        x = i * dx + dx / 2  # center
        for a, b in intervals:
            if a <= x <= b:
                h[i] = 1
                break
    
    # enforce sum = n/2
    cur_sum = np.sum(h)
    needed = n // 2
    if cur_sum != needed:
        delta = needed - cur_sum
        if delta > 0:
            # need to flip 0->1
            # simplest: find first 0 and flip it to 1
            for i in range(n):
                if h[i] == 0:
                    h[i] = 1
                    delta -= 1
                    if delta == 0:
                        break
        else:
            # need to flip 1->0
            for i in range(n):
                if h[i] == 1:
                    h[i] = 0
                    delta += 1
                    if delta == 0:
                        break
    
    return h

def binary_hill_climb(h, max_iter=2000):
    """Fast hill climbing on discrete step function."""
    best_h = h.copy()
    last_c5 = evaluate_erdos_solution(h)
    max_iter = min(max_iter, 2000)
    iter_count = 0
    
    for _ in range(max_iter):
        iter_count += 1
        if iter_count % 100 == 0:
            # periodic check
            c5_assess = evaluate_erdos_solution(best_h)
            if c5_assess < last_c5:
                last_c5 = c5_assess
        # Random swap
        i = np.random.randint(len(h))
        j = np.random.randint(len(h))
        if i == j:
            continue
        if h[i] == h[j]:
            continue
        # swap and evaluate
        h[i], h[j] = h[j], h[i]
        c5_new = evaluate_erdos_solution(h)
        if c5_new < last_c5:
            last_c5 = c5_new
            best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    
    return best_h, last_c5

def evaluate_erdos_solution(h):
    """Compute the overlap integral as per original specification."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)
