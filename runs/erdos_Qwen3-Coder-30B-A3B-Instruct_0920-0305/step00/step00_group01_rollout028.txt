def evaluate_h(h_values):
    """Compute the C5 bound of h_values."""
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute correlation: integral h(x)(1 - h(x+k)) dx
    correlation = np.correlate(h, 1 - h, mode='full') * dx
    # The maximum over all k values (from -1 to 1, corresponding to indices n-1 to 2n-2)
    max_over_k = np.max(correlation[n-1:])
    return max_over_k

def objective(h_values):
    """Return negative of the C5 bound to minimize it"""
    return -evaluate_h(h_values)

def constraint_eq(h_values):
    """Equality constraint: integral h(x) dx = 1"""
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    return np.sum(h) * dx - 1.0

def process_solution(h_values, n_points):
    """Process final solution to ensure bounds and compute final C5"""
    h = np.array(h_values)
    # Ensure bounds on h
    h = np.clip(h, 0.0, 1.0)
    # Normalize 
    sum_h = np.sum(h)
    dx = 2.0 / n_points
    if sum_h != 0.0:
        h = h * (n_points / 2.0) / sum_h
    # Evaluate final C5
    c5 = evaluate_h(h)
    return h, c5
