import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n = 300
    dx = 2.0 / n
    max_time = budget_s * 0.9  # Leave margin
    alpha_start = 1.0
    alpha_max = 10000.0
    alpha_factor = 2.0
    eta_start = 1.0
    line_search_tol = 1e-8
    proj_tol = 1e-9
    
    # Initialize h from Beatty sequence
    beta = (np.sqrt(5) - 1) / 2
    h = np.array([1 if (i * beta) % 1 < 0.5 else 0 for i in range(n)])
    h_sum = np.sum(h)
    if h_sum != n / 2:
        # Adjust to exact sum by swapping
        need_ones = int(n / 2)
        flip_indices = np.argsort(h)[:int(h_sum - need_ones)]
        h[flip_indices] = 1
    
    # Project to feasible set (just in case)
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum - n/2) > proj_tol:
        # Water-filling adjustment
        step = (n/2 - h_sum) / n
        h = np.clip(h + step, 0, 1)
    
    # Track best
    h_best = h.copy()
    best_c5 = float('inf')
    
    # Compute C5 from current h
    c_full = np.correlate(h, 1 - h, mode='full')
    c5_current = np.max(c_full) * dx
    if c5_current < best_c5:
        best_c5 = c5_current
        h_best = h.copy()
    
    # Gradient ascent
    alpha = alpha_start
    t_start = time.time()
    iter_count = 0
    
    while time.time() - t_start < max_time - 0.5:
        # Compute overlaps S_k = sum h_i * h_{i+k}
        c_full = np.correlate(h, h, mode='full')
        shifts = np.arange(0, len(c_full))
        S_all = c_full * dx
        S_all = S_all[n-1:]  # Only the valid shifts k = 0 to n-1
        
        # Soft minimum using log-sum-exp
        weights = np.exp(-alpha * S_all)
        W = np.sum(weights)
        if W == 0:
            break

        # Gradient ∇Φ_α = (Σ k w_k ∇S_k) / W 
        # ∇S_k[i] = h[i+k] + h[i-k] (for k shift)
        grad = np.zeros_like(h)
        for k in range(len(S_all)):
            shifted = np.roll(h, -k, axis=0)
            grad += weights[k] * shifted
            if k > 0:
                shifted = np.roll(h, k, axis=0)
                grad += weights[k] * shifted
                
        grad /= W
        grad = grad[:len(h)]  # Trim leading/trailing zeros
        
        # Backtracking line search
        eta = eta_start
        new_h = h + eta * grad
        while True:
            new_h = np.clip(new_h, 0, 1)
            # Project new_h onto sum constraint
            h_sum = np.sum(new_h)
            if abs(h_sum - n/2) > proj_tol:
                step = (n/2 - h_sum) / n
                new_h = np.clip(new_h + step, 0, 1)
            
            new_S_all = np.correlate(new_h, new_h, mode='full')[n-1:] * dx
            new_phi = -np.log(np.sum(np.exp(-alpha * new_S_all))) / alpha
            old_phi = -np.log(np.sum(np.exp(-alpha * S_all))) / alpha
            if new_phi >= old_phi - line_search_tol or eta < 1e-10:
                break
            eta *= 0.5
            new_h = h + eta * grad
        
        h = new_h
        
        # Update alpha
        if iter_count % 200 == 0 and alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)
        
        # Evaluate
        c_full = np.correlate(h, 1 - h, mode='full')
        c5_current = np.max(c_full) * dx
        if c5_current < best_c5:
            best_c5 = c5_current
            h_best = h.copy()
        
        iter_count += 1
        
    # Final projection and rounding step
    h_best = np.clip(h_best, 0, 1)
    
    # Do one final rounding to binary
    sort_vals = np.sort(h_best)
    median_val = sort_vals[n//2]
    h_final = (h_best > median_val).astype(float)
    
    # Adjust to exact sum if needed
    h_sum_final = np.sum(h_final)
    if abs(h_sum_final - n/2) > 1e-6:
        diff = int(h_sum_final - n/2)
        if diff > 0:
            # Find ones to flip
            ones = np.where(h_final == 1)[0]
            flips = ones[:diff]
            h_final[flips] = 0
        else:
            # Find zeros to flip
            zeros = np.where(h_final == 0)[0]
            flips = zeros[:abs(diff)]
            h_final[flips] = 1
    
    # Local optimization
    # Swap hill-climbing
    c_full = np.correlate(h_final, 1 - h_final, mode='full')
    best_c5_final = np.max(c_full) * dx
    
    for _ in range(50000):
        if time.time() - t_start > max_time:
            break
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # Try swap
        new_h = h_final.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        # Recompute without FFT convolution for small n
        c_full = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(c_full) * dx
        if new_c5 < best_c5_final:
            h_final = new_h
            best_c5_final = new_c5

    return (h_final.tolist(), float(best_c5_final), int(n))
