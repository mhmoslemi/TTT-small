import numpy as np
import time
import random

def project_to_feasible(h, target_sum):
    """Project h to feasible set: sum=h, 0 <= h[i] <= 1"""
    # Simple projection onto the set {h | sum(h) = target_sum, 0 <= h <= 1}
    h = np.array(h)
    n = len(h)
    # Clip values to [0,1]
    clipped_h = np.clip(h, 0, 1)
    current_sum = np.sum(clipped_h)
    
    if abs(current_sum - target_sum) < 1e-10:
        return clipped_h
    
    # Adjust values to hit target sum
    if current_sum > target_sum:
        # Need to reduce sum: decrease larger values
        indices = np.argsort(clipped_h)[::-1]
        diff = current_sum - target_sum
        for i in indices:
            if diff <= 0:
                break
            reduce = min(diff, clipped_h[i])
            clipped_h[i] -= reduce
            diff -= reduce
    else:
        # Need to increase sum: increase smaller values
        indices = np.argsort(clipped_h)
        diff = target_sum - current_sum
        for i in indices:
            if diff <= 0:
                break
            increase = min(diff, 1 - clipped_h[i])
            clipped_h[i] += increase
            diff -= increase
            
    return clipped_h

def compute_max_overlap(h):
    """Compute max of ∫ h(x)(1 - h(x+k)) dx over all k"""
    # Note: correlation using full mode includes all possible lags, which 
    # effectively considers all shifts possible within finite domain
    # This corresponds to zero-padding
    N = len(h)
    dx = 2.0 / N
    # h(x) -> h(i) for i = 0 to N-1 (on [0,2])
    # In order to compute ∫ h(x) (1-h(x+k)) dx, we use correlation:
    # Consider h as finite sequence, correlation with 1-h.
    # Then unpacking the mode='full' correlation: corresponds to return size 2*N-1.
    reversed_h = 1 - h
    correlations = np.correlate(h, reversed_h, mode="full")
    # Discard first and last N-1 elements since those are correlation outside 
    # the meaningful overlapping region due to zero-padding compared to domain size.
    correlations_half = correlations[N-1:2*N-1]
    # Each element corresponds to a shift (lag) from -(N-1) to (N-1)
    # The corresponding shift k = (lag_index - (N-1)) * dx
    # In the return value, we might need to take the max of overlaps
    overlaps = 1 - correlations_half * dx  # Because overlap = 1 - correlation  
    return np.max(overlaps)

def evaluate_erdos_solution(h):
    """Use helper function to compute max overlap (ensures compatibility with challenge)"""
    return compute_max_overlap(h)

def gradient_descend_step(h, step_size=0.01):
    """Modify h to decrease max overlap by applying projected gradient descent on one k"""
    N = len(h)
    dx = 2.0 / N
    # Compute all overlaps: O(k) = 1 - C(k)
    reversed_h = 1 - h
    correlations = np.correlate(h, reversed_h, mode="full")
    correlations_half = correlations[N-1:2*N-1]
    overlaps = 1 - correlations_half * dx
    
    # Find max overlap and corresponding k index
    max_idx = np.argmax(overlaps)
    max_k = (max_idx - (N-1)) * dx
    
    # Compute gradient of max overlap w.r.t h using finite differences for simplicity
    # For shift offset = max_k/dx, we need to represent some of the correlation term:
    # Recall that O(k) = 1 - ∑ h[i] h[i+offset] * dx
    # So gradient w.r.t h[j] = -dx * (h[j+offset] + h[j-offset]) if those indices exist
    offset = int(max_k / dx)
    
    gradient = np.zeros(N)
    # Advance = some case of j -> j+offset case
    for j in range(N):
        if 0 <= j + offset < N:
            gradient[j] -= dx * h[j + offset]
        if 0 <= j - offset < N:
            gradient[j] -= dx * h[j - offset]
        
        # Add constant 1 from the derivative of 1
        gradient[j] += 1.0
          
    # Update h: 
    h_new = h - step_size * gradient
    # Project to feasible set
    target_sum = N / 2.0
    h_new = project_to_feasible(h_new, target_sum)
    
    return h_new
