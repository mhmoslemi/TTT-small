from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The full correlation has 2*n - 1 elements
    # We want the maximum value among shifts from -1 to 1
    max_correlation = np.max(correlation[n//2 : 3*n//2])
    return max_correlation * dx
