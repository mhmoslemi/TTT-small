import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time
import math

def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant for given h values."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the one we want
    max_val = np.max(corr) * dx
    return max_val

def initial_h_values():
    """Return initial configuration from the problem."""
    n_points = 45
    dx = 2.0 / n_points
    h = np.ones(n_points) * 0.5
    # Shift the initial guess to be more structured
    h[:20] = 1.0
    h[20:30] = 0.0
    h[30:] = 1.0
    h = h * (1.0 / (np.sum(h) * dx))
    return h

def objective_function(h_values):
    """Objective function to minimize."""
    return evaluate_erdos_solution(h_values)

def constraint_sum(h_values):
    """Constraint that integral equals 1."""
    n = len(h_values)
    dx = 2.0 / n
    return np.sum(h_values) * dx - 1.0

def run(seed=42, budget_s=600, **kwargs):
    """Run optimization to find improved upper bound for C5."""
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Use the initial solution as a starting point
    h_init = initial_h_values()
    n_points = len(h_init)
    
    # Define bounds for each variable: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization using differential evolution with constraints
    def constrained_objective(h_vals):
        # Ensure h_vals is within bounds
        h_vals = np.clip(h_vals, 0, 1)
        return objective_function(h_vals)
    
    # Add constraint that sum(h)*dx = 1 (i.e., sum(h) = n_points/2)
    def constraint_func(h_vals):
        dx = 2.0 / n_points
        return np.sum(h_vals) * dx - 1.0
    
    # Define constraint dictionary for scipy.optimize
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Optimization parameters
    maxiter = 1000
    disp = False
    
    # Run optimization
    best_h = None
    best_c5 = float('inf')
    
    # Try multiple random restarts to improve results
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Random initial guess
        h_start = np.random.rand(n_points) * 0.5
        # Apply sum constraint
        dx = 2.0 / n_points
        h_start = h_start * (n_points / 2.0 / np.sum(h_start))
        
        try:
            result = minimize(
                constrained_objective,
                h_start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': maxiter, 'disp': disp}
            )
            
            if result.success:
                h_opt = np.clip(result.x, 0, 1)
                c5_val = evaluate_erdos_solution(h_opt)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt.copy()
        except:
            continue
            
        # Add a few more iterations, if time allows
        if time.time() - start_time > budget_s - 10:
            break
            
    # Final optimization using differential evolution if we still have time
    if best_h is None or time.time() - start_time < budget_s - 10:
        try:
            result_de = differential_evolution(
                constrained_objective,
                bounds,
                seed=seed,
                maxiter=50,
                popsize=15,
                mutation=(0.5, 1),
                recombination=0.7,
                disp=False
            )
            
            if result_de.success:
                h_de = np.clip(result_de.x, 0, 1)
                c5_de = evaluate_erdos_solution(h_de)
                if c5_de < best_c5:
                    best_c5 = c5_de
                    best_h = h_de.copy()
        except:
            pass
    
    # If none better than initial, return initial
    if best_h is None:
        best_h = initial_h_values()
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
