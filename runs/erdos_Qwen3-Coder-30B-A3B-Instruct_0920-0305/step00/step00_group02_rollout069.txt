from scipy import signal

def evaluate_overlap(h, dx):
    # Compute the overlap integral C_5 = max_k int h(x)(1-h(x+k)) dx
    # Using full convolution to get all lags
    conv = signal.correlate(h, 1 - h, mode='full')
    # The maximum is over all possible shifts k
    max_overlap = np.max(conv) * dx
    return max_overlap

    # Penalize if not exactly 1.0
    penalty = 1000 * abs(integral - 1.0)
    # Compute the overlap
    overlap = evaluate_overlap(h, dx)
    return overlap + penalty
