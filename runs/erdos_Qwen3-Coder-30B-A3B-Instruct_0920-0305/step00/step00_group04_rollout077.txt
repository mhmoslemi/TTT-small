def evaluate_erdos_solution(h_values):
    # This evaluates the true C5 value using the provided tool
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    # Compute correlations for all lags (full mode)
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def pad_and_correlate(h_vec, g_vec, n):
    # Pad to next power of two length for efficient FFT
    L = 1
    while L < 2 * n - 1:
        L *= 2
    h_padded = np.pad(h_vec, (0, L - len(h_vec)), mode="constant")
    g_padded = np.pad(g_vec, (0, L - len(g_vec)), mode="constant")
    H = fft(h_padded)
    G = fft(g_padded)
    corr_fft = ifft(H * np.conj(G))
    # Return first 2*n - 1 values (correlations at all lags)
    corr = np.real(corr_fft[:2*n - 1])
    return corr

def compute_c5(h, n, dx):
    corr = pad_and_correlate(h, 1 - h, n)
    # keep values for lags in [-n+1, n-1]
    max_corr = np.max(corr)
    return max_corr * dx

def generate_binary_seed(n, method='random'):
    """Generate a random binary vector with sum n/2."""
    half = n // 2
    b = np.zeros(n)
    b[:half] = 1.0
    np.random.shuffle(b)
    return b

def generate_binary_seed_from_initial(n, h0):
    """Embed an initial construction into a larger grid."""
    # Pad or downsample h0 to size n
    if len(h0) > n:
        h = h0[:n]
    elif len(h0) < n:
        # Replicate pattern with periodicity
        h = np.tile(h0, (n // len(h0)) + 1)[:n]
    else:
        h = h0.copy()
    
    # Count 1s and 0s
    ones = np.sum(h)
    if ones > n // 2:
        # We have too many 1s, need to convert some to 0s
        indices = np.where(h == 1)[0]
        num_to_flip = int(ones - n // 2)
        to_flip = np.random.choice(indices, size=num_to_flip, replace=False)
        h[to_flip] = 0
    elif ones < n // 2:
        # We have too few 1s, need to convert some to 1s
        indices = np.where(h == 0)[0]
        num_to_flip = int(n // 2 - ones)
        to_flip = np.random.choice(indices, size=num_to_flip, replace=False)
        h[to_flip] = 1
    
    return h

def simulated_annealing_binary(h0, n, dx, max_time_s, seed=42):
    # For performance, only a few restarts are practical
    np.random.seed(seed)
    
    best_h = h0.copy()
    best_c5 = compute_c5(h0, n, dx)
    
    # Phase A: Simulated Annealing with warm restarts
    start = time.time()
    time_left = lambda: max_time_s - (time.time() - start)
    
    # from Strategy, set a full SA time slice (let's say 30 sec/round for 600s)
    tenure = min(30, time_left() * 0.7)
    T0 = 0.05
    alpha = 0.9995
    
    restarts = 0
    iteration = 0
    
    while time_left() > 10:
        restarts += 1
        local_h = h0.copy()
        local_c5 = compute_c5(local_h, n, dx)
        
        T = T0
        
        inner_iter = 0
        while time_left() > 0.1 and T > 1e-6:
            inner_iter += 1
            # Pick a 1 and 0 to swap
            ones = np.where(local_h == 1)[0]
            zeros = np.where(local_h == 0)[0]
            if len(ones) > 0 and len(zeros) > 0:
                pos1 = np.random.choice(ones)
                pos0 = np.random.choice(zeros)
                # do the swap
                local_h[pos1] = 0
                local_h[pos0] = 1
                # Compute new C5
                new_c5 = compute_c5(local_h, n, dx)
                delta = new_c5 - local_c5
                
                accept_prob = math.exp(-delta / T) if delta > 0 else 1.0
                
                if np.random.rand() < accept_prob:
                    local_c5 = new_c5
                else:
                    # undo swap
                    local_h[pos1] = 1
                    local_h[pos0] = 0
                
                if local_c5 < best_c5:
                    best_c5 = local_c5
                    best_h = local_h.copy()
                    
                T *= alpha
            
            if inner_iter > 100000 or (inner_iter % 10000 == 0 and time_left() < 5):
                break
                
    return best_h, best_c5

def project_onto_simplex(y, target_sum):
    """Project vector y onto capped simplex with sum = target_sum"""
    def S(lam):
        # S(lam) = sum of clipped values
        return np.sum(np.clip(y - lam, 0, 1)) - target_sum

    # find zero of S by bisection
    low, high = np.min(y) - 1, np.max(y) + 1
    try:
        sol = root_scalar(S, bracket=[low, high], method='brentq')
        lam = sol.root
    except ValueError:
        # fallback
        lam = np.mean(y)
    x = np.clip(y - lam, 0, 1)
    # adjust to ensure correct sum
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) > 1e-10:
        # Adjust by shifting values
        diff = target_sum - current_sum
        # find indices with room to increase/decrease
        low_indices = np.where(x < 1)[0]
        high_indices = np.where(x > 0)[0]
        if len(low_indices) > 0 and diff > 0:
            # distribute diff to indices that can take more
            if len(low_indices) >= len(high_indices):
                idx = np.random.choice(low_indices, size=min(int(diff * 5),len(low_indices)), replace=True)
                x[idx] += diff / len(idx)
            else:
                idx = np.random.choice(low_indices, size=max(1, int(diff*10)), replace=True)
                x[idx] += diff / len(idx)
        elif len(high_indices) > 0 and diff < 0:
            # decrease some
            if len(high_indices) >= len(low_indices):
                idx = np.random.choice(high_indices, size=min(int(-diff*5), len(high_indices)), replace=True)
                x[idx] += diff / len(idx)
            else:
                idx = np.random.choice(high_indices, size=max(1, int(-diff*10)), replace=True)
                x[idx] += diff / len(idx)

    return np.clip(x, 0, 1)

def smooth_max_surrogate(f, gamma):
    """Compute \frac{1}{\gamma}\log\sum e^{\gamma f} """
    return (1 / gamma) * np.log(np.sum(np.exp(gamma * f)))

def gradient_of_smooth_max(h, gamma):
    """Compute gradient of smooth max over all correlations."""
    n = len(h)
    # Compute forward contribution: h_i * (1 - h_{i+delta})
    # Recursive sums
    dx = 2.0 / n
    df = np.zeros(2 * n - 1)  # correlation values for each lag
    h_adj = np.pad(h, (0, n - 1), mode="linear_ramp", end_values=(0, 0))
    inv_h_adj = 1 - h_adj
    # We compute all lag correlations
    # use correntive FFT
    f_fft = np.zeros(2 * n - 1, dtype=complex)
    #perform only for those that make sense
    for l in range(2*n - 1):
        # Equivalent to convolution of h and (1-h) with lag
        if l < n:
            # forward corr
            c = np.sum(h[:n-l] * (1 - h[l:]))
        else:
            # backward corr
            c = np.sum(h[l-n+1:] * (1 - h[:n-(l-n+1)]))
        df[l] = c * dx
    
    # Compute weights
    exp_f = np.exp(gamma * df)
    w = exp_f / np.sum(exp_f)

    # Now compute the gradient
    # \frac{\partial}{\partial b_i} \sum_l w_l f_l
    grad_f = np.zeros(n)
    # Add contributions for each lag l:
    # f_l = \sum_i h_i (1 - h_{i+l}) for lth lag
    # grad_df/dh_i = (1 - h_{i+l}) * indicator_left + (-h_{i-l}) * indicator_right
    for l in range(2 * n - 1):
        # Non-zero part when l corresponds to valid positions
        if l == 0:
            # Both sides of identity value
            grad_f += w[l] * (1 - h)  # since h_{i+0} = h_i
        elif l < n:
            # Positive lag: forward offset
            # gradient contrib for h[i] = 1 - h[i+l], where i <= n-l-1
            grad_f[:-l] += w[l] * (1 - h[l:])
        else:
            # Negative lag: backward offset
            # gradient contrib for h[i] = -h[i-l], where i >= l
            grad_f[l:] -= w[l] * h[:n-l]
    return grad_f

def refine_with_gradient_descent(h0, n, dx, max_time_s):
    # Initialize with binary h0
    h = h0.copy()
    h = h / np.sum(h) * (n / 2)  # force exact sum
    c5_current = compute_c5(h, n, dx)
    
    gamma = 5.0
    alpha = 0.1 * dx
    gamma_max = 200.0
    gamma_inc = 1.5
    patience = 5
    patience_count = 0
    
    start = time.time()
    
    # First evaluate exact C5
    c5_exact = evaluate_erdos_solution(h)
    
    while time.time() - start < max_time_s * 0.25:
        # Smooth surrogate
        corr = pad_and_correlate(h, 1 - h, n)
        f_vals = corr * dx
        
        # Compute smooth max
        Phi = smooth_max_surrogate(f_vals, gamma)
        
        # Get gradient
        grad = gradient_of_smooth_max(h, gamma)
        
        # Gradient step
        h_new = h - alpha * grad
        
        # Project back onto feasible set
        h_new = project_onto_simplex(h_new, n / 2)
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluating true C5 after project step
        c5_new_exact = evaluate_erdos_solution(h_new)
        h = h_new.copy()
        c5_exact = c5_new_exact
        
        # Assess improvement
        if abs(c5_exact - c5_current) < 1e-8:
            patience_count += 1
        else:
            patience_count = 0
            c5_current = c5_exact
            
        if patience_count >= patience:
            break
            
        if time.time() - start > 0.1 * max_time_s:
            break
            
        # Increase gamma
        if gamma < gamma_max:
            gamma *= gamma_inc

    # Ensure final projection
    h_clipped = np.clip(h, 0, 1)
    h_final = project_onto_simplex(h_clipped, n / 2)
    
    # Final C5
    c5_final = evaluate_erdos_solution(h_final)
    
    return h_final, c5_final

def run(seed=42, budget_s=600):
    start = time.time()
    
    # Select appropriate grid size
    n = 60  # good compromise between speed and resolution
    dx = 2.0 / n
    time_left = lambda: budget_s - (time.time() - start)

    # Phase A: Binary SA
    h0 = generate_binary_seed_from_initial(n, initial_h_values())
    h_best, c5_best = simulated_annealing_binary(h0, n, dx, time_left(), seed=seed)
    
    # Phase B: refinement
    if time_left() > 10 and c5_best <= 0.3807:
        # Refine continuous
        h_refined, c5_refined = refine_with_gradient_descent(h_best, n, dx, time_left())
        if c5_refined < c5_best:
            h_best = h_refined
            c5_best = c5_refined
    
    # Ensure final projection with exact constraints
    h_final = project_onto_simplex(h_best, n / 2)
    c5_eval = evaluate_erdos_solution(h_final)
    
    return (h_final, c5_eval, n)
