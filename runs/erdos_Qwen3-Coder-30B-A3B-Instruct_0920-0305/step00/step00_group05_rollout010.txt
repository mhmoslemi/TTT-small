import numpy as np
import time
import random

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    time_end = start + budget_s - 0.5
    
    n_points = 200
    dx = 2.0 / n_points
    
    best_h = None
    best_c5 = float('inf')
    
    # Random restarts
    restarts = 0
    while time.time() < time_end:
        restarts += 1
        # Generate random binary vector with exactly half ones
        h = np.zeros(n_points, dtype=np.float64)
        ones_indices = np.random.choice(n_points, n_points // 2, replace=False)
        h[ones_indices] = 1.0
        
        # Local annealing
        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        
        # Parameters for simulated annealing
        max_local_iters = 20000
        T0 = 0.05
        local_iter = 0
        
        while time.time() < time_end and local_iter < max_local_iters:
            local_iter += 1
            # Choose two random indices with different values
            i = np.random.randint(n_points)
            j = np.random.randint(n_points)
            while h[i] == h[j]:
                i = np.random.randint(n_points)
                j = np.random.randint(n_points)
            
            # Swap values
            h[i], h[j] = h[j], h[i]
            
            # Compute new C5
            new_c5 = compute_c5(h)
            delta = new_c5 - cur_c5
            
            # Accept with probability based on simulated annealing
            T = max(1e-6, T0 * (1 - local_iter / max_local_iters))
            if delta <= 0 or np.random.rand() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                # Revert swap
                h[i], h[j] = h[j], h[i]
                
            if best_c5 <= 0.3807:
                break
                
    return (best_h.tolist(), best_c5, n_points)
