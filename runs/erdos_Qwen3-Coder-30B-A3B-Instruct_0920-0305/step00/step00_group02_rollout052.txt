import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize_scalar

def project_to_feasible(h, target_sum):
    """
    Projects h onto the set {h : 0 ≤ h[i] ≤ 1, sum(h) = target_sum}
    Using water-filling for sum constraint.
    """
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Water-filling: find λ such that sum(clip(h + λ, 0, 1)) = target_sum
    def objective(lam):
        h_new = h + lam
        h_clipped = np.clip(h_new, 0, 1)
        return np.sum(h_clipped) - target_sum

    # Find λ such that objective(lam) = 0
    a, b = -1.0, 1.0
    while objective(a) > 0: a -= 1.0
    while objective(b) < 0: b += 1.0
    lam = minimize_scalar(objective, bounds=(a, b), method='bounded').x
    h = np.clip(h + lam, 0, 1)
    return h

def beatty_initialization(n, irrational=0.618033988749895):
    """
    Initialize h using the Beatty sequence with an irrational β.
    """
    h = np.zeros(n)
    for i in range(n):
        if (i * irrational) % 1 < 0.5:
            h[i] = 1.0
    h = project_to_feasible(h, n / 2)
    return h

def compute_overlaps(h, dx):
    """
    Compute all overlaps S_k = ∫ h(x) h(x+k) dx using convolution.
    """
    # 1 - h for correlation
    one_minus_h = 1 - h
    cross_corr = np.correlate(h, one_minus_h, mode='full')
    # Keep only valid shifts
    n = len(h)
    valid_overlaps = cross_corr[n - 1:]
    return valid_overlaps * dx

def logsumexp_smooth_min(h, alpha, dx):
    """
    Smooth approximation of min_k S_k using log-sum-exp.
    """
    overlaps = compute_overlaps(h, dx)
    w = np.exp(-alpha * overlaps)
    total_w = np.sum(w)
    if total_w == 0:
        return 0.0
    return - (1.0 / alpha) * np.log(total_w)

def gradient_of_smooth_min(h, alpha, dx):
    """
    Compute gradient of logsumexp_smooth_min with respect to h.
    """
    n = len(h)
    overlaps = compute_overlaps(h, dx)
    w = np.exp(-alpha * overlaps)
    total_w = np.sum(w)
    if total_w == 0:
        return np.zeros_like(h)

    weights = w / total_w

    # Compute gradient efficiently using convolutions
    g = np.zeros(n)
    for k in range(n):
        start = max(0, k)
        end = min(n, n - k)
        if end > start:
            g[start:end] += weights[k] * h[k:end-k]
            if k > 0:
                g[k:end] += weights[k] * h[:end-k]
    return g

def backtracking_line_search(h, grad, alpha, dx, step_size=1.0, decay=0.5, tol=1e-8):
    """
    Backtracking line search to ensure monotonicity.
    """
    h_test = h + step_size * grad
    h_test = project_to_feasible(h_test, len(h) / 2)
    new_value = logsumexp_smooth_min(h_test, alpha, dx)
    old_value = logsumexp_smooth_min(h, alpha, dx)
    while new_value < old_value - tol and step_size > 1e-15:
        step_size *= decay
        h_test = h + step_size * grad
        h_test = project_to_feasible(h_test, len(h) / 2)
        new_value = logsumexp_smooth_min(h_test, alpha, dx)
    return step_size
