import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h_values."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute full correlation of h with (1-h)
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value gives the C5 bound
    c5_bound = np.max(corr) * dx
    return c5_bound

def objective(h_values):
    """Minimize negative of C5 bound (since we want to minimize C5)."""
    return -evaluate_h(h_values)

def constraint_eq(h_values):
    """Equality constraint: integral of h should be 1."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    return np.sum(h_values) * dx - 1.0

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start timer
    start_time = time.time()

    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = initial_h_values.copy()
    else:
        # Default: uniform distribution
        n_points = 45
        h_init = np.ones(n_points) * (2.0 / n_points)  # Ensures integral = 1
    
    # Prepare bounds (0 to 1 for each h[i])
    bounds = [(0, 1) for _ in range(len(h_init))]
    
    # Use differential evolution for global optimization
    # We define constraints in the way scipy wants
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Run optimization with time budget
    best_c5 = float('inf')
    best_h = h_init.copy()
    
    # Sample a few random initial points to diversify search
    n_evals = 0
    max_evals = 1000  # Limit for safety
    
    while time.time() - start_time < budget_s and n_evals < max_evals:
        try:
            # Use differential evolution with current h_init
            result = differential_evolution(
                objective,
                bounds,
                args=(),
                constraint=cons,
                seed=seed + n_evals,
                maxiter=100,
                popsize=15,
                mutation=(0.5, 1),
                recombination=0.7,
                disp=False,
                polish=True
            )
            
            if result.success:
                h_opt = result.x
                c5 = -result.fun  # Because we minimized negative
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                
                # Early stopping if we have a good value
                if best_c5 < 0.379:
                    break
                    
            n_evals += 1
            
        except Exception as e:
            # If there's an error we just skip it
            n_evals += 1
            continue
    
    # Final verification of best solution
    final_c5 = evaluate_h(best_h)
    final_h_values = best_h.copy()
    
    # Return final result
    return (final_h_values, final_c5, len(final_h_values))
