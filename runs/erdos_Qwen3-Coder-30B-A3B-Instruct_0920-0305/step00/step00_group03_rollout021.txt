def project_onto_simplex_box(h, target_sum):
    """Project h onto the simplex with sum=target_sum and bounds [0,1]"""
    # Sort values in descending order
    h_sorted = np.sort(h)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(h_sorted)
    # Find the threshold
    n = len(h)
    threshold = (cumsum - target_sum) / np.arange(1, n + 1)
    # Find the largest k such that threshold[k] <= h_sorted[k]
    k = np.argmax(h_sorted >= threshold)
    if h_sorted[k] < threshold[k]:
        k += 1
    # Compute Lambda
    Lambda = threshold[k-1] if k > 0 else threshold[0]
    # Apply projection
    h_proj = np.clip(h - Lambda, 0, 1)
    # Adjust to maintain sum
    adjustment = (target_sum - np.sum(h_proj)) / len(h_proj)
    h_proj += adjustment
    # Ensure bounds
    h_proj = np.clip(h_proj, 0, 1)
    return h_proj

    # Ensure h is correctly shaped for correlate
    if len(h) == 0:
        return float('inf')

    corr = np.correlate(h, 1-h, mode='full')
    # Scale by dx
    dx = 2.0 / len(h)
    corr *= dx
    # Return maximum value
    return np.max(corr)
