def project_to_feasible(v, n_points):
    """Project vector v onto feasible set: sum = n_points/2, each component in [0,1]"""

    h = v.copy()
    
    # Sum constraint
    sum_h = np.sum(h)
    desired_sum = n_points / 2.0
    
    # Adjust via projection of all values
    diff = desired_sum - sum_h
    
    if abs(diff) < 1e-10:
        # Already satisfies sum constraint, clip to [0,1]
        h = np.clip(h, 0, 1)
        return h

    h = np.clip(h, 0, 1)

    if abs(actual_sum - desired_sum) > 1e-6:
        # Get indices where we can increase or decrease

        # Direct method: add correction
        correction = (desired_sum - actual_sum) / len(h)
        h += correction
        
        # Re-clip to [0,1]
        h = np.clip(h, 0, 1)
        
    return h

    # Return the maximum value
    return np.max(c)
