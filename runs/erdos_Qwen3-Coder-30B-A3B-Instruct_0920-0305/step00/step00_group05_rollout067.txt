from scipy.optimize import differential_evolution
import time

def evaluate_overlap(h):
    """Evaluate the maximum overlap integral for given h"""
    n_points = len(h)
    dx = 2.0 / n_points

    # The maximum value is the desired C5
    max_correlation = np.max(correlation)
    return max_correlation * dx
