import numpy as np
import time
import math
from scipy import fft
from scipy.optimize import minimize


def evaluate_c5(h, dx):
    """Evaluate C5 using FFT for speed"""
    n = len(h)
    # Pad to next power of 2
    L = 1 << (int(math.log2(2*n-1)) + 1)
    h_padded = np.pad(h, (0, L - n), 'constant')
    g_padded = np.pad(1 - h, (0, L - n), 'constant')
    
    H = fft.fft(h_padded)
    G = fft.fft(g_padded)
    corr = fft.ifft(H * np.conj(G)).real
    corr = corr[:2*n-1]
    return np.max(corr) * dx

            corr = 0
            for i in range(n):
                j = i + k
                if 0 <= j < n:
                    corr += h[i] * (1 - h[j])
            max_corr = max(max_corr, corr * dx)
        return max_corr
