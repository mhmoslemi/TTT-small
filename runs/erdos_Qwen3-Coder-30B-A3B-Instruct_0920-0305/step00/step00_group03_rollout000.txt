import numpy as np
import time
from scipy.optimize import minimize

def project_onto_simplex_box(h, target_sum):
    """Project h onto the simplex with sum = target_sum and bounds [0,1]"""
    # First clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Then adjust to meet sum constraint using iterative method
    h_sum = np.sum(h)
    if abs(h_sum - target_sum) < 1e-10:
        return h
    
    # Sort h in descending order
    sorted_indices = np.argsort(h)[::-1]
    sorted_h = h[sorted_indices]
    
    # Find threshold
    cumsum = np.cumsum(sorted_h)
    threshold = (cumsum - target_sum) / np.arange(1, len(h) + 1)
    
    # Find where to cut
    pos = np.argmax(threshold >= sorted_h)
    if pos == 0 and threshold[0] >= sorted_h[0]:
        pos = len(h)
    
    # Apply threshold
    if pos == 0:
        lambda_star = threshold[0]
    else:
        lambda_star = threshold[pos-1]
    
    # Project using threshold
    result = np.maximum(0, h - lambda_star)
    
    # Correct slight deviation from target sum
    current_sum = np.sum(result)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Distribute diff to elements that are not at bounds
        indices = np.where((result > 0) & (result < 1))[0]
        if len(indices) > 0:
            step = diff / min(len(indices), abs(diff)*100)  # small step
            for i in indices[:int(abs(diff)*100)]:
                result[i] += step
                if result[i] > 1:
                    result[i] = 1
                elif result[i] < 0:
                    result[i] = 0
    return result

def evaluate(h):

    n = len(h)
    dx = 2.0 / n
    S_raw = np.correlate(h, 1 - h, mode='full')
    return np.max(S_raw) * dx

            grad[i + l] -= h[i]

        shift = -l
        # i from shift to n-1
        for i in range(shift, n):
            grad[i] -= h[i - shift]
            grad[i - shift] += 1 - h[i]
    return grad
