import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    # Compute all cross-correlations
    correlations = np.correlate(h_values, 1 - np.array(h_values), mode="full")
    # The overlap integral is 1 - correlation
    overlaps = 1 - correlations * dx
    # Return the maximum overlap
    return np.max(overlaps)

def initial_h_values():
    # Standard construction from 47 samples as starting point
    n = 47
    h = np.zeros(n)
    # Set some values to 1, others to 0
    # Allow some fractional values for later optimization
    h[:23] = 1
    h[23:24] = 0.5
    h[24:] = 0
    return h / np.sum(h) * 2.0

def solve_lp_feasibility(h_vars, p_vars, shift_constraints, sum_constraint, w_target):
    # Constraint: sum h = sum_constraint
    constraints = [cp.sum(h_vars) == sum_constraint]
    # Constraint: 0 <= h <= 1
    constraints += [0 <= h_vars, h_vars <= 1]
    # Constraints from McCormick envelopes for each shift:
    for d_idx, d in enumerate(shift_constraints):
        constraints += d
    # Constraint from target overlap
    for d_idx, p_list in enumerate(p_vars):
        constraints += [cp.sum(p_list) >= w_target[d_idx]]
    
    # Solve the linear program (feasibility check)
    # Objective is irrelevant here, just check if any solution exists
    prob = cp.Problem(cp.Minimize(0), constraints)
    try:
        prob.solve(solver=cp.GLPK, verbose=False)
        return prob.status == cp.OPTIMAL or prob.status == cp.INFEASIBLE
    except:
        return False

def bisection_lp_method(n_points, budget_s):
    dx = 2.0 / n_points
    # Precompute all possible shift indices
    shifts_list = list(range(-n_points+1, n_points)) # that's 2n-1 offsets total
    w_targets = []
    p_vars = []
    shift_constraints = []
    
    start_time = time.time()
    h_star = None
    lower_bound = 0.0
    upper_bound = 0.38092  # Known bound from problem statement
    best_bound = upper_bound

    # Precompute shifts and constraints for each offset
    # For offset d, some indices overlap
    for d in shifts_list:
        # Define indices that can be constrained
        # p_{i,d} for i = 0 to n-1-d for d>0, or i = -d to n-1 for d < 0
        if d >= 0:
            valid_indices = list(range(0, n_points - d))
            p_list = [cp.Variable(1, name=f"p_{i}_{d}") for i in valid_indices]
            p_vars.append(p_list)
            constraints = []
            for i in valid_indices:
                u = cp.Variable(1, name=f"h_{i}")
                v = cp.Variable(1, name=f"h_{i+d}")
                p = p_list[i]
                constraints.append(p >= 0)
                constraints.append(p >= u + v - 1)
                constraints.append(p <= u)
                constraints.append(p <= v)
            shift_constraints.append(constraints)
        else:
            # symmetric case but interpreted as reverse index
            valid_indices = list(range(-d, n_points))
            p_list = [cp.Variable(1, name=f"p_{i}_{d}") for i in valid_indices]
            p_vars.append(p_list)
            constraints = []
            for i in valid_indices:
                u = cp.Variable(1, name=f"h_{i}")
                v = cp.Variable(1, name=f"h_{i+d}")
                p = p_list[i - (-d)]
                constraints.append(p >= 0)
                constraints.append(p >= u + v - 1)
                constraints.append(p <= u)
                constraints.append(p <= v)
            shift_constraints.append(constraints)

    max_iterations = 100
    iteration = 0
    while iteration < max_iterations and time.time() - start_time < budget_s - 1.0:
        iteration += 1
        mid_point = (lower_bound + upper_bound) / 2.0
        w_target = [(1 - mid_point) / dx] * len(shifts_list)

        # Construct LP again with updated weight
        # Variables
        h_vars = cp.Variable(n_points, name="h")
        p_vars_new = []
        shift_constraints_new = []

        # This needs to be simpler - restructure using limits correctly
        all_p_vars = []
        all_constraints = []

        total_shifts = len(shifts_list)
        for d_idx, d in enumerate(shifts_list):
            if d >= 0:
                indices = list(range(0, n_points - d))
            else:
                indices = list(range(-d, n_points))
            p_list = [cp.Variable((1,), name=f"p_{i}_{d}") for i in indices]
            all_p_vars.append(p_list)
            constraints = []
            for i in indices:
                if d >= 0:
                    u = h_vars[i]
                    v = h_vars[i + d]
                else:
                    u = h_vars[i]
                    v = h_vars[i + d]
                p = p_list[i if d >= 0 else i + d]
                constraints.append(p >= 0)
                constraints.append(p >= u + v - 1)
                constraints.append(p <= u)
                constraints.append(p <= v)
            all_constraints.append(constraints)

        # Construct sum constraint
        sum_constraint = cp.sum(h_vars) == n_points / 2.0

        # Build up constraints
        n_vars = [sum_constraint]
        for constraints in all_constraints:
            n_vars += constraints
        
        # Add objective (arbitrary - just check feasibility)
        prob = cp.Problem(cp.Minimize(0), n_vars)
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status == cp.OPTIMAL:
                # Feasible; save h solution and update upper bound
                h_solution = h_vars.value
                c5_bound = evaluate_erdos_solution(h_solution)
                if c5_bound < best_bound:
                    best_bound = c5_bound
                    h_star = h_solution.copy()
                upper_bound = mid_point
            else:
                # Infeasible; increase lower bound
                lower_bound = mid_point
        except:
            # Solving failed, use higher lower bound
            lower_bound = mid_point

        # Break if progress stalls significantly
        if abs(upper_bound - lower_bound) < 1e-5:
            break

    if h_star is None:
        # Fallback to current best known
        h_star = initial_h_values()
        best_bound = evaluate_erdos_solution(h_star)
    
    # Apply an additional postprocess step to smooth and refine
    h_final = h_star.copy()
    # Clip values to [0,1] and scale to sum to 1
    h_final = np.clip(h_final, 0, 1)
    h_final *= 2.0 / np.sum(h_final)
    
    # Local refinement via a simple hill climb on a coarse grid
    def local_refinement(h_current, dx):
        h_copy = h_current.copy()
        for _ in range(1000):  # limit iterations to avoid overuse
            if time.time() - start_time > budget_s - 1.0:
                break
            # Small perturbations
            i = np.random.randint(0, len(h_copy))
            old_val = h_copy[i]
            perturbation = np.random.normal(0, 0.01)
            h_copy[i] = np.clip(old_val + perturbation, 0, 1)
            
            # Renormalize
            h_copy *= 2.0 / np.sum(h_copy)
            
            # Evaluate the bound
            new_bound = evaluate_erdos_solution(h_copy)
            if new_bound < best_bound:
                best_bound = new_bound
                h_final = h_copy.copy()
            elif np.random.random() > 0.95:
                # Sometimes revert if no improvement
                h_copy[i] = old_val
                h_copy *= 2.0 / np.sum(h_copy)
        return h_final, best_bound

    h_final, final_bound = local_refinement(h_final, dx)
    return h_final, final_bound, n_points

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    # Try different resolution for better result
    n_points = 50  # moderate value for performance and solution quality
    h_result, c5_bound, n = bisection_lp_method(n_points, budget_s)
    end_time = time.time()
    if end_time - start_time > budget_s - 1:
        print("Warning: Exceeded time budget")
    return h_result, c5_bound, n
