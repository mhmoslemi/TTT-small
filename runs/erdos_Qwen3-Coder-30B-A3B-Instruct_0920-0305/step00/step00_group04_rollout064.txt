    # We want maximum of the positive lags (k in [0,2])
    # In our discretization, k = i * dx for i = 0 to n-1
    # So lags for k from 0 to 2 are 0 to n-1
    max_val = np.max(correlation[n_points-1:]) * dx
    return max_val
