import numpy as np
from scipy.optimize import differential_evolution
import time
import math

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for a given step function h."""
    # Ensure h is a numpy array
    h = np.array(h)
    # Compute the overlap integral
    # This is equivalent to max_k ∫ h(x)(1 - h(x+k)) dx
    # Using full convolution
    conv = np.correlate(h, 1 - h, mode='full')
    # The maximum value of the convolution
    max_conv = np.max(conv)
    # Scale by dx = 2/n
    dx = 2.0 / len(h)
    return max_conv * dx

def initialize_h_values(n_points):
    """Initialize with the current best construction or uniform distribution."""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) == n_points:
            return h.copy()
        else:
            # Rescale to correct length
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
            return h
    else:
        # Uniform distribution as fallback
        return np.ones(n_points) * 0.5

def objective(h_values):
    """Objective function to minimize."""
    return evaluate_erdos_solution(h_values)

def constraint_sum(h_values):
    """Constraint that integral of h equals 1."""
    dx = 2.0 / len(h_values)
    return np.sum(h_values) * dx - 1.0

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Use 45 points as suggested by the initial construction
    n_points = 45
    dx = 2.0 / n_points
    
    # Get starting point
    h_start = initialize_h_values(n_points)
    
    # Optimization bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # DiffEvo settings
    maxiter = 500
    popsize = 15
    mutation = (0.5, 1)
    recombination = 0.7
    
    # Track best solution
    best_h = h_start.copy()
    best_score = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    
    # We'll modify the approach: use a hybrid method
    # Try several local search approaches starting from different points
    
    for iter_step in range(maxiter):
        # Skip too many iterations
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.99:
            break
            
        # Perturb the current best
        h_trial = best_h.copy()
        # Add noise to a few random points
        indices = np.random.choice(n_points, size=5, replace=False)
        for i in indices:
            # Add small random perturbation
            new_val = h_trial[i] + np.random.normal(0, 0.05)
            # Clip to [0, 1]
            h_trial[i] = np.clip(new_val, 0, 1)
        
        # Renormalize
        current_sum = np.sum(h_trial) * dx
        h_trial = h_trial * (1.0 / current_sum)
        
        # Evaluate
        score = evaluate_erdos_solution(h_trial)
        if score < best_score:
            best_score = score
            best_h = h_trial.copy()
    
    # Final tuning with scipy.optimize
    def constrained_objective(h):
        # Make sure it's in [0,1]
        h = np.clip(h, 0, 1)
        # Renormalize
        h = h * (1.0 / np.sum(h) * dx)
        return evaluate_erdos_solution(h)
    
    # Run optimization starting with the best solution
    try:
        # Use differential evolution with constraints
        result = differential_evolution(
            constrained_objective,
            bounds,
            seed=seed,
            maxiter=100,
            popsize=15,
            mutation=mutation,
            recombination=recombination,
            disp=False
        )
        
        # Evaluate final solution
        final_h = np.clip(result.x, 0, 1)
        final_sum = np.sum(final_h) * dx
        final_h = final_h * (1.0 / final_sum)
        final_score = evaluate_erdos_solution(final_h)
        
        if final_score < best_score:
            best_score = final_score
            best_h = final_h.copy()
    except Exception as e:
        pass  # Fall back to previous best
    
    # Final check and return
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h) * dx
    if abs(final_sum - 1.0) > 1e-6:
        best_h = best_h * (1.0 / final_sum)
    
    return (best_h.tolist(), best_score, n_points)
