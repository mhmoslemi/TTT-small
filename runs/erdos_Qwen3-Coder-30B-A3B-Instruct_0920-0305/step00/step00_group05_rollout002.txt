import numpy as np
import time
import random
from scipy.optimize import minimize
from math import exp, log

# Helper functions
def evaluate(h):
    """Return C5 value computed exactly as in evaluator."""
    dx = 2.0 / len(h)
    correlation = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(correlation)
    return max_overlap * dx

def project_onto_capped_simplex(y, target_sum):
    """Project y onto the capped simplex: [0,1]^N with sum = target_sum"""
    y = np.clip(y, 0, 1)
    y_sum = np.sum(y)
    if y_sum == target_sum:
        return y
    # Duchi et al. projection 
    # Sort y in descending order
    sorted_y = np.sort(y)[::-1]
    # Find the threshold
    cumsum = np.cumsum(sorted_y)
    threshold = (cumsum - target_sum) / np.arange(1, len(sorted_y) + 1)
    # Select the largest index satisfying threshold[i] >= sorted_y[i]
    idx = np.argmax(sorted_y >= threshold)
    if idx == 0:
        lambda_val = threshold[0]
    else:
        lambda_val = threshold[idx - 1]
    # Apply projection
    projected_y = np.maximum(y - lambda_val, 0)
    return projected_y

def gradient_of_max(x, beta):
    """Return gradient of smooth max of overlaps with parameter beta."""
    n = len(x)
    # Compute all overlaps O_k
    O = np.correlate(x, 1 - x, mode='full')
    K = len(O)
    # Compute softmax weights
    O_beta = beta * O
    max_O = np.max(O_beta)
    exp_O = np.exp(O_beta - max_O)
    Z = np.sum(exp_O)
    w = exp_O / Z
    # Compute gradient (with more care about bounds and case)
    g = np.zeros(n)
    for k in range(K):
        # For shift k, gradient is sum of w_k * dO_k/dx_i
        # dO_k/dx_i = (1 - x_{i+k}) if i <= N-1-k and 0 otherwise
        # and -x_{i-k} if i >= k and 0 otherwise
        offset = k - (K - 1) // 2  # k maps from - (n-1) to n-1
        # active region for this shift k
        start = max(0, -offset)
        end = min(n, n - abs(offset))
        if offset > 0:
            # Shift is positive: 
            # dO_k/dx_i = (1 - x_{i+k}) for i = 0 to n-1-k
            for i in range(start, end):
                g[i] += -w[k] * x[i + offset]
        elif offset < 0:
            # Negative shift
            # dO_k/dx_i = (1 - x_{i+k}) = 1 - x_{i+offset} for i = 0 to -offset
            # and -x_{i-offset} for i = -offset to n-1
            for i in range(start, end):
                g[i] += w[k] * (1 - x[i - offset])
        else:
            # offset = 0: dO_0/dx_i = 0 everywhere, because
            # sum_i x_i * (1 - x_{i+0}) = sum_i x_i * (1 - x_i) = sum_i x_i - sum_i x_i^2  
            # Its gradient wrt x_i is 1 - 2*x_i
            for i in range(start, end):
                g[i] += w[k] * (1 - 2 * x[i])
    return g

def swap_move(x, i, j):
    """Swap x[i] and x[j]"""
    x_copy = x.copy()
    x_copy[i], x_copy[j] = x_copy[j], x_copy[i]
    return x_copy

def fractional_move(x, i, j, delta):
    """Move delta from x[j] to x[i], maintaining value and bounds."""
    x_copy = x.copy()
    x_copy[i] += delta
    x_copy[j] -= delta
    x_copy[i] = max(0, min(1, x_copy[i]))
    x_copy[j] = max(0, min(1, x_copy[j]))
    return x_copy

def worst_shift_info(x):
    """Return (k_star, offending_indices) from the worst shift."""
    n = len(x)
    correlation = np.correlate(x, 1 - x, mode='full')
    k_star = np.argmax(correlation) - (len(correlation) - 1) // 2
    max_val = correlation[np.argmax(correlation)]
    # Find indices for the worst shift
    offsets = np.arange(n)
    if k_star > 0:
        # shift is positive  
        # take x_i * (1 - x_{i+k}) for i = 0 to n-1-k 
        mask = (offsets + k_star < n)
        i_indices = offsets[mask]
        j_indices = offsets[mask] + k_star
        # Check which are good: x_i = 1, x_{i+k} = 0
        mask_bad = (x[i_indices] == 1) & (x[j_indices] == 0)
        offending = i_indices[mask_bad]
    elif k_star < 0:
        # shift is negative
        mask = (offsets - k_star >= 0)
        i_indices = offsets[mask] 
        j_indices = offsets[mask] - k_star
        mask_bad = (x[i_indices] == 1) & (x[j_indices] == 0)
        offending = i_indices[mask_bad]
    else:
        # shift = 0
        # x_i * (1 - x_i) for all i
        # good if x_i == 1 and x_{i+0} == 0. But checking that is always false since x_i == x_i
        # So just count all x_i == 1, i too often
        # Rewriting O_0 = sum x_i * (1 - x_i) = sum x_i - sum x_i^2
        # Yes it's not straightforward but not necessary for this stage.
        # Instead, just return none
        offending = np.where(x == 1)[0]
    return k_star, offending

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    t_start = time.time()
    safety_margin = 5.0
    t_dead = t_start + budget_s - safety_margin
    
    # Candidate N values to try, scaled to budget
    N_vals = [70, 100, 140, 200, 260]
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    for N in N_vals:
        if time.time() >= t_dead:
            break
        n_time_budget = (t_dead - time.time()) / len(N_vals)
        t_next_cycle = time.time() + n_time_budget
        
        # SA Phase 1
        best_binary = None
        best_binary_c5 = float('inf')
        T0 = 0.12
        t_max = 30000  # or a limit based on time
        
        # Initial random binary vector
        h = np.zeros(N)
        ones_indices = np.random.choice(N, N//2, replace=False)
        h[ones_indices] = 1
        
        current_c5 = evaluate(h)
        accepted_c5 = current_c5
        best_candidate = h.copy()
        best_candidate_c5 = current_c5
        
        # Time allowed for this N
        t_cycle = time.time() + n_time_budget
        t_iter = 0
        
        while time.time() < t_cycle:
            t_iter += 1
            if t_iter > t_max or time.time() > t_cycle:
                break
            T = T0 * (1 - t_iter / t_max)
            # Attempt swap
            i = np.random.randint(N)
            j = np.random.randint(N)
            # When either is same or both same values, retry
            if h[i] == h[j]:
                continue
            h_proposed = swap_move(h, i, j)
            new_c5 = evaluate(h_proposed)
            delta = new_c5 - current_c5
            if delta <= 0:
                h = h_proposed
                current_c5 = new_c5
                if current_c5 < accepted_c5:
                    accepted_c5 = current_c5
                if current_c5 < best_candidate_c5:
                    best_candidate = h.copy()
                    best_candidate_c5 = current_c5
            elif np.random.rand() < exp(-delta / T) and T > 0:
                h = h_proposed
                current_c5 = new_c5
                if current_c5 < accepted_c5:
                    accepted_c5 = current_c5
                    
        if best_candidate_c5 < best_binary_c5:
            best_binary = best_candidate
            best_binary_c5 = best_candidate_c5
            
        # Phase 2a: Directed local improvement
        if (time.time() < t_dead) and (best_binary_c5 < best_c5):
            h = best_binary.copy()
            iter_limit = 5000  # Number of directed swaps to perform
            for _ in range(iter_limit):
                if time.time() >= t_cycle:
                    break
                k_star, offending = worst_shift_info(h)
                if len(offending) == 0:
                    break
                # Pick a random offending i
                i = np.random.choice(offending)
                # Find random zero j that's not in any pair (i, i+k)
                candidates = np.where(h == 0)[0]
                valid_cand = candidates[(candidates != i) & (candidates != i + k_star)]
                if len(valid_cand) == 0:
                    continue
                j = np.random.choice(valid_cand)
                h_proposed = swap_move(h, i, j)
                new_c5 = evaluate(h_proposed)
                if new_c5 <= current_c5:
                    h = h_proposed
                    current_c5 = new_c5
                    if current_c5 < best_binary_c5:
                        best_binary_c5 = current_c5
                        best_binary = h.copy()
                # Else revert
        # Phase 2b: Continuous refinement
        if (time.time() < t_dead) and (best_binary_c5 < best_c5) and N >= 100:
            # Refine the binary vector
            h_bin = best_binary.copy()
            x = h_bin.copy()  # Entries in [0,1]
            steps = 1000
            β = 5.0  # Initial smoothness
            α = 0.02 # Step size
            for step in range(steps):
                if time.time() >= t_cycle:
                    break
                if step % 50 == 0:
                    β *= 1.02
                grad = gradient_of_max(x, β)
                x -= α * grad
                # Project onto (continuous capped simplex)
                x = project_onto_capped_simplex(x, N / 2)
                # Evaluate if good
                c5 = evaluate(x)
                if c5 < best_binary_c5:
                    best_binary_c5 = c5
                    best_binary = x.copy()
            # Final best suspect
            current_c5 = best_binary_c5
            h = best_binary.copy()
        # Update global best
        if best_binary_c5 < best_c5:
            best_c5 = best_binary_c5
            best_h = h.copy()
            best_n = N
            if best_c5 <= 0.379:
                break
                
    # Final evaluation with true method
    final_c5 = evaluate(best_h)
    return (best_h.tolist(), final_c5, best_n)
