import numpy as np

# Fixed parameters
M = 6  # Number of intervals
n = 400  # Grid size for final discretization
dx = 2.0 / n

def decode(z):
    # z is a vector of 2*M+1 real numbers
    z_ell = z[:M]
    z_g = z[M:]
    
    # Softmax to ensure positive lengths and gaps with correct sum
    ell = np.exp(z_ell) / np.sum(np.exp(z_ell))
    g = np.exp(z_g) / np.sum(np.exp(z_g))
    
    return ell, g

def build_intervals(ell, g):
    # Compute the start and end points of intervals
    a = np.zeros(M+1)  # starts of intervals
    a[0] = g[0]  # g[0] is the gap before first interval
    for i in range(1, M):
        a[i] = a[i-1] + ell[i-1] + g[i]
    a[M] = a[M-1] + ell[M-1] + g[M]
    b = a + ell
    return a, b

def candidate_shifts(a, b):
    # Generate all valid shift candidates
    E = list(a) + list(b)
    T = set([0.0])
    for i in range(len(E)):
        for j in range(len(E)):
            t = abs(E[i] - E[j])
            if t <= 2.0:
                T.add(t)
    return sorted(list(T))

def intersection_length(t, a, b):
    # For given shift t, compute overlap
    # Shift intervals to [a_i - t, b_i - t] for positive t
    total = 0.0
    for i in range(len(a)):
        interval_start = a[i] - t
        interval_end = b[i] - t
        if interval_start > 2.0 or interval_end < 0.0:
            continue
        overlap_start = max(0.0, interval_start)
        overlap_end = min(2.0, interval_end)
        if overlap_end > overlap_start:
            total += overlap_end - overlap_start
    return total

def objective(z):
    ell, g = decode(z)
    a, b = build_intervals(ell, g)
    shifts = candidate_shifts(a, b)
    I_values = [intersection_length(t, a, b) for t in shifts]
    min_I = min(I_values)
    # Since C5 = 1 - min_I
    return 1.0 - min_I

        popsize=15 * (2 * M + 1),
        seed=42,
        disp=False,

        strategy='best1bin'
    )
    
    return result

def gridify(intervals, n):
    h_grid = np.zeros(n)
    ell, g = intervals
    a, b = build_intervals(ell, g)
    for i in range(M):
        start_index = int(np.ceil(a[i] / dx))
        end_index = min(n, int(np.floor(b[i] / dx)) + 1)
        h_grid[start_index:end_index] = 1.0
    
    # Adjust sum to be exactly n/2
    current_sum = np.sum(h_grid)
    if abs(current_sum - n/2) < 1e-9:
        return h_grid
    elif abs(current_sum - n/2) <= 1:
        # Flip one
        flip_i = np.random.randint(0, n)
        h_grid[flip_i] = 1 - h_grid[flip_i]
        return h_grid
    else:
        # Adjust sum significantly, flip multiple indices
        diff = int(current_sum - n/2)
        for _ in range(abs(diff)):
            flip_i = np.random.randint(0, n)
            h_grid[flip_i] = 1 - h_grid[flip_i]
        return h_grid

def binary_hill_climb(h, time_remaining):
    best_h = h.copy()
    best_c5 = evaluate_solution(best_h)
    start_time = time.time()
    
    max_swaps = 5000  # Should be small enough to finish within time
    # Approx. 1000 swaps since they are near instant with overlap correlation
    swaps_per_loop = 1000
    loops = max_swaps // swaps_per_loop
    
    for loop in range(loops):
        if time.time() - start_time > time_remaining:
            break
            
        for _ in range(swaps_per_loop):
            i = np.random.randint(0, len(h))
            j = np.random.randint(0, len(h))
            if (h[i] == 1 and h[j] == 0) or (h[i] == 0 and h[j] == 1):
                h[i], h[j] = h[j], h[i]
                new_c5 = evaluate_solution(h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                else:
                    h[i], h[j] = h[j], h[i]  # revert swap
    
    return best_h, best_c5

def evaluate_solution(h):
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlaps = np.max(corr)
    # Since the overlap is the maximum, we use it directly
    # Normalization comes from the dx in the definition
    return 1.0 - max_overlaps * dx

    # Phase 3: Discretization to grid
    h_grid = gridify((ell, g), n)
    
    # Phase 4: Binary hill climbing
    time_remaining = budget_s - (time.time() - start_time)
    h_final, final_c5 = binary_hill_climb(h_grid, time_remaining)
    
    return (h_final, final_c5, n)
