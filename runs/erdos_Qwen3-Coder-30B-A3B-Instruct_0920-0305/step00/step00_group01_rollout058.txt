import numpy as np
from scipy.optimize import differential_evolution, minimize
import time
from cvxpy import *
import math

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the one we care about
    max_overlap = np.max(correlation) * dx
    return max_overlap

def initial_h_values():
    # Initial construction from the problem description
    # Using a step function approach similar to known constructions
    n = 47
    h = np.zeros(n)
    # Simple construction: 1 on [0, 1] and 0 on [1, 2]
    for i in range(n):
        x = i * 2.0 / n
        if x < 1.0:
            h[i] = 1.0
        else:
            h[i] = 0.0
    # Normalize so that integral is 1
    h = h * (2.0 / np.sum(h) * 0.5)  # Scale to get integral = 1
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the provided initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    print(f"Starting with initial construction of length {n_points}")
    
    # Define optimization constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Global optimization function
    def objective(h):
        # Jitter to avoid numerical issues
        h = np.clip(h, 0, 1)
        # Rescale so that integral = 1
        sum_h = np.sum(h) * dx
        if sum_h == 0:
            return 1e10
        h = h * (1.0 / sum_h)
        return evaluate_erdos_solution(h)
    
    # Optimize using differential evolution with limited time
    results = []
    
    # Try a few random restarts and also the initial guess
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    print(f"Initial C5: {best_c5}")
    
    # Try to improve the initial solution using local optimization
    # First, grid search with nearby solutions to bypass local optima
    try:
        # Differential evolution is good for global search
        de_res = differential_evolution(objective, bounds, maxiter=50, 
                                       popsize=10, seed=seed, 
                                       tol=1e-6, 
                                       disp=False)
        if de_res.success:
            h_opt = de_res.x.copy()
            h_opt = np.clip(h_opt, 0, 1)
            sum_h = np.sum(h_opt) * dx
            if sum_h > 0:
                h_opt = h_opt * (1.0 / sum_h)  # Normalize to integration 1
                c5 = evaluate_erdos_solution(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                print(f"Differential evolution C5: {c5}")
    except Exception as e:
        print(f"Differential evolution failed: {e}")

    # Local search with gradients: Use a more direct method if time allows
    # Let's use Nelder-Mead on subsampled versions
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.9:
        try:
            for trial in range(10):
                # Slightly perturb the solution
                h_pert = best_h + (np.random.rand(n_points) - 0.5) * 0.1
                h_pert = np.clip(h_pert, 0, 1)
                # Normalize
                sum_h = np.sum(h_pert) * dx
                if sum_h > 0:
                    h_pert = h_pert * (1.0 / sum_h)
                c5 = evaluate_erdos_solution(h_pert)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_pert.copy()
                    print(f"Improvement found: {c5}")
        except Exception as e:
            print(f"Perturbation failed: {e}")
    
    # Return the best result found
    final_c5 = evaluate_erdos_solution(best_h)
    print(f"Final C5: {final_c5}")
    
    return (best_h.tolist(), final_c5, n_points)
