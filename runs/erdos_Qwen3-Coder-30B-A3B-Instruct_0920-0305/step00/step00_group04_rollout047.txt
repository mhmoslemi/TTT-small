import numpy as np
import time
import scipy.fft as fft
from scipy.optimize import minimize

# Predefined helper functions from the environment
def evaluate_erdos_solution(h):
    """Evaluate the actual C5 value using the reference implementation"""
    n = len(h)
    dx = 2.0 / n
    # correlate h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum of all correlation values
    max_corr = np.max(corr)
    return max_corr * dx

def initial_h_values():
    """Return initial h from the reference solution"""
    # This is a placeholder - in practice this would load from the environment
    # For now, return a simple construction to start with
    n = 42
    h = np.zeros(n)
    h[:n//2] = 1
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Grid size to use
    n = 60
    dx = 2.0 / n
    
    # Phase A: Binary optimization (Simulated Annealing)
    def compute_corr(h):
        """Compute correlation values efficiently with FFT"""
        # Pad to next power of 2
        L = 1
        while L < 2 * n - 1:
            L *= 2
        h_padded = np.pad(h, (0, L - n), 'constant')
        g_padded = np.pad(1 - h, (0, L - n), 'constant')
        H = fft.fft(h_padded)
        G = fft.fft(g_padded)
        corr_full = fft.ifft(H * np.conj(G)).real
        # Take only the valid range, corresponding to lags -n+1 to n-1
        corr_vals = corr_full[:2*n-1]
        # The max over all lags, scaled by dx
        return np.max(corr_vals) * dx
    
    def binary_random(n):
        """Generate a random binary vector with exactly n//2 ones"""
        h = np.zeros(n)
        ones = np.random.choice(n, n//2, replace=False)
        h[ones] = 1
        return h
    
    def binary_swap_move(h, idx1, idx2):
        """Swap positions idx1 and idx2 in binary vector h"""
        h_new = h.copy()
        h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
        return h_new
    
    def binary_double_swap_move(h):
        """Perform two independent swaps"""
        h_new = h.copy()
        indices_1 = np.where(h_new == 1)[0]
        indices_0 = np.where(h_new == 0)[0]
        if len(indices_1) < 1 or len(indices_0) < 1:
            return h_new
        i1 = np.random.choice(indices_1)
        i2 = np.random.choice(indices_0)
        h_new[i1], h_new[i2] = h_new[i2], h_new[i1]
        if len(indices_1) > 1 and len(indices_0) > 1:
            # Try second swap
            indices_1 = np.where(h_new == 1)[0]
            indices_0 = np.where(h_new == 0)[0]
            i1 = np.random.choice(indices_1)
            i2 = np.random.choice(indices_0)
            h_new[i1], h_new[i2] = h_new[i2], h_new[i1]
        return h_new
    
    def sa_binary_optimization(n_points, max_time):
        best_h = None
        best_c5 = float('inf')
        num_restarts = 0
        
        def run_sa_iteration(h_seed, restart_id):
            h = h_seed.copy()
            current_c5 = compute_corr(h)
            T0 = 0.05
            alpha = 0.9995
            t0 = time.time()
            t_max = max_time * 0.7 / 10  # 10 restarts
            step = 0
            
            while time.time() - t0 < t_max:
                # Even if near end, take at least one step
                if time.time() - start_time > max_time - 5:
                    break
                    
                # Try a double-swap move
                h_new = binary_double_swap_move(h)
                new_c5 = compute_corr(h_new)
                delta = new_c5 - current_c5
                t = T0 * (alpha ** step)
                
                if delta <= 0 or np.random.rand() < np.exp(-delta / max(t, 1e-8)):
                    h = h_new
                    current_c5 = new_c5
                    
                step += 1
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    
            return best_h, best_c5
        
        # Multiple restarts
        while time.time() - start_time < max_time - 5:
            num_restarts += 1
            h_seed = binary_random(n_points)
            h_best, c5_best = run_sa_iteration(h_seed, num_restarts)
            if h_best is not None and c5_best < best_c5:
                best_c5 = c5_best
                best_h = h_best.copy()
        
        return best_h, best_c5
    
    # Use initial construction if available
    try:
        init_h = initial_h_values()[:n]  # Truncate to n points
        init_h = init_h / np.sum(init_h) * (n / 2)  # Normalize to sum = n/2
        init_h = np.round(init_h).astype(int).astype(float)  # binary
        init_h[np.where(init_h == 0)[0][:n//2 - np.sum(init_h).astype(int)]] = 1  # make sure it sums to n/2
    except:
        init_h = binary_random(n)
        
    # Run binary optimization first
    binary_h, binary_c5 = sa_binary_optimization(n, budget_s * 0.7)
    
    if binary_c5 <= 0.3807:
        # Early return
        return (binary_h, binary_c5, n)
    
    # Phase B: Continuous refinement using gradient descent
    def soft_max(x, gamma):
        """Compute smooth max with parameter gamma"""
        return np.log(np.sum(np.exp(gamma * x))) / gamma
    
    def compute_correlation_parts(h):
        """Compute correlation values as a vector"""
        n = len(h)
        dx = 2.0 / n
        # pad
        L = 1
        while L < 2 * n - 1:
            L *= 2
        h_padded = np.pad(h, (0, L - n), 'constant')
        g_padded = np.pad(1 - h, (0, L - n), 'constant')
        H = fft.fft(h_padded)
        G = fft.fft(g_padded)
        corr_full = fft.ifft(H * np.conj(G)).real
        corr_vals = corr_full[:2*n-1] * dx
        return corr_vals
    
    def compute_gradient(h, gamma):
        """Compute the gradient of smooth max"""
        n = len(h)
        dx = 2.0 / n
        
        # Correlations
        cors = compute_correlation_parts(h)
        exp_gamma = np.exp(gamma * cors)
        Z = np.sum(exp_gamma)
        weights = exp_gamma / Z
        
        # Gradient calculation
        # For each i, compute ∂/∂h_i of Φ = (1/γ)*log(sum exp(γ * f_ℓ))
        # which equals sum_c w_c * ∂f_c / ∂h_i
        
        grad = np.zeros(n)
        for l in range(2*n - 1):
            # we adjust for l from -n+1 to n-1
            lag = l - (n - 1)
            for i in range(n):
                # Derivative from forward term (like indexing corr[i], corr[i+lag])
                end_c = i + lag
                if end_c >= 0 and end_c < n:
                    grad[i] += weights[l] * (- dx * (1 - h[end_c]))
                # Derivative from backward term  
                start_c = i - lag
                if start_c >= 0 and start_c < n:
                    grad[i] += weights[l] * (- dx * h[start_c]) 
        return grad
    
    def project_to_simplex(y, target_sum):
        """Project y onto the capped simplex"""
        # Find lambda such that sum(clamp(y - lambda, 0, 1)) = target_sum
        def sum_of_clamped(x, t):
            return np.sum(np.clip(x - t, 0, 1))
        
        low, high = np.min(y) - 1, np.max(y) + 1
        eps = 1e-8
        try:
            while high - low > eps:
                mid = (low + high) / 2
                s = sum_of_clamped(y, mid)
                if s > target_sum:
                    low = mid
                else:
                    high = mid
            lambda_opt = (low + high) / 2
        except:
            lambda_opt = 0  # Default fallback
            
        z = np.clip(y - lambda_opt, 0, 1)
        scale = target_sum / (np.sum(z) + 1e-12)  # avoid division by 0
        return z * scale
    
    def continuous_refinement(h_start, max_time):
        h = h_start.copy()
        c5_prev = compute_corr(h)
        gamma = 5.0
        alpha = 0.1 * dx
        t_end = time.time() + max_time
        
        # Loop convergence flags
        last_improvement = 0
        patience = 10
        
        for iter_count in range(100000):
            if time.time() > t_end or (iter_count - last_improvement) > patience:
                break
            
            # Compute gradient
            grad = compute_gradient(h, gamma)
            # Step in reverse direction (as we minimize)
            h_trial = h - alpha * grad
            # Project back
            h_proj = project_to_simplex(h_trial, n / 2)
            
            # Evaluate true C5
            c5_new = compute_corr(h_proj)
            
            if c5_new < c5_prev - 1e-8:  # Improvement
                h = h_proj
                c5_prev = c5_new
                last_improvement = iter_count
                # If we're getting close to goal, try more fine steps
                if c5_new < 0.3807:
                    alpha = max(alpha * 0.95, 1e-6)
            else:
                # No improvement, reduce step size
                alpha *= 0.9  
                if alpha < 1e-8:
                    break
                    
            # Increase gamma for tighter surrogate
            if iter_count % 10 == 0 and gamma < 150:
                gamma *= 1.5
            
        return h, c5_prev
    
    # Main code ends here
    try:
        final_h, final_c5 = continuous_refinement(binary_h, budget_s * 0.25)
    except:
        final_h, final_c5 = binary_h, binary_c5
    
    # Final validation and cleanup
    final_h = project_to_simplex(final_h, n/2)
    final_c5 = compute_corr(final_h)
    
    return (final_h, final_c5, n)
