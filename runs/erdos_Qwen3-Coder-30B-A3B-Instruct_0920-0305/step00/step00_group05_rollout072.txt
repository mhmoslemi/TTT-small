import numpy as np
import time
import random
import math
from scipy.optimize import minimize
from scipy.special import softmax

def evaluate(x):
    """Evaluate C5 using the same procedure as the evaluator"""
    dx = 2.0 / len(x)
    corr = np.correlate(x, 1 - x, mode="full")
    return np.max(corr) * dx

def project_onto_capped_simplex(y, target_sum):
    """Project y onto the capped simplex: [0,1]^n with sum = target_sum"""
    y = np.clip(y, 0, 1)
    sorted_y = np.sort(y)[::-1]
    n = len(y)
    
    cumsum = np.cumsum(sorted_y)
    rho = np.argmax(sorted_y * np.arange(1, n+1) > (target_sum - cumsum + sorted_y)) + 1
    if rho == 0:
        theta = (target_sum - np.sum(sorted_y)) / n
    else:
        theta = (target_sum - cumsum[rho - 1]) / rho
    
    x = np.maximum(y - theta, 0)
    
    # Ensure sum constraint is satisfied by clipping
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-10:
        # Adjust slightly to satisfy constraint
        idx = np.argsort(x)
        for i in range(1, min(10, len(x))):
            if diff > 0:
                if x[idx[-i]] < 1:
                    added = min(diff, 1 - x[idx[-i]])
                    x[idx[-i]] += added
                    diff -= added
            else:
                if x[idx[i-1]] > 0:
                    subtracted = min(-diff, x[idx[i-1]])
                    x[idx[i-1]] -= subtracted
                    diff += subtracted
            if abs(diff) < 1e-10:
                break
    
    return x

def gradient_of_max(x, beta):
    """Compute gradient of log-sum-exp of overlaps"""
    corr = np.correlate(x, 1 - x, mode="full")
    k = len(corr) - len(x)  # offset k = len(x) - 1
    n = len(x)
    weights = softmax(beta * corr)
    
    # Use finite difference approach for the gradient
    g = np.zeros_like(x)
    h = 1e-4
    for i in range(n):
        # Compute ∂O_k / ∂x_i for all k
        einsum_term = np.zeros_like(x)
        for shift in range(-n+1, n):
            lshift = abs(shift)
            if shift >= 0:
                # For positive shift, x[i] contributes to O_{shift} if i + shift < n
                if i + lshift < n:
                    einsum_term[i] += 1 - x[i + lshift] if i + lshift < n else 0
                    einsum_term[i + lshift] -= x[i] if i < n else 0
            else:
                # For negative shift, x[i] contributes to O_{shift} if i - shift < n
                if i - lshift >= 0:
                    einsum_term[i] += 1 - x[i - lshift] if i - lshift >= 0 else 0
                    einsum_term[i - lshift] -= x[i] if i >= 0 else 0
                    
        g[i] = np.dot(weights, einsum_term)
    return g

def swap_move(x, i, j):
    """Swap elements at positions i and j"""
    y = x.copy()
    y[i], y[j] = y[j], y[i]
    return y

def fractional_move(x, i, j, delta):
    """Move delta from j to i while respecting bounds"""
    y = x.copy()
    delta = min(delta, y[j])
    y[i] += delta
    y[j] -= delta
    return y

def worst_shift_info(x):
    """Return worst shift and 'offending' indices associated with it"""
    dx = 2.0 / len(x)
    corr = np.correlate(x, 1 - x, mode="full")
    k_idx = np.argmax(corr)
    worst_k = k_idx - len(x) + 1  # From correlation indexing
    max_val = corr[k_idx]
    
    n = len(x)
    offenders = []
    abs_k = abs(worst_k)
    if worst_k >= 0:
        for i in range(n - abs_k):
            if x[i] == 1 and x[i + abs_k] == 0:
                offenders.append(i)
    else:
        for i in range(abs_k, n):
            if x[i] == 1 and x[i - abs_k] == 0:
                offenders.append(i)
    return worst_k, offenders

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    t_start = time.time()
    t_dead = t_start + budget_s - 5  # Safety margin
    
    # Candidate sizes
    candidateNs = [140, 200, 260]
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    for N in candidateNs:
        if time.time() >= t_dead:
            break
        
        t_slice = (t_dead - time.time()) / (len(candidateNs) - candidateNs.index(N))
        if t_slice <= 0:
            break
            
        np.random.seed(seed + N)
        np.random.shuffle(range(N))
        
        # Phase 2.1: Binary SA
        best_x = None
        best_c5_phase = float('inf')
        T0 = 0.15
        max_iter = max(1000, int(t_slice * 9000 // N))
        restarts = 0
        last_improvement = 0
        
        while restarts < 50 and time.time() < t_dead:
            # Initialize binary vector
            x = np.zeros(N)
            ones = np.random.choice(N, N//2, replace=False)
            x[ones] = 1
            
            current_c5 = evaluate(x)
            best_x_local = x.copy()
            best_c5_local = current_c5
            
            t_slice_remaining = t_dead - time.time()
            if t_slice_remaining <= 0:
                break
        
            for iter in range(max_iter):
                if time.time() >= t_dead:
                    break
                    
                # Choose two indices with different values
                ones = np.where(x == 1)[0]
                zeros = np.where(x == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                
                x_new = swap_move(x, i, j)
                new_c5 = evaluate(x_new)
                
                # Acceptance rule
                delta = new_c5 - current_c5
                T = T0 * (1.0 - iter / max_iter)
                if delta <= 0 or np.random.rand() < np.exp(-delta / max(1e-12, T)):
                    x = x_new
                    current_c5 = new_c5
                    if current_c5 < best_c5_local:
                        best_c5_local = current_c5
                        best_x_local = x.copy()
                        last_improvement = iter
                        
            # Check if better than current global best
            if best_c5_local < best_c5_phase:
                best_c5_phase = best_c5_local
                best_x = best_x_local.copy()
            
            if last_improvement > max_iter * 0.8 and time.time() < t_dead:
                # Early stop or restart
                break
            else:
                restarts += 1
        
        if best_x is not None and best_c5_phase < best_c5:
            best_h = best_x.copy()
            best_c5 = best_c5_phase
            best_n = N
            
        # Phase 2.2: Directed local improvement (on best binary)
        if best_x is not None and time.time() < t_dead:
            x = best_x.copy()
            for iter in range(10000):  # Default to directed local improvement with early-stop
                if time.time() >= t_dead:
                    break
                    
                k, offenders = worst_shift_info(x)
                if len(offenders) == 0:
                    break
                    
                i = random.choice(offenders)
                # Find a zero that is not in the same pair
                valid_indices = list(set(range(len(x))) - {i, i + abs(k)})
                if not valid_indices:
                    break
                j = random.choice(valid_indices)
                
                x_new = swap_move(x, i, j)
                new_c5 = evaluate(x_new)
                
                if new_c5 <= current_c5:
                    x = x_new
                    current_c5 = new_c5
                else:
                    # Revert if worsening
                    pass
                    
            # Update best if better
            if current_c5 < best_c5:
                best_h = x.copy()
                best_c5 = current_c5
                best_n = N
        else:
            continue
        
        # Phase 2.3: Continuous refinement
        if time.time() >= t_dead:
            break
            
        x = best_h.copy()
        target_sum = N / 2
        beta = 10.0
        step_size = 0.01
        max_iter_cont = 3000
        prev_c5 = current_c5 
        t_remaining = t_dead - time.time()
        
        if t_remaining > 5:
            max_iter_cont = int(min(max_iter_cont, t_remaining * 500))
            for step in range(max_iter_cont):
                if time.time() >= t_dead:
                    break
                # Gradient computation with smoothing
                grad = gradient_of_max(x, beta)
                y = x - step_size * grad
                x = project_onto_capped_simplex(y, target_sum)
                current_c5 = evaluate(x)
                # Adjust β (continuation)
                if beta < 500:
                    beta *= 1.01
                if abs(current_c5 - prev_c5) < 1e-6:
                    break
                prev_c5 = current_c5
            
            if current_c5 < best_c5:
                best_h = x.copy()
                best_c5 = current_c5
                best_n = N
                
        # Optimize last step with full evaluation
        if time.time() < t_dead:
            final_c5 = evaluate(best_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
    
    return best_h.tolist(), best_c5, best_n
