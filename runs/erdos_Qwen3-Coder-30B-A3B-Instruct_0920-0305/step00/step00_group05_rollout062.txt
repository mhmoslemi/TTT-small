import numpy as np
import time
import random
import math
from scipy.optimize import minimize
from scipy.special import softmax

def project_onto_capped_simplex(y, target_sum):
    """Projects y onto the capped simplex {x: 0 <= x <= 1, sum(x) = target_sum}"""
    y = np.clip(y, 0, 1)
    sorted_y = np.sort(y)[::-1]
    cumsum = np.cumsum(sorted_y)
    idx = np.arange(1, len(sorted_y) + 1)
    condition = sorted_y - (target_sum - cumsum) / idx > 0
    rho = np.sum(condition)
    if rho == 0:
        return np.zeros_like(y)
    theta = (cumsum[rho - 1] - target_sum) / rho
    return np.clip(y - theta, 0, 1)

def evaluate(x):
    """Evaluates C5 = max_k int h(x)(1 - h(x+k))dx"""
    dx = 2.0 / len(x)
    corr = np.correlate(x, 1 - x, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def gradient_of_max(x, beta):
    """Returns gradient of log-sum-exp approximation to max_k O_k"""
    N = len(x)
    corr = np.correlate(x, 1 - x, mode="full")
    indices = np.arange(-N + 1, N)
    exp_corr = np.exp(beta * corr)
    weights = exp_corr / np.sum(exp_corr)
    
    grad = np.zeros_like(x)
    for k in range(-N + 1, N):
        if k < 0:
            # k is negative, so we're looking at x[i] * (1 - x[i-k]) where i-k = i+|k|
            offset = -k
            for i in range(N - offset):
                grad[i] += weights[k + N - 1] * (1 - x[i + offset])
                if i + offset < N:
                    grad[i + offset] -= weights[k + N - 1] * x[i]
        elif k > 0:
            # k is positive, so we're looking at x[i] * (1 - x[i+k])  
            for i in range(N - k):
                grad[i] += weights[k + N - 1] * (1 - x[i + k])
                if i + k < N:
                    grad[i + k] -= weights[k + N - 1] * x[i]
        else:
            # k = 0
            for i in range(N):
                grad[i] += weights[k + N - 1] * (1 - x[i]) * (i < N-1) # Favor first terms
                # But it's 0 since we only add it in conv w.r.t. 1-x, so x*0=0
    return grad

def swap_move(x, i, j):
    """Swaps values at indices i and j"""
    x_new = x.copy()
    x_new[i], x_new[j] = x_new[j], x_new[i]
    return x_new

def fractional_move(x, i, j, delta):
    """Moves delta from x[j] to x[i] respecting box constraints"""
    x_new = x.copy()
    if delta < 0:
        delta = -delta
        i, j = j, i
    if x_new[j] < delta:
        delta = x_new[j]
    x_new[i] += delta
    x_new[j] -= delta
    x_new[i] = max(0, min(1, x_new[i]))
    x_new[j] = max(0, min(1, x_new[j]))
    return x_new

def worst_shift_info(x):
    """Returns (shift_idx, set_of_offending_indices) that causes max overlap"""
    corr = np.correlate(x, 1 - x, mode="full")
    max_idx = np.argmax(corr)
    best_shift = max_idx - len(x) + 1
    # Find accusations:
    if best_shift < 0:
        # For shift k<0: look for positions i such that x[i]=1 and x[i+|k|]=0
        k = -best_shift
        offending = []
        for i in range(len(x) - k):
            if x[i] == 1 and x[i + k] == 0:
                offending.append(i)
        return best_shift, offending
    elif best_shift > 0:
        # For shift k>0: look for positions i such that x[i]=1 and x[i-k]=0
        k = best_shift
        offending = []
        for i in range(k, len(x)):
            if x[i] == 1 and x[i - k] == 0:
                offending.append(i)
        return best_shift, offending
    else:
        # k == 0: no shift
        return best_shift, []

def run(seed=42, budget_s=600, **kwargs):
    # Random seed set
    random.seed(seed)
    np.random.seed(seed)
    
    t_start = time.time()
    t_dead = t_start + budget_s - 5  # Safety margin
    
    # Candidate discretization sizes
    Ns = [140, 200, 260]
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    for i, N in enumerate(Ns):
        if time.time() >= t_dead:
            break
        
        t_slice_end = t_dead if i == len(Ns) - 1 else t_dead - (len(Ns) - 1 - i) * (budget_s - 5) / (len(Ns) - 1)
        
        dx = 2.0 / N
        best_binary_solution = None
        best_binary_c5 = float('inf')
        t_slice_end = t_dead if i == len(Ns) - 1 else t_dead - (len(Ns) - 1 - i) * (budget_s - 5) / (len(Ns) - 1)
        
        while time.time() < t_slice_end:
            # Phase 2.1: binary simulated annealing
            # Initialize a random binary solution
            h = np.zeros(N)
            ones_indices = random.sample(range(N), N//2)
            h[ones_indices] = 1
            
            T0 = 0.12
            t_max = 30000
            T = T0
            
            current_c5 = evaluate(h)
            best_local_c5 = current_c5
            best_local_sol = h.copy()
            
            for t in range(t_max):
                if time.time() >= t_slice_end:
                    break
                    
                # Select two distinct indices with opposite values
                one_pos = np.where(h==1)[0]
                zero_pos = np.where(h==0)[0]
                if len(one_pos) == 0 or len(zero_pos) == 0:
                    break
                i = random.choice(one_pos)
                j = random.choice(zero_pos)
                
                # Do the swap
                h_new = swap_move(h, i, j)
                c5_new = evaluate(h_new)
                
                Δ = c5_new - current_c5
                
                if Δ <= 0 or np.random.rand() < np.exp(-Δ / T):
                    h = h_new
                    current_c5 = c5_new
                
                if current_c5 < best_local_c5:
                    best_local_c5 = current_c5
                    best_local_sol = h.copy()
                
                # Anneal temperature
                T = T0 * (1 - t / t_max)
            
            # Phase 2.2: directed local refinement
            h = best_local_sol.copy()
            max_iter = 10000
            for iter in range(max_iter):
                if time.time() >= t_slice_end:
                    break
                k_star, offending = worst_shift_info(h)
                if len(offending) == 0:
                    break
                # Get a random offending index
                i_offend = random.choice(offending)
                # Find a valid j to swap with (not involved in current pair)
                valid_j = []
                for idx in range(N):
                    if idx != i_offend and idx != i_offend + abs(k_star) and h[idx] == 0:
                        valid_j.append(idx)
                if len(valid_j) == 0:
                    break
                j = random.choice(valid_j)
                new_h = swap_move(h, i_offend, j)
                new_c5 = evaluate(new_h)
                if new_c5 <= current_c5:
                    h = new_h
                    current_c5 = new_c5
            
            # Phase 3: continuous refinement
            if time.time() < t_slice_end:
                h_f = h.astype(float)
                # Initially smooth β to get even landscape
                β = 5
                β_multi = 1.02
                α = 0.02
                max_iter_cont = 5000
                
                for iter in range(max_iter_cont):
                    if time.time() >= t_slice_end:
                        break
                    grad = gradient_of_max(h_f, β)
                    h_f = h_f - α * grad
                    h_f = project_onto_capped_simplex(h_f, N / 2)
                    _ = evaluate(h_f)
                    β *= β_multi
                
                c5_cont = evaluate(h_f)
                if c5_cont < best_local_c5:
                    best_local_c5 = c5_cont
                    h = h_f
            
            if best_local_c5 < best_binary_c5:
                best_binary_c5 = best_local_c5
                best_binary_solution = best_local_sol.copy()
        
        # Final evaluation
        final_c5 = evaluate(best_binary_solution)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = best_binary_solution
            best_n = N
    
    return (best_h.tolist(), best_c5, best_n)
