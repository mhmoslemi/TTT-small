import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Time management: spend 90% of time on MILP, 10% on local polish
    milp_time = 0.9 * budget_s
    polish_time = 0.1 * budget_s
    
    # Try a range of n values, starting with more promising ones
    n_values = [42, 60, 70, 80]
    
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    start_time = time.time()
    
    # Focus on a manageable size to keep within budget
    for n in n_values:
        if time.time() - start_time > milp_time:
            break
            
        dx = 2.0 / n
        # Target is ceiling of 0.3807 * n/2
        target_c5 = 0.3807
        target_int = int(np.ceil(target_c5 * n / 2))
        
        # Build MILP model
        try:
            # Variables
            b = cp.Variable(n, boolean=True)
            t = cp.Variable()
            
            # Constraints
            constraints = [cp.sum(b) == n // 2]
            # Symmetry breaking
            constraints.append(b[0] == 1)
            # Lexicographic ordering
            for i in range(n - 1):
                constraints.append(b[i] >= b[i+1])
            
            # Add lag constraints
            # Create auxiliary variables for b_i * b_{i+ℓ} 
            # Note: we will treat each non-zero ℓ separately
            # and let t = max over all relevant ℓ of (sum b - sum w_ℓ) = sum b - min sum w_ℓ
            
            for ell in range(1, n):
                if time.time() - start_time > milp_time:
                    break
                    
                # w[i] = b[i] AND b[i+ell] (only when defined)
                # w ell: [0, ..., (n-ell-1)] -> length (n-ell)
                w = cp.Variable(n - ell, boolean=True)
                # w[i] <= b[i], w[i] <= b[i+ell], w[i] >= b[i] + b[i+ell] - 1
                for i in range(n - ell):
                    constraints.append(w[i] <= b[i])
                    constraints.append(w[i] <= b[i + ell])
                    constraints.append(w[i] >= b[i] + b[i + ell] - 1)
                
                # N_ell = sum b - sum w
                N_ell = cp.sum(b[:n-ell]) - cp.sum(w)
                constraints.append(t >= N_ell)
                    
            # Objective
            objective = cp.Minimize(t)
            prob = cp.Problem(objective, constraints)
            
            # Solve with a time limit
            prob.solve(solver=cp.CBC, max_seconds=milp_time - (time.time() - start_time))
            
            # Check if solution is found
            if prob.status in [cp.OPTIMAL, cp.FEASIBLE]:
                # Extract solution
                b_val = b.value
                t_val = t.value
                h = b_val.astype(float)
                c5 = t_val * dx
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    best_n = n

                if c5 <= target_c5:
                    break
        except Exception as e:
            continue

    # If we haven't found a good enough solution yet, we might allow fractional relaxation
    if best_h is None:
        # Use continuous relaxation and floating point approach
        try:
            n = 60
            dx = 2.0 / n
            h = cp.Variable(n)
            constr = [h >= 0, h <= 1, cp.sum(h) == 1.0]
            t = cp.Variable()
            # Add pretty small regressions of overlap constraints
            for ell in range(1, n):
                # Just use shifted correlation terms as expansion
                # N_ell = sum(h_i * (1 - h_{i+ell})) for all i when i+ell < n
                if ell > n // 2: break  # Usual tightening
                overlap = 0
                for i in range(n - ell):
                    overlap += h[i] * (1 - h[i + ell])
                constr.append(t >= overlap)
            constr.append(t >= 0)
            obj = cp.Minimize(t)
            prob = cp.Problem(obj, constr)
            prob.solve(solver=cp.SCS, eps=1e-5)
            if prob.status in [cp.OPTIMAL, cp.FEASIBLE]:
                h_val = h.value
                t_val = t.value
                c5 = t_val * dx
                best_c5 = c5
                best_h = h_val.copy()
                best_n = n
        except:
            pass

    # Polishing step
    if polish_time > 0 and best_h is not None:
        h_orig = best_h.copy()
        current_c5 = best_c5

        # Very brief local change attempt (gradient-free)
        max_iter = 1000
        iter_count = 0
        start_polish = time.time()
        while iter_count < max_iter and time.time() - start_polish < polish_time:
            i1 = np.random.randint(0, len(best_h))
            # Tweak a small value in h and see if we drop max overlap
            delta = np.random.normal(0, 0.001)
            original = best_h[i1]
            new = np.clip(original + delta, 0, 1)
            best_h[i1] = new
            # Re-evaluate max correlation
            max_corr = 0
            for k in range(1, len(best_h)):
                corr = 0
                for i in range(len(best_h) - k):
                    corr += best_h[i] * (1 - best_h[i+k])
                max_corr = max(max_corr, corr)
            ps = 0.0
            if len(best_h) > 0: 
                ps = 1.0 / len(best_h)  # dx is already incorporated in max_corr
            # Sometimes we don't penalize constraint violation but just try to reduce peak correlation
            new_c5_bound = max_corr * ps  # (Actually this is an approximation)
            # The rest is tricky with continuous values and doesn't cleanly extend
            # For now keep best as evidence is still too noisy for not knowing C5 without max 
            # Let's correct this approach...
            # We are going to simulate a basic coordinate descent path using only a total swap
            # to make small adjustment for effective smooth form
            iter_count += 1
        # Use original result unless supported improvement is made by random trial
        # Re-evaluate full manually on existing result
        def compute_c5_manual(h_arr):
            # Approximate C5 calculation - compute maximum auto-correlation
            n = len(h_arr)
            dx = 2.0 / n
            c5 = 0.0
            # Brute force over lags 1 to n-1
            for k in range(1, n):
                corr = 0.0
                for i in range(n - k):
                    corr += h_arr[i] * (1 - h_arr[i+k])
                c5 = max(c5, corr)
            return c5 * dx
        manual_c5 = compute_c5_manual(best_h) if best_h is not None else float('inf')
        if manual_c5 < current_c5:
            best_c5 = manual_c5
    
    # Final computation of C5 if needed
    if best_h is None:
        best_h = np.ones(42) * 0.5
        best_n = 42
        dx = 2.0 / best_n
        best_c5 = 0.5 * dx  # Placeholder
    
    # We want to make sure we evaluate properly
    final_c5 = 0.0
    try:
        final_c5 = compute_c5_manual(best_h)
    except:
        final_c5 = 0.5  # Fallback estimate
    
    return (best_h.tolist(), final_c5, best_n)
