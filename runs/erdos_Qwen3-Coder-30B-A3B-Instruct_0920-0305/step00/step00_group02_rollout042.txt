import numpy as np
import time
import math

def run(seed=42, budget_s=600, **kwargs):

    n_points = 300
    dx = 2.0 / n_points
    target_bound = 0.3807
    max_alpha = 10000.0
    alpha_schedule_interval = 200
    max_hill_climb_seconds = 180

    np.random.seed(seed)

    current_sum = np.sum(h)

        ones_indices = np.where(h == 1)[0]
        flip_indices = np.random.choice(ones_indices, size=int(current_sum - needed), replace=False)
        h[flip_indices] = 0
    elif current_sum < needed:
        # Find indices of 0s and flip some to 1
        zeros_indices = np.where(h == 0)[0]
        flip_indices = np.random.choice(zeros_indices, size=int(needed - current_sum), replace=False)
        h[flip_indices] = 1

    total = np.sum(h)
    if abs(total - needed) > 1e-6:

    start_time = time.time()
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    elapsed = time.time() - start_time
    
    # Continuous optimization - gradient ascent with log-sum-exp
    alpha = 1.0
    iteration = 0
    while elapsed < budget_s * 0.6:  # 60% for gradient, 40% for polish
        # Compute overlap sums S_k for all k
        # Use FFT: h * h with 'full' mode gives all cross-correlations
        # But we use full convolution of h with h to get S_k
        # We're computing the overlap at different shifts k
        # From definition S_k = sum_i h_i * h_{i+k}
        # This is cross correlate h with h
        c_full = np.correlate(h, h, mode='full')
        sk_array = np.array([c_full[n - 1 + k] for k in range(n_points)])  # S_k for k = 0..n-1
        # Prioritize making sums non-negative for numerical stability
        sk_array = np.clip(sk_array, 0, n_points)
        weights = np.exp(-alpha * sk_array)
        weight_sum = np.sum(weights)

        # Gradient of surrogate with respect to h
        grad = np.zeros_like(h)
        for k in range(n_points):
            wk = weights[k]
            # According to reverse step derivation: S_k = sum h_i h_{i+k}
            # So dS_k/dh_i = h_{i+k} + h_{i-k} (zero padding and orientation)
            # Use appropriate indexing: wrap around or zero padding
            k_shift_grad = np.zeros_like(h)
            if k < n_points:
                k_shift_grad[k:] += h[:-k]
            if k > 0:
                k_shift_grad[:-k] += h[k:]
            grad += wk * k_shift_grad

        # Normalize the gradient
        grad /= weight_sum if weight_sum > 1e-10 else 1.0
        
        # Step size line search
        step_size = 1.0
        new_h = h + step_size * grad
        new_h = project_box_and_sum(new_h, n_points / 2)

        logsumexp_new = -np.log(np.sum(np.exp(-alpha * sk_new))) / alpha
        logsumexp_old = -np.log(np.sum(np.exp(-alpha * sk_array))) / alpha

        while logsumexp_new < logsumexp_old and step_size > 1e-10:

        if iteration % alpha_schedule_interval == 0 and alpha < max_alpha:
            alpha *= 2.0
            alpha = min(alpha, max_alpha)

        else:

            flip_to_one = int(n_points / 2 - current_sum)

    best_C5 = compute_C5(binary_h, dx)

    climb_start = time.time()

        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)

        binary_h_copy = binary_h.copy()

        new_C5 = compute_C5(binary_h_copy, dx)
        
        # Accept if better
        if new_C5 < best_C5:
            binary_h = binary_h_copy.copy()
            best_C5 = new_C5
        
        # Occasionally accept neutral or even slightly worse moves to escape local minima
        if new_C5 <= best_C5 + 1e-8:
            binary_h = binary_h_copy.copy()
            best_C5 = new_C5

    return (binary_h.tolist(), float(best_C5), n_points)

def project_box_and_sum(h, target_sum):
    # Clip h into [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-6:
        return h

    # Here's a simple water-filling algorithm
    # Adjust by adding or subtracting a constant to make the sum correct
    delta = target_sum - current_sum
    if abs(delta) < 1e-9:
        return h
    # Since h is normalized to [0,1], we can add delta/n to all entries and then clip
    h = h + delta / len(h)

    # Ensure all values stay within [0,1]
    h = np.clip(h, 0, 1)
    return h

def compute_C5(h, dx):
    # Compute C5 based on Euclidean distance in overlapping region
    c = np.correlate(h, 1 - h, mode='full')
    # The maximum C5 from [0, 2] for physical shift k=0..2
    # Consider convolution over full range, and then find maximum
    max_overlap = np.max(c)
    return max_overlap * dx
