import numpy as np
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the Erdos overlap constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr) * dx
    return max_val

def initial_h_values():
    """Return an initial construction for h."""
    # Placeholder if needed, but should be imported externally.
    return np.array([0.5] * 42)  # Default placeholder; real h is pre-imported

def compute_c5_fast(h):
    """Compute the maximum overlap integral using FFT for fast computation."""
    n = h.shape[0]
    size = 1 << ((2*n-1).bit_length())
    f_h = np.fft.fft(h, size)
    f_g = np.fft.fft(1 - h, size)
    corr = np.fft.ifft(f_h * np.conj(f_g)).real
    corr = corr[:2*n-1]
    dx = 2.0 / n
    max_val = np.max(corr) * dx
    return max_val

def random_binary_vector(n):
    """Generate a random binary vector of length n with exactly n//2 ones."""
    h = np.zeros(n)
    ones_indices = random.sample(range(n), n // 2)
    h[ones_indices] = 1
    return h

def propose_swap(h):
    """Propose a swap between a 1 and a 0."""
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return None, None
    i = random.choice(ones)
    j = random.choice(zeros)
    return i, j

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 60  # Starting with n=60 which seems reasonable
    dx = 2.0 / n_points

    # Use initial_h_values if available; fallback to random if unavailable.
    try:
        h = initial_h_values()
        if len(h) != n_points:
            h = random_binary_vector(n_points)
    except:
        h = random_binary_vector(n_points)
    
    current_c5 = compute_c5_fast(h)
    best_c5 = current_c5
    best_h = h.copy()

    # Simple simulated annealing with linear temperature decay
    max_iter = 200000
    temp_start = 0.05
    temp_end = 1e-4
    time_limit = budget_s - 0.2  # Allow some buffer
    
    for iteration in range(max_iter):
        if time.time() - start_time > time_limit:
            break
            
        # Temperature schedule
        elapsed = time.time() - start_time
        temp = temp_start * (temp_end / temp_start) ** (elapsed / time_limit)
        
        # Propose a swap
        i, j = propose_swap(h)
        if i is None:
            # If no valid swap, try again or restart
            continue
            
        # Perform the swap
        h[i], h[j] = h[j], h[i]
        
        # Compute new overlap
        new_c5 = compute_c5_fast(h)
        
        # Accept or reject
        delta = new_c5 - current_c5
        if delta <= 0 or random.random() < np.exp(-delta / temp):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Revert swap
            h[i], h[j] = h[j], h[i]

    # Final check
    final_c5 = compute_c5_fast(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, n_points)
