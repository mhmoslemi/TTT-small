import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize
import time

def compute_gradient(h, w, dx):
    n = len(h)
    grad = np.zeros_like(h)
    offset = n - 1
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        shift = idx - offset
        if shift >= 0:
            # first term: grad[i] += wk * (1 - h[i + shift])
            grad[:n - shift] += wk * (1 - h[shift:])
            # second term: grad[i] -= wk * h[i - shift]
            grad[shift:] -= wk * h[:n - shift]
        else:
            # shift < 0
            s = -shift
            # first term: grad[i] += wk * (1 - h[i + shift]) = grad[i] += wk * (1 - h[i - s])
            grad[s:] += wk * (1 - h[:n - s])
            # second term: grad[i] -= wk * h[i - shift] = grad[i] -= wk * h[i + s]
            grad[:n - s] -= wk * h[s:]
    return grad * dx

def project_to_constraints(v, target_sum=0.5):
    # Binary search for lambda to satisfy sum constraint
    lo = -np.max(v) - 1.0
    hi = 1 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2.0, 0.0, 1.0)

def loss_and_grad(h, beta, dx):
    p = 1 - h
    c = np.correlate(h, p, mode='full') * dx
    scaled = beta * c
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w_sum = np.sum(w)
    w = w / w_sum
    loss = (max_scaled + np.log(w_sum)) / beta
    grad = compute_gradient(h, w, dx)
    return loss, grad

        # Project onto sum = target_sum

                loss, grad = loss_and_grad(x, beta, dx)
