import time
import numpy as np
import cvxpy as cp
import random
import math

def random_binary_vector(n):
    h = np.zeros(n, dtype=int)
    ones_indices = np.random.choice(n, n//2, replace=False)
    h[ones_indices] = 1
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Max over all shifts
    max_corr = np.max(corr)
    return max_corr * dx

def solve_milp(n, time_limit=10):
    # Create variables: h binary, t continuous
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h) == n // 2]
    
    # For all shifts k from 0 to n-1 (compute overlaps for each)
    # shift k means h[i] * h[i+k], over range where both indices valid
    for k in range(n):
        if n - k == 0:
            continue
        # indices for i: from 0 to n-k-1, respective partner indices are i+k
        # Create product variables for this shift
        y = cp.Variable(n - k, nonneg=True)
        # Linearize product constraint: y[i] = h[i] * h[i+k]
        for i in range(n - k):
            constraints += [y[i] <= h[i]]
            constraints += [y[i] <= h[i + k]]
            constraints += [y[i] >= h[i] + h[i + k] - 1]
        # Sum of products for this shift
        overlap_k = cp.sum(y)
        # Ensure t <= overlap_k
        constraints += [t <= overlap_k]
    
    # Objective: maximize t (minimum overlap)
    prob = cp.Problem(cp.Maximize(t), constraints)
    
    # Try to solve with CBC within time limit
    try:
        prob.solve(solver=cp.CBC, maxSeconds=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURACY]:
            # Check if we have valid binary solution
            h_sol = np.round(h.value).astype(int)
            if not np.all(np.isin(h_sol, [0, 1])):
                h_sol = (h_sol > 0.5).astype(int)
            # Compute actual C5 bound
            dx = 2.0 / n
            total = 1.0
            corr_full = np.correlate(h_sol, 1 - h_sol, mode='full')
            min_overlap = np.min(corr_full)
            c5 = total - min_overlap * dx
            return h_sol, float(c5)
        else:
            # Fallback to heuristic if MILP could not be solved optimally
            h_sol = np.round(h.value).astype(int)
            if not np.all(np.isin(h_sol, [0, 1])):
                h_sol = (h_sol > 0.5).astype(int)
            dx = 2.0 / n
            total = 1.0
            corr_full = np.correlate(h_sol, 1 - h_sol, mode='full')
            min_overlap = np.min(corr_full)
            c5 = total - min_overlap * dx
            return h_sol, float(c5)
    except Exception as e:
        # Return fallback
        h_sol = random_binary_vector(n)
        c5 = compute_c5(h_sol)
        return h_sol, c5

def local_search(h_initial, max_iters, time_limit):
    start = time.time()
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h_initial.copy()
    cur_c5 = compute_c5(cur_h)
    
    while time.time() - start < time_limit:
        # Random swap: choose one 1 and one 0 to swap
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Perform swap
        cur_h[i] = 0
        cur_h[j] = 1
        
        new_c5 = compute_c5(cur_h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # Revert swap
            cur_h[i] = 1
            cur_h[j] = 0
        
        max_iters -= 1
        if max_iters <= 0:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    budget_s = max(10, budget_s - 5) # Leave margin for cleanup
    
    # Try to solve MILP for small n
    n = 100
    try:
        h_opt, c5_opt = solve_milp(n, time_limit=min(10, budget_s - 10))
    except:
        h_opt = random_binary_vector(n)
        c5_opt = compute_c5(h_opt)
    
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    
    if remaining > 5 and False:  # Disable pure local search for now
        # Local search with better solution
        h_opt, c5_opt = local_search(h_opt, max_iters=50000, time_limit=remaining) 
    
    # Ensure all operations completed within time limit
    elapsed = time.time() - start
    if elapsed >= budget_s:
        pass # Best effort completed
    
    return h_opt.tolist(), float(c5_opt), int(n)
