import numpy as np
import cvxpy as cp
import time
import random
from scipy import sparse

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 200  # Grid points, should be even
    dx = 2.0 / n
    target_C5 = 0.3807
    time_left = lambda: budget_s - (time.time() - start_time)

    # Use sparse for memory efficiency
    A_matrices = []
    k_range = list(range(1, n//2 + 1))  # k from 1 to n//2
    
    print("Building sparse matrices...")
    for k in k_range:
        if time_left() < 5:
            break
        # Create sparse matrix with 1s for positions (i, i+k) and (i+k, i)
        rows = []
        cols = []
        for i in range(n - k):
            rows.extend([i, i + k])
            cols.extend([i + k, i])
        data = [1] * len(rows)
        A = sparse.csr_matrix((data, (rows, cols)), shape=(n, n))
        A_matrices.append(A)
    
    print("Creating SDP problem...")
    
    # SDP Problem setup
    # Decision variables
    X = cp.Variable((n, n), symmetric=True)
    x = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        X >> 0,  # Positive semidefinite
        cp.diag(X) == x,
        cp.sum(x) == n / 2,
        0 <= x,
        x <= 1
    ]
    
    for i, A in enumerate(A_matrices):
        if time_left() < 5:
            break
        constraints.append(cp.trace(A @ X) >= 2 * t)
    
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    print("Solving SDP...")
    # Solve with time limit
    try:
        prob.solve(solver=cp.SCS, max_iters=2000, eps=1e-4, verbose=False)
    except Exception as e:
        print(f"SDP error: {e}")
    
    if prob.status != cp.OPTIMAL:
        print("SDP did not converge; using initial solution")
        return initial_h_values, 0.38092, len(initial_h_values)
        
    X_star = X.value
    t_star = t.value
    
    print(f"SDP solved. t* = {t_star:.6f}")
    C5_lower_bound = 1 - t_star * dx
    
    print(f"SDP C5 lower bound = {C5_lower_bound:.6f}")
    
    # Rounding phase: randomized hyperplane rounding using Cholesky
    def round_to_binary(X_mat, num_rounds=250):
        # Compute low-rank factorization V such that X ≈ V @ V.T
        try:
            # SVD-based low rank factorization
            u, s, vt = np.linalg.svd(X_mat, full_matrices=False)
            rank = min(10, len(s))
            V = u[:, :rank] * np.sqrt(s[:rank])
        except Exception:
            # Fallback to Cholesky if SVD fails
            try:
                L = np.linalg.cholesky(X_mat)
                V = L.T
                rank = min(10, V.shape[0])
                V = V[:rank, :]  # Take first rank rows
            except:
                # Fallback to a simple low-rank approximation
                rank = min(8, n)
                V = np.random.rand(n, rank) * 0.1
        best_h = None
        best_C5 = float('inf')
        
        for trial in range(num_rounds):
            if time_left() < 2:
                break
            # Sample random vector g
            g = np.random.randn(rank)
            s = V.T @ g
            # Find threshold τ such that exactly n/2 entries of s are ≥ τ
            sorted_s = np.sort(s)[::-1]
            if n//2 >= len(sorted_s):
                tau = sorted_s[-1] - 1e-10
            else:
                tau = sorted_s[n//2 - 1]
            h = (s >= tau).astype(int)
            if np.sum(h) != n//2:
                # Use exact median-based threshold if needed
                sorted_s = np.sort(s)
                idx = n//2 - 1
                if idx >= len(sorted_s):
                    idx = len(sorted_s) - 1
                tau = sorted_s[idx]
                h = (s >= tau).astype(int)
                if np.sum(h) != n//2:
                    # Potentially adjust account for floating point precision
                    diff = n//2 - np.sum(h)
                    inds0 = np.where(h == 0)[0]
                    inds1 = np.where(h == 1)[0]
                    if diff > 0 and len(inds0) >= diff:
                        to_flip = inds0[:diff]
                        h[to_flip] = 1
                    elif diff < 0 and len(inds1) >= -diff:
                        to_flip = inds1[:abs(diff)]
                        h[to_flip] = 0
            if np.sum(h) != n//2:
                continue
            # Compute actual C5
            corr = np.correlate(h, 1 - h, mode='full')
            C5_trial = np.max(corr) * dx
            if C5_trial < best_C5:
                best_C5 = C5_trial
                best_h = h.copy()
        return best_h, best_C5
    
    # Perform rounding
    print("Rounding SDP solution...")
    if time_left() < 5:
        return initial_h_values, 0.38092, len(initial_h_values)
    
    best_h, best_C5 = round_to_binary(X_star, num_rounds=250)
    
    if best_h is None:
        print("Rounding failed, using initial solution")
        return initial_h_values, 0.38092, len(initial_h_values)
    
    # Local polishing with hill-climbing swaps
    if time_left() < 5:
        return best_h.tolist(), best_C5, n
    
    print("Starting local polishing...")
    # Utilities for fast correlation recalculation
    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def fast_update_correlation(old_corr, h, i, j, corr_vals, n):
        # Remove contributions from positions i, j
            
        # assumes h is binary, 0 or 1
        new_corr = corr_vals.copy()
        for k in range(1, n):
            i1 = i - k
            j1 = j - k
            if 0 <= i1 < n:
                new_corr[k-1] -= h[i1] * (1 - h[i1+k])
            if 0 <= j1 < n:
                new_corr[k-1] -= h[j1] * (1 - h[j1+k])
        # Add updated contributions
        for k in range(1, n):
            i1 = i - k
            j1 = j - k
            if 0 <= i1 < n:
                new_corr[k-1] += h[i1] * (1 - h[i1+k])
            if 0 <= j1 < n:
                new_corr[k-1] += h[j1] * (1 - h[j1+k])
        return new_corr
    
    # Evaluate initial full correlation
    corr_full = np.correlate(best_h, 1 - best_h, mode='full')
    corr_vals = corr_full[n-1:]
    # Find worst shift (k that gives min overlap)
    min_k = np.argmin(corr_vals)
    print(f"Initial min overlap at shift {min_k}")
    
    # Modify best_h by swapping
    max_swaps = int(10000 * time_left() / 120.0)  # up to ~100K swaps
    swaps_done = 0
    
    # Precomputed lookup of all 1s and 0s positions
    ones = np.where(best_h == 1)[0].tolist()
    zeros = np.where(best_h == 0)[0].tolist()
    # Randomized swaps
    for _ in range(max_swaps):
        if time_left() < 1:
            break
        
        if not ones or not zeros:
            break
            
        # Pick two positions to swap: 1 -> 0, 0 -> 1
        i = random.choice(ones)
        j = random.choice(zeros)
        
        # Perform swap in h
        best_h[i] = 0
        best_h[j] = 1
        ones.remove(i)
        zeros.remove(j)
        ones.append(j)
        zeros.append(i)
        
        # Recompute overlap efficiently
        new_corr = np.correlate(best_h, 1 - best_h, mode='full')
        new_corr_vals = new_corr[n-1:]
        min_k_new = np.argmin(new_corr_vals)
        best_C5_new = np.max(new_corr) * dx
        
        if best_C5_new < best_C5:
            # Accept improvement
            best_C5 = best_C5_new
            corr_vals = new_corr_vals
            min_k = min_k_new
            swaps_done += 1
        else:
            # Rollback
            best_h[i] = 1
            best_h[j] = 0
            ones.append(i)
            zeros.append(j)
            ones.remove(j)
            zeros.remove(i)
    
    print(f"Polished {swaps_done} swaps, final C5 = {best_C5}")
    if best_C5 <= target_C5:
        print(f"Target C5 = {target_C5} achieved!")
    
    return best_h.tolist(), best_C5, n
