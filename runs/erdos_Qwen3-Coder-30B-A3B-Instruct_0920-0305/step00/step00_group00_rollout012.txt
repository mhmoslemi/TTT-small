import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize
import time

def compute_gradient(h, w, dx):
    """Compute gradient of smooth approximation to max correlation."""
    n = len(h)
    grad = np.zeros_like(h)
    offset = n - 1
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        shift = idx - offset
        if shift >= 0:
            grad[:n - shift] += wk * (1 - h[shift:])
            grad[shift:] -= wk * h[:n - shift]
        else:
            s = -shift
            grad[s:] += wk * (1 - h[:-s])
            grad[:-s] -= wk * h[s:]
    return grad * dx

def project_to_constraints(v, lower=0.0, upper=1.0, target_sum=None):
    """Project v onto box constraints [lower, upper] and sum constraint."""
    if target_sum is None:
        target_sum = len(v) / 2.0
    lo = -np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, lower, upper)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo+hi)/2.0, lower, upper)
    return x

def loss_and_grad(h, beta, dx):
    """Compute smooth approximation to max correlation and its gradient."""
    p = 1 - h
    c = np.correlate(h, p, mode='full') * dx
    scaled = beta * c
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w_sum = np.sum(w)
    w = w / w_sum
    loss = (max_scaled + np.log(w_sum)) / beta
    grad = compute_gradient(h, w, dx)
    return loss, grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    n = 140  # Moderate resolution
    dx = 2.0 / n
    target_sum = n / 2.0
    max_restarts = 10
    time_per_restart = budget_s / max_restarts * 0.9
    
    # Use initial_h_values if available
    try:
        from initial_h_values import initial_h_values
        old_h = np.array(initial_h_values)
        old_n = len(old_h)
        old_x = np.linspace(0, 2, old_n, endpoint=False) + 1.0 / old_n
        new_x = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        h0 = np.interp(new_x, old_x, old_h)
        h0 = project_to_constraints(h0 - 0.5, target_sum=target_sum)
    except ImportError:
        h0 = project_to_constraints(np.random.random(n) - 0.5, target_sum=target_sum)
    
    for restart in range(max_restarts):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
            
        # Initialize
        h = h0.copy()
        beta_schedule = [1.0, 5.0, 15.0, 30.0]
        max_iters = 200
        
        for beta in beta_schedule:
            if time.time() - start_time > budget_s * 0.95:
                break
            for iter in range(max_iters):
                if time.time() - start_time > budget_s * 0.95:
                    break
                try:
                    loss, grad = loss_and_grad(h, beta, dx)
                    h_new = project_to_constraints(h - 0.1 * grad, target_sum=target_sum)
                    c = np.correlate(h_new, 1 - h_new, mode='full') * dx
                    c5 = c.max()
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_new.copy()
                    h = h_new
                except Exception:
                    break
                
        # Random local search
        h = project_to_constraints(np.random.random(n) - 0.5, target_sum=target_sum)
        for iter in range(50):
            if time.time() - start_time > budget_s * 0.95:
                break
            c = np.correlate(h, 1 - h, mode='full') * dx
            c5 = c.max()
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            i, j = np.random.randint(0, n, 2)
            if (h[i] < 1) and (h[j] > 0):
                delta = 0.02 * np.random.random() - 0.01
                if abs(h[i] + delta) <= 1 and abs(h[j] - delta) <= 1:
                    h[i] += delta
                    h[j] -= delta
                    h = project_to_constraints(h, target_sum=target_sum)
    
    # Final evaluation
    if best_h is not None:
        try:
            from evaluate_erdos_solution import evaluate_erdos_solution
            final_c5 = evaluate_erdos_solution(best_h)
            return (best_h.tolist(), final_c5, n)
        except ImportError:
            return (best_h.tolist(), best_c5, n)
    else:
        return ([0.5]*n, 0.5, n)
