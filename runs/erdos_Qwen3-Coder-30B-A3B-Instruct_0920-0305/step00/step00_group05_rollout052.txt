import numpy as np
import time
import random
from math import exp, log

# Helper functions
def project_onto_capped_simplex(y, target_sum):
    """Project y onto the capped simplex: {x ∈ [0,1]^N | sum(x) = target_sum}"""
    y = np.clip(y, 0, 1)
    sorted_y = np.sort(y)[::-1]
    cumsum = np.cumsum(sorted_y)
    n = len(y)
    # Find the largest k such that sum(sorted_y[:k]) <= target_sum
    # We check for the element where the condition changes
    temp = np.arange(1, n + 1) * sorted_y - cumsum + target_sum
    k = np.sum(temp > 0)
    if k == 0:
        lambda_val = 0.0
    else:
        lambda_val = (cumsum[k-1] - target_sum) / k
    x = np.clip(y - lambda_val, 0, 1)
    return x

def evaluate(x):
    """Returns C5 = max_k int h(x)(1 - h(x+k)) dx"""
    dx = 2.0 / len(x)
    # Compute correlations for all lags
    corr = np.correlate(x, 1 - x, mode="full")
    # Get the max of all valid correlations
    max_corr = np.max(corr)
    return max_corr * dx

def gradient_of_max(x, beta):
    """Gradient of smooth approximation to max_k O_k(x)"""
    n = len(x)
    dx = 2.0 / n
    # Compute all O_k(x)
    correlations = np.correlate(x, 1 - x, mode="full")
    # Scale back for dx
    correlations *= dx
    
    # Compute softmax weights
    exp_corr = np.exp(beta * correlations)
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradients of each O_k w.r.t. x_i
    # Initialize gradient vector
    grad = np.zeros(n)
    
    # Compute each dO_k/dx_i for all k:  
    # |O_k| = n - |k|, so we have k from -(n-1) to (n-1)
    # For k = 0, dO_0/dx_i = 1-x_{i+0} - x_{i-0}
    # For k > 0, dO_k/dx_i = (1-x_{i+k}) - x_{i-k}
    # For k < 0, dO_k/dx_i = (1-x_{i+k}) - x_{i-k}
    
    for k_idx, k in enumerate(range(-n+1, n)):
        if k >= 0:
            # For k >= 0
            # Contribution to x_i: (1 - x_{i+k})
            # Contribution to x_{i-k}: -x_{i-k}
            # Only valid when: i+k < n and i-k >= 0
            # Valid i are: k <= i <= n-1-k
            
            if k == 0:
                # dO_0/dx_i = (1-x_{i+0}) - x_{i-0} = 1 - 2*x_i
                # Only valid for 0 <= i <= n-1
                # But we don't have x_{i-0} for i=0 etc..., 
                # So take a convention that-out-of-bound indices are 0.
                # Actually, orginalemvariance is just shuffling within [0, n)
                # and it avoids out-of-bound in real math.
                # We can compile this in proper % scaling style.
                # Equivalently, here, we consider chains of 0s in ends.
                for i in range(n):
                    grad[i] += weights[k_idx] * (1 - 2 * x[i])  
            else:
                # for x_{i+k} and x_{i-k} energies:
                # i goes up to (n-1-k) for i+k < n
                # i goes down from (k) to avoid (i-k) < 0
                # valid range: max(0,k) <= i <= n-1-k
                valid_start = max(0, k)
                valid_stop = n - k
                for i in range(valid_start, valid_stop):
                    grad[i] += weights[k_idx] * (1 - x[i+k])
                    grad[i-k] -= weights[k_idx] * x[i]
        else:
            # k < 0
            # symmetrical but in opposite way
            # x_{i+k} = x_{i-(-k)}  and x_{i-k} = x_{i-(-k)},
            # just in both directions
            # dO_k/dx_i = (1 - x_{i+k}) - x_{i-k}
            k_abs = -k
            # i goes up to (n-1+k_abs) for i+k_abs < n  
            # i goes down from (k_abs) to avoid i - k_abs < 0
            # valid range: k_abs <= i <= n-1-k_abs
            valid_start = k_abs
            valid_stop = n - k_abs
            for i in range(valid_start, valid_stop):
                grad[i] += weights[k_idx] * (1 - x[i+k])
                grad[i-k] -= weights[k_idx] * x[i]
    
    return grad

def swap_move(x, i, j):
    """Swap x[i] and x[j]"""
    x = x.copy()
    x[i], x[j] = x[j], x[i]
    return x

def fractional_move(x, i, j, delta):
    """Move delta from x[j] to x[i] while keeping constraints"""
    x = x.copy()
    new_i = x[i] + delta
    new_j = x[j] - delta
    if new_i < 0:
        # Need to adjust
        delta = -x[i]  # full amount from i to j
        new_i = 0.0
        new_j = x[j] + delta
    elif new_j < 0:
        # Need to adjust
        delta = x[j]  # full amount from j to i
        new_j = 0.0
        new_i = x[i] + delta
    if new_i > 1:
        delta = 1 - x[i]
        new_i = 1.0
        new_j = x[j] - delta
    elif new_j > 1:
        delta = x[j] - 1
        new_j = 1.0
        new_i = x[i] + delta
    x[i] = new_i
    x[j] = new_j
    return x

def worst_shift_info(x):
    """Return (k_star, offenses) where k_star is the worst shift and offenses are indices of 1's followed by 0's at that shift"""
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    max_corr = np.max(corr)
    k_star = np.argmax(corr) - (n - 1)
    # find all positions where x_i=1 and x_{i+|k_star|} = 0
    off_indices = []
    k_abs = abs(k_star)
    if k_star >= 0:
        # then x_{i+k_abs} must be after i, i+k_abs < n
        for i in range(n - k_abs):
            if x[i] == 1 and x[i + k_abs] == 0:
                off_indices.append(i)
    else:
        # x_{i+k_abs} = x_{i-|k_star|} is before i
        for i in range(k_abs, n):
            if x[i] == 1 and x[i - k_abs] == 0:
                off_indices.append(i)
    return (k_star, off_indices)

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    t_start = time.time()
    t_dead = t_start + budget_s - 5  # Safety margin
    
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    # Candidate sizes to try
    candidate_ns = [140, 200, 260]
    
    for n in candidate_ns:
        if time.time() >= t_dead:
            break
            
        t_slice_end = t_dead
        loop_time = 0
        while time.time() < t_slice_end:
            if time.time() >= t_dead:
                break
                
            # Phase 2.1: Binary Simulated Annealing
            t_slice_start = time.time()
            tries = 0
            while time.time() < t_slice_end:
                tries += 1
                # Initialize random binary vector
                h = np.zeros(n)
                ones_indices = np.random.choice(n, size=n//2, replace=False)
                h[ones_indices] = 1
                
                # Initial evaluation
                current_c5 = evaluate(h)
                best_so_far = current_c5
                x_best = h.copy()
                
                # Temperature schedule
                T0 = 0.12
                T_min = 1e-5
                iterations = 30000
                t0 = time.time()
                t_last_improvement = t0
                
                for iter_num in range(iterations):
                    if time.time() >= t_slice_end or time.time() >= t_dead:
                        break
                        
                    # Calculate current T
                    t_elapsed = time.time() - t0
                    if iter_num < iterations:
                        T = T0 * (1.0 - t_elapsed / (iterations * 0.01))
                    else:
                        T = T_min
                    
                    if T <= T_min:
                        T = T_min
                    
                    # Generate move: find two distinct positions with opposite values
                    indices = list(range(n))
                    random.shuffle(indices)
                    i, j = None, None
                    for idx in indices:
                        if i is None and h[idx] == 1:
                            i = idx
                        elif j is None and h[idx] == 0:
                            j = idx
                        if i is not None and j is not None:
                            break
                    
                    if i is None or j is None:
                        continue
                    
                    # Make the swap
                    h_temp = swap_move(h, i, j)
                    
                    # Evaluate new solution
                    new_c5 = evaluate(h_temp)
                    
                    # Accept or reject
                    delta = new_c5 - current_c5
                    if delta <= 0:
                        h = h_temp
                        current_c5 = new_c5
                        if current_c5 < best_so_far:
                            best_so_far = current_c5
                            x_best = h.copy()
                    else:
                        if random.random() < exp(-delta / T):
                            h = h_temp
                            current_c5 = new_c5
                            if current_c5 < best_so_far:
                                best_so_far = current_c5
                                x_best = h.copy()
                
                # Store best found binary vector for this N
                if best_so_far < best_c5:
                    best_c5 = best_so_far
                    best_h = x_best.copy()
                    best_n = n

                # Phase 2.2: Directed local refinement
                if time.time() < t_slice_end:
                    for _ in range(2000):
                        if time.time() >= t_slice_end or time.time() >= t_dead:
                            break
                        # Compute the worst shift
                        k_star, off_indices = worst_shift_info(best_h) 
                        if not off_indices:
                            break
                            
                        # Pick a random one in off_indices
                        i = random.choice(off_indices)
                        # Find a j = 0 that is not part of the pair (i+|k-star|) 
                        # or equal to i
                        valid_j_indices = [j for j in range(len(best_h)) if 
                                           best_h[j] == 0 and 
                                           j != i and 
                                           j != (i + abs(k_star))]
                        if not valid_j_indices:
                            break  # no eligible j found
                        j = random.choice(valid_j_indices)
                        
                        # Perform swap
                        h_temp = swap_move(best_h, i, j)
                        new_c5 = evaluate(h_temp)
                        
                        if new_c5 <= best_c5:
                            best_c5 = new_c5
                            best_h = h_temp
                        else:
                            # Use the previous one
                            pass
                
                # Phase 2.3: Continuous fine-tuning
                if time.time() < t_slice_end:
                    # Start from the best binary solution
                    h = best_h.copy()
                    initial_c5 = best_c5
                    # Scale it to get values in [0,1], sum should be N/2
                    # Already true since binary
                    beta = 5.0
                    alpha = 0.02
                    max_iter = 5000
                    for i in range(max_iter):
                        if time.time() >= t_slice_end or time.time() >= t_dead:
                            break
                        
                        if i % 100 == 0 and i != 0:
                            beta *= 1.02
                            if beta > 200:
                                beta = 200
                            
                        grad = gradient_of_max(h, beta)
                        h_new = h - alpha * grad
                        h_new = project_onto_capped_simplex(h_new, n / 2)
                        h = h_new
                        
                        new_c5 = evaluate(h)
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
                        
                        if abs(new_c5 - initial_c5) < 1e-5:
                            # converged
                            break
                
                if best_c5 < 0.379:
                    break
                
            if best_c5 < 0.379:
                break
                
        # If found a good enough bound, we can end early
        if best_c5 < 0.379:
            break
            
    # Final evaluation
    final_c5 = evaluate(best_h)
    
    # Report result
    return (best_h.tolist(), final_c5, best_n)
