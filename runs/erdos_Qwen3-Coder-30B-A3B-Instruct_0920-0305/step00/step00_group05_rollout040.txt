import numpy as np
import time
import random

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    time_end = start + budget_s - 0.5
    
    n_points = 200
    dx = 2.0 / n_points
    
    best_h = None
    best_c5 = float('inf')
    
    # Number of restarts to try
    max_restarts = 30
    restart_count = 0
    
    while time.time() < time_end and restart_count < max_restarts:
        # Random binary vector
        h = np.zeros(n_points, dtype=np.float64)
        ones = np.random.choice(n_points, n_points // 2, replace=False)
        h[ones] = 1.0
        
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Local simulated annealing
        T0 = 0.05
        max_iter = 20000
        for i in range(max_iter):
            if time.time() >= time_end:
                break
                
            # Choose two different indices
            idx1 = np.random.randint(n_points)
            idx2 = np.random.randint(n_points)
            while h[idx1] == h[idx2]:
                idx1 = np.random.randint(n_points)
                idx2 = np.random.randint(n_points)
            
            # Swap the values
            h[idx1], h[idx2] = h[idx2], h[idx1]
            
            # Compute new C5
            new_c5 = compute_c5(h)
            
            # Accept or reject
            delta = new_c5 - current_c5
            T = T0 * (1 - i / max_iter)
            if delta <= 0 or np.random.rand() < np.exp(-delta / (T + 1e-12)):
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            else:
                # Swap back
                h[idx1], h[idx2] = h[idx2], h[idx1]
        
        restart_count += 1
    
    if best_h is None:
        # Fallback to random solution
        h = np.zeros(n_points, dtype=np.float64)
        ones = np.random.choice(n_points, n_points // 2, replace=False)
        h[ones] = 1.0
        best_h = h
        best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
