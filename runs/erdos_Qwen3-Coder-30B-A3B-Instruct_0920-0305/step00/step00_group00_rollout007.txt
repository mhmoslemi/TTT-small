import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute cross-correlation of h with (1-h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value is what we want
    max_val = np.max(correlation) * dx
    return max_val

def initial_h_values(n_points=45):
    """Initial construction provided by the problem"""
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    # Simple construction: piecewise constant with jump at 1
    h[x < 1] = 1
    h[x >= 1] = 0
    # Scale so that integral = 1
    h = h * 2.0 / n_points
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45  # Use the provided initial construction size
    dx = 2.0 / n_points
    
    # Start from the initial solution
    current_h = initial_h_values(n_points).copy()
    current_c5 = evaluate_erdos_solution(current_h)
    
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Define constraints for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization parameters
    max_iter = 10000
    iter_count = 0
    
    # Store optimization history
    history = []
    
    # Gradient-free optimization approach
    # Use a hybrid: small perturbations, then optimize if beneficial
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Make small perturbation to current solution
        h_new = current_h.copy()
        # Randomly perturb a few entries
        n_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        for i in indices:
            # Add small random change
            delta = np.random.normal(0, 0.01)
            h_new[i] += delta
            # Clamp to bounds
            h_new[i] = max(0, min(1, h_new[i]))
        
        # Rescale to satisfy constraint
        h_new_sum = np.sum(h_new) * dx
        h_new = h_new * (1.0 / h_new_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new)
        
        # Check if we should accept the new solution
        # Accept if it's better or sometimes randomly
        if new_c5 < current_c5 or np.random.random() < 0.05:
            current_h = h_new
            current_c5 = new_c5
            
        # Update best solution if needed
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        history.append(current_c5)
        
        # Print progress every 100 iterations
        if iter_count % 100 == 0:
            elapsed = time.time() - start_time
            remaining = budget_s - elapsed
            print(f"Iter {iter_count}, C5: {current_c5:.6f}, Best: {best_c5:.6f}, Time: {elapsed:.1f}s, Remaining: {remaining:.1f}s")
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
