import numpy as np
import time
from scipy.optimize import dual_annealing

def project_onto_simplex_box(h, target_sum):
    # Project h onto the simplex with sum = target_sum and bounds [0,1]
    # Use iterative projection method
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    if current_sum > target_sum:
        # Need to decrease values
        # Sort h descending to find threshold
        sorted_indices = np.argsort(h)[::-1]
        cumsum = 0.0
        for i in range(len(h)):
            idx = sorted_indices[i]
            cumsum += h[idx]
            if cumsum > target_sum + (i + 1) * 1:
                # Threshold is (cumsum - target_sum) / (i + 1)
                threshold = (cumsum - target_sum) / (i + 1)
                for j in range(i + 1):
                    idx_j = sorted_indices[j]
                    h[idx_j] = max(0, h[idx_j] - threshold)
                break
        else:
            # All values need to be 0
            h = np.zeros_like(h)
    else:
        # Need to increase values
        diff = target_sum - current_sum
        sorted_indices = np.argsort(h)
        cumsum = 0.0
        for i in range(len(h)-1, -1, -1):
            idx = sorted_indices[i]
            cumsum += 1 - h[idx]  # amount we can increase this
            if cumsum >= diff:
                amount = diff
                h[idx] += amount
                diff = 0
                break
            else:
                h[idx] = 1
                diff -= cumsum
        if diff > 0:
            # Increase uniformly - this should not happen under constraint
            h += diff / len(h)
    return h

def evaluate(h):
    """Evaluate the maximum overlap integral"""
    dx = 2.0 / len(h)
    result = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(result)

def optimize_step(h, steps=500, lr=0.01):
    """Optimize h using gradient descent"""
    n_points = len(h)
    target_sum = n_points / 2
    
    # Initialize with projected h (makes sure sum constraint satisfied)
    h = project_onto_simplex_box(h, target_sum)
    best_h = h.copy()
    best_c5 = evaluate(h)
    
    for step in range(steps):
        # Compute raw overlaps (no dx yet)
        S_raw = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(S_raw)
        l = idx - (n_points - 1)  # convert index to lag value
        
        # Compute gradient for the maximum overlap
        grad = np.zeros(n_points)
        if l >= 0:
            i_start = 0
            i_end = n_points - l - 1
            for i in range(i_start, i_end + 1):
                grad[i] += 1 - h[i + l]
                grad[i + l] -= h[i]
        else:
            i_start = -l
            i_end = n_points - 1
            for i in range(i_start, i_end + 1):
                grad[i] += 1 - h[i + l]
                grad[i + l] -= h[i]
        
        # Update with learning rate
        h = h - lr * grad
        h = project_onto_simplex_box(h, target_sum)
        
        # Update best solution
        current_c5 = evaluate(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
    return best_h, best_c5
