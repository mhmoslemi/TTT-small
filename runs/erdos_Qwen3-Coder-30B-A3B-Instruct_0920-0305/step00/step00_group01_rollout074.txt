import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute the correlation integral
    conv = np.correlate(h_values, 1 - np.array(h_values), mode="full")
    # The maximum of the convolution gives us the C5 bound
    c5 = np.max(conv) * dx
    return c5

def optimize_h(n_points, max_time_s):
    """Optimize h using differential evolution with bounds."""
    # Start with a reasonable initial solution (from given initial_h_values or a simple one)
    # Use a simple construction that might give us a good starting point
    h0 = np.full(n_points, 0.5)
    # Try to use the initial_h_values if it's available and has the correct length
    try:
        # Refer to global initial_h_values if available
        global initial_h_values
        if len(initial_h_values) == n_points:
            h0 = np.array(initial_h_values)
    except:
        pass
    
    # Define bounds for each h[i] between 0 and 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Add the constraint that sum(h)*dx = 1
    # This means sum(h) should be n_points / 2
    
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Define objective function
    def objective(h):
        return evaluate_h(h)
    
    # Define constraint dictionary
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Optimize using differential evolution
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        constraints=constraints,
        maxiter=int(10000),
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=42,
        disp=False
    )
    
    # Return the best solution
    best_h = result.x
    best_c5 = evaluate_h(best_h)
    
    # If not within time budget yet, try local optimization
    if time.time() - start_time < max_time_s * 0.7:
        # Use a local optimizer for further improvement
        # We'll do one iteration of Nelder-Mead
        try:
            local_result = minimize(
                objective,
                best_h,
                method='Nelder-Mead',
                constraints=constraints,
                bounds=bounds,
                options={'maxiter': 1000}
            )
            if local_result.success:
                better_h = local_result.x
                better_c5 = evaluate_h(better_h)
                if better_c5 < best_c5:
                    best_h = better_h
                    best_c5 = better_c5
        except:
            pass
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    start_time = time.time()
    
    # Perform optimization
    h_values, c5_bound = optimize_h(n_points, budget_s)
    
    # Make sure the constraint is satisfied
    dx = 2.0 / n_points
    if abs(np.sum(h_values) * dx - 1.0) > 1e-6:
        # Adjust h to satisfy the constraint
        target_sum = n_points / 2
        current_sum = np.sum(h_values)
        adjustment = (target_sum - current_sum) / n_points
        h_values += adjustment
        h_values = np.clip(h_values, 0, 1)  # Ensure within [0, 1]
        
    # Final evaluation
    final_c5 = evaluate_h(h_values)
    return h_values, final_c5, n_points
