```python
import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time
from math import sqrt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial parameters
    n_points = 80
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        # Compute max overlap
        h_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(h_corr) * dx
        return c5
    
    def solve_lp_with_bisection(t_guess):
        # Set up LP model
        # Variables
        h = cp.Variable(n_points, name="h")
        p = cp.Variable((n_points, 2*n_points-1), name="p")  # p[i,d] where d ranges from -(n-1) to n-1
        
        # Constraints
        constraints = [h >= 0, h <= 1]
        constraints.append(cp.sum(h) == target_sum)
        
        # McCormick envelopes for p[i,d]
        for d in range(-(n_points-1), n_points):
            for i in range(n_points):
                if 0 <= i+d < n_points:
                    # p[i,d] = h[i] * h[i+d] for valid indices
                    p_ij = p[i, d + n_points - 1]  # Adjust index
                    constraints.extend([
                        p_ij <= h[i],
                        p_ij <= h[i+d],
                        p_ij >= h[i] + h[i+d] - 1,
                        p_ij >= 0
                    ])
        
        # Overlap constraints: sum(p[i,d]) >= (1-t)/dx for each d
        for d in range(-(n_points-1), n_points):
            if d >= 0:
                idx = d + n_points - 1
                constraints.append(cp.sum(p[:, idx]) >= (1 - t_guess) / dx)
        
        # Objective implicitly handled by feasibility check
        
        # Solve the LP
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.GLPK)
            if prob.status == cp.OPTIMAL:
                return h.value, True
            else:
                return None, False
        except:
            return None, False
            
    def round_solution(h_solution):
        # Round h to binary values, preserving sum
        h_sorted_indices = np.argsort(h_solution)[::-1]
        h_binary = np.zeros_like(h_solution)
        ones_count = int(round(target_sum))
        for i in range(ones_count):
            h_binary[h_sorted_indices[i]] = 1
        return h_binary
        
    # Bisection search on t
    lower_bound = 0.0
    upper_bound = 0.38092

    best_h = None
    best_c5 = float("inf")
    
    last_update = start_time
    while upper_bound - lower_bound > 1e-6:
        if time.time() - start_time > budget_s - 5:
            break
            
        t_mid = (lower_bound + upper_bound) / 2.0
        h_sol, is_feasible = solve_lp_with_bisection(t_mid)
        
        if is_feasible and h_sol is not None:
            h_rounded = round_solution(h_sol)
            actual_c5 = evaluate(h_rounded)
            
            if actual_c5 < best_c5:
                best_h = h_rounded
                best_c5 = actual_c5
            
            upper_bound = t_mid
        else:
            lower_bound = t_mid
            
        if time.time() - last_update > 10:
            last_update = time.time()

    if best_h is None:
        # Fallback if nothing found
        best_h = np.ones(n_points) * 0.5
        best_h[:n_points//2] = 1
        best_h[n_points//2:] = 0
        best_c5 = evaluate(best_h)
            
    return (best_h, best_c5, n_points)
```
