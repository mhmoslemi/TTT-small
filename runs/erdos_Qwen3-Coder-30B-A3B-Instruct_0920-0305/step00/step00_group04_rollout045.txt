import numpy as np
from scipy.optimize import differential_evolution, minimize
import time
from math import floor

def evaluate_erdos_solution(h):
    # Compute the overlap integral for the given step function h
    n = len(h)
    dx = 2.0 / n
    # Correlate h with (1-h) using full mode
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value, scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

    h[:n//2] = 0

    def objective(h):
        # Normalize h to satisfy integral constraint
        h_normalized = h / (np.sum(h) * dx)
        return evaluate_erdos_solution(h_normalized)
