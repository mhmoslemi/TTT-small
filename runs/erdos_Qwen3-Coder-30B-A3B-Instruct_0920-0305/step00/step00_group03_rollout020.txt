import numpy as np
from scipy.optimize import minimize
import time

def project_onto_simplex_box(h, target_sum):
    """Project h onto the simplex with sum=target_sum and box constraints [0,1]"""
    # Sort h in descending order
    h_sorted = np.sort(h)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(h_sorted)
    # Find the threshold
    rho = np.argmax(h_sorted - (target_sum - cumsum + h_sorted) / np.arange(1, len(h)+1) < 0)
    if rho == 0:
        theta = (target_sum - cumsum[-1]) / len(h)
    else:
        theta = (target_sum - cumsum[rho-1]) / rho
    # Apply thresholding and clipping
    h_proj = np.clip(h - theta, 0, 1)
    # Adjust to ensure exact sum
    diff = target_sum - np.sum(h_proj)
    idx = np.argsort(h_proj)
    for i in range(len(idx)):
        if diff >= 0:
            delta = min(diff, 1 - h_proj[idx[i]])
            h_proj[idx[i]] += delta
            diff -= delta
        else:
            break
    return h_proj

def evaluate(h_values):
    """Compute C5 using the provided evaluation method"""
    dx = 2.0 / len(h_values)
    # Compute np.correlate(h, 1-h, mode="full") * dx
    c = np.correlate(h_values, 1 - h_values, mode="full") * dx
    return np.max(c)

def compute_gradient(h_values, l):
    """Compute gradient of S_l with respect to h, where l is the lag"""
    N = len(h_values)
    grad = np.zeros(N)
    if l >= 0:
        # i from 0 to N-1-l
        for i in range(N - l):
            grad[i] += 1 - h_values[i + l]
            grad[i + l] -= h_values[i]
    else:
        # l < 0, we need to be careful
        l_abs = -l
        # i from l_abs to N-1
        for i in range(l_abs, N):
            grad[i] += 1 - h_values[i - l_abs]
            grad[i - l_abs] -= h_values[i]
    return grad

def optimize_step(h_values, lr=0.05, beta=1.0):
    """One optimization step"""
    N = len(h_values)
    dx = 2.0 / N
    
    # Compute all S values
    c = np.correlate(h_values, 1 - h_values, mode="full")
    # We compute for efficiency, store as raw counts
    max_idx = np.argmax(c)
    max_lag = max_idx - (N - 1)
    
    # Compute gradient for worst case lag
    grad = compute_gradient(h_values, max_lag)
    
    # Update h with gradient step
    h_new = h_values - lr * grad
    
    # Project onto feasible region
    h_projected = project_onto_simplex_box(h_new, N / 2)
    
    return h_projected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # parameters
    min_n = 100
    max_n = 300
    step_n = 50
    lr = 0.02
    steps_per_n = 10000
    
    best_h = None
    best_c5 = float('inf')
    n_points = 0
    
    for n in range(min_n, max_n + 1, step_n):
        if time.time() - start_time > budget_s - 10:  # Leave some buffer
            break
            
        # Initialize
        h = np.full(n, 0.5)
        h = project_onto_simplex_box(h, n / 2)
        
        # Optimize
        for step in range(steps_per_n):
            if time.time() - start_time > budget_s - 5:
                break
            h = optimize_step(h, lr=lr)
            if step % max(1, steps_per_n // 100) == 0:
                c5 = evaluate(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    n_points = n
                    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
            
    return (best_h.tolist(), best_c5, n_points)
