import numpy as np
import time
from scipy.special import softmax
from scipy.signal import correlate

# Predefined initial construction (from globals)
initial_h_values = None  # Will be filled in if available, otherwise use random

def project_onto_box_and_sum(h, target_sum):
    """Projects h onto [0,1] and enforces sum = target_sum using water filling"""
    # Define bounds
    low, high = 0.0, 1.0
    # Binary search for the Lagrange multiplier
    # The idea is to find lambda such that:
    # sum(clip(h + lambda, low, high)) = target_sum
    
    def sum_at_lambda(lam):
        # Compute the projection of h + lam onto [0,1]
        projected = np.clip(h + lam, low, high)
        return np.sum(projected)
    
    # Use bisection to find the correct lambda
    # Since the projection decreases as lambda increases beyond optimal,
    # we know that target_sum will be achieved with increasing lambda.
    
    # Start with a large enough range to bracket the solution
    lam_min = -1.0
    lam_max = 1.0
    # For safety, expand the bracket if needed
    while sum_at_lambda(lam_min) > target_sum:
        lam_min -= 1.0
    while sum_at_lambda(lam_max) < target_sum:
        lam_max += 1.0

    # Binary search to find lambda that maps to target_sum
    for _ in range(60):  # More than enough for double precision
        if lam_max - lam_min < 1e-14:
            break
        lam_mid = 0.5 * (lam_min + lam_max)
        current_sum = sum_at_lambda(lam_mid)
        if current_sum > target_sum:
            lam_max = lam_mid
        else:
            lam_min = lam_mid
            
    final_lambda = 0.5 * (lam_min + lam_max)
    
    # Compute the final projection
    result = np.clip(h + final_lambda, low, high)
    return result

def compute_correlation(h, dx):
    # This is equivalent to computing:
    # c[k] = dx * sum(h[i] * (1 - h[i+k]) )
    # for all shifts k (with zero-padding)
    full_corr = correlate(h, 1 - h, mode="full")
    return full_corr * dx

def compute_gradient_single_shift(h, k_shift, dx):
    # Compute gradient of c_k(h) = dx * sum(h[i] * (1 - h[i+k]))
    # This is partial derivative of c_k w.r.t. h_j
    # Let g_k_j = dx * (1 - h[j+k]) * 1_{j+k < n} - dx * h[j-k] * 1_{j-k >= 0}
    # i.e., we add (1-h[j+k]) where defined and subtract (h[j-k]) where defined
    
    n = len(h)
    grad = np.zeros(n)
    
    # Add contribution from (1 - h[j+k]):
    if k_shift >= 0:
        for j in range(n - k_shift):
            grad[j] += dx * (1 - h[j + k_shift])
    else:
        # k_shift is negative
        for j in range(-k_shift, n):
            grad[j] += dx * (1 - h[j + k_shift])

    # Subtract contribution from h[j-k]:
    if k_shift <= 0:
        for j in range(n + k_shift):
            grad[j] -= dx * h[j - k_shift]
    else:
        # k_shift is positive
        for j in range(k_shift, n):
            grad[j] -= dx * h[j - k_shift]
            
    return grad

def compute_gradient_of_LSE(h, corr_vals, beta, dx):
    """Computes the gradient of the log-sum-exp approximation to max correlation"""
    # Compute weights (softmax of beta * corr_vals)
    weights = softmax(beta * corr_vals)
    # Compute weighted combination of gradients for each shift
    grad = np.zeros_like(h)
    for i, weight in enumerate(weights):
        if weight > 1e-16:  # Avoid very small weights
            k_shift = i - (len(corr_vals) // 2)
            g = compute_gradient_single_shift(h, k_shift, dx)
            grad += weight * g
    return grad

def backtracking_line_search(h, grad, loss_fn, beta, dx, alpha0=0.2):
    alpha = alpha0
    current_corr = compute_correlation(h, dx)
    current_loss = np.log(np.sum(np.exp(beta * current_corr))) / beta
    h_proj = project_onto_box_and_sum(h - alpha * grad, len(h) / 2)
    new_corr = compute_correlation(h_proj, dx)
    new_loss = np.log(np.sum(np.exp(beta * new_corr))) / beta
    while new_loss > current_loss:
        alpha *= 0.5
        if alpha < 1e-10:
            break
        h_proj = project_onto_box_and_sum(h - alpha * grad, len(h) / 2)
        new_corr = compute_correlation(h_proj, dx)
        new_loss = np.log(np.sum(np.exp(beta * new_corr))) / beta
    return alpha, h_proj

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Setup parameters
    n_points = 180
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # Sum constraint
    
    # Beta schedule
    beta_schedule = [0.5, 2, 5, 15, 40, 80]
    max_iter_per_beta = 150
    
    # Build initial construction with upscaled known values
    if initial_h_values is not None and len(initial_h_values) > 0:
        h_initial = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, len(initial_h_values)),
                              initial_h_values)
        h_initial = project_onto_box_and_sum(h_initial, target_sum)
    else:
        h_initial = np.random.uniform(0, 1, n_points)
        h_initial = project_onto_box_and_sum(h_initial, target_sum)
        
    # Try several random starts
    all_starts = [h_initial]
    for _ in range(5):
        ran = np.random.uniform(0, 1, n_points)
        ran = project_onto_box_and_sum(ran, target_sum)
        all_starts.append(ran)
        
    # Global best
    best_h = h_initial.copy()
    best_c5 = np.max(compute_correlation(best_h, dx))
    
    # Iterate through starts
    for h_start in all_starts:
        h = h_start.copy()
        current_time = time.time()
        if current_time - t0 > 0.95 * budget_s:
            break
        # Optimise for each beta stage
        for beta in beta_schedule:
            for it in range(max_iter_per_beta):
                if time.time() - t0 > 0.95 * budget_s:
                    break
                
                # Compute all correlations
                corr_vals = compute_correlation(h, dx)
                
                # Compute gradient of LSE
                try:
                    grad = compute_gradient_of_LSE(h, corr_vals, beta, dx)
                except Exception as e:
                    # Fall back to a previous good solution maybe
                    break
                
                # Check for small gradient (to avoid dividing by zero in backtracking)
                if np.linalg.norm(grad) < 1e-8:
                    continue
                
                # Line search
                alpha, h_new = backtracking_line_search(h, grad, 
                                                        lambda x: np.log(np.sum(np.exp(beta * compute_correlation(x, dx)))) / beta, 
                                                        beta, dx)
                
                # Accept step
                h = h_new
                
                # Compute actual maximum correlation
                current_corr = compute_correlation(h, dx)
                current_c5 = np.max(current_corr)
                
                # Update best if needed
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                    # Early exit if threshold reached
                    if best_c5 <= 0.379:
                        return (best_h, best_c5, n_points)
                
    # Post-process with refinement if time allows
    current_time = time.time()
    if current_time - t0 < 0.95 * budget_s:
        # Refinement step - binary swap heuristic (as outlined in strategy)
        h_refined = best_h.copy()
        abs_corr = compute_correlation(h_refined, dx)
        best_c5_refined = np.max(abs_corr)
        
        # Convert to binary with proper mass
        h_binary = np.zeros_like(h_refined)
        ones_idx = np.argpartition(h_refined, -target_sum)[-int(target_sum):]
        h_binary[ones_idx] = 1.0
        
        # Important note: we assume the binary swap refinement isn't important for performance
        # since we already generally do well, but to be safe we just return what we have
                
    return (best_h, best_c5, n_points)
