import numpy as np
import time
import random

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    time_end = start + budget_s - 0.5
    
    n_points = 200
    best_h = None
    best_c5 = float('inf')
    dx = 2.0 / n_points

    # binary hill climbing with simulated annealing
    max_restarts = 30
    max_local_iters = 20000
    
    restart_count = 0
    
    while time.time() < time_end and restart_count < max_restarts:
        # Random binary solution with exactly n_points/2 ones
        h = np.zeros(n_points, dtype=np.float64)
        ones = np.random.choice(n_points, n_points//2, replace=False)
        h[ones] = 1.0

        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        # Local search (simulated annealing)
        for it in range(max_local_iters):
            if time.time() >= time_end:
                break
                
            # Choose two random indices with different values
            i = np.random.randint(n_points)
            j = np.random.randint(n_points)
            while h[i] == h[j]:
                i = np.random.randint(n_points)
                j = np.random.randint(n_points)
                
            # Swap values
            h[i], h[j] = h[j], h[i]
            
            new_c5 = compute_c5(h)
            d = new_c5 - cur_c5
            
            # Temperature schedule
            T = 0.05 * (1 - it / max_local_iters)
            T = max(1e-6, T)
            
            if d <= 0 or np.random.rand() < np.exp(-d / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                # Revert the swap
                h[i], h[j] = h[j], h[i]
        
        restart_count += 1

    # Final check
    if best_h is not None:
        final_c5 = compute_c5(best_h)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to random solution
        h = np.zeros(n_points, dtype=np.float64)
        ones = np.random.choice(n_points, n_points//2, replace=False)
        h[ones] = 1.0
        return (h.tolist(), compute_c5(h), n_points)
