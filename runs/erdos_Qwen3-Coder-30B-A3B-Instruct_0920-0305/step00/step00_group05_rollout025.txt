def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    correlation = np.correlate(h, 1 - h, mode="full")
    # The correlation is of length 2*n - 1, we only need the middle part [n-1:2*n-1]
    # corresponding to k from 0 to 2 in steps of dx

    return np.max(c5_values)
