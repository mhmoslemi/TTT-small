import numpy as np
import time
import random

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    time_end = start + budget_s - 0.5
    
    n_points = 200
    best_h = None
    best_c5 = float('inf')
    dx = 2.0 / n_points
    
    # Simple binary hill climbing approach with simulated annealing
    max_restarts = 50
    max_iter_per_restart = 20000
    
    for restart in range(max_restarts):
        if time.time() >= time_end:
            break
            
        # Initialize binary vector with exactly n_points//2 ones
        h = np.zeros(n_points, dtype=np.float64)
        ones = np.random.choice(n_points, n_points//2, replace=False)
        h[ones] = 1.0
        
        cur_c5 = compute_c5(h)
        
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            
        # Local annealing for this restart
        T0 = 0.05
        for it in range(max_iter_per_restart):
            if time.time() >= time_end:
                break
                
            # Choose two different random positions
            idx1 = np.random.randint(n_points)
            idx2 = np.random.randint(n_points)
            while h[idx1] == h[idx2]:
                idx1 = np.random.randint(n_points)
                idx2 = np.random.randint(n_points)
                
            # Propose swap
            h_new = h.copy()
            h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
            
            new_c5 = compute_c5(h_new)
            d = new_c5 - cur_c5
            
            # Simulated annealing acceptance probability
            T = max(1e-6, T0 * (1 - it / max_iter_per_restart))
            if d <= 0 or np.random.rand() < np.exp(-d / T):
                h = h_new
                cur_c5 = new_c5
                
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
