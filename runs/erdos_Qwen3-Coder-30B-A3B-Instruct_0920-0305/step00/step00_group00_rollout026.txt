def evaluate_erdos_solution(h):
    """
    Computes the C5 constant given a step function h.
    h: array of values in [0,1]
    """
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    conv = np.correlate(h, 1 - h, mode='full')
    # The maximum value is what we need (multiply by dx)
    return np.max(conv) * dx

def objective(h):
    """Objective function to minimize"""
    return evaluate_erdos_solution(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Parameters
    n_points = 45  # Use the given initial construction size
    dx = 2.0 / n_points
    
    # We'll use a bounds array for optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define constraint for integral = 1
    # We ensure that sum(h)*dx = 1, which is equivalent to sum(h) = n_points/2
    def constraint_eq(x):
        return np.sum(x) - n_points / 2.0
    
    # Define constraint for integral = 1, as a dict for scipy
    constr = {'type': 'eq', 'fun': constraint_eq}
    
    # Initialize with the given initial solution or use random
    h_init = initial_h_values.copy()
    if not any(h_init):
        h_init = np.random.rand(n_points) * 0.5  # Random between 0 and 0.5
        # Normalize to satisfy constraint
        h_init = h_init * (n_points / 2.0 / np.sum(h_init))
    
    # Use differential evolution which is good for discrete-like optimization
    best_h = h_init.copy()
    best_value = evaluate_erdos_solution(best_h)
    
    # Let's set the maximum number of iterations to a reasonable value
    # that will allow us to explore the space reasonably within budget
    maxiter = 2000
    max_evals = 100000
    popsize = min(max(5, n_points), 20)
    
    # Number of iterations based on budget
    # Use a rough estimate of iterations per second
    time_per_iter = 0.01  # estimate time per iteration to be safe
    
    # Optimization
    try:
        result = differential_evolution(
            func=objective,
            bounds=bounds,
            args=(),
            maxiter=maxiter,
            popsize=popsize,
            mutation=(0.5, 1.0),
            recombination=0.7,
            seed=seed,
            atol=1e-10,
            rtol=1e-10,
            constraint=constr,
            disp=False,
            callback=None,
            polish=True,
            init='latinhypercube'
        )
        
        h_opt = result.x
        final_value = result.fun
        if final_value < best_value:
            best_h = h_opt
            best_value = final_value
        
        # The optimization is complete; let's make sure the returned is correct
        h_final = best_h
        c5_bound = best_value
        n_points_final = n_points
        
        return (h_final, c5_bound, n_points_final)
        
    except Exception as e:
        # fallback to the original if anything goes wrong
        pass

    # Fallback to the initial solution with scaling if no improvement
    return (best_h, best_value, n_points)
