import numpy as np
import time
import random
import math
from scipy.optimize import minimize
from scipy.special import logsumexp

def project_onto_capped_simplex(y, target_sum):
    """Projects y onto the capped simplex {x: 0 <= x <= 1, sum(x) = target_sum}."""
    y = np.clip(y, 0, 1)
    sorted_y = np.sort(y)[::-1]
    cumsum = np.cumsum(sorted_y)
    temp = (cumsum - target_sum) / np.arange(1, len(y) + 1)
    theta = temp[np.argmax(sorted_y >= temp)]
    return np.clip(y - theta, 0, 1)

def evaluate(x):
    """Evaluates C5 = max_k ∫ h(x)(1 - h(x+k)) dx"""
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def gradient_of_max(x, beta):
    """Estimated gradient of log-sum-exp approximation to max_k O_k(x)"""
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full')
    weights = np.exp(beta * corr)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for k in range(-n + 1, n):
        if k == 0:
            continue
        if k > 0:
            grad[:n-k] += weights[k+n-1] * (1 - x[k:])
        else:
            grad[-k:] += weights[k+n-1] * (1 - x[:n+k])
    return grad

def swap_move(x, i, j):
    """Swap elements at indices i and j in binary vector x"""
    x = x.copy()
    x[i], x[j] = x[j], x[i]
    return x

def fractional_move(x, i, j, delta):
    """Move `delta` from x[j] to x[i] while preserving bounds and sum"""
    x = x.copy()
    new_i = max(0, min(1, x[i] + delta))
    new_j = max(0, min(1, x[j] - delta))
    delta_actual = new_i - x[i]
    x[i], x[j] = new_i, new_j
    return x

def worst_shift_info(x):
    """Returns the shift k* with highest overlap and the indices contributing to it"""
    n = len(x)
    corr = np.correlate(x, 1 - x, mode='full')
    max_k = np.argmax(corr)
    k = max_k - (n - 1)
    if k >= 0:
        # k is positive
        mask = (x[:n-k] == 1) & (x[k:] == 0)
    else:
        # k is negative
        mask = (x[-k:] == 1) & (x[:-k] == 0)
    offending = np.where(mask)[0]
    return k, offending

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    t_start = time.time()
    t_dead = t_start + budget_s - 5  # safety margin
    
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    candidate_n = [140, 200, 260]
    for n in candidate_n:
        if time.time() >= t_dead:
            break

        t_slice = (t_dead - time.time()) / (len(candidate_n) - list(candidate_n).index(n))
        t_end_slice = time.time() + t_slice
        
        # Phase 2.1: Binary simulated annealing
        # Initial binary vector
        x = np.zeros(n)
        ones_indices = random.sample(range(n), n // 2)
        x[ones_indices] = 1
        
        best_x = x.copy()
        best_c5_slice = evaluate(x)
        
        # Cooling parameters
        temp = 0.12
        temp_min = 1e-6
        cooling_rate = 0.9995
        iteration = 0
        max_iterations = 10000
        
        while iteration < max_iterations and time.time() < t_end_slice:
            # Get a pair of indices with opposite values
            i = random.randint(0, n - 1)
            j = random.randint(0, n - 1)
            while x[i] == x[j]:
                i = random.randint(0, n - 1)
                j = random.randint(0, n - 1)
            
            x_new = swap_move(x, i, j)
            c5_new = evaluate(x_new)
            
            delta = c5_new - best_c5_slice
            if delta <= 0 or random.random() < math.exp(-delta / temp):
                x = x_new
                if c5_new < best_c5_slice:
                    best_c5_slice = c5_new
                    best_x = x.copy()
            
            temp = max(temp_min, temp * cooling_rate)
            iteration += 1
            
        # Update even if best is not global best
        if best_c5_slice < best_c5:
            best_c5 = best_c5_slice
            best_h = best_x.copy()
            best_n = n
        
        # Phase 2.2: Directed local improvement
        if time.time() >= t_end_slice:
            continue
            
        x = best_x.copy()
        for _ in range(5000):
            if time.time() >= t_end_slice:
                break
            k, offending_indices = worst_shift_info(x)
            if len(offending_indices) == 0:
                break
            i = random.choice(offending_indices)
            # Choose j such that x[j] == 0 and j is not part of the pair (i, i+|k|)
            candidates = np.where(x == 0)[0]
            candidates = candidates[candidates != i]
            if k >= 0:
                candidates = candidates[candidates != (i + k)]
            else:
                candidates = candidates[candidates != (i - k)]
            if len(candidates) == 0:
                break
            j = random.choice(candidates)
            x_new = swap_move(x, i, j)
            c5_new = evaluate(x_new)
            if c5_new <= evaluate(x):
                x = x_new
        
        c5_current = evaluate(x)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = x.copy()
            best_n = n
            
        # Phase 2.3: Continuous refinement using projected gradient descent
        if time.time() >= t_end_slice:
            continue
            
        x = best_x.copy()
        x = project_onto_capped_simplex(x, n / 2)
        beta = 5
        steps = 0
        max_steps = 5000
        alpha = 0.02
        
        while steps < max_steps and time.time() < t_end_slice:
            grad = gradient_of_max(x, beta)
            x_new = x - alpha * grad
            x_new = project_onto_capped_simplex(x_new, n / 2)
            c5_new = evaluate(x_new)
            
            if c5_new <= evaluate(x):
                x = x_new
                steps += 1
                beta *= 1.02
            
        if evaluate(x) < best_c5:
            best_c5 = evaluate(x)
            best_h = x.copy()
            best_n = n

    return (best_h.tolist(), best_c5, best_n)
