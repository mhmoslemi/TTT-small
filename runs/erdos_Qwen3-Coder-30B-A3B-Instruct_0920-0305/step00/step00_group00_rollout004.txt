def softmax(x):
    """Stable softmax computation"""
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    return exp_x / np.sum(exp_x)

def project_to_feasible(h, target_sum):
    """Project h onto the set {0 ≤ h[i] ≤ 1, sum(h) = target_sum}"""
    # Use binary search to find the right lambda
    h = np.array(h)
    low, high = -100, 100
    eps = 1e-10
    while high - low > eps:
        mid = (low + high) / 2
        x = np.clip(h + mid, 0, 1)
        sum_x = np.sum(x)
        if sum_x < target_sum:
            high = mid
        else:
            low = mid
    x = np.clip(h + low, 0, 1)
    
    # Adjust to ensure exact sum
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-10:
        # Distribute the difference among non-bound elements
        idxs = np.where((x > 0) & (x < 1))[0]
        if len(idxs) > 0:
            incr = diff / len(idxs)
            x[idxs] = np.clip(x[idxs] + incr, 0, 1)
    
    return x

def compute_correlations(h, dx):
    """Compute all correlations c_k = dx * sum(h[i] * (1 - h[i+k]))"""
    p = 1 - h
    correlations = np.correlate(h, p, mode='full')
    correlations = correlations * dx
    return correlations

def gradient_of_correlation(h, k, dx):
    """Compute gradient of correlation with lag k w.r.t. h"""
    n = len(h)
    grad = np.zeros_like(h)
    # Add term for h[i] * (1 - h[i+k])
    if k >= 0:
        grad[k:] += dx * (1 - h[k:])  # h[i+k] contributes when i+k < n
        # Subtract term for h[i] * h[i-k]
        if k > 0:
            grad[:-k] -= dx * h[:-k]   # -h[i-k] when i-k >= 0
    else:
        # k < 0 case
        grad[:k] += dx * (1 - h[:k])
        if k < 0:
            grad[-k:] -= dx * h[-k:]
    return grad

def log_sum_exp_weights(c, beta):
    """Compute weights for log-sum-exp approximation"""
    exp_c = np.exp(beta * c)
    return exp_c / np.sum(exp_c)

def compute_smooth_max_loss(h, dx, beta):
    """Compute smooth approximation of max over all shifts"""
    c = compute_correlations(h, dx)
    L = np.log(np.sum(np.exp(beta * c))) / beta
    return L

def compute_gradient_smooth_max(h, dx, beta):
    """Compute gradient of smooth max using weights"""
    c = compute_correlations(h, dx)
    weights = log_sum_exp_weights(c, beta)
    grad = np.zeros_like(h)
    n = len(h)
    for k in range(-n+1, n):
        if k >= 0:
            gg = np.zeros(n)
            if k < n:
                gg[k:] = dx * (1 - h[k:])  # Add
                if k > 0:
                    gg[:-k] -= dx * h[:-k]  # Subtract
            else:
                continue
        else:
            gg = np.zeros(n)
            if -k < n:
                gg[:k] = dx * (1 - h[:k])
                gg[-k:] -= dx * h[-k:]
            else:
                continue
        grad += weights[k + n - 1] * gg
    return grad

def gradient_descent_step(h, dx, beta, alpha=0.2):
    """Single gradient descent step with backtracking line search"""
    grad = compute_gradient_smooth_max(h, dx, beta)
    
    # Initial loss
    c = compute_correlations(h, dx)
    old_loss = np.max(c)
    
    # Backtracking line search
    while True:
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, len(h) / 2)
        c_new = compute_correlations(h_new, dx)
        new_loss = np.max(c_new)
        
        if new_loss <= old_loss:
            break
        alpha *= 0.5
        if alpha < 1e-10:
            break
    return h_new
