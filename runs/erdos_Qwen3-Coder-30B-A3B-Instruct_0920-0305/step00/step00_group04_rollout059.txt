import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 60
    dx = 2.0 / n_points
    target_c5 = 0.3807
    
    # Symmetry breaking and time limit
    time_limit = 0.9 * budget_s
    
    # Create variables
    b = cp.Variable(n_points, boolean=True)
    
    # Ensure exactly half are 1s
    constraints = [cp.sum(b) == n_points // 2]
    
    # Build auxiliary variables and constraints for each lag
    # We'll use the fact that lag k and -k give same max
    all_lags = list(range(1, n_points))  # Ignore 0
    w_vars = {}  # Dictionary to hold auxiliary variables
    
    # For each k, there are (n - k) pairs
    for k in all_lags:
        w_vars[k] = cp.Variable(n_points - k, boolean=True)
        # w_vars[-k] = cp.Variable(n_points - k, boolean=True)  # Will reuse
    
    # Add constraints for each lag
    for k in all_lags:
        w = w_vars[k]
        # w[i] <= b[i]
        # w[i] <= b[i + k]
        # w[i] >= b[i] + b[i + k] - 1
        for i in range(n_points - k):
            constraints.extend([
                w[i] <= b[i],
                w[i] <= b[i + k],
                w[i] >= b[i] + b[i + k] - 1
            ])
    
    # Define N_k for each lag
    N_vars = {}
    for k in all_lags:
        # N_k = sum(b) - sum(w_k)
        w = w_vars[k]
        N_vars[k] = cp.sum(b[:n_points - k]) - cp.sum(w)
    
    # Introduce scalar variable t and constrain it
    t = cp.Variable()
    for k in all_lags:
        constraints.append(t >= N_vars[k])
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.CBC, maxSeconds=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_EITHER]:
            b_solution = b.value
            t_solution = t.value
            if t_solution is not None:
                c5_bound = t_solution * dx
                h_values = b_solution.astype(float)
                return (h_values, c5_bound, n_points)
        else:
            # Fallback to solver's best feasible solution
            # Re-run with limited time, extract infeasible solution
            pass
    except Exception:

    # Fallback plan: use local search
    # Initialize h from the binary solution (or from initial_h_values if available)
    try:
        h_values = b_solution.astype(float)
        h_values = h_values / np.sum(h_values) * 1.0  # ensure integral = 1
    except:

            h_values = np.array(initial_h_values)
        except:
            h_values = np.ones(n_points) * 0.5  # fallback to central
        # Normalize
        h_values = h_values / np.sum(h_values) * 1.0
    
    # Local search for small improvements (very limited)
    def compute_C5(h):
        h = np.asarray(h)
        h_conv = np.correlate(h, 1-h, mode="full")
        return np.max(h_conv) * dx
    
    best_h = h_values.copy()
    best_c5 = compute_C5(best_h)
    
    end_time = time.time()
    remaining_time = budget_s - (end_time - start_time)
    
    # Quick local search on some swaps (this is still fast)
    max_swaps = 5000
    for _ in range(max_swaps):
        if time.time() - start_time > budget_s * 0.95:
            break
        i, j = np.random.randint(0, n_points, 2)
        if i == j:
            continue
        # Clone but only swap one element (keep 1s and 0s)
        h_try = best_h.copy()
        if h_try[i] > h_try[j]:  # make sure h[i] gets smaller to allow 0s
            h_try[i], h_try[j] = h_try[j], h_try[i]
        c5_try = compute_C5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
    
    return (best_h, best_c5, n_points)
