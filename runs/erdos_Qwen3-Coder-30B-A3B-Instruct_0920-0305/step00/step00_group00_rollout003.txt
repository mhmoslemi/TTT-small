import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize
import time

def project_to_constraints(v, lower=0.0, upper=1.0, target_sum=None):
    if target_sum is None:
        target_sum = len(v) / 2.0
    # binary search for lambda
    lo = -np.max(v) - 1.0  # ensures v+lo <= lower
    hi = upper - np.min(v) + 1.0  # ensures v+hi >= upper
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, lower, upper)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo+hi)/2.0, lower, upper)
    return x

def compute_gradient(h, w, dx):
    n = len(h)
    grad = np.zeros_like(h)
    offset = n - 1
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        shift = idx - offset
        if shift >= 0:
            # first term: j from 0 to n-shift-1: grad[j] += wk * (1 - h[j+shift])
            grad[:n-shift] += wk * (1 - h[shift:])
            # second term: j from shift to n-1: grad[j] -= wk * h[j-shift]
            grad[shift:] -= wk * h[:n-shift]
        else:
            s = -shift
            # first term: j from s to n-1: grad[j] += wk * (1 - h[j+s])
            grad[s:] += wk * (1 - h[:n-s])
            # second term: j from 0 to n-s-1: grad[j] -= wk * h[j+s]
            grad[:n-s] -= wk * h[s:]
    return grad * dx

def loss_and_grad(h, beta, dx):
    p = 1 - h
    c = np.correlate(h, p, mode='full') * dx
    scaled = beta * c
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w_sum = np.sum(w)
    w = w / w_sum
    loss = (max_scaled + np.log(w_sum)) / beta

    grad = compute_gradient(h, w, dx)
    return loss, grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = np.inf
    n_points = 140
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Initialize with upscale from initial_h_values if available
    initial_h = None
    if 'initial_h_values' in globals():
        old_h = np.array(initial_h_values)
        old_n = len(old_h)
        old_x = np.linspace(0, 2, old_n, endpoint=False) + 1.0 / old_n
        new_x = np.linspace(0, 2, n_points, endpoint=False)
        initial_h = np.interp(new_x, old_x, old_h)
        initial_h = project_to_constraints(initial_h, target_sum=target_sum)
    else:
        initial_h = np.random.random(n_points)
        initial_h = project_to_constraints(initial_h, target_sum=target_sum)

    restarts = 15
    for restart in range(restarts):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
            
        h = initial_h.copy()
        beta_schedule = [1, 5, 15, 30]
        
        for beta in beta_schedule:
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.95:
                break
                
            # Define objective
            def fun(x):
                loss, grad = loss_and_grad(x, beta, dx)
                return loss, grad
                
            # Use SLSQP
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
            bounds = [(0.0, 1.0)] * n_points
            
            res = minimize(fun, h, method='SLSQP', jac=True, bounds=bounds,
                           constraints=cons, options={'maxiter': 200, 'ftol':1e-9, 'disp':False})
            
            h = res.x
            
        # Evaluate actual C5
        c = np.correlate(h, 1-h, mode='full') * dx
        actual_c5 = c.max()
        
        if actual_c5 < best_c5:
            best_c5 = actual_c5
            best_h = h.copy()
            
        # Terminate early if we're close to target
        if best_c5 <= 0.3795:
            break

    if best_h is None:
        best_h = initial_h.copy()
        
    # Final check
    c = np.correlate(best_h, 1-best_h, mode='full') * dx
    final_c5 = c.max()

    return (best_h.tolist(), final_c5, n_points)
