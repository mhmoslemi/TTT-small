```python
import numpy as np
import cvxpy as cp
import time
import math

def full_overlap(x):
    return np.correlate(x, 1 - x, mode="full")

def worst_shift(x):
    overlaps = full_overlap(x)
    k = np.argmax(overlaps)
    max_overlap = overlaps[k]
    return (k, max_overlap)

def build_mip(N, active_shifts):
    x = cp.Variable(N, integer=True)
    t = cp.Variable(integer=True)
    w = {}
    
    # Symmetry breaking
    constraints = [
        x[0] == 1,
        cp.sum(x) == N // 2,
        x >= 0,
        x <= 1
    ]
    
    # Create auxiliary variables and constraints for active shifts
    for k in active_shifts:
        w[k] = cp.Variable(N - k, integer=True)
        for i in range(N - k):
            # wij,k <= xi
            constraints.append(w[k][i] <= x[i])
            # wij,k <= x_{i+k}
            constraints.append(w[k][i] <= x[i + k])
            # wij,k >= xi + x_{i+k} - 1
            constraints.append(w[k][i] >= x[i] + x[i + k] - 1)
    
    # Add overlap constraints
    for k in active_shifts:
        left_sum = cp.sum(x[:N-k])
        right_sum = cp.sum(w[k])
        constraints.append(left_sum - right_sum <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    vars_dict = {'x': x, 't': t, 'w': w}
    return prob, vars_dict

def solve_mip(prob, vars_dict, time_limit):
    prob.solve(solver=cp.CBC, max_iters=10000, TimeLimit=time_limit, verbose=False)
    x_sol = vars_dict['x'].value
    t_sol = vars_dict['t'].value
    
    if x_sol is None:
        return None, None
    x_sol = np.round(x_sol).astype(int)
    return x_sol, t_sol

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_bound = float('inf')
    best_h = None
    best_n = None
    
    # Try different values of N
    candidates = [120, 140, 160]
    
    for N in candidates:
        if time.time() - start_time > budget_s - 10:
            break
            
        active_shifts = {1, 2, 3, 4, 5}
        prob, vars_dict = build_mip(N, active_shifts)
        time_left = budget_s - (time.time() - start_time)
        
        while time_left > 5:
            x_sol, t_sol = solve_mip(prob, vars_dict, min(30, time_left))
            if x_sol is None:
                break
            
            # Record current best
            if t_sol < best_bound:
                best_bound = t_sol
                best_h = x_sol.copy()
                best_n = N
            
            # Check if current solution is optimal for all shifts
            overlaps = full_overlap(x_sol)
            max_k = np.argmax(overlaps)
            max_ov = overlaps[max_k]
            
            if max_ov <= t_sol:
                break  # Found feasible solution for all shifts
            
            # Add the worst shift
            active_shifts.add(max_k)
            # Rebuild MIP with new shift
            prob, vars_dict = build_mip(N, active_shifts)
            time_left = budget_s - (time.time() - start_time)
            
        # Binary search to tighten
        if best_h is not None:
            b = best_bound - 1
            while b >= 1 and time.time() - start_time < budget_s - 10:
                # Feasibility check
                prob, vars_dict = build_mip(N, active_shifts)
                prob.constraints[-1].rhs = b
                x_sol_new, t_sol_new = solve_mip(prob, vars_dict, 3)
                if x_sol_new is not None:
                    best_bound = b
                    best_h = x_sol_new
                else:
                    break
                b -= 1
    
    if best_h is None:
        # Fallback to a simple construction
        h = np.ones(70) * 0.5
        h[0:35] = 1.0
        h[35:70] = 0.0
        c5_est = max(np.correlate(h, 1-h, mode="full")) / 70
        return (h.tolist(), c5_est, 70)
        
    dx = 2.0 / best_n
    c5 = best_bound * dx
    return (best_h.tolist(), c5, best_n)
```
