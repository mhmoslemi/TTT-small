import numpy as np
import time
import math
import random
from scipy.optimize import minimize
from scipy.special import logsumexp

# Helper functions defined at module level

def evaluate(x):
    """Evaluate C5 for a step function represented as vector x"""
    dx = 2.0 / len(x)
    # Compute full correlation of x with (1-x)
    corr = np.correlate(x, 1 - x, mode='full')
    # The maximum correlation is the largest value in corr array
    max_corr = np.max(corr)
    return max_corr * dx

def project_onto_capped_simplex(y, target_sum):
    """Project y onto [0,1]^N with sum = target_sum, using Duchi et al. method"""
    y = np.clip(y, 0, 1)
    # Sort in descending order
    sorted_y = np.sort(y)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_y)
    # Find the index where lambda should be
    n = len(sorted_y)
    idx = np.arange(1, n+1)
    lambda_val = (cumsum - target_sum) / idx
    # Find the largest index where lambda > sorted_y
    valid_indices = idx - 1
    valid_mask = lambda_val > sorted_y
    if np.any(valid_mask):
        k = np.max(valid_indices[valid_mask])
    else:
        k = 0
    # Compute lambda
    lambda_star = (cumsum[k] - target_sum) / (k + 1)
    # Apply lambda
    z = np.clip(y - lambda_star, 0, 1)
    return z

def gradient_of_max(x, beta=10.0):
    """Compute gradient of smooth approximation to max_k O_k(x)"""
    N = len(x)
    dx = 2.0 / N
    # Compute all O_k values for shift k = -N+1 to N-1
    corr_x = np.correlate(x, 1 - x, mode='full')
    corr_x = corr_x[N-1:] # Take the right half to get shifts 0 to N-1 and -N+1 to -1
    # For shift k=-N+1 to -1 (negative shifts), consider the part of the correlation
    # and for k=0 to N-1 (non-negative shifts), consider the complete part
    # Correct approach: each shift gives O_k = sum_{i=0}^{N-1-|k|} x_i*(1-x_{i+|k|}) for k non-negative
    # and O_{-k} = sum_{i=0}^{N-1-|k|} x_{i+k}*(1-x_{i}) for k positive
    # There's a simpler and reliable method using np.correlate properly
    all_O = []
    for k in range(-N+1, N):
        if k >= 0:
            # This will compute the standard correlation but we have to slice it properly
            # O_k = sum_{i=0}^{N-1-k} x[i]*y[i+k]
            # where y[i] = 1 - x[i]
            # So we do it via np.correlate:
            corr_vals = np.correlate(x, 1 - x, mode='full')
            shift_idx = k + N - 1  # index in corr array for correlation at shift k
            if shift_idx < len(corr_vals):
                all_O.append(corr_vals[shift_idx])
            else:
                all_O.append(0)
        else:
            # k < 0
            corr_vals = np.correlate(x, 1 - x, mode='full')
            shift_idx = k + N - 1
            if shift_idx >= 0 and shift_idx < len(corr_vals):
                all_O.append(corr_vals[shift_idx])
            else:
                all_O.append(0)
    # Now compute the smooth max and its gradient  
    O_array = np.array(all_O)
    w = np.exp(beta * O_array)
    w = w / np.sum(w)
    # Now compute gradient of O_k with respect to x_i
    # For k (positive shift) - same as for backward shift by |k|
    # For -k shift (negative) - reverse the indices  
    # Instead, build the gradient using for-loop
    grad = np.zeros(N)
    # For each shift, compute gradients
    for k in range(-N+1, N):
        # k = shift. In full correlation, O_k = \sum_{i=0}^{N-1-k} x[i] ** (1 - x[i+k])
        # Gradient with respect to x_i = (1 - x[i+k]) if i+k <= N-1 and i <= N-1-k and i >=0  
        # = - x[i-k] if i >= k and i <= N-1 and i-k >= 0 and i-k <= N-1  
        if k >= 0:
            # when k is positive, i starts from 0 to N-1-k
            for i in range(N-k):
                # Add gradient contribution for term x[i](1 - x[i+k])
                # term O_k = x[i] * (1 - x[i+k])
                # d/dx_i  = (1 - x[i+k])
                if i+k < N:
                    grad[i] += w[k+N-1] * (1 - x[i+k])
                # d/dx[i+k] = -x[i]  
                if i+k < N and i+k >= 0:
                    grad[i+k] += w[k+N-1] * (-x[i])
        else:
            # k < 0, shift: i starts from -k to N-1
            for i in range(-k, N):
                # For this, O_k = x[i+k] * (1 - x[i])
                # d/dx[i+k] = 1 - x[i]
                # d/dx[i] = -x[i+k]
                grad[i+k] += w[k+N-1] * (1 - x[i])
                grad[i] -= w[k+N-1] * x[i+k]
    return grad

def swap_move(x, i, j):
    """Swap entries i and j in x (binary case)"""
    new_x = x.copy()
    new_x[i], new_x[j] = new_x[j], new_x[i]
    return new_x

def fractional_move(x, i, j, delta):
    """Perform fractional move: add delta to x[i], subtract from x[j]"""
    new_x = x.copy()
    new_x[i] = new_x[i] + delta
    new_x[j] = new_x[j] - delta
    # Ensure bounds:
    new_x[i] = np.clip(new_x[i], 0, 1)
    new_x[j] = np.clip(new_x[j], 0, 1)
    return new_x

def worst_shift_info(x):
    """Return shift k* and list of indices that are part of the worst pair"""
    corr = np.correlate(x, 1 - x, mode='full')
    k_max = np.argmax(corr) - len(x) + 1  # shift index from -N+1 to N-1
    max_val = corr[np.argmax(corr)]
    # Identify "bad" pairs, i.e. indices where x_i = 1 and x_{i+|k_max|} = 0 
    # Search for pairs causing maximum overlap.
    # For simplicity, use direct computation of shift values
    N = len(x)
    # Define shifts properly for negative and positive ones
    # For shift k (e.g. 1): sum over i where x_i*x_{i+1}  
    # For negative shift -1: sum over i where x_{i-1}*x_i
    # Let's compute full correlation along with positions
    # The index in the full correlation array for shift k is k + N - 1
    true_k = k_max
    # Identify indices that contribute to this maximum 
    if true_k >= 0:
        # O_k = sum_{i=0}^{N-1-k} x[i] * (1 - x[i+k])
        bad_pairs = []
        for i in range(N - true_k):
            if x[i] == 1 and x[i + true_k] == 0:
                bad_pairs.append(i)
    else:
        # Negative shift, required k = -|k|
        bad_pairs = []
        for i in range(N + true_k):  # N - (-true_k) = N + true_k
            if x[i + true_k] == 1 and x[i] == 0:
                bad_pairs.append(i)
    return true_k, bad_pairs

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    t_start = time.time()
    t_dead = t_start + budget_s - 5  # safety margin:
    
    # Candidate N values
    candidate_N = [140, 200, 260]
    
    best_h = None
    best_c5 = float('inf')
    best_N = 0
    
    # Loop over different N values
    for N in candidate_N:
        if time.time() >= t_dead:
            break
            
        # allocate time slice per N:
        time_slice = (t_dead - time.time()) / (len(candidate_N) - candidate_N.index(N))
        t_slice_end = time.time() + time_slice
        
        # Phase 2.1: Coarse binary simulated annealing
        best_binary = None
        best_c5_binary = float('inf')
        
        # Start with a random binary solution
        x = np.zeros(N)
        ones = random.sample(range(N), N // 2)
        x[ones] = 1.0
        
        T0 = 0.12
        T_min = 1e-6
        T = T0
        iter_count = 0
        n_iter_SA = 5000  # Iteration bound per run
        # Initial score
        c5 = evaluate(x)
        if c5 < best_c5_binary:
            best_c5_binary = c5 
            best_binary = x.copy()
            
        # SA main loop
        while time.time() < t_slice_end and iter_count < n_iter_SA:
            # Get two random indices with opposite values
            while True:
                i = random.randint(0, N - 1)
                j = random.randint(0, N - 1)
                if i != j and x[i] != x[j]:
                    break
                    
            # Make move
            x_new = swap_move(x, i, j)
            c5_new = evaluate(x_new)
            
            # Accept according to SA rules
            delta = c5_new - c5
            if delta <= 0 or random.random() < math.exp(-delta / T):
                x = x_new
                c5 = c5_new
                if c5 < best_c5_binary:
                    best_c5_binary = c5
                    best_binary = x.copy()
                    
            # Decay temp
            T = max(T_min, T0 * (1 - iter_count / n_iter_SA))
            iter_count += 1
            
        if time.time() >= t_slice_end:
            break
        
        # Phase 2.2: Directed local refinement from the best binary in this N
        if best_binary is not None:
            x_refined = best_binary.copy()
            c5_refined = evaluate(x_refined)
            
            # Run several iterations of directed improvement:
            for it in range(2000):
                if time.time() >= t_slice_end:
                    break
                k_max, bad_indices = worst_shift_info(x_refined)
                if len(bad_indices) == 0:
                    break
                # Pick a random ordinate and a random zero not in neighbor
                i = random.choice(bad_indices)
                j_index = i + abs(k_max)
                # Avoid swap with point that is also part of contribs
                candidates = [idx for idx in range(N) if x_refined[idx] == 0 and idx != i and idx != j_index]
                if not candidates:
                    break
                j = random.choice(candidates)
                # Make the swap or revert if value doesn't improve
                x_new = swap_move(x_refined, i, j)
                c5_new = evaluate(x_new)
                if c5_new <= c5_refined:
                    x_refined = x_new
                    c5_refined = c5_new
                if c5_refined < best_c5_binary:
                    best_c5_binary = c5_refined
                    best_binary = x_refined.copy()
                    
        if time.time() >= t_slice_end:
            break
                
        # Phase 2.3: Continuous refinement
        if best_binary is not None and best_binary is not None and time.time() < t_slice_end - 1:
            x_cont = best_binary.copy()
            # Set beta to a moderate value first
            beta = 5.0
            c5_cont = evaluate(x_cont)
            
            # Gradient descent with log-sum-exp smooth max
            iter_cnt = 0
            max_iter = 2000  # Limit iterations for continuous refinement
            
            while iter_cnt < max_iter and time.time() < t_slice_end - 1:
                grad = gradient_of_max(x_cont, beta=beta)
                # Compute delta step
                step_size = 0.01
                x_candidate = x_cont - step_size * grad
                # Project back to feasible set
                x_projected = project_onto_capped_simplex(x_candidate, N / 2)
                c5_candidate = evaluate(x_projected)
                
                # Sometimes accept step anyway if no strict improvement
                if c5_candidate <= c5_cont + 1e-10:
                    x_cont = x_projected
                    c5_cont = c5_candidate
                elif random.random() < 0.05 :
                    # Occasionally make a (worse) step to escape local optima
                    x_cont = x_projected
                    c5_cont = c5_candidate
                # Update smoothness parameter
                beta = min(100, beta * 1.02)
                iter_cnt += 1
                
            if c5_cont < best_c5_binary:
                best_c5_binary = c5_cont
                best_binary = x_cont.copy()
                
        # Update global best
        if best_c5_binary < best_c5:
            best_c5 = best_c5_binary
            best_h = best_binary.copy()
            best_N = N
            
    # Wrap-up: make sure it's the final evaluation
    final_c5 = evaluate(best_h)
    
    return (best_h.tolist(), final_c5, best_N)
