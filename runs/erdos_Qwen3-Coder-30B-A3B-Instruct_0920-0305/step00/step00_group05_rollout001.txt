import numpy as np
import time
import random
import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    time_end = start + budget_s - 0.5
    
    n_points = 200
    dx = 2.0 / n_points
    
    best_h = None
    best_c5 = float('inf')
    
    # Main loop with restarts
    restart_count = 0
    while time.time() < time_end:
        # Generate new random binary vector with exactly n/2 ones
        h = np.zeros(n_points, dtype=np.float64)
        ones = np.random.choice(n_points, n_points//2, replace=False)
        h[ones] = 1.0
        
        # Local optimization for this restart
        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        
        # Simulated annealing local search
        max_local_iters = 30000
        T0 = 0.05
        local_iter = 0
        
        while time.time() < time_end and local_iter < max_local_iters:
            # Choose two random indices with different values
            i = np.random.randint(n_points)
            j = np.random.randint(n_points)
            while h[i] == h[j]:
                i = np.random.randint(n_points)
                j = np.random.randint(n_points)
            
            # Propose swap
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
            new_c5 = compute_c5(h_new)
            
            # Accept or reject
            delta = new_c5 - cur_c5
            T = T0 * (1 - local_iter / max_local_iters)
            if delta <= 0 or np.random.rand() < math.exp(-delta / (T + 1e-12)):
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            
            local_iter += 1
            restart_count += 1
            
            # Break early if we're very close to target and time is running low
            if best_c5 <= 0.3795 and time.time() > time_end - 10:
                break
                
    return (best_h.tolist(), best_c5, n_points)
