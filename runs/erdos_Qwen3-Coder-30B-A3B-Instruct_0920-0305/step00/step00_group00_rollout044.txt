def project_onto_feasible(h, n_half):
    """Project h onto the feasible set: 0 <= h[i] <= 1, sum(h) = n_half."""
    # Water-filling approach to find lambda
    h = np.array(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    # Find the right threshold
    n = len(h)
    lambda_star = None
    for i in range(n):
        if cumsum[i] <= n_half:
            continue
        else:
            # Lambda such that sum of clipped values = n_half
            lambda_star = (cumsum[i] - n_half) / (i+1)
            # Adjust to make sum exactly n_half, handle numerical imprecision
            break
    
    if lambda_star is None:
        # All values are too small - must sum up
        lambda_star = (np.sum(h) - n_half) / n

    # Generate h_proj = clip(h - lambda_star, 0, 1)
    h_proj = np.clip(h - lambda_star, 0, 1)
    # Ensure sum is exactly n_half
    current_sum = np.sum(h_proj)
    diff = n_half - current_sum
    # Adjust smallest elements by small increments
    if abs(diff) > 1e-10:
        # get indices that are strictly less than 1
        idx_less_one = np.where(h_proj < 1)[0]
        if len(idx_less_one) != 0:
            # distribute diff evenly to those
            delta = diff / len(idx_less_one)
            for i in idx_less_one:
                h_proj[i] = min(1.0, h_proj[i] + delta)
    
    return h_proj

def correlate_h_complement(h, dx):
    """Compute c_k = dx * Σ_i h[i] * (1 - h[i+k]) for all valid k."""
    n = len(h)
    # Compute 1 - h
    p = 1 - h
    
    # corr[k] = Σ_i h[i] * p[i+k]
    # We use numpy.correlate in full mode
    corr = np.correlate(h, p, mode='full') * dx
    return corr

def compute_C5(h, dx):
    """Compute the C5 value for the given h."""
    corr = correlate_h_complement(h, dx)
    return np.max(corr)

def gradient_c_k(h, k, dx):
    """Compute ∇c_k for a fixed shift k."""
    n = len(h)
    g = np.zeros(n)
    # Add terms for +h[i] * (1 - h[i+k])
    if k >= 0:
        g[k:] += dx * (1 - h[:-k] if k > 0 else dx * (1 - h))
    else:
        g[:k] += dx * (1 - h[-k:])
    
    # Subtract terms for h[i] * h[i+k] 
    if k >= 0:
        g[:-k] -= dx * h[k:] if k > 0 else dx * h
    else:
        g[-k:] -= dx * h[:k] if k < 0 else dx * h
    
    return g

def gradient_LSE(h, c_vals, beta, dx):
    """Compute ∇L_β where L_β = (1/β) log Σ_k exp(β c_k)"""
    w = softmax(beta * c_vals)
    # g = Σ_k w_k * ∇c_k 
    n = len(h)
    g = np.zeros(n)
    # Use vectorized computation for each k
    for k in range(len(c_vals) - n + 1):
        g += w[k] * gradient_c_k(h, k - (len(c_vals) - n), dx)
    return g
