import numpy as np

def project_to_feasible(h, target_sum):
    """Project h onto sum = target_sum and bounds [0,1]"""
    # Ensure bounds
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Simple projection: adjust until sum matches
    # More robust method: iterative scaling
    # Actually, for [0,1] and fixed sum, we can use the simple algorithm
    # that finds root of the bisection projection
    
    # Find the lambda that makes sum = target_sum when clipping
    def sum_at_lambda(lam):
        return np.sum(np.clip(h - lam, 0, 1))
    
    # Binary search for lambda
    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high) / 2
        if sum_at_lambda(mid) > target_sum:
            low = mid
        else:
            high = mid
    
    return np.clip(h - low, 0, 1)

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Use np.correlate to compute overlap integral
    corr = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to normalize
    corr *= dx
    # Take max over all shifts
    max_overlap = np.max(corr)
    return max_overlap

        delta_min = max(-h_i, h_j - 1.0)
        delta_max = min(1.0 - h_i, h_j)
        
        if abs(delta_min - delta_max) < 1e-10:
            continue

        delta = np.random.uniform(delta_min, delta_max)

        new_h = h_values.copy()

        new_h[j] = h_j - delta
