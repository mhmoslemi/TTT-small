def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant for a given step function h"""
    n_points = len(h)

    # Compute the overlap integral for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap integral is what we want to minimize
    max_overlap = np.max(conv) * dx
    return max_overlap
