import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize
import time

def project_to_constraints(v, lower=0.0, upper=1.0, target_sum=None):
    if target_sum is None:
        target_sum = len(v) / 2.0
    # binary search for lambda
    lo = -np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, lower, upper)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo+hi)/2.0, lower, upper)
    return x

def compute_gradient(h, w, dx):
    n = len(h)
    grad = np.zeros_like(h)
    offset = n - 1
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        shift = idx - offset
        if shift >= 0:
            # first term: ∂C_k/∂h_j = dx * (1 - h_{j+k})
            idx1 = slice(0, n - shift)
            grad[idx1] += wk * (1 - h[idx1 + shift])
            # second term: ∂C_k/∂h_j = dx * (- h_{j-k})
            idx2 = slice(shift, n)
            grad[idx2] -= wk * h[idx2 - shift]
        else:
            s = -shift  # shift is negative
            # first term: ∂C_k/∂h_j = dx * (1 - h_{j+k})
            idx1 = slice(s, n)
            grad[idx1] += wk * (1 - h[idx1 + shift])
            # second term: ∂C_k/∂h_j = dx * (- h_{j-k})
            idx2 = slice(0, n - s)
            grad[idx2] -= wk * h[idx2 - shift]
    return grad * dx

def loss_and_grad(h, beta, dx):
    p = 1 - h
    c = np.correlate(h, p, mode='full') * dx
    scaled = beta * c
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w_sum = np.sum(w)
    w = w / w_sum
    loss = (max_scaled + np.log(w_sum)) / beta
    grad = compute_gradient(h, w, dx)
    return loss, grad

def run(seed=42, budget_s=600, **kwargs):
    # Set seed and parameters
    np.random.seed(seed)
    start_time = time.time()
    n_points = 140
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_c5 = np.inf
    best_h = None
    
    # Use initial values if available
    try:
        import initial_h_values
        old_h = initial_h_values
        old_n = len(old_h)
        old_x = np.linspace(0, 2, old_n, endpoint=False) + 1.0 / old_n
        new_x = np.linspace(0, 2, n_points, endpoint=False)
        # Linear interpolation
        init_h = np.interp(new_x, old_x, old_h)
        init_h = project_to_constraints(init_h)
    except ImportError:
        # Generate initial h with target sum
        init_h = np.random.random(n_points)
        init_h = project_to_constraints(init_h)
    
    max_restarts = 10
    resample_factor = 0.1
    
    for restart in range(max_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
        # Initialize h
        rng = np.random.default_rng(seed + restart)
        h = init_h.copy()
        # Add noise to break symmetry
        if restart > 0:
            noise = rng.normal(0, 0.01, n_points)
            h = project_to_constraints(h + noise)
        
        # Optimize with different beta values
        beta_schedule = [1.0, 5.0, 10.0, 20.0]
        for beta in beta_schedule:
            # Use SLSQP optimizer
            try:
                def func(x):
                    loss, grad = loss_and_grad(x, beta, dx)
                    return loss, grad
                
                cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
                bounds = [(0.0, 1.0)] * n_points
                res = minimize(func, h, method='SLSQP', jac=True, bounds=bounds,
                               constraints=cons, options={'maxiter': 200, 'ftol': 1e-9, 'disp': False})
                if res.success:
                    h = res.x
                else:
                    # Fallback: gradient descent within bounds
                    for _ in range(200):
                        if time.time() - start_time > budget_s * 0.95:
                            break
                        loss, grad = loss_and_grad(h, beta, dx)
                        step = 0.2 / (1.0 + beta)
                        h = project_to_constraints(h - step * grad)
            except:
                # Fallback to gradient descent if SLSQP fails
                for _ in range(200):
                    if time.time() - start_time > budget_s * 0.95:
                        break
                    loss, grad = loss_and_grad(h, beta, dx)
                    step = 0.2 / (1.0 + beta)
                    h = project_to_constraints(h - step * grad)
            
            # Check C5 value
            if time.time() - start_time > budget_s * 0.95:
                break

        # Evaluate final C5
        try:
            import evaluate_erdos_solution
            actual_c5 = evaluate_erdos_solution(h)
            if actual_c5 < best_c5:
                best_c5 = actual_c5
                best_h = h.copy()
        except:
            # Compute manually if evaluate_erdos_solution not available
            p = 1 - h
            c = np.correlate(h, p, mode='full') * dx
            actual_c5 = np.max(c)
            if actual_c5 < best_c5:
                best_c5 = actual_c5
                best_h = h.copy()

    if best_h is None:
        best_h = init_h
    
    return (best_h.tolist(), best_c5, n_points)
