from scipy.optimize import differential_evolution
import time

def evaluate_c5(h):
    """Evaluate C5 bound for given h values"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation: h * (1 - h_{shifted})
    corr = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_corr = np.max(corr)
    return max_corr * dx
