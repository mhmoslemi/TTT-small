import numpy as np
import time
from scipy.special import softmax
from scipy.optimize import minimize

def project_to_feasible_space(h, target_sum):
    """Project h to the set {h : 0 <= h[i] <= 1, sum(h) = target_sum}"""
    # Binary search on Lagrange multiplier
    low, high = -100, 100
    tolerance = 1e-10
    h = np.array(h)
    
    def count_sum(lam):
        temp = np.clip(h + lam, 0, 1)
        return np.sum(temp) - target_sum
    
    while high - low > tolerance:
        mid = (low + high) / 2
        if count_sum(mid) > 0:
            high = mid
        else:
            low = mid
            
    lam = (low + high) / 2
    proj_h = np.clip(h + lam, 0, 1)
    
    # Fine-tune to exactly target_sum
    diff = target_sum - np.sum(proj_h)
    if abs(diff) > 1e-10:
        # Add the difference to elements that can still grow
        indices = np.where(proj_h < 1)[0]
        if len(indices) > 0:
            delta = diff / len(indices)
            proj_h[indices] = np.minimum(proj_h[indices] + delta, 1.0)
    
    return proj_h

def compute_correlations(h, dx):
    """Compute correlations for all shifts"""
    # Compute the correlation vector for all lags
    correlation = np.correlate(h, 1 - h, mode="full")
    # Since we only care about lags in [0,2], we take from the middle
    # The full correlation is size 2*n - 1, so middle is around n-1
    start_idx = len(h) - 1
    c = correlation[start_idx:] * dx  # Take up to 2*n - 1 lags
    return c

def compute_SmoothMax(h, beta, dx):
    """Compute log-sum-exp approximation to max of correlations"""
    c = compute_correlations(h, dx)
    weights = softmax(beta * c)
    return (1 / beta) * np.log(np.sum(np.exp(beta * c))), weights

def compute_gradient_single(h, k, dx):
    """Compute gradient of single correlation c_k"""
    n = len(h)
    grad = np.zeros(n)
    # Add contribution from h[i] * (1-h[i+k])
    for i in range(n):
        if i + k < n:
            grad[i] += dx * (1 - h[i + k])
        if i - k >= 0:
            grad[i] -= dx * h[i - k]
    return grad

def compute_gradient(h, weights, dx):
    """Compute weighted sum of gradients"""
    g = np.zeros(len(h))
    for k, w in enumerate(weights):
        if k == 0:
            # Shift zero to the end for correlation indexing
            continue
        # Compute full correlation, then shift values for gradient calculation
        g += w * compute_gradient_single(h, k, dx)
    return g

def line_search(h, g, beta, dx, max_iter=10):
    """Simple backtracking line search"""
    alpha = 0.2 / (1.0 + 1)
    h_new = project_to_feasible_space(h - alpha * g, len(h) / 2)
    loss_prev = np.max(compute_correlations(h, dx))
    loss_new = np.max(compute_correlations(h_new, dx))
    iter_count = 0
    while iter_count < max_iter and loss_new > loss_prev:
        alpha *= 0.5
        h_new = project_to_feasible_space(h - alpha * g, len(h) / 2)
        loss_new = np.max(compute_correlations(h_new, dx))
        iter_count += 1
    return h_new, alpha

def optimize_one_start(h0, beta_schedule, max_iter_per_beta, dx):
    """Optimize one starting point"""
    h = h0.copy()
    best_C5 = np.max(compute_correlations(h, dx))
    
    for beta in beta_schedule:
        for it in range(max_iter_per_beta):
            c = compute_correlations(h, dx)
            # Smooth approximation of maximum
            loss = np.max(c)
            weights = softmax(beta * c)
            g = compute_gradient(h, weights, dx)
            if np.linalg.norm(g, ord=2) < 1e-8:
                break
            h_new, _ = line_search(h, g, beta, dx)
            h = h_new.copy()
            
            C5_now = np.max(compute_correlations(h, dx))
            if C5_now < best_C5:
                best_C5 = C5_now
    return h, best_C5

def run(seed=42, budget_s=600, **kwargs):
    # Set fixed parameters
    np.random.seed(seed)
    n_points = 180  # close to 150-200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    beta_schedule = [0.5, 2, 5, 15, 40, 80]
    max_iter_per_beta = 150
    t0 = time.time()
    
    # Create initial points
    # Up-scaled from initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h_values = globals()['initial_h_values']
        # Interpolate to our desired resolution
        if len(initial_h_values) > 0:
            x_orig = np.linspace(0, 2, len(initial_h_values))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, initial_h_values)
        else:
            # If empty, create simple ramp
            h_init = np.linspace(0, 1, n_points)
    else:
        # Basic initial guess
        h_init = np.ones(n_points) * 0.5
        
    # Project to feasible set by mass conservation
    h_init = project_to_feasible_space(h_init, target_sum)
    
    # Try multiple restarts
    best_h = h_init.copy()
    best_C5 = np.max(compute_correlations(best_h, dx))
    
    # Try several random restarts
    num_starts = 10
    for start_idx in range(num_starts):
        if time.time() - t0 > 0.95 * budget_s:
            break
            
        # Random initial point
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = project_to_feasible_space(h_rand, target_sum)
        h_opt, C5_opt = optimize_one_start(h_rand, beta_schedule, max_iter_per_beta, dx)
        if C5_opt < best_C5:
            best_C5 = C5_opt
            best_h = h_opt.copy()
            
        # Early exit if we found a good enough bound
        if best_C5 <= 0.379:
            break
            
    # Refinement with binary swaps if time allows
    if time.time() - t0 < 0.95 * budget_s:
        # Binary rounding of best_h
        h_binary = np.zeros(n_points)
        ones_count = 0
        for i in range(n_points):
            if np.random.random() < best_h[i]:
                h_binary[i] = 1
                ones_count += 1
            else:
                h_binary[i] = 0
                
        # Adjust to make sure sum = target_sum
        if ones_count > target_sum:
            # Decrease some 1s to 0s
            indices = np.where(h_binary == 1)[0]
            np.random.shuffle(indices)
            for i in range(ones_count - int(target_sum)):
                h_binary[indices[i]] = 0
        elif ones_count < target_sum:
            # Increase some 0s to 1s
            indices = np.where(h_binary == 0)[0]
            np.random.shuffle(indices)
            for i in range(int(target_sum) - ones_count):
                h_binary[indices[i]] = 1
                
        best_binary = h_binary.copy()
        best_C5_binary = np.max(compute_correlations(best_binary, dx))
        
        # Local search refinement
        iterations = 0
        while time.time() - t0 < 0.95 * budget_s and iterations < 1000:
            # Pick two random indices (one from 1s, one from 0s)
            ones = np.where(h_binary == 1)[0]
            zeros = np.where(h_binary == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            # Swap them temporarily
            test_h = h_binary.copy()
            test_h[i] = 0
            test_h[j] = 1
            
            # Check if this is an improvement
            c_test = compute_correlations(test_h, dx)
            C5_test = np.max(c_test)
            if C5_test < best_C5_binary:
                h_binary = test_h.copy()
                best_C5_binary = C5_test
            iterations += 1
            
        if best_C5_binary < best_C5:
            best_C5 = best_C5_binary
            best_h = h_binary.copy()
            
    return (best_h, round(best_C5, 5), n_points)
