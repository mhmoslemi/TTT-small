import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C5 for given step function h"""
    dx = 2.0 / len(h)
    correlation = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(correlation)

def project_onto_simplex_box(h, target_sum):
    """Project h onto the simplex with sum = target_sum and bounds [0,1]"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust sum to target
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple projection: scale and then clip
    scale_factor = target_sum / current_sum
    h_scaled = h * scale_factor
    if np.sum(h_scaled) > target_sum:
        # Need to reduce
        # Use binary search for suitable lambda
        low, high = -1, 1
        while high - low > 1e-6:
            mid = (low + high) / 2
            h_test = np.clip(h - mid, 0, 1)
            if np.sum(h_test) >= target_sum:
                low = mid
            else:
                high = mid
        h = np.clip(h - low, 0, 1)
    else:
        # Need to increase or adjust by clipping and redistribution
        # A much simpler approach: just adjust vector
        diff = target_sum - np.sum(h)
        if diff > 0:
            # Increase values
            h += diff / len(h)
        h = np.clip(h, 0, 1)
        # Correct sum again
        h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Starting point configs
    n_points_list = [100, 150, 200]
    best_h, best_c5 = None, float('inf')
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Multiple random initializations
        for init_iter in range(5):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Initialize
            h = np.random.rand(n_points) * 0.8 + 0.1  # values between 0.1 and 0.9
            h = project_onto_simplex_box(h, target_sum=n_points / 2)
            
            # Optimization loop
            steps = 0
            max_steps = 5000
            eta = 0.1
            beta = 1.0
            
            while steps < max_steps and time.time() - start_time < budget_s * 0.95:
                steps += 1
                
                # Compute correlations
                S_raw = np.correlate(h, 1 - h, mode='full')
                # Find worst correlation
                idx_max = np.argmax(S_raw)
                l = idx_max - (n_points - 1)
                
                # Compute gradient for this specific lag
                grad = np.zeros(n_points)
                if l >= 0:
                    # i from 0 to n_points - 1 - l
                    for i in range(n_points - l):
                        grad[i] += 1 - h[i + l]
                        grad[i + l] -= h[i]
                else:
                    # l < 0 -> shift left
                    l_abs = -l
                    # i from l_abs to n_points - 1
                    for i in range(l_abs, n_points):
                        grad[i] += 1 - h[i - l_abs]
                        grad[i - l_abs] -= h[i]
                
                # Apply gradient step and project
                h_new = h - eta * grad
                h_new = project_onto_simplex_box(h_new, target_sum=n_points / 2)
                
                # Check if improvement
                h = h_new
                current_c5 = evaluate_erdos_solution(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                
                # Reduce learning rate gradually
                if steps % 500 == 0:
                    eta *= 0.9

                best_c5 = c5_init

    if best_h is None:
        # Fallback to default initialization
        n_points = 150
        h = np.ones(n_points) * 0.5
