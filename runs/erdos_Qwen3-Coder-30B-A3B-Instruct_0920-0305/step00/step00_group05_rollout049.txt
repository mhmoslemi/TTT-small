import numpy as np

def full_overlap(x):
    """Compute full correlation of x with 1-x (mode='full')."""
    return np.correlate(x, 1 - x, mode='full')

def worst_shift(x):
    """Return (k, max_overlap) where k is the shift attaining max overlap."""
    overlaps = full_overlap(x)
    k = np.argmax(overlaps)
    max_overlap = overlaps[k]
    return (k, max_overlap)

def build_mip(N, active_shifts):
    """Build the MIP formulation with active_shifts only."""
    x = cp.Variable(N, integer=True)
    t = cp.Variable(integer=True)
    
    # Variables w[i,k] for each active shift k and valid i
    w = {}
    constraints = []
    
    # Half-ones constraint
    constraints.append(cp.sum(x) == N // 2)
    
    # Symmetry breaking: fix first element
    constraints.append(x[0] == 1)
    
    # Bounds on x
    constraints.extend([x[i] >= 0 for i in range(N)])
    constraints.extend([x[i] <= 1 for i in range(N)])
    
    # Add auxiliary binary variables and constraints for each active_shift
    for k in active_shifts:
        for i in range(N - k):
            w_key = (i, k)
            w[w_key] = cp.Variable(integer=True)
            # Enforce w[i,k] == x[i] AND x[i+k]
            # w <= x[i], w <= x[i+k]
            constraints.append(w[w_key] <= x[i])
            constraints.append(w[w_key] <= x[i + k])
            # w >= x[i] + x[i+k] - 1
            constraints.append(w[w_key] >= x[i] + x[i + k] - 1)
    
    # Overlap constraints for active shifts
    for k in active_shifts:
        # Sum_{i=0}^{N-1-k} x_i - Sum_{i=0}^{N-1-k} w[i,k] <= t
        left_sum = cp.sum([x[i] for i in range(N - k)])
        right_sum = cp.sum([w[(i, k)] for i in range(N - k)])
        constraints.append(left_sum - right_sum <= t)

    # Objective: minimize t
    prob = cp.Problem(cp.Minimize(t), constraints)

    vars_dict = {'x': x, 't': t, 'w': w}
    return prob, vars_dict

def solve_mip(prob, vars_dict, time_limit):
    """Solve the MIP and return solution tuple (x_opt, t_opt) or (None, None) if timeout."""
    try:
        prob.solve(solver=cp.CBC, max_iters=10000, TimeLimit=time_limit)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            x_opt = [int(round(vars_dict["x"][i].value)) for i in range(len(vars_dict["x"]))] if vars_dict["x"].value is not None else None
            t_opt = int(round(vars_dict["t"].value)) if vars_dict["t"].value is not None else None
            return x_opt, t_opt
        else:
            return None, None
    except Exception as e:
        return None, None
