import time
import numpy as np
from scipy import fftpack

def run(seed=42, budget_s=600, **kwargs):
    # Set up timing and parameters
    np.random.seed(seed)
    start_time = time.time()
    
    n = 300  # samples
    dx = 2.0 / n
    target_sum = n / 2
    
    # Initialisation using Beatty sequence
    beta = (np.sqrt(5) - 1) / 2
    h = np.array([1 if (i * beta) % 1 < 0.5 else 0 for i in range(n)])
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    # Adjust to get exact sum
    if diff > 0:
        # Increase some zeros to ones
        flip_indices = np.argsort(h)[::-1][:int(diff)]
        for idx in flip_indices:
            h[idx] = 1
    elif diff < 0:
        # Decrease some ones to zeros
        flip_indices = np.argsort(h)[:int(-diff)]
        for idx in flip_indices:
            h[idx] = 0
    
    # Ensure exact sum
    adjusted_sum = np.sum(h)
    if abs(adjusted_sum - target_sum) > 1e-10:
        # If still off, fix with median threshold rounding
        h = (h > np.median(h)).astype(float)
        h = project_onto_box_and_sum(h, target_sum, n)
    else:
        h = h.astype(float)

    # Project onto feasible set
    h = project_onto_box_and_sum(h, target_sum, n)

    # Gradient ascent hyperparameters
    max_alpha = 10000
    alpha = 1.0
    eta = 1.0
    min_eta = 1e-8
    temperature = 1.0
    tabu_size = 20
    
    # Best tracking
    best_C5 = float('inf')
    best_h = h.copy()
    
    # Gradient ascent
    max_iter = 50000
    iter_cnt = 0
    alpha_increase_interval = 200
    last_alpha_increase = 0
    
    while time.time() - start_time < 0.6 * budget_s:
        iter_cnt += 1
        if iter_cnt >= max_iter:
            break
            
        # Compute all overlaps S_k by convolution
        corr_full = np.correlate(h, h, mode='full')
        # Take the k-shift part (shift = 0 .. n-1)
        S_k = corr_full[n-1:n-1+n] * dx  # Corrected term for 1-d domain
        
        # Compute weights and gradient of surrogate
        weights = np.exp(-alpha * S_k)
        W = np.sum(weights)
        if W == 0:
            W = 1e-10  # avoid 0

        # Compute one convolution to get the partial derivatives efficiently (approx)
        # Direct gradient estimate for all directions using corr
        grad = np.zeros_like(h)
        for k in range(n):
            if k < n:
                # Pad the shift to match size
                shift_h = np.zeros_like(h)
                shift_h[k:] = h[:-k] if k != 0 else h[:]
                grad += weights[k] * shift_h / W
            # Reverse shift for the other term
            rev_k = k if k < n else n - k + n - 1
            if k < n and k != 0:
                shift_h_rev = np.zeros_like(h)
                shift_h_rev[:-k] = h[k:] if k > 0 else h[:]
                grad += weights[k] * shift_h_rev / W

        # Check for non-negligible gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
            
        # Line search
        h_new = h + eta * grad
        
        # Project new iterate
        h_new = project_onto_box_and_sum(h_new, target_sum, n)
        
        # Compute new surrogate
        corr_full_new = np.correlate(h_new, h_new, mode='full')
        S_k_new = corr_full_new[n-1:n-1+n] * dx
        Phi_new = - (1.0 / alpha) * np.log(np.sum(np.exp(-alpha * S_k_new)))
        
        while Phi_new < (np.sum(weights * S_k) / W - 1e-12) and eta > min_eta:
            eta /= 2
            h_new = h + eta * grad
            h_new = project_onto_box_and_sum(h_new, target_sum, n)
            corr_full_new = np.correlate(h_new, h_new, mode='full')
            S_k_new = corr_full_new[n-1:n-1+n] * dx
            Phi_new = - (1.0 / alpha) * np.log(np.sum(np.exp(-alpha * S_k_new)))
                    
        # Update h
        h = h_new
        
        # Update alpha periodically
        if iter_cnt - last_alpha_increase > alpha_increase_interval and alpha < max_alpha:
            alpha *= 2
            last_alpha_increase = iter_cnt

        # Update best solution
        c5_val = 1.0 - np.min(S_k)
        if c5_val < best_C5:
            best_C5 = c5_val
            best_h = h.copy()
    
    # Discrete polishing with local search
    # Start from better continuous solution
    h_best = best_h.copy()
    rounding_threshold = np.median(np.sort(h_best)[::-1][:int(n/2)+1])
    h_binary = (h_best > rounding_threshold).astype(int)
    
    # Normalize it to have sum = n/2
    current_sum = np.sum(h_binary)
    diff = int(target_sum - current_sum)
    if diff > 0:
        # Increase zeros to ones
        flip_indices = np.argsort(h_best)[::-1][:diff]
        for idx in flip_indices:
            h_binary[idx] = 1
    elif diff < 0:
        # Decrease ones to zeros
        flip_indices = np.argsort(h_best)[:abs(diff)]
        for idx in flip_indices:
            h_binary[idx] = 0
    
    # Project again to ensure exact sum
    h_binary = project_onto_box_and_sum(h_binary.astype(float), target_sum, n)

    current_C5 = 1.0 - np.max(np.correlate(h_binary, h_binary, mode='full')[n-1:n-1+n]) * dx
    tabu_list = []

    max_local_iter = 20000  # roughly 180 seconds at 100 iters/sec
    local_iter = 0
    while local_iter < max_local_iter and time.time() - start_time < budget_s - 20:
        local_iter += 1
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
                
        # Record so we avoid very recent swaps
        if (i, j) in tabu_list or (j, i) in tabu_list:
            continue
            
        # Swap in binary world
        h_trial = h_binary.copy()
        val_i = h_trial[i]
        val_j = h_trial[j]
        if val_i == val_j:
            continue  # no change
        h_trial[i], h_trial[j] = val_j, val_i
        
        # Evaluate new C5
        corr = np.correlate(h_trial, 1 - h_trial, mode='full')
        trial_C5 = 1.0 - np.max(corr[n-1:n-1+n]) * dx 
        if trial_C5 < current_C5:
            current_C5 = trial_C5
            h_binary = h_trial.copy()
            tabu_list.append((i, j))
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)

    # Final evaluation of true C5 bound
    corr = np.correlate(h_binary, 1 - h_binary, mode='full')
    true_C5 = 1.0 - np.max(corr[n-1:n-1+n]) * dx
    
    return (h_binary.tolist(), float(true_C5), n)
    
def project_onto_box_and_sum(h, target_sum, n):
    """Project h onto [0,1]^n and sum = target_sum"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Adjust sum
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Water-filling approach to adjust the sum
    # This is very approximate, but sufficient here
    tol = 1e-9
    l_min = -1.0
    l_max = 1.0
    for _ in range(100):
        l = (l_min + l_max) / 2.0
        new_h = np.clip(h + l, 0, 1)
        new_sum = np.sum(new_h)
        if new_sum > target_sum:
            l_max = l
        else:
            l_min = l
        if abs(new_sum - target_sum) < tol:
            break
    
    # Finally, project exactly
    h = np.clip(h + (target_sum - np.sum(h)) / n, 0, 1)
    return h
