def evaluate_h(h_values):
    """compute the C_5 bound of a given h function"""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Use full mode correlation to get all shifts
    corr = correlate(h_values, 1 - np.array(h_values), mode='full')
    # Compute the maximum of the overlap integral
    max_overlap = np.max(corr) * dx
    return max_overlap

def evaluate_h_with_constraint(h_values):
    """compute the C_5 bound with normalization constraint"""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Normalize h to satisfy integral constraint
    h_norm = np.array(h_values) * (n_points / 2) / np.sum(h_values)
    # Evaluate using full convolution
    corr = correlate(h_norm, 1 - h_norm, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def optimize_h(seed=42, budget_s=600):

        current_h = h0.copy()

        # Fallback to initial solution if nothing else was found
        best_h = h0
        best_c5 = evaluate_h_with_constraint(best_h)
    
    return (best_h, best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
