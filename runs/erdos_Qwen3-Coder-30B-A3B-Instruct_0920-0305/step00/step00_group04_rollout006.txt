def binary_to_continuous(b, dx):

def compute_correlations(h, dx):
    """Compute correlations using FFT."""
    n = len(h)
    L = 2**int(np.ceil(np.log2(2*n-1)))
    h_pad = np.pad(h, (0, L-n), mode='constant')
    g_pad = np.pad(1-h, (0, L-n), mode='constant')
    H = fftpack.fft(h_pad)
    G = fftpack.fft(g_pad)
    corr = fftpack.ifft(H * np.conj(G)).real[:2*n-1]
    return corr * dx

def correlation_max(h, dx):
    """Compute maximum correlation."""
    corr = compute_correlations(h, dx)
    return np.max(corr)

def project_to_simplex(y, target_sum):
    """Project y onto the simplex with sum = target_sum and bounds [0,1]."""
    def S(lam):
        return np.sum(np.clip(y - lam, 0, 1)) - target_sum

    y_min, y_max = np.min(y), np.max(y)
    lambda_low = y_min - 1
    lambda_high = y_max + 1
    
    try:
        sol = root_scalar(S, bracket=[lambda_low, lambda_high], method='brentq')
        lam = sol.root
    except:
        # fallback
        lam = (np.sum(y) - target_sum) / len(y)
        f = lambda x: np.sum(np.clip(y - x, 0, 1)) - target_sum
        a, b = y_min - 1, y_max + 1
        while f(a) * f(b) > 0:
            a -= 1
            b += 1
        try:
            sol = root_scalar(f, bracket=[a, b], method='brentq')
            lam = sol.root
        except:
            lam = 0
    x = np.clip(y - lam, 0, 1)
    return x

def gradient_of_smooth_max(h, dx, gamma=5.0):
    """Compute gradient of smooth max approximation of correlation max."""
    n = len(h)
    corr = compute_correlations(h, dx)
    
    # Softmax weights
    exp_corr = np.exp(gamma * corr)
    weights = exp_corr / np.sum(exp_corr)
    
    # Construct gradient element by element using finite differences / convolutions
    # But for simplicity and speed, we'll aim for a vectorized but approximate version
    
    # For shared terms we compute gradient using convolution operations
    ones = np.ones_like(h)
    
    grad = np.zeros_like(h)
    for l in range(1-n, n):
        # efficiency considerations -- handle l with less work
        if l == 0:
            # no shift
            continue
        elif l > 0:
            # right shift
            # d/dh_i (h_i * (1 - h_{i+l}))
            temp = np.zeros_like(h)
            temp[l:] = 1 - h[:-l]
            # contribution from forward
            grad += weights * temp
            # back diff - contribution from reverse (h_i * h_{i-l})
            temp2 = np.zeros_like(h)
            temp2[:-l] = h[l:]
            grad -= weights * temp2
        else:
            # l < 0, left shift
            # d/dh_i (h_i * (1 - h_{i+l}))
            temp = np.zeros_like(h)
            temp[:l] = 1 - h[-l:]
            grad += weights * temp
            temp2 = np.zeros_like(h)
            temp2[-l:] = h[:l]
            grad -= weights * temp2
            
    return grad

def gradient_descent_refinement(h_bin, dx, budget_s=250):
    """Refine binary solution to continuous."""
    h = h_bin.copy().astype(float)
    max_iter = 5000
    gamma, gamma_max = 5.0, 200.0
    alpha = 0.1 * dx
    x_i = h.copy()
    best_x = h.copy()
    
    start = time.time()
    for i in range(max_iter):
        if time.time() - start > budget_s - 5:
            break
        # compute the smooth surrogate and gradient
        c_sm = compute_correlations(x_i, dx)
        c_max = np.max(c_sm)
        try:
            phi = (1.0 / gamma) * np.log(np.sum(np.exp(gamma * c_sm)))
        except:
            phi = c_max
        
        grad = gradient_of_smooth_max(x_i, dx, gamma)
        
        new_x = x_i - alpha * grad
        new_x = project_to_simplex(new_x, len(x_i)/2)
        
        # Accept new point if true C5 improves
        c5_new = correlation_max(new_x, dx)
        if c5_new < correlation_max(x_i, dx):
            x_i = new_x
        else:
            alpha *= 0.95
            
        # Update gamma
        if i % 10 == 0 and gamma < gamma_max:
            gamma *= 1.2
            
        # Update best
        c5_best = correlation_max(best_x, dx)
        if c5_new < c5_best:
            best_x = new_x.copy()
            
    return best_x

def simulate_annealing(n_points, dx, budget_s=400):
    """Simulated Annealing for binary optimisation."""
    best_h = None
    best_c5 = np.inf
    start = time.time() 
    
    # start with a known good seed
    try:
        seed_h = initial_h_values()
        if len(seed_h) != n_points:
            # pad or truncate   to n_points
            if len(seed_h) < n_points:
                h = np.pad(seed_h, (0, n_points - len(seed_h)), constant_values=0)
            else:
                # truncate?
                h = seed_h[:n_points]
        else:
            h = seed_h.copy()
        h = (h > 0.5).astype(int)
        h = h.astype(float) * dx  # now in [0,1]...
        # Reset to ensure exactly n/2 ones
        ones_count = int(n_points / 2)
        h = np.zeros(n_points)
        h[:ones_count] = 1
        np.random.shuffle(h)
        h = h * dx
        h = h.astype(float)
    except:
        # Random seed
        h = np.zeros(n_points)
        h[:int(n_points/2)] = 1
        np.random.shuffle(h)
        h = h * dx

    def get_correlation_max(h):
        # Convert to full-range input
        return correlation_max(h/dx, dx)

    temp0 = 0.05
    alpha = 0.9995
    T_max = 0.7 * budget_s
    time_per_restart = 10  # min per restart
    restart_time = 0
    num_restarts = 0
    
    while time.time() - start + time_per_restart < budget_s * 0.7:
        # Simulated Annealing loop
        T = temp0
        k = 0
        
        # Per restart, do long-run SA
        t_start_restart = time.time()
        
        while time.time() - t_start_restart < time_per_restart and time.time() - start < budget_s * 0.7:
            # mutation: choose two indices, swap 0<->1
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            idx1 = np.random.choice(ones)
            idx2 = np.random.choice(zeros)
            
            new_h = h.copy()
            new_h[idx1] = 0
            new_h[idx2] = 1
            
            c5_before = get_correlation_max(h)
            c5_after = get_correlation_max(new_h)
            
            delta = c5_after - c5_before
            
            if delta <= 0 or np.random.rand() < np.exp(- delta / T):
                h = new_h
                
            # update temperature based on elapsed time
            t_elapsed = time.time() - start
            T = temp0 * (1.0 - t_elapsed / T_max) ** 2
            
            k += 1
            
        c5 = get_correlation_max(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        num_restarts += 1

    return best_h, best_c5
