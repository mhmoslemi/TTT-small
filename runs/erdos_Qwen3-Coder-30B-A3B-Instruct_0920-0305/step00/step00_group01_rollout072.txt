import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Extract the relevant part (k from -1 to 1, which gives max length of 2*n-1)
    # We take the middle part to get the max value, already shifted appropriately
    max_corr = np.max(corr)
    return max_corr * dx
