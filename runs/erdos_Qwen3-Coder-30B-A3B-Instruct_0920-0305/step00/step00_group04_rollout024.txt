def evaluate_erdos_solution(h_values):
    """Compute C5 bound for given h_values"""
    h = np.array(h_values)

    # Compute all correlations with 1-h
    corr = np.correlate(h, 1 - h, mode="full")
    # The full correlation has length 2*n - 1
    # The maximum is the middle element (at index n-1) for kernel size n
    max_corr = np.max(corr)
    return max_corr * dx
