import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist
import time
import math

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 bound for given h_values"""
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all k
    correlation = np.correlate(h, 1 - h, mode='full')
    # The maximum value (excluding trivial zeros) is at the center
    max_overlap = np.max(correlation)
    # Normalize by dx
    c5 = max_overlap * dx
    return c5

def initial_h_values():
    """Return a good initial construction, using n=45 samples"""
    n = 45
    h = np.zeros(n)
    dx = 2.0 / n
    # Simple construction: step function 
    # Place a step at 0.5, making it a 1/2 height step from 0.5 to 1.5
    start_idx = int(0.5 / dx)
    end_idx = int(1.5 / dx)
    if start_idx < n and end_idx <= n:
        h[start_idx:end_idx] = 1.0
    # Normalize
    h = h * 2.0 / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    h_initial = initial_h_values()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    
    # Precompute the correlation terms for faster evaluation
    def objective(h_vals):
        h = np.array(h_vals)
        if np.any(h < 0) or np.any(h > 1):
            return float('inf')
        # Ensure constraint: integral = 1
        if abs(np.sum(h) * dx - 1.0) > 1e-10:
            # Penalize constraint violation
            penalty = 1e6 * abs(np.sum(h) * dx - 1.0)
        else:
            penalty = 0.0
        correlation = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(correlation)
        return max_overlap * dx + penalty
    
    def constraint_fun(h_vals):
        h = np.array(h_vals)
        return np.sum(h) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_fun}
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization: start with Nelder-Mead 
    result = minimize(objective, h_initial, method='Nelder-Mead', 
                      options={'maxiter': 5000, 'disp': False})
    
    best_h = result.x
    
    # Try different starting points
    for i in range(5):
        if time.time() - start_time > budget_s * 0.95:
            break
        # Random perturbation of the current solution
        h_perturbed = best_h + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Normalize
        h_perturbed = h_perturbed * (1.0 / np.sum(h_perturbed) * n_points * dx)
        
        result_temp = minimize(objective, h_perturbed, method='Nelder-Mead',
                              options={'maxiter': 2000, 'disp': False})
        
        # Check if it's better
        c5_current = evaluate_erdos_solution(best_h)
        c5_temp = evaluate_erdos_solution(result_temp.x)
        if c5_temp < c5_current:
            best_h = result_temp.x
    
    # Final check and optimization with L-BFGS if we have time
    if time.time() - start_time < budget_s * 0.95:
        result_final = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds,
                               constraints=constraints, options={'maxiter': 3000, 'disp': False})
        if not result_final.success or np.isnan(result_final.fun):
            # Fall back to nelder-mead if L-BFGS fails
            result_final = minimize(objective, best_h, method='Nelder-Mead', 
                                   options={'maxiter': 1000, 'disp': False})
        
        c5_final = evaluate_erdos_solution(result_final.x)
        if c5_final < evaluate_erdos_solution(best_h):
            best_h = result_final.x
    
    # Clean up result to ensure it meets all constraints
    best_h = np.array(best_h)
    # Ensure sums are correct
    best_h = best_h * (1.0 / np.sum(best_h) * n_points * dx)
    # Clip to [0,1]
    best_h = np.clip(best_h, 0, 1)
    
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
