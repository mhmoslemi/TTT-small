import numpy as np

def evaluate_overlap(h, dx):
    """Compute maximum overlap integral C5 = max_k ∫ h(x)(1 - h(x+k)) dx"""
    n = len(h)
    # Compute cross-correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Get the values where the shift is from 0 to 2 (corresponding to k=0 to 2)
    # The overlap integral is 1 - correlation (since ∫ h(x)h(x+k)dx)
    # But we want ∫ h(x)(1-h(x+k)) dx = ∫ h(x) dx - ∫ h(x)h(x+k)dx = 1 - correlation
    # So overlap = 1 - correlation
    # We want to maximize this for all possible k in [0,2] with 0 <= k <= 2
    # The indices in 'corr' represent shift values from -(n-1) to (n-1)
    # So the valid shifts are from 0 to (n-1) (representing k = 0 to 2)
    # For k=0 onwards, shift indices start from (n-1) towards (n-1) + (n-1) = 2*(n-1)
    # We only take the elements where k = 0 to 2, so we need to map correctly
    # Since the correlation is from -2 to 2, and we sample stroking dx, we need
    # Exclude the negative entries (k < 0) and consider k = 0 to 2
    
    # Extract overlap values (as 1 - correlation)
    overlaps = []
    # valid k from 0 to 2 (in steps of dx)
    # Only consider positive shifts corresponding to k in [0,2]
    # The max index in corr is 2*(n-1) for shift 2.0
    # The shift at index i of full correlation is (i - (n-1)) * dx
    # So shift 0 is index (n-1) 
    # Shifts up to 2 from (n-1) to end of the array
    # So k values are from 0 to 2 exactly, corresponding to shift indices 
    # from (n-1) to (n-1) + 2*(n-1) based on sample spacing dx.
    
    # Actually easier: get all correlations (positive shifts, since that is what we want)
    # shift indices from 0 to (2*(n-1)) for shifts 0 to 2
    shift_start = n - 1
    shift_end = shift_start + 2*(n-1)  # That's 2*(n-1) + n - 1 = n + 2*(n-1) - 1 = 3*n - 3
    
    # k from 0 to 2 in steps of dx
    k_values = np.arange(0, 2 + dx, dx)
    
    # Now we compute for each valid k, the overlap integral (1 - correlation)
    # but we should take only the k values in valid grid:
    
    # Correlations at positive indices
    correlations = corr[shift_start:]
    
    # If have more correlations than k required (should narrow down to k=0 to 2)
    if len(correlations) > len(k_values):
        # Take every dx step as needed
        step = int(np.round(2 / dx))  # not correct... need to be smarter!
        # Correct approach: compute centers
        indices = np.linspace(0, len(correlations)-1, len(k_values), dtype=int)
        correlations_samples = correlations[indices]
    else:
        correlations_samples = correlations
    
    overlaps = 1 - correlations_samples * dx
    return np.max(overlaps)


def optimize_lp_step(n_points, t, dx):
    """Solve the LP feasibility problem for a given t"""
    # Variables:
    # h[i], i=0..n-1 (bounds 0 <= h[i] <= 1)
    # p[i,d] = h[i] * h[i+d], where d = 0..n-1 
    # We'll only include d that are valid (i+d < n), so d can range up to n-i-1.
    
    n = n_points
    # List of all index pairs (i, d) for valid products
    # which is, for d = 0 to n-1 (or max valid d for a given i)  
    # and i = 0 to n-1-d  
    pairs = []
    pmap = {} # mapping from pair (i,d) to p index in flattened list
    pcount = 0
    
    for i in range(n):
        for d in range(n - i):  # d from 0 to n-i-1
            pairs.append((i, d))
            pmap[(i, d)] = pcount
            pcount += 1
    
    # Number of p variables
    p_len = len(pairs)
    
    # Total number of variables = n + p_len
    # Each p must be constrained, and sum h = n/2
    A_eq = np.zeros((1, n + p_len))
    A_eq[0, :n] = 1  # sum h = n/2
    b_eq = np.array([n/2])
    
    # Add p inequalities
    # For each p[i,d], add its McCormick constraints
    # p <= h[i], p <= h[i+d], p >= h[i] + h[i+d] - 1, p >= 0
    A_ub = []
    b_ub = []
    
    # For each p[i,d] constraint
    for i, d in pairs:
        p_idx = pmap[(i, d)]
        row1 = np.zeros(n + p_len)
        row1[i] = 1
        row1[n + p_idx] = -1  # p <= h[i]
        A_ub.append(row1)
        b_ub.append(0)
        
        row2 = np.zeros(n + p_len)
        row2[i+d] = 1
        row2[n + p_idx] = -1  # p <= h[i+d]
        A_ub.append(row2)
        b_ub.append(0)
        
        row3 = np.zeros(n + p_len)
        row3[i] = -1
        row3[i+d] = -1
        row3[n + p_idx] = 1  # p >= h[i] + h[i+d] - 1
        A_ub.append(row3)
        b_ub.append(1 - 1)  # actually 1-1 = 0
		
        # p >= 0
        row4 = np.zeros(n + p_len)
        row4[n + p_idx] = 1
        A_ub.append(row4)
        b_ub.append(0) 
    
    # Add sum p constraint for each shift d
    # shifts go from -n+1 to n-1 (actually we modulate, but just consider due to periodicity)  
    # But we'll simplify and consider each see all components from 0 to n-1 and build valid matrix form
    # Let's switch to simulating shifts more carefully to convert to complete shifts from 0 to 2
    # So we consider for all valid time shifts 0 to 2 and make the cross-correlation condition
    
    # Better approach: for each shift d (in samples, useful to build correlations),
    # define the effect on overlap constraint as:
    # C(d) = ∑_{i=0}^{n-1-d} h[i] * h[i+d] ≈ p[i,d]  
    # constraint: 1 - C(d)/dx >= 1 - t => C(d) <= dx * t  
    # So ∑_{i} p[i,d] >= w where w = (1 - t)/dx -- this is now a e.g. 2*N for shifts
    #  
    # So for each valid shift d (from 0 to n-1 for positive), compute which indices add
    # by requiring there is a valid i such that i+d < n, i.e., d < n-i, for i=0 to n-1-d
    # But easier to create pairs for the full shifts [0,n-1] in dx steps, or systematically
    
    # Alternative: for full correlation domain on [-n+1, n-1], we will extract positive shifts
    # from 0 to 2: so we just consider both positive and negative shifts in the integral domain
    # provided k ∈ [0,2] ⇒ shift within [0,2] should be possible on positive side only, constrained for k = 0 to 2 
    # But our convolution is over full range in corr (but shift 0 to 2 = 0 to n-1)
    # so let's focus on average points
    
    # Let's just operate directly with positive shifts (i.e., patterns matching index ranges) in terms of correlation
    # Using the approach that for each shift (e.g., if i=0 to N-1, we compute left shift = i to N-1 shift compactly)
    # So i is pointing to left, shift d = i+1 to i+j where j < n  
    # So let's modify solver this way:
    # Correlation of shift d (from -n+1 to n-1) 
    # At positive d, indices go from 0 to n-1-d 
    # Compute using the product sum for all valid d (representing all valid k from 0 to 2)
    # For each d (shift between 0 and n-1), compute
    # Moment readjustment for coherence:
    # Let's think carefully:
    # Given that correlation of h with (1-h) over full range (with appropriate zero padding etc.)
    # When we take shift d of the form h[i+d], we form products h[i] * h[i+d] 
    # So we look at all valid i where i+d < n: that is i = 0 to n-1-d for d=0 to n-1.
    # That is the set we used already, by iterating i=0..n-1, d=0..n-1-i
    # Which makes p variables index 0..(sum_{l=0}^{n-2}(n-1-l))   
    # Ok, so now for a given d=0..n-1:
    # we want:
    # ∑_{i=0}^{n-1-d} h[i] * h[i+d] >= w = (1-t)/dx
    # In terms of p:[i,d], this is: 
    #  ∑_{i=0}^{n-1-d} p[i,d] >= w
    
    # Create linear constraints
    shift_indices = range(n)  # d = 0 to n-1
    shift_num = len(shift_indices)
    # Total constraints = n(eq) + 4np + (shift_num)
    num_total_ub = n + 4 * p_len + shift_num
    
    A_full = np.zeros((num_total_ub, n + p_len))
    b_full = np.zeros(num_total_ub)
    
    # First n constraints for h bounds
    idx_counter = 0
    for i in range(n):
        # 0 <= h_i <= 1
        A_full[idx_counter, i] = 1
        b_full[idx_counter] = 1
        idx_counter += 1
        A_full[idx_counter, i] = -1
        b_full[idx_counter] = 0
        idx_counter += 1
        
    # Next 4*p_len McCormick constraints for each p variable
    pair_idx = 0
    for i, d in pairs:
        p_idx = pmap[(i, d)]
        # p <= h[i]
        A_full[idx_counter, i] = 1
        A_full[idx_counter, n + p_idx] = -1
        b_full[idx_counter] = 0
        idx_counter += 1
        
        # p <= h[i+d]
        A_full[idx_counter, i+d] = 1
        A_full[idx_counter, n + p_idx] = -1
        b_full[idx_counter] = 0
        idx_counter += 1
        
        # p >= h[i] + h[i+d] - 1
        A_full[idx_counter, i] = -1
        A_full[idx_counter, i+d] = -1
        A_full[idx_counter, n + p_idx] = 1
        b_full[idx_counter] = 0
        idx_counter += 1
        
        # p >= 0
        A_full[idx_counter, n + p_idx] = 1  # p >= 0  meaningful only when promoted to inequality constraints
        b_full[idx_counter] = 0
        idx_counter += 1
    
    # Finally sum shift constraints:
    # for d (i from 0 to n-1-d): ∑ p[i,d] >= w = (1-t)/dx
    for d in range(n):
        # filter indices with i from 0 to n-1-d
        rows = [] # list of vars indices
        for i in range(n - d):
            rows.append(pmap[(i, d)])
        
        # Add constraint: sum of vars >= (1-t)/dx
        A_full[idx_counter, n + np.array(rows)] = 1
        b_full[idx_counter] = (1-t) / dx
        idx_counter += 1

    # All variables bounded: h_i ∈ [0,1], p_i,d ≥ 0 (checked via inequalities)  
    # Hasty Solver: Gurobi or CVXOPT or SciPy only = let’s use SciPy just to try
    
    # Actually let's avoid complex assertions far.  
    # For testing JOS, we'll reduce to more-understandable case (smaller n), test it within constraints
    return A_full, b_full, A_eq, b_eq
