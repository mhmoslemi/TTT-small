import numpy as np
import cvxpy as cp
import time
import math

def full_overlap(x):
    return np.correlate(x, 1 - x, mode='full')

def worst_shift(x):
    overlaps = full_overlap(x)
    k = np.argmax(overlaps)
    return (k, overlaps[k])

def build_mip(N, active_shifts):
    # Variables
    x = cp.Variable(N, integer=True)
    t = cp.Variable(integer=True)
    
    # Auxiliary variables for AND constraints
    w = {}
    constraints = []

    # Symmetry breaking
    constraints.append(x[0] == 1)
    
    # Sum constraint
    constraints.append(cp.sum(x) == N // 2)
    
    # Bounds on x
    constraints.extend([x[i] >= 0 for i in range(N)])
    constraints.extend([x[i] <= 1 for i in range(N)])
    
    # AND constraints
    for k in active_shifts:
        for i in range(N - k):
            wij = cp.Variable(integer=True)
            w[(i, k)] = wij
            constraints.append(wij <= x[i])
            constraints.append(wij <= x[i + k])
            constraints.append(wij >= x[i] + x[i + k] - 1)
    
    # Overlap constraints
    for k in active_shifts:
        lhs = cp.sum(x[:N-k]) - cp.sum([w[(i, k)] for i in range(N - k)])
        constraints.append(lhs <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create problem
    prob = cp.Problem(objective, constraints)
    return prob, {'x': x, 't': t, 'w': w}

def solve_mip(prob, vars_dict, time_limit):
    try:
        prob.solve(solver=cp.CBC, max_iters=10000, TimeLimit=time_limit)
        return vars_dict['x'].value, vars_dict['t'].value
    except Exception:
        return None, None
