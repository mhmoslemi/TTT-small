```python
import time
import numpy as np
import cvxpy as cp
import random
import math

def random_binary_vector(n):
    """Generate a random binary vector of length n with exactly n/2 ones."""
    if n % 2 != 0:
        raise ValueError("n must be even")
    h = np.zeros(n, dtype=int)
    ones = np.random.choice(n, n//2, replace=False)
    h[ones] = 1
    return h

def compute_c5(h):
    """Compute C5 bound for given step function h."""
    n = len(h)
    dx = 2.0 / n
    total = 1.0
    # Compute correlation: full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # Max over all shifts (k from -(n-1) to n-1)
    max_corr = np.max(corr)
    c5 = max_corr * dx
    return c5

def solve_milp(n, time_limit=10):
    """Solve MILP to find optimal binary h and return corresponding C5 bound."""
    # Create variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()  # represents minimum overlap

    # Constraint: exactly n/2 ones
    constraints = [cp.sum(h) == n // 2]
    
    # For each shift k = 0 to n-1
    for k in range(n):
        # Skip k=0 (since it's always n/2 which is too large to be minimal)
        # We only need constraints that are useful; for the min we mainly care about k=1..n-1
        # However to get strict bound, include all.
        # Number of valid pairs for this shift is n-k
        if n - k <= 0:
            continue
        # For shift k, compute sum_{i=0}^{n-k-1} h_i h_{i+k}
        # Define auxiliary variables for product terms
        y = cp.Variable(n - k)
        # Linearization constraints for y_i = h_i * h_{i+k}
        # y_i <= h_i
        constraints.extend([y[i] <= h[i] for i in range(n - k)])
        # y_i <= h_{i+k}
        constraints.extend([y[i] <= h[i + k] for i in range(n - k)])
        # y_i >= h_i + h_{i+k} - 1
        constraints.extend([y[i] >= h[i] + h[i + k] - 1 for i in range(n - k)])
        # y_i >= 0 (already implied by the above)
        # Set the constraint that overlap at shift k should be at least t
        constraints.append(cp.sum(y) >= t)
    
    # Additional bound to prevent t from exceeding n/2 (not needed strictly)
    constraints.append(t <= n // 2)
    
    # Objective: maximize the minimum overlap
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CBC, time_limit=time_limit, verbose=False)
        h_sol = np.round(h.value).astype(int)
        dx = 2.0 / n
        # Compute actual C5 from this solution
        # To compute C5 = max_k integral h(x) (1 - h(x+k)) dx
        # We compute sum_{i=0}^{n-1} h_i * (1 - h_{i+k}) for all k
        # i.e., consider as correlation with 1-h 
        corr = np.correlate(h_sol, 1 - h_sol, mode='full')
        max_corr = np.max(corr)
        c5_bound = max_corr * dx
        return h_sol, c5_bound
    except Exception:
        # Fallback to heuristic
        h_sol = random_binary_vector(n)
        c5_bound = compute_c5(h_sol)
        return h_sol, c5_bound

def local_search(h_initial, max_iters, time_limit):
    """Perform greedy local search to improve initial solution."""
    cur = h_initial.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    start = time.time()
    
    for i in range(max_iters):
        if time.time() - start > time_limit:
            break
        # Random swap: pick one index with h[i] = 1 and one with h[j] = 0
        ones_indices = np.where(cur == 1)[0]
        zeros_indices = np.where(cur == 0)[0]
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            continue
        i_idx = np.random.choice(ones_indices)
        j_idx = np.random.choice(zeros_indices)
        # Perform swap
        cur[i_idx] = 0
        cur[j_idx] = 1
        # Compute new C5
        new_c5 = compute_c5(cur)
        # Accept improvement
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best = cur.copy()
                best_c5 = new_c5
        else:
            # Revert swap
            cur[i_idx] = 1
            cur[j_idx] = 0

    return best, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    n = 100  # More is more accurate, but wait for better solver/tech
    try:
        # First attempt exact MILP solution with MILP only if fast
        h_opt, c5_opt = solve_milp(n, time_limit=budget_s / 3.0)
    except Exception:
        h_opt = random_binary_vector(n)
        c5_opt = compute_c5(h_opt)

    elapsed = time.time() - start
    remaining = max(0, budget_s - elapsed - 2)

    # Local search as refinement
    if remaining > 2:
        try:
            h_final, c5_final = local_search(h_opt, int(2e5), remaining)
        except Exception:
            h_final, c5_final = h_opt, c5_opt
    else:
        h_final = h_opt
        c5_final = c5_opt

    return h_final.tolist(), float(c5_final), n
```
