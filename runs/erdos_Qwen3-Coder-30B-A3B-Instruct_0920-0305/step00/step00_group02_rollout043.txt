import numpy as np
import time
from scipy import fftpack

def project_onto_simplex(h, target_sum):
    """Project h onto the hyperplane sum(h) = target_sum and box [0,1]^n."""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-9:
        return h
    # Compute λ to shift the sum to target
    # We want new_h = clip(h + λ, 0, 1), with sum(new_h) = target_sum
    # This leads to solving a 1-d root finding problem.
    # Simple Newton-style binary search:
    low, high = -1.0, 1.0
    eps = 1e-10
    for _ in range(50):
        mid = (low + high) / 2.0
        t = np.clip(h + mid, 0, 1)
        if np.sum(t) > target_sum:
            high = mid
        else:
            low = mid
        if abs(np.sum(t) - target_sum) < eps:
            break
    return np.clip(h + low, 0, 1)

def compute_Sk(h, n):
    """Compute all overlap sums S_k for k = 0, ..., n-1 using FFT-based convolution."""
    h_pad = np.concatenate([h, np.zeros(n)]);
    # convolution with reverse of itself
    c = np.convolve(h_pad, h[::-1], mode='full')
    # Now, for shift k, we want sum_{i=0}^{n-1-k} h_i h_{i+k}
    # the above convolution gives S_k = c[n-1+k] for k=0,1,...,n-1.
    # if we do C = np.correlate(h,h,mode='full'), we get same values.
    # The index entries are shifted to be i = 0,1,...n-1
    len_c = len(c)
    # S_k = c[n-1+k] for k=0, ..., n-1
    Sk = np.array([c[n-1+k] for k in range(n)])
    return Sk

def compute_grad_Sk(h, k, n):
    """Compute gradient ∂S_k/∂h_i. Apply zero padding implicitly."""
    grad = np.zeros(n)
    # The terms are h[...] * h[...+k] - which is term at i contributes to 
    # grad[i] += h[i+k]  and grad[i+k] += h[i]
    if k < n:
        for i in range(n-k):
            grad[i] += h[i+k]
        for i in range(n-k):
            grad[i+k] += h[i]
    return grad

def n_step_grad_ascent(h, n, dx, alpha, max_iter=2000, eta=1e0):
    """Performs projected gradient ascent on Φ_α approximation to min_k S_k."""
    best_val = -np.inf
    best_h = h.copy()
    alpha = 1.0
    for it in range(max_iter):
        # Compute all Sk
        Sk = compute_Sk(h, n)

        # Avoid numerical overflow in exp
        Sk = np.clip(Sk, 0, 1e6)
        exp_neg_alpha_Sk = np.exp(-alpha * Sk)
        w_sum = np.sum(exp_neg_alpha_Sk)

        if w_sum == 0:
            break

        # Gradient of Φ_α = - (1/α) log(sum(exp(-α Sk)))
        # ∇Φ_α = (sum w_k ∇S_k) / (sum w_k)
        grad = np.zeros(n)
        for k in range(n):
            g = compute_grad_Sk(h, k, n)
            grad += g * exp_neg_alpha_Sk[k]
        grad /= w_sum

        # Take a step
        h_new = h + eta * grad
        h_new = project_onto_simplex(h_new, n/2)

        # Check if new solution is better
        s_min_new = np.min(Sk)
        if s_min_new > best_val:
            best_val = s_min_new
            best_h = h_new.copy()

        # Setup for next iteration
        h = h_new
        if it > 0 and it % 200 == 0:
            alpha *= 2
            if alpha > 10000:
                alpha = 10000

        # Early stopping on gradient norm smallness
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-8:
            break

    return best_h, best_val

def round_to_binary(h_best, n):
    """Rounds continuous values h_best to binary vector with exactly n/2 ones."""
    # Find threshold to get exactly n/2 ones
    h_sorted = np.sort(h_best)[::-1]
    threshold = h_sorted[n//2]
    h_bin = (h_best >= threshold).astype(float)
    diffs = np.sum(h_bin) - n/2
    if diffs > 0:
        # Need to flip some 1s to 0s
        idxs = np.where(h_bin == 1)[0]
        idxs_sorted_by_val = idxs[np.argsort(h_best[idxs])[::-1]]
        for i in range(int(diffs)):
            h_bin[idxs_sorted_by_val[i]] = 0
    elif diffs < 0:
        # Flip some 0s to 1s
        idxs = np.where(h_bin == 0)[0]
        idxs_sorted_by_val = idxs[np.argsort(h_best[idxs])]
        for i in range(-int(diffs)):
            h_bin[idxs_sorted_by_val[i]] = 1
    return h_bin

def hill_climb_binary(h_bin, n, dx, tries_per_sec=200):
    """Optimize the binary h_bin by swaps to reduce max overlap."""
    # compute C5 = max_k S_k = max_k sum h_i * h_{i+k}
    c = np.correlate(h_bin, 1-h_bin, mode='full')
    C5 = np.max(c) * dx
    best_C5 = C5
    best_h = h_bin.copy()

    for _ in range(tries_per_sec * 180):
        # Pick random i with h[i]=1 and j with h[j]=0
        one_idx = np.where(h_bin == 1)[0]
        zero_idx = np.where(h_bin == 0)[0]
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break
        i = np.random.choice(one_idx)
        j = np.random.choice(zero_idx)
        # Swap h[i] and h[j]
        h_trial = h_bin.copy()
        h_trial[i], h_trial[j] = h_trial[j], h_trial[i]
        # Evaluate
        c_trial = np.correlate(h_trial, 1 - h_trial, mode='full')
        C5_trial = np.max(c_trial) * dx
        if C5_trial < best_C5:
            best_C5 = C5_trial
            best_h = h_trial.copy()
            h_bin = h_trial.copy()
            C5 = C5_trial
    return best_h, best_C5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 300
    dx = 2.0 / n
    target_sum = n / 2.
    
    # Step 1: Initialize with Beatty-sequence seed
    beta = (np.sqrt(5) - 1.0) / 2.0
    h0 = np.array([1 if (i * beta) % 1 < 0.5 else 0 for i in range(n)])
    h0 = project_onto_simplex(h0, target_sum)  # Wrapper for case were it doesn't satisfy sum exactly
    h = h0.copy()

    # Step 2: Perform ascent optimization
    max_iter = int(360 / 0.001)  # about 360 seconds worth, reasonable for a gradient descent over n=300
    h_opt, val_min = n_step_grad_ascent(h, n, dx, alpha=1.0, max_iter=max_iter)

    # Step 3: Rounding to binary
    h_bin = round_to_binary(h_opt, n)

    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    tries_per_sec = 200

    # Step 4: Hill climbing on binary vector
    if remaining > 1.0:
        h_final, C5 = hill_climb_binary(h_bin, n, dx, tries_per_sec=int(remaining/180 * tries_per_sec))
    else:
        h_final = h_bin
        c = np.correlate(h_final, 1 - h_final, mode='full')
        C5 = np.max(c) * dx

    return (h_final.tolist(), float(C5), n)
