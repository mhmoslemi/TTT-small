def evaluate(x):
    """Computes C5 using the same logic as the evaluator"""
    dx = 2.0 / len(x)
    # Compute all correlations for all shifts
    correlation = np.correlate(x, 1 - x, mode="full")
    # Take the maximum of all overlaps
    max_overlap = np.max(correlation)
    return max_overlap * dx

def project_onto_capped_simplex(y, target_sum):
    """Project point y onto the capped simplex with target_sum"""
    # Clip to [0, 1]
    y = np.clip(y, 0, 1)
    # Sort clipped values in descending order
    sorted_y = np.sort(y)[::-1]
    # Compute partial sums from the back
    cumsum = np.cumsum(sorted_y[::-1])[::-1]
    # Find the index that makes the projection valid
    n = len(y)
    # modified = sorted_y - (cumsum - target_sum) / np.arange(1, n+1)
    # Find the index where (sorted_y - (sum - target_sum)/k) > 0, with k being index + 1
    # The lagrange multiplier lambda is the value that makes sum = target_sum after projection
    # This is done by finding where (sorted_y[k] - (cumsum[k-1] - target_sum) / k) > 0
    tmp = sorted_y - (cumsum - target_sum) / np.arange(1, n+1)
    new_idx = np.argmax(tmp > 0)
    if tmp[new_idx] <= 0:
        new_idx = n
    lambda_star = (cumsum[new_idx-1] - target_sum) / new_idx if new_idx > 0 else 0
    
    result = np.maximum(y - lambda_star, 0)
    # Ensure this sums to target_sum
    # There is a small numerical error, so we adjust it
    diff = target_sum - np.sum(result)
    if abs(diff) < 1e-12:
        return result
    # Add adjustment to the portion with largest y (to keep it in [0,1])
    # This happens because of numerical precision issues
    if diff > 0:
        idx = np.argsort(result)[::-1]
        for i in idx:
            if diff <= 0:
                break
            add = min(diff, 1 - result[i])
            result[i] += add
            diff -= add
    elif diff < 0:
        idx = np.argsort(result)
        for i in idx:
            if diff >= 0:
                break
            sub = min(-diff, result[i])
            result[i] -= sub
            diff += sub
    return result

def gradient_of_max(x, beta):
    """Computes gradient of the smooth max with parameter beta"""
    n = len(x)
    # Compute all correlations for all shifts
    correlation = np.correlate(x, 1 - x, mode="full")
    shifts = np.arange(-n + 1, n)
    # Compute softmax weights
    exp_val = np.exp(beta * correlation)
    weights = exp_val / np.sum(exp_val)
    
    # Gradient computation
    # Use simpler approximation, since actual gradient computation is complex
    # 1 - x_{i+k} if make contribution, -x_{i-k} if contributes
    # It's better to directly compute it:
    grad = np.zeros_like(x)
    for k in shifts:
        if k > 0:
            # Contribution of x[i] (from x[i] * (1 - x[i+k]) for i = 0,...,N-1-k)
            for i in range(n-k):
                grad[i] += weights[k + n - 1] * (1 - x[i+k])
            # Contribution of x[i+k] (from x[i] * (1 - x[i+k]) for i = 0,...,N-1-k)
            for i in range(n-k):
                if i + k < n:
                    grad[i+k] -= weights[k + n - 1] * x[i]
        elif k < 0:
            k = -k
            # Contribution of x[i] (from x[i] * (1 - x[i-k]) for i = k,...,N-1)
            for i in range(k, n):
                grad[i] += weights[-k + n - 1] * (1 - x[i-k])
            # Contribution of x[i-k]  (from x[i] * (1 - x[i-k]) for i = k,...,N-1)
            for i in range(k, n):
                if i-k >= 0:
                    grad[i-k] -= weights[-k + n - 1] * x[i]
        else:
            # k = 0 then value is 0
            pass

    return grad

def swap_move(x, i, j):
    """Perform a swap move"""
    new_x = x.copy()
    new_x[i], new_x[j] = new_x[j], new_x[i]
    return new_x

def fractional_move(x, i, j, delta):
    """Perform a fractional move: add delta.x[i], subtract from x[j]"""
    new_x = x.copy()
    new_x[i] = min(1.0, new_x[i] + delta)
    new_x[j] = max(0.0, new_x[j] - delta)
    return new_x

def worst_shift_info(x):
    """Returns (k*, bad_indices) where k* is the worst shift and bad_indices are indices where a 1 is followed by a 0 at k*"""
    n = len(x)
    correlation = np.correlate(x, 1 - x, mode="full")
    # Find the index of the maximum overlap (an index in [-n+1, n-1])
    # Find max and its shift
    idx = np.argmax(correlation)
    k_star = idx - (n - 1)  # This maps correlation index back to actual shift
    overlapping_pairs = []
    if k_star > 0:
        # Positions in [0, n-k_star) where x[i] = 1 and x[i+k_star] = 0
        for i in range(n - k_star):
            if x[i] == 1 and x[i + k_star] == 0:
                overlapping_pairs.append(i)
    elif k_star < 0:
        k_abs = -k_star
        # Positions in [k_abs, n) where x[i] = 1 and x[i-k_abs] = 0
        for i in range(k_abs, n):
            if x[i] == 1 and x[i - k_abs] == 0:
                overlapping_pairs.append(i)
    else:
        pass  # k_star = 0, no such pairs under this condition

    return k_star, overlapping_pairs

def binary_sanneal(N, T0=0.12, T_min=1e-15, decay_rate=0.95):
    """Perform binary simulated annealing"""
    global t_start, t_dead

    best_x = None
    best_c5 = float('inf')
    
    num_iters = 30_000
    T_start = T0
    T_end = T_min
    step_size = (T_start - T_end) / num_iters

    # Initialize binary vector
    h0 = np.zeros(N)
    one_indices = np.random.choice(N, size=N//2, replace=False)
    h0[one_indices] = 1
    x = h0.copy()

    current_c5 = evaluate(x)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_x = x.copy()

    # Run until time is gone
    count = 0
    while count < num_iters and time.time() < t_dead:
        count += 1
        # Update T linearly
        t = count / num_iters
        T = T0 * (1 - t) if t < 1 else T0  # So T keeps decreasing

        # Select keys to swap
        ones = np.where(x == 1)[0]
        zeros = np.where(x == 0)[0]

        if len(ones) == 0 or len(zeros) == 0:
            continue

        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # Make move
        x_new = swap_move(x, i, j)
        new_c5 = evaluate(x_new)
        delta = new_c5 - current_c5

        # Accept or reject
        if delta <= 0:
            accept = True
        else:
            accept = np.log(np.random.rand()) < -delta / T
        if accept:
            x = x_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_x = x.copy()
    return best_x, best_c5

def directed_local_improvement(x_binary, max_iter=5000):
    """Performs targeted local improvement"""
    global t_dead
    x = x_binary.copy()
    c5 = evaluate(x)
    count = 0
    while count < max_iter and time.time() < t_dead:
        count += 1
        k_star, offending = worst_shift_info(x)
        if not offending:
            break  # No more offending positions
        i = np.random.choice(offending)
        # Find valid j (x_j = 0 and not i or i+k_star) 
        valid_mask = (x == 0) & (np.arange(len(x)) != i) & (np.arange(len(x)) != (i + abs(k_star) if k_star > 0 else i - abs(k_star)))
        if not np.any(valid_mask):
            break
        j = np.random.choice(np.where(valid_mask)[0])
        # Swap i and j
        x_new = swap_move(x, i, j)
        new_c5 = evaluate(x_new)
        if new_c5 <= c5:
            x = x_new
            c5 = new_c5
    return x, c5

def continuous_fine_tune(x_binary, max_iter=1000, beta_inc=1.02, beta_start=5.0):
