import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize
import time

def compute_gradient(h, w, dx):
    n = len(h)
    grad = np.zeros_like(h)
    offset = n - 1
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        shift = idx - offset
        if shift >= 0:
            # first term: h_j * (1 - h_{j + shift})
            if shift < n:
                grad[:n - shift] += wk * (1 - h[shift:])
            # second term: - h_{j - shift} * h_j
            if shift > 0:
                grad[shift:] -= wk * h[:-shift]
        else:
            # shift < 0
            s = -shift  # positive
            # first term: h_j * (1 - h_{j + shift})
            if s < n:
                grad[s:] += wk * (1 - h[:-s])
            # second term: - h_{j - shift} * h_j
            if s > 0:
                grad[:n - s] -= wk * h[s:]
    return grad * dx

def project_to_constraints(v, target_sum=0.5):
    # Project onto [0,1] and sum = target_sum using binary search
    n = len(v)
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def evaluate_solution(h):
    # Assuming evaluate_erdos_solution is given
    try:
        import evaluate_erdos_solution
        return evaluate_erdos_solution(h)
    except:
        # Fallback for demonstration (this will not work in actual run)
        return np.inf

def upscale_h(old_h, new_n):
    # linearly interpolate old_h to new_n points
    old_n = len(old_h)
    old_x = np.linspace(0, 2, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    h_interp = np.interp(new_x, old_x, old_h)
    # projection to sum = n/2
    h_interp = project_to_constraints(h_interp - 0.5, 0.5)
    return h_interp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    
    n_points = 140
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize
    try:
        import initial_h_values
        # upscale initial solution from known good one
        h_init = upscale_h(initial_h_values, n_points)
    except:
        # fallback to random if no initial provided
        h_init = np.random.random(n_points)
        h_init = project_to_constraints(h_init - 0.5, 0.5)
    
    max_restarts = 10
    for restart in range(max_restarts):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
        if restart == 0:
            h = h_init.copy()
        else:
            h = np.random.random(n_points)
            h = project_to_constraints(h - 0.5, 0.5)  # start from a random valid solution
        beta_schedule = [1.0, 5.0, 15.0, 30.0]
        for beta in beta_schedule:
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.95:
                break
            # use scipy minimize
            def obj_grad(x):
                p = 1 - x
                c = np.correlate(x, p, mode='full') * dx
                # compute loss with numerical stability
                scaled = beta * c
                max_scaled = np.max(scaled)
                w = np.exp(scaled - max_scaled)
                w_sum = np.sum(w)
                w = w / w_sum
                loss = (max_scaled + np.log(w_sum)) / beta
                # compute gradient
                grad = compute_gradient(x, w, dx)
                return loss, grad

            # equality constraint for sum
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
            bounds = [(0, 1)] * n_points
            
            # Initial step: simulated annealing style for stability
            try:
                res = minimize(obj_grad, h, method='SLSQP', jac=True, 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 100, 'ftol':1e-9})
                h = res.x.copy()
            except:
                # in case of error proceed with next restart or set as fallback
                h = h_init.copy()
            
        # Final evaluation of C5
        c = np.correlate(h, 1 - h, mode='full') * dx
        actual_c5 = np.max(c)
        if actual_c5 < best_c5:
            best_c5 = actual_c5
            best_h = h.copy()
    
    if best_h is None:
        print("No solution found within time budget.")
        return [], float('inf'), 0
    
    return list(best_h), best_c5, n_points
