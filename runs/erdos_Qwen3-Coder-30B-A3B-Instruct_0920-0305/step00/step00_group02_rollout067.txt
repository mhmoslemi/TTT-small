```python
import time
import numpy as np
import cvxpy as cp
from scipy import sparse
import random

def run(seed=42, budget_s=600):
    # Fix random seeds
    np.random.seed(seed)
    random.seed(seed)

    elapsed = time.time
    start_time = elapsed()

    # Parameters
    n = 200  # Even number of points in [0, 2] with dx = 2/n
    dx = 2.0 / n
    total_h = n // 2  # integral h = 1 => sum h = n/2
    num_rounds = 250
    max_swap_iters = 100000  # ~100k swaps in 120s at ~1000 swaps/sec
    r = 8  # Low rank for factorization

    def time_left():
        return budget_s - (elapsed() - start_time)

    # Build sparse shift matrices A_k
    A_list = []
    for k in range(1, n // 2 + 1):  # symmetry reduction
        if time_left() < 30:
            break
        rows, cols = [], []
        for i in range(n - k):
            rows.append(i)
            cols.append(i + k)
        A = sparse.csr_matrix((np.ones(len(rows)), (rows, cols)), shape=(n, n))
        A = A + A.T  # symmetric
        A_list.append(A)

    if not A_list:
        # fallback
        n = 100
        dx = 2.0 / n
        total_h = n // 2
        A_list = []
        for k in range(1, n // 2 + 1):
            rows, cols = [], []
            for i in range(n - k):
                rows.append(i)
                cols.append(i + k)
            A = sparse.csr_matrix((np.ones(len(rows)), (rows, cols)), shape=(n, n))
            A = A + A.T
            A_list.append(A)

    # At this point, we must have A_list
    print(f"Building SDP with {len(A_list)} constraints on n={n}.")

    # Define SDP variables
    X = cp.Variable((n, n), symmetric=True)
    x = cp.Variable(n)
    t = cp.Variable()

    # Constraints
    constraints = [
        X >> 0,  # PSD
        cp.diag(X) == x,  # diagonals
        cp.sum(x) == total_h,
        0 <= x,
        x <= 1,
    ]

    # Add overlap constraints
    for A_k in A_list:
        constraints.append(cp.trace(A_k @ X) >= 2 * t)

    # Objective
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)

    # Solve SDP
    print("Solving SDP...")
    time_remaining = time_left()
    if time_remaining < 60:
        print("Not enough time for SDP solving.")
        return None, None, 0

    try:
        # Solve with SCS or fallback to ECOS if it's too slow
        prob.solve(solver=cp.SCS, max_iters=2000, eps=1e-4, verbose=False)
        if prob.status != cp.OPTIMAL:
            print(f"SDP solve status: {prob.status}")
            return None, None, 0
    except Exception as e:
        print(f"SDP failed with error: {e}")
        return None, None, 0

    print(f"SDP done. Optimal t* = {t.value:.6f}, C5 bounds = {1 - t.value * dx:.6f}")

    # Extract solution
    X_star = X.value
    x_star = np.diag(X_star)

    # Factorize X_star = V V^T using SVD
    U, s, Vt = np.linalg.svd(X_star)
    r = min(r, len(s))
    V = U[:, :r] * np.sqrt(s[:r])
    print(f"Low-rank factorization with r={r}")

    # Random rounding
    best_h = None
    best_C5 = float('inf')

    print("Starting rounding iterations...")
    iter_start = elapsed()
    for rit in range(num_rounds):
        if time_left() < 10:
            break
        g = np.random.randn(r)
        scores = V @ g
        # sort scores and find threshold that makes exactly total_h ones
        sorted_scores = np.sort(scores)[::-1]
        if total_h > 0:
            tau = sorted_scores[total_h - 1]
        else:
            tau = sorted_scores[-1] - 1
        h = (scores >= tau).astype(float)

        # Evaluate C5
        # Use the given method: np.correlate(h, 1-h, mode="full")
        corr = np.correlate(h, 1 - h, mode="full") * dx
        C5 = np.max(corr)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()

    print(f"Best C5 from rounding = {best_C5:.6f}")

    # Local polishing with swaps
    if best_h is None or time_left() < 15:
        return best_h.tolist(), best_C5, n

    print("Starting local polishing...")
    h = best_h.copy()
    S = np.correlate(h, h, mode='full')[n - 1:]
    min_k = np.argmin(S)
    min_S = S[min_k]
    swap_iter_count = 0
    iter_start = elapsed()

    # Switching to optimized swap ops with pruning
    while elapsed() - iter_start < 120 and swap_iter_count < max_swap_iters:
        if time_left() < 5:
            break
        # Identify 1s and 0s
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # Perform swap
        h[i], h[j] = 0, 1
        swap_iter_count += 1
        # Compute new overlaps with rolling update
        # from sliding window; but recompute because fast enough
        S_new = np.correlate(h, h, mode='full')[n - 1:]
        min_S_new = np.min(S_new)
        if min_S_new > min_S:
            # Accept swap
            min_S = min_S_new
            min_k = np.argmin(S_new)
        else:
            # Revert
            h[i], h[j] = 1, 0

    # Final evaluation
    corr_final = np.correlate(h, 1 - h, mode="full") * dx
    final_C5 = np.max(corr_final)

    return h.tolist(), final_C5, n
```
