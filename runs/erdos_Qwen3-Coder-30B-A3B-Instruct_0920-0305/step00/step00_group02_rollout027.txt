import numpy as np
import cvxpy as cp
import time
import random
from scipy import sparse

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    n = 200
    dx = 2.0 / n
    r = 8

    # Create shift matrices A_k
    def construct_A_k(k):
        rows, cols = [], []
        for i in range(n - k):
            rows.extend([i, i + k])
            cols.extend([i + k, i])
        data = [1] * len(rows)
        A_k = sparse.csr_matrix((data, (rows, cols)), shape=(n, n))
        return A_k
    
    A_list = [construct_A_k(k) for k in range(1, n//2 + 1)]
    n_cons = len(A_list)
    
    # SDP Setup
    X = cp.Variable((n, n), PSD=True)
    x = cp.Variable(n)
    t = cp.Variable()

    constraints = [cp.diag(X) == x,
                   cp.sum(x) == n/2,
                   0 <= x, x <= 1]
    
    for i, A_k in enumerate(A_list):
        constraints.append(cp.trace(A_k @ X) >= 2 * t)

    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve SDP
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time <= 10:
        return initial_h_values, 0.38092, len(initial_h_values)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=2000, eps=1e-4, time_limit=remaining_time - 10)
    except Exception:
        return initial_h_values, 0.38092, len(initial_h_values)
    
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return initial_h_values, 0.38092, len(initial_h_values)
        
    X_star = X.value
    x_star = np.diag(X_star)
    t_star = t.value

    # Rounding
    remaining_time = budget_s - (time.time() - start_time)
    rnd_trials = min(300, max(100, int(remaining_time * 1000 / (n * r))))
    
    best_h = None
    best_c5 = float('inf')
    
    for trial in range(rnd_trials):
        if time.time() - start_time > budget_s - 2:
            break
        V, _ = np.linalg.eigh(X_star)
        V = V[:, -r:]
        g = np.random.randn(r)
        s = V @ g
        sorted_s = np.sort(s)[::-1]
        tau = sorted_s[int(n/2)]
        h = (s >= tau).astype(int)
        if np.sum(h) != n/2:
            # Adjust threshold to satisfy sum
            indices = np.argsort(s)[::-1]
            h = np.zeros(n, dtype=int)
            h[indices[:int(n/2)]] = 1
        corr = np.correlate(h, 1-h, mode='full')
        c5 = max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = 0.38092
    
    # Local search enhancement with swaps
    if time.time() - start_time < budget_s - 2:
        h_current = best_h.copy()
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k_min = np.argmin(corr[n-1:])
        c5_cur = max(corr) * dx
        iters = 0
        max_iters = 100000

        while time.time() - start_time < budget_s - 2 and iters < max_iters:
            iters += 1
            # Find a 1 and a 0 to swap
            ones = np.where(h_current == 1)[0]
            zeros = np.where(h_current == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Temporarily swap
            h_new = h_current.copy()
            h_new[i] = 0
            h_new[j] = 1

            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = max(corr_new) * dx
            k_min_new = np.argmin(corr_new[n-1:])

            if c5_new < c5_cur:
                h_current = h_new
                c5_cur = c5_new
                k_min = k_min_new

        best_h = h_current
        corr = np.correlate(best_h, 1 - best_h, mode='full') 
        best_c5 = max(corr) * dx

    return best_h.tolist(), best_c5, n
