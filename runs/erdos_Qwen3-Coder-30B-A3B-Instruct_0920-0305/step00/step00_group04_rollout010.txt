import numpy as np
import time
import random

def compute_c5_fast(h):
    n = h.shape[0]
    size = 1 << ((2*n-1).bit_length())
    f_h = np.fft.fft(h, size)
    f_g = np.fft.fft(1 - h, size)
    corr = np.fft.ifft(f_h * np.conj(f_g)).real
    corr = corr[:2*n-1]
    dx = 2.0 / n
    max_val = np.max(corr) * dx
    return max_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 60
    
    # Initialization with binary pattern
    def random_binary_vector(n):
        h = np.zeros(n)
        ones_indices = np.random.choice(n, size=n//2, replace=False)
        h[ones_indices] = 1
        return h
    
    def evaluate(h):
        return compute_c5_fast(h)
    
    # Simulated Annealing
    max_iterations = 200000
    current_h = random_binary_vector(n_points)
    current_c5 = evaluate(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Temperature schedule
    T0 = 0.1
    T_min = 1e-6
    iterations = 0
    
    while iterations < max_iterations and (time.time() - start_time) < budget_s - 0.5:
        #邻居 Sampling
        ones_indices = np.where(current_h > 0.5)[0]
        zeros_indices = np.where(current_h < 0.5)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            # Restart if constraint violated
            current_h = random_binary_vector(n_points)
            current_c5 = evaluate(current_h)
            continue
            
        i = np.random.choice(ones_indices)
        j = np.random.choice(zeros_indices)
        
        # Swap
        current_h[i], current_h[j] = current_h[j], current_h[i]
        
        # Evaluate new solution
        new_c5 = evaluate(current_h)
        delta = new_c5 - current_c5
        
        # Accept or reject
        if delta < 0 or random.random() < np.exp(-delta / T0):
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        else:
            # Revert the swap
            current_h[i], current_h[j] = current_h[j], current_h[i]
        
        # Temperature update
        iterations += 1
        # Adaptive temperature
        elapsed = time.time() - start_time
        T0 = max(T_min, 0.1 * (1 - elapsed / budget_s))
    
    best_h = best_h if best_h is not None else current_h
    return (best_h, best_c5, n_points)
