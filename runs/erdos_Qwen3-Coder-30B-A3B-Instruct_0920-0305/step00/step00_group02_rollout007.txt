import numpy as np
import cvxpy as cp
import time
import random
from scipy import sparse

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    n = 200  # moderate size, dx=0.01
    dx = 2.0 / n
    
    if n % 2 != 0:
        n = n - 1  # enforce even n
    half_n = n // 2
    
    # Construct sparse shift matrices A_k for k=1..half_n
    # A_k[i,j] = 1 if |i-j| = k, else 0
    
    time_spent = time.time() - start_time
    if time_spent > budget_s * 0.95:
        h = initial_h_values.copy() if 'initial_h_values' in globals() else np.ones(n) * 0.5
        return (h.tolist(), 0.5, n)
    
    A_list = []
    for k in range(1, half_n + 1):
        rows = []
        cols = []
        data = []
        for i in range(n - k):
            rows.append(i)
            cols.append(i + k)
            data.append(1)
            rows.append(i + k)
            cols.append(i)
            data.append(1)
        A = sparse.csr_matrix((data, (rows, cols)), shape=(n, n))
        A_list.append(A)
        time_spent = time.time() - start_time
        if time_spent > budget_s * 0.95:
            break
    
    # SDP relaxation
    time_spent = time.time() - start_time
    if time_spent > budget_s * 0.95:
        h = initial_h_values.copy() if 'initial_h_values' in globals() else np.ones(n) * 0.5
        return (h.tolist(), 0.5, n)
    
    X = cp.Variable((n, n), PSD=True)
    x = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.diag(X) == x,
        cp.sum(x) == n / 2,  # sum of h = n/2
        0 <= x, x <= 1,      # h in [0,1]
    ]
    
    # Add overlap constraints
    for A_k in A_list:
        constraints.append(cp.trace(A_k @ X) >= 2 * t)  # trace(A_k * X) >= 2t
    
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve with SCS (?)
    try:
        prob.solve(solver=cp.SCS, max_iters=2000, eps=1e-4, verbose=False)
    except Exception:
        # fallback to a simpler or cached version
        h = initial_h_values.copy() if 'initial_h_values' in globals() else np.ones(n) * 0.5
        return (h.tolist(), 0.5, n)
        
    # Extract and check
    if prob.status != cp.OPTIMAL:
        h = initial_h_values.copy() if 'initial_h_values' in globals() else np.ones(n) * 0.5
        return (h.tolist(), 0.5, n)
    
    X_star = X.value
    t_star = t.value
    
    # Now perform randomized rounding
    # Factorize X_star = V V^T using SVD or Cholesky
    cutoff = 1e-6
    r = 8  # keep rank 8 for fast sampling
    
    try:
        # SVD approximation
        U, s, Vt = np.linalg.svd(X_star, full_matrices=False)
        # Keep only components where s > cutoff
        s = s[:r]
        if s.size < r:
            r = s.size
        V = U[:, :r] * np.sqrt(s[:r])
    except Exception:
        # Fall back on eigendecomposition (PSD)
        try:
            w, v = np.linalg.eigh(X_star)
            keep = np.where(w > cutoff)[0][::-1][:r]
            V = v[:, keep] * np.sqrt(w[keep])
        except Exception:
            # fallback
            h = initial_h_values.copy() if 'initial_h_values' in globals() else np.ones(n) * 0.5
            return (h.tolist(), 0.5, n)
        
    best_h = None
    best_C5 = float('inf')
    
    # Rounding trials
    num_rounds = 200
    for _ in range(num_rounds):
        g = np.random.randn(r)
        s = V @ g  # This gives scores for rounding
        # find threshold to get exactly n/2 ones
        indices_sorted = np.argsort(-s)  # descending
        threshold_idx = n // 2
        tau = s[indices_sorted[threshold_idx]]
        h_trial = (s >= tau).astype(float)
        # Check feasibility
        if abs(np.sum(h_trial) - n / 2) > 1e-6:
            # recompute with tight threshold
            u = np.sort(s)[::-1]
            if len(u) >= n // 2:
                tau = u[n // 2 - 1]
                h_trial = (s >= tau).astype(float)
                # adjust in case of ties
                still_needs = n // 2 - np.sum(h_trial)
                if still_needs > 0:
                    for idx in range(n):
                        if h_trial[idx] == 0 and s[idx] == tau:
                            h_trial[idx] = 1
                            still_needs -= 1
                            if still_needs <= 0:
                                break
                elif still_needs < 0:
                    # correct excess
                    for idx in range(n-1, -1, -1):
                        if h_trial[idx] == 1 and s[idx] == tau:
                            h_trial[idx] = 0
                            still_needs += 1
                            if still_needs >= 0:
                                break
        # Evaluate the actual C5 using np.correlate - we do full correlation
        corr = np.correlate(h_trial, 1 - h_trial, mode='full')
        c5_val = np.max(corr) * dx
        if c5_val < best_C5:
            best_C5 = c5_val
            best_h = h_trial.copy()
            
        time_spent = time.time() - start_time
        if time_spent > budget_s * 0.95:
            break
    
    if best_h is None:
        h = initial_h_values.copy() if 'initial_h_values' in globals() else np.ones(n) * 0.5
        return (h.tolist(), 0.5, n)
    
    # Postprocess with swap hill-climbing
    if time.time() - start_time > budget_s * 0.92:
        return (best_h.tolist(), best_C5, n)
    
    # Reevaluate to get exact correlation
    h = best_h.copy()
    corr = np.correlate(h, 1 - h, mode='full')
    C5 = np.max(corr) * dx
    best_C5 = C5
    
    def eval_C5(h_arr):
        corr = np.correlate(h_arr, 1 - h_arr, mode='full')
        return np.max(corr) * dx
    
    # Simple swap: try to increase the minimum S_k
    max_iter = 1_000_000
    swaps = 0
    last_improvement = 0
    max_swaps = 10_000  # don't get too aggressive
    
    while swaps < max_swaps and time.time() - start_time < budget_s * 0.95:
        # Find the current minimum overlap index
        corr = np.correlate(h, 1 - h, mode='full')
        min_k = np.argmin(corr[2*n//3:])  # Slightly offset due to leading zeros
        # Pick random 1 and 0 idx
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # Swap
        h_trial = h.copy()
        h_trial[i] = 0
        h_trial[j] = 1
        # Evaluate
        new_C5 = eval_C5(h_trial)
        if new_C5 < best_C5:
            h = h_trial
            best_C5 = new_C5
            last_improvement = swaps
        swaps += 1
        if swaps - last_improvement > 1000:  # early stopping
            break
    
    # Final evaluation
    corr = np.correlate(h, 1 - h, mode='full')
    final_C5 = np.max(corr) * dx
    
    return (h.tolist(), final_C5, n)
