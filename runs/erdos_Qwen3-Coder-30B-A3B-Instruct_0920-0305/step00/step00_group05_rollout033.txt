import numpy as np
import time

from scipy.optimize import minimize

def evaluate(x):
    """Evaluate the Erdos constant C5 for given step function x"""
    dx = 2.0 / len(x)
    # Compute all correlations (full mode)
    cor = np.correlate(x, 1 - x, mode="full")
    # The maximum overlap
    max_overlap = np.max(cor)
    return max_overlap * dx

def project_onto_capped_simplex(y, target_sum):
    """Project y onto the capped simplex [0,1]^N with sum = target_sum"""
    y = np.clip(y, 0, 1)
    u = np.sort(y)[::-1]
    cup = np.cumsum(u) - target_sum
    rho = np.argmax(u > cup / np.arange(1, len(u) + 1)) if u[0] > cup[0] else len(u) - 1
    if rho == 0:
        theta = 0.0
    else:
        theta = cup[rho - 1] / rho
    return np.clip(y - theta, 0, 1)

def gradient_of_max(x, beta):
    """Compute gradient of log-sum-exp approximation to max"""
    N = len(x)
    dx = 2.0 / N
    # Compute all O_k values for all shifts k
    O_shifts = []
    # For each shift k, compute O_k
    for k in range(-N+1, N):
        if k >= 0:
            # O_k = sum_{i=0}^{N-1-k} x_i * (1 - x_{i+k})
            term1 = x[:N-k]
            term2 = 1 - x[k:]
            O_k = np.sum(term1 * term2)
        else:
            # k < 0 => O_k = sum_{i=-k}^{N-1} x_i * (1 - x_{i+k})
            term1 = x[-k:]
            term2 = 1 - x[:N+k]
            O_k = np.sum(term1 * term2)
        O_shifts.append(O_k)
    
    # Compute softmax weights for log-sum-exp smoothing
    O_vals = np.array(O_shifts)
    weights = np.exp(beta * O_vals)
    weights = weights / np.sum(weights)
    
    # Compute the gradient w.r.t. x_i
    grad = np.zeros_like(x)
    for i in range(N):
        for k in range(-N+1, N):
            if k >= 0:
                if i < N - k and i + k < N:
                    if i < N - k:
                        grad[i] += weights[k+N-1] * (1 - x[i+k])
                    if i >= k:
                        grad[i] -= weights[k+N-1] * x[i-k]
            else:
                # k < 0
                if i < N + k and i - k < N:
                    if i < N + k:
                        grad[i] += weights[k+N-1] * (1 - x[i+k])
                    if i >= -k:
                        grad[i] -= weights[k+N-1] * x[i-k]
        
    return grad

def swap_move(x, i, j):
    """Change x[i], x[j] via a swap - maintaining integer values"""
    x = x.copy()
    x[i], x[j] = x[j], x[i]
    return x

def fractional_move(x, i, j, delta):
    """Add delta to x[i], subtract from x[j], respecting bounds [0,1]"""
    x = x.copy()
    x[i] = min(1.0, x[i] + delta)
    x[j] = max(0.0, x[j] - delta)
    return x

def worst_shift_info(x):
    """Return (k_star, offending_indices) - telling us which shift causes max"""
    N = len(x)
    dx = 2.0 / N
    # Compute all correlations
    cor = np.correlate(x, 1 - x, mode="full")
    # Find maximum correlation
    max_cor = np.max(cor)
    # The shift k that attains max_overlaps
    k_star = np.argmax(cor) - (N - 1)
    # Find all violating pairs
    E = []
    if k_star >= 0:
        for i in range(N - k_star):
            if x[i] == 1 and x[i + k_star] == 0:
                E.append(i)
    else:
        for i in range(-k_star, N):
            if x[i] == 1 and x[i + k_star] == 0:
                E.append(i)
    return k_star, E

def phase_21_binary_sa(x, n_iter, T0=0.12, T_min=1e-6):
    """Phase 2.1: Binary simulated annealing"""
    current_c5 = evaluate(x)
    best_x = x.copy()
    best_c5 = current_c5
    T = T0
    iter_since_improvement = 0
    for t in range(n_iter):
        T = T0 * (1 - t / n_iter)
        if T < T_min:
            break
        # Try a new binary move
        i = np.random.randint(0, len(x))
        j = np.random.randint(0, len(x))
        if x[i] != x[j]:
            x_new = swap_move(x, i, j)
            new_c5 = evaluate(x_new)
            Δ = new_c5 - current_c5
            if Δ <= 0 or np.random.rand() < np.exp(-Δ / T):
                x = x_new
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_x = x.copy()
                    iter_since_improvement = 0
                else:
                    iter_since_improvement += 1
            else:
                # revert
                pass
        if iter_since_improvement > 1000:
            break
    return best_x, best_c5

def phase_22_local_improvement(x, n_iter=5000):
    """Phase 2.2: Local improvement targeting worst shift"""
    current_c5 = evaluate(x)
    for _ in range(n_iter):
        k_star, E = worst_shift_info(x)
        if len(E) == 0:
            break
        e = np.random.choice(E)
        # Find a random index j that is 0 and not part of the violating pair
        valid_indices = []
        for idx in range(len(x)):
            if x[idx] == 0 and idx != e and idx != (e + abs(k_star)):
                valid_indices.append(idx)
        if not valid_indices:
            break
        j = np.random.choice(valid_indices)
        # Perform the swap and re-evaluate
        x_new = swap_move(x, e, j)
        new_c5 = evaluate(x_new)
        if new_c5 <= current_c5:
            x = x_new
            current_c5 = new_c5
        else:
            pass  # revert the move
    return x, current_c5

def phase_3_continuous_refinement(x, n_iter=2000, beta_start=5.0, beta_increment=1.02):
    """Phase 3: Continuous optimization with smoothing"""
    dx = 2.0 / len(x)
    beta = beta_start
    current_c5 = evaluate(x)
    for i in range(n_iter):
        # Progressively increase β
        if i % 100 == 0 and i > 0:
            beta *= beta_increment
        # Compute gradient
        grad = gradient_of_max(x, beta)
        step_size = 0.02
        x_new = x - step_size * grad
        # Project to feasible region
        x_new = project_onto_capped_simplex(x_new, len(x) / 2)
        new_c5 = evaluate(x_new)
        if new_c5 < current_c5:
            x = x_new
            current_c5 = new_c5
    return x, current_c5
