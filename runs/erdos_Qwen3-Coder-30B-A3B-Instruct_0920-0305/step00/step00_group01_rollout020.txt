import numpy as np
import time
from scipy.optimize import minimize
import math

def project_to_feasible(v, target_sum):
    """Project v onto the simplex with sum = target_sum and bounds [0,1]"""
    # Clip values to [0,1]
    v = np.clip(v, 0, 1)
    # Check if already feasible
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-10:
        return v
    elif current_sum > target_sum:
        # Need to reduce values
        # Use binary search to find lambda
        lower, upper = 0, 1
        while upper - lower > 1e-12:
            mid = (lower + upper) / 2
            new_v = np.clip(v - mid, 0, 1)
            if np.sum(new_v) >= target_sum:
                lower = mid
            else:
                upper = mid
        return np.clip(v - lower, 0, 1)
    else:
        # Need to increase values, but cannot exceed 1
        # Distribute deficit
        diff = target_sum - current_sum
        indices = np.argsort(v)
        new_v = v.copy()
        for i in range(len(indices)-1, -1, -1):
            idx = indices[i]
            if diff <= 0:
                break
            can_add = 1 - new_v[idx]
            add = min(can_add, diff)
            new_v[idx] += add
            diff -= add
        return new_v

def evaluate_overlap(h):
    """Evaluate C5 = max_k ∫ h(x)(1 - h(x+k)) dx using correlation"""
    # np.correlate with mode='full' computes cross-correlation
    # for h and (1-h), but let's compute it manually for accuracy
    # Compute O(k) = ∫ h(x)(1 - h(x+k)) dx = ∫ h(x) dx - ∫ h(x)h(x+k) dx
    # Since ∫ h(x) dx = 1, O(k) = 1 - ∫ h(x)h(x+k) dx
    # so max O(k) = 1 - min ∫ h(x)h(x+k) dx
    
    # Use numpy correlate with mode full of h and (1-h)
    correl = np.correlate(h, 1 - h, mode='full')
    # correlation size = 2*N - 1, corresponding to k from -(N-1) to (N-1)
    # dx = 2 / len(h)
    dx = 2.0 / len(h)
    # Convert to O(k) values
    O = 1.0 - correl * dx
    # Return max O over all k
    return np.max(O)

def run(seed=42, budget_s=600, **kwargs):
    # Initialize from global initial_h_values if available
    # n_points should be 47 or similar (use defautl if it's a full array)
    try:
        # Assume initial_h_values is set by import
        h_values = np.array(initial_h_values)
        n_points = len(h_values)
    except:
        # Create a simple construction
        n_points = 47
        h_values = np.ones(n_points) * 0.5  # Simple constant case
    
    # Setup:
    # 1. with dx = 2.0 / n_points, we need sum(h) = n_points / 2 = total mass
    # 2. each h[i] in [0,1].  
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    # Initial evaluate
    elapsed = 0.0
    start_time = time.time()
    current_val = evaluate_overlap(h_values)
    best_val = current_val
    best_h = h_values.copy()
    
    # Use simple simulated annealing with pair moves
    # Convert to float for uniformity
    h_values = h_values.astype(float)
    
    # Annealing parameters
    T0 = 0.1  # Initial temperature
    decay_rate = 0.995  # Geometric decay
    min_T = 1e-12
    max_iter = 500000  # Max iterations
    update_interval = 1000  # Update timer every so often
    
    # Main loop
    for iteration in range(max_iter):
        # Track elapsed time
        if iteration % update_interval == 0:
            elapsed = time.time() - start_time
            if elapsed >= budget_s * 0.98:
                break
                
        T = T0 * (decay_rate ** iteration)
        if T < min_T:
            T = min_T
            
        # Select two random indices
        i1 = np.random.randint(0, n_points)
        i2 = np.random.randint(0, n_points)
        if i1 == i2:
            continue
            
        # Compute allowed delta range for moving mass from i1 to i2
        # Require new_h[i1] = h[i1] - delta >= 0 => delta <= h[i1]
        # Require new_h[i2] = h[i2] + delta <= 1 => delta <= (1 - h[i2])
        # Also require new_h[i1] >= 0, new_h[i2] <= 1
        delta_min = -h_values[i1]  # Can't go below 0
        delta_max = min(1 - h_values[i2], h_values[i1])  # Can't exceed 1
        
        if delta_min >= delta_max: 
            continue
            
        # Sample delta from uniform
        delta = np.random.uniform(low=delta_min, high=delta_max)
        if delta == 0:
            continue
            
        # Create new values
        new_h_values = h_values.copy()
        new_h_values[i1] -= delta
        new_h_values[i2] += delta
        
        # Project to feasible set (ensure sum=target_sum and bounds [0,1])
        new_h_values = project_to_feasible(new_h_values, target_sum)
        
        # Evaluate
        new_val = evaluate_overlap(new_h_values)
        
        # Accept or reject based on simulated annealing
        if new_val < current_val:
            # Always accept improvement
            h_values = new_h_values
            current_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h_values.copy()
        else:
            # Accept with probability
            delta_E = new_val - current_val
            if np.random.rand() < np.exp(-delta_E / T):
                h_values = new_h_values
                current_val = new_val
                
    return (best_h, best_val, n_points)
