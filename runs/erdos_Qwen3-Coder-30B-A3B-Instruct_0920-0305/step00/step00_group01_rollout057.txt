import numpy as np
from scipy.optimize import linprog
import time
from cvxpy import *
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 80
    dx = 2.0 / n_points
    target_mass = n_points / 2.0

    # bisection parameters
    low_bound = 0.0
    high_bound = 0.381  # current record is ~0.38092, so use a bit higher to start with
    tolerance = 1e-4

    best_h = None
    best_c5 = float('inf')

    def solve_feasible_lp(t_bound):
        # Prepare variables
        h = Variable(n_points)
        # Variables for bilinear terms
        p = {}
        all_p_vars = []
        all_constraints = []

        # Add bounds on h
        all_constraints.extend([h >= 0, h <= 1])
        # Add mass constraint
        all_constraints.append(sum(h) == target_mass)

        # Define shifts and constraints
        shift_range = range(-(n_points-1), n_points)
        w_target = (1.0 - t_bound) / dx

        for d in shift_range:
            if d >= 0:
                p_vars = Variable(n_points - d)
            else:
                p_vars = Variable(n_points + d)

            p[(d,)] = p_vars
            all_p_vars.extend(p_vars)

            # Add McCormick envelope constraints for product
            for i in range(len(p_vars)):
                i_start = i if d >= 0 else i - d
                j_start = i + d if d >= 0 else i
                if 0 <= i_start < n_points and 0 <= j_start < n_points:
                    if d >= 0:
                        h_i = h[i_start]
                        h_j = h[j_start]
                    else:
                        h_i = h[i_start]
                        h_j = h[j_start]
                    p_ij = p_vars[i]

                    # McCormick envelopes:
                    # p_ij <= h_i, p_ij <= h_j, p_ij >= h_i + h_j - 1, p_ij >= 0
                    all_constraints.extend([
                        p_ij <= h_i,
                        p_ij <= h_j,
                        p_ij >= h_i + h_j - 1,
                        p_ij >= 0
                    ])
            # Constraint that sum of products >= w_target
            if d >= 0:
                all_constraints.append(sum(p[(d,)][:]) >= w_target)
            else:
                all_constraints.append(sum(p[(d,)][:]) >= w_target)

        # Minimize dummy objective (to just check feasibility)
        objective = Minimize(0)
        prob = Problem(objective, all_constraints)
        try:
            # Try to solve
            result = prob.solve(solver=CVXOPT, verbose=False)
            if result != float('inf') and result != float('-inf'):
                h_sol = h.value
                if h_sol is not None:
                    return True, h_sol
        except Exception:
            pass
        return False, None

    def compute_C5(h_vals):
        # Allow only up to max shifts
        n = len(h_vals)
        c5_val = -1
        for k in range(1, n):
            # Compute integral ∫ h(x) * (1 - h(x+k)) dx
            integral = 0.0
            for i in range(n - k):
                integral += h_vals[i] * (1.0 - h_vals[i + k])
            integral *= dx
            c5_val = max(c5_val, integral)
        return c5_val

    # Rounding scheme: pick elements to set to 1 based on h values
    def round_to_binary(h_sol):
        # Create sorted indices by h values descending
        indices = list(range(len(h_sol)))
        indices.sort(key=lambda i: h_sol[i], reverse=True)
        h_binary = np.zeros(len(h_sol))
        current_sum = 0
        # Set top indices until total mass is reached
        for i in indices:
            if current_sum + h_sol[i] >= target_mass:
                # Assign remaining
                remaining = target_mass - current_sum
                h_binary[i] = remaining
                current_sum += remaining
                break
            h_binary[i] = h_sol[i]
            current_sum += h_sol[i]
        return h_binary

    # Perform binary search
    while high_bound - low_bound > tolerance and time.time() - start_time < budget_s - 5:
        t_mid = (low_bound + high_bound) / 2
        is_feasible, h_sol = solve_feasible_lp(t_mid)
        if is_feasible:
            # We have a feasible solution, now get a binary version and check actual value
            h_binary = round_to_binary(h_sol)
            actual_c5 = compute_C5(h_binary)
            if actual_c5 < best_c5:
                best_c5 = actual_c5
                best_h = h_binary.copy()
            high_bound = t_mid
        else:
            low_bound = t_mid

    if best_h is None:
        # fallback with initial construction
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h, 47)  # for 47 samples

    return (best_h, best_c5, len(best_h))
