import numpy as np
from scipy.optimize import minimize
import time
from math import floor

def evaluate_h(h_values, n_points):
    """Evaluate the C5 constant for given h values."""
    dx = 2.0 / n_points
    # Compute the overlap integral using convolution
    conv = np.correlate(h_values, 1 - h_values, mode="full")
    # The maximum value of the convolution gives the C5 bound
    c5_bound = np.max(conv) * dx
    return c5_bound

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial solution as a starting point

    # Bigger initial solution

    # Ensure the constraint sum(h) * dx = 1
    dx = 2.0 / n_points
    current_sum = sum(initial_h_values)
    if abs(current_sum * dx - 1.0) > 1e-10:
        scale = 1.0 / (current_sum * dx)
        initial_h_values = [x * scale for x in initial_h_values]
    
    # Now we can optimize: minimize max of overlap integrals
    # We'll sample a range of points to decide whether or not to perturb

    # The problem becomes: find h that minimizes max_k ∫ h(x)(1 - h(x+k)) dx
    # This is equivalent to minimizing the largest convolution output
    # i.e. min max(conv(h, 1-h)) where conv is full convolution

    # We will work with the normalized version where sum(h)*dx = 1
    # Devise a smooth optimization approach using scipy optimize
    
    # Starting point, we'll perturb slightly with lower/upper bounds
    def objective(h_values):
        # Bound h to [0,1] by clipping
        h = np.clip(h_values, 0, 1)
        c5 = evaluate_h(h, n_points)
        return c5

    # Constrain h to be in [0, 1], and sum(h)*dx = 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Sum constraint using scipy's custom constraint
    def constraint_func(h_values):
        return np.sum(h_values) * dx - 1.0
    
    cons = {'type': 'eq', 'fun': constraint_func}

    # Option 1: use SLSQP or trust-constr for smooth optimization
    # Option 2: use a LBFGSB that handles bounds
    method = 'L-BFGS-B'

    try:

        x0 = np.array(initial_h_values)
        res = minimize(objective, x0, method=method, bounds=bounds, constraints=cons, options={'maxiter': 10000})

        final_c5 = evaluate_h(best_h, n_points)
    except Exception as e:
        # Fallback to simple perturbation if optimization fails

    # Run small local search
    best_c5 = final_c5

    for i in range(500):

            break

        h_copy = best_h.copy()
