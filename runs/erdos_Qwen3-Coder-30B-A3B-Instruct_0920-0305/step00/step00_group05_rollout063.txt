def project_onto_capped_simplex(y, target_sum):
    """Projects y onto the capped simplex {x ∈ [0,1]^n | sum x = target_sum}"""
    y = np.clip(y, 0, 1)
    sorted_y = np.sort(y)[::-1]
    cumulative = np.cumsum(sorted_y)
    n = len(y)
    # Find the largest k such that (sum(sorted_y[:k]) - target_sum) / k <= sorted_y[k-1]
    # But it's more direct to use the formula from Duchi et al.
    # Find lambda such that sum(max(0, y_i - lambda)) = target_sum
    # sorted_y[i] - lambda >= 0 => lambda <= sorted_y[i]
    # We want sum((y_i - lambda)_+) = target_sum
    # After Eq 1 in Duchi et al. (2008)
    # lambda = (sum(sorted_y) - target_sum) / n if n <= sum(sorted_y)
    # Otherwise, use the precise algorithm
    # Let m be largest index where sorted(m) > (cumsum(sorted) - target_sum) / (m+1)
    # But computing correctly using O(n log n) approach:
    # After sorting in descending order y[0] >= y[1] >= ... >= y[n-1]
    # Then λ is such that sum(max(y[i]-λ,0)) = target_sum
    # the correct lambda is solution to that equation
    # Duchi et al. formula:
    # Let sum_y = sum(y)
    # If sum_y <= target_sum, then set all to target_sum/n
    # Else, find largest k such that sum(y[:k]) - (k * λ) >= target_sum
    # But, proper implementation:
    # Projected point z = max(0, y - λ)
    # sum(z) = target_sum
    # sum(max(y - λ, 0)) = target_sum 
    # LAMBDA - the largest λ such that above holds
    # It's derived by finding the point where the projection stops decreasing
    # A standard method: we realize that taking a sorted version,
    # then subtracting a unique λ from same index positions works.
    # Order by descending value
    idx = np.argsort(y)[::-1]
    sorted_y = y[idx]
    # Calculate cumulative sums of sorted values
    sum_y = np.sum(sorted_y)
    if sum_y <= target_sum:
        # Just scale to target
        return np.full_like(y, target_sum / len(y))
    
    # Find the right lambda for projection
    # Based on Duchi et al. Method
    # The approach for proper projection:
    # sort y descendingly as  y0 >= y1 >= ... >= y_{n-1}
    # Find lowest index k such that sum(y[0:k]) - k*lambda >= target_sum
    # => lambda = (sum(y[0:k]) - target_sum)/k  
    # But since we want sum(max(y_i - lambda, 0)) = target_sum  
    # rearrange the sum to match the connective fewer or equal:
    # sum(max(y[i] - lambda, 0)) = target_sum
    # So lambda is the smallest value that makes sum equal target
    # It's simpler: we want to find lambda within range 
    # such that sum = target
    # We compute cumulative sums of sorted y:
    # We check threshold starting from the top
    # We're after an index k such that (sum of first k elements) - k*λ >= target_sum
    # i.e. k*λ <= (sum of first k elements) - target_sum  
    # i.e. λ <= (sum first k elements - target_sum) / k
    # But    sum(max(0, y[i] - λ)) = sum_{i=0}^{k-1} (y[i] - λ) = k*lambda - sum(y[0:k]) >=  target_sum 
    # KICKSTART 1:Small problem case
    # If sum(y) <= target_sum, then the projection is truncated (elements exceed target)  
    # For target < sum(y):
    # Find highest k such that s_k - (k * lambda) <= (sum(y) - target_sum) where s_k are cumulative sums
    # So that s_k >= (k * lambda) + (sum(y) - target_sum)  
    # When is that?  
    # There is a closed form via the typical approach:
    # We sort all in descending order
    # sum of projected = target_sum  
    # Let's iterate to find exactly the threshold where it helps:
    # Sort the result descendingly
    temp = np.concatenate([[0], np.cumsum(sorted_y)])
    nvec = np.arange(1, len(sorted_y)+1)
    values = temp[1:] - target_sum
    threshold = (values / nvec)  # lambda values
    # max def for this case, we need the largest k where the threshold is after sorted_y[k]
    # First index where sorted_y[k] <= threshold[k] 
    # i.e. sorted_y[k] - lambda >= 0 AND sorted_y[k] <= lambda
    # sorted_y[k] <= threshold[k] => this condition is that we need lambda <= sorted_y[k] 
    # But that's true when lambda < sorted_y[k], but when sorted_y[k] <= threshold[k]
    # => Need highest k where sorted_y[k] <= threshold[k]
    # In range of indices
    # Compute mask of threshold[k] >= sorted_y[k]  
    # lambda_k = (temp[k+1] - target_sum)/k  
    # if this lambda >= sorted_y[k] it's okay (but we're minimizing the projection value)
    # While sorted_y[k] - lambda_k >= 0, okay, let's find such k
    mask = sorted_y <= threshold
    if np.any(mask):
        indices = np.where(mask)[0] #_sorted indices
        k = indices[-1] # Last such index where duchy criteria satisfied
        lambda_opt = (temp[k+1] - target_sum)/k
        z = np.maximum(0, sorted_y - lambda_opt) 
        # Now map back to original indices 
        result = y.copy()
        result[idx] = z
        return result
    else: # else every y was >= threshold => it's in the underweighted case
        z = sorted_y - threshold[-1]
        z = np.maximum(z, 0) 
        result = y.copy()
        result[idx] = z
        return result
    # We actually have a finished procedure

    # We compute:
    # form cumulative sums cumsum_t
    # test exponents of form: (cumsum_t[i] - target_sum)/i

def gradient_of_max(x, β):
    """Computes the gradient of the smooth max (log-sum-exp) approximation"""
    N = len(x)
    # Compute all O_k(x) 
    # This is like taking correlation (h, 1-h) for all shifts
    # full correlation with mode="full"     len = 2N-1
    # [k=-N+1,...,N-1] i.e. treated as an array
    # Want O_k = sum x_i (1-x_{i+k}) for valid i
    o_k = []
    # generate all values of O_k(x), i.e. correlations
    for k in range(1-N, N):
        if k >= 0:
            # sum_{i=0}^{N-1-k} x_i (1 - x_{i+k})
            sum_val = 0
            for i in range(0, N-k):
                sum_val += x[i] * (1-x[i+k])
            o_k.append(sum_val)
        else: # k < 0  
            # sum_{i=-k}^{N-1} x_i (1 - x_{i+k})
            sum_val = 0
            for i in range(-k, N):
                sum_val += x[i] * (1-x[i+k])
            o_k.append(sum_val)
    
    # exp(β * O_k)
    exps = np.exp(β * np.array(o_k))
    exps_sum = np.sum(exps) 
    if exps_sum == 0: 
        return np.zeros_like(x)
    
    weights = exps / exps_sum
    
    # Gradient is sum of weights * ∇O_k  
    # each ∇O_k[i] = (1 - x[i+k])·I[i+|k| < N] - x[i-k]·I[i >= k]
    # ∇[O_k] at position i  
    # it's { (1 - x[i+k]) for valid i, k >= 0, else 0 } + { -x[i-k] for valid i, k < 0, else 0 }
    grad = np.zeros_like(x)
    # For each shift k, compute its contribution
    
    for k_idx, k in enumerate(range(1-N, N)):  
        exp_wt = weights[k_idx]  
        if k >= 0:  
            # rightmost can be N-1-k, i from 0 to N-1-k
            # ∂/∂x_i  = (1 - x_{i+k}) * (i+k < N) 
            for i in range(0, N-k): 
                grad[i] += exp_wt * (1 - x[i+k]) 
                
                # subtract if x_{i-k} ليس null i.e i >= k  
                # but we also need to account that the negation gets added from the terms
                # which have in virtual x_{i-k}, but this is correct due to dual structure  
                #  vs other gradient dual authorization  
                if i >= k:  
                    grad[i] -= exp_wt * x[i-k] 
        else: # k < 0
            # i ranges from 0 to N+k-1
            # ∂/∂x_i = (1 - x_{i+k}) * (i+k >=0)  if i >= -k  
            #         = -x_{i-k} * (i < N+k)  
            k_abs = -k
            for i in range(0, N+k_abs): 
                # gradient for shift k 
                if i >= k_abs: 
                    grad[i] -= exp_wt * x[i-k] 
                if i+k_abs < N:
                    grad[i] += exp_wt * (1 - x[i+k])
    return grad

def evaluate(x):
    """Computes C5 = max_k ∫ h(x)(1 - h(x+k)) dx"""
    # Use numpy for fast correlated operation
    dx = 2.0 / len(x)
    # reverse x for compatibility with correlate equivalent to 1-h
    corr = np.correlate(x, 1-x, mode='full')
    # mode full gives 2N-1 correlations, so we take from middle and normalize the max
    max_val = np.max(corr) * dx
    return max_val

def swap_move(x, i, j):
    """Swap x[i] and x[j]"""
    x_new = x.copy()
    x_new[i], x_new[j] = x_new[j], x_new[i]
    return x_new

def fractional_move(x, i, j, delta):
    """Move delta from index j to i, preserving bounds"""
    x_new = x.copy()
    x_new[i] += delta
    x_new[j] -= delta
    x_new[i] = np.clip(x_new[i], 0, 1)
    x_new[j] = np.clip(x_new[j], 0, 1)
    return x_new

def worst_shift_info(x):
    """Returns shift_k* that gives max overlap and list of offending indices"""
    # compute overlaps for all shifts
    N = len(x)
    o_k = np.zeros(2*N-1)  # from -N+1 to N-1
    
    for k in range(1-N, N): 
        if k >= 0:
            # O_k = sum x_i (1-x_{i+k}) for i=0 to N-1-k
            start = 0
            end = N - k
            # Both i and i+k should be valid
            o_k[k+N-1] = sum(x[i] * (1 - x[i+k]) for i in range(start, end)) 
        else: # k < 0
            # O_k = sum x_i (1-x_{i+k}) for i=-k to N-1
            start = -k
            end = N
            o_k[k+N-1] = sum(x[i] * (1 - x[i+k]) for i in range(start, end))
    
    best_shift_idx = np.argmax(o_k)
    max_val = o_k[best_shift_idx]
    shift = best_shift_idx - (N-1)
    
    # identify indices where the violation happens
    # i needs to be such that i+k is valid and both x_i = 1 and x_{i+k} = 0  
    # also i ranges well 0 to N-k-1 if k>=0
    # etc.
    if shift >= 0:
        start = 0
        end = N - shift
        # identify where x_i = 1 and x_{i+shift} = 0 
        offending = [(i, i+shift) for i in range(start, end) if x[i] == 1 and x[i+shift] == 0]
    else:
        start = -shift
        end = N
        # x_i = 1 and x_{i+shift} = 0 means x_{i-shift} = 0, because x_{i-shift} = x_{i+(-shift)}
        # shift < 0, i shifted forward, so i-shift => i + |shift| => is x_{i+|shift|}  
        # FIX: if i-shift < 0, do not use
        # x_i=1 and x_{i-shift} = 0 means x_{i+|shift|}=0
        offending = [(i, i-shift) for i in range(start, end) if x[i] == 1 and x[i+(-shift)] == 0]  
    
    return shift, offending, max_val

def binary_sa(N, T0=0.12, iters_per_run=30000):
    """Perform binary simulated annealing"""
    x = np.zeros(N)  
    # sample random set of indices with 1's
    ones_pos = np.random.choice(N, N//2, replace=False)
    x[ones_pos] = 1
    
    best_x = x.copy()
    best_C5 = evaluate(x)
    
    T = T0
    decay_rate = T0 / iters_per_run
    cnt = 0
    
    while T > 0 and cnt < iters_per_run:
        # pick two random indices with the same value (which is impossible, cannot swap)
        # pick two random indices such that one is 1 and other 0
        # Select three indices uniformly: if one is 1 (and opposite not 0) or vice versa, try again
        pos_a = -1
        pos_b = -1
        steps = 0
        while True:
            a, b = np.random.choice(N, 2, replace=False)
            # try to make sure the colors are opposite
            if x[a] != x[b]: 
                pos_a, pos_b = a, b
                break
            steps += 1
            if steps > 100:
                raise RuntimeError("cannot find suitable positions after many tries")
        
        # attempt swap
        try:
            x_new = swap_move(x, pos_a, pos_b)
        except:
            print(f"Swap error at step {cnt}: pos_a={pos_a}, pos_b={pos_b}")
            continue
            
        new_C5 = evaluate(x_new)
        Δ = new_C5 - best_C5
        if Δ < 0 or np.random.rand() < np.exp(-Δ / T):
            x = x_new
            if new_C5 < best_C5:
                best_x = x_new.copy()
                best_C5 = new_C5
        cnt += 1
        T -= decay_rate
    return best_x, best_C5

def directed_local_improvement(x, iterations=5000):
    """Perform greedy swap improvement targeting the worst shift"""
    best_x = x.copy()
    C5 = evaluate(x)
    for _ in range(iterations):
        k_star, offending, max_val = worst_shift_info(best_x)
        if len(offending) == 0: break
        i_idx, i_k = random.choice(offending)
        # pick j such that x[j]=0 and j not part of current pair
        # search for any valid j other than i or i+k
        valid = [j for j in range(len(best_x)) if j != i_idx and j != i_k]
        if not valid: continue
        j_idx = np.random.choice(valid) 
        # Perform swap  
        x_new = swap_move(best_x, i_idx, j_idx)
        C5_new = evaluate(x_new)
        # Only accept unless worse (not randomized)
        if C5_new <= C5:
            best_x = x_new 
            C5 = C5_new
    return best_x, C5

def continuous_refinement(x, iterations=5000):

    α = 0.02 
    # run until small improvement or iterations
    for it in range(iterations):
