import numpy as np
import time
import random

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    time_end = start + budget_s - 0.5
    n_points = 200
    dx = 2.0 / n_points
    
    best_h = None
    best_c5 = float('inf')
    
    max_restarts = 50
    max_iters_per_restart = 20000
    
    # Multi-restart simulated annealing
    restart_count = 0
    while time.time() < time_end and restart_count < max_restarts:
        # Generate new binary solution (exactly half ones)
        h = np.zeros(n_points, dtype=np.float64)
        ones = np.random.choice(n_points, n_points // 2, replace=False)
        h[ones] = 1.0
        
        cur_c5 = compute_c5(h)
        
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        
        # Local simulated annealing
        T_start = 0.05
        for it in range(max_iters_per_restart):
            if time.time() >= time_end:
                break
                
            # Choose random indices with different values
            idx1 = np.random.randint(n_points)
            idx2 = np.random.randint(n_points)
            while h[idx1] == h[idx2]:
                idx1 = np.random.randint(n_points)
                idx2 = np.random.randint(n_points)
            
            # Propose swap
            h_new = h.copy()
            h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
            new_c5 = compute_c5(h_new)
            
            # Accept with simulated annealing probability
            d = new_c5 - cur_c5
            T = max(1e-6, T_start * (1 - it / max_iters_per_restart))
            if d <= 0 or np.random.rand() < np.exp(-d / T):
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
        
        restart_count += 1
    
    # Final check
    if best_h is not None:
        final_c5 = compute_c5(best_h)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Return a random valid solution if no better found
        h = np.zeros(n_points, dtype=np.float64)
        ones = np.random.choice(n_points, n_points // 2, replace=False)
        h[ones] = 1.0
        final_c5 = compute_c5(h)
        return (h.tolist(), final_c5, n_points)
