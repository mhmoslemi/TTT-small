import numpy as np

from scipy.signal import correlate
import time

def evaluate_h(h):
    """Compute C5 = max_k ∫ h(x)(1 - h(x+k)) dx"""
    # Correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # The integral is approximated by summing the correlation values and scaling by dx
    dx = 2.0 / len(h)
    C5 = np.max(corr) * dx
    return C5

def setup_lp_constraints(n_points, t):
    """Set up constraints for the linear programming feasibility problem.
    Variables: h_i, and p_{i,d} for d = -n+1 to n-1
    """
    dx = 2.0 / n_points
    # Target value for the correlation constraint
    w_target = (1.0 - t) / dx

    # Build the constraints matrix A_ub, b_ub for Ax <= b
    # Each shift constraint gives a new constraint
    # The constraints are of the form: sum(p_{i,d}) >= w_target for each d
    # But for LP format we need <=, so reverse signs.
    
    # We'll build p indices and variable mappings
    # First, list all the valid (i,d) pairs
    pairs = []
    for d in range(-n_points+1, n_points):
        for i in range(n_points):
            if 0 <= i+d < n_points:
                pairs.append((i, d))
    
    # Now setup LP with proper constraints
    # Variables: [h0, h1, ..., h_{n-1}, p_{0,-n+1}, p_{0,-n+2}, ..., p_{n-1,n-2}]
    # We'll flatten all h, then p in order
    
    n_vars = n_points + len(pairs)
    A = []
    b = []
    
    # Sum constraint: sum h_i = n/2 * dx
    row_sum = [0] * n_vars
    for i in range(n_points):
        row_sum[i] = 1
    A.append(row_sum)
    b.append(n_points / 2.0)  # note: sum * dx = 1 -> sum = n/2
    
    # McCormick envelope constraints for p_{i,d}
    # For each (i,d): p_{i,d} <= h_i, p_{i,d} <= h_{i+d}, 
    # p_{i,d} >= h_i + h_{i+d} - 1, p_{i,d} >= 0
    # These will be added as part of the constraint matrix
    p_idx_map = {}
    idx = n_points
    for (i, d) in pairs:
        p_idx_map[(i, d)] = idx
        idx += 1
    
    # Add all McCormick constraints per variable
    # Each p_{i,d} requires 4 constraints 
    p_constraints = 0
    for (i, d) in pairs:
        #row for p_{i,d} <= h_i
        row = [0] * n_vars
        row[p_idx_map[(i, d)]] = 1
        row[i] = -1
        A.append(row)
        b.append(0)
        p_constraints += 1
        #row for p_{i,d} <= h_{i+d}
        row = [0] * n_vars
        row[p_idx_map[(i, d)]] = 1
        row[i+d] = -1
        A.append(row)
        b.append(0)
        p_constraints += 1
        #row for p_{i,d} >= h_i + h_{i+d} - 1
        row = [0] * n_vars
        row[p_idx_map[(i, d)]] = -1
        row[i] = 1
        row[i+d] = 1
        A.append(row)
        b.append(1)
        p_constraints += 1
        #row for p_{i,d} >= 0
        row = [0] * n_vars
        row[p_idx_map[(i, d)]] = -1
        A.append(row)
        b.append(0)
        p_constraints += 1
    
    # Now the shift constraints -- Σ_{i} p_{i,d} ≥ w_target
    # For each d:
    # we're looking at shift d (i->i+d). For valid pairs, we know them.
    # Let's compute shift index
    # shifts from -n+1 to n-1, so start at 0
    shift_map = {}
    for i in range(-n_points+1, n_points):
        shift_map[i] = i + (n_points - 1)
    
    for d in range(-n_points+1, n_points):
        # Compute mass here - but we're doing total constraints
        # Build constraint vector for shift d
        row = [0] * n_vars
        # Add p_{i,d} coefficients for valid indices
        for i in range(n_points):
            if 0 <= i+d < n_points:
                row[p_idx_map[(i, d)]] = 1
        if np.sum(np.abs(row)) > 0:
            A.append(row)
            b.append(w_target)
    
    # Precompute inverse mapping for efficiency
    return np.array(A), np.array(b), p_idx_map, shift_map

def round_solution(h_lp, n_points):
    """Round the fractional h solution to binary values satisfying the mass constraint."""
    # Compute M = n_points / 2 
    M = int(n_points / 2)
    
    # Rank indices based on h values descending
    h_sorted_idx = np.argsort(h_lp)[::-1]
    
    h_binary = np.zeros(n_points)
    for i in range(M):
        h_binary[h_sorted_idx[i]] = 1.0
    return h_binary
