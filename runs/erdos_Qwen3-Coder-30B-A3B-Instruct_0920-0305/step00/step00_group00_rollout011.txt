import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_C5(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the correlation using full mode
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the one we want to minimize
    C5 = np.max(correlation) * dx
    return C5

def initial_h_values():
    """Provide an initial construction for h."""
    n = 45
    h = np.zeros(n)
    # Create piecewise constant function
    # A good starting construction
    h[0:15] = 0.5
    h[15:30] = 1.0
    h[30:45] = 0.0
    return h

def objective(h):
    """Objective function to minimize."""
    return evaluate_C5(h)

def run(seed=42, budget_s=600, **kwargs):
    """Run optimization to find improved upper bound for C5."""
    np.random.seed(seed)
    
    n_points = 45  # Start with 45 points as in the initial construction
    dx = 2.0 / n_points

    # Function representing the constraint that integral equals 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0

    # Bounds for each variable: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    current_h = initial_h_values()
    
    best_h = current_h.copy()
    best_C5 = evaluate_C5(best_h)
    start_time = time.time()
    
    # Use a simple differential evolution for optimization
    # This is a local search that's suitable for this discrete-like space
    def diff_evo_objective(h):
        # Normalize sum to be correct
        h = h.copy()
        current_sum = np.sum(h) * dx
        if current_sum > 0:
            h = h * (1.0 / current_sum)
        h = np.clip(h, 0, 1)
        return evaluate_C5(h)
        
    # We use bounds and constraints carefully
    # Set up the optimization with the objective that requires normalization
    def constrained_objective(h):
        # Ensure it's within bounds
        h = np.clip(h, 0, 1)
        # Ensure sum constraint
        h_sum = np.sum(h) * dx
        if abs(h_sum - 1.0) > 1e-8:
            # Renormalize to satisfy the constraint
            if h_sum > 0:
                h = h * (1.0 / h_sum)
                h = np.clip(h, 0, 1)
        return evaluate_C5(h)
        
    # Use differential evolution which usually does well on such problems
    # Keep track of time
    try:
        result = differential_evolution(
            constrained_objective,
            bounds,
            maxiter=500,
            popsize=15,
            seed=seed,
            strategy='best1bin',
            tol=1e-6
        )
        if not result.success:
            # If optimization failed, just return previous best
            h_opt = best_h.copy()
            c5_opt = best_C5
        else:
            h_opt = np.clip(result.x, 0, 1)
            c5_opt = evaluate_C5(h_opt)
            # Renormalize to make sure integral constraint is satisfied
            correct_sum = np.sum(h_opt) * dx
            if abs(correct_sum - 1.0) > 1e-8:
                h_opt = h_opt * (1.0 / correct_sum)
                h_opt = np.clip(h_opt, 0, 1)
                c5_opt = evaluate_C5(h_opt)
        
        # Final check to see if this is better
        if c5_opt < best_C5:
            best_h = h_opt.copy()
            best_C5 = c5_opt
    except Exception:
        pass
        
    # Even if optimization failed, return the best known (which is initial)
    h_final = np.clip(best_h, 0, 1)
    
    # Make sure the integral constraint is satisfied
    if abs(np.sum(h_final) * dx - 1.0) > 1e-8:
        h_final = h_final * (1.0 / (np.sum(h_final) * dx))
        h_final = np.clip(h_final, 0, 1)
    
    c5_final = evaluate_C5(h_final)
    
    # At this point, if we haven't improved much, still return the result immediately
    return (h_final.tolist(), c5_final, n_points)
