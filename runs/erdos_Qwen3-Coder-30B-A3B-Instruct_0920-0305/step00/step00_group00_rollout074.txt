import time
import numpy as np
from scipy.special import softmax
from scipy.signal import correlate

def project_onto_feasible(h, target_sum):
    """Project h onto the set {0 <= h[i] <= 1, sum h = target_sum}"""
    # Binary search for lambda
    low, high = -100, 100
    tol = 1e-12
    for _ in range(60):
        mid = (low + high) / 2
        new_h = np.clip(h + mid, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > target_sum:
            low = mid
        else:
            high = mid
    h_proj = np.clip(h + (low + high)/2, 0, 1)
    # Adjust to ensure exact sum
    diff = target_sum - np.sum(h_proj)
    # Add to smallest values
    if abs(diff) > 1e-10:
        indices = np.argsort(h_proj)
        i = 0
        while diff > 1e-12 and i < len(indices):
            idx = indices[i]
            to_add = min(diff, 1 - h_proj[idx])
            h_proj[idx] += to_add
            diff -= to_add
            i += 1
    return h_proj

def compute_correlations(h, dx):
    """Compute all correlations c_k = dx * sum h[i] * (1 - h[i+k])"""
    # This computes the full correlation with zero padding
    # Results = [c_0, c_1, ..., c_{2n-2}]
    # But we are interested only in c_k for small k = 0,1,...,n-1
    h_rev = np.flip(h)
    conv = correlate(h, h_rev, mode="full") # This is h * h_rev
    c = np.zeros(2*len(h) - 1)
    c[len(h)-1] = np.sum(h)  # term h[i] * h[i] sum
    c[len(h):] = np.sum(h) - conv[len(h):]  # term h[i] * h[i+k]
    c[:len(h)-1] = np.sum(h) - conv[:len(h)-1] # term h[i] * h[i-k]
    # Correct the off-diagonal terms:
    # For i+k < len(h), c[k] = dx * sum h[i] * (1 - h[i+k])
    # We have conv[i] = sum h[j] * h[j-i]
    # But we want conv[i] = sum h[j] * (1 - h[j+i]) = sum h[j] - sum h[j]*h[j+i]
    # So: d = sum h[j] - conv[i] = sum (1 - h[j]) * h[j+i] = sum h'[j] * h[j+i]
    # Which means we compute correctly the auto-correlation of h', 
    # hence we compute the cross-correlation between h and (1-h)
    # But we need sum h[i] * (1 - h[i+k])
    # v = 1 - h
    # compute h * v_k (v shifted by k) for all k
    # This is h * (1-h)_{shifted}
    # correlation of h and (1-h) is \sum h[i] * (1 - h[i+k])
    # We expect we do it correctly using scipy.signal.correlate correctly 
    # in a simplified way:
    one_minus_h = 1 - h
    full_cross_corr = correlate(h, one_minus_h, mode="full")
    # full_cross_corr should be double the width:
    # first len(h)-1 of it are like at negative shifts, then len(h) are positive
    # so c[i] = full_cross_corr[len(h)-1 + i] * dx
    c = full_cross_corr * dx
    # Only the positive lags contribute to the max
    # because for lags >=n we have zero-padding
    # c[0] is for k=0, and c[1] ... c[n-1] are for k=1, ..., n-1
    # c[n+1] ... are for negative shifts, but only positive ones matter
    return c[:len(h)]

def compute_gradient(h, c, weights, dx):
    """Compute gradient of L_β = (1/β)*log(sum_k exp(β*c_k))"""
    n = len(h)
    g = np.zeros(n)
    
    # Loop over shifts k
    for k in range(n):
        # Gradient: \frac{\partial c_k}{\partial h_j} = dx * (1 - h[j+k]) * 1_{j+k < n} - dx * h[j-k] * 1_{j-k >= 0}
        # which is pdf via slicing:
        if k == 0:
            # \frac{\partial c_0}{\partial h_j} = dx * (1 - h[j+0]) 
            # i.e. dx * 1_{0 <= j < n} = dx * ones[n]
            g += dx * (1 - h)
        else:
            # Subtract contribution from h[j-k]*delta_{j-k>=0}  
            # i.e. -dx * h[j-k] where j-k >= 0 => j >= k 
            # Add contribution from (1-h[j+k]) * delta_{j+k < n} 
            # i.e. dx [1 - h[j+k]] where j+k < n => j < n-k
            if k < n:
                g += dx * (1 - h[k:])  # from the positive convolution part
                if k > 0:
                    g[:n-k] -= dx * h[:n-k] # from the negative part

    # Return weighted sum over all slopes (similar to softmax weighting)
    # Actually compute actually that:
    # g = sum_k w_k * grad_k(c_k) where grad_k(c_k) = (1/h) * dg/dh_k  
    # No, then I have to rewrite the computation of gradient to get 
    # d/dh_j of c_k = dx * h_i*(1-h_{i+k})
    # J = dx * sum h_i*(1-h_{i+k}) = dx * (sum h_i - sum h_i*h_{i+k})
    # \frac{\partial J}{\partial h_j} = dx * (1 - sum_{i=j} h_{i+k} - sum_{i=j} h_i * 1_{i+k = j})
    # = dx * (1 - h_{j+k} - h_{j-k})
    # But for i+k, if j+k >= n it has no contribution
    # So we restrict to valid lags and compute:
    
    g_total = np.zeros_like(h)
    for k in range(n):
        grad = np.zeros_like(h)
        if k == 0:
            grad = dx * (1 - h)
        else:
            # Add - dx * h_{i+k} where i+k < n (i from 0 to n-k-1)
            # Subtract dx * h_{j-k} where j-k >= 0 (j from k to n-1)
            if k > 0:
                if k < n:
                    grad[k:] = -dx * h[:n-k]   # Subtract on [k, n-1]  
                    grad[:n-k] += dx * (1 - h[k:]) # Add on [0, n-k)
                else:
                    grad = -dx * h
        g_total += weights[k] * grad  
    return g_total
