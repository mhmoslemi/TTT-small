def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant C5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # For each shift k, compute integral value
    max_val = 0
    for k in range(n):
        val = corr[n-1+k] * dx
        if val > max_val:
            max_val = val
    return max_val
