```python
import numpy as np
import time
from scipy.optimize import differential_evolution
from cvxpy import *
import math

def decode(z):
    M = len(z) // 2
    z_ell = z[:M]
    z_g = z[M:]
    ell = np.exp(z_ell) / np.sum(np.exp(z_ell))
    g = np.exp(z_g) / np.sum(np.exp(z_g))
    return ell, g

def build_intervals(ell, g):
    M = len(ell)
    a = np.zeros(M)
    a[0] = g[0]
    for i in range(1, M):
        a[i] = a[i-1] + ell[i-1] + g[i]
    b = a + ell
    return a, b

def candidate_shifts(a, b):
    E = list(a) + list(b)
    T = set([0])
    for i in range(len(E)):
        for j in range(len(E)):
            diff = abs(E[i] - E[j])
            if diff <= 2:
                T.add(diff)
    return sorted(list(T))

def intersection_length(t, a, b):
    total = 0
    for i in range(len(a)):
        left = max(a[i], a[i] - t)
        right = min(b[i], b[i] - t)
        if right > left:
            total += (right - left)
    return total

def objective(z):
    ell, g = decode(z)
    a, b = build_intervals(ell, g)
    T = candidate_shifts(a, b)
    min_I = float('inf')
    for t in T:
        I = intersection_length(t, a, b)
        if I < min_I:
            min_I = I
    return 1 - min_I

def run_DE(budget_s, seed, M):
    np.random.seed(seed)
    bounds = [(-5, 5)] * (2*M + 1)
    result = differential_evolution(objective, bounds, seed=seed,
                                    popsize=15*(2*M+1), maxiter=2000,
                                    disp=False, atol=1e-8, ftol=1e-8)
    return result.x

def local_LP(z_best, M):
    ell, g = decode(z_best)
    a, b = build_intervals(ell, g)
    T = candidate_shifts(a, b)
    min_I = float('inf')
    for t in T:
        I = intersection_length(t, a, b)
        if I < min_I:
            min_I = I
    active_t = [t for t in T if abs(intersection_length(t, a, b) - min_I) < 1e-10]
    
    # Build LP
    x = Variable(2*M)
    coeffs = {}
    for t in active_t:
        c_t = 0
        alphas = np.zeros(2*M)
        for i in range(M):
            left = max(a[i], a[i] - t)
            right = min(b[i], b[i] - t)
            width = max(0, right - left)
            c_t += width
            if abs(left - a[i]) < 1e-10:
                alphas[i] = 1
            if abs(right - b[i]) < 1e-10:
                alphas[i + M] = -1
            if abs(left - (a[i] - t)) < 1e-10:
                alphas[i] = 1
            if abs(right - (b[i] - t)) < 1e-10:
                alphas[i + M] = -1
        coeffs[t] = (c_t, alphas)

    objective = Maximize(Variable() - 1e-12)
    constraints = []
    for t in active_t:
        c_t, alpha = coeffs[t]
        constraints.append(c_t + alpha.T @ x >= objective.args[0])
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=ECOS, verbose=False)
    except:
        pass

    # Rebuild best
    new_z = z_best.copy()
    old_a, old_b = build_intervals(ell, g)
    delta = 1e-3
    new_a = np.array([a[i] + x.value[i] if abs(x.value[i]) < delta else a[i] for i in range(M)])
    new_b = np.array([b[i] + x.value[i+M] if abs(x.value[i+M]) < delta else b[i] for i in range(M)])
    new_ell = new_b - new_a
    new_g = np.zeros(M+1)
    new_g[0] = new_a[0]
    for i in range(1, M):
        new_g[i] = new_a[i] - new_b[i-1]
    new_g[-1] = 2 - new_b[-1]
    
    new_ell_norm = new_ell / np.sum(new_ell) * 1
    new_g_norm = new_g / np.sum(new_g) * 1
    new_z[:M] = np.log(new_ell_norm)
    new_z[M:] = np.log(new_g_norm)
    return new_z

def gridify(intervals, heights, n):
    h = np.zeros(n)
    dx = 2.0 / n
    for i in range(len(intervals)):
        a, b = intervals[i]
        start = max(0, int(np.ceil((a - 0) / dx)))
        end = min(n, int(np.floor((b - 0) / dx)) + 1)
        for j in range(start, end):
            h[j] = heights[i]
    return h

def binary_hill_climb(h, time_left_s, start_time):
    # Simple hill climbing on binary swaps
    dx = 2.0 / len(h)
    def compute_C5(h):
        return np.max(np.correlate(h, 1-h, mode="full")) * dx
    
    C5 = compute_C5(h)
    best = h.copy()
    best_C5 = C5
    count = 0
    while time.time() - start_time < time_left_s - 0.5 and count < 50000:
        # Random swap
        i = np.random.randint(0, len(h))
        j = np.random.randint(0, len(h))
        if i != j:
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
            C5_new = compute_C5(h_new)
            if C5_new < best_C5:
                best = h_new.copy()
                best_C5 = C5_new
                h = h_new
        count += 1
    return best, best_C5

def run(seed=42, budget_s=600):
    start_time = time.time()
    M = 6
    n = 400
    
    # Phase 1: DE tuning
    remaining = budget_s * 0.7
    z_best = run_DE(remaining, seed, M)
    
    # Phase 2: LP refinement
    remaining = budget_s * 0.7 + budget_s * 0.05
    z_refined = local_LP(z_best, M)
    
    # Phase 3: Gridify
    ell, g = decode(z_refined)
    a, b = build_intervals(ell, g)
    intervals = list(zip(a, b))
    heights = np.ones(len(ell))  # assuming all heights are 1
    h = gridify(intervals, heights, n)
    
    # Adjust to ensure integral = 1
    dx = 2.0 / n
    h_sum = np.sum(h) * dx
    diff = h_sum - 1
    if abs(diff) > 1e-6:
        if diff > 0:
            # Find a 1 and change to 0
            for i in range(n):
                if h[i] > 0:
                    h[i] = 0
                    break
        else:
            # Find a 0 and change to 1
            for i in range(n):
                if h[i] < 1:
                    h[i] = 1
                    break
    h_sum = np.sum(h) * dx
    if abs(h_sum - 1) > 1e-6:
        # final adjustment
        indices = np.where(h == 0)[0]
        if len(indices) > 0:
            h[indices[0]] = 1

    # Phase 4: Binary hill climbing
    remaining = budget_s - (time.time() - start_time)
    h_final, c5_final = binary_hill_climb(h, remaining, start_time)
    
    return h_final.tolist(), c5_final, n
```
