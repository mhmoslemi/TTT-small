def run(seed=42, budget_s=600, **kwargs):
    import time
    import numpy as np
    import cvxpy as cp
    import random
    import math

    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    
    # Try to solve MILP for n=100
    n = 100
    dx = 2.0 / n
    total = 1.0
    
    h_sol = None
    c5_bound = float('inf')

    try:
        # Create binary variables
        h = cp.Variable(n, boolean=True)
        t = cp.Variable()
        
        # Constraints
        constraints = [cp.sum(h) == n // 2]
        
        # Linearization for each shift k
        for k in range(1, n):  # shifted from 1 to avoid k=0 which is always n//2
            # For shift k, valid indices are 0..n-k-1
            if n - k == 0:
                continue
            # Binary product variables
            y = cp.Variable(n - k)
            # Linear constraints to enforce y[i] = h[i] * h[i+k]
            constraints += [y <= h[:n-k]]
            constraints += [y <= h[k:]]
            constraints += [y >= h[:n-k] + h[k:] - 1]
            constraints += [y >= 0]
            # Add constraint: sum_y >= t
            constraints += [cp.sum(y) >= t]
            
        # Ensure t <= n/2
        constraints += [t <= n/2]
        
        # Solve to maximize t
        prob = cp.Problem(cp.Maximize(t), constraints)
        prob.solve(solver=cp.CBC, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTAINED]:
            h_sol = np.round(h.value).astype(float)
            # Compute actual C5 from h
            corr_full = np.correlate(h_sol, 1 - h_sol, mode='full')
            c5_bound = np.max(corr_full) * dx
            
    except Exception as e:
        pass

    # Fallback to heuristic if MILP fails or takes too much time
    if h_sol is None or c5_bound >= 0.3807:
        n = 200  # Try larger n for heuristic search
        dx = 2.0 / n
        # Initialize with a random binary vector
        h_initial = np.zeros(n)
        ones_indices = np.random.choice(n, size=n//2, replace=False)
        h_initial[ones_indices] = 1.0
        h_sol = h_initial.copy()

        # Compute initial C5
        corr_full = np.correlate(h_sol, 1 - h_sol, mode='full')
        c5_bound = np.max(corr_full) * dx

        # Hill climbing local search
        elapsed = time.time() - start
        remaining_time = budget_s - elapsed - 5  # Leave some buffer

        if remaining_time > 0:
            max_iters = int(remaining_time * 2000)  # Rough estimate of iterations per second

            for iteration in range(max_iters):
                # Pick two positions to swap
                ones_idx = np.where(h_sol == 1)[0]
                zeros_idx = np.where(h_sol == 0)[0]
                if len(ones_idx) == 0 or len(zeros_idx) == 0:
                    break

                i = np.random.choice(ones_idx)
                j = np.random.choice(zeros_idx)
                
                # Try the swap
                h_new = h_sol.copy()
                h_new[i] = 0
                h_new[j] = 1
                
                # Compute new C5
                corr_full = np.correlate(h_new, 1 - h_new, mode='full')
                new_c5 = np.max(corr_full) * dx

                # Accept if better
                if new_c5 < c5_bound:
                    h_sol = h_new
                    c5_bound = new_c5
                # Accept with small probability if not better for escaping local minima
                elif np.random.rand() < 0.001:
                    h_sol = h_new
                    c5_bound = new_c5

    return h_sol.tolist(), float(c5_bound), int(n)
