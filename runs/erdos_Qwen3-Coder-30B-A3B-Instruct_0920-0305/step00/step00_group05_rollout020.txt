import numpy as np
import time
import random

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    time_end = start + budget_s - 0.5
    n = 200

    best_h = None
    best_c5 = float('inf')
    dx = 2.0 / n

    max_restarts = 50
    max_local_iters = 10000

    restart_count = 0
    while time.time() < time_end and restart_count < max_restarts:
        # Generate new random binary with half ones
        h = np.zeros(n, dtype=np.float64)
        ones = np.random.choice(n, n//2, replace=False)
        h[ones] = 1.0

        cur_c5 = compute_c5(h)

        # Update global best
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        # Local simulated annealing for this restart
        T0 = 0.05
        for it in range(max_local_iters):
            if time.time() >= time_end:
                break

            # Choose two different indices randomly
            idx1 = np.random.randint(n)
            idx2 = np.random.randint(n)
            while h[idx1] == h[idx2]:
                idx1 = np.random.randint(n)
                idx2 = np.random.randint(n)

            # Propose a swap
            h_new = h.copy()
            h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
            new_c5 = compute_c5(h_new)

            d = new_c5 - cur_c5
            T = max(1e-6, T0 * (1 - it / max_local_iters))
            
            if d <= 0 or np.random.rand() < np.exp(-d / T):
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    
        restart_count += 1

    # Final evaluation to ensure best_c5 is correct
    if best_h is not None:
        final_c5 = compute_c5(best_h)
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n)
