from scipy.special import softmax

def evaluate(x):
    """Evaluate C5 for a given step function h represented as x"""
    dx = 2.0 / len(x)
    # Correlation of x with 1-x using full mode
    conv = np.correlate(x, 1 - x, mode='full')
    max_conv = np.max(conv)
    return max_conv * dx

def project_onto_capped_simplex(y, target_sum):
    """Project y onto the capped simplex: [0,1]^N with sum = target_sum"""
    # Clip to [0, 1]
    y = np.clip(y, 0, 1)
    # Sort clipped values in descending order
    sorted_y = np.sort(y)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_y)
    # Find the threshold index
    n = len(y)
    idx = np.arange(1, n + 1)
    # Check where (sorted_y - (target_sum - cumsum + sorted_y) / idx) > 0
    threshold = (target_sum - cumsum + sorted_y) / idx
    # Handle the boundary case where we should take the last value such that threshold[i] > sorted_y[i]
    # Start from the end
    if sorted_y[-1] >= threshold[-1]:
        mu = threshold[-1]
    else:
        # Find largest k such that sorted_y[k] > threshold[k]
        k = np.argmax(sorted_y < threshold)
        if k == 0:
            mu = threshold[0]
        else:
            mu = threshold[k-1]
    # Projected point
    projected = np.maximum(y - mu, 0)
    return projected

def gradient_of_max(x, beta):
    """Compute gradient of log-sum-exp of overlaps"""
    N = len(x)
    # Compute overlaps for all shifts k (from -N+1 to N-1)
    overlaps = []
    for k in range(-(N-1), N):
        if k >= 0:
            # sum_{i=0}^{N-1-k} x_i * (1 - x_{i+k})
            # x_{i+k} where i = 0..N-1-k, so i+k = k..N-1
            cov = np.sum(x[:N-k] * (1 - x[k:]))
        else:
            # sum_{i=0}^{-k-1} x_i * (1 - x_{i+|k|})  
            # This shift goes into negative portion of x spread
            cov = np.sum(x[-k:] * (1 - x[:-k]))
        overlaps.append(cov)
    # Softmax weights: exp(beta * O_k) / sum(exp(beta * O_j))
    weights = softmax(np.array(overlaps) * beta)
    # Gradient with respect to x_i
    # For each shift k, the contribution to gradient of x_i is:
    # w_k * (1 - x_{i+k}) * I[i <= N-1-k] - w_k * x_{i-k} * I[i >= k]
    # Let's compute component-wise
    grad = np.zeros(N)
    for k in range(-(N-1), N):
        if k >= 0:
            # Gradient contribution to x_i from shift k:
            # Takes term (1 - x_{i+k}) and x_{i-k}
            for i in range(N):
                # Avoid bounds issues
                if i <= N-1-k:
                    grad[i] += weights[k+N-1] * (1 - x[i+k])  # Positive
                if i >= k:
                    grad[i] -= weights[k+N-1] * x[i-k]  # Negative
        else:
            # Same logic with k < 0
            for i in range(N):
                # Check bounds
                if i >= -k:
                    grad[i] += weights[k+N-1] * (1 - x[i+(-k)])  # Positive
                if i <= N - 1 + k:
                    grad[i] -= weights[k+N-1] * x[i-k]  # Negative
    return grad

def swap_move(x, i, j):
    """Swap the elements at positions i and j (binary vectors)"""
    x = x.copy()
    x[i], x[j] = x[j], x[i]
    return x

def fractional_move(x, i, j, delta):
    """Transfer delta from x[j] to x[i] maintaining bounds [0,1]"""
    x = x.copy()
    new_i = x[i] + delta
    new_j = x[j] - delta
    # Update with bounds
    x[i] = np.clip(new_i, 0, 1)
    x[j] = np.clip(new_j, 0, 1)
    return x

def worst_shift_info(x):
    """Find the worst shift k and the set of 'offending' indices"""
    N = len(x)
    max_shift_val = -1
    best_shift = 0
    offending = []
    for k in range(-(N-1), N):
        # Note: indices increase left to right
        if k >= 0:
            cov = np.sum(x[:N-k] * (1 - x[k:]))
        else:
            cov = np.sum(x[-k:] * (1 - x[:-k]))
        if cov > max_shift_val:
            max_shift_val = cov
            best_shift = k
            # Find offending positions for this shift
            offending = []
            if k >= 0:
                for i in range(N-k):
                    if x[i] == 1 and x[i+k] == 0:
                        offending.append(i)
            else:
                for i in range(N+k):
                    if x[i] == 1 and x[i-k] == 0:
                        offending.append(i)
    return best_shift, offending
