import numpy as np

from scipy.special import softmax

def project_to_feasible(h, target_sum):
    """
    Project h onto the set {x : 0 ≤ x ≤ 1, sum(x) = target_sum}
    Uses binary search on the Lagrange multiplier.
    """
    low, high = -1000.0, 1000.0
    x = h.copy()
    while high - low > 1e-12:
        mid = (low + high) / 2.0
        x_new = np.clip(h + mid, 0, 1)
        sum_new = np.sum(x_new)
        if sum_new < target_sum:
            high = mid
        else:
            low = mid
    x = np.clip(h + low, 0, 1)
    # Ensure exact sum by adjusting free components
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-10:
        # Distribute the difference to components that can take it
        free_indices = np.where((x > 0) & (x < 1))[0]
        if len(free_indices) > 0:
            # Add diff evenly among free components
            delta = diff / len(free_indices)
            for i in free_indices:
                x[i] = min(1.0, max(0.0, x[i] + delta))
    return x

def compute_gradient(h, k):
    """Compute gradient of c_k(h) with respect to h."""
    n = len(h)
    grad = np.zeros_like(h)
    # Add (1 - h_{i+k}) where valid
    if k >= 0:
        if k < n:
            grad[k:] += (1 - h[:-k])
    else:
        if -k < n:
            grad[:k] += (1 - h[-k:])
    # Subtract h_{i-k} where valid
    if k <= 0:
        if -k < n:
            grad[-k:] -= h[:-k]
    else:
        if k < n:
            grad[k:] -= h[:-k]
    return grad

def compute_all_gradients(h):
    """Compute all gradients for all possible shifts."""
    n = len(h)
    gradients = []
    for k in range(1 - n, n):
        grad = compute_gradient(h, k)
        gradients.append(grad)
    return gradients
