import numpy as np
import time
from scipy.special import softmax
from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    """Compute C5 using the standard evaluation method."""
    n = len(h)
    dx = 2.0 / n
    h = np.array(h)
    result = 0.0
    for k in range(2*n - 1):
        c = 0.0
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                c += h[i] * (1 - h[j])
        c *= dx
        result = max(result, c)
    return result

def project_to_feasible(h):
    """
    Projects h onto the set {0 <= h[i] <= 1, sum(h) = n/2}
    using binary search on Lagrange multiplier
    """
    n = len(h)
    target_sum = n / 2.0

    # Binary search for lambda
    def lagrange_sum(lam):
        return np.sum(np.clip(h + lam, 0, 1))
    
    low, high = -1.0, 1.0
    for _ in range(100):
        mid = (low + high) / 2
        s = lagrange_sum(mid)
        if s < target_sum:
            high = mid
        else:
            low = mid
    lam = (low + high) / 2
    
    # Compute projection
    proj = np.clip(h + lam, 0, 1)
    
    # Adjust to hit exact sum
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-10:
        # If needed, adjust by adding to smallest values
        idx = np.argsort(proj)
        idx = idx[:int(np.floor(abs(diff) * n))]  # Choose indices to adjust
        if diff > 0:
            proj[idx] = np.clip(proj[idx] + diff / len(idx), 0, 1)
        else:
            proj[idx] = np.clip(proj[idx] + diff / len(idx), 0, 1)
    return proj

def compute_correlation(h, mode='full'):
    # The correlation array needs to be consistent with the overlap computation
    # We want sum(h[i] * (1 - h[i+k])) for all valid shifts k
    n = len(h)
    dx = 2.0 / n
    correlation = np.zeros(2*n - 1)
    one_minus_h = 1 - h
    for k in range(2*n - 1):
        sum_val = 0.0
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                sum_val += h[i] * one_minus_h[j]
        correlation[k] = sum_val * dx
    return correlation

def grad_c_k(h, k):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    # Add contribution for (1 - h[j+k]) with zero padding
    if k >= 0:
        for i in range(n - k):
            grad[i] += dx * (1 - h[i + k])
    else:
        k = -k
        for i in range(k, n):
            grad[i] += dx * (1 - h[i - k])
    # Subtract contribution for h[j-k] with zero padding
    if k >= 0:
        for i in range(k, n):
            grad[i] -= dx * h[i - k]
    else:
        k = -k
        for i in range(n - k):
            grad[i] += dx * h[i + k]
    return grad

def compute_gradient(h, c, beta=1.0):
    # Compute w = softmax(beta * c)
    w = softmax(beta * c)
    # Compute gradient of loss = log-sum-exp
    n = len(h)
    g = np.zeros(n)
    for k in range(len(c)):
        g += w[k] * grad_c_k(h, k - (n - 1))
    return g

    # Additional random starts
    for _ in range(10):
        if time.time() - t0 > 0.95 * budget_s:

        h = project_to_feasible(h)
        c = compute_correlation(h)
        c5 = np.max(c)

            best_c5 = c5
            best_h = h.copy()
    
    # Smoothed gradient descent
    beta_schedule = [0.5, 2, 5, 15, 40, 80]
    max_iter_per_beta = 150

        for iteration in range(max_iter_per_beta):

            g = compute_gradient(best_h, c, beta)
            
            # Back-tracking line search
            alpha = alpha0 / (1.0 + iteration)
            converged = False
            while not converged:

                    break
                h_new = best_h - alpha * g
                h_new = project_to_feasible(h_new)
                c_new = compute_correlation(h_new)
                c5_new = np.max(c_new)
                if c5_new <= c5:

    # Binary rounding refinement
    if time.time() - t0 < 0.95 * budget_s:
        h_binary = np.round(best_h).astype(float)

        c_new = compute_correlation(h_binary)
        c5_new = np.max(c_new)

            best_c5 = c5_new
