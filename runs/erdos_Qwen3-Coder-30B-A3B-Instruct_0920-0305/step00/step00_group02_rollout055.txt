import numpy as np

def decode(z):
    # Softmax normalization to ensure sums are correct
    M = (len(z) - 1) // 2
    z_ell = z[:M]
    z_g = z[M:]
    
    ell = np.exp(z_ell)
    g = np.exp(z_g)
    
    ell = ell / np.sum(ell)  # total integral = 1
    g = g / np.sum(g)       # total gap = 1
    
    return ell, g

def build_intervals(ell, g):
    # Build interval starts and ends from lengths and gaps
    a = [g[0]]
    for i in range(1, len(ell)):
        a.append(a[-1] + ell[i-1] + g[i])
    b = [a[i] + ell[i] for i in range(len(ell))]
    return np.array(a), np.array(b)

def candidate_shifts(a, b):
    # Generate list of candidate shift values
    E = np.concatenate([a, b])
    T = set()
    for i in range(len(E)):
        for j in range(len(E)):
            t = abs(E[i] - E[j])
            if t <= 2:
                T.add(t)
    return sorted(list(T))

def intersection_length(t, a, b):
    # Calculate the intersection length of S and S+t
    total = 0
    for i in range(len(a)):
        # Intersection of [a[i], b[i]] and [a[i]-t, b[i]-t]
        # If t < 0, compute [a[i], b[i]] ∩ [a[i]+t, b[i]+t]
        start = max(a[i], a[i] - t)
        end = min(b[i], b[i] - t)
        if start <= end:
            total += end - start
    return total

def objective(z):
    ell, g = decode(z)
    a, b = build_intervals(ell, g)
    shifts = candidate_shifts(a, b)
    min_I = float('inf')
    for t in shifts:
        I_t = intersection_length(t, a, b)
        if I_t < min_I:
            min_I = I_t
    # Return 1 - min_I as the C5 bound
    return 1 - min_I

def local_LP(z_best, max_iter=5, delta=0.001):
    # Run LP refinement around the best solution
    
    ell, g = decode(z_best)
    a, b = build_intervals(ell, g)
    
    shifts = candidate_shifts(a, b)
    min_I = float('inf')
    best_t = 0
    for t in shifts:

        if I_t < min_I:

            best_t = t
    
    # Set up the LP for the active interval
    # For the active t, model I(t) as linear in parameters
    
    # Setup variables for LP
    num_vars = len(a) * 2  # each a, b is a variable
    # The variable is actually a vector [a_0, b_0, a_1, b_1, ...)
    x_vars = []
    for i in range(len(a)):
        x_vars.append(a[i])
        x_vars.append(b[i])
    
    # Build linear equations for the current active t
    # But we're just going to do one refinement for simplicity
    
    # Run one iteration of perturbing slightly:
    new_z = z_best.copy()
    new_z[:len(z_best)//2] += 0.001 * np.random.rand(len(z_best)//2)
    new_z[len(z_best)//2:] += 0.001 * np.random.rand(len(z_best)//2)
    
    # We will apply this adjustement safely and double check if it leads to a better result
    return new_z

def gridify(intervals, n):
    # Discretize intervals to a binary grid of n points
    a, b = build_intervals(intervals[0], intervals[1])
    
    dx = 2.0 / n
    h = np.zeros(n)
    
    # Fill in the grid
    for i in range(n):
        x = i * dx + dx / 2  # center of grid point
        for j in range(len(a)):
            if a[j] <= x <= b[j]:
                h[i] = 1
                break
    
    # Adjust to make sum equal to n/2
    current_sum = np.sum(h)
    target = n / 2
    if current_sum < target:
        # Try to find a zero to turn into 1
        idx1 = np.where(h == 0)[0]
        if len(idx1) > 0:
            # Find index closest to middle
            mid = n // 2
            idxsel = np.argmin(np.abs(idx1 - mid))
            h[idx1[idxsel]] = 1
    elif current_sum > target:
        # Find a 1 to turn into 0
        idx0 = np.where(h == 1)[0]
        if len(idx0) > 0:
            idxsel = np.argmin(np.abs(idx0 - mid))  # e.g. middle
            h[idx0[idxsel]] = 0
            
    return h

def binary_hill_climb(h, budget_s):
    # Simplified hill climbing over the grid
    # We take a fixed number of steps (smaller loop for efficiency)
    total_time = 0
    start_time = time.time()
    
    def eval_current(h):
        dx = 2.0 / len(h)
        C5 = max(np.correlate(h, 1-h, mode="full")) * dx
        return C5
    
    best_h = h.copy()
    best_C5 = eval_current(h)
    
    # Perform swaps for a fixed number of times
    max_swaps = 200  # limit based on budget
    for _ in range(max_swaps):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        i, j = np.random.choice(len(h), 2, replace=False)
        if h[i] == h[j]:
            continue
            
        # Swap
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        
        new_C5 = eval_current(h_new)
        
        if new_C5 < best_C5:
            h = h_new
            best_h = h_new
            best_C5 = new_C5
    
    return best_h, best_C5

def run(seed=42, budget_s=600):
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Parameters
    M = 6  # number of intervals
    n = 400  # discretization grid size
    dx = 2.0 / n
    
    # Initial search range for DE
    bounds = [(-5, 5)] * (2 * M + 1)
    
    # 70 % time for global DE
    de_time = 0.7 * budget_s
    de_start = elapsed()
    
    # Run DE with multiple restarts
    best_z = None
    best_score = float('inf')
    
    # First run
    res = differential_evolution(objective, bounds, maxiter=1000, popsize=210, seed=seed)
    if res.fun < best_score:
        best_score = res.fun
        best_z = res.x
    
    # Second run with different settings
    if elapsed() < de_time:
        res2 = differential_evolution(objective, bounds, maxiter=1000, popsize=210, seed=seed+1)
        if res2.fun < best_score:
            best_score = res2.fun
            best_z = res2.x
    
    if elapsed() < de_time:
        res3 = differential_evolution(objective, bounds, maxiter=1000, popsize=210, seed=seed+2)
        if res3.fun < best_score:
            best_score = res3.fun
            best_z = res3.x
    
    # Local refinement with LP
    if elapsed() < budget_s * 0.9:
        for _ in range(3):
            try:
                best_z = local_LP(best_z)  # Simple local refinement
                current_score = objective(best_z)
                if current_score < best_score:
                    best_score = current_score
            except:
                break
    
    # Convert to intervals
    ell, g = decode(best_z)
    a, b = build_intervals(ell, g)
    
    # Discretize
    h = gridify((ell, g), n)
    
    # Final binary hill climbing
    if elapsed() < budget_s:
        h, C5 = binary_hill_climb(h, budget_s - elapsed())
    else:
        C5 = max(np.correlate(h, 1-h, mode="full")) * dx
    
    return h.tolist(), C5, len(h)
