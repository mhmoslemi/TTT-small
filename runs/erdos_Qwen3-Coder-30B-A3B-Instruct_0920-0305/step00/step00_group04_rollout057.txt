import numpy as np

def evaluate_erdos_solution(h):
    """Provided evaluation function for C5."""
    # This is a placeholder; in practice it would be imported from the environment
    # We'll simulate it for now
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    # Correlate h with h1 (full mode)
    corr = np.correlate(h, h1, mode='full')
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    """Provide initial construction via the global variable."""
    # Placeholder; assume it's available as a global variable in the environment

def fft_correlation(h1, h2, dx):
    """Use FFT for fast correlation computation"""
    n = len(h1)
    L = 2**int(np.ceil(np.log2(2*n - 1)))
    H1 = fftpack.fft(h1, L)
    H2 = fftpack.fft(h2, L)
    corr = np.real(fftpack.ifft(H1 * np.conj(H2)))
    return corr[:2*n-1] * dx


def binary_SA(n, max_time_s, seed):
    """Binary Simulated Annealing to minimize max overlap"""
    np.random.seed(seed)
    max_iter = 1000000
    t0 = time.time()
    T0 = 0.05
    alpha = 0.9995
    target_N = int(0.3807 * n / 2)
    best_h = None
    best_C5 = float('inf')
    done = False

    def evaluate_binary(h_b):
        h = h_b.astype(float)
        return evaluate_erdos_solution(h)

    def get_swap_moves(h):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            return []
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        return [(i, j)]

    # Start with a random binary string with exactly n/2 ones
    h = np.zeros(n, dtype=int)
    ones_indices = np.random.choice(n, size=n//2, replace=False)
    h[ones_indices] = 1

    T = T0
    last_improvement = 0
    for i in range(max_iter):
        if time.time() - t0 > max_time_s * 0.7:
            break

        moves = get_swap_moves(h)
        if not moves:
            continue
            
        (i1, i2) = moves[0]
        h_new = h.copy()
        h_new[i1], h_new[i2] = h_new[i2], h_new[i1]

        c_old = evaluate_binary(h)
        c_new = evaluate_binary(h_new)
        
        delta = c_new - c_old
        accept = delta <= 0 or np.random.rand() < np.exp(-delta / T)
        
        if accept:
            h = h_new
            if c_new < best_C5:
                best_C5 = c_new
                best_h = h.copy()
                last_improvement = i

        # Cool down
        if (i - last_improvement) > 5000:
            T = max(T * alpha, 1e-6)
            
    # If we didn't find a valid move, use latest h
    if best_h is None or not (best_h == h).all():
        c = evaluate_binary(h)
        if c < best_C5:
            best_C5 = c
            best_h = h.copy()

    return best_h, best_C5


def project_simplex(y, S=0.5):
    """Project y onto the capped simplex {x : sum x = S, 0≤x≤1}"""
    n = len(y)
    y = np.clip(y, 0, 1)
    s = np.sum(y)
    if np.abs(s - S) < 1e-10:
        return y
    elif s > S:
        # Find lambda such that sum clip(y - lambda, 0, 1) = S
        def sum_func(lambda_val):
            return np.sum(np.clip(y - lambda_val, 0, 1)) - S

        lambda_low = np.min(y) - 1
        lambda_high = np.max(y) + 1
        tol = 1e-12
        while lambda_high - lambda_low > tol:
            mid = (lambda_low + lambda_high) / 2
            val = sum_func(mid)
            if val > 0:
                lambda_low = mid
            else:
                lambda_high = mid
        lambda_opt = (lambda_low + lambda_high) / 2
        return np.clip(y - lambda_opt, 0, 1)
    else:  # s < S
        # Add the shortfall in uniform way to all non-zero, non-one components
        excess = S - s
        if excess <= 0:
            return y
        add_per_unit = excess / np.sum(y < 1)
        y = np.clip(y + add_per_unit, 0, 1)
        # Re-adjust for sum < S due to clipping
        if np.abs(np.sum(y) - S) > 1e-10:
            # Perform one final adjustment to meet the sum constraint strongly
            y = np.clip(y + (S - np.sum(y)) / np.count_nonzero(y), 0, 1)
        return y


def continue_pattern(h_binary, target_c5):
    """
    Refine a binary vector to a continuous one using projected gradient.
    """
    n = len(h_binary)
    dx = 2.0 / n
    h = h_binary.astype(float)
    max_iter = 1000
    gamma = 5.0
    gamma_increase = 1.2
    max_gamma = 200.
    alpha = 0.1 * dx
    alpha_decay = 0.9
    min_improvement = 1e-8
    last_improvement_step = 0
    step = 0
    last_c5 = evaluate_erdos_solution(h)
    t0 = time.time()
    safety_margin = 5.0  # seconds

    def compute_pseudo_overlap(h):
        h1 = 1 - h
        # Full correlation via FFT
        corr = fft_correlation(h, h1, dx)
        return np.max(corr)

    def gradient_of_pseudo_overlap(h, gamma):
        h1 = 1 - h
        L = 2**int(np.ceil(np.log2(2*n - 1)))
        
        H = fftpack.fft(h, L)
        H1 = fftpack.fft(h1, L)
       _corr = fftpack.ifft(H * np.conj(H1)).real
        corr = _corr[:2*n-1] * dx

        # Softmax of gradients
        corr_vec = corr
        exp_corr = np.exp(gamma * corr_vec)
        exp_sum = np.sum(exp_corr)
        weights = exp_corr / exp_sum
        
        # Compute ∇f_ℓ = dx * (indicator forward term - backward term)
        # We actually compute efficient convolutions
        grad = np.zeros_like(h)
        for k in range(len(corr_vec)):
            if k >= n:
                continue  # Avoid aliasing from FFT
            if k == 0:
                # Direct computation for k=0 shows same as full convolution, but with the logic here it's all-zero except long ones.
                # Better to work with these explicitly
                continue
            # Forward
            start_shift = k
            # Backward term
            end_shift = k
            # Only partial convolution
            # Compute: prod_i h[i] * (1 - h[i+k]) | only valid i where i+k < n
            for i in range(n - k):
                grad[i] += weights[k] * (1 - h[i + k]) * dx
            for i in range(k, n):
                grad[i] -= weights[n - 1 - k] * h[i - k] * dx
        return grad

    while step < max_iter and time.time() - t0 < 600 - safety_margin:
        grad = gradient_of_pseudo_overlap(h, gamma)
        new_h = h - alpha * grad
        new_h = project_simplex(new_h, n/2)

        # Check actual value
        new_c5 = compute_pseudo_overlap(new_h)

        if new_c5 < last_c5:
            h = new_h
            last_c5 = new_c5
            last_improvement_step = step
            # Reduce step size if it's taking off
            if alpha < 0.1 * dx:
                alpha *= 1.1
        else:  # Not improved
            if step - last_improvement_step > 10:
                if gamma < max_gamma:
                    gamma *= gamma_increase
                else:
                    if alpha > 1e-10:
                        alpha *= alpha_decay
                    else:
                        break
        step += 1

    # Recompute exact value
    final_c5 = evaluate_erdos_solution(h)
    return h, final_c5


def run(seed=42, budget_s=600, **kwargs):

        h_orig_sum = np.sum(h_original)
        h_scaled = (n / 2) * h_original / h_orig_sum
        # Convert to binary
        h_binary = (h_scaled > 0.5).astype(int)
        # Adjust sum to exactly n/2
        current_sum = np.sum(h_binary)
        if current_sum > n/2:
            # Find indices to set to 0
            diff = current_sum - n/2
            zero_indices = np.random.choice(np.where(h_binary == 1)[0], size=int(diff), replace=False)
            for idx in zero_indices:
                h_binary[idx] = 0
        elif current_sum < n/2:
            # Find indices to set to 1
            diff = n/2 - current_sum
            one_indices = np.random.choice(np.where(h_binary == 0)[0], size=int(diff), replace=False)
            for idx in one_indices:
                h_binary[idx] = 1
    else:
        h_original = initial
        h_binary = (h_original > 0.5).astype(int)
        # Adjust sum
        current_sum = np.sum(h_binary)
        if current_sum > n/2:
            diff = current_sum - n/2
            zero_indices = np.random.choice(np.where(h_binary == 1)[0], size=int(diff), replace=False)
            for idx in zero_indices:
                h_binary[idx] = 0
        elif current_sum < n/2:
            diff = n/2 - current_sum
            one_indices = np.random.choice(np.where(h_binary == 0)[0], size=int(diff), replace=False)
            for idx in one_indices:
                h_binary[idx] = 1

    best_h = h_binary.copy()
    best_c5 = evaluate_erdos_solution(h_binary)

    # Phase A: Binary SA
    print(f"Starting Phase A binary optimization with n={n}...")
    restarts = 0
    t_a = 0.7 * budget_s
    best_h_a = best_h.copy()
    best_c5_a = best_c5
    start_time = time.time()

            h_from_a, c5_from_a = binary_SA(n, t_a, seed + restarts)

            best_h = h_refined
            best_c5 = c5_refined

    # Final projection
    final_h = project_simplex(best_h, n/2)
    final_c5 = evaluate_erdos_solution(final_h)

    return (final_h.copy(), float(final_c5), int(n))
