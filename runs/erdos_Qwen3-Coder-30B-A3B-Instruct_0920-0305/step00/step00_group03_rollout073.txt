import numpy as np
import time
import math

def project_onto_simplex_box(v, target):
    # binary search for lambda
    lo = -1.0
    hi = 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return proj

def max_overlap(h):
    n = len(h)
    dx = 2.0 / n
    # correlation returns array len 2n-1
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_subgradient(h, shift):
    n = len(h)
    grad = np.zeros_like(h)
    start = max(0, -shift)
    end = min(n - 1, n - 1 - shift)
    if start > end:
        return grad
    i_slice = slice(start, end + 1)
    j_slice = slice(start + shift, end + shift + 1)
    grad[i_slice] += 1.0 - h[j_slice]
    grad[j_slice] -= h[i_slice]
    return grad

def optimize_start(h0, time_limit):
    h = np.asarray(h0, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_val = max_overlap(h)
    # parameters
    start_time = time.time()
    iteration = 0
    step0 = 0.5
    # Use schedule for step and tolerance
    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed > time_limit - 0.1:
            break
        iteration += 1
        # compute correlation and identify worst shift(s)
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = corr.max()
        # compute set of shifts within tolerance 1e-12 relative
        tol = 1e-12 * max_val + 1e-14
        indices = np.where(corr >= max_val - tol)[0]
        # compute gradient as sum of subgradients of these shifts
        grad = np.zeros_like(h)
        for idx in indices:
            shift = idx - (n - 1)
            # add subgradient for this shift
            start = max(0, -shift)
            end = min(n - 1, n - 1 - shift)
            if start > end:
                continue
            i_slice = slice(start, end + 1)
            j_slice = slice(start + shift, end + shift + 1)
            grad[i_slice] += 1.0 - h[j_slice]
            grad[j_slice] -= h[i_slice]
        # average gradient if multiple worst shifts
        if len(indices) > 1:
            grad /= len(indices)
        # step size decreasing
        eta = step0 / math.sqrt(iteration)
        # propose new h
        h_new = project_onto_simplex_box(h - eta * grad, target)
        new_val = max_overlap(h_new)
        # accept if improved
        if new_val < best_val - 1e-12:
            best_val = new_val
            best_h = h_new.copy()
        # accept move if it does not increase objective
        cur_val = max_overlap(h)
        if new_val <= cur_val + 1e-12:
            h = h_new
        else:
            # reduce step size on failure
            step0 *= 0.7
            # optionally random perturbation if step too small
            if step0 < 1e-5:
                # random jitter
                perturb = (np.random.rand(n) - 0.5) * 0.01
                h_j = project_onto_simplex_box(h + perturb, target)
                h = h_j
                step0 = 0.5
    return best_h, best_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    global initial_h_values
    # Determine number of points to use (prefer N up to ~150)
    original = np.array(initial_h_values, dtype=np.float64)
    n0 = len(original)
    # Choose target N: either n0 or 120 if n0 < 120 else n0
    N = max(n0, 120)
    # Build initial candidates
    candidates = []

    # 1: upsample original if needed
    if N != n0:
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, N, endpoint=False)
        up = np.interp(x_new, x_old, original)
        up = project_onto_simplex_box(up, N/2.0)
        candidates.append(up)
    else:
        candidates.append(original)

    # 2: triangular function
    xs = np.linspace(0.0, 2.0, N, endpoint=False) + (1.0 / N)  # shift tocentre
    tri = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))
    tri = project_onto_simplex_box(tri, N/2.0)
    candidates.append(tri)

    # 3: constant 0.5
    const = np.full(N, 0.5, dtype=np.float64)
    const = project_onto_simplex_box(const, N/2.0)  # should already satisfy
    candidates.append(const)

    # 4: random (few)
    for _ in range(3):
        rand = np.random.rand(N)
        rand = project_onto_simplex_box(rand, N/2.0)
        candidates.append(rand)

    # 5: binary (random half ones)
    bin_vec = np.zeros(N, dtype=np.float64)
    bin_vec[:N//2] = 1.0
    np.random.shuffle(bin_vec)
    candidates.append(bin_vec)

    # Time allocation
    total_time = budget_s - 4.0  # buffer
    start_time_all = time.time()
    best_h_overall = None
    best_val_overall = float('inf')
    n_candidates = len(candidates)
    # allocate equal time slices initially, may adjust
    time_per = total_time / n_candidates
    for h0 in candidates:
        remaining = total_time - (time.time() - start_time_all)
        if remaining <= 0:
            break
        allotted = min(time_per, remaining)
        h_opt, val_opt = optimize_start(h0, allotted)
        if val_opt < best_val_overall:
            best_val_overall = val_opt
            best_h_overall = h_opt.copy()
    # final refinement with remaining time
    remaining = total_time - (time.time() - start_time_all)
    if remaining > 2.0 and best_h_overall is not None:
        h_refined, val_refined = optimize_start(best_h_overall, remaining)
        if val_refined < best_val_overall:
            best_val_overall = val_refined
            best_h_overall = h_refined.copy()

    # Return results
    return (list(best_h_overall), float(best_val_overall), len(best_h_overall))
