import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time
from math import sqrt

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant for a given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode='full')
    # The maximum value is the one we want
    max_overlap = np.max(correlation) * dx
    return max_overlap

def initial_h_values():
    """Return initial construction for the problem"""
    # Using the initial construction from the problem
    n_points = 45
    h = np.zeros(n_points)
    # Simple construction that should be reasonably good
    for i in range(n_points):
        x = i * 2.0 / n_points
        if x < 1.0:
            h[i] = min(1.0, x * 2.0)
        else:
            h[i] = max(0.0, 2.0 - (x - 1.0) * 2.0)
    # Normalize to have integral = 1
    h = h * (n_points / (2.0 * np.sum(h)))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45  # Small enough to be fast
    dx = 2.0 / n_points
    max_iter = 10000
    tol = 1e-6
    
    # Start with initial construction
    h = initial_h_values()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Define optimization variables
    h_vars = cp.Variable(n_points)
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == n_points / 2  # integral = 1
    ]
    
    # For optimization, use gradient-based method as such problems are smooth
    # We will use scipy.optimize with bounds
    
    # Objective function for scipy minimize
    def objective(h_values):
        return evaluate_erdos_solution(h_values)
        
    # Callback to track best solution
    def callback(h_values):
        nonlocal best_h, best_c5
        c5 = evaluate_erdos_solution(h_values)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_values.copy()
    
    # Bounds for h[i] in [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization using scipy minimize approach
    iterations = 0
    
    # Add small random perturbations to avoid local minimums
    # This will make multiple attempts with different starting points
    # But within budget, we do multiple runs
    for attempt in range(10):  # 10 different starting points
        if time.time() - start_time > budget_s - 2:
            break
            
        # Create a random starting point in valid ranges
        h0 = np.random.rand(n_points)
        # Adjust integral to exactly 1  
        h0 = h0 * (n_points / (2.0 * np.sum(h0)))
        # Clip to [0,1]
        h0 = np.clip(h0, 0, 1)
        # Ensure normalization (exact integral = 1)
        h0 = h0 * (n_points / (2.0 * np.sum(h0)))
        
        # Minimize with bounds using L-BFGS-B
        res = minimize(objective, h0, method='L-BFGS-B', bounds=bounds, 
                      callback=callback, options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-6})
        
        iterations += 1
        if time.time() - start_time > budget_s - 2:
            break
    
    # Final check if there's an improvement in the last few steps by doing more careful local search
    # Create a cache for objective values to avoid recomputing
    def fast_evaluate(h_vals):
        return evaluate_erdos_solution(h_vals)
    
    # Do a final optimization using a local search method
    # Try another random set to see if we can do better
    h_new = np.random.rand(n_points)

    try:
        result = minimize(fast_evaluate, h_new, method='L-BFGS-B', bounds=bounds,

    # Ensure that the result is within bounds (though it should be)
