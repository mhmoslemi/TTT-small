import numpy as np
import scipy.fftpack as fftpack
import time
from scipy.optimize import minimize
from math import floor

def evaluate_erdos_solution(h):
    """Evaluate C5 using the official evaluator."""
    h = np.array(h)
    n = len(h)
    dx = 2.0 / n
    # Compute convolution of h with (1-h)
    corr = np.correlate(h, 1-h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def compute_C5(h, dx):
    """Fast computation of C5 using FFT."""
    h = np.array(h)
    n = len(h)
    # Correlation using FFT
    L = 2 * n - 1
    L = 1 << (L - 1).bit_length()  # next power of 2
    H = fftpack.fft(h, L)
    G = fftpack.fft(1 - h, L)
    corr = fftpack.ifft(H * np.conj(G)).real
    corr = corr[:2*n-1]
    return np.max(corr) * dx

def binary_to_continuous(b):
    """Convert binary vector to continuous with sum constraint."""
    b = np.array(b)
    n = len(b)
    h = b.astype(float)
    sum_diff = np.sum(h) - n / 2
    h += -sum_diff / n  # Adjust to sum to n/2
    h = np.clip(h, 0, 1)
    return h

def project_to_simplex(y):
    """Project vector y onto the capped simplex."""
    y = np.array(y)
    n = len(y)
    # Use bisection to find lambda
    def S(lam):
        return np.sum(np.clip(y - lam, 0, 1)) - n/2
    
    low, high = np.min(y) - 1, np.max(y) + 1
    while S(low) * S(high) > 0:
        low -= 1
        high += 1
    
    for _ in range(60):
        mid = (low + high) / 2
        s = S(mid)
        if s > 0:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2
    x = np.clip(y - lam, 0, 1)
    return x

def smooth_max(f, gamma):
    """Smooth approximation of max function."""
    return np.log(np.sum(np.exp(gamma * f))) / gamma

def gradient_smooth_max(f, gamma):
    """Gradient of smooth max."""
    w = np.exp(gamma * f)
    w = w / np.sum(w)
    # Approximate gradient; max element will have most weight
    return w

def gradient_descent_refinement(b, max_iter=200, gamma_start=5, gamma_increase=1.5, gamma_max=200):
    """Refine binary solution to continuous with gradient descent."""
    n = len(b)
    h = binary_to_continuous(b)
    dx = 2.0 / n
    gamma = gamma_start
    
    for it in range(max_iter):
        # Evaluate correlation for all shifts
        corr_all = []
        for shift in range(-n+1, n):
            corr = 0
            for i in range(n):
                j = i + shift
                if 0 <= j < n:
                    corr += h[i] * (1 - h[j])
            corr_all.append(corr)
        f_vals = np.array(corr_all) * dx
        C5_current = np.max(f_vals)
        
        # Compute smooth max
        Phi = smooth_max(f_vals, gamma)
        g = gradient_smooth_max(f_vals, gamma)
        
        # Update
        grad = np.zeros(n)
        for shift_idx, shift in enumerate(range(-n+1, n)):
            for i in range(n):
                j = i + shift
                if 0 <= j < n:
                    if shift < 0:
                        grad[i] += g[shift_idx] * (1 - h[j])
                    else:
                        grad[i] += g[shift_idx] * (-h[j - shift])
        
        # Simple update
        alpha = 0.1 * dx
        h_new = h - alpha * grad
        h_new = project_to_simplex(h_new)
        h = h_new
        
        if gamma < gamma_max:
            gamma *= gamma_increase
    return h

def binary_sa_search(n, time_limit=400):
    """Perform binary simulated annealing search."""
    dx = 2.0 / n
    h_current = np.random.permutation([1]*int(n/2) + [0]*int(n/2))
    C5_best = compute_C5(h_current, dx)
    h_best = h_current.copy()
    
    def accept(C_new, C_old, temp):
        if C_new <= C_old:
            return True
        return np.random.rand() < np.exp((C_old - C_new) / temp)
    
    start = time.time()
    temp = 0.05
    cooling_rate = 0.9995
    it = 0
    while time.time() - start < time_limit:
        it += 1
        
        # Make a move
        ones = np.where(h_current == 1)[0]
        zeros = np.where(h_current == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            h_current = np.random.permutation([1]*int(n/2) + [0]*int(n/2))
            continue
            
        idx1 = np.random.choice(ones)
        idx2 = np.random.choice(zeros)
        h_new = h_current.copy()
        h_new[idx1] = 0
        h_new[idx2] = 1
        
        C_new = compute_C5(h_new, dx)
        C_old = compute_C5(h_current, dx)
        
        if accept(C_new, C_old, temp):
            h_current = h_new
            if C_new < C5_best:
                C5_best = C_new
                h_best = h_new.copy()
        
        temp *= cooling_rate
    
    return h_best, C5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 60
    dx = 2.0 / n
    
    # Phase A: Binary Simulated Annealing
    t_left = budget_s * 0.7
    h_binary, C5_binary = binary_sa_search(n, t_left)
    
    if C5_binary <= 0.3807:
        h_final = binary_to_continuous(h_binary)
        C5_final = evaluate_erdos_solution(h_final)
        return h_final, C5_final, n
    
    # Phase B: Continuous refinement
    h_refined = gradient_descent_refinement(h_binary, max_iter=200)
    C5_refined = evaluate_erdos_solution(h_refined)
    
    # Phase C: Optional MILP solver
    # We skip MILP for now
    
    return h_refined, C5_refined, n
