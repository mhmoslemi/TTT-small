import numpy as np
from scipy.optimize import differential_evolution
from scipy.optimize import minimize
import time
import math

# Precomputed initial solution
initial_h_values = None

def evaluate_h(h):
    """Evaluate C5 for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # The correlation is of length 2*n_points - 1, we need the middle part
    # corresponding to shifts from -1 to 1 (i.e., from 0 to n_points-1
    # and from n_points-1 to 2*n_points-2)
    middle_idx = len(corr) // 2
    max_val = max(corr[middle_idx:middle_idx + n_points]) * dx
    return max_val

def obj_func(h):
    """Objective function to minimize"""
    n_points = len(h)
    # Minimize the negative of C5 (to turn it into a minimization problem)
    return evaluate_h(h)

def callback(x, *args):

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    n_points = 45
    try:
        h0 = initial_h_values
        if len(h0) != n_points:
            h0 = np.array(h0)
            # interpolate to desired length
            x = np.linspace(0, 2, len(h0))
            x_new = np.linspace(0, 2, n_points)
            h0 = np.interp(x_new, x, h0)
    except:
        # if initial_h_values is not available, start with a simple pattern
        h0 = np.ones(n_points) * 0.5
    
    # Constraint: sum(h) * dx = 1, so sum(h) = n_points / 2
    def constraint_fun(h):
        return np.sum(h) - n_points / 2.0
    
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint for constraint_fun
    constraint = {'type': 'eq', 'fun': constraint_fun}

    # Initial evaluation
    h_current = h0.copy()
    current_c5 = evaluate_h(h_current)
    best_h = h_current.copy()
    best_c5 = current_c5

    # Use a local optimizer with callbacks
    def minimize_chunk(start_idx, end_idx, sub_h0, num_iters=50):
        nonlocal h_current, current_c5, best_h, best_c5
        
        # Optimizer options
        options = {'maxiter': num_iters, 'disp': False}
        
        # Local optimization
        result = minimize(obj_func, sub_h0, method='L-BFGS-B',
                          bounds=bounds[start_idx:end_idx],
                          options=options, callback=callback)
        if not result.success:
            return
        updated_h = h_current.copy()
        updated_h[start_idx:end_idx] = result.x
        converted_c5 = evaluate_h(updated_h)
        if converted_c5 < best_c5:
            best_h = updated_h.copy()
            best_c5 = converted_c5

    # Use a global optimization approach
    # Use differential evolution over chunked subspace with overlapping windows
    try:
        # Try differential evolution with a reasonable budget
        # First, we'll do a basic discretized optimization
        
        # optimize the whole vector by chunks in subspaces using grids
        # We do full optimization over a small number of subsampled elements
        opt_indices = np.linspace(0, n_points-1, min(20, n_points)).astype(int)
        
        # Group the indices and solve with local optimization
        for _ in range(1000):  # Loop with time checking
            if time.time() - start_time > budget_s - 1:
                break
            
            # Copy current solution
            local_h = h_current.copy()
            
            # Select random chunk to optimize
            num_to_optimize = min(10, len(opt_indices) - 1)
            opt_indices_sub = np.random.choice(opt_indices, size=num_to_optimize, replace=False)
            opt_indices_sub = np.sort(opt_indices_sub)
            
            # Make local copy of h
            old_vals = local_h[opt_indices_sub]
            local_h[opt_indices_sub] = old_vals
            
            # Local optimization for selected block
            result = minimize(obj_func, local_h[opt_indices_sub], method='L-BFGS-B', 
                              bounds=[(0,1) for _ in range(len(opt_indices_sub))])
            if result.success:
                local_h[opt_indices_sub] = result.x
                
                # Evaluate
                c5_new = evaluate_h(local_h)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = local_h.copy()
            
            # Update current solution
            h_current = best_h.copy()
            
    except Exception:
        pass

    # Final check
    if time.time() - start_time > budget_s - 1:
        pass
    
    # Ensure final solution satisfies constraint
    h_final = best_h.copy()
    h_final = h_final * (n_points / 2.0) / np.sum(h_final)
    
    # Evaluate the final result
    final_c5 = evaluate_h(h_final)
    
    return (list(h_final), final_c5, n_points)
