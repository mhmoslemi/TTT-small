def decode(z):
    """Decode parameters from raw vector z to lengths and gaps"""
    M = len(z) // 2
    z_ell = z[:M]
    z_g = z[M:]
    ell = np.exp(z_ell) / np.sum(np.exp(z_ell))
    g = np.exp(z_g) / np.sum(np.exp(z_g))
    return ell, g

def build_intervals(ell, g):
    """Build interval positions from lengths and gaps"""
    a = [g[0]]
    for i in range(1, len(ell)):
        a.append(a[-1] + ell[i-1] + g[i])
    return np.array(a), np.array(a) + ell

def candidate_shifts(a, b):
    """Generate all candidate shifts"""
    E = set()
    for i in range(len(a)):
        E.add(a[i])
        E.add(b[i])
    E = sorted(list(E))
    T = {0}
    for i in range(len(E)):
        for j in range(i, len(E)):
            diff = abs(E[i] - E[j])
            if diff <= 2:
                T.add(diff)
    return sorted(list(T))

def intersection_length(t, a, b):
    """Calculate intersection length at shift t"""
    if t == 0:
        return np.sum(b - a)
    total = 0
    for i in range(len(a)):
        left = max(a[i], a[i] - t)
        right = min(b[i], b[i] - t)
        if right > left:
            total += (right - left)
    return total

def compute_C5_from_intervals(ell, g):
    """Compute C5 from interval lengths and gaps"""
    a, b = build_intervals(ell, g)
    T = candidate_shifts(a, b)
    I_vals = [intersection_length(t, a, b) for t in T]
    min_I = min(I_vals)
    return 1 - min_I

def objective(z):
    """Objective function to minimize"""
    ell, g = decode(z)
    return compute_C5_from_intervals(ell, g)

    # Use bounds from [-5, 5] for each parameter
    bounds = [(-5, 5)] * (2 * M + 1)
    
    # Logging
    best_C5 = float('inf')
    best_z = None

        nonlocal best_C5, best_z
        C5 = objective(xk)
        if C5 < best_C5:
            best_C5 = C5
            best_z = xk.copy()
    
    # Run DE 
    try:
        result = differential_evolution(
            objective, bounds, maxiter=2000, popsize=15*(2*M+1), 
            callback=callback, seed=42, disp=False, polish=False,
            init='latinhypercube'  # better for small dimensions
        )
        if result.success and result.fun < best_C5:
            best_C5 = result.fun
            best_z = result.x.copy()
    except:
        pass
    
    time_left = budget_s - (time.time() - start_time)
    return best_z, best_C5, time_left

def gridify(intervals, n):
    """Convert continuous intervals to step function"""
    h = np.zeros(n)
    dx = 2.0 / n
    
    a, b = intervals
    a = a.copy()
    b = b.copy()
    
    # Fill grid
    for i in range(n):
        x_center = dx * (i + 0.5)
        for j in range(len(a)):
            if a[j] <= x_center <= b[j]:
                h[i] = 1.0
                break
    
    # Adjust to sum to n/2 exact
    current_sum = np.sum(h)
    target_sum = n / 2
    
    if abs(current_sum - target_sum) > 0.5:
        return h  # Too far off
    
    diff = int(round(target_sum - current_sum))
    if diff == 0:
        return h
    
    # Find indices with least impact on C5 and flip
    best_indices = []
    for i in range(n):
        if (h[i] == 0 and a[0] - dx <= i * dx <= b[-1] + dx):
            # Rough heuristic - add or remove if within reasonable range
            best_indices.append(i)
    
    if len(best_indices) == 0:
        # Fallback flip one at random that doesn't go too far
        for i in range(n):
            if h[i] == 0 or h[i] == 1:
                best_indices.append(i)

    # Neighbors by indices
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    import random
    
    for _ in range(max_swaps):
        if time.time() - start_time > time_budget_s - 0.1:

            break
