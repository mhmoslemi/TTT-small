def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for step function h"""

    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value (scaled by dx)
    max_overlap = np.max(correlation) * dx
    return max_overlap
