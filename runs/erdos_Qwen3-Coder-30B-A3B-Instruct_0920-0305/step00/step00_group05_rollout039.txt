import numpy as np
import cvxpy as cp
import time

def full_overlap(x):
    """Compute full correlation of x with 1-x"""
    return np.correlate(x, 1 - x, mode="full")

def worst_shift(x):
    """Return (k*, O_k*) where O_k* is the maximum overlap for shift k"""
    overlaps = full_overlap(x)
    k_max = np.argmax(overlaps)
    return (k_max, overlaps[k_max])

def build_mip(N, active_shifts):
    """Build a CVXPY MIP model with active_shifts and symmetry-breaking"""
    # Create variables
    x = cp.Variable(N, integer=True)
    t = cp.Variable(integer=True)
    
    # Track variables for easy access
    vars_dict = {'x': x, 't': t}
    
    # Build auxiliary w variables and constraints
    w_vars = {}
    constraints = []
    
    # Half-ones constraint
    constraints.append(cp.sum(x) == N // 2)
    
    # Symmetry-breaking constraints
    constraints.append(x[0] == 1)  # Fix first element
    
    # Add lexicographic ordering for symmetry breaking
    if N >= 3:
        constraints.append(x[0] >= x[N-1])
        constraints.append(x[1] >= x[N-2])
    
    # Build constraints for each active shift
    for k in active_shifts:
        # Create w variables for this shift
        w = cp.Variable(N - k, integer=True)
        w_vars[k] = w
        vars_dict[f'w_{k}'] = w
        
        # The continuous overlap for this shift
        for i in range(N - k):
            # w[i,k] = x[i] AND x[i+k]
            constraints.append(w[i] <= x[i])
            constraints.append(w[i] <= x[i + k])
            constraints.append(w[i] >= x[i] + x[i + k] - 1)
        
        # Overlap constraint: sum(xi) - sum(w[i,k]) <= t
        overlap_constraint = cp.sum(x[:N-k]) - cp.sum(w) <= t
        constraints.append(overlap_constraint)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Build problem
    prob = cp.Problem(objective, constraints)
    
    return prob, vars_dict

def solve_mip(prob, vars_dict, time_limit=30):
    """Solve the MIP problem with specified time limit"""
    try:
        result = prob.solve(solver=cp.CBC, max_iters=10000, TimeLimit=time_limit)
        x_val = vars_dict['x'].value
        t_val = vars_dict['t'].value if vars_dict['t'].value is not None else 0
        return x_val, t_val
    except Exception as e:
        print(f"Solving error: {e}")
        return None, None
