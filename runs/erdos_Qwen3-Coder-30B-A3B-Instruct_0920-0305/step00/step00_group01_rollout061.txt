import numpy as np

# Global variables

def project_to_feasible(x, target_sum):
    """Project x onto the set where sum = target_sum and 0 <= x[i] <= 1"""
    # Ensure all values are in [0,1]
    x = np.clip(x, 0, 1)
    
    # If sum equals target, we're done
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x
    
    # Simple projection method: iterative scaling with dual ascent
    # We want to project to simplex with upper bounds
    # For simplicity, we directly adjust boundary violations
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return x
    
    # If diff > 0, we want to increase values
    # If diff < 0, we want to decrease values
    # Just scale according to rank ordering 
    if diff > 0:
        # Adding mass, prioritize small values
        # But also maintain differences
        excess = diff
        for i in range(len(x)):
            if excess <= 0:
                break
            can_add = 1 - x[i]
            add = min(can_add, excess)
            x[i] += add
            excess -= add
    else:
        # Removing mass
        deficit = -diff
        for i in range(len(x)):
            if deficit <= 0:
                break
            can_remove = x[i] 
            remove = min(can_remove, deficit)
            x[i] -= remove
            deficit -= remove
    
    return x

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if initial_h_values is not None:
        h = np.array(initial_h_values)
    else:
        N = 47
        h = np.random.rand(N)
        h = project_to_feasible(h, N/2)
    
    N = len(h)
    dx = 2.0 / N
    cur_best = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_val = cur_best
    
    # Simulated Annealing main loop
    temp = 0.05  # Initial temperature
    cooling_rate = 0.999
    max_iter = int(2e6)
    
    iter_count = 0
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 0.1:
        iter_count += 1
        
        # Random neighbor: swap mass between two indices
        i, j = np.random.choice(N, 2, replace=False)
        
        # Compute allowed delta
        old_i = h[i]
        old_j = h[j]
        delta = np.random.uniform(-min(old_i, 1 - old_j), min(1 - old_i, old_j))
        
        # Apply change
        new_h = h.copy()
        new_h[i] = old_i + delta
        new_h[j] = old_j - delta
        
        # Project to feasible set
        new_h = project_to_feasible(new_h, N/2)
        
        # Evaluate
        new_val = evaluate_erdos_solution(new_h)
        
        # Accept or reject
        if new_val < cur_best:
            h = new_h
            cur_best = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val
        else:
            # Accept with probability
            prob = math.exp(-(new_val - cur_best) / temp)
            if np.random.rand() < prob:
                h = new_h
                cur_best = new_val
                
        # Cool down
        temp *= cooling_rate
        
    # Final evaluation
    final_val = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_val, len(best_h))
