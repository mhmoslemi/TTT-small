import numpy as np
import time
import random
from scipy.optimize import minimize

def project_to_feasible(h, target_sum):
    """Project h onto feasible region: sum = target_sum, 0 <= h[i] <= 1"""
    # Clip values to [0,1] and adjust to match target sum
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Use bisection to find lambda for projection onto simplex
    def obj_lambda(lam):
        h_proj = np.clip(h - lam, 0, 1)
        return np.sum(h_proj) - target_sum
    
    # Binary search for lambda in reasonable range
    lam_min, lam_max = -10.0, 10.0
    eps = 1e-8
    for _ in range(100):
        lam = (lam_min + lam_max) / 2
        val = obj_lambda(lam)
        if abs(val) < eps:
            break
        elif val > 0:
            lam_min = lam
        else:
            lam_max = lam
    
    h = np.clip(h - lam, 0, 1)
    return h

def compute_overlap(h):
    """Compute C_5 = max_k ∫ h(x)(1 - h(x+k)) dx using np.correlate"""
    if len(h) < 2:
        return 0.0
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    overlap = corr * dx
    # Max over all lags (k values)
    max_overlap = np.max(overlap)
    return max_overlap
