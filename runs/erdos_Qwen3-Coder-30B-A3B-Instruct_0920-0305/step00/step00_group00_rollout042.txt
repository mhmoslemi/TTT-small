import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize
import time

def project_to_constraints(v, lower=0.0, upper=1.0, target_sum=None):
    if target_sum is None:
        target_sum = len(v) / 2.0
    # binary search for lambda
    lo = -np.max(v) - 1.0  # ensures v+lo <=0
    hi = upper - np.min(v) + 1.0  # ensures v+hi >= upper
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, lower, upper)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo+hi)/2.0, lower, upper)
    return x

def compute_gradient(h, w, dx):
    n = len(h)
    grad = np.zeros_like(h)
    offset = n - 1
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        shift = idx - offset
        if shift >= 0:
            # first term
            grad[:n - shift] += wk * (1 - h[shift:])
            # second term
            grad[shift:] -= wk * h[:n - shift]
        else:
            s = -shift
            # first term
            grad[s:] += wk * (1 - h[:-s])
            # second term
            grad[:-s] -= wk * h[s:]
    return grad * dx

def loss_and_grad(h, beta, dx):
    p = 1 - h
    c = np.correlate(h, p, mode='full') * dx
    scaled = beta * c
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w_sum = np.sum(w)
    w = w / w_sum
    loss = (max_scaled + np.log(w_sum)) / beta
    grad = compute_gradient(h, w, dx)
    return loss, grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    n = 140
    dx = 2.0 / n
    best_c5 = np.inf
    best_h = None
    max_restarts = 10
    target_sum = n / 2.0
    
    # Get initial_h_values if available
    try:
        initial_h_values
    except NameError:
        initial_h_values = None
    
    for restart in range(max_restarts):
        elapsed = time.time() - start
        if elapsed > budget_s * 0.95:
            break
        
        # Initialize
        if restart == 0 and initial_h_values is not None:
            # upscale initial_h_values
            old_n = len(initial_h_values)
            old_dx = 2.0 / old_n
            old_x = np.linspace(0, 2, old_n, endpoint=False) + old_dx / 2
            new_x = np.linspace(0, 2, n, endpoint=False) + dx / 2
            h0 = np.interp(new_x, old_x, initial_h_values)
        else:
            h0 = np.random.random(n)  # uniform [0,1)
        
        # Project onto box and sum constraints  
        h0 = project_to_constraints(h0, target_sum=target_sum)
        h = h0.copy()
        current_loss = np.inf
        beta_schedule = [1.0, 5.0, 15.0, 30.0]
        
        for beta in beta_schedule:
            elapsed = time.time() - start
            if elapsed > budget_s * 0.95:
                break
            
            # Optimization loop for this beta
            for it in range(200):
                elapsed = time.time() - start
                if elapsed > budget_s * 0.95:
                    break
                
                loss, grad = loss_and_grad(h, beta, dx)
                
                # Project onto box and sum constraints
                step = 0.05 / (1 + it/10)
                new_h = project_to_constraints(h - step * grad, target_sum=target_sum)
                
                # linesearch
                for ls_iter in range(5):
                    new_loss, _ = loss_and_grad(new_h, beta, dx)
                    if new_loss <= loss:
                        break
                    step *= 0.5
                    new_h = project_to_constraints(h - step * grad, target_sum=target_sum)
                
                h = new_h
                current_loss = loss
                
                # Compute actual C5
                c = np.correlate(h, 1-h, mode='full') * dx
                c5 = c.max()
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        
        # Final evaluation
        if best_h is not None:
            try:
                from evaluate_erdos_solution import evaluate_erdos_solution
                final_c5 = evaluate_erdos_solution(best_h)
                if final_c5 < best_c5:
                    best_c5 = final_c5
            except:
                pass
    
    if best_h is None:
        # Fallback to random initialization if no improvement
        h0 = np.random.random(n)
        h0 = project_to_constraints(h0, target_sum=target_sum)
        c = np.correlate(h0, 1-h0, mode='full') * dx
        best_c5 = c.max()
        best_h = h0
    
    return (best_h.tolist(), best_c5, n)
