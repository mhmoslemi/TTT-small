import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize
import time

def project_to_constraints(v, lower=0.0, upper=1.0, target_sum=None):
    if target_sum is None:
        target_sum = len(v) / 2.0
    # binary search for lambda
    lo = -np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, lower, upper)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo+hi)/2.0, lower, upper)
    return x

def compute_gradient(h, w, dx):
    n = len(h)
    grad = np.zeros_like(h)
    offset = n - 1
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        shift = idx - offset
        if shift >= 0:
            # first term
            grad[:n - shift] += wk * (1 - h[shift:])
            # second term
            grad[shift:] -= wk * h[:n - shift]
        else:
            s = -shift
            # first term
            grad[s:] += wk * (1 - h[:n - s])
            # second term
            grad[:n - s] -= wk * h[s:]
    return grad * dx

def loss_and_grad(h, beta, dx):
    p = 1 - h
    c = np.correlate(h, p, mode='full') * dx
    # compute log-sum-exp loss
    scaled = beta * c
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w_sum = np.sum(w)
    w = w / w_sum
    loss = (max_scaled + np.log(w_sum)) / beta

    # compute gradient via w
    grad = compute_gradient(h, w, dx)
    return loss, grad

            h0 = np.interp(new_x, old_x, init)
            h0 = project_to_constraints(h0)
        else:
            h0 = np.random.random(n_points)
            h0 = project_to_constraints(h0)
            
        h = h0.copy()
        beta_schedule = [1., 5., 15., 30.]
        for beta in beta_schedule:
            elapsed = time.time() - start_time

            def fun(x):
                loss, grad = loss_and_grad(x, beta, dx)

            cons = ({'type':'eq', 'fun': lambda x: np.sum(x) - target_sum,
