import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
import math

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h_values."""
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # The maximum value gives the upper bound
    max_corr = np.max(corr)
    return max_corr * dx

def initial_h_values(n=47):
    """Produce a starting point for the optimization."""
    # Use the initial construction - a step function that's 1 on [0, 1] and 0 on (1, 2]
    h = np.zeros(n)
    # Scale to ensure integral is 1
    dx = 2.0 / n
    # Place 1's on first half of the domain
    half_points = int(n / 2)
    h[:half_points] = 1.0
    return h

def objective_function(h_values):
    """Computes the C5 constant for given h values."""
    return evaluate_erdos_solution(h_values)

def run(seed=42, budget_s=600):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Initial solution
    n_points = 47
    x0 = initial_h_values(n_points)
    # Make a copy for optimization
    h_start = x0.copy()
    
    # Define bounds for each variable (0 <= h[i] <= 1)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Ensure that sum(h)*dx = 1 which is equivalent to sum(h) = n_points/2
    # We can enforce this constraint via a custom constraint:
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization settings
    maxiter = 10000
    disp = False
    
    # Custom step function that also ensures the integral constraint
    def modify_h(h):
        """Ensure the integral constraint is met."""
        h = np.array(h)
        current_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(current_sum - target_sum) > 1e-6:
            # Adjust values to meet target sum
            diff = target_sum - current_sum
            # Distribute the difference evenly
            h += diff / len(h)
            # Clip to [0, 1]
            h = np.clip(h, 0, 1)
        return h
    
    # Function to help with bounds 
    def bounded_h(h):
        """Ensure each value is in [0,1] for constraints."""
        return np.clip(h, 0, 1)
    
    # Try a few strategies:
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    last_improvement = 0
    
    # Try simulated annealing with some bounds checking
    iter_num = 0
    while time.time() - start_time < budget_s - 1:
        try:
            # Small random changes to explore nearby solutions
            h_trial = best_h.copy()
            idx = np.random.randint(0, n_points)
            # Mutate one value with small change
            delta = np.random.normal(0, 0.02)
            h_trial[idx] += delta
            h_trial = bounded_h(h_trial)
            
            # Enforce the integral constraint precisely
            h_trial = modify_h(h_trial)
            
            trial_c5 = evaluate_erdos_solution(h_trial)
            
            if trial_c5 < best_c5:
                best_h = h_trial.copy()
                best_c5 = trial_c5
                last_improvement = 0
            else:
                last_improvement += 1
            
            iter_num += 1
            
            if last_improvement > 1200:
                # Restart with a new point
                h_start = initial_h_values(n_points)
                h_start = modify_h(h_start)
                best_h = h_start.copy()
                best_c5 = evaluate_erdos_solution(best_h)
                last_improvement = 0
            
        except Exception as e:
            # Safeguard against issues
            continue
    
    # Final check on best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
