```python
import time
import numpy as np
import cvxpy as cp
import random
import math

def random_binary_vector(n):
    # Create a random binary vector with exactly n/2 ones
    h = np.zeros(n)
    ones_indices = np.random.choice(n, n//2, replace=False)
    h[ones_indices] = 1
    return h.astype(float)

def compute_c5(h):
    # Compute the overlap integral C5
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum of the correlation values is the desired C5
    C5 = np.max(corr) * dx
    return C5

def solve_milp(n, time_limit=500):
    # Solve the MILP to find a good solution for small n
    # Variables: h (binary), w (continuous), t (continuous)
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Constraints sum = n/2
    constraints = [cp.sum(h) == n//2]
    
    # Add constraints for each shift k (0 to n-1)
    for k in range(1, n):  # skip k=0 as it always contributes n/2
        # Number of valid positions: n - k
        if n - k == 0:
            continue
        # Create continuous variables for products h_i * h_{i+k} for i = 0..n-k-1
        y = cp.Variable(n - k)
        # Linearization constraints
        constraints += [y >= 0]
        constraints += [y <= h[:n-k]]
        constraints += [y <= h[k:]]
        constraints += [y >= h[:n-k] + h[k:] - 1]
        # Sum constraint for this shift
        constraints += [cp.sum(y) >= t]
    
    # Objective: maximize t
    prob = cp.Problem(cp.Maximize(t), constraints)
    
    # Solve with CBC (MILP solver)
    try:
        prob.solve(solver=cp.CBC, time_limit=time_limit)
        h_sol = np.round(h.value).astype(float)
        # Recompute C5 using the actual h_sol
        C5 = compute_c5(h_sol)
        return h_sol, C5
    except Exception as e:
        # fallback to heuristic if MILP fails
        h_sol = random_binary_vector(n)
        C5 = compute_c5(h_sol)
        return h_sol, C5

def local_search(h_initial, max_iters, time_limit):
    start = time.time()
    cur = h_initial.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    
    while time.time() - start < time_limit and max_iters > 0:
        # Random swap
        ones_idx = np.where(cur == 1)[0]
        zeros_idx = np.where(cur == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        
        # Swap
        cur[i] = 0
        cur[j] = 1
        new_c5 = compute_c5(cur)
        
        if new_c5 < cur_c5:  # Improvement
            cur_c5 = new_c5
        else:
            # Revert swap
            cur[i] = 1
            cur[j] = 0
            
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best = cur.copy()
            
        max_iters -= 1
        
    return best, best_c5

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    n = 100  # Smaller value for speed and stability
    h_opt = None
    c5_opt = float('inf')
    
    # Try to solve MILP for small n
    try:
        remaining_time = budget_s - (time.time() - start) - 5
        if remaining_time > 5:
            h_opt, c5_opt = solve_milp(n, time_limit=remaining_time)
    except Exception:
        # Fallback to random if MILP fails
        h_opt = random_binary_vector(n)
        c5_opt = compute_c5(h_opt)
    
    # Use local search with remaining time
    remaining_time = budget_s - (time.time() - start) - 5
    if remaining_time > 5:
        try:
            best_h, best_c5 = local_search(h_opt, max_iters=100000, time_limit=remaining_time)
            return best_h.tolist(), float(best_c5), n
        except Exception:
            pass
    
    # Return best found solution
    return h_opt.tolist(), float(c5_opt), n
```
