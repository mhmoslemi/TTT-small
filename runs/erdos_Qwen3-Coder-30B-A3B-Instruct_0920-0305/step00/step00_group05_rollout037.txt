import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_overlap(h):
    """Evaluate the maximum overlap integral C_5 for a given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the result
    c5 = np.max(corr) * dx
    return c5

    final_h = np.array(result.x)
