import numpy as np
import time
from scipy.optimize import minimize
from math import floor, ceil

def project_to_feasible(h, target_sum):
    """Project h onto [0,1] and sum=target_sum constraint"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase some values
        idx = np.argsort(h)
        for i in reversed(range(len(h))):
            if diff <= 0:
                break
            incr = min(diff, 1 - h[idx[i]])
            h[idx[i]] += incr
            diff -= incr
    else:
        # Need to decrease some values
        idx = np.argsort(h, kind='mergesort')
        for i in range(len(h)):
            if diff >= 0:
                break
            decr = min(-diff, h[idx[i]])
            h[idx[i]] -= decr
            diff += decr
    return h

def evaluate_erdos_solution(h):
    """Evaluate the maximum overlap for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation with 1-h using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # Since numpy correlate is for 1D arrays, we get correlation for all lags
    # The full mode gives 2*n - 1 elements
    # The maximum overlap over all lags is what we want
    max_overlap = np.max(corr) * dx
    return max_overlap

    max_overlap = evaluate_erdos_solution(h)
    best_h = h.copy()

        # Generate random pair move
        i1, i2 = np.random.choice(n_points, 2, replace=False)
        h1, h2 = h[i1], h[i2]
        
        # Compute allowable delta
        low_bound = max(-h1, h2 - 1)
        high_bound = min(1 - h1, h2)
        if low_bound < high_bound:
            delta = np.random.uniform(low_bound, high_bound)
            # Update h
            h[i1] += delta
            h[i2] -= delta
                
            # Re-project to maintain constraints
            h = project_to_feasible(h, n_points / 2)
            
            # Evaluate new solution
            new_max_overlap = evaluate_erdos_solution(h)

            if new_max_overlap < max_overlap:

                else:
                    # Revert changes
                    h[i1] = h1
                    h[i2] = h2
                    
            # Cool down temperature
            temp *= cooling_rate
    
    # Final re-evaluation on the best solution found
    best_val = evaluate_erdos_solution(best_h)
    return (best_h, best_val, len(best_h))
