import numpy as np
import time
import math
import random

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    time_end = start + budget_s - 0.5
    
    n_points = 200
    best_h = None
    best_c5 = float('inf')
    
    # Run multiple restarts
    restart_count = 0
    while time.time() < time_end:
        restart_count += 1
        # Generate random binary vector with exactly n_points/2 ones
        h = np.zeros(n_points, dtype=np.float64)
        ones = np.random.choice(n_points, n_points//2, replace=False)
        h[ones] = 1.0
        
        cur_c5 = compute_c5(h)
        
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        
        # Simulated annealing within this restart
        max_local_iters = 10000
        T0 = 0.05
        for it in range(max_local_iters):
            if time.time() >= time_end:
                break
            # Choose two random indices with different values
            i, j = np.random.randint(n_points, size=2)
            while h[i] == h[j]:
                i, j = np.random.randint(n_points, size=2)
            
            # Propose swap
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
            new_c5 = compute_c5(h_new)
            
            d = new_c5 - cur_c5
            T = max(1e-6, T0 * (1 - it / max_local_iters))
            
            if d <= 0 or np.random.rand() < math.exp(-d / T):
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
    
    # Ensure result is a list
    return (best_h.tolist(), best_c5, n_points)
