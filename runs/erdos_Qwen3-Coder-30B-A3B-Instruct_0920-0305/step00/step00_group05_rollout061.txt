import numpy as np
import time
import math
import random

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    time_end = start + budget_s - 0.5
    
    n_points = 200
    best_h = None
    best_c5 = float('inf')
    
    max_restarts = 50
    max_local_iters = 10000
    
    restart_count = 0
    iteration_count = 0
    
    while time.time() < time_end and restart_count < max_restarts:
        # generate new random binary with half ones
        h = np.zeros(n_points, dtype=np.float64)
        ones = np.random.choice(n_points, n_points//2, replace=False)
        h[ones] = 1.0

        cur_c5 = compute_c5(h)

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        # local annealing for this start
        T0 = 0.05
        for it in range(max_local_iters):
            if time.time() >= time_end:
                break
                
            # Choose random i with 1 and j with 0
            idx1 = np.random.randint(n_points)
            idx2 = np.random.randint(n_points)
            while h[idx1] == h[idx2]:
                idx1 = np.random.randint(n_points)
                idx2 = np.random.randint(n_points)
                
            # propose swap
            h_new = h.copy()
            h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
            new_c5 = compute_c5(h_new)
            
            d = new_c5 - cur_c5
            T = T0 * (1 - it / max_local_iters)
            
            if d <= 0 or np.random.rand() < math.exp(-d / (T + 1e-12)):
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    
        restart_count += 1
        iteration_count += max_local_iters
                        
    return (best_h.tolist(), best_c5, n_points)
