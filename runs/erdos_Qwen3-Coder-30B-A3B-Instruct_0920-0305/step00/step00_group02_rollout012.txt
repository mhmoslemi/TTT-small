import numpy as np
import time
from scipy import fftpack

def run(seed=42, budget_s=600, **kwargs):
    # Initialize parameters
    np.random.seed(seed)
    n = 300
    dx = 2.0 / n
    total_sum = n / 2.0
    max_time = budget_s
    start_time = time.time()

    # Generate Beatty sequence seed: h[i] = 1 if fractional part of i * β < 0.5
    beta = (np.sqrt(5) - 1) / 2
    h0 = np.array([1 if (i * beta) % 1 < 0.5 else 0 for i in range(n)])
    current_sum = np.sum(h0)
    
    # Adjust to enforce exact sum
    if current_sum > total_sum:
        # Need to flip some 1s to 0s
        indices_1 = np.where(h0 == 1)[0]
        to_flip = int(current_sum - total_sum)
        for idx in sorted(np.random.choice(indices_1, to_flip, replace=False)):
            h0[idx] = 0
    elif current_sum < total_sum:
        # Need to flip some 0s to 1s
        indices_0 = np.where(h0 == 0)[0]
        to_flip = int(total_sum - current_sum)
        for idx in sorted(np.random.choice(indices_0, to_flip, replace=False)):
            h0[idx] = 1
            
    # Initialize 'h' with the Beatty seed
    h = h0.astype(float)
    
    # Project to unit box and enforce sum
    def project_to_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if np.abs(current_sum - total_sum) < 1e-10:
            return h_vec
        # Water filling to adjust mean
        z = h_vec - current_sum / n
        sorted_z = np.sort(z)[::-1]
        z_cumsum = np.cumsum(sorted_z)
        # Find index where z_cumsum <= -z[i]/(i+1)
        threshold_idx = None
        for i in range(len(sorted_z)):
            if z_cumsum[i] + sorted_z[i] * (i + 1) < 0:
                threshold_idx = i
                break
        if threshold_idx is None:
            threshold_idx = len(sorted_z)
        gamma = (z_cumsum[threshold_idx - 1] if threshold_idx > 0 else 0.0) / (threshold_idx + 1.0)
        offset = (total_sum / n) - gamma
        h_vec = np.clip(h_vec + offset, 0, 1)
        return h_vec

    # Initialize
    h = project_to_feasible(h)

    # Sampler for evaluating the final function
    def evaluate_c5(h_vec):
        full_corr = np.correlate(h_vec, 1.0 - h_vec, mode='full') * dx
        return np.max(full_corr)

    # Surrogate smooth minimum
    def smooth_min_surrogate(h_vals, alpha):
        # Compute all S_k values
        conv = np.correlate(h_vals, h_vals, mode='full')
        s_vals = conv * dx
        # Heuristic for handling numerical instability
        s_vals = np.clip(s_vals, 0, np.max(s_vals))  # Prevent negative inf during exp
        exp_vals = np.exp(-alpha * s_vals)
        sum_exp = np.sum(exp_vals)
        if sum_exp <= 0:
            return np.min(s_vals)
        log_sum_exp = -np.log(sum_exp) / alpha
        return log_sum_exp

    # Compute overlap vectors for gradient.
    def compute_all_sk(h_vals):
        # Use FFT for efficiency
        conv_full = np.correlate(h_vals, h_vals, mode='full')
        s_vals = conv_full * dx
        return s_vals

    # Compute gradient with respect to smooth surrogate
    def compute_grad_smooth(h_vals, alpha):
        s_vals = compute_all_sk(h_vals)
        w_vals = np.exp(-alpha * s_vals)
        sum_w = np.sum(w_vals)
        w_vals = w_vals / (sum_w + 1e-16)  # Avoid division by 0

        # Now compute ∇Sk + use them weighted to form the full gradient
        grad = np.zeros_like(h_vals)
        # Plane sweep: for each k, compute ∇Sk
        for k in range(len(h_vals)):
            if k == 0:
                # ∇S0 = h + h_rev shifted
                shift_h = np.roll(h_vals, 0)
                grad += w_vals[k] * shift_h
            else:
                # ∇Sk+i = h_{i+k} + h_{i-k}; compute with sliding window
                shifted_right = np.zeros_like(h_vals)
                shifted_left = np.zeros_like(h_vals)
                shifted_right[:-k] = h_vals[k:]
                shifted_left[k:] = h_vals[:-k]
                grad += w_vals[k] * (shifted_right + shifted_left)

        # Each ∇Sk contains partial derivatives w.r.t h_i, from overlapping computation:
        # Take full vector weight sum ∇S_k
        # Normalize sum and return
        return grad

    # Voigt explosion fallback if needed
    def backtracking_line_search(h_old, grad, alpha, step_size=1.0):
        c1 = 1e-4
        phi_old = smooth_min_surrogate(h_old, alpha)
        h_new = h_old + step_size * grad
        h_new = project_to_feasible(h_new)
        phi_new = smooth_min_surrogate(h_new, alpha)
        while phi_new <= phi_old + c1 * step_size * np.dot(grad, grad):
            step_size *= 0.5
            h_new = h_old + step_size * grad
            h_new = project_to_feasible(h_new)
            phi_new = smooth_min_surrogate(h_new, alpha)
        return h_new, step_size

    # Main iterations
    eta = 1.0
    alpha = 1.0
    max_alpha = 10000
    alpha_multiplier = 2.0
    iteration = 0
    best_h = h.copy()
    best_value = smooth_min_surrogate(h, alpha)
    min_val = best_value
    
    # Keep track of best in whole process
    continue_optimizing = True
    while continue_optimizing:
        elapsed = time.time() - start_time
        if elapsed >= max_time * 0.6:
            continue_optimizing = False  # Leave some time for local search
            break
        
        # Compute gradient
        step_grad = compute_grad_smooth(h, alpha)
        # Apply line search to ensure ascent
        h_new, eta = backtracking_line_search(h, step_grad, alpha, eta)
        # Project
        h = project_to_feasible(h_new)
        
        # Record best
        current_value = smooth_min_surrogate(h, alpha)
        min_val = min(min_val, current_value)
        if current_value > best_value:
            best_value = current_value
            best_h = h.copy()
        
        # Update alpha (slowly)
        if iteration % 100 == 0 and alpha < max_alpha:
            alpha = min(alpha * alpha_multiplier, max_alpha)
        iteration += 1

    # Discrete polishing step: perform hill climbing on binary vector
    def binary_swap_hill_climb(best_h_before):
        # Round to binary using median threshold
        sorted_h = np.sort(best_h_before)
        median_threshold = sorted_h[int(len(sorted_h)/2)]
        binary_h = (best_h_before > median_threshold).astype(float)
        current_sum = np.sum(binary_h)
        if np.abs(current_sum - total_sum) < 1e-6:
            pass  # Good sum
        elif current_sum > total_sum:
            # Flip some 1 to 0
            ones_indices = np.where(binary_h == 1)[0]
            to_flip = int(current_sum - total_sum)
            for i in range(to_flip):
                binary_h[ones_indices[i]] = 0
        else:
            # Flip some 0 to 1
            zeros_indices = np.where(binary_h == 0)[0]
            to_flip = int(total_sum - current_sum)
            for i in range(to_flip):
                binary_h[zeros_indices[i]] = 1

        # Final local search using swaps
        h_current = binary_h.copy()
        abs_best = evaluate_c5(h_current)
        best_repr = h_current
        max_iterations = int(180 * 100)  # 180s * 100 iterations/sec
        burn_in = max_iterations // 10
        for _ in range(max_iterations):
            elapsed = time.time() - start_time
            if elapsed >= max_time * 0.95:
                break
            # Good random swap
            i1 = np.random.randint(0, n)
            i2 = np.random.randint(0, n)
            if i1 == i2:
                continue
            if h_current[i1] == h_current[i2]:
                continue  # No point swapping same value
            # Swap
            h_current[i1], h_current[i2] = h_current[i2], h_current[i1]
            new_value = evaluate_c5(h_current)
            if new_value < abs_best:
                abs_best = new_value
                best_repr = h_current.copy()
            elif np.random.rand() < 0.1 and _ > burn_in:
                # Occasionally accept worse move for sampling
                pass
            else:
                # Undo swap
                h_current[i1], h_current[i2] = h_current[i2], h_current[i1]

        return best_repr, abs_best

    # Run binary polishing
    if time.time() - start_time < max_time * 0.9:
        h_final, c5_bound = binary_swap_hill_climb(best_h)
    else:
        h_final = best_h
        c5_bound = evaluate_c5(h_final)
        
    return (h_final.tolist(), float(c5_bound), n)
