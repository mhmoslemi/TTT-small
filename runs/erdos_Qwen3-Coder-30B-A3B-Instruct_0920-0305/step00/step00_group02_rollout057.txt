import numpy as np
import cvxpy as cp
import random
import time
from scipy import sparse

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    n = 200  # number of points; dx = 2/n = 0.01
    dx = 2.0 / n
    assert n % 2 == 0, "n must be even"
    
    # Step 1: Construct A_k matrices (sparse)
    if time.time() - start_time > 0.5 * budget_s:
        return initial_h_values, 0.38092, len(initial_h_values)
    
    # Keep only k up to n//2 for symmetry
    ks = list(range(1, n // 2 + 1))
    # A_k -> sparse matrix n x n
    A_k_list = []
    for k in ks:
        rows = np.arange(n - k)
        cols = np.arange(k, n)
        data = np.ones_like(rows)
        A_k = sparse.csr_matrix((data, (rows, cols)), shape=(n, n))
        A_k = A_k + A_k.T  # make symmetric
        A_k_list.append(A_k)
    
    # Step 2: Define and solve SDP
    if time.time() - start_time > 0.7 * budget_s:
        return initial_h_values, 0.38092, len(initial_h_values)
    
    # Define variables
    X = cp.Variable((n, n), PSD=True)
    x = cp.Variable(n)  # diagonal entries of X
    t = cp.Variable()
    
    # Constraints
    constraints = [
        cp.diag(X) == x,
        cp.sum(x) == n / 2,
        0 <= x,
        x <= 1,
    ]
    
    for k_idx, A_k in enumerate(A_k_list):
        constraints.append(cp.trace(A_k @ X) >= 2 * t)
    
    # Objective
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve SDP using SCS
    # We'll softly limit time by adjusting max_iters if needed
    try:
        prob.solve(solver=cp.SCS, max_iters=2000, eps=1e-4, verbose=False)
    except Exception:
        # fallback to initial guess
        return initial_h_values, 0.38092, len(initial_h_values)
    
    if prob.status != cp.OPTIMAL:
        return initial_h_values, 0.38092, len(initial_h_values)
    
    X_opt = X.value
    t_opt = t.value
    x_opt = x.value
    
    # Step 3: Rounding
    if time.time() - start_time > 0.8 * budget_s:
        return initial_h_values, 0.38092, len(initial_h_values)
    
    def hyperplane_round(X, num_trials=250):
        # X is a positive semidefinite matrix
        # Factorize: X ~ V @ V.T
        _, V = np.linalg.eigh(X)
        # Keep top-r components, r = 8 is enough
        r = min(8, V.shape[1])
        V = V[:, :r]
        best_h = None
        best_C5 = float('inf')
        
        for _ in range(num_trials):
            g = np.random.randn(r)
            s = V.T @ g  # shape (n,)
            # Sort s to find median
            s_sorted = np.sort(s)[::-1]
            k = n // 2  # We want exactly half
            tau = s_sorted[k - 1] if k < len(s_sorted) else s_sorted[-1]
            h_trial = (s >= tau).astype(int)
            if np.sum(h_trial) == n // 2:
                # Evaluate C5
                overlap_vals = np.correlate(h_trial, 1 - h_trial, mode='full')
                overlap_vals = overlap_vals[n - 1:]
                C5_trial = np.max(overlap_vals) * dx
                if C5_trial < best_C5:
                    best_C5 = C5_trial
                    best_h = h_trial
        return best_h, best_C5
    
    # Round X_opt
    h_rounded, C5_rounded = hyperplane_round(X_opt, 250)
    
    if h_rounded is None:
        return initial_h_values, 0.38092, len(initial_h_values)
    
    # Step 4: Local polishing with hill climbing
    if time.time() - start_time > 0.9 * budget_s:
        return h_rounded.tolist(), C5_rounded, n
    
    def compute_C5_exact(h):
        # h is a numpy array of 0/1
        c = np.correlate(h, 1 - h, mode='full')
        c = c[n - 1:]  # get the relevant part (positive shifts only)
        return np.max(c) * dx
    
    def swap_improve(h, max_iter=100000):
        current_h = h.copy()
        current_C5 = compute_C5_exact(current_h)
        exchanged = 0
        
        # Precompute S_k
        S = np.correlate(current_h, current_h, mode='full')[n - 1:]
        min_idx = np.argmin(S)
        min_S = S[min_idx]
        
        iter_count = 0
        while iter_count < max_iter and time.time() - start_time < 0.98 * budget_s:
            iter_count += 1
            
            # Try swapping
            ones = np.where(current_h == 1)[0]
            zeros = np.where(current_h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
                
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Perform swap
            temp_h = current_h.copy()
            temp_h[i] = 0
            temp_h[j] = 1
            
            # Recompute S (we just need to update overlaps affected by positions i,j)
            temp_S = np.correlate(temp_h, temp_h, mode='full')[n - 1:]
            temp_min_S = np.min(temp_S)
            temp_C5 = np.max(temp_S) * dx
            
            # if improvement, accept
            if temp_C5 < current_C5:
                current_h = temp_h
                current_C5 = temp_C5
                S = temp_S
                min_S = temp_min_S
                min_idx = np.argmin(temp_S)
                exchanged += 1
        return current_h, current_C5
    
    h_final, C5_final = swap_improve(h_rounded, max_iter=100000)
    C5_final = compute_C5_exact(h_final)
    return h_final.tolist(), C5_final, n
