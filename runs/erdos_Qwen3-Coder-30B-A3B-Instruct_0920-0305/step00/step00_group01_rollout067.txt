import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 80
    dx = 2.0 / n_points
    target_max = 0.38092

    def evaluate_bound(h_vals):
        # Compute overlaps using full convolution
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        corr = corr * dx
        return np.max(corr)

    # Define LP variables
    def build_and_solve_lp(t_target):
        h = cp.Variable(n_points, name="h")
        p = cp.Variable((n_points, 2*n_points - 1), name="p")
        # Sum constraint
        constraints = [cp.sum(h) == n_points / 2]
        # Bound constraints
        for i in range(n_points):
            constraints.append(h[i] >= 0)
            constraints.append(h[i] <= 1)

        # McCormick envelopes: p[i, d] = h[i] * h[i+d]
        # shift index from [0, 2n-2] to [-n+1, n-1]
        for d in range(-(n_points-1), n_points):
            if d >= 0:
                for i in range(n_points - d):
                    p_idx = i + d
                    constraints.append(p[i, d + n_points - 1] <= h[i])
                    constraints.append(p[i, d + n_points - 1] <= h[p_idx])
                    constraints.append(p[i, d + n_points - 1] >= h[i] + h[p_idx] - 1)
                    constraints.append(p[i, d + n_points - 1] >= 0)
            else:  # negative shifts
                for i in range(-d, n_points):
                    p_idx = i + d
                    constraints.append(p[i, d + n_points - 1] <= h[i])
                    constraints.append(p[i, d + n_points - 1] <= h[p_idx])
                    constraints.append(p[i, d + n_points - 1] >= h[i] + h[p_idx] - 1)
                    constraints.append(p[i, d + n_points - 1] >= 0)
        # Add constraints from t_target
        for d in range(-(n_points-1), n_points):
            target_val = (1 - t_target) / dx
            if d >= 0:
                constraints.append(cp.sum(p[:, d + n_points - 1]) >= target_val)
            else:
                constraints.append(cp.sum(p[:, d + n_points - 1]) >= target_val)
        
        # Solve LP (simplest belief for consistency: try feasibility using dummy objective)
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                return h.value
            else:
                return None
        except Exception:
            return None

    # Binary search over t
    lower = 0.0
    upper = target_max
    best_h = None
    final_bound = upper

    while upper - lower > 1e-5 and time.time() - start_time < budget_s - 5:
        t_mid = (lower + upper) / 2
        h_sol = build_and_solve_lp(t_mid)
        if h_sol is not None:
            # Verify the solution actually satisfies the constraint for true bound
            actual_bound = evaluate_bound(h_sol)
            if actual_bound <= t_mid:
                best_h = h_sol
                final_bound = actual_bound
                upper = t_mid  # tighten upper bound
            else:
                lower = t_mid  # cannot satisfy constraint
        else:
            lower = t_mid  # cannot solve LP
            
    # If we still got a valid solution but need a final clean-up
    if best_h is not None:
        h_opt = best_h.copy()
    else:
        # fallback to binary search until found
        h_opt = None
        for i in range(8):
            t = (upper + lower) / 2
            h_res = build_and_solve_lp(t)
            if h_res is not None:
                if evaluate_bound(h_res) <= t:
                    h_opt = h_res
                    final_bound = evaluate_bound(h_res)
                    break
            lower = t

    # Final rounding and refinement
    if h_opt is None:
        h_opt = np.ones(n_points) * 0.5
    else:
        h_opt = np.clip(h_opt, 0, 1)
    
    # Use local refinement with sum constraint
    def objective(h):
        h = np.clip(h, 0, 1)
        return evaluate_bound(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points/2
    
    # Optimize the current solution using scipy
    initial_h = h_opt.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Local optimization
    try:
        res = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 200})
        h_opt = res.x
        final_bound = res.fun
    except:
        pass
        
    # Final rounding
    h_final = h_opt.copy()
    total_mass = np.sum(h_final)
    desired_mass = n_points / 2
    
    # Round to assign 0 or 1 values
    sorted_indices = np.argsort(h_final)[::-1]
    assigned = 0
    
    h_rounded = np.zeros(n_points)
    for i in sorted_indices:
        if assigned < desired_mass:
            h_rounded[i] = 1
            assigned += 1
        else:
            h_rounded[i] = 0
    if assigned < desired_mass:
        # assign remaining 1's to smallest h values that were 0
        for i in range(n_points - 1, -1, -1):
            if assigned >= desired_mass:
                break
            if h_rounded[i] == 0:
                h_rounded[i] = 1
                assigned += 1

    # Compute final maximum overlap
    final_bound = evaluate_bound(h_rounded)
    
    return (h_rounded, final_bound, n_points)
