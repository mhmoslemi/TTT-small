import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.signal import correlate

def project_to_feasible(h, target_sum):
    """Projects h onto the feasible set {h: 0 <= h <= 1, sum(h) = target_sum}"""
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if abs(total - target_sum) < 1e-10:
        return h
    
    # Water-filling approach
    # Sort h in descending order to find how much we must adjust
    h_sorted = np.sort(h)[::-1]
    cumulative = np.cumsum(h_sorted)
    
    # Find the threshold such that sum of clipped(h + lambda) == target_sum
    # where lambda is added to each element
    diff = target_sum - total
    if diff > 0:
        # Need to raise values
        for i in range(len(h_sorted)):
            if i == 0:
                frac = 1.0
            else:
                frac = 1.0 - (cumulative[i-1] - cumulative[i]) / (i+1)
            if frac * (i+1) >= diff:
                lambda_val = diff / (i+1)
                break
        else:
            lambda_val = diff / len(h_sorted)
        
        # Apply adjustment
        h_new = h + lambda_val
        h_new = np.clip(h_new, 0, 1)
        return h_new
    else:
        # Need to lower values, but not go below 0
        diff = -diff
        # Find how much to reduce each to meet target
        h_new = h.copy()
        #   Start with full lowering of lowest values
        for i in range(len(h_sorted)):
            if i == 0:
                sub_total = 0
            else:
                sub_total = cumulative[i-1]
            # Reduced allowed lambda = (total - target_sum - sub_total) / (n-i)
            test_lambda = (np.sum(h_new) - target_sum - sub_total) / (len(h) - i)
            if test_lambda > 0 and abs(test_lambda) > 1e-10:
                if i == 0:
                    lambda_adjust = test_lambda
                else:
                    lambda_adjust = min(test_lambda, lambda_adjust)
        if '__i' not in locals():
            lambda_adjust = 0
        h_new = h - lambda_adjust
        h_new = np.clip(h_new, 0, 1)
        return h_new

def compute_Sk(h, dx):
    # Computes all S_k = ∫h(x) h(x+k) dx using overlap convex correlation operator
    # At the same time, gets the maximum
    c = correlate(h, h, mode='full')
    S = np.array([c[len(h)-1+k] for k in range(len(h)) if k < len(h)])
    return S * dx

def compute_C5(h, dx):
    """Computes C5 as 1 - min(S_k) via direct full overlap"""
    c = correlate(h, 1-h, mode='full')
    return (1.0 - np.max(c) * dx)

# Uses a local hill climber for binary optimalization
def local_search(h_binary, iterations):
    n = len(h_binary)
    best_h = h_binary.copy()
    best_C5 = compute_C5(best_h, 2.0/n)

    idx_ones = np.argwhere(h_binary == 1).ravel()
    idx_zeros = np.argwhere(h_binary == 0).ravel()

    tabu_list = []

    for _ in range(iterations):
        if len(idx_ones) < 1 or len(idx_zeros) < 1:
            break

        # Pick random one and zero
        i = np.random.choice(idx_ones)
        j = np.random.choice(idx_zeros)

        # Check tabu
        if (i, j) in tabu_list or (j, i) in tabu_list:
            continue

        # Make a swap
        h_trial = best_h.copy()
        h_trial[i] = 0
        h_trial[j] = 1

        # Evaluate
        trial_C5 = compute_C5(h_trial, 2.0/n)
        if trial_C5 < best_C5:
            best_h = h_trial
            best_C5 = trial_C5
            idx_ones = np.argwhere(best_h == 1).ravel()
            idx_zeros = np.argwhere(best_h == 0).ravel()
        else:
            # Accept with probability to escape stalemate at low temp
            pass

        tabu_list.append((i, j))
        if len(tabu_list) > 20:
            tabu_list.pop(0)

    return best_h, best_C5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n = 300  # Large enough to resolve structure; small enough to be fast
    dx = 2.0 / n
    target_sum = n / 2.0

    # Initial Beatty sequence seed (golden ratio)
    beta = (np.sqrt(5) - 1.0) / 2.0
    h0 = np.array([1 if ((i * beta) % 1.0) < 0.5 else 0 for i in range(n)])
    
    # Adjust to nonzero sum
    actual_sum = np.sum(h0)
    if abs(actual_sum - target_sum) > 1e-5:
        diff = int(round(target_sum - actual_sum))
        if diff > 0:
            # Need to make more 1s
            flip_idx = np.argwhere(h0 == 0).ravel()
            if len(flip_idx) >= diff:
                for i in np.random.choice(flip_idx, diff, replace=False):
                    h0[i] = 1
        else:
            # Need to make more 0s
            flip_idx = np.argwhere(h0 == 1).ravel()
            if len(flip_idx) >= -diff:
                # Select random positions to flip to 0s
                for i in np.random.choice(flip_idx, -diff, replace=False):
                    h0[i] = 0
    
    h = project_to_feasible(h0.astype(float), target_sum)
    
    # Continuous optimization via smoothed surrogate
    alpha = 1.0
    alpha_max = 10000.0
    k_restart = 400  # Restart alpha when knowing to be close
    
    best_h = h.copy()
    best_C5 = compute_C5(best_h, dx)
    
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)
    
    while time_left() > 5:
        # Compute overlaps S_k with circular shift using overlap FFT
        c = correlate(h, h, mode='full')
        S_all = np.array([c[len(h) - 1 + k] for k in range(len(h)) if k < len(h)]) * dx
        
        # Smooth min surrogate
        w = np.exp(-alpha * S_all)
        W = np.sum(w)
        if abs(W) < 1e-15: # Avoid numerical explosion
            break
        
        # Compute gradient \nabla phi_alpha(h) = (1/W) * sum w_k * \nabla S_k
        g = np.zeros_like(h)
        for k in range(len(h)):
            # Compute h_i + h_{i-k}
            if k == 0:
                grad_i = h.copy()
            else:
                # h_{i+k} from truncation (forward)
                g_fwd = np.zeros_like(h)
                g_fwd[:-k] = h[k:]
                # h_{i-k} from backward
                g_bwd = np.zeros_like(h)
                g_bwd[k:] = h[:-k]
                grad_i = g_fwd + g_bwd
            g += w[k] * grad_i  # Tensor product convolution will yield another h-like vector
        grad = g / W
        
        # Update
        h_new = h + grad * 0.15  # stepsize
        h = project_to_feasible(h_new, target_sum)
        
        # Update alpha
        if (time.time() - start_time) > 200 and alpha < alpha_max:
            alpha *= 2.0
        
        # Update best
        C5_trial = compute_C5(h, dx)
        if C5_trial < best_C5:
            best_h = h.copy()
            best_C5 = C5_trial
            
    # Discretization polishing
    # Momentary best continuous solution to binary
    threshold = np.sort(best_h)[int(n/2)]
    h_binary = (best_h > threshold).astype(int)
    # Ensure sum = n/2
    current_sum = np.sum(h_binary)
    need_change = int(current_sum - target_sum)
    if need_change > 0:
        flip1 = np.argwhere(h_binary == 1).ravel()
        if len(flip1) > need_change:
            idx_to_zero = np.random.choice(flip1, need_change, replace=False)
            for i in idx_to_zero:
                h_binary[i] = 0
    elif need_change < 0:
        flip0 = np.argwhere(h_binary == 0).ravel()
        if len(flip0) > -need_change:
            idx_to_one = np.random.choice(flip0, -need_change, replace=False)
            for i in idx_to_one:
                h_binary[i] = 1
    
    # Final local hill climb
    local_iter = 100 * n
    final_h, final_C5 = local_search(h_binary, local_iter)
    
    # Verify final bounds
    if final_C5 < best_C5:
        best_h = final_h
        best_C5 = final_C5
    
    return (best_h.tolist(), float(best_C5), n)
