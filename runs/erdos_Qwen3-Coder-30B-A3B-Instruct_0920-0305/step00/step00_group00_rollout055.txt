import numpy as np
import time
from scipy.special import softmax

def project_onto_box_and_sum(h, target_sum):
    """Project h onto [0,1] and enforce sum = target_sum using water filling."""
    # Binary search on Lagrange multiplier
    low, high = -100, 100
    for _ in range(60):  # Ensure convergence
        lam = (low + high) / 2
        x = np.clip(h + lam, 0, 1)
        s = np.sum(x)
        if s > target_sum:
            low = lam
        else:
            high = lam
    lam = (low + high) / 2
    x = np.clip(h + lam, 0, 1)
    
    # Final adjustment to ensure exact sum
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-10:
        # Find indices where we can increase
        free_indices = np.where((x > 0) & (x < 1))[0]
        if len(free_indices) > 0:
            increase = min(diff, 1 - np.min(x[free_indices]))
            for i in free_indices:
                if x[i] + increase <= 1:
                    x[i] += increase
                    break
    return np.clip(x, 0, 1)

def evaluate_correlation(h, dx):
    """Evaluate all correlations c_k = dx * sum h[i] * (1 - h[i+k])."""
    p = 1 - h
    n = len(h)
    c = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        # For shift k, we compute sum_{i} h[i] * p[i+k]
        # Equals sum_{i in range with valid indices} h[i]*p[i+k]
        if k < n:
            # Forward shift: i runs from 0 to n-k-1
            c[k] = np.sum(h[:n-k] * p[k:])
        else:
            # Backward shift: i runs from k-n+1 to n-1
            offset = k - n + 1
            c[k] = np.sum(h[offset:] * p[:n-offset])
    return c * dx

def gradient_of_correlation(h, k, dx):
    """Return gradient of correlation at shift k w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    # Add contribution from h[i] * (1 - h[i+k]) for valid i
    if k < n:
        g[:n-k] += dx * (1 - h[k:])
    else:
        offset = k - n + 1
        g[offset:] -= dx * h[:n-offset]
    return g

def evaluate_and_gradient_lse(h, dx, beta, target_sum):
    """Compute the LSE approximation c = max c_k and its gradient."""
    c = evaluate_correlation(h, dx)
    # We are approximating max_k c_k with (1/beta) * log(sum_k exp(beta * c_k))
    # We compute weights from softmax over beta*c
    w = softmax(beta * c)
    
    # Now compute the gradient of LSE: sum_k w_k * grad(c_k)
    g = np.zeros_like(h)
    for k in range(len(c)):
        g += w[k] * gradient_of_correlation(h, k, dx)
    
    # LSE value
    lse = (1 / beta) * np.log(np.sum(np.exp(beta * c)))
    
    # Apply projection to normalize the sum to target_sum
    # But since we're projecting onto sum=1 and using dx=2/n, target_sum = n/2
    # Here we ensure the sum is maintained
    h_projected = project_onto_box_and_sum(h, target_sum)
    
    # Weighted sum of correlations (for loss)
    loss = lse
    
    return loss, g, h_projected
