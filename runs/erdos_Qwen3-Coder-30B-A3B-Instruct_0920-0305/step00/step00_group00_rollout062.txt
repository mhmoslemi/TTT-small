import numpy as np
import time
from scipy.special import logsumexp
from scipy.optimize import minimize

def project_to_constraints(v, lower=0.0, upper=1.0, target_sum=None):
    if target_sum is None:
        target_sum = len(v) / 2.0
    # binary search for lambda
    lo = -np.max(v) - 1.0  
    hi = upper - np.min(v) + 1.0  
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, lower, upper)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo+hi)/2.0, lower, upper)
    return x

def compute_gradient(h, w, dx):
    n = len(h)
    grad = np.zeros_like(h)
    offset = n - 1
    for idx, wk in enumerate(w):
        if wk <= 1e-12:
            continue
        shift = idx - offset
        if shift >= 0:
            # first term: add wk * (1 - h[idx + shift])
            grad[:n-shift] += wk * (1 - h[shift:])
            # second term: subtract wk * h[idx - shift]
            grad[shift:] -= wk * h[:n-shift]
        else:
            s = -shift
            # first term: add wk * (1 - h[idx + shift]) = wk * (1 - h[idx - s])
            grad[s:] += wk * (1 - h[:n-s])
            # second term: subtract wk * h[idx - shift] = wk * h[idx + s]
            grad[:n-s] -= wk * h[s:]
    return grad * dx

def loss_and_grad(h, beta, dx):
    p = 1 - h
    c = np.correlate(h, p, mode='full') * dx
    # Compute log-sum-exp loss  
    scaled = beta * c
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w_sum = np.sum(w)
    w = w / w_sum
    loss = (max_scaled + np.log(w_sum)) / beta
    # compute gradient via w
    grad = compute_gradient(h, w, dx)
    return loss, grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 140
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    best_h = None
    best_c5 = np.inf
    
    # Try 5 restarts
    for restart in range(5):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
            
        # Initialize h
        h = np.random.random(n_points)
        h = project_to_constraints(h)
        
        # Vary beta to smooth the max
        beta_schedule = [1.0, 5.0, 15.0, 30.0]
        for beta in beta_schedule:
            # SLSQP optimization
            def fun(x):
                loss, grad = loss_and_grad(x, beta, dx)
                return loss, grad
            
            # Define constraint for sum = target_sum
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
            
            bounds = [(0.0, 1.0)] * n_points
            
            res = minimize(fun, h, method='SLSQP', jac=True, bounds=bounds,
                           constraints=cons, options={'maxiter': 100, 'ftol': 1e-9, 'disp': False})
            
            h = res.x
            
            # Check for improved solution
            if time.time() - start_time > budget_s * 0.95:
                break
                
        # Check final solution
        p = 1 - h
        c = np.correlate(h, p, mode='full') * dx
        c5 = c.max()
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final evaluation using provided function
    try:
        from evaluate_erdos_solution import evaluate_erdos_solution
        c5_eval = evaluate_erdos_solution(best_h)
    except Exception:
        # If the function is not available, use correlation directly
        c5_eval = best_c5
    
    return (best_h.tolist(), c5_eval, n_points)
