def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Phase 1: Seeding
    M = 6  # intervals
    n = 400  # grid size
    dx = 2.0 / n
    
    # Phase 2: Global DE
    def decode(z):
        z_ell, z_g = z[:M], z[M:]
        ell = np.exp(z_ell) / np.sum(np.exp(z_ell))
        g = np.exp(z_g) / np.sum(np.exp(z_g))
        return ell, g

    def build_intervals(ell, g):
        a = np.zeros(M + 1)
        a[0] = g[0]
        for i in range(1, M):
            a[i] = a[i-1] + ell[i-1] + g[i]
        a[M] = a[M-1] + ell[M-1]
        return a

    def candidate_shifts(a):
        E = []
        for i in range(len(a)-1):
            E.append(a[i])
            E.append(a[i+1])
        T = set([0.0])
        for i in range(len(E)):
            for j in range(len(E)):
                diff = abs(E[i] - E[j]) 
                if diff <= 2.0:
                    T.add(diff)
        return list(T)

    def intersection_length(t, a, b):
        if t == 0.0:
            return sum(b[i] - a[i] for i in range(len(a)-1))
        total = 0.0
        for i in range(len(a)-1):
            t1, t2 = a[i] - t, b[i] - t
            total += max(0.0, min(b[i], t2) - max(a[i], t1))
        return total

    def objective(z):
        ell, g = decode(z)
        a = build_intervals(ell, g)
        shifts = candidate_shifts(a)
        min_I_t = float('inf')
        for t in shifts:
            I_t = intersection_length(t, a, np.append(a[:-1], a[-1]))
            if I_t < min_I_t:
                min_I_t = I_t
        return 1.0 - min_I_t

    # DE with constraints in the objective (soft constraint support via exponential normalization)
    bounds = [(-5, 5)] * (2 * M + 1)
    de_results = []
    for _ in range(3):
        if time.time() - start_time > budget_s * 0.7:
            break
        res = differential_evolution(objective, bounds, maxiter=1000, popsize=15 * (2 * M + 1), seed=seed + _)
        de_results.append((res.fun, res.x))

    if not de_results:
        best_z = np.zeros(2 * M + 1)
        best_c5 = 1.0
    else:
        # Find best result so far from restarts
        best_c5, best_z = min(de_results, key=lambda x: x[0])

    time_remaining = budget_s * 0.7 - (time.time() - start_time)
    if time_remaining < 5:
        h_values = np.zeros(n)
        return h_values, best_c5, n

    # Phase 3: Local LP refinement
    ell, g = decode(best_z)
    a = build_intervals(ell, g)
    shifts = candidate_shifts(a)
    I_vals = [intersection_length(t, a, np.append(a[:-1], a[-1])) for t in shifts]

    min_I = min(I_vals)
    active_shifts = [t for t, i_val in zip(shifts, I_vals) if abs(i_val - min_I) < 1e-8]

    # Simplified attempt at constraint LP solving: directly adjust a little
    # Use simple numerical adjustments due to complexity of proper LP solving in pure environment
    # Start from the decoded solution and do some rounding or entropy minimization

    # Phase 4: Discretization
    h_values = np.zeros(n)
    a = build_intervals(ell, g)
    b = np.append(a[:-1], a[-1])
    # Value is 1 all along the intervals
    for i in range(n):
        x = (i + 0.5) * dx
        for j in range(M):
            if a[j] <= x <= b[j]:
                h_values[i] = 1.0
                break
    # Adjust the sum to be exactly n/2
    h_sum = np.sum(h_values)
    if abs(h_sum - n / 2) > 1e-3:
        # Find smallest contribution of 1 and reduce it, find smallest contribution of 0 and increase it
        # Try to fix it by swinging a few points
        diff = int(h_sum - n/2)  # -1 or +1     (with 400 grid, should be at most ±1)
        indices_of_1s = [i for i in range(n) if h_values[i] == 1.0]
        if diff > 0:
            # Reduce one 1 to 0
            idx_to_reduce = np.random.choice(indices_of_1s)
            h_values[idx_to_reduce] = 0.0
        elif diff < 0:
            # Increase one 0 to 1
            idx_to_inc = np.random.choice([i for i in range(n) if h_values[i] == 0.0])
            h_values[idx_to_inc] = 1.0

    # Phase 5: Binary hill-climbing
    time_remaining = budget_s - (time.time() - start_time)
    if time_remaining > 20:
        def compute_c5(h):
            # Use np.correlate the way it is done in the test script
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5

        best_c5_val = compute_c5(h_values)
        it = 0
        for it in range(10000):
            # Try to change any 1 to 0 or 0 to 1
            pos1 = np.random.choice(np.where(h_values == 1)[0])
            pos0 = np.random.choice(np.where(h_values == 0)[0])
            # Simulate swap
            h_new = h_values.copy()
            h_new[pos1] = 0.0
            h_new[pos0] = 1.0
            c5_new = compute_c5(h_new)
            if c5_new < best_c5_val:
                h_values = h_new
                best_c5_val = c5_new
            if time.time() - start_time > budget_s:
                break

    # Return final result
    c5_final = compute_c5(h_values)
    return h_values, c5_final, n
