import numpy as np
from scipy.optimize import differential_evolution
from cvxpy import Problem, Minimize, Variable, norm
import time
import random
from math import exp

def decode(z):
    """Decode raw parameters into interval lengths and gaps using softmax."""
    M = (len(z) - 1) // 2
    z_ell = z[:M]
    z_g = z[M:]
    ell = np.exp(z_ell) / np.sum(np.exp(z_ell))
    g = np.exp(z_g) / np.sum(np.exp(z_g))
    return ell, g

def build_intervals(ell, g):
    """Compute interval starts based on lengths and gaps."""
    a = np.zeros(len(ell))
    a[0] = g[0]
    for i in range(1, len(ell)):
        a[i] = a[i-1] + ell[i-1] + g[i]
    b = a + ell
    return a, b

def candidate_shifts(a, b):
    """Compute all candidate shift values that could affect overlap."""
    E = set()
    for x in a:
        E.add(x)
    for x in b:
        E.add(x)
    E = sorted(list(E))
    T = set([0.0])
    for i in range(len(E)):
        for j in range(len(E)):
            diff = abs(E[i] - E[j])
            if diff <= 2.0:
                T.add(diff)
    return sorted(list(T))

def intersection_length(t, a, b):
    """Compute overlap length for a given shift t."""
    total = 0.0
    for i in range(len(a)):
        # Compute intersection of [a[i], b[i]] and [a[i]-t, b[i]-t] (for t>0)
        # or [a[i]+t, b[i]+t] (for t<0)
        left = max(a[i], a[i] - t)
        right = min(b[i], b[i] - t)
        if t < 0:
            left = max(a[i], a[i] + t)
            right = min(b[i], b[i] + t)
        if left < right:
            total += right - left
    return total

def objective(z):
    """Evaluate the C5 bound for given parameters."""
    ell, g = decode(z)
    a, b = build_intervals(ell, g)
    shifts = candidate_shifts(a, b)
    min_overlap = float('inf')
    for t in shifts:
        overlap = intersection_length(t, a, b)
        min_overlap = min(min_overlap, overlap)
    return 1.0 - min_overlap

def run_DE(budget_s):
    """Run differential evolution with time limit."""
    M = 6 
    n_params = 2 * M + 1
    bounds = [(-5, 5)] * n_params
    
    start_time = time.time()
    best_result = None
    best_value = float('inf')
    time_limit = budget_s * 0.7
    
    while time.time() - start_time < time_limit:
        # Run DE with a fixed number of generations to control time usage
        maxiter = 2000
        popsize = 15 * n_params
        
        try:
            result = differential_evolution(
                objective,
                bounds,
                maxiter=maxiter,
                popsize=popsize,
                seed=hash(str(time.time()) + str(random.random())) % 1000000,
                mutation=(0.5, 1.0),
                recombination=0.7,
                disp=False,
                polish=False
            )
            
            if result.fun < best_value:
                best_value = result.fun
                best_result = result
        except Exception:
            break
            
    if best_result is None:
        # Fallback
        z0 = np.zeros(n_params)
        z0 = np.random.uniform(-2, 2, n_params)
        best_result = differential_evolution(objective, bounds, maxiter=500, popsize=20, seed=42, disp=False)
    
    return best_result

def local_LP_refinement(z_best, max_iter=5):
    """Improve the solution using local linear programming technique."""
    ell, g = decode(z_best)
    a, b = build_intervals(ell, g)
    shifts = candidate_shifts(a, b)
    min_overlap = float('inf')
    selected_t = 0.0
    for t in shifts:
        overlap = intersection_length(t, a, b)
        if overlap < min_overlap:
            min_overlap = overlap
            selected_t = t
    
    # Refine only if we still have a good solution
    for _ in range(max_iter):
        try:
            break  # Just return as placeholder for now
        except:
            pass
    return z_best

def gridify(intervals, n):
    """Convert intervals to a step function on a grid."""
    ell, g = decode(intervals)
    a, b = build_intervals(ell, g)
    h = np.zeros(n)
    dx = 2.0 / n
    for i in range(len(a)):
        start_idx = int(np.ceil(a[i] / dx))
        end_idx = int(np.floor(b[i] / dx))
        if start_idx < 0:
            start_idx = 0
        if end_idx > n:
            end_idx = n
        h[start_idx:end_idx] = 1.0
    total = np.sum(h)
    target = n / 2
    diff = int(target - total)
    if abs(diff) < 2:  # Only adjust if necessary
        if diff > 0:
            idx = np.argmin(h)
            h[idx] = 1.0
        elif diff < 0:
            idx = np.argmax(h)
            h[idx] = 0.0
    return h

def binary_hill_climb(h, time_remaining):
    """Perform hill climbing to slightly improve the binary solution."""
    start = time.time()
    current_c5 = np.correlate(h, 1-h, mode="full")
    dx = 2.0 / len(h)
    c5_val = np.max(current_c5) * dx
    best_c5 = c5_val
    best_h = h.copy()
    
    while time.time() - start < time_remaining:
        # Random swap
        i, j = random.sample(range(len(h)), 2)
        if h[i] == h[j]:
            continue
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_c5 = np.correlate(h_new, 1-h_new, mode="full")
        new_c5_val = np.max(new_c5) * dx
        
        if new_c5_val < c5_val:
            h = h_new
            c5_val = new_c5_val
            
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    M = 6
    n_params = 2 * M + 1
    n = 400
    
    # Phase 1: Global DE
    de_result = run_DE(budget_s)
    z_best = de_result.x
    # Phase 2: Local refinement
    z_best = local_LP_refinement(z_best)
    # Phase 3: Discretization
    h_values = gridify(z_best, n)
    # Phase 4: Binary hill climbing
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        h_values, c5_bound = binary_hill_climb(h_values, remaining_time)
    else:
        # Compute final C5
        c5_bound = np.max(np.correlate(h_values, 1-h_values, mode="full")) * (2.0 / n)
    
    n_points = len(h_values)
    return h_values, c5_bound, n_points
