import numpy as np
import cvxpy as cp
import time
import math

def full_overlap(x):
    return np.correlate(x, 1 - x, mode="full")

def worst_shift(x):
    overlaps = full_overlap(x)
    k_max = np.argmax(overlaps)
    return k_max, overlaps[k_max]

def build_mip(N, active_shifts):
    # Create binary variables
    x = cp.Variable(N, boolean=True)
    t = cp.Variable(integer=True)
    
    # Auxiliary variables wij,k
    w = {}
    constraints = []
    
    # Add half-ones constraint
    constraints.append(cp.sum(x) == N // 2)
    
    # Add symmetry breaking
    constraints.append(x[0] == 1)
    
    # Add constraints for each active shift
    for k in active_shifts:
        # Create auxiliary variables for shift k
        w[k] = cp.Variable((N - k), boolean=True)
        for i in range(N - k):
            # wij,k <= xi
            constraints.append(w[k][i] <= x[i])
            # wij,k <= x_{i+k}
            constraints.append(w[k][i] <= x[i + k])
            # wij,k >= xi + x_{i+k} - 1
            constraints.append(w[k][i] >= x[i] + x[i + k] - 1)
        
        # Ok = sum(xi) - sum(wij,k) <= t
        overlap_k = cp.sum(x[:N-k]) - cp.sum(w[k])
        constraints.append(overlap_k <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Build the problem
    prob = cp.Problem(objective, constraints)
    
    return prob, (x, t, w)

def solve_mip(prob, vars_dict, time_limit):
    x, t, w = vars_dict
    try:
        prob.solve(solver=cp.CBC, max_iters=10000, TimeLimit=time_limit, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value.astype(int), t.value, w
        return None, None, None
    except Exception:
        return None, None, None

            # We have optimal solution for current set
            final_c5 = t_sol * dx
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = x_sol.copy()
        else:
