import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize_scalar

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Problem setup
    n = 300  # samples, even for stability
    dx = 2.0 / n
    target_sum = n / 2.0
    max_time = time.time() + budget_s
    best_C5 = float('inf')
    best_h = None

    # Step 1: Generate a Beatty-based binary initialization (Golden ratio)
    beta = (np.sqrt(5) - 1) / 2
    initial_h = np.array([1 if (i * beta) % 1 < 0.5 else 0 for i in range(n)])
    
    # Ensure the sum is correct (adjust if needed)
    current_sum = np.sum(initial_h)
    diff = int(target_sum - current_sum)
    if diff != 0:
        # Flip some 0s to 1s, or 1s to 0s to get the right sum
        if diff > 0:
            indices = np.where(initial_h == 0)[0]
            flip_indices = indices[:diff]
            for idx in flip_indices:
                initial_h[idx] = 1
        else:
            indices = np.where(initial_h == 1)[0]
            flip_indices = indices[:abs(diff)]
            for idx in flip_indices:
                initial_h[idx] = 0

    # Project the initial binary vector onto the feasible set (identity operation)
    h = initial_h.copy().astype(float)
    h = _project_onto_box_and_hyperplane(h, target_sum)
    
    # Step 2: Gradient ascent on smooth surrogate
    α = 1.0
    α_max = 10000.0
    α_step = 100  # At every α_step steps, increase α
    iteration = 0


    def compute_signal_overlap(h_vec):
        # compute all S_k by FFT-based cross-correlation
        # h_vec is a 1D numpy array of length n
        # Compute s[i] = sum h_j * h_{j+i} where i is shift
        # Let's do convolution of h with h, shifted (all in the full mode of numpy.md)
        # Using a, b = h_vec, h_vec:
        # The correlation at shift k is sum_i h[i] * h[i+k] 
        # which is the full cross-correlation with mode="full", and we take indices from n-1 to 2n-2
        # The shifted version is to avoid confusion so let's use np.correlate with 1-h
        # Actually: (lengths n) 
        # conv_h = np.correlate(h_vec, h_vec, mode='full')
        # Here we want to match h(l) * h(l+k) over all valid indices.
        # index k (shift) is at pos n-1+k in conv_h vector
        # so need S_k = conv_h[n-1+k]
        conv_h = np.correlate(h_vec, h_vec, mode='full')
        S = np.array([conv_h[n-1 + k] for k in range(n)])  # valid range: k from 0 to n-1
        return S


    def compute_smooth_min(h_vec, alpha):
        S = compute_signal_overlap(h_vec)
        # Avoid numerical issues: clip S to be non-negative
        S = np.clip(S, 0, np.inf)
        w = np.exp(-alpha * S)
        w_sum = np.sum(w)
        if w_sum == 0:
            return float('inf')
        normalized_w = w / w_sum
        # Construct the gradient weight over h indices for each S_k
        # The gradient's i-th component is:
        # sum_k w_k * (h_{i+k} + h_{i-k})
        grad_sum = np.zeros_like(h_vec)
        for k in range(n):
            offset = k
            # Gradient contribution from S_k: 
            # ∂S_k/∂h_i = h_{i+k} + h_{i-k}
            # pad zeros in both directions
            h_rev = h_vec[::-1]
            w_k = w[k]
            # Left contributions
            left = np.zeros_like(h_vec)
            right = np.zeros_like(h_vec)
            # Consider 1d array as periodic in order to handle shift properly
            # Clean manual way for prod with shifted h
            h_lag = np.concatenate([h_vec, np.zeros(k)])
            h_lead = np.concatenate([np.zeros(k), h_vec])
            grad_sum += w_k * (h_lag[:n] + h_lead[:n])  # Add contribution from k shift 

        # The smooth surrogate function minus sign: to maximize it
        phi = - (1 / alpha) * np.log(w_sum)
        if np.isnan(phi) or np.isinf(phi):
            phi = float('inf')
        # Now compute the gradient directly using a clear method:
        # The smooth surrogate is Φ_α = - (1/α) log(Σ exp(-α S_k))
        # dΦ/dh_i = (1/α) * ( Σ w_k * ∂S_k/∂h_i ) / (Σ w_k) = (1/α) * (grad_sum / w_sum)  
        gradient = (1/alpha) * (grad_sum / w_sum) 
        return phi, gradient


    # Core gradient ascent loop
    τ = time.time()
    while time.time() < max_time:
        if iteration % (α_step) == 0 and α < α_max:
            α = min(α * 2.0, α_max)
        iteration += 1

        # Compute current smooth objective and gradient
        try:
            phi, grad = compute_smooth_min(h, α)
        except Exception as e:
            continue

        # Perform gradient ascent
        ε = 1.0  # step size
        while True:
            h_new = h + ε * grad
            h_new = _project_onto_box_and_hyperplane(h_new, target_sum)
            try:
                phi_new, _ = compute_smooth_min(h_new, α)
            except:
                ε /= 2.0
                continue
            if phi_new >= phi + 1e-10:
                h = h_new
                if phi_new < 0:
                    phi_new = 0
                break
            else:
                ε /= 2.0
                if ε < 1e-8:
                    break


        # Compute true C5 (max overlap integral)
        c_full = np.correlate(h, 1-h, mode='full') 
        # Max over all possible shifts including full
        phase_shift = n - 1
        C5_test = np.max(c_full) * dx
        # Record the best solution
        if C5_test < best_C5:
            best_h = h.copy()
            best_C5 = C5_test

    # Step 3: Discrete polishing
    # Threshold rounding to binary
    thresh_val = np.partition(best_h, int(target_sum))[int(target_sum) - 1]  # median
    h_final = np.array([1.0 if x > thresh_val else 0.0 for x in best_h])
    
    # Fix sum exactly to target_sum (some rounding error)
    true_ones = np.sum(h_final)
    diff = int(target_sum - true_ones)
    if diff > 0:
        flip_indices = np.where(h_final == 0)[0][:diff]
        for idx in flip_indices:
            h_final[idx] = 1.0
    elif diff < 0:
        flip_indices = np.where(h_final == 1)[0][:abs(diff)]
        for idx in flip_indices:
            h_final[idx] = 0.0

    # Swap-based local search
    def compute_C5(h_vec):
        c_full = np.correlate(h_vec, 1-h_vec, mode='full') 
        return np.max(c_full) * dx

    # Hill climbing
    current_C5 = compute_C5(h_final)
    remaining_time = max_time - time.time()

    while time.time() < max_time - 2 and remaining_time > 0.1:
        # pick two random indices to swap
        idx1, idx2 = np.random.choice(n, size=2, replace=False)
        if idx1 == idx2:
            continue
        h_new = h_final.copy()
        h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
        new_C5 = compute_C5(h_new)
        if new_C5 < current_C5:
            h_final = h_new
            current_C5 = new_C5

    # Final C5
    final_C5 = compute_C5(h_final)

    return h_final.tolist(), float(final_C5), n

def _project_onto_box_and_hyperplane(h, target_sum):
    """Project h ∈ ℝⁿ onto the set {x ∈ ℝⁿ: 0 ≤ x_i ≤ 1, sum(x) = target_sum}"""
    # First clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-9:
        return h.copy()
    # Waterfilling
    # We want a λ such that sum_i clip(h_i + λ, 0, 1) = target_sum
    # The function g(λ) = sum_i clip(h_i + λ, 0, 1) is piecewise affine
    # Reasonable bounds for λ: λ ∈ [-1,1] in practice
    λ_lo = -1.0
    λ_hi = 1.0
    tol = 1e-9
    max_iter = 100

    for _ in range(max_iter):
        λ_m = (λ_lo + λ_hi)/2  
        h_test = np.clip(h + λ_m, 0, 1)
        test_sum = np.sum(h_test)
        if abs(test_sum - target_sum) < tol:
            break
        elif test_sum > target_sum:
            λ_hi = λ_m
        else:
            λ_lo = λ_m

    h = np.clip(h + λ_m, 0, 1)
    return h
