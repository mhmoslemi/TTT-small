import numpy as np

def project_onto_simplex_box(h, target_sum):
    """Project h onto the simplex {x: sum(x) = target_sum, 0 <= x <= 1}"""
    # First clip to [0,1]
    h = np.clip(h, 0, 1)
    # Then adjust to meet sum constraint
    if np.sum(h) > target_sum:
        # Reduce values to decrease sum
        diff = np.sum(h) - target_sum
        # Sort indices in decreasing order of values
        sort_idx = np.argsort(h)[::-1]
        cumsum = 0
        for i, idx in enumerate(sort_idx):
            if i == len(sort_idx) - 1:
                # remaining amount equally distributed
                h[idx] = max(0, h[idx] - diff)
                break
            # amount that can be reduced from h[idx] to make sum = target_sum
            amount = min(h[idx], (target_sum - cumsum) / (len(sort_idx) - i))
            h[idx] = max(0, h[idx] - amount)
            cumsum += amount
            diff = target_sum - cumsum
    elif np.sum(h) < target_sum:
        # Increase values to increase sum
        diff = target_sum - np.sum(h)
        # Sort indices in increasing order of values
        sort_idx = np.argsort(h)
        cumsum = 0
        for i, idx in enumerate(sort_idx):
            if i == len(sort_idx) - 1:
                h[idx] = min(1, h[idx] + diff)
                break
            # amount that can be added to h[idx] to make sum = target_sum
            amount = min(1 - h[idx], (target_sum - cumsum) / (len(sort_idx) - i))
            h[idx] = min(1, h[idx] + amount)
            cumsum += amount
            diff = target_sum - cumsum
    return h

def compute_c5(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode="full") * dx)"""
    dx = 2.0 / len(h)
    S_raw = np.correlate(h, 1 - h, mode='full')
    return np.max(S_raw) * dx

def optimize_step(h, step_size=0.01, beta=1.0, max_iter=50):
    """Perform one optimization step"""
    N = len(h)
    dx = 2.0 / N
    
    # Compute all correlation values
    S_raw = np.correlate(h, 1 - h, mode='full') 
    max_idx = np.argmax(S_raw)
    max_value = S_raw[max_idx]
    
    # Derivative computation
    l = max_idx - (N - 1)  # lag index
    
    # Compute gradient for lag l
    grad = np.zeros(N)
    
    if l >= 0:
        # indices where both terms are valid:
        i_start = 0
        i_end = N - l - 1
        for i in range(i_start, i_end + 1):
            grad[i] += 1 - h[i + l]
            grad[i + l] -= h[i]
    else:
        # For negative l
        shift = -l  # make it positive
        i_start = shift
        i_end = N - 1
        for i in range(i_start, i_end + 1):
            grad[i] += 1 - h[i - shift]
            grad[i - shift] -= h[i]
    
    # Update step
    h_new = h - step_size * grad
    
    # Project onto feasible region
    h_new = project_onto_simplex_box(h_new, target_sum=N / 2)
    
    return h_new
