import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h"""
    n = len(h)

    # Compute the overlap integral using full convolution
    correlation = np.correlate(h, 1 - h, mode='full')
    # The maximum value among all possible shifts
    max_correlation = np.max(correlation)
    return max_correlation * dx
