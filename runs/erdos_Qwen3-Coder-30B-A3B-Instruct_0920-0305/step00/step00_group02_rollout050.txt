```python
import time
import numpy as np
import cvxpy as cp
import random
import math

def random_binary_vector(n):
    """Generate a random binary vector with exactly n/2 ones."""
    if n % 2 != 0:
        raise ValueError("n must be even")
    h = np.zeros(n, dtype=int)
    ones_indices = np.random.choice(n, n//2, replace=False)
    h[ones_indices] = 1
    return h

def compute_c5(h):
    """Compute C5 bound for the given step function h."""
    n = len(h)
    dx = 2.0 / n
    total = 1.0
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Maximum correlation is at indices where full overlap occurs
    # We are interested in all shifts, so take max from full correlation
    max_corr = np.max(corr)
    c5 = max_corr * dx
    return c5

def solve_milp(n, time_limit=10):
    """Solve MILP for optimal binary h to minimize C5."""
    # Set up variables
    h = cp.Variable(n, boolean=True)  # h must be binary
    t = cp.Variable()  # represents minimum overlap value

    # Constraint: sum(h) = n/2
    constraints = [cp.sum(h) == n // 2]
    
    # For each shift, introduce linearized products
    # Shift k indexes that overlap: i from 0 to n-k-1
    for k in range(1, n):  # skipping k=0 to give t expected upper bound which is not critical
        # This corresponds to sum_{i=0}^{n-k-1} h_i * h_{i+k}
        # Constrain sum to be at least t
        # y represents h_i * h_{i+k} for i such that i+k < n
        y = cp.Variable(n - k, nonneg=True)  # This boolean constraint in solve
        # Linearization
        # y <= h_i, y <= h_{i+k}, y >= h_i + h_{i+k} - 1, y >= 0
        # So we define y = h_i * h_{i+k} with given constraints
        constraints += [y <= h[:n-k]]
        constraints += [y <= h[k:]]
        constraints += [y >= h[:n-k] + h[k:] - 1]
        constraints += [cp.sum(y) >= t]

    # Optimization objective: maximize t subject to constraints
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)

    # Solve with limited time
    try:
        prob.solve(solver=cp.CBC, max_seconds=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_EOF]:
            h_sol = np.round(h.value).astype(int)
            # Compute real C5 value
            dx = 2.0 / n
            total = 1.0
            corr = np.correlate(h_sol, 1 - h_sol, mode="full")
            c5 = np.max(corr) * dx
            return h_sol, c5
        else:
            raise Exception("MILP solver did not find an optimal solution")
    except Exception:
        # Fallback to heuristic if MILP fails
        h_sol = random_binary_vector(n)
        c5 = compute_c5(h_sol)
        return h_sol, c5


def local_search(h_initial, max_iters, time_limit):
    """Perform local search to improve solution."""
    start = time.time()
    cur = h_initial.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5

    # Precompute initial overlap vector
    n = len(cur)
    dx = 2.0 / n
    total = 1.0
    
    # For fast comp. of correlation
    def compute_overlap():
        corr = np.correlate(cur, cur, mode="full")
        return corr[n-1:]  # Overlaps for k=0 to n-1

    while time.time() - start < time_limit:
        for _ in range(max_iters):
            # Sample positions where current value is 1 or 0
            ones_idx = np.where(cur == 1)[0]
            zeros_idx = np.where(cur == 0)[0]
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
            i = np.random.choice(ones_idx)
            j = np.random.choice(zeros_idx)
            
            # Swap
            cur[i] = 0
            cur[j] = 1
            
            # Recompute C5
            new_c5 = compute_c5(cur)
            if new_c5 < cur_c5:
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best = cur.copy()
            else:
                # Revert
                cur[i] = 1
                cur[j] = 0
            
            if time.time() - start >= time_limit:
                break

    return best, best_c5


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    n = 100
    try:
        # Use MILP for a good starting point quickly
        milp_time = min(10, budget_s // 4)
        h_opt, c5_opt = solve_milp(n, time_limit=milp_time)
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining > 5:
            # Use local search for remaining time
            best_h, best_c5 = local_search(h_opt, 100000, remaining)
        else:
            best_h, best_c5 = h_opt, c5_opt

    except Exception:
        # fallback to random large-scale local search
        h_opt = random_binary_vector(n)
        c5_opt = compute_c5(h_opt)
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining > 5:
            best_h, best_c5 = local_search(h_opt, 100000, remaining)
        else:
            best_h, best_c5 = h_opt, c5_opt

    return best_h.tolist(), float(best_c5), n
```
