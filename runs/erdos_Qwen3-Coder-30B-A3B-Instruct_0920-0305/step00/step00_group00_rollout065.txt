import numpy as np
import time
from scipy.special import softmax
from scipy.signal import correlate

def project_onto_feasible(h, target_sum):
    """Project h onto {0 ≤ h[i] ≤ 1, sum h[i] = target_sum} using bisection."""
    h = np.asarray(h).copy()
    
    def check_sum(lam):
        ph = np.clip(h + lam, 0, 1)
        return np.sum(ph) - target_sum
    
    a, b = -1.5, 1.5
    while check_sum(a) < 0:
        a -= 1.0
    while check_sum(b) > 0:
        b += 1.0

    epsilon = 1e-12
    for _ in range(60):
        mid = (a + b) / 2
        if check_sum(mid) < 0:
            a = mid
        else:
            b = mid
        if abs(a - b) < epsilon:
            break
    
    lam = (a + b) / 2
    h_proj = np.clip(h + lam, 0, 1)
    
    # Ensure exact sum using water-filling on free components
    diff = target_sum - np.sum(h_proj)
    free = np.where((h_proj > 0) & (h_proj < 1))[0]
    if len(free) > 0:
        delta = diff / len(free)
        if delta < 0:
            h_proj[free] = np.clip(h_proj[free] + delta, 0, 1)
        else:
            h_proj[free] = np.clip(h_proj[free] + delta, 0, 1)
            if np.sum(h_proj) > target_sum:
                h_proj[free] = np.clip(h_proj[free] - (np.sum(h_proj) - target_sum) / len(free), 0, 1)
    
    # Final protection
    if np.sum(h_proj) < target_sum:
        # Add to 0 components
        zeros = np.where(h_proj == 0)[0]
        if len(zeros) > 0:
            delta = (target_sum - np.sum(h_proj)) / len(zeros)
            h_proj[zeros] = np.clip(h_proj[zeros] + delta, 0, 1)
    elif np.sum(h_proj) > target_sum:
        # Subtract from 1 components
        ones = np.where(h_proj == 1)[0]
        if len(ones) > 0:
            delta = (np.sum(h_proj) - target_sum) / len(ones)
            h_proj[ones] = np.clip(h_proj[ones] - delta, 0, 1)

    return h_proj

def compute_correlations(h, dx):
    """Compute all correlations c_k = dx * sum h[i] * (1 - h[i+k]) for all k."""
    h1 = 1 - h
    # correlate computes full convolution (zero-padded) of h1 with h
    # c[k] = sum_{i} h[i]*h1[i+k] = sum_{i} h[i]* (1 - h[i+k])
    c = correlate(h, h1, mode='full') * dx
    
    # Extract valid correlations (only those in the range [0, n-1])
    n = len(h)
    return c[n-1:]

def compute_gradient(c, h, dx):
    """Compute the gradient of LSE loss with weights from c."""
    n = len(h)
    g = np.zeros_like(h)
    
    # Compute weights
    weights = softmax(c)  # Normalize to avoid overflow
    if np.sum(np.isnan(weights)) > 0:
        weights = np.ones_like(weights) / len(weights)
    
    # Build gradient as weighted sum of individual gradients for each shift k
    for k in range(len(c)):
        # Consistent with correlate result
        if k == 0:
            # i+k = i, so g[i] += dx*(1-h[i])
            g += dx * (1 - h)
        else:
            # i+k < n -> h2[i+k] contributes, i-k >=0 -> h[i-k] contributes
            # g[i] += dx*(1-h[i+k]) if i+k < n
            # g[i] -= dx*h[i-k] if i-k >= 0
            g_shift_plus = np.zeros_like(g)
            g_shift_minus = np.zeros_like(g)
            
            if k < n:
                g_shift_plus[k:] = dx * (1 - h[:-k])
            if k < n:
                g_shift_minus[:-k] = dx * h[k:]
            g += weights[k] * (g_shift_plus - g_shift_minus)
    return g

def evaluate_erdos_solution(h):
    return max(correlate(h, 1-h, mode='full') * (2.0/len(h)))

    max_iter_per_beta = 150
    alpha0 = 0.2
    
    best_C5 = float('inf')

    else:
        h_init = project_onto_feasible(np.ones(n) * 0.5, target_sum)
    
    # Try multiple random starts
    starts = [h_init]
    for _ in range(3):
        h_rand = np.random.random(n)
        h_rand = project_onto_feasible(h_rand, target_sum)
        starts.append(h_rand)
    
    for h in starts:
        # Make sure we are within our time budget
        if time.time() - t0 > 0.95 * budget_s:
            break
        
        h_current = h.copy()
        loss_current = 0
        
        for beta in beta_schedule:
            for it in range(max_iter_per_beta):
                # Early exit if good enough
                if time.time() - t0 > 0.95 * budget_s:
                    break
                
                # Evaluate correlations (c_k = dx * sum h[i]*h1[i+k])
                c = compute_correlations(h_current, dx)

                # True C5
                current_C5 = max(c)
                if current_C5 < best_C5:
                    best_C5 = current_C5
                    best_h = h_current.copy()
                
                if current_C5 < 0.379:
                    break

                # Approximate gradient w.r.t. log-sum-exp loss (L_beta)
                grad = compute_gradient(c, h_current, dx)
                
                if np.linalg.norm(grad) < 1e-8:
                    break
                
                # Line search with backtracking
                alpha = alpha0 / (1.0 + it)
                max_trials = 5
                for _ in range(max_trials):
                    h_new = project_onto_feasible(h_current - alpha * grad, target_sum)
                    c_new = compute_correlations(h_new, dx)
                    loss_new = max(c_new)  # Just use the actual maximum for convergence check
                    if loss_new <= loss_current or loss_new < 0.380:
                        break
                    alpha *= 0.5
                
                h_current = h_new
                loss_current = loss_new

    # Optional refinement further
    if best_h is not None and time.time() - t0 < 0.95 * budget_s:
        # Binary rounding + local swapping
        h_binary = np.zeros_like(best_h)
        ones = np.argsort(best_h)[::-1][:int(np.round(target_sum))]
        h_binary[ones] = 1.0
        h_binary *= target_sum / np.sum(h_binary)
        
        # Refinement via local swaps
        c = compute_correlations(h_binary, dx)
        current_C5 = max(c)
        success = True
        
        while success and time.time() - t0 < 0.95 * budget_s:
            success = False
            for i in range(n):
                for j in range(n):
                    if h_binary[i] > 0.5 and h_binary[j] < 0.5:
                        # Swap
                        h_binary[i] = 0
                        h_binary[j] = 1
                        
                        c_new = compute_correlations(h_binary, dx)
                        new_C5 = max(c_new)
                        
                        if new_C5 < current_C5:
                            current_C5 = new_C5
                            success = True
                        else:
                            h_binary[i] = 1
                            h_binary[j] = 0
                    elif h_binary[i] < 0.5 and h_binary[j] > 0.5:
                        # Swap
                        h_binary[i] = 1
                        h_binary[j] = 0
                        
                        c_new = compute_correlations(h_binary, dx)
                        new_C5 = max(c_new)
                        
                        if new_C5 < current_C5:
                            current_C5 = new_C5
                            success = True
                        else:
                            h_binary[i] = 0
                            h_binary[j] = 1

        # Final evaluation
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_binary.copy()

    # Final check
    if best_h is None:
        h_initial = np.ones(n) * 0.5
        h_initial = project_onto_feasible(h_initial, target_sum)
        best_C5 = evaluate_erdos_solution(h_initial)
        best_h = h_initial
    
    # Final evaluation to make sure things are clean
    evaluated_bound = evaluate_erdos_solution(best_h)
    return (best_h, evaluated_bound, n)
