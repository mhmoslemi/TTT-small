import numpy as np
import time
from scipy.special import softmax

def project_onto_box_and_sum(h, target_sum):
    """Project h onto the set {0 <= h[i] <= 1, sum(h) = target_sum}."""
    # Binary search for the right lambda
    h = h.copy()
    lb, ub = -100.0, 100.0
    tol = 1e-12
    for _ in range(100):
        lam = (lb + ub) / 2.0
        x = np.clip(h + lam, 0, 1)
        s = np.sum(x)
        if s < target_sum:
            ub = lam
        else:
            lb = lam
        if abs(s - target_sum) < tol:
            break
    # Final adjustment to ensure exact sum
    x = np.clip(h + lam, 0, 1)
    s = np.sum(x)
    diff = target_sum - s
    # Add diff to components that are not already at 1
    mask = x < 1
    if np.sum(mask) > 0:
        x[mask] += diff / np.sum(mask)
    return np.clip(x, 0, 1)

def correlate_h_with_1_minus_h(h, dx):
    """Compute cross-correlations c_k = dx * sum(h[i] * (1 - h[i+k]))"""
    # This is slightly more efficient than using np.correlate directly
    n = len(h)
    c = np.zeros(2*n - 1)
    for k in range(-n+1, n):
        # For k >= 0, h[i] * (1 - h[i+k]) for i in [0, n-k-1]
        # For k < 0, h[i+k] * (1 - h[i]) for i in [0, n+k-1]
        if k >= 0:
            for i in range(n - k):
                c[k] += h[i] * (1 - h[i + k])
        else:
            for i in range(n + k):
                c[k] += h[i + k] * (1 - h[i])
        c[k] *= dx
    return c

def gradient_of_correlation(h, k):
    """Returns ∇c_k(h), where c_k = dx * sum(h[i] * (1 - h[i+k]))"""
    n = len(h)
    g = np.zeros(n)
    # Add (1 - h[i+k]) * dx for valid i values where i+k is in range
    if k >= 0:
        for i in range(n - k):
            g[i] += (1 - h[i + k]) * (-dx)
    else:
        for i in range(-k, n):
            g[i] += (1 - h[i + k]) * (-dx)
    # Subtract h[i-k] * dx for valid i values where i-k is in range
    if k <= 0:
        for i in range(-k, n):
            g[i] -= h[i - k] * dx
    else:
        for i in range(n - k):
            g[i] -= h[i - k] * dx
    return g

def gradient_of_LSE(h, c, w, dx):
    """Returns gradient of LSE β * c_k weighted by w_k"""
    n = len(h)
    g = np.zeros(n)
    for k in range(-n + 1, n):
        if k >= 0:
            # Add w_k * (1 - h[i+k]) * dx
            g[:n-k] += w[k + n - 1] * (1 - h[k:]) * dx
            # Subtract w_k * h[i-k] * dx
            g[k:] -= w[k + n - 1] * h[:n-k] * dx
        else:
            # Add w_k * (1 - h[i+k]) * dx
            g[-k:] += w[k + n - 1] * (1 - h[:n+k]) * dx
            # Subtract w_k * h[i-k] * dx
            g[:n+k] -= w[k + n - 1] * h[-k:] * dx
    return g

def evaluate_erdos_solution(h):
    """Copied from eval function for checking"""
    h = np.array(h)
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(c)

    max_iter_per_β = 150
    α0 = 0.2

    best_C5 = 100.0
    best_h = initial_h.copy()

    initial_starts = [initial_h]
    for _ in range(3):
        h_random = np.random.rand(n)
        h_random = project_onto_box_and_sum(h_random, target_sum)
        initial_starts.append(h_random)
    
    for start_i, h in enumerate(initial_starts):
        if time.time() - t0 > 0.95 * budget_s:
            break
            
        h = h.copy()
        # compute initial c vector to initialize things
        c = correlate_h_with_1_minus_h(h, dx)
        current_C5 = np.max(c)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
        
        # For each β stage
        for β_idx, β in enumerate(β_schedule):
            if time.time() - t0 > 0.95 * budget_s:
                break
            for it in range(max_iter_per_β):
                if time.time() - t0 > 0.95 * budget_s:
                    break
                    
                # Evaluate c vector
                c = correlate_h_with_1_minus_h(h, dx)
                
                # LSE weights
                w = softmax(β * c)
                
                # gradient of LSE
                g = gradient_of_LSE(h, c, w, dx)
                
                # early stopping if no improvement
                g_norm = np.sqrt(np.sum(g**2))
                if g_norm < 1e-8:
                    break
                    
                # backtracking line-search
                α = α0 / (1 + it)
                max_trials = 20
                for trial in range(max_trials):
                    h_new = project_onto_box_and_sum(h - α * g, target_sum)
                    
                    c_new = correlate_h_with_1_minus_h(h_new, dx)
                    loss_new = np.max(c_new)
                    
                    if loss_new <= np.max(c) * 1.01:
                        break
                    α *= 0.5
                if trial == max_trials - 1:
                    # fallback, don't get caught out by loops
                    h_new = project_onto_box_and_sum(h, target_sum)
                
                h = h_new
                current_C5 = np.max(c_new)
                if current_C5 < best_C5:

    if time.time() - t0 < 0.95 * budget_s:
        # Keep 1s and 0s, make a binary version and swap
        h_best = best_h.copy()
        # Round to binary values while preserving sum
        h_binary = np.round(h_best).astype(float)
        if np.sum(h_binary) != target_sum:
            # Adjust to correct sum
            diff = int(target_sum - np.sum(h_binary))
            if diff > 0:
                # Get indices where we can flip from 0 to 1
                indices = np.where(h_binary == 0)[0]
                indices = indices[:diff]
                for i in indices:
                    h_binary[i] = 1
            else:
                # Get indices where we can flip from 1 to 0
                indices = np.where(h_binary == 1)[0]
                indices = indices[:abs(diff)]
                for i in indices:
                    h_binary[i] = 0
        # Recompute the C5
        c_binary = correlate_h_with_1_minus_h(h_binary, dx)
        current_C5_refine = np.max(c_binary)
        if current_C5_refine < best_C5:

            best_h = h_binary.copy()
    
    if time.time() - t0 > 0.95 * budget_s:

        return (best_h, final_c5, n)
    
    # Final computation
