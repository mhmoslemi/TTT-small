```python
import numpy as np
from scipy.optimize import linprog
from cvxpy import *
import time
import math

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all cross-correlations
    c = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the negative of the cross-correlation
    # So we need to find
    return max(c) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 80  # Moderate number to keep LP manageable
    dx = 2.0 / n_points
    total_mass = n_points / 2.0 
    
    # Initialize h from initial construction (if available)
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = h_init / np.sum(h_init) * total_mass
    else:
        h_init = np.full(n_points, 0.5)
    
    # Bisection parameters
    lower_t = 0.0
    upper_t = 0.4  # Start with a reasonable upper bound
    tolerance = 1e-5
    best_solution = None
    best_c5 = float('inf')
    
    # For damping to prevent too aggressive rounding
    epsilon = 1e-3 
    
    while upper_t - lower_t > tolerance and (time.time() - start_time) < budget_s - 1:
        t_mid = (lower_t + upper_t) / 2.0
        print(f"Testing t = {t_mid:.6f}")
        
        # Solve LP feasibility problem
        try:
            # Setup LP variables
            h_vars = [Variable() for _ in range(n_points)]
            p_vars = {}
            for d in range(-n_points+1, n_points):
                if d >= 0:
                    p_vars[d] = [Variable() for i in range(n_points - d)]
                else:
                    p_vars[d] = [Variable() for i in range(n_points + d)]
            
            # Constraints for each variable
            constraints = []
            # Bound on h
            for i in range(n_points):
                constraints.extend([h_vars[i] >= 0, h_vars[i] <= 1])
                
            # Mass constraint
            mass_constraint = sum(h_vars) == total_mass
            constraints.append(mass_constraint)
            
            # McCormick envelope constraints for each d and overlapping pair
            for d in range(-n_points + 1, n_points):
                if d >= 0:
                    for i in range(n_points - d):
                        h_i = h_vars[i]
                        h_j = h_vars[i + d]
                        p = p_vars[d][i]
                        constraints.extend([
                            p <= h_i,            # p <= h_i
                            p <= h_j,            # p <= h_j
                            p >= h_i + h_j - 1,  # p >= h_i + h_j - 1
                            p >= 0               # p >= 0
                        ])
                else:
                    for i in range(n_points + d):
                        h_i = h_vars[i]
                        h_j = h_vars[i - d]
                        p = p_vars[d][i]
                        constraints.extend([
                            p <= h_i,
                            p <= h_j,
                            p >= h_i + h_j - 1,
                            p >= 0
                        ])
                        
            # Additional shift integral constraint
            w_target = (1.0 - t_mid) / dx
            for d in range(-n_points + 1, n_points):
                if d >= 0:
                    sum_p = sum(p_vars[d])
                else:
                    sum_p = sum(p_vars[d])
                constraints.append(sum_p >= w_target)
                
            # Objective function (minimize target t): we don't minimize per se, 
            # but just check feasibility
            prob = Problem(Minimize(0), constraints)
            prob.solve(solver=CPLEX, verbose=False)
            
            if prob.status == 'optimal':  # Feasible
                print(f"Set t = {t_mid:.6f} is feasible.")
                # Extract h values
                h_solution = np.array([float(h_var.value) for h_var in h_vars])
                # Round to binary
                h_binary = np.zeros(n_points)
                sorted_indices = np.argsort(h_solution)[::-1]
                mass_left = total_mass
                for i in sorted_indices:
                    if mass_left >= 1:
                        h_binary[i] = 1
                        mass_left -= 1
                    elif mass_left > 0:
                        h_binary[i] = mass_left
                        mass_left = 0
                    else:
                        break
                if np.abs(np.sum(h_binary) - total_mass) < 1e-5:
                    c5_approx = evaluate_erdos_solution(h_binary)
                    print(f"Binary solution achieves C₅ = {c5_approx:.6f}")
                    if c5_approx < best_c5:
                        best_c5 = c5_approx
                        best_solution = h_binary.copy()
                        print(f"New best C₅ = {best_c5:.6f}")
                else:
                    print("Sum check failed in rounding")
                upper_t = t_mid
            else:
                print(f"Set t = {t_mid:.6f} is infeasible.")
                lower_t = t_mid
                
        except Exception as e:
            print(f"Error solving LP at t = {t_mid}: {e}")
            lower_t = t_mid

    # Final refinement search if needed
    if best_solution is None:
        h_best = h_init
        best_c5 = evaluate_erdos_solution(h_init)
    else:
        h_best = best_solution
    
    print(f"Final solution with C₅ = {best_c5:.6f}")
    return (h_best.tolist(), best_c5, n_points)
```
