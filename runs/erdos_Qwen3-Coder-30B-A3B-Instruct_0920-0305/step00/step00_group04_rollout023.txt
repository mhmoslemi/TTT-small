def evaluate_erdos_solution(h_vals, n_points=None):
    """Evaluate the C5 constant for given h values"""

    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum of these values scaled by dx
    c5 = max(corr) * dx
    return c5
