def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    # compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode='full')
    # take the max correlation (over shifts k=0,...,n-1)
    max_corr = np.max(corr[n-1:])
    return max_corr * dx

        A = sparse.csr_matrix((data, (row, col)), shape=(n,n))
        A_k_list.append(A)
    print("Done.")

    constraints += [x[i] <= 1 for i in range(n)]
    # sum constraint
