```python
import numpy as np
import cvxpy as cp
import random
import time
from scipy.sparse import csr_matrix
import math

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    n = 200  # Grid points, even and moderate for SDP feasibility
    dx = 2.0 / n
    half_n = n // 2
    
    # Time allocation
    # Phase 1: SDP construction
    if time.time() - start_time > budget_s - 200:
        return None, float('inf'), n
    A_k_list = []
    for k in range(1, half_n + 1):
        row = np.arange(n - k)
        col = np.arange(k, n)
        data = np.ones(n - k)
        A_k = csr_matrix((data, (row, col)), shape=(n, n))
        A_k = A_k + A_k.T  # Symmetrize
        A_k_list.append(A_k)
    
    # Phase 2: SDP solve
    if time.time() - start_time > budget_s - 180:
        return None, float('inf'), n
    X = cp.Variable((n, n), symmetric=True)
    x = cp.Variable(n)
    t = cp.Variable()
    
    # Constraints
    constrs = [X >> 0]  # PSD
    constrs.append(cp.diag(X) == x)  # diagonal
    constrs.append(cp.sum(x) == half_n)  # sum = n/2
    constrs.append(0 <= x)
    constrs.append(x <= 1)  # box constraint
    
    for A_k in A_k_list:
        constrs.append(cp.trace(A_k @ X) >= 2 * t)
    
    prob = cp.Problem(cp.Maximize(t), constrs)
    try:
        prob.solve(solver=cp.SCS, max_iters=2000, eps=1e-4)
        t_opt = t.value
        if t_opt is None:
            t_opt = 0.0
    except Exception:
        t_opt = 0.0
    X_opt = X.value
    
    # Phase 3: Rounding
    if time.time() - start_time > budget_s - 150:
        return None, float('inf'), n
    
    best_h = np.zeros(n)
    best_C5 = float('inf')

    # Factorization with low rank
    K = min(10, n)
    try:
        # May not be exactly low-rank in strict sense, but approximate
        U, s, Vt = np.linalg.svd(X_opt, full_matrices=False)
        V = U[:, :K] * np.sqrt(s[:K])
    except Exception:
        # fallback to Cholesky-like
        V = np.random.rand(n, 5)  # General case
        
    num_rounds = 250
    for _ in range(num_rounds):
        if time.time() - start_time > budget_s - 150:
            break
        g = np.random.normal(0, 1, K)
        s = V @ g
        sorted_s = np.sort(s)[::-1]
        # Find threshold for exactly half
        if half_n == 0:
            continue
        threshold = sorted_s[half_n-1]
        h = (s >= threshold).astype(int)
        # Ensure sum constraint exactly
        sum_h = np.sum(h)
        while sum_h < half_n:
            # Brute force increase 0s to 1s
            for i in range(n):
                if h[i] == 0:
                    h[i] = 1
                    sum_h += 1
                    if sum_h >= half_n:
                        break
        while sum_h > half_n:
            # Find a 1 to set to 0
            for i in range(n):
                if h[i] == 1:
                    h[i] = 0
                    sum_h -= 1
                    if sum_h <= half_n:
                        break
        # Compute C5
        # Convert to list to use np.correlate
        h_list = h.tolist()
        corr_full = np.correlate(h_list, [1 - x for x in h_list], mode='full')
        c5_val = np.max(corr_full) * dx
        if c5_val < best_C5:
            best_C5 = c5_val
            best_h = h.copy()

    # Phase 4: Local polishing by swap moves
    if time.time() - start_time > budget_s - 150:
        return best_h.tolist(), best_C5, n
    max_swaps = 100000  # Rough estimate based on remaining time
    swaps_done = 0
    h_best = best_h.copy()
    while swaps_done < max_swaps and time.time() - start_time < budget_s - 10:
        corr_full = np.correlate(h_best.tolist(), [1 - x for x in h_best.tolist()], mode='full')
        min_k = np.argmin(corr_full)
        # now we want to swap 
        ones = [i for i in range(n) if h_best[i] == 1]
        zeros = [i for i in range(n) if h_best[i] == 0]
        if not ones or not zeros:
            break
        i = random.choice(ones)
        j = random.choice(zeros)
        h_new = h_best.copy()
        h_new[i] = 0
        h_new[j] = 1
        corr_new = np.correlate(h_new.tolist(), [1 - x for x in h_new.tolist()], mode='full')
        min_k_new = np.argmin(corr_new)
        if min_k_new > min_k:
            h_best = h_new
            swaps_done += 1
        else:
            swaps_done += 1

    # Final evaluation
    h_final = h_best
    corr_full = np.correlate(h_final.tolist(), [1 - x for x in h_final.tolist()], mode='full')
    final_c5 = np.max(corr_full) * dx
    return h_final.tolist(), final_c5, n
```
