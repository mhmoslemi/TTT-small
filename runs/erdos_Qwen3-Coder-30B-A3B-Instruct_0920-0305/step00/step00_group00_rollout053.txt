import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize
import time

def compute_gradient(h, w, dx):
    # w is a vector of length 2*n-1; corresponds to shifts from -(n-1) to n-1
    n = len(h)
    grad = np.zeros_like(h)
    offset = n - 1  # w[0] corresponds to shift -(n-1)
    
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        shift = idx - offset
        
        if shift >= 0:
            # First term: sum_{j: 0 <= j <= n-shift-1} wk * (1 - h[j+shift])
            grad[:n-shift] += wk * (1 - h[shift:])
            # Second term: sum_{j: shift <= j <= n-1} wk * (-h[j-shift])
            grad[shift:] -= wk * h[:n-shift]
        else:
            # shift < 0, so shift = -|shift|
            s = -shift
            # First term: sum_{j: s <= j <= n-1} wk * (1 - h[j+s])
            grad[s:] += wk * (1 - h[:n-s])
            # Second term: sum_{j: 0 <= j <= n-s-1} wk * (-h[j+s])
            grad[:n-s] -= wk * h[s:]
    return grad * dx

def project_to_constraints(v, lower=0.0, upper=1.0, target_sum=None):
    if target_sum is None:
        target_sum = len(v) / 2.0
    # binary search for lambda
    lo = -np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, lower, upper)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo+hi)/2.0, lower, upper)
    return x

def loss_and_grad(h, beta, dx):
    # Compute correlation vector
    p = 1 - h
    c = np.correlate(h, p, mode='full') * dx
    
    # Compute log-sum-exp of beta * c
    scaled = beta * c
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w_sum = np.sum(w)
    w = w / w_sum
    
    # Compute loss (already stabilized)
    loss = (max_scaled + np.log(w_sum)) / beta
    
    # Compute gradient
    grad = compute_gradient(h, w, dx)
    
    return loss, grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use 140 points to balance between resolution and speed
    n = 140
    dx = 2.0 / n
    target_sum = n / 2.0
    
    # Import here
    try:
        from evaluate_erdos_solution import evaluate_erdos_solution
        has_evaluate = True
    except ImportError:

        # Fallback to manual method if needed
        
    # Initialize best values
    best_h = None
    best_c5 = np.inf
    
    # Number of restarts
    max_restarts = 20
    
    for restart in range(max_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Initialize with random values in [0,1]
        h = np.random.random(n)
        h = project_to_constraints(h, target_sum=target_sum)
        
        # Beta schedule
        beta_schedule = [1.0, 5.0, 15.0, 30.0]
        
        for beta in beta_schedule:
            # Create closure for function and gradient
            def fun_and_grad(x):
                return loss_and_grad(x, beta, dx)
            
            # Optimize using SLSQP
            bounds = [(0.0, 1.0)] * n
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
            
            res = minimize(fun_and_grad, h, method='SLSQP', jac=True, bounds=bounds, 
                           constraints=cons, options={'maxiter': 200, 'ftol': 1e-9, 'disp': False})
            
            if res.success:
                h = res.x
            else:
                # If optimization fails, use the best we have
                h = project_to_constraints(h, target_sum=target_sum)
                
            # Evaluate actual C5
            c = np.correlate(h, 1 - h, mode='full') * dx
            c5 = c.max()
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Early stopping if we're close to the record
            if best_c5 < 0.379:
                break
                
        # Break if we've converged
        if best_c5 < 0.379:
            break
    
    # Final check if needed
    if best_h is not None:
        c = np.correlate(best_h, 1 - best_h, mode='full') * dx
        actual_c5 = c.max()
        if actual_c5 < best_c5:
            best_c5 = actual_c5
    
    if best_h is None:
        # Fallback to random initialization with projection
        h = np.random.random(n)
        h = project_to_constraints(h, target_sum=target_sum)
        c = np.correlate(h, 1 - h, mode='full') * dx
        best_c5 = c.max()
        best_h = h
        
    return (best_h.tolist(), float(best_c5), n)
