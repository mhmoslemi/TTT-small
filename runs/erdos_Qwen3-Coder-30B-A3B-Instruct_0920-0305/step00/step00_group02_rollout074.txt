def decode(z, M):
    # Softmax to get positive lengths and gaps
    z_ell = z[:M]
    z_g = z[M:]
    ell = np.exp(z_ell)
    ell = ell / np.sum(ell)
    g = np.exp(z_g)
    g = g / np.sum(g)
    return ell, g

def build_intervals(ell, g):
    # Build intervals [a_i, a_i + ell_i]
    a = np.zeros(len(ell))
    a[0] = g[0]
    for i in range(1, len(ell)):
        a[i] = a[i-1] + ell[i-1] + g[i]
    b = a + ell
    return a, b

def candidate_shifts(a, b):
    # All possible shift values where overlaps can change
    E = list(a) + list(b)
    T = {0.0}
    for i in range(len(E)):
        for j in range(len(E)):
            if abs(E[i] - E[j]) <= 2.0:
                T.add(abs(E[i] - E[j]))
    return sorted(list(T))

def intersection_length(t, a, b):
    # Compute the intersection length of S and S+t
    if t <= 0:
        return 0.0
    
    total = 0.0
    for i in range(len(a)):
        # Compute overlap of [a[i], b[i]] with [a[i] - t, b[i] - t]
        # Since t>0, we have [a[i]-t, b[i]-t] = [a[i]-t, a[i]-t+ell[i]]
        start = max(a[i] - t, a[i])
        end = min(b[i] - t, b[i])
        if end > start:
            total += end - start
    return total

def objective(z):
    ell, g = decode(z, M)
    a, b = build_intervals(ell, g)
    T = candidate_shifts(a, b)
    min_I = float('inf')
    for t in T:
        I = intersection_length(t, a, b)
        if I < min_I:
            min_I = I
    return 1.0 - min_I

    )

def gridify(intervals, n, h_values=None):
    # Use the interval data passed in
    a, b = intervals
    h_grid = np.zeros(n)
    dx = 2.0 / n
    
    if h_values is not None:
        h = h_values
        for i, (a_i, b_i) in enumerate(zip(a, b)):
            start_idx = max(0, int(np.floor(a_i / dx)))
            end_idx = min(n, int(np.ceil(b_i / dx)))
            h_grid[start_idx:end_idx] = h[i]
    else:
        for i, (a_i, b_i) in enumerate(zip(a, b)):
            start_idx = max(0, int(np.floor(a_i / dx)))
            end_idx = min(n, int(np.ceil(b_i / dx)))
            h_grid[start_idx:end_idx] = 1.0
            
    # Ensure sum is exactly n/2 (avoid rounding errors)
    current_sum = np.sum(h_grid)
    delta = current_sum - n / 2
    if abs(delta) > 1e-10:
        # Adjust:
        if delta > 0:
            # Remove one 1
            indices_ones = np.where(h_grid == 1.0)[0]
            if len(indices_ones) > 0:
                idx = indices_ones[0]
                h_grid[idx] = 0.0
        else:
            # Add one 1
            indices_zeros = np.where(h_grid == 0.0)[0]
            if len(indices_zeros) > 0:
                idx = indices_zeros[0]
                h_grid[idx] = 1.0
                
    # Final check
    current_sum = np.sum(h_grid)
    if abs(current_sum - n / 2) > 1e-10:
        # Fix with smallest impact
        if delta > 0:
            diff = np.where(h_grid == 1.0)[0]
            i = diff[0]
            h_grid[i] = 0.0
        else:
            # Add one
            diff = np.where(h_grid == 0.0)[0]
            i = diff[0]
            h_grid[i] = 1.0
            
    return h_grid

def binary_hill_climb(h, time_left):
