import numpy as np
import time

# Predefined constants and functions
def evaluate_erdos_solution(h, dx=2.0/len(h)):
    # Compute overlaps using direct convolution
    c = np.correlate(h, 1-h, mode="full")
    return np.max(c) * dx

def initial_h_values(n=45):
    # Use the initial construction: 1s in first half regions, 0s in second half
    h = np.zeros(n)
    h[:n//2] = 1
    return h

def project_onto_feasible(h, target_sum):
    """
    Project h onto the set {h : 0 <= h_i <= 1, sum(h) = target_sum}
    Clip to [0,1] and rescale.
    """
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple water-filling
    if current_sum > target_sum:
        # Need to decrease values
        diff = current_sum - target_sum
        indices = np.argsort(h)[::-1]  # descending order
        for i in indices:
            if diff <= 0: break
            take = min(diff, h[i])
            h[i] -= take
            diff -= take
    else:
        # Need to increase values
        diff = target_sum - current_sum
        indices = np.argsort(h)  # ascending order
        for i in indices:
            if diff <= 0: break
            take = min(diff, 1 - h[i])
            h[i] += take
            diff -= take
    return h

def logsumexp_neg_weights(S_k_values, alpha):
    """
    Log-sum-exp of -alpha * S_k values
    """
    exp_terms = np.exp(-alpha * np.array(S_k_values))
    return -np.log(np.sum(exp_terms)) / alpha

def compute_overlap_S_k(h, dx):
    n = len(h)
    # The overlap for shift k is sum(h_i * h_{i+k})
    c = np.correlate(h, h, mode='full')
    S_k = np.array([c[n-1+k] * dx for k in range(n)])
    return S_k

def compute_gradient(h, S_k_vals, alpha):
    """
    Compute ∇ Φ_α = Σ_k w_k ∇S_k / Σ_k w_k
    Gradient: ∇S_k[i] = h[i-k] + h[i+k] (with padding)
    """
    n = len(h)
    w_vals = np.exp(-alpha * np.array(S_k_vals))
    W = np.sum(w_vals)
    if W == 0:
        # Avoid division by zero
        return np.zeros_like(h)
    grad = np.zeros_like(h)

    # Efficient O(n log n) convolution
    h_rev = h[::-1]
    for k in range(n):
        if k == 0:
            # S_k = sum h_i h_{i+k}
            batch_grad = np.zeros_like(h)
            # For each i compute h_{i-k} + h_{i+k} if exists
            # We compute this using two shifted versions of h
            if n >= 2:
                # h' = h[-k:] + h[:-k] (k>0) but since only one side is known,
                # we do it in two parts
                # graduate = current + left-via-shift
                pass
            # Or to expand: vectors like:
            # For a vector h = [h[0], h[1], ..., h[n-1]]
            #   h[:,k] = h[k:] + h[:n-k]
            # Use existence and offset indexing explicitly.
            # We only directly compute deltas vs providing a loop.
            # Direct implementation for python loops
            pass
        else:
            # Alternative, using partial convolutions via np.roll?
            pass

    # Use FFT approach for arriving at ∇S_k more efficiently.
    # Instead of writing full formula for gradients per shift k,
    # warp in convolution
    # ∂S_k/∂h_i = h_{i+shift} + h_{i-shift}
    # So loop over k and do two convolutions
    # ergo use numpy correlate with specific shifts
    expected_grad = np.zeros_like(h)
    # For each k:
    for k in range(n):
        # ∂S_k/∂h_i = h_{i+k} + h_{i-k}
        term1 = np.zeros_like(h)
        term2 = np.zeros_like(h)
        if k < n:
            term1[k:] = h[:-k]
        if k < n:
            term2[:-k] = h[k:]   # shifted j-k forward
        grad += w_vals[k] * (term1 + term2)
    return grad / W

def backtracking_line_search(h, grad, func_val, alpha, eta=1.0, c1=1e-4):
    """
    Backtracking line search for armijo condition:
    f(h + eta*grad) <= f(h) + c1*eta*grad @ grad
    If this is false we reduce eta.
    """
    # Because we're building Φ_α, and want to ensure Φ_α(h_new) >= Φ_α(h)
    # Actually for maximisation, we increase. So this is used upward.
    # Relax condition to just monotonically improving a bit
    while True:
        h_new = project_onto_feasible(h + eta * grad, len(h)/2)
        func_val_new = logsumexp_neg_weights(compute_overlap_S_k(h_new, 2.0/len(h)), alpha)
        if func_val_new >= func_val:
            return h_new, func_val_new, eta
        else:
            eta *= 0.5
            if eta < 1e-10:
                return h, func_val, eta

def beatty_construction(n):
    """Create Beatty sequence with β=(sqrt(5)-1)/2 in [0,1]^n."""
    beta = ( np.sqrt(5) - 1 ) / 2
    h = np.array([int(np.frac(i * beta) < 0.5) for i in range(n)])
    # Adjust if necessary to have sum = n/2
    ones_count = np.sum(h)
    target = n / 2
    if ones_count != target:
        # Need to flip a few 1s to 0s or vice versa to reach target
        diff = int(target - ones_count)
        if diff > 0:
            # Need more 1s, flip zeros to 1s at the end
            zeros = np.where(h == 0)[0][:diff]
            h[zeros] = 1
        else:
            # Need fewer 1s, flip 1s to 0s at the end
            ones = np.where(h == 1)[0][-diff:]
            h[ones] = 0
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    # Choose a reasonable n (smaller than 1000; prefer <800 for speed)
    n = 300  # For tracking n in output
    dx = 2.0 / n
    
    # Start timing
    start_time = time.time()

    # Step 1: Initialize from Beatty sequence
    h = beatty_construction(n)
    h = project_onto_feasible(h, n/2)

    # Step 2: Continuous relax and solve using projected gradient ascent
    alpha = 1.0
    max_alpha = 10000
    step_size = 1.0
    
    best_h = h.copy()
    best_objective = -logsumexp_neg_weights(compute_overlap_S_k(h, dx), alpha)
    
    # Gradient phase: about 360 sec or until time runs short
    grad_budget = 360.0
    iteration = 0
    while time.time() - start_time < grad_budget + 10:
        # Compute overlaps
        s_k_vals = compute_overlap_S_k(h, dx)
        # Compute objective via surrogate logsumexp
        surr = logsumexp_neg_weights(s_k_vals, alpha)
        # Compute gradient
        try:
            grad = compute_gradient(h, s_k_vals, alpha)
        except Exception as e:
            print(f"Gradient computation failed: {e}")
            break

        # Take step
        h_new, new_surr, eta = backtracking_line_search(
            h, grad, surr, alpha, eta=step_size)
        
        h = h_new
        # If the surrogate wasn't really increasing, reduce step size
        if new_surr <= surr:
            step_size *= 0.5
        else:
            step_size = 1.0 
        
        # Update the best
        if new_surr > best_objective:
            best_objective = new_surr
            best_h = h.copy()

        # Increase alpha
        if alpha < max_alpha:
            if iteration % 200 == 0:
                alpha *= 2.0
                alpha = min(alpha, max_alpha)

        iteration += 1
        
        if time.time() - start_time > grad_budget:
            break

    h_solution = best_h.copy()

    # Step 3: Discrete polishing (ribinizing + hill-climbing)
    # Threshold at median to get binary
    median_thresh = np.median(h_solution)
    h_binary = (h_solution > median_thresh).astype(float)
    h_binary = project_onto_feasible(h_binary, n/2)

    # Now apply hill climbing on the binary version
    current_c5 = evaluate_erdos_solution(h_binary, dx)
    best_binary = h_binary.copy()
    best_c5 = current_c5

    # Project to ensure valid binary vector with exact sum
    if abs(np.sum(h_binary) - n/2) >= 1e-5:
        ones = np.where(h_binary == 1)[0]
        zeros = np.where(h_binary == 0)[0]
        needed_flips = int(np.sum(h_binary)) - int(n/2)
        if needed_flips > 0:
            for _ in range(needed_flips):
                if len(zeros) > 0:
                    idx = zeros[-1]
                    h_binary[idx] = 1
                    zeros = zeros[:-1]
                    ones = np.append(ones, idx)
        elif needed_flips < 0:
            for _ in range(-needed_flips):
                if len(ones) > 0:
                    idx = ones[-1]
                    h_binary[idx] = 0
                    ones = ones[:-1]
                    zeros = np.append(zeros, idx)
        h_binary = project_onto_feasible(h_binary, n/2)
        
    # Post-processing: local search
    # Hill-climbing
    search_budget = 180.0
    time_left = lambda: max(0.0, search_budget - (time.time() - start_time - grad_budget))
    
    tabu_list = set()
    tabu_size = 100
    while time.time() - start_time < budget_s - 10:
        if time_left() <= 0:
            break
        # Select a swap between 1 and 0
        ones = np.where(h_binary == 1)[0]
        zeros = np.where(h_binary == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = np.random.choice(ones)
        b = np.random.choice(zeros)
        # Make a swap in h_binary
        h_test = h_binary.copy()
        h_test[a] = 0
        h_test[b] = 1
        # Check new C5
        c5 = evaluate_erdos_solution(h_test, dx)
        # If this is better, accept it
        if c5 < current_c5:
            h_binary = h_test
            current_c5 = c5
            best_c5 = min(best_c5, c5)
        else:
            # Accept with low probability if better than fail
            # Butfor keep things structured, always accept strictly improving
            pass

    final_c5 = evaluate_erdos_solution(h_binary, dx)
    return (h_binary.tolist(), float(final_c5), n)
