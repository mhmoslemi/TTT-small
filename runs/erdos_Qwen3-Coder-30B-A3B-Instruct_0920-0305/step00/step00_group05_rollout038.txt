import numpy as np
import cvxpy as cp
import time
import math

def full_overlap(x):
    # Compute full correlation of x with 1-x
    return np.correlate(x, 1 - x, mode='full')

def worst_shift(x):
    # Returns (k, max_overlap) for worst shift
    overlaps = full_overlap(x)
    k = np.argmax(overlaps)
    return (k, overlaps[k])

def build_mip(N, active_shifts):
    # Build MIP problem for given N and active shifts
    # x[i] are binary variables
    x = cp.Variable(N, boolean=True)
    t = cp.Variable(integer=True)
    
    # sym break
    constraints = [x[0] == 1]
    constraints.append(cp.sum(x) == N // 2)
    # w[i,k] = x[i] * x[i+k]
    w_vars = {}
    overlap_constraints = []
    
    for k in active_shifts:
        w = cp.Variable(N - k, boolean=True)
        w_vars[(k,)] = w
        for i in range(N - k):
            constraints.append(w[i] <= x[i])
            constraints.append(w[i] <= x[i + k])
            constraints.append(w[i] >= x[i] + x[i + k] - 1)
        # overlap for shift k
        overlap_expr = cp.sum(x[:N - k]) - cp.sum(w)
        overlap_constraints.append(overlap_expr <= t)
    
    constraints.extend(overlap_constraints)
    # objective
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    return prob, {'x': x, 't': t, 'w': w_vars}

def solve_mip(prob, vars_dict, time_limit=30):
    # Solve the MIP problem, extracting x and t values
    x = vars_dict['x']
    t = vars_dict['t']
    
    try:
        prob.solve(solver=cp.CBC, max_iters=10000, TimeLimit=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_vals = [int(round(val)) for val in x.value]
            t_val = int(round(t.value))
            return x_vals, t_val
        else:
            # Fallback to checking if we can still extract a reasonable solution
            x_vals = [int(round(val)) if val is not None else 0 for val in x.value]
            t_val = int(round(t.value)) if t.value is not None else 0
            return x_vals, t_val
    except Exception:
        x_vals = [int(round(val)) if val is not None else 0 for val in x.value]
        t_val = int(round(t.value)) if t.value is not None else 0
        return x_vals, t_val

            if len(indices) > 0:

                h[flip_indices] = 1
        else:

    remaining = budget_s - (time.time() - start_time)

            break  # Something went wrong

    # Evaluate final performance
    dx = 2.0 / N
    final_c5 = best_t * dx
    
    return (best_x.tolist(), final_c5, N)
