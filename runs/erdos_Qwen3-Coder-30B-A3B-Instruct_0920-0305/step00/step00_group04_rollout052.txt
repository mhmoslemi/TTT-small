import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.ndimage import convolve
import time
import math

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for a given step function h"""
    # Compute the full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode="full")
    # Maximum value over all shifts
    max_val = np.max(conv) * dx
    return max_val

def initial_h_values(n=42):
    """Return initial step function values"""
    # Use a simple construction
    h = np.ones(n)
    # Normalize to have integral = 1
    h = h * 2.0 / n
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction as starting point
    n_points = 42
    h0 = initial_h_values(n_points)
    dx = 2.0 / n_points
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_vec):
        # Convert to numpy array
        h = np.array(h_vec)
        return evaluate_overlap(h, dx)
    
    def constraint_eq(h_vec):
        # Integral constraint: sum(h) * dx = 1
        h = np.array(h_vec)
        return np.sum(h) * dx - 1.0
    
    # Optimization using differential evolution
    constraint = {'type': 'eq', 'fun': constraint_eq}
    
    # Run optimization with time limiting
    result = differential_evolution(
        objective, 
        bounds, 
        args=(),
        constraints=constraint,
        seed=seed,
        maxiter=int(500000 / n_points),
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        atol=1e-6,
        tol=1e-6
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Refine with local optimization if time allows
    elapsed = time.time() - start_time
    if elapsed < budget_s - 1:
        # Try local refinement using L-BFGS-B
        try:
            result_refine = minimize(
                objective,
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraint,
                options={'maxiter': 1000, 'ftol': 1e-9}
            )
            if result_refine.success and result_refine.fun < best_c5:
                best_h = result_refine.x
                best_c5 = result_refine.fun
        except:
            pass
    
    # Ensure constraints are satisfied
    best_h = np.clip(best_h, 0, 1)
    integral_check = np.sum(best_h) * dx
    if abs(integral_check - 1.0) > 1e-5:
        # Scale to satisfy constraint
        best_h = best_h * (1.0 / integral_check)
    
    # Return best solution
    return (best_h.tolist(), float(best_c5), n_points)
