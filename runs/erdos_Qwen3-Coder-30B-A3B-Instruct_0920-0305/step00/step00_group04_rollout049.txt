import cvxpy as cp
import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n = 60
    dx = 2.0 / n

    b = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Constraints that fix the sum
    constraints = [cp.sum(b) == n // 2]
    
    # For all non-zero lags, we need to compute overlaps
    # The overlap at lag ℓ is sum of (b[i] * (1 - b[i+ℓ])) for all valid i
    # This can be rewritten as sum(b[i]) - sum(b[i] * b[i+ℓ])
    # Introduce auxiliary variables w[i,ℓ] = b[i] * b[i+ℓ] (AND)
    
    # Since lag ℓ and -ℓ are symmetric for the max, we only need ℓ in {1,2,...,n-1}
    max_overlaps = []
    
    for lag in range(1, n):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # We constrain w[i, ℓ] = b[i] AND b[i+ℓ]
        # We only consider valid indices; ignore the gap at the boundaries.
        # For lag ℓ, valid indices are 0 to n - ℓ - 1
        n_valid = n - lag
        w = cp.Variable(n_valid, boolean=True)
        
        # Constraints for w[i, ℓ] = b[i] AND b[i+ℓ]
        for i in range(n_valid):
            constraints.append(w[i] <= b[i])
            constraints.append(w[i] <= b[i + lag])
            constraints.append(w[i] >= b[i] + b[i + lag] - 1)

        # because we compute sum of b[i] * (1 - b[i+lag]) = sum(b[i]) - sum(b[i] * b[i+lag])
        N_ell = cp.sum(b[:n_valid]) - cp.sum(w)
        max_overlaps.append(N_ell)
        constraints.append(t >= N_ell)

    objective = cp.Minimize(t)
    
    # Problem formulation
    prob = cp.Problem(objective, constraints)

    try:
