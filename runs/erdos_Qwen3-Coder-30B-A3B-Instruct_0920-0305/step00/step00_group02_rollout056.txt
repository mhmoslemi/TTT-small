import cvxpy as cp
import random
import time
from scipy import sparse

    random.seed(seed)

    start_time = time.time()

    # Parameters

    r = 8   # rank for low-rank factorization

    if time.time() - start_time > budget_s - 10:
        # Try a small n
        n = 100
        r = 5

    # Time allocated to SDP solve (in seconds)
    sdp_time = max(100, budget_s / 3)
    
    # Build shift matrices A_k
    def build_shift_matrix(k, n):
        """Build sparse shift matrix A_k for shift k."""
        rows = []
        cols = []
        data = []
        for i in range(n - k):
            rows.append(i)
            cols.append(i + k)
            data.append(1)
            rows.append(i + k)
            cols.append(i)
            data.append(1)
        return sparse.csr_matrix((data, (rows, cols)), shape=(n, n))

    A_matrices = []
    k_list = list(range(1, (n // 2) + 1))  # Only k=1 to n//2 due to symmetry
    for k in k_list:
        A_matrices.append(build_shift_matrix(k, n))

    # SDP Model
    X = cp.Variable((n, n), PSD=True)
    x = cp.Variable(n)
    t = cp.Variable()

    # Diagonal constraints
    constraints = [cp.diag(X) == x]

    # Constraints for x to be in [0,1]
    constraints += [x >= 0, x <= 1]

    # Sum constraint
    constraints += [cp.sum(x) == n / 2]

    # Overlap constraints
    for A in A_matrices:
        constraints.append(cp.trace(A @ X) >= 2 * t)

    # Objective
    objective = cp.Maximize(t)

    # Solve the SDP
    prob = cp.Problem(objective, constraints)

    try:
        # Allow SCS to use at most sdp_time seconds
        prob.solve(solver=cp.SCS, max_iters=2000, time_limit=sdp_time)

    if prob.status != cp.OPTIMAL:

    # Compute low-rank decomposition for rounding
    U, s, Vt = np.linalg.svd(X_opt, full_matrices=False)
    if r < len(s):

    U = U[:, :r]

    min_c5 = float('inf')

            break
        g = np.random.randn(r)
        scores = U @ g  # Shape (n,)
        # Sort to find p-th percentile to get exactly n/2 ones
        sorted_scores = np.sort(scores)[::-1]
        threshold = sorted_scores[n // 2 - 1]
        h_trial = (scores >= threshold).astype(int)
        if np.sum(h_trial) != n // 2:
            # Adjust if necessary
            diff = np.sum(h_trial) - n // 2
            if diff > 0:  # Too many 1s
                zero_indices = np.where(h_trial == 0)[0]
