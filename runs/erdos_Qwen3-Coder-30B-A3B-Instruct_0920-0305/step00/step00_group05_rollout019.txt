import numpy as np
import cvxpy as cp
import time
from math import floor, ceil

def full_overlap(x):
    """Returns the array of overlaps for all shifts."""
    return np.correlate(x, 1 - np.array(x), mode="full")

def worst_shift(x):
    """Returns (k, max_overlap) where overlap is for shift k."""
    overlaps = full_overlap(x)
    k_max = np.argmax(overlaps)
    max_overlap = overlaps[k_max]
    return (k_max, max_overlap)

def build_mip(N, active_shifts):
    """Build a MIP model for given active shifts."""
    # Variables: x[i] in {0,1}, t integer, w[i,k] binary
    x = cp.Variable(N, integer=True)
    t = cp.Variable(integer=True)
    
    # Dictionary to store w variables for indexing later
    w = {}
    constraints = []
    
    # Constraints: x in [0,1], sum(x) = N/2, x0 = 1
    constraints.append(x >= 0)
    constraints.append(x <= 1)
    constraints.append(cp.sum(x) == N // 2)
    constraints.append(x[0] == 1)
    
    # For active shifts, add auxiliary AND variables and constraints
    for k in active_shifts:
        w[k] = cp.Variable(N - k, integer=True)
        for i in range(N - k):
            # Link w[i,k] with x[i], x[i+k]
            constraints.append(w[k][i] <= x[i])
            constraints.append(w[k][i] <= x[i + k])
            constraints.append(w[k][i] >= x[i] + x[i + k] - 1)
        
        # Constraint: sum(x[i] - w[i,k]) <= t
        ones_sum = cp.sum(x[:N-k])
        and_sum = cp.sum(w[k])
        constraints.append(ones_sum - and_sum <= t)
    
    # Objective: minimise t
    obj = cp.Minimize(t)
    
    # Build the problem
    prob = cp.Problem(obj, constraints)
    return prob, (x, t, w)

def solve_mip(prob, vars_dict, time_limit=30):
    """Solve the MIP with a given time limit."""
    x, t, w = vars_dict
    try:
        prob.solve(solver=cp.CBC, time_limit=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_vals = [int(round(val)) for val in x.value]
            t_val = int(round(t.value))
            return x_vals, t_val
        else:
            # Try to get a feasible solution if not optimal
            x_vals = [int(round(val)) for val in x.value]
            t_val = None
            return x_vals, t_val
    except Exception as e:
        # Return a feasible solution from x.value if problem
        x_vals = [int(round(val)) if val is not None else 0 for val in x.value]
        t_val = None
        return x_vals, t_val

def local_search(x_start, N):
    x = np.array(x_start)
    best_x = x.copy()
    best_overlap = np.max(full_overlap(x))
    best_t = floor(best_overlap / (2.0 / N))
    
    for trial in range(10000):
        i = np.random.randint(0, N)
        j = np.random.randint(0, N)
        if i == j:
            continue
        x_copy = x.copy()
        x_copy[i] = 1 - x_copy[i]
        x_copy[j] = 1 - x_copy[j]
        if sum(x_copy) != N // 2:
            continue  # Skip invalid solutions
        # Fix symmetry
        x_copy[0] = 1
        x_copy = x_copy.astype(int)
        
        # Evaluate the new overlap
        overlaps = full_overlap(x_copy)
        max_overlap = np.max(overlaps)
        t = floor(max_overlap / (2.0 / N))
        
        if t < best_t:
            best_t = t
            best_x = x_copy.copy()
    
    return best_x
