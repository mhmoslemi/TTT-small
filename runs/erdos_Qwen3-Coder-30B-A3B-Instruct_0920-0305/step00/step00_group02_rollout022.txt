import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600):
    np.random.seed(seed)

    # Parameters
    n = 300  # Number of samples; must be even for exact half-sum
    dx = 2.0 / n
    target = 0.3807
    max_alpha = 10000
    initial_alpha = 1.0
    alpha_growth = 2.0
    max_iterations = 4000
    gradient_time_limit = 0.6 * budget_s
    local_search_time_limit = 0.3 * budget_s

    # Initial Beatty sequence
    beta = (np.sqrt(5) - 1) / 2
    h_initial = np.array([1 if (i * beta) % 1 < 0.5 else 0 for i in range(n)])
    current_sum = np.sum(h_initial)
    # Adjust to exact half-sum by flipping extras
    needed = n // 2 - current_sum
    if needed > 0:
        # Find zeros and flip some to 1
        zeros = np.where(h_initial == 0)[0]
        for i in range(needed):
            h_initial[zeros[i]] = 1
    elif needed < 0:
        # Find ones and flip some to 0
        ones = np.where(h_initial == 1)[0]
        for i in range(-needed):
            h_initial[ones[i]] = 0

    # Project into feasible set (ensure sum is exactly n/2 and bounds are enforced)
    h = np.clip(h_initial, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n/2) > 1e-8:
        # Use water filling fix
        diff = n/2 - current_sum
        h = np.clip(h + diff / n, 0, 1)

    # Initialize key variables
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    alpha = initial_alpha
    last_improvement = 0

    # Timing
    start_time = time.time()
    elapsed = 0

    while elapsed < gradient_time_limit and last_improvement < max_iterations:
        # Compute overlaps S_k via correlation
        # We will compute s_k = sum h_i * h_{i+k}
        # Let g = h * h (cross correlation) so that g[k] = sum_{i} h[i] * h[i+k]
        # But the kernel must be flipped: we compute correlate(h, reverse(h)) this gives shifted cross correlation as needed
        # Actually h * h would be auto correlation, but as needed for shift_right, use correlation(h,h)
        # For our specific case that's compute h_i * h_{i+k}, we do: s_k = corr(h, shift_k(h)) = corr(h, shift_k(h))
        # Let's use different API: use np.correlate(h, h, mode='full') to get all possible overlaps including boundary

        # But what we want is variable length correlations, with leading zeros or boundary padding, i.e. the k-shifted alignment
        # Real type correlations with zero padding at boundaries.
        # So better to compute cross-correlation with h and reverse(h): we'll get h_i * h_{i+k} as cross-correlation, for k = 0..n-1
        # But for the windowed version, e.g., shift 0: h[0]*h[0]… to h[n-1]*h[n-1],
        # Now if wanted two ways of computing same but more efficiently: FFT convolution

        # In our setup:
        # shift_k: h[i] is paired with h[i+k] (not h[i-k])
        # so we want `cross-correlation(h,h)` or `corr(h, reverse(h))` by standard definition
        # shifted left by k means key to index k
        # corr(h, h) gives same result?
        # Ah, I see: h[i]h[i+k] so we want 'full' mode and select appropriate sub
        # Scalable for big data: use numpy's native FFT-based correlate which matters
        Corr = np.correlate(h, h, mode='full')
        # Offset = n - 1 (because full convolution).
        # So: Corr[n-1+k] = sum_{i=0}^{n-1-k} h[i] h[i+k] = S_k
        S = Corr[n-1:n-1+n]  # corresponds to s_0...s_{n-1}
        # Compute weights: w_k = exp(-alpha * S_k)
        # Need to avoid overflow in exponent: wrap before applying exp
        # Since the inputs are bounded, min(S) ≥ 0 and max(S) ≤ n, hence exponent within [-n α, 0]
        # Large negative ones of exp are ~0; we can set tolerance small to avoid numerical problems

        # Reset weights based on current alpha
        if alpha > max_alpha:
            weights = np.exp(-alpha * S)
        else:
            weights = np.exp(-alpha * S)

        # Normalize weights
        W = np.sum(weights)
        if W < 1e-15:
            W = 1e-15  # Avoid div zero

        weight_grad = np.zeros(n)
        for k in range(n):
            start_i = k
            end_i = n - k
            if k == 0:
                # Gradient for S_0 = sum_{i} h_i^2 → d/dh_i = 2 h_i
                weight_grad += weights[k] * 2 * h
            else:
                # For S_k = sum h[i] h[i+k] → d/dh_i = h[i+k]
                # d/dh_{i+k} = h[i]  (same row)
                # Use slightly manual loop here:
                # pad S_k and other related values to match relevance:
                # If we had corr(h, h_rev), then get corresponding gradient:
                #      For k = 0: all ones;
                #      For k>0, then dS_k/dh_i = h[i+k] when i < n-k; dS_k/dh_{i} = h[i-k] when i>=k
                # fill in upper diagonal for d/dh_i of S_k:
                # U(k) = array of ones (1 if h_i increase input of S_k due to increase h_i) shape n,
                # It’s 1 only at positions i where h_i appears in the corresponding sum S_k = sum h[i] h[i+k],
                # So which i satisfy i+k < n, i.e., i < n - k. At i = {0, ..., n-k-1} => gradient += weights[k] * U(k) = weights[k] * (1 everywhere here)
                # V(k) = h trimmed at right (i.e., h[i-k]) => take h[...] in range with index match for right
                # The dS_k/dh_i = h[i+k] if i < n-k
                # So construct a weighting vector V_k such that V_k[i] = h[i] when i+k is valid
                # Let w = a; w_i = h_{i+k} if i+k < n => shift s_i = h_{i+k}
                # So what is derivative of sum_{i} h_i s_i? => s_i for h_i, s_i for h_i
                # That is, we want the part where k is included, e.g.
                # For example, d(S_k) = sum_j h_j h_{j+k} gives gradient with respect to h_l = {h_{l-k} if l ≥ k; h_{l+k} if l+k < n }
                # Why? It's the Jacobian of dot product. Exactly two contributions:
                # i=l:          contributes h_{l+k}
                # i=l+k:        contributes h_l
                # In total: matrix S_k = h h^T (except partial)
                # Split out k=0 separately
                if k >= n:
                    continue
                # Apply convolution trick using even more-improved human solution:
                # dS_k/h_l = h_{l+k} + h_{l-k}
                # Using shifted indices:
                # v_l = h_{l+k}, u_l = h_{l-k}
                # 2convolution path: both shifts applied in either direction:
                # constructs a projected result dS_k/dh via efficient single backward/backward
                # read h from start=0 to n-k, stop at n-k (right extend):  h[...n-k] → h_order (len = n-k)
                # size = n
                # If i = k..n-1, then h_i appears in S_k under its position
                # Also in addition defined the i = 0 to k-1, then h_{i-k} = 0. So unless shift k==0:
                # Actually, simpler and safe if just put zero padding
                # Define:
                # V_k = shift(k) of h (only place where some unit is added)
                # U_k = reverse(shift(n-k) of h)
                # Then they give two forms so full can be simplied or blocked;-1
                # But simpler is separate code:
                # Consider h_k[0] -> h_k[n-k-1], i'll be from 0 to n-k-1
                # Now accum from full-h vector = h[0] when calculating S_k (when i range 0 to n-k-1)
                # Let vector p = h[n-k-1:n-1] = h_k_shifted_right
                # And u = v = h_flip[end-k:end]
                # Take: h[0:n-k] - is the overlapping part with h_on_right_side
                # Start better approach once many have overlapping parts.
                # Simple struct logic with two as thresholds:
                # Feed via shift, add back to weight_grad:
                # Let's just loop over finite subprocess:
                # i counts positions where h_i appears in sum
                # There are two derivatives possible:
                # 1. dd/h_i = h_{i+k} [when i+k<n]
                # 2. dd/h_i = h_{i-k} [when i-k >= 0]
                # To compute chained: elementary structure on mod access

                # Long form:
                temp_grad = np.zeros(n)
                for i in range(n - k):
                    temp_grad[i] += weights[k] * h[i + k]
                for i in range(k, n):
                    temp_grad[i] += weights[k] * h[i - k]
                weight_grad += temp_grad

        # Full gradient
        grad = weight_grad / W

        # Perform backtracking line search to update h:
        # Start with step size η = 1 and reduce it until an ascent is observed
        eta = 1.0
        min_step = 1e-10  # minimum learning rate
        tolerance = 1e-8

        while eta >= min_step:
            # Apply update: h_new = h + eta * grad
            proposed_h = h + eta * grad

            # Project onto feasible set: [0, 1] and sum=1
            proposed_h = np.clip(proposed_h, 0, 1)
            s = np.sum(proposed_h)
            diff = n/2 - s
            if abs(diff) > 1e-9:
                proposed_h = np.clip(proposed_h + diff / n, 0, 1)

            # Compute the new Phi_alpha:
            Corr_new = np.correlate(proposed_h, proposed_h, mode='full')
            S_new = Corr_new[n-1:n-1+n]

            # Test computation
            if alpha > max_alpha:
                new_weights = np.exp(-alpha * S_new)
            else:
                new_weights = np.exp(-alpha * S_new)
            W_new = np.sum(new_weights)
            if W_new < 1e-15:
                W_new = 1e-15
            new_grad = np.zeros_like(grad)
            for k in range(n):
                if k == 0:
                    new_grad += new_weights[k] * 2 * proposed_h
                else:
                    temp_grad = np.zeros(n)
                    for i in range(n - k):
                        temp_grad[i] += new_weights[k] * proposed_h[i + k]
                    for i in range(k, n):
                        temp_grad[i] += new_weights[k] * proposed_h[i - k]
                    new_grad += temp_grad

            new_phi = -1.0 / alpha * np.log(W_new) if W_new > 0 else -1e10
            current_phi = -1.0 / alpha * np.log(W) if W > 0 else -1e10

            if new_phi > current_phi + tolerance:
                h = proposed_h.copy()
                break  # Accept descent; stop decreasing step size
            else:
                eta *= 0.5  # Reduce step size

        # Get Phi_alpha estimate (approximation to actual min overlap + small corrections)
        # FULL VERIFICATION (use exact version)
        corr_val = np.correlate(h, h, mode='full')
        S_val = corr_val[n-1:n-1+n]
        Phi_val = -1.0 / alpha * np.log(np.sum(np.exp(-alpha * S_val)))

        # Update best solution
        c5 = 1.0 - np.min(S_val) * dx
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            last_improvement = 0
        else:
            last_improvement += 1

        # Adaptive increase of alpha
        if len(S_val) > 0 and last_improvement % 100 == 0:
            if alpha < max_alpha:
                alpha *= 1.1  # Gradual growth in α

        # Update time
        elapsed = time.time() - start_time
        if elapsed >= gradient_time_limit:
            break

    # Local pairwise swapping based hill climbing on discrete version
    max_trials = 10 * n
    converge = 0
    trial_counter = 0

    # Binary rounding based on median threshold
    median_val = np.sort(best_h)[n//2]
    h_discrete = np.array([1 if x >= median_val else 0 for x in best_h])
    cur_sum = np.sum(h_discrete)
    if cur_sum != n/2:
        # adjust by flipping
        diff = n/2 - cur_sum
        if diff > 0:
            # Flip 0s to 1s
            flipped = 0
            for i in range(n):
                if flipped >= diff:
                    break
                if h_discrete[i] == 0:
                    h_discrete[i] = 1
                    flipped += 1
        else:
            # Flip 1s to 0s
            flipped = 0
            for i in range(n):
                if flipped >= -diff:
                    break
                if h_discrete[i] == 1:
                    h_discrete[i] = 0
                    flipped += 1

    c5_final = evaluate_c5(h_discrete, dx)
    h_best = h_discrete.copy()
    c5_best = c5_final

    # Run local search for up to remaining time
    start_climb = time.time()
    while time.time() - start_climb < local_search_time_limit:
        # Tentative swap
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h_best[i] == h_best[j]:
            continue  # No benefit if both same
        # Compute new h after swap
        h_swap = h_best.copy()
        h_swap[i], h_swap[j] = h_swap[j], h_swap[i]
        c5_swap = evaluate_c5(h_swap, dx)
        if c5_swap < c5_best:
            h_best = h_swap
            c5_best = c5_swap
        trial_counter += 1
        if trial_counter % 100 == 0:
            if c5_best < target:
                break  # Early exit if desired bound achieved

    return (h_best.tolist(), float(c5_best), n)

def evaluate_c5(h, dx):
    """Evaluate the max overlap for the given step function h"""
    # Get overlaps for all shifts
    correlation = np.correlate(h, [1 - x for x in h], mode="full")
    max_overlap = np.max(correlation)
    return max_overlap * dx
