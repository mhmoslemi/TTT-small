def project_to_feasible(v, target_sum):
    """Project v to the feasible set: 0 <= v[i] <= 1, sum(v) = target_sum"""
    # Simple projection: clip and renormalize
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    
    if abs(current_sum - target_sum) < 1e-10:
        return v
    
    # Add a small amount to avoid zero division
    if current_sum > target_sum:
        # We need to reduce
        diff = current_sum - target_sum
        # Find the largest elements to reduce
        indices = np.argsort(v)[::-1]
        for idx in indices:
            if diff <= 0:
                break
            reduce_amount = min(v[idx], diff)
            v[idx] -= reduce_amount
            diff -= reduce_amount
    else:
        # We need to increase
        diff = target_sum - current_sum
        indices = np.argsort(v)
        for idx in indices:
            if diff <= 0:
                break
            increase_amount = min(1 - v[idx], diff)
            v[idx] += increase_amount
            diff -= increase_amount
            
    return v

def evaluate_erdos_solution(h):
    """Evaluate the overlap integral for the given h"""
    # Ensure h is valid (within [0,1] and sum constraint satisfied)
    h = np.array(h)
    dx = 2.0 / len(h)
    
    # Compute correlation of h with (1 - h) using full mode
    # np.correlate computes convolution of h with reversed 1-h
    corr = np.correlate(h, 1 - h, mode="full")
    # This returns correlation for shifts from -(N-1) to (N-1)
    # and we want the maximum overlap value
    C5 = np.max(corr) * dx
    
    return C5
