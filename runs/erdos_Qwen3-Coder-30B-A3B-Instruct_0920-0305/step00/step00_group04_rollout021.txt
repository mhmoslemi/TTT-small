import numpy as np
import time
import random

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr) * dx
    return max_val

def compute_c5_fast(h):
    n = h.shape[0]
    size = 1 << ((2*n-1).bit_length())
    f_h = np.fft.fft(h, size)
    f_g = np.fft.fft(1 - h, size)
    corr = np.fft.ifft(f_h * np.conj(f_g)).real
    corr = corr[:2*n-1]
    dx = 2.0 / n
    max_val = np.max(corr) * dx
    return max_val

def random_binary_vector(n):
    h = np.zeros(n)
    ones_indices = np.random.choice(n, size=n//2, replace=False)
    h[ones_indices] = 1
    return h

        # Choose a random 1 and 0 to swap

        j = zeros_indices[j_idx]

        h[i], h[j] = h[j], h[i]
        
        # Update lists
        ones_indices[i_idx] = j
