def project_to_feasible(h, target_sum):

    high_lambda = 2.0
    epsilon = 1e-10
    
    def project_with_lambda(lam):
        proj = np.clip(h - lam, 0, 1)
        return np.sum(proj) - target_sum
    
    # Binary search for lambda such that sum equals target
    for _ in range(100):
        mid_lambda = (low_lambda + high_lambda) / 2
        proj_sum = project_with_lambda(mid_lambda)
        if proj_sum > 0:
            low_lambda = mid_lambda
        else:
            high_lambda = mid_lambda
    
    # Final projection with computed lambda
    lambda_star = (low_lambda + high_lambda) / 2
    h = np.clip(h - lambda_star, 0, 1)
    # Adjust to exactly match target sum
    h_sum = np.sum(h)
    diff = target_sum - h_sum
    if abs(diff) > 1e-10:
        # Find small index to adjust
        # just add diff to first index
        h[0] += diff
        # And clip to [0,1]
        h = np.clip(h, 0, 1)
        
    return h

def evaluate_erdos_solution(h):
    N = len(h)
    dx = 2.0 / N

    # Compute correlation with 1-h using full mode
    # correlate is same as convolution with 1-h
    corr = np.correlate(h, 1 - h, mode='full')
    # scale by dx for integration
    corr *= dx

    # Return the maximum across all shifts
    return np.max(corr)
