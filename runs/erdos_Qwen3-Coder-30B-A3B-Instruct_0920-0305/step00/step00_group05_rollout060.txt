import numpy as np
import time
import random

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    time_end = start + budget_s - 0.5
    
    n = 200
    best_h = None
    best_c5 = float('inf')
    dx = 2.0 / n

    max_restarts = 50
    max_iters_per_restart = 20000
    
    restart_count = 0
    total_iterations = 0

    while time.time() < time_end and restart_count < max_restarts:
        # generate new random binary with half ones
        h = np.zeros(n, dtype=np.float64)
        ones = np.random.choice(n, n//2, replace=False)
        h[ones] = 1.0

        cur_c5 = compute_c5(h)

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        # local annealing for this start
        T0 = 0.05
        T_end = 1e-6
        for it in range(max_iters_per_restart):
            if time.time() >= time_end:
                break
            total_iterations += 1

            # Choose random i with 1 and j with 0
            idx1 = np.random.randint(n)
            idx2 = np.random.randint(n)
            while h[idx1] == h[idx2]:
                idx1 = np.random.randint(n)
                idx2 = np.random.randint(n)
            
            # propose swap
            h[idx1], h[idx2] = h[idx2], h[idx1]
            
            new_c5 = compute_c5(h)
            d = new_c5 - cur_c5
            
            # Temperature decays linearly
            T = max(T_end, T0 * (1 - it / max_iters_per_restart))
            
            if d <= 0 or np.random.rand() < np.exp(-d / (T + 1e-12)):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                # revert swap
                h[idx1], h[idx2] = h[idx2], h[idx1]
        
        restart_count += 1

    return (best_h.tolist(), best_c5, n)
