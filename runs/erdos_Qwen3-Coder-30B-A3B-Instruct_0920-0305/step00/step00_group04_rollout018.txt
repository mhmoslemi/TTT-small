        b = cp.Variable(n, boolean=True)
        t = cp.Variable(1, integer=True)  # integer variable for max overlap
        
        # Symmetry breaking constraints to reduce search space
        constraints = [
            cp.sum(b) == n // 2,
            b[0] == 1,  # force b[0] to be 1 to break symmetry
        ]
        
        # Create auxiliary variables for each lag
        lag_indices = [i for i in range(1, n)]
        all_w = {}  # will store w_ell for each lag ell
        
        # Add lag constraints
        for ell in lag_indices:
            # w_ell[i] = b[i] AND b[i+ell]  
            # we create w_ell that is sliced by ell (to avoid indexing error)
            w_ell = cp.Variable(n - ell, boolean=True)
            all_w[ell] = w_ell
            
            # Constraint enforcement
            for i in range(n - ell):
                constraints.append(w_ell[i] <= b[i])
                constraints.append(w_ell[i] <= b[i + ell])
                constraints.append(w_ell[i] >= b[i] + b[i + ell] - 1)

            # But let's directly compute all min/max combinations.

        #      N_ell = sum b_i - sum (b_i AND b_{i+ell})

        # So final clause:
        # add constraints: t >= sum b_i - sum w_ell[i]

        # Actually we keep it symmetric:
        # For each ell from 1 to n-1 (excluding 0):  
        #   w_ell is vector of length n-ell.  
        #   w_ell[i] = b[i] AND b[i+ell]
        #   overlap = sum(b[:n-ell]) - sum(w_ell)
        #   constrain: t >= overlap   
        # We use ell in 1 to n-1 only; negatives are handled

        # Now add one for each ell in 1 to n-1

            N_ell = cp.sum(b[:n - ell]) - cp.sum(w_ell)
            constraints.append(t >= N_ell)
