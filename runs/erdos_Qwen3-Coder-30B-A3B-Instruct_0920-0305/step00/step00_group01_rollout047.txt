import numpy as np

    def compute_c5(h_vals):
        h = np.array(h_vals)
        # Compute all correlations (full convolution with 1-h)
        corr = np.correlate(h, 1 - h, mode='full')
        # Extract the relevant part for shifts k ∈ [0,2]
        # For each offset k from 0 to 2, we compute ∫ h(x)(1-h(x+k)) dx
        # The convolution indices correspond to shift k in [0,2] 
        # Let's extract just the positive lags corresponding to k ∈ [0,2]
        half_len = len(corr) // 2
        # corr indices correspond to shift -half_len <= k <= half_len  
        max_val = 0
        for k_idx in range(len(corr)):
            if k_idx <= half_len:  # corresponds to positive k shift
                k = k_idx * dx
            else:  # corresponds to negative k shift  
                k = (k_idx - len(corr)) * dx  # so k = k_idx - n_points for negative shifts
                if k < 0:
                    k = (len(corr) - k_idx) * dx  # correct sign

            if 0 <= k <= 2:
                local_correlation = corr[k_idx] * dx
                max_val = max(max_val, local_correlation)
        return max_val

        b_eq.append(target_sum)

        for d in range(n_points):
            for i in range(n_points - d):  # valid indices to align
                p_indices.append((i, d))

        # Constraints per pair p_{i,d}
        # But we don't want to build matrices too large or forward this using direct problem setup
        
        # Let's simplify:
        # Empirical use of simple subset LP to save time and code:
        # We will solve smaller LP by adapting the matrix form.

        # Use dual approach manually

    # Using more direct algorithm sampling:
    # Let's look elements for bin packing-like rounding / reconstruction
    def round_to_feasible(h_vals):
        h_new = list(h_vals)
        total_mass = sum(h_new) * dx
        target_mass = 1.0
        
        # Approximate rounding to binary {0,1} while matching sum
        # Strategy: sort, place highest ones at 1, until mass is met
        # This is ROUNDING between fea_solutions
        
        indices_sorted = sorted(range(len(h_new)), key=lambda i: h_new[i], reverse=True)
        h_out = [0.0] * len(h_new)
        current_sum = 0.0
        
        for i in indices_sorted:
            if current_sum + h_new[i] * dx <= target_mass:
                h_out[i] = 1.0
                current_sum += h_new[i] * dx
            elif current_sum < target_mass:
                # last fractional 1.0
                proportion = (target_mass - current_sum) / (h_new[i] * dx)
                h_out[i] = proportion
                current_sum = target_mass
            else:
                h_out[i] = 0.0
                
        return h_out

                        best_i = i
                        old_c5 = new_c5
                        
            if best_i != -1:
                h[best_i] = max(0.0, min(1.0, h[best_i] + best_delta))
        
        return h

    # Bisection with trusted checks based on baby steps and computing results
    def binary_search_lpd():
        nonlocal best_c5, best_h
        lower_bound = 0
        upper_bound = 0.4  # empirical from known lower judges

        while time.time() - start_time < budget_s - 1:
            mid_bound = (lower_bound + upper_bound) / 2.0
            
            # clearly use a substituted LP model to test feasibility (approximate)
            # repeatedly iterate within tight frame
            
            # Use a heuristic regular grid to test with already viable h_vals
            h_test = construct_h_basic()
            # Refine slightly toward minimum one, to start better goto
            h_local = refine_h(h_test)
            current_c5 = compute_c5(h_local)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_local.copy()
                print(f"[{time.time() - start_time:.1f}s] Improved c5 = {best_c5:.5f}")
                
            # Avoid very small ==, rather search for cut point around there
            if current_c5 <= mid_bound + 1e-5:
                upper_bound = mid_bound
            else:
                lower_bound = mid_bound
                
        return best_c5, best_h

    # Run Search
    final_c5, final_h = binary_search_lpd()

    return final_h, final_c5, n_points
