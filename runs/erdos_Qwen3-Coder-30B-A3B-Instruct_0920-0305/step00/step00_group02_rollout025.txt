def decode(z):
    """
    Decode raw parameter vector into interval lengths and gaps.
    Input: z = [l1, ..., lM, g0, ..., gM] where each is unbounded.
    Returns: lengths, gaps (both normalized to sum to 1)
    """
    M = (len(z) - 1) // 2
    z_ℓ = z[:M]
    z_g = z[M:]
    ℓ = np.exp(z_ℓ)
    g = np.exp(z_g)
    ℓ /= np.sum(ℓ)
    g /= np.sum(g)
    return ℓ, g

def build_intervals(ℓ, g):
    """Compute interval start points a_i from lengths and gaps."""
    M = len(ℓ)
    a = np.zeros(M)
    a[0] = g[0]
    for i in range(1, M):
        a[i] = a[i-1] + ℓ[i-1] + g[i]
    return a, a + ℓ

def candidate_shifts(a, b):
    """Generate all potential shift values that cause overlaps to change."""
    E = list(a) + list(b)
    T = {0.0}
    for i in range(len(E)):
        for j in range(len(E)):
            diff = abs(E[i] - E[j])
            if diff <= 2:
                T.add(diff)
    return sorted(list(T))

def intersection_length(t, a, b):
    """Compute the measure of intersection of S and S+t."""
    total = 0.0
    for i in range(len(a)):
        # Interval [a[i], b[i]]
        # Shifted interval [a[i]-t, b[i]-t]
        # Compute intersection
        left = max(a[i], a[i] - t)
        right = min(b[i], b[i] - t)
        if right > left:
            total += right - left
    return total

def objective(z):
    """Objective function to minimize: 1 - min_t I(t)"""
    ℓ, g = decode(z)
    a, b = build_intervals(ℓ, g)
    T = candidate_shifts(a, b)
    min_I = float('inf')
    for t in T:
        I_t = intersection_length(t, a, b)
        min_I = min(min_I, I_t)
    return 1 - min_I

def gridify(intervals, n):
    """Convert the intervals to a step function on grid."""
    h = np.zeros(n)
    dx = 2.0 / n
    cut_points = [(a, b) for a, b in intervals]
    # fill grid
    for i in range(n):
        x = dx * (i + 0.5)  # center of cell
        total = 0
        for a, b in cut_points:
            if a <= x <= b:
                total += 1
        h[i] = total
    # adjust to sum = n/2
    current_sum = np.sum(h)
    diff = current_sum - n/2
    if abs(diff) > 0.5:
        # find indices to modify
        if diff > 0:
            idxs = np.argsort(h)[::-1]  # descending order
        else:
            idxs = np.argsort(h)  # increasing order
        for idx in idxs:
            if abs(diff) < 0.5:
                break
            if diff > 0:
                h[idx] = 0
                diff -= 1
            else:
                h[idx] = 1
                diff += 1
    # at most one element off in rounding
    # check correctness and fix if needed
    current_sum = np.sum(h)
    if current_sum != n / 2:
        idx = int((n/2 - current_sum) + n/2) % n
        h[idx] = 1 if h[idx] == 0 else 0
    return h

def binary_hill_climb(h, time_left, max_iter=1000):
    """Simple hill climbing on binary function."""
    start_time = time()
    best = h.copy()
    c5_best = evaluate_erdos_solution(best)
    min_iter = 10
    iter_count = 0
    while (time() - start_time < time_left) and iter_count < max_iter:
        i = np.random.randint(len(h))
        j = np.random.randint(len(h))
        if i == j:
            continue
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        c5_new = evaluate_erdos_solution(h)
        if c5_new < c5_best:
            best = h.copy()
            c5_best = c5_new
        else:
            h[i], h[j] = h[j], h[i]  # revert
        iter_count += 1
    return best, c5_best

def evaluate_erdos_solution(h_vals):
    """Evaluate the solution using the built-in."""
    n = len(h_vals)
    dx = 2.0 / n
    corr = np.correlate(h_vals, 1 - np.array(h_vals), mode="full")
    max_corr = np.max(corr)
    return max_corr * dx
