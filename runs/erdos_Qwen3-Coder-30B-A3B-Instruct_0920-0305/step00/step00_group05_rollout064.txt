import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_h(h_values):
    # Compute C5 = max_k int h(x)(1-h(x+k)) dx
    # Using convolution with 1-h
    h = np.array(h_values)
    one_minus_h = 1 - h
    # Full convolution
    conv = correlate(h, one_minus_h, mode="full")
    # The convolution is of length 2*n - 1
    # The maximum is over all possible shifts k from -n+1 to n-1
    # Which corresponds to indices from 0 to 2*n-2 in the conv array
    # Index n-1 in conv corresponds to shift k = 0
    c5 = np.max(conv)
    return c5

def evaluate_erdos_solution(h_values):
    """Evaluate the solution and return the C5 bound"""
    return evaluate_h(h_values)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction - use provided initial_h_values if available
    # We'll use a discrete implementation approach using scipy.optimize
    n_points = 70
    dx = 2.0 / n_points
    
    # Start from the provided initial solution if available
    try:
        h_init = initial_h_values.copy()
        h_init = h_init / np.sum(h_init) * (n_points / 2)
    except:
        # Otherwise, create a random initial solution
        h_init = np.random.rand(n_points)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
    
    # Create bounds - all values between 0 and 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # The constraint is sum(h) * dx = 1, i.e. sum(h) = n_points/2
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Function to optimize
    def objective(h):
        # Return the negative of C5 (because we minimize, but want minimize max)
        # Converting correlation to the C5 value
        return evaluate_h(h)
    
    # Define constraints
    cons = {'type': 'eq', 'fun': constraint_func}
    
    # Use scipy.optimize.minimize with L-BFGS-B or SLSQP algorithm
    # Budget is for total time
    best_h = None
    best_c5 = float('inf')
    
    # Start with initial solution
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
    
    # Try a few different starting points (霰弹式优化)
    attempts = 0
    while time.time() - start_time < budget_s - 1 and attempts < 50:
        attempts += 1
        # Generate a random noisy version of current best solution
        noise = np.random.normal(scale=0.1, size=n_points)
        h_candidate = best_h + noise
        
        # Clamp to [0, 1]
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Rescale to meet integral constraint
        h_candidate = h_candidate / np.sum(h_candidate) * (n_points / 2)
        
        # Evaluate
        c5 = evaluate_h(h_candidate)
        
        # See if this is better
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()
        
        # Check if we have some time left
        if time.time() - start_time > budget_s - 1:
            break
    
    # Also do some gradient-based refinement if time allows  
    if time.time() - start_time < budget_s - 10:
        try:
            # Alternative optimization approach - Nelder-Mead
            # Convert to flat parameters
            initial_flat = best_h.copy()
            result = minimize(objective, initial_flat, method='Nelder-Mead', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 1000})
            if result.success:
                c5_candidate = evaluate_h(result.x)
                if c5_candidate < best_c5:
                    best_h = result.x.copy()
                    best_c5 = c5_candidate                    
        except:
            pass
    
    # Final check
    final_c5 = evaluate_h(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
