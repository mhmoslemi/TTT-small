import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given h"""
    dx = 2.0 / len(h)
    correlations = np.correlate(h, 1 - h, mode='full')
    return np.max(correlations) * dx

def project_onto_simplex_box(h, target_sum):
    """Project h onto the feasible region: 0 <= h <= 1 and sum(h) = target_sum"""
    # Clip to box constraints
    h = np.clip(h, 0, 1)
    # Adjust to sum constraint using simple rescaling
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple scaling adjustment
    if current_sum > 0:
        h = h * target_sum / current_sum
        h = np.clip(h, 0, 1)
    return h
