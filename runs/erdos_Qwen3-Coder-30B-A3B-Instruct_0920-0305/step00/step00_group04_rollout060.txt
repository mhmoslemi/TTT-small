import numpy as np
import time
import random

def compute_c5_fast(h):
    n = h.shape[0]
    size = 1 << ((2*n-1).bit_length())
    f_h = np.fft.fft(h, size)
    f_g = np.fft.fft(1 - h, size)
    corr = np.fft.ifft(f_h * np.conj(f_g)).real
    corr = corr[:2*n-1]
    dx = 2.0 / n
    max_val = np.max(corr) * dx
    return max_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    
    n_points = 60
    dx = 2.0 / n_points
    target_c5 = 0.3807
    
    # Initialize with a random binary pattern (exactly n/2 ones)
    def random_binary():
        h = np.zeros(n_points)
        ones_indices = np.random.choice(n_points, n_points//2, replace=False)
        h[ones_indices] = 1
        return h
    
    # Simulated Annealing
    max_iterations = 200000
    temperature_start = 0.1
    temperature_end = 1e-6
    alpha = 0.999
    
    iteration = 0
    
    # Track time and compute temperature
    while time.time() - start_time < budget_s - 0.2 and iteration < max_iterations:
        # Generate new candidate
        h = random_binary()
        current_c5 = compute_c5_fast(h)
        
        # Track best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if best_c5 <= target_c5:
            break
        
        iteration += 1
        
        # Adaptive temp update
        elapsed_ratio = (time.time() - start_time) / budget_s
        temperature = temperature_start * (temperature_end / temperature_start) ** elapsed_ratio
        
        if temperature < 1e-6:
            temperature = 1e-6
    
    if best_h is None:
        # If nothing was found, use a random pattern as fallback
        best_h = random_binary()
        best_c5 = compute_c5_fast(best_h)
    
    return (best_h, best_c5, n_points)
