import numpy as np
import time
import random
import math
from scipy.special import softmax

# Helper functions
def evaluate(x):
    """Evaluate C5 for given step function x"""
    dx = 2.0 / len(x)
    correlations = np.correlate(x, 1 - x, mode="full")
    return np.max(correlations) * dx

def project_onto_capped_simplex(y, target_sum):
    """Project y onto the capped simplex {x : 0 <= x <= 1, sum(x) = target_sum}"""
    y = np.clip(y, 0, 1)
    y_sorted = np.sort(y)[::-1]
    rho = np.sum(y_sorted > 0)
    while rho > 0 and np.sum(y_sorted[:rho]) > target_sum:
        rho -= 1
    lambda_val = (np.sum(y_sorted[:rho]) - target_sum) / rho if rho > 0 else 0
    x = np.clip(y - lambda_val, 0, 1)
    return x

def gradient_of_max(x, beta):
    """Compute gradient of smoothed max"""
    N = len(x)
    correlations = np.correlate(x, 1 - x, mode="full")
    weights = softmax(beta * correlations)
    grad = np.zeros(N)
    for k in range(-N + 1, N):
        if k >= 0:
            i_indices = range(N - k)
            j_indices = range(k, N)
        else:
            i_indices = range(-k, N)
            j_indices = range(N + k)
        
        for i, j in zip(i_indices, j_indices):
            grad[i] += weights[k + N - 1] * (1 - x[j])
            grad[j] -= weights[k + N - 1] * x[i]
    return grad

def swap_move(x, i, j):
    """Perform a swap move between positions i and j"""
    x_new = x.copy()
    x_new[i], x_new[j] = x_new[j], x_new[i]
    return x_new

def fractional_move(x, i, j, delta):
    """Move mass delta from j to i"""
    x_new = x.copy()
    if (x_new[i] + delta <= 1.0 and x_new[j] - delta >= 0.0):
        x_new[i] += delta
        x_new[j] -= delta
    else:
        x_new[i] = np.clip(x_new[i] + delta, 0.0, 1.0)
        x_new[j] = np.clip(x_new[j] - delta, 0.0, 1.0)
    return x_new

def worst_shift_info(x):
    """Return the worst shift and offending indices"""
    dx = 2.0 / len(x)
    correlations = np.correlate(x, 1 - x, mode="full")
    k_star = np.argmax(correlations)
    shift = k_star - len(x) + 1
    corr = correlations[k_star]
    # Find the indices
    if shift >= 0:
        offending = [i for i in range(len(x)-shift) if x[i] == 1 and x[i+shift] == 0]
    else:
        shift_abs = abs(shift)
        offending = [i for i in range(shift_abs, len(x)) if x[i] == 1 and x[i-shift_abs] == 0]
    return shift, offending, corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Candidate sizes to try
    candidate_sizes = [140, 200, 260]
    best_c5_bound = float('inf')
    best_h = None
    best_n_points = None
    
    t_start = time.time()
    safety_margin = 5.0
    t_dead = t_start + budget_s - safety_margin
    
    for N in candidate_sizes:
        if time.time() > t_dead:
            break
        
        # Time allocation per N
        remaining_time = t_dead - time.time()
        if remaining_time <= 0:
            break
            
        # Allocate time slice for this N
        t_slice = remaining_time / len(candidate_sizes)
        t_end = time.time() + t_slice
        best_binary = None
        best_binary_c5 = float('inf')
        
        # Phase 2.1: Binary simulated annealing
        T0 = 0.12
        max_iter = 30000
        current_iter = 0
        T = T0
        
        while time.time() < t_end and current_iter < max_iter:
            # Generate random binary vector
            h = np.zeros(N)
            ones_indices = np.random.choice(N, N//2, replace=False)
            h[ones_indices] = 1
            
            # Initial evaluation
            current_c5 = evaluate(h)
            best_h_local = h.copy()
            best_c5_local = current_c5
            
            # Some state to track improvement
            no_improvement_iter = 0
            last_best_c5 = current_c5
            
            # Simulated Annealing loop
            sa_iter = 0
            while time.time() < t_end and sa_iter < 10000 and no_improvement_iter < 2000:
                # Pick two random indices with different values
                i, j = -1, -1
                while i == j or h[i] == h[j]:
                    i = np.random.randint(0, N)
                    j = np.random.randint(0, N)
                
                # Move
                h_new = swap_move(h, i, j)
                new_c5 = evaluate(h_new)
                
                # Accept or reject
                delta = new_c5 - current_c5
                if delta <= 0 or np.random.rand() < np.exp(-delta / T):
                    h = h_new
                    current_c5 = new_c5
                    if current_c5 < best_c5_local:
                        best_h_local = h.copy()
                        best_c5_local = current_c5
                        last_best_c5 = current_c5
                        no_improvement_iter = 0
                    else:
                        no_improvement_iter += 1
                
                sa_iter += 1
                current_iter += 1
                
                # Cooling
                if sa_iter % 500 == 0:
                    T = T0 * (1 - sa_iter / 10000)
        
        # Update the best binary solution for this N
        if best_c5_local < best_binary_c5:
            best_binary_c5 = best_c5_local
            best_binary = best_h_local.copy()
        
        # Phase 2.2: Directed local search for the best binary vector
        if time.time() < t_dead and best_binary_c5 < float('inf'):
            # Convert binary to float for projected gradient step
            h = best_binary.astype(float)
            
            # Local improvement using worst shift targeting
            local_iter = 0
            max_local_iter = 10000
            while time.time() < t_dead and local_iter < max_local_iter:
                k_star, offending_indices, min_corr = worst_shift_info(h)
                # Sample a bad index
                if len(offending_indices) > 0:
                    i = random.choice(offending_indices)
                    # Find a valid j
                    valid_j = []
                    for j in range(len(h)):
                        if h[j] == 0 and (j != i) and (j != i + abs(k_star)):
                            valid_j.append(j)
                    if valid_j:
                        j = random.choice(valid_j)
                        h_old = h.copy()
                        h = swap_move(h, i, j)
                        new_c5 = evaluate(h)
                        if new_c5 >= min_corr:
                            # Revert if no improvement
                            h = h_old
                local_iter += 1
        
        # Phase 3: Continuous refinement
        if time.time() < t_dead and best_binary_c5 < float('inf'):
            h = best_binary.astype(float)
            # Refine continuously
            beta = 5.0
            max_refinement_iter = 2000
            refinement_iter = 0
            while time.time() < t_dead and refinement_iter < max_refinement_iter:
                # Project
                h = project_onto_capped_simplex(h, N / 2)
                
                # Compute gradient of smoothed max
                grad = gradient_of_max(h, beta)
                
                # Step
                h = h - 0.01 * grad
                
                # Re-project after gradient step
                h = project_onto_capped_simplex(h, N / 2)
                
                # Increase beta for smoothing
                beta *= 1.02
                
                refinement_iter += 1
        
        # Final evaluation
        h_final = best_binary.astype(float) if best_binary is not None else None
        if h_final is not None:
            final_c5 = evaluate(h_final)
            # If this result is better, update global best
            if final_c5 < best_c5_bound:
                best_c5_bound = final_c5
                best_h = h_final.copy()
                best_n_points = N
                
    return (best_h.tolist(), best_c5_bound, best_n_points)
