import time

def evaluate_h(h):
    """Evaluate the overlap integral for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the full convolution of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum value is the one we want
    c5 = np.max(corr) * dx
    return c5
