import numpy as np
import scipy.optimize as opt
from scipy.optimize import minimize
import time
import math

# Predefined initial construction
initial_h_values = None

def evaluate_h(h):
    """Evaluate the C5 bound for a given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The correlation is of length 2*n-1
    # The valid overlaps are from index n-1 to 2*n-2
    overlaps = corr[n-1:2*n-1] * dx
    c5 = np.max(overlaps)
    return c5

def objective(h):
    """Objective function to minimize."""
    # Just return the C5 value
    return evaluate_h(h)

def constraint_eq(h):
    """Equality constraint: integral of h should be 1."""
    n = len(h)
    dx = 2.0 / n
    return np.sum(h) * dx - 1

def constraint_ineq(h):
    """Inequality constraints: h[i] should be in [0,1]."""
    n = len(h)
    lower = -h  # h[i] >= 0
    upper = 1 - h  # h[i] <= 1
    return np.concatenate([lower, upper])

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    
    # Start time tracking
    start_time = time.time()
    
    # Get initial h from the pre-imported variable or create one
    if initial_h_values is None:
        # Create a smooth initial guess
        n_points = 70
        x = np.linspace(0, 2, n_points)
        # Try an initial solution like a smooth bump
        h0 = np.exp(-10 * (x - 1)**2)
        h0 = h0 / np.sum(h0) * 2  # normalize to sum to 2
    else:
        h0 = np.array(initial_h_values)
        n_points = len(h0)

    # Make sure h0 is in [0,1]
    h0 = np.clip(h0, 0, 1)
    # Adjust it to integral = 1
    dx = 2.0 / n_points
    integral = np.sum(h0) * dx
    h0 = h0 / integral

    # Define constraints for scipy.optimize
    # For the bounds on h[i] in [0,1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    # Equality constraint: integral of h should be 1
    eq_con = {'type': 'eq', 'fun': constraint_eq}
    # Inequality constraints for bounds
    ineq_con = {'type': 'ineq', 'fun': constraint_ineq}

    # We'll use L-BFGS-B which handles bounds well and supports constraints
    # Initialize optimizer with simple method first for a quick start
    min_res = minimize(objective, h0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': int(budget_s*0.4)}, 
                       callback=lambda x: None if time.time() - start_time > budget_s*0.9 else None)
    
    best_h = min_res.x
    best_c5 = evaluate_h(best_h)
    
    # Continue refining with a more exploratory search
    # Try gradient-based optimization with different random restarts
    for i in range(20):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Random perturbation of current solution
        h = best_h + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        integral = np.sum(h) * dx
        h = h / integral  # renormalize
        
        # Optimize from this perturbed point
        try:
            res = minimize(objective, h, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': int(budget_s*0.02)})
            c5 = evaluate_h(res.x)
            if c5 < best_c5:
                best_h = res.x
                best_c5 = c5
        except:
            continue

    # Final evaluation
    final_c5 = evaluate_h(best_h)
    
    return (list(best_h), final_c5, n_points)
