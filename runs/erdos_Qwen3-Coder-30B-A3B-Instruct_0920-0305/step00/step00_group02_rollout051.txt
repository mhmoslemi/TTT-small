import time
import numpy as np
import cvxpy as cp
import random
import math

def random_binary_vector(n):
    # Create binary vector with exactly n//2 ones
    h = np.zeros(n, dtype=int)
    ones_indices = np.random.choice(n, size=n//2, replace=False)
    h[ones_indices] = 1
    return h

def compute_c5(h):
    # Compute dx
    dx = 2.0 / len(h)
    # Compute correlation using np.correlate with mode='full'
    corr = np.correlate(h, 1 - h, mode='full')
    # Take only the positive shifts (k from 0 to n-1)
    # Full correlation indices go from -(n-1) to +(n-1)
    # Positive shifts are at indices n-1 to 2n-2
    positive_shifts = corr[n-1:]
    # The maximum value is the C5 bound
    c5 = np.max(positive_shifts) * dx
    return c5

def solve_milp(n, time_limit=300):
    # Define variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Constraint: sum of h equals n/2
    constraints = [cp.sum(h) == n//2]
    
    # For each shift k, define overlap sum as sum of products
    for k in range(1, n):  # Skip k=0 as it's always n/2
        # Number of valid positions for shift k
        valid_positions = n - k
        if valid_positions <= 0:
            continue
        # Linearize h_i * h_{i+k} for i=0..valid_positions-1
        y = cp.Variable(valid_positions)
        # Constraints for linearization (product constraint)
        constraints += [y >= 0]
        constraints += [y <= h[:valid_positions]]
        constraints += [y <= h[k:]]
        constraints += [y >= h[:valid_positions] + h[k:] - 1]
        # Constraint: sum of y should be >= t
        constraints += [cp.sum(y) >= t]
    
    # Constraint on t: it should not exceed n/2 (since shift 0 term is guaranteed to be n/2)
    constraints += [t <= n//2]
    
    # Define the problem
    prob = cp.Problem(cp.Maximize(t), constraints)
    
    # Use CBC solver with time limit
    try:
        prob.solve(solver=cp.CBC, maxSeconds=time_limit)
    except:
        # If solver fails, fallback to heuristic approach
        raise ValueError("MILP solver timed out or failed")
    
    # Extract solution
    h_sol = np.round(h.value).astype(int)
    # Compute actual C5 bound using the solution
    dx = 2.0 / n
    corr = np.correlate(h_sol, 1 - h_sol, mode='full')
    positive_shifts = corr[n-1:]
    c5 = np.max(positive_shifts) * dx
    
    return h_sol, c5

def local_search(h, max_iters, time_limit):
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    iteration = 0
    while iteration < max_iters and (time.time() - start) < time_limit:
        # Generate random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Perform swap
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)
        
        # Accept improvement
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
        else:
            # Revert swap
            h[i] = 1
            h[j] = 0
        
        iteration += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()

    # Try to solve with MILP using reasonably small n
    n = 100
    best_c5 = float('inf')
    best_h = None
    
    try:
        elapsed = time.time() - start
        # Use MILP solver with timeout
        h_opt, c5_opt = solve_milp(n, time_limit=min(300, budget_s - elapsed - 5))
        best_h = h_opt
        best_c5 = c5_opt
    except Exception:
        # If MILP fails, use heuristic with initial vector or random construction
        # Using initial_h_values if available
        try:
            # Initialize with a copy of initial_h_values if available
            if 'initial_h_values' in globals():
                h_init = np.array(initial_h_values)
                # Check if we need to pad or truncate
                if len(h_init) < n:
                    h_init = np.pad(h_init, (0, n - len(h_init)), constant_values=0)
                elif len(h_init) > n:
                    h_init = h_init[:n]
                # Ensure it has n//2 ones
                if np.sum(h_init) != n//2:
                    h_init = random_binary_vector(n)
            else:
                h_init = random_binary_vector(n)
        except:
            h_init = random_binary_vector(n)
        h_opt = h_init.copy()
        c5_opt = compute_c5(h_init)
        best_h = h_opt
        best_c5 = c5_opt

    # Run local search on the result
    elapsed = time.time() - start
    remaining = max(0, budget_s - elapsed)
    
    if remaining > 5:
        try:
            h_final, c5_final = local_search(best_h.copy(), max_iters=50000, time_limit=remaining - 2)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
        except Exception:
            pass  # Continue with best so far

    return best_h.tolist(), float(best_c5), n
