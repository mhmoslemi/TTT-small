import numpy as np
import cvxpy as cp
import time


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Configuration
    n = 60  # Number of sample points
    dx = 2.0 / n
    target_c5 = 0.3807
    target_t = int(target_c5 * n / 2)
    
    # Time budgeting
    solve_time = 0.5 * budget_s
    polish_time = 0.5 * budget_s
    
    # Variables
    b = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(b) == n // 2]
    
    # Create auxiliary variables and constraints for each lag
    w_vars = {}
    N_exprs = []
    
    # We only iterate over positive lags and use symmetry
    for lag in range(1, n):
        if time.time() - start_time > solve_time:
            break
        # For lag l, w[i] = b[i] AND b[i+l]
        w = cp.Variable(n - lag, boolean=True)
        w_vars[lag] = w
        
        # Add constraints for each pair
        for i in range(n - lag):
            constraints.append(w[i] <= b[i])
            constraints.append(w[i] <= b[i + lag])
            constraints.append(w[i] >= b[i] + b[i + lag] - 1)
        
        # Compute N_lag = sum(b) - sum(w)
        N_lag = cp.sum(b[:n-lag]) - cp.sum(w)
        N_exprs.append(N_lag)
        constraints.append(t >= N_lag)
    
    # Symmetry breaking
    constraints.append(b[0] == 1)
    for i in range(n - 1):
        constraints.append(b[i] >= b[i + 1])
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.CBC, maxSeconds=solve_time, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_EOF]:
            b_val = b.value
            t_val = t.value
            
            # Convert back to h_values
            h_values = b_val.astype(float)
            c5_bound = t_val * dx

def compute_overlap(h_vals, dx):
    h = np.array(h_vals)
    n = len(h)
    max_overlap = 0.0
    for k in range(-n+1, n):
        correlation = 0.0
        if k >= 0:
            correlation = np.sum(h[:n-k] * (1 - h[k:]))
        else:
            correlation = np.sum(h[-k:] * (1 - h[:-k]))
        max_overlap = max(max_overlap, correlation)
