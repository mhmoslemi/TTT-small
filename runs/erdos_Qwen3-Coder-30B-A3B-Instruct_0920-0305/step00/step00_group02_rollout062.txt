import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):

    n = 300  # number of samples
    dx = 2.0 / n
    max_iter = 50000  # max iterations before timeout
    domain = np.arange(n)

    beta = (np.sqrt(5) - 1) / 2
    h0 = np.array([1 if (i * beta) % 1 < 0.5 else 0 for i in range(n)])
    sum_h0 = np.sum(h0)
    target_sum = n // 2

    if sum_h0 != target_sum:
        diff = target_sum - sum_h0

            # Need to flip zeros to ones

            flipped = np.random.choice(zeros, diff, replace=False)
            h0[flipped] = 1
        else:
            # Need to flip ones to zeros
            ones = np.where(h0 == 1)[0]
            flipped = np.random.choice(ones, -diff, replace=False)
            h0[flipped] = 0
    
    # Return early if seed ensures direct usage

    h = h0.astype(float)
    h = project_onto_feasible(h, target_sum)
    
    # Gradient ascent parameters
    alpha = 1.0
    alpha_max = 10000.0
    eta = 1.0
    tol = 1e-8
    best_C5 = np.inf
    best_h = h.copy()
    
    # Start timer
    start_time = time.time()
    
    current_time = start_time
    iter_count = 0
    while (current_time - start_time) < budget_s * 0.6 and iter_count < max_iter:
        current_time = time.time()
        iter_count += 1
        
        # Compute overlaps S_k via correlation
        overlaps = np.correlate(h, h, mode='full')
        # Correct indexing
        k_vals = list(range(len(overlaps)))
        S_vals = overlaps[len(overlaps)//2:]
        
        # Smooth min using log-sum-exp
        exp_neg_alpha_S = np.exp(-alpha * np.array(S_vals))
        Z = np.sum(exp_neg_alpha_S)
        if Z <= 0:
            break
        Phi_alpha = - (1.0 / alpha) * np.log(Z)
        
        # Gradient of Phi_alpha
        grad = np.zeros_like(h)
        for i in range(len(S_vals)):
            # Self-correlation terms
            k = i
            if k < len(h):
                grad += exp_neg_alpha_S[i] * h[k]  # h_{i+k}
            if i > 0:
                k = i - len(h)
                if k < len(h) and k >= 0:
                    grad += exp_neg_alpha_S[i] * h[k]  # h_{i-k}
        
        grad /= Z
        
        # Backtracking line search for eta
        old_phi = Phi_alpha
        
        # Step size backtracking
        while True:
            h_new = h + eta * grad
            h_new = project_onto_feasible(h_new, target_sum)
            
            # Evaluate new Phi_alpha
            overlaps_new = np.correlate(h_new, h_new, mode='full')
            S_vals_new = overlaps_new[len(overlaps_new)//2:]
            
            exp_neg_alpha_S_new = np.exp(-alpha * np.array(S_vals_new))
            Z_new = np.sum(exp_neg_alpha_S_new)
            if Z_new <= 0:
                eta *= 0.5
                continue
            
            Phi_alpha_new = - (1.0 / alpha) * np.log(Z_new)
            if Phi_alpha_new > old_phi + 1e-10 or eta < 1e-10:
                break
            eta *= 0.5
        
        h = h_new
        
        # Increase alpha
        if iter_count % 200 == 0 and alpha < alpha_max:
            alpha *= 2
        
        # Check for improvement

        max_overlap = np.max(overlaps_final) * dx
        if max_overlap < best_C5:
            best_C5 = max_overlap
            best_h = h.copy()

    h_bin = h.copy()
    median_val = np.median(h_bin)

    diff = int(target_sum - sum_bin)
    
    if diff > 0:
        zeros = np.where(h_bin == 0)[0]
        chosen = np.random.choice(zeros, diff, replace=False)
        h_bin[chosen] = 1
    elif diff < 0:
        ones = np.where(h_bin == 1)[0]
        chosen = np.random.choice(ones, -diff, replace=False)
        h_bin[chosen] = 0
    
    # Local hill climbing (swap-based)
    current_time = time.time()
    while (time.time() - current_time) < budget_s * 0.3:
        # Swap a 1 and 0 more carefully: avoid restarts
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break

        j = np.random.choice(zeros)
        
        # Create test vector
        h_test = h_bin.copy()
        h_test[i] = 0
        h_test[j] = 1
        
        c_old = np.correlate(h_bin, 1 - h_bin, mode='full')

        old_max = np.max(c_old) * dx
        new_max = np.max(c_new) * dx

        if new_max < old_max:
            h_bin = h_test

    # Final evaluation
    c_final = np.correlate(h_bin, 1 - h_bin, mode='full')
    final_C5 = np.max(c_final) * dx
    
    return (h_bin.tolist(), float(final_C5), n)

def project_onto_feasible(h, target_sum):
    """Project onto [0,1]^n and sum = target_sum"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum - target_sum) < 1e-9:
        return h
    
    # Waterfilling algorithm to adjust to sum = target_sum
    h_sorted = np.sort(h)[::-1]  # descending order
    
    # Find partition threshold
    left_sum = 0
    for i in range(len(h_sorted)):
        right_sum = np.sum(h_sorted[i+1:])
        total = left_sum + (len(h_sorted) - i) * h_sorted[i]
        if total >= target_sum:
            # Interpolate
            left_sum = left_sum + h_sorted[i] * i
            diff = target_sum - left_sum
            threshold = diff / (len(h_sorted) - i)
            break
        else:
            left_sum += h_sorted[i]
    else:
        threshold = h_sorted[-1] if len(h_sorted) > 0 else 0
    
    # Clip to threshold, then re-scale
    adjusted = np.clip(h, 0, threshold)
    rescale = target_sum / np.sum(adjusted)
    adjusted = adjusted * rescale
    
    return adjusted
