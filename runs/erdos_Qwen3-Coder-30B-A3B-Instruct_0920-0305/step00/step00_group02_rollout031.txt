import time
import numpy as np
import cvxpy as cp
import random
import math

def random_binary_vector(n):
    # Create binary vector with exactly n/2 ones and n/2 zeros
    h = np.zeros(n, dtype=int)
    indices = list(range(n))
    random.shuffle(indices)
    for i in range(n//2):
        h[indices[i]] = 1
    return h

def compute_c5(h):
    # Compute dx
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # C5 is the maximum of the correlation values
    C5 = np.max(corr) * dx
    return C5

def solve_milp(n):
    # Sleep briefly to avoid solver startup time interference
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Constraint: sum of h equals n/2
    constraints = [cp.sum(h) == n // 2]
    
    # Linearization: for each shift k, define overlap_k = sum_{i} h_i * h_{i+k}
    for k in range(1, n):  # Skip k=0 as it's always high
        if n - k == 0:
            continue
        # y is 1 if both positions i and i+k are 1
        y = cp.Variable(n - k, boolean=False)  # Keep as continuous but bounded
        # Constraints: linearize h_i * h_{i+k}
        constraints += [y <= h[:n-k]]
        constraints += [y <= h[k:]]
        constraints += [y >= h[:n-k] + h[k:] - 1]
        constraints += [y >= 0]
        # The overlap for this shift
        constraints += [cp.sum(y) >= t]
    
    # Objective: maximize t such that it is bounded by each overlap
    prob = cp.Problem(cp.Maximize(t), constraints)
    
    # Solve with CBC for MILP
    try:
        prob.solve(solver=cp.CBC, verbose=False)
        # Extract solution
        h_sol = np.round(h.value).astype(float)
        # Recompute C5 based on integer solution
        dx = 2.0 / n
        corr = np.correlate(h_sol, 1 - h_sol, mode='full')
        C5 = np.max(corr) * dx
        # Also ensure the solution has correct constraint sum
        if abs(np.sum(h_sol) - n/2) > 1e-5:
            # Handle case if rounding gives invalid sum
            # Reset to binary random solution
            h_sol = random_binary_vector(n)
            corr = np.correlate(h_sol, 1 - h_sol, mode='full')
            C5 = np.max(corr) * dx
        return h_sol, C5
    except Exception as e:
        # Fallback if MILP fails
        h_sol = random_binary_vector(n)
        corr = np.correlate(h_sol, 1 - h_sol, mode='full')
        C5 = np.max(corr) * dx
        return h_sol, C5

def local_search(h_initial, time_limit, target_c5):
    # Initialize
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h_initial.copy()
    cur_c5 = best_c5
    
    start = time.time()
    num_iters = 0
    
    # Perform swaps until timeout
    while time.time() - start < time_limit:
        num_iters += 1
        # Select two random indices: one 1 and one 0
        ones_idx = np.where(cur_h == 1)[0]
        zeros_idx = np.where(cur_h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        
        # Swap
        cur_h[i] = 0
        cur_h[j] = 1
        
        # Compute new C5
        new_c5 = compute_c5(cur_h)
        
        # Accept if better
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # Revert swap
            cur_h[i] = 1
            cur_h[j] = 0
        
        # Occasionally check for early exit
        if num_iters % 10000 == 0 and best_c5 <= target_c5:
            break
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    
    # Determine initial n - small enough for probabilistic MILP
    n = 100  # Empirical selection to keep within 600 seconds
    
    # Solve MILP for a small n to get a good starting point
    try:
        h_opt, c5_opt = solve_milp(n)
        elapsed = time.time() - start
        remaining_time = budget_s - elapsed - 10  # Leave 10s for cleanup
        if remaining_time > 5:
            h_opt, c5_opt = local_search(h_opt, remaining_time, 0.3807)
        return h_opt.tolist(), float(c5_opt), n
    except:
        # Fallback pattern for setup issues
        h = random_binary_vector(n)
        c5 = compute_c5(h)
        return h.tolist(), float(c5), n
