import numpy as np
import time
import scipy.signal as signal
from scipy.fftpack import fft, ifft
from scipy.optimize import root_scalar
import math

# Helper functions
def evaluate_correlation_fft(h, dx, n):
    L = 2 * n - 1
    L = 1 << (int(math.log2(L)) + 1)  # next power of 2
    H = fft(h, L)
    G = fft(1 - h, L)
    corr = ifft(H * np.conj(G)).real[:2*n-1]
    return corr * dx

def evaluate_c5(h, dx, n):
    corr = evaluate_correlation_fft(h, dx, n)
    return np.max(corr)

def project_to_simplex(y, target_sum):
    """Project y onto capped simplex with sum = target_sum"""
    def S(lam):
        return np.sum(np.clip(y - lam, 0, 1)) - target_sum

    low, high = np.min(y) - 1, np.max(y) + 1
    try:
        res = root_scalar(S, bracket=[low, high], method="brentq")
        lam = res.root
    except:
        lam = (np.sum(y) - target_sum) / len(y)
        # fallback projection with simpler method if root_scalar fails
        if lam > 0:
            lam = np.max([np.min(y) - 1, lam - 0.1])
        else:
            lam = np.min([np.max(y) + 1, lam + 0.1])
        for _ in range(50):
            if S(lam) > 0:
                lam += (high - lam) / 2
            else:
                lam -= (lam - low) / 2
    x = np.clip(y - lam, 0, 1)
    return x

def compute_gradient(h, dx, n, gamma=5.0):
    # evaluate_all_correlations
    L = 2 * n - 1
    L = 1 << (int(math.log2(L)) + 1)
    H = fft(h, L)
    G = fft(1 - h, L)
    corr = ifft(H * np.conj(G)).real[:2*n-1]
    # Compute gradient w.r.t. h
    # using the method derivisions from the optimization problems:

    # we're computing:
    #  \partial / \partial h_i \sum_{\ell} w_\ell \cdot \Delta f_\ell(h)
    # where \Delta f_\ell = dx [ \sum_{j} h_j (1 - h_{j+\ell}) ] = dx [ \sum_{j} h_j - \sum_{j} h_j h_{j+\ell}]

    # f = dx \sum h[j] (1 - h[j+\ell])
    #  df_i / dh_j = dx (1 - h_{j+\ell}) \delta_{i,j} - dx h_{i-\ell} \delta_{j, i-\ell}  

    # So: df_i / dh_i = dx (1 - h_i), and df_i / dh_{i+\ell} = - dx h_i
    # idxs for valid correlations
    df = np.zeros_like(h)

    # update all positions
    grad = np.zeros_like(h)
    for k in range(n):
        corr_vals = 0.0
        if k > 0:
            # h_i \sum_{j} h_j (1 - h_{j+k})
            # contribution from h_{i-k}
            corr_vals += np.sum(h[k:] * (1 - h[:n-k]))
        for l in range(-n+1, n):
            if abs(l) <= k:
                if l < 0:
                    val = h[k-l] * (1 - h[k])      
                else:
                    val = -h[k] * h[k+l]
                grad[k] += val * dx
            if l >= 0:
                break
        # include contribution term for current position
    # now compute weights
    corr = corr * dx
    exp_arg = gamma * corr
    exp_arg -= np.max(exp_arg)  
    w = np.exp(exp_arg)
    w /= np.sum(w)

    # compute gradient via proxy
    grad = np.zeros_like(h)
    for l in range(-n+1, n):
        # actual inner loop for finite convolution
        # could reuse from signal.correlate or build a vectorized loop
        start = max(0, l)
        end = min(n, n + l)
        if l >= 0:
            # slide h_{i-l}
            # for i in range(start, end):
            #     grad[i] += w[l+n-1] * (1 - h[i-l]) * dx # corr 
            #     grad[i-l] += w[l+n-1] * (-h[i]) * dx # corr
            indices1 = np.arange(start, end)  # i
            indices2 = indices1 - l  # i - l
            grad[indices1] += w[l+n-1] * (1 - h[indices2]) * dx
            grad[indices2] += w[l+n-1] * (-h[indices1]) * dx
        else:
            # l < 0, i from -l to n-l, i - l just shifts
            indices1 = np.arange(start, end)
            indices2 = indices1 - l
            grad[indices2] += w[l+n-1] * (1 - h[indices1]) * dx
            grad[indices1] += w[l+n-1] * (-h[indices2]) * dx

    return grad

def binary_sa(n, max_time_per_restart, total_time, start_time, dt):
    # Generate seed from initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        while len(initial_h) < n:
            _ = np.concatenate([initial_h, initial_h[:n - len(initial_h)]])
        h = initial_h[:n]
    else:
        h = np.random.rand(n) < 0.5
    h = h.astype(int)
    # ensure sum = n/2
    n2 = n // 2
    if np.sum(h) != n2:
        diff = n2 - np.sum(h)
        if diff > 0:
            # Add 1s to fewer indices
            idxs = np.argsort(h)[::-1][:diff]
            h[idxs] = 1
        else:
            # Convert to 0s
            idxs = np.argsort(h)[:abs(diff)]
            h[idxs] = 0
    assert np.sum(h) == n2
    dx = 2 / n

    current_c5 = evaluate_c5(h, dx, n)
    best_h = h.copy()
    best_c5 = current_c5

    # parameters
    temp = 0.05
    cooling_rate = 0.9995
    max_iter = 1000000
    iters = 0
    start_restart = time.time()

    while (time.time() - start_time) < (0.7 * total_time - 1.0) and time.time() - start_restart < max_time_per_restart:
        # system temperature cooling
        temp = max(1e-5, temp * cooling_rate)
        # propose random swap
        idx1 = np.random.choice(np.where(h == 1)[0])
        idx2 = np.random.choice(np.where(h == 0)[0])
        h_new = h.copy()
        h_new[idx1], h_new[idx2] = 0, 1
        c5_new = evaluate_c5(h_new, dx, n)
        delta = c5_new - current_c5
        if delta <= 0 or np.random.rand() < np.exp(-delta / temp):
            h = h_new
            current_c5 = c5_new
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        iters += 1
        # Optionally reset timing if needed but restart logic handled by outer loop
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n = 60  # Use a reasonable value
    dx = 2.0 / n
    n2 = n // 2

    # Phase A: Binary Search
    max_time_per_restart = 10.0
    num_restarts = 0
    while (time.time() - start_time) < (0.7 * budget_s - 2.0):
        restart_start = time.time()
        h_best, c5_best = binary_sa(n, max_time_per_restart, budget_s, start_time, dx)
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best.copy()
        num_restarts += 1
        if (time.time() - start_time) > (0.7 * budget_s - 1.0):
            break

    # If we've reached the goal or nearly so, maybe proceed to phase B
    if best_c5 <= 0.3807:
        # Do not run Phase B, bring to the correct sum and return
        h_final = np.array(best_h, dtype=float)
        # Project to the correct sum
        # If the sum is exactly n/2 and elements are already in [0,1], we're good
        h_final = h_final / np.sum(h_final) * n2
        h_final = np.clip(h_final, 0, 1)
        # Force sum to be exactly n2 by projecting to simplex
        h_final = project_to_simplex(h_final, n2)
        c5_computed = evaluate_c5(h_final, dx, n)
        return (h_final, c5_computed, n)

    # Phase B: continuous optimization (only if we didn't already reach goal)
    
    # refine with full continuous optimization based on binary solution
    h = best_h.astype(float)
    h = h / np.sum(h) * n2
    h = np.clip(h, 0, 1)
    h = project_to_simplex(h, n2)
    
    dx = 2.0 / n
    max_iter = int(2000)  # Allow enough steps for convergence
    gamma = 5.0
    gamma_max = 200.0
    alpha = 0.1 * dx
    patience = 0
    patience_limit = 5
    last_improvement = 0
    old_c5 = float('inf')
    iter_count = 0
    
    # Embedded simple gradient loop to minimize the smooth max
    while iter_count < max_iter and (time.time() - start_time) < (budget_s - 2.0):
        # Compute gradient of smooth max
        grad = compute_gradient(h, dx, n, gamma)
        
        # approximate gradient with finite differences
        # we can select a better approach if needed
        h_proposed = h - alpha * grad
        
        # project back to simplex
        h_new = project_to_simplex(h_proposed, n2)
        
        # evaluate true C5 after projecting
        c5_new = evaluate_c5(h_new, dx, n)
        old_c5 = evaluate_c5(h, dx, n)
        
        if c5_new < old_c5:
            h = h_new
            last_improvement = iter_count
            patience = 0
        else:
            patience += 1
            if patience == 2:
                alpha *= 0.9
                patience = 0
        
        iter_count += 1
        if gamma < gamma_max and iter_count % 10 == 0:
            gamma *= 1.5
        
        if iter_count - last_improvement > patience_limit:
            break
    if len(h) == n:
        h_final = h
        c5_final = evaluate_c5(h_final, dx, n)
        return (h_final, c5_final, n)
    else:
        # Return default one if something unexpected happened
        h_final = np.zeros(n)
        h_final[:n2] = 1
        c5_final = evaluate_c5(h_final, dx, n)
        return (h_final, c5_final, n)
