import numpy as np
from scipy.optimize import minimize
import time

def project_onto_simplex_box(h, target_sum):
    """Project h onto the simplex with sum=target_sum and bounds [0,1]"""
    # Make a copy to avoid modifying input
    h = np.array(h, dtype=float)
    
    # Use a simple iterative approach:
    # First ensure sum is correct by scaling if needed
    h_sum = np.sum(h)
    if abs(h_sum - target_sum) > 1e-10:
        # Scale to get correct sum
        if h_sum > 0:
            h = h * target_sum / h_sum
    
    # Clip to [0,1] bounds
    h = np.clip(h, 0, 1)
    
    # Now adjust to enforce sum exactly using simple heuristic:
    # Reduce values and increase others to match target
    h_sum = np.sum(h)
    diff = target_sum - h_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Simple strategy: sort and adjust
    if diff > 0:
        # Need to increase some values
        # Add diff to smallest values (without exceeding 1)
        sorted_indices = np.argsort(h)
        for i in range(len(h)):
            idx = sorted_indices[i]
            if diff <= 0:
                break
            can_add = min(diff, 1.0 - h[idx])
            h[idx] += can_add
            diff -= can_add
    else:
        # Need to reduce some values
        diff = -diff
        sorted_indices = np.argsort(h)[::-1]  # Reverse sorted by descending
        for i in range(len(h)):
            idx = sorted_indices[i]
            if diff <= 0:
                break
            can_subtract = min(diff, h[idx])
            h[idx] -= can_subtract
            diff -= can_subtract
    
    return h

def evaluate(h):
    """Evaluate C5 for a given h"""
    dx = 2.0 / len(h)
    # Compute the correlation of h with 1-h using full mode
    # Result will be length 2*N - 1
    correlation = np.correlate(h, 1 - h, mode='full')
    # The maximum value is the one we want to minimize
    max_overlap = np.max(correlation) * dx
    return max_overlap

def optimize_step(h, lr=0.01, beta=1.0):
    """Perform one optimization step:
    1. Compute overlap integral for all shifts
    2. Find worst shift
    3. Compute gradient for that shift and update
    4. Project back onto feasible set
    """
    N = len(h)
    dx = 2.0 / N
    # Compute overlaps for all lags: S_ℓ = sum h_i * (1 - h_{i+ℓ})
    correlation = np.correlate(h, 1 - h, mode='full')
    # max is over the full array
    max_idx = np.argmax(correlation)
    max_value = correlation[max_idx]
    
    # Get lag index from max_idx
    lag = max_idx - (N - 1)  # Convert from 0-based index to actual lag
    
    # Compute gradient for the max shift
    grad = np.zeros(N)
    
    # Handle positive and negative lags correctly:
    # We iterate over indices i in range where both i and i+lag are valid:
    # i needs i ≥ 0 and i+lag < N, that is
    # i ≥ 0 and i < N - lag
    # i from max(0, 0) = 0 to min(N-1, N-1-lag)
    if lag >= 0:
        # Relevant indices for i: starting from 0 to N-1-lag
        start = 0
        end = N - lag
        for i in range(start, end):
            # Derivative w.r.t. h[i]: (1 - h[i+lag])
            grad[i] += 1 - h[i + lag]
            # Derivative w.r.t. h[i+lag]: -h[i]
            grad[i + lag] -= h[i]
    else:
        # Negative lag: equivalent to shift by negative amount
        lag = -lag  # Now make it positive
        # i from lag to N-1
        start = lag
        end = N
        for i in range(start, end):
            # i corresponds to i-lag being the first coordinate
            first_i = i - lag
            # Derivative w.r.t. h[first_i]: (1 - h[i])
            grad[first_i] += 1 - h[i]
            # Derivative w.r.t. h[i]: -h[first_i]
            grad[i] -= h[first_i]
    
    # Update h by gradient step
    h_new = h - lr * grad
    
    # Verify that h_new is still feasible
    # Apply projection and adjust sum to be exactly N/2
    h_proj = project_onto_simplex_box(h_new, N/2)
    
    # Now evaluate and return updated h and value
    return h_proj, evaluate(h_proj)
