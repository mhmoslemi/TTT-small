def project_onto_capped_simplex(y, target_sum):
    """Project y onto the capped simplex: {x ∈ [0,1]^n | sum(x) = target_sum}"""
    y = np.clip(y, 0, 1)
    sorted_y = np.sort(y)[::-1]
    cumulative = np.cumsum(sorted_y)
    n = len(y)
    # Find the largest index where (sorted_y[i] - (cumulative[i] - target_sum) / (i + 1)) > 0
    # This is a standard projection algorithm from Duchi et al. (2008)
    temp = sorted_y - (cumulative - target_sum) / np.arange(1, n + 1)
    idx = np.argmax(temp > 0)
    if temp[idx] > 0:
        lambda_star = (cumulative[idx] - target_sum) / (idx + 1)
        x = np.clip(y - lambda_star, 0, 1)
    else:
        x = np.clip(y, 0, 1)
    return x

def evaluate(h):
    """Return C5 as reported by the evaluator"""
    dx = 2.0 / len(h)
    result = np.correlate(h, 1 - h, mode="full")
    return np.max(result) * dx

def worst_shift_info(h):

    k_star = np.argmax(result) - (len(h) - 1)

    # The shift k* = k_star (from np.correlate) positions 
    # and that constant k_star = k_star (if k_star < 0)
    # means that i goes from 0 to N-1-|k_star|
    sign_k = 1 if k_star >= 0 else -1
    offset = abs(k_star)
    offending_indices = []
    if k_star >= 0:
        for i in range(len(h) - offset):

                offending_indices.append(i)
    else:
        for i in range(offset, len(h)):
            if h[i] == 1 and h[i - offset] == 0:
                offending_indices.append(i)
    return k_star, offending_indices

def gradient_of_max(x, beta):
    """Return gradient of log-sum-exp approximation to max_k O_k(x)"""
    N = len(x)
    # compute all O_k with correlations
    dx = 2.0 / N
    corr1 = np.correlate(x, 1 - x, mode="full")
    corr2 = np.correlate(1 - x, x, mode="full")
    shifts = np.arange(-(N - 1), N)
    # Correct indexing: O_k = sum_i x_i * (1 - x_{i+k}) for i from 0 to N-1-|k|
    # We can compute 
    O_values = np.zeros_like(shifts, dtype=float)
    for k_idx, k in enumerate(shifts):
        if k >= 0:
            # Sum over spaces i from 0 to N-1-k
            O_values[k_idx] = np.sum(x[:N-k] * (1 - x[k:]))
        else:
            # Sum over spaces i from -k to N-1 
            O_values[k_idx] = np.sum(x[-k:] * (1 - x[:-k]))
    # softmax weights: exp(beta * O_k) / sum exp(beta * O_j)
    weights = np.exp(beta * O_values)
    weights /= np.sum(weights)
    # gradient computation
    gradient = np.zeros_like(x)
    for k_idx, k in enumerate(shifts):
        if k >= 0:
            # O_k = sum_i x_i * (1 - x_{i+k}) 
            # dO_k/dx_i = (1 - x_{i+k}) * I[i < N-k] - x_{i-k} * I[k < i]
            # But this is for shift k where x_{i+k}
            # the full term: gradients are different: dO_k/dx_i = (1-x_{i+k}) 
            # if i < N-k and x_{i-k} if i-k >= 0, but with signs:
            # weight * [ (1 - x[i+k]) * I[i < N-k] - x[i-k] * I[i-k >= 0] ] 
            # = weight * [ gradient[ i ] + gradient[ i-k ] ] if k > 0
            temp_grad = np.zeros_like(x)
            if k < len(x):  # i < N-k
                temp_grad[:len(x)-k] += 1.0 - x[k:]  # the +1 part equivalent to derivative of
            # then
            if k < len(x):  # for i < N-k in sum_i x_i * (1 - x_{i+k})
                # Also, there is also derivative term x_{i-k} if i >= k:
                temp_grad[k:] -= x[:len(x)-k]  # - derivative of x_{i-k}
            # So, we are summing over k_idx weights[k_idx] * all[i]
            gradient += weights[k_idx] * temp_grad
        else:  # k < 0
            # reverse logic: O_k = x_{i-k} * (1 - x_{i}) for i from 0 to N-1+k 
            # dO_k/dx_i = (1-x_{i}) * I[i >= -k] - x_{i+k} * I[i+k < N]  
            temp_grad = np.zeros_like(x)          
            if -k <= len(x):  # i >= -k
                temp_grad[-k:] += 1.0 - x[:-k]  # from derivative of first term  
            if k + len(x) <= len(x):  # i+k < N
                temp_grad[:-k] -= x[k:]  # from derivative of second term
            gradient += weights[k_idx] * temp_grad
    return gradient

def swap_move(x, i, j):
    """swap x[i] and x[j] in binary vector"""
    x = x.copy()
    x[i], x[j] = x[j], x[i]
    return x

def fractional_move(x, i, j, delta):
    """Move delta from x[j] to x[i] respecting bounds [0,1]"""
    x = x.copy()
    x[i] += delta
    x[j] -= delta
    x[i] = max(0, min(1, x[i]))
    x[j] = max(0, min(1, x[j]))
    return x
