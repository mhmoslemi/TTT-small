import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import math

def evaluate_erdos_solution(h_values):
    """Compute C5 for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute the correlation integral
    corr = np.correlate(h_values, 1 - np.array(h_values), mode='full')
    # Take the maximum value and multiply by dx
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    """Return initial construction for h"""
    n = 70
    h = np.zeros(n)
    # Simple construction: step function
    # Fill first half with 0.5, second half with 0.5
    h[:n//2] = 1.0
    h[n//2:] = 0.0
    return h

def optimize_h_c5(h_start, n_points, max_time):
    """Optimize h to minimize C5"""
    start_time = time.time()
    
    # Define bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_vec):
        # Ensure the integral constraint
        integral = np.sum(h_vec) * (2.0 / n_points)
        if not abs(integral - 1.0) < 1e-6:
            # Penalty for violating the integral constraint
            penalty = 1000.0 * abs(integral - 1.0)
        else:
            penalty = 0.0
            
        # Evaluate C5
        c5 = evaluate_erdos_solution(h_vec)
        return c5 + penalty
    
    # Simple local search: random perturbations with constraint enforcement
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    def project_to_constraint(h_vec):
        """Project h_vec to satisfy integral constraint"""
        integral = np.sum(h_vec) * (2.0 / n_points)
        if abs(integral) < 1e-10:  # Prevent division by zero
            return h_vec
        scale_factor = 1.0 / integral
        new_h = h_vec * scale_factor
        # Ensure bounds [0,1]
        new_h = np.clip(new_h, 0, 1)
        return new_h
    
    # Gradient-free optimization with wrapper for constraint satisfaction 
    num_iterations = 100000
    iter_count = 0
    last_improvement = 0
    patience = 1000
    
    while iter_count < num_iterations and time.time() - start_time < max_time - 5:
        # Slightly mutate the current solution
        mutant = best_h.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.02)
        mutant[idx] += delta
        # Apply bounds
        mutant = np.clip(mutant, 0, 1)
        
        # Correct the constraint via projection
        mutant = project_to_constraint(mutant)
        
        # Evaluate
        new_c5 = objective(mutant)
        if new_c5 < best_c5:
            best_h = mutant
            best_c5 = new_c5
            last_improvement = 0
        else:
            last_improvement += 1
            if last_improvement > patience:
                break
                
        iter_count += 1
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    
    # Try several starting points
    n_points = 100  # Start with a manageable size
    trial_count = 0
    
    while time.time() - start_time < budget_s - 10:
        trial_count += 1
        
        # Different starting configurations
        if trial_count == 1:
            # Original construction
            h_start = initial_h_values()
        elif trial_count == 2:
            # Alternating construction
            h_start = np.ones(n_points) * 0.5
            for i in range(0, n_points, 2):
                h_start[i] = 1.0
        else:
            # Random start
            h_start = np.random.rand(n_points)
            integral = np.sum(h_start) * (2.0 / n_points)
            h_start = h_start / integral
            
        # Ensure the constraint is met
        h_start = h_start / np.sum(h_start) * (n_points / 2)
        h_start = np.clip(h_start, 0, 1)
            
        # Optimize
        try:
            h_opt, c5_opt = optimize_h_c5(h_start, n_points, budget_s - (time.time() - start_time))
            
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                
        except Exception as e:
            pass  # Skip this trial if error
    
    # Final evaluation
    if best_h is None:
        h_start = initial_h_values()
        best_h, best_c5 = optimize_h_c5(h_start, n_points, budget_s)
        
    # Further cleanup and constraint projection if needed
    integral = np.sum(best_h) * (2.0 / len(best_h))
    if abs(integral - 1.0) > 1e-6:
        best_h = best_h / integral * 1.0
    
    best_h = np.clip(best_h, 0, 1)
        
    return (list(best_h), best_c5, len(best_h))
