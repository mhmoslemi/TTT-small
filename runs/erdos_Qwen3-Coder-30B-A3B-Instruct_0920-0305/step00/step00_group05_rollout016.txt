from scipy.signal import correlate
import time

def evaluate_c5(h_vals, dx):
    """Evaluate C5 for given h values"""
    # Compute correlation
    corr = correlate(h_vals, 1 - np.array(h_vals), mode="full")
    # The maximum over all shifts
    max_corr = np.max(corr)
    return max_corr * dx
