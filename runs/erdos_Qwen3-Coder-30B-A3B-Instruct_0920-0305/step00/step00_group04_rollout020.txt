import numpy as np
import time
import random

def compute_c5_fast(h):
    n = len(h)
    size = 1 << ((2*n-1).bit_length())
    f_h = np.fft.fft(h, size)
    f_g = np.fft.fft(1 - h, size)
    corr = np.fft.ifft(f_h * np.conj(f_g)).real
    corr = corr[:2*n-1]
    dx = 2.0 / n
    max_val = np.max(corr) * dx
    return max_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 60  # Using 60 points for better resolution
    
    # Set parameters
    max_iter = 200000
    temp_start = 0.1
    temp_decay = 0.9995
    
    # Generate initial binary vector
    def random_binary_vector(n):
        h = np.zeros(n)
        ones_indices = np.random.choice(n, size=n//2, replace=False)
        h[ones_indices] = 1
        return h
    
    # Hill climbing with simulated annealing
    time_buffer = 0.2
    iteration = 0
    temp = temp_start
    
    while time.time() - start_time < budget_s - time_buffer and iteration < max_iter:
        # Create a new random binary vector
        h = random_binary_vector(n_points)
        current_c5 = compute_c5_fast(h)
        
        # If better solution, store it
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Now perform simulated annealing on the current solution
        current_h = h.copy()
        current_c5 = current_c5
        
        # Inner loop for SA
        inner_iter = 0
        while time.time() - start_time < budget_s - time_buffer and inner_iter < 10000:
            # Randomly select indices to swap
            ones = np.where(current_h > 0.5)[0]
            zeros = np.where(current_h < 0.5)[0]
            
            if len(ones) == 0 or len(zeros) == 0:
                break
                
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Swap values
            current_h[i], current_h[j] = current_h[j], current_h[i]
            
            # Compute new C5
            new_c5 = compute_c5_fast(current_h)
            
            # Accept or reject
            delta = new_c5 - current_c5
            if delta <= 0 or random.random() < np.exp(-delta / temp):
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
            else:
                # Revert swap
                current_h[i], current_h[j] = current_h[j], current_h[i]
            
            inner_iter += 1
            iteration += 1
            
            # Adjust temperature
            temp = max(temp_start * (temp_decay ** iteration), 1e-8)
    
    return (best_h, best_c5, n_points)
