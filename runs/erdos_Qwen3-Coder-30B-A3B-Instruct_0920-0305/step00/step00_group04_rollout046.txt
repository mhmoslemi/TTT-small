import numpy as np
import time
from scipy import fftpack
from scipy.optimize import minimize
from math import floor

# Precomputed constants
initial_h = np.array(initial_h_values)
n_initial = len(initial_h)
n_points_target = 60
dx_target = 2.0 / n_points_target
target_int = floor(0.3807 * n_points_target / 2)

def evaluate_C5(h):
    """Evaluate the maximum overlap integral for given h."""
    n = len(h)
    dx = 2.0 / n
    h_padded = np.pad(h, (0, n-1), mode='constant')
    g_padded = np.pad(1-h, (0, n-1), mode='constant')
    corr = fftpack.fft(h_padded) * np.conj(fftpack.fft(g_padded))
    corr = np.real(fftpack.ifft(corr))
    max_overlap = np.max(corr[:2*n-1]) * dx
    return max_overlap

def binary_search_projection(y, target_sum):
    """Project y onto the simplex with sum=target_sum using bisection."""
    y = np.array(y)
    low = np.min(y) - 1
    high = np.max(y) + 1
    tol = 1e-12
    max_iter = 100
    for _ in range(max_iter):
        mid = (low + high) / 2
        x = np.clip(y - mid, 0, 1)
        current_sum = np.sum(x)
        if current_sum < target_sum:
            high = mid
        else:
            low = mid
        if abs(current_sum - target_sum) < tol:
            break
    return np.clip(y - low, 0, 1)

def smooth_max_surrogate(h, gamma=5.0):
    """Compute the smooth maximum of overlaps with given gamma."""
    h = np.array(h)
    n = len(h)
    dx = 2.0 / n
    h_padded = np.pad(h, (0, n-1), mode='constant')
    g_padded = np.pad(1-h, (0, n-1), mode='constant')
    corr = fftpack.fft(h_padded) * np.conj(fftpack.fft(g_padded))
    corr = np.real(fftpack.ifft(corr))
    values = corr[:2*n-1] * dx
    exp_values = np.exp(gamma * values)
    Z = np.sum(exp_values)
    return np.log(Z) / gamma

def gradient_smooth_max(h, gamma=5.0):
    """Compute the gradient of smooth max surrogate."""
    h = np.array(h)
    n = len(h)
    dx = 2.0 / n
    # Compute correlation values with all lags
    h_padded = np.pad(h, (0, n-1), mode='constant')
    g_padded = np.pad(1-h, (0, n-1), mode='constant')
    corr_fft = fftpack.fft(h_padded) * np.conj(fftpack.fft(g_padded))
    corr = np.real(fftpack.ifft(corr_fft))
    values = corr[:2*n-1] * dx
    
    w = np.exp(gamma * values)
    w_sum = np.sum(w)
    if w_sum == 0:
        return np.zeros_like(h)
    w = w / w_sum
    
    # Construct the gradient via causal and anti-causal convolutions
    grad = np.zeros(n)
    # Forward contribution
    # h[i] * (1 - h[i + l]) for l >= 0 that is i + l < n => l < n - i
    for l in range(n):
        if l == 0:
            grad += w[l] * (1 - h)
        else:  # l > 0
            if l < n:
                h_slice = h[l:]
                grad[:n-l] += w[l] * (1 - h_slice)
            # Now for negative l
            if -l >= 0:
                h_slice = h[:-l]
                grad[-l:] += w[n-1+l] * (1 - h_slice)
        # negative l case is same in reversed form:
    # Backward contribution from opposite lags:
    for l in range(n):
        h_slice = h[l:]
        if l == 0:
            grad += w[n-1] * h
        else:
            if l < n:
                grad[l:] += w[n-1-l] * h_slice
    return grad

def inexact_gradient_descent_step(h, gamma=5.0, step_size=0.01):
    """Single gradient descent step."""
    grad = gradient_smooth_max(h, gamma)
    new_h = h - step_size * grad
    # Project to [0,1] and adjust sum
    new_h = np.clip(new_h, 0, 1)
    new_h = binary_search_projection(new_h, n_points_target / 2)
    return new_h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start = time.time()
    
    # Phase A: Binary Simulated Annealing
    # Initial binary vector
    h_original = np.array(initial_h)
    h_original = h_original * (n_points_target / n_initial)
    # Reinterpolate to target length
    h_interp = np.interp(np.linspace(0, 2, n_points_target), np.linspace(0, 2, n_initial), h_original)
    # Threshold for binary: convert to binary with exact count
    h_binary = np.zeros(n_points_target)
    x = np.argsort(h_interp)[::-1]  # descending order
    h_binary[x[:n_points_target//2]] = 1
    h_binary = h_binary.astype(float)
    
    best_h = h_binary.copy()
    best_C5 = evaluate_C5(best_h)
    best_int = int(best_C5 / dx_target)
    if best_int <= target_int:
        # Already good enough!
        return (best_h, best_C5, n_points_target)
        
    # Start optimization with binary search
    # For simplicity, using good binary seed
    h_current = h_binary.copy()
    current_C5 = best_C5
    
    time_phaseA = 0.7 * budget_s
    while time.time() - start < time_phaseA:
        # Apply random swaps
        swaps = 10
        for _ in range(swaps):
            ones = np.where(h_current == 1)[0]
            zeros = np.where(h_current == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i1 = np.random.choice(ones)
            i2 = np.random.choice(zeros)
            # Swap
            h_current[i1], h_current[i2] = h_current[i2], h_current[i1]
        new_C5 = evaluate_C5(h_current)
        if new_C5 < current_C5:
            current_C5 = new_C5
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_current.copy()
        if current_C5 <= 0.3807:
            break

    # Phase B: Continuous refinement
    if time.time() - start < budget_s - 5:
        # If we can try continuous refinement
        h_refined = best_h.copy()
        gamma = 5.0
        step_size = 0.01 * dx_target
        iters_no_improve = 0
        max_iters_no_improve = 10
        
        while time.time() - start < budget_s - 5 and iters_no_improve < max_iters_no_improve:
            h_prev = h_refined.copy()
            h_refined = inexact_gradient_descent_step(h_refined, gamma, step_size)
            new_C5 = evaluate_C5(h_refined)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h_refined.copy()
            else:
                iters_no_improve += 1
                step_size *= 0.9
                if step_size < 1e-8:
                    break
            gamma *= 1.1
            if gamma > 150:
                gamma = 150  # cap early
            
        # Final projection and evaluate
        best_h = binary_search_projection(best_h, n_points_target / 2)
        final_C5 = evaluate_C5(best_h)
        if final_C5 < best_C5:
            best_C5 = final_C5

    return (best_h, best_C5, n_points_target)
