def evaluate_erdos_solution(h):
    """Return the full C5 of the solution."""
    n = len(h)
    dx = 2.0 / n
    # Do correlation between h and (1 - h) = -h + 1
    c = np.correlate(h, 1 - h, mode='full')
    c = c * dx
    c5 = max(c)
    return c5

def project_onto_constraints(h, target_sum):
    """Project h onto [0,1] and sum = target_sum using water filling."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum

    if abs(diff) < 1e-10:
        return h

    if diff > 0:
        # Fill up
        idx = np.argsort(h)
        for i in range(len(h)-1, -1, -1):
            j = idx[i]
            if h[j] < 1.0:
                can_add = min(diff, 1.0 - h[j])
                h[j] += can_add
                diff -= can_add
                if abs(diff) < 1e-10:
                    break
    else:
        # Remove
        diff = -diff
        idx = np.argsort(h)
        for i in range(len(h)):
            j = idx[i]
            if h[j] > 0:
                can_subtract = min(diff, h[j])
                h[j] -= can_subtract
                diff -= can_subtract
                if abs(diff) < 1e-10:
                    break
    return h

def softmax(x):
    """Numerically stable softmax."""
    x = np.array(x)
    max_x = np.max(x)
    exp_x = np.exp(x - max_x)
    return exp_x / np.sum(exp_x)

def compute_correlations(h, dx):
    """Compute all c_k = dx * sum h[i] * (1 - h[i+k]) for k = -n+1,...,n-1."""
    n = len(h)
    h_rev = 1 - h
    # Use correlation of h with h_rev with zero padding
    c = correlate(h, h_rev, mode='full') * dx
    return c

def compute_gradient_of_c_k(h, k, dx):
    """Return the gradient ∇c_k(h) for fixed k."""
    n = len(h)
    grad = np.zeros(n)
    # Add terms (1-h_{j+k}) where valid
    if k >= 0:
        # grad[j] += dx * (1 - h[j+k])  for j in [0, ..., n-k-1]
        for j in range(n - k):
            grad[j] += dx * (1 - h[j + k])
    else:
        # k < 0
        # grad[j] += dx * (1 - h[j+k])  for j in [0, ..., n+k-1]
        for j in range(n + k):
            grad[j] += dx * (1 - h[j + k])

    # Subtract terms h_{j-k} where valid
    if k <= 0:
        # grad[j] -= dx * h[j-k]  for j in [-k, ..., n-1]
        # Use j = -k + i -> so i = j + k, j = i - k,  fixed i = 0,...n-1-k
        # But we're considering j in [0, ..., -k-1]
        # i in [k, ..., n-k-1]
        for j in range(n - (-k)):
            grad[j] -= dx * h[j - k]
    else:
        # k > 0
        # grad[j] -= dx * h[j-k] for j in [k, ..., n-1]
        for j in range(k, n):
            grad[j] -= dx * h[j - k]
    return grad

def computeLSE_gradient(weights, gradients):
    """Return the weighted sum of gradients."""
    return np.dot(weights, gradients)

def evaluate_loss(h, dx, beta):
    """Compute LSE at given h for given beta."""
    c = compute_correlations(h, dx)
    return (1.0 / beta) * log(np.sum(np.exp(beta * c)))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    # Initial parameters
    n = 180  # Target number of points, reasonable for performance/time
    dx = 2.0 / n
    target_sum = n / 2.0  # constrain sum(h) * dx = 1

    # Parameters for optimization
    beta_schedule = [0.5, 2.0, 5.0, 15.0, 40.0, 80.0]
    max_iter_per_beta = 150
    alpha_0 = 0.2

    def project_and_clamp(h):
        # Initialize with projection
        h = project_onto_constraints(h, target_sum)
        h = np.clip(h, 0, 1)
        return h

    # Set up initial points
    starts = []
    # Upscale initial construction
    h_up = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    starts.append(project_and_clamp(h_up))

    # Add some random starts
    for _ in range(5):
        h_random = np.random.rand(n)
        h_random = (h_random - h_random.mean()) * 0.5 + 0.5
        h_random = project_and_clamp(h_random)
        starts.append(h_random)

    # Optimizer loop
    best_h = None
    best_C5 = float('inf')

    for start_idx, h in enumerate(starts):
        if time.time() - t0 > 0.95 * budget_s:
            break
        h = h.copy()
        best_local = evaluate_erdos_solution(h)

        for beta in beta_schedule:
            if time.time() - t0 > 0.95 * budget_s:
                break
            for it in range(max_iter_per_beta):
                if time.time() - t0 > 0.95 * budget_s:
                    break

                # Evaluate current correlations
                c = compute_correlations(h, dx)
                # Evaluate LSE
                loss_current = evaluate_loss(h, dx, beta)

                # Compute per-k gradients
                n_lags = len(c)
                k_vals = np.arange(-n+1, n)
                gradients = []
                for k in k_vals:
                    grad = compute_gradient_of_c_k(h, k, dx)
                    gradients.append(grad)

                # Compute weights (soft max of beta * c)
                weights = softmax(beta * c)

                # Compute LSE gradient
                g = computeLSE_gradient(weights, gradients)

                # Line search
                alpha = alpha_0 / (1.0 + it)

                while True:
                    h_new = h - alpha * g
                    h_new = project_and_clamp(h_new)
                    c_new = compute_correlations(h_new, dx)
                    loss_new = evaluate_loss(h_new, dx, beta)
                    if loss_new <= loss_current:
                        break
                    alpha *= 0.5

                # Accept
                h = h_new
                loss_current = loss_new

                # Eval actual C5
                current_C5 = max(c_new)
                if current_C5 < best_local:
                    best_local = current_C5
                    best_h = h.copy()
                if current_C5 < best_C5:
                    best_C5 = current_C5

                # Early exit if already good
                if best_C5 <= 0.379:
                    break

    # Further refinement
    if time.time() - t0 < 0.95 * budget_s and best_h is not None:
        # Binary swap refinement: reset or iteration done
        best_h_copy = best_h.copy()
        # Target mass is still n/2
        current_sum = np.sum(best_h_copy)
        if abs(current_sum - target_sum) > 1e-8:
            # Re-project
            best_h_copy = project_and_clamp(best_h_copy)
            current_sum = np.sum(best_h_copy)

        # Convert to binary (0 or 1) preserving the sum
        binary_h = np.zeros_like(best_h_copy)
        ones_indices = np.argsort(best_h_copy)[::-1]  # descending order of values
        # Set the first `target_sum` indices to 1
        num_ones = int(np.round(target_sum))
        for i in range(num_ones):
            binary_h[ones_indices[i]] = 1.0

        # Then check if we can improve by swapping
        try:
            current_C5 = evaluate_erdos_solution(binary_h)
            if current_C5 < best_C5:
                best_h = binary_h.copy()
                best_C5 = current_C5

            # Local swaps
            best_h_refined = binary_h.copy()
            best_C5_refined = current_C5
            for _ in range(200):
                # Slightly randomize to avoid deadlocks
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i == j:
                    continue
                # Only swap if they are different
                if best_h_refined[i] != best_h_refined[j]:
                    # Swap
                    old_i = best_h_refined[i]
                    old_j = best_h_refined[j]
                    best_h_refined[i], best_h_refined[j] = best_h_refined[j], best_h_refined[i]
                    # Eval new C5
                    new_C5 = evaluate_erdos_solution(best_h_refined)
                    # If worse, revert, otherwise keep
                    if new_C5 >= best_C5_refined:
                        best_h_refined[i], best_h_refined[j] = old_i, old_j
                    else:
                        best_C5_refined = new_C5
                # Check if we want to break an early exit for here too
                if time.time() - t0 > 0.98 * budget_s:
                    break
            if best_C5_refined < best_C5:
                best_h = best_h_refined
                best_C5 = best_C5_refined
        except:
            pass  # continue with previous best

    # Final evaluation
    final_C5 = evaluate_erdos_solution(best_h)

    return (best_h, final_C5, n)
