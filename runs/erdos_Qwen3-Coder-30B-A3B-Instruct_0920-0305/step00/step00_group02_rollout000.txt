```python
import time
import numpy as np
import cvxpy as cp
import random
import math

def random_binary_vector(n):
    """Generate a random binary vector of length n with exactly n/2 ones."""
    if n % 2 != 0:
        raise ValueError("n must be even")
    h = np.zeros(n, dtype=int)
    ones_indices = np.random.choice(n, size=n//2, replace=False)
    h[ones_indices] = 1
    return h

def compute_c5(h):
    """Compute C5 value for given step function h."""
    dx = 2.0 / len(h)
    total = 1.0
    # Compute correlations with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum is the maximum of all shifts
    C5 = np.max(corr) * dx
    return C5

def solve_milp(n, time_limit=10):
    """Solve MILP formulation for small n."""
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == n // 2]
    
    # We linearize product h_i * h_{i+k} with w_{i,k}
    # For each shift k, define correlation k
    for k in range(1, n):  # start from 1 to exclude shift 0
        if n - k == 0:
            continue
        y = cp.Variable(n - k, nonneg=True)
        # y[i] = h[i] * h[i+k] for i=0..n-k-1
        constraints += [y <= h[:n-k]]
        constraints += [y <= h[k:]]
        constraints += [y >= h[:n-k] + h[k:] - 1]
        constraints += [cp.sum(y) >= t]
    
    # Additional constraint on t
    constraints += [t <= n // 2]
    
    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, max_seconds=time_limit)
        h_sol = np.round(h.value).astype(float)
        dx = 2.0 / n
        total = 1.0
        min_overlap = np.min([np.sum(h_sol[:n-k] * h_sol[k:]) for k in range(1, n)])
        C5 = total - min_overlap * dx
        return h_sol, C5
    except Exception as e:
        # fallback to heuristic if MILP fails
        h = random_binary_vector(n)
        C5 = compute_c5(h)
        return h, C5

def local_search(h_initial, max_iters, time_limit):
    """Perform local search with simulated annealing."""
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    cur = h_initial.copy()
    cur_c5 = best_c5
    start = time.time()
    T = 0.1
    
    for iteration in range(max_iters):
        if time.time() - start > time_limit:
            break
            
        ones_idx = np.where(cur == 1)[0]
        zeros_idx = np.where(cur == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            continue
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        
        # swap
        cur[i] = 0
        cur[j] = 1
        
        new_c5 = compute_c5(cur)
        if new_c5 < cur_c5:
            # accept improvement
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur.copy()
        else:
            # accept with probability
            if np.random.rand() < math.exp(-(new_c5 - cur_c5) / T):
                cur_c5 = new_c5
            else:
                # revert swap
                cur[i] = 1
                cur[j] = 0
        
        T *= 0.9999
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    n = 100  # Starting with a medium size to avoid solvers being too slow
    
    try:
        h_solved, c5_solved = solve_milp(n, time_limit=max(5, budget_s // 4))
    except Exception as e:
        h_solved = random_binary_vector(n)
        c5_solved = compute_c5(h_solved)
    
    elapsed = time.time() - start
    remaining = max(0, budget_s - elapsed)
    
    if remaining > 10:
        best_h, best_c5 = local_search(h_solved, 100000, remaining)
    else:
        best_h, best_c5 = h_solved, c5_solved
    
    return best_h.tolist(), float(best_c5), n
```
