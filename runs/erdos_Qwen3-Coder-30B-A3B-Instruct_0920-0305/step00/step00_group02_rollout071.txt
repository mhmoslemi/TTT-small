def random_binary_vector(n):
    h = np.zeros(n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    total = 1.0
    # Compute correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # The max value corresponds to the largest C5
    max_c5 = np.max(corr) * dx
    return max_c5

def local_search(start_h, max_iters=100000, time_limit=300):
    start = time.time()
    cur = start_h.copy()
    cur_c5 = compute_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    
    # Simple greedy approach: swap one 1 with one 0
    iteration = 0
    while time.time() - start < time_limit and iteration < max_iters:
        # Randomly pick a 1 and 0 to swap
        ones_idx = np.where(cur == 1)[0]
        zeros_idx = np.where(cur == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
            
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        
        # Temporary swap
        cur[i] = 0
        cur[j] = 1
        
        # Compute new C5
        new_c5 = compute_c5(cur)
        
        # Accept it if better or probabilistically
        if new_c5 < cur_c5:
            cur_c5 = new_c5
        else:
            # Revert swap
            cur[i] = 1
            cur[j] = 0
            
        # Update best
        if new_c5 < best_c5:
            best_h = cur.copy()
            best_c5 = new_c5
            
        iteration += 1
        
    return best_h, best_c5
