def evaluate_erdos_solution(h_values):
    """Evaluate the current solution and return C5 bound."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute the full correlation between h and (1-h)
    correlation = np.correlate(h_values, 1 - h_values, mode="full")
    # The maximum overlap is at the center
    max_overlap = np.max(correlation)
    # Multiply by dx to get C5 constant
    c5 = max_overlap * dx
    return c5
