import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the Erdős constant for given h values."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute correlation of h with (1-h)
    correlation = np.correlate(h_values, 1 - h_values, mode="full")
    # The maximum value among all lags gives C5
    c5 = np.max(correlation) * dx
    return c5

def initial_h_values():
    """Return initial construction for h(x) on [0,2]."""
    n = 42
    dx = 2.0 / n
    h = np.zeros(n)
    # Simple step function
    h[:n//2] = 1.0
    return h

def objective(h_values):
    """Objective function to minimize: -C5 (since we want to minimize C5)"""
    return -evaluate_erdos_solution(h_values)

def constraint_func(h_values):
    """Constraint: integral of h(x) = 1"""
    n_points = len(h_values)
    dx = 2.0 / n_points
    return 1.0 - np.sum(h_values) * dx
