import numpy as np
import time
import math
from scipy.optimize import minimize
from scipy.special import softmax

def project_onto_capped_simplex(y, target_sum):
    """Project y onto the capped simplex: {x in [0,1]^N | sum x = target_sum}"""
    y = np.clip(y, 0, 1)
    s = np.sum(y)
    if s == target_sum:
        return y
    elif s < target_sum:
        # Need to add more to reach target_sum
        # All entries are already in [0,1], so we can add directly
        delta = (target_sum - s) / len(y)
        y = np.clip(y + delta, 0, 1)
        return y
    else:
        # sort elements and find threshold
        sorted_y = np.sort(y)[::-1]
        cumsum = np.cumsum(sorted_y)
        # Find threshold lambda using Duchi et al. 2008
        # We're looking for the largest index such that sorted_y[i] - (target_sum - cumsum[i]) / (i+1) >= 0.
        # We compute this implicitly.
        diff = target_sum - cumsum
        threshold_indices = np.where(diff <= 0)[0]
        if len(threshold_indices) == 0:
            # all elements are greater than lambda
            # subtract from all
            delta = (s - target_sum) / len(y)
            return np.clip(y - delta, 0, 1)
        else:
            # Find the largest valid index
            i_star = threshold_indices[-1]
            if i_star == len(y) - 1:
                # all elements are greater than lambda
                delta = (s - target_sum) / len(y)
                return np.clip(y - delta, 0, 1)
            else:
                # There is a boundary: compute lambda and subtract
                # Lambda is approximately (s - target_sum) / (i_star + 1)
                # Involve correct constraint
                lambda_val = (s - target_sum) / (i_star + 1)
                y_clipped = np.clip(y - lambda_val, 0, 1)
                return y_clipped

def evaluate(x):
    """Returns C5 = max_k integral of h(x) * (1 - h(x+k)) dx"""
    # Correlation (full mode) computes overlaps for k from -len+1 to len-1
    # For each k, O_k = sum x_i * (1 - x_{i+k})
    # The correlation length handling is implicit. We compute the full correlation with mode='full'
    # This will give us correlation coefficients for shifts k from -L to L where L = len(x) - 1
    dx = 2.0 / len(x)
    corr = np.correlate(x, 1 - x, mode="full")
    # Take only non-negative shifts (k from 0 to len-1)
    max_corr = np.max(corr[len(x)-1:])
    return max_corr * dx

def swap_move(x, i, j):
    """swap elements at i and j (binary vectors only)"""
    x = x.copy()
    x[i], x[j] = x[j], x[i]
    return x

def fractional_move(x, i, j, delta):
    """Move delta amount from j to i (ensuring bounds)"""
    x = x.copy()
    if delta > 0:
        # Transfer delta from j to i
        actual_delta = min(delta, x[j])
        x[i] += actual_delta
        x[j] -= actual_delta
    else:
        actual_delta = min(-delta, 1 - x[i])
        x[i] += actual_delta
        x[j] -= actual_delta
    return x

def gradient_of_max(x, beta):
    """returns gradient of smooth surrogate S(x;beta)"""
    n = len(x)
    corr_indices = np.arange(2*n - 1) - (n - 1)
    opt_k = []
    opt_g = np.zeros(n)
    
    for i, k in enumerate(corr_indices):
        if abs(k) >= n:
            continue
        if k >= 0:
            # sum x_i * (1 - x_{i+k})
            # grad w.r.t. x_i = (1 - x_{i+k}) * I[i <= n-1-k] - x_{i-k} * I[i >= k]
            corr_v = np.sum(x[:n-k] * (1 - x[k:]))
            g = np.zeros(n)
            # Part of gradient from first term (1 - x_{i+k})
            if k == 0:
                g[:n] = 1 - x[:n]
            else:
                g[:-k] += 1 - x[k:]
                # Part from -x_{i-k}
                if k > 0:
                    g[k:] -= x[:-k]
        else:  # k < 0
            # negative shift
            corr_v = np.sum(x[-k:] * (1 - x[:-k]))
            # Compute gradient w.r.t. positions
            g = np.zeros(n)
            g[-k:] += 1 - x[:-k]
            g[:-k] -= x[-k:]
        
        g = np.clip(g, -1e6, 1e6)
        opt_k.append((corr_v, g))
    
    if not opt_k:
        return np.zeros_like(x)
    # softmax weights
    vals = [v[0] for v in opt_k]
    weights = softmax(np.array(vals) * beta)
    grad = np.sum([weights[i] * opt_k[i][1] for i in range(len(opt_k))], axis=0)
    return grad

def worst_shift_info(x):
    """returns the shift k* and indices where (x_i=1 and x_{i+k}=0) """
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    # largest shift index
    mid = len(corr) // 2
    max_shift_idx = np.argmax(corr[mid:]) + mid  # corresponding to k = 0, ..., n-1 (shifts)
    max_k = max_shift_idx - mid  # convert to actual shift value between -n+1 to n-1
    max_val = corr[max_shift_idx]
    
    # Find indices where we have 1 followed by 0 at that distance
    # In case of k>0, it is x_i * (1 - x_{i+k})
    # So x_i = 1, x_{i+k} = 0 
    if max_k > 0:
        #
        off_indices = np.where((x[:-max_k] == 1) & (x[max_k:] == 0))[0]
        return max_k, off_indices.tolist()
    elif max_k < 0:
        #
        off_indices = np.where((x[-max_k:] == 1) & (x[:-max_k] == 0))[0]
        return max_k, off_indices.tolist()
    else:
        # When k = 0, x_i * (1 - x_i) = 0 if x = 0 or 1
        # There are no relevant correlations
        off_indices = np.where((x == 1) & (x == 0))[0]  # Empty
        return max_k, off_indices.tolist()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    np.seterr(divide='ignore', invalid='ignore')
    t_start = time.time()
    t_dead = t_start + budget_s - 5.0  # leave some safety margin
    
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    candidate_Ns = [140, 200, 260]
    
    for N in candidate_Ns:
        print("Optimizing for N=", N)
        # Initialize solution with initial_h_values or random binary
        if 'initial_h_values' in globals() and len(initial_h_values) == N:
            h = np.array(initial_h_values)
        else:
            h = np.zeros(N, dtype=int)
            one_indices = np.random.choice(N, N//2, replace=False)
            h[one_indices] = 1
            
        dx = 2.0 / N
        h = h.astype(float)
        h = h * (N/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        if not (abs(np.sum(h) - N/2) < 1e-6):
            # If not compliant, re-project
            h = project_onto_capped_simplex(h, N/2)
        
        # Phase 2.1: Binary Simulated Annealing
        current_t = time.time()
        temp = 0.12
        T0 = 0.12
        t_max_SA = 30000  # number of SA iterations per run
        best_SA = float('inf')
        best_SA_binary = h.copy()
        
        # Phase 2.2: Local refinement based on worst shift
        # Sets are computed in sequence, loop to look for binary solutions with dynamic improvements
        # Phase 2.3: Continuous optimization (after SA or local refinement)
        
        # Phase 2.1: SA binary optimization
        count = 0
        while time.time() < t_dead and count < t_max_SA:
            # Sample two indices that are 1 and 0 respectively
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue
            
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Perform swap and compute new value
            h_prop = swap_move(h, i, j)
            c5_new = evaluate(h_prop)
            
            c5_old = evaluate(h)
            
            delta = c5_new - c5_old
            if delta <= 0 or np.random.rand() < np.exp(-delta / temp):
                h = h_prop
                if c5_new < best_SA:
                    best_SA = c5_new
                    best_SA_binary = h.copy()
            
            count += 1
            temp = T0 * (1 - count / t_max_SA)
            if count % 1000 == 0:
                # Check time
                if time.time() > t_dead:
                    break
        
        h = best_SA_binary.copy()
        # Phase 2.2: Directed local improvement on the binary solution
        reached = 0
        while time.time() < t_dead and reached < 1000:
            k_star, off_indices = worst_shift_info(h)
            if len(off_indices) == 0 or abs(k_star) == 0:
                break
            
            i = np.random.choice(off_indices)
            j = np.random.choice(np.where(h == 0)[0])  # not in {i, i+k*}
            if abs(i - j) <= abs(k_star):
                # This makes k_star, so find another j
                valid_j = np.where((h == 0) & (np.abs(np.arange(N) - j) > abs(k_star)))[0]
                if len(valid_j) > 0:
                    j = np.random.choice(valid_j)
                else:
                    continue
                
            h_prop = swap_move(h, i, j)
            c5_new = evaluate(h_prop)
            c5_old = evaluate(h)
            if c5_new <= c5_old:
                h = h_prop
                reached += 1
            else:
                reached += 1  # still move on
            if time.time() > t_dead:
                break

        # Phase 3: Continuous refinement
        h = project_onto_capped_simplex(h, N/2)
        beta = 5.0
        alpha = 0.02
        max_cont_iter = 1000
        cont_iter = 0
        while time.time() < t_dead and cont_iter < max_cont_iter:
            if beta > 500:
                break
            g = gradient_of_max(h, beta)
            h_prop = h - alpha * g
            # manhattan projection
            h_prop = project_onto_capped_simplex(h_prop, N/2)
            c5_prop = evaluate(h_prop)
            c5_old = evaluate(h)
            if c5_prop < c5_old or np.random.rand() < 0.1:
                h = h_prop
            cont_iter += 1
            beta *= 1.02
            if time.time() > t_dead:
                break
        
        # Final evaluation
        final_c5 = evaluate(h)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h.copy()
            best_n = N
            
        if time.time() > t_dead:
            break
        
    return (best_h.tolist(), best_c5, best_n)
