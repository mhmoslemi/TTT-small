import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

def project_onto_simplex_box(h, target_sum):
    """Project h onto the simplex with sum = target_sum and bounds [0,1]"""
    # First clip to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to hit target_sum
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple approach: rescale to get exact sum
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.full_like(h, target_sum / len(h))
    # Re-clip
    h = np.clip(h, 0, 1)
    # Final adjustment to ensure sum exactly
    h_sum = np.sum(h)
    diff = target_sum - h_sum
    if abs(diff) > 1e-10:
        # Add a small amount to the largest entries
        idx = np.argsort(h)[-min(len(h), max(1, int(abs(diff) * len(h))//0.1)):]
        h[idx] = np.clip(h[idx] + diff / len(idx), 0, 1)
        
    return h

def evaluate_erdos_solution(h):
    """Evaluate the solution h for Erdos problem"""
    # Note: h is not required to be normalized; we handle it internally
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def gradient_of_max(h, max_idx):
    """Compute gradient of max S_{ℓ} with respect to h"""
    N = len(h)
    l = max_idx - (N - 1)  # get actual shift value
    grad = np.zeros(N)
    
    if l >= 0:
        # i ranges from 0 to N-l-1
        for i in range(N - l):
            # first term: d/dh_i(h_i * (1 - h_{i+l})) = 1 - h_{i+l}
            grad[i] += 1 - h[i + l]
            # second term: d/dh_{i+l}(h_i * (1 - h_{i+l})) = -h_i
            grad[i + l] -= h[i]
    else:
        # l < 0, i ranging from -l to N-1
        l_pos = -l  # l_pos > 0
        for i in range(N - l_pos):
            idx1 = i
            idx2 = i + l_pos  # i + l = i - l_pos
            grad[idx1] += 1 - h[idx2]
            grad[idx2] -= h[idx1]
    
    return grad
