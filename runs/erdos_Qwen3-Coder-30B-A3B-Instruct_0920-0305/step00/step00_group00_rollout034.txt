import numpy as np
import time
from scipy.special import softmax

def project_to_feasible(h, target_sum):
    """Project h onto the set {0 <= h[i] <= 1, sum(h) = target_sum}"""
    # Binary search for the right lambda
    lower, upper = -1.0, 2.0
    for _ in range(60):  # Conservative bound
        mid = (lower + upper) / 2.0
        trial = np.clip(h + mid, 0, 1)
        trial_sum = np.sum(trial)
        if trial_sum < target_sum:
            upper = mid
        else:
            lower = mid
    lambda_val = (lower + upper) / 2.0
    result = np.clip(h + lambda_val, 0, 1)
    
    # Ensure correct sum by adjusting free components
    actual_sum = np.sum(result)
    diff = target_sum - actual_sum
    if abs(diff) > 1e-10:
        # Find indices not at bounds
        free_indices = []
        for i in range(len(result)):
            if not (abs(result[i] - 0) < 1e-10 or abs(result[i] - 1) < 1e-10):
                free_indices.append(i)
        # Distribute diff among free indices (or not if all are at bounds)
        if len(free_indices) > 0:
            to_add = diff / len(free_indices)
            for i in free_indices:
                result[i] += to_add
                result[i] = np.clip(result[i], 0, 1)
    
    # Clamp final result
    result = np.clip(result, 0, 1)
    return result

def compute_correlation(h, dx):
    """Compute all correlations c_k for all shifts k."""
    p = 1 - h
    # Full correlation of h with p (so we will compute h[i] * (1-h[i+k]))
    full_corr = np.correlate(h, p, mode='full') * dx
    # The valid correlations are from -n+1 to n-1, but we only care about 0 to 2n-2
    # Actually, the effective shifts for the integral along [0,2] with n points
    # are shifted to align properly. We take endpoints.
    # The full correlation is of length 2*n-1 for symmetric 1-d correlation (symmetric)
    # But we need max_k c_k for k=0 to n-1, actually from k=0 to 2*n-2, 
    # but in valid integral range 0 to 2
    # However,   max_k c_k  where c_k = dx * sum(h[i]*h[i+k])  isn't  
    # correct, since it's integration over domain [0,2], forming the shifted convolutions.
    # Let's reflect the approach from strategy:
    # For the shift k, c_k(h) = dx * sum h[i]*(1-h[i+k]) 
    # = dx * sum h[i] - dx*sum h[i] h[i+k]
    # Only wrong, we're using h instead of h[i+k] offset - we must compute it as
    # the correlation of h with (1-h). But we use "forward" conv and/or defined in the domain
    # Treat as correlation with 'p'; need full convolution 
    # Another reflection: Let's compute the right way:
    # Integrate h(x) * (1 - h(x + k)) (x in [0,2])
    # Discretize and treat it similarly to autocorrelations but of h with (1-h)
    # As defined in problem:
    # c[k] = dx * sum h[i] * (1 - h[i+k]) 
    # Summing over all i, starting from k=0 to max valid, gives different indexing
    # For n points, we have [0, 2] divided into n intervals, so stride dx = 2/n
    # E.g., at k=0: h[0]*1 + h[1]*1 + ... 
    # At k=1: h[0]* (1-h[1]) + h[1] (1-h[2]) + ...
    # We will compute this directly via shifting slices
    # Full shift (i.e. convolution with flip, with zero padding as necessary)
    # But this gives h(i) * (1-h(i-k)) which is not what we want
    # We want h(i) * (1 - h(i+k)) over full domain.
    # We compute c[k] = dx * sum over i of h[i] * (1 - h[i+k])  
    # For k=0: h[0]*(1-h[0]) + h[1]*(1-h[1]) + ...
    # For k=1: h[0]*(1-h[1]) + h[1]*(1-h[2]) + ...  
    # ...
    # 0 <= k <= n-1 (n possible shifts with conv on full domain)
    # But we need only the non-trivial result of the full convolution over range supported,
    # which is of length [0, 2n-2], and [0, n-1] and [n-1, 2n-2] are mirrored
    # But for simplicity, let's just use a reliable method using loops or numpy sliding window.
    # This is recoverable via the full correlation of h with (1-h) interpreted correctly.
    # strlen = len(h) = n, so we'll compute full conv. Then take correct offsets. 
    # One way: use 1 - h as the second, conv it with h with full mode.
    # Example for 3 points: h[0] h[1] h[2]  
    # Full should start at shift -2 up to shift 2 (length 5)
    #   k=-2: h[0]*(1-h[2]) etc
    #   k=-1: h[0]*(1-h[1]) + h[1]*(1-h[2]) etc
    #   k=0: h[0]*1 + h[1]*1 + h[2]*1 etc
    #   k=1: h[0]*(1-h[1]) + h[1]*(1-h[2]) etc 
    #   k=2: h[0]*(1-h[2]) etc
    # Note that the sequence slice indices must be correct per shift.
    # The trick is to compute all of them in one operation using np.correlate
    # But no, we need diag of shifted products.
    # So we compute for each k from 0 to n-1:
    #  sum h[i] * (1 - h[i+k]) for valid i 
    # This sounds like dot(h, 1-h[k:]) if k>0, and same with reverse-going.
    # Python indexing:
    # full_output is 2*n-1 in length
    p = 1 - h
    corr = np.correlate(h, p, mode='full')  # length=2*n-1
    # Now we map element corr[n-1+k] to shift k
    corr_dx = corr * dx
    # We only care about non-negative shifts, in [0, n-1], indices from n-1 to 2*(n-1)
    # Correctly identified segments:
    k_values = np.arange(0, n)  # from 0 to n-1, shifts
    c = corr_dx[n-1:n-1+k_values.shape[0]] 
    return c

def compute_gradient_of_correlation(h, k, dx):
    """Get gradient of c_k = dx * sum(h[i] * (1 - h[i+k])) wrt h_j"""
    # df/dh_j = dx * [(1-h_j+k)*1_{j+k < n} - h_j-k * 1_{j-k < n}]
    # This is a compact way to express the partial derivative w.r.t h_j
    g = np.zeros_like(h)
    # Add contribution from (1-h_{j+k}) part  
    if k < len(h):
        # For indices where j+k < n => j < n-k
        j_indices = np.arange(len(h) - k)
        g[j_indices] += dx * (1 - h[k + j_indices])
    # Subtract contribution from h_{j-k}
    if k > 0:
        # For indices where j-k >= 0 => j >= k
        j_indices = np.arange(k, len(h))
        g[j_indices] -= dx * h[j_indices - k]
    return g

def compute_gradients_all(h, dx):
    """Return the full gradient matrix - [[grad_c_k(h)] for k in range(n)]"""
    n = len(h)
    # Very fast, but memory-heavy; let’s do in turn instead for now.
    return np.array([compute_gradient_of_correlation(h, k, dx) for k in range(n)])

def compute_gradient_lse(h, c_vals, beta, dx):
    """Weighted average of gradients, use softmax to get weights w"""
    # c_vals = [c_k(h) for all k] 
    # Weight each gradient by the softmax of beta*c_k
    w = softmax(beta * c_vals)
    # Compute weighted sum of gradients grad(c_k) = dx* [1-h_{j+k} - h_{j-k}]
    # Simplified directly with full conv
    all_grads = np.array([compute_gradient_of_correlation(h, k, dx) for k in range(len(h))])
    weighted_grad = np.average(all_grads, axis=0, weights=w)  # could perhaps optimize
    return weighted_grad

def evaluate_loss_lse(h, beta, c_vals, dx):
    """Smooth loss: L_β(h) = (1/β)*log(sum(exp(β * c_k)))"""
    return (1/beta) * np.log(np.sum(np.exp(beta * c_vals)))

def backtracking_line_search(h, grad, c_vals, beta, dx, alpha_init=0.2):
    """Backtracking line search for projected update"""

            break
        alpha = alpha * 0.5
