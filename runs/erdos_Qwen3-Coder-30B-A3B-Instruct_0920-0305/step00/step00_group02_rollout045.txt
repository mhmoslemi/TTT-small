def decode(z, M):
    """Decode raw parameters to interval lengths and gaps."""
    z_ell = z[:M]
    z_g = z[M:]
    ell = np.exp(z_ell)
    g = np.exp(z_g)
    ell = ell / np.sum(ell)
    g = g / np.sum(g)
    return ell, g

def build_intervals(ell, g):
    """Construct intervals [a_i, a_i + ell_i] from lengths and gaps."""
    a = np.zeros(len(ell) + 1)
    a[0] = g[0]
    for i in range(len(ell)):
        a[i+1] = a[i] + ell[i] + g[i+1]
    return a

def candidate_shifts(a, b):
    """Generate all candidate shift values."""
    E = list(a) + list(b)
    T = set([0.0])
    for i in range(len(E)):
        for j in range(len(E)):
            diff = abs(E[i] - E[j])
            if diff <= 2:
                T.add(diff)
    return sorted(list(T))

def intersection_length(t, a, b):
    """Compute intersection length of S and S+t."""
    total = 0.0
    for i in range(len(a)):
        # Interval [a[i], b[i]]
        # Shifted to [a[i] + t, b[i] + t] or [a[i] - t, b[i] - t]
        if t >= 0:
            start = max(a[i], a[i] + t)
            end = min(b[i], b[i] + t)
        else:
            start = max(a[i], a[i] - t)
            end = min(b[i], b[i] - t)
        if end > start:
            total += end - start
    return total

def objective(z):
    """Evaluate objective C5 from raw parameters."""
    ell, g = decode(z, M)
    a = build_intervals(ell, g)
    b = a + ell
    shifts = candidate_shifts(a, b)
    intersections = [intersection_length(t, a, b) for t in shifts]
    min_inter = min(intersections)
    return 1 - min_inter

def run_DE(budget_s, seed=42):

    start = time.time()

def local_LP(z_best, max_iter=5):
    """Refine solution using local LP-based adjustment."""
    ell, g = decode(z_best, M)
    a = build_intervals(ell, g)
    b = a + ell
    
    for _ in range(max_iter):
        shifts = candidate_shifts(a, b)
        intersections = [intersection_length(t, a, b) for t in shifts]
        min_val = min(intersections)
        active = [s for i, s in enumerate(shifts) if abs(intersections[i] - min_val) < 1e-10]
        
        if not active:
            break
            
        # Set up LP for adjustment
        a_vars = [Variable(1) for _ in range(len(a))]
        b_vars = [Variable(1) for _ in range(len(b))]
        
        # Create constraints
        # Minimize tau subject to I(t) >= tau for all active t
        
        obj = Minimize(Variable(1))
        constraints = []
        tau = Variable(1)
        
        # Build intersection expressions
        for t in active:
            term = 0
            for i in range(len(a_vars)):
                # Overlap of [a[i], b[i]] with [a[i]+t, b[i]+t]
                start = max(a[i], a[i] + t)
                end = min(b[i], b[i] + t)
                if end > start:
                    term += end - start
                else:
                    term += 0
            constraints.append(term >= tau)
            
        prob = Problem(obj, constraints)
        try:
            prob.solve(solver="ECOS", verbose=False)
            if prob.status == "optimal":
                # Update a and b
                a_new = [float(a[i] + 0) for i in range(len(a))]
                b_new = [float(b[i] + 0) for i in range(len(b))]
                break  # Assuming fixed for case
        except:
            pass
            
    return z_best

def gridify(intervals, n):
    """Convert intervals to grid step function."""
    dx = 2.0 / n
    h = np.zeros(n)
    a = intervals[0]
    ell = intervals[1]
    for i in range(len(a)-1):
        start = int(np.floor((a[i] - 0) / dx))
        end = int(np.ceil((a[i] + ell[i] - 0) / dx))
        if start < 0: start = 0
        if end > n: end = n
        h[start:end] = 1
    h = h.astype(float)
    sum_h = np.sum(h)
    target = n / 2
    if abs(sum_h - target) > 1e-6:
        if sum_h < target:
            # Add ones
            for i in range(n):
                if h[i] == 0:
                    h[i] = 1
                    sum_h += 1
                    if abs(sum_h - target) < 1e-6:
                        break
        else:
            # Remove ones
            for i in range(n):
                if h[i] == 1:
                    h[i] = 0
                    sum_h -= 1
                    if abs(sum_h - target) < 1e-6:
                        break
    return h

def binary_hill_climb(h, time_left):
    """Fast hill-climbing swap optimization."""
    dx = 2.0 / len(h)
    start_time = time.time()
    best_h = h.copy()
    best_C = float("inf")
    while time.time() - start_time < time_left - 0.1:
        i = np.random.randint(len(h))
        j = np.random.randint(len(h))
        if h[i] == 1 and h[j] == 0:
            h[i] = 0
            h[j] = 1
            c5 = np.max(np.correlate(h, 1-h, mode="full")) * dx
            if c5 < best_C:
                best_C = c5
                best_h = h.copy()
            else:
                h[i] = 1
                h[j] = 0
        elif h[i] == 0 and h[j] == 1:
            h[i] = 1
            h[j] = 0
            c5 = np.max(np.correlate(h, 1-h, mode="full")) * dx
            if c5 < best_C:
                best_C = c5
                best_h = h.copy()
            else:
                h[i] = 0
                h[j] = 1
    return best_h, best_C

    return h, c5_bound, n
