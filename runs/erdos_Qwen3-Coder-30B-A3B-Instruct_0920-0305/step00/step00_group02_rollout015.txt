def decode(z):
    """Decode raw vector z into interval lengths ℓ and gaps g with softmax normalization"""
    M = len(z) // 2
    z_ell = z[:M]
    z_g = z[M:]
    ell = np.exp(z_ell) / np.sum(np.exp(z_ell))
    g = np.exp(z_g) / np.sum(np.exp(z_g))
    return ell, g

def build_intervals(ell, g):
    """Build interval start and end points from lengths and gaps"""
    M = len(ell)
    a = np.zeros(M)
    a[0] = g[0]
    for i in range(1, M):
        a[i] = a[i-1] + ell[i-1] + g[i]
    b = a + ell
    return a, b

def candidate_shifts(a, b):
    """Generate all possible shift values that can make an endpoint align with another"""
    E = list(a) + list(b)
    E = [x for x in E if 0 <= x <= 2]
    T = {0}
    for i in range(len(E)):
        for j in range(len(E)):
            diff = abs(E[i]-E[j])
            if diff <= 2:
                T.add(diff)
    return sorted(list(T))

def intersection_length(t, a, b):
    """Compute intersection length of S and S+t for a given t"""
    total = 0.0
    for i in range(len(a)):
        start1, end1 = a[i], b[i]
        start2, end2 = start1 - t, end1 - t
        if t < 0:
            start2, end2 = start1 + t, end1 + t
        
        overlap_start = max(start1, start2)
        overlap_end = min(end1, end2)
        if overlap_end > overlap_start:
            total += overlap_end - overlap_start
    return total

def objective(z):
    """Evaluate objective function C5 for a given parameter vector z"""
    ell, g = decode(z)
    a, b = build_intervals(ell, g)
    
    shifts = candidate_shifts(a, b)
    min_I = float('inf')
    for t in shifts:
        I_t = intersection_length(t, a, b)
        if I_t < min_I:
            min_I = I_t
    
    return 1 - min_I

def gridify(intervals, n):
    """Convert intervals to a grid step function h"""
    h_grid = np.zeros(n)
    dx = 2.0 / n
    grid_centers = np.array([i*dx + dx/2 for i in range(n)])
    
    a, ell = intervals
    for i, (start, l) in enumerate(zip(a, ell)):
        for j, center in enumerate(grid_centers):
            if start <= center <= start + l:
                h_grid[j] = 1.0
    
    # Adjust for exact constraint: sum must be n/2
    sum_h = np.sum(h_grid)
    if abs(sum_h - n/2) > 1:
        raise ValueError("Sum too far off")
    while sum_h > n/2:
        # Flip smallest number of ones to zero
        grid_indices = np.where(h_grid == 1)[0]
        if len(grid_indices) > 0:
            idx = grid_indices[0]  # Arbitrarily reduce
            h_grid[idx] = 0.0
            sum_h -= 1
    while sum_h < n/2:
        # Flip smallest number of zero to one
        grid_indices = np.where(h_grid == 0)[0]
        if len(grid_indices) > 0:
            idx = grid_indices[0]  # Possibility to flip arbitrarily
            h_grid[idx] = 1.0
            sum_h += 1
    
    return h_grid

def binary_hill_climb(h, budget_s):
    """Fast hill-climbing to refine grid"""
    start_time = time.time()
    dx = 2.0 / len(h)
    current_C5 = np.correlate(h, 1-h, mode="full")[len(h)-1:] @ dx
    
    best_h = h.copy()
    best_C5 = current_C5
    iteration = 0
    
    while time.time() - start_time < budget_s:
        # Pick two indices
        i, j = np.random.randint(0, len(h), size=2)
        if i == j:
            continue
        if h[i] == h[j]:
            continue
            
        # Swap i and j values
        h[i], h[j] = h[j], h[i]
        
        # Compute change in C5
        test_C5 = np.correlate(h, 1-h, mode="full")[len(h)-1:] @ dx
        
        if test_C5 < current_C5:
            current_C5 = test_C5
            if test_C5 < best_C5:
                best_C5 = test_C5
                best_h = h.copy()
        else:
            # Undo swap
            h[i], h[j] = h[j], h[i]
        iteration += 1
        
        # Give a sense of progress
        if iteration % 100 == 0:
            # Periodically check if still inside budget
            if time.time() - start_time > budget_s:
                break
            
    return best_h, best_C5

    return h_final, final_c5, n
