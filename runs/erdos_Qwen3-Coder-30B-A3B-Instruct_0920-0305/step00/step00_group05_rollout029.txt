import numpy as np
import cvxpy as cp
import time
import math

def full_overlap(x):
    return np.correlate(x, 1 - x, mode='full')

def worst_shift(x):
    overlaps = full_overlap(x)
    k_max = np.argmax(overlaps)
    return (k_max, overlaps[k_max])

def build_mip(N, active_shifts):
    x = cp.Variable(N, boolean=True)
    t = cp.Variable(integer=True)
    
    # Symmetry breaking
    constraints = [x[0] == 1]
    
    # Half-ones constraint
    constraints.append(cp.sum(x) == N // 2)
    
    # Auxiliary variables wij,k for active shifts (i,k)
    w = {}
    for k in active_shifts:
        for i in range(N - k):
            w[i, k] = cp.Variable(boolean=True)
            constraints.append(w[i, k] <= x[i])
            constraints.append(w[i, k] <= x[i + k])
            constraints.append(w[i, k] >= x[i] + x[i + k] - 1)
    
    # Overlap constraints
    for k in active_shifts:
        left_term = cp.sum(x[:N - k])
        right_term = cp.sum([w[i, k] for i in range(N - k)])
        constraints.append(left_term - right_term <= t)
    
    # Objective
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    return prob, (x, t, w)

def solve_mip(prob, vars_dict, time_limit=30):
    x, t, w = vars_dict
    try:
        prob.solve(solver=cp.CBC, time_limit=time_limit, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return (x.value.astype(int), t.value)
        else:
            # Return a feasible but possibly suboptimal solution
            x_val = x.value
            return (np.round(x_val).astype(int), t.value)
    except Exception:
        # Fallback: return integer approximation and residual value
        x_val = np.round(x.value).astype(int)
        return (x_val, t.value)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Use a reasonably large N for a better solution
    N = 120
    dx = 2.0 / N
    
    # Start timer
    start_time = time.time()
    
    # Initial set of shifts to consider
    active_shifts = {1, 2, 3, 4, 5}
    
    # Try to solve the initial MIP within the budget
    prob, vars_dict = build_mip(N, active_shifts)
    x_star, t_star = solve_mip(prob, vars_dict, time_limit=30)
    
    iteration = 0
    while time.time() - start_time < budget_s - 5:
        # Check if x_star is a true optimum for all shifts
        _, max_overlap = worst_shift(x_star)
        overlap_is_bounded_by_t = True
        for k in active_shifts:
            if k >= len(x_star):
                continue
            # Evaluate the full overlap function to check all k
            # Fast version to avoid recalculation
            y = np.correlate(x_star, 1 - x_star, mode='full')
            if y[k] > t_star:
                overlap_is_bounded_by_t = False
                break
        
        if overlap_is_bounded_by_t:
            break
        
        # If not, find the worst shift for the current solution
        k_max, _ = worst_shift(x_star)
        active_shifts.add(k_max)
        
        # Rebuild and solve MIP
        prob, vars_dict = build_mip(N, active_shifts)
        
        # Warm start
        x0 = x_star.astype(float)
        for idx in range(len(x_star)):
            vars_dict[0][idx].warm_start = x0[idx]
        
        x_star, t_star = solve_mip(prob, vars_dict, time_limit=20)
        iteration += 1
        
        if iteration > 20:
            break
    
    # Binary search to tighten the bound
    candidate_t = int(t_star)
    while candidate_t >= 1:
        if time.time() - start_time > budget_s - 5:
            break
        # Set up feasibility problem
        prob_test, vars_dict_test = build_mip(N, active_shifts)
        # Reuse all constraints but modify objective to None (feasibility)
        # and add bound constraint
        x_test, t_test, w_test = vars_dict_test
        for k in active_shifts:
            if k >= len(x_test):
                continue
            left_term = cp.sum(x_test[:N - k])
            right_term = cp.sum([w_test[i, k] for i in range(N - k)])
            prob_test.constraints.append(left_term - right_term <= candidate_t)
        
        try:
            prob_test.solve(solver=cp.CBC, time_limit=5, verbose=False)
            if prob_test.status in ["optimal", "optimal_inaccurate"]:
                t_star = candidate_t
            else:
                break
        except:
            break
        
        candidate_t -= 1
    
    # Final computation of C5
    final_c5 = t_star * dx
    
    return (x_star.tolist(), final_c5, N)
