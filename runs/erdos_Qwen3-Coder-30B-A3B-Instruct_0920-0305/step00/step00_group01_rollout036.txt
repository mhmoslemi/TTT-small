            j = i + d
            if 0 <= j < n_points:

            break

        h_vars = cp.Variable(n_points)
        p_vars = {}

        for d in range(1 - n_points, n_points):
            if d != 0:
                p_vars[d] = cp.Variable(len(shift_indices[d]))

        constraints = [
            cp.sum(h_vars) == target_mass,
        ]
        for i in range(n_points):
            constraints.append(0 <= h_vars[i])
            constraints.append(h_vars[i] <= 1)

        for d in range(1 - n_points, n_points):
            if d == 0:
                continue
            for idx, (i, j) in enumerate(shift_indices[d]):
                p = p_vars[d][idx]
                h_i = h_vars[i]
                h_j = h_vars[j]

                # McCormick envelopes for p = h_i * h_j
                constraints.append(p <= h_i)
                constraints.append(p <= h_j)
                constraints.append(p >= h_i + h_j - 1)
                constraints.append(p >= 0)

        # ∫ h(x) h(x + k) dx = sum(p[i,k] * dx)

        for d in range(1 - n_points, n_points):
            if d == 0:
                continue
            w = (1 - t) / dx
            constraints.append(cp.sum(p_vars[d]) >= w)

        # Minimize 0 (dummy objective)
        prob = cp.Problem(cp.Minimize(0), constraints)

        try:
            # Solve the LP
            prob.solve(solver=cp.GLPK, verbose=False)

        except Exception as e:

    # Final refinement with local optimization
    if time.time() - start_time < time_limit - 10:
        best_h = refine_solution(best_h, target_mass)

def round_solution(h_float, mass_target):
    # Create binary approximation
    h_rounded = np.zeros_like(h_float)
    # Sort indices by h_float value descending
    indices = np.argsort(h_float)[::-1]
    mass_used = 0
    for i in indices:
        if mass_used < mass_target:
            h_rounded[i] = 1
            mass_used += 1
        else:
            break
    # Adjust to match exactly target_mass
    if mass_used > mass_target:
        # Turn on some 1s to 0s
        diff = mass_used - mass_target
        for i in range(len(h_rounded)-1, -1, -1):
            if h_rounded[i] == 1 and diff > 0:
                h_rounded[i] = 0
                diff -= 1
    elif mass_used < mass_target:
        # Turn on some 0s to 1s
        diff = mass_target - mass_used
        for i in range(len(h_rounded)):
            if h_rounded[i] == 0 and diff > 0:
                h_rounded[i] = 1
                diff -= 1
    return h_rounded

def refine_solution(h, mass_target):
    # Simple local refinement using projected gradient
    h = h.copy()
    
    def eval_c5(h_test):
        return evaluate_erdos_solution(h_test)

    def objective(x):
        # Use negative to minimize
        return -eval_c5(x)

    def gradient(x):
        # Approximate gradient using finite difference
        eps = 1e-5
        grad = np.zeros_like(x)
        for i in range(len(x)):
            x_plus = x.copy()
            x_plus[i] += eps
            x_minus = x.copy()
            x_minus[i] -= eps
            grad[i] = (eval_c5(x_plus) - eval_c5(x_minus)) / (2 * eps)
        return grad

    # Directions for projecting onto [0,1]^n
    bounds = [(0, 1) for _ in range(len(h))]
    # Project h to feasible set
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum != mass_target:
        # Simple adjust: add offset to larger values
        adjust = (mass_target - h_sum) / len(h)
        h += adjust
        h = np.clip(h, 0, 1)
    h_prev = h.copy()
    for _ in range(30):
        # Try gradient descent step
        g = gradient(h_prev)
        step_size = 0.001
        h_new = h_prev - step_size * g
        # Project onto [0,1]^n
        h_new = np.clip(h_new, 0, 1)
        # Enforce sum constraint
        h_scale = mass_target / np.sum(h_new)
        h_new = h_new * h_scale
        h_new = np.clip(h_new, 0, 1)
        if np.sum(h_new) > 0:
            h_new = h_new * (mass_target / np.sum(h_new))
        diff = np.linalg.norm(h_new - h_prev)
        h_prev = h_new
        if diff < 1e-6:
            break
    return h_prev

def evaluate_erdos_solution(h):
    # Your pre-imported evaluation function
    # Compute maximum overlap integral
    n = len(h)
    dx = 2.0 / n
    max_val = 0
    # Compute full convolution h * (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    for k in range(len(corr)):
        val = corr[k] * dx
        max_val = max(max_val, val)
    return max_val
