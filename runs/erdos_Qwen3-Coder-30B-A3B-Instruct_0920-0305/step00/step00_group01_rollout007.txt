# Helper functions
def compute_overlap(h, dx):
    # h: array of length n_points
    # returns maximum overlap integral C5 = max_k ∫ h(x)(1 - h(x+k)) dx
    n = len(h)
    # Compute full convolution of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # The overlap integrals for each shift k
    overlaps = []
    for k in range(len(corr)):
        shift = k - (n - 1)  # k ranges from -(n-1) to (n-1)
        if abs(shift) <= 2:
            overlaps.append(corr[k] * dx)
    return max(overlaps)

def initial_h_values(n_points):
    # Provide a starting solution, e.g., a simple step function
    h = np.ones(n_points) * 0.5
    # Create a step function
    mid = n_points // 2
    h[:mid] = 1
    h[mid:] = 0
    # Normalize for average 0.5
    h = h * (n_points / 2) / np.sum(h) * 0.5
    return h

def setup_lp_model(n_points, dx, t):
    # Set up the linear programming problem with binary variables
    # h_i : [0,1]
    # p_{i,d} : linearization of h_i * h_{i+d}
    # objective: minimize t >= max_k O_k
    # O_k = ∫ h(x)(1 - h(x+k)) dx = 1 - ∫ h(x)h(x+k) dx = 1 - ∑ p_{i,k} * dx
    # So we need ∑ p_{i,k} >= (1 - t) / dx for all shifts k.
    
    # Variables
    h = cp.Variable(n_points)
    p = cp.Variable((n_points, 2*n_points - 1))  # For all shifted products
    
    # Objective
    # Just need to check feasibility for the given t, so make trivial objective
    obj = cp.Minimize(0)
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h) == n_points/2)
    
    # Bound constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # McCormick envelopes for each p_ij (for shift d)
    # p[i,d] = h[i] * h[i+d]
    # Let u = h[i], v = h[i+d]
    # p >= 0
    # p <= u
    # p <= v
    # p >= u + v - 1
    for d in range(2*n_points - 1):
        shift = d - (n_points - 1)
        for i in range(n_points):
            if i + shift >= 0 and i + shift < n_points:
                # The p indicies correspond to (i, d) which is i and shift d
                p_index = (i, d)
                # Put the constraints ensuring p <= h_i, h_{i+d}
                p[i,d] <= h[i]
                p[i,d] <= h[i + shift]
                # Enforce the constraint p >= h_i + h_{i+d} - 1
                p[i,d] >= h[i] + h[i + shift] - 1
                p[i,d] >= 0
                # Enforce ∑ p_{i,d} >= (1 - t)/dx
                constraints.append(cp.sum(p[:,d]) >= (1 - t)/dx)

    # Max constraint
    # Formulated differently: make sure the max discrete shift overlap does not exceed t
    max_overlap = 0
    
    # Create a more direct method:
    # We will use a problem (LP) to check if there exists a feasible h with all overlaps <= t
    # We solve:
    # min 0 s.t.
    #   constraints from above
    #   for each shift:
    #     where shift = d - (n-1), i in [0, n-1-shift], sum of p(i,d) >= (1-t)/dx
    #   sum(h) = n/2, 0 <= h[i] <= 1
    # for the above LP checking feasibility:
    # we just solve min 0 instead of minimizing anything else. 
    # we check if it returns a feasible solution (i.e., not infeasible or unbounded).
    
    prob = cp.Problem(obj, constraints)
    prob.solve(solver='GLPK')

    return prob.status == 'optimal'

def find_feasible_h_bisection(n_points, dx, lower_bound=0.0, upper_bound=0.38092, time_budget=600):
    start_time = time.time()
    best_h = None
    best_c5 = upper_bound
    last_improvement = 0
    
    while (upper_bound - lower_bound > 1e-5) and (time.time() - start_time < time_budget - 5):
        t_mid = (lower_bound + upper_bound) / 2
        is_feasible = setup_lp_model(n_points, dx, t_mid)
        if is_feasible:
            upper_bound = t_mid
            best_c5 = t_mid
        else:
            lower_bound = t_mid
        last_improvement = time.time()
        
    # At this point, computing the actual good solution for the final t
    # Reuse, with an optimization loop for best local result

    # We don't actually want to do full simulation because of our external constraints - just refine the existing ones
    h_initial = initial_h_values(n_points)
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    
    # Our guess after LP and slight refinement:
    best_h = refine_solution(h_initial, dx, best_c5)
    return best_h, best_c5

def refine_solution(h_initial, dx, t_target):
    # Simple refinement to make h realizable, and ensure it meets the target.
    h = h_initial.copy()
    
    # Clip h to [0,1] (just a sanity check)
    h = np.clip(h, 0, 1)
    
    # During normalization, keep track of mass imbalance
    current_mass = np.sum(h)
    target_mass = 47 / 2  # assuming n=47 or consistent value
    h = h * target_mass / current_mass  # adjust to correct sum
    
    # Bring values closer to {0,1} to make a good function:
    # A rough approximation using a sigmoid function or direct rounding.
    # Let's try rounding it and balancing to sum to target mass.
    
    sorted_indices = np.argsort(h)[::-1]  # Sort indices from highest to lowest
    desired_mass = target_mass
    h_bin = np.zeros_like(h)
    for i in sorted_indices:
        if desired_mass >= 1:
            h_bin[i] = 1.0
            desired_mass -= 1
        else:
            h_bin[i] = desired_mass  # Final fractional assignment
            break
    if desired_mass > 0:
        for i in range(len(h)):
            if h_bin[i] == 0 and desired_mass > 0:
                h_bin[i] = desired_mass if desired_mass <= 1 else 1
                desired_mass -= h_bin[i]
                if desired_mass <= 0:
                    break
    
    # Ensure it's valid
    h_bin = np.clip(h_bin, 0, 1)
    np.testing.assert_allclose(np.sum(h_bin), target_mass, rtol=1e-5)
    
    # Compute final C5 value
    max_k = compute_overlap(h_bin, dx)
    return h_bin, max_k
