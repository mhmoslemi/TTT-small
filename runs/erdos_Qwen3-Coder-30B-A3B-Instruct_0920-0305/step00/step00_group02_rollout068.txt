def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Using convolution to compute all overlaps at once
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    # Note: the convolution has length 2*n - 1, and the valid overlaps are 
    # in the middle part (n-1 elements) corresponding to k from 0 to 2
    # For our problem, we need k from 0 to 2, so we look at the middle part
    # The full convolution has length 2*n - 1, the overlapping part is from
    # index n-1 : 2*n-2 (which is indices 0 to n-1 of the full convolution)
    # Actually, let's be precise: for k from 0 to 2 (i.e. 2*dx step from 0 to 2) 
    # we have steps 0 to n-1. The actual k values are 2*dx * (0 to n-1) = 0 to 2 - 2*dx
    # Overlaps calculated with convolve h and 1-h (i.e. k from 0 to 2), so maximum 
    # overlap should be the maximal value in the full convolution
    max_overlap = np.max(conv) * dx
    return max_overlap
