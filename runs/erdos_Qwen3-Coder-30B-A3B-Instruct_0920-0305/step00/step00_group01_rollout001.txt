import numpy as np
import time
from scipy.optimize import minimize
import math

def project_to_feasible(v, target_sum):
    """Project vector v to feasible region with sum = target_sum and bounds [0,1]"""
    # Ensure bounds
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-10:
        return v
    
    # Use bisection to find lambda for projection
    # Project v to sum = target_sum with 0,1 bounds
    # This is equivalent to finding lambda such that 
    # sum_i clip(v_i - lambda, 0, 1) = target_sum
    low, high = -1.0, 1.0
    for _ in range(100):
        mid = (low + high) / 2.0
        proj = np.clip(v - mid, 0, 1)
        proj_sum = np.sum(proj)
        if proj_sum > target_sum:
            low = mid
        else:
            high = mid
    lambda_val = (low + high) / 2.0
    final_v = np.clip(v - lambda_val, 0, 1)
    return final_v

def compute_overlaps(h):
    """Compute overlaps for all shifts"""
    # Compute correlation of h with (1-h)
    # This will give us the integral over each lag
    corr = np.correlate(h, 1 - h, mode="full")
    # Transform to actual domain shifts of size dx
    dx = 2.0 / len(h)
    return corr * dx

def evaluate_erdos_solution(h):
    """Evaluate C_5 = max_k integral of h(x)(1 - h(x+k)) dx"""
    # Compute all overlaps
    overlaps = compute_overlaps(h)
    # Return maximum overlap
    return np.max(overlaps)

    # Parameters

        # Generate a candidate solution by modifying two points
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        if i == j:
            continue
        delta = np.random.uniform(-step_size, step_size)
        
        # Compute constraints for delta (how much we can change)
        # h[i] + delta should be in [0,1]
        # h[j] - delta should be in [0,1] 
        d_low = max(-h[i], h[j] - 1.0)
        d_high = min(1.0 - h[i], h[j])
        
        if d_low > d_high:
            continue
            
        actual_delta = np.random.uniform(max(d_low, delta*0.1), min(d_high, delta*1.5))
        
        # Apply changes
        h_new = h.copy()

        h_new[j] -= actual_delta

        h_new = project_to_feasible(h_new, n_points/2)

        new_c5 = evaluate_erdos_solution(h_new)

            # Periodic check of timeout
            elapsed = time.time() - start_time
