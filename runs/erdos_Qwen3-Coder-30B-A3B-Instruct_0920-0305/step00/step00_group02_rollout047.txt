import time
import numpy as np
import cvxpy as cp
from scipy import sparse
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    
    # --- Parameters ---
    n = 200  # grid points, even
    dx = 2.0 / n
    max_time = budget_s
    
    # --- Helper functions ---
    def build_shift_matrices(n):
        """Build sparse A_k matrices for k=1..floor(n/2)"""
        A_list = []
        for k in range(1, (n//2) + 1):
            # Create indices for A_k
            i_indices, j_indices = [], []
            for i in range(n - k):
                i_indices.append(i)
                j_indices.append(i + k)
            # Symmetric matrix
            rows = np.concatenate([i_indices, j_indices])
            cols = np.concatenate([j_indices, i_indices])
            data = np.ones(len(rows), dtype=int)
            A_k = sparse.csr_matrix((data, (rows, cols)), shape=(n, n))
            A_list.append(A_k)
        return A_list

    # --- SDP relaxation ---
    print("Building shift matrices...")
    A_list = build_shift_matrices(n)
    if time.time() - start_time > 0.9 * max_time:
        print("Early exit: not enough time to build matrices.")
        return initial_h_values, 0.38092, 45
    print("Matrices built.")

    # --- Define SDP variables ---
    X = cp.Variable((n, n), symmetric=True)
    x = cp.Variable(n)

    # --- Constraints ---
    constraints = [
        cp.diag(X) == x,
        cp.sum(x) == n / 2,
        x >= 0,
        x <= 1,
        X >> 0  # Positive semidefinite
    ]

    # Add overlap constraints
    t = cp.Variable()
    for A_k in A_list:
        constraints.append(cp.trace(A_k @ X) >= 2 * t)

    # --- Objective ---
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    # --- Solve SDP ---
    print("Solving SDP...")
    try:
        prob.solve(solver=cp.SCS, max_iters=2000, eps=1e-4, verbose=False)
        # Check if solved
        if prob.status != cp.OPTIMAL:
            print("SDP did not converge to optimal.")
            # Fallback to a basic strategy
            return initial_h_values, 0.38092, 45
        t_star = t.value
        X_star = X.value
        x_star = x.value
        print(f"SDP solved. t* = {t_star:.6f}")
    except Exception as e:
        print("Exception in SDP solve:", e)
        return initial_h_values, 0.38092, 45
    end_sdp = time.time()
    print(f"SDP solve took {end_sdp - start_time:.2f}s")

    # --- Rounding via Randomized Hyperplane Round ---
    print("Performing rounding...")
    best_C5 = 1.0
    best_h = None
    num_rounds = 250
    r = 8  # Rank for factorization

    V, _ = np.linalg.eigh(X_star)
    V = V[:, -r:]  # Top r components
    signs = np.sign(V @ np.random.randn(r))
    V = V * signs  # Fix signs for consistency

    for _ in range(num_rounds):
        if time.time() - start_time > 0.8 * max_time:
            break
        g = np.random.randn(r)
        s = V @ g
        s_sorted = np.sort(s)[::-1]
        # Find threshold so exactly n/2 entries are >=
        tau = s_sorted[n // 2 - 1]  # But we might need more careful handling.
        # Better: find exact threshold after sorting
        indices_sorted = np.argsort(s)[::-1]
        if n // 2 <= len(indices_sorted):
            tau = s[indices_sorted[n // 2 - 1]]
        else:
            tau = s_sorted[-1]
        h = (s >= tau).astype(int)
        if np.sum(h) != n // 2:
            # Adjust to get exactly n/2 ones (give preference to highest scores)
            diffs = s - tau
            top_indices = np.argsort(diffs)[::-1][:n//2]
            h = np.zeros(n, dtype=int)
            h[top_indices] = 1
        # Compute actual C5
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_C5:
            best_C5 = c5
            best_h = h.copy()

    if best_h is None:
        # fallback
        return initial_h_values, 0.38092, 45

    # --- Local polishing ---
    print("Polishing solution...")
    h = best_h
    corr = np.correlate(h, 1 - h, mode='full')
    min_overlap = np.min(corr)
    corr_full = np.correlate(h, h, mode='full')[n-1:]
    best_corr = np.copy(corr_full)

    # Hill-climbing swaps
    max_swaps = 50000
    swaps_done = 0
    last_improvement = 0

    while swaps_done < max_swaps and time.time() - start_time < 0.95 * max_time:
        i = np.random.choice(np.where(h == 1)[0])
        j = np.random.choice(np.where(h == 0)[0])
        # Try swap
        h_copy = h.copy()
        h_copy[i] = 0
        h_copy[j] = 1
        corr_new = np.correlate(h_copy, 1 - h_copy, mode='full')
        min_new = np.min(corr_new)
        if min_new > min_overlap or (min_new == min_overlap and np.random.rand() < 0.01):
            h = h_copy
            corr = corr_new
            min_overlap = min_new
            last_improvement = swaps_done
        swaps_done += 1

    # Compute final C5
    final_corr = np.correlate(h, 1 - h, mode='full')
    final_C5 = np.max(final_corr) * dx

    return h.tolist(), final_C5, n
