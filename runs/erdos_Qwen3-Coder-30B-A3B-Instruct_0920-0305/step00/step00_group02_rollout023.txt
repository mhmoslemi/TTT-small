import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Hyperparameters
    n = 300  # number of samples (must be even)
    dx = 2.0 / n
    max_time = budget_s
    alpha_start = 1.0
    alpha_max = 10000.0
    eta_init = 1.0
    patience = 100
    iter_per_sec = 1000  # for scheduling
    
    # Generate beatty sequence seed
    beta = (np.sqrt(5) - 1) / 2  # golden ratio
    h0 = np.array([1 if (i * beta) % 1 < 0.5 else 0 for i in range(n)])
    # Project to sum = n/2
    current_sum = np.sum(h0)
    target_sum = n / 2
    if abs(current_sum - target_sum) > 1e-6:
        if current_sum > target_sum:
            # Need to turn some 1s into 0s
            ones_idx = np.where(h0 == 1)[0]
            num_to_flip = int(current_sum - target_sum)
            flip_idx = np.random.choice(ones_idx, size=num_to_flip, replace=False)
            h0[flip_idx] = 0
        else:
            # Need to turn some 0s into 1s
            zeros_idx = np.where(h0 == 0)[0]
            num_to_flip = int(target_sum - current_sum)
            flip_idx = np.random.choice(zeros_idx, size=num_to_flip, replace=False)
            h0[flip_idx] = 1
    h = h0.astype(float)
    
    # Optimization parameters
    alpha = alpha_start
    best_h = h.copy()
    best_C5 = np.inf
    t0 = time.time()
    
    def project_onto_F(h, target_sum):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-9:
            return h
        # Waterfilling
        diff = target_sum - current_sum
        if abs(diff) < 1e-12:
            return h
            
        # Binary search for the watermark
        min_val = 0.0
        max_val = 1.0
        for _ in range(30):
            mid = (min_val + max_val) / 2.0
            new_s = np.sum(np.clip(h + mid, 0, 1))
            if new_s < target_sum:
                min_val = mid
            else:
                max_val = mid
        lambda_val = (min_val + max_val) / 2.0
        h = np.clip(h + lambda_val, 0, 1)
        return h
    
    def compute_S(h, k):
        return np.sum(h[:-k] * h[k:]) if k < len(h) else 0.0
    
    def compute_all_S(h):
        # Use convolution
        h_rev = h[::-1]
        c_full = np.correlate(h, h_rev, mode='full')
        S = []
        for k in range(n):
            S.append(c_full[n - 1 + k] * dx)
        return np.array(S)
    
    def logsumexp_minus(S, alpha):
        exp_terms = np.exp(-alpha * S)
        return - (1.0/alpha) * np.log(np.sum(exp_terms))
    
    def grad_logsumexp_minus(S, alpha):
        exp_terms = np.exp(-alpha * S)
        weights = exp_terms / np.sum(exp_terms)
        grad = np.zeros_like(h)
        for k in range(n):
            grad += weights[k] * np.roll(h, k) * dx  # left shift
        return grad
    
    last_improvement = 0
    iter_count = 0
    
    while time.time() - t0 < max_time * 0.6:
        # Compute overlaps directly via correlation
        # This part is fast using numpy.convolve
        c = np.correlate(h, h[::-1], mode='full')
        S = np.array([c[n - 1 + k] * dx for k in range(n)])
        
        # Use smoother version
        Phi_alpha = logsumexp_minus(S, alpha)
        
        grad = grad_logsumexp_minus(S, alpha)
        
        # Adaptive step size selection (line search)
        eta = eta_init
        while eta > 1e-8:
            h_new = h + eta * grad
            h_new = project_onto_F(h_new, n / 2)
            c_new = np.correlate(h_new, h_new[::-1], mode='full')
            S_new = np.array([c_new[n - 1 + k] * dx for k in range(n)])
            Phi_alpha_new = logsumexp_minus(S_new, alpha)
            if Phi_alpha_new >= Phi_alpha - 1e-8 or eta < 1e-10:
                break
            eta *= 0.5
        h = h_new
        if Phi_alpha_new > Phi_alpha + 1e-8:
            last_improvement = iter_count
        
        # Update alpha
        if iter_count > 0 and iter_count % 200 == 0 and alpha < alpha_max:
            alpha *= 2.0
        if alpha > alpha_max:
            alpha = alpha_max
        iter_count += 1
        
        # Keep best solution
        C5 = 1.0 - np.min(S)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
        if iter_count - last_improvement > patience * 2:
            break
    
    # After gradient ascent, perform local hill climbing on binary version
    h_binary = np.zeros(n)
    h_sorted = np.sort(best_h)
    cutoff = h_sorted[n//2]
    for i in range(n):
        h_binary[i] = 1.0 if best_h[i] >= cutoff else 0.0
    h_binary = project_onto_F(h_binary, n/2)
    
    def compute_C5_binary(h_bin):
        c_full = np.correlate(h_bin, 1 - h_bin, mode='full')
        return np.max(c_full) * dx
    
    current_C5 = compute_C5_binary(h_binary)
    best_binary = h_binary.copy()
    best_binary_C5 = current_C5
    # Simulated annealing neighbourhood search (swap-based)
    t1 = time.time()
    max_swap_seconds = max_time * 0.3
    swapped = set()
    tabu_size = 20
    tabu_list = []
    T = 1.0
    c_temp = 1.0  # cooling factor
    attempts = 0
    
    while time.time() - t1 < max_swap_seconds:
        if attempts % 1000 == 0:
            T *= c_temp
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        if (i,j) in tabu_list or (j,i) in tabu_list:
            continue
        # Try swap
        h_swap = h_binary.copy()
        h_swap[i], h_swap[j] = h_swap[j], h_swap[i]
        # Ensure sum constraint is maintained
        s_old = np.sum(h_binary)
        s_new = np.sum(h_swap)
        if abs(s_old - s_new) != 0:
            # this shouldn't happen but just in case, adjust
            h_swap = project_onto_F(h_swap, n/2)
        
        C5_new = compute_C5_binary(h_swap)
        if C5_new < current_C5:
            h_binary = h_swap
            current_C5 = C5_new
            # Update best_solution
            if current_C5 < best_binary_C5:
                best_binary_C5 = current_C5
                best_binary = h_binary.copy()
            # Add to tabu list
            tabu_list.append((i, j))
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        elif np.random.rand() < np.exp(-(C5_new - current_C5) / T):
            h_binary = h_swap
            tabu_list.append((i, j))
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        attempts += 1

    h_final = best_binary
    metrics_final = compute_C5_binary(h_final)
    
    # Final validation
    assert np.isclose(np.sum(h_final), n / 2, atol=1e-6)
    assert np.all((h_final >= 0) & (h_final <= 1))
    
    return (h_final.tolist(), float(metrics_final), n)
