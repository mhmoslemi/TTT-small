import numpy as np
import time
from scipy.optimize import minimize

def project_onto_simplex_box(h, target_sum):
    """Project h onto the simplex with sum = target_sum and bounds [0,1]"""
    # Clip to [0,1] bounds
    h = np.clip(h, 0, 1)
    # Adjust sum to exact target using simple rescaling
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    elif current_sum > target_sum:
        # Reduce values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values
        diff = target_sum - current_sum
        # Distribute increment greedily to values that can increase
        remaining = diff
        indices = np.argsort(h)
        for i in indices:
            if remaining <= 0:
                break
            increase = min(1 - h[i], remaining)
            h[i] += increase
            remaining -= increase
    return h

def evaluate(h):
    """Evaluate C5 = max ∫ h(x)(1 - h(x+k)) dx"""
    dx = 2.0 / len(h)
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def gradient_step(h, step_size=0.01):
    """Compute gradient and take a step to reduce max overlap"""
    N = len(h)
    dx = 2.0 / N
    # Compute all correlations
    correlation = np.correlate(h, 1 - h, mode="full")
    # Find maximum correlation index
    max_idx = np.argmax(correlation)
    max_corr = correlation[max_idx]
    
    # Compute gradient for the max overlap lag
    grad = np.zeros(N)
    lag = max_idx - (N - 1)  # convert index to actual lag value
    
    # Compute gradient contribution for this lag
    if lag >= 0:
        start = 0
        end = N - lag
        for i in range(start, end):
            j = i + lag
            grad[i] += 1 - h[j]
            grad[j] -= h[i]
    else:
        # lag < 0, shift left
        shift = -lag
        start = shift
        end = N
        for i in range(start, end):
            j = i - shift
            grad[i] += 1 - h[j]
            grad[j] -= h[i]
    
    # Update h with gradient step
    new_h = h - step_size * grad
    # Project back to feasible space
    new_h = project_onto_simplex_box(new_h, target_sum=N/2)
    return new_h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    
    # Try different discretizations
    for N in [100, 150, 200]:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Start with initial construction if provided
        if 'initial_h_values' in globals() and initial_h_values is not None:
            h = np.array(initial_h_values)
            h = h[:N] if len(h) > N else np.pad(h, (0, N - len(h)), 'constant')
        else:
            h = np.full(N, 0.5)
        
        # Ensure proper sum constraint
        h = project_onto_simplex_box(h, target_sum=N/2)
        
        # Optimization loop
        iterations = 0
        max_iterations = min(10000, int(1000000 / N))
        
        while time.time() - start_time < budget_s - 5 and iterations < max_iterations:
            iterations += 1
            h = gradient_step(h, step_size=0.05)
            c5 = evaluate(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if iterations % 1000 == 0:
                pass  # progress estimation
    
    if best_h is None:
        # Fallback to constant solution
        N = 200
        best_h = np.full(N, 0.5)
        best_c5 = evaluate(best_h)
    
    # Ensure it is within acceptable range
    best_h = np.clip(best_h, 0, 1)
    best_h = project_onto_simplex_box(best_h, target_sum=len(best_h)/2)
    best_c5 = evaluate(best_h)
    
    return (best_h.tolist(), float(best_c5), len(best_h))
