def evaluate_erdos_solution(h):
    # Provided function to evaluate the exact C5 value
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr_full = np.correlate(h, 1 - h, mode='full')
    # Take only the relevant parts (lags from -n+1 to n-1)
    corr_vals = corr_full[n-1:2*n-1]
    # Scale by dx and return max
    return np.max(corr_vals) * dx

def fast_correlation(h, dx, L=None):
    # Compute correlation efficiently using FFT
    if L is None:
        L = 2**(int(np.log2(2*len(h)-1)) + 1)
    h_padded = np.pad(h, (0, L - len(h)), 'constant')
    g_padded = np.pad(1 - h, (0, L - len(h)), 'constant')
    H = fftpack.fft(h_padded, L)
    G = fftpack.fft(g_padded, L)
    corr = fftpack.ifft(H * np.conj(G)).real[:2*len(h)-1]
    return corr * dx

def project_to_simplex(y, target_sum):
    # Euclidean projection onto capped simplex
    low = np.min(y) - 1.0
    high = np.max(y) + 1.0
    def S(lam):
        return np.sum(np.clip(y - lam, 0, 1)) - target_sum
    # Binary search for lambda
    for _ in range(60):
        lam = (low + high) / 2.0
        s = S(lam)
        if s > 0:
            low = lam
        else:
            high = lam
    lam = (low + high) / 2.0
    x = np.clip(y - lam, 0, 1)
    return x

def smooth_max_grad(h, dx, gamma=5.0):
    # Compute the gradient of smooth maximum approximation
    n = len(h)
    L = 2**(int(np.log2(2*n - 1)) + 1)
    h_padded = np.pad(h, (0, L - n), 'constant')
    g_padded = np.pad(1 - h, (0, L - n), 'constant')
    H = fftpack.fft(h_padded, L)
    G = fftpack.fft(g_padded, L)
    corr = fftpack.ifft(H * np.conj(G)).real[:2*n-1]
    corr = corr * dx  # scaled correlation values
    # Softmax weights
    efc = np.exp(gamma * corr)
    w = efc / np.sum(efc)
    w = np.pad(w, (n-1, n-1), 'constant')  # for full correlation length
    # Compute gradient per position
    grad = np.zeros(n)
    for i in range(n):
        # Forward terms: (1-h[i+ell])
        # Backward terms: h[i-ell]
        for ell in range(n):
            if i + ell < n:
                grad[i] += w[ell+n-1] * (-h[i+ell]) * dx
            if i >= ell:
                grad[i] += w[ell+n-1] * (1-h[i-ell]) * dx
    return grad

def continuous_refinement(initial_h, n, dx, t_max):
    h = initial_h.copy()
    t_start = time.time()
    # Initial C5
    current_C5 = evaluate_erdos_solution(h)
    gamma = 5.0
    step_size = 0.1 * dx
    patience = 0
    max_patience = 5
    
    while time.time() - t_start < t_max:
        grad = smooth_max_grad(h, dx, gamma)
        # gradient step
        trial_h = h - step_size * grad
        # project
        projected = project_to_simplex(trial_h, n/2)
        # recompute C5
        new_C5 = evaluate_erdos_solution(projected)
        
        if new_C5 < current_C5:
            h = projected
            current_C5 = new_C5
            patience = 0
        else:
            patience += 1
            if patience >= max_patience:
                # Reduce step size if happened too many times
                step_size *= 0.9
                patience = 0
        
        # Increasing smoothness
        if gamma < 200:
            gamma *= 1.5
        
        if patience < max_patience:
            continue  # Keep going
        else:
            if gamma >= 200:
                break
            continue  # Stop early if we're stuck

    return h, current_C5
