import numpy as np
import time
import random

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    time_end = start + budget_s - 0.5
    n_points = 200
    
    # Initialize best solution
    best_h = None
    best_c5 = float('inf')
    
    # Parameters for simulated annealing
    max_restarts = 50
    max_local_iters = 10000
    T0 = 0.05
    
    restart_count = 0
    while time.time() < time_end and restart_count < max_restarts:
        if time.time() >= time_end:
            break
            
        # Generate new random binary solution with exactly n/2 ones
        h = np.zeros(n_points, dtype=np.float64)
        ones = np.random.choice(n_points, n_points//2, replace=False)
        h[ones] = 1.0
        
        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            
        # Local annealing search
        for it in range(max_local_iters):
            if time.time() >= time_end:
                break
                
            # Choose two random indices with different values
            i = np.random.randint(n_points)
            j = np.random.randint(n_points)
            while h[i] == h[j]:
                i = np.random.randint(n_points)
                j = np.random.randint(n_points)
                
            # Propose swap
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
            
            new_c5 = compute_c5(h_new)
            d = new_c5 - cur_c5
            
            # Simulated annealing acceptance probability
            T = T0 * (1 - it / max_local_iters)
            if d <= 0 or np.random.rand() < np.exp(-d / max(1e-12, T)):
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    
        restart_count += 1
        
    # Final evaluation to ensure correctness
    if best_h is not None:
        final_c5 = compute_c5(best_h)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback if no solution found
        h_fallback = np.zeros(n_points, dtype=np.float64)
        h_fallback[:n_points//2] = 1.0
        c5_fallback = compute_c5(h_fallback)
        return (h_fallback.tolist(), c5_fallback, n_points)
