def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    # h is an array of values over [0, 2]

    n_points = len(h)
    dx = 2.0 / n_points
    # For each k, compute integral h(x)(1 - h(x+k))
    # We compute the correlation of h with (1-h)
    # The full convolution gives us all possible overlaps
    h1 = 1 - h
    correl = np.correlate(h, h1, mode='full')
    # The maximum of these values times dx is our C5
    return np.max(correl) * dx
