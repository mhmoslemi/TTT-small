import numpy as np
import time
import random
import math
from scipy.optimize import minimize

def evaluate(x):
    """Evaluate the C5 constant using the same method as the official evaluator."""
    dx = 2.0 / len(x)
    corr = np.correlate(x, 1 - x, mode='full')
    return np.max(corr) * dx

def project_onto_capped_simplex(y, target_sum):
    """Project y onto the capped simplex: {x in [0,1]^n | sum(x) = target_sum}."""
    y = np.clip(y, 0, 1)
    n = len(y)
    if abs(np.sum(y) - target_sum) < 1e-10:
        return y
    # Duchi's algorithm for projection
    y_sorted = np.sort(y)[::-1]
    tmp_sum = 0
    for i in range(n):
        tmp_sum += y_sorted[i]
        if tmp_sum > target_sum * (i + 1):
            continue
        else:
            # Compute lambda
            lambda_val = (tmp_sum - target_sum) / (i + 1)
            x = np.maximum(y - lambda_val, 0)
            return x
    # If we get here, all values are zero or they are (target_sum/n) - which should
    # be clamped to [0,1]. So, return the capped version.
    offset = (np.sum(y) - target_sum) / n
    return np.clip(y - offset, 0, 1)

def compute_gradient(x, k):
    """Compute the gradient dO_k/dx_i.
    x: the vector
    k: the shift
    gradient: list of dO_k/dx_i for i=0,...,n-1.
    """
    n = len(x)
    grad = np.zeros(n)
    if k > 0:
        # k > 0: O_k = sum_{i=0}^{n-k-1} x_i*(1-x_{i+k})
        for i in range(n - k):
            grad[i] += (1 - x[i + k])
            grad[i + k] -= x[i]
    else:
        # k < 0: O_{-k} = sum_{i=0}^{n+k-1} x_{i+k}*(1-x_i)
        k_abs = -k
        for i in range(n - k_abs):
            grad[i + k_abs] += (1 - x[i])
            grad[i] -= x[i + k_abs]
    return grad

def gradient_of_max(x, beta):
    """Compute the gradient of the smooth approximation S(x;beta) = (1/beta)*log(sum_k exp(beta * O_k))."""
    n = len(x)
    # Compute all O_k
    corr = np.correlate(x, 1 - x, mode='full')
    shifts = np.arange(-n + 1, n)
    O_k = corr
    # Softmax weights
    exp_beta_O = np.exp(beta * O_k)
    Z = np.sum(exp_beta_O)
    weights = exp_beta_O / Z
    # Gradient
    grad = np.zeros(n)
    for i in range(len(shifts)):
        k = shifts[i]
        g = compute_gradient(x, k)
        grad += weights[i] * g
    return grad

def swap_move(x, i, j):
    """Swap x[i] and x[j]."""
    x = x.copy()
    x[i], x[j] = x[j], x[i]
    return x

def fractional_move(x, i, j, delta):
    """Move delta units from x[j] to x[i] ensuring bounds."""
    x = x.copy()
    x[i] += delta
    x[j] -= delta
    x[i] = max(0, min(1, x[i]))
    x[j] = max(0, min(1, x[j]))
    return x

def worst_shift_info(x):
    """Return the index of the shift that gives the maximum O_k and list of indices where the mismatch occurs."""
    n = len(x)
    corr = np.correlate(x, 1 - x, mode='full')
    shifts = np.arange(-n + 1, n)
    max_k = shifts[np.argmax(corr)]
    max_val = np.max(corr)
    # Find positions where it's a "1" followed by a "0" with distance k*
    k_abs = abs(max_k)
    offending = []
    if max_k >= 0:
        for i in range(n - k_abs):
            if x[i] == 1 and x[i + k_abs] == 0:
                offending.append(i)
    else:
        for i in range(k_abs, n):
            if x[i] == 1 and x[i - k_abs] == 0:
                offending.append(i)
    return max_k, offending

def binary_simulated_annealing(N, max_iter, T0=0.12):
    """Perform simulated annealing on binary vectors."""
    np.random.seed(42)
    x = np.zeros(N)
    ones_count = N // 2
    ones_indices = np.random.choice(N, size=ones_count, replace=False)
    x[ones_indices] = 1
    best_x = x.copy()
    best_val = evaluate(x)
    T = T0
    i = 0
    while i < max_iter:
        i += 1
        # pick random valid swap
        while True:
            a = np.random.randint(N)
            b = np.random.randint(N)
            if x[a] != x[b]:
                break
        new_x = swap_move(x, a, b)
        new_val = evaluate(new_x)
        delta = new_val - best_val
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            x = new_x
            if new_val < best_val:
                best_val = new_val
                best_x = x.copy()
        if T > 1e-10:
            T *= 0.99995
    return best_x, best_val

def directed_local_improvement(x, max_iter=5000):
    """Perform a local hill-climb targeting the worst shift."""
    x = x.copy()
    val = evaluate(x)
    for i in range(max_iter):
        # Find worst shift
        max_k, offend_indices = worst_shift_info(x)
        if len(offend_indices) == 0:
            break
        # Pick a random offending position
        idx = np.random.choice(offend_indices)
        # Look for a swap
        while True:
            j = np.random.randint(len(x))
            if x[j] == 0 and j != idx and j != (idx + abs(max_k)):
                break
        # Swap
        new_x = swap_move(x, idx, j)
        new_val = evaluate(new_x)
        if new_val <= val:
            x = new_x
            val = new_val
    return x, val

def continuous_refinement(x, max_iter=1000, beta0=5, beta_factor=1.02):
    """Refine using continuous method with smooth max function."""
    x = x.copy()
    val = evaluate(x)
    beta = beta0
    for i in range(max_iter):
        # Gradient of smooth approximation
        grad = gradient_of_max(x, beta)
        step_size = 0.02
        y = x - step_size * grad
        x = project_onto_capped_simplex(y, len(x) / 2)
        new_val = evaluate(x)
        if new_val < val:
            val = new_val
        beta *= beta_factor
    return x, val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    t_start = time.time()
    safety_margin = 5.0
    t_dead = t_start + budget_s - safety_margin

    candidate_sizes = [140, 200, 260]
    for N in candidate_sizes:
        if time.time() >= t_dead:
            break
        t_slice = (t_dead - time.time()) / (len(candidate_sizes) - candidate_sizes.index(N))
        if t_slice < 5:
            break
        
        # Phase 1: Binary Simulated Annealing
        max_sa_iter = int(30000 * (t_slice / 600))
        x, val = binary_simulated_annealing(N, max_sa_iter)
        if val < best_c5:
            best_h = x.copy()
            best_c5 = val
            best_n = N

        # Phase 2: Directed Local Improvement
        if time.time() < t_dead:
            x2, val2 = directed_local_improvement(x, max_iter=1000)
            if val2 < best_c5:
                best_h = x2.copy()
                best_c5 = val2
                best_n = N

        # Phase 3: Continuous Refinement
        if time.time() < t_dead:
            x3, val3 = continuous_refinement(x2, max_iter=500)
            if val3 < best_c5:
                best_h = x3.copy()
                best_c5 = val3
                best_n = N
    
    # Final evaluation to ensure correctness
    final_c5 = evaluate(best_h)
    return (list(best_h), final_c5, best_n)
