def evaluate(x):
    """Evaluate C5 from a step function x (discretized)"""
    dx = 2.0 / len(x)
    n = len(x)
    # This computes the full correlation: correlation of x with 1-x
    corr = np.correlate(x, 1 - x, mode='full')
    # The maximum is at the center of the correlation array
    max_corr = np.max(corr)
    return max_corr * dx

def project_onto_capped_simplex(y, target_sum):
    """
    Project y onto the capped simplex: {x : 0 <= x <= 1, sum(x) = target_sum}
    Uses Duchi et al. (2008) algorithm
    """
    y = np.clip(y, 0, 1)
    n = len(y)
    
    # Sort y in decreasing order
    sorted_y = np.sort(y)[::-1]
    
    # Find the threshold lambda for sum constraint
    # We need to find lambda such that sum(max(y_i - lambda, 0)) = target_sum
    
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_y)
    # For index k, we set everything <= k to maximum of target_sum / n (since we want mean = target_sum / n)
    # But for a more precise computation:
    # For threshold lambda = sorted_y[k] - (cumsum[k] - target_sum) / (k + 1)
    # We check all possible thresholds to find lambda such that sum(max(y_i - lambda, 0)) = target_sum
    
    # Simple way: clip and adjust
    # We know the projection exists and is unique.
    # Duchi's method:
    # 1. Sort y in descending order
    # 2. Compute cumulative sum of sorted y
    # 3. Find the critical index
    
    # Compute lambda_precise:
    # We want lambda such that sum(max(y[i] - lambda, 0)) = target_sum
    # Rearranging:
    # sum(y_i) - lambda * count(y_i > lambda) = target_sum
    # lambda = (sum(y_i) - target_sum) / count(y_i > lambda)
    # So solve for lambda in a way that both constraints are satisfied:
    
    # Alternative: Use sorting approach from Duchi:
    # t = (sum(y) - target_sum) / n
    # y = max(y - t, 0)
    # then scale on 1 to target_sum if it's not the case:
    # This is a simplified version that will not exactly work due to variance
    
    # Use direct approach:
    # 1. Compute full cumulative sums
    # 2. For each k (index)
    #    Define cutoff as  (sum(sorted_y[:k+1]) - target_sum) / (k + 1)
    #    This is origin of lambda at that level
    #    But more precisely:
    #    The lambda we want to find such that after projection, the sum is target_sum
    
    # Another clean way:
    # A) Sort y descending
    # B) For k in 0..n-1, compute delta = sum(sorted_y[:k+1]) - target_sum
    #    The lambda can be set to some point from sorted_y[k] to sorted_y[k+1]
    #    Let's say lambda ~ (sum_sorted_i - target_sum) / (k)
    #    Actually, it's simpler to follow  
    #    If we define lambda as (C - target_sum) / k where C is cumulative sum up to k
    #    Then we project, but only so that sum is correct
    
    # Algo: 
    # 1. Sort in descending order
    # 2. Scalar project iteratively (not efficient but correct)
    # Better approach: from Duchi algorithm
    temp = np.copy(y)
    temp = np.sort(temp)[::-1]
    
    # Polyak's method:
    # 1. Sort descending
    # 2. Incrementally compute lambda such that projection equals target_sum
    # 3. We compute lambda such that below that point, new values are > 0 and 
    #    above that point are 0 and sum in descending item is exactly target_sum

    # My implementation:
    # Implementation adapted from Duchi:  
    # Based on paper, it's not trivial to get correct lambda
    
    # Let us use the monotonic decrease method:
    # We compute: for i = 0, ... n-1
    # sum_vals = sum(sorted_y[:i+1])
    # lambda = (sum_vals - target_sum) / (i + 1)  
    # if lambda <= sorted_y[i], then lambda_i works, otherwise iterate further
    
    # Another method: averaged projection using bisection on the sum
    
    # Simpler: let's use the sophisticated two-loop algorithm:
    # Since this is just for small dimension, regular simple approach works
    
    # 1. start from 0 and compute cumulative sum
    # 2. For each position: define lambda = (sum(u[:k]) - target_sum)/(k+1)  
    # 3. Check if lambda was violated: avoid cutting prior points when used
    #    So: if i>=1:  lambda > u_sorted[i-1] violates assumption
    # Condensed abstract validation approach
    r = len(y)
    # sort descending
    sorted_indices = np.argsort(temp)[::-1]
    u = temp[sorted_indices]
    # cumsum u
    s = np.cumsum(u)
    # compute lambda values that would make the first k+1 entries give the correct sum
    # skip trivial cases
    # iterate to find the right partitioning
    
    # Method as per Duchi:
    # Compute from largest to smallest
    # λ = (s[i] – target_sum) / (i+1)
    # As long as λ <= u_ij , continue adding i 
    # so we use lambda from current level:
    for i in range(len(u)):
        # we compute lambda for this level
        lambda_level = (s[i] - target_sum) / (i + 1) 
        if lambda_level < u[i]:
            # We can only set elements below i if below target lambda level
            # accumulate.
            # Final lambda is a constant value where sum( (u - lambda)_+ ) = target_sum
            final_lambda = lambda_level
            break
    else: # normal case  
        final_lambda = (s[-1] - target_sum) / len(u)

    # Project onto [0,1]
    z = np.clip(np.array(u) - final_lambda, 0, 1)
    # Reorder back to original indices
    final = np.empty_like(y)
    final[sorted_indices] = z
    return final


def gradient_of_max(x, beta):
    """
    Compute the gradient of the smooth max approximation S(x;β) = (1/β)*log(sum(exp(β*O_k(x))))
    """
    n = len(x)
    dx = 2.0 / n
    # Compute all O_k values as a list
    corr = np.correlate(x, 1 - x, mode='full')
    k_vals = np.arange(-n+1, n)
    max_ks = []
    for k in k_vals:
        #the value O_k for index k
        start = max(0, -k) if k >= 0 else max(0, k)
        end = n - abs(k)  
        O_k = 0.0
        for i in range(start, end):
            if k >= 0:
                # x[i] * (1 - x[i+k])
                O_k += x[i] * (1 - x[i+k]) 
            else:
                # x[i] * (1 - x[i+k])  
                O_k += x[i] * (1 - x[i+k])
        max_ks.append(O_k)
    # Now compute softmax weight
    exp_ks = np.exp(np.array(max_ks) * beta)
    total = np.sum(exp_ks)
    if total == 0: 
        weights = np.ones_like(max_ks) / len(max_ks)
    else:
        weights = exp_ks / total
    # Compute gradient w.r.t. each x_i over all shifts k where it's used
    # ∂O_k/∂x_i is 1 - x_{i+k} if i <= N-1-k and i+|k| < N, and -x_{i-|k|} if i>=k  
    grad = np.zeros(n)
    for i in range(n):
        for k_idx, k in enumerate(k_vals):
            if k >= 0:
                i_prime = i + k
                if 0 <= i_prime < n:
                    grad[i] += weights[k_idx] * (1 - x[i_prime])
            else:
                i_prime = i + k
                if 0 <= i_prime < n:
                    grad[i] += weights[k_idx] * (- x[i_prime])
    return grad


def swap_move(x, i, j):
    """Swap two positions in x"""
    x_new = np.copy(x)
    x_new[i], x_new[j] = x_new[j], x_new[i]
    return x_new


def fractional_move(x, i, j, d):
    """Move amount d (signed) from x[j] to x[i]"""
    x_new = np.copy(x)
    if x[j] < d or x[i] + d > 1:
        return x # Do nothing
    x_new[i] += d
    x_new[j] -= d
    return x_new


def worst_shift_info(x):
    """Find the worst shift and the set of offending indices"""
    n = len(x)
    corr = np.correlate(x, 1 - x, mode='full')
    k_vals = np.arange(-n+1, n)
    max_corr = -np.inf
    best_k = 0
    best_set = []
    for k in k_vals:
        start_i = max(0, -k) if k >= 0 else max(0, k)
        end_i = n - abs(k)
        if end_i - start_i <= 0:
            continue
        # Compute the overlap for this shift
        o = 0.0
        pos_list = []
        for i in range(start_i, end_i):
            if k >= 0:
                o += x[i] * (1 - x[i + k])
                if x[i] == 1 and x[i + k] == 0:
                    pos_list.append(i)
            else:
                o += x[i] * (1 - x[i + k])
                if x[i] == 1 and x[i + k] == 0:
                    pos_list.append(i)
        
        if o > max_corr:
            max_corr = o
            best_k = k
            best_set = pos_list
    
    return best_k, best_set, max_corr
