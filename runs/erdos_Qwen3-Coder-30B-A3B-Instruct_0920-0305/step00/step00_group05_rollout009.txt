import numpy as np
import cvxpy as cp
import time
import math
from collections import defaultdict

def full_overlap(x):
    """Return the full overlap array."""
    return np.correlate(x, 1 - np.array(x), mode="full")

def worst_shift(x):
    """Return (k, max_overlap) where k is the shift that maximizes overlap."""
    overlaps = full_overlap(x)
    max_k = np.argmax(overlaps)
    max_val = overlaps[max_k]
    return max_k, max_val

def build_mip(N, active_shifts):
    """Build the MIP problem for given active shifts."""
    # Variables
    x = cp.Variable(N, integer=True)
    t = cp.Variable(integer=True)
    
    # Variables for AND constraints (wij,k = xi AND x_{i+k})
    w = {}
    constraints = []
    
    # Constraints for half-ones
    constraints.append(cp.sum(x) == N // 2)
    constraints.append(x >= 0)
    constraints.append(x <= 1)
    
    # Symmetry breaking
    constraints.append(x[0] == 1)
    
    # Add auxilliary binary variables for each active shift
    for k in active_shifts:
        for i in range(N - k):
            w_key = (i, k)
            w[w_key] = cp.Variable(integer=True)
            constraints.append(w[w_key] <= x[i])
            constraints.append(w[w_key] <= x[i + k])
            constraints.append(w[w_key] >= x[i] + x[i + k] - 1)
    
    # Overlap bounds
    for k in active_shifts:
        overlap_expr = sum(x[i] for i in range(N - k)) - sum(w[(i, k)] for i in range(N - k))
        constraints.append(overlap_expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Build problem
    prob = cp.Problem(objective, constraints)
    return prob, {'x': x, 't': t, 'w': w}

def solve_mip(prob, vars_dict, time_limit=30):
    """Solve the problem with a time limit."""
    x = vars_dict['x']
    t = vars_dict['t']
    prob.solve(solver=cp.CBC, max_iters=10000, TimeLimit=time_limit)
    x_val = np.round(x.value).astype(int)
    t_val = int(t.value)
    return x_val, t_val
