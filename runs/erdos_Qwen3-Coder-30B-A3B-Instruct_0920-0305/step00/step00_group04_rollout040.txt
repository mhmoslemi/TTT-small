import numpy as np
import time
import random

def compute_c5_fast(h):
    """Compute C5 using FFT for speed"""
    n = len(h)
    size = 1 << ((2*n-1).bit_length())
    f_h = np.fft.fft(h, size)
    f_g = np.fft.fft(1 - h, size)
    corr = np.fft.ifft(f_h * np.conj(f_g)).real
    corr = corr[:2*n-1]
    dx = 2.0 / n
    max_val = np.max(corr) * dx
    return max_val

def random_binary_vector(n):
    """Generate random binary vector with exactly n/2 ones"""
    h = np.zeros(n)
    ones_indices = random.sample(range(n), n//2)
    h[ones_indices] = 1
    return h
