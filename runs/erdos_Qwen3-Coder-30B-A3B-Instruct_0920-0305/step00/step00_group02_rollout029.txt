def evaluate_c5(h):
    """Evaluate the C5 constant for a given h function."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all k values
    correlation = np.correlate(h, 1 - h, mode="full")
    # The full correlation has length 2*n_points - 1
    # We want the maximum value (which occurs at k=0, k=1, ..., k=n_points-1)
    # but actually the max is across all possible k values (0 to 2).
    # When k is in [0,2], the valid indices for correlation are 
    # from offset = n_points-1 to n_points-1 + (n_points-1) = 2*n_points - 2
    # which spans from (n_points-1) downto (n_points-1) + (n_points-1) = 2*n_points - 2
    # The relevant maximum is indexed by k = 0 to 2*(n_points-1) but we want 
    # the maximum over the full range of correlation
    # We want: max_{k in [0,2]} integral over [0,2-k] of h(x) * (1 - h(x+k)) dx
    
    # For floating point interpretation:
    # correlation index j corresponds to k = (j - (n_points-1)) * dx
    # So for k in [0,2], we have j in [n_points-1, n_points-1 + 2*(n_points-1)] = [n_points-1, 3*n_points-3]
    # The max is over all valid indices, but only those corresponding to k in [0,2] 
    # Or we can just say, we take the maximum of the correlation, scaled by dx
    max_corr = np.max(correlation)
    return max_corr * dx

def evaluate_c5_full(h):
    """Evaluates C5 as max of all correlations, corresponding to 
       each possible k shift value from range [0,2]"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlation between h and (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Full correlation has length 2*n_points - 1
    # The original problem states that the maximum of the integral is
    # taken over all k values in [0,2], which means we take 
    # the maximum across overlapping sections where:
    # 0 <= k <= 2  
    # So k at each shifted index j includes all shifts:
    # When correlation index j corresponds to k = (j - (n_points - 1)) * dx 
    # The valid k range is [0,2] which gives us valid j indices from:
    # 0 = (j - (n_points-1)) * dx    =>  j = (n_points-1) 
    # 2 = (j - (n_points-1)) * dx    =>  j = (n_points-1) + (2/dx) = (n_points-1) + 2*(n_points/2) = (n_points-1) + n_points = 2*n_points - 1
    # But correlation is of length 2*n_points - 1, indexed from 0 to 2*n_points - 2
    # We are looking at middle part of correlation with indices k=0..2
    # Thus valid indices of correlation between k=0 and k=2, i.e., when 
    # 0 <= (j - (n_points-1)) * dx <= 2, 
    # 0 <= j - (n_points-1) <= 2/dx = 2*(n_points/2) = n_points,  
    # (n_points-1) <= j <= (n_points-1) + n_points = 2*n_points - 1 
    # But only valid indices go up to 2*n_points - 2 (= full length - 1).
    # So we consider indices from (n_points-1) to (2*n_points-2)
    valid_indices = slice(n_points-1, 2*n_points-1)
    # If we want to limit to actual k values from [0, 2], let's actually 
    # filter indices appropriately. But since 2*n_points - 2 < 2*n_points - 1 in some cases,
    # we'll just use the real correlation values over the middle, which computes correct max over valid shift.
    valid_corr = correlation[valid_indices]
    max_corr = np.max(valid_corr)
    return max_corr * dx
