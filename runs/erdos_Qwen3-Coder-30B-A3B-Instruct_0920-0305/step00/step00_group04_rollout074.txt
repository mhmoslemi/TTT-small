import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import math

def initial_h_values(n_points):
    # Create an initial step function
    h = np.ones(n_points)
    # Set first half to 0.5 and second half to 0 to satisfy integral constraint
    half = n_points // 2
    h[:half] = 0.0
    h[half:] = 1.0
    # Normalize
    h *= 2.0 / n_points
    return h

def evaluate_c5(h):
    """Compute the C5 value for given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlation for all lags
    corr = correlate(h, 1 - h, mode='full')
    # Only consider the valid overlap range [0, 2]
    # This corresponds to the middle part of the full correlation
    mid = len(corr) // 2
    valid_correlation = corr[mid : mid + n_points]
    return np.max(valid_correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 42  # Start with reasonable size
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Start with initial construction
    h = initial_h_values(n_points)
    current_c5 = evaluate_c5(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    # Optimization parameters
    max_iter = 1000
    iterations = 0
    
    def objective(h_flat):
        # Reshape and normalize
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        # Apply integral constraint
        current_sum = np.sum(h) * dx
        if current_sum > 0:
            h = h * (1.0 / current_sum) * 0.5  # Normalize to make sum = 1
        # Evaluate
        c5_val = evaluate_c5(h)
        return c5_val
    
    # Define constraints for optimization
    def constraint1(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * dx - 1.0  # Should be 0
    
    def constraint2(h_flat):
        h = h_flat.reshape(n_points)
        return 1 - np.sum(h) * dx  # Should be >= 0
    
    constraints = [
        {'type': 'eq', 'fun': constraint1},
        {'type': 'ineq', 'fun': constraint2}
    ]
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Local search approach
    while time.time() - start_time < budget_s and iterations < max_iter:
        # Create a perturbed version of the current best solution
        h_new = best_h.copy()
        # Random perturbation
        i = np.random.randint(0, n_points)
        h_new[i] = np.clip(h_new[i] + np.random.normal(0, 0.05), 0, 1)
        
        # Re-normalize to satisfy integral constraint
        current_sum = np.sum(h_new) * dx
        if current_sum > 0:
            h_new = h_new * (1.0 / current_sum) * 0.5
            
        # Evaluate
        c5_val = evaluate_c5(h_new)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_new.copy()
            
        iterations += 1
        # Early stopping if improvement is negligible
        if iterations % 50 == 0:
            if best_c5 < 0.3795:  # Early exit if acceptable
                break

    # Final validation
    if best_h is not None:
        # Re-normalize one last time
        current_sum = np.sum(best_h) * dx
        if current_sum > 0:
            best_h = best_h * (1.0 / current_sum) * 0.5
            
        validated_c5 = evaluate_c5(best_h)
        return (best_h, validated_c5, n_points)
    else:
        # Return initial solution if no better found
        h = initial_h_values(n_points)
        return (h, evaluate_c5(h), n_points)
