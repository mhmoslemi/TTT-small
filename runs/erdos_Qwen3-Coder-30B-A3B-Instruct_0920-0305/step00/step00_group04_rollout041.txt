import numpy as np
import time
import random

def compute_c5_fast(h):
    n = len(h)
    size = 1 << ((2*n-1).bit_length())
    f_h = np.fft.fft(h, size)
    f_g = np.fft.fft(1 - h, size)
    corr = np.fft.ifft(f_h * np.conj(f_g)).real
    corr = corr[:2*n-1]
    dx = 2.0 / n
    max_val = np.max(corr) * dx
    return max_val

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr) * dx
    return max_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    best_c5 = float('inf')
    best_h = None
    n_points = 60  # Use 60 points to balance speed and accuracy
    dx = 2.0 / n_points
    max_count = int(0.3807 * n_points / 2)  # Maximum acceptable integer count for target C5

    # Simulated Annealing parameters
    initial_temp = 0.1
    final_temp = 1e-6
    iterations = 0
    max_iterations = 200000

    start_time = time.time()
    
    # Initialize with random binary pattern (exactly 30 ones)
    h = np.zeros(n_points)
    ones_indices = random.sample(range(n_points), n_points // 2)
    h[ones_indices] = 1.0

    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()

    # Randomized hill climbing with simulated annealing
    while iterations < max_iterations and (time.time() - start_time) < budget_s - 0.1:
        iterations += 1
        
        # Random swap between a 1 and a 0
        ones = np.where(current_h > 0.5)[0]
        zeros = np.where(current_h < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            continue
            
        i = random.choice(ones)
        j = random.choice(zeros)
        
        # Swap values
        current_h[i], current_h[j] = current_h[j], current_h[i]
        
        # Compute new C5
        new_c5 = evaluate_c5(current_h)
        
        # Compute acceptance probability with temperature
        elapsed_time = time.time() - start_time
        temperature = initial_temp * (final_temp / initial_temp) ** (elapsed_time / budget_s)
        
        if new_c5 < current_c5 or random.random() < np.exp(-(new_c5 - current_c5) / max(temperature, 1e-10)):
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        else:
            # Revert the swap
            current_h[i], current_h[j] = current_h[j], current_h[i]
    
    if best_h is None:
        best_h = h

    return (best_h, best_c5, n_points)
