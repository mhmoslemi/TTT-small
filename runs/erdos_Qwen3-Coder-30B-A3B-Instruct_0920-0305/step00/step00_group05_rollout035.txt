import numpy as np
from scipy.optimize import differential_evolution
from scipy.optimize import minimize
from math import sqrt
import time

def evaluate_erdos_solution(h, n_points=70):
    """Evaluate the Erdős constant for given h values"""
    dx = 2.0 / n_points
    # Compute the correlation integral
    correlation = np.correlate(h, 1 - h, mode='full')
    # Find maximum value
    max_corr = np.max(correlation)
    return max_corr * dx

def initial_h_values():
    """Return a good starting point - the initial construction"""
    # Using the known good solution from the problem description
    n_points = 70
    h = np.ones(n_points) * 0.5
    # Set first 35 points to 1 and last 35 points to 0 to satisfy constraint
    h[:35] = 1.0
    h[35:] = 0.0
    return h

def objective(h, n_points=70):
    """Objective function to minimize"""
    dx = 2.0 / n_points
    # Compute the correlation integral
    correlation = np.correlate(h, 1 - h, mode='full')
    # Find maximum value and multiply by dx
    max_corr = np.max(correlation)
    return max_corr * dx

def constraint_sum(h, n_points=70):
    """Constraint that sum(h) * dx = 1, equivalent to sum(h) == n_points/2"""
    dx = 2.0 / n_points
    return np.sum(h) * dx - 1.0

def run(seed=42, budget_s=600, **kwargs):
    """Run optimization to find better h values for Erdős constant C5"""
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Use smaller problem sizes for faster optimization
    n_points = 70
    dx = 2.0 / n_points
    
    # Initial point
    h0 = initial_h_values()
    
    # Define bounds (0 <= h[i] <= 1)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints
    cons = {'type': 'eq', 'fun': constraint_sum, 'args': (n_points,)}
    
    # Optimization approach: Differential evolution with constraints
    # Start with a simpler approach using scipy's minimize
    options = {'maxiter': 500, 'disp': False}
    best_h = h0.copy()
    best_c5 = objective(h0, n_points)
    
    # Use a hybrid optimization: differential evolution + local minimizer
    # First perform a fast multi-start optimization
    for _ in range(10):  # Multiple restarts with different seeds
        if time.time() - start_time > budget_s - 5:  # leave buffer
            break
        # Slightly perturb initial solution
        perturbation = np.random.normal(scale=0.05, size=n_points)
        h0_perturbed = np.clip(h0 + perturbation, 0, 1)
        
        # Project to satisfy constraint
        current_sum = np.sum(h0_perturbed) * dx
        h0_perturbed = h0_perturbed * (n_points / 2) / current_sum
        
        # Local optimization
        res = minimize(objective, h0_perturbed, method='L-BFGS-B', 
                      bounds=bounds, constraints=cons, options=options)
        
        if res.success:
            h_test = res.x
            c5_test = objective(h_test, n_points)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()
    
    # Final cleanup
    best_c5 = objective(best_h, n_points)
    
    return (list(best_h), best_c5, n_points)
