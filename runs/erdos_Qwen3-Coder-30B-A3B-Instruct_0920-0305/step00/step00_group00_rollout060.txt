from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the correlation integral
    conv = np.correlate(h, 1 - h, mode='full')
    # The maximum value is what we want
    max_val = np.max(conv) * dx
    return max_val
