import numpy as np
import time
from scipy.optimize import minimize
from math import sqrt

def project_to_feasible(v):
    """Project vector v onto the feasible set: sum = n_points/2, 0 <= v[i] <= 1."""
    n_points = len(v)
    target_sum = n_points / 2.0
    v = np.array(v)
    
    # Simple projection onto simplex with box constraints [0,1]
    # We'll use iterative approach: first project onto box, then onto sum constraint
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    
    if abs(current_sum - target_sum) < 1e-10:
        return v
    
    # Simple iterative projection
    # If sum < target_sum, we want to increase some values
    # If sum > target_sum, we want to decrease some values
    
    # We do simple binary search for the lambda
    # So we try to find lambda such that 
    # sum_i = clip(v_i - lambda, 0, 1) = target_sum
    # Problem is convex, we can solve it.
    
    # More direct approach: 
    # sort in descending order, and adjust all values below a cutoff
    idx = np.argsort(v)[::-1]  # sorted descending
    if current_sum < target_sum:
        # Need to increase, we add to smallest entries
        # to preserve order of difference, first increase all until total hits target
        v_sorted = v[idx]
        new_sum = 0
        for i in range(n_points):
            if new_sum + (1 - v_sorted[i]) < target_sum - current_sum:
                new_sum += 1 - v_sorted[i]
                v_sorted[i] = 1
            else:
                delta = target_sum - current_sum - new_sum
                v_sorted[i] += delta
                new_sum = target_sum - current_sum
                for j in range(i+1, n_points):
                    v_sorted[j] = 1
                break
        # Restore order
        v[idx] = v_sorted
    else:
        # Need to decrease
        v_sorted = v[idx]
        new_sum = 0
        for i in range(n_points):
            if new_sum + v_sorted[i] - 0 < current_sum - target_sum:
                new_sum += v_sorted[i] 
                v_sorted[i] = 0
            else:
                delta = current_sum - target_sum - new_sum
                v_sorted[i] -= delta
                new_sum = current_sum - target_sum
                for j in range(i+1, n_points):
                    v_sorted[j] = 0
                break
        v[idx] = v_sorted

    # Reprojecting in different approach for safety:
    # apply clamping on each step 0-1 individually and ensure sum matches
    v = np.clip(v, 0, 1)
    # Correct sum 
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) > 1e-10:
        # Distribute too much or too little evenly, or adjust 
        diff = target_sum - current_sum
        if diff > 0:
            # Increase by small amount equally to all, keeping under 1
            # Per element increments
            step = min(1.0 - np.max(v), diff / len(v))
            v += step
            diff -= step * len(v) 
            # Now corrected excess will be corrected
            while abs(diff) > 1e-10:
                step = min(1.0 - np.max(v), diff / len(v))
                if step <= 0: break
                v += step
                diff -= step * len(v)
        else:
            # Decrease all by equal steps
            step = min(np.min(v), -diff / len(v))
            if step > 0:
                v -= step
                diff += step * len(v)
                # Now corrected
                while abs(diff) > 1e-10 and max(v) > 0:
                    # Adjust more carefully
                    if diff < 0:
                        step = min(np.min(v), -diff)
                        if step > 0:
                            v -= step
                            diff += step * len(v)
                    else:
                        break
        # Final check
        v = np.clip(v, 0, 1)
        
    return v

def evaluate_erdos_solution(h):
    """Compute C_5 = max_k ∫ h(x)(1 - h(x+k)) dx"""
    # Normalize dx 
    n_points = len(h)
    dx = 2.0 / n_points
    # Evaluate correlation using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # Only consider valid shifts that overlap
    corr = corr * dx
    # Return max correlation value (which represents max O(k) = 1 - min C(k))
    return np.max(corr)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the given initial values if possible
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        # Default initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Debug: try alternating for start (since it's simple and infeasible, project)
    
    # Ensure sum is correct (n_points / 2.0)  
    current_sum = np.sum(h)
    h = h * (0.5 * len(h)) / current_sum

    n_points = len(h)
    dx = 2.0 / n_points
    
    # Evaluate initial
    best_val = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T_start = 0.01
    T_min = 1e-8
    cooling_rate = 0.99  # will decay T gradually  
    max_iter = 500000  # Number of search steps to make
    
    iter_count = 0
    T = T_start
    
    iter_since_improve = 0
    max_iter_since_improve = 10000
    
    while time.time() - start_time < budget_s - 0.1 and iter_count < max_iter:
        # Make a small random change: pick two distinct indices and try to move some mass
        i, j = np.random.choice(n_points, size=2, replace=False)
        
        # Get current values
        h_i = h[i]
        h_j = h[j]
        
        # Determine allowed delta
        delta_min = max(-h_i, h_j - 1.0)
        delta_max = min(1.0 - h_i, h_j)
        
        if abs(delta_max - delta_min) < 1e-10:
            iter_count += 1
            continue
        
        # Sample delta
        delta = np.random.uniform(delta_min, delta_max)
        
        # Make change
        new_h = h.copy()
        new_h[i] = h[i] + delta
        new_h[j] = h[j] - delta
        
        # Ensure feasible - we will project
        new_h = project_to_feasible(new_h)
        
        # Check objective
        new_val = evaluate_erdos_solution(new_h)
        
        # Accept or reject
        if new_val < best_val:
            # Always accept if improvement
            h = new_h
            best_val = new_val
            best_h = new_h.copy()
            iter_since_improve = 0
        elif np.random.rand() < np.exp(-(new_val - best_val) / T):
            # Accept with probability based on temperature
            h = new_h
            iter_since_improve = 0
        else:
            iter_since_improve += 1
        
        # Cooling schedule - reduce T gradually  
        if iter_count % 1000 == 0:
            if T > T_min:
                T *= cooling_rate
        
        # Periodically check if we should stop improving after long idle
        if iter_since_improve > max_iter_since_improve:
            break
            
        iter_count += 1
    
    return (best_h, best_val, len(best_h))
