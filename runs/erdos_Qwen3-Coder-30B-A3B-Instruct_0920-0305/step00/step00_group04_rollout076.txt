import numpy as np
import time
import cvxpy as cp
from scipy import fftpack
from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    # Computes the exact C5 value using the provided function
    return max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def get_fft_plan(n):
    # Returns the FFT plan and padding size
    L = 1
    while L < 2 * n - 1:
        L *= 2
    return L

def compute_correlation_fft(h, L):
    # Compute correlation using FFT for speed
    h_padded = np.pad(h, (0, L - len(h)), 'constant')
    g_padded = np.pad(1 - h, (0, L - len(h)), 'constant')
    H = fftpack.fft(h_padded)
    G = fftpack.fft(g_padded)
    corr = fftpack.ifft(H * np.conj(G)).real
    return corr[:2 * len(h) - 1]

def compute_c5_bound(h, dx):
    # Computes the maximum overlap integral using FFT
    L = get_fft_plan(len(h))
    corr = compute_correlation_fft(h, L)
    return np.max(corr) * dx

def binary_to_float(b, n):
    # Converts binary vector to float vector, preserving sum
    h = b.astype(float)
    current_sum = np.sum(h)
    target_sum = n / 2
    scale = target_sum / current_sum
    return np.clip(h * scale, 0, 1)

def swap_move(b, i, j):
    # Perform a swap between positions i and j
    b = b.copy()
    b[i], b[j] = b[j], b[i]
    return b

def binary_sa_step(b, T, L, dx):
    # Performs one SA step on binary vector
    n = len(b)
    # Choose two indices to swap
    i = np.random.randint(0, n)
    while b[i] == 0:
        i = np.random.randint(0, n)
    j = np.random.randint(0, n)
    while b[j] == 1:
        j = np.random.randint(0, n)
    new_b = swap_move(b, i, j)
    c_old = compute_c5_bound(binary_to_float(b, n), dx)
    c_new = compute_c5_bound(binary_to_float(new_b, n), dx)
    delta = c_new - c_old
    if delta <= 0 or np.random.rand() < np.exp(-delta / T):
        return new_b, c_new
    else:
        return b, c_old

def binary_sa_restart(seed, n, budget_s):
    # Perform a restart of simulated annealing on binary vectors
    np.random.seed(seed)
    L = get_fft_plan(n)
    dx = 2.0 / n
    # Initialize binary vector with n/2 ones
    b = np.zeros(n, dtype=int)
    b[:n//2] = 1
    np.random.shuffle(b)
    best_b = b.copy()
    best_c5 = compute_c5_bound(binary_to_float(b, n), dx)
    T0 = 0.05
    alpha = 0.9995
    start = time.time()
    it = 0
    while time.time() - start < budget_s:
        T = T0 * (alpha ** it)
        new_b, new_c5 = binary_sa_step(best_b, T, L, dx)
        if new_c5 < best_c5:
            best_b = new_b
            best_c5 = new_c5
        it += 1
    return best_b, best_c5

def project_to_simplex(y, target_sum=1.0):
    # Project point y onto the simplex with sum = target_sum
    # Using bisection to find the correct lambda
    y = np.asarray(y)
    low = np.min(y) - 1.0
    high = np.max(y) + 1.0
    tol = 1e-12
    while high - low > tol:
        mid = (low + high) / 2.0
        x = np.clip(y - mid, 0, 1)
        s = np.sum(x)
        if s >= target_sum:
            low = mid
        else:
            high = mid
    x = np.clip(y - low, 0, 1)
    x = x * target_sum / np.sum(x)
    return x

def step_gradient_descent(h, L, dx, gamma=5.0, iterations=20):
    # Perform a few iterations of projected gradient descent using soft-max surrogate
    n = len(h)
    h = np.asarray(h).copy()
    for i in range(iterations):
        # Compute correlation values f_l for all lags
        corr = compute_correlation_fft(h, L)
        f = corr * dx
        # Soft-max
        exp_f = np.exp(gamma * f)
        s = np.sum(exp_f)
        w = exp_f / s
        # Gradient computation: d/dh_i = sum_l w_l * d/dh_i f_l
        grad = np.zeros(n)
        for l in range(1 - n, n):
            if l > 0:
                # Forward difference terms
                for i in range(n - l):
                    grad[i] += w[l + n - 1] * (1 - h[i + l])
                # Backward difference terms (offset)
            else:
                # Backward difference terms
                for i in range(-l, n):
                    grad[i] += w[l + n - 1] * (-h[i + l])  # similar to bicgstab
        # Gradient step
        alpha = 0.05 * dx
        h_new = h - alpha * grad
        h_new = project_to_simplex(h_new)
        # Evaluate new C5
        c5_new = compute_c5_bound(h_new, dx)
        # If improved, accept
        if c5_new < np.max(compute_correlation_fft(h, L)) * dx:
            h = h_new
        else:
            break
    return h

def run(seed=42, budget_s=600, **kwargs):
    start = time.time()
    n = 60  # Try n=60 first for balance of speed and accuracy
    dx = 2.0 / n
    t_phaseA = 0.7 * budget_s
    t_phaseB = 0.25 * budget_s
    t_milp = 0.05 * budget_s
    best_c5 = float('inf')
    best_h = None
    # Phase A: Binary SA with multiple restarts
    restarts = 0
    while time.time() - start < t_phaseA:
        remaining = t_phaseA - (time.time() - start)
        if remaining < 1:
            break
        new_b, new_c5 = binary_sa_restart(seed + restarts, n, min(remaining, 30))
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_b = new_b
        restarts += 1
    # Convert best binary to floating point
    h_float = binary_to_float(best_b, n)
    # Phase B: Gradient descent refinement
    L = get_fft_plan(n)
    if time.time() - start < t_phaseA + t_phaseB:
        h_tmp = step_gradient_descent(h_float, L, dx, gamma=5.0, iterations=30)
        c5_after = compute_c5_bound(h_tmp, dx)
        if c5_after < best_c5:
            best_h = h_tmp
            best_c5 = c5_after
    # Final projection to enforce sum = n/2
    if best_h is None:
        best_h = h_float
    best_h = project_to_simplex(best_h)
    # Verify with exact computation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, final_c5, n)
