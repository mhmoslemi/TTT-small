import numpy as np
import time

def evaluate_erdos_solution(h):
    """Evaluates the ERDOS constant given a step function h."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    C5 = np.max(c) * dx
    return C5

def initial_h_values(n):
    """Generate the initial Beatty-sequence based step function."""
    beta = (np.sqrt(5) - 1) / 2
    h = np.zeros(n)
    for i in range(n):
        if (i * beta) % 1 < 0.5:
            h[i] = 1
    # Adjust to make sum = n/2
    s = np.sum(h)
    if s > n/2:
        # find the indices with 1s to flip to 0
        idx_ones = np.where(h == 1)[0]
        num_to_flip = int(s - n/2)
        for i in range(num_to_flip):
            h[idx_ones[i]] = 0
    elif s < n/2:
        # find the indices with 0s to flip to 1
        idx_zeros = np.where(h == 0)[0]
        num_to_flip = int(n/2 - s)
        for i in range(num_to_flip):
            h[idx_zeros[i]] = 1
    return h

def project_onto_feasible(h, n_half):
    """Project h onto the set {h: 0 <= h_i <= 1, sum(h) = n_half}."""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - n_half) < 1e-12:
        return h
    # Water-filling approach
    delta = n_half - s
    if abs(delta) < 1e-12:
        return h
    # solve for lambda such that clip(h + lambda, 0, 1) has sum = n_half
    # к = sum(clip(h_i + lambda, 0, 1) - h_i)
    # simplify: for i such that h_i + lambda is in [0, 1], k_i = lambda
    # for those not in [0,1], they contribute 0 or -lambda or +lambda
    # Simple iterative method instead:
    # Try to adjust values minimally
    h -= delta / n_half
    h = np.clip(h, 0, 1)
    # Re-solve for exact sum using bisection search:
    a = -1.0
    b = 1.0
    while b - a > 1e-10:
        mid = (a + b) / 2
        temp_h = np.clip(h + mid, 0, 1)
        sum_h = np.sum(temp_h)
        if sum_h - n_half > 0:
            b = mid
        else:
            a = mid
    final_h = np.clip(h + a, 0, 1)
    return final_h

def compute_gradients(h, S_k_vals, alpha):
    """Compute the gradient vector of the smooth surrogate."""
    n = len(h)
    w = np.exp(-alpha * S_k_vals)
    W = np.sum(w)
    if W < 1e-16:
        return np.zeros_like(h)
    # Compute ∂S_k/∂h_i
    # Each ∂S_k/∂h_i = h_{i+k} + h_{i-k} (check bounds)
    # Time-consuming via explicit loop for each k, better via convolution:
    # we compute ∂S_k/h_i = [h rev]_{i+k} + [h rev]_{i-k}
    # That is: 
    # For i=0.. n-1 -k: h[i+k]
    # For i=0.. n-1 +k: h[i-k] 
    # So compute via two convolutions.
    grad = np.zeros_like(h)
    # Compute ∂S_k/∂h = h * w_rev (but using correct index alignment)
    h_rev = h[::-1]
    # Expanded version manually:
    for k in range(n):   # k goes from 0 to n-1
        der = np.zeros_like(h)
        for i in range(n-k):
            der[i] += h_rev[i+k]  # h[i+k] contribution
        for i in range(k, n):
            der[i] += h_rev[i-k]  # h[i-k] contribution
        grad += w[k] * der
    return grad / W

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n = 300  # sample size, even number
    n_half = n / 2
    max_iter = 100000
    dx = 2.0 / n
    alpha = 1.0
    alpha_max = 10000
    step_size = 1.0
    iter_per_alpha = 200
    max_hill_trials = int(0.3 * budget_s * 200)

    # Initial h from low-discrepancy sequence
    h = initial_h_values(n)
    h = project_onto_feasible(h, n_half)
    h_best = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    best_phi = float('-inf')
    
    # Timing
    start_time = time.time()
    elapsed = 0
    
    # Gradient ascent loop
    for iteration in range(max_iter):
        if elapsed >= 0.6 * budget_s:
            break
        elapsed = time.time() - start_time
        
        # Compute S_k values
        c = np.correlate(h, h, mode="full")
        S_k_vals = np.array([c[n-1+j] for j in range(n)])  # First n terms of self-correlation
        S_k_vals *= dx  # get ΔS_k in real units
        
        # Maximal min value before transformation
        min_S = np.min(S_k_vals)
        
        # Smooth surrogate objective Φ_α (We want to maximize this)
        # Don't compute on small α
        if alpha < 1e-6:
            t_s = np.exp(-alpha * np.array(S_k_vals))
            Phi = - (1/alpha) * np.log(np.sum(t_s))
        else:
            t_s = np.exp(-alpha * np.array(S_k_vals))
            Phi = - (1/alpha) * np.log(np.sum(t_s))
        
        # Use derivative approximation from paper
        dPhi = np.zeros_like(h)
        w = np.exp(-alpha * np.array(S_k_vals))  # Note: we use direct computation for gradient
        W = np.sum(w)
        if W < 1e-16:
            w = np.ones_like(w)  # avoid division by zero
            W = len(w)
        # Reuse FFT for more robustness:
        h_rev = h[::-1]
        # Compute quantized gradient estimates (for k=0..n-1):
        grad_part = np.zeros_like(h)
        for kidx in range(n):
            temp = np.zeros_like(h)
            if kidx < n:
                temp[kidx:] = h[:-kidx]
            if kidx > 0:
                temp[:-kidx] += h[kidx:]
            grad_part += w[kidx] * temp
        dPhi = grad_part / W
        
        old_h = h.copy()
        h += 0.1 * dPhi  # line search not needed by general rule
        h = project_onto_feasible(h, n_half)
        
        # Update α if needed
        if iteration % iter_per_alpha == 0 and alpha < alpha_max:
            alpha = min(alpha*2.0, alpha_max)
        
        # Track best solution
        current_C5 = evaluate_erdos_solution(h)
        if current_C5 < best_C5:
            h_best = h.copy()
            best_C5 = current_C5
        
        # Housekeeping
        np.random.seed(iteration + seed)  # Temp fix for randomness
        if elapsed >= 0.6 * budget_s:
            break
    
    # Polishing step using randomized hill climbing on binary function
    start_polish = time.time()
    
    # Start from best continuous h
    h_rounded = np.zeros(n)
    # Set threshold such that exactly n_half ones are chosen
    sorted_h = np.sort(h_best)[::-1]
    threshold = sorted_h[int(n_half)]
    h_rounded = (h_best >= threshold).astype(float)
    # ensure sum is exactly n_half
    sum_current = np.sum(h_rounded)
    if sum_current < n_half:
        idx_zeros = np.where(h_rounded == 0)[0]
        for i in range(int(n_half - sum_current)):
            h_rounded[idx_zeros[i]] = 1
    elif sum_current > n_half:
        idx_ones = np.where(h_rounded == 1)[0]
        for i in range(int(sum_current - n_half)):
            h_rounded[idx_ones[i]] = 0
    h_final = h_rounded.copy()
    best_binary_C5 = evaluate_erdos_solution(h_final)
    
    # Hill climbing
    trial_count = 0
    while time.time() - start_polish < 0.3 * budget_s and trial_count < max_hill_trials:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        if h_final[i] == h_final[j]:
            continue
        # Swap bit
        h_new = h_final.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        c_new = evaluate_erdos_solution(h_new)
        if c_new < best_binary_C5:
            h_final = h_new
            best_binary_C5 = c_new
        trial_count += 1
    
    final_c5 = evaluate_erdos_solution(h_final)
    
    return (h_final.tolist(), float(final_c5), n)
