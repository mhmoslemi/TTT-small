import numpy as np
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant C5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using FFT for efficiency
    size = 1 << ((2*n-1).bit_length())
    Fh = np.fft.fft(h, size)
    Fg = np.fft.fft(1 - h, size)
    corr = np.fft.ifft(Fh * np.conj(Fg)).real
    # Take first 2*n-1 elements corresponding to full correlation
    corr = corr[:2*n-1]
    max_val = np.max(corr) * dx
    return max_val

def compute_c5_fast(h):
    """Fast computation of C5 using FFT"""
    n = h.shape[0]
    size = 1 << ((2*n-1).bit_length())
    f_h = np.fft.fft(h, size)
    f_g = np.fft.fft(1 - h, size)
    corr = np.fft.ifft(f_h * np.conj(f_g)).real
    corr = corr[:2*n-1]
    dx = 2.0 / n
    max_val = np.max(corr) * dx
    return max_val

def random_binary_vector(n):
    """Generate a random binary vector with exactly n//2 ones"""
    h = np.zeros(n)
    ones_indices = np.random.choice(n, size=n//2, replace=False)
    h[ones_indices] = 1
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    n_points = 60
    
    # Use a binary approach with simulated annealing
    max_iter = 200000
    
    # Simulated Annealing parameters
    T0 = 0.05
    T_min = 1e-6
    decay_rate = 0.9995
    
    # Initialize statistics tracking
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 0.2 and iter_count < max_iter:
        # Start with a random binary vector
        h = random_binary_vector(n_points)
        current_c5 = compute_c5_fast(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Begin Simulated Annealing
        T = T0
        iter_since_improvement = 0
        
        while time.time() - start_time < budget_s - 0.2 and iter_since_improvement < 50000 and iter_count < max_iter:
            # Find indices with 1s and 0s
            ones_idx = np.where(h > 0.5)[0]
            zeros_idx = np.where(h < 0.5)[0]
            
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
            
            # Choose two indices to swap
            i = np.random.choice(ones_idx)
            j = np.random.choice(zeros_idx)
            
            # Swap values
            h[i], h[j] = h[j], h[i]
            
            # Compute new C5
            new_c5 = compute_c5_fast(h)
            
            # Accept or reject
            delta = new_c5 - current_c5
            
            if delta <= 0 or random.random() < np.exp(-delta / T):
                current_c5 = new_c5
                iter_since_improvement = 0
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    last_improvement = iter_count
            else:
                # Revert swap
                h[i], h[j] = h[j], h[i]
            
            iter_count += 1
            iter_since_improvement += 1
            
            # Decay temperature
            T = max(T_min, T * decay_rate)
    
    # If no solution found, work with initial_h_values if exists
    if best_h is None:
        if 'initial_h_values' in globals():
            best_h = initial_h_values.copy()
            best_c5 = evaluate_erdos_solution(best_h)
        else:
            # Fallback solution
            h = random_binary_vector(n_points)
            best_c5 = compute_c5_fast(h)
            best_h = h
    
    return (best_h, best_c5, n_points)
