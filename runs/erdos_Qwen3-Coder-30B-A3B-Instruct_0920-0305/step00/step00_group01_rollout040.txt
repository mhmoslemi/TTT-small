def project_to_feasible(v, target_sum):
    """Project vector v onto the set of vectors with sum = target_sum and 0 <= v[i] <= 1"""
    # Use the method of Lagrange multipliers with clipping
    v = np.array(v, dtype=float)
    
    # Simple project: make sure all v[i] are in [0,1], adjust sum
    v = np.clip(v, 0, 1)
    
    # Adjust to force correct sum
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-10:
        return v
    
    # Determine how much to adjust
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase some values
        # Sort indices by current value (to increase smallest first)
        idx = np.argsort(v)
        for i in idx:
            if diff <= 0:
                break
            # Increase v[i] by as much as possible without exceeding 1
            increase = min(diff, 1 - v[i])
            v[i] += increase
            diff -= increase
    else:
        # Need to decrease some values  
        # Sort indices by current value (to decrease largest first)
        idx = np.argsort(v)[::-1]
        for i in idx:
            if diff >= 0:
                break
            # Decrease v[i] by as much as possible without going below 0
            decrease = min(-diff, v[i])
            v[i] -= decrease
            diff += decrease
    
    return v

def compute_gradient(h, max_shift_idx, dx):
    """Compute gradient of O(max_shift_idx) w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    
    offset = max_shift_idx - (n - 1)
    
    # For max shift idx, compute gradient:
    # O(k) = 1 - sum_i h[i]*h[i+offset] * dx
    # dO/dh_j = -dx * (h[j+offset] + h[j-offset])  
    # (where neighbors may not exist)
    
    for i in range(n):
        neighbor_sum = 0
        if i + offset < n and i + offset >= 0:
            neighbor_sum += h[i + offset]
        if i - offset < n and i - offset >= 0:
            neighbor_sum += h[i - offset]
        
        grad[i] = -dx * neighbor_sum
    
    return grad
