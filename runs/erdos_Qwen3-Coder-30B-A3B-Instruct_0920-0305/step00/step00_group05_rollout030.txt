import numpy as np
import time
import random

def compute_c5(h):
    """Compute C5 bound for given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    """Find improved upper bound for Erdős minimum overlap problem constant C₅"""
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    time_end = start + budget_s - 0.5
    
    # Use 200 points as a good balance between computational efficiency and solution quality
    n_points = 200
    best_h = None
    best_c5 = float('inf')
    dx = 2.0 / n_points
    
    # Binary optimization with simulated annealing
    max_restarts = 30
    max_local_iters = 20000
    
    for restart in range(max_restarts):
        if time.time() >= time_end:
            break
            
        # Generate random binary vector with exactly half ones
        h = np.zeros(n_points, dtype=np.float64)
        ones = np.random.choice(n_points, n_points // 2, replace=False)
        h[ones] = 1.0
        
        cur_c5 = compute_c5(h)
        
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            
        # Local simulated annealing
        T0 = 0.05
        for it in range(max_local_iters):
            if time.time() >= time_end:
                break
                
            # Choose two different indices
            idx1 = np.random.randint(n_points)
            idx2 = np.random.randint(n_points)
            while h[idx1] == h[idx2]:
                idx1 = np.random.randint(n_points)
                idx2 = np.random.randint(n_points)
                
            # Perform swap
            h[idx1], h[idx2] = h[idx2], h[idx1]
            
            new_c5 = compute_c5(h)
            d = new_c5 - cur_c5
            
            # Temperature schedule
            T = max(1e-6, T0 * (1 - it / max_local_iters))
            
            # Accept or reject
            if d <= 0 or np.random.rand() < np.exp(-d / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                # Revert swap
                h[idx1], h[idx2] = h[idx2], h[idx1]
    
    # Ensure the final result is returned
    return (best_h.tolist(), best_c5, n_points)
