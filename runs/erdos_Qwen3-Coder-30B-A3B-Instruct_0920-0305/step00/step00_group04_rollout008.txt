def build_milp_model(n, target_c5):
    # The constraints:
    # minimize t
    # subject to:
    #   sum(b) == n/2
    #   for each lag ℓ != 0:
    #     w[ℓ][i] <= b[i]
    #     w[ℓ][i] <= b[i+ℓ]
    #     w[ℓ][i] >= b[i] + b[i+ℓ] - 1
    #     N_ℓ = sum(b) - sum(w[ℓ])
    #     t >= N_ℓ
    
    # Define variables
    b = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Constraints to keep model tight: symmetry breaking and binary sum
    constraints = [cp.sum(b) == n/2]
    
    # Add symmetry breaking: fix first element to 1
    # and make sequence non-increasing
    constraints.append(b[0] == 1)
    
    # Make sequence non-increasing for symmetry breaking
    for i in range(n-1):
        constraints.append(b[i] >= b[i+1])
    
    # Add auxiliary AND constraints for each lag
    lags = list(range(1, n))  # Positive lags only
    for ℓ in lags:
        # w_ℓ is a vector of size (n-ℓ) 
        w_ell = cp.Variable(n - ℓ, boolean=True)
        
        # Add w constraints
        for i in range(n - ℓ):
            constraints.append(w_ell[i] <= b[i])
            constraints.append(w_ell[i] <= b[i + ℓ])
            constraints.append(w_ell[i] >= b[i] + b[i + ℓ] - 1)
        
        # Compute N_ℓ = sum(b[:n-ℓ]) - sum(w_ell)
        N_ell = cp.sum(b[:n-ℓ]) - cp.sum(w_ell)
        
        # Add constraint: t >= N_ℓ
        constraints.append(t >= N_ell)
    
    # obj: minimize t
    obj = cp.Minimize(t)
    
    prob = cp.Problem(obj, constraints)
    return prob

def compute_c5(h, dx):
    # Compute maximum over all lags of int h(x)(1-h(x+k))dx
    n = len(h)
    # Convolution h * (1-h) for all lag shift k
    # Using full mode to get maximum overlap lengths
    overlaps = np.correlate(h, 1 - h, mode='full')
    C5 = np.max(overlaps) * dx
    return C5
