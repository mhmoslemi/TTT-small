import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import cvxpy as cp

# Helper functions
def evaluate_overlap(h, dx):
    """Fast evaluation of C5 using FFT."""
    n = len(h)
    L = 2**int(np.ceil(np.log2(2*n - 1)))
    H = fft(h, L)
    G = fft(1 - h, L)
    corr = np.real(ifft(H * np.conj(G)))
    corr = corr[:2*n - 1]
    return np.max(corr) * dx

def project_simplex(y, target_sum):
    """Project y onto the capped simplex with sum = target_sum."""
    # Clip to [0,1]
    y = np.clip(y, 0, 1)
    # Ensure sum equals target_sum
    current_sum = np.sum(y)
    if abs(current_sum - target_sum) < 1e-10:
        return y
    # Bisection to find the right lambda
    def sum_at_lambda(lam):
        return np.sum(np.clip(y - lam, 0, 1))
    low, high = np.min(y) - 1, np.max(y) + 1
    while high - low > 1e-10:
        mid = (low + high) / 2
        s = sum_at_lambda(mid)
        if s > target_sum:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2
    x = np.clip(y - lam, 0, 1)
    return x

def soft_max_grad(h, dx, gamma=5.0):
    """Compute gradient of smooth max of correlations."""
    n = len(h)
    L = 2**int(np.ceil(np.log2(2*n - 1)))
    H = fft(h, L)
    G = fft(1 - h, L)
    corr = np.real(ifft(H * np.conj(G)))
    corr = corr[:2*n - 1]
    # Compute soft weights
    exp_gamma_f = np.exp(gamma * corr * dx)
    w = exp_gamma_f / np.sum(exp_gamma_f)
    # Gradient computation
    grad = np.zeros(n)
    for l in range(-(n-1), n):
        if l >= 0:
            # Forward part
            part = (1 - h[l:]) * h[:n-l]
        elif l < 0:
            # Backward part
            part = (1 - h[:l]) * h[-l:]
        # Multiply with weight
        if l >= 0:
            grad[:n-l] += w[l+n-1] * part
        else:
            grad[-l:] += w[l+n-1] * part
    return grad * dx

def gradient_descent_refinement(h, dx, max_iter=500):
    """Refine the solution using projected gradient descent."""
    h = h.copy()
    gamma = 5.0
    alpha = 0.1 * dx
    for i in range(max_iter):
        grad = soft_max_grad(h, dx, gamma)
        # Take step
        h_new = h - alpha * grad
        # Project
        h_new = project_simplex(h_new, len(h) / 2)
        # Check improvement
        c5_before = evaluate_overlap(h, dx)
        c5_after = evaluate_overlap(h_new, dx)
        if c5_after > c5_before:
            alpha *= 0.9
        else:
            h = h_new
            if abs(c5_after - c5_before) < 1e-8:
                break
        # Increase gamma
        if i % 10 == 0:
            gamma *= 1.5
            if gamma > 200:
                gamma = 200
    return h

def binary_sa(n, n_restarts=15, time_per_restart=20, max_time=400):
    """Binary simulated annealing (discrete optimization)."""
    def swap_move(b):
        """Single swap move."""
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            return b.copy()
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        b = b.copy()
        b[i] = 0
        b[j] = 1
        return b

    def acceptance_prob(delta, T):
        if delta <= 0:
            return 1
        return np.exp(-delta / T)

    def compute_c5_binary(b, dx):
        n = len(b)
        L = 2**int(np.ceil(np.log2(2*n - 1)))
        H = fft(b, L)
        G = fft(1 - b, L)
        corr = np.real(ifft(H * np.conj(G)))
        corr = corr[:2*n - 1]
        return np.max(corr) * dx

    best_b = None
    best_c5 = np.inf
    start_time = time.time()
    for _ in range(n_restarts):
        if time.time() - start_time > max_time:
            break
        # Initialize binary
        b = np.zeros(n)
        ones_indices = np.random.permutation(n)[:n//2]
        b[ones_indices] = 1
        c5 = compute_c5_binary(b, 2.0 / n)
        T = 0.05
        step_count = 0
        while time.time() - start_time < max_time:
            step_count += 1
            if T < 1e-4:
                break
            b_new = swap_move(b)
            c5_new = compute_c5_binary(b_new, 2.0 / n)
            delta = c5_new - c5
            if acceptance_prob(delta, T) > np.random.rand():
                b = b_new
                c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_b = b.copy()
            #Cooling
            T *= 0.9995
    return best_b, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 60
    dx = 2.0 / n
    # Phase A: Binary optimization
    b_best, c5_best = binary_sa(n, time_per_restart=20, max_time=0.7 * budget_s)
    if c5_best <= 0.3807:
        # Continue with refinement if needed
        h = b_best.copy().astype(float)
        if time.time() - start_time < 0.9 * budget_s:
            h_final = gradient_descent_refinement(h, dx)
            c5_final = evaluate_overlap(h_final, dx)
            if c5_final < c5_best:
                h = h_final
                c5_best = c5_final
    else:
        # Gradient descent start from best binary
        h = b_best.copy().astype(float)
        h = gradient_descent_refinement(h, dx)
        c5_best = evaluate_overlap(h, dx)
    # Final projection to ensure sum = n/2
    h = project_simplex(h, n / 2)
    c5_best = evaluate_overlap(h, dx)
    return h, c5_best, n
