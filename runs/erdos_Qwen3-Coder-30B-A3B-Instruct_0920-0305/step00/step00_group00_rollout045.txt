def softmax(x):
    """Numerically stable softmax."""
    x = np.array(x)
    x = x - np.max(x)
    exp_x = np.exp(x)
    return exp_x / np.sum(exp_x)

def project_h_to_feasible(h, n):
    """Project h to satisfy 0 <= h[i] <= 1 and sum(h) = n/2 using water filling."""
    h = np.array(h)
    target_sum = n / 2.0
    mean_val = target_sum / n
    
    if np.sum(h) == target_sum:
        return np.clip(h, 0, 1)
        
    # Sort in descending order
    sorted_h = np.sort(h)[::-1]
    
    # Find the threshold
    cumsum = np.cumsum(sorted_h)
    threshold_idx = np.argmax(cumsum <= target_sum)
    if cumsum[threshold_idx] >= target_sum:
        threshold_idx += 1
    
    # Adjust if necessary
    if threshold_idx == 0:
        # All need to be adjusted
        new_h = np.clip(h, 0, 1)
        diff = target_sum - np.sum(new_h)
        if diff > 0:
            # If sum is less than target, increase values
            # but respect clipping
            while diff > 0:
                idx = np.argmax(new_h)
                if new_h[idx] < 1.0:
                    incr = min(diff, 1.0 - new_h[idx])
                    new_h[idx] += incr
                    diff -= incr
                else:
                    break
        return new_h
    else:
        # Some values are pinned at 1 or 0
        # Compute the average of the threshold
        threshold = (target_sum - cumsum[threshold_idx-1]) / (n - threshold_idx)
        if threshold < mean_val:
            # Do not over fill
            threshold = mean_val
        
        # Create a new vector
        new_h = np.copy(h)
        new_h[new_h > 1] = 1.0
        new_h[new_h < 0] = 0.0
        # Now adjust values
        for i in range(n):
            if new_h[i] >= threshold:
                new_h[i] = 1.0
            elif new_h[i] <= 0:
                new_h[i] = 0.0
            else:
                new_h[i] = new_h[i] 
            
        # Rescale so that sum is correct
        current_sum = np.sum(new_h)
        if np.abs(current_sum - target_sum) < 1e-10:
            return new_h
        else:
            # Adjust by scaling/adding to appropriate positions
            diff = target_sum - current_sum
            if diff > 0:
                # Need to add to smaller ones
                shifted_idx = np.argsort(new_h)
                for i in shifted_idx:
                    if new_h[i] < 1 and diff > 0:
                        add = min(diff, 1 - new_h[i])
                        new_h[i] += add
                        diff -= add
            else:
                # Need to reduce larger ones
                shifted_idx = np.argsort(new_h)[::-1]
                for i in shifted_idx:
                    if new_h[i] > 0 and diff < 0:
                        subtract = min(-diff, new_h[i])
                        new_h[i] -= subtract
                        diff += subtract
            return new_h

def compute_correlations(h, dx):
    """Compute all correlations c_k for each shift k."""
    p = 1 - h
    # Using full mode of correlate as specified
    c = correlate(h, p, mode='full') * dx
    # Only the positive lags (corresponding to 0 to n-1 indices in full output)
    # but we compute all because we're going to find max over all
    return c

def band_grad_c_k(h, k, dx):
    """Compute the gradient of correlation c_k w.r.t. h using matrix backward."""
    # This method will be clean and use direct difference approach:
    # d/dh_j c_k = dx [ (1-h_{j+k}) 1_{j+k < n} - h_{j-k} 1_{j-k >= 0} ]
    # = dx * (1 - h[i+k])        if i+k < n  
    #      - dx * h[i-k]         if i-k >= 0
    
    grad = np.zeros_like(h)
    # Forward contribution (1 - h[j+k])
    if k >= 0:
        # Forward part
        if k < len(h):
            grad[k:] += -dx * h[:-k]  # [i+k] part
        grad += dx  # (i) part (adding 1)
    else:
        # Backward part
        if -k < len(h):
            grad[:-k] += -dx * h[k:]  # [i-k] part  
        grad += dx  # Add dx things (i=0 to n-1 each gets dx*1)
        # But again we are evaluating d/dh_j c_k where c_k = ∫ h(x) (1-h(x+k)) dx
        # = dx ∑ h[i] (1 - h[i+k])
        # This needs only:
        # 1. derivative of h[i+k] from h[i], which contributes -dx  
        #    only if i+k is valid: (i+k < n) and index k = i+k in (0, n-1)
        # 2. derivative of h[i] itself: which contributes dx*1 unconditionally (i.e. for all i) 
        grad += dx  # contribution of h[i] to c_k from 1*1
        # But our computed gradient also subtracts the incorrectly added one used by full line
        
    return grad  # needs to match logic w.r.t. integral output on periodic boundary

def compute_grad_c_k_v2(h, k, dx):
    """Compute d/dh_j c_k explicitly."""
    n = len(h)
    # d/dh_j [dx * sum h[i] * (1-h[i+k])] 
    # = dx [sum_{i} d/dh_j (h[i] * (1-h[i+k])) ]
    # = dx [ (1 - h[i+k]) 1_{i = j} - h[i] 1_{i+k = j} ]
    # For all j: if i == j then we get (1 - h[i+k]) right when valid
    # and if i+k == j (i==j-k), we get -h[i]                  
    #                     = -h[j-k] if j >= k
    # For j:
    # Gradient should be dx * [ (1 - h[j+k] if j+k < n) - h[j-k] if j-k >= 0 ]
    # Want to compute it for fixed k
    grad = np.zeros_like(h)
    if (k >= 0):
        # 1 if i = j possible in h[j+k]  -> we add dx to the j-th element
        # -h[j-k] if it is valid to compute j-k
        grad += dx
        # Check if j+k is valid for shift: j remains valid for index j+k in [0, n)
        # i = j
        # We drop x-x (no h[i] = h[j]), only contribution is from (j+k) that exists
        # Hence what we want:
        # We want (1 - h[j+k]) term when that's valid (k >= 0) and we shift one position forward
        # So we add term to index `j` containing (+dx) if the i = j+k, then we want: 
        # For i = j+k → (1-h[i]) = (1-h[j+k])
        # But what we compute is:
        # For i = j -> when j+k is valid, we want contribution from term (j+k-th)
        # So sum h[i] * (1-h[i+k]), when differentiated w.r.t. h_j produces
        # ( ) when i = j, which contributes (1-h[j+k]) or doesn't exist
        # ( ) when i = j-k which contributes ( - h[j-k] ) or doesn't exist
        if k < n - 1:
            grad[:-k] -= dx * h[k:]   # relevant indices are j-k and j==x elements
    else:
        # k < 0: backwards shift: we are computing component j -> term related to (j+k) which would go backwards
        # It's to extract i = j+k → gives us (1 - h[i]) = (1 - h[j+k]), j still here
        # But apply to form:  contribute (1 - h[j+k]) to the exact channel (i)  
        # Auditing: for i = j in this term for d/dh_j c_k = dx [1 - h[i+k] for i = j where i+k < n ] 
        grad += dx
        if k > -n and k < 0:
            grad[-k:] -= dx * h[:-k]
    
    return grad

def smooth_max(h, c, beta):
    """Compute log-sum-exp of beta * c."""
    if beta == 0:
        return np.max(c)
    # Use stable implementation of logsumexp
    log_exp = beta * c
    log_exp = log_exp - np.max(log_exp)
    exps = np.exp(log_exp)
    return (1/beta) * np.log(np.sum(exps))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define parameters  
    n_points = 180  # To match highest resolution expected in computation
    dx = 2.0 / n_points
    sum_target = n_points / 2.0
    
    # Constants for optimization schedule  
    beta_schedule = [0.5, 2, 5, 15, 40, 80]
    max_iter_per_beta = 150
    
    final_best_h = None 
    final_best_c5 = float('inf')
    
    t0 = time.time()
    
    # Start from upscaled initial_h_values, but provide fallback
    if 'initial_h_values' in globals():
        # Interpolate initial_h_values to match n_points (linearly)
        initial_h_vals = np.array(initial_h_values)
        n0 = len(initial_h_vals)
        x_orig = np.linspace(0, 2, n0)
        x_new = np.linspace(0, 2, n_points)
        h_interp = np.interp(x_new, x_orig, initial_h_vals)
        init_start_h = project_h_to_feasible(h_interp, n_points)
    else:
        # Fallback to random initialization
        h_initial = np.random.rand(n_points)
        h_initial = project_h_to_feasible(h_initial, n_points)
        init_start_h = h_initial
    
    # Collect starting positions
    start_positions = [init_start_h]
    # Add more random starting points  
    n_starts = 2
    for _ in range(n_starts): 
        h_rand = np.random.rand(n_points) 
        h_rand = project_h_to_feasible(h_rand, n_points)
        start_positions.append(h_rand)
    
    # For each start position
    for start_idx, h_start in enumerate(start_positions):
        h = h_start.copy()
        # Also track best for this local search
        best_here = float('inf')
        best_here_h = h.copy().
        best_here_c5 = float('inf')
        
        # Reflect the final c5 value
        c = compute_correlations(h, dx)
        true_c5 = np.max(c)    
        if true_c5 < best_here:
            best_here = true_c5
            best_here_c5 = true_c5
            best_here_h = h.copy()
            
        # Change a global parameter for final tracking:  
        beta_idx = 0
        # Optimization per β
        for beta in beta_schedule:
            if time.time() - t0 > 0.95 * budget_s:
                break
                
            # Go a few steps in this stage
            # For now, never exit early for the same beta - fixed iteration  
            for it in range(max_iter_per_beta):
                if time.time() - t0 > 0.95 * budget_s:
                    break
                    
                # Compute c for all k shifts
                c = compute_correlations(h, dx)
                
                # Normalize to bounds  
                if np.max(c) == 0:
                    break
                    
                # Compute weights from LSE smoothing
                w = softmax(beta * c)
                
                # Compute gradient vector sum
                g = np.zeros_like(h)
                for k_idx in range(2 * n_points - 1):
                    if k_idx < n_points:
                        k = k_idx  # lag shift from 0 to n-1 
                    else:
                        k = k_idx - n_points + 1 # from -n+1 to n-1
                    if k >= 0:
                        grad_term = np.zeros_like(h)
                        if k < n_points:
                            grad_term[k:] -= dx * h[:-k]
                        grad_term += dx
                    else:
                        grad_term = np.zeros_like(h)
                        if -k < n_points:
                            grad_term[:-k] -= dx * h[k:]
                        grad_term += dx
                    
                    g += w[k_idx] * grad_term
                
                # Step size decay
                step_size = 0.05 / (1 + it)
                
                # Line search
                step_accept = False
                tries = 0
                while not step_accept:
                    h_new = h - step_size * g
                    h_new = project_h_to_feasible(h_new, n_points)
                    c_new = compute_correlations(h_new, dx)
                    c5_new = np.max(c_new)
                    
                    # Allow acceptance if slightly better or if doing reasonable descent
                    if c5_new < true_c5 or it < 80:
                        # Always accept for first step or big improvement
                        h = h_new 
                        true_c5 = c5_new
                        if true_c5 < best_here:
                            best_here = true_c5
                            best_here_h = h.copy()
                            best_here_c5 = true_c5
                        step_accept = True
                    else:
                        step_size = step_size * 0.5
                        tries += 1
                        if tries > 10:
                            step_accept = True  # Break if stuck in local place

        # After all β stages for this start, store final version
        if best_here_c5 < final_best_c5:
            final_best_c5 = best_here_c5
            final_best_h = best_here_h.copy()
    
    # Refinement step if time allows
    if time.time() - t0 < 0.95 * budget_s:
        # Binary rounding approach: round to nearest binary value while maintaining sum
        h_bin = np.round(final_best_h).astype(float)
        # Rebalance to keep sum = n/2
        sum_diff = np.sum(h_bin) - sum_target
        if sum_diff != 0:
            # Adjust by flipping the lowest difference bits
            # Pick elements that can be flipped
            flip_bits = np.argsort(np.abs(final_best_h - 0.5))[:int(abs(sum_diff))]
            if sum_diff < 0:
                h_bin[flip_bits] = 1.0
            else:
                h_bin[flip_bits] = 0.0
                
        # Recompute and potentially accept improved result
        c_bin = compute_correlations(h_bin, dx)
        bin_c5 = np.max(c_bin)
        
        if bin_c5 < final_best_c5:
            final_best_c5 = bin_c5
            final_best_h = h_bin.copy()
            
    # Ensure that final computed C5 represents the correct allowed output shape
    if final_best_h is None:
        final_best_h = np.ones(n_points) * 0.5
    final_best_c5 = evaluate_erdos_solution(final_best_h) if 'evaluate_erdos_solution' in globals() else final_best_c5
    
    return (final_best_h, final_best_c5, n_points)
