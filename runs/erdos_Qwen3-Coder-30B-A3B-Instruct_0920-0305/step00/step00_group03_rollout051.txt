import numpy as np
import time
from scipy.optimize import minimize

def project_onto_simplex_box(h, target_sum):
    """Project h onto the simplex with sum=target_sum and bounds [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple projection using sorting
    # Find the lambda that makes the sum equal to target_sum
    # h' = max(0, min(1, h_i - lambda))
    # Sum = sum_i min(1, max(0, h_i - lambda)) = sum_i max(0, min(1, h_i - lambda))
    # This is equivalent to sum_{i : h_i - lambda >= 0} min(1, h_i - lambda) 
    # clearly sorted h_i, we can just drop results < 0, then if sum < target, we are done
    # else we find a threshold value and adjust
    
    # More robust approach: sort and adjust
    sorted_indices = np.argsort(h)[::-1]  # descending order
    remaining_sum = target_sum
    for i in range(len(h)):
        idx = sorted_indices[i]
        if h[idx] <= remaining_sum / (len(h) - i):
            # We can set all remaining elements to this value
            # But actually all elements from sorted_indices[i:] are at this level
            if i == 0:
                return np.ones(len(h)) * (target_sum / len(h))
            else:
                # Bring them down
                avg_val = remaining_sum / (len(h) - i)
                new_h = h.copy()
                for j in range(i, len(h)):
                    new_h[sorted_indices[j]] = max(0, min(1, avg_val))
                return new_h
        remaining_sum -= h[idx]
    # All value can be set to same avg
    new_h = np.ones(len(h)) * (target_sum / len(h))
    return np.clip(new_h, 0, 1)

def evaluate(h):
    """Evaluate C5 from h using the provided method."""
    # The evaluation code as exactly stated in task:
    # "The evaluation computes: C5 = max(np.correlate(h, 1-h, mode="full") * dx)"
    # with dx = 2.0 / len(h) 
    dx = 2.0 / len(h)
    conv = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(conv) * dx
    return c5

def compute_gradient_S_max(h, l_max):
    """Compute gradient for the max term S_l."""
    N = len(h)
    grad = np.zeros(N)
    
    if l_max >= 0:
        # i from 0 to N - l_max - 1
        for i in range(N - l_max):
            # contribution to S_l = h_i * (1 - h_{i+l_max})
            grad[i] += 1 - h[i + l_max]
            grad[i + l_max] -= h[i]
    else:
        # l_max is negative, shift made by taking (i + l_max) which is later index i -|l_max|
        labs = -l_max
        for i in range(N - labs):
            # i is index, i+l_max = i-|l_max| is second index
            # contribution is h_i * (1 - h_{i+l_max}) = h_i * (1 - h_{i-labs})
            # gradient is:
            # d/dh_i ...
            grad[i] += 1 - h[i - labs]
            grad[i - labs] -= h[i]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Start with the provided initial construction if available
    INITIAL_H = getattr(__import__('__main__', fromlist=['initial_h_values']), 'initial_h_values')
    initial_h = np.array(INITIAL_H)
    n_points = len(initial_h)
    # Normalize sum to be exactly 1
    dx = 2.0 / n_points
    initial_h = initial_h * (1.0 / np.sum(initial_h)) * n_points / 2.0
    # Project onto feasible region
    initial_h = project_onto_simplex_box(initial_h, target_sum=n_points / 2.0)
    
    h = initial_h.copy()
    # Alpha should be small as it's poorly scaled
    # Start with a small step size
    lr = 0.01
    step = 0
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate the function
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute S_raw
        S_raw = np.correlate(h, 1 - h, mode='full')
        # Find the worst lag
        idx = np.argmax(S_raw)
        l_max = idx - (n_points - 1)
        
        # Compute gradient
        grad = compute_gradient_S_max(h, l_max)
        
        # Gradient step
        h_new = h - lr * grad
        
        # Project onto feasible region
        h_new = project_onto_simplex_box(h_new, target_sum=n_points / 2.0)
        
        h = h_new
        step += 1
        
        # Reduce step size slowly to help convergence
        if step % 500 == 0:
            lr = lr * 0.99
        
    # Final evaluation
    if best_c5 == float('inf'):
        final_c5 = evaluate(h)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h.copy()
    
    return (best_h, best_c5, len(best_h))
