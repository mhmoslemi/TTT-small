import numpy as np
import time
import random

def evaluate_c5(h):
    """Compute C5 using FFT for efficiency."""
    n = len(h)
    size = 1 << ((2*n-1).bit_length())
    f_h = np.fft.fft(h, size)
    f_g = np.fft.fft(1 - h, size)
    corr = np.fft.ifft(f_h * np.conj(f_g)).real
    corr = corr[:2*n-1]
    dx = 2.0 / n
    max_val = np.max(corr) * dx
    return max_val

def random_binary_vector(n):
    """Generate a random binary vector with exactly n//2 ones."""
    h = np.zeros(n)
    ones_indices = np.random.choice(n, n//2, replace=False)
    h[ones_indices] = 1
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    best_h = None
    best_c5 = float('inf')
    
    time_start = time.time()
    elapsed = lambda: time.time() - time_start
    
    # Try different n values
    n_options = [60, 80, 100]
    
    for n in n_options:
        if elapsed() > budget_s - 1.0:
            break
            
        # Initialize
        restart_count = 0
        max_restarts = 5
        max_iter_per_restart = 50000
        
        while restart_count < max_restarts and elapsed() < budget_s - 1.0:
            # Generate initial binary vector
            h = random_binary_vector(n)
            current_c5 = evaluate_c5(h)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            # Simulated Annealing
            T = 0.1
            min_T = 1e-6
            decay = 0.9995
            iteration = 0
            
            while iteration < max_iter_per_restart and elapsed() < budget_s - 1.0:
                # Choose two indices to swap
                ones_indices = np.where(h == 1)[0]
                zeros_indices = np.where(h == 0)[0]
                
                if len(ones_indices) == 0 or len(zeros_indices) == 0:
                    break
                    
                i = np.random.choice(ones_indices)
                j = np.random.choice(zeros_indices)
                
                # Swap values
                h[i], h[j] = h[j], h[i]
                
                # Evaluate new C5
                new_c5 = evaluate_c5(h)
                
                # Accept or reject
                delta = new_c5 - current_c5
                if delta <= 0 or np.random.random() < np.exp(-delta / T):
                    current_c5 = new_c5
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()
                else:
                    # Revert swap
                    h[i], h[j] = h[j], h[i]
                
                iteration += 1
                T = max(min_T, T * decay)
                
            restart_count += 1
