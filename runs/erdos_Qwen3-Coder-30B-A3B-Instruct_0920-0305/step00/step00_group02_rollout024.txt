import numpy as np
from scipy.optimize import differential_evolution
import cvxpy as cp
from time import time
import math

def decode(z):
    M = len(z) // 2
    z_l = z[:M]
    z_g = z[M:]
    
    # Softmax normalization for lengths
    exp_l = np.exp(z_l - np.max(z_l))
    l = exp_l / np.sum(exp_l)
    
    # Softmax normalization for gaps
    exp_g = np.exp(z_g - np.max(z_g))
    g = exp_g / np.sum(exp_g)
    
    return l, g

def build_intervals(ℓ, g):
    M = len(ℓ)
    a = np.zeros(M)
    for i in range(M):
        if i == 0:
            a[i] = g[0]
        else:
            a[i] = a[i-1] + ℓ[i-1] + g[i]
    b = a + ℓ
    return a, b

def candidate_shifts(a, b):
    E = np.concatenate([a, b])
    shifts = set([0.0])
    for i in range(len(E)):
        for j in range(len(E)):
            t = abs(E[i] - E[j])
            if t <= 2.0:
                shifts.add(t)
    return sorted(shifts)

def intersection_length(t, a, b):
    I = 0.0
    for i in range(len(a)):
        if t >= 0:
            # Interval [a[i], b[i]] intersected with [a[i]+t, b[i]+t]
            l = max(0, min(b[i], b[i]+t) - max(a[i], a[i]+t))
        else:
            # Interval [a[i], b[i]] intersected with [a[i]-t, b[i]-t]
            l = max(0, min(b[i], b[i]-t) - max(a[i], a[i]-t))
        I += l
    return I

def objective(z):
    ℓ, g = decode(z)
    a, b = build_intervals(ℓ, g)
    shifts = candidate_shifts(a, b)
    min_I = float('inf')
    for t in shifts:
        I_t = intersection_length(t, a, b)
        if I_t < min_I:
            min_I = I_t
    return 1.0 - min_I

def run_DE(budget_s):
    M = 6
    n_params = 2 * M + 1
    bounds = [(-5.0, 5.0)] * n_params
    start_time = time()
    res = differential_evolution(objective, bounds, seed=42, maxiter=2000, popsize=150, mutation=(0.5, 1.0), recombination=0.7)
    
    return res, time() - start_time

def local_LP(z_best, max_iter=5):
    for _ in range(max_iter):
        ℓ, g = decode(z_best)
        a, b = build_intervals(ℓ, g)
        shifts = candidate_shifts(a, b)
        
        # Find the minimum intersection for these shift points
        min_I = float('inf')
        min_t_values = []
        for t in shifts:
            I_t = intersection_length(t, a, b)
            if I_t < min_I:
                min_I = I_t
                min_t_values = [t]
            elif abs(I_t - min_I) < 1e-10:
                min_t_values.append(t)
        
        # Build linear model using active breakpoints
        # For each active t, we need the derivative with respect to interval positions
        # This is a simplication. The full linear model is constructed as in the strategy.
        # For now, we just use a single step improvement
        
        # Try small changes to endpoints
        # We'll do a parameter `Δ` and subtract it from the start position of an interval
        # and see if we can make I(t) smaller for those active t
        
        # This is a small local step to improve things. In full generation:
        # - We identify all active t (breakpoints that realize the min)
        # - We construct the linear expressions I(t) = c_t + sum_k α_{t,k} · x_k
        # - We solve a small LP to maximize minimum among the active t.
        # Here we do a quick and rough approximation:
        delta = 0.001
        improved = False
        for k in range(len(a)):
            # Try shifting left and right
            new_z = z_best.copy()
            new_z[len(ℓ) + k] -= delta  # shift a[k] to the left by delta
            try:
                new_ℓ, new_g = decode(new_z)
                new_a, new_b = build_intervals(new_ℓ, new_g)
                new_shifts = candidate_shifts(new_a, new_b)
                new_min_I = float('inf')
                for t in new_shifts:
                    I_t = intersection_length(t, new_a, new_b)
                    if I_t < new_min_I:
                        new_min_I = I_t
                if 1.0 - new_min_I < 1.0 - min_I:
                    z_best = new_z
                    improved = True
            except Exception:
                pass
                
        if not improved:
            break
            
    return z_best

def gridify(a, b, n=400):
    h = np.zeros(n)
    dx = 2.0 / n
    for i in range(n):
        x_center = (i + 0.5) * dx
        for k in range(len(a)):
            if a[k] <= x_center <= b[k]:
                h[i] = 1.0
                break
    
    actual_sum = np.sum(h)
    target_sum = n / 2
    
    # Adjust if needed
    if abs(actual_sum - target_sum) <= 1:
        diff = int(target_sum - actual_sum)
        if diff != 0:
            # Find the element whose flip has least impact
            diffs = []
            for i in range(n):
                if h[i] == 1.0 and diff < 0:
                    h[i] = 0.0
                    break
                elif h[i] == 0.0 and diff > 0:
                    h[i] = 1.0
                    break
            else:
                # fallback - just toggle one arbitrary element
                for i in range(n):
                    if h[i] == 1.0:
                        h[i] = 0.0
                        break
                else:
                    h[0] = 1.0
    return h

def binary_hill_climb(h, time_left):
    dx = 2.0 / len(h)
    start_time = time()
    best_h = h.copy()
    best_C5 = float('inf')
    try:
        best_C5 = np.correlate(best_h, 1 - best_h, mode="full") * dx
        best_C5 = max(best_C5)
    except:
        pass

    MAX_SWAPS = 20000
    swaps = 0
    while (time() - start_time) < time_left and swaps < MAX_SWAPS:
        swaps += 1
        # Select two random positions
        i = np.random.randint(0, len(h))
        j = np.random.randint(0, len(h))
        if i == j:
            continue
        if h[i] == h[j]:
            continue

        if (h[i] == 1 and h[j] == 0) or (h[i] == 0 and h[j] == 1):
            old_val = best_C5
            h[i], h[j] = h[j], h[i]
            try:
                new_C5 = np.correlate(h, 1 - h, mode="full") * dx
                new_C5 = max(new_C5)
            except:
                h[i], h[j] = h[j], h[i]
                continue

            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h.copy()
            else:
                h[i], h[j] = h[j], h[i]

    return best_h, best_C5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()

    # Seeding phase
    M = 6
    n_points = 400
    n_params = 2 * M + 1
    time_left = budget_s - 5  # Reserve 5 seconds for other phases
    
    # First global optimization
    res, elapsed = run_DE(time_left)
    z_opt = res.x
    time_left -= elapsed
    
    # Local refinement using LP-like method
    z_refined = local_LP(z_opt)
    time_left -= 30
    
    # Decoding
    ℓ, g = decode(z_refined)
    a, b = build_intervals(ℓ, g)
    
    # Discretisation
    h = gridify(a, b, n_points)
    dx = 2.0 / n_points
    
    # Binary hill climbing phase
    best_h, final_C5 = binary_hill_climb(h, time_left - 5)  # Reserve a bit more for last steps
    
    # Final evaluation
    final_C5 = np.correlate(best_h, 1 - best_h, mode="full") * dx
    final_C5 = max(final_C5)
    
    return (best_h, final_C5, n_points)
