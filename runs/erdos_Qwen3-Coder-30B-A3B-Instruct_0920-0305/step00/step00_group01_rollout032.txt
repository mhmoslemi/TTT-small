def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)

    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value gives us the C5 bound
    max_correlation = np.max(correlation)
    return max_correlation * dx
