import numpy as np
import time
import random
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def project_onto_simplex_box(h, target_sum):
    # Project h onto the set {x: 0 <= x_i <= 1, sum x_i = target_sum}
    # Simple iterative approach:Clip then adjust sum
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return h
    # Simple proportional adjustment to fix sum
    if diff > 0:
        # Need to add mass
        # Find indices that can take more
        candidates = np.where(h < 1)[0]
        if len(candidates) == 0:
            return h
        ave_add = diff / len(candidates)
        for i in candidates:
            delta = min(ave_add, 1 - h[i])
            h[i] += delta
    else:
        # Need to remove mass
        diff = -diff
        candidates = np.where(h > 0)[0]
        if len(candidates) == 0:
            return h
        ave_sub = diff / len(candidates)
        for i in candidates:
            delta = min(ave_sub, h[i])
            h[i] -= delta
    return h

def compute_gradient(h, l):
    # Compute gradient of S[l] = sum_i h_i * (1 - h_{i+l})
    # l can be negative, positive or zero
    N = len(h)
    grad = np.zeros_like(h)
    
    # Determine range that increases with l
    if l >= 0:
        # i from 0 to N-1-l, compute h_i * (1 - h_{i+l})
        i_start = 0
        i_end = N - l
    else:
        # l < 0, i from -l to N-1, compute h_i * (1 - h_{i+l})
        i_start = -l
        i_end = N
    
    for i in range(i_start, i_end):
        # contribution of h_i to S[l]: derivative = 1 - h_{i+l}
        grad[i] += 1 - h[i + l]
        # contribution of h_{i+l} to S[l]: derivative = - h_i
        grad[i + l] -= h[i]
    
    return grad

def compute_objective(h):
    # Return max(np.correlate(h, 1 - h, mode='full'))
    result = np.correlate(h, 1 - h, mode='full')
    return np.max(result)

    # Common setup
