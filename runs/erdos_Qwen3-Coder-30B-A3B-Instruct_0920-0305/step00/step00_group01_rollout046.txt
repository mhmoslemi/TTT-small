import numpy as np
from scipy.optimize import linprog
from scipy.signal import correlate
import time

def evaluate_h(h_values, dx):
    """Evaluate the overlap integral C5 for given h_values."""
    # The overlap integral is the maximum of ∫ h(x)(1 - h(x+k)) dx over all k
    # This is equivalent to 1 - min_k ∫ h(x) h(x+k) dx, which is 1 - min_k C_k
    # So C5 = max_k O_k = max_k (1 - C_k) = 1 - min_k C_k
    # But it's better to compute directly how large O_k can be:
    # O_k = ∫ h(x) (1 - h(x+k)) dx = ∫ h(x) dx - ∫ h(x) h(x+k) dx = 1 - ∫ h(x) h(x+k) dx
    # So we seek max_k O_k = 1 - min_k ∫ h(x) h(x+k) dx
    corr = correlate(h_values, h_values, mode="full")
    # The correlation contribution (sum over x) is scaled to be over [0,2] interval.
    # The convolution results in full correlation shifts from -n+1 to n-1
    # We're correlating h with (1-h) though:
    corr_1_minus_h = correlate(h_values, 1 - h_values, mode="full")
    # The actual index in the full array corresponds to:
    # when the binary shift is +k, it's at index n-1+k in full array
    # when the binary shift is -k, it's at index n-1-k in full array
    # For a function of length n and full mode:
    # indices from -n+1 to n-1.
    n = len(h_values)
    # We compute size of correlation array
    full_size = 2 * n - 1
    shifts = np.arange(-n + 1, n)
    
    # Evaluate each O(k) = 1 - ∫ h(x) h(x+k) dx;
    # The ∫ h(x) h(x+k) dx integrates over 2 length in both directions unless boundary.
    # Actually the integral in one call is already double-writing, so it goes to double sum
    # and we must normalize. But note that for discrete samples:
    # ∫ h(x) h(x+k) dx ≈ sum h[i] h[i+k] * dx
    # The central term in convolution does not give doubles, so:
    # To avoid double counting: compute properly symmetric convolution of h with h
    # and use clever indexing.
    # Approach: correlate h with itself, then compute ∫ h(x) h(x+k) dx = sum(h[i]h[i+k])/dx
    # Similarly for h with (1-h).
    
    # What we actually want is the actual integral values for various shifts:
    # O(k) = 1 - ∫ h(x) h(x+k) dx = 1 - (sum h[i] h[i+k] * dx) * dx
    # Since we're in discrete approximation, the integral is (sum h[i] h[i+k]) * dx
    
    # So given full correlation array of size full_size = 2*n-1
    # Shift index corresponds to:
    # i, i+1: a 1-shift in shift key. When shift is 0, it's center.
    
    # We compute all O(k) values for all valid k shifts from -n+1 to n-1.
    #     O(k) = 1 - [sum h[i]*h[i+k] * dx] * dx, for k = -n+1 to +n-1
    
    # The common way is to let b = h, c = (1-h).
    # Then O_k = 1 - sum b_i * c_{i+k} * dx^2 = [sum b_i * c_{i+k}] *dx - 1
    
    # But again we compute... Therefore, we compute array of integrals:
    # Let us define correct overlap calculations using formula:
    # O_k = ∫ h(x) (1 - h(x+k)) dx = ∫ h(x) dx - ∫ h(x) h(x+k) dx = 1 - C_k
    
    # Recheck the standard way:
    # Overlap O_k = ∫ h(x+h) (1 - h(x)) dx = ∫ h(x) dx - ∫ h(x) h(x+k) dx = 1 - ∫ h(x) h(x+k) dx
    # Define C(k) = ∫ h(x) h(x+k) dx. We want max_k (1 - C(k)) which is:
    # 1 - min_k C(k).
    
    # Correlation gives us convolution of h with h, but we want correlation h with (1-h).
    # So we need to reinterpret properly.
    
    # The final idea:
    # Define C_k = sum(h[i] * h[i + k]) * dx, taking valid k range only.
    # But the sum must be from i=0 to n-1 such that i+k also in [0,n) 
    # so sum starts from max(i=0, -k) to min(i=n-1, n-1-k) over valid i.
    
    # Actually compute it via the full mode convolution, then extract key values:
    # Full correlate gives correlation from index = -n+1 to n-1
    # Shift centers at index = n-1 to shift=0
    # So index in full array j indexes shift = j - (n-1) 
    #       mapping to actual shift = [-n+1, -n+2, ..., -1, 0, +1, ..., n-1]
    
    # Handle both h*h and h*(1-h) convolution, but prefer reusable:
    # We want ∫ h(x) h(x+k) dx = sum h[i]*h[i+k]*dx.
    # Also want ∫ h(x)(1-h(x+k)) dx = sum h[i]*(1-h[i+k])*dx
    
    # So the overlapped sums are sums of h[i] *  h[i+k]   and  h[i] * (1-h[i+k])
    
    # One way:
    # C_k = correlate(h, h, mode="full") using full mode
    # Let's recompute:
    # Efficient recomputation modularity is good.
    corr_h1 = correlate(h_values, h_values, mode="full")
    corr_h_1_minus_h = correlate(h_values, 1 - h_values, mode="full")
    
    # Extract overlapping values corresponding to shifts i to n-1
    shift_indices_for_k = np.arange(-n+1, n)
    
    # O_k = 1 - (corr_h_1_minus_h[j] * dx) where j = n-1 + k.
    # Reconstruct digital O_k values for all k.
    # When shift d = k, we use index in correlation array which is = n-1 + k
    overlaps = []
    for k in range(-n+1, n):
        idx = n - 1 + k  # Mapping from shift k to correlation index
        # compute integral ∫ h(x) h(x+k) dx = sum h[i] h[i+k] * dx
        c_k = corr_h1[idx]  # no need to multiply by dx here because we'll use it when reconstructing overlap
        # compute integral ∫ h(x) (1 - h(x+k)) dx = sum h[i] (1 - h[i+k]) * dx
        o_k = 1 - corr_h_1_minus_h[idx] * dx  # multiply corr by dx, but corr sums with dx factor
        overlaps.append(o_k)
        
    # But we must properly interpret:
    # correlation h*h without (1-): also gives sum h[i] h[i+k] for i + k valid ranges
    # if we use correlation of h with (1-h) using the standard binary:
    # correlations may actually share same indices because we're computing hooverlap of:
    # how much h(i) interacts with h(i+k) in terms of area overlap in definition.
    # This is necessarily that we use full correlate and analyze well.
    
    # Further ideal treatment:
    # For each shift k, we compute the overlapping integral:
    # ∫ h(x) h(x+k) dx.
    #
    # Using correlation for discrete values:
    # let dx = 2/n, for real integration scaling.
    # Supposed to compute ∫ h(x) h(x+k) dx = sum h[i]*h[i+k] * dx * dx = sum h[i]*h[i+k] * dx^2.
    # Huh, but that doesn't match the form. Good let us follow the standard:
    #
    # Relation is in standard application of correlation index:
    # Let x_i = h[i] within [0,n), index i -> x_i = h[i]
    # Then ∫ h(x)*h(x+k)dx (which is actually C_k) ≈ 
    # ∑_{i=0}^{n-1} h_i h_{i+k} dx. (hang on, assuming discrete interval of size dx)
    # Confusing, so we define discrete version and scale. Over overlap same size, not蜣alien cluster.
    # OK. let n = 80. dx = 2/80
    # We compute overlap of h shifted k positions to the right.
    # Using standard convolution algorithm, it's easier to define what we really mean:
    # O_k = 1 - integral h(x) h(x+k) dx, so integral = ∫ h h(x+k) dx = ∫ h_i h_{i+k} dx
    # The sum ∑ h_i h_{i+k} is what you get in corr between h and h.
    # corri = ∑ h_i * h_{i+k} dx
    
    # Confidence in approach:
    # align accuracy: If dx = 2 / n, then:
    # Let us define: correct correlation definition for dx scaling.
    # c_k = ∑ h[i] h[i+k] * dx, where i+k ranges within [0,n-1].
    # This could also be: c_k = ∑ h_i h_{i+k} is from nonzero values.
    # So this makes:
    # o_k = 1 - c_k = 1 - ∑ h_i h_{i+k} dx
    # But o_k = 1 - ∫ h(x) h(x+k) dx = 1 - ∫ h h(x+k) dx (if h(x) is same as h_i)
    
    # So in convolution mode="full", we effectively get the correct overlap values.
    # What the right scaling is? If we compute sum h_i h_{i+k} on indices i,k such that both in [0,n-1]:
    # Then it is shifted k adjustment. As standard in description of shifting.
    # Denote S_k = sum h[i]h[i+k] = corr_h_h[n-1 + k] 
    # Then ∫ h(x) h(x+k) dx = S_k * dx => O_k = 1 - S_k * dx
    
    # Thus: full correlate gives S_k values. O_k = 1 - S_k * dx
    # Should also include fully? No, overlaps are inherently convolution with same h.
    # Check derived principle: Cumulative hit between h with h shifted by k.
    
    # Final correct assignment.
    C_k_values = corr_h1  # This is already sum per shift
    O_k_values = 1 - C_k_values * dx  # factor dx correctly applied here.
    
    return np.max(O_k_values)
