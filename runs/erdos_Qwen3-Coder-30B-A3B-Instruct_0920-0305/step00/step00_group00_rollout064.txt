import time
import numpy as np
from scipy.signal import correlate
from scipy.special import softmax

def project_onto_simplex(v, target_sum):
    """Project vector v onto the simplex {x: sum(x) = target_sum, x >= 0}"""
    # Sort v in descending order
    v_sorted = np.sort(v)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(v_sorted)
    # Find the threshold
    n = len(v)
    threshold = (cumsum - target_sum) / np.arange(1, n + 1)
    # Find largest index where threshold[i] <= v_sorted[i]
    idx = np.where(v_sorted > threshold)[0]
    if len(idx) == 0:
        # All elements are below threshold, set them to 0
        return np.clip(v, 0, None)
    theta = threshold[idx[-1]]
    # Project
    x = np.maximum(v - theta, 0)
    # Adjust to ensure sum = target_sum
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-10:
        # Add the difference to the smallest non-zero elements (to avoid zeroing out)
        non_zero_indices = np.nonzero(x)[0]
        if len(non_zero_indices) > 0:
            # Distribute the difference among non-zero elements
            sorted_indices = np.argsort(x[non_zero_indices])[::-1]
            sorted_non_zero = non_zero_indices[sorted_indices]
            for i in range(min(int(np.ceil(abs(diff))), len(sorted_non_zero))):
                if diff > 0:
                    x[sorted_non_zero[i]] += 1
                    diff -= 1
                elif diff < 0:
                    x[sorted_non_zero[i]] += diff
                    diff = 0
    return np.clip(x, 0, 1)

def evaluate_C5(h, dx):
    """Evaluate maximum overlap integral (C5) for given h."""
    c = correlate(h, 1 - h, mode='full') * dx
    return np.max(c)

def compute_gradient(h, k, dx):
    """Compute gradient of c_k(h) with respect to h"""
    g = np.zeros_like(h)
    if k >= 0:
        # Add contribution from h[i] * (1 - h[i+k]) for i, i+k within range
        g[:len(h)-k] += dx * (1 - h[k:])
        # Subtract contribution from h[i] * h[i-k] for i-k within range
        g[k:] -= dx * h[:len(h)-k]
    else:
        # Simulate the opposite case
        # Add g[i] += dx * (1 - h[i-k]) term for i-k within range,
        # i.e., h[i-k] -> valid for i >= -k, and also i-k < n, so i < n+k;
        # or i >= max(0, -k) and i < min(n-1, n+k)
        g[-k:] += dx * (1 - h[:len(h)+k])
        # Subtract h[i] * h[i-k] term for i >= max(0, k), i-k within range
        g[:len(h)+k] -= dx * h[-k:]
    return g

def smooth_max_gradient(c, weights, dx, h):
    """Compute gradient of log-sum-exp of c values."""
    # weights here are softmax(β * c) ≈ exp(β * c) / sum(exp(β * c))
    grad = np.zeros_like(h)
    for k, w in enumerate(weights):
        grad += w * compute_gradient(h, k - len(h) + 1, dx)
    return grad
