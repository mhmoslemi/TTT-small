import numpy as np
import time
from scipy.optimize import minimize
from math import exp

def project_to_feasible(h, target_sum):
    """Project h onto box [0,1] and sum = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        idx = np.argsort(h)
        for i in range(len(h)):
            if diff <= 0:
                break
            max_increase = 1 - h[idx[i]]
            actual_increase = min(max_increase, diff)
            h[idx[i]] += actual_increase
            diff -= actual_increase
    else:
        # Need to decrease values
        idx = np.argsort(h)[::-1]
        for i in range(len(h)):
            if diff >= 0:
                break
            max_decrease = h[idx[i]]
            actual_decrease = max(max_decrease + diff, 0)
            h[idx[i]] -= actual_decrease
            diff += actual_decrease
    return h

def evaluate_erdos_solution(h):
    """Evaluate the maximum overlap for the given h using numpy correlation"""
    n_points = len(h)
    dx = 2.0 / n_points
    result = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(result) * dx
    return max_overlap
