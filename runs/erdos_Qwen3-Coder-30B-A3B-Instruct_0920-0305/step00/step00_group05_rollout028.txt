import numpy as np
import cvxpy as cp
import time
from math import floor
import random

def full_overlap(x):
    return np.correlate(x, 1 - np.array(x), mode="full")

def worst_shift(x):
    overlaps = full_overlap(x)
    k_max = np.argmax(overlaps)
    return k_max, overlaps[k_max]

def build_mip(N, active_shifts):
    x = cp.Variable(N, integer=True)
    t = cp.Variable(integer=True)
    w_vars = {}
    
    # Constraints for x in [0,1]
    constraints = [x >= 0, x <= 1]
    
    # Half-ones constraint
    constraints.append(cp.sum(x) == N // 2)
    
    # Symmetry breaking
    constraints.append(x[0] == 1)
    
    # Build auxiliary variables and constraints for active shifts
    for k in active_shifts:
        w_vars[k] = cp.Variable(N - k, integer=True)
        for i in range(N - k):
            # wij,k <= xi
            constraints.append(w_vars[k][i] <= x[i])
            # wij,k <= x_{i+k}
            constraints.append(w_vars[k][i] <= x[i + k])
            # wij,k >= xi + x_{i+k} - 1
            constraints.append(w_vars[k][i] >= x[i] + x[i + k] - 1)
    
    # Overlap constraints for all active shifts
    for k in active_shifts:
        overlap_expr = cp.sum(x[:N-k]) - cp.sum(w_vars[k])
        constraints.append(overlap_expr <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    return prob, {'x': x, 't': t, 'w': w_vars}

def solve_mip(prob, vars_dict, time_limit):
    # Extract variables
    x = vars_dict['x']
    t = vars_dict['t']
    
    # Solve the problem
    try:
        prob.solve(solver=cp.CBC, max_iters=10000, TimeLimit=time_limit, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_sol = x.value
            t_sol = t.value
            return x_sol, t_sol
        else:
            return None, None
    except:
        return None, None

    iter_count = 0

    # Start the cutting-plane loop
    while time.time() - start_time < budget_s - 20 and iter_count < max_iterations:
        # Build MIP
        prob, vars_dict = build_mip(N, active_shifts)
        
        # Solve MIP
        x_sol, t_sol = solve_mip(prob, vars_dict, time_limit=max_time_per_solve)
        
        if x_sol is None or t_sol is None:
            break
            
        # Update best solution
        if t_sol < best_t:
            best_t = t_sol
            best_h = x_sol
            
        # Check full overlaps
        overlaps = full_overlap(x_sol)
        max_overlap = np.max(overlaps)
        
        if max_overlap <= t_sol:
            break  # Feasible for all shifts
            
        # Add the worst shift to active shifts
        max_k = np.argmax(overlaps)
        active_shifts.add(max_k)
        
        # Ensure we don't have too many auxiliary variables
        if len(active_shifts) > 200:
            # Pick the largest few shifts that were added most recently
            sorted_shifts = sorted(active_shifts, key=lambda k: -k)
            active_shifts = set(sorted_shifts[:200])

        iter_count += 1
    
    # Binary search to tighten the bound
    while time.time() - start_time < budget_s - 10 and best_t > 0:
        test_t = floor(best_t) - 1
        if test_t <= 0:
            break
            
        # Check feasibility at test_t
        try:

            # Add constraint that t must be at most test_t

            prob.solve(solver=cp.CBC, max_iters=10000, TimeLimit=5, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                best_t = test_t
            else:

        except:

    # Final evaluate and return
    final_c5 = best_t * dx
    return (best_h.tolist(), final_c5, N)
