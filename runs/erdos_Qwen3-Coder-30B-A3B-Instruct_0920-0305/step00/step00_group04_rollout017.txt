def evaluate_erdos_solution(h):

    # Compute the full correlation array
    corr = np.correlate(h, 1 - h, mode="full")
    # Extract only valid correlations (can be full length of 2n-1)
    corr = corr[n - 1:]
    # The C_5 value is the maximum rescaled by dx
    return max(corr) * dx

def compute_correlation_fft(h):
    """Compute overlaps using FFT."""
    n = len(h)
    dx = 2.0 / n
    # zero pad for full correlation
    npad = 2 * n - 1
    # pad to length that is power of 2 for fast FFT
    npad = 1 << (npad - 1).bit_length()
    h_padded = np.pad(h, (0, npad - n), mode='constant')
    g_padded = np.pad(1 - h, (0, npad - n), mode='constant')
    # compute FFTs
    H = fftpack.fft(h_padded)
    G = fftpack.fft(g_padded)
    corr_full = np.real(fftpack.ifft(H * np.conj(G)))
    # take only first 2n-1 values (or npad if padded)
    corr_full = corr_full[:2*n - 1]
    # rescale to get integral approximation * dx
    corr = corr_full[n-1:] * dx
    return np.max(corr)

def create_binary_vector(n, sum_n_half=True):
    """Create a random binary vector with exactly n/2 ones."""
    vec = np.zeros(n, dtype=int)
    ones_indices = np.random.choice(n, n // 2, replace=False)
    vec[ones_indices] = 1
    return vec

def binary_step_swap(vec):
    """Perform a random swap move (1 -> 0, 0 -> 1)."""
    n = len(vec)
    ones = np.where(vec == 1)[0]
    zeros = np.where(vec == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return vec.copy()
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    new_vec = vec.copy()
    new_vec[i] = 0
    new_vec[j] = 1
    return new_vec

def sa_binary_optimization(start_vec, T0=0.05, alpha=0.9995, max_iter=10000, time_limit=30):
    """Simulated Annealing for binary optimization."""
    best_vec = start_vec.copy()
    best_c5 = compute_correlation_fft(best_vec)
    current_vec = start_vec.copy()
    current_c5 = best_c5
    T = T0
    start_time = time.time()
    iteration = 0
    while time.time() - start_time < time_limit and iteration < max_iter:
        new_vec = binary_step_swap(current_vec)
        new_c5 = compute_correlation_fft(new_vec)
        delta = new_c5 - current_c5
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            current_vec = new_vec
            current_c5 = new_c5
        if current_c5 < best_c5:
            best_vec = current_vec.copy()
            best_c5 = current_c5
        T *= alpha
        iteration += 1
    return best_vec, best_c5

def project_simplex(y, target_sum):
    """Projected gradient onto capped simplex [0,1]."""
    n = len(y)
    # try to find lambda such that sum(clamp(y - lambda, 0, 1)) = target_sum
    y = np.array(y, dtype=float)
    def s_lambda(lam):
        return np.sum(np.clip(y - lam, 0, 1)) - target_sum
    low = np.min(y) - 1.0
    high = np.max(y) + 1.0
    while s_lambda(low) * s_lambda(high) > 0:
        low -= 1.0
        high += 1.0
    for _ in range(100):
        lam = (low + high) / 2
        if s_lambda(lam) > 0:
            low = lam
        else:
            high = lam
    # now correct to actually get the exact sum
    lambda_sol = lam = (low + high) / 2
    x = np.clip(y - lam, 0, 1)
    # if we're off, adjust proportionally for final correction
    extra = target_sum - np.sum(x)
    if abs(extra) > 1e-10:
        x += extra / n
    return np.clip(x, 0, 1)

def compute_smooth_max(h, gamma=5):
    """Compute the soft maximum of correlation values."""
    n = len(h)
    dx = 2.0 / n
    # Reuse the FFT-based computation from earlier
    npad = 2 * n - 1
    npad = 1 << (npad - 1).bit_length()
    h_padded = np.pad(h, (0, npad - n), mode='constant')
    g_padded = np.pad(1 - h, (0, npad - n), mode='constant')
    H = fftpack.fft(h_padded)
    G = fftpack.fft(g_padded)
    corr_full = np.real(fftpack.ifft(H * np.conj(G)))
    corr = corr_full[n-1:] * dx
    exp_term = np.exp(gamma * corr)
    Z = np.sum(exp_term)
    if Z == 0:
        result = 0
    else:
        result = np.log(Z) / gamma
    return result

def smooth_grad(h, gamma=5, step=0.01):
    """Compute the gradient of soft max, used to optimize h."""
    n = len(h)
    dx = 2.0 / n
    # First compute the correlation terms needed
    npad = 2 * n - 1
    npad = 1 << (npad - 1).bit_length()
    # pad h and 1-h
    h_padded = np.pad(h, (0, npad - n), mode='constant')
    g_padded = np.pad(1 - h, (0, npad - n), mode='constant')
    H = fftpack.fft(h_padded)
    G = fftpack.fft(g_padded)
    corr_full = np.real(fftpack.ifft(H * np.conj(G)))
    corr = corr_full[n-1:] * dx  # true correlation values
    # compute softmax weights
    exp_corr = np.exp(gamma * corr)
    Z = np.sum(exp_corr)
    weights = exp_corr / Z
    # compute gradient terms (with trilinear indexing)
    grad = np.zeros(n)
    # loop through lag indices (corr is length 2*n-1)
    for l in range(2 * n - 1):
        lag = l - n + 1
        start_idx = max(0, -lag)
        end_idx = min(n, n - lag)
        if start_idx >= end_idx:
            continue
        if lag > 0:
            grad[start_idx:end_idx] += weights[l] * (1 - h[start_idx + lag:end_idx + lag])
        elif lag < 0:
            grad[start_idx:end_idx] += weights[l] * (-h[start_idx + lag:end_idx + lag])
        else:  # lag == 0
            grad[start_idx:end_idx] += weights[l] * (1 - h[start_idx:end_idx]) - weights[l] * h[start_idx:end_idx]
    grad *= dx
    return grad

def project_gradient_descent(h0, max_iter=500, gamma_factor=1.5, min_gamma=5, max_gamma=200, time_limit=150):
    """Continuously refine h using smooth surrogate."""
    start_time = time.time()
    h = h0.copy()
    current_c5 = compute_correlation_fft(h)
    gamma = min_gamma
    step = 0.01
    patience = 0
    best_h = h.copy()
    best_c5 = current_c5
    last_c5 = current_c5
    for i in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        grad = smooth_grad(h, gamma)
        h_new = h - step * grad
        h_new = project_simplex(h_new, len(h) / 2)
        h_new = np.clip(h_new, 0, 1)
        new_c5 = compute_correlation_fft(h_new)
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            patience = 0
        else:
            patience += 1
            if patience > 5:
                step *= 0.9
                patience = 0
        # continue to smooth max
        if i % 10 == 0:
            gamma *= gamma_factor
            if gamma > max_gamma:
                gamma = max_gamma
        last_c5 = new_c5
    return best_h, best_c5
