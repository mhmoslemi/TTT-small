def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant C_5 for given step function h"""
    # Normalize h to have integral = 1
    n_points = len(h)
    dx = 2.0 / n_points
    h = h / np.sum(h) * (2.0 / n_points)
    
    # Compute the overlap integral using full convolution
    conv_result = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the one we want
    max_val = np.max(conv_result) * dx
    return max_val
