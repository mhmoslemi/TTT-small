import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_overlap(h, dx):
    # Compute the overlap integral as max_k of ∫ h(x)(1 - h(x+k)) dx
    n = len(h)
    corrs = np.correlate(h, 1 - h, mode='full')
    # Extract the relevant part (shifts from -n+1 to n-1)
    start = n - 1
    end = 2 * n - 2
    overlaps = corrs[start:end] * dx
    return np.max(overlaps)

def solve_lp_feasibility(h_init, t, n_points, dx):
    # Variables
    h = cp.Variable(n_points)
    p = cp.Variable((n_points, 2 * n_points - 1))

    # Constraints
    constrs = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
    ]

    # Product constraints using McCormick envelopes
    for k in range(2 * n_points - 1):
        d = k - n_points + 1
        for i in range(n_points):
            if 0 <= i + d < n_points:
                # Define p[i,k] = h[i] * h[i+d]
                constrs += [
                    p[i, k] >= h[i] + h[i + d] - 1,
                    p[i, k] <= h[i],
                    p[i, k] <= h[i + d],
                    p[i, k] >= 0,
                ]

    # Overlap constraints: sum p_{i,d} >= (1-t)/dx
    for k in range(2 * n_points - 1):
        d = k - n_points + 1
        if d >= 0:
            # Sum over valid indices i
            expr = cp.sum([p[i, k] for i in range(n_points - d)])
        else:
            expr = cp.sum([p[i, k] for i in range(-d, n_points)])
        w_target = (1 - t) / dx
        constrs += [expr >= w_target]

    # Solve the feasibility LP
    prob = cp.Problem(cp.Minimize(0), constrs)
    try:
        prob.solve(solver=CPLEX, verbose=False)
        if prob.status == "optimal":
            return h.value
        else:
            return None
    except Exception:
        return None

def round_to_binary(h_continuous, n_points):
    # Round to binary values, preserving mass
    h_bin = np.zeros(n_points)
    total_mass = n_points / 2
    sorted_indices = np.argsort(h_continuous)[::-1]
    mass = 0
    for i in sorted_indices:
        if mass + 1 <= total_mass:
            h_bin[i] = 1
            mass += 1
        else:
            break
    return h_bin

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 80
    dx = 2.0 / n_points
    best_c5 = 1.0
    best_h = None

    # Binary search on t
    lower_t = 0.0
    upper_t = 0.38092  # known bound
    tolerance = 1e-4
    max_iterations = 50
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 2:
            break
        t = (lower_t + upper_t) / 2.0
        h_sol = solve_lp_feasibility(None, t, n_points, dx)
        if h_sol is not None:
            h_round = round_to_binary(h_sol, n_points)
            c5_val = evaluate_overlap(h_round, dx)
            if c5_val <= t:
                upper_t = t
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_round.copy()
            else:
                lower_t = t
        else:
            lower_t = t

    # Refinement using local optimization
    if best_h is not None:
        h_refined = best_h.copy()
        # Try to improve using gradient-based local search
        def objective(h):
            if np.any((h < 0) | (h > 1)):
                return 1e6
            return -evaluate_overlap(h, dx)
        
        result = minimize(objective, h_refined, method='L-BFGS-B', bounds=[(0,1)]*n_points, options={'maxiter':100})
        if not result.success:
            h_final = h_refined
        else:
            h_final = np.clip(result.x, 0, 1)
        final_c5 = evaluate_overlap(h_final, dx)
        best_h = h_final
        best_c5 = final_c5

    return (best_h, best_c5, n_points)
