import numpy as np
import cvxpy as cp
import time

def full_overlap(x):
    return np.correlate(x, 1 - x, mode="full")

def worst_shift(x):
    overlaps = full_overlap(x)
    k_star = np.argmax(overlaps)
    return k_star, overlaps[k_star]

def build_mip(N, active_shifts):
    x = cp.Variable(N, integer=True)
    t = cp.Variable(integer=True)
    w = {}
    constraints = []
    
    # Half-ones constraint
    constraints.append(cp.sum(x) == N // 2)
    
    # Symmetry breaking
    constraints.append(x[0] == 1)
    constraints.append(x[1] >= x[N-2])
    constraints.append(x[0] >= x[N-1])
    
    # Binary bounds
    constraints.append(x >= 0)
    constraints.append(x <= 1)
    
    # Auxiliary variables and constraints for active shifts
    for k in active_shifts:
        w[k] = cp.Variable(N - k, integer=True)
        for i in range(N - k):
            # wij,k <= xi
            constraints.append(w[k][i] <= x[i])
            # wij,k <= x_{i+k}
            constraints.append(w[k][i] <= x[i + k])
            # wij,k >= xi + x_{i+k} - 1
            constraints.append(w[k][i] >= x[i] + x[i + k] - 1)
    
    # Overlap constraints
    for k in active_shifts:
        left = cp.sum(x[:N - k])
        right = cp.sum(w[k])
        constraints.append(left - right <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    prob = cp.Problem(objective, constraints)
    return prob, {'x': x, 't': t, 'w': w}

def solve_mip(prob, vars_dict, time_limit):
    try:
        result = prob.solve(solver=cp.CBC, max_iters=10000, TimeLimit=time_limit)
        x_val = vars_dict['x'].value
        t_val = vars_dict['t'].value
        return x_val, t_val
    except Exception:
        return None, None
