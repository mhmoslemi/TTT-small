import numpy as np
import time
from scipy import fft

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n = 300   # Good balance between resolution and speed
    dx = 2.0 / n
    max_time = budget_s * 0.6  # for gradient ascent
    polishing_time = budget_s * 0.3
    evaluation_time = budget_s * 0.1
    
    # Beatty sequence initialization
    beta = (np.sqrt(5) - 1) / 2.0
    h = np.array([1 if (i * beta) % 1 < 0.5 else 0 for i in range(n)])
    sum_h = np.sum(h)
    target_sum = n // 2
    if sum_h != target_sum:
        # Flip some 0s to 1s to get exactly target_sum ones
        to_flip = abs(sum_h - target_sum)
        indices_0 = np.where(h == 0)[0][:to_flip]
        h[indices_0] = 1
        # Adjust in case we over-flipped
        if np.sum(h) > target_sum:
            indices_1 = np.where(h == 1)[0][:(np.sum(h) - target_sum)]
            h[indices_1] = 0
    
    h = h.astype(float)
    
    # Helper function for projection onto the constraints
    def project(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if np.abs(current_sum - n/2) < 1e-8:
            return h
        # Solve for λ such that sum(clip(h + λ, 0, 1)) = n/2
        def sum_at_lambda(lam):
            return np.sum(np.clip(h + lam, 0, 1))
        # Binary search for correct lambda
        low, high = -1, 1
        while high - low > 1e-10:
            mid = (low + high) / 2
            sum_test = sum_at_lambda(mid)
            if sum_test < n/2:
                low = mid
            else:
                high = mid
        h = np.clip(h + low, 0, 1)
        return h
    
    # Smooth objective
    def Phi_alpha(h, alpha=1.0):
        # Compute all S_k via fft
        h_pad = np.concatenate([h, [0]*(n-1)])
        h_rev = h[::-1]
        h_rev_pad = np.concatenate([h_rev, [0]*(n-1)])
        c = fft.ifft(fft.fft(h_pad) * fft.fft(h_rev_pad)).real[:n]
        s_k = c[n-1:] * dx  # shift k from 0 to n-1
        # Avoid numerical underflow
        s_k = np.clip(s_k, 0, np.max(s_k) + 1)
        w = np.exp(-alpha * s_k)
        return -(1/alpha) * np.log(np.sum(w))
    
    def grad_Phi_alpha(h, alpha=1.0):
        # Compute S_k (already done in Phi_alpha above)
        h_pad = np.concatenate([h, [0]*(n-1)])
        h_rev = h[::-1]
        h_rev_pad = np.concatenate([h_rev, [0]*(n-1)])
        c = fft.ifft(fft.fft(h_pad) * fft.fft(h_rev_pad)).real[:n]
        s_k = c[n-1:] * dx  # shift k from 0 to n-1
        # Avoid numerical underflow
        s_k = np.clip(s_k, 0, np.max(s_k) + 1)
        w = np.exp(-alpha * s_k)
        W = np.sum(w)
        if W < 1e-12:
            return np.zeros_like(h)
        # Gradient vector: grad_i = sum_k w_k * (h_{i+k} + h_{i-k}) / W
        grad = np.zeros(n)
        h_extended = np.pad(h, (0, n-1), mode='constant')
        for k in range(n):
            if k == 0:
                grad += w[k] * h_extended[:n]
            else:
                grad += w[k] * (h_extended[k:n+k] + h_extended[n-k:n])
        return grad / W
    
    # Gradient ascent loop
    h_current = h.copy()
    h_best = h_current.copy()
    phi_best = Phi_alpha(h_current)
    start = time.time()
    
    alpha = 1.0
    next_alpha_update = 200
    update_count = 0
    
    while time.time() - start < max_time:
        # Compute gradient and update
        grad = grad_Phi_alpha(h_current, alpha)
        step = 1.0
        while step > 1e-8:
            h_new = h_current + step * grad
            h_new = project(h_new)
            phi_new = Phi_alpha(h_new, alpha)
            if phi_new > phi_best:
                h_best = h_new.copy()
                phi_best = phi_new
            if phi_new >= phi_best - 1e-8:  # Accept change if as good or better
                h_current = h_new
                break
            else:
                step *= 0.5  # Backtracking
        
        update_count += 1
        if update_count == next_alpha_update:
            alpha *= 2
            next_alpha_update += 200
            if alpha >= 10000:
                alpha = 10000
    
    # Discrete polishing: rounding + hill climbing
    h_final = h_best.copy()
    # Round to binary
    median_val = np.partition(h_final, n//2)[n//2]
    h_round = (h_final > median_val).astype(float)
    # Ensure sum = n//2
    sum_h = np.sum(h_round)
    if sum_h != n//2:
        to_flip = int(np.abs(sum_h - n//2))
        indices_0 = np.where(h_round == 0)[0][:to_flip]
        indices_1 = np.where(h_round == 1)[0][:to_flip]
        h_round[indices_0] = 1
        h_round[indices_1] = 0
    
    # Hill climbing
    h_best_final = h_round.copy()
    best_c5 = float('inf')
    
    # Evaluate current result
    c_final = np.correlate(h_best_final, 1 - h_best_final, mode='full')
    c5_current = np.max(c_final) * dx
    if c5_current < best_c5:
        best_c5 = c5_current
        h_best_final[:] = h_round[:]
    
    start_polish = time.time()
    while time.time() - start_polish < polishing_time:
        # Quick and biased swap moves
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if h_best_final[i] == h_best_final[j]:
            continue
        # Swap i and j
        h_best_final[i], h_best_final[j] = h_best_final[j], h_best_final[i]
        # Compute new C5
        c_new = np.correlate(h_best_final, 1 - h_best_final, mode='full')
        new_c5 = np.max(c_new) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            # Revert if worse
            h_best_final[i], h_best_final[j] = h_best_final[j], h_best_final[i]
    
    # Final evaluation
    final_corr = np.correlate(h_best_final, 1 - h_best_final, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return h_best_final.tolist(), float(final_c5), n
