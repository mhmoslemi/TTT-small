def decode(z):
    """Decode raw vector z into interval lengths and gaps"""
    M = len(z) // 2  # number of intervals
    z_ell = z[:M]
    z_g = z[M:]
    
    ell = np.exp(z_ell)
    ell = ell / np.sum(ell)
    
    g = np.exp(z_g)
    g = g / np.sum(g)
    
    return ell, g

def build_intervals(ell, g):
    """Build the intervals [a_i, a_i + ell_i] from lengths and gaps"""
    a = np.zeros(len(ell))
    a[0] = g[0]
    for i in range(1, len(ell)):
        a[i] = a[i-1] + ell[i-1] + g[i]
    return a, a + ell

def candidate_shifts(a, b):
    """Generate all possible shift values that could be critical"""
    E = list(a) + list(b)
    T = {0.0}
    for i in range(len(E)):
        for j in range(len(E)):
            t = abs(E[i] - E[j])
            if t <= 2.0:
                T.add(t)
    return sorted(list(T))

def intersection_length(t, a, b):
    """Compute the intersection length of S and S+t"""
    if t == 0:
        return np.sum(b - a)
    
    total = 0.0
    for i in range(len(a)):
        left = max(a[i], a[i] - t)
        right = min(b[i], b[i] - t)
        if right > left:
            total += right - left
    return total

def objective(z):
    """Objective: compute C5 = 1 - min_t I(t)"""
    ell, g = decode(z)
    a, b = build_intervals(ell, g)
    shifts = candidate_shifts(a, b)
    
    min_I = float('inf')
    for t in shifts:
        I = intersection_length(t, a, b)
        if I < min_I:
            min_I = I
    
    return 1.0 - min_I

def gridify(intervals, n):

    h = np.zeros(n)
    dx = 2.0 / n
    
    # Ensure exact sum = 1 (i.e., n/2)

        a, b = intervals[i]
        total_length += b - a
    
    if abs(total_length - 1.0) > 1e-10:
        # Correct lengths to exactly sum to 1 using simple distribution
        adjustments = (1.0 - total_length) / len(intervals)
        new_intervals = []
        for i in range(len(intervals)):
            new_intervals.append([intervals[i][0], intervals[i][1] + adjustments])
        intervals = new_intervals
    
    # Fill grid
    for i in range(len(intervals)):
        a, b = intervals[i]
        start_idx = max(int(np.floor(a/dx)), 0)
        end_idx = min(int(np.ceil(b/dx)), n)
        for j in range(start_idx, end_idx):
            h[j] = 1  # binary intervals in this version
    
    # Adjust sum to exactly n/2
    current_sum = np.sum(h)
    if abs(current_sum - n/2) > 1e-6:
        diff = int(n/2 - current_sum)
        if diff > 0:
            # Need to flip some 0s to 1s
            ones = np.where(h == 0)[0]
            if len(ones) >= diff:
                for i in range(diff):
                    h[ones[i]] = 1
        elif diff < 0:
            # Need to flip some 1s to 0s
            zeros = np.where(h == 1)[0]
            if len(zeros) >= abs(diff):
                for i in range(abs(diff)):
                    h[zeros[i]] = 0
    
    return h
