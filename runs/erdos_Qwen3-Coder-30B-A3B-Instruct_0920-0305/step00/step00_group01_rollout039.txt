import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time
import math

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all possible k values
    # Using correlation approach
    corr = np.correlate(h, 1 - h, mode='full')
    # Select the maximum value (corresponds to the maximum k)
    max_val = np.max(corr)
    return max_val * dx

def initial_h_values():
    """Initial construction - a step function with 47 samples"""
    n = 47
    h = np.zeros(n)
    # Simple construction: small spikes
    h[10:15] = 1.0
    h[25:30] = 1.0
    # Normalize
    dx = 2.0 / n
    h = h * (1.0 / (np.sum(h) * dx))
    return h

def setup_optimization(n_points=47):
    """Set up optimization variables"""
    h = np.ones(n_points) * 0.5  # Initial guess
    return h

def project_to_feasible(h, n_points):
    """Project h to satisfy the constraints"""
    dx = 2.0 / n_points
    # Force h to be between 0 and 1
    h = np.clip(h, 0, 1)
    # Adjust to satisfy integral constraint
    current_integral = np.sum(h) * dx
    if abs(current_integral - 1.0) > 1e-10:
        h = h * (1.0 / current_integral)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Initial values
    n_points = 47
    h = initial_h_values()
    
    # Store current solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization approach: gradient-based with projection
    def objective(h_flat):
        # Reshape and project
        h = h_flat.reshape(-1)
        h = project_to_feasible(h, n_points)
        return evaluate_erdos_solution(h)
    
    def constraint_func(h_flat):
        h = h_flat.reshape(-1)
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0  # Should return 0 at feasible points
    
    # Consider a few strategies for search
    method = 'l-bfgs-b'  # Simple, efficient
    
    # Keep track of best solution so far
    global best_h, best_c5
    
    # Do a few iterations with local search
    for i in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb the current solution to form a new start
        new_h = h + np.random.normal(0, 0.01, n_points) 
        new_h = project_to_feasible(new_h, n_points)
        
        # Evaluate new point
        new_c5 = evaluate_erdos_solution(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            if best_c5 < 0.379:  # Early termination if good enough
                break
        
        # If reasonably close to an optimal, sample a few more times
        if i % 50 == 0:
            if time.time() - start_time > budget_s - 2:
                break
    
    # More precise local refinement if we haven't spent much time yet
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1.0:
        # Try gradient-based approach
        bounds = [(0, 1)] * n_points  # Each h[i] between 0 and 1
        
        # Do gradient-based search starting from the current best
        h_flat = best_h.copy()
        try:
            opt_result = minimize(
                objective,
                h_flat,
                method=method,
                bounds=bounds,
                options={'maxiter': 100, 'disp': False}
            )
            if opt_result.success:
                h_refined = opt_result.x.reshape(-1)
                final_h = project_to_feasible(h_refined, n_points)
                final_c5 = evaluate_erdos_solution(final_h)
                if final_c5 < best_c5:
                    best_c5 = final_c5
                    best_h = final_h
        except:
            pass
    
    return (best_h.tolist(), best_c5, n_points)
