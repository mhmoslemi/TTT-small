def project_onto_simplex_box(h, target_sum):
    """Project h onto box [0,1]^N and simplex {sum h = target_sum}"""
    # First project onto box
    h = np.clip(h, 0, 1)
    # Then project onto simplex
    h_sum = np.sum(h)
    if abs(h_sum - target_sum) < 1e-10:
        return h
    # Use simple adjustment: scale down or up to hit target
    scale = target_sum / h_sum
    h = np.clip(h * scale, 0, 1)
    return h

    # Compute correlation - this is how the code does it
    corr = np.correlate(h, 1 - h, mode='full')
    # Return maximum integral of overlap
    return np.max(corr) * dx
