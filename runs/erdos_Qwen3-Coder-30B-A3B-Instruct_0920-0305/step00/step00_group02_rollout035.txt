import numpy as np

def decode(z):
    """map a vector z in R^{2M+1} to positive lengths and gaps that sum to 1"""
    M = (len(z) - 1) // 2
    zl, zg = z[:M], z[M:]
    l = np.exp(zl)
    g = np.exp(zg)
    return l / np.sum(l), g / np.sum(g)

def build_intervals(l, g):
    """compute interval starts a_i and ends b_i = a_i + l_i"""
    a = np.empty(len(l))
    a[0] = g[0]
    for i in range(1, len(l)):
        a[i] = a[i-1] + l[i-1] + g[i]
    b = a + l
    return a, b

def candidate_shifts(a, b):
    """generate all candidate shifts that induce breakpoints"""
    E = np.concatenate((a, b))  # all interval endpoints
    T = set([0.0])
    for ei in E:
        for ej in E:
            t = abs(ei - ej)
            if t <= 2.0:
                T.add(t)
    return sorted(list(T))

def intersection_length(t, a, b):
    """compute |S ∩ (S+t)| where S = ⋃ [a_i, b_i]"""
    I = 0.0
    for i in range(len(a)):
        # intersection of [a[i], b[i]] and [a[i]-t, b[i]-t] if t>0
        # or [a[i]+t, b[i]+t] if t<0
        left = a[i] + t
        right = b[i] + t
        # overlap with [left, right]
        if t > 0:  # shift right
            common_left = max(a[i], left)
            common_right = min(b[i], right)
        else:  # shift left or none
            common_left = max(a[i], left)
            common_right = min(b[i], right)
        if common_right > common_left:
            I += common_right - common_left
    return I

def objective(z):
    """evaluate the objective function for given z"""
    l, g = decode(z)
    a, b = build_intervals(l, g)
    T = candidate_shifts(a, b)
    I_values = [intersection_length(t, a, b) for t in T]
    min_I = min(I_values)
    # C5 = 1 - min |S ∩ (S+t)|
    return 1.0 - min_I

def run_DE(budget_s, seed, M):
    np.random.seed(seed)
    bounds = [(-5.0, 5.0)] * (2*M + 1)
    start = time.time()
    res = differential_evolution(objective, bounds, maxiter=2000, popsize=15*(2*M+1),
                                seed=seed, updating='immediate', disp=False)
    elapsed = time.time() - start
    return res

def gridify(intervals, n):
    """Convert intervals to binary array with length n"""
    a, b = intervals
    l = b - a
    h = np.zeros(n)
    dx = 2.0 / n
    for i in range(len(a)):
        # integer indices covered by interval [a[i], b[i]]
        ileft = int(np.ceil(a[i] / dx))
        iright = int(np.floor(b[i] / dx))
        # Test whether it's within [0,2]
        if ileft < 0: ileft = 0
        if iright >= n: iright = n-1
        for j in range(ileft, iright + 1):
            h[j] = max(h[j], 1.0)  # in set case, height=1

    # correct to ensure correct sum:
    h_sum = np.sum(h)
    target = n / 2.0
    
    # If sum was off due to rounding, flip some bit
    diff = h_sum - target
    if abs(diff) > 0.5:
        I = np.where(h == 1.0)[0]
        J = np.where(h == 0.0)[0]
        # Fix it with one 1->0 and one 0->1
        if diff > 0:
            indflip1 = I[0]
            h[indflip1] = 0.0
        else:
            # flip only one 0->1
            indflip2 = J[0]
            h[indflip2] = 1.0

        # flip again if still not exact
        if abs(np.sum(h) - n/2) > 1e-6:
            # More if needed.
            # try efficient index pick
            I = np.where(h == 1.0)[0]
            J = np.where(h == 0.0)[0]
            if len(J) > 0:
                h[J[0]] = 1.0
            if len(I) > 0:
                h[I[0]] = 0.0

    return h

def binary_hill_climb(h, time_left):
    """Perform quick hill-climbing on the discrete grid"""
    # Allow up to 1000 swaps (in tight memory use account)
    # 1000 swaps on 400 will cover space well
    np.random.seed(42)
    start = time.time()
    best_h = h.copy()
    n = len(best_h)
    # Evaluation function
    def evaluate(h):
        dx = 2.0 / n
        cor = np.correlate(h, 1 - h, mode="full")
        return max(cor) * dx

    best_c5 = evaluate(best_h)
    nswap = min(1000, int((time_left - 2) * 1000 / 2))
    for i in range(nswap):
        if time.time() - start > time_left - 0.5:
            break
        # Pick indices i and j
        i1 = np.random.randint(0, n)
        i2 = np.random.randint(0, n)
        if i1 == i2: continue
        if h[i1] == h[i2]: continue
        # Create copy  
        hnew = h.copy()
        hnew[i1], hnew[i2] = hnew[i2], hnew[i1]
        c5_new = evaluate(hnew)
        if c5_new < best_c5:
            best_h[:] = hnew
            best_c5 = c5_new
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    # Allocate time
    start = time.time()
    # Parameters
    M = 6
    n_grid = 400
    branching = 0.5
    rest = budget_s - 5  # initial seed time
    DE_budget = int(rest * 0.7)
    LP_budget = int(rest * 0.05)
    refine_budget = int(rest * 0.05)
    bin_polish_budget = int(rest * 0.2) 
    
    # 1. DE an initial solution
    res = run_DE(DE_budget, seed, M)
    zbest = res.x.copy()
    
    if len(zbest) < 1:
        zbest = np.random.uniform(-3, 3, size=2*M + 1)

    # Optional: Refine using local LP
    if LP_budget > 5:
        try:
            zbest = local_LP(zbest, M, delta=0.001)
        except:
            pass

    # 2. Decode zbest to intervals
    l, g = decode(zbest)
    
    # Now build step function
    # integrate the continuous design to a discrete one:
    a, b = build_intervals(l, g)
    h = gridify((a, b), n_grid)
    dx = 2.0 / n_grid
    cor = np.correlate(h, 1 - h, mode="full")
    c5 = max(cor) * dx

    # 3. Binary Hill-climbing polishing
    if bin_polish_budget > 1:
        h_final, c5_final = binary_hill_climb(h, bin_polish_budget)
    else:
        h_final, c5_final = h, c5

    # Debug final:
    return (h_final, c5_final, len(h_final))
