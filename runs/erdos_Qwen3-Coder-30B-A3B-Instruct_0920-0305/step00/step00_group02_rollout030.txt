```python
import time
import numpy as np
import cvxpy as cp
import random
import math

def random_binary_vector(n):
    """Create random binary vector with exactly n/2 ones."""
    if n % 2 != 0:
        raise ValueError("n must be even")
    h = np.zeros(n, dtype=int)
    ones_indices = np.random.choice(n, n//2, replace=False)
    h[ones_indices] = 1
    return h

def compute_c5(h):
    """Compute C5 value for given h."""
    n = len(h)
    dx = 2.0 / n
    # compute full correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # max correlation is max of all valid shifts
    # valid shift indices are from n-1 to 2*n-2 (to make sure no wrap)
    c5 = np.max(corr) * dx
    return c5

def solve_milp(n, time_limit=10):
    """Solve MILP to find optimal binary h vector."""
    # Create boolean variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()  # continuous variable for min overlap
    constraints = [cp.sum(h) == n//2]
    
    # For each shift k, create variables for product terms
    # We consider all shifts from 0 to n-1
    shift_constraints = []
    for k in range(n):
        # number of positions where product makes sense is length (n-k)
        if n - k == 0:
            continue
        # Variables for products at this shift
        y = cp.Variable(n - k, boolean=True)
        # y[i] represents h[i] * h[i+k] 
        # constraint: y[i] <= h[i], y[i] <= h[i+k], y[i] >= h[i] + h[i+k] - 1
        # and y[i] >= 0 (implicitly by definition, but include for completeness)
        # y >= 0 is automatic; we'll handle other constraints for the product linearization
        # We need exactly that: y[i] = h[i] * h[i+k] for binary h
        # The hull of {y = x1*x2 : x1, x2 binary} is {y <= x1, y <= x2, y >= x1 + x2 -1}
        # Hence the constraints:
        # y >= h[i] + h[i+k] - 1
        # y <= h[i] 
        # y <= h[i+k]
        # (no y >= 0 because that's default for variables)
        for i in range(n - k):
            constraints.append(y[i] >= h[i] + h[i+k] - 1)
            constraints.append(y[i] <= h[i])
            constraints.append(y[i] <= h[i+k])
        
        # sum of y's is the overlap for this shift
        constraints.append(cp.sum(y) >= t)

    prob = cp.Problem(cp.Maximize(t), constraints)
    
    # Use solver with timeout
    try:
        prob.solve(solver=cp.CBC, maxTime=time_limit)
    except:
        # fallback if solver fails
        raise RuntimeError("MILP solver failed")
        
    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
        h_sol = np.round(h.value).astype(float)
        dx = 2.0 / n
        total = 1.0
        corr = np.correlate(h_sol, 1 - h_sol, mode='full')
        c5 = np.max(corr) * dx
        return h_sol, c5
    else:
        # fallback to random solution if MILP fails to find good solution
        h_sol = random_binary_vector(n)
        c5 = compute_c5(h_sol)
        return h_sol, c5

def local_search(h_initial, max_iters, time_limit):
    """Use greedy local search with random swaps to reduce C5."""
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    
    start = time.time()
    for i in range(max_iters):
        if time.time() - start > time_limit:
            break
            
        # get indices of 1's and 0's
        ones_idx = np.where(best_h == 1)[0]
        zeros_idx = np.where(best_h == 0)[0]
        
        # pick random swap
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            continue
            
        i_idx = np.random.choice(ones_idx)
        j_idx = np.random.choice(zeros_idx)
        
        # perform swap
        candidate_h = best_h.copy()
        candidate_h[i_idx] = 0
        candidate_h[j_idx] = 1
        
        candidate_c5 = compute_c5(candidate_h)
        
        # accept improvement
        if candidate_c5 < best_c5:
            best_h = candidate_h
            best_c5 = candidate_c5
        elif np.random.rand() < 0.01:  # occasional acceptance of worse solution
            best_h = candidate_h
            best_c5 = candidate_c5

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    n = 100  # chosen for reasonable memory and computation time
    
    try:
        # Try MILP for high quality upper bound
        h_opt, c5_opt = solve_milp(n, time_limit=min(10, budget_s/3))
        elapsed = time.time() - start
        remaining = budget_s - elapsed - 1
        
        # Local search improvement if time remains
        if remaining > 5:
            best_h, best_c5 = local_search(h_opt, 50000, remaining)
        else:
            best_h, best_c5 = h_opt, c5_opt
    except Exception as e:
        # Fallback
        h_opt = random_binary_vector(n)
        c5_opt = compute_c5(h_opt)
        best_h, best_c5 = local_search(h_opt, 30000, 300)
    
    return best_h.tolist(), float(best_c5), int(n)
```
