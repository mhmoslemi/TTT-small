def project_onto_capped_simplex(y, target_sum):
    """Projects y onto the capped simplex {x : 0 <= x <= 1, sum(x) = target_sum}."""
    y = np.clip(y, 0, 1)
    sorted_y = np.sort(y)[::-1]
    cumulative_sum = np.cumsum(sorted_y)
    n = len(y)
    # Find the threshold lambda
    lambda_ = (cumulative_sum[-1] - target_sum) / n
    if lambda_ < 0:
        lambda_ = 0
        for i in range(n):
            if cumulative_sum[i] <= target_sum:
                lambda_ = (cumulative_sum[i] - target_sum) / (i+1)
            else:
                break
    proj = np.clip(y - lambda_, 0, 1)
    return proj

def evaluate(x):
    """Computes the C5 value of the step function x."""
    N = len(x)
    dx = 2.0 / N
    # Correlation of x with (1 - x) for all shifts
    correlation = np.correlate(x, 1 - x, mode='full')
    # Extract the valid correlations (shifts from -(N-1) to N-1)
    max_corr = np.max(correlation)
    return max_corr * dx

def gradient_of_max(x, beta):
    """Computes the gradient of the smooth max approximation with parameter beta."""
    N = len(x)
    dx = 2.0 / N
    # Compute all O_k values
    correlations = np.correlate(x, 1 - x, mode='full')
    # Shift from -(N-1) to N-1, so the actual shift index in the array is from 0 to 2*N-2
    O_values = []
    for k in range(-N+1, N):
        start_idx = max(0, -k) if k < 0 else 0
        end_idx = N - abs(k)
        if k < 0:
            k_corr = np.sum(x[start_idx:end_idx] * (1 - x[start_idx - k:end_idx - k]))
        else:
            k_corr = np.sum(x[start_idx:end_idx] * (1 - x[start_idx + k:end_idx]))
        O_values.append(k_corr * dx)
    # Softmax weights
    exp_vals = np.exp(np.array(O_values) * beta)
    weights = exp_vals / np.sum(exp_vals)
    # Gradient of smooth max = sum_k weights_k * grad O_k
    grad = np.zeros(N)
    for k in range(-N+1, N):
        idx = k + N - 1  # index in [0, 2*N-2]
        start_i = max(0, -k) if k < 0 else 0
        end_i = N - abs(k)
        if k < 0:
            # grad O_k[i] = (1 - x[i+k]) - x[i-k] = (1 - x[i-k]) - x[i-k]
            # For i=0..end_i-1, x[i-k] term contributes from the left
            for i in range(start_i, end_i):
                if i - k >= 0 and i - k < N:
                    grad[i] += weights[idx] * (1 - 2 * x[i-k])
        else:
            # grad O_k[i] = (1 - x[i+k]) - x[i-k]  
            # For i=0..end_i-1, x[i-k] term doesn't exist to the left
            for i in range(start_i, end_i):
                if i + k < N:
                    grad[i] += weights[idx] * (1 - 2 * x[i+k])
    return grad

def swap_move(x, i, j):
    """Swaps entries i and j in a binary vector x."""
    x = x.copy()
    x[i], x[j] = x[j], x[i]
    return x

def fractional_move(x, i, j, delta):
    """Moves delta from x[j] to x[i], respecting bounds."""
    x = x.copy()
    if delta <= x[j]:
        x[i] += delta
        x[j] -= delta
    else:
        x[i] += x[j]
        x[j] = 0.0
    return np.clip(x, 0, 1)

def worst_shift_info(x):
    """Returns shift k* and list of indices i whose mismatch causes largest overlap."""
    N = len(x)
    correlation = np.correlate(x, 1-x, mode='full')
    max_shift = np.argmax(correlation)
    max_val = correlation[max_shift]
    shift_k = max_shift - (N - 1)
    # Find all indices i such that x[i] = 1 and x[i+shift_k] = 0 if shift_k > 0,
    # or x[i-shift_k] = 0 if shift_k < 0
    if shift_k >= 0:
        # Project x[i] * (1 - x[i+shift_k])  
        off_indices = []
        for i in range(N - shift_k):
            if x[i] == 1 and x[i + shift_k] == 0:
                off_indices.append(i)
    else:
        # shift_k < 0 so we have i and i-shift_k position
        off_indices = []
        shift_abs = abs(shift_k)
        for i in range(shift_abs, N):
            if x[i] == 1 and x[i - shift_abs] == 0:
                off_indices.append(i)
    return shift_k, off_indices
