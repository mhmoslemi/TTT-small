import numpy as np
from scipy.optimize import linprog
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Discretization parameters
    n_points = 80        # reasonable compromise for speed and quality
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # For searching upper bound on C_5 via bisection
    lower_bound = 0.0
    upper_bound = 0.38092
    tolerance = 1e-5
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Helper to convert h into a binary vector that sums to target_sum
    def round_to_binary(h_vals):
        # create binary vector with same sum (approximately)
        indices = np.argsort(h_vals)[::-1]  # descending order
        h_binary = np.zeros_like(h_vals)
        h_sum = np.sum(h_vals)
        count = 0
        target = int(round(target_sum))
        for i in indices:
            if count < target:
                h_binary[i] = 1
                count += 1
            else:
                break
        # if not reached exactly, adjust any difference with remaining indices
        current_sum = np.sum(h_binary)
        if current_sum < target:
            for i in indices:
                if current_sum >= target:
                    break
                if h_binary[i] == 0:
                    h_binary[i] = 1
                    current_sum += 1
        elif current_sum > target:
            for i in indices[::-1]:
                if current_sum <= target:
                    break
                if h_binary[i] == 1:
                    h_binary[i] = 0
                    current_sum -= 1
        return h_binary

    # Helper to compute overlaps for given h (uses scipy convolution)
    def compute_overlap(h_vals):
        # compute all possible overlaps for shifts k
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        # collapse to positive shifts: corresponds to [0,2] domain
        shift_len = len(corr)
        half_length = (shift_len - 1) // 2
        # only take effective overlaps, the first half shifting from 0 to 2
        overlaps = corr[half_length:]
        return np.max(overlaps) * dx

    # Main loop for bisection
    while upper_bound - lower_bound > tolerance:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.1:
            break
            
        t_mid = (lower_bound + upper_bound) / 2.0
        # Define LP problem variables for feasibility check
        # For shift k we need sum of h[i]*h[i+k] >= (1-t_mid)/dx
        
        A_eq = []
        b_eq = []
        
        # Constraint: sum h = target_sum
        row_sum = np.ones(n_points)
        A_eq.append(row_sum)
        b_eq.append(target_sum)
        
        # Build constraints for maximizing overlap
        N = n_points
        total_shifts = 2*N - 1
        # shift indices correspond to reach shift index
        # Correlation index offsets for each shift: -N+1 to N-1
        p_vars = {}
        # Create decision variables for p_{i,d}
        # These can be larger than h_i * h_{i+d} due to McCormick envelopes
        # How to track negatives?  
        # Loop over shifts from -N+1 to N-1
        
        # LtX: we'll use matrix representation to avoid complex variable tracking
        # We separate out main logic for LP, let LP solver handle constraints efficiently
        # Total rows = 1 (sum) + (2*N-1) (shifts)
        num_constraints = 1 + (2*N - 1) 

        # Assemble matrix A_ub and b_ub with the following structure:
        # All rows are:
        # [1, 1, ..., 1] for sum constraint
        # [0, ..., 0, p_{i,d}, 0, ..., 0] for shift d, with 1 at some valid positions
        # ... But better modeled in complete linear form.
        
        # Simple representation
        # Each correlated pair (k,i) is added once to coefficient matrix A
        # But with shifted scalings (for symmetry of h below)
        
        from scipy.sparse import csr_matrix
        # To make trusty double loop safe
        # Don't make it O(N^2). Make smarter.
        # Only consider actual valid for each subset
        # Use component-wise variable lists
        
        # We will use a slightly different structure:
        # Represent with proper LP matrix approach:
        constraints = []
        
        # First constraint: sum(h) = 1
        h_vars = [cp.Variable() for _ in range(N)]
        # Unconstrained
        constraints.append(cp.sum(h_vars) == target_sum)
        
        # All cross terms can be represented with p_vars
        # Goal: find h s.t. for all shifts d: sum_{i} h_i h_{i+d} >= (1 - t_mid) / dx
        # So each correlation term is: sum_{j=-N+1}^{N-1} e_d^T (h . h_shifted_d)
        # But it's easier to sample correlations 
        # Let C_d = sum_{i} h_i h_{i+d}
        # We want max_d (1 - C_d * dx) <= t -> min_d C_d * dx >= (1 - t)
        # i.e. sum h_i h_{i+d} >= (1 - t)/dx for all valid shifts d
        
        # The slow'ish part: build constraints for correlations
        # For dataset of size N, shifts are from -N+1 to N-1:
        # For each d in [-N+1, N-1] we compute valid pairs (i, i+d)
        # Problem with Taylor solution is not tractable easily with linear algebra if helpers
        
        # To simplify, we work with general system reconstruction helper
        # Use instead only definite computational formulation
        
        # For indices i, j = i + d:
        # i + d < N ->  i < N - d
        # i >= 0
        # So i ∈ [0, min(N - d - 1, N - 1)]
        
        # Use very careful and known symbolic constraints (action)
        # For now urge the weak but reliable-type construction
        
        # Method:
        # For large N (say 60-80) use CVXPY's outer+reduction on constraints
        # We predefine symbolic LP matrices, far simpler logic otherwise
        
        # Define all sum and constraints:
        expr_list = []
        
        # For               d from -N+1 to N-1, return indices
        def get_valid_indices(shift_d):
            # i valid when: i in [0, N), and i+d in [0, N)
            # -> 0 <= i <= N-1, and 0 <= i+d <= N-1
            # -> max(0, -d) <= i <= min(N-1, N-1-d)
            start = max(0, -shift_d)
            end = min(N - 1, N - 1 - shift_d)
            return range(start, end + 1) if start <= end else []
        
        # Set up variables: only create if less than many?
        hj = [cp.Variable(lower_bound=0, upper_bound=1) for _ in range(N)]
        
        # Now loop over shifts to add correlations (these define p_vars) 
        all_constraints = []
        for d in range(-N+1, N):  
            valid_indices = list(get_valid_indices(d))  # folds to valid [0, N) i
            if d >= 0:
                # case d positive: h_i h_{i+d} → for i from 0 to N-1-d
                valid_pairs = [(i, i+d) for i in valid_indices]
                p_ij = []
                for i, j in valid_pairs:
                    # Introduce McCormick approximation variables
                    p_ij_var = cp.Variable()
                    # instead of using big products of others:
                    # use four mils of constraints from the McCormick envelopes:
                    # p_ij >= 0  
                    # p_ij >= hj[i] + hj[j] - 1  
                    # p_ij <= hj[i]  
                    # p_ij <= hj[j]
                    all_constraints.append(p_ij_var >= 0)
                    all_constraints.append(p_ij_var >= hj[i] + hj[j] - 1)
                    all_constraints.append(p_ij_var <= hj[i])
                    all_constraints.append(p_ij_var <= hj[j])
                    p_ij.append(p_ij_var)
                
                # total sum over a shift must be ≥ (1 - t_mid)/dx
                if len(p_ij) > 0:
                    w_target = (1 - t_mid) / dx
                    sum_p = cp.sum(p_ij)
                    all_constraints.append(sum_p >= w_target)
            
            else:
                # d negative: same idea, but swap indices
                valid_pairs = [(i+d, i) for i in valid_indices]
                p_ij = []
                for i, j in valid_pairs:
                    p_ij_var = cp.Variable()
                    all_constraints.append(p_ij_var >= 0)
                    all_constraints.append(p_ij_var >= hj[i] + hj[j] - 1)
                    all_constraints.append(p_ij_var <= hj[i])
                    all_constraints.append(p_ij_var <= hj[j])
                    p_ij.append(p_ij_var)
                
                if len(p_ij) > 0: 
                    w_target = (1 - t_mid) / dx
                    sum_p = cp.sum(p_ij)
                    all_constraints.append(sum_p >= w_target)
                    
        # Build the LP equation
        prob = cp.Problem(cp.Minimize(0), all_constraints + [cp.sum(hj) == target_sum])
        try:
            prob.solve(solver='ECOS', verbose=False)
        except Exception as e:
            # print(f"Solver error with t={t_mid}: {e}")
            # Early terminate if solver can't handle
            lower_bound = t_mid
            continue

        if prob.status == 'optimal':
            # Retrieve h values from solver
            h_sol = np.array([v.value for v in hj]) 
            # post-adjustments
            if h_sol is not None:
                h_sol = np.clip(h_sol, 0, 1)
                # Re-scale to sum to target_sum
                h_sol = ((target_sum / np.sum(h_sol)) * h_sol)
                actual_c5 = compute_overlap(h_sol)
                if actual_c5 < best_c5:
                    best_c5 = actual_c5
                    best_h = h_sol.copy()
                upper_bound = t_mid
                # print(f"Feasible at t={t_mid}, C5 = {actual_c5:.6f}")
            else:
                lower_bound = t_mid
        else:
            lower_bound = t_mid
            
    # Final refinement: round to a valid binary function and verify
    if best_h is not None:
        # round to integer (as a group-based rounding)
        h_bin = round_to_binary(best_h)
        final_c5 = compute_overlap(h_bin)
        final_h = h_bin
        return final_h, final_c5, len(final_h)
    else:

        h_init = (target_sum / np.sum(h_init)) * h_init
        final_c5 = compute_overlap(h_init)
        if final_c5 < 0.38092:
            return h_init, final_c5, n_points
        else:
            return initial_h_values[:n_points], 0.38092, n_points
