import numpy as np
from scipy.optimize import differential_evolution, minimize
import time
from math import sqrt

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 bound for given h values."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute the correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum value of the correlation scaled by dx
    max_corr = np.max(corr)
    return max_corr * dx

def initial_h_values():
    """Return initial construction for h values."""
    # Use the initial construction from the problem description
    n = 70
    h = np.zeros(n)
    dx = 2.0 / n
    # Simple step function
    step_start = 0.5
    step_end = 1.5
    for i in range(n):
        x = i * dx
        if step_start <= x < step_end:
            h[i] = 1.0
        else:
            h[i] = 0.0
    # Normalize to have integral = 1
    h = h * (1.0 / np.sum(h) * dx)
    return h.tolist()

def objective_function(h_values):
    """Objective function: minimize the maximum overlap."""
    return evaluate_erdos_solution(h_values)

def constraint_sum(h_values):
    """Constraint: sum of h_values * dx = 1."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    return np.sum(h_values) * dx - 1.0

def run(seed=42, budget_s=600, **kwargs):
    """Main function to find improved upper bound for C5."""
    np.random.seed(seed)
    
    # Initialize variables
    n_points = 70
    dx = 2.0 / n_points
    
    # Start with initial construction
    h_initial = initial_h_values()
    
    # Set bounds for optimization: [0, 1] for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Add the constraint that sum(h) * dx = 1 -> sum(h) = n_points / 2
    # We'll use a penalty method or let optimizer handle it
    
    # Initialize best solution
    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Optimize using differential evolution
    start_time = time.time()
    max_time = budget_s
    
    # Perform optimization with constraints handled via penalty
    def penalized_objective(h):
        penalty = 0
        # Add penalty for violating sum constraint
        sum_h = np.sum(h) * dx
        if abs(sum_h - 1.0) > 1e-5:
            penalty += 1e6 * (abs(sum_h - 1.0))**2
        return objective_function(h) + penalty
    
    try:
        # Use differential evolution for global optimization
        result = differential_evolution(
            penalized_objective,
            bounds,
            maxiter=500,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            disp=False
        )
        
        if not result.success:
            return best_h, best_c5, n_points
            
        new_h = result.x
        new_c5 = objective_function(new_h)
        
        # Validate the solution
        if new_c5 < best_c5:
            best_h = new_h.tolist()
            best_c5 = new_c5
            
    except Exception as e:
        pass
    
    # Try a few individual optimizations from different starting points
    for i in range(10):
        if time.time() - start_time > max_time * 0.9:
            break
            
        # Perturb initial solution slightly
        h_perturbed = h_initial.copy()
        for j in range(len(h_perturbed)):
            if np.random.random() < 0.1:
                h_perturbed[j] = max(0, min(1, h_perturbed[j] + np.random.normal(0, 0.05)))
        
        # Normalize
        sum_h = np.sum(h_perturbed) * dx
        h_perturbed = [x / sum_h * (n_points / 2) for x in h_perturbed]
        
        # Local optimization using Nelder-Mead
        try:
            res = minimize(
                penalized_objective,
                h_perturbed,
                method='Nelder-Mead',
                options={'maxiter': 500}
            )
            
            if not res.success:
                continue
                
            new_h = res.x
            if len(new_h) != n_points:
                continue
                
            new_c5 = objective_function(new_h)
            if new_c5 < best_c5:
                best_h = new_h.tolist()
                best_c5 = new_c5
                
        except Exception:
            continue
    
    # If there's a better solution from initial, keep it
    init_c5 = evaluate_erdos_solution(h_initial)
    if init_c5 < best_c5:
        best_h = h_initial
        best_c5 = init_c5

    return best_h, best_c5, n_points
