def compute_c5_and_shift(h):
    N = len(h)
    padded_len = 2*N
    H = np.fft.rfft(h, n=padded_len)
    G = np.fft.rfft(1 - h, n=padded_len)
    conv = np.fft.irfft(H * G, n=padded_len)
    # conv length padded_len; real.
    # want full correlation: start index N-1 (shift 0)
    # we compute offsets = conv[N-1: N*2-1]
    c = conv[N-1:]  # length N
    max_idx = np.argmax(c)  # shift with max correlation
    max_val = c[max_idx]
    dx = 2.0 / N
    C5 = max_val * dx
    return C5, max_idx, max_val

def project_to_sum_and_bounds(v, target_sum):
    # clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # iterative waterfilling
    diff = target_sum - v.sum()
    # tolerance
    tol = 1e-12
    max_iter = 100
    for _ in range(max_iter):
        if abs(diff) <= tol:
            break
        # free indices: not at bound
        free = (v > 0 + 1e-12) & (v < 1 - 1e-12)
        nfree = np.count_nonzero(free)
        if nfree == 0:
            # All at bounds, cannot adjust further; assign to return current and break.
            break
        delta = diff / nfree
        # Update free indices
        v[free] += delta
        # Clip again (some might violate bounds)
        v = np.clip(v, 0.0, 1.0)
        diff = target_sum - v.sum()
    # final adjustment may cause small residual; distribute among any variable not at bound.
    if abs(diff) > tol:
        # adjust by adding diff to the variable with largest slack
        idx_possible = np.where((v > 0) & (v < 1))[0]
        if idx_possible.size:
            v[idx_possible[0]] += diff
            # clip if needed
            v = np.clip(v, 0.0, 1.0)
        else:
            # if all at bounds, fallback: adjust the first variable within bounds
            v[0] = max(0, min(1, v[0] + diff))
    # final clip
    v = np.clip(v, 0.0, 1.0)
    return v
