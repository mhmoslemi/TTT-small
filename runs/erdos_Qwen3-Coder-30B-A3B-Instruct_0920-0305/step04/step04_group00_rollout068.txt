import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def project_onto_box_and_simplex(h, target_sum, dx):
    """Project h onto the set 0 <= h <= 1 and sum(h)*dx = target_sum"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h) * dx
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Waterfilling algorithm
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 30
    for _ in range(max_iter):
        mid = (low + high) / 2
        p = np.clip(h - mid, 0, 1)
        p_sum = np.sum(p) * dx
        if p_sum > target_sum:
            low = mid
        else:
            high = mid
        if abs(p_sum - target_sum) < tol:
            break
    p = np.clip(h - low, 0, 1)
    return p

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def decode_symmetric_vector(heights, x_points, n):
    """Interpolate a symmetric decreasing vector from heights at x_points"""
    # Assume x_points are ordered from 0 to 1
    # Create full symmetric vector on [0,2]
    left = np.interp(np.linspace(0, 1, n//2), x_points, heights, left=1, right=0)
    right = left[::-1]
    full_vector = np.concatenate([left, right])
    return full_vector

def objective_de(x):
    """DE objective for low-dimensional symmetric family"""
    K = len(x) - 1  # Number of breakpoints minus one
    x_points = np.linspace(0, 1, K + 1)
    heights = np.sort(x)[::-1]  # sort descending
    n = 200
    dx = 2.0 / n
    h = decode_symmetric_vector(heights, x_points, n)
    # Scale to ensure integral=1
    h_sum = np.sum(h) * dx
    h = h * (0.5 * n / h_sum)
    # Clip overflow
    h = np.clip(h, 0, 1)
    return evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n = 200
    dx = 2.0 / n

    # The parametric shape
    K = 5  # Number of breakpoints for symmetric decaying piecewise constant shape

    # Define time slices
    t_de = 0.7 * budget_s
    t_pgd = 0.2 * budget_s
    t_final = 0.1 * budget_s
    t_start = time.time()

    # DE stage
    bounds = [(0, 1) for _ in range(K + 1)]
    de_res = differential_evolution(
        objective_de,
        bounds,
        seed=seed,
        maxiter=max(100, int(t_de / 2)),  # Roughly 2 seconds per iter
        popsize=15,
        strategy='best1bin',
        recombination=0.7,
        atol=1e-8,
        rtol=1e-8
    )

    best_x = de_res.x
    best_heights = np.sort(best_x)[::-1]
    x_points = np.linspace(0, 1, K + 1)
    hbest = decode_symmetric_vector(best_heights, x_points, n)
    hbest = hbest * (0.5 * n / (np.sum(hbest) * dx))
    hbest = np.clip(hbest, 0, 1)
    best_c5 = evaluate_c5(hbest)

    # PGD stage with smooth max surrogate
    def smooth_max(h, beta=15):
        corr = correlate(h, 1-h, mode='full')
        corr *= dx
        exp_ = np.exp(beta * corr)
        return np.log(np.sum(exp_)) / beta

    def smooth_max_grad(h, beta=15):
        # Get correlations for all shifts
        corr = correlate(h, 1-h, mode='full')
        corr *= dx
        exp_ = np.exp(beta*corr)
        weights = exp_ / np.sum(exp_)
        # Now compute gradient of each term and form weighted combo
        # Not very efficient but acceptable for our small n
        grad = np.zeros_like(h)
        for shift in range(-n+1, n):
            if shift >= 0:
                # h_i * (1 - h_{i+shift})
                indices = slice(0, n-shift)
                shifted_indices = slice(shift, n)
                grad[indices] += weights[shift + n - 1] * dx * (1 - h[shifted_indices])
                grad[shifted_indices] += weights[shift + n - 1] * dx * (-h[indices])
            else:
                # shift is negative
                indices = slice(-shift, n)
                shifted_indices = slice(0, n + shift)
                grad[indices] += weights[shift + n - 1] * dx * (1 - h[shifted_indices])
                grad[shifted_indices] += weights[shift + n - 1] * dx * (-h[indices])
        return grad

    # Refine via projected gradient descent
    h = hbest.copy()
    h = project_onto_box_and_simplex(h, 0.5 * n, dx)
    beta = 10.0
    eta = 0.03
    max_pgd_iter = int(t_pgd / 0.05)
    for i in range(max_pgd_iter):
        if time.time() - t_start > t_start + t_de + t_pgd:
            break
        grad = smooth_max_grad(h, beta)
        h = h - eta * grad
        # Project
        h = project_onto_box_and_simplex(h, 0.5 * n, dx)
        # Update beta if needed
        if beta < 20:
            beta += 0.5
        new_c5 = evaluate_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            hbest = h.copy()

    # Final polishing
    if time.time() - t_start < t_start + t_de + t_pgd + t_final:
        def objective_final(h):
            return evaluate_c5(h)

        def constraint_eq(h):
            return np.sum(h) * dx - 0.5 * n

        bounds = [(0, 1) for _ in range(n)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective_final,
                hbest,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10}
            )
            if res.success:
                h_final = res.x.copy()
                h_final = h_final * (0.5 * n / (np.sum(h_final) * dx))
                h_final = np.clip(h_final, 0, 1)
                c5_final = evaluate_c5(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    hbest = h_final
        except Exception:
            pass

    return (hbest.tolist(), best_c5, n)
