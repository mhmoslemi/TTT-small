import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import GLPK_MI, CBC, ECOS_BB

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Convert to MILP formulation
    dx = 2.0 / n_points
    
    # Define the model
    h = cp.Variable(n_points, integer=True)
    p = cp.Variable((n_points, n_points), integer=True)
    M = cp.Variable(integer=True)
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        M >= 0,
        cp.sum(h) == n_points / 2,  # exactly 21 ones
        h[0] == 1,  # symmetry breaking constraint
    ]
    
    # Logical AND constraints for p[i,l] = h[i] AND h[i+l] for l > 0
    for l in range(1, n_points):
        for i in range(n_points - l):
            constraints += [
                p[i, l] <= h[i],
                p[i, l] <= h[i + l],
                p[i, l] >= h[i] + h[i + l] - 1,
            ]
    
    # Compute A_l for non-zero l
    for l in range(1, n_points):
        A_l_plus = cp.sum(h[:n_points-l])
        A_l_minus = cp.sum(h[l:])
        S_l = cp.sum(p[:n_points-l, l])
        constraints += [
            A_l_plus - S_l <= M,
            A_l_minus - S_l <= M,
        ]
    
    # Objective
    objective = cp.Minimize(M)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Set remaining time limit
    remaining_time = budget_s - (time.time() - start_time)

        best_M = float('inf')

            if (time.time() - start_time) > budget_s - 2:
                break
            
            try:
                prob.solve(solver=solver, time_limit=remaining_time)
            except:
                continue
                
            if prob.status in ["optimal", "optimal_inaccurate"]:

                    h_values = h.value
