import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

# Global variables for the constraint
TARGET_SUM = None

def compute_corr_fft(h):
    """Compute the full convolution using FFT for faster evaluation"""
    n = len(h)
    h_rev = h[::-1]
    b = 1 - h
    # Use real FFT with proper length
    L = 2 * n - 1
    h_fft = fft(h_rev, n=L)
    b_fft = fft(b, n=L)
    corr = ifft(h_fft * b_fft).real
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from the correlation output"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using fast FFT convolution"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Project x onto the box-constrained sum equality hyperplane using shift-and-clip"""
    # Binary search for lambda
    a, b = -1.0, 1.0
    for _ in range(50):  # binary search well enough
        lam = (a + b) / 2
        y = np.clip(x + lam, 0, 1)
        sum_y = np.sum(y)
        if sum_y < target_sum:
            a = lam
        else:
            b = lam
    y = np.clip(x + a, 0, 1)
    return y

def random_pair_move(h):
    """Perform a feasible pairwise mass transfer move"""
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    delta_max = min(h[i], 1 - h[j])
    delta_min = max(-h[j], h[i] - 1)
    if delta_min < delta_max:
        delta = np.random.uniform(delta_min, delta_max)
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        return h_new
    else:
        return h.copy()

def guided_move(h, worst_lag):
    """Perform a deterministic move targeting the worst lag"""
    n = len(h)
    g = np.zeros(n)
    # Compute gradient as sum of neighbors of lag
    # h[i] contributes to g[i+lag] and g[i-lag]
    if worst_lag > 0:
        g[worst_lag:] += h[:-worst_lag]
        g[:-worst_lag] += h[worst_lag:]
    else:
        g[:worst_lag] += h[-worst_lag:]
        g[-worst_lag:] += h[:worst_lag]
    
    # Avoid causing infeasibility
    min_idx = np.argmin(g)
    max_idx = np.argmax(g)
    if min_idx == max_idx:
        return h.copy()
    # Determine feasible step size
    delta_max = min(h[min_idx], 1 - h[max_idx])
    delta_min = max(-h[max_idx], h[min_idx] - 1)
    if delta_min >= delta_max:
        return h.copy()
    elif delta_max > 0:
        # Pick an amount to transfer
        delta = min(0.1, delta_max)
        h_new = h.copy()
        h_new[min_idx] -= delta
        h_new[max_idx] += delta
        return h_new
    else:
        return h.copy()

def anneal(h0, run_seconds, rng):
    """Run simulated annealing from a starting point"""
    # Set initial and final temperature
    T0 = 0.005
    Tf = 1e-6
    current_h = h0.copy()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    start_time = time.time()
    step_count = 0
    while time.time() - start_time < run_seconds:
        T = T0 * (1 - (time.time() - start_time) / run_seconds) + Tf * (time.time() - start_time) / run_seconds
        # 1. Random move
        trial_h = random_pair_move(current_h)
        trial_h = project_to_sum_box(trial_h, TARGET_SUM)
        trial_c5 = evaluate_c5(trial_h)
        if trial_c5 < current_c5 or np.random.rand() < np.exp(-(trial_c5 - current_c5) / T):
            current_h = trial_h
            current_c5 = trial_c5
        # 2. Periodic guided move (every 10 random steps)
        if step_count % 10 == 0:
            corr = compute_corr_fft(current_h)
            worst_lag = np.argmax(corr) - (len(corr) // 2)
            guided_h = guided_move(current_h, worst_lag)
            if not np.array_equal(current_h, guided_h):  # Only if move was valid
                guided_h = project_to_sum_box(guided_h, TARGET_SUM)
                guided_c5 = evaluate_c5(guided_h)
                if guided_c5 < current_c5:
                    current_h = guided_h
                    current_c5 = guided_c5
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
        step_count += 1
    return best_h, best_c5

def constraint_eq(h):
    """Equality constraint for optimization"""
    return np.sum(h) - TARGET_SUM

def objective(h):
    """Objective function for scipy optimizer"""
    return evaluate_c5(h)

def local_refine(h):
    """Use SLSQP to refine the solution"""
    n = len(h)
    bounds = [(0, 1) for _ in range(n)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12})
    if res.success:
        refined = res.x
        refined = project_to_sum_box(refined, TARGET_SUM)
        return refined, evaluate_c5(refined)
    return h, evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 120  # fine grid
    TARGET_SUM = n / 2.0
    # Initial construction from the provided one
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Interpolate to the new n-point grid
        h0 = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h_init)), h_init)
        h0 = project_to_sum_box(h0, TARGET_SUM)
    else:
        # Fallback - random with correct sum
        h0 = np.random.uniform(0, 1, n)
        h0 = project_to_sum_box(h0, TARGET_SUM)

    # Phase A: Multi-restart simulated annealing
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    phase_a_time = 0.7 * budget_s
    restart_count = 0
    while time.time() - start_time < phase_a_time:
        # Initializers
        init_type = restart_count % 3
        if init_type == 0:
            # Interpolated from initial_h_values if available, else random
            if 'initial_h_values' in globals():
                h0 = np.array(initial_h_values)
                h0 = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h0)), h0)
            else:
                h0 = np.random.uniform(0, 1, n)
            h0 = project_to_sum_box(h0, TARGET_SUM)
        elif init_type == 1:
            # Binary vector
            h0 = np.zeros(n)
            ones = np.random.choice(n, n//2, replace=False)
            h0[ones] = 1
            h0 = project_to_sum_box(h0, TARGET_SUM)
        else:  # init_type == 2
            # Random continuous
            h0 = np.random.uniform(0, 1, n)
            h0 = project_to_sum_box(h0, TARGET_SUM)
        h0 = project_to_sum_box(h0, TARGET_SUM)
        # Run 5-second annealing
        current_h, current_c5 = anneal(h0, 5, np.random.RandomState(seed + restart_count))
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
        restart_count += 1

    # Phase B: Local refinement with SLSQP
    max_refinements = 10
    for i in range(max_refinements):
        if time.time() - start_time > 0.95 * budget_s:
            break
        try:
            refined_h, refined_c5 = local_refine(best_h)
            if refined_c5 < best_c5:
                best_h = refined_h.copy()
                best_c5 = refined_c5
        except:
            pass

    # Phase C: Polishing with tweaks
    sub_phase_time = budget_s - (time.time() - start_time)
    while time.time() - start_time < budget_s - 0.5:
        # Add small random noise, reproject, and refine
        noise_level = 0.02
        h_perturbed = best_h + np.random.normal(0, noise_level, n)
        h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
        try:
            refined_h, refined_c5 = local_refine(h_perturbed)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
        except:
            pass
    return (best_h.tolist(), float(best_c5), n)
