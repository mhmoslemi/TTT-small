import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(x, n_points):
    """Project x onto feasible set: [0,1]^n and sum = n/2"""
    y = np.clip(x, 0, 1)
    target = n_points / 2.0
    # Simple iterative adjustment
    for _ in range(100):
        s = np.sum(y)
        if abs(s - target) < 1e-9:
            break
        if s > target:
            # Reduce uniformly from values not at lower bound
            free = y > 0
            if not free.any():
                break
            decrement = (s - target) / np.sum(free)
            y[free] = y[free] - decrement
        else:
            # Increase uniformly from values not at upper bound
            free = y < 1
            if not free.any():
                break
            increment = (target - s) / np.sum(free)
            y[free] = y[free] + increment
        y = np.clip(y, 0, 1)
    return y
