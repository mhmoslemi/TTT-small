import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use a parameterized symmetric shape: triangular with variable width
    def parametric_h(cp):
        """Generate h values based on plateau width parameter cp in [0,1]"""
        n = 42  # Always use 42 points as initial
        x = np.linspace(0, 2, n)
        h = np.zeros(n)
        
        # Central plateau width = 2 * cp, height = 1/(1 + cp)
        # If cp=0, triangle; if cp=1, uniform height 0.5
        H = 1.0 / (1.0 + cp)
        cp_width = cp
        
        for i in range(n):
            dist = abs(x[i] - 1.0)
            if dist <= cp_width:
                h[i] = H
            else:
                # Linear decay from plateau edge to boundary
                remaining = dist - cp_width
                if remaining <= 1.0 - cp_width:
                    h[i] = H * (1.0 - remaining / (1.0 - cp_width))
                else:
                    h[i] = 0.0
        
        # Renormalize to integral = 1
        integral = np.sum(h) * (2.0 / n)
        h = h * (1.0 / integral)
        
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        return h
    
    def objective(cp):
        h = parametric_h(cp)
        return evaluate_c5(h)
    
    # Optimize parameter cp in [0, 1]
    result = differential_evolution(
        objective,
        bounds=[(0.0, 1.0)],
        maxiter=500,
        popsize=15,
        seed=seed,
        disp=False
    )
    
    best_cp = result.x[0]
    best_h = parametric_h(best_cp)
    best_c5 = evaluate_c5(best_h)
    
    # Local refinement with SLSQP
    def objective_local(h):
        # Normalize to integral = 1
        dx = 2.0 / len(h)
        integral = np.sum(h) * dx
        if integral > 0:
            h = h * (1.0 / integral)
        h = np.clip(h, 0, 1)
        return evaluate_c5(h)
    
    bounds = [(0, 1) for _ in range(len(best_h))]
    
    from scipy.optimize import minimize
    # Try to improve with local search
    try:
        res = minimize(
            objective_local,
            best_h,
            method='SLSQP',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            refined_h = res.x
            # Normalize
            dx = 2.0 / len(refined_h)
            integral = np.sum(refined_h) * dx
            if integral > 0:
                refined_h = refined_h * (1.0 / integral)
            refined_h = np.clip(refined_h, 0, 1)
            refined_c5 = evaluate_c5(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass
    
    return (best_h.tolist(), best_c5, len(best_h))
