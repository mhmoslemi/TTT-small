import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_binary_milp(n):
    """Solve binary MILP to find minimum t such that max N_d <= t"""
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(x) == n // 2]
    
    # Add constraints for each shift
    for d in range(-n+1, n):
        if d == 0:
            continue
        w_vars = []
        if d > 0:
            rng = range(n - d)
        else:
            rng = range(-d, n)
        for i in rng:
            w = cp.Variable(boolean=True)
            constraints.append(w <= x[i])
            constraints.append(w <= x[i+d])
            constraints.append(w >= x[i] + x[i+d] - 1)
            w_vars.append(w)
        N_d = cp.sum(x) - cp.sum(w_vars)
        constraints.append(N_d <= t)
    constraints.append(t >= 0)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    # Try different solvers
    solvers = [cp.CBC, cp.GLPK_MI, cp.ECOS_BB]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False, max_iters=5000)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                break
        except:
            continue
    
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        raise RuntimeError("Failed to solve binary MILP")
        
    x_vals = np.round(x.value).astype(float)
    t_val = int(round(t.value))
    
    return x_vals, t_val
