import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Parameters
    n_points = 200
    k_breakpoints = 5  # Number of breakpoints (excluding 0 and 1)
    dx = 2.0 / n_points

    def decode_h(u_values):
        """Decode k+1 breakpoint heights into full h vector"""
        # Sort the values to enforce monotonic decreasing
        heights = np.sort(u_values)[::-1]
        # Ensure boundary values are 0 and 1
        heights = np.clip(np.append([1.0], heights), 0, 1.0)
        # Get breakpoint locations
        breakpoints = np.linspace(0, 1, k_breakpoints + 1)
        # Interpolate heights to full domain [0,1]
        h_half = np.interp(np.linspace(0, 1, n_points//2 + 1), breakpoints, heights)
        # Mirror to full domain [0,2]
        h_full = np.concatenate([h_half, h_half[-2::-1]])
        # Normalize the integral constraint
        integral = np.sum(h_full) * dx
        scale = 1.0 / integral
        h_full *= scale
        # Clip to [0,1]
        h_full = np.clip(h_full, 0, 1)
        # Renormalize to ensure exact area constraint
        integral = np.sum(h_full) * dx
        h_full *= (1.0 / integral)
        return h_full

    def evaluate_c5_full(h):
        """Compute C5 using np.correlate"""
        corr = correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    def objective(u):
        h = decode_h(u)
        return evaluate_c5_full(h)

    # Global optimization with DE
    bounds = [(0, 1) for _ in range(k_breakpoints)]
    t_left = budget_s * 0.7
    if t_left < 10:
        t_left = 10

    # Run DE
    res = differential_evolution(
        func=objective,
        bounds=bounds,
        seed=seed,
        maxiter=100,
        popsize=min(15, 2*k_breakpoints),
        disp=False,
        time_limit=t_left,
        worker=None,
        options={"workers": 1},
    )

    best_u = res.x
    h_opt = decode_h(best_u)
    best_c5 = evaluate_c5_full(h_opt)

    # Smooth approximation for gradient-based refinement
    def smooth_max(h, beta=15.0):
        corr = correlate(h, 1 - h, mode="full")
        corr *= dx
        exp_corr = np.exp(beta * corr)
        return np.log(np.sum(exp_corr)) / beta

    def smooth_max_grad(h, beta=15.0):
        corr = correlate(h, 1 - h, mode="full")
        corr *= dx
        exp_corr = np.exp(beta * corr)
        w = exp_corr / np.sum(exp_corr)

        # Build gradient per shift
        grad = np.zeros(n_points)
        for k in range(-n_points + 1, n_points):
            # Index k in full correlation: range [-n+1, n-1]
            idx = k + n_points - 1
            if idx < 0 or idx >= 2*n_points - 1:
                continue
            w_k = w[idx]
            # Compute ∂f_k / ∂h_i = dx * (1 - h_{i+k} - h_{i-k})
            if k == 0:
                grad += w_k * 1.0
            else:
                # Only handle finite differences manually here
                corr_grad = np.zeros_like(h)
                # No loop to avoid issue, just direct assignment for valid shifts
                for i in range(n_points):
                    if i + k >= 0 and i + k < n_points:
                        corr_grad[i] += -w_k * dx
                    if i - k >= 0 and i - k < n_points:
                        corr_grad[i] += -w_k * dx
                grad += corr_grad
        return grad

    # Gradient-based update with projection
    def project_onto_box_simplex(h, target_sum):
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Water filling algorithm to normalize to exactly target_sum
        S = np.sum(h) * dx
        if abs(S - target_sum) < 1e-10:
            return h

        def waterfilling(lam):
            h_clipped = np.clip(h - lam, 0, 1)
            return np.sum(h_clipped) * dx - target_sum

        # Binary search for lambda
        low, high = -2.0, 2.0
        for _ in range(50):
            mid = (low + high) / 2
            val = waterfilling(mid)
            if val > 0:
                low = mid
            else:
                high = mid
        lam = (low + high) / 2
        h = np.clip(h - lam, 0, 1)
        return h

    # Gradient descent
    t_left = budget_s * 0.2
    eta_decay = 1000
    beta = 15.0
    h_current = h_opt.copy()

    i = 0
    while i < 200 and (time.time() - start_time) < budget_s * 0.95:
        grad = smooth_max_grad(h_current, beta)
        eta = 0.01 / (1 + i / eta_decay)
        h_new = h_current - eta * grad
        # Project back onto feasible set
        h_new = project_onto_box_simplex(h_new, 1.0)
        current_c5 = evaluate_c5_full(h_new)
        if current_c5 < best_c5:
            h_opt = h_new.copy()
            best_c5 = current_c5
        h_current = h_new.copy()
        i += 1

        if (time.time() - start_time) > budget_s * 0.95:
            break

    # Final SLSQP polishing (if time remains)
    if (time.time() - start_time) < budget_s * 0.99:
        from scipy.optimize import minimize

        def constraint_eq(h):
            return np.sum(h) * dx - 1.0

        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}

        res = minimize(
            evaluate_c5_full,
            h_opt,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_final = res.x
            c5_final = evaluate_c5_full(h_final)
            if c5_final < best_c5:
                h_opt = h_final
                best_c5 = c5_final

    return (h_opt.tolist(), best_c5, n_points)
