import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def construct_h_from_params(params, n_points):
    x1, x2, h1, h2 = params
    # Ensure monotonic shape
    x1 = max(1e-6, min(0.499999, x1))
    x2 = max(x1 + 1e-6, min(0.499999, x2))
    h1 = min(1.0, max(0.0, h1))
    h2 = min(1.0, max(0.0, h2))
    
    # Non-symmetric `h` construction
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    xvals = np.linspace(0, 2, n_points)
    
    # Define the piecewise linear part from (0,0) to (x1,h1)
    # Then to (x2,h2), then to (1,0)
    for i in range(n_points):
        x = xvals[i]
        if x <= x1:
            h[i] = h1 * x / x1
        elif x <= x2:
            h[i] = h2 + (h1 - h2) * (x2 - x) / (x2 - x1)
        elif x <= 1:
            h[i] = (1 - x) * h2 / (1 - x2)
        else:
            # Beyond 1; mirror to [0,1] with symmetry but reflected accordingly
            x_ref = 2 - x
            if x_ref <= x1:
                h[i] = h1 * x_ref / x1
            elif x_ref <= x2:
                h[i] = h2 + (h1 - h2) * (x2 - x_ref) / (x2 - x1)
            else:
                h[i] = (1 - x_ref) * h2 / (1 - x2)
    
    # Compute normalization
    integral = np.sum(h) * dx
    if integral > 0:
        h = h * (n_points / 2) / integral
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def objective(params):
    h = construct_h_from_params(params, n_points)
    return evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 200
    
    # Bounds: x1, x2 in [0.001, 0.499], h1, h2 in [0,1]
    bounds = [(1e-6, 0.499), (1e-6, 0.499), (0, 1), (0, 1)]
    
    # Global optimization with DE
    de_result = differential_evolution(objective, bounds, seed=seed, maxiter=100, popsize=16)
    best_params = de_result.x
    best_c5 = de_result.fun
    
    # Fine-tuning with local optimizer
    res = minimize(objective, best_params, method='L-BFGS-B', bounds=bounds)
    if res.success:
        refined_params = res.x
        refined_h = construct_h_from_params(refined_params, n_points)
        refined_c5 = evaluate_c5(refined_h)
        if refined_c5 < best_c5:
            best_params = refined_params
            best_c5 = refined_c5

    # Final full-scan evaluation
    final_h = construct_h_from_params(best_params, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
