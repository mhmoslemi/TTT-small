import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the same grid size as initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Time budget for MIP solver (half the total budget)
    mip_time_limit = budget_s / 2
    remaining_time = budget_s - (time.time() - start_time)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    try:
        # Define the MILP model
        h = cp.Variable(n_points, boolean=True)
        M = cp.Variable(integer=True, name="M")
        
        # Construct constraints
        constraints = []
        
        # Integral constraint
        constraints.append(cp.sum(h) == n_points / 2)
        
        # Symmetry breaking
        constraints.append(h[0] == 1)
        
        # Product variables p_{i,l}
        p = {}
        for l in range(1, n_points):
            for i in range(n_points - l):
                p[(i, l)] = cp.Variable(boolean=True)
                # McCormick constraints
                constraints.append(p[(i, l)] <= h[i])
                constraints.append(p[(i, l)] <= h[i + l])
                constraints.append(p[(i, l)] >= h[i] + h[i + l] - 1)
                
        # Worst-case overlap constraints
        for l in range(1, n_points):
            # A_l^+ = sum of h_i for i in 0..n-l-1
            A_plus = cp.sum(h[:n_points-l])
            # A_l^- = sum of h_i for i in l..n-1 = total_ones - sum of h_i for i in 0..l-1
            A_minus = cp.sum(h[l:])
            # S_l = sum of p_{i,l}
            S_l = cp.sum([p[(i, l)] for i in range(n_points - l)])
            
            constraints.append(A_plus - S_l <= M)
            constraints.append(A_minus - S_l <= M)
            
        # Objective
        objective = cp.Minimize(M)
        
        # Solve the MILP problem
        prob = cp.Problem(objective, constraints)
        
        # Set time limit
        prob.solve(solver=CPLEX, time_limit=mip_time_limit)
        
        if prob.status in ['optimal', 'feasible']:
            # Get the binary solution
            h_solution = np.array(h.value).astype(int)
            M_solution = M.value
            
            # Compute C5 from M_solution
            dx = 2.0 / n_points
            c5_milp = M_solution * dx
            
            # Use this as starting point for polishing
            h_float = h_solution.astype(float)
            best_h = h_float.copy()
            best_c5 = evaluate_c5(h_float)
        else:
            # If not solved, fall back to heuristic
            pass
            
    except:
        # If MIP solver fails or is not available, fallback to heuristic
        pass
        
    # Perform polishing with SLSQP on the best binary solution found
    # Now also do polishing using continuous optimization
    if (time.time() - start_time) < (budget_s - 2):
        for restart in range(1000):  # More restarts for better search
            if (time.time() - start_time) > (budget_s - 2):
                break
                
            # Try different starting points
            # Start from current best
            start_h = best_h.copy()
            
            # Start with slightly perturbed version
            noise = np.random.normal(0, 0.02, n_points)
            start_h_perturbed = start_h + noise
            start_h_perturbed = np.clip(start_h_perturbed, 0, 1)
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(start_h_perturbed) * dx)
            start_h_perturbed = start_h_perturbed * scale_factor
            
            def objective_continuous(h):
                return evaluate_c5(h)
                
            bounds = [(0, 1) for _ in range(n_points)]
            
            def constraint_eq(h):
                dx = 2.0 / n_points
                return np.sum(h) * dx - 1.0
                
            cons = {'type': 'eq', 'fun': constraint_eq}
            
            try:
                res = minimize(
                    objective_continuous,
                    start_h_perturbed,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
                )
                if res.success:
                    h_candidate = res.x
                    scale = 1.0 / (np.sum(h_candidate) * dx)
                    h_candidate = h_candidate * scale
                    c5_candidate = objective_continuous(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
            except:
                pass

    return (best_h.tolist(), best_c5, n_points)
