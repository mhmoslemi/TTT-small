def evaluate_c5_fft(h):

    max_corr = np.max(conv)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # Choose n_points = 84 for moderate size
    n_initial = len(initial_h_values)
    n_points = max(84, n_initial*2)   # maybe 84.
    # Resample initial shape to new n
    x_original = np.linspace(0, 2, n_initial)
    x_new = np.linspace(0, 2, n_points)
    # Interpolate initial shape
    init_h = np.interp(x_new, x_original, initial_h_values)
    # Project to constraints
    target_sum = n_points / 2
    init_h = project_to_sum_box(init_h, target_sum)

    best_h = init_h.copy()
    best_c5 = evaluate_c5_fft(best_h)

    # Phase 1: Simulated annealing with multiple restarts
    # Allocate time: 0.7 * budget
    phase1_budget = budget_s * 0.70
    # Determine time per restart: at least maybe 5 sec => 0.05*budget => 30 sec each => about 20 restarts.
    # We'll break on time.

    restart = 0
    while time.time() - start_time < phase1_budget:
        # assign time for this run, say 5 sec
        remaining = phase1_budget - (time.time() - start_time)
        run_time = min(5.0, remaining)
        # Random initial h for this run
        # Choose mode based on restart index
        if restart % 3 == 0:
            # start from best_h
            h0 = best_h.copy()
            # Apply small random perturbation
            h0 += np.random.normal(scale=0.05, size=n_points)
            h0 = np.clip(h0, 0, 1)
            # Reproject to sum
            h0 = project_to_sum_box(h0, target_sum)
        elif restart % 3 == 1:
            # start from random binary pattern
            h0 = np.zeros(n_points)
            # randomly choose n_points/2 positions
            ones_idx = np.random.choice(n_points, size=n_points//2, replace=False)
            h0[ones_idx] = 1.0
        else:
            # random continuous uniform
            h0 = np.random.rand(n_points)
            # project
            h0 = project_to_sum_box(h0, target_sum)

        # run annealing
        h_candidate, c_candidate = simulated_annealing(h0, target_sum, run_time, seed + restart)
        if c_candidate < best_c5:
            best_c5 = c_candidate
            best_h = h_candidate.copy()
        restart += 1

    # Phase 2: Local refinement using SLSQP
    # Allocate remaining time
    remaining_time = budget_s - (time.time() - start_time)
    # Use a few attempts within remaining time
    attempts = 0
    while remaining_time > 5 and attempts < 10:
        # start from best_h with slight noise
        h0 = best_h + np.random.normal(scale=0.02, size=n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = project_to_sum_box(h0, target_sum)
        # refine
        try:
            cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(evaluate_c5_fft, h0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 2000, 'ftol':1e-12})
            if res.success:
                h_ref = res.x
                # ensure feasibility (some numerical slack)
                h_ref = np.clip(h_ref, 0, 1)
                h_ref = project_to_sum_box(h_ref, target_sum)
                c_ref = evaluate_c5_fft(h_ref)
                if c_ref < best_c5:
                    best_c5 = c_ref
                    best_h = h_ref.copy()
        except Exception as e:
            pass
        attempts += 1
        remaining_time = budget_s - (time.time() - start_time)

    # Ensure final best_h satisfies constraints accurately
    best_h = project_to_sum_box(best_h, target_sum)
    best_c5 = evaluate_c5_fft(best_h)

    # Return result as list
    return (best_h.tolist(), float(best_c5), n_points)

    n = len(h)
