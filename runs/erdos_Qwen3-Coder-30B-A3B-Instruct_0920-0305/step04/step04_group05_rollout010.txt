import numpy as np
from scipy.optimize import differential_evolution, basinhopping, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(x):
    """Project x onto feasible set: bounds [0,1] and sum = n/2"""
    y = np.clip(x, 0, 1)
    target = len(x) / 2.0
    # Iteratively adjust to meet sum constraint
    for _ in range(100):
        current_sum = np.sum(y)
        if abs(current_sum - target) < 1e-9:
            break
        diff = current_sum - target
        if diff > 0:  # Too much mass, reduce proportionally
            free = y > 0
            if not free.any():
                break
            # Reduce each non-zero element evenly
            reduction = diff / np.sum(free)
            y[free] -= reduction
            y = np.clip(y, 0, 1)
        else:  # Too little mass, increase proportionally
            free = y < 1
            if not free.any():
                break
            # Increase each non-one element evenly
            increment = -diff / np.sum(free)
            y[free] += increment
            y = np.clip(y, 0, 1)
    return y

def objective_with_projection(h):
    """Objective function that projects to feasible set"""
    h_projected = project_to_feasible(h)
    return evaluate_c5(h_projected)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use 80 points for better resolution
    n_points = 80
    
    # Initial heuristic construction: tent function
    x = np.linspace(0, 2, n_points)
    h0 = np.maximum(0, 1 - np.abs(x - 1) * 2)
    
    # Normalize to integral = 1
    dx = 2.0 / n_points
    scale = 1.0 / (np.sum(h0) * dx)
    h0 = h0 * scale
    
    # Best solution tracker
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Custom basinhopping with projection
    def my_take_step(x):
        """Take a random step then project back to feasible set"""
        step_size = 0.05
        x_new = x + np.random.normal(0, step_size, len(x))
        return project_to_feasible(x_new)
    
    # Define bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run basinhopping with SLSQP local minimizer
    maxiter = 400
    niter_success = 0
    
    for i in range(maxiter):
        if (time.time() - start_time) > budget_s * 0.9:
            break
            
        # Random restart with projection
        h_start = np.random.uniform(0, 1, n_points)
        h_start = project_to_feasible(h_start)
        
        # Local optimization with SLSQP
        try:
            res = minimize(
                objective_with_projection,
                h_start,
                method='SLSQP',
                bounds=bounds,
                options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-10},
                callback=lambda x: None
            )
            
            if res.success:
                h_candidate = project_to_feasible(res.x)
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
            
        niter_success += 1
    
    # Final comprehensive local optimization using DE
    try:
        # DE with constraint projection
        bounds_de = [(0, 1) for _ in range(n_points)]
        
        def de_objective(h):
            return objective_with_projection(h)
        
        # Run DE with bounds
        result = differential_evolution(
            de_objective,
            bounds_de,
            maxiter=150,
            popsize=20,
            strategy='best1bin',
            seed=seed,
            polish=False
        )
        
        if result.success:
            h_final = project_to_feasible(result.x)
            c5_final = evaluate_c5(h_final)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
                
    except Exception as e:
        pass

    # Final refinement with multiple local searches
    for i in range(50):
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
        # Generate initial random solution with projection
        h_trial = np.random.uniform(0, 1, n_points)
        h_trial = project_to_feasible(h_trial)
        
        # Local refinement with SLSQP
        try:
            res = minimize(
                objective_with_projection,
                h_trial,
                method='SLSQP',
                bounds=bounds,
                options={'maxiter': 150, 'ftol': 1e-11, 'gtol': 1e-11}
            )
            
            if res.success:
                h_candidate = project_to_feasible(res.x)
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
            
    # Ensure final solution meets constraints
    best_h = project_to_feasible(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
