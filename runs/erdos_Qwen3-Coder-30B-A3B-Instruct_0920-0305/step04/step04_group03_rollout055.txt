import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def generate_initial_h():
    """Generate an initial construction for h"""
    # Using the provided initial_h_values as base, but we'll create a more informed version
    # Since we're targeting C5 < 0.3807, we need to be more careful
    
    # Use a smooth construction that's likely to be better
    n_points = 42
    x = np.linspace(0, 2, n_points)
    
    # Start with a more concentrated shape that dips in the middle (as seen in optimized solutions)
    # Trying a construction with less overall overlap
    h = np.zeros(n_points)
    
    # Add some structure to force concentration
    # Try to balance around the center
    midpoint = 1.0
    width = 0.4
    
    # Create a more focused function near the center
    for i in range(n_points):
        x_i = x[i]
        # Gaussian-like shape with peak around center
        h[i] = np.exp(-((x_i - midpoint) / width)**2) * 0.8
        
        # Add some structure to avoid uniform distribution
        if i < n_points // 2:
            h[i] *= 1.0 + 0.2 * np.sin(3 * np.pi * x_i)
        else:
            h[i] *= 1.0 + 0.2 * np.sin(3 * np.pi * (x_i - 1))
    
    # Normalize to area = 1
    dx = 2.0 / n_points
    h = h * (1.0 / (np.sum(h) * dx))
    
    # Bound between 0 and 1
    h = np.clip(h, 0, 1)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with a better initial construction
    h0 = generate_initial_h()
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use a hybrid approach combining DE and local optimization
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    max_iter = 10000
    
    # Counter for adaptive parameters
    de_count = 0
    
    # Adaptive cooling
    while (time.time() - start_time) < budget_s * 0.98:
        # Differential Evolution parameters
        current_iter = de_count % 500
        popsize = max(15, 20 - current_iter // 50)
        mutation_rate = max(0.1, 0.9 - current_iter / 2000)
        recomb_rate = min(0.9, 0.5 + current_iter / 500)
        
        try:
            # Use differential evolution with adaptive parameters
            de_res = differential_evolution(
                objective, 
                bounds, 
                seed=seed + de_count, 
                maxiter=min(50, 10 + current_iter // 10),
                popsize=popsize,
                mutation=(mutation_rate, 1),
                recombination=recomb_rate,
                strategy='best1bin'
            )
            
            if de_res.success:
                h_candidate = de_res.x
                # Normalize to satisfy constraint
                dx = 2.0 / n_points
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
                    
        except Exception as e:
            # Just skip if something goes wrong
            pass
        
        # Local refinement with SLSQP
        ref_h = best_h + np.random.normal(0, 0.005, n_points)
        ref_h = np.clip(ref_h, 0, 1)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(ref_h) * dx)
        ref_h = ref_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                ref_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 50, 'ftol': 1e-13, 'gtol': 1e-13}
            )
            
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
                    
        except Exception as e:
            # Skip if anything fails
            pass
        
        # Add a random perturbation occasionally
        if de_count % 20 == 0:
            pert_h = best_h + np.random.normal(0, 0.002, n_points)
            pert_h = np.clip(pert_h, 0, 1)
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(pert_h) * dx)
            pert_h = pert_h * scale_factor
            
            try:
                res = minimize(
                    objective,
                    pert_h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 20, 'ftol': 1e-14, 'gtol': 1e-14}
                )
                
                if res.success:
                    h_candidate = res.x
                    scale = 1.0 / (np.sum(h_candidate) * dx)
                    h_candidate = h_candidate * scale
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
            except:
                pass
        
        de_count += 1
        
        # Break early if no improvement recently
        if de_count > 1000 and best_c5 < 0.3808:
            break
            
        if (time.time() - start_time) > budget_s * 0.98:
            break

    # Final local optimization
    try:
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-15, 'gtol': 1e-15}
        )
        
        if res.success:
            h_final = res.x
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_final) * dx)
            h_final = h_final * scale
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
