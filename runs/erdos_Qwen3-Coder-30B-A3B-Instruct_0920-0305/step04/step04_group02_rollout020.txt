import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_binary_milp(n):
    """Solve binary MILP to find minimum t such that max overlap <= t"""
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(x) == n // 2]
    
    # Add constraints for each shift d
    for d in range(-n+1, n):
        if d == 0:
            continue
        w_vars = []
        if d > 0:
            rng = range(n - d)
        else:
            rng = range(-d, n)
        for i in rng:
            w = cp.Variable(boolean=True)
            constraints.append(w <= x[i])
            constraints.append(w <= x[i+d])
            constraints.append(w >= x[i] + x[i+d] - 1)
            w_vars.append(w)
        N_d = cp.sum(x) - cp.sum(w_vars)
        constraints.append(N_d <= t)
    constraints.append(t >= 0)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    # Try different solvers
    solver_list = [cp.CBC, cp.GLPK_MI]
    solved = False
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False, max_iters=10000)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                solved = True
                break
        except:
            continue
    
    if not solved:
        raise RuntimeError("Binary MILP solver failed to find solution")
        
    x_vals = np.round(x.value).astype(float)
    t_val = int(round(t.value))
    
    # Ensure the sum constraint is exactly satisfied
    if abs(np.sum(x_vals) - n // 2) > 1e-6:
        # Try to correct by setting some values
        diff = int(n // 2 - np.sum(x_vals))
        if diff > 0:
            # Need to add ones
            zero_indices = np.where(x_vals < 0.5)[0]
            for i in range(min(diff, len(zero_indices))):
                x_vals[zero_indices[i]] = 1.0
        elif diff < 0:
            # Need to subtract ones
            one_indices = np.where(x_vals > 0.5)[0]
            for i in range(min(-diff, len(one_indices))):
                x_vals[one_indices[i]] = 0.0

    return x_vals, t_val
