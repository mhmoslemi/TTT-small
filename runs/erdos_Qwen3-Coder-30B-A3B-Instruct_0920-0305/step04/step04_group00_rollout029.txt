import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
from scipy.spatial.distance import cdist
import time

def project_onto_box_simplex(h, target_sum, bounds=(0, 1)):
    """Projects vector h onto the set {x: bounds[0] <= x <= bounds[1], sum(x) = target_sum}"""
    a, b = bounds
    # Dual water-filling procedure to find critical lambda
    def sum_with_lambda(lam):
        p = np.clip(h - lam, a, b)
        return np.sum(p)
    # Find lambda such that sum_with_lambda(lambda) = target_sum
    # Binary search on lambda
    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        s = sum_with_lambda(mid)
        if s > target_sum:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2.0
    # Return projected vector
    return np.clip(h - lam, a, b)

def evaluate_c5(h):
    """Evaluate the C5 constant using exact max of correlation"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def build_symmetric_piecewise_linear(n, breakpoints, heights):
    """
    Each of the two halves are interpolated piecewise-linearly from (breakpoints, heights)
    Then profiles are reflected and concatenated, maintaining the full length n
    """
    x = np.linspace(0, 1, n//2 + 1)[:-1]  # Length n//2
    # Linear interpolation from breakpoints to heights
    heights_in = np.interp(x, breakpoints, heights)
    heights_out = heights_in[::-1]  # Reverse for [1,2] interval
    h = np.concatenate([heights_in, heights_out])
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    total_time = budget_s

    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # ensures integral constraint
    K = 5  # Number of breakpoints for piecewise-linear construction

    # Prepare stages
    t_de = int(0.70 * total_time)
    t_pgd = int(0.20 * total_time)
    t_final_opt = int(0.10 * total_time)
    t_start = time.time()

    # DE stage on low-dimensional simplex of the form:
    # Parameters are H = [h0,...,hK] where h0 ≥ h1 ≥ ... ≥ hK ≥ 0
    # Build a scaled, symmetric linear piecewise function
    def objective_de(u_vec):
        # Safe and sorted heights
        heights = np.sort(u_vec)[::-1]
        # breakpoints from 0 to 1, linearly spaced with K intervals
        breakpoints = np.linspace(0, 1, K + 1)
        h_raw = build_symmetric_piecewise_linear(n_points, breakpoints, heights)
        # Rescale to sum to target_sum = n/2
        sum_raw = np.sum(h_raw)
        if sum_raw == 0:
            h = np.zeros_like(h_raw)
        else:
            h = h_raw * (target_sum / sum_raw)
        h = np.clip(h, 0, 1)  # avoid numerical stuff
        return evaluate_c5(h)

    bounds = [(0, 1) for _ in range(K)]

    # Run DE
    res_de = differential_evolution(
        objective_de,
        bounds,
        seed=seed,
        maxiter=max(100, int(t_de / 50)),
        popsize=15,
        mutation=0.8,
        recombination=0.7,
        disp=False,
        polish=False,
    )

    best_u = res_de.x
    best_heights = np.sort(best_u)[::-1]
    best_breakpoints = np.linspace(0, 1, K + 1)
    h_final = build_symmetric_piecewise_linear(n_points, best_breakpoints, best_heights)
    h_final = h_final * (target_sum / np.sum(h_final))
    h_final = np.clip(h_final, 0, 1)
    current_c5 = evaluate_c5(h_final)

    # PGD stage
    # Smoothing parameter for log-sum-exp trick at 20 to avoid steep gradients
    beta = 20.0

    def surrogate_objective(h):
        n = h.shape[0]
        dx = 2.0 / n
        corr = correlate(h, 1 - h, mode="full")
        # Use log-sum-exp approximation of max
        f_vals = corr * dx
        f_vals -= np.max(f_vals)  # For numerical stability
        exp_vals = np.exp(beta * f_vals)
        return np.log(np.sum(exp_vals)) / beta

    def surrogate_gradient(h):
        n = h.shape[0]
        dx = 2.0 / n
        corr = correlate(h, 1 - h, mode="full")
        f_vals = corr * dx
        f_vals -= np.max(f_vals)
        exp_vals = np.exp(beta * f_vals)
        w = exp_vals / np.sum(exp_vals)
        # Compute ∂f_k/∂h_j = dx * (1 - h_{j+k} - h_{j-k})
        grad = np.zeros_like(h)
        # For each shift k, compute contribution
        for k in range(len(f_vals)):
            shift_indices = np.arange(len(h))
            left_idx = shift_indices - k
            right_idx = shift_indices + k
            left_val = np.roll(h, k)  # h_{j+k}
            right_val = np.roll(h, -k)  # h_{j-k}
            f_grad = dx * (1 - left_val - right_val)
            grad += w[k] * f_grad
        return grad

    # Start PGD with refined solution from DE
    h_pgd = h_final.copy()
    h_pgd = project_onto_box_simplex(h_pgd, target_sum)

    pgd_time_start = time.time()
    epochs = 0
    while time.time() - pgd_time_start < t_pgd - 1 and epochs < 2000:
        grad = surrogate_gradient(h_pgd)
        h_new = h_pgd - 0.02 * grad
        h_pgd = project_onto_box_simplex(h_new, target_sum)
        epochs += 1

    # Final polishing with SLSQP if time permits
    current_c5 = evaluate_c5(h_pgd)
    if time.time() - t_start < t_start + t_final_opt - 1:
        def obj_slsqp(h):
            return evaluate_c5(h)

        def esc_h(h):
            return np.sum(h) * dx - 1.0

        cons = {'type': 'eq', 'fun': esc_h}

        res = minimize(obj_slsqp, h_pgd, method='SLSQP',
                       bounds=[(0, 1)] * len(h_pgd),
                       constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12})

        if res.success:
            h_opt = res.x
            h_opt = h_opt * (target_sum / np.sum(h_opt))
            c5_opt = evaluate_c5(h_opt)
            if c5_opt < current_c5:
                h_pgd = h_opt
                current_c5 = c5_opt

    return (h_pgd.tolist(), current_c5, n_points)
