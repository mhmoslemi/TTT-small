import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(x):

    target = len(y) / 2

        current_sum = np.sum(y)

            # Need to increase sum, add to values that aren't at 1
            free = y < 1

                break

            free = y > 0
            if not free.any():
                break
            decrement = -diff / np.sum(free)
            y[free] = np.maximum(y[free] - decrement, 0.0)
    return y
