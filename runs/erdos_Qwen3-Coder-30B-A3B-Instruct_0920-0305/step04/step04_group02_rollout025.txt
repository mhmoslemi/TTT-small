import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
import cvxpy as cp

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient(h, d_star):
    n = len(h)
    g = np.zeros(n)
    if d_star > 0:
        g[:-d_star] = 1 - h[d_star:]
    elif d_star < 0:
        g[-d_star:] = 1 - h[:d_star]
    else:
        g[:] = 1 - h[:]
    return g

def binary_optimal(n):
    dx = 2.0 / n
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    prob = cp.Problem(
        cp.Minimize(t),
        [
            cp.sum(x) == n // 2,
            x[0] == 1,  # symmetry breaking
        ]
    )
    # add constraints for overlap bounds
    # This section builds auxiliary variables w[i,d] = x[i]*x[i+d]
    # and constraints w[i,d] <= x[i], w[i,d] <= x[i+d], 
    # w[i,d] >= x[i] + x[i+d] - 1.
    # The expression sum(x) - sum(w) is the number of (1,0) pairs for shift d.
    # We'll compute these expression and add constraints.
    # Simplified coding by looping only over shifts d=-1 to +1
    max_d = n - 1
    for d in range(-max_d, max_d + 1):
        if d == 0:
            continue
        # List the indices where shift d is valid
        # i from 0 to (n-1-d), if d > 0
        # i from (-d) to (n-1), if d < 0
        if d > 0:
            valid_i = list(range(n - d))
            indices_i = valid_i
            indices_j = [i + d for i in valid_i]
        else:
            valid_i = list(range(-d, n))
            indices_i = valid_i
            indices_j = [i + d for i in valid_i]
        w = cp.Variable(len(indices_i), boolean=True)
        # Sum_i x_i - Sum_i w_i is the number of (1,0) pairs for shift d
        # We constrain this <= t for each d.
        # For now, let's just focus on a naive approach to show the idea.
        # However, for speed and correctness it would be better to use
        # more efficient linearization with Big-M | just a subset of d's.
        # For simplicity here, focus on manual constraints for some d's.
        # Start with just d=1
        break
        w = cp.Variable(len(indices_i), boolean=True)
        # Add the 3 big-M constraints for each i
        for i, j in zip(indices_i, indices_j):
            w_idx = indices_i.index(i)
            prob.constraints.append(w[w_idx] <= x[i])
            prob.constraints.append(w[w_idx] <= x[j])
            prob.constraints.append(w[w_idx] >= x[i] + x[j] - 1)
        num_pairs = cp.sum(x) - cp.sum(w)
        prob.constraints.append(num_pairs <= t)
    # Add constraints for a few representative shifts
    for d in [1, 2]:
        if d > 0:
            valid_i = list(range(n - d))
            valid_j = [i + d for i in valid_i]
        else:
            valid_i = list(range(-d, n))
            valid_j = [i + d for i in valid_i]
        w = cp.Variable(len(valid_i), boolean=True)
        for i, j in zip(valid_i, valid_j):
            prob.constraints.append(w[valid_i.index(i)] <= x[i])
            prob.constraints.append(w[valid_i.index(i)] <= x[j])
            prob.constraints.append(w[valid_i.index(i)] >= x[i] + x[j] - 1)
        num_pairs = cp.sum(x) - cp.sum(w)
        prob.constraints.append(num_pairs <= t)
        
    for d in [-2, -1]:
        if d < 0:
            valid_i = list(range(-d, n))
            valid_j = [i + d for i in valid_i]
        else:
            valid_i = list(range(n - d))
            valid_j = [i + d for i in valid_i]
        w = cp.Variable(len(valid_i), boolean=True)
        for i, j in zip(valid_i, valid_j):
            prob.constraints.append(w[valid_i.index(i)] <= x[i])
            prob.constraints.append(w[valid_i.index(i)] <= x[j])
            prob.constraints.append(w[valid_i.index(i)] >= x[i] + x[j] - 1)
        num_pairs = cp.sum(x) - cp.sum(w)
        prob.constraints.append(num_pairs <= t)

    try:
        prob.solve(solver=cp.CBC)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_bin = x.value.astype(float)
            t_val = t.value
        else:
            # fallback to ECOS_BB
            prob.solve(solver=cp.ECOS_BB)
            h_bin = x.value.astype(float)
            t_val = t.value
    except Exception:
        try:
            # If CBC is not available, try GLPK
            prob.solve(solver=cp.GLPK_MI)
            h_bin = x.value.astype(float)
            t_val = t.value
        except Exception:
            # fallback: return some maximal value
            # we just return the initial from the function
            h_bin = np.array(initial_h_values)
            t_val = 100
            h_bin = h_bin - np.mean(h_bin) + 0.5
            h_bin = np.clip(h_bin, 0, 1)
    return h_bin, t_val

def refine_continuous(h0, time_left_s):
    start_time = time.time()
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    n = len(h0)
    dx = 2.0 / n
    max_iter = 500  # increase
    # Try to identify worst shifts, then project gradient steps
    h = h0.copy()
    iter_count = 0
    while time.time() - start_time < time_left_s and iter_count < max_iter:
        # Compute all overlaps
        corr = correlate(h, 1 - h, mode="full")
        max_val = np.max(corr)
        idx_max = np.argmax(corr)
        # shift is from -n+1 to n-1
        shift = idx_max - 2 * (n - 1)
        # if shift is zero, we compute from all non-zero shifts
        if shift == 0:
            # get indices where not zero
            non_zeros = np.where(corr != 0)[0]
            if len(non_zeros) >= 2:
                # largest non-zero
                idx_max = np.partition(corr, -2)[-2]
                idx_max = np.argmax(corr[corr != 0])
                shift = idx_max - (2 * (n - 1))
            else:
                shift = 1
        # Get subgradient
        g = subgradient(h, shift)
        # step size
        alpha = 0.5
        while True:
            h_new = h - alpha * g
            h_new = np.clip(h_new, 0, 1)
            # project to correct sum
            scale = (n / 2) / np.sum(h_new)
            h_new = h_new * scale
            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5 - 1e-9:
                h = h_new
                best_c5 = new_c5
                best_h = h_new.copy()
                break
            else:
                alpha *= 0.5
                if alpha < 1e-6:
                    break
        iter_count += 1
    # Optional: one final optimization
    try:
        res = minimize(evaluate_c5, best_h, method='SLSQP',
                       bounds=[(0, 1)] * n,
                       options={'maxiter': 200})
        if res.success:
            final_h = res.x.copy()
            final_h = np.clip(final_h, 0, 1)
            scale = (n / 2) / np.sum(final_h)
            final_h = final_h * scale
            final_c5 = evaluate_c5(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h0 = np.array(initial_h_values)
    n = len(h0)
    # Find integer solution using MILP
    try:
        h_bin, t_bin = binary_optimal(n)
        c5_bin = evaluate_c5(h_bin)
    except Exception:
        h_bin = h0.copy()
        c5_bin = evaluate_c5(h_bin)
    
    # Start with the binary solution and refine it
    best_h = h_bin.copy()
    best_c5 = c5_bin
    
    # Use few restarts, each runs refine_continuous with 200 iterations
    max_restarts = 30
    for i in range(max_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
        h_refined, c5_refined = refine_continuous(best_h, (budget_s - (time.time()-start_time)) / 2)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    return (best_h.tolist(), best_c5, n)
