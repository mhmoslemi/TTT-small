import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient(h, d):
    """Compute subgradient for shift d"""
    n = len(h)
    g = np.zeros(n)
    if d > 0:
        g[:n-d] = 1 - h[d:]
    elif d < 0:
        g[-d:] = 1 - h[:-d]
    else:
        g[:] = 1  # This case should not happen since d != 0
    return g

def binary_optimal(n):
    """Solve the binary version of the problem via MILP"""
    # Create CVXPY variables
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    dx = 2.0 / n

    # Add mass constraint
    constraints = [cp.sum(x) == n / 2]

    # Add symmetry breaking (fix x[0] = 1 to reduce symmetry)
    constraints.append(x[0] == 1)

    # Add McCormick constraints for each shift
    shift_range = range(-n+1, n)
    shift_range = [d for d in shift_range if d != 0]

    shift_to_w = {}

    for d in shift_range:
        w = cp.Variable(n, boolean=True)
        # Linearization using McCormick envelopes
        for i in range(n):
            if i+d >= 0 and i+d < n:
                # w[i] <= x[i]  
                constraints.append(w[i] <= x[i])
                # w[i] <= x[i+d]
                constraints.append(w[i] <= x[i+d])
                # w[i] >= x[i] + x[i+d] - 1
                constraints.append(w[i] >= x[i] + x[i+d] - 1)
        shift_to_w[d] = w

    # Set the overlap bound for each shift
    for d in shift_range:
        w = shift_to_w[d]
        N_d = cp.sum(x) - cp.sum(w)  # Number of (1,0) pairs
        constraints.append(N_d <= t)

    # Objective function: minimize t
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)

    # Try different solvers
    solvers_to_try = [cp.CBC, cp.GLPK_MI, cp.ECOS_BB]
    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                # Extract integer solution
                x_val = np.round(x.value).astype(np.float64)
                t_val = t.value
                break
        except Exception:
            continue
    else:
        # Fallback to a continuous relaxation and use stub solution
        prob.solve(solver=cp.ECOS_BB, verbose=False)
        x_val = np.round(x.value).astype(np.float64)
        t_val = t.value

    # Correct the exact sum constraint (it might not be exactly n/2 due to rounding)
    scale = (n / 2) / np.sum(x_val)
    x_val *= scale
    return x_val, t_val

def refine_continuous(h0, remaining_time):
    """Refine the continuous solution"""
    dx = 2.0 / len(h0)
    h = h0.copy().astype(np.float64)
    best_h = h.copy()
    best_val = evaluate_c5(h)
    
    start_time = time.time()
    max_iter = 300
    iter_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < remaining_time * 0.9:
        # Evaluate all correlations
        corr_vals = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr_vals)
        # Find indices corresponding to max
        indices = np.where(np.abs(corr_vals - max_corr) < 1e-9)[0]
        # Get the actual shift values
        shift_indices = indices - (len(h) - 1)
        # Pick one of the worst shifts
        d_star = shift_indices[0]
        
        # Compute the subgradient
        g = subgradient(h, d_star)
        
        # Step size (decaying)
        alpha = 0.5 / (1 + iter_count/100)
        
        # Update
        h_new = h - alpha * g
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Scale to satisfy sum constraint
        scale = (len(h) / 2) / np.sum(h_new)
        h_new *= scale
        
        # Evaluate new
        new_val = evaluate_c5(h_new)
        
        # Accept if improvement
        if new_val < best_val - 1e-9:
            h = h_new.copy()
            best_h = h_new.copy()
            best_val = new_val
        
        # Occasionally add noise to escape local minima
        if iter_count % 10 == 0 and np.random.rand() < 0.2:
            noise = np.random.normal(0, 0.005, len(h))
            h_new = h + noise
            h_new = np.clip(h_new, 0, 1)
            scale = (len(h) / 2) / np.sum(h_new)
            h_new *= scale
            new_val = evaluate_c5(h_new)
            if new_val < best_val - 1e-9:
                h = h_new.copy()
                best_h = h_new.copy()
                best_val = new_val
        
        iter_count += 1
    
    return best_h, best_val
