import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def evaluate_c5_exact(h):
    """Evaluate the exact C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def decode_shape(params, n):
    """Decode the parameter vector into a full h vector"""
    # Parameters are the heights at breakpoints
    heights = np.array(params)
    # Sort heights in decreasing order to ensure monotonicity
    heights = np.sort(heights)[::-1]
    
    # Extend 0 at both ends for proper mirroring
    left = np.linspace(0, 1, n//2+1)[:-1]
    right = np.linspace(1, 0, n//2+1)[1:]

    # Linear interpolation between breakpoints
    heights_left = np.interp(left, np.linspace(0, 1, len(heights)), heights)
    heights_right = np.interp(right, np.linspace(0, 1, len(heights)), heights[::-1])
    
    # Combine into full array
    h = np.concatenate([np.flipud(heights_left[:-1]), heights_right])
    
    # Normalize to area 1
    area = np.sum(h) * 2.0 / n
    h = h / area * 1.0
    
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    
    return h

def objective(params, n):
    h = decode_shape(params, n)
    return evaluate_c5_exact(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 200
    k = 5  # Number of breakpoints
    
    # Adjacent interval array (0,1) parameterization
    bounds = [(0, 1) for _ in range(k)]
    
    # Use DE to explore global shape parameter space with reduced dimensionality
    t_de = 0.7 * budget_s
    result = differential_evolution(
        objective,
        bounds,
        args=(n_points,),
        seed=seed,
        maxiter=int(1000*t_de/budget_s),
        popsize=15,
        disp=False,
        atol=1e-8
    )
    
    best_h = decode_shape(result.x, n_points)
    
    # Smooth approximation objective and refine with projected gradient descent
    def smooth_objective(h, beta=20):
        """Smooth approximation to max using log-sum-exp"""
        dx = 2.0 / n_points
        corr = correlate(h, 1 - h, mode="full")
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * corr * dx)))
    
    def smooth_gradient(h, beta=20):
        """Compute gradient of smooth objective using correlation"""
        dx = 2.0 / n_points
        corr = correlate(h, 1 - h, mode="full")
        exp_corr = np.exp(beta * corr * dx)
        w = exp_corr / np.sum(exp_corr)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            for m in range(len(corr)):
                if 0 <= i-m < len(h):
                    grad[i] += w[m] * dx * (-h[i-m] - h[i+m] if 0 <= i+m < len(h) else -h[i-m])
                if 0 <= i+m < len(h):
                    grad[i] += w[m] * dx * (-h[i+m] - h[i-m] if 0 <= i-m < len(h) else -h[i+m])
        # Grad includes contribution from 1 terms, so should be subtracting
        return grad
    
    def project_to_constraint_space(h, dx):
        """Project onto feasible set (bounded [0,1] and normalized)"""
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Normalize
        area = np.sum(h) * dx
        h = h / area * 1.0
        return h
    
    # Adaptive step-size for gradient descent
    eta = 0.01
    max_grad_iter = 200
    last_improvement = np.inf
    
    for i in range(max_grad_iter):
        if time.time() - start_time > 0.95 * budget_s:
            break
        g = smooth_gradient(best_h)
        best_h = best_h - eta * g
        best_h = project_to_constraint_space(best_h, 2.0/n_points)
        last_improvement = evaluate_c5_exact(best_h)
        eta *= 0.98  # Momentum-like decay
    
    # Final exact evaluation
    final_c5 = evaluate_c5_exact(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
