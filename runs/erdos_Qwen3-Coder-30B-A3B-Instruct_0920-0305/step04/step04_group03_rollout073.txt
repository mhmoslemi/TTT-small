import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_boxed_simplex(v, target_sum):
    """Project vector v onto the simplex with box constraints [0,1] and sum = target_sum"""
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-10:
        return v
    
    # Iteratively adjust to meet sum constraint
    tol = 1e-10
    max_iter = 1000
    for _ in range(max_iter):
        diff = target_sum - current_sum
        if abs(diff) < tol:
            break
        free = (v > 0) & (v < 1)
        if not np.any(free):
            break
        nfree = np.sum(free)
        delta = diff / nfree
        v[free] += delta
        v = np.clip(v, 0, 1)
        current_sum = np.sum(v)
    
    return v
