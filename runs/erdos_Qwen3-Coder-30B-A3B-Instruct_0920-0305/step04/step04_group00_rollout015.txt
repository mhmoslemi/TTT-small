import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def h_from_params(params, n_points=200):
    """
    Generate a step function h(x) from 4 parameters:
    params = [center_height, slope_in, slope_out, plateau_width]
    """
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)

    # Parameters: central height, slopes, plateau width
    h_center, slope_in, slope_out, width = params

    # Ensure parameters are valid
    h_center = np.clip(h_center, 0, 1)
    width = np.clip(width, 0, 1)
    slope_in = np.clip(slope_in, 0, 1)
    slope_out = np.clip(slope_out, 0, 1)

    # Compute center and bounds for the plateau
    center = 1.0
    left_edge = center - width / 2
    right_edge = center + width / 2

    # Build the piecewise-linear function with slopes
    for i in range(n_points):
        xi = x[i]
        if xi <= left_edge:
            h[i] = h_center - slope_in * (left_edge - xi)
        elif xi >= right_edge:
            h[i] = h_center - slope_out * (xi - right_edge)
        else:
            h[i] = h_center
        # Clip at [0,1]
        h[i] = np.clip(h[i], 0, 1)

    # Symmetrize h about the center
    for i in range(n_points):
        h[i] = 0.5 * (h[i] + h[n_points - i - 1])

    # Normalize integral to 1
    integral = np.sum(h) * dx
    if integral > 0:
        h *= (1.0 / integral)

    return h

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def objective(params):
    # Generate h from parameters
    h = h_from_params(params)
    # Evaluate
    return evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 200  # Higher resolution
    bounds = [
        (0.1, 0.9),     # h_center
        (0.0, 1.0),     # slope_in
        (0.0, 1.0),     # slope_out
        (0.0, 1.0)      # plateau_width
    ]

    # Objective with bounds
    def obj(p):
        return objective(p)

    # Global search using differential evolution
    res_global = differential_evolution(
        obj,
        bounds,
        seed=seed,
        maxiter=int(budget_s * 0.9 / 5),  # Rough estimate
        popsize=16,
        mutation=(0.5, 1.0),
        recombination=0.7,
        tol=1e-5,
        callback=None
    )
    best_params = res_global.x
    h_globals = h_from_params(best_params)
    c5_global = evaluate_c5(h_globals)

    # Refinement: SLSQP from best global solution
    res_local = minimize(
        obj,
        best_params,
        method='L-BFGS-B',
        bounds=bounds,
        options={'gtol': 1e-7, 'ftol': 1e-9}
    )

    if res_local.success:
        h_final = h_from_params(res_local.x)
        c5_final = evaluate_c5(h_final)
        if c5_final < c5_global:
            h_globals = h_final
            c5_global = c5_final

    # Return final result
    return (h_globals.tolist(), c5_global, n_points)
