import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_binary_milp(n):
    """Solve the binary version of the problem using MILP to get integer optimal."""
    # Create binary variables
    x = cp.Variable(n, boolean=True)
    
    # Add constraint that sum equals n/2
    sum_constraint = cp.sum(x) == n // 2
    
    # Create auxiliary variables y[i,d] for the product x[i]*x[i+d]
    y = {}
    for d in range(-n+1, n):
        if d == 0:
            continue
        y[d] = cp.Variable((n - abs(d)), boolean=True)
    
    # Add linearization constraints
    constraints = [sum_constraint]
    for d in range(-n+1, n):
        if d == 0:
            continue
        for i in range(n - abs(d)):
            idx1 = i
            idx2 = i + d
            if idx2 < 0 or idx2 >= n:
                continue
            constraints.extend([
                y[d][i] <= x[idx1],
                y[d][i] <= x[idx2],
                y[d][i] >= x[idx1] + x[idx2] - 1
            ])

    # Add bounded overlap variable
    t = cp.Variable(integer=True)
    for d in range(-n+1, n):
        if d == 0:
            continue
        # We want N_d = (n/2) - sum(y[d]) <= t
        constraints.append((n//2) - cp.sum(y[d]) <= t)
    
    constraints.append(t >= 0)
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CBC, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            x_val = np.array(x.value).astype(int)
            t_val = int(t.value)
            return x_val, t_val, True
    except:
        return None, None, False
    return None, None, False

def log_sum_exp_smoothed_objective(h, shifts, tau=0.1):  
    """
    Compute surrogate of max overlap using log-sum-exp smoothing
    """
    n = len(h)
    n_shifts = len(shifts)
    c_values = np.zeros(n_shifts)
    for idx, d in enumerate(shifts):
        c = 0.0
        if d < 0:
            # slide h forward by |d|, then compute dot product with (1 - h[-d:])
            c1 = h[-d:]  # overlapping part on right
            c2 = (1.0 - h[:d]) if d != -n else np.zeros(0)  # overlapping part on left
            c = np.sum(c1) - np.sum(c1 * c2)
        elif d > 0:
            # slide h backward by d
            c = np.sum(h[:-d]) - np.sum((1.0 - h[d:]) * h[:-d])
        else:
            continue
        c_values[idx] = c
    
    # compute soft-max over all shift contributions
    exp_vals = np.exp(c_values / tau)
    sum_exp = np.sum(exp_vals)
    smooth_obj = tau * np.log(sum_exp)
    return smooth_obj

def gradient_of_smoothed_max(h, shifts, tau=0.1):
    """
    Compute the gradient of the log-sum-exp surrogate
    """
    n = len(h) 
    scores = np.zeros_like(h)
    n_shifts = len(shifts)
    
    c_values = np.zeros(n_shifts)
    for idx, d in enumerate(shifts):
        c = 0.0
        if d < 0:
            # slide h forward by |d|, then compute dot product with (1 - h[-d:])
            c1 = h[-d:]  # overlapping part on right
            c2 = (1.0 - h[:d]) if d != -n else np.zeros(0)  # overlapping part on left
            c = np.sum(c1) - np.sum(c1 * c2)
        elif d > 0:
            # slide h backward by d
            c = np.sum(h[:-d]) - np.sum((1.0 - h[d:]) * h[:-d])
        else:
            continue
        c_values[idx] = c
        
    exp_vals = np.exp(c_values / tau)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp  # normalizing the weights

    # Compute gradient vector of Φ_τ(h) = τ * log (Σ exp(C_d(h)/τ))
    grad = np.zeros_like(h)
    for idx, d in enumerate(shifts):
        if d == 0:
            continue
        if d < 0:
            # gradient of C_d = sum h - sum h[i] * h[i+d]
            for i in range(len(h[:d])):  # only i such that i+d < len(h)
                grad[i] += weights[idx]
            for i in range(-d, len(h)):
                grad[i] -= weights[idx] * h[i-d]
        elif d > 0:
            for i in range(0, len(h[d:])):  # only i such that i+d < len(h)
                grad[i] += weights[idx]
            for i in range(d, len(h)):
                grad[i] -= weights[idx] * h[i-d]
                
    return grad

def project_to_feasible_simplex(h, target_sum=1.0):
    """Project h onto simplex with sum = target_sum and bounds [0,1]"""
    h = np.clip(h, 0, 1)
    cur_sum = np.sum(h)
    if cur_sum == 0.0:
        return np.zeros_like(h)  # avoid division by zero
    h = h * target_sum / cur_sum  # scale to exactly target_sum
    h = np.clip(h, 0, 1)  # ensure bounds still hold
    return h

def smooth_max_descent(h_start, shifts, max_restarts=10, max_iter_per_restart=300, tau_schedule=[0.5, 0.25, 0.125, 0.0625, 0.03125]):
    n = len(h_start)
    best_h = np.array(h_start)
    best_c5 = evaluate_c5(best_h)
    
    # RBFGS optimizer with backtracking step size selection
    alphas = [1.0]
    for restart_count in range(max_restarts):
        h_current = np.array(best_h)
        for t_idx, tau in enumerate(tau_schedule):
            for step in range(max_iter_per_restart):  # Fixed number of steps per tau level
                grad = gradient_of_smoothed_max(h_current, shifts, tau)
                new_h = h_current - (alphas[-1] * grad)
                new_h = project_to_feasible_simplex(new_h, target_sum=1.0)

                # Adaptive step size via backtracking
                candidate_c5 = evaluate_c5(new_h)
                if candidate_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = candidate_c5
                # Check if we improved
                if candidate_c5 < evaluate_c5(h_current):
                    h_current = new_h.copy()
                    
                # Test safety boundaries and early exit if no improvements
                if step > 10 and evaluate_c5(h_current) >= best_c5:
                    break  # No further improvement: break inner loop (continue to next tau)

                # Scale back if needed to be safe
                if np.abs(candidate_c5 - evaluate_c5(h_current)) < 1e-8:
                    break

            # Restart with slightly perturbed result (good if fell into flat region)
            if len(tau_schedule) - t_idx > 1:
                h_current += np.random.normal(0, 0.01, len(h_current))  
                h_current = project_to_feasible_simplex(h_current, target_sum=1.0)

        if best_c5 <= 0.3807:
            break

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    init_h = np.array(initial_h_values)
    n_points = len(init_h)
    
    best_h = init_h.copy()
    best_c5 = evaluate_c5(best_h)

    # STEP 1: Solve Exact MILP (Binary Optimization)
    try:
        binary_solution, t_opt, found = solve_binary_milp(n_points)
        if found and binary_solution is not None:
            # Reconstruct it to float rather than int
            h_bin = binary_solution.astype(float)
            h_bin = project_to_feasible_simplex(h_bin, target_sum=1.0)
            c5_bin = evaluate_c5(h_bin)
            if c5_bin < best_c5:
                best_h = h_bin.copy()
                best_c5 = c5_bin
    except:
        pass

    # Define all shifts
    shifts_list = list(range(-n_points+1, n_points))  
    shifts_list.remove(0)  # remove zero shift (that's not a shift)

    # If already solved to acceptable level, stop early
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)

    # STEP 2: Smooth Max Descent Refinement with Restart Scheme
    time_budget_remaining = budget_s * 0.95 - (time.time() - start_time)

    if time_budget_remaining > 50:
        refined_h, refined_c5 = smooth_max_descent(
            best_h, 
            shifts_list, 
            max_restarts=20, 
            max_iter_per_restart=200
        )
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5

    # STEP 3: Final polishing with standard NLP solver (SLSQP)
    def objective_slqp(h):
        return evaluate_c5(h)

    def constraint_sq(h):
        return np.sum(h) - 1.0  # maintain integral

    bounds = [(0, 1) for _ in range(n_points)]
    
    cons = {'type': 'eq', 'fun': constraint_sq}

    try:
        res = minimize(
            objective_slqp,
            best_h.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_final = res.x
            c5_final = evaluate_c5(h_final)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
