import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient(h, d_star):
    """Compute subgradient for shift d_star"""
    n = len(h)
    g = np.zeros_like(h)
    if d_star > 0:
        # h_i - h_{i+d_star}
        g[:-d_star] = 1 - h[d_star:]
    elif d_star < 0:
        # h_i - h_{i+d_star} (d_star is negative)
        g[-d_star:] = 1 - h[:-d_star]
    else:
        # d_star == 0, this shouldn't happen here
        pass
    return g

def binary_optimal(n):
    """Solve the binary integer program to find minimal t"""
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)

    # Mass constraint
    mass = cp.sum(x) == n // 2
    # Symmetry breaking
    symmetry = x[0] == 1

    # Add constraints for each shift d
    constraints = [mass, symmetry, t >= 0]
    
    # Assume 0 <= d <= n-2, ignore d=0
    for d in range(1, n):
        w = cp.Variable(n - d, boolean=True)
        # Product constraints (McCormick)
        for i in range(n - d):
            constraints.append(w[i] >= x[i] + x[i + d] - 1)
            constraints.append(w[i] <= x[i])
            constraints.append(w[i] <= x[i + d])
        # Overlap count = sum(x) - sum(w), we want this <= t
        N_d = cp.sum(x) - cp.sum(w)
        constraints.append(N_d <= t)
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Try CBC first, fallback
    try:
        prob.solve(solver=cp.CBC)
    except:
        try:
            prob.solve(solver=cp.GLPK_MI)
        except:
            try:
                prob.solve(solver=cp.ECOS_BB)
            except:
                # If all fail, return initial h
                return None, None

    if prob.status in ["optimal", "optimal_inaccurate"]:
        h_bin = np.round(x.value).astype(float)
        return h_bin, int(t.value)
    else:
        return None, None

def refine_continuous(h0, time_left):
    """Refine a continuous solution using projected sub-gradient descent"""
    best = h0.copy()
    best_val = evaluate_c5(best)
    n = len(h0)
    dx = 2.0 / n
    start_time = time.time()
    
    iterations = 0
    max_iter = 200
    
    while iterations < max_iter and (time.time() - start_time) < time_left * 0.95:
        # compute correlations and get maximal shift(s)
        corr = correlate(best, 1 - best, mode='full')
        max_val = np.max(corr)
        
        # Identify which shifts achieve the maximum
        indices = np.where(np.abs(corr - max_val) < 1e-9)[0]
        
        # Since correlation is symmetric, just consider shifts that are positive or zero
        # and then deal with the negative case by symmetry
        d_max_indices = indices[(indices >= n-1) & (indices < 2*n-2)]
        d_max_indices = [i - n + 1 for i in d_max_indices]  # Normalize to shift indices
        d_max_indices = [d for d in d_max_indices if d > 0]  # Only positive shifts
        
        if not d_max_indices:
            break
            
        # Average gradient over all worst shifts
        g = np.zeros_like(best)
        for d in d_max_indices:
            g += subgradient(best, d)
        
        # step size: try adaptive approach
        alpha = 0.5
        while True:
            h_trial = best - alpha * g
            h_trial = np.clip(h_trial, 0, 1)
            # scale to enforce sum constraint
            scale = (n / 2) / np.sum(h_trial)
            h_trial = h_trial * scale
            trial_val = evaluate_c5(h_trial)
            if trial_val < best_val - 1e-9:
                best = h_trial
                best_val = trial_val
                break
            alpha *= 0.5
            if alpha < 1e-9:
                break
        
        # Occasionally add random noise
        if np.random.rand() < 0.2:
            noise = np.random.normal(0, 0.01, n)
            h_trial = best + noise
            h_trial = np.clip(h_trial, 0, 1)
            # scale to enforce sum constraint
            scale = (n / 2) / np.sum(h_trial)
            h_trial = h_trial * scale
            trial_val = evaluate_c5(h_trial)
            if trial_val < best_val - 1e-9:
                best = h_trial
                best_val = trial_val
        iterations += 1
    
    # Final polishing
    try:
        res = minimize(lambda h: evaluate_c5(h), best, method='SLSQP',
                       bounds=[(0, 1) for _ in range(n)],
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0},
                       options={'maxiter': 50, 'ftol': 1e-12, 'gtol': 1e-12})
        if res.success:
            scaled_res = res.x * (n/2) / np.sum(res.x)
            final_val = evaluate_c5(scaled_res)
            if final_val < best_val:
                best = scaled_res
                best_val = final_val
    except:
        pass
        
    return best, best_val
