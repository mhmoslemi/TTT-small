import numpy as np
from scipy.optimize import differential_evolution, minimize, basinhopping
from scipy.signal import correlate
import time

def project_to_feasible(h, target_sum):
    """Project h onto the feasible set: [0,1]^n and sum = target_sum"""
    y = np.clip(h, 0, 1)
    current_sum = np.sum(y)
    if abs(current_sum - target_sum) < 1e-9:
        return y
    
    # Iterative adjustment to meet sum constraint
    for _ in range(100):
        diff = current_sum - target_sum
        if abs(diff) < 1e-9:
            break
        if diff > 0:  # Sum too high
            free = y > 0
            if not free.any():
                break
            decrement = diff / np.sum(free)
            y[free] = np.maximum(0, y[free] - decrement)
        else:  # Sum too low
            free = y < 1
            if not free.any():
                break
            increment = -diff / np.sum(free)
            y[free] = np.minimum(1, y[free] + increment)
        current_sum = np.sum(y)
    
    return y

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx
