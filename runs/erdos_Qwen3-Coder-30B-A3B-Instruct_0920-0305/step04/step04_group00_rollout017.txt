import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Discretization parameters
    n_points = 200  # Smooth resolution
    dx = 2.0 / n_points
    target_area = 1.0

    # Parametric representation: monotonic heights
    # Use 5 breakpoints (4 interior heights) on [0,1] half-interval
    k = 4   
    # Parametric space: [h_0, h_1, ..., h_k] where h_i in [0,1] and h_0 >= ... >= h_k
    # h_0 must be 1 due to normalization
    # So we optimize only k values: h_1, ..., h_k
    
    def decode_h(params):
        # params: [h_1, ..., h_k] 
        h = np.ones(n_points)
        h[0] = 1.0
        h[1:n_points//2] = np.interp(np.linspace(0, 1, n_points//2), 
                                    np.linspace(0, 1, k+1), 
                                    [1.0] + list(params))
        h[n_points//2:] = h[n_points//2-1::-1]  # symmetric
        # If not correctly normalized, re-scale
        integral = np.sum(h)*dx
        if integral > 0:
            h *= (target_area / integral)
        h = np.clip(h, 0, 1)
        return h
    
    def smooth_max_corr(h, beta=20):
        # t := h[0]*dx = 1/n_points, simulate t in the full vector
        # But just compute the correlation manually via partial sums,
        # or use scipy correlate, but apply an energy-based version
    
        # Full correlation matrix approach for large vector: possible slow
        # But the data is finite we can precompute shifted indices
        corr_vals = []
        for shift in range(-n_points+1, n_points):
            corr_sum = 0.0
            for i in range(n_points):
                j = i + shift
                if j >= 0 and j < n_points:
                    corr_sum += h[i]*(1-h[j])
            corr_vals.append(corr_sum*dx)
            
        if len(corr_vals) <= 0:
            return 0.0
        arr_vals = np.array(corr_vals)
        # Smooth max approximation with beta scaling to avoid numerical issues
        exp_vals = np.exp(beta * arr_vals)
        if np.allclose(np.log(exp_vals), -np.inf):
            # In case of numerical underflow avoid log of 0
            return np.max(arr_vals)  # fallback
        smooth_max = (1./beta)*np.log(np.sum(exp_vals))
        return smooth_max

    def objective(params):
        h = decode_h(params)
        c5 = np.max(np.correlate(h, 1-h, mode='full'))*dx
        return c5
        
    def smooth_objective(params):
        # This is smoothed version for gradient-based refinement
        h = decode_h(params)
        return smooth_max_corr(h, beta=20)

    # Optimization: Use DE to explore low-dimensional space
    bounds = [(0, 1) for _ in range(k)]
    
    de_time = 0.7 * budget_s
    de_start = time.time()
    
    # Important: Wrap objective so it's compatible with expected signature for scipy minimize
    def obj_wrapper(params):
        return objective(params)

    de_res = differential_evolution(
        obj_wrapper,
        bounds,
        seed=seed,
        maxiter=100,
        popsize=15,
        strategy='best1bin',
        atol=1e-7,
        rtol=1e-7
    )
    
    if not de_res.success:
        print("DE failed or not converging properly.")

    best_params = de_res.x.copy()
    best_h = decode_h(best_params)
    best_c5 = objective(best_params)

    # Gradient refinement using smooth surrogate
    t_grad = 0.2 * budget_s
    t_refine_start = time.time()

    # Initialize from DE result
    h_cur = best_h.copy()

    # Projected gradient descent on full space (shadowed later with more precise smooth approx)
    def project_h_on_simplex_and_box(h, eps=1e-10):
        # Clip to [0,1]
        h_clipped = np.clip(h, eps, 1.0-eps)
        # Scale to required area
        integral = np.sum(h_clipped)*dx
        if integral > 0:
            h_clipped = h_clipped * (target_area / integral)
        # Ensure it's within [0,1] still after scaling
        h_clipped = np.clip(h_clipped, 0, 1)
        return h_clipped

    def compute_f_gradients(h, k_idx):
        # f_k = sum_{i} h_i * (1 - h_{i+k})
        # grad h_i = sum_k w_k * (1 - h_{i+k} - h_{i-k})
        # But we need to be more careful with boundary indices here.
        # Simple approximation via correlation with shifted versions
        return np.zeros_like(h)

    # Define minimum step size and max iterations
    num_iter = 200
    lr = 0.03
    decay_rate = 100.0
    
    for i in range(num_iter):
        if time.time() - t_refine_start > t_grad:
            break

        # Gradient on the smooth surrogate via memory shortcuts (approximate)
        # Use derivative from finite difference of corr value around this h
        eps = 1e-4
        cost = smooth_max_corr(h_cur, beta=20)

        # Approximate gradient with small finite differences (in curve shape)
        grad = np.zeros_like(h_cur)
        for idx in range(n_points):
            # Find all shift values k where both i, i+k are valid
            h_plus = h_cur.copy()
            h_plus[idx] += eps
            h_plus = project_h_on_simplex_and_box(h_plus)
            cost_plus = smooth_max_corr(h_plus, beta=20)
            grad[idx] = (cost_plus - cost) / eps

        # Update: gradient step
        h_new = h_cur - lr * grad
        
        # Project onto constraints
        h_new = project_h_on_simplex_and_box(h_new)
        
        # Update current iterate
        h_cur = h_new
        
        # Update learning rate decay
        lr = lr / (1.0 + (i+1)/decay_rate)

    # Final C5 determination using exact max  
    final_corr = np.correlate(h_cur, 1-h_cur, mode='full')*dx
    final_c5 = np.max(final_corr)

    # Final refinement with local logic:
    lsqp_start = time.time()
    t_lsqp = 0.1 * budget_s
    if time.time() - lsqp_start < t_lsqp:
        # Prepare a smaller "<= 200" control grid
        h_local_trial = h_cur.copy()
        try:
            from scipy.optimize import minimize
            def objective_local(h):
                corr = np.correlate(h, 1-h, mode='full')*dx
                return np.max(corr) 
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x)*dx - 1.0}
            bounds = [(0, 1) for _ in range(n_points)]
            # Avoid improve only if improvement is very small
            res = minimize(objective_local, h_local_trial, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter':100})
            if res.success:
                h_final = res.x.copy()
                h_final = project_h_on_simplex_and_box(h_final)
                c5_final = np.max(np.correlate(h_final, 1-h_final, mode='full')*dx)
                if c5_final < final_c5:
                    final_c5 = c5_final
                    h_cur = h_final.copy()
        except Exception as e:
            pass

    return (h_cur.tolist(), final_c5, n_points)
