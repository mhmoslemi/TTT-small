import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Phase 1: Integer Optimal via MILP (if available)
    h_binary = None
    try:
        # Define variables (binary)
        x = Variable(n_points, boolean=True)
        
        # Constraint: Sum of x equals n/2
        constraints = [sum(x) == n_points / 2]
        
        # Precompute valid shifts and create auxiliary variables
        y_vars = {}
        for d in range(1, n_points):
            for i in range(n_points):
                if i + d < n_points:
                    y_vars[(i, d)] = Variable(boolean=True)
                    constraints.extend([
                        y_vars[(i, d)] <= x[i],
                        y_vars[(i, d)] <= x[i+d],
                        y_vars[(i, d)] >= x[i] + x[i+d] - 1
                    ])
        
        # Constraint: For each shift d, the number of mismatches should not exceed t
        t = Variable(integer=True)
        for d in range(1, n_points):
            sum_mismatch = 0
            for i in range(n_points):
                if i + d < n_points:
                    sum_mismatch += y_vars[(i, d)]
            constraints.append(sum(x) - sum_mismatch <= t)
        
        # Objective function: minimize t
        objective = Minimize(t)
        prob = Problem(objective, constraints)
        
        # Try various solvers
        solvers = ['CBC', 'GLPK_MI', 'ECOS_BB']
        for solver in solvers:
            try:
                prob.solve(solver=solver, warm_start=True, max_iters=10000)
                if prob.status == 'optimal':
                    x_vals = [1 if v.value >= 0.5 else 0 for v in x]
                    h_binary = np.array(x_vals)
                    break
            except:
                continue
    
        if h_binary is None:
            # Fallback: use initial values
            # The goal is to use integer solution as a base
            h_binary = h0.copy()
            h_binary = (h_binary > 0.5).astype(int)
            
    except Exception as e:
        print(f"MILP failed with error: {e}")
        h_binary = h0.copy()
        h_binary = (h_binary > 0.5).astype(int)
    
    # Correct h_binary to ensure it sums to n/2
    actual_sum = np.sum(h_binary)
    if actual_sum != n_points / 2:
        diff = int(n_points/2 - actual_sum)
        if diff > 0:  # Need to flip more 0s to 1s
            indices = np.where(h_binary == 0)[0]
            for i in range(min(diff, len(indices))):
                h_binary[indices[i]] = 1
        elif diff < 0:  # Need to flip more 1s to 0s
            indices = np.where(h_binary == 1)[0]
            for i in range(min(-diff, len(indices))):
                h_binary[indices[i]] = 0

    # Evaluate the binary result
    c5_binary = evaluate_c5(h_binary)
    best_h = h_binary.copy()
    best_c5 = c5_binary

    # Phase 2: Smoothed gradient descent with a continuation schedule
    def smooth_max(h, shifts, tau):
        """Compute the smooth max of overlaps with a temperature tau"""
        overlaps = []
        for d in shifts:
            if d < 0:
                corr = np.sum(h[:d] * (1 - h[-d:]))
            elif d > 0:
                corr = np.sum(h[d:] * (1 - h[:-d]))
            else:
                corr = np.sum(h * (1 - h))  # Should not happen for d=0
            overlaps.append(corr)
        overlaps = np.array(overlaps)
        return tau * np.log(np.sum(np.exp(overlaps / tau)))

    def grad_smooth_max(h, shifts, tau):
        """Compute gradient of smooth max"""
        overlaps = []
        for d in shifts:
            if d < 0:
                corr = np.sum(h[:d] * (1 - h[-d:]))
            elif d > 0:
                corr = np.sum(h[d:] * (1 - h[:-d]))
            else:
                corr = np.sum(h * (1 - h))  # Should not happen for d=0
            overlaps.append(corr)
        overlaps = np.array(overlaps)
        exps = np.exp(overlaps / tau)
        weights = exps / np.sum(exps)
        
        grad = np.zeros_like(h)
        for k, d in enumerate(shifts):
            if d < 0:
                term = (1 - h[-d:]) - h[:d]
            elif d > 0:
                term = (1 - h[:-d]) - h[d:]
            else:
                term = (1 - h) - h
            grad += weights[k] * term
            
        return grad
    
    # Set up shifts (exclude d=0)
    shifts = list(range(1, n_points)) + list(range(-n_points+1, 0))
    dx = 2.0 / n_points
    
    # Continuation schedule for temperature
    tau_schedule = [0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125, 0.00390625, 0.001953125, 0.0009765625, 0.00048828125, 0.000244140625]
    
    # Start with a refined version of the binary solution
    h_refined = h_binary.astype(float)
    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
    
    # Project to simplex
    def project_simplex(v, target_sum):
        v = np.clip(v, 0, 1)
        current_sum = np.sum(v)
        if current_sum > 0:
            v = v * (target_sum / current_sum)
        return np.clip(v, 0, 1)
    
    max_restarts = 10
    restart_count = 0
    last_best = best_c5
    
    # Loop over continuation schedule
    for idx, tau in enumerate(tau_schedule):
        if (time.time() - start_time) > 0.95 * budget_s:
            break
        num_inner_iters = 100
        for _ in range(num_inner_iters):
            if (time.time() - start_time) > 0.95 * budget_s:
                break
            grad = grad_smooth_max(h_refined, shifts, tau)
            h_new = h_refined - 0.1 * grad  # Step size 0.1
            h_new = project_simplex(h_new, n_points / 2)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_refined = h_new.copy()

        # Early stopping if improvement is too slow
        if abs(best_c5 - last_best) < 1e-5:
            break
        last_best = best_c5

    # Final polish with non-smooth optimization
    try:
        def objective_non_smooth(h):
            return evaluate_c5(h)
            
        def constraint_eq(h):
            return np.sum(h) * dx - 1.0
        
        cons = {'type': 'eq', 'fun': constraint_eq}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(
            objective_non_smooth,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_final = res.x
            normalizer = np.sum(h_final) * dx
            h_final = h_final * (1.0 / normalizer)
            c5_final = evaluate_c5(h_final)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
    except Exception as e:
        pass  # Ignore polish failure

    return (best_h.tolist(), best_c5, n_points)
