import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Minimize, Problem

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

        h_vars = cp.Variable(n_points, boolean=True)
        p_vars = cp.Variable((n_points, n_points), boolean=True)
        M = cp.Variable(integer=True, name="M")
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == n_points // 2,
            M >= 0,
        ]
        
        # Add logical AND constraints
        for l in range(1, n_points):
            for i in range(n_points - l):
                # p[i,l] <= h[i] and p[i,l] <= h[i+l]
                constraints.append(p_vars[i, l] <= h_vars[i])
                constraints.append(p_vars[i, l] <= h_vars[i + l])
                # p[i,l] >= h[i] + h[i+l] - 1
                constraints.append(p_vars[i, l] >= h_vars[i] + h_vars[i + l] - 1)
        
        # Add overlap constraints
        for l in range(1, n_points):
            # Left side
            A_plus = cp.sum(h_vars[:n_points - l])
            # Right side 
            A_minus = cp.sum(h_vars[l:])
            S = cp.sum(p_vars[:n_points - l, l])
            
            constraints.append(A_plus - S <= M)
            constraints.append(A_minus - S <= M)
            
        # Set symmetry-breaking constraint
        constraints.append(h_vars[0] == 1)
        
        # Objective
        objective = Minimize(M)
        prob = Problem(objective, constraints)
        
        # Solve with specified time limit
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=remaining_time)

        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(h_trial) * dx)

                    best_c5 = c5_candidate
        except:
            continue

    return (best_h.tolist(), best_c5, n_points)
