```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def build_symmetric_piecewise_linear(half_breaks, half_heights, n):
    """Build full vector from piecewise linear symmetric construction"""
    # Sort heights in descending order for monotonicity
    sorted_heights = np.sort(half_heights)[::-1]
    
    # Create the left half including middle point
    left_points = np.linspace(0, 1, len(half_breaks), endpoint=True)
    heights_left = np.interp(left_points, half_breaks, sorted_heights)
    
    # Mirror to get full function
    full_h = np.concatenate([heights_left, heights_left[::-1][1:]])
    
    # Scale to satisfy integral constraint
    scale = (n / 2) / np.sum(full_h)
    full_h = full_h * scale
    # Ensure boundary constraints
    full_h = np.clip(full_h, 0, 1)
    # Final re-normalization if needed
    scale = (n / 2) / np.sum(full_h)
    full_h = full_h * scale
    # Ensure it sums to exactly n/2
    full_h = full_h * (n / 2) / np.sum(full_h)
    
    return full_h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 200  # Number of samples
    dx = 2.0 / n
    
    # Proposed parameterization: 5 points (K=4) for the half interval
    K = 4
    def objective(params):
        half_breaks = np.linspace(0, 1, K+1)
        half_heights = params[:K+1]
        try:
            h = build_symmetric_piecewise_linear(half_breaks, half_heights, n)
            return evaluate_c5(h)
        except:
            return float('inf')
    
    bounds = [(0, 1) for _ in range(K+1)]
    
    # DE stage - allocate 70% of time
    t_end_de = start_time + 0.7 * budget_s
    de_result = differential_evolution(
        objective, bounds,
        maxiter=100,
        popsize=15,
        mutation=0.8,
        recombination=0.7,
        seed=seed,
        callback=lambda x, convergence: time.time() > t_end_de,
        workers=1,
        updating='deferred'
    )
    
    # Decode best DE result to full vector
    if de_result.success:
        half_breaks = np.linspace(0, 1, K+1)
        best_params = de_result.x
        try:
            h_final = build_symmetric_piecewise_linear(half_breaks, best_params, n)
        except:
            # Fallback to uniform init if decode fails
            h_final = np.ones(n) * 0.5
    else:
        h_final = np.ones(n) * 0.5

    c5_current = evaluate_c5(h_final)
    
    # PGD stage with smooth max - 20% of time
    t_end_pgd = start_time + 0.9 * budget_s
    
    # Smooth max surrogate parameters
    beta = 10.0
    lr = 0.01
    
    def smooth_max_surrogate(h, beta=10.0):
        corr = correlate(h, 1 - h, mode="full")
        # Normalize correlation by dx
        corr *= dx
        # Smooth max using log-sum-exp
        max_val = np.max(corr)
        exp_vals = np.exp(beta * (corr - max_val))
        log_sum = np.log(np.sum(exp_vals))
        return max_val + log_sum / beta
    
    def compute_gradient(h, beta=10.0):
        corr = correlate(h, 1 - h, mode="full")
        corr *= dx
        max_val = np.max(corr)
        exp_vals = np.exp(beta * (corr - max_val))
        weights = exp_vals / np.sum(exp_vals)
        # Compute gradient of each correlation term w.r.t. h
        grad = np.zeros_like(h)
        for i, w in enumerate(weights):
            # Weights correspond to shifts k in -n+1 to n-1
            k = i - (n - 1)
            # Gradient contribution of this term
            # grad[i] += dx * (1 - h[(i+k)%n] - h[(i-k)%n])
            # Take proper wraps to avoid oversized indexing
            h_shifted_plus = np.roll(h, k)
            h_shifted_minus = np.roll(h, -k)
            if k == 0:
                local_grad = dx * (1 - 2 * h)
            else:
                local_grad = dx * (1 - h_shifted_plus - h_shifted_minus)
            grad += w * local_grad
        return grad
    
    def project_onto_feasible(h):
        # Project onto box [0,1] and simplex (sum = n/2)
        # Use water-filling to project onto the simplex, then clip
        total = np.sum(h)
        target = n / 2
        ratio = target / total
        h_clipped = np.clip(h * ratio, 0, 1) 
        sum_h = np.sum(h_clipped)
        if np.abs(sum_h - target) < 1e-8:
            return h_clipped
        # Water filling with equal increments
        # Minimize the amount we need to add to meet target
        def solve_lambda(lam):
            t = np.clip(h_clipped - lam, 0, 1)
            return np.sum(t) - target
        # Binary search for lambda (Newton/Rho modifications)
        # Bound lambda appropriately for simplicity
        low, high = -1.0, 1.0
        for i in range(60):
            mid = (low + high) / 2.0
            val = solve_lambda(mid)
            if val < 0:
                high = mid
            else:
                low = mid
        lam = (low + high) / 2.0
        h_proj = np.clip(h_clipped - lam, 0, 1)
        # Recompute sum and fix via rescaling
        s = np.sum(h_proj)
        if s > 0:
            h_proj = h_proj * (target / s)
        return h_proj

    h = h_final.copy()
    best_c5 = c5_current
    iteration = 0
    while time.time() < t_end_pgd:
        if iteration > 1000:
            break
        try:
            # Compute surrogate gradient and update
            grad = compute_gradient(h, beta)
            h_new = h - lr * grad
            h_proj = project_onto_feasible(h_new)
            c5_new = evaluate_c5(h_proj)
            if c5_new < best_c5:
                best_c5 = c5_new
                h = h_proj.copy()
        except:
            pass
        iteration += 1
        # Increase softness of log-sum-exp
        beta = min(25.0, beta + 0.1)
        if (time.time() - start_time) > t_end_pgd - 5:
            break

    # Final polishing with SLSQP if time permits - 10% of time
    t_end_final = time.time() + 0.1 * budget_s
    if time.time() < t_end_final:
        bounds_slsqp = [(0, 1) for _ in range(n)]
        def obj_slsqp(h_array):
            return evaluate_c5(h_array)
        def con_slsqp(h_array):
            return np.sum(h_array) * dx - 1.0

        cons = {'type': 'eq', 'fun': con_slsqp}
        try:
            res = minimize(obj_slsqp, h, method='SLSQP',
                           bounds=bounds_slsqp, constraints=cons,
                           options={'maxiter': 50, 'ftol': 1e-10, 'gtol': 1e-10})
            if res.success:
                h_final = res.x.copy()
                c5_final = evaluate_c5(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
        except:
            pass

    return (h_final.tolist(), best_c5, n)
```
