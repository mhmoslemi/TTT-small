def project_onto_feasible(h, target_sum, dx):
    """
    Project h onto the feasible set: 0 <= h_i <= 1 and sum(h_i * dx) = target_sum
    Using water-filling algorithm.
    """
    h = np.array(h)
    # Binary search for the correct lambda
    lower, upper = -1.0, 1.0
    for _ in range(30):
        mid = (lower + upper) / 2.0
        p = np.clip(h - mid, 0, 1)
        s = np.sum(p) * dx
        if s > target_sum:
            lower = mid
        else:
            upper = mid
    lambda_ = (lower + upper) / 2.0
    p = np.clip(h - lambda_, 0, 1)
    scale = target_sum / (np.sum(p) * dx)
    p = p * scale
    return p

def piecewise_linear_symmetric(h_points, x_breaks, n):
    """Generate symmetric piecewise linear function from given points."""
    # Linearly interpolate between break points
    # Only generate one half plus the center
    half = np.zeros(n // 2 + 1)
    x_half = np.linspace(0, 1, n // 2 + 1)
    for i in range(len(x_breaks) - 1):
        idx = np.where((x_half >= x_breaks[i]) & (x_half <= x_breaks[i+1]))[0]
        if len(idx) == 0:
            continue
        slope = (h_points[i+1] - h_points[i]) / (x_breaks[i+1] - x_breaks[i])
        intercept = h_points[i] - slope * x_breaks[i]
        half[idx] = slope * x_half[idx] + intercept
    
    # Mirror the values for symmetry
    h_full = np.concatenate([half[::-1], half[1:]])
    return h_full

def compute_c5_exact(h):
    """Compute the exact C5 value using max of correlations."""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def compute_c5_smooth(h, beta=20):
    """Compute a smooth approximation to C5 using log-sum-exp."""
    n = len(h)
    dx = 2.0 / n
    # Compute all shift correlations
    corr = correlate(h, 1 - h, mode="full")
    f_vals = corr * dx
    # Smooth max
    f_max = np.max(f_vals)
    exp_vals = np.exp(beta * (f_vals - f_max))
    w = exp_vals / np.sum(exp_vals)
    # Gradient of smooth max
    grad = np.zeros(n)
    for k in range(len(corr)):
        shift = k - (n - 1)
        slice_idx = slice(max(0, -shift), min(n, n-shift))
        shift_idx = slice(max(0, shift), min(n, n+shift))
        grad[slice_idx] += w[k] * dx * (1 - h[shift_idx])
    return np.sum(w * f_vals), grad
