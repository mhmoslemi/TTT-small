```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use a hybrid optimization approach
    # Phase 1: Use global optimization
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Use different evolutionary strategies
    strategies = ['best1bin', 'best2bin', 'rand1bin', 'rand2bin', 'currenttobestbin']
    
    for strategy in strategies:
        if (time.time() - start_time) > budget_s:
            break
        try:
            de_res = differential_evolution(
                objective, 
                bounds, 
                seed=seed, 
                maxiter=1000,  # Reduced maxiter for faster iteration
                popsize=15,    
                mutation=(0.8, 1),  
                recombination=0.9,   
                strategy=strategy
            )
            if de_res.success:
                h_candidate = de_res.x
                # Normalize to satisfy constraint
                dx = 2.0 / n_points
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
    
    # Phase 2: Local search from good global points
    local_restart_bound = 500
    
    for restart in range(local_restart_bound):
        if (time.time() - start_time) > budget_s:
            break
            
        # Start with small random perturbations around previous best
        new_h = best_h + np.random.normal(0, 0.01, n_points)
        new_h = np.clip(new_h, 0, 1)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue

    # Phase 3: Try to improve with structured pattern
    if (time.time() - start_time) < budget_s * 0.9:
        try:
            # Generate patterns with different structures
            patterns_to_try = []
            
            # Generate random binary with constraint
            binary_pattern = np.random.rand(n_points) > 0.5
            binary_pattern = binary_pattern.astype(float)
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(binary_pattern) * dx)
            binary_pattern = binary_pattern * scale_factor
            patterns_to_try.append(binary_pattern)
            
            # Generate counting pattern with gentle variation
            x = np.linspace(0, 2, n_points)
            # Use sine to create smooth variation
            smooth_pattern = np.clip(0.5 + 0.5 * np.sin(np.pi * x), 0, 1)
            scale_factor = 1.0 / (np.sum(smooth_pattern) * dx)
            smooth_pattern = smooth_pattern * scale_factor
            patterns_to_try.append(smooth_pattern)
            
            # Try a piecewise constant step pattern
            step_pattern = np.zeros(n_points)
            # Use larger intervals
            intervals = 6
            segment_size = n_points // intervals
            for i in range(intervals):
                start_idx = i * segment_size
                end_idx = (i + 1) * segment_size if i < intervals - 1 else n_points
                # Alternate between 0 and 1 to prevent extreme values
                if i % 2 == 0:
                    step_pattern[start_idx:end_idx] = 1.0
                else:
                    step_pattern[start_idx:end_idx] = 0.0

            # Fix very unequal proportions and rescale
            scale_factor = 1.0 / (np.sum(step_pattern) * dx)
            step_pattern = step_pattern * scale_factor
            patterns_to_try.append(step_pattern)
            
            # Try a hybrid pattern with manual checks
            # Start with initial solution, modify slightly
            hybrid_pattern = best_h.copy()
            # Perform a few swaps of high low values to explore
            indices = list(range(n_points))
            random.shuffle(indices)
            
            # Change some values to explore
            for i in range(n_points // 10):
                idx1 = indices[i]
                idx2 = indices[i+1] if i+1 < n_points else indices[0]
                # Swap jiggles
                val1 = hybrid_pattern[idx1]
                val2 = hybrid_pattern[idx2]
                hybrid_pattern[idx1] = val2
                hybrid_pattern[idx2] = val1

            # Rescale the hybrid
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(hybrid_pattern) * dx)
            hybrid_pattern = hybrid_pattern * scale_factor
            patterns_to_try.append(hybrid_pattern)
                
            # Evaluate all patterns
            for pattern in patterns_to_try:
                if len(pattern) != n_points:
                    continue
                c5_candidate = objective(pattern)
                if c5_candidate < best_c5:
                    best_h = pattern.copy()
                    best_c5 = c5_candidate
                
        except:
            pass

    return (best_h.tolist(), best_c5, n_points)
```
