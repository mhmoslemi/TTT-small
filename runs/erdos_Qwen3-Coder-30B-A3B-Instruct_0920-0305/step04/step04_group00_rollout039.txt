import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_onto_feasible(h, target_sum):
    """Project h onto the feasible set: 0 <= h[i] <= 1 and sum(h) = target_sum"""
    # Use waterfilling to project onto the simplex with constraint sum(h) = target_sum
    h = np.array(h)
    lmbda_low, lmbda_high = -1.0, 1.0
    # Ensure bounds are large enough for cases where projection is needed
    while np.sum(np.clip(h - lmbda_low, 0, 1)) < target_sum:
        lmbda_low -= 1.0
    while np.sum(np.clip(h - lmbda_high, 0, 1)) > target_sum:
        lmbda_high += 1.0
    
    def sum_at_lambda(lmbda):
        return np.sum(np.clip(h - lmbda, 0, 1))
    
    # Binary search
    for _ in range(50):
        lmbda = (lmbda_low + lmbda_high) / 2.0
        current_sum = sum_at_lambda(lmbda)
        if current_sum > target_sum:
            lmbda_low = lmbda
        else:
            lmbda_high = lmbda
    lmbda = (lmbda_low + lmbda_high) / 2.0
    h_projected = np.clip(h - lmbda, 0, 1)
    scale = target_sum / np.sum(h_projected)
    h_projected = h_projected * scale
    return h_projected

def build_piecewise_linear_symmetric(K, heights, n):
    """Build a piecewise linear symmetric function with K+1 breakpoints."""
    # Sorted (descending) heights
    heights = np.sort(heights)[::-1]
    xs = np.linspace(0, 1, K+1)
    # Linear interpolation on [0,1]
    ax = np.linspace(0, 1, n//2)
    heights_interp = np.interp(ax, xs, heights)
    # Mirror and concatenate to [0,2]
    full_h = np.concatenate([heights_interp, heights_interp[::-1]])
    return full_h

def smooth_max_surrogate(h, beta=20):
    """Return a smooth approximation of max_k sum_i h_i (1 - h_{i+k})"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    # Correct the scale: correlate returns un-normalized values
    corr = corr * dx
    
    # Use log-sum-exp approximation
    max_corr = np.max(corr)
    # subtract max_corr to prevent overflow
    exp_terms = np.exp(beta * (corr - max_corr))
    log_sum_exp = max_corr + np.log(np.sum(exp_terms)) / beta
    return log_sum_exp

def smooth_max_gradient(h, beta=20):
    """Return gradient of smooth max surrogate."""
    n = len(h)
    dx = 2.0 / n
    # compute correlations manually for gradients
    corr = correlate(h, 1 - h, mode="full")
    corr = corr * dx
    max_corr = np.max(corr)
    exp_terms = np.exp(beta * (corr - max_corr))
    weights = exp_terms / np.sum(exp_terms)
    
    # Construct gradient per each shift
    grad = np.zeros(n)
    for k in range(len(corr)):
        shift_k = k-n+1
        # This is a compact way of computing d/dh_i of sum_j h_j (1 - h_{j+k})
        # (but indices need to be handled carefully, assuming 0 outside the range)
        # In correlate, we compute sum_j h[j] * (1 - h[j+k]) for shift k (with offset)
        # So gradient wrt h[i] is (1 - h[i+k]) for every j that contributes
        # But let's compute it more cleanly for efficiency
        s = 0
        for i in range(n):
            j = i + shift_k
            if 0 <= j < n:
                grad[i] += weights[s] * (-dx * (1 - h[j]))
        s += 1
    
    # More direct way:
    # compute each gradient entry by accumulating all terms contributing to it
    grad = np.zeros(n)
    for k_idx, weight in enumerate(weights):
        shift = k_idx - (n-1)
        for i in range(n):
            j = i + shift
            if 0 <= j < n:
                grad[i] -= weight * dx * (1 - h[j])
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 200
    dx = 2.0 / n
    
    # Number of breakpoints
    K = 5
    # Optimization variables are heights [h0, h1, ..., hK] with h0 >= h1 >= ... >= hK >= 0
    bounds = [(0, 1) for _ in range(K+1)]
    
    def decode_params(params):
        heights = np.array(params)
        # sort descending for monotonicity
        heights = np.sort(heights)[::-1]
        h = build_piecewise_linear_symmetric(K, heights, n)
        # enforce integral constraint
        h_sum = np.sum(h) * dx
        h = h * (n/2) / h_sum
        # clip if necessary to ensure <= 1
        h = np.clip(h, 0, 1)
        # final renormalization if still needed
        h_sum = np.sum(h) * dx
        if np.abs(h_sum - 1.0) > 1e-10:
            h = h * (n/2) / h_sum
        h = np.clip(h, 0, 1)
        return h
    
    def objective(params):
        h = decode_params(params)
        return evaluate_c5(h)
    
    # Global optimization
    de_result = differential_evolution(
        objective,
        bounds,
        seed=seed,
        maxiter=int(1000 * (1 - 0.7)),  # Use 70% time for DE
        popsize=15,
        mutation=0.8,
        recombination=0.7,
        strategy='best1bin'
    )
    
    if not de_result.success:
        raise RuntimeError("DE optimization failed")
    
    # Best solution from DE
    best_params = de_result.x
    best_h = decode_params(best_params)
    current_best_c5 = evaluate_c5(best_h)
    
    # Now refine with smooth surrogate gradient descent
    t_pgd = 0.2 * budget_s
    start_pgd = time.time()
    h_current = best_h.copy()
    
    learning_rates = np.linspace(0.05, 0.002, 1000)
    x = 0
    
    while time.time() - start_pgd < t_pgd and x < len(learning_rates):
        # Compute gradient with soft max
        try:
            # Fixed small beta for smoothness
            grad = smooth_max_gradient(h_current, beta=20)
            # Take step
            lr = learning_rates[x]
            h_new = h_current - lr * grad
            # Project back
            h_current = project_onto_feasible(h_new, n/2)
            # Evaluate new value
            c5_val = evaluate_c5(h_current)
            if c5_val < current_best_c5:
                current_best_c5 = c5_val
                best_h = h_current.copy()
            x += 1
        except Exception as e:
            x += 1
            continue
    
    # Optionally do final polishing with exact optimization
    t_final = 0.1 * budget_s
    start_final = time.time()
    
    # Final polishing step
    bounds_final = [(0, 1) for _ in range(n)]
    cons_eq = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}
    
    try:
        res = minimize(
            evaluate_c5,
            best_h,
            method='SLSQP',
            bounds=bounds_final,
            constraints=cons_eq,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            final_h = res.x
            scale = (n / 2) / (np.sum(final_h) * dx)
            final_h = final_h * scale
            final_h = np.clip(final_h, 0, 1)
            final_c5 = evaluate_c5(final_h)
            if final_c5 < current_best_c5:
                current_best_c5 = final_c5
                best_h = final_h
    except:
        pass  # Fallback to best found so far
    
    return (best_h.tolist(), current_best_c5, n)
