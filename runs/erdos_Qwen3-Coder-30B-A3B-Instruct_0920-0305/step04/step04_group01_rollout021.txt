import numpy as np
from scipy.optimize import minimize
import time

# Global variable for constraint
TARGET_SUM = None

def evaluate_c5_fft(h):
    """Evaluate the C5 constant using FFT for speed"""
    n = len(h)
    if n == 0:
        return 0.0
    # Reverse h
    h_rev = h[::-1]
    b = 1.0 - h
    L = 2 * n - 1
    # Compute FFT lengths
    hf = np.fft.rfft(h_rev, n=L)
    bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(hf * bf, n=L)
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)

def compute_corr_fft(h):
    """Compute the full correlation array for gradient calculation"""
    n = len(h)
    h_rev = h[::-1]
    b = 1.0 - h
    L = 2 * n - 1
    hf = np.fft.rfft(h_rev, n=L)
    bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(hf * bf, n=L)
    return conv

def project_to_sum_box(x, target_sum):
    """Project vector to box [0,1] and sum constraint using binary search"""
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x
    
    # Binary search for the shift
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(x + mid, 0, 1)
        s = proj.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) / 2.0, 0, 1)

def simulated_annealing(h_initial, target_sum, time_limit, seed):
    """Run simulated annealing with temperature schedule"""

    n = len(h_initial)
    h = np.copy(h_initial)
    # Ensure feasibility via project

    best_c = cur_c

    # Set cooling schedule parameters

    Tmin = 1e-6
    
    start = time.time()
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start < time_limit:
        # Determine temperature
        elapsed = time.time() - start
        progress = elapsed / time_limit
        T = T0 * (1 - progress) + Tmin * progress

        # Random move
        i, j = np.random.choice(n, size=2, replace=False)
        
        # Compute valid delta range
        delta_max = min(h[i], 1 - h[j])
        delta_min = -min(1 - h[i], h[j])
        
        if delta_max <= delta_min:
            continue
            
        # Choose delta uniformly in [delta_min, delta_max]
        delta = np.random.uniform(delta_min, delta_max)
        
        # Update h
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # evaluate
        c_new = evaluate_c5_fft(h_new)
        
        # Accept or reject
        if c_new < cur_c or np.random.rand() < np.exp(-(c_new - cur_c) / (T + 1e-12)):
            h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                last_improvement = iter_count
                
        iter_count += 1
        
    return best_h, best_c

    """Constraint function to enforce sum(h) = TARGET_SUM"""
    return np.sum(h) - TARGET_SUM
