import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_binary_milp(n):
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(x) == n // 2]
    
    for d in range(-n+1, n):
        if d == 0:
            continue
        w_vars = []
        if d > 0:
            rng = range(n - d)
        else:
            rng = range(-d, n)
        for i in rng:
            w = cp.Variable(boolean=True)
            constraints.append(w <= x[i])
            constraints.append(w <= x[i+d])
            constraints.append(w >= x[i] + x[i+d] - 1)
            w_vars.append(w)
        N_d = cp.sum(x) - cp.sum(w_vars)
        constraints.append(N_d <= t)
    constraints.append(t >= 0)

    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        prob.solve(solver=cp.CBC, verbose=False, warm_start=True, max_iters=5000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_vals = np.round(x.value).astype(float)
            t_val = int(round(t.value))
            return x_vals, t_val
    except:
        pass
    
    try:
        prob.solve(solver=cp.GLPK_MI, verbose=False, warm_start=True, max_iters=5000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_vals = np.round(x.value).astype(float)
            t_val = int(round(t.value))
            return x_vals, t_val
    except:
        pass
    
    # fallback to continuous relaxation if integer fails
    try:
        prob.solve(solver=cp.ECOS_BB, verbose=False, warm_start=True, max_iters=5000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_vals = np.round(x.value).astype(float)
            t_val = int(round(t.value))
            return x_vals, t_val
    except:
        pass
        
    raise RuntimeError("MILP solve failed")
