import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial values
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Step 1: Solve binary optimization using MILP
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Create binary optimization problem with CVXPY
    try:
        x = Variable(n_points, boolean=True)
        t = Variable(integer=True)
        
        # Sum constraint
        prob = Problem(Minimize(t), [
            sum(x) == n_points // 2,
            t >= 0
        ])
        
        # Add overlap constraints
        for d in range(-n_points + 1, n_points):
            if d == 0:
                continue
            sum_h = 0
            for i in range(n_points):
                if i + d >= 0 and i + d < n_points:
                    sum_h += x[i] * x[i + d]
            prob.constraints.append(n_points // 2 - sum_h <= t)
        
        # Solve
        prob.solve(solver=CBC, verbose=False)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            x_sol = np.array([float(xi.value) for xi in x]).round()
            if np.allclose(np.sum(x_sol), n_points / 2):
                h_binary = x_sol
                c5_binary = evaluate_c5(h_binary)
                if c5_binary < best_c5:
                    best_h = h_binary.copy()
                    best_c5 = c5_binary
                    
    except Exception as e:
        pass
        
    # Step 2: Refine with smooth approximation
    def smooth_max_overlaps(h, tau=0.1):
        """Smoothed version of max overlap"""
        n = len(h)
        dx = 2.0 / n
        overlaps = []
        for d in range(-n + 1, n):
            if d == 0:
                continue
            overlap = 0
            for i in range(n):
                j = i + d
                if 0 <= j < n:
                    overlap += h[i] * h[j]
            overlaps.append(overlap)
        
        # Use log-sum-exp to smooth
        overlaps = np.array(overlaps)
        log_sum_exp = tau * np.log(np.sum(np.exp(overlaps / tau)))
        return log_sum_exp / dx
    
    def softmax_gradient(h, tau=0.1):
        """Compute gradient of smooth max overlap"""
        n = len(h)
        dx = 2.0 / n
        weights = np.zeros(len(h))
        d = 0
        while d < len(h):
            if d == 0:
                d += 1
                continue
            x = 0
            for i in range(n):
                j = i + d
                if 0 <= j < n:
                    x += h[i] * h[j]
            weights[d] = np.exp(-x / tau)
            d += 1
        weights /= np.sum(weights)  # normalize 
        # Not exact gradient, but approximation
        g = np.zeros_like(h)
        for i in range(n):
            for d in range(1, n):
                j = i + d
                if 0 <= j < n:
                    g[i] += weights[d] * h[j]
        g = g * (-1.0 / dx)
        return g

    def project_onto_simplex(h, total=1.0):
        """Project a vector onto the simplex"""
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum <= 0:
            return np.full_like(h, 0)
        h = h * (total / h_sum)
        return np.clip(h, 0, 1)
    
    # Initialize with best binary solution
    h_continuous = best_h.copy().astype(float)
    h_continuous = project_onto_simplex(h_continuous, total=n_points / 2)

    # Smooth max descent with continuation
    max_iter = 1000
    max_restart = 20
    restart_count = 0
    tau_schedule = [0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625]
    
    while restart_count < max_restart and (time.time() - start_time) < budget_s * 0.9:
        h_current = h_continuous.copy()
        
        for tau in tau_schedule:
            if (time.time() - start_time) > budget_s * 0.9:
                break
                
            iterations = int(200 * (0.5 if tau > 0.1 else 1.0))
            for _ in range(iterations):
                if (time.time() - start_time) > budget_s * 0.9:
                    break
                    
                grad = softmax_gradient(h_current, tau)
                h_trial = h_current - 0.01 * grad
                h_trial = project_onto_simplex(h_trial, total=n_points / 2)
                
                c5_trial = evaluate_c5(h_trial)
                c5_current = evaluate_c5(h_current)
                
                if c5_trial < c5_current:
                    h_current = h_trial.copy()
                elif np.random.rand() < 0.05:
                    # Random perturbation
                    noise = np.random.normal(0, 0.02, n_points)
                    h_trial = h_current + noise
                    h_trial = project_onto_simplex(h_trial, total=n_points / 2)
                    if evaluate_c5(h_trial) < c5_current:
                        h_current = h_trial.copy()
        
        # Final polishing step
        def obj_func(h):
            return evaluate_c5(h)
            
        try:
            res = minimize(
                obj_func,
                h_current,
                method='SLSQP',
                bounds=[(0, 1)] * n_points,
                options={'maxiter': 100}
            )
            if res.success or res.fun < evaluate_c5(h_current):
                c5_polished = res.fun if res.success else evaluate_c5(h_current)
                if c5_polished < best_c5:
                    best_h = res.x.copy() if res.success else h_current.copy()
                    best_c5 = c5_polished
        except:
            pass
            
        # Increment restart count
        restart_count += 1
        
        # Perturb for next iteration
        noise = np.random.normal(0, 0.01, n_points)
        h_continuous = h_current + noise
        h_continuous = project_onto_simplex(h_continuous, total=n_points / 2)
    
    return (best_h.tolist(), best_c5, n_points)
