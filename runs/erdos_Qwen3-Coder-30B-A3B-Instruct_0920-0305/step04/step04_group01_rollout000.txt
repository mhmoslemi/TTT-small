import numpy as np
from scipy.optimize import minimize
import time

TARGET_SUM = None

def evaluate_c5_fft(h):
    """Evaluate the C5 constant using FFT for speed"""
    n = len(h)
    if n == 0:
        return 0.0
    # Reverse h
    h_rev = h[::-1]
    b = 1.0 - h
    L = 2 * n - 1
    # Compute FFT lengths
    hf = np.fft.rfft(h_rev, n=L)
    bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(hf * bf, n=L)
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)

def project_to_sum_box(x, target_sum):
    """Project vector x to sum constraint and box [0,1]"""
    # Clip x to [0,1]
    x = np.clip(x, 0, 1)
    # Adjust sum
    current_sum = np.sum(x)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return x.copy()
    
    # Simple approach: push towards target sum
    if diff > 0:
        # Need to add more values
        # Find indices that can be increased
        free_indices = np.where(x < 1)[0]
        if len(free_indices) == 0:
            return x
        # Distribute the increase equally among free indices
        to_add = diff / len(free_indices)
        
        # But ensure we don't exceed 1
        for idx in free_indices:
            increase = min(to_add, 1 - x[idx])
            x[idx] += increase
            diff -= increase
            if abs(diff) < 1e-10:
                break
    else:
        # Need to reduce
        # Find indices that can be decreased
        free_indices = np.where(x > 0)[0]
        if len(free_indices) == 0:
            return x
        # Distribute decrease equally
        to_remove = -diff / len(free_indices)
        for idx in free_indices:
            decrease = min(to_remove, x[idx])
            x[idx] -= decrease
            diff += decrease
            if abs(diff) < 1e-10:
                break
        
    return x.copy()

def simulated_annealing(h0, target_sum, time_limit, seed):
    """Run simulated annealing with temperature schedule"""
    np.random.seed(seed)
    n = len(h0)
    
    # Initial project
    h = project_to_sum_box(h0, target_sum)
    cur_c = evaluate_c5_fft(h)
    best_h = h.copy()
    best_c = cur_c
    
    start_time = time.time()
    
    # Temporal schedule
    T_initial = max(0.005, best_c * 0.05)
    T_final = 1e-6
    
    # Simple loop with time control
    iter_count = 0
    while time.time() - start_time < time_limit:
        # Update temperature
        elapsed = time.time() - start_time
        progress = elapsed / time_limit
        T = T_initial * (1 - progress) + T_final * progress
        
        # Propose random move (swap mass between two indices)
        i, j = np.random.choice(n, size=2, replace=False)
        
        # Compute valid delta range
        delta_min = max(-h[i], h[j] - 1.0)
        delta_max = min(h[i], 1.0 - h[j])
        
        if delta_min >= delta_max:
            continue
            
        # Choose delta
        delta = np.random.uniform(delta_min, delta_max)
        
        # Make move
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # Evaluate
        c_new = evaluate_c5_fft(h_new)
        
        # Acceptance probability
        if c_new < cur_c or np.random.rand() < np.exp(-(c_new - cur_c) / (T + 1e-12)):
            h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        
        iter_count += 1
        
    return best_h, best_c
