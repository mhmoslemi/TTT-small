import numpy as np

from scipy.fft import fft, ifft
import time
from scipy.optimize import minimize

# Global variable for constraint
TARGET_SUM = None

def compute_corr_fft(h):
    """Compute full correlation using FFT. Returns correlation array."""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute 1 - h
    b = 1 - h
    # Use FFT for convolution (real inputs, use real-FFT algorithms)
    # Length L = 2*n - 1
    L = 2 * n - 1
    # Pad with zeros to length L
    h_pad = np.pad(h_rev, (0, L - n), mode='constant')
    b_pad = np.pad(b, (0, L - n), mode='constant')
    # Use real FFT for efficiency
    f1 = fft(h_pad)
    f2 = fft(b_pad)
    corr_full = ifft(f1 * np.conj(f2))
    # Take only real part, because result should be real (by symmetry)
    corr_real = np.real(corr_full)
    return corr_real

def c5_from_corr(corr, n):
    """Convert correlation array to C5 value with proper scaling"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT-based correlation"""
    n = len(h)
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n)

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection onto box and sum constraints"""
    x = np.array(x)
    low = 0.0
    high = 1.0
    
    def f(lam):
        # Project x + lam to [0,1] and compute sum - target_sum
        y = np.clip(x + lam, low, high)
        return np.sum(y) - target_sum
    
    # Binary search for lambda in [-1, 1]
    a = -1.0
    b = 1.0
    eps = 1e-12
    for _ in range(100):
        if b - a < eps:
            break
        lam = (a + b) / 2.0
        val = f(lam)
        if val < 0:
            a = lam
        else:
            b = lam
    lam = (a + b) / 2.0
    result = np.clip(x + lam, low, high)
    return result

def random_pair_move(h, valid_indices=None):
    """Generate a feasible random pairwise move (i,j, delta) for h"""
    if valid_indices is None:
        valid_indices = list(range(len(h)))
    i, j = np.random.choice(valid_indices, size=2, replace=False)
    delta_min = max(-h[j], h[i] - 1.0)
    delta_max = min(h[i], 1.0 - h[j])
    if delta_min >= delta_max:
        return None
    delta = np.random.uniform(delta_min, delta_max)
    return (i, j, delta)

def guided_move(h, worst_lag):
    """Perform a deterministic guided move towards reduction in worst correlation"""
    n = len(h)
    # Compute subgradient for lag: g_i = h_{i-lag} + h_{i+lag} 
    # (with out-of-bounds treated as 0)
    g = np.zeros(n)
    if worst_lag > 0:
        g[worst_lag:] += h[:-worst_lag]  # h_{i-lag} term
        g[:-worst_lag] += h[worst_lag:]  # h_{i+lag} term
    elif worst_lag < 0:
        # For negative lag, shift h in the positive direction to get h_{i-lag}
        g[:worst_lag] += h[-worst_lag:]  
        g[-worst_lag:] += h[:worst_lag]
    else:
        # lag = 0: g_i = h_i + h_i = 2 h_i
        g = 2.0 * h
    
    # Find indices of min and max g values
    i_min = np.argmin(g)
    i_max = np.argmax(g)
    if i_min == i_max:
        return None
    
    # Determine transfer amount:
    # Transfer a small amount from i_min (lower g) to i_max (higher g) 
    delta_min = max(-h[i_min], 1 - h[i_max])
    delta_max = min(h[i_min], h[i_max])
    if delta_min >= delta_max:
        return None
    # Slightly scale it down, avoid exact full transfer which is not guaranteed feasible
    transfer_amount = min(0.2, (delta_max - delta_min) * 0.5) + delta_min
    # Ensure it's between bounds
    delta = np.random.uniform(delta_min, delta_max)
    
    return (i_min, i_max, delta)

def anneal(h0, run_seconds, rng):
    """Single simulated annealing run"""
    global TARGET_SUM
    n = len(h0)
    TARGET_SUM = n / 2
    h = h0.copy()
    h = project_to_sum_box(h, TARGET_SUM)
    c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    start_time = time.time()
    T0 = 5e-3
    Tf = 1e-6
    t0 = start_time

    # Each iteration does either a guided or random move
    move_counter = 0
    while True:
        t = time.time()
        if t - t0 >= run_seconds:
            break

        # Temperature schedule
        elapsed = t - start_time
        T = T0 * (1.0 - elapsed / run_seconds) + Tf * (elapsed / run_seconds)
        if T < Tf:
            T = Tf
            
        # Alternate moves
        if move_counter % 10 == 0:
            # guided move
            # Find worst lag 
            corr = compute_corr_fft(h)
            idx = np.argmax(corr)
            # The lag is `idx - n + 1`
            lag = idx - n + 1
            move = guided_move(h, lag)
        else:
            # random pairwise move
            move = random_pair_move(h)
        
        if move is None:
            move_counter += 1
            continue
            
        i, j, delta = move
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # Project back to feasible set
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)

        # Accept decision
        if c5_new <= c5 or np.random.rand() < np.exp(-(c5_new - c5) / T):
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

        move_counter += 1

    return best_h, best_c5
