import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

    integral_constraint = cp.sum(h) == n_points // 2

    symmetry_constraint = h[0] == 1
    
    # Add logical AND constraints
    constraints = [integral_constraint, symmetry_constraint]

            # p[i,l] <= h[i]
            constraints.append(p[i, l] <= h[i])
            # p[i,l] <= h[i+l]
            constraints.append(p[i, l] <= h[i + l])
            # p[i,l] >= h[i] + h[i+l] - 1
            constraints.append(p[i, l] >= h[i] + h[i + l] - 1)

        A_l = cp.sum(h[:n_points-l])  # Sum of h_i for i = 0..n-1-l
