import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Problem, Minimize, Variable, Binary, sum as cp_sum

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

        # Model parameters
        n = n_points
        half_n = n // 2
        M = Variable(integer=True, name='M')
        
        # Model variables
        h = Variable(n, boolean=True, name='h')
        
        # Create p variables for each lag l
        p = {}
        for l in range(1, n):
            p[l] = Variable((n - l), boolean=True, name=f'p_{l}')
        
        # Constraints: p[l][i] = h[i] AND h[i+l]
        constraints = []
        for l in range(1, n):
            for i in range(n - l):
                # p[i,l] <= h[i]
                constraints.append(p[l][i] <= h[i])
                # p[i,l] <= h[i+l]  
                constraints.append(p[l][i] <= h[i + l])
                # p[i,l] >= h[i] + h[i+l] - 1
                constraints.append(p[l][i] >= h[i] + h[i + l] - 1)
        
        # Integral constraint
        constraints.append(cp_sum(h) == half_n)
        
        # Minimum overlap constraints (to ensure M captures the max overlap)
        for l in range(1, n):
            S_l = cp_sum(p[l])
            A_l_plus = cp_sum(h[:n-l])
            A_l_minus = cp_sum(h[l:])
            constraints.append(A_l_plus - S_l <= M)
            constraints.append(A_l_minus - S_l <= M)
        
        # Objective
        objective = Minimize(M)
        problem = Problem(objective, constraints)
        
        # Try solving with CBC (free)
        problem.solve(solver=cp.CBC, max_seconds=budget_s/2, verbose=False)
        
        if problem.status in ["optimal", "optimal_inaccurate"]:

            M_val = M.value

            if M_val is not None:
                # Compute actual C5
                dx = 2.0 / n
                c5_milp = M_val * dx
                
                # Convert to floating point
                h_float = h_solution.astype(float)
                
                # Polish with SLSQP
                def obj_func(x):
                    # Clip to [0,1]
                    x = np.clip(x, 0, 1)
                    return evaluate_c5(x)
                
                def constraint_eq(x):
                    return np.sum(x) * dx - 1.0
                
                bounds = [(0, 1) for _ in range(n)]
                cons = {'type': 'eq', 'fun': constraint_eq}

                if current_c5 < best_c5_so_far:
                    best_c5_so_far = current_c5
