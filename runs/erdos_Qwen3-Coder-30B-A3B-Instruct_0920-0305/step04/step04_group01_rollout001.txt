import time

# Global variable for constraint
TARGET_SUM = None

def evaluate_c5_fft(h):
    """Evaluate the C5 constant for a given step function h using FFT."""
    n = len(h)
    if n == 0:
        return 0.0
    # Reverse h
    h_rev = h[::-1]
    b = 1.0 - h
    L = 2 * n - 1
    # Compute FFT lengths
    hf = np.fft.rfft(h_rev, n=L)
    bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(hf * bf, n=L)
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)

def project_to_sum_box(x, target_sum):
    """Project vector x onto sum constraint and box [0,1]"""
    # Clip to [0,1]
    x = np.clip(x, 0, 1)
    # Adjust sum via shift
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x
    
    # Compute lambda for projection
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(x + mid, 0, 1)
        s = proj.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo+hi)/2.0, 0, 1)

def simulated_annealing(h_initial, target_sum, time_limit, seed):
    """Perform simulated annealing on the h vector."""
    np.random.seed(seed)
    n = len(h_initial)
    h = np.copy(h_initial)
    # Ensure feasibility via project
    h = project_to_sum_box(h, target_sum)
    
    cur_c = evaluate_c5_fft(h)
    best_h = h.copy()
    best_c = cur_c

    # Set temperature schedule
    T0 = cur_c * 0.005 if cur_c > 0 else 0.005
    Tmin = 1e-6
    start = time.time()
    iter_count = 0
    
    while time.time() - start < time_limit:
        # Random move
        i, j = np.random.choice(n, size=2, replace=False)
        
        # Delta range
        delta_max = min(h[i], 1 - h[j])
        delta_min = -min(1 - h[i], h[j])
        
        if delta_max <= delta_min:
            continue
        
        # Choose delta uniformly in [delta_min, delta_max]
        delta = np.random.uniform(delta_min, delta_max)
        
        # Update h
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # Evaluate
        c_new = evaluate_c5_fft(h_new)
        
        # Compute temperature based on proportion of elapsed time
        elapsed = time.time() - start
        progress = elapsed / time_limit
        T = T0 * (1 - progress) + Tmin * progress
        
        # Accept with probability
        if c_new < cur_c or np.random.rand() < np.exp(-(c_new - cur_c) / (T + 1e-12)):
            h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        
        iter_count += 1
    
    return best_h, best_c

    """Constraint function for sum equality."""
    return np.sum(h) - TARGET_SUM
