import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

# Global variable to hold the target sum for constraint
TARGET_SUM = None

def compute_corr_fft(h):
    """Compute the full correlation using FFT."""
    n = len(h)
    # Reverse the input
    h_rev = h[::-1]
    # Create the second vector (1 - h)
    b = 1 - h
    # Use FFT based correlation with length 2*n-1
    L = 2*n - 1
    H_rev = fft(h_rev, L)
    B = fft(b, L)
    corr = ifft(H_rev * B.conj()).real
    # Return only the valid part (corresponding to full correlations)
    return corr[:L]

def c5_from_corr(corr, n):
    """Convert correlation result to C5 bound."""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT."""
    n = len(h)
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n)

def project_to_sum_box(x, target_sum):
    """Project a vector onto the box and sum constraints using shift-and-clip."""
    x = np.asarray(x)
    n = len(x)
    
    # Binary search over lambda to find the right shift
    def sum_with_shift(lmbda):
        clipped = np.clip(x + lmbda, 0, 1)
        return np.sum(clipped) - target_sum
    
    low, high = -1.0, 1.0
    for _ in range(50):  # Binary search iterations
        mid = (low + high) / 2.0
        if sum_with_shift(mid) > 0:
            high = mid
        else:
            low = mid
    
    lambda_opt = (low + high) / 2.0
    projected = np.clip(x + lambda_opt, 0, 1)
    return projected

def random_pair_move(h, max_delta=0.1):
    """Generate a random pairwise move."""
    n = len(h)
    i = np.random.randint(0, n)
    j = np.random.randint(0, n)
    if i == j:
        return None
    
    # Compute feasible range
    delta_min = max(-h[j], h[i] - 1)
    delta_max = min(h[i], 1 - h[j])
    
    if delta_min >= delta_max:
        return None
    
    delta = np.random.uniform(delta_min, delta_max)
    
    return (i, j, delta)

def guided_move(h, worst_lag):
    """execute a directed move based on worst lag gradient."""
    n = len(h)
    # Compute the "subgradient" for the worst lag
    zero_pad = np.zeros(n)
    h_padded = np.concatenate([zero_pad, h, zero_pad])
    g = h_padded[worst_lag + n: 2 * n + worst_lag] + h_padded[n - worst_lag: n + n - worst_lag]
    
    # Accuracy: sometimes due to edge effects, g may have nan/inf
    g = np.nan_to_num(g, copy=False, nan=0.0, posinf=1e10, neginf=-1e10)
    
    # Find minimal and maximal indices of g
    g_min_idx = np.argmin(g)
    g_max_idx = np.argmax(g)
    
    if g_max_idx == g_min_idx:
        return False, h, 0.0  # No progress

    # Execute transfer from g_min to g_max
    delta = min(0.1, min(h[g_min_idx], 1 - h[g_max_idx]))
    if delta <= 0:
        return False, h, 0.0  # No valid transfer

    h_new = h.copy()
    h_new[g_min_idx] -= delta
    h_new[g_max_idx] += delta
    
    # Project to satisfy constraint
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    
    return True, h_new, delta

def anneal(h0, run_seconds, rng):
    """Perform a simulated annealing run."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    
    T0 = 5e-3
    Tf = 1e-6
    n_iter = 0
    max_iter = 1000000 if run_seconds > 10 else 100000  # Adaptive based on run time
    
    while n_iter < max_iter and (time.time() - start_time) < run_seconds:
        # Update temperature
        t = (time.time() - start_time) / run_seconds
        T = T0 * (1 - t) + Tf * t
        
        if n_iter % 10 == 0:
            # Guided move
            corr = compute_corr_fft(h)
            worst_lag_idx = np.argmax(corr)
            # The full correlation index needs remapping:
            # corr[0] is lag = -(n-1)
            # corr[i] corresponds to lag i - (n-1)
            true_lag = worst_lag_idx - (len(corr) - 1) // 2
            
            # Decrement the gamut on lucky indication
            guided_accept, h_trial, delta = guided_move(h, true_lag)
            if guided_accept:
                c5_trial = evaluate_c5(h_trial)
                if c5_trial < best_c5:
                    best_h = h_trial.copy()
                    best_c5 = c5_trial
                h = h_trial
        else:
            # Random move
            move = random_pair_move(h)
            if move is not None:
                i, j, delta = move
                h_trial = h.copy()
                h_trial[i] -= delta
                h_trial[j] += delta
                h_trial = project_to_sum_box(h_trial, TARGET_SUM)
                c5_trial = evaluate_c5(h_trial)
                # Accept or reject based on Metropolis criterion
                if c5_trial < best_c5 or np.random.rand() < np.exp(-(c5_trial - best_c5)/T):
                    h = h_trial
                    if c5_trial < best_c5:
                        best_h = h.copy()
                        best_c5 = c5_trial
        n_iter += 1
    
    return best_h, best_c5

def local_refine(h):
    """Refine with SLSQP."""
    n = len(h)
    bounds = [(0, 1) for _ in range(n)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = project_to_sum_box(res.x, TARGET_SUM)
            return h_refined
    except:
        pass
    return h

        h0 = init_func()
        h_final, c5_final = anneal(h0, 5.0, rng)

    if (time.time() - start_time) < budget_s * 0.95:
        h_refined = local_refine(best_h)
        c5_refined = evaluate_c5(h_refined)
        if c5_refined < best_c5:
            best_h = h_refined
