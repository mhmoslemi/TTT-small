```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def build_symmetric_piecewise_linear(h_raw, n_points):
    """Build a symmetric piecewise linear function from raw heights"""
    # h_raw needs to be sorted descending
    h_sorted = np.sort(h_raw)[::-1]
    K = len(h_sorted) - 1
    x = np.linspace(0, 1, K + 1)
    
    # Interpolate over [0,1] using the sorted heights
    x_full = np.linspace(0, 1, n_points // 2)
    h_interp = np.interp(x_full, x, h_sorted, left=h_sorted[0], right=h_sorted[-1])
    
    # Mirror for the second half [1,2]
    h_left = h_interp
    h_right = h_interp[::-1]
    h_full = np.concatenate([h_left, h_right])
    
    return h_full

def project_onto_feasible(h, target_sum):
    """Project h onto the feasible set: 0 <= h <= 1, sum(h) = target_sum"""
    # Water-filling algorithm
    h = np.asarray(h).copy()
    h = np.clip(h, 0, 1)
    
    def waterfill(lambda_val):
        # return sum(max(0, h[i] - lambda_val) for i in range(len(h))) - target_sum
        p = np.clip(h - lambda_val, 0, 1)
        return np.sum(p) - target_sum
    
    # Let's find lambda using binary search
    lambda_low = -1.0
    lambda_high = 1.0
    # Approximate bounds based on initial sum
    initial_sum = np.sum(h)
    if initial_sum < target_sum:
        lambda_high = 0.0
        # try to increase lambda while sum < target_sum
    else:
        lambda_low = 0.0
    
    # Binary search to find correct lambda
    for _ in range(50):  # max iterations
        lambda_mid = (lambda_low + lambda_high) / 2.0
        val = waterfill(lambda_mid)
        if val > 0:
            lambda_low = lambda_mid
        else:
            lambda_high = lambda_mid
    lambda_opt = (lambda_low + lambda_high) / 2.0
    projected = np.clip(h - lambda_opt, 0, 1)
    scale_factor = target_sum / np.sum(projected)
    return projected * scale_factor

def smooth_max_over_shifts(h, beta=10.0):
    """Compute a smooth approximation of the maximum correlation"""
    dx = 2.0 / len(h)
    corr_full = correlate(h, 1 - h, mode="full")
    # Possible k: -(n-1) to (n-1)
    shifts = np.arange(-len(h)+1, len(h))
    f_vals = corr_full * dx
    max_f = np.max(f_vals)
    exp_vals = np.exp(beta * (f_vals - max_f))
    Z = np.sum(exp_vals)
    # The Hessian is not needed - just the gradient
    exp_w = exp_vals / Z
    # Gradient w.r.t. h (this is an approximation)
    grad = np.zeros_like(h)
    for k in range(len(corr_full)):
        if k >= len(h):
            continue
        shift = k - len(h) + 1
        # Compute gradient contribution from f_k = sum_i h_i * (1 - h_{i+k})
        for i in range(len(h)):
            i_shift = i + shift
            if i_shift >= 0 and i_shift < len(h):
                grad[i] += exp_w[k] * dx * (1 - h[i_shift])
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 200  # Moderately fine discretization
    
    # Parameterise with 5 breakpoints (K=4)
    K = 5
    initial_guess = np.array([1.0] * K + [0.0])  # Start with uniform then adjust
    
    def objective(params):
        raw_heights = params.copy()
        h_full = build_symmetric_piecewise_linear(raw_heights, n_points)
        h_clipped = np.clip(h_full, 0, 1)
        # Ensure sum == n_points/2 (i.e., 100)
        h_scaled = project_onto_feasible(h_clipped, n_points / 2)
        return evaluate_c5(h_scaled)
    
    bounds = [(0, 1) for _ in range(K)]
    
    # Phase 1: Differential Evolution (about 70% of time)
    t_de = 0.7 * budget_s
    res_de = differential_evolution(
        objective,
        bounds,
        seed=seed,
        maxiter=int(1000 * (t_de/600)),
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.7,
        strategy='best1bin'
    )
    
    if not res_de.success:
        # Fallback
        best_params = initial_guess
    else:
        best_params = res_de.x
    
    # Build from best parameters
    h_full = build_symmetric_piecewise_linear(best_params, n_points)
    h_clipped = np.clip(h_full, 0, 1)
    h_opt = project_onto_feasible(h_clipped, n_points / 2)
    current_c5 = evaluate_c5(h_opt)
    
    # Phase 2: Validate smooth surrogate + projected gradient (20% of time)
    t_pgd = 0.2 * budget_s
    beta = 10.0
    
    max_iter_pgd = 1000
    step_size = 0.05
    # Start from current solution
    h_current = h_opt.copy()
    
    for i in range(max_iter_pgd):
        if time.time() - start_time > start_time + t_pgd:
            break
        # Compute gradients via smooth surrogate
        grad = smooth_max_over_shifts(h_current, beta=beta)
        # Gradient ascent step (since we minimized C5, we want to minimize it)
        # But to be consistent with code below, we assume it is descent direction
        # so we move in the opposite of gradient direction to reduce the loss
        # i.e. C5 = max(-overlaps) (so we want to reduce max overlaps)
        # Optimization faces: we negate the overlap Grad to achieve this in a way consistent with the code
        # delta = step_size * grad
        h_new = h_current - step_size * grad
        # Project back
        h_new = project_onto_feasible(h_new, n_points / 2)
        # Evaluate new value
        c5_new = evaluate_c5(h_new)
        # Update if better
        if c5_new < current_c5:
            h_current = h_new.copy()
            current_c5 = c5_new
        # Adjust beta if necessary
        if beta < 20 and i > 500:
            beta = 20
    
    # Phase 3: Final polishing (10% of time)
    t_final = 0.1 * budget_s
    if time.time() - start_time < start_time + t_final:
        # Run short SLSQP optimization with best from PGD
        h_init = h_current.copy()
        def obj_slsqp(h):
            return evaluate_c5(h)
        bounds_slsqp = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) * (2.0 / n_points) - 1.0}
        res_slsqp = minimize(
            obj_slsqp,
            h_init,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10},
            tol=1e-10
        )
        if res_slsqp.success:
            h_final = res_slsqp.x
            c5_final = evaluate_c5(h_final)
            if c5_final < current_c5:
                h_current = h_final
                current_c5 = c5_final
    
    # Final evaluation
    return (h_current.tolist(), current_c5, n_points)
```
