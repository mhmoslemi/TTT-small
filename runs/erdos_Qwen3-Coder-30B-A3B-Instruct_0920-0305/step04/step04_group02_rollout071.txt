import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_binary_milp(n):
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(x) == n // 2]
    
    # iterate shifts
    for d in range(-n+1, n):
        if d == 0:
            continue
        w_vars = []
        if d > 0:
            rng = range(n - d)
        else:
            rng = range(-d, n)
        for i in rng:
            w = cp.Variable(boolean=True)
            # link
            constraints.append(w <= x[i])
            constraints.append(w <= x[i+d])
            constraints.append(w >= x[i] + x[i+d] - 1)
            w_vars.append(w)
        # N_d = sum(x) - sum(w_vars)
        N_d = cp.sum(x) - cp.sum(w_vars)
        constraints.append(N_d <= t)
    constraints.append(t >= 0)

    prob = cp.Problem(cp.Minimize(t), constraints)
    # attempt solvers
    for solver in [cp.CBC, cp.GLPK_MI, cp.ECOS_BB]:
        try:
            prob.solve(solver=solver, verbose=False, warm_start=True, max_iters=5000)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                break
        except Exception as e:
            continue
            
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        # Fallback to continuous relaxation if MILP fails
        x = cp.Variable(n)
        t = cp.Variable()
        constraints = [cp.sum(x) == n // 2, x >= 0, x <= 1]
        
        for d in range(-n+1, n):
            if d == 0:
                continue
            w_vars = []
            if d > 0:
                rng = range(n - d)
            else:
                rng = range(-d, n)
            for i in rng:
                w = cp.Variable()
                constraints.append(w <= x[i])
                constraints.append(w <= x[i+d])
                constraints.append(w >= x[i] + x[i+d] - 1)
                constraints.append(w >= 0)
                w_vars.append(w)
            N_d = cp.sum(x) - cp.sum(w_vars)
            constraints.append(N_d <= t)
        constraints.append(t >= 0)
        prob = cp.Problem(cp.Minimize(t), constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        # Convert to binary by thresholding
        x_vals = np.round(x.value).astype(float)
        t_val = t.value
    else:
        x_vals = np.round(x.value).astype(float)
        t_val = int(round(t.value))
    
    return x_vals, t_val

        dx = 2.0 / n_points

        new_h = new_h * scale
        
        try:
            res = minimize(
                evaluate_c5, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-12}
            )

                if c5_candidate < best_c5 - 1e-8:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate

            pass

        if (time.time() - start_time) > budget_s * 0.95:
