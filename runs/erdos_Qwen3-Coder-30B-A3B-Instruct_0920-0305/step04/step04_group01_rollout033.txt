import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

# Global variables
TARGET_SUM = None
n_points = 120 # this is the discretization parameter

def compute_corr_fft(h):
    """Compute full correlation using FFT."""
    n = len(h)
    # Compute b = 1 - h
    b = 1 - h
    # Reverse h
    h_rev = h[::-1]
    corr = np.real(ifft(fft(h_rev, 2*n - 1) * fft(b, 2*n - 1)))
    return corr[:2*n - 1]

def c5_from_corr(corr, n):
    """Convert correlation array to C5 value."""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x to satisfy sum(x) = target_sum with box constraints [0,1]."""
    def f(lam):
        proj = np.clip(x + lam, 0, 1)
        return np.sum(proj) - target_sum

    a, b = -1.0, 1.0
    for _ in range(50):
        lam = (a + b) / 2
        if f(lam) > 0:
            b = lam
        else:
            a = lam

    proj = np.clip(x + lam, 0, 1)
    return proj

def random_pair_move(h, rng):
    """Generate a random mass transfer move."""
    i, j = rng.choice(n_points, 2, replace=False)
    h_i, h_j = h[i], h[j]
    delta_min = max(-h_j, h_i - 1)
    delta_max = min(h_i, 1 - h_j)
    if delta_min < delta_max:
        delta = rng.uniform(delta_min, delta_max)
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        return h_new
    return h

def guided_move(h, worst_lag):
    """Perform a guided move that minimizes the worst correlation."""
    n = len(h)
    if worst_lag == 0:
        # No improvement from this move
        return h
    grad = np.zeros(n)
    # Recreate the equivalent partial derivative of correlation w.r.t. h_i
    # for lag = worst_lag
    if worst_lag > 0:
        grad[worst_lag:] += h[:-worst_lag]
        grad[:worst_lag] += h[-worst_lag:]
    else:
        # worst_lag < 0
        grad[:worst_lag] += h[-worst_lag:]
        grad[worst_lag:] += h[:-worst_lag]
    # minimize(grad) is a good starting point for the antisymmetric worst-case sum
    i_min = np.argmin(grad)
    i_max = np.argmax(grad)
    if i_min == i_max or grad[i_min] >= grad[i_max]:
        # No meaningful improvement
        return h
    # Move from min to max
    h_new = h.copy()
    delta = min(h[i_min], 1 - h[i_max])
    if delta > 0:
        h_new[i_min] -= delta
        h_new[i_max] += delta
    return h_new

def anneal(h0, run_seconds, seed):
    """Perform simulated annealing."""
    rng = np.random.default_rng(seed)
    h_curr = h0.copy()
    c5_curr = evaluate_c5(h_curr)
    h_best = h_curr.copy()
    c5_best = c5_curr
    t_start = time.time()
    T0 = 5e-3
    Tf = 1e-6
    t_max = run_seconds
    iter_since_improve = 0
    max_wait = 1000 # don't wait too long
    while time.time() - t_start < t_max:
        t = time.time() - t_start
        T = T0 * (1 - t / t_max) + Tf * (t / t_max)
        # Evaluate current worst lag
        corr = compute_corr_fft(h_curr)
        worst_lag = np.argmax(corr) - (n_points - 1)
        # determine change
        if iter_since_improve < 50:
            h_trial = random_pair_move(h_curr, rng)
        else:
            h_trial = guided_move(h_curr, worst_lag)
        # Project to feasible set
        h_trial = project_to_sum_box(h_trial, TARGET_SUM)
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < c5_curr or rng.uniform() < np.exp(-(c5_trial - c5_curr) / T):
            h_curr = h_trial
            c5_curr = c5_trial
            if c5_curr < c5_best:
                c5_best = c5_curr
                h_best = h_curr.copy()
                iter_since_improve = 0
            else:
                iter_since_improve += 1
        else:
            iter_since_improve += 1
        if iter_since_improve > max_wait:
            break
    return h_best, c5_best

def constraint_eq(h):
    return np.sum(h) - TARGET_SUM

def objective(h):
    return evaluate_c5(h)

def local_refine(h):
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    res = minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12}
    )
    if res.success:
        h_refined = res.x
        h_refined = project_to_sum_box(h_refined, TARGET_SUM)
        return h_refined, evaluate_c5(h_refined)
    return h, evaluate_c5(h)

    global TARGET_SUM, n_points
    TARGET_SUM = n_points / 2
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = np.inf
    n_points = 120
    dx = 2.0 / n_points
    initial_h_values = np.array(initial_h_values)
    if len(initial_h_values) < n_points:
        # interpolate initial_h_values to n_points
        x_orig = np.linspace(0, 2, len(initial_h_values))
        x_new = np.linspace(0, 2, n_points)
        h_interpolated = np.interp(x_new, x_orig, initial_h_values)
        h0 = project_to_sum_box(h_interpolated, TARGET_SUM)
    else:
        h0 = project_to_sum_box(initial_h_values[:n_points], TARGET_SUM)
    # restart strategy
    restart_count = 0
    time_left = lambda: budget_s - (time.time() - start_time)
    while time_left() > 10:
        sub_seed = seed + restart_count
        if restart_count % 5 == 0:
            # binary pattern
            h0 = np.zeros(n_points)
            indices = np.random.choice(n_points, int(n_points / 2), replace=False)
            h0[indices] = 1.0
            h0 = project_to_sum_box(h0, TARGET_SUM)
        elif restart_count % 5 == 1:
            # uniform random
            h0 = np.random.uniform(0, 1, n_points)
            h0 = project_to_sum_box(h0, TARGET_SUM)
        else:
            # interpolated and projected
            x_orig = np.linspace(0, 2, len(initial_h_values))
            x_new = np.linspace(0, 2, n_points)
            h_interpolated = np.interp(x_new, x_orig, initial_h_values)
            h0 = project_to_sum_box(h_interpolated, TARGET_SUM)
        h_new, c5_new = anneal(h0, min(5, time_left() - 10), sub_seed)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        restart_count += 1
        if time_left() < 10:
            break
    # local refinement phase
    t_remaining = time_left()
    if t_remaining > 0:
        refined_h, refined_c5 = local_refine(best_h)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h
        # polish mode
        attempts = 0
        polish_time = min(0.2 * budget_s, 5)
        while attempts < 10 and time_left() > 1:
            h_perturb = best_h + np.random.normal(0, 0.02, n_points)
            h_perturb = project_to_sum_box(h_perturb, TARGET_SUM)
            h_refine_p, c5_refine_p = local_refine(h_perturb)
            if c5_refine_p < best_c5:
                best_c5 = c5_refine_p
                best_h = h_refine_p
            attempts += 1

    return (best_h.tolist(), float(best_c5), n_points)
