import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Phase 1: Try to solve exact MILP formulation
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    remaining_time = budget_s - (time.time() - start_time)
    
    # MILP formulation
    if remaining_time > 10:
        try:
            # Create CVXPY problem
            h = Variable(n_points, boolean=True)
            p = Variable((n_points, n_points), boolean=True)
            M = Variable(integer=True, nonneg=True)
            
            # Constraints
            # Integral constraint
            constraints = [sum(h) == n_points // 2]
            
            # Symmetry breaking
            constraints.append(h[0] == 1)
            
            # Logical AND constraints
            for l in range(1, n_points):
                for i in range(n_points - l):
                    # p[i,l] <= h[i]
                    constraints.append(p[i,l] <= h[i])
                    # p[i,l] <= h[i+l]
                    constraints.append(p[i,l] <= h[i+l])
                    # p[i,l] >= h[i] + h[i+l] - 1
                    constraints.append(p[i,l] >= h[i] + h[i+l] - 1)
            
            # Worst-case overlap constraints
            for l in range(1, n_points):
                # A_l^+ - S_l <= M
                A_l_plus = sum(h[i] for i in range(n_points - l))
                S_l = sum(p[i,l] for i in range(n_points - l))
                constraints.append(A_l_plus - S_l <= M)
                
                # A_l^- - S_l <= M
                A_l_minus = sum(h[i] for i in range(l, n_points))
                constraints.append(A_l_minus - S_l <= M)
            
            # Objective
            objective = Minimize(M)
            prob = Problem(objective, constraints)
            
            # Solve with reasonable time limit
            prob.solve(solver=CPLEX, verbose=False, time_limit=remaining_time/2)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_val = np.array([float(h[i].value) for i in range(n_points)])
                # Ensure integral constraint is satisfied
                dx = 2.0 / n_points
                scale = 1.0 / (np.sum(h_val) * dx)
                h_val = h_val * scale
                c5_val = evaluate_c5(h_val)
                if c5_val < best_c5:
                    best_h = h_val.copy()
                    best_c5 = c5_val
            
        except Exception as e:
            # Fall through to heuristic if MILP fails
            pass
    
    # Phase 2: Continuous polishing if we have some solution
    if (time.time() - start_time) < budget_s - 1:
        # Convert to continuous problem for polishing
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            dx = 2.0 / n_points
            return np.sum(h) * dx - 1.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        # Multiple restarts with different starting points
        restarts = 100
        for restart in range(restarts):
            if (time.time() - start_time) > budget_s - 2:
                break
                
            # Try different starting points
            temp_h = best_h.copy()
            randomness = np.random.normal(0, 0.05, n_points)
            temp_h = temp_h + randomness
            temp_h = np.clip(temp_h, 0, 1)
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(temp_h) * dx)
            temp_h = temp_h * scale_factor
            
            try:
                res = minimize(
                    objective,
                    temp_h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 50, 'ftol': 1e-12, 'gtol': 1e-12}
                )
                if res.success:
                    h_candidate = res.x
                    scale = 1.0 / (np.sum(h_candidate) * dx)
                    h_candidate = h_candidate * scale
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
            except:
                continue
    
    # Phase 3: Final fallback to simulated annealing if needed
    if (time.time() - start_time) < budget_s - 1:
        def acceptance_probability(delta, temperature):
            if delta < 0:
                return 1.0
            return np.exp(-delta / temperature)
        
        def get_neighbor(h):
            h_new = h.copy()
            ones = np.where(h_new > 0.5)[0]
            zeros = np.where(h_new < 0.5)[0]
            if len(ones) > 0 and len(zeros) > 0:
                # Pick random 1 and 0 to swap
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                h_new[i] = 0
                h_new[j] = 1
            return h_new
        
        def compute_overlap(h, dx):
            corr = correlate(h, 1 - h, mode="full")
            return np.max(corr) * dx
        
        # Start with existing best
        current_h = best_h.copy()
        dx = 2.0 / n_points
        current_c5 = compute_overlap(current_h, dx)
        best_c5 = current_c5
        best_h = current_h.copy()
        
        # Simulated annealing
        temperature = 1.0
        cooling_rate = 0.999
        min_temperature = 1e-10
        
        # Easy initial condition: binary values for correctness check
        binary_h = (best_h > 0.5).astype(float)
        dx = 2.0 / n_points
        scale = 1.0 / (np.sum(binary_h) * dx)
        binary_h = binary_h * scale
        current_c5 = evaluate_c5(binary_h)
        if current_c5 < best_c5:
            best_h = binary_h.copy()
            best_c5 = current_c5
            
        # Now do simulated annealing on binary states with slight perturbation
        iterations = 0
        max_iterations = 50000
        
        while temperature > min_temperature and (time.time() - start_time) < budget_s - 1:
            if iterations >= max_iterations:
                break
                
            # Get neighbor by swapping two points
            neighbor_h = get_neighbor(best_h)
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(neighbor_h) * dx)
            neighbor_h = neighbor_h * scale
            neighbor_c5 = evaluate_c5(neighbor_h)
            
            delta = neighbor_c5 - best_c5
            if acceptance_probability(delta, temperature) > np.random.random():
                best_h = neighbor_h.copy()
                best_c5 = neighbor_c5
            
            temperature *= cooling_rate
            iterations += 1
    
    return (best_h.tolist(), best_c5, n_points)
