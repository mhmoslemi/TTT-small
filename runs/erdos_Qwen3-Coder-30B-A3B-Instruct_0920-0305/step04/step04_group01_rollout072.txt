import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import minimize
import time

# Global variables for constraint handling
TARGET_SUM = 60.0  # n / 2 for n = 120

def compute_corr_fft(h):
    """Fast computation of full correlation using FFT"""
    n = len(h)
    h_rev = h[::-1]
    b = 1 - h
    # Use rfft with appropriate length
    corr_full = fftconvolve(h_rev, b, mode='full')
    return corr_full

def c5_from_corr(corr, n):
    """Convert correlation array to C5 value"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast C5 evaluation using FFT"""
    n = len(h)
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n)

def project_to_sum_box(x, target_sum):
    """Project x onto sum = target_sum with box constraints [0,1]"""
    # Convert to numpy array if needed
    x = np.array(x)
    x = np.clip(x, 0, 1)
    total = np.sum(x)
    diff = target_sum - total
    
    if abs(diff) < 1e-12:
        return x
    
    # Binary search for lambda to make sum equal target_sum
    # f(lambda) = sum(clip(x + lambda, 0, 1)) - target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    
    for _ in range(100):  # Safe loop
        mid = (low + high) / 2
        projected = np.clip(x + mid, 0, 1)
        val = np.sum(projected) - target_sum
        if val > 0:
            high = mid
        else:
            low = mid
        if abs(val) < tol:
            break
    
    return np.clip(x + low, 0, 1)

def random_pair_move(h):
    """Generate a random feasible pairwise mass transfer move"""
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    delta_min = max(-h[j], h[i] - 1)
    delta_max = min(h[i], 1 - h[j])
    if delta_min < delta_max:
        delta = np.random.uniform(delta_min, delta_max)
        return i, j, delta
    else:
        return i, j, 0.0  # Return no change if invalid

def guided_move(h, worst_lag):
    """ guided mass transfer assuming convolution representation is fast """
    # Compute gradient-like subgradient (indices i, i-lag, i+lag)
    n = len(h)
    grad = np.zeros_like(h)
    
    # left shift (i -> i-lag) part
    if worst_lag > 0:
        grad[worst_lag:] += h[:-worst_lag]
    elif worst_lag < 0:
        grad[:worst_lag] += h[-worst_lag:]
    else:
        grad += h
    
    # right shift (i -> i+lag) part  
    if worst_lag < 0:
        grad[-worst_lag:] += h[:-worst_lag]
    elif worst_lag > 0:
        grad[:worst_lag] += h[worst_lag:]
    else:
        grad += h

    # Now transfer from minimum to maximum
    i_min = np.argmin(grad)
    i_max = np.argmax(grad)
    
    if i_min == i_max:
        return h.copy()
        
    delta_min = max(-h[i_max], h[i_min] - 1)
    delta_max = min(h[i_min], 1 - h[i_max])
    if delta_min >= delta_max:
        return h.copy()
    
    delta = np.clip(np.random.uniform(delta_min, delta_max), delta_min, delta_max)
    
    h_new = h.copy()
    h_new[i_min] -= delta
    h_new[i_max] += delta
    
    return project_to_sum_box(h_new, TARGET_SUM)

def anneal(h0, run_seconds, rng):
    """Single anneal run with both guided and random moves"""
    h = h0.copy()
    c5_current = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5_current
    
    T0 = 5e-3
    Tf = 1e-6
    t0 = time.time()
    
    step = 0
    while time.time() - t0 < run_seconds:
        # Alternate between guided and random moves
        if step % 10 == 0:  # Guided step
            corr = compute_corr_fft(h)
            worst_lag = np.argmax(corr)
            h_next = guided_move(h, worst_lag)
        else:  # Random step
            i, j, delta = random_pair_move(h)
            h_next = h.copy()
            if delta != 0:
                h_next[i] -= delta
                h_next[j] += delta
                h_next = project_to_sum_box(h_next, TARGET_SUM)
        
        c5_next = evaluate_c5(h_next)
        
        # Accept
        if c5_next <= c5_current:
            h = h_next
            c5_current = c5_next
        else:
            # Accept with probability
            T = T0 * (1 - (time.time() - t0) / run_seconds) + Tf * (time.time() - t0) / run_seconds
            prob = np.exp(-(c5_next - c5_current) / T)
            if np.random.rand() < prob:
                h = h_next
                c5_current = c5_next
                
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
        
        step += 1
        
    return best_h, best_c5

def local_refine(h0):
    """Local refinement with SLSQP"""
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    bounds = [(0, 1) for _ in range(len(h0))]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    res = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12})
    
    if res.success:
        return res.x.copy()
    else:
        # if it fails, return the original as fallback
        return h0.copy()
