import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_binary_milp(h0, n):
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    y = {}
    constraints = []

    # Integral constraint
    constraints.append(cp_sum(x) == n // 2)

    # Product constraints y[i,d] = x[i] * x[i+d]
    for d in range(-n+1, n):
        if d == 0: continue
        for i in range(n):
            if i+d >= 0 and i+d < n:
                y[(i,d)] = cp.Variable(boolean=True)
                constraints.append(y[(i,d)] <= x[i])
                constraints.append(y[(i,d)] <= x[i+d])
                constraints.append(y[(i,d)] >= x[i] + x[i+d] - 1)

    # Upper bound constraint for each shift
    for d in range(-n+1, n):
        if d == 0: continue
        sum_y = cp_sum([y[(i,d)] for i in range(n) if i+d >= 0 and i+d < n])
        constraints.append(n//2 - sum_y <= t)

    # Objective
    objective = cp.Minimize(t)

    # Problem
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.CBC)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = x.value.astype(float)
            t_opt = t.value
            return h_opt, float(t_opt)
        else:
            return h0, evaluate_c5(h0)
    except:
        return h0, evaluate_c5(h0)

def smooth_max_objective(h, shifts, tau=0.1):
    # Compute overlaps for each shift
    overlaps = []
    dx = 2.0 / len(h)
    for d in shifts:
        if d > 0:
            left = h[:-d]
            right = h[d:]
        else:
            left = h[-d:]
            right = h[:d]
        overlap = np.sum(left * (1 - right)) * dx
        overlaps.append(overlap)
    # log-sum-exp smooth approximation
    overlaps = np.array(overlaps)
    return tau * np.log(np.sum(np.exp(overlaps / tau)))

def smooth_max_gradient(h, shifts, tau=0.1):
    dx = 2.0 / len(h)
    overlaps = []
    for d in shifts:
        if d > 0:
            left = h[:-d]
            right = h[d:]
        else:
            left = h[-d:]
            right = h[:d]
        overlap = np.sum(left * (1 - right)) * dx
        overlaps.append(overlap)
    overlaps = np.array(overlaps)
    exp_overlaps = np.exp(overlaps / tau)
    weights = exp_overlaps / np.sum(exp_overlaps)
    
    grad = np.zeros_like(h)
    for i, d in enumerate(shifts):
        if d > 0:
            left = h[:-d]
            right = h[d:]
            grad[:-d] += weights[i] * (1 - right)
            grad[d:] -= weights[i] * left
        else:
            left = h[-d:]
            right = h[:d]
            grad[-d:] += weights[i] * (1 - right)
            grad[:d] -= weights[i] * left
    return grad

def project_simplex(h, target_sum=1.0):
    # Project onto simplex using the softmax trick
    n = len(h)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    
    # Scale to target sum
    h = h * target_sum / current_sum
    return h
