import numpy as np

from scipy.signal import correlate
import time

# Global variables for constraint and objective
TARGET_SUM = 60.0  # n/2 for n=120

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Build b = 1 - h
    b = 1 - h
    # Perform convolution with FFT
    corr = np.convolve(h_rev, b, mode='full')
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation array"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5"""
    n = len(h)
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n)

def project_to_sum_box(x, target_sum):
    """Project vector x to the sum constraint and box constraints [0,1]"""
    x = np.array(x)
    n = len(x)
    
    # Find lambda such that sum(x + lambda) = target_sum
    # f(lambda) = sum(x + lambda) - target_sum = sum(x) + lambda*n - target_sum
    # lambda = (target_sum - sum(x)) / n
    
    lambda_val = (target_sum - np.sum(x)) / n
    
    # Apply to each component
    projected = x + lambda_val
    
    # Clip to [0,1]
    projected = np.clip(projected, 0, 1)
    
    return projected

def random_pair_move(h):
    """Perform a random feasible pairwise mass transfer"""
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    
    h_i = h[i]
    h_j = h[j]
    
    # Compute feasible delta range
    delta_min = max(-h_j, h_i - 1)
    delta_max = min(h_i, 1 - h_j)
    
    if delta_min < delta_max:
        delta = np.random.uniform(delta_min, delta_max)
        return [i, j, delta]
    else:
        return None

def guided_move(h, corr):
    """Perform a guided move along the worst direction"""
    # Determine worst lag (index in correlation that gives max)
    n = len(h)
    # corr has length 2*n-1
    max_idx = np.argmax(corr)
    # correlation index -> lag: -n+1 to n-1
    lag = max_idx - (n - 1)  # lag = max_idx - (n-1)
    
    # Compute subgradient for this lag
    grad = np.zeros(n)
    
    # grad[i] = h[i+lag] + h[i-lag] where out-of-range = 0
    for i in range(n):
        term1 = h[i+lag] if 0 <= i+lag < n else 0
        term2 = h[i-lag] if 0 <= i-lag < n else 0
        grad[i] = term1 + term2
    
    # Find indices with min and max gradient
    min_idx = np.argmin(grad)
    max_idx = np.argmax(grad)
    
    # Transfer from min to max
    h_i = h[min_idx]
    h_j = h[max_idx]
    
    delta_min = max(-h_j, h_i - 1)
    delta_max = min(h_i, 1 - h_j)
    
    if delta_min < delta_max:
        delta = (delta_min + delta_max) * 0.5
        return [min_idx, max_idx, delta]
    else:
        return None

def anneal(h0, run_seconds, rng):
    """Run simulated annealing for a fixed time"""
    # Initialize
    h = h0.copy()
    h = project_to_sum_box(h, TARGET_SUM)  # Ensure constraint
    
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    t0 = time.time()
    
    # Schedule
    T0 = 0.005
    Tf = 1e-6
    
    step_size = 0.02 / np.sqrt(2)
    step_count = 0
    guided_count = 0
    
    while time.time() - t0 < run_seconds:
        # Compute full correlation
        corr = compute_corr_fft(h)
        # Compute current max index
        max_idx = np.argmax(corr)
        
        # Decide on move type
        move_type = "guided" if (guided_count % 10 == 0) else "random"
        
        if move_type == "guided":
            # Perform guided move
            move = guided_move(h, corr)
            if move is not None:
                i, j, delta = move
                h[i] -= delta
                h[j] += delta
                h = project_to_sum_box(h, TARGET_SUM)
        else:
            # Perform random pair move
            move = random_pair_move(h)
            if move is not None:
                i, j, delta = move
                h[i] -= delta
                h[j] += delta
                h = project_to_sum_box(h, TARGET_SUM)
        
        # Evaluate
        new_c5 = evaluate_c5(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
        
        step_count += 1
        guided_count += 1
    
    return best_h, best_c5

def local_refine(h):
    """Refine with a bound-constrained optimizer"""
    def objective(h):
        return evaluate_c5(h)
    
    bounds = [(0, 1) for _ in range(len(h))]
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    res = minimize(
        objective, 
        h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons,
        options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12}
    )
    
    if res.success:
        return res.x
    else:
        return h
