import numpy as np

def smooth_max_corr(h, beta=15.0):
    """Smooth approximation to max of correlation values"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlation values
    corr = correlate(h, 1 - h, mode="full")
    # Take smooth maximum
    corr_scaled = beta * corr
    max_corr = np.log(np.sum(np.exp(corr_scaled))) / beta
    return max_corr * dx

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def decode_params(params, n=200):
    """Convert shape parameters to full h vector"""
    # params is a list of K heights
    K = len(params)
    # crude interpolation to make a symmetric shape with n points
    x = np.linspace(0, 1, n//2 + 1)
    heights = np.array(params)
    # Ensure monotonic decreasing
    heights = np.sort(heights)[::-1]
    # Build left half increasing to max
    y = np.interp(x, np.linspace(0, 1, K+1), np.concatenate([[1], heights]))
    # Mirror to right half
    h = np.concatenate([y, y[-2::-1]])
    # Normalise
    area = np.sum(h) * 2.0 / n
    h = h / area * 1.0  # ensure integral is 1
    h = np.clip(h, 0, 1)
    return h

def objective(params):
    h = decode_params(params)
    return evaluate_c5(h)
