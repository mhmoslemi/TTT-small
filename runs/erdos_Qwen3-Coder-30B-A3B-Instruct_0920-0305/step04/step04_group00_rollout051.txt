from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def triangular_shape(x, a):
    """Triangular shape parameterized by a (half-width)"""
    return np.maximum(0, 1 - np.abs(x - 1) / a)

def build_h_from_params(params, n_points):
    """Build h from parameters: [a] where a is half-width"""

    # Normalize to integral = 1
    integral = np.sum(h) * (2.0 / n_points)
    h = h / integral * 1.0
    h = np.clip(h, 0, 1)
    return h
