import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use a hybrid optimization approach
    # Phase 1: Use global optimization
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Use differential evolution with a different strategy
    try:
        de_res = differential_evolution(
            objective, 
            bounds, 
            seed=seed, 
            maxiter=500,  # Reduced maxiter for faster iteration
            popsize=10,    
            mutation=(0.8, 1),  
            recombination=0.9,   
            strategy='best1bin'
        )
        if de_res.success:
            h_candidate = de_res.x
            # Normalize to satisfy constraint
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass
    
    # Phase 2: Local search with improved method
    # Use L-BFGS-B with bounds and constraint
    if (time.time() - start_time) < budget_s * 0.8:
        try:
            res = minimize(
                objective, 
                best_h, 
                method='L-BFGS-B', 
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}  # More precise tolerances
            )
            if res.success:
                h_candidate = res.x
                dx = 2.0 / n_points
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass

    # Phase 3: Add structure using a pattern based on golden ratio
    if (time.time() - start_time) < budget_s * 0.9:
        try:
            # Golden ratio based pattern
            golden_ratio = (1 + np.sqrt(5)) / 2
            x = np.arange(n_points)
            # Compute fractional parts for golden ratio
            positions = (x * golden_ratio) % 1
            # Create pattern based on golden ratio positions
            golden_h = (positions < 0.5).astype(float)
            # Scale to match integral
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(golden_h) * dx)
            golden_h = golden_h * scale_factor
            
            # Try different combinations  
            mixing_ratios = [0.8, 0.9]
            for ratio in mixing_ratios:
                mixed_h = ratio * best_h + (1 - ratio) * golden_h
                # Normalize
                scale_factor = 1.0 / (np.sum(mixed_h) * dx)
                mixed_h = mixed_h * scale_factor
                
                # Local optimization
                try:
                    res = minimize(
                        objective, 
                        mixed_h, 
                        method='L-BFGS-B', 
                        bounds=bounds, 
                        constraints=cons,
                        options={'maxiter': 150, 'ftol': 1e-13, 'gtol': 1e-13}
                    )
                    if res.success:
                        h_candidate = res.x
                        scale = 1.0 / (np.sum(h_candidate) * dx)
                        h_candidate = h_candidate * scale
                        c5_candidate = objective(h_candidate)
                        if c5_candidate < best_c5:
                            best_h = h_candidate.copy()
                            best_c5 = c5_candidate
                except:
                    continue
        except:
            pass

    # Phase 4: Final refined optimization
    if (time.time() - start_time) < budget_s * 0.95:
        try:
            # Try multiple starting points and local optimization
            for i in range(10):
                if (time.time() - start_time) > budget_s:
                    break
                # Random perturbation around best solution
                new_h = best_h + np.random.normal(0, 0.03, n_points)
                new_h = np.clip(new_h, 0, 1)
                dx = 2.0 / n_points
                scale_factor = 1.0 / (np.sum(new_h) * dx)
                new_h = new_h * scale_factor
                
                res = minimize(
                    objective, 
                    new_h, 
                    method='L-BFGS-B', 
                    bounds=bounds, 
                    constraints=cons,
                    options={'maxiter': 200, 'ftol': 1e-13, 'gtol': 1e-13}
                )
                if res.success:
                    h_candidate = res.x
                    scale = 1.0 / (np.sum(h_candidate) * dx)
                    h_candidate = h_candidate * scale
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
        except:
            pass

    return (best_h.tolist(), best_c5, n_points)
