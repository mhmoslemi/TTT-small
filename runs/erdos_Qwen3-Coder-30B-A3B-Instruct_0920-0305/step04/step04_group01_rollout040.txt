def evaluate_c5_fft(h):
    """Evaluate the C5 constant for a given step function h using FFT for speed."""
    n = len(h)
    if n == 0:
        return 0.0
    # Reverse h
    h_rev = h[::-1]
    b = 1.0 - h
    L = 2 * n - 1
    # Compute FFT lengths
    hf = np.fft.rfft(h_rev, n=L)
    bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(hf * bf, n=L)
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)

def compute_corr_fft(h):
    """Compute the full correlation array for h using FFT."""
    n = len(h)
    h_rev = h[::-1]
    b = 1.0 - h
    L = 2 * n - 1
    hf = np.fft.rfft(h_rev, n=L)
    bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(hf * bf, n=L)
    return conv

def project_to_sum_box(x, target_sum):
    """Project vector x to be within [0,1]^n and sum to target_sum."""
    # Clip to [0,1]
    x = np.clip(x, 0, 1)
    # Adjust to reach target sum
    current_sum = np.sum(x)
    diff = target_sum - current_sum

    if abs(diff) < 1e-10:
        return x

    # Simple redistributive adjustment (water-filling approach)
    if diff > 0:
        # We need to increase sum, find indices that can be increased
        free_indices = np.where(x < 1)[0]
        if len(free_indices) == 0:
            return x
        # Distribute diff equally
        delta = diff / len(free_indices)
        step_size = min(delta, 1 - x[free_indices].min())
        if step_size > 0:
            x[free_indices] += step_size
            # If too much, reduce evenly until sum matches
            if np.sum(x) > target_sum:
                remaining = np.sum(x) - target_sum
                x[free_indices] -= remaining / len(free_indices)
    elif diff < 0:
        # We need to decrease sum, find indices that can be decreased
        free_indices = np.where(x > 0)[0]
        if len(free_indices) == 0:
            return x
        # Distribute diff equally
        delta = -diff / len(free_indices)
        step_size = min(delta, x[free_indices].min())
        if step_size > 0:
            x[free_indices] -= step_size
            # If too little, correct
            if np.sum(x) < target_sum:
                remaining = target_sum - np.sum(x)
                x[free_indices] += remaining / len(free_indices)
    
    return x

def simulated_annealing(h0, target_sum, time_limit, seed):
    """Perform simulated annealing to minimize C5."""
    np.random.seed(seed)
    n = len(h0)
    h = project_to_sum_box(h0.copy(), target_sum)
    cur_c = evaluate_c5_fft(h)
    best_h = h.copy()
    best_c = cur_c

    # Set temperature schedule
    T0 = 0.005
    Tmin = 1e-6
    start = time.time()
    iter_count = 0

    while time.time() - start < time_limit:
        # Random move
        i, j = np.random.choice(n, size=2, replace=False)
        h_i, h_j = h[i], h[j]

        # Determine delta range
        delta_max = min(h_i, 1 - h_j)
        delta_min = -min(1 - h_i, h_j)
        
        if delta_max <= delta_min:
            continue
            
        # Choose delta uniformly
        delta = np.random.uniform(delta_min, delta_max)
        
        # Update h
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # evaluate
        c_new = evaluate_c5_fft(h_new)
        
        # Compute temperature
        elapsed = time.time() - start
        progress = elapsed / time_limit
        T = T0 * (1 - progress) + Tmin * progress

        # Acceptance probability
        if c_new < cur_c or np.random.rand() < np.exp(-(c_new - cur_c) / (T + 1e-12)):
            h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        iter_count += 1

    return best_h, best_c
