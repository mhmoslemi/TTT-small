import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_binary_milp(n):
    """Solve binary MILP to find minimal t such that max overlap <= t"""
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    
    constraints = [cp.sum(x) == n // 2]
    
    # For each shift d
    for d in range(-n+1, n):
        if d == 0:
            continue
        w_vars = []
        if d > 0:
            rng = range(n - d)
        else:
            rng = range(-d, n)
        for i in rng:
            w = cp.Variable(boolean=True)
            # Link w to x[i] and x[i+d]
            constraints.append(w <= x[i])
            constraints.append(w <= x[i+d])
            constraints.append(w >= x[i] + x[i+d] - 1)
            w_vars.append(w)
        # N_d = sum(x) - sum(w_vars)
        N_d = cp.sum(x) - cp.sum(w_vars)
        constraints.append(N_d <= t)
    
    constraints.append(t >= 0)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    # Try different solvers
    solvers = [cp.CBC, cp.GLPK_MI, cp.ECOS_BB]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False, warm_start=True, max_iters=5000)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                break
        except:
            continue
    
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        # If no solver works, fall back to solve continuous relaxation to get 0.380952
        # (often good enough for experimental)
        raise RuntimeError("Could not solve binary MILP")
    
    x_vals = np.round(x.value).astype(float)
    t_val = int(round(t.value))
    return x_vals, t_val

        binary_h, t_val = solve_binary_milp(n_points)

        dx = 2.0 / n_points
        c5_binary = t_val * dx
    except Exception as _:
        # If MILP fails, fallback to the initial construction
        binary_h = np.array(initial_h_values)
        n_points = len(binary_h)
        dx = 2.0 / n_points
        c5_binary = evaluate_c5(binary_h)
        t_val = int(round(c5_binary / dx))  # approximate
    
    # Set initial best solution
    best_h = binary_h.copy()
    best_c5 = c5_binary
    
    # Local refinement with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Run a few iterations of local search
    max_restarts = 50  # fewer restarts to stay within budget
    restart_count = 0
    
    while restart_count < max_restarts and (time.time() - start_time) < budget_s * 0.95:
        # Perturb the best solution
        new_h = best_h + np.random.normal(0, 0.02, n_points)
        new_h = np.clip(new_h, 0, 1)
        
        # Project onto constraint
        scale = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale
        new_h = np.clip(new_h, 0, 1)
        
        # Final rescaling
        scale = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale
        
        try:
            res = minimize(
                evaluate_c5,
                new_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_candidate = res.x
                # Normalize to satisfy constraint
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                h_candidate = np.clip(h_candidate, 0, 1)
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5 - 1e-8:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
        
        restart_count += 1
        if (time.time() - start_time) > budget_s * 0.95:
            break

    return (best_h.tolist(), best_c5, n_points)
