import numpy as np
from scipy.signal import convolve
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

# Global constants
TARGET_SUM = 60.0  # n/2 when n = 120
N_POINTS = 120
DX = 2.0 / N_POINTS

def compute_corr_fft(h):
    """Compute the full correlation using FFT."""
    h_rev = h[::-1]
    b = 1 - h
    corr = convolve(h_rev, b, mode='full')
    return corr

def c5_from_corr(corr):
    """Return the C5 constant from the full correlation array."""
    return DX * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT-based convolution."""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto hyperplane sum = target_sum, respecting [0,1] bounds."""
    # Binary search for shift
    a, b = -1.0, 1.0
    while b - a > 1e-10:
        mid = (a + b) / 2.0
        trial = np.clip(x + mid, 0, 1)
        if np.sum(trial) >= target_sum:
            b = mid
        else:
            a = mid
    return np.clip(x + b, 0, 1)

def random_pair_move(h):
    """Generate a feasible pairwise mass transfer move."""
    i, j = np.random.choice(N_POINTS, 2, replace=False)
    h_i, h_j = h[i], h[j]
    delta_min = max(-h_j, h_i - 1)
    delta_max = min(h_i, 1 - h_j)
    if delta_min >= delta_max:
        return None  # No feasible move
    delta = np.random.uniform(delta_min, delta_max)
    return i, j, delta

def guided_move(h, corr):
    """Execute a deterministic move that reduces the correlation with the worst lag."""
    # Find worst lag
    max_corr_idx = np.argmax(corr)
    # Assume corr is of length (2*n - 1) = 239
    # index middle is n-1 = 119
    worst_lag = max_corr_idx - (N_POINTS - 1)
    # Compute gradient
    grad = np.zeros_like(h)
    for i in range(N_POINTS):
        if i - worst_lag >= 0:
            grad[i] += h[i - worst_lag]
        if i + worst_lag < N_POINTS:
            grad[i] += h[i + worst_lag]
    # Find indices of min and max gradients
    min_grad_idx = np.argmin(grad)
    max_grad_idx = np.argmax(grad)
    if grad[min_grad_idx] >= grad[max_grad_idx]:
        return None  # No improvement possible
    delta_min = max(-h[min_grad_idx], h[max_grad_idx] - 1)
    delta_max = min(h[min_grad_idx], 1 - h[max_grad_idx])
    if delta_min >= delta_max:
        return None  # No feasible transfer
    delta = 0.5 * (delta_min + delta_max)
    return min_grad_idx, max_grad_idx, delta

def anneal(h0, run_seconds, rng):
    """Run simulated annealing with guided moves."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    start_t = time.time()
    T0 = 5e-3
    Tf = 1e-6
    # Generate initial random state to fill probabilities
    np.random.seed(rng)
    # Run for a fixed number of seconds
    while time.time() - start_t < run_seconds:
        T = T0 * (1 - (time.time() - start_t) / run_seconds) + Tf * (time.time() - start_t) / run_seconds
        # Try guided move every 10 steps
        if rng % 10 == 0:
            corr = compute_corr_fft(h)
            move = guided_move(h, corr)
            if move is not None:
                i, j, delta = move
                h[i] -= delta
                h[j] += delta
                h = project_to_sum_box(h, TARGET_SUM)
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        else:
            move = random_pair_move(h)
            if move is not None:
                i, j, delta = move
                h[i] -= delta
                h[j] += delta
                h = project_to_sum_box(h, TARGET_SUM)
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                elif np.random.rand() < np.exp(-(c5 - best_c5) / T):
                    # Accept worse solution
                    pass
        rng += 1
    return best_h, best_c5

def local_refine(h0):
    """Call scipy optimizer to refine."""
    bounds = [(0, 1) for _ in range(N_POINTS)]

    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM

    cons = {'type': 'eq', 'fun': constraint_eq}

    def objective(h):
        return evaluate_c5(h)

    res = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 2000, 'ftol': 1e-12})
    if res.success:
        h_refined = res.x
        h_refined = project_to_sum_box(h_refined, TARGET_SUM)
        return h_refined, evaluate_c5(h_refined)
    else:
        return h0, evaluate_c5(h0)
