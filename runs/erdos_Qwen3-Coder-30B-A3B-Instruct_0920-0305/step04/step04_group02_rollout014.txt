import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def compute_subgradient(h, d_star):
    """Compute the subgradient for the overlap at shift d_star"""
    n = len(h)
    g = np.zeros(n)
    if d_star > 0:  # h[i] * (1 - h[i+d_star])
        for i in range(n - d_star):
            g[i] += 1 - h[i + d_star]
        for i in range(d_star, n):
            g[i] += -h[i - d_star]
    elif d_star < 0:  # h[i] * (1 - h[i+d_star])
        for i in range(n + d_star):
            g[i] += 1 - h[i - d_star]
        for i in range(-d_star):
            g[i] += -h[i + (-d_star)]
    else:
        # d_star == 0 case is not used here as we're looking at non-zero shifts
        pass
    return g

def binary_optimal(n):
    """Solve the binary version of the problem using MILP"""
    # Variables
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    
    # Problem data
    dx = 2.0 / n
    
    # Constraints
    # Mass constraint
    constraints = [cp.sum(x) == n / 2]
    
    # Symmetry breaking
    constraints.append(x[0] == 1)
    
    # Product constraints
    # For each valid shift d, we want to link w_{i,d} to x_i and x_{i+d}
    # Will define w_{i,d} for d != 0
    for d in range(-n + 1, n):
        if d == 0:
            continue
        for i in range(n):
            if 0 <= i + d < n:
                # Define auxiliary variable w_{i,d}
                w = cp.Variable(boolean=True)
                # McCormick envelope
                constraints.append(w <= x[i])
                constraints.append(w <= x[i + d])
                constraints.append(w >= x[i] + x[i + d] - 1)
                # At the moment, we won't use w directly,
                # just use overlap count formula:
                # N_d = sum x_i - sum w_{i,d}
    
    # Overlap constraints
    # We compute overlap directly using dummy variables or better yet we'll just 
    # build auxiliary sums manually to satisfy the integer equivalence from [0,1] binary
    # But for efficiency, we can define each sum as a vector on shift d
    # Actually easier to just relax to integer constraint on number of (1,0) overlaps 
    # In this case, we take max_d over all non-zero shifts of (sum h_i - sum h_i h_{i+d})
    # effectively, we can think of auxiliary variables for each pair (i,i+d)  
    # but for rounding errors, just enforce inequalities appropriately  

    # More directly, for each non-zero shift d:
    # Let's manually derive and write constraints by looping over d:
    # So for each d, define: overlap_d = sum_i h_i - sum_i y_{i,d}
    # with constraint y_{i,d} <= h_i, y_{i,d} <= h_{i+d}, y_{i,d} >= h_i + h_{i+d} - 1
    # And have integer-overlap constraint: overlap_d <= t
    # y{i,d} := h_i h_{i+d} integer program for binary x_i
    
    constraints.append(t >= 0)
    
    # Add some overlap constraint based on sum tests for known shifts in BIPROD way
    # Here we directly model it like this:
    # For each non-zero shift d, 
    # consider:
    #   - all indices i such that i+d is also in [0, n-1]
    #   - compute sum of h_i * h_{i+d} over these pairs
    #   - The number of (1,0) pairs = sum(h_i) - sum(h_i * h_{i+d})
    
    # More accurately, we need a few tricks to avoid overcomplicating:
    # Instead, do full equivalent model using boolean product with extra variables:
    # For all pairs (i, i+d):
    # let w_{i,d} = x_i * x_{i+d} if both indices valid
    
    # Basis for solving exactly the minimal max overlap count:
    # - build binary MILP with:
    #   - boolean x_i  
    #   - integer t, bounds each of sum(x_i) - sum_{valid pairs} (h_i h_{i+d}) <= t
    #   - equality sum(x_i) = n/2 
    #   - symmetry breaking
    
    # Approximate reconstruct the model using practical expansion:
    all_shifts = list(range(-n+1, n))
    all_shifts.remove(0)
    
    # Create a symbolic variable for number of (1,0) pairs,
    # use an auxiliary variable sum_{i} (1 - h_{i+d} * h_i) within each shift domain:
    
    # Track all the overlap calculations more directly: 
    max_npairs = 0  # Arbitrary large bound
    
    for d in all_shifts:
        if d == 0:
            continue
        i_range = []
        for i in range(n):
            if 0 <= i + d < n:
                i_range.append(i)
        expr = 0
        for i in i_range:
            expr += x[i] * x[i + d]
        offset = cp.sum(x) - expr
        constraints.append(offset <= t)
        constraints.append(offset >= 0)
    
    # Objective
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        prob.solve(solver=cp.CBC)  # fast, may not be available
        # Check result
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value.astype(float), t.value
        else:
            # fallback to GLPK
            try:
                prob.solve(solver=cp.GLPK_MI)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    return x.value.astype(float), t.value
                else:
                    # Use ECOS_BB
                    prob.solve(solver=cp.ECOS_BB)
                    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                        return x.value.astype(float), t.value
            except:
                return None, None
    except:
        try:
            prob.solve(solver=cp.GLPK_MI)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return x.value.astype(float), t.value
        except:
            try:
                prob.solve(solver=cp.ECOS_BB)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    return x.value.astype(float), t.value
            except:
                pass
    return None, None  # Failed to solve exactly
                
def refine_continuous(h0, remaining_time):
    """Refine the continuous solution"""

    # Cast initial h0 into float and normalize
    h = np.array(h0, dtype=float)
    dx = 2.0 / len(h)
    
    # Force sum constraint
    scal = (len(h) / 2) / np.sum(h)
    h *= scal
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Keep a list of good randomly perturbed starting points
    perturbed_starts = []
    
    # Time bound: use a chunk of time
    for i in range(30):
        if time.time() - start_time > remaining_time * 0.9:
            break
        # Do a random perturbation of the best solution
        h_pert = h + np.random.normal(0, 0.01, len(h))
        h_pert = np.clip(h_pert, 0, 1)
        # Normalization again
        scal = (len(h) / 2) / np.sum(h_pert)
        h_pert *= scal
        
        # Now perform several rounds of projected gradient descent
        # From h_pert:
        h_trial = h_pert.copy()
        for iter in range(200):
            if time.time() - start_time > remaining_time * 0.9:
                break
            
            corr = np.correlate(h_trial, 1 - h_trial, mode="full")
            max_d = np.argmax(corr) - (len(h_trial) - 1)
            
            # Compute gradient for current worst shift
            g = compute_subgradient(h_trial, max_d)
            
            # Find best alpha (step size) by linesearch: try 0.1, 0.05, ..., 1e-6
            alpha = 0.5
            while alpha > 1e-6:
                h_next = h_trial - alpha * g
                h_next = np.clip(h_next, 0, 1)
                scal = (len(h) / 2) / np.sum(h_next)
                h_next *= scal
                new_c5 = evaluate_c5(h_next)
                if new_c5 < best_c5 - 1e-15:
                    h_trial = h_next
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h_next.copy()
                    break
                else:
                    alpha *= 0.5
                    
            # Occasionally jitter
            if np.random.rand() < 0.2:
                h_trial += np.random.normal(0, 0.002, len(h))
                h_trial = np.clip(h_trial, 0, 1)
                scal = (len(h) / 2) / np.sum(h_trial)
                h_trial *= scal
    
    return best_h, best_c5
