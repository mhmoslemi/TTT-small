```python
import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_c5(h):
    """Computes the C5 overlap constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Full correlation with 1-h
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Phase 1: Solve integer program (MILP) to get a binary solution
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    try:
        # Build variables
        x = Variable(n_points, boolean=True)
        t = Variable(1, integer=True)
        
        # Constraints
        constraints = [sum(x) == n_points // 2]
        
        # Compute y_{i,d} = x_i * x_{i+d}
        for d in range(1, n_points):
            for i in range(n_points - d):
                y = Variable(boolean=True)
                constraints.extend([
                    y <= x[i],
                    y <= x[i+d],
                    y >= x[i] + x[i+d] - 1
                ])
                # Shift d contributes to overlap
                # We'll use these explicitly in a later step for computing 
                # the actual value of overlap
                
        # Summarize: For each shift d, N_d = sum(x_i) - sum of y_{i,d} = n/2 - sum of y_{i,d}  
        # We enforce N_d <= t for all d by introducing t and achieving min t directly
        # Total number of such constraints needs to be summed carefully but here let's use a more direct approach
        # (Note: exact redundant constraints not needed here as this is a simplified version)
        # Instead we solve to optimize t directly using a trick.

        # Define an integer problem for the minimal t
        # The binary formulation makes N_d = n/2 - sum of y_{i,d} so we minimize max of all N_d
        # But solve the MILP version only over one variable t by enforcing constraints on each pair
        # A simpler way is to build the problem directly
        
        # Constraint: we don't want to use too complex an approach here  
        # just compute based on the KKT derived from constraints
        # Recap: only relevant for current problem are scalar bounds
        # and that sum_i h_i = n/2
        
        # Let's bypass complex ADMM sim by using direct exact MILP  
        # We define the complete matrix and translate (conceptually).
        # Instead, since this is a small combinatorial problem  
        # (42 variables), use a trick of direct solver calling
        # Assuming solver like CVXPY CBC exists and will work  

        print("Solving binary MILP (n=42)...")
        with Problem(Minimize(t), constraints) as prob:
            result = prob.solve(solver=CBC, verbose=False, max_iters=5000)
        
        if result is not None and prob.status in ["optimal", "optimal_inaccurate"]:
            x_sol = np.array([x[i].value for i in range(n_points)]).round().astype(int)
            h_binary = x_sol
            c5_binary = evaluate_c5(h_binary)
            if c5_binary < best_c5:
                best_h = h_binary.copy()
                best_c5 = c5_binary
                print(f"MILP achieved C5={c5_binary}")
        else:
            print("MILP not solvable; fallback to continuous.")

    except Exception as e:
        print("MILP failed or solver not available:", str(e))

    # If original binary solution is still good, keep it
    if best_c5 >= evaluate_c5(h0):
        best_h = h0.copy()
        best_c5 = evaluate_c5(h0)

    # Phase 2: Continuous refinement with smooth max
    print("Starting smooth-max projected gradient descent...")

    # Reset seed for consistency between two phases
    np.random.seed(seed)

    # Construct a mapping from each relevant shift index to whether it's active in the correlation
    # Build helper function to calculate overlap for a shift
    def compute_shift_overlap(h, shift):
        n = len(h)
        if shift > 0:
            return np.sum(h[:-shift] * (1 - h[shift:]))
        else:
            return np.sum(h[-shift:] * (1 - h[:shift]))

    def smoothed_max_overlaps(h, tau=0.5):
        """
        Compute smooth approximation to max over shifts
        """
        overlaps = []
        n = len(h)
        for d in range(1, n):
            overlap = np.sum(h * (1 - np.roll(h, d)))
            overlaps.append(overlap)
        overlaps = np.array(overlaps)
        return tau * np.log(np.sum(np.exp(overlaps / tau)))

    def gradient_smoothed_max(h, tau=0.5):
        """
        Gradient of the smooth max with respect to h
        """
        n = len(h)
        g = np.zeros_like(h)
        overlaps = []
        dx = 2.0 / n

        for d in range(1, n):
            if d < n:
                # Compute gradient per shift 
                # d: finite difference for h_{i+d}
                # derivative w.r.t. h_i of h_i * (1 - h_{i+d}) is 1 - h_{i+d}
                # and w.r.t. h_{i+d} is -h_i
                grad_i = np.zeros_like(h)
                for i in range(n - d):
                    grad_i[i] += 1 - h[i + d]
                    grad_i[i + d] -= h[i]
                overlaps.append((np.sum(h * (1 - np.roll(h, d))), grad_i))
            else:
                break
        
        if not overlaps:
            return np.zeros_like(h)
        
        # Exponentials and weights  
        weights = []
        total_exp = 0.0
        for val, _ in overlaps:
            exp_val = np.exp(val / tau)
            total_exp += exp_val
            weights.append(exp_val)

        weights = np.array(weights) / total_exp

        # Gradient of log-sum-exp is weighted sum of gradients
        final_grad = np.zeros_like(h)
        for w, (val, grad) in zip(weights, overlaps):
            final_grad += w * grad

        return final_grad

    def project_onto_simplex(h, target_sum=21.0):
        """
        Projects h onto the simplex: sum(h) = target_sum, h_i in [0,1]
        """
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-8:
            return h
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        return h

    # Start with the best solution
    h = best_h.copy()

    # Smoothing parameters schedule
    temperatures = [0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125, 1e-4]
    iterations_per_temp = 200
    
    for tau in temperatures:
        for _ in range(iterations_per_temp):
            if time.time() - start_time > budget_s * 0.95:
                break

            # Gradient
            grad = gradient_smoothed_max(h, tau=tau)
            
            # Gradient step
            step_size = 1.0
            h_new = h - step_size * grad
            
            # Project back onto feasible set
            h_new = project_onto_simplex(h_new, target_sum=21.0)

            # Compute new value
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            h = h_new
        if time.time() - start_time > budget_s * 0.95:
            break

    # Final polishing step with exact non-smooth optimization using scipy
    if time.time() - start_time < budget_s * 0.98:
        def objective_exact(h):
            return evaluate_c5(h)

        def constraint_eq(h):
            return np.sum(h) * dx - 1.0

        cons = {'type': 'eq', 'fun': constraint_eq}

        bounds = [(0, 1) for _ in range(n_points)]

        # Use local optimization around the current best
        res = minimize(
            objective_exact,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_final = res.x
            # Normalize to satisfy constraint
            scale = 1.0 / (np.sum(h_final) * dx)
            h_final = h_final * scale
            c5_final = objective_exact(h_final)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final

    return (best_h.tolist(), best_c5, n_points)
```
