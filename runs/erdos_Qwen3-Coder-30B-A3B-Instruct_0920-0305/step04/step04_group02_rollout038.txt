import numpy as np
from cvxpy import *
import time
from scipy.optimize import minimize

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_binary_milp(n_points):
    """Solve the exact binary MILP for the Erdős problem."""
    # Create variables
    x = Variable(n_points, boolean=True)
    t = Variable(integer=True)
    
    # Constraints
    constraints = [sum(x) == n_points // 2]
    
    # Add constraints for y_{i,d} = x_i * x_{i+d}
    for d in range(-n_points + 1, n_points):
        if d == 0:
            continue
        for i in range(n_points):
            if 0 <= i + d < n_points:
                y = Variable(boolean=True)
                constraints += [
                    y <= x[i],
                    y <= x[i + d],
                    y >= x[i] + x[i + d] - 1
                ]
    
    # Add constraint that each overlap count is at most t
    for d in range(-n_points + 1, n_points):
        if d == 0:
            continue
        overlap_sum = 0
        for i in range(n_points):
            if 0 <= i + d < n_points:
                overlap_sum += x[i] * x[i + d]
        constraints += [n_points // 2 - overlap_sum <= t]
    
    # Objective
    objective = Minimize(t)
    
    # Solve
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=ECOS_BB, verbose=False)
        opt_t = int(t.value)
        x_val = np.round(x.value).astype(int)
    except:
        # Fallback to another approach if exact solver fails
        return None, None
    
    return x_val, opt_t

def smooth_max_gradient_descent(h_start, max_iter=2000, tau_schedule=[0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125, 0.00390625, 0.001953125, 0.0009765625, 0.00048828125, 1e-4], lr=0.01, tolerance=1e-9):
    """Smooth max projected gradient descent to refine fractional solution"""
    n = len(h_start)
    dx = 2.0 / n
    h = np.array(h_start, dtype=float)
    h /= np.sum(h) * dx  # Normalize
    h *= n / 2  # Make integral constraint hold

    best_c5 = float('inf')
    best_h = h.copy()

    for tau in tau_schedule:
        for _ in range(max_iter // len(tau_schedule)):
            # Compute gradients
            totals = []
            all_grad = np.zeros_like(h)
            for d in range(-n + 1, n):
                if d == 0:
                    continue
                grad = np.zeros_like(h)
                for i in range(n):
                    if i + d >= 0 and i + d < n:
                        # Gradient of sum_i h_i - sum_i h_i * h_{i+d}
                        grad[i] += 1.0
                        grad[i + d] -= h[i]
                # Renormalize and compute total overlap
                y = h - np.roll(h, d)
                total = np.sum(h) - np.sum(h * y)
                totals.append(total)
                all_grad += np.exp(total / tau) * grad
            
            # Normalize weights
            weights = np.exp(np.array(totals) / tau)
            weights /= np.sum(weights)
            grad_total = np.sum([w * g for w, g in zip(weights, np.gradient(h))], axis=0)
            
            # Gradient step
            h_new = h - lr * grad_total
            
            # Projection onto box [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Projection onto simplex ∑ h_i = n/2
            h_new = h_new * (n / 2) / np.sum(h_new)
            
            h = h_new
            
            # Evaluate current C5
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = len(initial_h_values)
    h0 = np.array(initial_h_values)

    # Phase 1: Binary MILP solver
    h_binary, t_opt = solve_binary_milp(n_points)
    if h_binary is not None:
        c5_binary = t_opt * 2.0 / n_points
    else:
        # Use initial non-binary solution
        c5_binary = evaluate_c5(h0)
        h_binary = h0.copy()
        t_opt = int(round(c5_binary * n_points / 2))

    # Phase 2: Smoothed gradient descent over refined vectors
    best_h = h_binary.copy()
    best_c5 = c5_binary
    
    # Generate a few random restarts for better results
    num_restarts = 10
    restart_count = 0
    while restart_count < num_restarts and (time.time() - start_time) < budget_s * 0.95:
        # Randomize starting point slightly
        h_perturbed = h_binary + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed *= (n_points / 2) / np.sum(h_perturbed)
        
        h_refined, c5_refined = smooth_max_gradient_descent(h_perturbed, max_iter=1000)
        
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
        
        restart_count += 1
        if (time.time() - start_time) > budget_s * 0.90:
            break

    # Final polishing with SLSQP if possible
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        def constraint_eq(h):
            dx = 2.0 / n_points
            return np.sum(h) * dx - 1.0
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        res = minimize(
            lambda h: evaluate_c5(h),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-12}
        )
        if res.success:
            h_final = res.x
            dx = 2.0 / n_points
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            c5_final = evaluate_c5(h_final)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
