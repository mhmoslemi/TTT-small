import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def create_symmetric_piecewise_linear_vector(h_j, x_j, n):
    """
    Create a symmetric piecewise linear function from breakpoints.
    x_j: array of n+1 breakpoints in [0,1], with x[0]=0, x[n]=1
    h_j: array of n+1 heights in decreasing order
    """
    x = np.linspace(0, 2, n+1)
    h = np.zeros(n)
    for i in range(n):
        if x[i] <= x_j[0]:
            h[i] = h_j[0]
        elif x[i] >= x_j[-1]:
            h[i] = h_j[-1]
        else:
            # Find the interval [x_j[k], x_j[k+1]] containing x[i]
            k = np.searchsorted(x_j, x[i], side='right') - 1
            if k < 0:
                k = 0
            if k >= len(x_j) - 1:
                k = len(x_j) - 2
            x_left = x_j[k]
            x_right = x_j[k+1]
            h_left = h_j[k]
            h_right = h_j[k+1]
            if x_right != x_left:
                h[i] = h_left + (h_right - h_left) * (x[i] - x_left) / (x_right - x_left)
            else:
                h[i] = h_left
    # Mirror the left half to form a symmetric function on [0,2]
    h_full = np.concatenate([h[:n//2], h[n//2-1::-1]])
    return h_full

def decode_parameters(u_j, n):
    """Decode u_j into a full vector h, enforcing constraints."""
    # Sort in descending order for monotonicity
    h_j = np.sort(u_j)[::-1]
    # Ensure total mass is preserved by applying to [0,1] only
    x_j = np.linspace(0, 1, len(u_j))
    # Create full vector with symmetric shape
    h_raw = create_symmetric_piecewise_linear_vector(h_j, x_j, n)
    # Adjust to satisfy integral constraint
    scaling = (n / 2) / np.sum(h_raw)
    h = h_raw * scaling
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Ensure constraint by rescaling
    final_scaling = (n / 2) / np.sum(h)
    h = h * final_scaling
    h = np.clip(h, 0, 1)
    return h
