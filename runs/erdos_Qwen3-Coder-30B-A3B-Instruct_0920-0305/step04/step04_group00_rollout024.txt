import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def construct_h_from_params(params, n_points=256):
    p1, p2, p3, p4 = params
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    
    # Central plateau
    center = n_points // 2
    width = int(p2 * n_points / 2)
    plateau_start = center - width
    plateau_end = center + width
    
    # Heights at endpoints
    h[0] = p4
    h[-1] = p4
    
    # Central part
    if plateau_start >= 0 and plateau_end < n_points:
        h[plateau_start:plateau_end] = p1
    
    # Slopes
    for i in range(plateau_start):
        ratio = (center - i) / (center - plateau_start)
        h[i] = p1 - p3 * (1 - ratio) * (p1 - p4)
    
    for i in range(plateau_end, n_points):
        ratio = (i - center) / (plateau_end - center)
        h[i] = p1 - p3 * (1 - ratio) * (p1 - p4)
    
    # Ensure symmetry
    for i in range(n_points // 2):
        h[n_points - 1 - i] = h[i]
    
    # Normalize integral to 1
    dx = 2.0 / n_points
    scaled_sum = np.sum(h) * dx
    if scaled_sum > 0:
        h = h * (1.0 / scaled_sum)
    
    # Clip h to [0, 1] to satisfy constraints
    h = np.clip(h, 0, 1)
    h = h * (1.0 / (np.sum(h) * dx))  # Ensure sum * dx = 1
    
    return h

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameter bounds for [p1, p2, p3, p4]
    bounds = [(0, 1), (0, 1), (0, 1), (0, 1)]
    
    # Objective function to minimize
    def objective(params):
        h = construct_h_from_params(params)
        return evaluate_c5(h)

    # Optimization using differential evolution
    de_result = differential_evolution(
        objective,
        bounds,
        maxiter=500,
        popsize=32,
        strategy='best1bin',
        tol=1e-6,
        seed=seed,
        callback=lambda x, convergence: None if (time.time() - start_time) < budget_s * 0.8 else None
    )

    best_h = construct_h_from_params(de_result.x)
    best_c5 = evaluate_c5(best_h)

    # Local refinement
    def objective_refine(params):
        h = construct_h_from_params(params)
        return evaluate_c5(h)

    refined_result = minimize(
        objective_refine,
        de_result.x,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-10},
        callback=lambda x: None if (time.time() - start_time) < budget_s * 0.95 else None
    )

    if refined_result.success:
        final_h = construct_h_from_params(refined_result.x)
        final_c5 = evaluate_c5(final_h)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5

    return (best_h.tolist(), best_c5, len(best_h))
