import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient(h, d_star):
    """Compute subgradient for a particular shift d_star"""
    n = len(h)
    g = np.zeros(n)
    if d_star > 0:
        # h[i] * (1 - h[i+d_star])
        g[:-d_star] = 1 - h[d_star:]
        g[-d_star:] = 1  # Not relevant but avoid error
    elif d_star < 0:
        # h[i] * (1 - h[i+d_star])
        g[-d_star:] = 1 - h[:-d_star]
        g[:-d_star] = 1  # Not relevant but avoid error
    else:
        g[:] = 0  # No shift
    return g

def binary_optimal(n):
    """Solve binary integer program to obtain the best integer vector"""
    try:
        import cvxpy as cp
    except ImportError:
        return None, None
        
    # Variables
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    
    # Constraints
    mass = cp.sum(x) == n // 2
    symmetry_breaker = x[0] == 1  # To break symmetry (arbitrary)
    
    constraints = [mass, symmetry_breaker]
    
    # For each shift d ≠ 0
    shift_indices = list(range(-n+1, 0)) + list(range(1, n))
    for d in shift_indices:
        w = cp.Variable(n, boolean=True)
        # Piecewise constraints to make w[i] = x[i] * x[i+d]
        for i in range(n):
            if i + d < 0 or i + d >= n:
                continue
            # Eq: w_i <= x_i
            constraints.append(w[i] <= x[i])
            # Eq: w_i <= x_{i+d}
            constraints.append(w[i] <= x[i + d])
            # Eq: w_i >= x_i + x_{i+d} - 1
            constraints.append(w[i] >= x[i] + x[i + d] - 1)
        
        # Count (1,0) pairs for shift d
        N_d = cp.sum(x) - cp.sum(w)
        # Add constraint to bound N_d by t
        constraints.append(N_d <= t)
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CBC, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return x.value.astype(float), int(t.value)
    except:
        pass
    # Try other solver if necessary
    try:
        prob.solve(solver=cp.GLPK_MI, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return x.value.astype(float), int(t.value)
    except:
        pass
    # Try ECOS_BB if needed
    try:
        prob.solve(solver=cp.ECOS_BB, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return x.value.astype(float), int(t.value)
    except:
        pass
        
    # If nothing works, return a rough estimate
    return None, None

def refine_continuous(h0, time_left):
    """Refine the solution with projected gradient descent"""
    n = len(h0)
    dx = 2.0 / n
    best = h0.copy()
    best_val = evaluate_c5(best)
    
    max_iter = 1000
    iter_count = 0
    while iter_count < max_iter and time_left > 0.1:
        start_time = time.time()
        # Compute all overlaps
        corr = correlate(best, 1 - best, mode="full")
        max_val = np.max(corr)
        d_indices = np.where(corr == max_val)[0]
        # Adjust from full to actual shift indices
        d_max = d_indices[0] - (n - 1)  # Convert index to shift value
        if d_max == 0:
            d_max = d_indices[1] - (n - 1)
        # Compute subgradient
        g = subgradient(best, d_max)  # Use first max shift
        
        # Simple gradient step with decrease in step size
        step = 0.2
        h_new = best - step * g
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n / 2) / np.sum(h_new)
        val_new = evaluate_c5(h_new)
        
        if val_new < best_val:
            best = h_new.copy()
            best_val = val_new
        # Occasionally perturb
        if np.random.rand() < 0.1:
            h_new = best + np.random.normal(0, 0.01, n)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n / 2) / np.sum(h_new)
            val_new = evaluate_c5(h_new)
            if val_new < best_val:
                best = h_new.copy()
                best_val = val_new
        iter_count += 1
        time_left -= (time.time() - start_time)
    
    return best, best_val

        def objective(h):
            return evaluate_c5(h)

            dx = 2.0 / n_points
            return np.sum(h) * dx - 1.0
        
        cons = {'type': 'eq', 'fun': constraint_eq}
