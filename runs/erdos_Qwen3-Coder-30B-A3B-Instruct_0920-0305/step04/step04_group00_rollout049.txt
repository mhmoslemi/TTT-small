import numpy as np

from scipy.signal import correlate
import time

def evaluate_c5_exact(h):
    """Evaluate the exact C5 constant using max of correlations"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def decode_h(u_vals, n):
    """Decode a low-dimensional vector of heights to full vector of length n"""
    # Input: u_vals = [h0, h1, ..., hK]  (sorted descending)
    # construct piecewise linear symmetric function on [0,2]
    K = len(u_vals) - 1  # breakpoints
    x_vals = np.linspace(0, 1, K+1)  # [0, 1/K, ..., 1]
    heights = np.array(u_vals)
    # Interpolate to full interval [0,1]
    x_full = np.linspace(0, 1, n//2)
    # linear interpolation
    h_half = np.interp(x_full, x_vals[::-1], heights[::-1])

    # Now get h_full on [0,2] - symmetric
    h_full = np.concatenate([h_half, h_half[-2::-1]])
    
    # global scale to sum = n/2
    current_sum = np.sum(h_full)
    if current_sum > 0:
        scale = (n/2) / current_sum
        h_full *= scale
        h_full = np.clip(h_full, 0, 1)
        # renormalize again if necessary (this covers potential clipping)
        current_sum = np.sum(h_full)
        scale = (n/2) / current_sum
        h_full *= scale
        h_full = np.clip(h_full, 0, 1)
        # Ensure sum exactly n/2
        h_full *= (n/2) / np.sum(h_full)
    else:
        h_full = np.ones(n) * 0.5  # fallback
        h_full *= (n/2) / np.sum(h_full)

    return h_full

    """Objective for DE optimization"""

    n = 200
    h = decode_h(u_vals, n)
