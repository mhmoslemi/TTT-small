import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_boxed_simplex(v, target_sum):
    """Project vector onto the region {0 <= x <= 1, sum(x) = target_sum}"""
    # First clip to [0,1]
    v = np.clip(v, 0, 1)
    # Then adjust to meet sum target while staying within [0,1]
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-12:
        return v
    
    # Distribute correction evenly among unfixed entries
    diff = target_sum - current_sum
    free = (v > 0) & (v < 1)
    n_free = np.sum(free)
    if n_free > 0:
        v[free] += diff / n_free
        v = np.clip(v, 0, 1)
    
    return v

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use a moderate resolution
    n_points = 100
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    # Initialize with coarse grid approximation
    # Try a slightly more uniform distribution to start with
    h0 = np.zeros(n_points)
    # Create variable number of intervals
    interval_count = 40  # half of points to make it dense
    interval_width = n_points // interval_count
    for i in range(interval_count):
        start_idx = i * interval_width
        end_idx = start_idx + interval_width
        if i % 2 == 0:  # alternate pattern
            h0[start_idx:end_idx] = 1.0
    
    # Normalize to target integral
    current_sum = np.sum(h0)
    h0 = h0 * (target_sum / current_sum)
    
    # Clip to valid range
    h0 = np.clip(h0, 0, 1)

    # Track best solution
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Gradient-based refinement with smooth approximation
    # Alpha scheduling for smooth approximation to max
    alpha_schedule = [5, 10, 20, 50, 100]
    
    step_size = 0.01
    max_iter = 5000
    iter_count = 0
    
    # Parameters for gradient descent
    max_inner_iter = 1000
    
    # Run gradient descent
    while (time.time() - start_time) < budget_s * 0.9:
        # For each alpha in schedule, do optimization step
        for alpha in alpha_schedule:
            if (time.time() - start_time) > budget_s * 0.9:
                break
                
            # Compute correlations
            corr_hh = correlate(h0, h0, mode='full')
            corr_hh = corr_hh[n_points-1:]  # Only positive shifts
            
            # Compute g values = sum_h - correlation
            g_vals = target_sum - corr_hh
            
            # Compute softmax weights - but avoid numerical overflow
            g_centered = g_vals - np.max(g_vals)
            exp_g = np.exp(alpha * g_centered)
            w = exp_g / np.sum(exp_g)
            
            # Compute gradient
            # ∇(log sum exp) = w * ∇(sum_i g_i) = w * (1 - (h_{i+shift} + h_{i-shift}))
            # Using convolution
            grad = np.ones_like(h0)
            
            # Compute cyclic shifts efficiently using convolution
            # Add small regularization to prevent numerical issues
            w_pad = np.zeros_like(h0)
            w_pad[:len(w)] = w
            # w_pad will be used to weight the convolution; we want to compute
            # the convolution of the weight vector with h, but carefully handling the indices
            # This is the gradient: 1 - 2 * sum over weights times shifted values
            # We compute convolution of w (centered at middle) with h
            
            # Manually compute gradient using shifts
            gradient = np.zeros_like(h0)
            for s in range(len(w)):
                shift = s - (len(w) // 2)
                if shift >= 0:
                    # Positive shift: h[i-shift] comes from earlier indices
                    gradient += w[s] * np.roll(h0, -shift)
                else:
                    # Negative shift: h[i-shift] is h[i+|shift|]
                    gradient += w[s] * np.roll(h0, abs(shift))
                    
            # Gradient = -
            grad = 1 - 2 * gradient
            
            # Update
            h0 -= step_size * grad
            
            # Project back to feasible region
            h0 = project_to_boxed_simplex(h0, target_sum)
            
            # Ensure valid range
            h0 = np.clip(h0, 0, 1)
            
            # Update step size
            step_size *= 0.999
            
            # Evaluate current solution
            c5_candidate = evaluate_c5(h0)
            if c5_candidate < best_c5:
                best_h = h0.copy()
                best_c5 = c5_candidate
            
            iter_count += 1
            if (time.time() - start_time) > budget_s * 0.9:
                break

        # Local refinement using scipy optimizer
        bounds = [(0, 1) for _ in range(n_points)]
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) * dx - 1.0
        
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            # Do a quick local optimization with SLSQP
            ref_h = h0 + np.random.normal(0, 0.01, n_points)
            ref_h = np.clip(ref_h, 0, 1)
            scale = 1.0 / (np.sum(ref_h) * dx)
            ref_h = ref_h * scale
            res = minimize(
                objective, 
                ref_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
        
        # Periodic check of improvement
        if iter_count % 100 == 0:
            print(f"Current time: {time.time() - start_time:.1f}s, Best C5: {best_c5:.6f}")
            
        # Early exit if we've improved significantly
        if best_c5 < 0.3807 and (time.time() - start_time) > 300:
            break
            
    return (best_h.tolist(), best_c5, n_points)
