import numpy as np

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_c5_binary(x):
    """Evaluate binary C5 = max_d Σ_i x_i * (1 - x_{i+d})"""
    n = len(x)
    max_overlap = 0
    # Loop over non-zero shifts d = -(n-1) to -(n-1)
    for d in range(-n+1, n):
        if d == 0: continue
        overlap = 0
        for i in range(n):
            if d < 0 and i + d < 0:
                continue
            if d > 0 and i + d >= n:
                continue
            if d < 0:
                overlap += x[i] * (1 - x[i + d])
            else:
                overlap += x[i] * (1 - x[i + d])
        max_overlap = max(max_overlap, overlap)
    return max_overlap

def solve_milp_binary(n):
    """Solve the MILP for binary case using CVXPY"""
    # Variables: x binary, y binary, t integer
    x = Variable(n, boolean=True)
    y = Variable((n, 2*(n-1)), boolean=True)  # for each overlapping pair
    t = Variable(integer=True)
    
    # Set up constraint matrices
    constraints = [
        cvx_sum(x) == n // 2,
        t >= 0,
        # Linearization of y[i,d] = x[i] * x[i+d] for all admissible d
    ]
    
    # Precalculate indices
    indices = []
    # For each non-zero shift d
    for d in range(-n+1, n):
        if d == 0: continue
        for i in range(n):
            if d < 0 and i + d < 0:
                continue
            if d > 0 and i + d >= n:
                continue
            indices.append((i, i+d, d))
    
    # Constraints
    # For each i, d where x_i and x_{i+d} are valid
    for i, i_d, d in indices:
        # y[i,d] <= x[i], y[i,d] <= x[i+d], y[i,d] >= x[i] + x[i+d] - 1
        j = indices.index((i, i_d, d))  # This should be unique buthard to index directly
        # We need to impose the constraints manually
        # We will encode them differently
    
    # Reproject index triples to flat array of y_ij variables
    y_vars = []
    index_map = {}
    for i in range(n):
        for d in range(-n+1, n):
            if d == 0: continue
            if d < 0 and i + d < 0: continue
            if d > 0 and i + d >= n: continue
            index_map[(i, i+d)] = len(y_vars)
            y_vars.append(Variable(boolean=True))
    
    # Build constraint list with proper indices
    for i in range(n):
        for d in range(-n+1, n):
            if d == 0: continue
            if d < 0 and i + d < 0: continue
            if d > 0 and i + d >= n: continue
            j = index_map[(i, i+d)]
            constraints += [
                y_vars[j] <= x[i],
                y_vars[j] <= x[i+d],
                y_vars[j] >= x[i] + x[i+d] - 1,
                ]
    
    # Build overlap for each shift and constraint on max
    overlaps_exprs = []
    for d in range(-n+1, n):
        if d == 0: continue
        overlap = n // 2 - cvx_sum([y_vars[index_map[(i, i+d)]] for i in range(n) if i+d < n and i+d >= 0])
        overlaps_exprs.append(overlap)
        
    # Force constraint that overlapping counts are all <= t
    for expr in overlaps_exprs:
        constraints.append(expr <= t)
    
    # Add trivial lower bound
    constraints.append(t >= 0)
    
    # Objective
    objective = Minimize(t)
    prob = Problem(objective, constraints)
    
    # Optimization
    sol = prob.solve(solver='CBC', verbose=False)
    if sol in (None, np.inf, -np.inf):
        return None, None
    
    x_sol = np.array([v.value for v in x])
    t_sol = int(t.value)
    return x_sol, t_sol

def compute_gradient(h, d):
    """ Compute ∂C_d/∂h_j """
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        # Forward index j+d  (if inside bounds)
        if j + d < n:
            grad[j] += 1 - h[j+d]
        # Backward index j-d  (if inside bounds)
        if j - d >= 0:
            grad[j] += 1 - h[j-d]
    grad -= 1  # because we have h_i - h_i * h_{i+d}
    return grad

def smooth_max_gradient(h, tau):
    """
    Compute smoothed gradient using log-sum-exp surrogate.
    Phi_tau(h) = tau * log(sum of exp(C_d/h))
    """
    n = len(h)
    # Compute all Correlation functions C_d
    corr_values = []
    for d in range(-n+1, n):
        if d == 0:
            continue
        corr = 0
        for i in range(n):
            if i + d < 0 or i + d >= n:
                continue
            corr += h[i] - h[i] * h[i+d]
        corr_values.append(corr)
    
    # Compute W_d weights
    exp_values = np.exp(np.array(corr_values) / tau)
    exp_sum = np.sum(exp_values)
    weights = exp_values / exp_sum
    
    # Now compute ∇Φ_τ(h) = sum_d w_d ∇C_d(h)
    grad = np.zeros(n)
    for i, d in enumerate(range(-n+1, n)):
        if d == 0:
            continue
        grad += weights[i] * compute_gradient(h, d)
    return grad

def project_simplex(h, target_sum=1.0):
    """ Project h onto the simplex: h >= 0, sum(h) = target_sum """
    n = len(h)
    # clip to [0,1]
    h = np.clip(h, 0, 1)
    # scale to sum to target_sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def smooth_max_descent(h0, n, tau0=0.5, max_restarts=10):
    """ Run smooth max gradient descent + restarts """
    h = h0.copy()
    # Normalize
    h = project_simplex(h, n/2)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for restart in range(max_restarts):
        hcur = h.copy()
        tau = tau0
        while tau > 1e-4:
            # 200 steps at current tau
            for i in range(200):
                grad = smooth_max_gradient(hcur, tau)
                hcur = hcur - 0.1 * grad  # LR = 0.1
                hcur = project_simplex(hcur, n/2)
            
            # Reduce temperature
            tau *= 0.5
            
            # Evaluate new value and possibly update best
            cur_c5 = evaluate_c5(hcur)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = hcur.copy()
                
        # Random restart
        h = best_h.copy()
        h += np.random.normal(0, 0.02, n)
        h = project_simplex(h, n/2)
    
    return best_h, best_c5
