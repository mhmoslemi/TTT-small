import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initialize from the given initial_h_values
    h0 = np.array(initial_h_values)
    n_points = len(h0)

    # Phase 1: Binary MILP optimization
    # Model as mixed integer quadratic program
    h_bin = cp.Variable(n_points, boolean=True)
    t = cp.Variable(integer=True)

    # Constraints
    mass_constraint = cp.sum(h_bin) == n_points // 2  # integral constraint
    constraints = [mass_constraint]

    # For all non-zero shifts, add overlap constraints
    for d in range(1, n_points):
        # yi,d = xi * x_{i+d}
        # sum over i where both indices valid
        y = cp.Variable(n_points - d, boolean=True)
        constraints += [
            y <= h_bin[d:],
            y <= h_bin[:n_points-d],
            y >= h_bin[d:] + h_bin[:n_points-d] - 1
        ]

        # Overlap for shift d
        N_d = cp.sum(h_bin) - cp.sum(y)
        # Add the inequality that overlap must be <= t
        constraints.append(N_d <= t)

    # Objective
    problem = cp.Problem(cp.Minimize(t), constraints)

    best_h_bin = h0.astype(bool)  # Starting here
    best_t = float('inf')
    
    # Assume best_t is computed via MILP solver at time T
    # If T > budget, abort and fallback on continuous heuristic
    
    # Approximate search of binary pattern that matches our integer criteria
    # Use combinatorial idea to find binary vector with t = 8
    # For now: directly subtract and normalize or return from existing h0

    # We fix a binary solution as a reasonable starting point
    # In an actual MILP, we would have solved it here
    # Since exact MILP not easy in this context (solver not guaranteed),
    # we proceed to continuous relaxation + gradient descent refinement

    # Our fallback heuristic
    def compute_overlap(h):
        dx = 2.0 / n_points
        corr = correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    # We directly start with refined continuous solution
    # Heuristically perturb the initial values towards a binary pattern

    # We compute the original evaluation directly
    best_h = h0.copy()
    best_c5 = compute_overlap(best_h)

    # Phase 2: Continuous refinement using smooth max
    def smooth_max_overlaps(h, temp=0.1):
        """Use log-sum-exp smooth approximation over overlaps"""
        dx = 2.0 / n_points
        overlaps = []
        for d in range(-n_points+1, n_points):
            if d == 0:
                continue
            # Slide h backward by d
            h_d = np.roll(h, d)
            ov = np.sum(h * (1 - h_d))
            overlaps.append(ov * dx)
        total = np.log(np.sum(np.exp(np.array(overlaps) / temp)))
        return temp * total

    def smooth_grad(h, temp=0.1):
        """Gradient of log-sum-exp approximation"""
        dx = 2.0 / n_points
        overlaps = []
        weights = []
        for d in range(-n_points+1, n_points):
            if d == 0:
                continue
            h_d = np.roll(h, d)
            ov = np.sum(h * (1 - h_d))
            overlaps.append(ov * dx)
        overlaps = np.array(overlaps)
        exps = np.exp(overlaps / temp)
        weights = exps / np.sum(exps)
        grad = np.zeros_like(h)
        for i, d in enumerate(range(-n_points+1, n_points)):
            if d == 0:
                continue
            # df/dh_j = (1 - h_{j+d}) + (1 - h_{j-d})
            # Adjust indices for circular shift
            h_d = np.roll(h, d)
            grad += weights[i] * (1 - h_d)
            h_neg_d = np.roll(h, -d)
            grad += weights[i] * (1 - h_neg_d)
        return grad

    # Proximal gradient descent with continuation
    max_restart = 5
    restart_count = 0
    step_size = 0.005
    max_iter = 300
    tau = 0.5  # temperature for smoothing

    h_refined = best_h.copy()
    cur_c5 = best_c5

    while (time.time() - start_time) < budget_s * 0.9 and restart_count < max_restart:
        h_current = h_refined.copy()
        # Outer loop - decrease temperature to approximate max
        for i in range(3):
            # Inner loop - gradient descent
            for _ in range(max_iter):
                if (time.time() - start_time) > budget_s * 0.9:
                    break
                # Check if we are within range of the final schedule
                if tau < 1e-4:
                    break
                # Gradient step
                grad = smooth_grad(h_current, tau)
                h_new = h_current - step_size * grad
                # Clip and project
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)  # maintain sum = n/2
                h_current = h_new.copy()
                # Evaluate
                c5 = compute_overlap(h_current)
                if c5 < cur_c5:
                    cur_c5 = c5
                    h_refined = h_current.copy()
            # Reduce temperature
            tau = max(tau * 0.5, 1e-5)
        
        # Add small restart perturbation
        h_current = h_refined + np.random.normal(0, 0.02, n_points)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
        restart_count += 1
    
    # Final polishing with SLSQP
    def obj_func(h):
        return compute_overlap(h)

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) * 2.0 / n_points - 1.0}
    
    # Optimization with reduced learning rate for robustness
    res = minimize(
        obj_func, 
        h_refined, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-14}
    )

    if res.success:
        h_final = res.x
        h_final = h_final * (n_points / 2) / np.sum(h_final)
        c5_final = compute_overlap(h_final)
        if c5_final < cur_c5:
            cur_c5 = c5_final
            h_refined = h_final
    else:
        h_final = h_refined

    return (h_refined.tolist(), cur_c5, n_points)
