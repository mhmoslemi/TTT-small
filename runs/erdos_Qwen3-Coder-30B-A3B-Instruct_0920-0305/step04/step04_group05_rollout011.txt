import numpy as np
from scipy.optimize import differential_evolution, minimize, basinhopping
from scipy.signal import correlate
import time

def project_to_feasible(h, target_sum):
    """Project h to feasible set: [0,1]^n and sum = target_sum"""
    y = np.clip(h, 0, 1)
    current_sum = np.sum(y)
    if abs(current_sum - target_sum) < 1e-9:
        return y
    if current_sum == 0:
        return np.full_like(h, target_sum / len(h))
    
    # Use iterative approach to adjust
    y = y * (target_sum / current_sum)
    y = np.clip(y, 0, 1)
    return y

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

    # Target sum (area constraint)

    def eval_and_project(h):
        projected = project_to_feasible(h, target_sum)
        return evaluate_c5(projected)  # return projected version

    def take_step(x):

        return project_to_feasible(perturbed, target_sum)

    bh_result = basinhopping(

    try:
        result = minimize(

            method='SLSQP',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
        )

    # Final check with more optimization
    for _ in range(100):
        if (time.time() - start_time) > budget_s * 0.99:
            break
            
        # Random restart
        new_h = np.random.uniform(0, 1, n_points)
        new_h = project_to_feasible(new_h, target_sum)
        c5 = evaluate_c5(new_h)
        if c5 < best_c5:
