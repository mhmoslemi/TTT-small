import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: Binary optimization via MILP
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Create binary variables for h
    x = cp.Variable(n_points, boolean=True)
    
    # Constraint: sum h_i = n/2 
    constraints = [cp.sum(x) == n_points / 2]
    
    # Add auxiliary variables y_{i,d} for products h_i * h_{i+d}
    # These are introduced for d = 1 to n-1 only (positive shifts)
    # and we'll construct all Y[d][i] where i+d < n as needed
    # For simplicity of construction, consider all d in [1, n-1]
    # and use 2D array Y[d-1][i] for d shifts, i indices.
    Y = []
    for d in range(1, n_points):
        var_row = []
        for i in range(n_points):
            if i + d < n_points:
                y = cp.Variable(boolean=True)
                var_row.append(y)
                # McCormick constraints:
                constraints.extend([
                    y <= x[i],
                    y <= x[i+d],
                    y >= x[i] + x[i+d] - 1
                ])
            else:
                var_row.append(None)
        Y.append(var_row)

    # Objective: minimize t such that for each shift d:
    #  sum_{i} h_i - sum_{i} h_i h_{i+d} <= t
    # The first sum is constant = n/2.
    t = cp.Variable(integer=True)
    
    # Iterate over all shifts d = 1,...,n-1
    for d in range(1, n_points):
        # Compute sum of products h_i * h_{i+d} for valid indices
        sum_y = cp.sum([Y[d-1][i] for i in range(n_points) if i+d < n_points])
        # Overlap = sum h_i - sum y_ij = n/2 - sum y_ij
        overlap = n_points/2 - sum_y
        # Constraint: overlap <= t
        constraints.append(overlap <= t)

    # Build the MILP problem
    prob = cp.Problem(cp.Minimize(t), constraints)

    # Try to solve the MILP
    try:
        prob.solve(solver=cp.CBC, verbose=False, max_iter=10000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_sol = x.value.round().astype(int)
            t_opt = int(t.value)
        else:
            # Fallback to using initial solution if MILP fails
            x_sol = np.array(initial_h_values)
            t_opt = 8  # empirical known value for binary case at n=42
    except Exception:
        # Fallback to initial solution
        x_sol = np.array(initial_h_values)
        t_opt = 8  # known minimum

    # Phase 2: Refinement using smooth max projected gradient descent
    h_best = x_sol.copy().astype(float)
    h_best = h_best / np.sum(h_best) * (n_points / 2)  # Normalize sum to 1
    c5_best = evaluate_c5(h_best)

    # Smoothed objective function
    def smooth_max_objective(h, tau=0.5):
        shifted_corr = np.array([np.sum(h * np.roll(1 - h, d)) for d in range(-n_points+1, n_points) if d != 0])
        return tau * np.log(np.sum(np.exp(shifted_corr / tau)))

    def smooth_grad(h, tau=0.5):
        # Compute all C_d(h)
        cd_vals = np.array([np.sum(h * np.roll(1 - h, d)) for d in range(-n_points+1, n_points) if d != 0])
        # Compute weights
        exp_vals = np.exp(cd_vals / tau)
        w_vals = exp_vals / np.sum(exp_vals)
        # Gradient for each position j
        grad = np.zeros_like(h)
        for d_idx, d in enumerate(range(-n_points+1, n_points) if d != 0):
            if d != 0:
                w = w_vals[d_idx]
                rolled_h = np.roll(h, d)
                grad += w * (1 - (rolled_h if d > 0 else np.roll(h, -d)))
        return grad

    def project_simplex(h, target_sum=1.0):
        # Clip values to [0,1]
        h = np.clip(h, 0, 1)
        # Scale to sum to target_sum
        scale = target_sum / np.sum(h)
        return h * scale

    def backtracking_line_search(h, grad, alpha_max=1.0, tau=0.5, tolerance=1e-12):
        alpha = alpha_max
        old_value = smooth_max_objective(h, tau)
        for _ in range(10):  # max halvings
            h_new = h - alpha * grad
            h_new = project_simplex(h_new)
            new_value = smooth_max_objective(h_new, tau)
            if new_value <= old_value - 0.5 * alpha * np.dot(grad, grad)*dx:
                return h_new
            alpha /= 2
        return h

    # Radius for perturbations
    epsilon = 0.02
    max_restarts = 15
    restarts = 0
    
    while time.time() - start_time < 0.95 * budget_s and restarts < max_restarts:
        # Initialize h for this restart
        h = h_best.copy()
        
        # Schedule for temperature
        tau_vals = [0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125, 0.00390625]
        
        for tau in tau_vals:
            if time.time() - start_time > 0.9 * budget_s:
                break
            for _ in range(200):
                if time.time() - start_time > 0.9 * budget_s:
                    break
                    
                grad = smooth_grad(h, tau)
                h_new = backtracking_line_search(h, grad, tau=tau)
                h = h_new
                
                # Early stopping based on improvement
                if time.time() - start_time > 0.9 * budget_s:
                    break
                    
        # Did we improve?
        c5_new = evaluate_c5(h)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h.copy()
            
        # Random restart with small perturbation
        h = h_best + np.random.normal(0, epsilon, size=n_points)
        h = project_simplex(h)
        restarts += 1

    # Final polishing using standard optimizers
    if time.time() - start_time < 0.8 * budget_s:
        bounds = [(0, 1)] * n_points
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}

        def obj_func(h):
            return evaluate_c5(h)
            
        # SLSQP refinement starting from best solution
        try:
            res = minimize(
                obj_func,
                h_best,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-15}
            )
            if res.success:
                h_final = res.x
                # Normalize again
                h_final = h_final / np.sum(h_final) * (n_points / 2)
                c5_final = evaluate_c5(h_final)
                if c5_final < c5_best:
                    c5_best = c5_final
                    h_best = h_final
        except Exception:
            pass

    return (h_best.tolist(), c5_best, n_points)
