import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the provided initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Time for MILP solve
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time < 1:
        # fallback to heuristic if no time for MILP
        return run_heuristic(seed, budget_s)
    
    # Try to solve as MILP first
    try:
        # Define the MILP model
        h = cp.Variable(n_points, boolean=True)
        p = cp.Variable((n_points, n_points), boolean=True)
        M = cp.Variable(integer=True, name="M")
        
        # Constraints for logical AND
        constraints = []
        for l in range(1, n_points):
            for i in range(n_points - l):
                # p[i,l] <= h[i] 
                constraints.append(p[i,l] <= h[i])
                # p[i,l] <= h[i+l]
                constraints.append(p[i,l] <= h[i+l])
                # p[i,l] >= h[i] + h[i+l] - 1
                constraints.append(p[i,l] >= h[i] + h[i+l] - 1)
                
        # Intersection counts  
        for l in range(1, n_points):
            S_l = cp.sum(p[:,l])  # sum of products
            A_l_plus = cp.sum(h[:n_points-l])   # left overlap
            A_l_minus = cp.sum(h[l:])          # right overlap
            
            # Add worst-case overlap constraints
            constraints.append(A_l_plus - S_l <= M)
            constraints.append(A_l_minus - S_l <= M)
            
        # Integral constraint: exactly 21 ones
        constraints.append(cp.sum(h) == n_points // 2)
        
        # Symmetry breaking
        constraints.append(h[0] == 1)
        
        # Objective
        objective = cp.Minimize(M)
        prob = cp.Problem(objective, constraints)
        
        # Solve with a time limit
        # Because cvxpy doesn't support timeout directly, but CBS/CBC does
        prob.solve(solver=cp.CBC, max_seconds=remaining_time - 0.1)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value.astype(int)
            # Compute the exact C5
            dx = 2.0 / n_points
            corr = correlate(h_opt, 1 - h_opt, mode="full")
            max_corr = np.max(corr)
            c5_bound = max_corr * dx
            h_values = h_opt.astype(float).tolist()
            return (h_values, c5_bound, n_points)
            
    except Exception as e:
        pass
        
    # If MILP fails, fallback to heuristic
    return run_heuristic(seed, budget_s)

def run_heuristic(seed, budget_s):
    np.random.seed(seed)
    start_time = time.time()
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Use a simulated annealing approach with specific bit flips
    def flip_bits(h, num_flips=1):
        h_copy = h.copy()
        ones = np.where(h_copy == 1)[0]
        zeros = np.where(h_copy == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            return h_copy
        for _ in range(num_flips):
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            h_copy[i] = 0
            h_copy[j] = 1
        return h_copy
    
    def objective(h):
        return evaluate_c5(h)
    
    # Initial best value
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    
    # For probability to accept worse solutions
    T = 1.0
    cooling_rate = 0.995
    min_T = 1e-5
    
    # Annealing loop
    max_iter = 1000000  # number of swaps
    iter_count = 0
    
    while iter_count < max_iter:
        if (time.time() - start_time) > budget_s - 2:
            break
            
        # Try a neighbor
        neighbor_h = flip_bits(best_h, 1)
        
        # scale to keep integral constraint
        scale = 1.0 / (np.sum(neighbor_h) * dx)
        neighbor_h = neighbor_h * scale
        
        neighbor_c5 = objective(neighbor_h)
        
        # Accept accordingly
        if neighbor_c5 < best_c5:
            best_h = neighbor_h.copy()
            best_c5 = neighbor_c5
        else:
            delta = neighbor_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                best_h = neighbor_h.copy()
                best_c5 = neighbor_c5
                
        T *= cooling_rate
        if T < min_T:
            T = min_T
            
        iter_count += 1
    
    # Try some local polishing
    if (time.time() - start_time) > budget_s - 2:
        pass
    else:
        # Local search around best
        bounds = [(0, 1) for _ in range(n_points)]
        def constraint_eq(h):
            return np.sum(h) * dx - 1.0
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        # Multiple restarts
        best_poly_h = best_h.copy()
        best_poly_c5 = best_c5
        
        for _ in range(100):  # few restarts of SLSQP
            if (time.time() - start_time) > budget_s - 2:
                break
                
            # Random perturbation
            new_h = best_h + np.random.normal(0, 0.02, n_points)
            new_h = np.clip(new_h, 0, 1)
            scale_factor = 1.0 / (np.sum(new_h) * dx)
            new_h = new_h * scale_factor
            
            try:
                res = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
                if res.success:
                    h_candidate = res.x
                    scale = 1.0 / (np.sum(h_candidate) * dx)
                    h_candidate = h_candidate * scale
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_poly_c5:
                        best_poly_h = h_candidate.copy()
                        best_poly_c5 = c5_candidate
            except:
                continue
        
        best_h = best_poly_h
        best_c5 = best_poly_c5
    
    return (best_h.tolist(), best_c5, n_points)
