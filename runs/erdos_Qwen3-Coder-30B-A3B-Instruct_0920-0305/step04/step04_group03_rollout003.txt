import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_boxed_simplex(v, target_sum):
    """Project vector v onto the simplex with box constraints [0,1] and sum=target_sum"""
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-12:
        return v
    
    # Use iterative adjustment to meet sum target
    iter_count = 0
    while iter_count < 100:
        current_sum = np.sum(v)
        if abs(current_sum - target_sum) < 1e-10:
            break
        diff = target_sum - current_sum
        free = (v > 0) & (v < 1)
        nfree = np.sum(free)
        if nfree == 0:
            break
        v[free] += diff / nfree
        v = np.clip(v, 0, 1)
        iter_count += 1
    return v

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use a higher resolution to get better bounds
    n_points = 80
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    best_h = None
    best_c5 = float('inf')
    
    # Phase 1: Binary search to find good starting point
    max_sa_iter = 100000
    
    # Initialize binary vector with exactly half ones
    binary_h = np.zeros(n_points, dtype=int)
    binary_h[:n_points//2] = 1
    np.random.shuffle(binary_h)
    
    # Simulated Annealing for binary variables
    current_binary = binary_h.copy()
    current_corr = correlate(current_binary, 1 - current_binary, mode="full")
    current_min_corr = np.min(current_corr)
    current_score = -current_min_corr  # Minimize -min_corr (maximize min_corr)
    
    best_binary = current_binary.copy()
    best_min_corr = current_min_corr
    
    # Annealing schedule
    temp = 10.0
    alpha = 0.9995
    
    iterations = 0
    while iterations < max_sa_iter and (time.time() - start_time) < budget_s * 0.7:
        # Pick two random indices to swap
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Ensure i and j are different
        if i == j: continue
            
        # Ensure one is 1 and other is 0
        if current_binary[i] == current_binary[j]: continue
        
        # Temporarily swap
        current_binary[i], current_binary[j] = current_binary[j], current_binary[i]
        
        # Compute new correlation efficiently by updating
        try:
            new_corr = correlate(current_binary, 1 - current_binary, mode="full")
            new_min_corr = np.min(new_corr)
            new_score = -new_min_corr
            
            # Accept or reject the change
            if new_score >= current_score or np.random.rand() < np.exp((new_score - current_score) / temp):
                current_score = new_score
                current_min_corr = new_min_corr
                if new_min_corr > best_min_corr:
                    best_min_corr = new_min_corr
                    best_binary = current_binary.copy()
            else:
                # Revert the swap
                current_binary[i], current_binary[j] = current_binary[j], current_binary[i]
        except:
            # Revert if error
            current_binary[i], current_binary[j] = current_binary[j], current_binary[i]
            
        temp *= alpha
        iterations += 1
    
    # Convert best binary to continuous
    h_continuous = best_binary.astype(float)
    h_continuous = project_to_boxed_simplex(h_continuous, target_sum)
    
    # Phase 2: Refine with gradient-based optimization
    # Use smooth approximation for gradient descent
    max_grad_iter = 5000
    learning_rate = 0.01
    
    # Smooth max approximation parameters
    alpha = 5.0  # Start with smooth approximation
    
    for i in range(max_grad_iter):
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
        # Compute current correlation
        corr = correlate(h_continuous, h_continuous, mode="full")
        g_values = target_sum - corr  # g_k = sum_h - correlation_k
        
        # Smooth max for better gradient approximation
        max_g = np.max(g_values)
        exp_vals = np.exp(alpha * (g_values - max_g))
        weights = exp_vals / np.sum(exp_vals)
        
        # Compute gradient
        # Gradient of g_k = 1 - (h shifted right by k + h shifted left by k)
        # Gradient = 1 - 2 * sum_{k} weight_k * h_shifted_by +k
        conv = np.convolve(weights, h_continuous, mode="full")
        gradient = 1.0 - 2.0 * conv[n_points-1:n_points-1+n_points]
        
        # Gradient step
        h_new = h_continuous - learning_rate * gradient
        h_new = project_to_boxed_simplex(h_new, target_sum)
        
        # Evaluate
        current_c5 = evaluate_c5(h_new)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            
        h_continuous = h_new.copy()
        learning_rate *= 0.9995  # Decrease learning rate
        alpha = min(alpha * 1.01, 100)  # Gradually increase smoothness
    
    # Final evaluation with the true C5 computation
    if best_h is None:
        # Fall back to random initialization if none found
        h0 = np.random.rand(n_points)
        h0 = project_to_boxed_simplex(h0, target_sum)
        best_c5 = evaluate_c5(h0)
        best_h = h0.copy()
    else:
        # Double-check the value
        best_c5 = evaluate_c5(best_h)
        
    return (best_h.tolist(), best_c5, n_points)
