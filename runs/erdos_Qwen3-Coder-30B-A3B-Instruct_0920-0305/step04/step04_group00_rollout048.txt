import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_onto_feasible(h, n, target_sum):
    """Project h onto the feasible set: 0 <= h <= 1 and sum(h) * dx = 1"""
    DX = 2.0 / n
    # Binary search for the correct lambda
    lower, upper = -1.0, 1.0
    for _ in range(30):
        lamb = (lower + upper) / 2.0
        ph = np.clip(h - lamb, 0, 1)
        current_sum = np.sum(ph) * DX
        if current_sum > target_sum:
            lower = lamb
        else:
            upper = lamb
    lamb = (lower + upper) / 2.0
    result = np.clip(h - lamb, 0, 1)
    # Renormalize to satisfy integral exactly
    actual_sum = np.sum(result) * DX
    if actual_sum > 0:
        result *= target_sum / actual_sum
    return result

    xseg = np.linspace(0, 1, K+1)

        x1, x2 = xseg[pos], xseg[pos+1]
        y1, y2 = u_vals[pos], u_vals[pos+1]

def smooth_max(h, beta=15.0):
    """Approximate max by log-sum-exp with temperature beta"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    corr *= dx
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * corr)))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 200  # Desired resolution
    # Use n+1 for K breakpoints
    K = 5           # Number of breakpoints for piecewise construction
    dx = 2.0 / n_points
    
    print(f"Initial run with K={K}")
    # Phase 1: DE to find a promising shape
    def objective_de(u_vals):
        # decode and compute C5
        h = decode_solution(u_vals, n_points)
        return evaluate_c5(h)
    
    bounds = [(0, 1) for _ in range(K+1)]
    
    # Limits
    max_iters_de = 1000
    popsize = 15
    de_res = differential_evolution(
        objective_de,
        bounds,
        maxiter=max_iters_de,
        popsize=popsize,
        seed=seed,
        strategy='best1bin',
        recombination=0.7) # Run DE for much of the time budget

    # After DE
    best_u = de_res.x
    best_h = decode_solution(best_u, n_points)
    best_c5 = evaluate_c5(best_h)
    print(f"DE improved to: {best_c5}") 
    
    # Phase 2: refine with PGD
    t_pgd_start = time.time()
    t_pgd_budget = budget_s * 0.2
    max_pgd_iters = 10000
    beta = 10.0
    
    # Helper function for gradient update with smoothing
    def interpolate_g(h):
        n = len(h)
        grad = np.zeros_like(h)
        for k in range(-n+1, n):
            k = k if k >=0 else -k  # Take only positive part of shift
            if k < n:
                # roll and compute partial gradient for h[i] contribution
                shift_h = np.roll(h, -k)
                for i in range(n):
                    grad[i] += (1 - shift_h[i]) * dx
        return grad
    
    for pgd_iter in range(max_pgd_iters):
        if time.time() - t_pgd_start > t_pgd_budget:
            break            
        # gradient wrt smooth max
        f_vals = []
        for k in range(-n_points + 1, n_points):
            shift = range(max(0, -k), min(n_points, n_points - k)) if k < 0 else range(max(0, k), n_points)
            f_vals.append(np.sum(h[shift] * (1 - h[np.arange(0, len(shift)) + k]))) 
        grad = interpolate_g(h)
        # Gradient step with learning rate
        lr = 0.02 / (1 + pgd_iter / 1000)
        h = h - lr * grad
        # Project onto feasible set
        h = project_onto_feasible(h, n_points, 1.0)
        # Check improvement
        new_c5 = evaluate_c5(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
        if pgd_iter % 1000 == 0:
            print(f"PGD iter {pgd_iter} C5 = {best_c5}") 
    
    # Final polishing with SLSQP
    t_polish_start = time.time()
    t_polish_budget = budget_s * 0.1
    # Warp back into optimization form
    h0 = best_h.copy()
    bounds_list = [(0, 1) for _ in range(n_points)]
    
    def objective_slsqp(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective_slsqp,
            h0,
            method='SLSQP',
            bounds=bounds_list,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12, 'disp': False}
        )
        if res.success and evaluate_c5(res.x) < best_c5:
            print("Refined by SLSQP")
            best_h = res.x
            best_c5 = evaluate_c5(res.x)
    except Exception as e:
        pass

    return (best_h.tolist(), best_c5, n_points)
