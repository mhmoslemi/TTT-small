import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 200
    n_breaks = 5  # Number of breakpoints in [0,1] (K in strategy)
    dx = 2.0 / n_points

    def decode_h(raw_heights):
        # Sort in descending order to ensure monotonic decrease
        sorted_heights = np.sort(raw_heights)[::-1]
        # Clamp to [0,1]
        sorted_heights = np.clip(sorted_heights, 0, 1)
        
        # Build piecewise linear symmetric vector
        x_half = np.linspace(0, 1, n_points//2)
        # Interpolate values to full interval [0,1]
        heights = np.interp(x_half, np.linspace(0, 1, n_breaks+1),
                            np.concatenate([[1], sorted_heights, [0]]))
        h_full = np.concatenate([heights, heights[-2::-1]])
        
        # Normalize to exact sum
        area_sum = np.sum(h_full) * dx
        h_full = h_full / area_sum * 1.0
        # Clip to [0,1] and renormalize
        h_full = np.clip(h_full, 0, 1)
        area_sum = np.sum(h_full) * dx
        h_full = h_full / area_sum * 1.0
        
        return h_full

    def evaluate_c5(h):
        corr = correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    # Define bounds for DE (raw heights in [0,1])
    bounds = [(0, 1) for _ in range(n_breaks)]

    def objective(params):
        h = decode_h(params)
        return evaluate_c5(h)

    # Run Differential Evolution on low-dimensional parameter space
    de_result = differential_evolution(
        objective,
        bounds,
        seed=seed,
        maxiter=int(100 + 300 * (budget_s / 600.0)),  # Scale iter based on budget
        popsize=15,
        disp=False,
        tol=1e-6
    )
    best_params = de_result.x.copy()
    best_h = decode_h(best_params)
    best_c5 = evaluate_c5(best_h)

    # Fine-tune using smooth surrogate and projected gradient descent
    def smooth_max(h, beta=20.0):
        corr = correlate(h, 1 - h, mode="full")
        corr *= dx
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * corr)))

    def smooth_max_grad(h, beta=20.0):
        corr = correlate(h, 1 - h, mode="full")
        corr *= dx
        exp_corr = np.exp(beta * corr)
        w = exp_corr / np.sum(exp_corr)
        # Compute gradient of corr * dx w.r.t. h
        grad = np.zeros_like(h)
        n = len(h)
        for k in range(len(corr)):
            i_start = max(0, k - n + 1)
            i_end = min(n, k)
            if k < n:
                i_start = 0
                i_end = k
            else:
                i_start = k - n + 1
                i_end = n - 1
            for i in range(i_start, i_end + 1):
                if i + k < n:
                    grad[i] += w[k] * dx * (1 - h[i+k] - h[i-k])
        return grad

    # Water filling projection onto simplex and box [0,1]
    def project_on_simplex_box(h, target_sum=1.0):
        h = np.clip(h, 0, 1)
        # Use water-filling with binary search for the lagrange multiplier lambda
        # such that sum of (h_i - lambda)_+ = target_sum
        h_sorted = np.sort(h)[::-1]
        h_cum = np.cumsum(h_sorted)        
        feasible_lambdas = []
        for i in range(len(h_sorted)):
            if (h_sorted[i] - (target_sum - h_cum[i])/(i+1)) >= 0:
                feasible_lambdas.append(h_sorted[i] - (target_sum - h_cum[i])/(i+1))

        if not feasible_lambdas:
            lambda_val = 0.0
        else:
            lambda_val = max(feasible_lambdas)
        
        h = np.clip(h - lambda_val, 0, 1)
        h_sum = np.sum(h) * dx
        h = h / h_sum * target_sum
        return h

    # Objectives for gradient descent
    def smooth_obj(h):
        return smooth_max(h, beta=15.0)
        
    def smooth_obj_grad(h):
        return smooth_max_grad(h, beta=15.0)

    # Descend with a diminishing stepsize
    h_curr = best_h.copy()
    max_iter = int(1000 * (budget_s / 600.0))
    eta = 0.01
    decay_factor = 1.0
    for iter_i in range(max_iter):
        if (time.time() - start_time) > 0.9 * budget_s:
            break
        grad = smooth_obj_grad(h_curr)
        # Clip to prevent explosion
        grad = np.clip(grad, -100, 100)
        h_curr = h_curr - eta * grad
        # Project onto feasible set
        h_curr = project_on_simplex_box(h_curr, 1.0)
        eta = max(1e-6, eta * 0.999)
        # Adjust learning rate gradually
        
    # Evaluate final result with exact max
    final_c5 = evaluate_c5(h_curr)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_curr.copy()

    # Optional final polish with SLSQP using a restart
    try:
        def constraint_eq(h):
            return np.sum(h) * dx - 1.0
        cons = {'type': 'eq', 'fun': constraint_eq}
        bounds_h = [(0, 1) for _ in range(n_points)]
        res = minimize(
            evaluate_c5,
            best_h,
            method='SLSQP',
            bounds=bounds_h,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            refined_h = res.x
            refined_c5 = evaluate_c5(refined_h)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    except Exception:
        pass

    return (best_h.tolist(), best_c5, n_points)
