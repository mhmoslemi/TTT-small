import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables to track the sum constraint
TARGET_SUM = None

def compute_corr_fft(h):
    """Compute full correlation using FFT."""
    n = len(h)
    # Reverse the array
    h_rev = h[::-1]
    # Create 1-h
    b = 1 - h
    # Full correlation using FFT
    corr = np.correlate(h_rev, b, mode='full')
    return corr

def c5_from_corr(corr, n):
    """Convert correlation to C5 value with proper scaling."""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT."""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto the constraint that sum = target_sum, with 0 <= x_i <= 1."""
    x = np.array(x, dtype=float)
    n = len(x)
    
    # Ensure target_sum is valid
    if target_sum < 0 or target_sum > n:
        raise ValueError("target_sum must be between 0 and n")
    
    # Check if already valid
    if np.isclose(np.sum(x), target_sum, atol=1e-10):
        return x
    
    # Binary search for lambda to shift
    def f(l):
        y = np.clip(x + l, 0, 1)
        return np.sum(y) - target_sum
    
    # Find lambda such that f(lambda) = 0
    a, b = -1.0, 1.0
    if f(a) > 0:  # target_sum too low
        a, b = b, a
    eps = 1e-12
    while abs(b - a) > eps:
        c = (a + b) / 2.0
        fc = f(c)
        if fc > 0:
            b = c
        else:
            a = c
    lambda_ = (a + b) / 2.0
    
    # Apply final projection
    x_proj = np.clip(x + lambda_, 0, 1)
    return x_proj

def random_pair_move(h):
    """Generate a valid pair transfer move. Returns (i, j, delta)."""
    n = len(h)
    # Pick two distinct indices
    i, j = np.random.choice(n, 2, replace=False)
    
    # Compute delta bounds
    delta_min = max(-h[j], h[i] - 1.0)
    delta_max = min(h[i], 1.0 - h[j])
    
    if delta_min < delta_max:
        delta = np.random.uniform(delta_min, delta_max)
        return i, j, delta
    else:
        return None, None, None

def guided_move(h, worst_lag):
    """Perform a deterministic guided move based on worst lag."""
    n = len(h)
    h = np.array(h, dtype=float)
    
    # Compute subgradient for the worst lag
    g = np.zeros(n)
    if worst_lag > 0:
        # g[i] = h[i - worst_lag] + h[i + worst_lag] if both exist
        g[:worst_lag] = 0.0
        g[n-worst_lag:] = 0.0
        if worst_lag < n:
            g[worst_lag:n-worst_lag] = h[:-worst_lag] + h[worst_lag:]
    elif worst_lag < 0:
        # Similar but with index differences handled properly
        # For simplicity: treat out-of-bounds as zero
        g[worst_lag:] = 0.0
        g[:n+worst_lag] = 0.0
        if worst_lag > -n:
            # Actually calculate the subgradient:
            # g[i] = h[i - worst_lag] + h[i + worst_lag]
            # where the first term is from previous indices and second from future
            h_shift = np.concatenate(([0]*(-worst_lag), h, [0]*worst_lag))
            g[:] = h_shift[:-worst_lag] + h_shift[-worst_lag:]
    else:
        # worst_lag == 0, which is whole overlap: 
        g = h
        # Note: do not modify anything

    # We compute the subgradient as above, but we can also compute an easily obtainable 1D case:
    # More direct: we shift indices properly for the subgradient of h[i]*h[i+|lag|] (or similar), which means objects in either h[i+lag] or h[i-lag] are used.
    # Better for this: work in array format and see where overlapping products occur
    # Let's simplify and assume that h[i] receives from the index that has lowest h[i+lag]

    # Also adjustment is: compute the lag orbit in absolute values -- this is simpler
    
    # 
    # Construct the correct subgradient leveraging the same difference should work:
    # if we are computing for the max of over all k of sum_i h[i] * h[i+k], 
    # then the derivative of that term w.r.t. h[i] is:
    #   if k == 0: h[i] 
    #   if k > 0: h[i-k] if i - k >= 0 else 0, + h[i+k] if i + k < len(h) else 0
    #   if k < 0: h[i+k] if i+k >= 0 else 0, + h[i-k] if i-k < len(h) else 0
    
    # Most rainbow inefficient recompute:
    corr = compute_corr_fft(h)
    # Reverse our corr array:
    corr_flipped = corr[::-1]
    # The loop to find the best correlation index
    start_lag = (len(corr) - 1) // 2
    max_lag_idx = np.argmax([corr_flipped[i] for i in range(len(corr))])
    actual_lag = max_lag_idx - start_lag
    
    # Use the overkill version of inputting the second one:
    # Instead, just work with i, compute just what's needed:
    # The gradient of the sum of h[i]*h[i+actual_lag] w.r.t. h[i] is:
    #   if actual_lag >= 0: h[i - actual_lag] if valid else 0 + h[i + actual_lag] if valid else 0  
    #   if actual_lag < 0: h[i - actual_lag] if valid else 0 + h[i + actual_lag] if valid else 0
    # 归纳为最终最优子结构: for each index, compute derivatives of that term,
    # and sum them up. This maps directly to:
    # B[i] = h[i+actual_lag] + h[i-actual_lag] if actual_lag >= 0  
    # = h[i+actual_lag] + h[i + (-actual_lag)] if actual_lag < 0
    # Except we have to mask out boundary values
    
    # We recompute what the true worst_lag is as it is an integer in [-n+1, n-1]
    # Calculate the full convolutioned form, i.e., corr returns len=2n-1
    # The best correlation index is [i], corresponds to lag=i-(n-1)
    corr_pos = np.argmax(corr)
    best_lag = corr_pos - (n-1)
    
    # Let's now find the maximal gradient locations by same procedure:
    # subgradient w.r.t h_i of correlation with some lag:
    # This is actually the "connection" from
    # the previous loop into:
    # i gets +h_prev[i+lag] +h_next[i-lag] in terms of products,
    # And the total gradient covers it all
    
    # Better:
    # Build subgradient g[i] as the total derivative from all lags.
    g = np.zeros(n)
    # For each lag position i-lag to i+lag" contribute:
    for i in range(n):
        for k in range(-n+1, n):  # Change to list within limits?
            if i+k >= 0 and i+k < n:
                g[i] += h[i+k]  # From identity h[i] * h[j] where j = i+k,
        
    # [But more precisely just we use outer presence only for i's own contribution]
    # For a fixed lag, e.g., +1:
    # We have i*1ün(i+1), and derivatives w.r.t h_i and h_{i+1} are 1,1 respectively
    
    # Simpler and accurate:
    # gradient vector for h[i] of sum_{all k} h[i] h[i+k] is 
    # sum_{k} (h[i+k] + h[i-k]) 
    # if we have the term at indices i, i+k,
    # and their partial derivatives give respective values h[i+k] and h[i-k].
    g_shift = np.zeros(n)
    # Fallback: they don't use out of bounds indices:
    # So for given lag, sum over (h[i] h[i+lag] over all i),
    # d/d h_i of that sum gives h[i+lag] (alias from forward), and h[i-lag] (from backward)
    # But logistic: if lag>0 then i-lag < i, so always valid in a way, and vice versa
    
    # Let's compute:
    g = np.zeros(n)
    if best_lag >= 0:
        g[best_lag:] += h[:-best_lag]  # Add h[i-best_lag]
        g[:-best_lag] += h[best_lag:]  # Add h[i+best_lag]
    else:
        g[best_lag:] += h[-best_lag:]  # Add h[i-best_lag]
        g[-best_lag:] += h[:best_lag]  # Add h[i+best_lag]
    
    # Determine where to transfer mass to reduce gradient: 
    # we find min g[i] and max g[i] in valid range
    i_min = np.argmin(g)
    i_max = np.argmax(g)
    
    # Create the move:
    delta = min(h[i_min], 1.0 - h[i_max])
    if delta <= 0:
        return h, worst_lag  # No move possible
    
    # Introduce the transfer
    new_h = h.copy()
    new_h[i_min] -= delta
    new_h[i_max] += delta
    return new_h, worst_lag

def anneal(h0, run_seconds, rng_seed):
    """Run simulated annealing with both guided and random moves."""
    np.random.seed(rng_seed)
    n = len(h0)
    h = np.array(h0, dtype=float)
    h = project_to_sum_box(h, TARGET_SUM)
    
    # Parameters
    T0 = 0.005
    Tf = 1e-6
    T_current = T0
    decay_rate = (T0 - Tf) / run_seconds if run_seconds > 0 else 1.0
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Main annealing loop
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < run_seconds:
        # Determine mode:
        mode = iteration % 10
        if mode == 0:
            # Guided movement
            corr = compute_corr_fft(h)
            corr_flipped = corr[::-1]
            idx_max = np.argmax(corr_flipped)
            lag = idx_max - (n - 1)
            h_new, lag_use = guided_move(h, lag)
            if h_new is not None:
                h = h_new
            else:
                h = h  # Keep original
        else:
            # Random move
            i, j, delta = random_pair_move(h)
            if i is not None:
                new_h = h.copy()
                new_h[i] -= delta
                new_h[j] += delta
                h = new_h
        
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Compute new objective
        c5_new = evaluate_c5(h)
        
        # Accept with Metropolis criterion
        if c5_new <= best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            T_current = T0
        else:
            delta_energy = c5_new - best_c5
            if np.random.rand() < np.exp(-delta_energy / T_current):
                # Accept even though worse
                pass  # But keep current h and c5
        
        # Adaptive cooling
        T_current = max(Tf, T_current - decay_rate)
        iteration += 1
    
    return best_h, best_c5

def local_refine(h):
    """Local refinement using SLSQP."""
    # Set up the simple constraints
    bounds = [(0, 1) for _ in range(len(h))]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Optimizer
    res = minimize(
        objective, h, method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12}
    )
    if res.success:
        return res.x
    else:
        return h

def constraint_eq(h):
    return np.sum(h) - TARGET_SUM

def objective(h):
    return evaluate_c5(h)
