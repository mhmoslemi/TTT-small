import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Phase 1: Use global optimization with better parameters and strategic restarts
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Use multiple evolutionary strategies with different parameters
    strategies = ['best1bin', 'best2bin', 'rand1bin', 'rand2bin', 'currenttobestbin']
    param_sets = [
        (1000, 15, 0.8, 0.9),  # Strategy 1
        (800, 20, 0.9, 0.8),   # Strategy 2  
        (700, 12, 0.7, 0.95),  # Strategy 3
        (600, 25, 0.95, 0.75), # Strategy 4
        (900, 18, 0.85, 0.85)  # Strategy 5
    ]
    
    for i, (maxiter, popsize, mutation, recomb) in enumerate(param_sets):
        if (time.time() - start_time) > budget_s:
            break
            
        strategy = strategies[i % len(strategies)]
        try:
            de_res = differential_evolution(
                objective, 
                bounds, 
                seed=seed + i, 
                maxiter=maxiter,
                popsize=popsize,    
                mutation=mutation,  
                recombination=recomb,   
                strategy=strategy
            )
            if de_res.success:
                h_candidate = de_res.x
                # Normalize to satisfy constraint
                dx = 2.0 / n_points
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue
    
    # Phase 2: Local search from good global points
    local_restart_bound = 100
    
    for restart in range(local_restart_bound):
        if (time.time() - start_time) > budget_s:
            break
            
        # Start with small random perturbations around previous best
        new_h = best_h + np.random.normal(0, 0.015, n_points)
        new_h = np.clip(new_h, 0, 1)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 150, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue

    # Phase 3: Additional heuristics to further exploit structure
    # Use a modified objective function encouraging miminization of max overlap
    
    # Try improving best solution using alternating pattern
    if (time.time() - start_time) < budget_s * 0.7:
        try:
            # Create alternating 1s and 0s pattern with fixed even pattern  
            alt_h = np.zeros(n_points)
            alt_h[0::2] = 1  # Set even indices to 1
            dx = 2.0 / n_points
            
            # Adjust to satisfy constraint
            scale_factor = 1.0 / (np.sum(alt_h) * dx)
            alt_h = alt_h * scale_factor
            alt_c5 = evaluate_c5(alt_h)
            if alt_c5 < best_c5:
                best_h = alt_h.copy()
                best_c5 = alt_c5
                
            # Try tent-like pattern
            x = np.linspace(0, 2, n_points)
            tent_pattern = np.maximum(0, 1 - np.abs(x - 1) * 2)
            mixed_h = 0.6 * best_h + 0.4 * tent_pattern
            
            # Normalize
            scale_factor = 1.0 / (np.sum(mixed_h) * dx)
            mixed_h = mixed_h * scale_factor
            
            res = minimize(
                objective, 
                mixed_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 150, 'ftol': 1e-13, 'gtol': 1e-13}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
                    
        except:
            pass

    return (best_h.tolist(), best_c5, n_points)
