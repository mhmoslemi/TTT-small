import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient(h, d_star):
    """Compute subgradient for shift d_star"""
    n = len(h)
    g = np.zeros(n)
    if d_star > 0:
        # h[i] * (1 - h[i + d_star])
        g[:-d_star] = 1.0 - h[d_star:]
    elif d_star < 0:
        # h[i] * (1 - h[i + d_star])
        g[-d_star:] = 1.0 - h[:-d_star]
    else:
        # For d_star == 0, gradient is 0
        pass
    return g

def binary_optimal(n, seed=42):
    np.random.seed(seed)
    dx = 2.0 / n
    # Variables
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    # Mass constraint
    mass = cp.sum(x) == n / 2
    # Bounds
    bounds = [x >= 0, x <= 1]
    # For each shift d ≠ 0, compute N_d = sum x_i - sum x_i * x_{i+d}
    # auxiliary variables w_{i,d} for x_i * x_{i+d} and then N_d = sum x_i - sum w
    constraints = [mass] + bounds
    # All shifts except d=0
    all_shifts = list(range(-n+1, 0)) + list(range(1, n))
    
    # Add McCormick constraints for each shift
    w_vars = {}
    for d in all_shifts:
        w_vars[d] = cp.Variable(n, boolean=True)
        # Restrict w_i,d to valid indices
        if d > 0:
            valid_indices = slice(0, n-d)
        else:
            valid_indices = slice(-d, n)
        # w_{i,d} <= x_i
        constraints.append(w_vars[d][valid_indices] <= x[valid_indices])
        # w_{i,d} <= x_{i+d}
        constraints.append(w_vars[d][valid_indices] <= x[valid_indices+d])
        # w_{i,d} >= x_i + x_{i+d} - 1
        constraints.append(w_vars[d][valid_indices] >= x[valid_indices] + x[valid_indices+d] - 1)
        # summing over valid indices, N_d = sum x_i - sum w_i,d
        # i.e., N_d = sum_i x_i - sum_{i+d valid} w_{i+d,d}
        # H = (identity + shift_by_d (with zeros for out-of-bounds))
        # Get inner product: sum_i x_i - sum_{i+d valid} w_{i+d,d}
        # But to make sure it matches correctly, model N_d
    
    # Alternative do this for entire set of shifts and add N_d <= t
    N_d_list = []
    for d in all_shifts:
        if d > 0: # x_i * x_{i+d}
            valid_indices = slice(0, n - d)
            N_d = cp.sum(x[valid_indices]) - cp.sum(w_vars[d][valid_indices])
        elif d < 0: # x_i * x_{i+d}
            valid_indices = slice(-d, n)
            N_d = cp.sum(x[valid_indices]) - cp.sum(w_vars[d][valid_indices])
        else:
            # skip i+d + d = i case, but this is not valid round
            # currently d ≠ 0, so skip
            continue
        N_d_list.append(N_d)
        # Add constraint N_d <= t
        constraints.append(N_d <= t)
    
    # Set up problem
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    # Try CBC first
    try:
        prob.solve(solver=cp.CBC, verbose=False)
        x_opt = x.value
        t_opt = t.value
    except:
        try:
            prob.solve(solver=cp.GLPK_MI, verbose=False)
            x_opt = x.value
            t_opt = t.value
        except:
            try:
                prob.solve(solver=cp.ECOS_BB, verbose=False)
                x_opt = x.value
                t_opt = t.value
            except:
                # fallback to heuristics if MILP solving fails entirely
                x_opt = np.array([1 if i < n//2 else 0 for i in range(n)]).astype(float)
                t_opt = 0  # placeholder, not relied upon
    
    return np.array(x_opt), t_opt

def refine_continuous(h0, time_left, seed=42):
    np.random.seed(seed)
    n = len(h0)
    dx = 2.0 / n
    best_h = h0.copy()
    best_val = evaluate_c5(best_h)
    
    # Gradient descent settings
    max_iter = 1000
    step_size = 0.5
    scale_factor = n / 2.0 / np.sum(h0)
    h = h0 * scale_factor
    
    # avoid np.max in a loop for time concerns
    corrs = correlate(h, 1 - h, mode="full")
    current_max = np.max(corrs)
    
    iter_count = 0
    start_time = time.time()
    
    while iter_count < max_iter and (time.time() - start_time) < time_left:
        # compute all overlaps
        corr = correlate(h, 1 - h, mode="full")
        # Get shifts with max overlap
        max_val = np.max(corr)
        # indices of maximum to access D_max            D = 1:2n-1
        # But need to index from n-1 up to 2n-2
        max_indices = np.where(corr == max_val)[0]
        max_shifts = max_indices - (n-1)
        # compute gradient using first element
        d_star = max_shifts[0]
        
        g = subgradient(h, d_star)
        # update step
        h_new = h - step_size * g
        h_new = np.clip(h_new, 0, 1)
        # scale to sum constraint
        scale_factor = n / 2.0 / np.sum(h_new)
        h_new = h_new * scale_factor
        
        if (evaluate_c5(h_new) < best_val):
            best_h = h_new.copy()
            best_val = evaluate_c5(h_new)
        
        h = h_new.copy()
        iter_count += 1

    # Optional final polish with SLSQP
    try:
        bounds = [(0, 1) for i in range(n)]
        def obj(h):
            return evaluate_c5(h)
        def constraint_eq(h):
            return np.sum(h) * dx - 1.0
        cons = {'type': 'eq', 'fun': constraint_eq}

        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 50, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            scale_factor = 1.0 / (np.sum(h_final) * dx)
            h_final = h_final * scale_factor
            new_val = evaluate_c5(h_final)
            if new_val < best_val:
                best_h = h_final.copy()
                best_val = new_val
    except:
        pass

    return best_h, best_val
