import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2  # Exactly 21 ones for n=42
    
    best_c5 = evaluate_c5(h0)
    best_h = h0.copy()
    
    # 1. Try solving MILP first
    try:
        # Create the MILP model
        h_vars = cp.Variable(n_points, boolean=True)
        p_vars = cp.Variable((n_points, n_points), boolean=True)  # p[i,l] = h[i] AND h[i+l]
        M = cp.Variable(integer=True, nonneg=True)
        
        # Constraints
        constraints = []
        
        # Sum constraint
        constraints.append(cp.sum(h_vars) == target_sum)
        
        # McCormick constraints for p_vars
        for l in range(1, n_points):
            for i in range(n_points - l):
                # p[i,l] ≤ h[i]
                constraints.append(p_vars[i, l] <= h_vars[i])
                # p[i,l] ≤ h[i+l]
                constraints.append(p_vars[i, l] <= h_vars[i + l])
                # p[i,l] ≥ h[i] + h[i+l] - 1
                constraints.append(p_vars[i, l] >= h_vars[i] + h_vars[i + l] - 1)
        
        # Overlap counting constraints
        for l in range(1, n_points):
            # S_l = sum of p[i,l]
            S_l = cp.sum(p_vars[:n_points - l, l])
            # A_l^+ = sum of h[i] over valid i
            A_l_plus = cp.sum(h_vars[:n_points - l])
            # A_l^- = sum of h[i] over i >= l
            A_l_minus = cp.sum(h_vars[l:])
            
            # For the maximum overlap count, all these must be bounded by M
            constraints.append(A_l_plus - S_l <= M)
            constraints.append(A_l_minus - S_l <= M)
            
        # Objective
        objective = cp.Minimize(M)
        prob = cp.Problem(objective, constraints)
        
        # Attempt solving with time limit
        remaining_budget = budget_s - (time.time() - start_time)
        if remaining_budget > 2:
            prob.solve(solver=cp.CBC, time_limit=remaining_budget - 1)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INFEASIBLE]:
                h_binary = h_vars.value.astype(int)
                # Convert binary to float (normalized)
                h_float = h_binary.astype(float)
                h_float /= np.sum(h_float) * dx
                c5 = evaluate_c5(h_float)
                if c5 < best_c5:
                    best_h = h_float
                    best_c5 = c5
    except Exception as e:
        pass

    # 2. Continuous optimization polishing
    if best_c5 >= 0.379:  # Only polish if needed
        remaining_budget = budget_s - (time.time() - start_time)
        
        # Multiple restarts for SLSQP optimization
        def objective_funct(h):
            return evaluate_c5(h)
            
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_eq(h):
            return np.sum(h) * dx - 1.0
            
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        max_restarts = 200
        for restart in range(max_restarts):
            if (time.time() - start_time) > budget_s - 2:
                break
            try:
                # Start from current best, with perturbation
                h_start = best_h + np.random.normal(0, 0.02, n_points)
                h_start = np.clip(h_start, 0, 1)
                scale = 1.0 / (np.sum(h_start) * dx)
                h_start = h_start * scale
                
                res = minimize(
                    objective_funct,
                    h_start,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 50, 'ftol': 1e-12, 'gtol': 1e-12}
                )
                if res.success:
                    h_candidate = res.x
                    scale1 = 1.0 / (np.sum(h_candidate) * dx)
                    h_candidate = h_candidate * scale1
                    c5_candidate = objective_funct(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
            except Exception as e:
                continue

    # 3. Fallback simulated annealing if needed
    if best_c5 >= 0.379:
        remaining_budget = budget_s - (time.time() - start_time)
        if remaining_budget > 2:
            # Annealing parameters
            temp = 1.0
            cooling_rate = 0.995
            min_temp = 1e-6
            max_iter = int(1e5)
            
            h_current = best_h.copy()
            c5_current = evaluate_c5(h_current)
            
            # For SA step, we'll work with binary form instead of float
            h_bin = (h_current >= 0.5).astype(int)
            if np.sum(h_bin) != target_sum:
                # adjust to make sure it's exactly target_sum
                if np.sum(h_bin) > target_sum:
                    # flip some 1s to 0s (random)
                    indices_ones = np.where(h_bin == 1)[0]
                    indices_to_flip = np.random.choice(indices_ones, size=int(np.sum(h_bin) - target_sum), replace=False)
                    h_bin[indices_to_flip] = 0
                else:
                    # flip some 0s to 1s (random)
                    indices_zeros = np.where(h_bin == 0)[0]  
                    indices_to_flip = np.random.choice(indices_zeros, size=int(target_sum - np.sum(h_bin)), replace=False)
                    h_bin[indices_to_flip] = 1
                h_current = h_bin.astype(float)
                h_current /= np.sum(h_current) * dx
                c5_current = evaluate_c5(h_current)
            
            for iteration in range(max_iter):
                if (time.time() - start_time) > budget_s - 2:
                    break
                    
                # Generate neighbor
                # pick one 1 and one 0 to swap
                indices_ones = np.where(h_bin == 1)[0]
                indices_zeros = np.where(h_bin == 0)[0]
                if len(indices_ones) > 0 and len(indices_zeros) > 0:
                    i = np.random.choice(indices_ones)  
                    j = np.random.choice(indices_zeros)
                    h_new = h_bin.copy()
                    h_new[i] = 0
                    h_new[j] = 1
                    
                    # compute C5 of new h
                    h_new_float = h_new.astype(float)
                    # normalize to keep constraint
                    h_new_float *= 1 / (np.sum(h_new_float) * dx)
                    c5_new = evaluate_c5(h_new_float)
                    
                    # Accept or reject
                    delta = c5_new - c5_current
                    if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                        h_bin = h_new
                        h_current = h_new_float
                        c5_current = c5_new
                        
                        if c5_current < best_c5:
                            best_c5 = c5_current
                            best_h = h_current.copy()
                    
                    temp *= cooling_rate
                    if temp < min_temp:
                        temp = min_temp
                    
            # Final polishing of the best binary solution
            h_bin_final = (best_h >= 0.5).astype(int)
            h_final_float = h_bin_final.astype(float)
            h_final_float /= np.sum(h_final_float) * dx
            c5_final = evaluate_c5(h_final_float)
            if c5_final < best_c5:
                best_h = h_final_float
                best_c5 = c5_final
    
    return (best_h.tolist(), best_c5, n_points)
