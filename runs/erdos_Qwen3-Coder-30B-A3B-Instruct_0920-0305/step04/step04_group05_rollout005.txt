import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use x0 as starting point
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Scipy's differential evolution is generally good at avoiding local minima
    # and we'll do a few rounds with it
    
    # Coordinate-descent like approach: optimize one variable at a time
    def local_optimize_one_var(h, var_idx):
        original = h[var_idx]
        dx = 2.0 / n_points
        best_h_temp = h.copy()
        best_c5_temp = evaluate_c5(h)
        
        # Try a few values around current
        for val in np.linspace(max(0, original-0.2), min(1, original+0.2), 10):
            h_test = h.copy()
            h_test[var_idx] = val
            
            # Normalize to satisfy constraint
            scale = 1.0 / (np.sum(h_test) * dx)
            h_test = h_test * scale
            
            c5_test = evaluate_c5(h_test)
            if c5_test < best_c5_temp:
                best_c5_temp = c5_test
                best_h_temp = h_test.copy()
        
        return best_h_temp, best_c5_temp
    
    # First use a global optimization approach to explore
    try:
        res = differential_evolution(
            objective, 
            bounds, 
            maxiter=100, 
            popsize=20,
            seed=seed,
            atol=1e-12,
            tol=1e-12,
            disp=False

            h_candidate = res.x.copy()
            scale = 1.0 / (np.sum(h_candidate) * (2.0/n_points))
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except Exception as e:
        pass

    remaining_time = budget_s - (time.time() - start_time)

                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception as e:
            pass

    return (best_h.tolist(), best_c5, n_points)
