import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def decode_h(u_vals, n_points):
    """Decode parametric heights into full vector h (symmetric, monotonic decreasing)."""
    # u_vals: unordered values, sorted descending to get heights
    heights = np.sort(u_vals)[::-1]
    # Ensure non-negative and bounded
    heights = np.clip(heights, 0, 1)
    
    # Now build piecewise linear function on [0,1] and mirror
    K = len(heights) - 1
    # Break points in [0,1], including 0 and 1
    x_break = np.linspace(0, 1, K+1)
    # Linear interpolation to full resolution
    h = np.interp(np.linspace(0, 1, n_points//2), x_break, heights)
    # Mirror
    h = np.concatenate((h, h[::-1]))
    return h

def compute_c5(h, n_points):
    """Compute C5 using full correlation and max."""
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h_to_feasible(h, n_points):
    """Project h to satisfy integral = 1 and bounds [0,1]."""
    # Ensure it's within [0,1]
    h = np.clip(h, 0, 1)
    dx = 2.0 / n_points
    scaled_sum = np.sum(h) * dx
    if abs(scaled_sum - 1.0) > 1e-10:
        # Scale to integral = 1 by clipping and renormalizing
        total = np.sum(h) * dx
        h = h / total  # Scale to unit integral
        h = np.clip(h, 0, 1)
    return h

def smooth_max_objective(h, n_points, beta=15):
    """Smooth approximation to max using log-sum-exp."""
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode="full")
    # Apply smoothing with beta parameter
    corr_scaled = corr * dx
    exp_vals = np.exp(beta * corr_scaled)
    log_sum = np.log(np.sum(exp_vals))
    return log_sum / beta

def gradient_of_smooth_max(h, n_points, beta=15):
    """Compute gradient of smooth max with respect to h."""
    # Use correlation matrix to compute current values
    dx = 2.0 / n_points
    # f_k = sum h_i (1 - h_{i+k})
    corr = correlate(h, 1 - h, mode="full")
    corr_scaled = corr * dx
    
    # Weights: w_k = exp(beta f_k) / sum exp(beta f_l)
    exp_vals = np.exp(beta * corr_scaled)
    weights = exp_vals / np.sum(exp_vals)
    
    # Gradient 
    grad = np.zeros_like(h)
    n = len(h)
    # This computes the sum over all k of w_k * df_k/dh_i
    for k in range(-n+1, n):
        # For shift k, df_k/dh_i
        # = dx * (1 - h_{i+k} - h_{i-k})
        if k == 0:
            # if k == 0, only 1 - h_{i+0} = 1 - h_i
            local_grad = -dx * np.ones_like(h)
        else:
            # construct shift vectors
            if k > 0:
                # left shift h to match x+k (i.e., h[i] -> h[i+k])
                shift_backward = np.concatenate([np.zeros(k), h[:-k]])
                shift_forward = np.concatenate([h[k:], np.zeros(k)])
            else:  # k < 0
                k_abs = -k
                shift_forward = np.concatenate([np.zeros(k_abs), h[:-k_abs]])
                shift_backward = np.concatenate([h[k_abs:], np.zeros(k_abs)])
            local_grad = -dx * (shift_backward + shift_forward)
        
        # Add to cumulative gradient weighted by w_k
        grad += weights[k + n - 1] * local_grad
    
    return grad

def waterfill_projection(h, dx):
    """Projection onto the constraints sum(h)*dx=1 and h in [0,1]."""
    h = np.clip(h, 0, 1)
    target_sum = 1.0
    current_sum = np.sum(h) * dx
    if abs(current_sum - target_sum) < 1e-10:
        return h

    # Waterfilling algorithm
    b = h.copy()
    # Use binary search to find lambda such that clipped sum = target_sum
    low, high = -1.0, 1.0
    while high - low > 1e-10:
        mid = (low + high) / 2.0
        clipped = np.clip(b - mid, 0, 1)
        summed = np.sum(clipped) * dx
        if summed > target_sum:
            low = mid
        else:
            high = mid
    lambda_val = (low + high) / 2.0
    projected = np.clip(b - lambda_val, 0, 1)
    return projected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # Set up dimensions
    n_points = 200
    n_params = 5  # number of breakpoints in [0,1] for symmetric shape
    # Manual parameter vector with initial values
    initial_u_vals = np.array([1.0] + [0.5] * (n_params - 1))
    
    # Define bounds: u in [0,1] for each parameter
    bounds = [(0.0, 1.0)] * n_params
    
    def objective(u_vals):
        h = decode_h(u_vals, n_points)
        h = project_h_to_feasible(h, n_points)
        c5 = compute_c5(h, n_points)
        return c5
    
    # Phase 1: Global DE
    de_budget = 0.70 * budget_s
    t_left = de_budget
    
    res_de = differential_evolution(
        objective,
        bounds,
        maxiter=100,
        popsize=15,
        seed=seed,
        disp=False
    )
    
    best_u = res_de.x
    best_c5 = res_de.fun

    # Phase 2: Smooth refinement with projected gradient descent
    t_left = 0.20 * budget_s
    h = decode_h(best_u, n_points)
    h = project_h_to_feasible(h, n_points)
    beta = 15
    eta = 0.1
    eta_decay = 0.98
    tol = 1e-6
    max_iter = 500
    t_stop = time.time() + t_left
    
    for i in range(max_iter):
        if time.time() > t_stop:
            break
        # Compute gradient
        grad = gradient_of_smooth_max(h, n_points, beta)
        # Gradient step
        h_new = h - eta * grad
        # Project
        dx = 2.0 / n_points
        h_new = waterfill_projection(h_new, dx)
        h = h_new
        # Check improvement
        c5_new = compute_c5(h, n_points)
        if abs(c5_new - best_c5) < tol:
            break
        if c5_new < best_c5:
            best_c5 = c5_new
        eta *= eta_decay

    # Final polishing step - SLSQP local search on full h
    t_left = 0.10 * budget_s
    final_time = time.time() + t_left

    def obj_full(h_full):
        return compute_c5(h_full, n_points)
    
    def constraint_full(h_full):
        dx = 2.0 / n_points
        return np.sum(h_full) * dx - 1.0

    cons_full = {'type': 'eq', 'fun': constraint_full}
    bounds_full = [(0, 1)] * n_points

    # starting from h and normalize
    h_start = project_h_to_feasible(h, n_points)
    
    try:
        res = minimize(
            obj_full,
            h_start,
            method='SLSQP',
            bounds=bounds_full,
            constraints=cons_full,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12},
            callback=lambda x: None
        )
        if res.success:
            h_final = project_h_to_feasible(res.x, n_points)
            final_c5 = compute_c5(h_final, n_points)
            if final_c5 < best_c5:
                h = h_final
                best_c5 = final_c5
    except Exception:
        pass

    return (h.tolist(), best_c5, n_points)
