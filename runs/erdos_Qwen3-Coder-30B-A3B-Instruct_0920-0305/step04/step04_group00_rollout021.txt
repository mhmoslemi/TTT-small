import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

        cp=1: constant height 0.5 across domain
        """
        h = np.zeros(n)
        cp = max(0, min(1, cp))  # Clip to valid range
        H = 1.0 / (1.0 + cp)
        
        # Central plateau from 1-cp to 1+cp
        left_edge = int((1 - cp) * n / 2)
        right_edge = int((1 + cp) * n / 2)
        
        # Make sure we don't overflow
        left_edge = max(0, left_edge)
        right_edge = min(n, right_edge)
        
        # Set plateau values
        if left_edge < right_edge:
            h[left_edge:right_edge] = H
            
        # Set edge values (triangular drops)
        # Left triangular part
        if left_edge > 0:
            x_vals = np.linspace(0, 1, left_edge)
            h[:left_edge] = H * (1 - x_vals)
            
        # Right triangular part
        if right_edge < n:
            x_vals = np.linspace(1, 0, n - right_edge)
            h[right_edge:] = H * x_vals
            
        return h

        h = triangular_shape(cp, n_points)
