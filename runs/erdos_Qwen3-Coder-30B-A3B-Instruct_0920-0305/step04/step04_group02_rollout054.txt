def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def estimate_max_overlap(h):
    """Estimate maximum overlap from a given h, without dx"""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr)

def subgradient(h, d_star):
    """Compute subgradient for the overlap with shift d_star"""
    n = len(h)
    g = np.zeros(n)
    if d_star > 0:
        # h_i * (1 - h_{i+d_star}) for i < n-d_star
        g[:n-d_star] += 1
        g[d_star:] -= h[:n-d_star]
    elif d_star < 0:
        # h_i * (1 - h_{i+d_star}) for i > -d_star
        g[-d_star:] += 1
        g[:-d_star] -= h[-d_star:]
    else:
        # d_star == 0
        g[:] = 1 - h[:]
    return g

def refine_continuous(h0, time_left_s):
    """Refine the binary solution with floating point gradient descent"""
    n = len(h0)
    dx = 2.0 / n
    best = np.array(h0, dtype=float)
    best_val = evaluate_c5(best)
    start = time.time()
    
    max_iter = 200
    iter_count = 0
    while iter_count < max_iter and (time.time() - start) < time_left_s:
        # Compute all overlaps to find max
        corr = correlate(best, 1 - best, mode="full")
        max_idx = np.argmax(corr)
        d_star = max_idx - (n - 1)
        
        # Compute subgradient
        g = subgradient(best, d_star)
        
        # Step size
        alpha = 0.5
        while True:
            h_new = best - alpha * g
            h_new = np.clip(h_new, 0, 1)
            # Fix the integral constraint
            scale = (n/2) / np.sum(h_new)
            h_new *= scale
            
            new_val = evaluate_c5(h_new)
            if new_val < best_val - 1e-10:
                best = h_new
                best_val = new_val
                break
            else:
                alpha /= 2
                if alpha < 1e-8:
                    break
        
        iter_count += 1
        if (time.time() - start) > 0.9*time_left_s and iter_count > 20:
            break
            
    # Final polish with SLSQP if we have time
    if (time.time() - start) < 0.8 * time_left_s:
        try:
            def objective(h):
                return evaluate_c5(h)
            cons = {'type': 'eq', 'fun': lambda h: np.sum(h) * dx - 1}
            bounds = [(0, 1) for _ in range(n)]
            
            res = minimize(objective, best, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 50, 'ftol': 1e-12, 'gtol': 1e-12})
            if res.success:
                final_val = evaluate_c5(res.x)
                if final_val < best_val:
                    return res.x, final_val
        except:
            pass
    
    return best, best_val
