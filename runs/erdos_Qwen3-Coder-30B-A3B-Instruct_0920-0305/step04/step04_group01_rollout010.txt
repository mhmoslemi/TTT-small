import numpy as np
import time
from scipy.optimize import minimize

# Global variable for constraint
TARGET_SUM = None

def evaluate_c5_fft(h):
    """Compute C5 using FFT for fast correlation"""
    n = len(h)
    if n == 0:
        return 0.0
    # Reverse h
    h_rev = h[::-1]
    b = 1.0 - h
    L = 2 * n - 1
    # Compute FFT
    hf = np.fft.rfft(h_rev, n=L)
    bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(hf * bf, n=L)
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box [0,1]^n and sum constraint"""
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x
    
    # Simple adjustment when sum differs
    diff = target_sum - current_sum
    if abs(diff) < 1e-12:
        return x
    
    # Distribute adjustment
    # Find indices that can be adjusted
    free_indices = np.where((x > 0) & (x < 1))[0]
    if len(free_indices) == 0:
        # All are at bounds, scale to sum
        x = x * (target_sum / current_sum)
        return np.clip(x, 0, 1)
    
    idx = free_indices[0]
    if diff > 0:
        # Need to increase values
        delta = min(diff, 1 - x[idx])
        x[idx] += delta
        diff -= delta
        # Spread remaining
        while abs(diff) > 1e-12 and len(free_indices) > 1:
            idx = free_indices[0]
            if x[idx] < 1:
                delta = min(diff, 1 - x[idx])
                x[idx] += delta
                diff -= delta
            free_indices = free_indices[1:]
    else:
        # Need to decrease values
        delta = min(-diff, x[idx])
        x[idx] -= delta
        diff += delta
        # Spread remaining
        while abs(diff) > 1e-12:
            idx = free_indices[0]
            if x[idx] > 0:
                delta = min(-diff, x[idx])
                x[idx] -= delta
                diff += delta
            free_indices = free_indices[1:]
    
    return np.clip(x, 0, 1)

def simulated_annealing(h_initial, target_sum, time_limit, seed):
    """Run simulated annealing for C5 optimization"""
    np.random.seed(seed)
    n = len(h_initial)
    h = np.copy(h_initial)
    h = project_to_sum_box(h, target_sum)
    cur_c = evaluate_c5_fft(h)
    best_h = h.copy()
    best_c = cur_c

    T0 = cur_c * 0.01
    Tmin = 1e-8
    start = time.time()
    iter_count = 0
    last_improve = 0
    max_iter = 50000  # Max iters to control time for single run
    
    # For deterministic moves
    deterministic_freq = 0.1
    
    while time.time() - start < time_limit and iter_count < max_iter:
        iter_count += 1
        # Determine temperature
        elapsed = time.time() - start
        progress = min(1.0, elapsed / time_limit)
        T = T0 * (1 - progress) + Tmin * progress
        
        # Random move
        i, j = np.random.choice(n, size=2, replace=False)
        delta_max = min(h[i], 1 - h[j])
        delta_min = -min(1 - h[i], h[j])
        
        if delta_max <= delta_min:
            continue
            
        delta = np.random.uniform(delta_min, delta_max)
        
        # Adjust h
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # Evaluate
        c_new = evaluate_c5_fft(h_new)
        
        # Acceptance probability
        if c_new < cur_c or np.random.rand() < np.exp(-(c_new - cur_c) / (T + 1e-15)):
            h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                last_improve = iter_count
        
    return best_h, best_c

def constraint_eq(h):
    """Sum constraint function"""
    return np.sum(h) - TARGET_SUM

def local_refine(h0, target_sum, max_iter=1000):
    """Refine solution using local optimization"""
    try:
        bounds = [(0, 1) for _ in range(len(h0))]
        cons = {'type': 'eq', 'fun': constraint_eq}
        res = minimize(evaluate_c5_fft, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.success and 'fun' in res:
            h_ref = res.x
            h_ref = np.clip(h_ref, 0, 1)
            h_ref = project_to_sum_box(h_ref, target_sum)
            c_ref = evaluate_c5_fft(h_ref)
            return h_ref, c_ref
    except Exception:
        pass
    return h0, evaluate_c5_fft(h0)
