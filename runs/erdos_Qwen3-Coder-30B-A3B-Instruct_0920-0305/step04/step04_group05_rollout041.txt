import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(x, target_sum):
    """Project x onto feasible set: [0,1]^n and sum = target_sum"""
    y = np.clip(x, 0, 1)
    current_sum = np.sum(y)
    if abs(current_sum - target_sum) < 1e-9:
        return y
    
    # Simple iterative scaling approach
    for _ in range(50):
        diff = current_sum - target_sum
        if abs(diff) < 1e-6:
            break
        if diff > 0:  # Too much sum
            free = y > 0
            if not free.any():
                break
            decrement = diff / np.sum(free)
            y[free] -= decrement
            y = np.clip(y, 0, 1)
        else:  # Too little sum
            free = y < 1
            if not free.any():
                break
            increment = -diff / np.sum(free)
            y[free] += increment
            y = np.clip(y, 0, 1)
        current_sum = np.sum(y)
    
    return y
