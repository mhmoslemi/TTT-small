import numpy as np
from scipy.optimize import minimize
import time

# Global variables for constraints
TARGET_SUM = None

def evaluate_c5_fft(h):
    """Evaluate the C5 constant for a given step function h using FFT for speed"""
    n = len(h)
    if n == 0:
        return 0.0
    # Reverse h
    h_rev = h[::-1]
    b = 1.0 - h
    L = 2 * n - 1
    # Use rfft for efficiency
    hf = np.fft.rfft(h_rev, n=L)
    bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(hf * bf, n=L)
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)

def project_to_sum_box(x, target_sum):
    """Project vector x onto the box [0,1]^n and sum constraint sum(x) = target_sum"""
    # Clip to [0,1]
    x = np.clip(x, 0, 1)
    # Adjust to match target sum via binary search for shift
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(x + mid, 0, 1)
        s = proj.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) / 2.0, 0, 1)

def simulated_annealing(h0, target_sum, time_limit, seed):
    """Run simulated annealing for a given time limit"""
    np.random.seed(seed)
    n = len(h0)
    h = np.copy(h0)
    h = project_to_sum_box(h, target_sum)
    cur_c = evaluate_c5_fft(h)
    best_h = h.copy()
    best_c = cur_c

    # Temperature schedule
    T0 = cur_c * 0.01
    Tmin = 1e-6
    start = time.time()
    iter_count = 0
    
    while time.time() - start < time_limit:
        # Random move
        i, j = np.random.choice(n, size=2, replace=False)
        
        # Calculate delta bounds
        delta_min = max(-h[j], h[i] - 1.0)
        delta_max = min(h[i], 1.0 - h[j])
        
        if delta_min >= delta_max:
            continue
            
        # Choose delta
        delta = np.random.uniform(delta_min, delta_max)
        
        # Update h
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # Evaluate
        c_new = evaluate_c5_fft(h_new)
        
        # Temperature based on progress
        elapsed = time.time() - start
        progress = min(elapsed / time_limit, 1.0)
        T = T0 * (1 - progress) + Tmin * progress
        
        # Accept with probability
        if c_new < cur_c or np.random.rand() < np.exp(-(c_new - cur_c) / (T + 1e-12)):
            h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        
        iter_count += 1

    return best_h, best_c

def constraint_eq(h):
    """Constraint function for sum = target_sum"""
    return np.sum(h) - TARGET_SUM
