import numpy as np
import time
from scipy.optimize import minimize

# Global variable for constraint
TARGET_SUM = None

def evaluate_c5_fft(h):
    """Evaluate C5 using FFT for speed"""
    n = len(h)
    if n == 0:
        return 0.0
    # Reverse h
    h_rev = h[::-1]
    b = 1.0 - h
    L = 2 * n - 1
    # Compute FFT lengths
    hf = np.fft.rfft(h_rev, n=L)
    bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(hf * bf, n=L)
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)

def project_to_sum_box(x, target_sum):
    """Project vector to box [0,1] with sum constraint"""
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase sum
        free_indices = np.where(x < 1)[0]
        if len(free_indices) == 0:
            return x
        # Distribute increment
        increment = diff / len(free_indices)
        x[free_indices] = np.minimum(x[free_indices] + increment, 1.0)
    else:
        # Need to decrease sum
        free_indices = np.where(x > 0)[0]
        if len(free_indices) == 0:
            return x
        decrement = -diff / len(free_indices)
        x[free_indices] = np.maximum(x[free_indices] - decrement, 0.0)
    return x

def constraint_eq(h):
    """Equality constraint sum(h) = TARGET_SUM"""
    return np.sum(h) - TARGET_SUM

def simulated_annealing(h0, target_sum, time_limit, seed):
    """Run simulated annealing with time limit"""
    np.random.seed(seed)
    n = len(h0)
    
    # Initial projection
    h = project_to_sum_box(h0, target_sum)
    cur_c = evaluate_c5_fft(h)
    best_h = h.copy()
    best_c = cur_c
    
    # Temperature schedule
    T_initial = max(0.005, cur_c * 0.05)
    T_final = 1e-6
    start = time.time()
    
    # Always make at least one move
    iter_count = 0
    while time.time() - start < time_limit:
        # Random move
        i, j = np.random.choice(n, size=2, replace=False)
        
        # Compute bounds
        delta_min = max(-h[j], h[i] - 1.0)
        delta_max = min(h[i], 1.0 - h[j])
        
        if delta_min >= delta_max:
            continue
            
        # Choose random delta
        delta = np.random.uniform(delta_min, delta_max)
        
        # Update h
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # Evaluate
        c_new = evaluate_c5_fft(h_new)
        
        # Compute temperature
        elapsed = time.time() - start
        progress = elapsed / time_limit
        T = T_initial * (1 - progress) + T_final * progress
        
        # Accept or reject
        if c_new < cur_c or np.random.rand() < np.exp(-(c_new - cur_c) / max(T, 1e-12)):
            h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                
        iter_count += 1

    return best_h, best_c
