import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Using correlate to compute all overlaps, find maximum
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s * 0.95 - (time.time() - start_time)
    
    # Phase 1: Mixed-Integer Program to get exact integer solution
    n_points = 42
    dx = 2.0 / n_points
    
    # Create binary variables
    x = cp.Variable(n_points, boolean=True)
    t = cp.Variable(integer=True)
    
    # Constraints: sum x_i = n/2
    constraints = [cp.sum(x) == n_points // 2]
    
    # Add auxiliary binary variables to model products
    
    # Construct all necessary indices and y_ij variables
    y_vars = {}
    for d in range(1, n_points):
        # For each shift, indices that can be shifted
        for i in range(n_points - d):
            key = (i, d)
            y_vars[key] = cp.Variable(boolean=True)
            # y_ij >= x_i + x_{i+d} - 1
            constraints.append(y_vars[key] >= x[i] + x[i+d] - 1)
            # y_ij <= x_i
            constraints.append(y_vars[key] <= x[i])
            # y_ij <= x_{i+d}
            constraints.append(y_vars[key] <= x[i+d])
    
    # Make sure each shift has at most t mismatches
    for d in range(1, n_points):
        # Compute sum of y_i,d over valid i
        mismatches = 0
        for i in range(n_points - d):
            mismatches += y_vars[(i, d)]
        # total ones are sum x = n/2; also total y is mismatches
        # number of (1,0) pairs = n/2 - mismatches
        # So (1,1) pairs = mismatches
        # The overlap for shift d =  n/2 - mismatches = x_total - mismatches
        # We want the largest value among d=1 to n-1, i.e., max over d of (x_sum - mismatches)
        constraints.append(n_points // 2 - mismatches <= t)
    
    # Objective: minimize t
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    # Try different solvers
    solver = None
    for s in [CBC, GLPK_MI, ECOS_BB]:
        try:
            prob.solve(solver=s)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                solver = s
                break
        except:
            continue
    
    if solver is None:
        # Used fallback to not crash program
        h0 = np.array(initial_h_values)
        # Clip to unit interval
        h0 = np.clip(h0, 0, 1)
        # Normalize
        h0 = h0 * (n_points / 2) / np.sum(h0)
        best_h = h0.copy()
        best_c5 = evaluate_c5(best_h)
    else:
        # Extract values
        x_val = np.array([float(v.value) for v in x])
        best_h = np.array(x_val)
        best_c5 = evaluate_c5(best_h)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)

    # Phase 2: Fractional refinement using log-sum-exp smoother
    def smooth_max_c5(h, temperature):
        # Re-compute overlaps efficiently
        h = np.array(h)
        dx = 2.0 / n_points
        
        c_values = []
        # Compute for d = 1,...,n_points-1
        for d in range(1, n_points):
            # Get i-th element and (i+d)-th element for i=0 to n_points-d-1
            if d < n_points:
                h_shifted = np.concatenate([h[d:], np.zeros(d)])
                overlap_d = np.sum(h) - np.sum(h * h_shifted)
                c_values.append(overlap_d)
        
        if not c_values:
            return 0.0
        
        c_array = np.array(c_values)
        # Smoothed max
        return temperature * np.log(np.sum(np.exp(c_array / temperature)))
    
    def smooth_max_c5_grad(h, temperature):
        h = np.array(h)
        dx = 2.0 / n_points
        n = len(h)
        
        # Compute gradients separately for all d
        grad_sum = np.zeros_like(h)
        c_vals = []
        weights = []
        
        # Compute all overlaps
        for d in range(1, n):
            if d < n:
                h_shifted = np.concatenate([h[d:], np.zeros(d)])
                c_d = np.sum(h) - np.sum(h*h_shifted)
                c_vals.append(c_d)
                
                # Compute gradient for shift d
                start_idx = d
                grad_i = np.zeros_like(h)
                for i in range(n - d):
                    grad_i[i] = 1.0
                    if i < n - d:
                        grad_i[i+d] -= 1.0
                grad_sum += np.exp(c_d / temperature) * grad_i
                weights.append(np.exp(c_d / temperature))
                
        weights = np.array(weights)
        weights = weights / np.sum(weights)
        
        return grad_sum * weights.mean() / temperature
    
    # Smoothed gradient descent
    h_current = best_h.copy()
    
    # Smoothing parameter schedule
    t_schedule = [0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125, 0.00390625, 0.001953125, 0.0009765625]
    
    # Restart policy
    restarts = 0
    max_restarts = 15
    best_c5_refined = best_c5
    
    while restarts < max_restarts and time_left() > 10:
        h_optimized = h_current.copy()
        for tau in t_schedule:
            if time_left() < 30:
                break
            # Inner loop of 200 steps
            for _ in range(200):
                if time_left() < 1:
                    break
                g = smooth_max_c5_grad(h_optimized, tau)
                h_new = h_optimized - 0.5 * g
                # Clip to [0,1]
                h_new = np.clip(h_new, 0, 1)
                # Project onto simplex constraint
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h_optimized = h_new
            
            # Re-evaluate
            c5_current = evaluate_c5(h_optimized)
            if c5_current < best_c5_refined:
                best_c5_refined = c5_current
                h_current = h_optimized.copy()
        
        # Randomized restart
        h_pert = h_current.copy() + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        h_current = h_pert
        restarts += 1
    
    # Final polish with scipy SLSQP
    if time_left() > 30:
        # Normalize starting h_current
        h_start = h_current.copy()
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        bounds = [(0, 1) for _ in range(n_points)]
        
        def obj(h):
            return evaluate_c5(h)
        
        def constraint(h):
            return np.sum(h) * dx - 1.0
        
        cons = {'type': 'eq', 'fun': constraint}
        
        try:
            res = minimize(obj, h_start, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 1000, 'ftol': 1e-12})
            if res.success:
                h_final = res.x.copy()
                h_final = h_final * (n_points / 2) / np.sum(h_final)
                final_c5 = evaluate_c5(h_final)
                if final_c5 < best_c5_refined:
                    best_c5_refined = final_c5
                    h_current = h_final
        except:
            pass

    return (h_current.tolist(), best_c5_refined, n_points)
