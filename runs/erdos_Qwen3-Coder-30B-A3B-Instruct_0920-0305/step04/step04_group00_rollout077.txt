from scipy.signal import correlate
import time

    n_points = 200
    k_breakpoints = 5  # Number of breakpoints (should be <= n_points//2)
    dx = 2.0 / n_points

    bounds = [(0, 1) for _ in range(k_breakpoints)]

        # Sort heights to enforce decreasing order
        heights = np.sort(params)[::-1]
        # Create step function on [0, 1] using linear interpolation
        x_half = np.linspace(0, 1, n_points // 2 + 1)
        h_half = np.interp(x_half, np.linspace(0, 1, k_breakpoints + 1), np.concatenate([[1], heights, [0]]))
        # Mirror for [1,2]
        h_full = np.concatenate([h_half[:-1], h_half[::-1][1:]])
        # Normalize
        area = np.sum(h_full) * dx
        h_full = h_full / area * 1.0
        # Clip to [0,1] (very unlikely after normalization but just in case)
        h_full = np.clip(h_full, 0, 1)
        return h_full

        corr = correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

        recombination=0.7,

        disp=False

    def compute_gradient(h, beta):
        corr_vals = correlate(h, 1 - h, mode="full")
        corr_vals *= dx
        # Weights from exp(β f_k)
        if beta > 100:
            weights = np.exp(beta * (corr_vals - np.max(corr_vals)))
        else:
            weights = np.exp(beta * corr_vals)
        sum_weights = np.sum(weights)

            weights = np.ones_like(weights)
        else:
            weights /= sum_weights
