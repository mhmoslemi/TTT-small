def evaluate_c5_fft(h):
    """Fast evaluation of C5 using FFT"""
    n = len(h)
    if n == 0:
        return 0.0
    # Reverse h
    h_rev = h[::-1]
    b = 1.0 - h
    L = 2 * n - 1
    # Compute FFT lengths
    hf = np.fft.rfft(h_rev, n=L)
    bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(hf * bf, n=L)
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)

def project_to_sum_box(x, target_sum):
    """Project vector to box [0,1]^n with sum = target_sum"""
    # Clip to bounds
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x
    
    # Adjust by shifting values
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return x
    
    # Simple proportional adjustment
    if diff > 0:
        # Need to add diff; distribute among values < 1
        free_indices = np.where(x < 1)[0]
        if len(free_indices) == 0:
            return x
        # Distribute equally
        add_per = diff / len(free_indices)
        for i in free_indices:
            x[i] = min(1.0, x[i] + add_per)
    else:
        # Need to subtract diff; distribute among values > 0
        free_indices = np.where(x > 0)[0]
        if len(free_indices) == 0:
            return x
        # Distribute equally
        subtract_per = -diff / len(free_indices)
        for i in free_indices:
            x[i] = max(0.0, x[i] - subtract_per)
    
    return x

def simulated_annealing(h0, target_sum, time_limit, seed):
    """Run simulated annealing with temperature schedule"""
    np.random.seed(seed)
    
    n = len(h0)
    h = h0.copy()
    h = project_to_sum_box(h, target_sum)
    cur_c = evaluate_c5_fft(h)
    best_h = h.copy()
    best_c = cur_c

    # Set temperature schedule: adjust to start from reasonable value
    T0 = max(0.005, cur_c * 0.05)
    Tmin = 1e-6
    start = time.time()
    iter_count = 0
    
    # For sub-iteration counts
    inner_iter = 0
    
    while time.time() - start < time_limit:
        # Random move with acceptance probability
        i, j = np.random.choice(n, 2, replace=False)
        
        # Compute valid delta range
        delta_max = min(h[i], 1 - h[j])
        delta_min = -min(1 - h[i], h[j])
        
        if delta_max <= delta_min:
            continue
            
        # Propose delta from [delta_min, delta_max]
        delta = np.random.uniform(delta_min, delta_max)
        
        # Update h (maintain sum and bounds)
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # Evaluate
        c_new = evaluate_c5_fft(h_new)
        
        # Compute temperature based on proportion of elapsed time
        elapsed = time.time() - start
        progress = min(elapsed / time_limit, 1.0)
        T = T0 * (1 - progress) + Tmin * progress
        
        # Accept or reject based on energy difference and temperature
        if c_new < cur_c or np.random.rand() < np.exp(-(c_new - cur_c) / max(T, 1e-12)):
            h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        
        iter_count += 1
        inner_iter += 1
        
    return best_h, best_c
