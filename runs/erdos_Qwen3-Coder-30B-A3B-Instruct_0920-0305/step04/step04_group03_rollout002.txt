def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_boxed_simplex(v, target_sum):
    # Clip to box first
    v_clip = np.clip(v, 0, 1)
    # Adjust to meet sum target while staying within [0,1]
    s = np.sum(v_clip)
    if abs(s - target_sum) < 1e-12:
        return v_clip
    
    # Simple iterative adjustment
    max_iter = 100
    for _ in range(max_iter):
        diff = target_sum - s
        if abs(diff) < 1e-10:
            break
        # Find free variables (not at bounds)
        free = (v_clip > 0) & (v_clip < 1)
        nfree = np.sum(free)
        if nfree == 0:
            break
        delta = diff / nfree
        v_clip[free] += delta
        v_clip = np.clip(v_clip, 0, 1)
        s = np.sum(v_clip)
    
    return v_clip
