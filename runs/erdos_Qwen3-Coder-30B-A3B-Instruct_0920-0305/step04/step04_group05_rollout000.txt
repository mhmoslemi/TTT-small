import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible set: [0,1]^n and sum = target_sum"""
    # Clip to [0,1]
    y = np.clip(h, 0, 1)
    # Adjust sum to target using iterative scaling
    for _ in range(100):
        current_sum = np.sum(y)
        if abs(current_sum - target_sum) < 1e-9:
            break
        if current_sum == 0:
            y[:] = target_sum / len(y)
            break
        # Scale to match target
        y = y * (target_sum / current_sum)
        # Clip again to [0,1]
        y = np.clip(y, 0, 1)
    return y

    bounds = [(0, 1) for _ in range(n_points)]

        projected_h = project_to_feasible(h, n_points / 2)
        return evaluate_c5(projected_h)
    
    # Use DE with constraints
    # For DE, we will use a penalty-based approach for the sum constraint

    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)

    # Run multiple restarts with DE

                objective,

            )
            if res.success:
                h_candidate = res.x
                h_candidate = project_to_feasible(h_candidate, n_points / 2)
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate

            pass

    attempts = 0

        if (time.time() - start_time) > budget_s * 0.98:
            break
        h_trial = best_h + np.random.normal(0, 0.01, n_points)
        h_trial = project_to_feasible(h_trial, n_points / 2)
        try:
            res = minimize(
                objective,
                h_trial,
                method='SLSQP',
                bounds=bounds,
                options={'maxiter': 200, 'ftol': 1e-13, 'gtol': 1e-13}

            pass
        attempts += 1
