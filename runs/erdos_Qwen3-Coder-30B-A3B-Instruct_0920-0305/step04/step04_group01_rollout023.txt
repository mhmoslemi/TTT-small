import numpy as np

from scipy.fft import fft, ifft
import time

# Global variable for constraint
TARGET_SUM = None

def compute_corr_fft(h):
    """Compute full correlation using FFT."""
    n = len(h)
    L = 2*n - 1
    h_rev = h[::-1]
    b = 1 - h
    # Use real FFT for efficiency
    fft_h = fft(h_rev, L)
    fft_b = fft(b, L)
    corr = ifft(fft_h * fft_b.conj()).real
    return corr[:L]

def c5_from_corr(corr, n):
    """Compute C5 from full correlation array with scaling."""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 for given h using FFT."""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto sum = target_sum and box [0,1]^n."""
    x = np.array(x, dtype=float)
    low, high = 0.0, 1.0
    # Binary search for lambda such that sum(x + lambda) = target_sum
    def f(lam):
        y = np.clip(x + lam, low, high)
        return np.sum(y) - target_sum
    
    a, b = -1.0, 1.0
    for _ in range(50):
        mid = (a + b) / 2.0
        if f(mid) > 0:
            b = mid
        else:
            a = mid
    lam = (a + b) / 2.0
    y = np.clip(x + lam, low, high)
    return y

def random_pair_move(h):
    """Generate a random feasible mass transfer move."""
    i, j = np.random.choice(len(h), size=2, replace=False)
    h_i, h_j = h[i], h[j]
    delta_min = max(-h_j, h_i - 1)
    delta_max = min(h_i, 1 - h_j)
    if delta_min < delta_max:
        delta = np.random.uniform(delta_min, delta_max)
        return i, j, delta
    else:
        # No valid move, return an invalid one
        return i, j, 0

def guided_move(h, worst_lag):
    """Perform deterministic move towards reducing the worst correlation."""
    n = len(h)
    # Compute sub-gradient for the worst lag
    # compute h_{i - worst_lag} + h_{i + worst_lag}
    grad = np.zeros(n)
    for i in range(n):
        left_idx = i - worst_lag
        right_idx = i + worst_lag
        if left_idx >= 0 and left_idx < n:
            grad[i] += h[left_idx]
        if right_idx >= 0 and right_idx < n:
            grad[i] += h[right_idx]
    
    # Find indices with minimal and maximal gradient
    idx_min = np.argmin(grad)
    idx_max = np.argmax(grad)
    
    if idx_min == idx_max:
        # No difference, fall back to random move
        return None
    
    h_i, h_j = h[idx_min], h[idx_max]
    delta_min = max(-h_j, h_i - 1)
    delta_max = min(h_i, 1 - h_j)
    
    if delta_min < delta_max:
        delta = (delta_max + delta_min) / 2.0
        h[idx_min] -= delta
        h[idx_max] += delta
        return (idx_min, idx_max, delta)
    else:
        return None

def anneal(h0, run_seconds, rng):
    """Perform simulated annealing with guided moves."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Temperature schedule
    T0 = 5e-3
    Tf = 1e-6
    t0 = time.time()
    
    while time.time() - t0 < run_seconds:
        T = T0 * (1.0 - (time.time() - t0) / run_seconds) + Tf * ((time.time() - t0) / run_seconds)
        
        # Perform a random move
        i, j, delta = random_pair_move(h)
        if delta == 0:
            continue
            
        h_trial = h.copy()
        h_trial[i] -= delta
        h_trial[j] += delta
        h_trial = project_to_sum_box(h_trial, TARGET_SUM)
        
        c5_trial = evaluate_c5(h_trial)
        
        # Accept with Metropolis probability
        if c5_trial <= best_c5 or np.random.rand() < np.exp(-(c5_trial - best_c5) / T):
            h = h_trial
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial
        
        # Every 10 steps, guided move
        if rng.randint(10) == 0:
            corr = compute_corr_fft(h)
            worst_lag = np.argmax(corr)
            gmove = guided_move(h, worst_lag)
            if gmove is not None:
                idx_min, idx_max, delta = gmove
                h[idx_min] -= delta
                h[idx_max] += delta
                h = project_to_sum_box(h, TARGET_SUM)
                c5_trial = evaluate_c5(h)
                if c5_trial < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_trial
                    
    return best_h, best_c5

def local_refine(h):
    """Perform local refinement with SLSQP."""
    from scipy.optimize import minimize
    n = len(h)
    
    def objective(h_);
        return evaluate_c5(h_);
    
    def constraint_eq(h_):
        return np.sum(h_) - TARGET_SUM
    
    bounds = [(0, 1) for _ in range(n)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    res = minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12}
    )
    
    if res.success:
        h_final = project_to_sum_box(res.x, TARGET_SUM)
        return h_final, evaluate_c5(h_final)
    else:
        return h, evaluate_c5(h)
