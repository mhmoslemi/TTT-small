import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_boxed_simplex(v, target_sum):
    """Project vector v onto the simplex with sum = target_sum and bounds [0,1]"""
    # Clip to bounds first
    v = np.clip(v, 0, 1)
    # Adjust to sum target_sum
    current_sum = np.sum(v)
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return v
    
    # Distribute the difference
    if diff > 0:  # Need to add to some values, distribute among ones
        free = np.where(v < 1)[0]
        if len(free) > 0:
            delta = diff / len(free)
            v[free] += delta
            v = np.clip(v, 0, 1)
    else:  # Need to subtract from some values, distribute among zeros
        free = np.where(v > 0)[0]
        if len(free) > 0:
            delta = -diff / len(free)
            v[free] -= delta
            v = np.clip(v, 0, 1)
    return v

    return (best_h_refined.tolist(), best_c5_refined, n_points)
