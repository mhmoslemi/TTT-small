def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def binary_milp_solve(n):
    """Solve the binary version of the problem with MILP"""
    try:
        import cvxpy as cp
    except ImportError:
        return None, None
    
    # Variables
    x = cp.Variable(n, boolean=True)
    y = cp.Variable((n, n), boolean=True)
    t = cp.Variable(integer=True)
    
    # Constraints
    constraints = [
        cp.sum(x) == n // 2
    ]
    
    # Binary product constraints
    for i in range(n):
        for d in range(-n+1, n):
            if 0 <= i+d < n:
                j = i + d
                constraints.append(y[i, j] <= x[i])
                constraints.append(y[i, j] <= x[j])
                constraints.append(y[i, j] >= x[i] + x[j] - 1)
    
    # Overlap constraint
    for d in range(-n+1, n):
        if d != 0:
            # Compute ∑ (x_i (1 - x_{i+d})) for shift d
            overlap = 0
            for i in range(n):
                if 0 <= i+d < n:
                    j = i + d
                    overlap += x[i] - y[i, j]
            constraints.append(overlap <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CBC, verbose=False)
        if prob.status == cp.OPTIMAL:
            return x.value, t.value
        else:
            return None, None
    except:
        return None, None

def smooth_max(h, shifts, tau=0.1):
    """Smooth approximation of the max function"""
    C_d = []
    for d in shifts:
        if d > 0:
            shift_h = np.concatenate([np.zeros(d), h[:-d]])
        else:
            shift_h = np.concatenate([h[-d:], np.zeros(-d)])
        C_d.append(np.sum(h) - np.sum(h * shift_h))
    
    C_d = np.array(C_d)
    exp_C = np.exp(C_d / tau)
    return tau * np.log(np.sum(exp_C)), exp_C / np.sum(exp_C)

def gradient_smooth_max(h, shifts, weights):
    """Gradient of the smooth max"""
    n = len(h)
    grad = np.zeros(n)
    for i, d in enumerate(shifts):
        if d > 0:
            shift_h = np.concatenate([np.zeros(d), h[:-d]])
        else:
            shift_h = np.concatenate([h[-d:], np.zeros(-d)])
        grad += weights[i] * (1 - shift_h)
    return grad

def project_h(h, n):
    """Project h onto the feasible set ∑h_i = n/2"""
    h = np.clip(h, 0, 1)
    scale = (n / 2) / np.sum(h)
    return h * scale

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()

    n = len(initial_h_values)

    # Phase 1: Solve binary version with MILP if available
    best_h = h_current.copy()

    try:

            c5_bin = evaluate_c5(h_bin)
            if c5_bin < best_c5:
                best_h = h_bin.copy()
                best_c5 = c5_bin
                print(f"MILP solution gives C5 = {c5_bin}")
    except:
        pass

    # Use a combinatorial approach for binary vectors
    # Use a heuristic that tries to minimize max overlap
    try:
        # Heuristic: Create a configuration with small overlaps
        # Start from initial_h_values but adjust to get less overlap
        h_adjust = h_current.copy()
        # Try simple optimization 
        dx = 2.0 / n
        shifts = list(range(-n+1, 0)) + list(range(1, n))  # All shifts except 0
        
        # Iterate with smoothing
        tau = 0.5
        restart_count = 0
        max_restarts = 10

        while restart_count < max_restarts and (time.time() - start_time) < budget_s * 0.95:
            if (time.time() - start_time) > budget_s * 0.95:
                break
            # Refine the solution in a smooth manner
            h_refined = project_h(best_h.copy(), n)
            for _ in range(200):  # Steps per restart
                if (time.time() - start_time) > budget_s * 0.95:
                    break
                # Compute smooth max
                _, weights = smooth_max(h_refined, shifts, tau)
                # Gradient of smooth max
                grad = gradient_smooth_max(h_refined, shifts, weights)
                # Update
                h_refined = h_refined - 0.001 * grad
                h_refined = project_h(h_refined, n)
                
            # Compute actual C5
            c5_new = evaluate_c5(h_refined)
            if c5_new < best_c5:
                best_h = h_refined.copy()
                best_c5 = c5_new
                
            # Decrement temperature
            tau *= 0.5
            if tau < 1e-4:
                tau = 1e-4
            
            # Random restart
            h_refined = h_current + np.random.normal(0, 0.02, n)
            h_refined = project_h(h_refined, n)
            restart_count += 1
    except:
        pass

    # Final polish using SLSQP (if necessary)
    if (time.time() - start_time) < budget_s * 0.9:
        try:
            result = minimize(
                evaluate_c5, 
                best_h, 
                method='SLSQP',
                bounds=[(0,1) for _ in range(n)],
                options={'maxiter': 100, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if result.success:
                h_final = project_h(result.x, n)
                c5_final = evaluate_c5(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass

    return (best_h.tolist(), best_c5, n)
