import numpy as np

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def solve_binary_milp(h0, n_points):
    # Prepare data
    dx = 2.0 / n_points

    # Build the MILP model
    x = Variable(n_points, boolean=True)
    t = Variable(integer=True)
    y = Variable((n_points, 2*n_points - 1), boolean=True)  # For pairs (i, i+d)

    # Set up auxiliary product variables
    # For shift d in range(-(n_points-1), n_points-1)
    # This creates the pairs (i, i+d) for all valid d
    constraints = []

    # Constraint that sum equals n/2 (so integral is 1)
    constraints.append(cp_sum(x) == n_points / 2)

    # Add constraint on t
    # For each d, calculate the number of 1,0 pairs
    # Also store the overlap value in terms of y
    # Total overlap per shift is (sum x_i) - sum_{i} y_{i,d} 
    # which equals n/2 - sum_{i} y_{i,d}
    # Need to bound this with t
    # This keeps n/2 - sum_{i} y_{i,d} <= t => sum y_{i,d} >= n/2 - t
    # Or equivalently, let C_d = n/2 - sum y_{i,d}, then C_d <= t 

    # But doing it differently: project to overlap at each shift
    shifts = list(range(-(n_points - 1), 0)) + list(range(1, n_points))
    overlaps = []
    d_map = {}

    for d in shifts:
        d_idx = d + n_points - 1  # index in full correlation array (which is 2*n-1 long)
        d_map[d] = d_idx
        overlap = 0
        # For each i, we compute if x_i and x_{i+d} are both 1
        for i in range(n_points):
            if i + d >= 0 and i + d < n_points:
                if i < i + d:
                    idx = (i, i + d)
                else:
                    idx = (i + d, i)
                # constraint that:
                # y_{i,d} <= x_i, y_{i,d} <= x_{i+d}
                # y_{i,d} >= x_i + x_{i+d} - 1
                # At integer optimum, y_{i,d} = x_i * x_{i+d}

                # These constraints are new for each shifting d
                # Also want to sum y over all i-d pairs for specific d
                # At this point, we'll add the product constraints as a loop
                # Do this post for-all-shifts block in a more readable code
                overlap += y[i, d_idx]
        # Tight constraint for overlap at shift d
        constraints.append(cp_sum(y[:, d_idx]) >= n_points / 2 - t)  # This sets upperbound on 1 - overlap = sum y
        overlaps.append(n_points / 2 - cp_sum(y[:, d_idx]))  # At shift d, overlap is above

    # So maximize sum_over_d overlaps = sum_over_d (n/2 - sum_i y_{i,d})
    # subject to  for each d,  sum_i y_{i,d} => n/2 - t => sum{y} <= n/2 - t 

    # Objective:
    # Minimise t such that for all d, sum_i y_{i,d} <= n/2 - t
    objective = Minimize(t)

    # Constraint to ensure binary relations between y and x
    # In practice, we'll compute these valid combos once
    for d in shifts:
        d_idx = d + n_points - 1
        for i in range(n_points):
            if i + d >= 0 and i + d < n_points:
                # y_{i,d} = x_i * x_{i+d}
                # This can be enforced via:
                # y_{i,d} <= x_i
                # y_{i,d} <= x_{i+d}
                # y_{i,d} >= x_i + x_{i+d} - 1
                # Ensure y is indexed right
                constraints.append(y[i, d_idx] <= x[i])
                constraints.append(y[i, d_idx] <= x[i + d])
                constraints.append(y[i, d_idx] >= x[i] + x[i + d] - 1)

    # Set up and solve the model
    # Add symmetry breaking if needed
    # For now, stick to not breaking symmetry because it may help avoid bias

    problem = Problem(objective, constraints)
    
    try:
        problem.solve(solver=cp.CBC, verbose=False)
        if problem.status in ["optimal", "optimal_inaccurate"]:
            sol_x = x.value 
            if sol_x is not None:
                return sol_x
    except Exception as e:
        # Silently fail to fallback (ofcourse we should make clear we are using fallback)
        pass
    # Fallback: just return initial solution if MILP failed
    return np.array(h0, dtype=bool)

        # Compute overlap for shift d
        shifted_h = np.roll(h, d)
        if d >= 0:
            return np.sum(h[:len(h)-d]) - np.sum(h[:len(h)-d] * shifted_h[d:])
        else:  # d < 0
            return np.sum(h[-d:]) - np.sum(h[-d:] * shifted_h[:-d])

        c_vals = [c_d(h, d) for d in range(-(n_points-1), 0) + range(1, n_points)]

        return tau * np.log(np.sum(exps))

    # Combine Our gradient-based optimizer.
    # Instead of matrix multiply use elementary loops:
    def smooth_grad(h, tau):
        c_vals = []
        for d in range(-(n_points-1), 0) + range(1, n_points):
            shifted_h = np.roll(h, d)
            if d >= 0:
                val = np.sum(h[:len(h)-d]) - np.sum(h[:len(h)-d] * shifted_h[d:])
            else:
                val = np.sum(h[-d:]) - np.sum(h[-d:] * shifted_h[:-d])
            c_vals.append(val)
        exps = np.exp(np.array(c_vals) / tau)
        w = exps / np.sum(exps)

        grad = np.zeros_like(h)
        for i, d in enumerate(range(-(n_points-1), 0) + range(1, n_points)):
            # ∂C_d/h_i is either 1 
            # or -h[i+d] for i+d<n and i not beyond h bound
            # Unroll
            # Direct gradient in ?

            # We compute the components of ∂C_d / ∂h_j  for all j
            # And each shift d contributes with weight w[i]
            # We want ∑_d w_d * ∂C_d / ∂h_j
            # 
            # Summed over d: ∂C_d(h)/∂h_j = 1 (if j = some position)
            # For product pair h[i]*h[i+d]:
            # if j=i, 1 - h[i+d]
            # if j=i+d, -h[i]
            # else 0
            for j in range(n_points):

                elif d < 0 and j >= -d:

                    v += 1  
                    if j + d >= 0:

                grad[j] += w[i] * v

        return grad

        h = np.clip(h, 0, 1)

        ratio = (n_points / 2) / np.sum(h)
        h *= ratio
        return h

        outer_iter += 1
        # Maybe some break condition or continue without expiration
        if outer_iter > 100:

    return best_h, best_c5

            num_restarts += 1
        except Exception as e:
            # fallback to initial
            num_restarts += 1

    # --- Final SLSQP Polish ---
    try:
        # Re-evaluate final to make a good accuracy element
        bounds = [(0, 1) for _ in range(n_points)]
        def obj_fn(h):
            return evaluate_c5(h)
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) * dx - 1.0}

            obj_fn,
            best_sol,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}

            h_final = res.x
            scale = 1.0 / (np.sum(h_final) * dx)
            h_final = h_final * scale
            c5_final = obj_fn(h_final)

        pass
