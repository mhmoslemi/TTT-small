import numpy as np
import time
from scipy.optimize import minimize

# Global variable for constraints
TARGET_SUM = None

def evaluate_c5_fft(h):
    """Evaluate the C5 constant for a given step function h using FFT for speed."""
    n = len(h)
    if n == 0:
        return 0.0
    # Reverse h
    h_rev = h[::-1]
    b = 1.0 - h
    L = 2 * n - 1
    # Compute FFT lengths
    hf = np.fft.rfft(h_rev, n=L)
    bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(hf * bf, n=L)
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] and sum constraint."""
    # Clip to [0,1]
    x = np.clip(x, 0, 1)
    # Adjust to match target sum
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x    
    # Simple adjustment: distribute difference evenly among elements
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return x
    # Add diff smoothly to elements (avoiding boundary violations)
    sorted_indices = np.argsort(x)
    # Need to add diff to elements that can absorb it
    remaining_diff = diff
    for i in sorted_indices:
        if remaining_diff > 0:
            max_increase = 1 - x[i]
            increase = min(remaining_diff, max_increase)
            x[i] += increase
            remaining_diff -= increase
        elif remaining_diff < 0:
            max_decrease = x[i]
            decrease = min(-remaining_diff, max_decrease)
            x[i] -= decrease
            remaining_diff += decrease
        else:
            break
    return x

def constraint_eq(h):
    """Equality constraint for sum(h) = TARGET_SUM."""
    return np.sum(h) - TARGET_SUM

def simulated_annealing(h_initial, target_sum, time_limit, seed):
    """Run simulated annealing with given initial h and time limit."""
    np.random.seed(seed)
    n = len(h_initial)
    h = np.copy(h_initial)
    # Ensure feasibility via project
    h = project_to_sum_box(h, target_sum)
    cur_c = evaluate_c5_fft(h)
    best_h = h.copy()
    best_c = cur_c

    # Set temperature schedule
    T0 = cur_c * 0.05 if cur_c > 0 else 0.01
    Tmin = 1e-8
    start = time.time()
    
    iter_count = 0
    while time.time() - start < time_limit:
        # Decaying temperature
        elapsed = time.time() - start
        progress = elapsed / time_limit
        T = T0 * (1 - progress) + Tmin * progress
        
        # Random move
        i, j = np.random.choice(n, size=2, replace=False)
        # Compute delta range
        delta_min = max(-h[j], h[i] - 1.0)
        delta_max = min(h[i], 1.0 - h[j])
        
        if delta_max <= delta_min:
            iter_count += 1
            continue
            
        # Choose delta uniformly in [delta_min, delta_max]
        delta = np.random.uniform(delta_min, delta_max)
        
        # Update h
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # evaluate
        c_new = evaluate_c5_fft(h_new)
        
        # Acceptance probability
        if c_new < cur_c or np.random.rand() < np.exp(-(c_new - cur_c) / (T + 1e-12)):
            h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        iter_count += 1
        
    return best_h, best_c
