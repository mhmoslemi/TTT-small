import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def project_to_boxed_simplex(v, target_sum):
    """Project v onto the simplex {0 <= x <= 1, sum(x) = target_sum}"""
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-10:
        return v
    
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values that are < 1
        free = v < 1
        if np.sum(free) > 0:
            delta = diff / np.sum(free)
            v[free] += delta
            v = np.clip(v, 0, 1)
    else:
        # Need to decrease values that are > 0
        free = v > 0
        if np.sum(free) > 0:
            delta = -diff / np.sum(free)
            v[free] -= delta
            v = np.clip(v, 0, 1)
            
    return v

def evaluate_c5(h):
    """Evaluate C5 using provided method"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx
