import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
import math

def decode_trapezoidal_h(cp, w, s, offset, n_points):
    """Decode trapezoidal parameters to full h vector."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Compute the width of plateau and edge regions
    half_width = 0.5 * w
    adjust_center = offset * 0.5
    
    center_idx = n_points // 2
    left_edge = int(max(0, center_idx - adjust_center * n_points - half_width * n_points))
    right_edge = int(min(n_points, center_idx + adjust_center * n_points + half_width * n_points))
    
    # Calculate edge indices for decay
    left_decay_start = left_edge
    left_decay_end = int(left_edge + s * n_points / 2)
    right_decay_start = right_edge
    right_decay_end = int(right_edge - s * n_points / 2)
    
    # Fill h
    # Plateau region
    h[left_edge:right_edge] = cp
    
    # Left decay region
    for i in range(left_decay_start, min(left_decay_end, left_edge)):
        if i < 0:
            continue
        factor = (i - left_decay_start) / (left_edge - left_decay_start)
        h[i] = cp * (1 - factor * 0.5)
    
    # Right decay region
    for i in range(max(right_decay_start, right_edge), right_decay_end, -1):
        if i >= n_points:
            continue
        factor = (right_decay_start - i) / (right_decay_start - right_edge)
        h[i] = cp * (1 - factor * 0.5)
    
    h = np.clip(h, 0, 1)
    return h

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

    total_time_used = 0

    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 10.0), (0.0, 1.0)]

    def objective(params):

        scale = (n_points / 2) / np.sum(h)
        h = np.clip(h * scale, 0, 1)
        c5 = evaluate_c5(h)
        return c5

    max_de_restarts = 50
    max_iter = 500

            break

        res = differential_evolution(
            func=objective,
            bounds=bounds,
            seed=seed + _,
            maxiter=max_iter,
            popsize=32,
            strategy='best1bin',
            tol=1e-6,
            recombination=0.9,
            callback=None
        )

        if not res.success:
            continue

        h = decode_trapezoidal_h(cp, w, s, offset, n_points)
        scale = (n_points / 2) / np.sum(h)
        h = np.clip(h * scale, 0, 1)
        c5 = evaluate_c5(h)
