import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def project_to_simplex_and_box(h, target_sum, dx):
    """Project h to the feasible set: 0 <= h <= 1 and sum(h)*dx = target_sum"""
    # Ensure h is within bounds
    h = np.clip(h, 0, 1)
    # Scale to target sum
    current_sum = np.sum(h) * dx
    if abs(current_sum) < 1e-12:
        # Avoid division by zero
        h.fill(0.0)
    else:
        h *= target_sum / current_sum
        # Re-clip to ensure projection remains valid
        h = np.clip(h, 0, 1)
        # Re-scale again to match target sum
        current_sum = np.sum(h) * dx
        h *= target_sum / current_sum
    return h

def build_symmetric_step_function(u, n):
    """Convert a vector u of heights to a symmetric, step-like function"""
    # sort to ensure decreasing
    u_sorted = np.sort(u)[::-1]
    K = len(u_sorted) - 1
    # grid points for [0,1], half
    x_half = np.linspace(0, 1, K+1)
    # values of heights at x_half
    heights = u_sorted
    # step locations
    # we interpolate linearly to get the full function h on [0,2]
    # define x_grid to go from 0 to 2
    x_full = np.linspace(0, 2, n)
    # in between each x_we have intervals of width dx
    dx = 2.0 / n
    h = np.zeros(n)
    # replicate the thing twice, once from 0 to 1, once from 1 to 2
    # we'll use interpolation between x_half and heights for [0,1]
    # the construction is either symmetric w.r.t. 1 or we can just mirror it
    # for i in range(K):
    #     indices = (x_full >= x_half[i]) & (x_full <= x_half[i+1])
    #     ratio = (x_full[indices] - x_half[i]) / (x_half[i+1] - x_half[i])
    #     h[indices] = (1-ratio)*heights[i] + ratio*heights[i+1]
    
    # more careful linear construction:
    # from 0 to 1:
    x_interp = np.linspace(0, 1, n//2)
    # map x from [0,1] to x_half to look up weights
    x_interp_2 = np.digitize(x_interp, x_half)  # this gives 1..K
    x_interp_2 = np.clip(x_interp_2, 0, K-1) # safe indexing
    # get fraction v between the two points
    x_left = x_half[x_interp_2]
    x_right = x_half[x_interp_2 + 1]
    v = (x_interp - x_left) / (x_right - x_left + 1e-15)  # avoid div-by-zero
    v = np.clip(v, 0, 1)
    h_left = (1 - v)*heights[x_interp_2] + v*heights[x_interp_2 + 1]
    # from 1 to 2: mirrored around 1
    x_interp_r = 2 - x_interp
    x_interp_r_2 = np.digitize(x_interp_r, x_half)
    x_interp_r_2 = np.clip(x_interp_r_2, 0, K-1) # safe indexing
    # get fraction
    x_left_r = x_half[x_interp_r_2]
    x_right_r = x_half[x_interp_r_2 + 1]
    v_r = (x_interp_r - x_left_r) / (x_right_r - x_left_r + 1e-15)
    v_r = np.clip(v_r, 0, 1)
    h_right = (1 - v_r)*heights[x_interp_r_2] + v_r*heights[x_interp_r_2 + 1]
    h[:n//2] = h_left
    h[n//2:] = h_right
    return h

def evaluate_C5(h):
    """Evaluate the C5 quantity of the given distribution h"""
    # correlation
    corr = correlate(h, 1-h, mode="full")
    # normalize the correlation by dx
    dx = 2.0 / len(h)
    corr = corr * dx
    # return max
    return np.max(corr)

def smoothed_C5(h, beta=15.0):
    """Compute smooth approximation of C5 using log-sum-exp to avoid non-differentiable max"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1-h, mode="full")
    corr = corr * dx
    
    # soft max
    exps = np.exp(beta * corr)
    soft_max = (1./beta) * np.log(np.sum(exps))
    return soft_max

def compute_gradient(h, beta=20.0):
    """Compute gradient w.r.t. h for smoothed_C5"""
    dx = 2.0 / len(h)
    # the values of f_k (this is what's in corr)
    f_vals = correlate(h, 1-h, mode="full")
    f_vals = f_vals * dx
    # weights
    exps = np.exp(beta * f_vals)
    exps_sum = np.sum(exps)
    weights = exps / exps_sum
    
    # compute gradient for each i:
    # d/dh_i sum_k w_k f_k = sum_k w_k d/dh_i f_k = sum_k w_k ( (1-h_{i+k}) - (1-h_{i-k}) ) * dx
    # if i+k >= n or i-k < 0, then h_{i+k}, h_{i-k} contribute zero weight
    
    D = len(h)
    # initialize gradient
    grad = np.zeros(D)
    # go through all k values that exist for the correlation...
    # we're interested in k from -(D-1) to (D-1)
    # the full range of correlation indices is -D+1 to D-1
    # each shift happens at idx = k + D-1
    for k in range(1 - D, D):
        k_idx = k + D - 1  # Index in the corr array
        if k_idx < 0 or k_idx >= 2*D - 1:
            continue  # out of bounds
        weight = weights[k_idx]
        # compute term in gradient (note that top index in corr is at k=D-1)
        for i in range(D):
            # partial derivative d/dh_i of f_k = (h_i (1-h_{i+k})) dx
            # term is -h_{i+k} dx if i+k is within [0,D) and k>0,
            # term is +h_{i-k} dx if i-k is within [0,D) and k<0
            term = 0.0
            if k > 0 and i + k < D:
                term += -h[i + k]  # this is what we get from 1-h_{i+k}
            if k < 0 and i >= -k:
                term += h[i - k]   # positive because it's h_{i-k}
            grad[i] += weight * term * dx
    return grad

def project_gradient_step(h, grad, step_size, dx):
    """Perform projected gradient descent step"""
    h_new = h - step_size * grad
    # project
    h_new = projection_simplex_box(h_new, dx)
    return h_new

def projection_simplex_box(h, dx):
    """Project h into the set {h: 0<=h<=1, sum(h)*dx=1}"""
    # clip
    h = np.clip(h, 0, 1)
    # rescale to sum = 1/dx
    s = np.sum(h) * dx
    if s == 0:
        h.fill(0)
    else:
        h *= 1.0 / s
        # clip again
        h = np.clip(h, 0, 1)
        # rescale again
        s = np.sum(h) * dx
        h *= 1.0 / s
    return h
