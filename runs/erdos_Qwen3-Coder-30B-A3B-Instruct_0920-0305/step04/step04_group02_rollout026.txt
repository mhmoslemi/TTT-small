import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from cvxpy import *
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def build_and_solve_milp(n_points):
    """Solve MILP to get binary solution"""
    # Binary variables
    x = Variable(n_points, boolean=True)
    # Auxiliary variables for products
    y = {}
    for d in range(-n_points+1, n_points):
        if d == 0:
            continue
        for i in range(n_points):
            if 0 <= i+d < n_points:
                y[(i, d)] = Variable(boolean=True)
    
    # Constraints for product
    constraints = []
    for d in range(-n_points+1, n_points):
        if d == 0:
            continue
        for i in range(n_points):
            if 0 <= i+d < n_points:
                constraints += [
                    y[(i, d)] <= x[i],
                    y[(i, d)] <= x[i+d],
                    y[(i, d)] >= x[i] + x[i+d] - 1
                ]
    
    # Integral constraint
    constraints.append(sum(x) == n_points // 2)
    
    # Objective variable
    t = Variable(integer=True)
    
    # Overlap constraint
    for d in range(-n_points+1, n_points):
        if d == 0:
            continue
        overlap_expr = n_points // 2 - sum(y[(i,d)] for i in range(n_points) if 0 <= i+d < n_points)
        constraints.append(overlap_expr <= t)
    
    # Objective
    objective = Minimize(t)
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=CBCCBF, verbose=False, max_seconds=30)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            binary_sol = [float(v.value) for v in x]
            return binary_sol, t.value
    except:
        pass
    return None, None

def smooth_max_gradient(h, shifts, tau=0.1):
    """Calculate gradient of smooth max of overlap integrals"""
    n = len(h)
    grads = np.zeros(n)
    total_exp = 0.0
    exp_vals = {}
    
    for d in shifts:
        if d == 0:
            continue
        corr = 0.0
        for i in range(n):
            if 0 <= i+d < n:
                corr += h[i] * (1 - h[i+d])
            else:
                corr += h[i]  # if not in range, assume null shift value
        exp_vals[d] = np.exp(corr / tau)
        total_exp += exp_vals[d]
    
    for d in shifts:
        if d == 0:
            continue
        w_d = exp_vals[d] / total_exp
        for i in range(n):
            if 0 <= i+d < n:
                grad_idx = i
                grads[grad_idx] += w_d * (1.0 - h[i+d])
            else:
                grad_idx = i
                grads[grad_idx] += w_d * 1.0

    return grads

def project_to_simplex(h, target_sum=1.0):
    """Project h to the simplex: sum(h) = target_sum, h_i in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-10:
        return h
    scale_factor = target_sum / current_sum
    h = h * scale_factor
    return h

def optimize_smooth_max(h_init, n_points, budget_s, start_time):
    """Use smooth max and gradient descent for refinement"""
    h = h_init.copy()
    dx = 2.0 / n_points
    shifts = list(range(-n_points+1, n_points))
    shifts.remove(0)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    tau = 0.5
    step_size = 0.01
    max_iter = 200
    restart_count = 0
    max_restarts = 5
    
    while (time.time() - start_time) < budget_s * 0.90 and restart_count < max_restarts:
        h = best_h.copy()
        # Inner loop with continuation
        last_c5 = best_c5
        for i in range(max_iter):
            if (time.time() - start_time) > budget_s * 0.90:
                break
            grad_h = smooth_max_gradient(h, shifts, tau)
            h = h - step_size * grad_h
            h = project_to_simplex(h, n_points * dx / 2)
            if i % 20 == 0:
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
                if abs(c5 - last_c5) < 1e-6:
                    break
                last_c5 = c5
        tau /= 2
        if tau < 1e-5:
            break
        restart_count += 1
    
    # Final refinement using SLSQP
    def obj_func(h_vals):
        return evaluate_c5(h_vals)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}
    
    try:
        res = minimize(obj_func, best_h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 500})
        if res.success:
            h_final = res.x.copy()
            h_final = project_to_simplex(h_final, n_points * dx / 2)
            final_c5 = evaluate_c5(h_final)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h0 = np.array(initial_h_values)

    # Phase 1: Solve binary program

    # Phase 2: Smooth maximization & refinement
    best_h, best_c5 = optimize_smooth_max(h_binary, n_points, budget_s, start_time)
    
    # Final polishing using SLSQP if time allows
    if (time.time() - start_time) < budget_s * 0.95:
        def obj_func(h_vals):
            return evaluate_c5(h_vals)

        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}
        
        try:
            res = minimize(obj_func, best_h, method='SLSQP',

                          options={'maxiter': 300})

                final_h = res.x.copy()
                final_h = project_to_simplex(final_h, n_points * dx / 2)

                if final_c5 < best_c5:
