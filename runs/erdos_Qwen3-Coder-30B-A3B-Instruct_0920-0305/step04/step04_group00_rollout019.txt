import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def build_symmetric_piecewise_linear(half_heights, n_points):
    # sorted by decreasing heights
    h_sorted = np.sort(half_heights)[::-1]
    
    # create the left half: piecewise linear from (0, 0) to (1, h_sorted[0])
    # then to (x_K, h_K), etc.
    x_bps = np.linspace(0, 1, len(h_sorted))
    x_vals = np.linspace(0, 1, n_points // 2 + 1) 
    y_vals = np.interp(x_vals, x_bps, h_sorted, left=0, right=0)
    
    # mirror to get full function
    full_h = np.concatenate([y_vals, y_vals[-2::-1]])
    
    # enforce integral constraint
    integral = np.sum(full_h) * 2.0 / n_points
    scale = (n_points / 2) / integral
    full_h = full_h * scale
    full_h = np.clip(full_h, 0, 1)
    
    return full_h

        # Smoothed approximation
        max_f = np.max(corr)
        exp_f = np.exp(beta * (corr - max_f))
        w = exp_f / np.sum(exp_f)
        # gradient
        grad = np.zeros_like(h)
        for k in range(len(corr)):
            # k corresponds to shift in integer domain
            shift = k - (len(h) - 1)
            # 1 is used as a fixed extra term of 1 for reflect boundary
            for i in range(len(h)):
                j = i + shift
                if 0 <= j < len(h):
                    # The gradient of h_i * (1 - h_j) w.r.t. h_i is (1 - h_j)  
                    grad[i] += w[k] * dx * (1 - h[j])
        
        return grad
