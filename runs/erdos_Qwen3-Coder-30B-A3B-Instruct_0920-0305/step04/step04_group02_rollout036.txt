import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from cvxpy import *
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    def solve_milp_binary():
        # Binary MILP: exact combinatorial optimisation
        x = Variable(n_points, boolean=True)
        y = {}
        t = Variable(integer=True)
        
        # Build auxiliary variables for products x[i] * x[i+d]
        for d in range(-n_points+1, n_points):
            if d == 0:
                continue
            # Define y[i,d] = x[i] * x[i+d] for valid indices
            y[d] = {}
            for i in range(n_points):
                j = i + d
                if 0 <= j < n_points:
                    y[d][i] = Variable(boolean=True)
        
        # Mass constraint
        constraints = [sum(x) == n_points // 2]

        # Product constraints via McCormick
        for d in range(-n_points+1, n_points):
            if d == 0:
                continue
            for i in range(n_points):
                j = i + d
                if 0 <= j < n_points:
                    constraints += [
                        y[d][i] <= x[i],
                        y[d][i] <= x[j],
                        y[d][i] >= x[i] + x[j] - 1
                    ]
        
        # Constraint: for each d, overlap ≤ t
        for d in range(-n_points+1, n_points):
            if d == 0:
                continue
            # Compute ∑ y_{i,d}  =  ∑ x[i]*x[i+d]
            sum_y = sum(y[d][i] for i in range(n_points) if 0 <= i+d < n_points)
            # N_d = n/2 - sum_y, so max(N_d) = max_d (n/2 - sum_y) = n/2 - min_d sum_y
            # But we want to minimize max(N_d) = max_d (n/2 - sum_y) = n/2 - min_d sum_y
            # So, we maximize sum_y over shifts d, to find min(N_d) = min_d (n/2 - sum_y) = n/2 - max_d sum_y
            # Or: have variable t, and for all d, (n/2 - sum_y) ≤ t  =>  sum_y ≥ n/2 - t
            constraints += [sum_y >= n_points // 2 - t]
        
        constraints += [t >= 0]
        
        # Objective
        objective = Minimize(t)
        
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=CBC, verbose=False)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                x_val = [int(v.value) for v in x]
                t_val = int(t.value)
                return np.array(x_val), t_val
        except:
            pass
        return None, None
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Phase 1: Solve binary problem using MILP
    binary_solution, t_opt = None, None
    try:
        binary_solution, t_opt = solve_milp_binary()
        if binary_solution is not None:
            scale = (n_points / 2) / np.sum(binary_solution)
            # Normalize to match sum constraint
            h_bin = binary_solution.astype(float) * scale
            c5_bin = evaluate_c5(h_bin)
            if c5_bin < best_c5:
                best_h = h_bin.copy()
                best_c5 = c5_bin
    except:
        pass
    
    if binary_solution is None:
        # Fallback to starting point
        h_bin = h0.copy()
        c5_bin = evaluate_c5(h0)
        best_h = h0.copy()
        best_c5 = c5_bin
        binary_solution, t_opt = h0.astype(bool).astype(int), 5  # Arbitrary fallback

    # Phase 2: Smooth optimization using log-sum-exp
    def smooth_max_overlaps(h, tau):
        dx = 2.0 / n_points
        # Compute all overlaps
        overlaps = []
        for d in range(-n_points+1, n_points):
            if d == 0:
                continue
            shift_h = np.roll(h, d)
            overlap = np.sum(h * (1 - shift_h))
            overlaps.append(overlap * dx)
        overlaps = np.array(overlaps)
        # Soft max
        return tau * np.log(np.sum(np.exp(overlaps / tau)))

    def gradient_smooth_max(h, tau):
        dx = 2.0 / n_points
        # Overlaps and weights
        overlap_vals = []
        weights = []
        all_overlaps = []
        for d in range(-n_points+1, n_points):
            if d == 0:
                continue
            shift_h = np.roll(h, d)
            overlap = np.sum(h * (1 - shift_h))
            overlap_vals.append(overlap * dx)
            all_overlaps.append(overlap * dx)
        
        overlap_vals = np.array(overlap_vals)
        exp_vals = np.exp(overlap_vals / tau)
        Z = np.sum(exp_vals)
        weights = exp_vals / Z
        # Gradient
        grad = np.zeros_like(h)
        for i, d in enumerate(range(-n_points+1, n_points)):
            if d == 0:
                continue
            shift_h = np.roll(h, d)
            grad_here = 1 - shift_h
            # Derivative of h_i * (1-h_{i+d}) w.r.t. h_i is 1 - h_{i+d}
            # But there's 2 parts: h_i * (1 - h_{i+d}) and (1-h_i) * h_{i+d}
            # But we're looking at d’th shift: h_i (1 - h_{i+d})
            # The contribution to ∂/∂h_i is (1 - h_{i+d}) when d = i - j (i.e. j = i - d)
            # Therefore for each shift d:
            #      ∇[ C_d(h) ] = (1 - h_{i+d})
            # So: ∑_d w_d ∇[C_d(h)] is ∑_d w_d (1 - h_{i+d})
            for k, d_shift in enumerate(range(-n_points+1, n_points)):
                if d_shift == 0:
                    continue
                contrib_indices = np.arange(n_points)  # This is i
                shifted_indices = contrib_indices + d_shift  # This is i+d
                # Only include valid indices (0 <= i+d < n_points)
                valid = (shifted_indices >= 0) & (shifted_indices < n_points)
                sth_grad = np.zeros_like(h)
                sth_grad[contrib_indices[valid]] = 1.0
                sth_grad[shifted_indices[valid]] -= 1.0  # Subtract h_{i+d} effect
                grad += weights[k] * sth_grad
        
        return grad

    def project_to_simplex(h, target_sum=1.0):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if np.isclose(current_sum, target_sum, rtol=1e-10):
            return h
        h *= target_sum / current_sum
        return np.clip(h, 0, 1)

    # Approximate solver using projected gradient with continuation
    def solve_continuous_method(h_start, max_iters=1000):
        h = h_start.copy()
        h = project_to_simplex(h)
        tau_vals = [0.8, 0.4, 0.2, 0.1, 0.05, 0.01, 1e-3]
        lr = 0.2
        min_tau = 1e-4
        restart_count = 0
        max_restarts = 30
        current_c5 = evaluate_c5(h)
        best_h = h.copy()
        best_c5 = current_c5

        while (time.time() - start_time) < (budget_s * 0.95) and restart_count < max_restarts:
            h = best_h.copy()
            for i, tau in enumerate(tau_vals):
                if (time.time() - start_time) > (budget_s * 0.95):
                    break
                if tau < min_tau:
                    break
                    
                # Time for this tau
                inner_iters = 100 if i < len(tau_vals) - 1 else 150
                
                for _ in range(inner_iters):
                    if (time.time() - start_time) > (budget_s * 0.95):
                        break
                        
                    grad = gradient_smooth_max(h, tau)
                    h = h - lr * grad
                    h = project_to_simplex(h)

                    # Compute new c5 and check improvement
                    temp_c5 = evaluate_c5(h)
                    if temp_c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = temp_c5

            # Random restart
            h = best_h.copy()
            noise = np.random.normal(0, 0.02, len(h))
            h += noise
            h = project_to_simplex(h)
            restart_count += 1
            
        # Final polish
        def objective(h):
            return evaluate_c5(h)
            
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) * (2.0 / n_points) - 1.0}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 500})
            if res.success:
                h_final = res.x
                scale = (n_points / 2) / np.sum(h_final)
                h_final = h_final * scale
                c5_final = evaluate_c5(h_final)
                if c5_final < best_c5:
                    best_h = h_final.copy()
                    best_c5 = c5_final
        except:
            pass
            
        return best_h, best_c5

    # Run continuous optimization from best solution
    h_final, c5_final = solve_continuous_method(best_h)
    
    # Improve it further with one more local min file call
    try:
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) * (2.0 / n_points) - 1.0}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(evaluate_c5, h_final, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 1000})
        if res.success:
            h_final = res.x
            scale = (n_points / 2) / np.sum(h_final)
            h_final = h_final * scale
            c5_final = evaluate_c5(h_final)
    except:
        pass

    return (h_final.tolist(), c5_final, n_points)
