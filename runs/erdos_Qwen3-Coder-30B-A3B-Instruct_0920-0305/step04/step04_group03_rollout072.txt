import numpy as np
from scipy.optimize import minimize
import time
from scipy.signal import correlate

def project_to_boxed_simplex(v, target_sum):
    # Project onto {0 <= x <= 1, sum(x) = target_sum}
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-10:
        return v
    diff = target_sum - current_sum
    if diff > 0:
        # Need to add diff; distribute to entries < 1
        free = (v < 1)
        if np.sum(free) > 0:
            delta = diff / np.sum(free)
            v[free] += delta
            v = np.clip(v, 0, 1)
    else:
        # Need to subtract diff; distribute from entries > 0
        free = (v > 0)
        if np.sum(free) > 0:
            delta = -diff / np.sum(free)
            v[free] -= delta
            v = np.clip(v, 0, 1)
    return v

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    # The maximum value over all shifts (including zero and negative shifts)
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Use a higher resolution
    n_points = 80
    target_sum = n_points / 2
    dx = 2.0 / n_points

    best_h_bin = None
    best_c5_bin = float('inf')

        # h_bin is binary vector 0/1
        corr = correlate(h_bin, 1 - h_bin, mode="full")
        # Only consider valid shifts with substantial overlap
        # This is adjusted based on interpretation: we consider all shifts
        return np.min(corr)
    
    # Initialize binary vector with exact target sum
    h_bin = np.zeros(n_points, dtype=int)
    h_bin[:int(target_sum)] = 1
    np.random.shuffle(h_bin)
    
    max_iter = 100000
    
    # Temperature schedule
    initial_temp = 10.0
    final_temp = 0.001
    cooling_rate = 0.9999
    
    current_h = h_bin.copy()
    current_min_corr = compute_min_corr(current_h)
    
    # Simulated Annealing loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s * 0.8:
            break
            
        # Make a move: swap one 1 with one 0
        ones = np.where(current_h == 1)[0]
        zeros = np.where(current_h == 0)[0]
        if len(ones) > 0 and len(zeros) > 0:
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            # Create new state
            new_h = current_h.copy()
            new_h[i] = 0
            new_h[j] = 1
        else:
            continue
            
        new_min_corr = compute_min_corr(new_h)
        delta = new_min_corr - current_min_corr
        
        # Accept with probability based on temperature
        if delta > 0 or np.random.rand() < np.exp(delta / initial_temp):
            current_h = new_h
            current_min_corr = new_min_corr
            
        # Update best solution
        if current_min_corr > best_c5_bin:
            best_h_bin = current_h.copy()
            best_c5_bin = current_min_corr
            
        # Annealing
        initial_temp *= cooling_rate
        
    # If we didn't find a better solution just use a constant initial one
    if best_h_bin is None:
        best_h_bin = h_bin.copy()
        best_c5_bin = compute_min_corr(best_h_bin)
        
    # Phase 2: Continuous refinement using gradient descent
    def smooth_max_gd(h, alpha=50.0, num_iter=2000):
        # Use smooth max: max_k g_k ≈ (1/alpha) * log(sum_k exp(alpha * g_k))
        h = np.array(h, dtype=float)
        h = project_to_boxed_simplex(h, target_sum)
        
        # Store best h found so far
        best_h = h.copy()
        best_c5 = evaluate_c5(h)

        eta = 0.1
        decay = 0.999
        
        for round in range(num_iter):
            if (time.time() - start_time) > budget_s * 0.95:
                break

            corr = correlate(h, h, mode='full')  # Shape 2*N-1 = 159
            g = target_sum - corr  # Negative because we want to minimize max(g) = maximize min(-g)
            # Compute actual values (since we have dx).
            # But we can minimize with 1-dx factor since same across all shifts
            
            # Compute smooth max (e.g., log-sum-exp) approximation
            # But it's the max of f_k terms which are negative to match other formulation.
            # f_k = sum h_i * (1 - h_{i+k}) = target_sum - sum h_i h_{i+k} = target_sum - corr_k
            
            # g_k = target_sum - corr_k  =>  we want to minimize max(g_k)

            g_scaled = g / alpha

            smooth_max_g = log_sum / alpha

            # For g_k = target_sum - corr_k, we have the gradient of smooth max:
            # ∇ smooth_max_g = 1/alpha * (sum_k exp(alpha * g_k) / sum_k exp(alpha * g_k)) * α * ∇g_k
            #                 = ∇g_k (the smoothing average of ∇g_k)

            # So ∇g_k = - (h_{i+k} + h_{i-k})
            
            # We compute ∇g_k explicitly, and weight them (softmax of g_k)
            softmax_g = np.exp(g_scaled)
            softmax_g = softmax_g / np.sum(softmax_g)
            
            # Use vectorized differentiation for gradient:
            # Convolver: use the fact that ∂ corr_k / ∂h_j = h_{j+k} + h_{j-k}
            # So ∂g_k / ∂h_j = -(h_{j+k} + h_{j-k})
            # But doing conv with weights is easier.
            
            # Compute weighted sum of differences with shifted versions of h:
            # ∇ g_k ∂ h_i = - (h_{i+k} + h_{i-k})
            # We are summing: ∂ h_i / ∂ h_i = - (weight * (h shifted forward + backward))

            # Do convolution with rolled h
            n = len(h)

                if k == 0: continue
                if 9000  < shift_k < 9000: # avoid computing for now

                # Most efficient: roll the array to approximate shifted version
                # scan both directions
                if shift_k > 0:
                    h_shifted = np.roll(h, shift_k)
                    h_shifted[:shift_k] = 0  # pad out of bounds

                    h_shifted = np.roll(h, shift_k)  
                    h_shifted[shift_k:] = 0
                
                # sum over k: weight_k * (h_shifted_full - h_shifted_back) (with proper bounds checking, not easy in vectorized form but we can)
                # let's instead compute state-sensitive directly using mask
        
            # Computing vectorized version:
            # ∇_i g_k = - (h_{i+k} + h_{i-k}) (only safe indices)
            # Gradient as double sum: ∑_k weight_k * (h_{i+k} + h_{i-k})
            
            # More fast way: 
            # Gradients= [1] - 2*sum_k weights_k * shift_k conv h 
            
            # Convolution of weights with h to obtain weighting for each h[i]
            # We compute the gradient efficiently
            
            # Direct computation of gradients using convolution for efficiency
            # This uses the underlying linear structure.
            # We compute g_sequence (shape 2N-1) = target_sum - corr (same size)
            # Gradient of ∇_h_i smooth_max_g = 1 - sum_k weights_k * (h_{i+k} + h_{i-k})
            
            grad = np.zeros(n)
            for k in range(2*n - 1):
                k_shift = k - n + 1
                if k_shift == 0:
                    # At shift 0, the gradient term doesn't contribute, so skip
                    continue
                if k_shift > 0:
                    shifted_h = np.roll(h, k_shift)
                    shifted_h[:k_shift] = 0
                else:
                    shifted_h = np.roll(h, k_shift)
                    shifted_h[k_shift:] = 0
                # Define weights with softmax applied to g
                temp_weight_k = softmax_g[k]
                # Add contribution: - temp_weight_k * (shifted_h)
                grad += -temp_weight_k * shifted_h
                
            grad = 1 - grad                
            
            # Step in opposite direction of gradients
            h = h - eta * grad
            
            # Normalize and project back
            h = project_to_boxed_simplex(h, target_sum)
            
            # Rescale/explore
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Periodic update of step size
            if round % 100 == 0:
                eta *= decay
                
            # Optional: add Nesterov momentum
            # But keep it simple for now
            
        return best_h, best_c5
    
    # Apply continuous refinement
    h_refined, c5_refined = smooth_max_gd(best_h_bin, num_iter=2000)
    
    # Return best overall solution
    return (h_refined.tolist(), c5_refined, n_points)
