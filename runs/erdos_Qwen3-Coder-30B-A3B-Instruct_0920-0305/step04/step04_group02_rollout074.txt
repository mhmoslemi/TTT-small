import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient(h, d_star):
    """Compute the sub-gradient for the overlap with shift d_star"""
    n = len(h)
    g = np.zeros(n)
    if d_star > 0:
        # h_i * (1 - h_{i+d_star}) contributes
        # for i in range(n - d_star):
        #   g[i] += 1 - h[i + d_star]
        g[:n - d_star] += 1
        g[d_star:] -= h[:n - d_star]
    elif d_star < 0:
        # h_i * (1 - h_{i+d_star})  = h_i * (1 - h_{i-d_star})
        # g[i] += 1 - h[i - d_star]
        g[-d_star:] += 1
        g[:-d_star] -= h[-d_star:]
    else:
        # d_star == 0, it's not a valid shift for overlap
        pass
    return g

def binary_optimal(n):
    """Solve the integer version with MILP to get minimum t s.t. all overlaps <= t"""
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    
    # Mass constraint: sum(h) = n/2
    mass_constraint = cp.sum(x) == n // 2
    # Symmetry breaking
    symmetry_constraint = x[0] == 1
    
    constraints = [mass_constraint, symmetry_constraint]
    
    # For each shift d in -n+1 to n-1, except 0
    for d in range(-n+1, n):
        if d == 0:
            continue
        w_list = []
        for i in range(n):
            if i + d >= 0 and i + d < n:
                w = cp.Variable(boolean=True)
                w_constr = [w <= x[i],
                            w <= x[i+d],
                            w >= x[i] + x[i+d] - 1]
                constraints.extend(w_constr)
                w_list.append(w)
        
        # Count (1,0) pairs for shift d.
        # Count = sum(x) - sum(w)
        N_d = cp.sum(x) - cp.sum(w_list)
        constraints.append(N_d <= t)
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    solver = cp.CBC
    try:
        prob.solve(solver=solver)
    except:
        solver = cp.GLPK_MI
        try:
            prob.solve(solver=solver)
        except:
            solver = cp.ECOS_BB
            try:
                prob.solve(solver=solver)
            except:
                # fallback to SLSQP
                return None, None

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INFEASIBLE]:
        x_opt = np.round(x.value).astype(int)
        t_opt = t.value
        return x_opt, t_opt
    else:
        return None, None

def refine_continuous(h0, remaining_time):
    # Run projected sub-gradient descent
    h = h0.astype(float)
    best = h.copy()
    best_val = evaluate_c5(h)
    
    start_time = time.time()
    max_iter = 200  # iterations per refinement call
    iter = 0
    
    while iter < max_iter and (time.time() - start_time) < remaining_time:
        # Compute overlaps for all shifts
        corr = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr)
        # Identify shifts that achieve the maximum
        all_max_d = np.where(corr == max_overlap)[0]
        # Convert indices to actual shifts d (negative idx means d is negative)
        D_max = []
        for idx in all_max_d:
            if idx < len(h):  # shift from 0 to n-1 range
                d = idx - (len(h) - 1)
            else:  # Handle wrap-around (this should not happen)
                d = idx - len(h) - (len(h) - 1)
            if d != 0:
                D_max.append(d)
        
        # Average subgradients over the set of worst shifts
        avg_grad = np.zeros_like(h)
        for d in D_max:
            g = subgradient(h, d)
            avg_grad += g
        avg_grad /= len(D_max) if len(D_max) > 0 else 1
        
        # Line search
        alpha = 0.5
        while alpha > 1e-10:
            h_new = h - alpha * avg_grad
            # Project to bounds
            h_new = np.clip(h_new, 0, 1)
            # Enforce sum constraint using scaling
            scaling = (len(h) // 2) / np.sum(h_new)
            h_new *= scaling

            val_new = evaluate_c5(h_new)
            if val_new < best_val - 1e-9:
                h = h_new
                best = h_new.copy()
                best_val = val_new
                break
            alpha /= 2

        # Occasionally add jitter
        if np.random.rand() < 0.2:
            h = h + np.random.normal(0, 0.005, len(h))
            h = np.clip(h, 0, 1)
            scaling = (len(h) // 2) / np.sum(h)
            h *= scaling
        
        iter += 1
    
    # Last resort: solver SLSQP
    try:
        def obj(h_arg):
            return evaluate_c5(h_arg)
        
        res = minimize(obj, best, method='SLSQP', 
                       bounds=[(0, 1) for _ in range(len(best))],
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - len(x)//2},
                       options={'maxiter': 20})
        if res.success:
            h_final = res.x
            val_final = evaluate_c5(h_final)
            if val_final < best_val:
                best = h_final
                best_val = val_final
    except:
        pass

    return best, best_val
