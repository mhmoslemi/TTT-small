import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5_fft(h):
    n = len(h)
    if n == 0:
        return 0.0
    # Reverse h
    h_rev = h[::-1]
    b = 1.0 - h
    L = 2 * n - 1
    # Compute FFT lengths
    hf = np.fft.rfft(h_rev, n=L)
    bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(hf * bf, n=L)
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)

def project_to_sum_box(x, target_sum):
    # x vector real numbers.
    # Solve for λ such that sum(min(1, max(0, x_i + λ))) = target_sum.
    # Use binary search.
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(x + mid, 0, 1)
        s = proj.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo+hi)/2.0, 0, 1)

def simulated_annealing(h_initial, target_sum, time_limit, seed):
    np.random.seed(seed)
    n = len(h_initial)
    h = np.copy(h_initial)
    # Ensure feasibility via project
    h = project_to_sum_box(h, target_sum)
    # Ensure values in bounds
    h = np.clip(h, 0, 1)  # after projection they already are in bounds.
    cur_c = evaluate_c5_fft(h)
    best_h = h.copy()
    best_c = cur_c

    # Set temperature schedule: start high 
    T0 = cur_c * 0.005 if cur_c > 0 else 0.005
    Tmin = 1e-6
    start = time.time()
    iter = 0
    # We'll compute max iter based on time.
    while time.time() - start < time_limit:
        # Random move
        i, j = np.random.choice(n, size=2, replace=False)
        # delta range
        delta_max = min(h[i], 1 - h[j])
        delta_min = -min(1 - h[i], h[j])
        if delta_max <= delta_min:
            continue
        # Choose delta uniform in [delta_min, delta_max]
        delta = np.random.uniform(delta_min, delta_max)
        # Update h
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # evaluate
        c_new = evaluate_c5_fft(h_new)
        # compute temperature based on proportion of elapsed time
        elapsed = time.time() - start
        progress = min(elapsed / time_limit, 1.0)
        T = T0 * (1 - progress) + Tmin * progress
        if c_new < cur_c or np.random.rand() < np.exp(-(c_new - cur_c) / (T + 1e-12)):
            h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        iter += 1
    return best_h, best_c
