import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def decode_h_from_params(params, n=200):
    """Decode a parameter vector into a full h vector.
    Params are heights at K breakpoints, arranged with indices 0..K."""
    K = len(params) - 1  # number of breakpoints
    # Place breakpoints in [0, 1]
    p = np.linspace(0, 1, K+1)
    # Sort params in decreasing order to enforce monotonicity
    h_vals = np.sort(params)[::-1]
    # Clip to [0,1]
    h_vals = np.clip(h_vals, 0, 1)
    # Interpolate to full vector, using symmetry
    x = np.linspace(0, 1, n//2 + 1)
    h_half = np.interp(x, p, h_vals)
    h_full = np.concatenate([h_half[:-1], h_half[::-1]])
    # Normalize to integral = 1
    total = np.sum(h_full) * (2.0 / n)
    if total > 0:
        h_full = h_full / total * 1.0
    # Ensure entries are in [0, 1] due to interpolation and scaling
    h_full = np.clip(h_full, 0, 1)
    return h_full

def smooth_max_surrogate(h, beta=20.0):
    """Evaluate a smooth approximation of the maximum as log-sum-exp"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    # corr indices range from -(n-1) to n-1
    corr = corr * dx
    # compute log-sum-exp with beta scaling
    exp_beta_corr = np.exp(beta * corr)
    log_sum_exp = np.log(np.sum(exp_beta_corr))
    return log_sum_exp / beta

def compute_gradient(h, beta=20.0):
    """Compute the gradient w.r.t. h of the smooth max."""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full") * dx
    exp_beta_corr = np.exp(beta * corr)
    Z = np.sum(exp_beta_corr)
    weights = exp_beta_corr / Z
    # compute gradients
    grad = np.zeros(n)
    # This involves computing ∂f_k/∂h_i = dx*(1 - h_{i+k} - h_{i-k})
    for k in range(-n+1, n):
        if k < 0:
            # indices for h_{i+k} and h_{i-k}
            hk_plus = h[-k:]  # overlap with h shifted forward
            hk_minus = h[:n+k]  # overlap with h shifted backward
            # construct full vector for h_{i+k} values
            h_k_plus = np.concatenate([np.zeros(-k), hk_plus])
            # construct full vector for h_{i-k} values
            h_k_minus = np.concatenate([hk_minus, np.zeros(k)])
        else:
            # For k >= 0
            hk_plus = h[k:] if k > 0 else h
            hk_minus = h[:n-k] if k < n-1 else h
            # construct full vector for h_{i+k} values
            h_k_plus = np.concatenate([np.zeros(k), hk_plus])
            # construct full vector for h_{i-k} values
            h_k_minus = np.concatenate([hk_minus, np.zeros(k)])
        # add contribution from shift k
        diff_contrib = (1 - h_k_plus - h_k_minus) * dx
        # element-wise product with weights
        for i in range(n):
            if k >= 0:
                grad[i] += weights[k + n - 1] * diff_contrib[i]
            else:
                grad[i] += weights[k + n - 1] * diff_contrib[i]
    return grad

def project_onto_simplex_and_box(h, tol=1e-10):
    """Project h onto the set {h: 0 <= h_i <= 1, sum(h)*dx = 1}"""
    n = len(h)
    dx = 2.0 / n
    S = n / 2.0
    # initial guess
    h_proj = np.array(h)
    # clip to [0,1]
    h_proj = np.clip(h_proj, 0, 1)
    # adjust scaling so that sum equals S
    total = np.sum(h_proj) * dx
    if abs(total) < tol:
        # avoid division by zero
        h_proj.fill(0.0)
        return h_proj
    h_proj = h_proj * S / total
    # clip again after rescaling
    h_proj = np.clip(h_proj, 0, 1)
    # sum should now be close to S, but re-project if needed
    total = np.sum(h_proj) * dx
    if abs(total - S) > 1e-10:
        # binary search for correct scale
        def scale_sum(scale):
            h_try = np.clip(scale * h_proj, 0, 1)
            return np.sum(h_try) * dx
        low, high = 0.99, 1.01
        for _ in range(30):
            mid = (low + high) / 2
            sums = scale_sum(mid)
            if abs(sums - S) < 1e-9:
                h_proj = np.clip(mid * h_proj, 0, 1)
                return h_proj
            elif sums < S:
                low = mid
            else:
                high = mid
        h_proj = np.clip(np.clip(S * h_proj / total, 0, 1), 0, 1)
    return h_proj

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    total_budget = budget_s
    t_de = 0.7 * total_budget
    t_grad = 0.2 * total_budget
    t_last = 0.1 * total_budget
    
    n_points = 200
    K = 5  # number of breakpoints for piecewise linear

    bounds = [(0, 1) for _ in range(K+1)]
    
    best_c5 = float('inf')
    best_h = None

    # Phase 1: Global Optimization using DE
    def obj_func(params):
        h = decode_h_from_params(params, n_points)
        return evaluate_c5(h)

    if (time.time() - start_time) < t_de:
        try:
            de_res = differential_evolution(
                obj_func,
                bounds,
                seed=seed,
                maxiter=200,
                popsize=15
            )
            best_sol = de_res.x
            h_candidate = decode_h_from_params(best_sol, n_points)
            c5_candidate = evaluate_c5(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
        except Exception as e:
            pass

    # otherwise use a simpler baseline
    if best_h is None:
        h_zero = np.zeros(n_points)
        # Start with simple symmetric profile
        x = np.linspace(0, 2, n_points)
        h0 = np.maximum(0, 1 - np.abs(x - 1) * 0.8)
        h0 = h0 / (np.sum(h0) * 2.0 / n_points)
        h0 = np.clip(h0, 0, 1)
        best_h = h0
        best_c5 = evaluate_c5(h0)

    # Phase 2: Smooth gradient refinement with projected updates
    if (time.time() - start_time) < t_grad + t_de:
        final_h = np.copy(best_h)
        dx = 2.0 / n_points
        beta = 20.0
        eta = 0.1
        max_iter = 500
        t_left = total_budget - (time.time() - start_time)
        t_iter = min(t_left / max_iter, 0.01)

        # Main loop
        for i in range(max_iter):
            if (time.time() - start_time) >= total_budget:
                break
            # Compute gradient
            try:
                grad = compute_gradient(final_h, beta)
                # Step
                new_h = final_h - eta * grad
                # Project
                final_h = project_onto_simplex_and_box(new_h)
                # Update if improved
                c5_test = evaluate_c5(final_h)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = np.copy(final_h)
            except Exception:
                pass

    # Phase 3: Final polishing with SLSQP if necessary
    if (time.time() - start_time) < total_budget and best_c5 > 0.379:
        try:
            bounds_full = [(0, 1) for _ in range(n_points)]
            def obj_slsqp(h):
                return evaluate_c5(h)
            def constraint(h):
                return np.sum(h) * dx - 1.0
            cons = {'type': 'eq', 'fun': constraint}
            res = minimize(
                obj_slsqp,
                best_h,
                method='SLSQP',
                bounds=bounds_full,
                constraints=cons,
                options={'ftol': 1e-12, 'gtol': 1e-12, 'maxiter': 200}
            )
            if res.success:
                h_slsqp = res.x
                c5_slsqp = evaluate_c5(h_slsqp)
                if c5_slsqp < best_c5:
                    best_c5 = c5_slsqp
                    best_h = h_slsqp
        except:
            pass
    
    return (best_h.tolist(), best_c5, n_points)
