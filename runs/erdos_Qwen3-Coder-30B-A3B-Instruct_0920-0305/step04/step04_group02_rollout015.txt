import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient(h, d_star):
    """Compute the subgradient for the shift d_star"""
    n = len(h)
    grad = np.zeros(n)
    if d_star > 0:
        # h[i] - h[i]*h[i+d_star] -> partial derivative wrt h[i] is 1 - h[i+d_star]
        # For i+d_star < n, the term h[i+d_star] contributes
        for i in range(n - d_star):
            grad[i] += 1 - h[i + d_star]
        # For i+d_star >= n, h[i+d_star] is beyond grid, so ignored
    elif d_star < 0:
        # Similar for negative shift
        abs_d = -d_star
        for i in range(abs_d, n):
            grad[i] += 1 - h[i - abs_d]
    else:
        # d_star = 0 not valid for overlap
        raise ValueError("shift d_star must be non-zero")
    return grad

def binary_optimal(n):
    """Solve the integer (binary) problem via MILP to find minimum t"""
    # Variables: x_i binary, t integer
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    
    # Mass constraint
    mass = cp.sum(x) == n // 2
    # Symmetry breaking
    symmetry = x[0] == 1
    
    # Components of the constraint: for each shift d != 0
    # For each shift d ≠ 0, compute N_d = sum(x_i) - sum(w_{i,d})
    max_shift = n - 1
    constraints = [mass, symmetry, t >= 0]
    
    for d in range(-max_shift, max_shift + 1):
        if d == 0:
            continue
        # Build w_{i,d} for relevant i
        w_list = []
        for i in range(n):
            if 0 <= i + d < n:
                w = cp.Variable(boolean=True)
                w_list.append(w)
                # Linearization of w = x_i * x_{i+d} via McCormick
                constraints.append(w <= x[i])
                constraints.append(w <= x[i + d])
                constraints.append(w >= x[i] + x[i + d] - 1)
            else:
                # Waste – don't define w variables that are out of flux
                pass
                
        # Count of (1,0) pairs for this shift is sum(x_i) - sum(w_{i,d})
        # Constraint: sum(x_i) - sum(w_{i,d}) <= t
        if w_list:
            sum_w = cp.sum(w_list)
            count = cp.sum(x) - sum_w
            constraints.append(count <= t)
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers with decreasing preference
    solver = None
    try:
        prob.solve(solver=cp.CBC, verbose=False)
        solver = 'CBC'
    except:
        try:
            prob.solve(solver=cp.GLPK_MI, verbose=False)
            solver = 'GLPK_MI'
        except:
            try:
                prob.solve(solver=cp.ECOS_BB, verbose=False)
                solver = 'ECOS_BB'
            except:
                # Out of patience
                return None, float('inf')
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        x_opt = x.value.astype(float)
        t_opt = t.value
        return x_opt, t_opt
    else:
        return None, float('inf')

def refine_continuous(h0, time_remaining):
    """Refine a binary H to a fractional one via subgradient descent"""
    h = h0.astype(float)
    best = h.copy()
    best_val = evaluate_c5(h)
    dx = 2.0 / len(h)
    max_iter = 200
    
    start = time.time()
    start_trial = time.time()
    it = 0
    
    while it < max_iter and (time.time() - start) < time_remaining - 0.5:
        it += 1
        # Find the current worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        max_shift_val = corr[max_shift_idx]
        # Indices are awkward, 0 -> shift n -1, ..., n//2 -> shift 0, ..., n-1 -> shift n-1
        max_shift = max_shift_idx - (len(corr) // 2)
        
        # Compute subgradient of that shift
        g = subgradient(h, max_shift)
        
        # Gradient descent step
        alpha = 0.5
        h_new = h - alpha * g
        h_new = np.clip(h_new, 0, 1)
        # Enforce sum constraint
        scale = 1.0 / (np.sum(h_new) * dx)
        h_new = h_new * scale
        
        if evaluate_c5(h_new) < best_val:
            h = h_new.copy()
            best = h_new.copy()
            best_val = evaluate_c5(h_new)
        else:
            # Occasionally add a small random perturbation
            if np.random.random() < 0.2:
                noise = np.random.normal(0, 0.01, len(h_new))
                h_new = h_new + noise
                h_new = np.clip(h_new, 0, 1)
                # Again enforce sum constraint
                scale = 1.0 / (np.sum(h_new) * dx)
                h_new = h_new * scale
                if evaluate_c5(h_new) < best_val:
                    h = h_new.copy()
                    best = h_new.copy()
                    best_val = evaluate_c5(h_new)
    return best, best_val

            return np.sum(h) * dx - 1.0
