import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlation
import time
import math

# Global variable for constraint
TARGET_SUM = None

# Helper for FFT-based correlation computation
def compute_corr_fft(h):
    n = len(h)
    # Pad to avoid wrap-around (length 2*n-1)
    h_rev = h[::-1]
    b = 1 - h
    # Convolve h_rev with b
    corr_full = np.convolve(h_rev, b, mode='full')
    # Return only the valid cross-correlation values
    return corr_full[n-1:2*n-1]

def c5_from_corr(corr, n):
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """
    Projects x onto the set {y : sum(y) = target_sum, 0 <= y_i <= 1}
    Uses shift-and-clip with binary search for lambda.
    """
    n = len(x)
    # Binary search for lambda such that sum(clip(x + lambda, 0, 1)) = target_sum.
    l, r = -1.0, 1.0
    for _ in range(50):  # convergence threshold
        mid = (l + r) / 2
        y = np.clip(x + mid, 0, 1)
        s = np.sum(y)
        if s < target_sum:
            l = mid
        else:
            r = mid
    y = np.clip(x + l, 0, 1)
    return y

def random_pair_move(h, rng):
    """Perform a single mass transfer move"""
    i, j = rng.choice(len(h), 2, replace=False)
    h_i, h_j = h[i], h[j]
    # Compute the range of valid delta values
    delta_min = max(-h_j, h_i - 1)
    delta_max = min(h_i, 1 - h_j)
    if delta_min >= delta_max:
        return None  # no valid transfer
    delta = rng.uniform(delta_min, delta_max)
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def guided_move(h, worst_lag):
    """A deterministic improvement step targeting the worst shift"""
    n = len(h)
    # Compute subgradient:
    # gradient[i] = h[i - worst_lag] + h[i + worst_lag]
    g = np.zeros(n)
    if worst_lag > 0:
        # h[i + worst_lag] for i = 0..n-1
        g[worst_lag:] += h[:-worst_lag]
        # h[i - worst_lag] for i = 0..n-1 (adding zeros where needed)
        g[:-worst_lag] += h[worst_lag:]
    else:
        # worst_lag <= 0
        idx = -worst_lag
        # h[i + worst_lag] = h[i - |worst_lag|] for i = 0..n-1
        if idx < n:
            g[:idx] += h[idx:]
        # h[i - worst_lag] for i = 0..n-1 = h[i + |worst_lag|] for i = 0..n-1
        if idx < n:
            g[idx:] += h[:-idx]

    # Find index of min and max for g
    i_min = np.argmin(g)
    i_max = np.argmax(g)
    # Transfer a bit
    h_new = h.copy()
    delta = min(h[i_min], 1 - h[i_max])
    h_new[i_min] -= delta
    h_new[i_max] += delta
    return h_new


def anneal(h0, run_seconds, rng):
    """Run a single simulated annealing phase"""
    n = len(h0)
    h = h0.copy()
    h = project_to_sum_box(h, TARGET_SUM)
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    T0 = 5e-3
    Tf = 1e-6
    t0 = time.time()
    iter_count = 0
    while time.time() - t0 < run_seconds:
        iter_count += 1
        # Compute current worst lag
        corr = compute_corr_fft(h)
        worst_lag = np.argmax(corr) - n + 1
        if iter_count % 10 == 0:
            # Do guided move
            h_trial = guided_move(h, worst_lag)
            if h_trial is not None:
                c5_trial = c5_from_corr(compute_corr_fft(h_trial), n)
                if c5_trial < current_c5 or rng.uniform() < math.exp(-(c5_trial - current_c5) / (T0 + (Tf - T0) * (time.time() - t0) / run_seconds)):
                    h = h_trial
                    current_c5 = c5_trial
        else:
            # Do random pair move
            h_trial = random_pair_move(h, rng)
            if h_trial is not None:
                c5_trial = c5_from_corr(compute_corr_fft(h_trial), n)
                if c5_trial < current_c5 or rng.uniform() < math.exp(-(c5_trial - current_c5) / (T0 + (Tf - T0) * (time.time() - t0) / run_seconds)):
                    h = h_trial
                    current_c5 = c5_trial
        # Update best
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    return best_h, best_c5


def local_refine(h):
    """Use SLSQP to refine h to reduce C5 """
    n = len(h)
    dx = 2.0 / n

    def obj(h):
        corr = compute_corr_fft(h)
        return np.max(corr) * dx

    def constraint_eq(h):
        return np.sum(h) * dx - 1.0

    bounds = [(0, 1)] * n
    cons = {'type': 'eq', 'fun': constraint_eq}
    result = minimize(
        obj,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12}
    )
    if result.success:
        final_h = result.x
        final_h = project_to_sum_box(final_h, TARGET_SUM)
        return final_h, obj(final_h)
    else:
        return h, obj(h)
