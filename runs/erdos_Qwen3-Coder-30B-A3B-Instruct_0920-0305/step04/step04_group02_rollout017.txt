import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def solve_binary_milp(n):
    # Build a binary MILP instance for the Erdos problem
    x = Variable(n, boolean=True)
    y = {}
    t = Variable(1, integer=True)
    
    # Set up binary constraints
    constraints = []
    for i in range(n):
        constraints.append(x[i] <= 1)
        constraints.append(x[i] >= 0)
    
    # For each pair (i, j) we have auxiliary variable y[i,j] = x[i]*x[j]
    # The shift indices we consider include all valid (i, i+d) pairs for d = 1..(n-1)
    shift_indices = []
    for d in range(1, n):  # shifts 1 through n-1
        for i in range(n):
            j = i + d
            if j < n:
                y[(i, j)] = Variable(1, boolean=True)
                constraints.append(y[(i, j)] <= x[i])
                constraints.append(y[(i, j)] <= x[j])
                constraints.append(y[(i, j)] >= x[i] + x[j] - 1)
                shift_indices.append((i, j, d))
    
    # Sum constraint (integral = 1)
    constraints.append(sum(x) == n / 2)
    
    # For each shift d, define N_d = sum_i x[i] * (1 - x[i+d])
    # which is sum_i x[i] - sum_{i,j} y[i,j]
    t_constraints = []
    for d in range(1, n):
        # Compute sum_i x[i] - sum_{i,i+d} y[i,i+d]
        N_d_terms = []
        for i in range(n):
            j = i + d
            if j < n:
                N_d_terms.append(y[(i, j)])
        # This is sum x[i] - sum y[i,i+d]
        N_d = sum(x) - sum(N_d_terms)
        t_constraints.append(N_d <= t)
    
    constraints += t_constraints
    
    # Objective function
    prob = Problem(Minimize(t), constraints)
    
    # Try available solvers
    for solver_name in ['CBC', 'GLPK_MI', 'ECOS_BB']:
        try:
            prob.solve(solver=solver_name)
            if prob.status == 'optimal':
                x_vals = np.array([x[i].value for i in range(n)])
                t_val = int(t.value)
                return x_vals, t_val
        except:
            continue
    
    # Fallback to approximate solver if needed
    try:
        prob.solve(solver='CBC', verbose=False)
        if prob.status == 'optimal':
            x_vals = np.array([x[i].value for i in range(n)])
            t_val = int(t.value)
            return x_vals, t_val
    except:
        pass
    
    # If MILP fails, return dummy result
    return np.ones(n) * 0.5, 8

def smooth_max(h, shifts, tau):
    n = len(h)
    c_values = []
    for k in shifts:
        c = 0.0
        for i in range(n):
            if 0 <= i + k < n:
                c += h[i] - h[i] * h[i + k]
            else:
                c += h[i]
        c_values.append(c)
    c_values = np.array(c_values)
    return tau * np.log(np.sum(np.exp(c_values / tau)))

def gradient_smooth_max(h, shifts, tau):
    n = len(h)
    # Precompute contributions to each shift
    c_values = []
    for k in shifts:
        c = 0.0
        for i in range(n):
            if 0 <= i + k < n:
                c += h[i] - h[i] * h[i + k]
            else:
                c += h[i]
        c_values.append(c)
    c_values = np.array(c_values)
    
    # Compute softmax weights
    exp_c = np.exp(c_values / tau)
    w = exp_c / np.sum(exp_c)
    
    # Compute gradients
    grad = np.zeros(n)
    for idx, k in enumerate(shifts):
        weight = w[idx]
        for i in range(n):
            if 0 <= i + k < n:
                grad[i] += weight * (1.0 - h[i + k])
            else:
                grad[i] += weight * 1.0
    
    return grad

def project_onto_simplex(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    scale = target_sum / current_sum
    h = h * scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = 1.0

    # Phase 1: Binary optimization using MILP
    h_bin, t_opt = solve_binary_milp(n_points)
    h_bin = h_bin.astype(float)
    
    # Format t_opt to ratio accordingly
    c5_bin = t_opt * dx
    
    # Initialize best as binary solution
    best_h = h_bin.copy()
    best_c5 = c5_bin
    
    # Now refine with smooth max gradient descent using continuation
    shifts = list(range(1, n_points))

    # If binary has achieved target, return early
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    time_left = lambda: budget_s * 0.95 - (time.time() - start_time)
    
    # Continue refinement for rest of the time
    num_restarts = 0
    max_restarts = 5
    tau = 0.5  # Initial temperature for smoothing
    max_iterations = 100
    decay_factor = 0.5
    
    while num_restarts < max_restarts and time_left() > 5:
        h = best_h.copy()
        # Reset tau for each restart
        current_tau = tau
        iter_count = 0
        
        iter_time = time.time()
        while current_tau > 1e-4 and time_left() > 5:
            if iter_count >= max_iterations:
                break
                
            iter_count += 1
            # Gradient descent step
            grad = gradient_smooth_max(h, shifts, current_tau)
            h = h - 0.2 * grad
            
            # Project onto feasible set
            h = project_onto_simplex(h, target_sum)
            
            # Update current_tau (continuation)
            if current_tau > 1e-3 and iter_count % 10 == 0:
                current_tau *= decay_factor
                
        # Re-evaluate after iteration
        h_c5 = evaluate_c5(h)
        if h_c5 < best_c5:
            best_h = h.copy()
            best_c5 = h_c5
            
        # Random restart
        h = best_h + np.random.normal(0, 0.02, n_points)
        h = project_onto_simplex(h, target_sum)
        num_restarts += 1
    
    # Final polishing with SLSQP
    h_final = best_h.copy()
    try:
        def obj_func(h):
            return evaluate_c5(h)
        
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            obj_func, 
            h_final, 
            method='SLSQP', 
            bounds=bounds, 
            options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_final = res.x
            final_c5 = evaluate_c5(h_final)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except Exception as e:
        pass
    
    # Should reach below target in time
    return (best_h.tolist(), best_c5, n_points)
