import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient(h, d_star, n):
    """Compute sub-gradient with respect to shift d_star"""
    g = np.zeros(n)
    if d_star > 0:
        # shift forward: h[i] contributes to h[i+d_star] term
        g[:n-d_star] = 1 - h[d_star:]
        g[n-d_star:] = 0
    elif d_star < 0:
        # shift backward: h[i] contributes to h[i+d_star] term
        g[-d_star:] = 1 - h[:-d_star]
        g[:-d_star] = 0
    else:
        # direct gradient of sum
        g = np.ones(n)
    return g

def binary_optimal(n):
    # Define variables
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)

    # Mass constraint
    constraints = [cp.sum(x) == n // 2, x >= 0, x <= 1]

    # Symmetry breaking
    constraints.append(x[0] == 1)

    # Add product variables and McCormick constraints
    for d in range(1, n):
        w = cp.Variable(n - d, boolean=True)  # w[i] = x[i]*x[i+d]
        for i in range(n - d):
            # McCormick relaxation for w[i] = x[i]*x[i+d]
            constraints.extend([
                w[i] <= x[i],
                w[i] <= x[i+d],
                w[i] >= x[i] + x[i+d] - 1
            ])
        # Overlap constraint
        N_d = cp.sum(x) - cp.sum(w)
        constraints.append(N_d <= t)

    # Optimize
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CBC)
    except:
        try:
            prob.solve(solver=cp.GLPK_MI)
        except:
            try:
                prob.solve(solver=cp.ECOS_BB)
            except:
                return None, None

    x_opt = x.value
    # Convert to float and scale properly
    h_f = x_opt.astype(float)
    return h_f, int(t.value)

def refine_continuous(h0, time_limit, start_time):
    n = len(h0)
    dx = 2.0 / n

    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    now = time.time()

    # Iteratively optimize with subgradient methods
    max_iter = 2000
    iter = 0

    while (time.time() - start_time) < (time_limit * 0.95) and iter < max_iter:
        # Compute all correlations
        corr = correlate(best_h, 1 - best_h, mode="full")
        # Find worst shift
        d_max = np.argmax(corr)
        max_val = corr[d_max]

        # Convert full index to shift value
        d_star = d_max - (n - 1)

        g = subgradient(best_h, d_star, n)

        # Try different step sizes
        for step_size in [0.1, 0.05, 0.01, 0.001]:
            h_new = best_h - step_size * g
            h_new = np.clip(h_new, 0, 1)

            # Renormalize
            scale = (n / 2) / np.sum(h_new)
            h_new = h_new * scale

            new_c5 = evaluate_c5(h_new)

            if new_c5 < best_c5 - 1e-10:
                best_h = h_new.copy()
                best_c5 = new_c5
                break  # Accept this improvement

        # Noise occasionally
        if np.random.rand() < 0.2 and (time.time() - start_time) < (time_limit * 0.90):
            h_new = best_h + np.random.normal(0, 0.005, n)
            h_new = np.clip(h_new, 0, 1)
            scale = (n / 2) / np.sum(h_new)
            h_new = h_new * scale
            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5 - 1e-10:
                best_h = h_new.copy()
                best_c5 = new_c5

        iter += 1

    # Final polishing with SLSQP if needed
    if time.time() - start_time < time_limit * 0.9:
        constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) * dx - 1.0}
        bounds = [(0, 1) for _ in range(n)]

        try:
            res = minimize(
                evaluate_c5, 
                best_h.copy(), 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass

    return best_h, best_c5
