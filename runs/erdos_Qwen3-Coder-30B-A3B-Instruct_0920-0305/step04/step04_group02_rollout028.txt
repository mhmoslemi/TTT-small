def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_binary_milp(n=42):

    # Variables
    x = cp.Variable(n, integer=True)
    t = cp.Variable(integer=True)

    # Constraints
    constraints = [x >= 0, x <= 1, cp.sum(x) == n/2]

    # Add bilinear constraints via McCormick envelopes
    # y[i,d] = x[i] * x[i+d]
    for d in range(-n+1, n):
        if d == 0:
            continue
        for i in range(n):
            if i+d >= 0 and i+d < n:
                y = cp.Variable(1, integer=True)
                constraints.extend([
                    y <= x[i],
                    y <= x[i+d],
                    y >= x[i] + x[i+d] - 1,
                ])

    # Add inequality constraints for shift bounds
    for d in range(-n+1, n):
        if d == 0:
            continue
        shift_sum = 0
        for i in range(n):
            if i+d >= 0 and i+d < n:
                shift_sum += x[i] * x[i+d]
        # N_d = sum x_i - sum (x_i * x_{i+d})
        # So we want sum x_i - sum (x_i * x_{i+d}) <= t => t >= n/2 - sum(x_i x_{i+d})
        # i.e., t >= n/2 - shift_sum => shift_sum <= n/2 - t => sum(x_i x_{i+d}) >= n/2 - t

        # Or: sum_{i,d} x_i x_{i+d} <= (n/2 - t) - max over remaining terms

        # So N_d = (n/2) - sum_{i,j s.t. j=i+d} x_i x_j <= t -> sum_{i,j s.t. j=i+d} x_i x_j >= (n/2) - t

    """

    def smooth_max_cost(h, tau):

        for d in range(-n+1, n):

                continue
            corr = 0.0
            for i in range(n):

                    corr += h[i] * (1 - h[i+d])
                else:

                    pass
            overlaps.append(corr)
        # Use logsumexp to compute smooth max log(\sum exp(x_i / tau))
        log_sum_exp = np.log(np.sum(np.exp(np.array(overlaps) / tau)))
        return tau * log_sum_exp

                else:

        grad = np.zeros(n)

                    # But more accurately, factor out everything for grad w respect to h_i 
                    # across multiple terms
                    pass
        return np.zeros(n)
    
    # Forward function for gradient update
    # We'll do this numerically using finite differences for now
    def fun(h):
        return smooth_max_cost(h, tau)

    def grad(h):
        # Approximate gradient
        eps = 1e-6
        grad = np.zeros(len(h))
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            grad[i] = (fun(h_plus) - fun(h_minus)) / (2 * eps)
        return grad

    for tau in tau_schedule:
