import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

        # sum constraint
        constraints = [cp.sum(x) == n_points // 2]
        
        # Define auxiliary variables for products: y[i,d] = x[i] * x[i+d]
        # Use linear constraints to model AND
        y = {}
        shift_indices = list(range(-n_points+1, 0)) + list(range(1, n_points))
        
        for d in shift_indices:
            for i in range(n_points):
                if (i + d) >= 0 and (i + d) < n_points:
                    y_key = (i, d)
                    y[y_key] = cp.Variable(boolean=True)
                    constraints += [
                        y[y_key] <= x[i],
                        y[y_key] <= x[i + d],
                        y[y_key] >= x[i] + x[i + d] - 1
                    ]
        
        # Maximize overlap across all shifts, use t as upper bound
        t = cp.Variable(integer=True)
        for d in shift_indices:
            if d == 0:
                continue
            expr = cp.sum(x) - cp.sum([y[(i, d)] for i in range(n_points) if (i+d) >= 0 and (i+d) < n_points])
            constraints += [expr <= t]
        
        # Minimize t
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, verbose=False, max_iters=5000)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_sol = np.round(x.value).astype(int)

    def smooth_max_heuristic(h_vals, tau=0.1):
        """Compute smooth approximation to overlap using log-sum-exp"""
        n = len(h_vals)
        overlaps = []
        for d in range(-n+1, n):
            if d == 0:

            o = 0
            for i in range(n):

                if 0 <= j < n:
                    o += h_vals[i] - h_vals[i] * h_vals[j]
            overlaps.append(o)
        if len(overlaps) == 0:
            return 0.0

        # Log-sum-exp of overlaps to get a smooth surrogate
        # Add small epsilon to prevent numerical issues
        overlaps = np.array(overlaps)
        return tau * np.log(np.sum(np.exp(overlaps / tau)) + 1e-15)

        """Compute gradient of smooth surrogate"""

        grad = np.zeros(n)
        overlap_vals = []

            d_vals.append(d)

        overlap_vals = np.array(overlap_vals) / tau
        weights = np.exp(overlap_vals - np.max(overlap_vals))  # stable
        weights = weights / np.sum(weights)
        
        # Compute gradient for each h_i
        for idx in range(len(d_vals)):
            d = d_vals[idx]
            w = weights[idx]
            for i in range(n):
                j = i + d
                if 0 <= j < n:
                    # for shift d: dC_d/dh_i = 1 - h[j] (if j is valid)
                    grad[i] += w * (1.0 - h_vals[j])
                    # And dC_d/dh_j = -h[i] (if i is valid)
                    grad[j] += w * (-h_vals[i])
                    
        return grad

        return np.clip(h, 0, 1)

        # integrate to get small dx
        h_scaled = h.copy()
        s = np.sum(h_scaled) * dx
        if s == 0:
            return h_scaled  # Avoid divide by zero
        return h_scaled * (0.5 / s)
