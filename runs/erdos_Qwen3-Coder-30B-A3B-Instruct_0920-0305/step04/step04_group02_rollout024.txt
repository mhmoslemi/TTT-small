import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient(h, d_star):
    """Compute subgradient of overlap for shift d_star"""
    n = len(h)
    g = np.zeros(n)
    if d_star > 0:
        # h_i contributes to h_{i+d_star} if i+d_star < n
        g[:-d_star] = 1 - h[d_star:]
    elif d_star < 0:
        # h_i contributes to h_{i+d_star} if i+d_star >= 0
        g[-d_star:] = 1 - h[:d_star]
    else:
        # d=0 case: this should not happen in our context
        pass
    return g

def binary_optimal(n):
    """Solve MILP to get the optimal integer solution"""
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    
    # Mass constraint
    mass = cp.sum(x) == n // 2
    
    # Symmetry breaking
    symmetry = x[0] == 1
    
    # McCormick constraints and overlap counts
    constraints = [mass, symmetry, t >= 0]
    
    # We create w_{i,d} variables for each non-zero shift d
    # and add constraints w_{i,d} <= x_i ∧ w_{i,d} <= x_{i+d} ∧ w_{i,d} >= x_i + x_{i+d} - 1
    
    all_overlaps = []
    
    for d in range(-n+1, n):
        if d == 0: continue
            
        # Create w_{i,d}
        w = cp.Variable(n, boolean=True)
        w_cons = []
        for i in range(n):
            if i + d >= 0 and i + d < n:
                w_cons.extend([
                    w[i] <= x[i],
                    w[i] <= x[i+d],
                    w[i] >= x[i] + x[i+d] - 1
                ])
        constraints.extend(w_cons)
        
        # Compute overlap = sum(x) - sum(w)
        overlap = cp.sum(x) - cp.sum(w)
        all_overlaps.append(overlap)
        constraints.append(overlap <= t)
    
    # Objective: minimize t
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    # Try CBC, GLPK_MI, ECOS_BB
    try:
        prob.solve(solver=cp.CBC, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_sol = np.round(x.value).astype(int)
            return x_sol, t.value
    except:
        pass
        
    try:
        prob.solve(solver=cp.GLPK_MI, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_sol = np.round(x.value).astype(int)
            return x_sol, t.value
    except:
        pass
        
    try:
        prob.solve(solver=cp.ECOS_BB, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_sol = np.round(x.value).astype(int)
            return x_sol, t.value
    except:
        pass
    
    # Fallback: fail to solve by letting y be integer approximation
    try:
        prob.solve(solver=cp.ECOS_BB)
        x_sol = np.round(x.value).astype(int)
        return x_sol, t.value
    except:
        # Likely failed or not solved; let's start from initial_h_values
        return None, None

def refine_continuous(h0, time_left):
    """Refine the continuous function starting from h0"""
    best_h = h0.copy()
    best_val = evaluate_c5(best_h)
    n = len(h0)
    dx = 2.0 / n
    max_iters = 500
    iter_count = 0
    start_time = time.time()
    
    h = h0.copy()
    
    while iter_count < max_iters and (time.time() - start_time) < time_left * 0.9:
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode="full")
        max_shift = np.argmax(corr)
        max_val = corr[max_shift]
        
        # Identify all shifts that achieve this maximum
        D_max = np.where(np.abs(corr - max_val) < 1e-9)[0]
        
        # Adjust index for negative to positive mapping
        d_max_list = []
        for idx in D_max:
            if idx < n:
                d_max_list.append(-(n-1) + idx)
            else:
                d_max_list.append(idx - (n-1))
                
        # Compute sub-gradient using average
        g = np.zeros(n)
        for d in d_max_list:
            if d > 0:
                # h_i contributes to h_{i+d} if i+d < n
                g[:-d] += 1 - h[d:]
            elif d < 0:
                # h_i contributes to h_{i+d} if i+d >= 0
                g[-d:] += 1 - h[:-d]
            else:
                pass

        # Gradient step
        alpha = 0.5
        h_new = h - alpha * g
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Enforce sum constraint
        current_sum = np.sum(h_new)
        if abs(current_sum) > 1e-10:
            scale = (n/2) / current_sum
            h_new *= scale
            
        # Evaluate
        new_val = evaluate_c5(h_new)
        if new_val < best_val - 1e-9:
            best_h = h_new.copy()
            best_val = new_val
        
        # Update h
        h = h_new.copy()
        iter_count += 1
    
    # Optional: Final SLSQP polish
    try:
        bounds = [(0, 1) for _ in range(n)]
        
        def objective(h):
            return evaluate_c5(h)
            
        def constraint_eq(h):
            dx = 2.0 / n
            return np.sum(h) * dx - 1.0
            
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 50})
        if res.success:
            h_candidate = res.x
            dx = 2.0 / n
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_val:
                return h_candidate, c5_candidate
    except:
        pass
    
    return best_h, best_val
