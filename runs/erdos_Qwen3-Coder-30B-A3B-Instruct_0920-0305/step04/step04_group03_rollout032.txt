import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def project_to_boxed_simplex(v, target_sum):
    # Clip to [0,1] first
    v = np.clip(v, 0, 1)
    # Then adjust to meet target sum
    s = np.sum(v)
    if abs(s - target_sum) < 1e-12:
        return v
    diff = target_sum - s
    # Add diff to entries not at bounds
    free = (v > 0) & (v < 1)
    nfree = np.sum(free)
    if nfree > 0:
        v[free] += diff / nfree
        v = np.clip(v, 0, 1)
    return v

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
