import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient(h, d_star):
    """Return subgradient for the overlap at shift d_star."""
    n = len(h)
    g = np.zeros(n)
    if d_star > 0:
        # h[i] * (1 - h[i+d_star])
        for i in range(n - d_star):
            g[i] += 1 - h[i + d_star]
        for i in range(d_star, n):
            g[i] -= h[i - d_star]
    elif d_star < 0:
        # h[i] * (1 - h[i+d_star])
        for i in range(n + d_star):
            g[i] -= h[i - d_star]
        for i in range(-d_star, n):
            g[i] += 1 - h[i + d_star]
    else:
        # d_star == 0
        # h[i] - h[i]^2
        for i in range(n):
            g[i] += 1 - 2 * h[i]
    return g

def binary_optimal(n):
    """Solve the integer optimisation problem using MILP."""
    # Variables
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)

    # Constraints
    mass = cp.sum(x) == n // 2
    constraints = [mass, t >= 0]
    # Enforce t >= Nd for each shift
    dx = 2.0 / n
    for d in range(1, n):
        w = cp.Variable(n, boolean=True)
        # Product constraints (McCormick)
        for i in range(n):
            if i + d < n:
                constraints.append(w[i] <= x[i])
                constraints.append(w[i] <= x[i + d])
                constraints.append(w[i] >= x[i] + x[i + d] - 1)
        # The count for shift d is sum(x) - sum(w)
        Nd = cp.sum(x) - cp.sum(w)
        constraints.append(Nd <= t)

    # Symmetry breaking to cut symmetry
    constraints.append(x[0] == 1)

    # Objective
    problem = cp.Problem(cp.Minimize(t), constraints)
    try:
        problem.solve(solver=cp.CBC)
    except Exception:
        try:
            problem.solve(solver=cp.GLPK_MI)
        except Exception:
            try:
                problem.solve(solver=cp.ECOS_BB)
            except Exception:
                return None, None

    if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        x_opt = np.round(x.value).astype(int)
        t_opt = int(t.value)
        return x_opt, t_opt
    return None, None

def refine_continuous(h0, time_left_s):
    """Perform gradient-based refinement."""
    # Ensure there is enough time
    if time_left_s < 0.01:
        return h0, evaluate_c5(h0)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()
    n = len(h0)
    dx = 2.0 / n
    max_iter = 500
    iter_count = 0
    step_size = 0.8
    
    while iter_count < max_iter and (time.time() - start_time) < time_left_s:
        # Correlation to find worst shift
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_val = np.max(corr)
        d_indices = np.where(corr == max_val)[0]
        d_star = d_indices[0] - (n - 1)  # Convert from full to shift index
        
        # Compute subgradient
        g = subgradient(best_h, d_star)
        
        # Gradient step
        h_new = best_h - step_size * g
        
        # Clip to bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Enforce sum constraint
        scale = (n // 2) / np.sum(h_new)
        h_new *= scale
        
        # Check if improved
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            step_size = 0.8  # reset step size
        else:
            step_size *= 0.99  # diminishing step size

        iter_count += 1

    # Final polish with SLSQP if useful
    bounds = [(0, 1) for _ in range(n)]
    def objective(h):
        return evaluate_c5(h)

    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    cons = {'type': 'eq', 'fun': constraint_eq}
    try:
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
        if res.success:
            h_candidate = res.x
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
    except:
        pass

    return best_h, best_c5
