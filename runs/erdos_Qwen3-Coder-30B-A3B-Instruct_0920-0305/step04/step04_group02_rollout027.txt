import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from cvxpy import *
import time
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_binary_milp(n):
    """Solve the binary version of the problem using MILP"""
    # Binary variables for h
    h_bin = cp.Variable(n, boolean=True)
    
    # Auxiliary variables for y_{i,d} = h_i * h_{i+d}
    # We'll compute and store all necessary pairs
    d_range = range(-n+1, n)  # From -(n-1) to n-1

    # All valid (i,d) pairs that lead to a feasible computation
    valid_pairs = []
    for d in d_range:
        for i in range(n):
            if 0 <= i + d < n:
                valid_pairs.append((i, d))
    
    # Create binary variables
    y = {}
    for i, d in valid_pairs:
        y[(i, d)] = cp.Variable(boolean=True)
    
    # Set the constraint
    # sum(h) = n/2
    constraints = [cp.sum(h_bin) == n // 2]
    
    # Product linearization: y[i,d] = h[i] * h[i+d]
    for i, d in valid_pairs:
        constraints += [
            y[(i, d)] <= h_bin[i],
            y[(i, d)] <= h_bin[i+d],
            y[(i, d)] >= h_bin[i] + h_bin[i+d] - 1
        ]
    
    # Compute overlap for each shift d <- compute overlap directly
    # N_d = (n/2) - sum y_{i,d}
    t = cp.Variable(integer=True)
    constraints += [t >= 0]
    
    # For all shifts calculate number of (1,0) pairs
    for d in d_range:
        if d != 0:
            support_i = [i for i in range(n) if 0 <= i + d < n]
            if support_i:
                # N_d = sum over i in valid range of h_i*(1-h_{i+d})
                # = sum of h_i - sum of h_i*h_{i+d}
                #   = n/2 - sum over correct indices of y_{i,d}
                overlap_expr = n//2 - cp.sum([y[(i,d)] for i in support_i])
                constraints += [overlap_expr <= t]
    
    # Objective
    objective = cp.Minimize(t)
    
    # Form and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with a continuous relaxion solver for integral solution
    prob.solve(solver=cp.CBC, verbose=False)
    
    # Get solution from boolean vector
    h_sol = h_bin.value.astype(float)
    return h_sol, n//2 - np.sum([v.value for v in y.values() if v.value is not None])

def smooth_max_overlaps(h, tau=0.1):
    """Compute smooth maximum overlap with log-sum-exp approximation."""
    n = len(h)
    # Collect all overlap counts
    overlaps = []
    shift_range = range(-n+1, n)
    for d in shift_range:
        if d != 0:
            # overlap = sum h_i (1 - h_{i+d})
            # This can be done using correlation
            # We are interested in: sum_i h_i * (1 - h_{i+d})
            # = sum_i h_i - sum_i h_i h_{i+d}
            h_shift = np.zeros_like(h)
            if d > 0:
                h_shift[d:] = h[:-d]
            elif d < 0:
                h_shift[:d] = h[-d:]
            else:
                h_shift = h
            overlap = np.sum(h) - np.sum(h * h_shift)
            overlaps.append(overlap)
    overlaps = np.array(overlaps)
    # log sum exp smoothing
    Z = np.sum(np.exp(overlaps/tau))
    return tau * np.log(Z)

def grad_smooth_max_overlaps(h, tau=0.1):
    """Compute the gradient of the log-sum-exp approximation of overlap."""
    n = len(h)
    d_range = range(-n+1, n)
    weights = np.zeros(len(d_range)-1)  # exclude d=0 from shift range
    shifts = []
    grads = []
    corr_idx = 0
    for i, d in enumerate(d_range):
        if d == 0: continue
        h_shifted = np.zeros_like(h)
        if d > 0:
            h_shifted[d:] = h[:-d]
        elif d < 0:
            h_shifted[:d] = h[-d:]
        else:
            h_shifted = h
        overlap = np.sum(h) - np.sum(h * h_shifted)
        shifts.append(overlap)
        weights[i - 1] = np.exp(overlap / tau)
        corr_idx += 1
    # Normalized weights
    W = weights / np.sum(weights)
    # Gradient = sum_w_i * (1 - h_{j+d} if d>0 else h_{j-d} if d<0)
    grad = np.zeros_like(h)
    idx = 0
    for d in d_range:
        if d == 0: continue
        h_shifted = np.zeros_like(h)
        if d > 0:
            h_shifted[d:] = h[:-d]
        elif d < 0:
            h_shifted[:d] = h[-d:]
        else:
            h_shifted = h
        g_i = 1 - h_shifted
        grad += W[idx] * g_i
        idx += 1
    return grad

def proj_simplex(h, target_sum=1.0):
    """Project a vector h onto the simplex with sum = target_sum."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-10:
        return h
    else:
        # Simple scaling approach
        h = (h / s) * target_sum
        return np.clip(h, 0, 1)

            best_h = h_bin.copy()

        has_milp_solved = True
    except:
        pass

    if not has_milp_solved:
        # start from initial_h_values
        h_current = np.clip(np.array(initial_h_values), 0, 1)
        # normalize the sum
        h_current = (h_current / np.sum(h_current)) * (n / 2)
        best_h = h_current.copy()
        best_c5 = evaluate_c5(best_h)

    # Phase 2 - Smooth Max Continuation Gradient Descent
    h0 = best_h.copy()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    # Running continuation with decreasing temperature
    tau = 0.5  # Initial smoothing parameter
    max_iters = 5000
    iter_count = 0

    while (time.time() - start_time) < budget_s * 0.95 and iter_count < max_iters:
        h_prev = h_current.copy()
        # one inner loop
        inner_grad_step = 100  # Number of gradient steps per continuation level
        for _ in range(inner_grad_step):
            if (time.time() - start_time) > budget_s * 0.95:
                break
            g = grad_smooth_max_overlaps(h_current, tau)
            # stepsizel
            lr = 0.01
            h_next = h_current - lr * g
            h_next = proj_simplex(h_next)
            h_current = h_next

        # Update if better
        c5_current = evaluate_c5(h_current)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()

        # decaying schedule
        tau = max(tau / 2, 1e-6)  # stop at fine scale
        if abs(tau) < 1e-6:
            break

        iter_count += 1

        # Random restart occasionally:
        if iter_count % 50 == 0 and (time.time() - start_time) < budget_s * 0.9:
            r_h = np.clip(best_h + np.random.normal(0, 0.02, n), 0, 1)
            # always re-normalize
            r_h = (r_h / np.sum(r_h)) * (n / 2)
            h_current = r_h

    # Final polishing with non-smooth optimization
    # Use non-smooth overlap function
    def smooth_obj(h):
        return smooth_max_overlaps(h, tau=1e-4)

    def obj_fn(h):
        return evaluate_c5(h)

    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) * dx - 1.0}

    bounds = [(0, 1) for _ in range(n)]

    try:
        res = minimize(obj_fn, best_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100, 'ftol': 1e-12})

            h_opt = proj_simplex(h_opt)
            c5_final = evaluate_c5(h_opt)
            if c5_final < best_c5:
                best_h = h_opt
                best_c5 = c5_final
    except:
        pass

    # Final validation and return
    return (best_h.tolist(), best_c5, n)
