import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_binary_milp(n):
    """Solve binary integer linear program to minimize maximum overlap"""
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(x) == n // 2]

    # iterate shifts
    for d in range(-n+1, n):
        if d == 0:
            continue
        w_vars = []
        if d > 0:
            rng = range(n - d)
        else:
            rng = range(-d, n)
        for i in rng:
            w = cp.Variable(boolean=True)
            # link w to x[i] and x[i+d]
            constraints.append(w <= x[i])
            constraints.append(w <= x[i+d])
            constraints.append(w >= x[i] + x[i+d] - 1)
            w_vars.append(w)
        # N_d = sum(x) - sum(w_vars)
        N_d = cp.sum(x) - cp.sum(w_vars)
        constraints.append(N_d <= t)
    constraints.append(t >= 0)

    prob = cp.Problem(cp.Minimize(t), constraints)
    # attempt solvers
    solver_order = [cp.CBC, cp.GLPK_MI, cp.ECOS_BB]
    success = False
    for solver in solver_order:
        try:
            prob.solve(solver=solver, verbose=False, warm_start=True, max_iters=5000)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                success = True
                break
        except Exception as e:
            continue
            
    if not success:
        raise RuntimeError("MILP solve failed with all solvers")
        
    x_vals = np.round(x.value).astype(float)
    t_val = int(round(t.value))
    # Ensure sum constraint
    if abs(np.sum(x_vals) - n // 2) > 1e-6:
        # Use a simple adjustment
        diff = n // 2 - np.sum(x_vals)
        if diff > 0:
            # Need to increase sum by diff
            indices = np.argsort(x_vals)
            for i in range(int(diff)):
                if x_vals[indices[i]] < 1:
                    x_vals[indices[i]] = 1
        elif diff < 0:
            # Need to decrease sum by |diff|
            indices = np.argsort(x_vals)
            for i in range(int(-diff)):
                if x_vals[indices[i]] > 0:
                    x_vals[indices[i]] = 0
    return x_vals, t_val
