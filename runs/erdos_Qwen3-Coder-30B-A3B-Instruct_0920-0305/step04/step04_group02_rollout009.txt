import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Phase 1: Solve MILP for exact binary solution
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Use a quick and rough way to find integer solution
    try:
        # If we assume we can construct an appropriate integer solution from known values
        # Minimal over the entire set using current h0 as a starting point
        # Then apply continuation approach to get better fractional solution
        
        # Our simplified approach is to return the given solution if time allows
        initial_c5 = evaluate_c5(h0)
        best_h = h0.copy()
        best_c5 = initial_c5
    except Exception as e:
        pass
    
    # Phase 2: Apply smooth max projected gradient descent
    def smooth_max_objective(h, tau=0.1):
        """Smooth approximation of max over shifts"""
        n = len(h)
        total = 0.0
        exps = []
        for d in range(-n+1, n):
            if d == 0: continue
            shift_sum = 0.0
            for i in range(n):
                if 0 <= i+d < n:
                    shift_sum += h[i] * h[i+d]
            cd = np.sum(h) - shift_sum
            exps.append(np.exp(cd / tau))
        return tau * np.log(np.sum(exps))
    
    def smooth_max_gradient(h, tau=0.1):
        """Gradient of smooth max objective"""
        n = len(h)
        grad = np.zeros(n)
        total_exp = 0.0
        exp_vals = []
        c_vals = []

        for d in range(-n+1, n):
            if d == 0: continue
            shift_sum = 0.0
            for i in range(n):
                if 0 <= i+d < n:
                    shift_sum += h[i] * h[i+d]
            cd = np.sum(h) - shift_sum
            exp_val = np.exp(cd / tau)
            exp_vals.append(exp_val)
            c_vals.append(cd)
            total_exp += exp_val
        
        # Compute gradient
        for i in range(n):
            grad_term = 0.0
            for j, (d, exp_val, cd) in enumerate(zip(range(-n+1, n), exp_vals, c_vals)):
                if d == 0: continue
                if 0 <= i+d < n:
                    term_h_i = h[i+d]
                    term_h_i_d = h[i]
                else:
                    term_h_i = 0.0
                    term_h_i_d = 0.0
                grad_term += (1.0 - term_h_i - term_h_i_d) * (exp_val / total_exp)
            grad[i] = grad_term
        
        return grad
    
    def project_onto_simplex(h):
        """Project h onto the constraint sum(h) * dx == 1"""
        h = np.clip(h, 0, 1)
        s = np.sum(h) * dx
        if abs(s - 1.0) > 1e-10:
            h = h * (1.0 / s)
        return h
    
    # Continue optimizing with gradient descent
    max_epochs = 500
    alpha_default = 0.01
    tau_init = 0.5
    tau_decay = 0.5
    min_tau = 1e-4
    n_restarts = 10
    restart_count = 0
    opt_h = best_h.copy()
    opt_c5 = best_c5
    
    while time.time() - start_time < 0.95 * budget_s and restart_count < n_restarts:
        h = opt_h.copy()
        tau = tau_init
        epoch = 0
        unchanged_count = 0
        last_c5 = float('inf')
        
        while tau > min_tau and epoch < max_epochs and time.time() - start_time < 0.95 * budget_s:
            grads = smooth_max_gradient(h, tau)
            h_new = h - alpha_default * grads
            h_new = project_onto_simplex(h_new)
            c5_new = evaluate_c5(h_new)
            
            if c5_new < opt_c5:
                opt_h = h_new.copy()
                opt_c5 = c5_new
                unchanged_count = 0
            else:
                unchanged_count += 1
            
            if unchanged_count > 5:
                # Reduce step size or temperature
                alpha_default /= 2
                unchanged_count = 0
            
            h = h_new
            tau *= tau_decay
            epoch += 1
            
        # Restart with random perturbation
        h_pert = opt_h + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = project_onto_simplex(h_pert)
        opt_h = h_pert
        restart_count += 1
    
    # Final polishing
    def exact_objective(h):
        return evaluate_c5(h)
    
    try:
        res = minimize(
            exact_objective, 
            opt_h, 
            method='SLSQP', 
            bounds=[(0, 1) for _ in range(n_points)], 
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0},
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            final_c5 = exact_objective(res.x)
            if final_c5 < opt_c5:
                opt_h = res.x
                opt_c5 = final_c5
    except Exception:
        pass
    
    return (opt_h.tolist(), opt_c5, n_points)
