import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible set: [0,1]^n with sum = target_sum"""
    y = np.clip(h, 0, 1)
    current_sum = np.sum(y)
    if abs(current_sum - target_sum) < 1e-9:
        return y
    
    # Use iterative approach to adjust sum
    for _ in range(100):
        diff = current_sum - target_sum
        if abs(diff) < 1e-6:
            break
            
        if diff > 0:  # sum too high
            # reduce values from those not at lower bound
            free = y > 0
            if not free.any():
                break
            # Reduce proportionally
            y[free] = y[free] * (target_sum / current_sum)
        else:  # sum too low
            # increase values from those not at upper bound
            free = y < 1
            if not free.any():
                break
            # Increase proportionally  
            y[free] = y[free] + (-diff) / np.sum(free)
            
        y = np.clip(y, 0, 1)
        current_sum = np.sum(y)
        
    return y
