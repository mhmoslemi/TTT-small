def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def decode_h(params, n):
    # Interpret params as sorted decreasing heights
    heights = np.sort(params)[::-1]
    heights = np.clip(heights, 0, 1)
    
    # Create piecewise linear symmetric shape
    x = np.linspace(0, 1, n // 2)
    heights_sym = np.concatenate([heights, heights[-2::-1]])
    
    # Ensure area constraint
    area = np.sum(heights_sym) * 2.0 / n
    scale = 1.0 / area
    h = np.clip(heights_sym * scale, 0, 1)
    
    return h

def objective_smooth(params, n):

    beta = 15.0
    exp_vals = np.exp(beta * corr * dx)
    smooth_max = np.log(np.sum(exp_vals)) / beta
    return -smooth_max  # We minimize negative to maximize
