import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient(h, d_star):
    """Compute the subgradient for a given shift d_star"""
    n = len(h)
    grad = np.zeros(n)
    if d_star > 0:
        # h_i - h_{i+d_star} for i = 0..n-d_star-1
        grad[:-d_star] = 1.0
        grad[d_star:] -= h[:-d_star]
    elif d_star < 0:
        # h_i - h_{i+d_star} for i = -d_star..n-1
        grad[-d_star:] = 1.0
        grad[:-d_star] -= h[-d_star:]
    else:
        # This should never happen
        grad = np.ones(n)
    return grad

def binary_optimal(n):
    """Solve the integer version of the problem via MILP"""
    # Variables
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    
    # Mass constraint (exactly half are 1s)
    mass_constraint = cp.sum(x) == n // 2
    
    #.symmetry breaking
    sym_break = x[0] == 1
    
    # Constraints for all shifts d ≠ 0
    constraints = [t >= 0]
    
    # Use the fact that the last shift is from index 0 to n-1
    # for d = 1 to n-1, and symmetric fashion
    for d in range(1, n):
        # Create w variables for product x[i] * x[i+d]
        w = cp.Variable(n - d, boolean=True)
        for i in range(n - d):
            # McCormick constraints for w[i] = x[i] * x[i+d]
            constraints.extend([
                w[i] <= x[i],
                w[i] <= x[i + d],
                w[i] >= x[i] + x[i + d] - 1
            ])
        # Overlap count for shift d
        N_d = cp.sum(x) - cp.sum(w)
        constraints.append(N_d <= t)
    # Symmetric d < 0 cases
    for d in range(-n+1, 0):
        # Create w variables for product x[i] * x[i+d]
        w = cp.Variable(n + d, boolean=True)
        for i in range(n + d):
            # McCormick constraints for w[i] = x[i] * x[i+d]
            constraints.extend([
                w[i] <= x[i],
                w[i] <= x[i - d],  # i-d is same index as i+d in the reversed case
                w[i] >= x[i] + x[i - d] - 1
            ])
        # Overlap count for shift d
        N_d = cp.sum(x) - cp.sum(w)
        constraints.append(N_d <= t)
        
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, [mass_constraint, sym_break] + constraints)

    try:
        prob.solve(solver=cp.CBC)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value.astype(float), t.value
    except:
        pass

    # fallback to GLPK
    try:
        prob.solve(solver=cp.GLPK_MI)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value.astype(float), t.value
    except:
        pass

    # fallback to ECOS_BB
    try:
        prob.solve(solver=cp.ECOS_BB)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value.astype(float), t.value
    except:

def refine_continuous(h0, max_time):
    """Refine the solution using projected gradient descent"""
    best_h = h0.copy()
    n = len(h0)
    dx = 2.0 / n

    # best_c5 = evaluate_c5(h0)
    best_c5 = evaluate_c5(best_h)
    
    start_time = time.time()
    iter_count = 0
    max_iter = 2000
    alpha_start = 0.5

    while (time.time() - start_time) < max_time and iter_count < max_iter:
        corr = correlate(best_h, 1 - best_h, mode="full")
        d_star = np.argmax(corr)
        d_vals = np.where(corr == np.max(corr))[0]
        # Adjust for the indexing shift in full mode
        d_star = d_vals[0] - (n - 1)

        g = subgradient(best_h, d_star)
        alpha = alpha_start
        step_size = alpha
        while True:
            h_new = best_h - step_size * g
            h_new = np.clip(h_new, 0, 1)
            # renormalise to satisfy sum constraint
            scale = (n / 2) / np.sum(h_new)
            h_new = h_new * scale
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new
                break
            else:
                step_size *= 0.5
                if step_size < 1e-10:
                    break
        iter_count += 1

        # Occasionally add noise to escape local minima
        if np.random.rand() < 0.2:
            noise = np.random.normal(0, 1e-3, n)
            h_new = best_h + noise
            h_new = np.clip(h_new, 0, 1)
            scale = (n / 2) / np.sum(h_new)
            h_new = h_new * scale
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # final polish with SLSQP
    try:
        from scipy.optimize import minimize
        res = minimize(
            evaluate_c5,
            best_h,
            method='SLSQP',
            bounds=[(0, 1) for _ in range(n)],
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0},
            options={'maxiter': 200}
        )
        if res.success:
            best_h = res.x
            best_c5 = evaluate_c5(best_h)
    except:
        pass

    return best_h, best_c5
