import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time
from math import floor

# Global variable for constraint
TARGET_SUM = None

def compute_corr_fft(h):
    """Compute full correlation using FFT, return raw correlation array"""
    n = len(h)
    # Reverse h to be compatible with correlation convention
    h_rev = h[::-1]
    # Create array for convolution (length 2n-1)
    L = 2*n - 1
    h_padded = np.zeros(L)
    h_padded[:n] = h_rev
    b_padded = np.zeros(L)
    b_padded[:n] = 1 - h
    # FFT-based convolution
    corr_fft = fft(h_padded) * fft(b_padded)
    corr = ifft(corr_fft).real
    # Return only the valid part of correlation (shifts from -n+1 to n-1)
    return corr

def c5_from_corr(corr, n):
    """Convert FFT-based correlation to C5 value"""
    dx = 2.0 / n
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT-based convolution"""
    n = len(h)
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n)

def project_to_sum_box(x, target_sum):
    """Project vector x onto sum = target_sum with box constraints [0,1]"""
    # Ensure bounded by [0,1]
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-12:
        return x.copy()
    # Binary search for lambda
    low, high = -1.0, 1.0
    tol = 1e-12
    for _ in range(100):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if sum_y < target_sum:
            low = mid
        else:
            high = mid
    # Use final value
    final_lambda = (low + high) / 2.0
    result = np.clip(x + final_lambda, 0, 1)
    return result

def random_pair_move(h):
    """Generate a valid pairwise mass transfer move (i, j, delta)"""
    n = len(h)
    i = np.random.randint(0, n)
    j = np.random.randint(0, n)
    if i == j:
        return None
    h_i, h_j = h[i], h[j]
    delta_min = max(-h_j, h_i - 1)
    delta_max = min(h_i, 1 - h_j)
    if delta_min >= delta_max:
        return None
    delta = np.random.uniform(delta_min, delta_max)
    return (i, j, delta)

def guided_move(h, corr):
    """Perform global improvement step based on current worst lag"""
    n = len(h)
    # Get the index at which the maximum overlap occurs
    max_idx = np.argmax(corr)
    # corr in [0, 2n-2] where 0 means -n+1, 2n-2 means +n-1
    worst_lag = max_idx - (n - 1)
    
    # Compute gradient of the objective (inner product)
    # g_i = sum over j of h_j * h_{j+l} where l is the lag
    # So for lag l, we find s_{i+l} = h_{i+l} 
    # But also we need to loop properly for out-of-bound conventions
    
    # Since corr = [0..n-1] pairs of centered correlation, we have:
    # corr[0] = h[0]*h[0]+...  corr[n] = h[0]*h[n]+...
    # -->
    # Lag -n+1 to n-1 corresponds to indices [0..2n-2], centered at index n-1
    
    # Compute gradient essentially as:
    # For all positions i, compute sum over j of h_{j+l} * h_j
    # This gradient points in direction of decreasing overlap, i.e., increasing inner product
    # Actually, the gradient of the term h_i(h_{i+k}) w.r.t. h_i is h_{i+k}
    # For the max term, it is the sum gradient of the overall max correlated with worst_lag
    # What we want: for each index, compute contribution from nearest neighbor shifts
    # Better approach: 
    # For lag \ell, we have \sum h_i h_{i+\ell}
    # Gradient w.r.t. h_i of this is h_{i+\ell} if \ell >= 0, or h_{i-\ell} if \ell < 0
    # But the max curl is over all lags! So we first determine the one that is max...

    # Our existing corr is already computed!
    # Axes mapping: index 0 for lag -n+1 up to index 2n-2 for lag n-1
    # Start from k = 0, which is lag -n+1+1 = -n+2 = 1-n
    # corr[n-1] is lag 0
    # corr[2n-2] is lag n-1
    # corr[0] is lag -n+1
    
    # We know that max is at index max_idx which is due to lag = max_idx - (n-1)
    # Let's compute the sub-gradient
    # $$ \frac{\partial}{\partial h_i} \sum_{i} h_i h_{i + \ell} = h_{i + \ell} + h_{i - \ell} $$
    # But constraint is to not move past the edge of the domain!
    # So we compute this vector of changes across i:
    
    # Get the effective contribution from the worst lag to which gradient relates:
    gradient = np.zeros(n)
    for i in range(n):
        # i + worst_lag, cycle if out of bounds (periodic)
        j1 = (i + worst_lag) % n
        # i - worst_lag is an out of bounds in negative direction, don't use that directly
        if worst_lag >= 0:
            # h_{i + l} is h[j1]
            gradient[i] = h[j1]
        else:
            # h_{i + l} is h[j1], with j1 possibly < 0 wrapping
            gradient[i] = h[j1]
            
    # Find min and max of gradients in the correct sense
    min_idx = np.argmin(gradient)
    max_idx_grad = np.argmax(gradient)
    # Try to move mass from min to max
    # this requires to control the sign of the move later
    # where move is increasing max_idx_grad by some amount, decreasing min_idx
    # but we must respect: 
    # 1. h_new[i] >= 0  and <= 1 
    # 2. sum is preserved
    
    # indirect form: let the flow from index min_idx to max_idx_grad with current h[min_idx] and h[max_idx_grad]
    h_min = h[min_idx]
    h_max = h[max_idx_grad]
    delta_max_transfer = min(h_min, 1 - h_max)
    if delta_max_transfer <= 0:
        return False, h
    
    delta = np.random.uniform(0, delta_max_transfer)
    
    # Modify h
    new_h = h.copy()
    new_h[min_idx] -= delta
    new_h[max_idx_grad] += delta
    new_h = project_to_sum_box(new_h, TARGET_SUM)
    return True, new_h

def anneal(h0, run_seconds, rng):
    """Run simulated annealing with guided steps"""
    h = h0.copy()
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Set up time management
    t0 = time.time()
    
    # Annealing parameters
    T0 = 5e-3  # initial temperature
    Tf = 1e-6  # final temperature
    n_steps = 0
    
    while time.time() - t0 < run_seconds:
        n_steps += 1
        if n_steps % 10 == 0:
            # Do guided move
            corr = compute_corr_fft(h)
            success, new_h = guided_move(h, corr)
            if success:
                new_c5 = evaluate_c5(new_h)
                if new_c5 < current_c5:
                    h = new_h
                    current_c5 = new_c5
                    if new_c5 < best_c5:
                        best_h = new_h.copy()
                        best_c5 = new_c5
                else:
                    # accept with probability
                    T = T0 * (1.0 - (time.time() - t0) / run_seconds) + Tf * ((time.time() - t0) / run_seconds)
                    if T > 0 and np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                        h = new_h
                        current_c5 = new_c5
        else:
            # Do random pair move
            move = random_pair_move(h)
            if move is not None:
                i, j, delta = move
                new_h = h.copy()
                new_h[i] -= delta
                new_h[j] += delta
                new_h = project_to_sum_box(new_h, TARGET_SUM)
                new_c5 = evaluate_c5(new_h)
                if new_c5 < current_c5:
                    h = new_h
                    current_c5 = new_c5
                    if new_c5 < best_c5:
                        best_h = new_h.copy()
                        best_c5 = new_c5
                else:
                    # accept with probability
                    T = T0 * (1.0 - (time.time() - t0) / run_seconds) + Tf * ((time.time() - t0) / run_seconds)
                    if T > 0 and np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                        h = new_h
                        current_c5 = new_c5
                        
    return best_h, best_c5

def local_refine(h):
    """Use SLSQP with bounds and equality to refine local"""
    bounds = [(0, 1) for _ in range(len(h))]
    
    def constraint_eq(h_vec):
        return np.sum(h_vec) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    def objective(h_vec):
        return evaluate_c5(h_vec)
    
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12})
        if res.success:
            return res.x
    except:
        pass
    return h
