import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_boxed_simplex(v, target_sum):
    """Project v onto the simplex with box constraints [0,1] and sum = target_sum"""
    # Clip to box constraints first
    v = np.clip(v, 0, 1)
    # Adjust to meet the sum constraint
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-10:
        return v
    diff = target_sum - current_sum
    # Distribute the difference among entries that are not at bounds
    free = (v > 0) & (v < 1)
    if np.sum(free) > 0:
        v[free] += diff / np.sum(free)
        # Clip again to ensure bounds are still respected
        v = np.clip(v, 0, 1)
    return v

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from the provided solution, but use a higher resolution for better accuracy
    n_points = 100  # Use more points for better resolution
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    # Phase 1: Binary Search using Simulated Annealing
    best_binary_h = None
    best_c5_binary = float('inf')
    
    def binary_correlation(binary_vec):
        # Compute the correlation
        corr = np.correlate(binary_vec, 1 - binary_vec, mode='full')
        return corr
    
    # Binary SA
    def binary_sa():
        current_binary = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
        # Ensure sum equals target_sum
        ones_count = np.sum(current_binary)
        if ones_count > target_sum:
            # Change some 1s to 0s
            zero_indices = np.where(current_binary == 1)[0]
            to_change = int(ones_count - target_sum)
            change_indices = np.random.choice(zero_indices, size=to_change, replace=False)
            current_binary[change_indices] = 0
        elif ones_count < target_sum:
            # Change some 0s to 1s
            one_indices = np.where(current_binary == 0)[0]
            to_change = int(target_sum - ones_count)
            change_indices = np.random.choice(one_indices, size=to_change, replace=False)
            current_binary[change_indices] = 1
        
        current_c5 = evaluate_c5(current_binary.astype(float))
        best_binary = current_binary.copy()
        best_c5 = current_c5
        temp = 2.0
        cooling_rate = 0.9995
        max_iter = 50000
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
            
            # Propose new solution by swapping two elements
            idx1 = np.random.randint(0, n_points)
            idx2 = np.random.randint(0, n_points)
            while current_binary[idx1] == current_binary[idx2]:
                idx2 = np.random.randint(0, n_points)
                
            new_binary = current_binary.copy()
            new_binary[idx1] = new_binary[idx2]
            new_binary[idx2] = current_binary[idx1]
            
            # Ensure sum is maintained
            ones_count = np.sum(new_binary)
            if ones_count > target_sum:
                # Change some 1s to 0s to preserve sum
                zero_indices = np.where(new_binary == 1)[0]
                to_change = int(ones_count - target_sum)
                change_indices = np.random.choice(zero_indices, size=to_change, replace=False)
                new_binary[change_indices] = 0
            elif ones_count < target_sum:
                # Change some 0s to 1s to preserve sum
                one_indices = np.where(new_binary == 0)[0]
                to_change = int(target_sum - ones_count)
                change_indices = np.random.choice(one_indices, size=to_change, replace=False)
                new_binary[change_indices] = 1
            
            new_c5 = evaluate_c5(new_binary.astype(float))
            delta = new_c5 - current_c5
            
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                current_binary = new_binary
                current_c5 = new_c5
                
                if current_c5 < best_c5:
                    best_binary = current_binary.copy()
                    best_c5 = current_c5
            
            temp *= cooling_rate
        
        return best_binary, best_c5
    
    # Run multiple SA trials
    num_sa_trials = 10
    for trial in range(num_sa_trials):
        if time.time() - start_time > budget_s * 0.95:
            break
        try:
            binary_h, c5_val = binary_sa()
            if c5_val < best_c5_binary:
                best_c5_binary = c5_val
                best_binary_h = binary_h
        except Exception:
            continue
    
    # Phase 2: Continuous Optimization from best binary
    if best_binary_h is not None:
        # Convert to float and refine with gradient descent
        h_start = best_binary_h.astype(float)
        
        # Smooth approximate for gradient descent
        def smooth_max_objective(h, alpha=50.0):
            # Compute t = max_k ∫ h(x) (1 - h(x+k)) dx
            corr = np.correlate(h, 1 - h, mode='full')
            # Apply smooth maximum
            f = np.exp(alpha * (corr - np.max(corr)))  # Softmin variant
            # Compute softmin of f
            softmin_val = np.log(np.sum(f)) / alpha + np.max(corr)
            return softmin_val
        
        def smooth_max_gradient(h, alpha=50.0):
            n = len(h)
            corr = np.correlate(h, 1 - h, mode='full')
            f = np.exp(alpha * (corr - np.max(corr)))
            w = f / np.sum(f)  # Softmax weights
            
            # Gradient computation
            grad = np.zeros(n)
            # Use numpy convolution for efficient computation
            # Consider that gradient of A * h where A depends on h
            # For each i, sum over shifts k such that both indices valid
            for i in range(n):
                grad[i] = 1.0
                for k in range(-n+1, n):
                    if 0 <= i+k < n:
                        grad[i] -= w[k+n-1] * (h[i + k] + h[i - k] if 0 <= i - k < n else 0)
            return grad
            
        # Refine using L-BFGS-B with bounds enforced via projection
        def refine_h():
            h_current = h_start.copy()
            best_h = h_current.copy()
            best_c5 = evaluate_c5(h_current)
            
            # Use scipy minimize with constraints and bounds
            def obj(h):
                return smooth_max_objective(h)
            
            def grad(h):
                return smooth_max_gradient(h)
            
            for _ in range(2000):
                if time.time() - start_time > budget_s * 0.95:
                    break
                    
                # Gradient descent step
                h_next = h_current - 0.01 * grad(h_current)
                
                # Project onto feasible set
                h_next = project_to_boxed_simplex(h_next, target_sum)
                
                # Evaluate
                c5_val = evaluate_c5(h_next)
                
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_next.copy()
                
                h_current = h_next
            
            return best_h, best_c5
        
        try:
            refined_h, refined_c5 = refine_h()
            if refined_c5 < best_c5_binary:
                best_c5_binary = refined_c5
                best_binary_h = refined_h
        except:
            pass
    
    return (best_binary_h.tolist(), best_c5_binary, n_points)
