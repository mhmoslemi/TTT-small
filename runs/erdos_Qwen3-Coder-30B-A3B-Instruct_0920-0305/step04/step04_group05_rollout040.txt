import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible set with sum=target and bounds [0,1]"""
    # Clip to [0,1]
    y = np.clip(h, 0, 1)
    # Adjust sum to target
    current_sum = np.sum(y)
    if current_sum == 0:
        # Avoid division by zero
        y = np.ones_like(y) * (target_sum / len(y))
    else:
        y = y * (target_sum / current_sum)
        # Clip again to ensure bounds
        y = np.clip(y, 0, 1)
    return y
