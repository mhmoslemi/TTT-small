import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Discretization
    n_points = 200
    dx = 2.0 / n_points
    
    # Parametric representation: breakpoints for symmetric decreasing shape
    # Use 5 breakpoints for the left half of [0,1], so 10 parameters total (5 heights + 5 locations)
    # For simplicity, fix the locations (breakpoints) and only optimize the heights
    breakpoints = np.linspace(0, 1, 6)[1:-1]  # Intermediate points 0.2, 0.4, 0.6, 0.8
    
    # Number of internal heights to optimize (this optimizes heights at breakpoints)
    K = 4  # Number of heights (excluding h[0]=1 and h[-1]=0, which are fixed)
    
    def decode_h(params):
        # Take the first K heights, sort descending to enforce monotonicity
        heights = np.array(params[:K])
        heights = np.sort(heights)[::-1]
        heights = np.clip(heights, 0, 1)
        # Prepend 1 and append 0
        h_full = np.concatenate(([1.0], heights, [0.0]))
        # Interpolate to full resolution on [0,1], symmetrically mirror to [1,2]
        x_full = np.linspace(0, 1, n_points//2 + 1)
        h_left = np.interp(x_full, breakpoints[:len(h_full)-1], h_full)
        h_right = h_left[::-1]
        h = np.concatenate((h_left[:-1], h_right))
        # Adjust for area constraint
        area = np.sum(h) * dx
        h = h / area * 1.0  # rescale so sum(h) * dx = 1
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        return h
    
    def evaluate_c5(h):
        corr = correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(params):
        h = decode_h(params)
        c5 = evaluate_c5(h)
        return c5

    # DE settings
    bounds = [(0, 1) for _ in range(K)]
    
    # Run DE with limited iterations to save time
    time_de = 0.70 * budget_s
    de_start = time.time()
    res_de = differential_evolution(
        objective,
        bounds,
        seed=seed,
        maxiter=int(200 * (time_de / 600.0)),
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.7,
        strategy='best1bin',
        disp=False
    )
    
    # Get best from DE
    best_params = res_de.x.copy()
    h_opt = decode_h(best_params)
    c5_opt = evaluate_c5(h_opt)
    
    refine_start = time.time()
    
    # Smooth surrogate optimization
    beta = 20.0  # smooth approximation parameter
    
    def smooth_max(h, beta):
        corr_vals = correlate(h, 1 - h, mode="full") * dx
        return np.log(np.sum(np.exp(beta * corr_vals))) / beta
    
    def smooth_max_grad(h, beta):
        corr_vals = correlate(h, 1 - h, mode="full") * dx
        w = np.exp(beta * corr_vals)
        w = w / np.sum(w)
        # Compute partial derivatives of each corr
        grad = np.zeros_like(h)
        n = len(h)
        for k in range(-n+1, n):
            # Derivative of f_k = sum_i h_i * (1 - h_{i+k})
            # df_k/dh_i = 1 - h_{i+k} - h_{i-k} (with boundary handling)
            if k == 0:
                # Just 1 for each h_i (no h_{i+k} since same position)
                grad += w[k + n - 1] * np.ones_like(h)
            else:
                # Derivative term 1 and 2 terms involving shifted values
                F = np.zeros_like(h)
                if k > 0:  # h_{i+k} sources
                    F[k:] = h[:-k]
                else:  # h_{i-k} sources
                    F[:k] = h[-k:]
                # Add -h_{i-k} terms
                F = 1.0 - F
                if k < 0:
                    F = F[:k]
                else:
                    F = F[k:]
                # Actual shifted values for full length
                f_grad_k = np.zeros_like(h)
                if k > 0:
                    f_grad_k[:-k] = F
                else:
                    f_grad_k[-k:] = F
                grad += w[k + n - 1] * f_grad_k
        return grad
    
    # Gradient descent refinement
    h_refine = h_opt.copy()
    rate = 0.01
    decay_factor = 1000.0
    
    while (time.time() - refine_start) < 0.20 * budget_s:
        # NOTE: Instead of actual analytical derivative, let's do a small gradient step manually
        # Approximate gradient based on finite differences
        eps = 1e-4
        grad_approx = np.zeros_like(h_refine)
        for i in range(len(h_refine)):
            h_plus = h_refine.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            area = np.sum(h_plus) * dx
            h_plus = h_plus / area * 1.0
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = h_refine.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            area = np.sum(h_minus) * dx
            h_minus = h_minus / area * 1.0
            c5_minus = evaluate_c5(h_minus)
            
            grad_approx[i] = (c5_plus - c5_minus) / (2.0 * eps)
        
        # Update step
        h_new = h_refine - rate * grad_approx
        # Projection: renormalize to meet sum constraint and stay in [0,1]
        h_new = np.clip(h_new, 0, 1)
        area = np.sum(h_new) * dx
        h_new = h_new / area * 1.0
        h_refine = h_new
        
        # Check improvement
        c5_new = evaluate_c5(h_refine)
        if c5_new < c5_opt:
            c5_opt = c5_new
            h_opt = h_refine.copy()
        
        # Decrease learning rate
        rate = rate * (1.0 - 1.0 / decay_factor)
        if rate < 1e-6:
            break
    
    # Final polishing with SLSQP if time allows
    if (time.time() - start_time) < 0.95 * budget_s:
        # Use full original vector for polish, avoiding resolution loss
        h0 = h_opt.copy()
        def obj_slqp(h):
            return evaluate_c5(h)
        def con_eq(h):
            return np.sum(h) * dx - 1.0
        
        cons = {'type': 'eq', 'fun': con_eq}
        bounds_slqp = [(0, 1) for _ in range(n_points)]
        
        try:
            res_slsqp = minimize(
                obj_slqp,
                h0,
                method='SLSQP',
                bounds=bounds_slqp,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res_slsqp.success:
                c5_slsqp = obj_slqp(res_slsqp.x)
                if c5_slsqp < c5_opt:
                    h_opt = res_slsqp.x
                    c5_opt = c5_slsqp
        except:
            pass

    return (h_opt.tolist(), c5_opt, n_points)
