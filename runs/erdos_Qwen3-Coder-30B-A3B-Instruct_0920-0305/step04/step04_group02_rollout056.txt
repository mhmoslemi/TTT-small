def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_binary_milp(h0, n):
    """Solve the binary version of the problem using MILP"""
    # Create matrices
    x = Variable(n, boolean=True)
    t = Variable(integer=True)
    
    # ot = np.zeros((n, n))
    ones = np.ones(n)
    
    # Objective
    objective = Minimize(t)
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp_sum(x) == n // 2)
    
    # Add additional constraints for y_{i,d}: y_{i,d} = x_i * x_{i+d}
    # For shift d from -n+1 to n-1, non-zero shifts in correlation
    for d in range(1, n):  # Only compute positive shifts
        # Calculate overlap for this shift d
        y = Variable(n, boolean=True)
        
        # Linearize y_i = x_i * x_{i+d}
        for i in range(n):
            if i+d < n:
                constraints.append(y[i] <= x[i])
                constraints.append(y[i] <= x[i+d])
                constraints.append(y[i] >= x[i] + x[i+d] - 1)
        
        # Compute N_d(x) = sum x_i - sum y_{i,d}
        # And constrain sum to be <= t
        N_d = cp_sum(x) - cp_sum(y)
        constraints.append(N_d <= t)
    
    # Symmetry breaking
    constraints.append(x[0] == 1)
    
    # Solve
    try:
        prob = Problem(objective, constraints)
        prob.solve(solver=cp.CBC, verbose=False)
        
        # Extract solution
        if prob.status in ["optimal", "optimal_inaccurate"]:
            sol_x = x.value.astype(int)
        else:
            sol_x = h0.astype(int)
            # try to make x sum to n//2 by rounding
            diff = int(n//2 - np.sum(sol_x))
            if diff != 0:
                # flip to balance
                indices = np.argsort(sol_x)
                if diff > 0:
                    sol_x[indices[:diff]] = 1
                else:
                    sol_x[indices[diff:]] = 0
    except Exception as e:
        # fallback
        sol_x = h0.astype(int)
        
    return sol_x

def smooth_max(x, shift_func, tau=0.02):
    """Compute smooth max of overlaps with temperature tau"""
    shift_vals = [shift_func(x, d) for d in range(1, len(x)) if d + len(x) <= len(x)*2]
    if len(shift_vals) == 0:
        return np.inf
    exp_vals = np.exp(np.array(shift_vals) / tau)
    return tau * np.log(np.sum(exp_vals))

def gradient_shift_overlap(x, d):
    """Compute overlap for shift d"""
    # Contribute to the shift by looking at x_i and 1 - x_{i+d}
    n = len(x)
    shift_vals = []
    for i in range(n):
        if i + d < n:
            shift_vals.append(x[i] * (1 - x[i + d]))
    return sum(shift_vals)

def project_simplex(x, target_sum):
    """Project onto simplex of sum=target_sum"""
    # Clip to [0, 1]
    x = np.clip(x, 0, 1)
    # Rescale to sum=target_sum
    current_sum = np.sum(x)
    if current_sum > 0:
        x = x * target_sum / current_sum
    return x

def smooth_descent_step(x, dt, tau=0.01):
    """One step of smooth gradient descent (without projection)"""
    g = np.zeros_like(x)
    n = len(x)
    
    # Compute gradients for each shift
    weights = {}
    total = 0
    for d in range(1, n):
        if d + len(x) > len(x)*2:  # equivalent to i + d < n
            continue
        e = 0
        for i in range(n):
            if i + d < n:
                e += x[i] * (1 - x[i+d])
        weights[d] = np.exp(e / tau)
        total += weights[d]
        
    # Normalize weights
    for d in range(1, n):
        if d + len(x) > len(x)*2:
            continue
        weights[d] /= total
    
    # Compute gradient wrt each variable
    for d in range(1, n):
        if d + len(x) <= len(x)*2:
            for i in range(n):
                if i + d < n:
                    dx1 = (1 - x[i+d]) * weights[d]
                    dx2 = (1 - x[i]) * weights[d]
                    # g[i] += dx1
                    # g[i+d] += dx2
                    g[i] -= dx1
                    g[i+d] -= dx2
    
    # Update step
    x_new = x - dt * g
    return x_new
