from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def decode_h_from_params(params, n_points=200):
    """
    Decodes 4 parameters into a full h vector of length n_points
    params = [h0, h1, x1, x2]
    For symmetric function:
    - x1: left edge of plateau
    - x2: right edge of plateau
    - h0: height at x = 0
    - h1: height at x = 1 (center)

    h0, h1, x1, x2 = params
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    x_vals = np.linspace(0, 2, n_points)
    
    for i in range(n_points):
        x = x_vals[i]
        ref_x = 1 - (1 - x) if x > 1 else x  # symmetry around x=1
        
        if ref_x <= x1:
            # Region before plateau
            if x1 != 0:
                h[i] = h0 + (h1 - h0) * (ref_x / x1)
            else:
                # Avoid division by zero, assume flat before plateau
                h[i] = h0
        elif ref_x <= x2:
            # Plateau region
            h[i] = h1
        else:
            # Region after plateau
            scaled_x = (ref_x - x2) / (1 - x2)
            if scaled_x >= 0:
                h[i] = h1 + (0.0 - h1) * scaled_x
            else:
                h[i] = 0.0
    
    # Reflect around center (symmetry)
    for i in range(n_points):
        j = n_points - 1 - i
        h[i] = max(0, min(1, h[i]))
    
    return h

def evaluate_c5_full(h):
    """Computes C5 = max_k ∫ h(x)(1 - h(x+k)) dx using full overlap computation"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlation values for shifts from -n to n
    corr = correlate(h, 1 - h, mode="full")
    # Maximum overlap integrated over full space
    max_corr = np.max(corr)
    return max_corr * dx

def optimized_objective(params):
    h = decode_h_from_params(params)
    # Normalize h to ensure ∫ h dx = 1
    integral = np.sum(h) * (2.0 / len(h))
    if integral > 0:
        h = h / integral * 1.0
    h = np.clip(h, 0, 1)
    return evaluate_c5_full(h)
