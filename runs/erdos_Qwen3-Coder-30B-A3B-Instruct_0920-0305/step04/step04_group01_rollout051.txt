import numpy as np
import time
from scipy.optimize import minimize

# Global variable for constraint sum
TARGET_SUM = None

def project_to_sum_box(x, target_sum):
    """Project vector x to the box [0,1]^n and sum constraint sum(x) = target_sum"""
    # Clip to [0,1] first
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x
    
    # Use simple iterative adjustment
    # Find indices that can be adjusted
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return x
        
    # Adjust proportionally to get closer to target
    if diff > 0:
        # Need to increase. Find indices < 1 that can be increased
        indices = np.where(x < 1)[0]
        if len(indices) == 0:
            return x
        # Distribute diff proportionally among available indices
        max_incr = np.min([1.0 - x[indices], diff / len(indices) * np.ones_like(indices)], axis=0)
        x[indices] += max_incr
    else:
        # Need to decrease. Find indices > 0 that can be decreased
        indices = np.where(x > 0)[0]
        if len(indices) == 0:
            return x
        # Distribute diff proportionally 
        max_decr = np.min([x[indices], -diff / len(indices) * np.ones_like(indices)], axis=0)
        x[indices] -= max_decr
        
    return x

def evaluate_c5_fft(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    if n == 0:
        return 0.0
    # Reverse h
    h_rev = h[::-1]
    b = 1.0 - h
    L = 2 * n - 1
    # Compute FFTs
    hf = np.fft.rfft(h_rev, n=L)
    bf = np.fft.rfft(b, n=L)
    # Compute convolution
    conv = np.fft.irfft(hf * bf, n=L)
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)

        return np.array([])

    h_rev = h[::-1]
    b = 1.0 - h
    L = 2 * n - 1
    # Compute FFTs
    hf = np.fft.rfft(h_rev, n=L)
    bf = np.fft.rfft(b, n=L)
    # Compute convolution
    conv = np.fft.irfft(hf * bf, n=L)
    return conv

def constraint_eq(h):
    """Equality constraint for sum(h) = TARGET_SUM"""
    return np.sum(h) - TARGET_SUM

def simulated_annealing(h_initial, target_sum, time_limit, seed):
    """Run simulated annealing with time limit"""
    np.random.seed(seed)
    n = len(h_initial)
    h = np.copy(h_initial)
    h = project_to_sum_box(h, target_sum)
    
    cur_c = evaluate_c5_fft(h)
    best_h = h.copy()
    best_c = cur_c

    T0 = 0.005
    Tmin = 1e-6
    start = time.time()
    
    iter_count = 0
    
    while time.time() - start < time_limit:
        # Random move
        i, j = np.random.choice(n, size=2, replace=False)
        
        # Compute bounds for delta
        delta_max = min(h[i], 1 - h[j])
        delta_min = -min(1 - h[i], h[j])
        
        if delta_max <= delta_min:
            continue
            
        # Choose delta uniformly
        delta = np.random.uniform(delta_min, delta_max)
        
        # Update h
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # evaluate
        c_new = evaluate_c5_fft(h_new)
        
        # compute temperature based on proportion of elapsed time
        elapsed = time.time() - start
        progress = elapsed / time_limit
        T = T0 * (1 - progress) + Tmin * progress
        
        # Acceptance probability
        if c_new < cur_c or np.random.rand() < np.exp(-(c_new - cur_c) / (T + 1e-12)):
            h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        
        iter_count += 1
    
    return best_h, best_c
