        """Decode the breakpoint heights into a full step function h."""
        # u are the raw heights, sorted descending to maintain symmetry and monotonicity.
        h_raw = np.zeros(n_points)
        x = np.linspace(0, 1, n_points//2 + 1)
        x = x[:-1]  # remove last element to prevent duplication at x=1
        # define points
        x_points = np.linspace(0, 1, k_breakpoints+1)
        y_vals = np.sort(u)[::-1]
        # linearly interpolate to h_raw (i=0 to n_points/2-1)
        y = np.interp(x, x_points, y_vals)
        h_raw[:len(x)] = y
        # mirror to the right half
        h_raw[len(x):] = y[::-1]
        return h_raw
    
    def project_h(h):
        """Ensure h is within [0,1] and integral equals 1."""
        # clip to [0,1]
        h = np.clip(h, 0, 1)
        # normalize
        integral = np.sum(h) * dx
        if integral > 0:
            h = h / integral * 1.0
        return h

        """Evaluate the C5 constant for a given step function h."""
        corr = correlate(h, 1 - h, mode="full")

        return max_corr * dx

        result = 0.0
        expsum = 0.0
        gradients = np.zeros_like(h)
        for k in shifts:

            grad_k = np.zeros_like(h)
