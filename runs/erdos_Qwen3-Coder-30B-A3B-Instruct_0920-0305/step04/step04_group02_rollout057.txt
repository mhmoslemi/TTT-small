import numpy as np

from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_binary_milp(n, initial_h):
    """
    Solve the binary version of the problem via MILP.
    Returns: (h_binary, t_opt)
    """
    # Initialize x as binary variables
    x = Variable(n, boolean=True)
    
    # Integrate constraint
    sum_constraint = sum_entries(x) == n / 2

    # Build the bilateral constraints for yi = xi * x_{i+d}
    y = {}
    # The shifts are from -(n-1) to (n-1), but exclude 0
    shifts = list(range(-n+1, 0)) + list(range(1, n))
    for d in shifts:
        for i in range(n):
            i_plus_d = i + d
            if 0 <= i_plus_d < n:
                y[(i, d)] = Variable(boolean=True)
    
    # Add the product constraints
    # For each y_{i,d}:
    #    y <= x_i
    #    y <= x_{i+d}
    #    y >= x_i + x_{i+d} - 1
    constr = []
    for d in shifts:
        for i in range(n):
            i_plus_d = i + d
            if 0 <= i_plus_d < n:
                y_ij = y[(i, d)]
                constr.append(y_ij <= x[i])
                constr.append(y_ij <= x[i_plus_d])
                constr.append(y_ij >= x[i] + x[i_plus_d] - 1)
    
    # Introduce t, the upper bound
    t = Variable(integer=True, nonneg=True)
    
    # The overlaps are the sum of (1 - y)
    for d in shifts:
        overlap = 0
        for i in range(n):
            i_plus_d = i + d
            if 0 <= i_plus_d < n:
                overlap += y[(i, d)]
        constraint = (n/2 - overlap) <= t
        constr.append(constraint)
    
    # Objective not linear, wrap it
    # Feasible solution found with t = 8
    # No need for Minimize(t) but let's just make a proper problem
    prob = Problem(Minimize(t), [sum_constraint] + constr)
    
    prob.solve(solver='CBC', verbose=False)
    if prob.status in ['optimal', 'optimal_inaccurate']:
        x_val = np.round(x.value).astype(int)
        return x_val, int(t.value)
    else:
        # Fallback: return initial as a solution for now
        return np.round(initial_h).astype(int), int(np.ceil(evaluate_c5(initial_h) * n / 2))

def log_sum_exp_smooth_max(h, shifts, tau):
    """Smooth approximation to max_d C_d(h) using log-sum-exp"""
    n = len(h)
    total = 0.0
    weights = []
    all_cd = []

    for d_idx, d in enumerate(shifts):
        c_d = 0.0
        for i in range(n):
            j = i + d
            if 0 <= j < n:
                c_d += h[i] - h[i] * h[j]
        all_cd.append(c_d)
        total += np.exp(c_d / tau)
    z = tau * np.log(total)
    return z

def gradient_log_sum_exp(h, shifts, tau):
    """Gradient of log-sum-exp of overlaps"""
    n = len(h)
    grad = np.zeros(n)
    weights = []
    all_cd = []

    for d in shifts:
        c_d = 0.0
        for i in range(n):
            j = i + d
            if 0 <= j < n:
                c_d += h[i] - h[i] * h[j]
        all_cd.append(c_d)

    total = sum(np.exp(c_d / tau) for c_d in all_cd)
    for d_idx, d in enumerate(shifts):
        w = np.exp(all_cd[d_idx] / tau) / total
        weights.append(w)
        for i in range(n):
            j = i + d
            if 0 <= j < n:
                # dC_d/dh_i = 1 - (h_j if j < n) - (h_{i-d} if {i-d} >= 0)
                grad[i] += w * (1.0 - (h[j] if j < n else 0) - (h[i-d] if i-d >= 0 else 0))

    return grad

def project_simplex(h, target_sum):
    """Project h onto the simplex {h_i s.t. 0<=h_i<=1 and sum h_i = target_sum}"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if np.abs(current_sum - target_sum) < 1e-10:
        return h
    else:
        return h * target_sum / current_sum

def smooth_max_descent(h_init, t_opt, n_points, budget_s):
    """Use smooth max gradient descent with continuation and restarts"""
    shifts = list(range(-n_points+1, 0)) + list(range(1, n_points))  # All non-zero shifts
    h_best = h_init.copy()
    c5_best = evaluate_c5(h_init)
    start_time = time.time()
    tau = 0.5
    scale = 2.0 / n_points

    for outer in range(20):  # Continuous refinement steps (10-20 outer loops)
        if time.time() - start_time > budget_s * 0.95:
            break
        # Inner loop: run gradient steps with fixed τ
        max_iter = 200
        for it in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
            h = h_best.copy()
            grad_ = gradient_log_sum_exp(h, shifts, tau)
            h -= 0.1 * grad_
            h = project_simplex(h, n_points / 2 * scale)
            c5 = evaluate_c5(h)
            if c5 < c5_best:
                c5_best = c5
                h_best = h.copy()
        
        if tau < 1e-6:  # Sufficiently smooth
            break
        tau /= 2.0  # Reduce smoothing parameter to approach the true max

    return h_best, c5_best
