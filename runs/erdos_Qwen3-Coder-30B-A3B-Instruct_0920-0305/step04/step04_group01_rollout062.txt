import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time
import random

# Global settings
TARGET_SUM = 60.0  # corresponds to n=120, sum=1
N_POINTS = 120
DX = 2.0 / N_POINTS

def compute_corr_fft(h):
    """Compute full correlation using FFT with no scaling."""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute 1-h
    b = 1.0 - h
    # Length of full correlation: 2n-1
    L = 2 * n - 1
    # Compute using FFT
    H_rev = fft(h_rev, L)
    B = fft(b, L)
    corr = ifft(H_rev * B).real
    return corr

def c5_from_corr(corr, n):
    """Return C5 from correlation."""
    max_corr = np.max(corr)
    return DX * max_corr

def evaluate_c5(h):
    """Evaluate C5 using fast FFT-based correlation."""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, N_POINTS)

def project_to_sum_box(x, target_sum):
    """Project x onto box constraints [0,1] and sum=target_sum using shift-and-clip."""
    # Binary search for lambda to shift x such that sum is target_sum
    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        s = np.sum(y)
        if s < target_sum:
            high = mid
        else:
            low = mid
    lambda_opt = (low + high) / 2.0
    y = np.clip(x + lambda_opt, 0, 1)
    return y

def random_pair_move(h):
    """Sample a valid pair move (i, j, delta) uniformly."""
    i = random.randint(0, N_POINTS - 1)
    j = random.randint(0, N_POINTS - 1)
    if i == j:
        return None
    h_i, h_j = h[i], h[j]
    # Compute feasible delta range
    delta_min = max(-h_j, h_i - 1)
    delta_max = min(h_i, 1 - h_j)
    if delta_min >= delta_max:
        return None
    delta = random.uniform(delta_min, delta_max)
    return (i, j, delta)

def guided_move(h):
    """Try to improve the current configuration by making a small gradient-like move."""
    corr = compute_corr_fft(h)
    # Find index of max correlation value
    max_lag_idx = np.argmax(corr)
    # Since corr corresponds to convolution with 1-h in reversed order
    # The actual lag is: it's centered at 0, so lag = max_lag_idx - (n-1) = max_lag_idx - N+1
    lag = max_lag_idx - (N_POINTS - 1)
    # Compute gradient w.r.t. h_i for the worst shift
    # dy/dh_i = h_{i+l} + h_{i-l} where out-of-bound values are zero
    gradient = np.zeros(N_POINTS)
    for i in range(N_POINTS):
        # h_{i-lag} term (forward in the relation of indices)  
        prev_idx = i - lag
        if 0 <= prev_idx < N_POINTS:
            gradient[i] += h[prev_idx]
        # h_{i+lag} term
        next_idx = i + lag
        if 0 <= next_idx < N_POINTS:
            gradient[i] += h[next_idx]
    
    # Find min and max gradient indices
    idx_min_grad, idx_max_grad = np.argmin(gradient), np.argmax(gradient)
    if gradient[idx_min_grad] == gradient[idx_max_grad]:
        return h.copy()  # No improvement possible
    # Transfer a bit from min to max, maintaining bounds
    h_new = h.copy()
    delta = min(h[idx_min_grad], 1 - h[idx_max_grad])
    if delta <= 0:
        return h  # No space for transfer
    h_new[idx_min_grad] -= delta
    h_new[idx_max_grad] += delta

    # Project to box and sum constraints
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    return h_new

def anneal(h0, run_seconds, rng_seed):
    """Run simulated annealing for a given time."""
    # Initialize algorithmic parameters
    np.random.seed(rng_seed)
    random.seed(rng_seed)

    T0 = 5e-3
    Tf = 1e-6
    t0 = time.time()

    h = h0.copy()
    c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5
    time_limit = t0 + run_seconds

    while time.time() < time_limit:
        # Cool temperature
        t_elapsed = time.time() - t0
        T = T0 * (1 - t_elapsed / run_seconds) + Tf * (t_elapsed / run_seconds)
        # Acceptance probability
        if T <= 0:
            T = 1e-6  # Ensure positive temperature

        # Switch between random and guided
        random_choice = random.random()
        if random_choice < 0.9 or True:
            move = random_pair_move(h)
            if move is None:
                continue
            i, j, delta = move
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            h_new = project_to_sum_box(h_new, TARGET_SUM)
        else:
            # Detetrministic guided move (always accept this step if it helps)
            h_new = guided_move(h)
            # Always accept if it reduces C5
            if evaluate_c5(h_new) < c5:
                h = h_new
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                continue
            else:
                # Accept with Metropolis probability
                if random.random() < np.exp(-(evaluate_c5(h_new) - c5) / T):
                    h = h_new
                    c5 = evaluate_c5(h_new)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                continue

        # Accept or reject
        new_c5 = evaluate_c5(h_new)
        if new_c5 < c5:
            h = h_new
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # Accept with Metropolis probability
            if random.random() < np.exp(-(new_c5 - c5) / T):
                h = h_new
                c5 = new_c5
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()

    return best_h, best_c5

def local_refine(h):
    """Refine h using SLSQP."""
    bounds = [(0, 1) for _ in range(N_POINTS)]
    
    def objective(x):
        return evaluate_c5(x)

    def constraint_eq(x):
        return np.sum(x) - TARGET_SUM

    cons = {'type': 'eq', 'fun': constraint_eq}

    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12})
    
    if result.success:
        return project_to_sum_box(result.x, TARGET_SUM)
    else:
        # Fallback: return original
        return h
