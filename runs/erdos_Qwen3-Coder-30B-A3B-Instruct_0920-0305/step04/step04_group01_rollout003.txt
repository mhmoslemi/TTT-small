import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = 60.0  # n/2 for n=120

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    h_rev = h[::-1]
    b = 1 - h
    corr = fftconvolve(h_rev, b, mode='full')
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation array"""
    dx = 2.0 / n
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_c5(h):
    """Fast evaluation of C5"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Project x to satisfy sum = target_sum with box constraints [0,1]"""
    # Binary search for the shift scalar lambda
    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if sum_y > target_sum:
            high = mid
        else:
            low = mid
    y = np.clip(x + low, 0, 1)
    return y

def random_pair_move(h):
    """Generate a random valid pairwise transfer move"""
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    delta_max = min(h[i], 1 - h[j])
    delta_min = max(-h[j], h[i] - 1)
    if delta_min < delta_max:
        delta = np.random.uniform(delta_min, delta_max)
        return i, j, delta
    else:
        return None, None, 0

def guided_move(h, worst_lag):
    """Apply deterministic improvement using gradient of worst lag"""
    n = len(h)
    # Compute the gradient g such that grad[0] = 0 * right + h[0+lag]  
    if worst_lag >= 0:
        g = np.concatenate([np.zeros(worst_lag), h[:-worst_lag]]) + np.concatenate([h[worst_lag:], np.zeros(worst_lag)])
    else:
        g = np.concatenate([np.zeros(-worst_lag), h[-worst_lag:]]) + np.concatenate([h[:-worst_lag], np.zeros(-worst_lag)])

    # Find indices to move from/to    
    min_idx = np.argmin(g)
    max_idx = np.argmax(g)

    # Default move (no-op)
    if min_idx == max_idx:
        return h, 0.0

    # Determine transfer amount
    delta_max = min(h[min_idx], 1 - h[max_idx])
    delta_min = max(-h[max_idx], h[min_idx] - 1)
    
    if delta_min < delta_max:
        delta = delta_max  # Move full amount if possible
    else:
        delta = 0.0  # No valid move

    if delta != 0.0:
        h_new = h.copy()
        h_new[min_idx] -= delta
        h_new[max_idx] += delta
        return h_new, delta
    else:
        return h, 0.0

def anneal(h0, run_seconds, rng_seed):
    """Run simulated annealing"""
    np.random.seed(rng_seed)
    start_time = time.time()
    
    h_current = project_to_sum_box(h0.copy(), TARGET_SUM)
    c5_current = evaluate_c5(h_current)
    
    # Initial temperature
    T0 = 5e-3
    Tf = 1e-6
    t = 0.0
    t_total = run_seconds
    
    best_h = h_current.copy()
    best_c5 = c5_current
    
    while time.time() - start_time < run_seconds:
        # Current temperature
        T = T0 * (1 - t / t_total) + Tf * (t / t_total)
        
        # Random Move
        i, j, delta = random_pair_move(h_current)
        if i is not None:
            h_new = h_current.copy()
            h_new[i] -= delta
            h_new[j] += delta
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            c5_new = evaluate_c5(h_new)
            
            # Accept step
            if c5_new <= c5_current or np.random.rand() < np.exp((c5_current - c5_new) / T):
                h_current = h_new
                c5_current = c5_new
                
                if c5_current < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_current
        
        t += 0.01  # Approximate time per iteration
        
        # Guided Movement (every 10 moves)
        if int(t) % 10 == 0:
            # Compute max correlation
            corr = compute_corr_fft(h_current)
            worst_lag = np.argmax(corr) - (len(h_current) - 1)
            h_new, delta = guided_move(h_current, worst_lag)
            if delta != 0:
                c5_new = evaluate_c5(h_new)
                if c5_new < c5_current:
                    h_current = h_new
                    c5_current = c5_new
                    
                    if c5_current < best_c5:
                        best_h = h_current.copy()
                        best_c5 = c5_current

    return best_h, best_c5

def constraint_eq(h):
    return np.sum(h) - TARGET_SUM

def objective(h):
    return evaluate_c5(h)

def local_refine(h):
    """Refine using SLSQP"""
    bounds = [(0, 1) for _ in range(len(h))]
    cons = {'type': 'eq', 'fun': constraint_eq}
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12})
    if result.success:
        h_result = project_to_sum_box(result.x, TARGET_SUM)
        return h_result
    else:
        h_result = project_to_sum_box(h, TARGET_SUM) 
        return h_result

    refined_h = local_refine(best_h)
    refined_c5 = evaluate_c5(refined_h)
    
    if refined_c5 < best_c5:
        best_h = refined_h
        best_c5 = refined_c5
