import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for a given step function h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def decode_h(params, n_points):
    """Decode parameters to full h vector"""
    K = len(params)
    # params are sorted in decreasing order to ensure monotonic decrease
    h_vals = np.sort(params)[::-1]
    h_vals = np.clip(h_vals, 0, 1)

    # Linear interpolation on [0,1] and mirror to [1,2]
    x_half = np.linspace(0, 1, n_points//2 + 1)
    h_half = np.interp(x_half, np.linspace(0, 1, K), h_vals)
    h_full = np.concatenate([h_half, h_half[-2::-1]])
    
    # Normalize
    integral = np.sum(h_full) * (2.0 / n_points)
    h_full = h_full / integral * 1.0
    
    # Clip to [0,1] and renormalize if needed
    h_full = np.clip(h_full, 0, 1)
    integral = np.sum(h_full) * (2.0 / n_points)
    h_full = h_full / integral * 1.0
    
    return h_full

def smooth_max_corr(h, beta=20.0):
    """Get smooth approximation of the max overlap"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    corr = corr * dx
    exp_corr = np.exp(beta * corr)
    return np.log(np.sum(exp_corr)) / beta

def project_simplex_box(x, s, l=0.0, u=1.0):
    """Project x onto the set {y | l <= y <= u, sum(y) = s}"""
    # Clip values
    x = np.clip(x, l, u)
    # Water filling
    total = np.sum(x)
    if total > s:
        # Need to reduce
        # Simple binary search for lambda
        low, high = -100, 100
        tol = 1e-12
        for _ in range(100):
            mid = (low + high) / 2
            y = np.clip(x - mid, l, u)
            total_y = np.sum(y)
            if total_y > s:
                low = mid
            else:
                high = mid
        y = np.clip(x - high, l, u)
        return y
    else:
        # Need to increase
        # Add to smallest values
        y = x.copy()
        diff = s - total
        # Add diff evenly to smallest
        idx = np.argsort(y)
        for i in idx:
            if diff <= 0:
                break
            add = min(y[i] - l, diff)
            y[i] -= add
            diff -= add
        return y

                        grad[i] += weights[k + n - 1] * dx * (-h[i + k])

                        grad[i] += weights[k + n - 1] * dx * (-h[i - k])

                best_c5 = c5_polish
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
