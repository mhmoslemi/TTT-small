```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5_exact(h):
    """Evaluate the exact C5 constant with true max operator"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def build_symmetric_piecewise_linear(n_points, break_points, heights):
    """Build a symmetric piecewise linear step function"""
    # Ensure heights are sorted decreasing
    heights = np.array(heights)
    heights = np.sort(heights)[::-1]
    
    # Create x coordinates for the left half [0,1]
    x_half = np.linspace(0, 1, n_points//2 + 1)
    
    # Interpolate on the left half
    x_full = np.linspace(0, 2, n_points)
    # Linearly interpolate heights at break points
    breakpoints_x = np.linspace(0, 1, len(break_points))
    h_left = np.interp(x_half[:-1], breakpoints_x, heights)
    
    # Mirror to full support
    h_full = np.concatenate([h_left, h_left[-2::-1]])
    
    # Scale to satisfy integral constraint
    total_mass = np.sum(h_full) * (2.0 / n_points)
    scale = (n_points / 2) / total_mass
    h_full = h_full * scale
    # Clip to [0,1]
    h_full = np.clip(h_full, 0, 1)
    
    return h_full

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parametrization: piecewise linear with symmetric structure
    n_points = 200
    K = 5  # Number of breakpoints in [0,1]
    
    # Define parameter bounds [0,1] for K+1 heights
    bounds = [(0, 1) for _ in range(K + 1)]
    
    # Set time slices
    t_de = 0.70 * budget_s
    t_pgd = 0.20 * budget_s
    t_final_opt = 0.10 * budget_s
    
    # Phase 1: Differential Evolution on low-dimensional parametrization
    def objective_de(params):
        h = build_symmetric_piecewise_linear(n_points, np.linspace(0, 1, K+1), params)
        return evaluate_c5_exact(h)
    
    print(f"Starting DE with {K+1} parameters, {n_points} points...")
    try:
        de_result = differential_evolution(
            objective_de,
            bounds,
            seed=seed,
            maxiter=int(5000 * (t_de / budget_s)),
            popsize=15,
            mutation=0.8,
            recombination=0.7,
            disp=False,
            pols=5000
        )
        print(f"DE completed, best value: {de_result.fun}")
    except Exception as e:
        print(f"DE failed with error: {e}")
        # Fall back to initial construction
        de_result = type('obj', (object,), {'x': np.ones(K+1) * 0.5, 'fun': 1.0})()
    
    # Initialize with the best from DE
    best_params = de_result.x.copy()
    h = build_symmetric_piecewise_linear(n_points, np.linspace(0, 1, K+1), best_params)
    current_c5 = evaluate_c5_exact(h)
    print(f"Initial best from DE: {current_c5}")
    
    # Phase 2: Smoothed PGD refinement
    t0 = time.time()
    beta = 10.0
    
    # Helper function to compute gradient
    def compute_grad(h, beta):
        n = len(h)
        dx = 2.0 / n
        corr = correlate(h, 1-h, mode="full")
        # compute f_k for k=0, ..., n-1
        f_vals = corr * dx
        # Normalize f_vals to avoid overflow
        f_max = np.max(f_vals)
        weights = np.exp(beta * (f_vals - f_max))
        weights = weights / np.sum(weights)
        # Compute gradient approximating max
        grad = np.zeros_like(h)
        for k in range(len(corr)):
            shift = k - (len(corr) - 1) // 2  # convert to standard shift
            # Compute derivative of f_k w.r.t h_i:
            # df_k/dh_i = dx * (1 - h_{i+k} - h_{i-k}_{+})
            # where indices are wrapped
            diff = np.zeros_like(h)
            # h_{i+k} (right shift)
            if shift > 0:
                diff[shift:] += h[:(n-shift)]
            elif shift < 0:
                diff[:(n+shift)] += h[-shift:]
            # h_{i-k} (left shift)
            if shift < 0:
                diff[-shift:] += h[:(n+shift)]
            elif shift > 0:
                diff[shift:] += h[:(n-shift)]
            # -h_{i+k} - h_{i-k} term here included already above
            term = dx * (np.ones_like(h) - diff)
            grad += weights[k] * term
        return grad
    
    # Water-filling projection
    def project_clipped(h, target_sum=1.0):
        # Apply a water-filling projection
        h = np.array(h)
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if np.abs(total - target_sum) < 1e-10:
            return h
        # Obvious case: just scale
        if total > target_sum:
            # Binary search for lambda such that sum of clip(h[i] - lambda, 0, 1) == target_sum
            # Let g(lambda) = sum of clip(h[i] - lambda, 0, 1)
            # g'(lambda) = - count of h[i] > lambda
            low, high = -1, 1
            # Ensure initial range
            while np.sum(np.clip(h - low, 0, 1)) < target_sum:
                low -= 0.1
            while np.sum(np.clip(h - high, 0, 1)) > target_sum:
                high += 0.1
            eps = 1e-8
            while high - low > 1e-3:
                mid = (low + high) / 2
                s = np.sum(np.clip(h - mid, 0, 1))
                if s >= target_sum:
                    low = mid
                else:
                    high = mid
            lambda_opt = (low + high) / 2
            return np.clip(h - lambda_opt, 0, 1)
        else:
            # Can't possibly fill
            return h
    
    # Main PGD loop with decaying learning rate
    lr = 0.05
    pgd_iter = 0
    num_pgd = int(t_pgd / (0.01 * 1e-3))  # Estimate of iterations per second
    
    while (time.time() - t0) < t_pgd and pgd_iter < num_pgd:
        # Compute gradient
        try:
            grad = compute_grad(h, beta)
            # Gradient step
            h_new = h - lr * grad
            # Project
            h_new = project_clipped(h_new, n_points / 2)
            # Compute new value
            c5_new = evaluate_c5_exact(h_new)
            if c5_new < current_c5:
                h = h_new.copy()
                current_c5 = c5_new
                if pgd_iter % 50 == 0:
                    print(f"PGD iter {pgd_iter}, C5 = {current_c5}")
            # Adapt learning rate
            lr = max(1e-6, lr * 0.9995)
            pgd_iter += 1
        except Exception as e:
            print(f"PGD error: {e}")
            break
    
    print(f"PGD finished, best C5: {current_c5}")
    
    # Phase 3: Final polishing
    final_opt_start = time.time()
    
    def objective_final(x):
        return evaluate_c5_exact(x)
    
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) * (2.0 / n_points) - 1.0}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(
            objective_final,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12},
            tol=1e-12
        )
        if res.success:
            h_final = res.x
            optimal_c5 = evaluate_c5_exact(h_final)
            if optimal_c5 < current_c5:
                current_c5 = optimal_c5
                h = h_final.copy()
                print(f"Final polishing improved C5 to: {current_c5}")
    except Exception as e:
        print(f"Final opt failed: {e}")
    
    print(f"Overall result: C5 = {current_c5}")
    return (h.tolist(), current_c5, n_points)
```
