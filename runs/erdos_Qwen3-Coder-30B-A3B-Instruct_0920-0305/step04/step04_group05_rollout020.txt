import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, target_sum):
    """Project h onto feasible set: [0,1]^n and sum = target_sum"""
    # Clip to [0,1]
    y = np.clip(h, 0, 1)
    # Adjust sum to target_sum
    current_sum = np.sum(y)
    if abs(current_sum - target_sum) < 1e-9:
        return y
    
    # Use iterative scaling
    for _ in range(100):
        diff = current_sum - target_sum
        if abs(diff) < 1e-9:
            break
        if diff > 0:  # sum too high
            # Reduce uniformly from values not at lower bound
            free = y > 0
            if not free.any():
                break
            scale = target_sum / current_sum
            y = y * scale
        else:  # sum too low
            # Increase uniformly from values not at upper bound
            free = y < 1
            if not free.any():
                break
            scale = target_sum / current_sum
            y = y * scale
        y = np.clip(y, 0, 1)
        current_sum = np.sum(y)
    
    return y
