import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5_exact(h):
    # Evaluate the C5 constant exactly for a given step function h
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def make_symmetric_step_function(u_vals, n):
    # u_vals: heights from the parametrized family
    # n: number of points in final vector
    k = len(u_vals) - 1  # number of breakpoints - 1
    # Construct piecewise linear symmetric function
    x_points = np.linspace(0, 1, k + 1)
    # Sorted and reversed to ensure decreasing heights
    u_sorted = np.sort(u_vals)[::-1]
    
    # Define a linear piecewise function for [0,1]
    # Use linear interpolation between (x_points[i], u_sorted[i]), (x_points[i+1], u_sorted[i+1])
    x_interp = np.linspace(0, 1, n//2 + 1)
    # Interpolate over [0, 1]
    h_half = np.interp(x_interp, x_points, u_sorted, left=u_sorted[0], right=u_sorted[-1])

    h_full = np.concatenate([h_half, h_half[-2::-1]])  # Mirror to get [0,2]

    # Scale to enforce ∫ h dx = 1
    total = np.sum(h_full) * (2.0 / n)
    if total != 0:
        h_full = h_full * (n / (2 * total))
    # Clip if needed
    h_full = np.clip(h_full, 0, 1)

    return h_full

def objective_de(u_vals):
    # For differential evolution
    h = make_symmetric_step_function(u_vals, n_points)
    
    # Return negative because we want to minimize C5
    return -evaluate_c5_exact(h)

def project_simplex(h, target_sum):
    # Project h onto the simplex {x: x >= 0, sum x = target_sum}
    # Using waterfilling method
    
    # Ensure it's not already clipped
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    
    if np.isclose(total, target_sum):
        return h
    
    # Binary search for lambda
    def project_with_lambda(lam):
        p = np.clip(h - lam, 0, 1)
        return np.sum(p) - target_sum

    # Some bounds on lambda
    lower = -1
    upper = 1
    
    # Ensure bounds
    while project_with_lambda(lower) > 0:
        lower -= 1
    while project_with_lambda(upper) < 0:
        upper += 1

    # Binary search for the correct lambda
    for _ in range(50):
        lam = (lower + upper) / 2
        diff = project_with_lambda(lam)
        if diff > 0:
            lower = lam
        else:
            upper = lam
    lam = (lower + upper) / 2
    
    h_proj = np.clip(h - lam, 0, 1)
    return h_proj

def smooth_max(h, beta=20):
    # Smooth approximation of max
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    corr *= dx

    # Add a small padding to prevent overflow
    max_val = np.max(corr)
    corr_shifted = corr - max_val
    
    exp_corr = np.exp(beta * corr_shifted)
    sum_exp = np.sum(exp_corr)
    if sum_exp == 0:
        return corr[0]  # fallback
    w = exp_corr / sum_exp
    return np.sum(w * corr)

def gradient_smooth_max(h, beta=20):
    # Return the gradient of the smooth max approximation
    n = len(h)
    dx = 2.0 / n
    # Compute each f_k separately
    f = []
    grad_f = []
    for k in range(-n+1, n):
        # compute correlation but not for the full mode
        # compute it via manual loop to avoid correlations
        fk = 0.0
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                fk += h[i] * (1 - h[j])
        fk *= dx
        f.append(fk)
    
    # Now compute gradient
    # For k=0, ..., n-1: 
    # grad_{h_i} (sum_k w_k * f_k)
    # w_k = exp(beta * f_k) / \sum_{ell} exp(beta * f_ell)
    
    max_f = np.max(f)
    f_shifted = np.array(f) - max_f  # subtract max to prevent overflow
    exp_f = np.exp(beta * f_shifted)
    sum_exp = np.sum(exp_f)
    if sum_exp == 0:
        return np.zeros_like(h)
    w = exp_f / sum_exp
    
    # Compute ∇f_k for each k
    grad_f = []
    for k in range(-n+1, n):
        grad_new = np.zeros(n)
        for i in range(n):
            j = i + k
            j2 = i - k
            val = 0.0
            # Only add contribution if indices are in valid range
            if 0 <= j < n:
                val += -dx * h[j]
            if 0 <= j2 < n:
                val += -dx * h[j2]
            grad_new[i] = val
        grad_f.append(grad_new)
    
    # Now compute weighted sum
    gradient = np.zeros(n)
    for k in range(len(w)):
        gradient += w[k] * grad_f[k]
        
    return gradient

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 200
    k_breakpoints = 5
    bounds_de = [(0, 1) for _ in range(k_breakpoints)]
    
    # Phase 1: Global Optimization with DE
    t_de = 0.7 * budget_s
    res_de = differential_evolution(
        objective_de,
        bounds_de,
        maxiter=int(t_de / 0.5),
        popsize=15,
        seed=seed,
        disp=False,
        tol=1e-6
    )
    
    # Extract best solution from DE
    best_u_vals = res_de.x
    
    # Convert to full resolution h vector
    h = make_symmetric_step_function(best_u_vals, n_points)
    c5_best = evaluate_c5_exact(h)
    
    # Phase 2: Local Refinement with Smoothed Max
    t_pgd = 0.2 * budget_s
    t_pgd_end = time.time() + t_pgd
    
    # Initialize PGD variables
    h_current = np.copy(h)
    beta = 10.0
    
    def pgd_step(h_prev, step_size=0.05):
        # Compute gradient of smooth max
        grad = gradient_smooth_max(h_prev, beta)
        h_new = h_prev - step_size * grad
        # Project onto feasible set
        h_new = project_simplex(h_new, n_points / 2.0)
        return h_new

    # PGD loop
    while time.time() < t_pgd_end:
        h_next = pgd_step(h_current, 0.02)
        c5_new = evaluate_c5_exact(h_next)
        if c5_new < c5_best - 1e-8:
            c5_best = c5_new
            h_current = h_next
        if beta < 20:
            beta += 0.5
        # Reduce step size
        if time.time() > t_pgd_end - 20:
            step_size = 0.01
        else:
            step_size = 0.02

    # Phase 3: Final polishing with SLSQP
    t_final = 0.1 * budget_s
    t_final_end = time.time() + t_final
    
    def objective_slsqp(h):
        return evaluate_c5_exact(h)
    
    def constraint_eq(h):
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(n_points)]

    try:
        res_slsqp = minimize(
            objective_slsqp,
            h_current.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        if res_slsqp.success:
            h_final = res_slsqp.x
            c5_final = evaluate_c5_exact(h_final)
            if c5_final < c5_best:
                c5_best = c5_final
                h_current = h_final
    except:
        pass

    return (h_current.tolist(), c5_best, n_points)
