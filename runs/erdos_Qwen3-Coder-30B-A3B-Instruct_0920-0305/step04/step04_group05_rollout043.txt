import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def generate_diverse_initialization(n_points, pattern):
    """Generate various patterns for initializing optimization"""
    if pattern == 0:
        # Smooth bump
        x = np.linspace(0, 2, n_points)
        return np.exp(-10 * (x - 1)**2)
    elif pattern == 1:
        # Tent function
        x = np.linspace(0, 2, n_points)
        return np.maximum(0, 1 - np.abs(x - 1) * 2)
    elif pattern == 2:
        # Two peaks
        x = np.linspace(0, 2, n_points)
        return np.exp(-5 * (x - 0.5)**2) + np.exp(-5 * (x - 1.5)**2)
    elif pattern == 3:
        # Sinusoidal
        x = np.linspace(0, 2, n_points)
        return 0.5 + 0.3 * np.sin(np.pi * x)
    elif pattern == 4:
        # Step pattern
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1
        return h
    elif pattern == 5:
        # Gaussian shaped
        x = np.linspace(0, 2, n_points)
        return np.exp(-20 * (x - 1)**2)
    elif pattern == 6:
        # Random
        return np.random.uniform(0, 1, n_points)
    elif pattern == 7:
        # Smooth bump shifted
        x = np.linspace(0, 2, n_points)
        return np.exp(-15 * (x - 0.7)**2)
    elif pattern == 8:
        # Narrow spike
        h = np.zeros(n_points)
        h[n_points//2] = 1
        return h
    else:
        # Random with some structure
        x = np.linspace(0, 2, n_points)
        return 0.4 + 0.2 * np.sin(3*np.pi*x) + 0.1 * np.random.random(n_points)

def local_optimization_with_variants(h0, bounds, cons, start_time, budget_s):
    """Perform local optimization with multiple restarts"""
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Multiple restarts with different perturbations
    for i in range(20):
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
        # Choose perturbation type
        pert_type = i % 6
        
        if pert_type == 0:
            # Small random noise
            new_h = h0 * (1.0 + np.random.normal(0, 0.02, n_points))
        elif pert_type == 1:
            # Pulse adjustment
            new_h = h0.copy()
            new_h[::5] = np.clip(new_h[::5] + np.random.normal(0, 0.1, len(new_h[::5])), 0, 1)
        elif pert_type == 2:
            # Fourier-like pattern
            x = np.linspace(0, 2, n_points)
            new_h = h0 + 0.05 * np.sin(5 * np.pi * x) + 0.03 * np.cos(3 * np.pi * x)
        elif pert_type == 3:
            # Peak modification
            new_h = h0.copy()
            peaks = np.argsort(new_h)[-3:]  # Top 3 values
            for peak in peaks:
                new_h[peak] += np.random.normal(0, 0.03)
        elif pert_type == 4:
            # Gradient-like modification
            new_h = h0.copy()
            for j in range(1, n_points-1):
                new_h[j] += 0.02 * (new_h[j+1] - new_h[j-1])
        else:
            # Slightly more substantial adjustment
            new_h = h0 + np.random.normal(0, 0.03, n_points)
        
        # Ensure valid range
        new_h = np.clip(new_h, 0, 1)
        
        # Normalize to satisfy integral constraint
        scale = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale
        
        # Optimize
        try:
            res = minimize(
                objective, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue
            
    return best_h, best_c5
