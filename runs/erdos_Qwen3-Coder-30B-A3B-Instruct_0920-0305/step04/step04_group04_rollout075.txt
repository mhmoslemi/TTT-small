import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    h0 = np.array(initial_h_values)
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # MILP approach
    try:
        # Create binary variables for h
        h = cp.Variable(n_points, boolean=True)
        # Create auxiliary variables for p_{i,l}
        p = {}
        M = cp.Variable(integer=True)
        
        # Add symmetry breaking constraint
        h[0].value = 1
        
        # Constraints for logical AND
        constraints = []
        
        # Add integral constraint
        constraints.append(cp.sum(h) == n_points / 2)
        
        # M is at least the worst case overlap count
        constraints.append(M >= 0)
        
        # For all lags
        for l in range(1, n_points):
            # maximum number of valid overlapping indices for lag l  
            max_i = n_points - l
            # index sets for the two windows of overlap
            left_indices = list(range(max_i))
            right_indices = list(range(l, n_points))
            # Create auxiliary variables p_{i,l} and require that
            # p[i,l] = h[i] * h[i+l] = h[i] AND h[i+l]
            for i in range(max_i):
                p[i,l] = cp.Variable(boolean=True)
                constraints.append(p[i,l] <= h[i])
                constraints.append(p[i,l] <= h[i+l])
                constraints.append(p[i,l] >= h[i] + h[i+l] - 1)
    
            # Count the number of 1s in the overlapping parts
            S_l = cp.sum([p[i,l] for i in range(max_i)])
            A_l_plus = cp.sum([h[i] for i in left_indices])
            A_l_minus = cp.sum([h[i] for i in right_indices])
            
            # Worst-case overlap constraints
            constraints.append(A_l_plus - S_l <= M)
            constraints.append(A_l_minus - S_l <= M)
        
        # Objective is to minimize M
        objective = cp.Minimize(M)
        
        # Create and solve the problem
        prob = cp.Problem(objective, constraints)
        
        # Determine remaining time for solving
        remaining_time = max(1, budget_s - (time.time() - start_time))
        
        # Solve with time limit
        try:
            prob.solve(solver=cp.CBC, time_limit=remaining_time)
        except:
            # If solver failed, return the original construction as fallback
            best_h = h0.copy()
            best_c5 = evaluate_c5(best_h)
        else:
            # Extract solution
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                best_h_values = [float(h_val.value) for h_val in h]
                best_h = np.array(best_h_values)
                # Check the scaled value of M obtained from the MILP
                # In the unscaled setting, we can compute M directly
                # since it's the maximum of A_l - S_l for all lags l
                M_val = int(M.value)
                best_c5 = M_val * dx
                # Convert to floating-point and polish
                h_float = best_h.astype(float)
                # Set up constraints for polishing
                bounds = [(0, 1) for _ in range(n_points)]
                def polish_objective(h):
                    return evaluate_c5(h)
                def polish_constraint_eq(h):
                    return np.sum(h) * dx - 1.0
                cons = {'type': 'eq', 'fun': polish_constraint_eq}
                
                # Multiple restarts for polishing
                best_polished_c5 = best_c5
                best_polished_h = h_float.copy()
                
                n_rep = min(300, int((budget_s * 0.1) / 0.02)) 
                
                for _ in range(n_rep):
                    if (time.time() - start_time) > budget_s * 0.98:
                        break
                    try:
                        # Start from the binary solution and add noise
                        new_h = h_float + np.random.normal(0, 0.01, n_points)
                        new_h = np.clip(new_h, 0, 1)
                        scale_factor = 1.0 / (np.sum(new_h) * dx)
                        new_h = new_h * scale_factor
                        
                        res = minimize(
                            polish_objective,
                            new_h,
                            method='SLSQP',
                            bounds=bounds,
                            constraints=cons,
                            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
                        )
                        
                        if res.success:
                            h_candidate = res.x
                            c5_candidate = polish_objective(h_candidate)
                            if c5_candidate < best_polished_c5:
                                best_polished_h = h_candidate.copy()
                                best_polished_c5 = c5_candidate
                    except:
                        continue
                        
                # Return the final best result
                return (best_polished_h.tolist(), best_polished_c5, n_points)
            else:
                # Fallback to the original solution if MILP didn't give a good result
                best_h = h0.copy()
                best_c5 = evaluate_c5(best_h)
                
    except Exception as e:

        best_h = h0.copy()
        best_c5 = evaluate_c5(best_h)
        
        # Simulated Annealing approach as fallback
        current_h = best_h.copy()
        current_c5 = best_c5
        best_h = current_h.copy()
        best_c5 = current_c5
        
        # Parameters for simulated annealing
        max_iter = int((budget_s * 0.8) / 0.001)  # iterations based on estimate
        temp = 1.0
        cooling_rate = 0.995
        min_temp = 1e-10
        
        def acceptance_probability(old_c5, new_c5, temperature):
            if new_c5 < old_c5:
                return 1.0
            else:
                return np.exp(-((new_c5 - old_c5) / temperature))
        
        for i in range(max_iter):
            if (time.time() - start_time) > budget_s * 0.98:
                break
                
            # Get a neighbor by swapping two positions
            # Find 1s and 0s
            ones = list(np.where(current_h > 0.5)[0])  # integer indices
            zeros = list(np.where(current_h < 0.5)[0])
            
            if len(ones) == 0 or len(zeros) == 0:
                continue
                
            i1 = np.random.choice(ones)
            i0 = np.random.choice(zeros)
            
            # Swap
            new_h = current_h.copy()
            new_h[i1] = 0
            new_h[i0] = 1
            
            # Normalize
            scale = 1.0 / (np.sum(new_h) * dx)
            new_h = new_h * scale
            
            # Evaluate new solution
            new_c5 = evaluate_c5(new_h)
            
            # Accept or not
            if acceptance_probability(current_c5, new_c5, temp) > np.random.rand():
                current_h = new_h
                current_c5 = new_c5
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
            
            temp *= cooling_rate
            if temp < min_temp:
                break
        
        # Post-process with polishing if needed
        h_float = best_h.astype(float)
        bounds = [(0, 1) for _ in range(n_points)]
        def polish_objective(h):
            return evaluate_c5(h)
        def polish_constraint_eq(h):
            return np.sum(h) * dx - 1.0
        cons = {'type': 'eq', 'fun': polish_constraint_eq}
        
        n_rep = 300

            if (time.time() - start_time) > budget_s * 0.98:
                break
            try:
                # Start from the binary solution and add noise
                new_h = h_float + np.random.normal(0, 0.01, n_points)
                new_h = np.clip(new_h, 0, 1)
                scale_factor = 1.0 / (np.sum(new_h) * dx)
                new_h = new_h * scale_factor
                
                res = minimize(
                    polish_objective,
                    new_h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
                )
                
                if res.success:
                    h_candidate = res.x
                    c5_candidate = polish_objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
            except:
                continue
    
    return (best_h.tolist(), best_c5, n_points)
