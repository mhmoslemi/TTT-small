import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use a hybrid optimization approach
    # Phase 1: Use global optimization
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Try using different optimization approaches like simulated annealing
    # First try local search as before
    strategies = ['best1bin', 'best2bin', 'rand1bin', 'rand2bin', 'currenttobestbin']
    
    for strategy in strategies:
        if (time.time() - start_time) > budget_s:
            break
        try:
            de_res = differential_evolution(
                objective, 
                bounds, 
                seed=seed, 
                maxiter=500,  
                popsize=15,    
                mutation=(0.8, 1),  
                recombination=0.9,   
                strategy=strategy
            )
            if de_res.success:
                h_candidate = de_res.x
                # Normalize to satisfy constraint
                dx = 2.0 / n_points
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
    
    # Phase 2: Local search from good global points
    local_restart_bound = 300
    
    for restart in range(local_restart_bound):
        if (time.time() - start_time) > budget_s:
            break
            
        # Start with small random perturbations around previous best
        new_h = best_h + np.random.normal(0, 0.01, n_points)
        new_h = np.clip(new_h, 0, 1)
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor
        
        try:
            res = minimize(
                objective, 
                new_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-13, 'gtol': 1e-13}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue

    # Phase 3: Try more advanced approach
    # Try other robust methods 
    if (time.time() - start_time) < budget_s * 0.8:
        try:
            import scipy.optimize as so
            x0 = best_h.copy()
            res = so.minimize(
                objective, 
                x0, 
                method='BFGS', 
                options={'maxiter': 100, 'gtol': 1e-13}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass

    # Try a hybrid construction technique based on suggested improvements
    if (time.time() - start_time) < budget_s * 0.9:
        # Try building pattern to improve min overlap
        try:
            # Attempt pattern with alternating high/low pattern with slight variance
            # Create smoother version that balances high/low strengths while normalizing
            new_pattern = np.zeros(n_points)
            # Set alternating pattern of ones and zeros manually to test modification
            for i in range(0, n_points, 2):
                if i < n_points:
                    new_pattern[i] = 1
            # Add some flexibility by adjusting a few values to improve overlap
            for i in range(0, n_points, 3):  # every third element
                if 0 <= i < n_points:
                    new_pattern[i] = np.clip(new_pattern[i] + 0.1, 0, 1)
            
            # Normalize to correct integral
            dx = 2.0 / n_points
            scale_factor = 1.0 / (np.sum(new_pattern) * dx)
            new_pattern = new_pattern * scale_factor
            
            # Evaluate
            c5_candidate = evaluate_c5(new_pattern)
            if c5_candidate < best_c5:
                best_h = new_pattern.copy()
                best_c5 = c5_candidate
        except:
            pass
    
    return (best_h.tolist(), best_c5, n_points)
