import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

        M = cp.Variable(integer=True, nonneg=True)
        
        # Constraints for logical AND
        constraints = []
        for l in range(1, n_points):
            for i in range(n_points - l):
                # p[i,l] <= h[i]
                constraints.append(p[i, l] <= h[i])
                # p[i,l] <= h[i+l]
                constraints.append(p[i, l] <= h[i + l])
                # p[i,l] >= h[i] + h[i+l] - 1
                constraints.append(p[i, l] >= h[i] + h[i + l] - 1)
        
        # Intersection counts
        A_plus = [cp.sum(h[:n_points-l]) for l in range(1, n_points)]
        A_minus = [cp.sum(h[l:]) for l in range(1, n_points)]
        S = [cp.sum(p[:, l][:n_points-l]) for l in range(1, n_points)]
        
        # Worst-case overlap constraints
        for l in range(1, n_points):
            constraints.append(A_plus[l-1] - S[l-1] <= M)
            constraints.append(A_minus[l-1] - S[l-1] <= M)

        constraints.append(cp.sum(h) == n_points // 2)
