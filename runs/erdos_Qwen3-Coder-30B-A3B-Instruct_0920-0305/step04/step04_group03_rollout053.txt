def project_to_boxed_simplex(v, target_sum):
    # Clip to box first
    v_clip = np.clip(v, 0, 1)
    # Adjust to meet sum target while staying within [0,1]
    # Iteratively adjust entries not at bounds
    while True:
        s = np.sum(v_clip)
        if abs(s - target_sum) < 1e-10:
            break
        diff = target_sum - s
        # Find free entries not at bounds
        free = (v_clip > 0) & (v_clip < 1)
        nfree = np.sum(free)
        if nfree == 0:
            break
        delta = diff / nfree
        v_clip[free] += delta
        v_clip = np.clip(v_clip, 0, 1)
    return v_clip

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def binary_to_float(binary_vec):
    # Convert binary vector to float and ensure correct sum
    h = binary_vec.astype(float)
    n = len(h)
    target_sum = n / 2.0
    if abs(np.sum(h) - target_sum) > 1e-10:
        scale = target_sum / np.sum(h)
        h = h * scale
    return h

def simulate_annealing_binary(n_points, max_iter, seed):
    np.random.seed(seed)
    
    # Initialize binary vector with exactly n_points/2 ones
    h = np.zeros(n_points, dtype=int)
    h[:n_points//2] = 1
    np.random.shuffle(h)
    
    # Compute initial min correlation
    corr = correlate(h, h, mode='full')
    min_corr = np.min(corr)
    
    best_h = h.copy()
    best_min_corr = min_corr
    
    temperature = max(10.0, n_points)
    cooling_rate = 0.99995
    
    for i in range(max_iter):
        # Generate neighbor by swapping two random positions
        idx1 = np.random.randint(0, n_points)
        idx2 = np.random.randint(0, n_points)
        if idx1 == idx2:
            continue
            
        # Create neighbor
        h_new = h.copy()
        h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
        
        # Compute min correlation for neighbor
        corr_new = correlate(h_new, h_new, mode='full')
        min_corr_new = np.min(corr_new)
        
        # Accept based on acceptance probability
        if min_corr_new >= min_corr or np.random.rand() < np.exp((min_corr_new - min_corr) / temperature):
            h = h_new
            min_corr = min_corr_new
            
            if min_corr_new > best_min_corr:
                best_h = h.copy()
                best_min_corr = min_corr_new
        
        # Decrease temperature
        temperature *= cooling_rate
        
        if i % 1000 == 0:
            # Early stopping if we hit history
            pass

    return best_h, best_min_corr

    # Set parameters
    n_points = 80
    max_iter_SA = 50000
    max_iter_gradient = 2000
    
    # Track elapsed time
    start_time = time.time()
    
    # Use initial construction to warm-start
    try:
        h0 = np.array(initial_h_values)
        if len(h0) == n_points:
            h_start = h0.copy()
        else:
            # Interpolate or use resample to adjust 
            h_start = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, len(h0)), h0)
    except:
        # Default initialization if no initial_h_values
        h_start = np.random.rand(n_points)
        h_start = np.clip(h_start, 0, 1)
        # Renormalize sum to match constraint
        current_sum = np.sum(h_start)
        h_start = h_start * (n_points / 2) / current_sum
    
    # Phase 1: Binary Simulated Annealing
    binary_h, min_corr = simulate_annealing_binary(n_points, max_iter_SA, seed)
    h_best = binary_to_float(binary_h)
    
    # Phase 2: Gradient descent
    if (time.time() - start_time) < budget_s * 0.8:
        h_final, c5_final = run_gradient_descent(h_best, n_points, max_iter_gradient)
        if c5_final < evaluate_c5(h_best):
            h_best = h_final
            c5_best = c5_final
        else:
            c5_best = evaluate_c5(h_best)
    else:
        c5_best = evaluate_c5(h_best)
    
    # Phase 3: Final local refinement if time allows
    if (time.time() - start_time) < budget_s * 0.9:
        try:
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
            bounds = [(0, 1) for _ in range(n_points)]
            
            def obj_func(x):
                return evaluate_c5(x)
                
            res = minimize(
                obj_func,
                h_best.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_c5(h_refined)
                if c5_refined < c5_best:
                    h_best = h_refined
                    c5_best = c5_refined
        except:
            pass
