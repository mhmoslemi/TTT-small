import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def logsumexp_smooth_max(f_values, beta=20.0):
    """Smooth approximation to the maximum."""
    f_max = np.max(f_values)
    shifted = f_values - f_max
    exp_shifted = np.exp(beta * shifted)
    return f_max + np.log(np.sum(exp_shifted)) / beta

def make_symmetric_monotone_vector(half_heights, n_points):
    """Generate full symmetric vector from decreasing half-heights."""
    # Sort heights to ensure they are decreasing
    heights = np.sort(half_heights)[::-1]
    
    # Interpolate the first half from index 0 to n_points//2
    left_indices = np.linspace(0, 1, n_points//2 + 1)
    left_heights_interp = np.interp(left_indices, np.linspace(0, 1, len(heights)), heights)
    
    # Mirror the right half for symmetry across center x=1
    right_heights_interp = left_heights_interp[::-1]
    
    # Combine the two parts (excluding duplicate center point)
    full_h = np.concatenate([left_heights_interp[:-1], right_heights_interp])
    
    return full_h

def scale_and_respect_bounds(h, target_sum):
    """Scale vector to match target sum while clipping to [0,1]."""
    total = np.sum(h)
    if total == 0:
        h = np.ones_like(h) * (target_sum / len(h))
    else:
        h *= target_sum / total
    np.clip(h, 0, 1, out=h)
    # If still over the target sum after clipping, enforce with binary search
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-8:
        # Perform iterative scaling to match exactly
        while abs(np.sum(h) - target_sum) > 1e-8:
            # Use water filling approach to re-scale after clipping
            raw_h = h.copy()
            total = np.sum(raw_h)
            if total <= 0:
                return np.full_like(h, target_sum/len(h))
            scaled_h = raw_h * target_sum / total
            np.clip(scaled_h, 0, 1, out=h)
            # Normalize to target again, with small iteration to fix exact sum
            current_sum = np.sum(h)
            # Avoid over-correction by adding only small adjustment
            if abs(current_sum - target_sum) > 1e-6:
                factor = target_sum / current_sum
                h[:] = np.minimum(np.maximum(h * factor, 0), 1)
                if np.sum(h) > target_sum:
                    break  # Already monotonic enough
        h = h * target_sum / np.sum(h)
    return h

def decode_params_to_h(params, n_points):
    """Convert symmetric monotone parameters to full vector."""
    K = len(params)
    # Build the piecewise-linear symmetric shape from breakpoint heights
    h = make_symmetric_monotone_vector(params, n_points)
    # For simplicity--ensure exact constraint
    h = scale_and_respect_bounds(h, n_points / 2)
    return h

def smooth_objective(params, n_points):
    """Evaluate the smooth objective without computing actual max."""
    h = decode_params_to_h(params, n_points)
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode="full")
    f_values = corr * dx
    # Smooth maximum with exponential weighting as described
    return -logsumexp_smooth_max(f_values, beta=15.0)

def projected_gradient_descent(h_start, n_points, max_iter=1000):
    """Use PGD with a smooth surrogate to refine the solution."""
    h_current = h_start.copy()
    dx = 2.0 / n_points
    beta = 15.0
    lr = 0.05
    history = []
    # smooth max (logsumexp)
    for iter_ in range(max_iter):
        # Compute all correlation values (f_k)
        corr = correlate(h_current, 1 - h_current, mode="full")
        f_values = corr * dx
        # Compute weights
        f_max = np.max(f_values)
        exp_f = np.exp(beta * (f_values - f_max))
        weights = exp_f / np.sum(exp_f)
        # Compute gradient: \nabla f_k(h) = dx * (1 - h_{j+k} - h_{j-k})
        grad = np.zeros_like(h_current)
        for k_idx, (k, weight) in enumerate(zip(range(-n_points+1, n_points), weights)):
            # use shifted versions of h
            if k == 0:
                shifted = h_current
            elif k > 0:
                shifted = np.pad(h_current[:-k], (k, 0), mode='constant', constant_values=0)
            else:
                shifted = np.pad(h_current[-k:], (0, -k), mode='constant', constant_values=0)
            grad += weight * dx * (1 - shifted - np.roll(h_current, -k))
        # Update step
        h_new = h_current - lr * grad
        # Project onto feasible set [0,1], preserving integral
        # Using binary search for exact sum match after clipping
        def compute_sum(lambda_shift):
            p = np.clip(h_new - lambda_shift, 0, 1)
            return np.sum(p) - n_points / 2
        
        # Binary search to find lambda
        low, high = -1.0, 1.0
        for _ in range(30):
            mid = (low + high) / 2.0
            val = compute_sum(mid)
            if val > 0:
                high = mid
            else:
                low = mid
        lambda_opt = (low + high) / 2.0
        h_new = np.clip(h_new - lambda_opt, 0, 1)
        scale_factor = (n_points / 2) / np.sum(h_new)
        h_current = h_new * scale_factor
        # Evaluate smooth objective on every 10th step
        if iter_ % 10 == 0:
            corr = correlate(h_current, 1 - h_current, mode="full")
            f_values = corr * dx
            history.append(-logsumexp_smooth_max(f_values, beta=beta))
        if len(history) > 5 and abs(history[-1] - history[-2]) < 1e-6:
            break
    return h_current
