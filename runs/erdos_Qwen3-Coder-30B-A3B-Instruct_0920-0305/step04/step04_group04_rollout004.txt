import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    n_points = len(initial_h)
    
    # Time for MILP solving
    milp_time = min(200, budget_s * 0.7)  # 200 seconds or 70% of budget
    remaining_time = budget_s - milp_time
    
    # MIP approach using CVXPY
    h_vars = cp.Variable(n_points, boolean=True)
    p_vars = cp.Variable((n_points, n_points), boolean=True)
    M = cp.Variable(integer=True)
    
    # Constraints for p_vars (logical AND)
    constraints = []
    
    # Integral constraint
    constraints.append(cp.sum(h_vars) == n_points // 2)
    
    # Logical AND constraints for all lags
    for l in range(1, n_points):
        for i in range(n_points - l):
            # p[i,l] <= h[i]
            constraints.append(p_vars[i, l] <= h_vars[i])
            # p[i,l] <= h[i+l]
            constraints.append(p_vars[i, l] <= h_vars[i + l])
            # p[i,l] >= h[i] + h[i+l] - 1
            constraints.append(p_vars[i, l] >= h_vars[i] + h_vars[i + l] - 1)
    
    # Overlap counting constraints - set M to be at least A_l - S_l for all lags
    for l in range(1, n_points):
        # A_l^+ = sum of h_i for i in 0..n-l-1
        A_plus = cp.sum(h_vars[:n_points-l])
        # A_l^- = sum of h_i for i in l..n-1
        A_minus = cp.sum(h_vars[l:])
        # S_l = sum of p[i,l]
        S_l = cp.sum(p_vars[:n_points-l, l])
        # Constraint: A_l^+ - S_l <= M and A_l^- - S_l <= M
        constraints.append(A_plus - S_l <= M)
        constraints.append(A_minus - S_l <= M)
    
    # Objective: Minimize M
    objective = cp.Minimize(M)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=milp_time)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_binary = h_vars.value.astype(int)
            M_value = M.value
            dx = 2.0 / n_points
            C5_bound_milp = M_value * dx
        else:
            # Fallback to simulated annealing
            h_binary = initial_h.copy()
            C5_bound_milp = evaluate_c5(h_binary)
    except:

        h_binary = initial_h.copy()
        C5_bound_milp = evaluate_c5(h_binary)
    
    # Convert to float and initialize
    h_float = h_binary.astype(float)
    dx = 2.0 / n_points
    scale = 1.0 / (np.sum(h_float) * dx)
    h_float = h_float * scale
    best_c5 = evaluate_c5(h_float)
    best_h = h_float.copy()
    
    # Local polishing with SLSQP over the continuous space
    def objective_continuous(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Multiple restarts
    num_restarts = 100
    for _ in range(num_restarts):
        if (time.time() - start_time) > budget_s - 2:
            break
            
        # Start with current best
        h_start = best_h.copy()
        
        # Add noise
        noise = np.random.normal(0, 0.03, n_points)
        h_perturbed = h_start + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        scale_factor = 1.0 / (np.sum(h_perturbed) * dx)
        h_perturbed = h_perturbed * scale_factor
        
        try:
            res = minimize(
                objective_continuous, 
                h_perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 50, 'ftol': 1e-10, 'gtol': 1e-10}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective_continuous(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue
    
    return (best_h.tolist(), best_c5, n_points)
