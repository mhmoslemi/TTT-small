import numpy as np

def build_symmetric_h(u_vals, n):
    """Build a symmetric, piecewise linear function h from the u values."""
    K = len(u_vals) - 1  # number of internal points
    x_vals = np.linspace(0, 1, K+1)
    
    # Clamped values to [0,1]
    u_vals = np.clip(u_vals, 0, 1)
    # Sort descending to maintain monotonicity
    u_vals = np.sort(u_vals)[::-1]
    
    # construct values over [0, 1]
    h_left = np.interp(np.linspace(0, 1, n//2 + 1), x_vals, u_vals)
    
    # Mirror to [1, 2]
    h_right = h_left[::-1]
    
    h = np.concatenate([h_left[:-1], h_right])
    
    return h

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, target_area=1.0):
    """Project h onto the feasible set of values in [0,1] with sum * dx = target_area."""
    dx = 2.0 / len(h)
    total = np.sum(h) * dx
    
    if abs(total - target_area) < 1e-10:
        return h
    
    # Rescale
    scale = target_area / total
    h = h * scale
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Renormalize if still not correct
    total = np.sum(h) * dx
    if abs(total - target_area) > 1e-8:
        # Use water-filling to project onto simplex
        # We aim for sum * dx = 1, so target_sum = n / 2
        target_sum = len(h) / 2
        # Perform bisection search for lambda
        h_clipped = h.copy()
        low = -np.inf
        high = np.inf
        for _ in range(10):
            mid = (low + high) / 2

            total_test = np.sum(h_test) * dx
            if total_test > target_sum:
                low = mid
            else:
                high = mid
        h = np.clip(h - low, 0, 1)
    return h

def objective_with_params(u_vals, n):
    """Objective function taking parameter values (u_vals) and returning C5."""
    h = build_symmetric_h(u_vals, n)
    h = project_to_feasible(h)
    return evaluate_c5(h)
