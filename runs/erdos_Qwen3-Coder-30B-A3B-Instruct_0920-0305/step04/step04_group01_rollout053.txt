import numpy as np
from scipy.fft import real_fft, real_ifft
from scipy.optimize import minimize
import time

# Global variable for constraint
TARGET_SUM = None

def compute_corr_fft(h):
    """Compute full correlation using FFT."""
    n = len(h)
    L = 2 * n - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = real_ifft(real_fft(h_rev, L) * real_fft(b, L))
    return corr[:L//2 + 1]  # Only positive lags (first half)


def c5_from_corr(corr, n):
    """Convert correlation to C5 value with proper scaling."""
    dx = 2.0 / n
    return dx * np.max(corr)


def evaluate_c5(h):
    """Fast evaluation of C5 using FFT."""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))


def project_to_sum_box(x, target_sum):
    """Project x onto box [0,1]^n and sum constraint."""
    lo, hi = 0.0, 1.0
    s = np.sum(x)
    if s == target_sum:
        return x.copy()

    def f(lam):
        y = np.clip(x + lam, lo, hi)
        return np.sum(y) - target_sum

    # Binary search for lambda
    a, b = -1.0, 1.0
    for _ in range(50):
        lam = (a + b) / 2.0
        val = f(lam)
        if val > 0:
            b = lam
        else:
            a = lam

    y = np.clip(x + (a+b)/2.0, lo, hi)
    return y


def random_pair_move(h):
    """Generate a random pairwise transfer move."""
    n = len(h)
    i = np.random.randint(n)
    j = np.random.randint(n)
    if i == j:
        j = (j + 1) % n
    
    h_i, h_j = h[i], h[j]
    delta_min = max(-h_j, h_i - 1.0)
    delta_max = min(h_i, 1.0 - h_j)
    
    if delta_min < delta_max:
        delta = np.random.uniform(delta_min, delta_max)
        return i, j, delta
    else:
        return None


def guided_move(h, worst_lag):
    """Directed move: transfer mass to reduce overlap at worst lag."""
    n = len(h)
    g = np.zeros(n)
    
    # Compute gradient contribution to each index
    for i in range(n):
        if 0 <= i - worst_lag < n:
            g[i] += h[i - worst_lag]
        if 0 <= i + worst_lag < n:
            g[i] += h[i + worst_lag]
    
    # Find indices with minimal and maximal gradient
    i_min = np.argmin(g)
    i_max = np.argmax(g)
    
    if i_min == i_max:
        return None
    
    h_i, h_j = h[i_min], h[i_max]
    delta_min = max(-h_j, h_i - 1.0)
    delta_max = min(h_i, 1.0 - h_j)
    
    if delta_min < delta_max:
        delta = (delta_min + delta_max) / 2.0
        return i_min, i_max, delta
    else:
        return None


def constraint_eq(h):
    """Equality constraint."""
    return np.sum(h) - TARGET_SUM


def objective(h):
    """Objective function."""
    return evaluate_c5(h)


def anneal(h0, run_seconds, rng):
    """Run simulated annealing with both random and guided moves."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    t0 = time.time()
    T0 = 5e-3
    Tf = 1e-6
    T = T0
    step_count = 0
    
    while time.time() - t0 < run_seconds:
        T = T0 * (1.0 - (time.time() - t0) / run_seconds) + Tf * ((time.time() - t0) / run_seconds)
        
        # Determine move type
        is_guided = (step_count % 10 == 0)
        
        if is_guided:
            # Compute current maximal correlation
            corr = compute_corr_fft(h)
            worst_lag = np.argmax(corr)
            move = guided_move(h, worst_lag)
        else:
            move = random_pair_move(h)
        
        if move is None:
            # Revert to random move
            move = random_pair_move(h)
            if move is not None:
                i, j, delta = move
                h[i] -= delta
                h[j] += delta
        else:
            i, j, delta = move
            h[i] -= delta
            h[j] += delta
        
        # Ensure feasibility
        h = project_to_sum_box(h, TARGET_SUM)
        
        c5_new = evaluate_c5(h)
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
        
        # Accept or reject
        if c5_new < best_c5 or np.exp(-(c5_new - best_c5) / T) > np.random.rand():
            pass  # accept
        else:
            h = best_h.copy()  # revert
            
        step_count += 1
    
    return best_h, best_c5


def local_refine(h):
    """Refine with SLSQP."""
    bounds = [(0, 1) for _ in range(len(h))]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
    except:
        pass
    return h, evaluate_c5(h)


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Setup
    n = 120
    TARGET_SUM = n / 2.0
    dx = 2.0 / n
    
    # Initial construction
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = h0 / np.sum(h0) * TARGET_SUM  # adjust sum
        h0 = project_to_sum_box(h0, TARGET_SUM)
    else:
        # fallback: use uniform
        h0 = np.ones(n) * 0.5
        h0 = project_to_sum_box(h0, TARGET_SUM)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Phase A: Multi-start simulated annealing
    restarts = 0
    remaining = budget_s * 0.7
    
    while remaining > 0 and (time.time() - start_time) < budget_s:
        # Determine when to break
        next_break = min(5.0, remaining)
        if (time.time() - start_time) + next_break > budget_s:
            next_break = budget_s - (time.time() - start_time)
        
        # Initializer method (cyclic)
        phase = restarts % 3
        if phase == 0:
            # Interpolated initial values (if available)
            if 'initial_h_values' in globals():
                h_init = np.array(initial_h_values)
                h_init = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(h_init)), h_init)
                h_init = project_to_sum_box(h_init, TARGET_SUM)
            else:
                h_init = np.ones(n) * 0.5
                h_init = project_to_sum_box(h_init, TARGET_SUM)
        elif phase == 1:
            # Binary pattern
            h_init = np.zeros(n)
            indices = np.random.permutation(n)[:n//2]
            h_init[indices] = 1.0
            h_init = project_to_sum_box(h_init, TARGET_SUM)
        else:  # phase == 2
            # Uniform continuous noise
            h_init = np.random.uniform(0, 1, n)
            h_init = project_to_sum_box(h_init, TARGET_SUM)
        
        h_curr, c5_curr = anneal(h_init, next_break, np.random.RandomState(seed + restarts))
        if c5_curr < best_c5:
            best_h = h_curr.copy()
            best_c5 = c5_curr
        
        remaining -= next_break
        restarts += 1
    
    # Phase B: Local Refinement
    h_ref, c5_ref = local_refine(best_h)
    if c5_ref < best_c5:
        best_h = h_ref
        best_c5 = c5_ref
    
    # Phase C: Final Polish
    for _ in range(5):
        if (time.time() - start_time) > budget_s - 10:
            break
        h_pert = best_h + np.random.normal(0, 0.02, n)
        h_pert = project_to_sum_box(h_pert, TARGET_SUM)
        h_ref, c5_ref = local_refine(h_pert)
        if c5_ref < best_c5:
            best_h = h_ref
            best_c5 = c5_ref
    
    return (best_h.tolist(), float(best_c5), n)
