import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def decode_h(params, n, dx):
    """Decode parameters into a full h vector."""
    # params contains K heights (monotonic decreasing)
    heights = np.sort(params)[::-1]  # Sort descending
    K = len(heights)
    
    # Set up the x positions in [0,1] (half-interval)
    x_half = np.linspace(0, 1, K+1)
    # Create piecewise linear profile on [0,1]
    h_half = np.interp(np.linspace(0, 1, n//2 + 1), x_half, np.concatenate([[1], heights, [0]]))
    # Mirror to get symmetric profile on [0,2]
    h_full = np.concatenate([h_half[:-1], h_half[::-1]])
    # Ensure values in [0,1]
    h_full = np.clip(h_full, 0, 1)
    
    # Scale to meet area constraint
    area = np.sum(h_full) * dx
    if area > 0:
        h_full = h_full / area * 1.0
        h_full = np.clip(h_full, 0, 1)
    
    return h_full

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def smooth_max_surrogate(h, beta=15.0):
    """Compute a smooth surrogate of the max over all shifts."""
    n = len(h)
    dx = 2.0 / n
    # Compute f_k for all k from -(n-1) to n-1
    corr = correlate(h, 1 - h, mode="full")
    # Each element of corr is ∫ h(x) (1 - h(x+k)) dx for shift k
    shift_indices = np.arange(-n+1, n)
    fs = corr * dx
    # Smooth max
    exp_fs = np.exp(beta * fs)
    Z = np.sum(exp_fs)
    if Z <= 0:
        return np.max(fs)
    return np.log(Z) / beta

def compute_grad(h, beta=15.0):
    """Compute the gradient of the smooth max surrogate."""
    n = len(h)
    dx = 2.0 / n
    # Compute f_k values
    corr = correlate(h, 1 - h, mode="full")
    shift_indices = np.arange(-n+1, n)
    fs = corr * dx

    exp_fs = np.exp(beta * fs)
    Z = np.sum(exp_fs)
    if Z <= 0:
        weights = np.ones_like(fs) / len(fs)
    else:
        weights = exp_fs / Z

            # The term is dx * (1 - h_{i+shift} - h_{i-shift}) (where h_{i-shift} out of range is 0)

            for i in range(n - shift):
                grad[i] += w * dx * (-h[i + shift])  # negative sign because we compute d/dh_i(h_i * (1-h_{i+shift}))

            for i in range(shift, n):
                grad[i] += w * dx * (-h[i - shift])
        elif shift < 0:
            # h_i contributes to f_k only if i < -shift
            # derivative components of f_k = dx * (1 - h_{i+shift} - h_{i-shift})
            for i in range(-shift, n):
                grad[i] += w * dx * (-h[i - shift])

                grad[i] += w * dx * (-h[i + shift])

        # Actually, since d/dh_i(h_i * (1-h_{i+k})) = (1 - h_{i+k})
        # We actually have: d/dh_i(Σ_i h_i * (1 - h_{i+k})) = Σ_i (1 - h_{i+k}) d_i^i = (1 - h_{i+k}) (this is wrong!)

        # d/dh_i [ h_i * (1 - h_{i+k})] = 1 - h_{i+k} if i+k valid, 0 otherwise
        # Also it's not wrong to think: derivative of another h_j w.r.t. h_i is 0 or 1.
        # To do it properly, re-express it as correlation
        # e.g., for a fixed shift k, clump the total into the i-th element: 
        #   for shift k,connect h_i to 1-h_{i+k}

        #   ∂f_k/∂h_i = (1 - h_{i+k})*(dx) if i + k in [0,n-1] else 0,

        #    f_k = dx * (sum of h_i * (1 - h_{i+k})) => ∂f_k / ∂h_i = dx * (1 - h_{i+k}) if i + k < n

        # We want full gradient computation over all terms: 
        # For a fixed smaller shift k:
        #   For i, we consider f_k = ∫ h(t) (1 - h(t+k)) dt. d/dh_i(f_k) depend on summand h_i(1 - h_{i+k}).
        #  这样的话，我们不直接操作，而是考虑每一个smoothiation term，注意和index匹配

        # For shift k (shifted start index) -> matcher

        #   - i >=0  for shift k positive (i+k in right range)

        #      左边移位逻辑: j = i-k (h_j) where j allows shifting to the left

        #         j = i + shift = i + 6, valid for j=0 ~ n-1, so i = -6 ~ n-7, now fix to labels: i >= 0, so i >= max(0,-6) = 0

            #     So the contribution to f_k from i is 1 - h_{i+k}

def project_onto_feasible(h, dx):
    """Project h onto [0,1] and ensure sum = 1."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Renormalize to sum = 1
    area = np.sum(h) * dx
    if area > 0:
        h = h / area * 1.0
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 200
    dx = 2.0 / n_points
    k = 5  # number of breakpoints (4 height values to optimize, plus 1 = h_0, 0 = h_n)
    params_lb = [0.0] * k
    params_ub = [1.0] * k

    def objective(params):
        h = decode_h(params, n_points, dx)
        c5 = evaluate_c5(h)
        return c5

    # DE stage
    t_de = 0.70 * budget_s
    de_bounds = list(zip(params_lb, params_ub))
    de_result = differential_evolution(
        objective,
        de_bounds,
        maxiter=int(100 + 1000 * (t_de / budget_s)),
        popsize=15,
        seed=seed,
        disp=False,
        tol=1e-6
    )

    best_params = de_result.x
    best_h = decode_h(best_params, n_points, dx)
    best_c5 = evaluate_c5(best_h)
    
    # Gradient refinement stage
    t_grad = 0.20 * budget_s
    t_current = time.time()
    if t_current - start_time < t_grad + 1:
        # Run line search or projected gradient descent with smooth max
        current_h = best_h.copy()
        current_c5 = best_c5
        beta = 15.0
        step_size = 0.3
        max_iter = 500
        for i in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
            grad = compute_grad(current_h, beta)
            # Update
            new_h = current_h - step_size * grad
            # Project back
            new_h = project_onto_feasible(new_h, dx)
            new_c5 = evaluate_c5(new_h)
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
            else:
                step_size *= 0.9  # Reduce step size if not helping
        best_h = current_h
        best_c5 = current_c5

    # Final polishing
    if time.time() - start_time < budget_s - 1:
        def constraint_eq(h):
            return np.sum(h) * dx - 1.0

        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}

        # Run SLSQP with final polish
        try:
            res = minimize(
                evaluate_c5,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10}
            )
            if res.success:
                best_h = res.x
                best_c5 = evaluate_c5(best_h)
        except Exception:
            pass

    return (best_h.tolist(), best_c5, n_points)
