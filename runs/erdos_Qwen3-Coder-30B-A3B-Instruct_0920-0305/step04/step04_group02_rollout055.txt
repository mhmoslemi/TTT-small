import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient(h, d_star):
    """Compute the subgradient of the overlap for shift d_star"""
    n = len(h)
    g = np.zeros(n)
    if d_star > 0:
        # h_i - h_{i+d_star} for i in 0..n-d_star-1
        g[:n-d_star] = 1 - h[d_star:]
    elif d_star < 0:
        # h_i - h_{i+d_star} for i in -d_star..n-1
        g[-d_star:] = 1 - h[:n+d_star]
    else:
        # d_star == 0: no shift, gradient should be 1 (but we ignore this anyway due to asymmetry)
        raise ValueError("d_star cannot be 0 in this function")
    return g

def binary_optimal(n):
    """Solve the exact binary optimization problem using CVXPY"""
    dx = 2.0 / n
    
    # Variables
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    
    # Constraints
    constraints = [
        cp.sum(x) == n // 2,  # Mass constraint (n=42, so 21)
        t >= 0,
    ]
    
    # Add symmetry breaking (x[0] = 1)
    constraints.append(x[0] == 1)
    
    # Add product constraints for all d
    # Create a mapping of shift indices to the corresponding variables
    for d in range(-n+1, n) if d != 0 else []:
        # Create auxiliary variables for the product x[i]*x[i+d]
        w = []
        if d > 0:
            # indices where both x[i] and x[i+d] are valid
            for i in range(n-d):
                # keep track of i and i+d for reconstruction
                pass
        elif d < 0:
            for i in range(-d, n):
                # keep track of i and i+d for reconstruction
                pass
        else:
            continue  # skip d=0
        
        w_loop = []
        for i in range(max(0, -d), min(n, n-d)):
            wi = cp.Variable(boolean=True)
            w_loop.append(wi)
            # McCormick constraints
            constraints.append(wi <= x[i])
            constraints.append(wi <= x[i+d])
            constraints.append(wi >= x[i] + x[i+d] - 1)
        
        # Now compute number of 1,0 pairs for this shift:
        # number of 1 in x - number of 1*1 in w = sum(x) - sum(w)
        # => sum(x) - sum(w) <= t
        if len(w_loop) > 0:
            constraints.append(cp.sum(x) - cp.sum(w_loop) <= t)
    
    # Objective to minimize t
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Try solvers in order
    solver_list = [cp.CBC, cp.GLPK_MI, cp.ECOS_BB]
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                break
        except:
            continue
    else:
        # Give up if no solver works
        return None, None

    h_opt = x.value.astype(float)
    t_opt = t.value
    
    return h_opt, t_opt

def refine_continuous(h0, remaining_time):
    """Perform continuous refinement based on sub-gradient descent."""
    n = len(h0)
    dx = 2.0 / n
    best = h0.copy()
    best_val = evaluate_c5(best)
    
    # Safeguard that time doesn't exceed
    start = time.time()
    
    iterations = 0
    while time.time() - start < remaining_time and iterations < 2000:
        # Compute all overlaps
        corr = np.correlate(best, 1 - best, mode="full")
        max_val = np.max(corr)
        # Find all shifts giving the maximum value
        D_max = np.where(corr == max_val)[0]
        # Adjust shift indices to proper relative indices
        D_max = [d - (len(corr) - 1)//2 for d in D_max if d != (len(corr) - 1)//2]
        
        # Compute the average gradient over this set
        g_avg = np.zeros(n)
        for d in D_max:
            # Convert to actual shift range indices
            g = subgradient(best, d)
            g_avg += g
            
        # Average of gradients
        if len(D_max) > 0:
            g_avg /= len(D_max)

        # Simple line search: try decreasing step size if no progress
        alpha = 0.5
        while alpha > 1e-8:
            h_new = best - alpha * g_avg
            # Clip to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Force the integral to be 1 by scaling
            scale = (n/2) / np.sum(h_new)
            h_new = h_new * scale
            
            new_val = evaluate_c5(h_new)
            if new_val < best_val - 1e-12:
                best = h_new
                best_val = new_val
                break
            else:
                alpha /= 2
            
        if alpha <= 1e-8 or np.linalg.norm(g_avg) < 1e-10:
            # Break or decrease step size variation
            pass
        iterations += 1
        
        # Occasional random perturbation to escape local optima
        if np.random.rand() < 0.05:
            noise = np.random.normal(0, 0.01, n)
            h_new = best + noise
            h_new = np.clip(h_new, 0, 1)
            scale = (n/2) / np.sum(h_new)
            h_new = h_new * scale
            new_val = evaluate_c5(h_new)
            if new_val < best_val - 1e-12:
                best = h_new
                best_val = new_val
                
    # Final polish with SLSQP
    try:
        res = minimize(
            evaluate_c5, 
            best, 
            method='SLSQP', 
            bounds=[(0, 1) for _ in range(n)],
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x)*dx - 1.0},
            options={'maxiter': 50}
        )
        if res.success:
            h_final = res.x
            scale = (n/2) / np.sum(h_final)
            h_final = h_final * scale
            val_final = evaluate_c5(h_final)
            if val_final < best_val:
                best = h_final
                best_val = val_final
    except:
        pass
        
    return best, best_val
