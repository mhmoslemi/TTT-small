import numpy as np

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def interpolate_symmetric_heights(u_vals, x_points, n):
    """
    Interpolate breakpoints (x_j, h_j) and replicate symmetrically over [0,2]
    u_vals is sorted descending, x_points = [0, x1, x2, ..., 1]
    """
    x = np.linspace(0, 1, n//2 + 1)
    heights = np.interp(x, x_points, u_vals)
    h = np.concatenate([heights, heights[-2::-1]])  # mirror without repeating the center
    return h

def project_onto_feasible(h, target_sum, dx):
    """Project h onto the set of vectors with sum = target_sum and 0 ≤ h_i ≤ 1"""
    h = np.array(h)
    while True:
        # Water-filling to get sum=target_sum, 0 ≤ h_i ≤ 1
        p = np.clip(h, 0, 1)
        current_sum = np.sum(p) * dx
        if abs(current_sum - target_sum) < 1e-10:
            return p
        elif current_sum > target_sum:
            # Need to reduce values
            # Binary search for lambda
            l, u = 0, 1
            for _ in range(50):
                lam = (l + u) / 2
                p_trial = np.clip(h - lam, 0, 1)
                if np.sum(p_trial) * dx > target_sum:
                    l = lam
                else:
                    u = lam
            h = p_trial
        else:  # sum < target_sum
            l, u = -1, 0
            for _ in range(50):
                lam = (l + u) / 2
                p_trial = np.clip(h - lam, 0, 1)
                if np.sum(p_trial) * dx < target_sum:
                    u = lam
                else:
                    l = lam
            h = p_trial

def smooth_max_fcn(h_vals, x_points, n, beta=20.0):
    """Evaluate smooth max of correlations using log-sum-exp for better numerical properties"""
    dx = 2.0 / n
    h = interpolate_symmetric_heights(h_vals, x_points, n)
    corr = correlate(h, 1 - h, mode="full")
    # Shift to avoid large exponentials
    max_shift = len(corr) - n
    corr_shifted = corr[max_shift: max_shift + n]  # Just the proper k values
    f_vals = corr_shifted * dx  # Normalized correlations
    max_f = np.max(f_vals)
    exp_vals = np.exp(beta * (f_vals - max_f))
    Z = np.sum(exp_vals)
    return (max_f + np.log(Z) / beta) * dx

def smooth_max_grad(h_vals, x_points, n, beta=20.0):
    """Gradient of the smooth max with respect to h_vals (approximate)"""
    dx = 2.0 / n
    h = interpolate_symmetric_heights(h_vals, x_points, n)
    corr = correlate(h, 1 - h, mode="full")
    max_shift = len(corr) - n
    f_vals = corr[max_shift: max_shift + n] * dx
    max_f = np.max(f_vals)
    exp_vals = np.exp(beta * (f_vals - max_f))
    Z = np.sum(exp_vals)
    w = exp_vals / Z

    # Gradient of sum w_k * f_k with respect to h_i
    # This is a bit tricky; iterate through each h_i and compute its effect
    # We compute the gradient via finite differences of the function that we are differentiating
    grad = np.zeros_like(h_vals)
    eps = 1e-4
    for i in range(len(h_vals)):
        # We're differentiating f_k(h) = sum h_j (1 - h_{j+k}) over shifts
        # Keep track of only part that is in range for now
        h_pert = h_vals.copy()
        h_pert[i] += eps
        h_pert = np.sort(h_pert)[::-1]  # re-sort descending, account for function invariance

        h_new = interpolate_symmetric_heights(h_pert, x_points, n)
        corr_new = correlate(h_new, 1 - h_new, mode="full")
        corr_new = corr_new[max_shift: max_shift + n] * dx
        g = np.sum(w * (corr_new - f_vals)) / eps
        grad[i] = g
    return grad
