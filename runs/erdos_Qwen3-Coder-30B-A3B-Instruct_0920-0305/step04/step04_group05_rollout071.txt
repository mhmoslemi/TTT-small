def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(x, target_sum):
    """Project x onto feasible set: [0,1]^n with sum = target_sum"""
    y = np.clip(x, 0, 1)
    current_sum = np.sum(y)
    
    if abs(current_sum - target_sum) < 1e-9:
        return y

    for _ in range(100):

            y[:] = target_sum / len(y)

        y = np.clip(y, 0, 1)
        current_sum = np.sum(y)
        if abs(current_sum - target_sum) < 1e-9:
            break
    
    return y
