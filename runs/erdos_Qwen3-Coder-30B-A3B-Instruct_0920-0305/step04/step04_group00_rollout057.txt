import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Setup
    n_points = 200
    dx = 2.0 / n_points
    k_breakpoints = 5  # Number of breakpoints in [0,1] (symmetric about 1)
    
    # 1. DE Optimizer: parameterized by heights u_i, sorted decreasingly
    def decode_h(u):
        # u contains k_breakpoints height parameters, unsorted
        h = np.zeros(n_points)
        # Build piecewise linear in [0,1], then mirror to [1,2]
        # Create breakpoints in [0,1]
        breakpoints = np.linspace(0, 1, k_breakpoints+1)
        h_vals = np.sort(u)[::-1]   # decrease order
        # Linear segments in [0,1]
        for i in range(k_breakpoints):
            x0, x1 = breakpoints[i], breakpoints[i+1]
            h0, h1 = h_vals[i], h_vals[i+1]
            idx = np.where((x0 <= np.linspace(0, 1, n_points//2)) & (np.linspace(0, 1, n_points//2) < x1))[0]
            if len(idx) > 0:
                h[idx] = h0 + (h1-h0)/(x1-x0) * (np.linspace(0, 1, n_points//2)[idx] - x0)
        # Mirror for [1,2]
        h[n_points//2:] = h[:n_points//2][::-1]
        return h
    
    def decode_h_with_area_norm(u):
        h = decode_h(u)
        # Normalize
        area = np.sum(h) * dx
        if area > 0:
            h = h / area * 1.0
        # Ensure max is 1
        h = np.clip(h, 0, 1)
        return h
    
    def objective(u):
        h = decode_h_with_area_norm(u)
        corr = correlate(h, 1 - h, mode='full')
        # Max overlap term
        return -np.max(corr) * dx  # Minimize negative for consistency with DE
    
    # 2. DE parameters
    bounds = [(0, 1) for _ in range(k_breakpoints)]
    t_de = 0.7 * budget_s
    result_de = differential_evolution(
        objective,
        bounds,
        seed=seed,
        maxiter=int((t_de / 0.7 / 200) * 30),  # Rough estimation
        popsize=15,
        disp=False
    )
    best_u = result_de.x
    best_h = decode_h_with_area_norm(best_u)
    best_c5 = -objective(best_u)
    
    # 3. Gradient refinement
    # Smoothed max function for differentiability
    beta = 20.0
    
    def smooth_max(f_vals, beta):
        return (1./beta) * np.log(np.sum(np.exp(beta * f_vals)))
    
    # Fully refined with gradient-based optimization
    t_grad = 0.2 * budget_s
    max_iter = min(500, int(t_grad / 0.01))
    
    # Use enhanced gradient descent
    h_refined = best_h.copy()
    eta = 0.001
    decay_rate = 50
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlations with 1-h
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        f_vals = corr * dx
        
        # Smoothed approximation gradient
        exp_fb = np.exp(beta * f_vals)
        w = exp_fb / np.sum(exp_fb)
        
        # Compute ∇f_m = dx*(1 - h_{i+k} - h_{i-k})
        grad = np.zeros_like(h_refined)
        for k_idx in range(len(f_vals)):
            k = k_idx - (len(f_vals) - 1) // 2
            if k == 0:
                continue
            fwd_start = max(0, -k)
            bkwd_start = max(0, k)
            if k > 0:
                grad[fwd_start:] += w[k_idx] * dx * (-h_refined[bkwd_start:])
            elif k < 0:
                grad[bkwd_start:] += w[k_idx] * dx * (-h_refined[fwd_start:])
            # And the 1 - h_{i+k} terms:
            grad += w[k_idx] * dx
        
        # Update
        h_new = h_refined - eta * grad
        
        # Project onto feasible set
        # i.e., [0,1] and ∫ h dx = 1
        h_new = np.clip(h_new, 0, 1)
        area = np.sum(h_new) * dx
        h_new = h_new / area * 1.0
        
        # Ensure sum again (clip more if needed)
        h_new = np.clip(h_new, 0, 1)
        area = np.sum(h_new) * dx
        h_new = h_new / area * 1.0  # This is the constraint: sum h[i] * dx = 1
        
        # Convergence check or update
        h_refined = h_new
        eta = eta * 0.995
    
    # Evaluate final
    final_corr = correlate(h_refined, 1 - h_refined, mode="full")
    final_c5 = np.max(final_corr) * dx
    
    return (h_refined.tolist(), final_c5, n_points)
