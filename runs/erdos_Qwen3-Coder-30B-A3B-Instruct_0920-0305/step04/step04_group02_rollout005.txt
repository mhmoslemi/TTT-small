import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient(h, d_star):
    """
    Compute the subgradient vector for the overlap with shift d_star.
    This is the gradient of sum_i h_i - sum_i h_i * h_{i+d_star}
    """
    n = len(h)
    g = np.ones(n)
    if d_star > 0:
        g[d_star:] -= h[:-d_star]
        g[:d_star] = 0
    elif d_star < 0:
        g[:d_star] -= h[-d_star:]
        g[d_star:] = 0
    else:
        g = np.zeros(n)
    return g

def binary_optimal(n):
    """
    Solve the MILP problem of minimizing the maximum overlap using integer variables.
    Returns the binary solution as a continuous vector and the optimal integer bound.
    """
    # Create binary x variables
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    
    # Define constraints
    constraints = [
        cp.sum(x) == n // 2,
        x[0] == 1,  # symmetry breaker
        t >= 0
    ]

    # Add product constraints and overlap bounds for each shift
    shifts = list(range(-n+1, 0)) + list(range(1, n))
    for d in shifts:
        # Create auxiliary variables and constraints
        d_abs = abs(d)
        if d < 0:
            w = cp.Variable(n + d, boolean=True)
            for i in range(n + d):
                constraints += [
                    w[i] <= x[i],
                    w[i] <= x[i - d],
                    w[i] >= x[i] + x[i - d] - 1
                ]
            # Compute overlap for shift d
            N_d = cp.sum(x) - cp.sum(w)
        else:
            w = cp.Variable(n - d, boolean=True)
            for i in range(n - d):
                constraints += [
                    w[i] <= x[i],
                    w[i] <= x[i + d],
                    w[i] >= x[i] + x[i + d] - 1
                ]
            N_d = cp.sum(x) - cp.sum(w)
        constraints.append(N_d <= t)
    
    # Set up and solve the problem
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CBC)
    except:
        try:
            prob.solve(solver=cp.GLPK_MI)
        except:
            try:
                prob.solve(solver=cp.ECOS_BB)
            except:
                return None, None

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        x_opt = x.value
        t_opt = t.value
        return np.array(x_opt, dtype=float), t_opt
    return None, None

def refine_continuous(h0, time_left):
    """
    Refine a continuous solution using projected subgradient descent.
    """
    best_h = h0.copy()
    best_val = evaluate_c5(best_h)
    n = len(h0)
    dx = 2.0 / n
    remaining = time_left
    start = time.time()
    restarts = 0
    max_restarts = 20

    while restarts < max_restarts and (time.time() - start) < time_left * 0.8:
        h = h0.copy()
        # Break any exact symmetry by slightly varying if needed
        if restarts > 0:
            h += np.random.normal(0, 0.001, n)
            h = np.clip(h, 0, 1)

        for _ in range(100):
            if time.time() - start > time_left * 0.9:
                break

            # Compute overlaps for all shifts
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            # Find all shifts that achieve maximum
            d_max_indices = np.where(corr == max_corr)[0]
            # Adjust to actual shift indices
            d_vals = d_max_indices - (n - 1)

            # Compute average gradient for worst shifts
            avg_grad = np.zeros(n)
            for d_val in d_vals:
                g = subgradient(h, d_val)
                avg_grad += g
            avg_grad /= len(d_vals)

            # Gradient step
            alpha = 0.5
            h_new = h - alpha * avg_grad
            h_new = np.clip(h_new, 0, 1)
            
            # Force sum constraint
            h_new = h_new * (n / 2) / np.sum(h_new)
            
            # Evaluate new point
            new_val = evaluate_c5(h_new)
            if new_val < best_val - 1e-9:
                best_val = new_val
                best_h = h_new.copy()

            # Occasionally random perturbation
            if np.random.random() < 0.2:
                h_new += np.random.normal(0, 0.002, n)
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n / 2) / np.sum(h_new)

            h = h_new

        restarts += 1

    # Final SLSQP polish
    try:
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}
        bounds = [(0, 1) for _ in range(n)]
        res = minimize(evaluate_c5, best_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 50, 'ftol': 1e-12, 'gtol': 1e-12})
        if res.success:
            final_val = evaluate_c5(res.x)
            if final_val < best_val:
                best_h = res.x
                best_val = final_val
    except:
        pass

    return best_h, best_val
