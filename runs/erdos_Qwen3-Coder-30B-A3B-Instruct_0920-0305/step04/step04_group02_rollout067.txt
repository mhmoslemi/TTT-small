import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    # Phase 1: Binary optimization using MILP
    try:
        # Variables: x (binary, length n), y (binary, for each overlapping pair), t (integer)
        x = cp.Variable(n_points, boolean=True)
        t = cp.Variable(integer=True)
        y = {} # Dictionary for y_{i,d}
        
        # Add constraints for every valid pair in each shift
        for d in range(-n_points+1, n_points):
            if d == 0:
                continue
            for i in range(n_points):
                if i + d < 0 or i + d >= n_points:
                    continue
                y[(i,d)] = cp.Variable(boolean=True)
        
        # Mass constraint
        constraints = [cp.sum(x) == n_points // 2]
        
        # McCormick constraints for y_{i,d} = x_i * x_{i+d}
        for d in range(-n_points+1, n_points):
            if d == 0:
                continue
            for i in range(n_points):
                if i + d < 0 or i + d >= n_points:
                    continue
                y[(i,d)] = cp.Variable(boolean=True)
                constraints += [
                    y[(i,d)] <= x[i],
                    y[(i,d)] <= x[i+d],
                    y[(i,d)] >= x[i] + x[i+d] - 1
                ]
        
        # Overlap bound: N_d(x) = sum x_i - sum y_{i,d} <= t
        for d in range(-n_points+1, n_points):
            if d == 0:
                continue
            N_d = cp.sum(x) - cp.sum([y[(i,d)] for i in range(n_points) if 0 <= i+d < n_points])
            constraints += [N_d <= t]
            if d < 0:
                continue
            N_d_neg = cp.sum(x) - cp.sum([y[(i+d,-d)] for i in range(n_points) if 0 <= i < n_points])
            constraints += [N_d_neg <= t]

        # Set objective to minimize t
        objective = cp.Minimize(t)
        
        # Create and solve the problem
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, verbose=False)
        
        # Check for successful solution
        if prob.status == cp.OPTIMAL:
            # Extract binary solution
            x_val = x.value.astype(int)
            x_recovered = x_val.astype(float)
            # Normalize to satisfy integral constraint: get x_i such that sum x_i * dx = 1
            total = np.sum(x_recovered) * dx
            scale = 1.0 / total
            x_recovered = x_recovered * scale
            mc5 = evaluate_c5(x_recovered)
            
            if mc5 < best_c5:
                best_h = x_recovered
                best_c5 = mc5
    except Exception as e:
        # Fallback to nothing or continue with no binary optimization
        pass
    
    # Phase 2: Continuous refinement using smooth max and gradient descent
    # Define the smooth max surrogate function
    def smooth_max(h, tau=0.1):
        n = len(h)
        shifts = list(range(-n+1, 0)) + list(range(1, n))
        C_d_vals = []
        for d in shifts:
            C_d = 0.0
            if d > 0:
                for i in range(n - d):
                    C_d += h[i] - h[i]*h[i+d]
            else:
                d = -d
                for i in range(n - d):
                    C_d += h[i+d] - h[i+d]*h[i]
            C_d_vals.append(C_d)
        C_d_vals = np.array(C_d_vals)
        return tau * np.log(np.sum(np.exp(C_d_vals/tau)))
    
    def smooth_max_grad(h, tau=0.1):
        n = len(h)
        shifts = list(range(-n+1, 0)) + list(range(1, n))
        weights = []
        grad = np.zeros(n)
        C_d_vals = []
        for d in shifts:
            C_d = 0.0
            if d > 0:
                for i in range(n - d):
                    C_d += h[i] - h[i]*h[i+d]
            else:
                d = -d
                for i in range(n - d):
                    C_d += h[i+d] - h[i+d]*h[i]
            C_d_vals.append(C_d)
        C_d_vals = np.array(C_d_vals)
        exp_vals = np.exp(C_d_vals/tau)
        total_exp = np.sum(exp_vals)
        weights = exp_vals / total_exp
        
        for i, (d, weight) in enumerate(zip(shifts, weights)):
            grad_component = 0.0
            if d > 0:
                for j in range(n - d):
                    dj = j + d
                    grad_component += weight * (1.0 - h[dj])
            else:
                d = -d
                for j in range(n - d):
                    dj = j + d
                    grad_component += weight * (1.0 - h[dj])
            grad += grad_component
        return grad
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h) * dx
        scale = 1.0 / total
        h_scaled = h * scale * (n_points / 2)
        return h_scaled

    # Sequential smoothing continuation
    def smooth_descent(h_initial, max_iter=200):
        h_current = h_initial.copy()
        h_current = project_h(h_current)
        tau = 0.5
        max_tau_reductions = 5
        for _ in range(max_tau_reductions):
            for _ in range(max_iter):
                if (time.time() - start_time) > budget_s * 0.95:
                    return h_current
                grad = smooth_max_grad(h_current, tau)
                h_current = h_current - 0.01 * grad
                h_current = project_h(h_current)
                # Early stop if τ is small
                if tau < 1e-4:
                    break
            tau = tau / 2.0
        return h_current

    # Use smooth refinement with continuation
    try:
        h_refined = smooth_descent(best_h.copy())
        c5_refined = evaluate_c5(h_refined)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    except Exception as e:
        pass

    # Final polishing with SLSQP
    try:
        def obj(h):
            return evaluate_c5(h)

        def constraint_eq(h):
            return np.sum(h) * dx - 1.0

        cons = {'type': 'eq', 'fun': constraint_eq}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12})
        if res.success:
            h_final = res.x
            scale = 1.0 / (np.sum(h_final) * dx)
            h_final = h_final * scale
            c5_final = evaluate_c5(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception as e:
        pass

    return (best_h.tolist(), best_c5, n_points)
