import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(x, n_points):
    """Project a vector to feasible region: [0,1] and sum = n_points/2"""
    y = np.clip(x, 0, 1)
    target = n_points / 2.0
    current_sum = np.sum(y)
    
    if abs(current_sum - target) < 1e-9:
        return y
    
    # Simple iterative approach
    for _ in range(100):
        diff = current_sum - target
        if abs(diff) < 1e-6:
            break
            
        if diff > 0:  # sum too high, need reduce
            free = y > 0
            if not free.any():
                break
            reduction = diff / np.sum(free)
            y[free] = np.maximum(0, y[free] - reduction)
        else:  # sum too low, need increase
            free = y < 1
            if not free.any():
                break
            increment = -diff / np.sum(free)
            y[free] = np.minimum(1, y[free] + increment)
            
        current_sum = np.sum(y)
    
    return y
