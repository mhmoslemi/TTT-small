def smooth_max(h, beta=20):
    """Compute smooth approximation of max over all shifts"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full") * dx
    # Numerically stable log-sum-exp
    max_f = np.max(corr)
    exp_f = np.exp(beta * (corr - max_f))
    return max_f + np.log(np.sum(exp_f)) / beta

def project_to_feasible(h, target_sum):
    """Project h onto the simplex and box constraints using water-filling"""
    h = np.array(h)
    # Ensure it's within [0,1]
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if abs(total - target_sum) < 1e-10:
        return h
        
    # Water-filling algorithm
    def water_fill(lambda_val):
        p = np.clip(h - lambda_val, 0, 1)
        return np.sum(p) - target_sum
    
    # Binary search for the right lambda
    a, b = -1.0, 1.0
    tol = 1e-12
    for _ in range(50):
        lam = (a + b) / 2
        val = water_fill(lam)
        if val > 0:
            a = lam
        else:
            b = lam
        if abs(val) < tol:
            break
    p = np.clip(h - (a+b)/2, 0, 1)
    return p

def piecewise_linear_symmetric(K, heights, n):
    """Construct piecewise linear symmetric function"""
    x = np.linspace(0, 1, K+1)

    heights = np.sort(heights)[::-1]
    y = np.array(heights)
    # Linear interpolation on [0,1], then mirrored to [1,2]
    x_full = np.linspace(0, 2, n)
    # Left part [0,1)
    indices = np.searchsorted(x, x_full[x_full <= 1], side='right') - 1
    indices = np.maximum(indices, 0)
    indices = np.minimum(indices, K-1)

    # Right part [1,2]

    y_right = np.interp(x_right, x, y)
    # Full vector
    y_full = np.concatenate([y_left, y_right])
    return y_full

    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx
