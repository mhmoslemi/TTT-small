import numpy as np
import time
from scipy.optimize import minimize

def compute_corr_fft(h):
    n = len(h)
    h_rev = h[::-1]
    b = 1.0 - h
    L = 2 * n - 1
    hf = np.fft.rfft(h_rev, n=L)
    bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(hf * bf, n=L)
    return conv

def evaluate_c5_fft(h):
    conv = compute_corr_fft(h)
    max_corr = np.max(conv)
    return max_corr * (2.0 / len(h))

def project_to_sum_box(x, target_sum):
    # Project vector x onto the box [0,1]^n and sum = target_sum
    # Use binary search for lambda parameter that satisfies sum.
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(x + mid, 0, 1)
        s = proj.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) / 2.0, 0, 1)

def simulated_annealing(h_initial, target_sum, time_limit, seed):
    np.random.seed(seed)
    n = len(h_initial)
    h = np.copy(h_initial)
    h = project_to_sum_box(h, target_sum)
    h = np.clip(h, 0, 1)
    cur_c = evaluate_c5_fft(h)
    best_h = h.copy()
    best_c = cur_c

    # Set parameters
    T0 = cur_c * 0.01  # Start temperature
    Tmin = 1e-8  # Minimum temperature
    start = time.time()
    
    iter = 0
    while time.time() - start < time_limit:
        # Random move
        i, j = np.random.choice(n, size=2, replace=False)
        
        # Compute delta bounds
        delta_max = min(h[i], 1 - h[j])
        delta_min = -min(1 - h[i], h[j])
        
        if delta_max <= delta_min:
            continue
            
        # Choose delta uniformly in [delta_min, delta_max]
        delta = np.random.uniform(delta_min, delta_max)
        
        # Update h
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # evaluate
        c_new = evaluate_c5_fft(h_new)
        
        # Compute temperature based on proportion of elapsed time
        elapsed = time.time() - start
        progress = elapsed / time_limit
        T = T0 * (1 - progress) + Tmin * progress
        
        # Acceptance probability
        if c_new < cur_c or np.random.rand() < np.exp(-(c_new - cur_c) / (T + 1e-12)):
            h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        iter += 1
        
    return best_h, best_c
