import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Build a more sophisticated initial solution
    # Use block pattern and binary optimization strategy
    n_points = 42
    
    # Try to construct a solution based on overlapping blocks
    h0 = np.zeros(n_points)
    
    # Distribute ones in a pattern to avoid large gaps
    # Use an objective and gradient-based search approach
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Refined optimization approach: combined global and local search
    best_h = None
    best_c5 = float('inf')
    
    # Phase 1: Try different local solutions from various seeds
    n_seeds = 10
    for s in range(n_seeds):
        np.random.seed(s)
        # Start with random feasible solution
        h = np.random.random(n_points)
        dx = 2.0 / n_points
        scale = 1.0 / (np.sum(h) * dx)
        h = h * scale
        try:
            # Local optimization
            res = minimize(
                objective, 
                h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue
    
    # Phase 2: Use Diffevolution with custom bounds and cutting edge strategies
    if (time.time() - start_time) < budget_s * 0.8:
        strategies = ['best1bin', 'best2bin', 'rand1bin', 'rand2bin', 'currenttobestbin']
        
        for strategy in strategies:
            if (time.time() - start_time) > budget_s:
                break
            try:
                de_res = differential_evolution(
                    objective,
                    bounds,
                    seed=seed + 100,
                    maxiter=500,
                    popsize=15,
                    mutation=(0.8, 1),
                    recombination=0.9,
                    strategy=strategy
                )
                if de_res.success:
                    h_candidate = de_res.x
                    # Normalize to satisfy constraint
                    dx = 2.0 / n_points
                    scale = 1.0 / (np.sum(h_candidate) * dx)
                    h_candidate = h_candidate * scale
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
            except:
                continue
    
    # Phase 3: Push towards better solutions with simulated annealing
    if (time.time() - start_time) < budget_s * 0.9:
        # Start from the best known solution
        if best_h is None:
            h_start = np.ones(n_points) * 0.5
        else:
            h_start = best_h.copy()
        
        # Simulated Annealing approach
        current_h = h_start.copy()
        current_c5 = evaluate_c5(current_h)
        best_h = current_h.copy()
        best_c5 = current_c5
        
        # Temperature schedule
        T = 0.1
        T_min = 1e-5
        alpha = 0.95
        max_iter = 10000
        
        for i in range(max_iter):
            if (time.time() - start_time) > budget_s:
                break
            
            # Make a small perturbation
            new_h = current_h.copy()
            
            # Choose two random indices to swap their values
            idx1, idx2 = np.random.choice(n_points, 2, replace=False)
            new_h[idx1], new_h[idx2] = new_h[idx2], new_h[idx1]
            
            # Adjust to maintain integral constraint
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(new_h) * dx)
            new_h = new_h * scale

            new_c5 = objective(new_h)
            
            # Accept or reject
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
            else:
                # Probabilistically accept based on temperature
                delta = new_c5 - current_c5
                if np.random.rand() < np.exp(-delta / T):
                    current_h = new_h
                    current_c5 = new_c5
            
            T = max(T_min, T * alpha)
    
    # Final local refine
    if (time.time() - start_time) < budget_s:
        try:
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-13, 'gtol': 1e-13}
            )
            if res.success:
                h_candidate = res.x
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
    
    # Clip the hull to [0,1] and normalize
    best_h = np.clip(best_h, 0, 1)
    dx = 2.0 / n_points
    scale = 1.0 / (np.sum(best_h) * dx)
    best_h = best_h * scale
    
    return (best_h.tolist(), best_c5, n_points)
