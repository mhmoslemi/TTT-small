import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, Boolean, Integral, sum as cp_sum
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def solve_binary_milp(n):
    """Solve the binary version as an MILP to get lower bound"""
    x = Variable(n, boolean=True)
    t = Variable(integer=True)
    
    # Need to handle product x_i * x_{i+d} in the overlap function
    # These constraints are non-linear, but we linearize them
        
    # For each possible shift d from -n+1 to n-1, we compute the sum
    # of indices (i, i+d) such that the product x_i * x_{i+d} should appear
    # in overlap counts. If d = 0, it would be x_i * x_i = x_i,
    # but in our formula, we take h_i*(1-h_{i+d}), so this is not a problem
    # But diagonal elements can be skipped in the overlap computation.
    
    # Break down shift indices to make constraints easier
    # For i such that 0 <= i+d < n, we introduce y_{i,d}
    
    start_time = time.time()
    constraints = [cp_sum(x) == n // 2]
    
    # Add a dummy upper-bound variable t that we want to minimize
    # and we want all overlap terms (sums over i) to be <= t  
    # Overlap for shift d = sum_i h_i*(1-h_{i+d}) = sum_i h_i - sum_i h_i*h_{i+d} 
    
    # We still use offsets to compute these sums; we compute each for a fixed shift d
    
    # First try a small example and bake up the computed variables.
    # Let's instanciate overlaps from each d
    total_shifts = 2*n - 1  # d from -(n-1) to + (n-1) inclusive
    y_vars = {}
    max_d_values = []
    shift_map = {}
    
    # This file assumes 'initial_h_values' is global
    # and we want to calculate overlaps full-size
    
    for i, d in enumerate(range(-(n - 1), n)):
        shift_map[i] = d
        if d == 0:
            continue
            
        # Create constraint for this shift
        constraints.append(0 <= x[0:n - abs(d)] + x[abs(d):n] - 2 * y_vars.get((d, 'active'), 0))
        # Need square of overlap elements to be bounded by t
        # Try to construct all y_{i,d} for valid indices
        # Instead: Model based on difference sum
        overlap_d = 0
        for idx in range(n - abs(d)):
            overlap_d += x[idx] * x[idx + d if d > 0 else idx - d]
        # Now we will make MIP equivalent by making y variables and linearizing the product
        # time to build:
        y_vars[(d, 'i')] = [Variable(1, boolean=True) for _ in range(n - abs(d))]
        for j in range(n - abs(d)):
            y = y_vars[(d, 'i')][j]
            
            constraints.append(y <= x[j])
            constraints.append(y <= x[j + d if d > 0 else j - d])
            constraints.append(y >= x[j] + x[j + d if d > 0 else j - d] - 1)
            overlap_d += y  # will do next iteration
    
        # Avoid double counting sum terms
        # Just use a tight bound on the maximum shift value directly
        max_d_values.append(overlap_d)  # Potential constraint
    
    # Now make 1-t constraint for max. For any valid overlap expression,
    # Now refine NumPy version:
    # Manually enforce each shift constraint
    for d in range(-(n-1), n):
        if d == 0: continue
        # Ensure shift stops at proper end. Let's compute and create linearization placeholders:
        valid_shift_indices = []
        for i in range(n):
            if 0 <= i+d < n:
                valid_shift_indices.append(i)
        
        # Sum of x[i] * x[i+d] over all valid i for shift d
        # Linearized as sum of boolean y_{i,d} s.t. y_{i,d} = x[i] * x[i+d]
        y_shift = [Variable(1, boolean=True) for _ in valid_shift_indices]
        shift_sum = cp_sum(y_shift)
        constraints.append(shift_sum <= t)
        
        for i, idx in enumerate(valid_shift_indices):
            y = y_shift[i]
            constraints.append(y <= x[idx])
            constraints.append(y <= x[idx + d])
            constraints.append(y >= x[idx] + x[idx + d] - 1)

    objective = Minimize(t)
    
    problem = Problem(objective, constraints)
    
    # Try using CBC first (should be available)
    try:
        problem.solve(solver=cp.CBC, verbose=False)
    except:
        try:
            problem.solve(solver=cp.GLPK_MI, verbose=False)
        except:
            try:
                problem.solve(solver=cp.ECOS_BB, verbose=False)
            except:
                return None, None  # Failed all solvers
    
    if not problem.is_dpp():
        return None, None  # Boxed
    
    # Return the approximate solution
    x_opt = [round(x[i].value) for i in range(n)]
    
    # Record final integer result
    # Compute the actual maximum from the binary just computed
    h_int = np.array(x_opt)
    dx = 2.0 / n
    corr = correlate(h_int, 1 - h_int, mode="full")
    max_corr = np.max(corr)
    return h_int, max_corr * dx

def smooth_max_gradient_descent(h_start, max_iter=200, tau=0.5, shrink_rate=0.5, epsilon=1e-6):
    """Perform gradient descent using smooth max"""
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.copy()
    
    # Compute shifts
    shift_indices = []
    for d in range(-(n-1), n):
        if d == 0: continue
        shift_indices.append(d)
    
    def compute_shift_d(d):
        # h_i * (1 - h_{i+d}) for valid pairs
        if d > 0:
            # h_i * (1 - h_{i+d}) for i = [0..n-d-1]
            return np.sum(h[:n-d] * (1 - h[d:]))
        else:
            # d < 0: h_i * (1 - h_{i+d}) for i = [0..n+d-1]
            return np.sum(h[-d:] * (1 - h[:-d]))
    
    def smooth_max_overlaps(h, tau):
        # Compute the smooth max of all overlaps
        overlaps = [compute_shift_d(d) for d in shift_indices]
        return tau * np.log(np.sum(np.exp(np.array(overlaps)/tau)))
    
    def gradient_of_smooth_max(h, tau):
        # Compute gradient of smoothed max
        overlaps = np.array([compute_shift_d(d) for d in shift_indices])
        weights = np.exp(overlaps/tau) / np.sum(np.exp(overlaps/tau))
        
        # Derive ∇C_d(h) = 1 - h_{i+d} - h_{i-d},  but we're summing over all d and valid pairs
        # Let's compute manually:
        grad = np.zeros_like(h)
        
        for d in shift_indices:
            if d > 0:
                #grad[:n-d] += weights[shift_indices.index(d)]
                # but instead make contribution properly for specific pairs
                # Simplified distinction-based summation:
                for k in range(n - d):
                    grad[k] += weights[shift_indices.index(d)] * (1 - h[k + d])
                    grad[k + d] -= weights[shift_indices.index(d)] * h[k]
            else:  # d < 0
                abs_d = -d
                for k in range(n - abs_d):
                    grad[k] -= weights[shift_indices.index(d)] * h[k - abs_d]
                    grad[k - abs_d] += weights[shift_indices.index(d)] * h[k]
        
        return grad
    
    def project_h(h, desired_sum=1.0):
        """Project onto the feasible place ∫ h dx = 1"""
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h) * dx
        if abs(current_sum - desired_sum) > epsilon:
            h = h * (desired_sum / current_sum)
        h = np.clip(h, 0.0, 1.0)
        return h
    
    for _ in range(max_iter):
        if tau < epsilon:
            break
        # Compute gradient
        grad = gradient_of_smooth_max(h, tau)
        # Projected gradient step
        h_new = h - 0.01 * grad # Fixed step size
        h_new = project_h(h_new, 1.0)
        h = h_new
        
        # Shrink smoothing parameter
        tau *= shrink_rate
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Use initial_h_values to avoid edge cases
    # if it's a global value
    try:
        h0 = np.array(initial_h_values)
    except:
        h0 = np.array([0.5] * 42)
    n_points = len(h0)
    
    # Phase 1: Binary MILP for optimal integer solution (if feasible)
    binary_h, c5_binary = solve_binary_milp(n_points)
    
    if binary_h is None or c5_binary is None:
        # Fallback
        best_h = h0.copy()
        best_c5 = evaluate_c5(best_h)
    else:
        best_h = binary_h.astype(float)
        best_c5 = c5_binary
    
    # Phase 2: Continuous refinement with smooth max
    while (time.time() - start_time) < budget_s * 0.9:
        h_refined = smooth_max_gradient_descent(best_h)
        c5_new = evaluate_c5(h_refined)
        if c5_new < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_new
            
        # Try a randomized restart
        h_rand = best_h + np.random.normal(0, 0.02, n_points)
        h_rand = np.clip(h_rand, 0, 1)
        # Project back
        h_rand = h_rand * (1.0 / np.sum(h_rand * (2.0/n_points))) * (1.0 * n_points / 2.0)
        c5_rand = evaluate_c5(h_rand)
        if c5_rand < best_c5:
            best_h = h_rand.copy()
            best_c5 = c5_rand
            
        if (time.time() - start_time) > budget_s * 0.9:
            break
    
    # Final polishing step with SLSQP
    def obj_slqp(h):
        return evaluate_c5(h)
    bounds = [(0, 1) for _ in range(n_points)]
    
    def con_eq(h):
        return np.sum(h) * (2.0 / n_points) - 1.0
    
    try:
        res = minimize(obj_slqp, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': con_eq},
                       options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12})
        if res.success:
            best_h = res.x
            best_c5 = obj_slqp(best_h)
    except Exception as e:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
