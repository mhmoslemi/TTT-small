import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
N = 60  # Binary problem size

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_binary_milp():
    """Solve binary MILP to get a combinatorial skeleton"""
    x = cp.Variable(N, boolean=True)
    t = cp.Variable(pos=True)
    
    # Constraints
    constraints = [cp.sum(x) == N/2]
    
    # Create auxiliary variables for products
    for k in range(1, N):
        # Only include valid pairs
        valid_pairs = [(i, i + k) for i in range(N - k)]
        y = cp.Variable(len(valid_pairs), boolean=True)
        
        # Add constraints for y[i,k] = x[i]*x[i+k]
        for idx, (i, j) in enumerate(valid_pairs):
            constraints.extend([
                y[idx] <= x[i],
                y[idx] <= x[j],
                y[idx] >= x[i] + x[j] - 1
            ])
        
        # Worst-case bound for this shift
        constraints.append(cp.sum(y) >= N/2 - t)
        
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Try to solve with GLPK_MI solver, but limit time
    try:
        prob.solve(solver=cp.GLPK_MI, time_limit=30)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return np.round(x.value).astype(int)
        else:
            # Fallback to feasible solution from solver
            return np.round(x.value).astype(int)
    except:
        # Resort to a random feasible solution if MILP fails
        print("MILP solver failed, fallback to random binary vector")
        h = np.random.choice([0, 1], size=N, p=[0.5, 0.5])
        while np.sum(h) != N/2:
            h = np.random.choice([0, 1], size=N, p=[0.5, 0.5])
        return h

def upsample_binary(h_bin):
    """Upsample the binary vector to full resolution"""
    h_full = np.zeros(n_points)
    for i in range(n_points):
        h_full[i] = h_bin[i // (n_points // N)]
    return h_full

def improve_skeleton_swap_search(h):
    """Worst-shift-focused swap improvement of binary skeleton"""
    # Convert to boolean array
    h_bool = (h > 0.5).astype(int)
    n = len(h_bool)
    
    # Precompute auto-correlation
    def compute_auto_corr(h_vec):
        corr = np.zeros(n)
        for k in range(n):
            corr[k] = np.sum(h_vec[:-k] * h_vec[k:]) if k > 0 else np.sum(h_vec * h_vec)
        return corr
    
    def update_corr_delta(i, j, corr, h_vec):
        # Update only problematic indices for efficiency
        delta = 0
        for k in range(1, n):
            if i + k < n:
                delta += (h_vec[i] ^ h_vec[i+k]) - (h_vec[j] ^ h_vec[j+k])
        return delta
    
    corr = compute_auto_corr(h_bool)
    max_corr = np.max(corr)
    
    # Perform swaps only if we're still not good enough
    for _ in range(5000):  # iteration limit
        k_max = np.argmax(corr)  # worst shift
        if corr[k_max] >= max_corr / 2:  # Stop if good enough
            break
            
        # Find candidate swaps
        A = []
        B = []
        for i in range(n - k_max):
            if h_bool[i] == 1 and h_bool[i + k_max] == 0:
                A.append(i)
            if h_bool[i] == 0 and h_bool[i + k_max] == 1:
                B.append(i)
        
        if not A or not B:
            # Random swap
            i, j = np.random.randint(0, n, 2)
            if h_bool[j] == h_bool[i]:
                continue
            h_bool[i] = 1 - h_bool[i]
            h_bool[j] = 1 - h_bool[j]
            corr = compute_auto_corr(h_bool)
        else:
            # Choose pair
            a = np.random.choice(A)
            b = np.random.choice(B)
            
            if corr[k_max] - corr[a] < 1e-8 and corr[k_max] - corr[b] < 1e-8:
                # Accept good ones
                h_bool[a] = 0
                h_bool[b] = 1
                corr = compute_auto_corr(h_bool)
    
    return h_bool.astype(float)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points / 2
    
    h_current = None
    c5_best = float('inf')
    
    # Phase A: Binary MILP
    if time.time() - start_time < budget_s * 0.05:
        h_bin = solve_binary_milp()
        h_current = upsample_binary(h_bin)
        h_current = project_to_sum_box(h_current, TARGET_SUM)
    else:
        # Fallback to initial solution
        h_current = np.array(initial_h_values)
        h_current = project_to_sum_box(h_current, TARGET_SUM)
    
    # Phase B: Optional swap-based refinement
    if time.time() - start_time < budget_s * 0.1:
        h_current = improve_skeleton_swap_search(h_current)
        h_current = project_to_sum_box(h_current, TARGET_SUM)
    
    # Phase C: Differential Evolution on high resolution
    def differential_evolution(h_start):
        pop_size = 30
        max_iter = 1000
        F = 0.8
        CR = 0.9
        best_eval = float('inf')
        best_h = h_start.copy()
        
        # Initialize population
        pop = []
        for _ in range(pop_size):
            h = h_start.copy()
            h += np.random.rand(len(h)) * 0.02
            h = project_to_sum_box(h, TARGET_SUM)
            pop.append(h)
        
        # Iterate
        for it in range(max_iter):
            if time.time() - start_time > budget_s * 0.8:
                break
                
            for i, h in enumerate(pop):
                # Select three other individuals
                idxs = np.random.choice(pop_size, 3, replace=False)
                a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
                
                # Mutation
                m = a + F * (b - c)
                m = np.clip(m, 0, 1)
                
                # Crossover
                trial = np.where(np.random.rand(len(h)) < CR, m, h)
                
                # Project
                trial = project_to_sum_box(trial, TARGET_SUM)
                
                # Evaluate
                c5_trial = evaluate_c5(trial)
                
                # Selection
                c5_h = evaluate_c5(h)
                if c5_trial < c5_h:
                    pop[i] = trial
                    if c5_trial < best_eval:
                        best_eval = c5_trial
                        best_h = trial.copy()
        
        return best_h, best_eval
    
    # Run DE
    h_de, c5_de = differential_evolution(h_current)
    if c5_de < c5_best:
        c5_best = c5_de
        h_current = h_de
    
    # Phase D: Simulated Annealing refinement
    def simulated_annealing(h_start, epsilon=0.05):
        h_current = h_start.copy()
        c5_current = evaluate_c5(h_current)
        best_h = h_current.copy()
        best_c5 = c5_current
        
        T = 1.0
        alpha = 0.99
        max_iter = 10000
        for i in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Choose two indices
            i1 = np.random.randint(0, len(h_current))
            i2 = np.random.randint(0, len(h_current))
            
            # Transfer amount
            delta = np.random.uniform(-epsilon, epsilon)
            if abs(h_current[i1] + delta) > 1 or abs(h_current[i2] - delta) > 1:
                continue
            
            # Make transfer
            h_trial = h_current.copy()
            h_trial[i1] += delta
            h_trial[i2] -= delta
            
            # Ensure it's in valid range for box constraints
            h_trial = np.clip(h_trial, 0, 1)
            
            # Project onto sum constraint
            h_trial = project_to_sum_box(h_trial, TARGET_SUM)
            
            # Evaluate
            c5_trial = evaluate_c5(h_trial)
            delta_c5 = c5_trial - c5_current
            
            # Accept or not
            if delta_c5 < 0 or np.exp(-delta_c5 / T) > np.random.rand():
                h_current = h_trial
                c5_current = c5_trial
                
                if c5_current < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_current
            
            # Cool down
            T *= alpha
            
        return best_h, best_c5
    
    # Run SA
    h_sa, c5_sa = simulated_annealing(h_current)
    if c5_sa < c5_best:
        c5_best = c5_sa
        h_current = h_sa
    
    # Phase E: Hill climbing
    def hill_climb(h_start, num_iterations=2000):
        h_current = h_start.copy()
        c5_current = evaluate_c5(h_current)
        best_h = h_current.copy()
        best_c5 = c5_current
        
        for _ in range(num_iterations):
            if time.time() - start_time > budget_s:
                break
                
            # Pick subset of indices
            indices = np.random.choice(len(h_current), size=len(h_current)//20, replace=False)
            h_trial = h_current.copy()
            
            # Perturb a few coordinates
            for i in indices:
                step = np.random.normal(0, 0.01 * h_current[i] * (1 - h_current[i]))
                h_trial[i] += step
            
            # Clip and project
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = project_to_sum_box(h_trial, TARGET_SUM)
            
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < c5_current:
                h_current = h_trial
                c5_current = c5_trial
                
                if c5_current < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_current
        
        return best_h, best_c5
    
    # Run hill climbing
    h_hc, c5_hc = hill_climb(h_current)
    if c5_hc < c5_best:
        c5_best = c5_hc
        h_current = h_hc
    
    # Final evaluation and return
    final_c5 = evaluate_c5(h_current)
    
    return (h_current.tolist(), float(final_c5), n_points)
