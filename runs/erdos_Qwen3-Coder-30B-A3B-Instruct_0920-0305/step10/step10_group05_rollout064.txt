import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def autocorr_fft(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    b = h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, dx):
    """Convert correlation result to C5 value"""
    return 1.0 - dx * np.min(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_objective(h, beta=50):
    """Smooth approximation to C5 using log-sum-exp"""
    corr = autocorr_fft(h)
    max_corr = np.max(corr)
    corr_shifted = corr - max_corr
    weights = np.exp(beta * corr_shifted)
    z = np.sum(weights)
    return 1.0 - (1.0 / beta) * np.log(z)

def smooth_gradient(h, beta=50):
    """Gradient of smooth objective"""
    corr = autocorr_fft(h)
    max_corr = np.max(corr)
    corr_shifted = corr - max_corr
    weights = np.exp(beta * corr_shifted)
    z = np.sum(weights)
    weights /= z

    # numerical gradient computation
    term = np.zeros_like(h)
    for i in range(len(h)):
        # Compute convolution of weights with h reversed
        win = weights * np.roll(h, i)
        term[i] = np.sum(win)
    
    # Multiply by -dx
    dx = 2.0 / len(h)
    return -dx * term

def parametric_shape(a, m):
    """Generate parametric piecewise constant shape"""
    h = np.ones(m)
    if a >= 1.0:
        a = 0.999
    width = int(np.round(a * m))
    if width == 0:
        width = 1
    height = 1.0 
    p = (1.0 - 2.0 * width / m) / (1 - 2.0 * width / m) if (1 - 2.0 * width / m) != 0 else 0.5
    p = max(0, min(p, 1))
    h[:width] = height
    h[-width:] = height
    h[width:-width] = p
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 200
    dx = 2.0 / n_points
    half_size = n_points // 2
    target_sum = n_points / 4
    
    best_C5 = float('inf')
    best_h = None

    # Stage 1: Parameter sweep
    a_vals = np.arange(0.01, 0.49, 0.002)
    for a in a_vals:
        if time.time() - start_time > budget_s * 0.8:
            break
        h_half = parametric_shape(a, half_size)
        h_full = np.concatenate([h_half, h_half[::-1]])
        h_full = project_to_sum_box(h_full, n_points / 2)
        c5_val = c5_from_corr(autocorr_fft(h_full), dx)
        if c5_val < best_C5:
            best_C5 = c5_val
            best_h = h_full.copy()

    # Stage 2: Local refinement with smooth surrogate
    if best_h is not None:
        h_refined = best_h.copy()
        bounds = [(0, 1) for _ in range(half_size)] + [(0, 1) for _ in range(half_size)]
        # maximize the smooth surrogate with minimize
        def obj_to_minimize(h_half):
            h_full = np.concatenate([h_half, h_half[::-1]])
            return smooth_objective(h_full, beta=50)
        def constraint_eq(h_half):
            h_full = np.concatenate([h_half, h_half[::-1]])
            return np.sum(h_full) - n_points / 2
        
        x0 = best_h[:half_size].copy()
        cons = {'type': 'eq', 'fun': constraint_eq}
        try:
            res = minimize(
                obj_to_minimize,
                x0,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(half_size)],
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_opt = np.concatenate([res.x, res.x[::-1]])
                h_opt = project_to_sum_box(h_opt, n_points / 2)
                c5 = c5_from_corr(autocorr_fft(h_opt), dx)
                if c5 < best_C5:
                    best_C5 = c5
                    best_h = h_opt.copy()
        except Exception:
            pass
            
    # Stage 3: Simulated Annealing
    def simulated_annealing(h_init, num_steps=150000):
        h_curr = h_init.copy()
        C5_curr = c5_from_corr(autocorr_fft(h_curr), dx)
        best_h = h_curr.copy()
        best_C5 = C5_curr

        T = 0.01
        T_min = 1e-6
        alpha = 0.9995
        delta_max = 0.05
        acc_count = 0
        move_count = 0
        
        while T > T_min and move_count < num_steps:
            if time.time() - start_time > budget_s * 0.95:
                break

            # Pick two random indices to swap
            i = np.random.randint(0, half_size)
            j = np.random.randint(0, half_size)
            if i == j:
                j = (j + 1) % half_size

            # Temporarily get neighbors
            delta = np.random.rand() * delta_max
            # Ensure we can move mass without violating box constraints
            if h_curr[i] >= delta and h_curr[j] <= 1 - delta:
                h_new = h_curr.copy()
                h_new[i] -= delta
                h_new[j] += delta

                C5_new = c5_from_corr(autocorr_fft(h_new), dx)
                if C5_new < C5_curr:
                    h_curr = h_new.copy()
                    C5_curr = C5_new
                    if C5_new < best_C5:
                        best_C5 = C5_new
                        best_h = h_curr.copy()
                    acc_count += 1
                else:
                    # Metropolis acceptance for temperature-sensitive moves
                    if np.random.rand() < np.exp(-(C5_new - C5_curr) / T):
                        h_curr = h_new.copy()
                        C5_curr = C5_new
                        acc_count += 1
                
                move_count += 1
                
                # Adaptive step size
                if move_count % 500 == 0:
                    ratio = acc_count / 500.0
                    if ratio > 0.4:
                        delta_max *= 1.1
                    elif ratio < 0.1:
                        delta_max *= 0.9
                    delta_max = np.clip(delta_max, 1e-4, 0.2)
                    acc_count = 0

            T *= alpha

        return best_h, best_C5
    
    if best_h is not None:
        h_sa, c5_sa = simulated_annealing(best_h)
        if c5_sa < best_C5:
            best_C5 = c5_sa
            best_h = h_sa.copy()
    
    # Stage 4: Final refinement with smooth L-BFGS on full vector
    if best_h is not None:
        try:
            def obj_to_minimize_full(h_full):
                return smooth_objective(h_full, beta=200)

            x0 = best_h.copy()
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(
                obj_to_minimize_full,
                x0,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_final = res.x
                h_final = project_to_sum_box(h_final, n_points / 2)
                c5_final = c5_from_corr(autocorr_fft(h_final), dx)
                if c5_final < best_C5:
                    best_C5 = c5_final
                    best_h = h_final.copy()
        except Exception:
            pass
            
    # Perform exact computation of C5
    if best_h is not None:
        corr = autocorr_fft(best_h)
        exact_c5 = c5_from_corr(corr, dx)
        return (best_h.tolist(), exact_c5, n_points)
    else:
        return ([0.0]*n_points, 1.0, n_points)
