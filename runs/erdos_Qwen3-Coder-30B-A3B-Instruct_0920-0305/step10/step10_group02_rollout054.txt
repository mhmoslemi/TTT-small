def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):

    corr = compute_corr_fft(h)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_boundaries(h, smoothing_radius=3):
    """Apply smoothing at boundaries to create continuous values"""
    h_smooth = h.copy()
    boundaries = np.where(np.diff(h) != 0)[0]
    
    for boundary in boundaries:
        start = max(0, boundary - smoothing_radius + 1)
        end = min(len(h), boundary + smoothing_radius)
        # Create linear interpolation
        if boundary < len(h) - 1:
            # From 0 to 1
            if h[boundary] == 0:
                h_smooth[start:end] = np.linspace(0, 1, end - start)
            # From 1 to 0  
            else:
                h_smooth[start:end] = np.linspace(1, 0, end - start)
    
    # Adjust sum to target
    current_sum = np.sum(h_smooth)
    adjustment = (n_points / 2 - current_sum) / n_points
    h_smooth += adjustment
    
    # Clip to [0,1]
    h_smooth = np.clip(h_smooth, 0, 1)
    return h_smooth

def gradient_descent_refinement(h, max_iter=200, learning_rate=0.01):
    """Refine solution using gradient descent on softmax surrogate"""
    h_refined = h.copy()
    
    for iteration in range(max_iter):
        # Compute correlation
        corr = compute_corr_fft(h_refined)
        max_corr = np.max(corr)
        
        # Compute softmax surrogate gradient
        alpha = 10.0
        max_corr = np.max(corr)
        shifted_corr = corr - max_corr
        exp_corr = np.exp(alpha * shifted_corr)
        exp_sum = np.sum(exp_corr)
        weights = exp_corr / exp_sum
        
        # Compute gradient contributions
        grad = np.zeros_like(h_refined)
        for k in range(len(corr)):
            if k == 0:
                continue
            # Gradient w.r.t. h_j at lag k
            # This is approximate - using forward/backward difference
            for i in range(len(h_refined)):
                # In a general sense
                if i + k < len(h_refined):
                    grad[i] += weights[k] * (1 - h_refined[i + k] - h_refined[i])
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad /= grad_norm
            
        # Update with gradient descent
        h_refined = h_refined - learning_rate * grad
        h_refined = np.clip(h_refined, 0, 1)
        
        # Project to correct sum
        h_refined = project_to_sum_box(h_refined, n_points / 2)
        
        # Early stopping if improvement is small
        if iteration > 10 and np.linalg.norm(h_refined - h) < 1e-6:
            break
            
        # Apply adaptive learning rate
        if iteration % 10 == 0:
            learning_rate *= 0.95
            
    return h_refined
