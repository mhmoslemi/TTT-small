import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import math

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Start with random binary initialization
    h0 = np.zeros(n_points)
    ones_indices = np.random.choice(n_points, size=int(n_points/2), replace=False)
    h0[ones_indices] = 1.0
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Simulated Annealing
    max_iters = 500000
    T0 = 0.02
    T_end = 1e-5
    
    # Keep track of the best solution found
    best_solution = best_h.copy()
    best_c5_value = best_c5
    
    for iteration in range(max_iters):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Compute temperature
        T = T0 * (1 - iteration / max_iters) + T_end
        
        # Select two random indices with distinct values
        i, j = np.random.randint(0, n_points, 2)
        while h0[i] == h0[j]:
            i, j = np.random.randint(0, n_points, 2)
            
        # Perform swap
        h0[i], h0[j] = h0[j], h0[i]
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h0)
        
        # Accept or reject the move
        delta = new_c5 - best_c5
        if delta <= 0 or np.random.rand() < math.exp(-delta / T):
            best_c5 = new_c5
            if new_c5 < best_c5_value:
                best_c5_value = new_c5
                best_solution = h0.copy()
        else:
            # Revert swap
            h0[i], h0[j] = h0[j], h0[i]
    
    # Final refinement with continuous optimization
    try:
        # Use the best binary solution as start point for continuous refinement
        refined_h = best_solution.copy()
        refined_h = project_to_sum_box(refined_h, TARGET_SUM)
        refined_c5 = evaluate_c5(refined_h)
        
        # Local optimization with SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
        
        res = minimize(
            evaluate_c5, 
            refined_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        
        if res.success:
            optimized_h = res.x
            optimized_h = project_to_sum_box(optimized_h, TARGET_SUM)
            optimized_c5 = evaluate_c5(optimized_h)
            
            if optimized_c5 < best_c5_value:
                best_c5_value = optimized_c5
                best_solution = optimized_h.copy()
                
    except Exception as e:
        pass
    
    return (best_solution.tolist(), float(best_c5_value), n_points)
