import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def genetic_algorithm(start_time, budget_s):
    """Genetic algorithm for global optimization"""
    pop_size = 50
    generations = 100
    mutation_rate = 0.1
    elite_size = 5
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = np.random.rand(n_points) * 0.5  # Start with values in [0, 0.5]
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    
    best_individual = population[0]
    best_fitness = evaluate_c5(best_individual)
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 30:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for individual in population:
            fitness = evaluate_c5(individual)
            fitness_scores.append(fitness)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        if fitness_scores[0] < best_fitness:
            best_fitness = fitness_scores[0]
            best_individual = population[0]
        
        # Create new population
        new_population = []
        
        # Elitism
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = tournament_select(population, fitness_scores)
            parent2 = tournament_select(population, fitness_scores)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            if np.random.random() < mutation_rate:
                child1 = mutate(child1)
            if np.random.random() < mutation_rate:
                child2 = mutate(child2)
            
            # Project to valid range
            child1 = project_to_sum_box(child1, TARGET_SUM)
            child2 = project_to_sum_box(child2, TARGET_SUM)
            
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
    
    return best_individual, best_fitness

def tournament_select(population, fitness_scores, tournament_size=3):
    """Tournament selection for genetic algorithm"""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitness = [fitness_scores[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_idx]

def crossover(parent1, parent2):
    """Single point crossover"""
    crossover_point = np.random.randint(1, n_points-1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual):
    """Gaussian mutation"""
    mutated = individual.copy()
    mutation_strength = 0.05
    for i in range(len(mutated)):
        if np.random.random() < 0.1:  # 10% chance to mutate each gene
            mutated[i] += np.random.normal(0, mutation_strength)
    return mutated

def basin_hopping_optimization(start_time, budget_s):
    """Basin hopping optimization"""
    # Start from best solution so far
    current_h = np.array(initial_h_values)

    current_c5 = evaluate_c5(current_h)

        if local_c5 < current_c5:

            temp = 1.0 / (i + 1)

        if perturbed_c5 < current_c5:

            current_c5 = perturbed_c5
    
    return best_h, best_c5

def local_optimization(h):
    """Use L-BFGS for local optimization"""
    try:
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        res = minimize(
            objective, 
            h, 
            method='L-BFGS-B', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
        c5 = evaluate_c5(h)
        return h, c5
    except:
        return h, evaluate_c5(h)
