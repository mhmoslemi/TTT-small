import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def upscale_h(original_h, n_new):
    """Upscale initial_h_values to new length using linear interpolation"""
    n_old = len(original_h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, original_h)

def propose_move(h, max_step, rng):
    """Propose a move that maintains sum constraint and box constraints"""
    n = len(h)
    # try up to few times to find feasible pair
    for _ in range(10):
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # compute feasible interval for delta added to i and subtracted from j
        lower = max(-h[i], h[j] - 1)
        upper = min(1 - h[i], h[j])
        if lower < upper:
            # limit delta magnitude
            lower = max(lower, -max_step)
            upper = min(upper, max_step)
            if lower >= upper:
                continue
            delta = rng.uniform(lower, upper)
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            return new_h
        # try opposite direction: add to j, subtract from i
        lower = max(-h[j], h[i] - 1)
        upper = min(1 - h[j], h[i])
        if lower < upper:
            lower = max(lower, -max_step)
            upper = min(upper, max_step)
            if lower >= upper:
                continue
            delta = rng.uniform(lower, upper)
            new_h = h.copy()
            new_h[j] += delta
            new_h[i] -= delta
            return new_h
    return None

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Get initial candidates
    candidates = []
    
    # From initial_h_values (upscale if needed)
    try:
        global initial_h_values
        upsampled_h = upscale_h(initial_h_values, n_points)
        upsampled_h = project_to_sum_box(upsampled_h, TARGET_SUM)
        candidates.append(("upscaled_initial", upsampled_h.copy()))
    except:
        pass
    
    # Cosine shape
    x = np.arange(n_points) * dx
    a = 0.5 - 1e-5
    cos_h = 0.5 + a * np.cos(np.pi * x / 2)
    cos_h = project_to_sum_box(cos_h, TARGET_SUM)
    candidates.append(("cosine", cos_h.copy()))
    
    # Random uniform
    rand_h = np.random.uniform(0, 1, n_points)
    rand_h = project_to_sum_box(rand_h, TARGET_SUM)
    candidates.append(("random_uniform", rand_h.copy()))
    
    # Random binary
    binary_h = np.zeros(n_points)
    ones_idx = np.random.choice(n_points, size=n_points // 2, replace=False)
    binary_h[ones_idx] = 1.0
    candidates.append(("binary_random", binary_h.copy()))
    
    # Evaluate all candidates and find the best
    best_candidate_h = None
    best_candidate_val = float('inf')
    for name, h in candidates:
        val = evaluate_c5(h)
        if val < best_candidate_val:
            best_candidate_val = val
            best_candidate_h = h.copy()
    
    # Set best_h to the best of all candidates
    current_h = best_candidate_h.copy()
    current_c5 = best_candidate_val
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
    
    # Simulated Annealing
    rng = np.random.default_rng(seed)
    
    # Parameters for SA
    total_time_allowed = budget_s * 0.9
    max_step_initial = 0.1
    max_step_final = 0.001
    T_max = 0.02
    T_min = 0.0001
    
    iteration = 0
    while time.time() - start_time < total_time_allowed:
        iteration += 1
        elapsed = time.time() - start_time
        frac = elapsed / total_time_allowed
        
        # Adaptive parameters
        T = T_max * (1 - frac) + T_min * frac
        max_step = max_step_initial * (1 - frac) + max_step_final
        
        # Propose move
        new_h = propose_move(current_h, max_step, rng)
        if new_h is None:
            continue
            
        new_c5 = evaluate_c5(new_h)
        
        # Accept or reject
        if new_c5 < current_c5:
            current_h = new_h.copy()
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
        else:
            # Accept with probability
            if np.random.random() < np.exp(-(new_c5 - current_c5) / (T + 1e-12)):
                current_h = new_h.copy()
                current_c5 = new_c5
        
        if iteration % 5000 == 0 and time.time() - start_time > total_time_allowed * 0.8:
            break
        
    # Final local refinement (greedy descent)
    if best_h is not None and time.time() - start_time < budget_s * 0.95:
        ref_iter = 0
        max_step_refine = 0.005
        current_h_ref = best_h.copy()
        current_c5_ref = best_c5
        
        while time.time() - start_time < budget_s - 5 and ref_iter < 50000:
            ref_iter += 1
            elapsed = time.time() - start_time
            frac = elapsed / budget_s
            max_step = max_step_refine * (1 - frac) + 1e-6
            
            new_h = propose_move(current_h_ref, max_step, rng)
            if new_h is None:
                continue
                
            new_c5 = evaluate_c5(new_h)
            
            if new_c5 < current_c5_ref:
                current_h_ref = new_h.copy()
                current_c5_ref = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = current_h_ref.copy()
    
    if best_h is None:
        best_h = current_h.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
