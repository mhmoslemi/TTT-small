import numpy as np
from scipy.signal import correlate
import time
import math

# Global constants
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points

def compute_auto_correlation_binary(x):
    """Compute auto-correlation using NumPy correlate (full mode)"""
    return correlate(x, x, mode='full')

def update_auto_correlation_incremental(A, x, i, j):
    """Update A by swapping x[i] and x[j] without recomputing all of A"""
    # A is a vector with length 2*n-1, representing correlation indices from -(n-1) to +(n-1)
    delta_A = np.zeros_like(A)
    n = len(x)
    for d in range(-n+1, n):
        # Compute contributions to delta_A[d] for indices i and j
        contrib_i = 0
        contrib_j = 0
        if 0 <= i+d < n:
            contrib_i += x[i+d]
        if 0 <= j+d < n:
            contrib_j += x[j+d]
        if 0 <= i-d < n:
            contrib_i += x[i-d]
        if 0 <= j-d < n:
            contrib_j += x[j-d]
        delta_A[d+n-1] = -contrib_i + contrib_j
    return A + delta_A

def compute_overlap_from_auto_correlation(A_full, x):
    """Compute overlap values for all shifts assuming x is binary and sum = n/2"""
    n = len(x)
    # We need to compute how many 1s in window for overlaps (at each shift d)
    # Prefix sum approach for prefix_sums of x, then compute overlap sums
    prefix = np.concatenate([[0], np.cumsum(x)])
    # For each shift d, define overlap as:
    # Sum of ones in overlap window - actual number of matching pairs
    # overlap = #ones in overlap window - #1s in window at position i
    # We will use A as the number of 1-1 matches; 
    # but this rewrite is simpler to make clear
    
    # A takes indices from -(n-1) to +n-1
    # A[d + (n-1)] = sum_{i} x[i]*x[i+d]
    # For shift d, the overlap contribution = prefix[shift_end] - prefix[shift_start] - A[d]
    
    # Efficient way: we know that if x[i] = 1, we are interested in
    # sum_{j} x[j] for all j such that j is in the support of shift d and is in the overlap
    
    # To get overlap at shift d: 
    #   HowManyOnesInOverlap - count of matched 1,1 pairs = num_ones / 2 - A[d] 
    # but let's compute it properly:
    # overlap_distance_d = sum_{i} x[i]*(1-x[i+d])
    # = sum_{i} x[i] - sum_{i} x[i]*x[i+d] 
    # = count_ones - A[d]

    # So we get total count_ones - A[i+n-1] for each d

    total_ones = np.sum(x)
    overlap_values = np.zeros(2*n-1)
    for d in range(-n+1, n):
        if d >= 0:
            # d >= 0: window starts at index d, length sum over (0 to n-d-1) in the shifted x
            # actually total - A[d + (n-1)] where d = (shift index)
            overlap_values[d + (n-1)] = total_ones - A_full[d + (n-1)]
        else:
            # d < 0: shifted window starts at index 0, (n+d) length in the shifted x
            overlap_values[d + (n-1)] = total_ones - A_full[d + (n-1)]
    
    return overlap_values

def fast_c5_from_binary(x):
    """Compute C5 quickly using binary vectors"""
    A = compute_auto_correlation_binary(x)
    overlaps = compute_overlap_from_auto_correlation(A, x)
    return dx * np.max(overlaps)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using binary search"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_to_fractional(x):
    """Convert binary to fractional (used after continuous refinement)"""
    return x.astype(float)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    TARGET_SUM = n_points / 2

    # Initialize binary vector (starting with a good known solution)
    if 'initial_h_values' in globals():
        x = np.array(initial_h_values)
        x = np.round(x).astype(int)
    else:
        x = np.zeros(n_points, dtype=int)
        x[:n_points//2] = 1
        np.random.shuffle(x)
    
    # Ensure correct sum
    sum_x = np.sum(x)
    if sum_x > TARGET_SUM:
        # Turn excess ones to zeros
        ones_indices = np.where(x == 1)[0]
        flips = int(sum_x - TARGET_SUM)
        for i in range(flips):
            x[ones_indices[i]] = 0
    elif sum_x < TARGET_SUM:
        # Turn excess zeros to ones
        zeros_indices = np.where(x == 0)[0]
        flips = int(TARGET_SUM - sum_x)
        for i in range(flips):
            x[zeros_indices[i]] = 1
    
    # The core objective being optimized is a binary vector 
    # not a real vector; so we keep x binary but evaluate the continuous c5 via fft
    
    # Fast evaluation
    def evaluate_c5_continuous(h):
        """Evaluate using standard method"""
        return dx * np.max(correlate(h, 1 - h, mode='full'))

    # Comptue initial overlaps
    current_C5 = fast_c5_from_binary(x)
    best_x = x.copy()
    best_C5 = current_C5

    # Start of Simulated Annealing with targeted swap strategy
    # Parameters
    T0 = 2 * dx
    Tf = 1e-6 * dx
    alpha = 0.9999  # cooling rate
    num_iter = 0
    max_iter_SA = int(300000)
    temperature = T0
    accepted_moves = 0
    stagnation_count = 0
    threshold = 1e-5

    def find_worst_shift(x, A):
        n = len(x)
        prefix = np.concatenate([[0], np.cumsum(x)])
        overlaps = np.zeros(2*n-1)
        for d in range(-n+1, n):
            if d >= 0:
                end_of_shift = min(n, n-d)
                ones_in_shift = prefix[end_of_shift] - prefix[0]
            else:
                start_of_shift = max(0, -d) 
                end_of_shift = n
                try:
                    ones_in_shift = prefix[end_of_shift] - prefix[start_of_shift]
                except:
                    ones_in_shift = 0
            overlaps[d + (n-1)] = ones_in_shift - A[d + (n-1)]
        return np.argmax(overlaps) - (n-1)

    # Major loop: wait until time permits, do annealing for each iteration
    while True:
        if time.time() - start_time > budget_s - 10:  # Calculate remaining time
            break

        # Temperature decay
        if accepted_moves > 0 and accepted_moves % 1000 == 0:
            temperature *= alpha
            if temperature < Tf:
                temperature = Tf

        # Choose a swap type with probability
        if np.random.rand() < 0.6:
            # Targeted swap
            A = compute_auto_correlation_binary(x)
            d_star = find_worst_shift(x, A)
            # Collect U and V
            U = []
            V = []
            n = len(x)
            for i in range(n):
                if x[i] == 1:
                    off = i + d_star
                    if 0 <= off < n and x[off] == 0:
                        U.append(i)
                if x[i] == 0:
                    off = i + d_star
                    if 0 <= off < n and x[off] == 1:
                        V.append(i)

            if len(U) > 0 and len(V) > 0:
                i = np.random.choice(U)
                j = np.random.choice(V)
                # Perform the swap and update index
                x[i], x[j] = x[j], x[i]
                A = update_auto_correlation_incremental(A, x, i, j)
                new_C5 = faster_C5_from_auto(A, x)
                delta = new_C5 - current_C5
                if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
                    current_C5 = new_C5
                    if current_C5 < best_C5:
                        best_x = x.copy()
                        best_C5 = current_C5
                    accepted_moves += 1
                else:
                    x[i], x[j] = x[j], x[i]  # revert
                num_iter += 1
            else:
                # Fallback to random swap
                i, j = np.random.choice(n, size=2, replace=False)
                old_i, old_j = x[i], x[j]
                x[i], x[j] = x[j], x[i]
                new_C5 = fast_c5_from_binary(x)
                delta = new_C5 - current_C5
                if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
                    current_C5 = new_C5
                    if current_C5 < best_C5:
                        best_x = x.copy()
                        best_C5 = current_C5
                    accepted_moves += 1
                else:
                    x[i], x[j] = old_i, old_j  # revert
                num_iter += 1
        else:
            # Random swap
            i, j = np.random.choice(n, size=2, replace=False)
            old_i, old_j = x[i], x[j]
            x[i], x[j] = x[j], x[i]
            new_C5 = fast_c5_from_binary(x)
            delta = new_C5 - current_C5
            if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
                current_C5 = new_C5
                if current_C5 < best_C5:
                    best_x = x.copy()
                    best_C5 = current_C5
                accepted_moves += 1
            else:
                x[i], x[j] = old_i, old_j  # revert
            num_iter += 1

        # Periodic restart or escape
        if num_iter % 20000 == 0:
            if abs(current_C5 - best_C5) < threshold:
                stagnation_count += 1
                # Inject diversity
                flip_num = int(n * 0.05)
                old_indices = np.where(x == 1)[0]
                new_indices = np.where(x == 0)[0]
                to_flip = np.random.choice(old_indices, size=flip_num, replace=False)
                to_fix = np.random.choice(new_indices, size=flip_num, replace=False)
                for old_idx, new_idx in zip(to_flip, to_fix):
                    x[old_idx], x[new_idx] = x[new_idx], x[old_idx]
                temperature *= 0.5
                c5_after_restart = fast_c5_from_binary(x)
                if c5_after_restart < best_C5:
                    best_C5 = c5_after_restart
                    best_x = x.copy()
        else:
            stagnation_count = 0

    # Convert to real vector for fine-tuning
    x_real = binary_to_fractional(best_x)
    x_real = project_to_sum_box(x_real, TARGET_SUM)
    
    # Fine-tune with scipy optimization
    from scipy.optimize import minimize
    
    def objective(h):
        # Renormalize to sum = TARGET_SUM
        h = project_to_sum_box(h, TARGET_SUM)
        return dx * np.max(correlate(h, 1 - h, mode='full'))

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - TARGET_SUM}
    
    try:
        res = minimize(objective, x_real, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12})
        if res.success:
            x_real = res.x.copy()
        else:
            # Fall back to projection
            x_real = project_to_sum_box(x_real, TARGET_SUM)
    except:
        x_real = project_to_sum_box(x_real, TARGET_SUM)

    # Constrain to [0,1] and sum constraint finally
    x_real = project_to_sum_box(x_real, TARGET_SUM)
    final_C5 = dx * np.max(correlate(x_real, 1 - x_real, mode='full'))

    # Return result
    return (x_real.tolist(), float(final_C5), n_points)

def faster_C5_from_auto(A, x):
    total_ones = np.sum(x)
    n = len(x)
    # Invalid since overlap is not A alone, but needs careful consideration
    # Instead: compute all O_d
    prefix = np.concatenate([[0], np.cumsum(x)])
    overlaps = np.zeros(2*n-1)
    for d in range(-n+1, n):
        if d >= 0:
            # From i=0 to i + d < n, the overlap
            start_idx = 0
            end_idx = min(n, n-d)
            length = end_idx - start_idx
        else:
            start_idx = -d
            end_idx = n
            length = end_idx - start_idx
        try:
            ones_in_window = prefix[end_idx] - prefix[start_idx]
        except:
            ones_in_window = 0
        
        A_at_d = A[d + (n-1)]
        overlap = ones_in_window - A_at_d
        overlaps[d + (n-1)] = overlap
    
    return dx * np.max(overlaps)
