import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

# Global variables
n_points = 200
dx = 2.0 / n_points
half_size = n_points // 2
target_sum = n_points / 4

def autocorr_fft(h):
    """Compute autocorrelation using FFT. Returns correlation values for all shifts."""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.real(ifft(fft(h_rev, L) * fft(h, L)))
    return corr

def compute_c5_exact(h):
    """Exact computation of C5 using max-over-k correlation."""
    corr = autocorr_fft(h)
    # Use correlate to get same shape as the existing approach
    full_corr = correlate(h, 1-h, mode="full")
    return dx * np.max(full_corr)

def compute_c5_smooth(h, beta=50.0):
    """Smooth approximation of C5 using log-sum-exp for differentiability."""
    corr = autocorr_fft(h)
    max_corr = np.max(corr)
    # Center the values to avoid overflow
    corr_centered = corr - max_corr
    exp_corr = np.exp(beta * corr_centered)
    log_sum = np.log(np.sum(exp_corr))
    return 1.0 - (dx / beta) * log_sum

def gradient_smooth(h, beta=50.0):
    """Gradient of the smooth C5 function."""
    corr = autocorr_fft(h)
    max_corr = np.max(corr)
    corr_centered = corr - max_corr
    exp_corr = np.exp(beta * corr_centered)
    log_sum = np.log(np.sum(exp_corr))

    # Compute derivatives w.r.t. each element
    # grad = -dx * (sum(exp(beta*corr) * d/dh_i corr) / sum(exp(beta*corr)))
    # Since corr_k = sum(h_i * h_{i+k}), d/dh_i corr_k = h_{i+k} + h_{i-k}
    # So we compute a convolution of the vector with itself (symmetric)

    # Compute the weight
    w = exp_corr / np.sum(exp_corr)
    # Gradient contribution: for each position i, we have
    # sum_k w_k * d corr_k / dh_i = sum_k w_k * h_{i+k} if k>=0
    #                             = sum_k w_k * h_{i-k} if k<0
    #             = sum_{k=1}^{n-1} w_k * (h_{i+k} + h_{i-k}) = 2*(h*conv_full)(i) - h(i) w(0)
    # This is a bit tricky in frequency domain - let's compute directly

    grad = np.zeros_like(h)
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    reuse_w = fft(w, L)
    reuse_h = fft(h_rev, L)
    for i in range(len(h)):
        # For each i we want to compute d(autocorr_i)/dh_i = h_{i+1} + h_{i-1} + ...
        # But that's not correct - let's go via definition
        # corr_k = sum_{j} h[j] * h[j+k]
        # so dcorr_k / dh_i = h_{i+k} if i+k is valid, h_{i-k} if i-k is valid
        deriv = 0
        for k in range(-len(h)+1, len(h)):
            j = i - k
            if 0 <= j < len(h):
                # d/dh_i (h[j] * h[j+k]) = 0 unless either of the terms is h_i, which is iff:
                # i = j or i = j+k, so j = i or j = i-k.
                # In first case: j = i -> h[j+k] = h[i+k]
                # In second case: j = i-k -> h[j] = h[i-k]
                if j == i:
                    if 0 <= i+k < len(h):
                        deriv += h[i+k]
                elif i == j + k:
                    if 0 <= i-k < len(h):
                        deriv += h[i-k]
        grad[i] = -dx * (deriv / np.sum(exp_corr)) 
    return grad

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints with sum = target_sum, **keeping sum constraint exact**."""
    # Ensure sum equals target by scaling
    current_sum = np.sum(x)
    if current_sum == 0:
        x = np.ones_like(x) * (target_sum / len(x))
    else:
        x = x * (target_sum / current_sum)
    # Clamp to [0, 1]
    x = np.clip(x, 0, 1)
    # Re-correct the sum with a water-filling style approach
    # We'll ensure that each x[i] is at most 1, and sum is as close to target as possible
    # We use sophisticated projection to have sum exactly target_sum with bound constraints
    # This is a known entropy-type problem with := P(C, d), where d = target_sum
    
    # Actually clip again to make sure
    x = np.clip(x, 0, 1)
    
    # Now ensure exact target sum by adjusting the vector to lie on the convex simplex
    # with constraints. This is a well known problem in optimization. 
    # For our case, we can do simple iterative scaling:
    # Keep making it project to simplex. Exact algorithm is detailed in numerical optimization
    # textbooks, but a simple way to maintain sum constraint is just to return x * target_sum / sum(x)
    # But above ensured 0 <= x[i] <= 1, and also preserves sum being close to target.
    # But since we want exact matching:
    # We have already made the sum approach roughly target_sum, and we know that 
    # we are strictly bounded in [0,1]. To adjust it exactly to target_sum, 
    # we can do a small fixed point iteration:
    while abs(np.sum(x) - target_sum) > 1e-10:
        current_sum = np.sum(x)
        # Adjust using gradient of sum w.r.t.x
        sd = target_sum - current_sum
        if abs(sd) < 1e-10:
            break
        # adjust proportional to current gradient: we want to change all equally to match target
        step = sd / len(x)
        # This will keep x in [0,1]. 
        x_new = x + step
        x_new = np.clip(x_new, 0, 1)
        x = x_new
        if np.allclose(np.sum(x), target_sum):
            break
  
    return x

def shape_parametric(half_h, a):
    """Build symmetric half-step function from edge width `a`."""
    if a <= 0 or a >= 0.5:
        return half_h
    p = (1 - 2 * a) / (2 * (1 - a))  # desired p for area = 1
    # Clamp p to [0,1]
    p = np.clip(p, 0, 1)
    n = len(half_h)
    # Exact widths: two left and right: a*2 = 2a; middle: 2 - 2a
    # Floor indices for bins...
    idx_left = int(np.floor(a / dx))
    idx_right = int(np.ceil((1 - a) / dx))
    
    # Place p in the middle
    # Construct bins in [0, 2] -> half in [0, 1] -> index: 0 to n-1
    # Create a function from this - place 1.0 at left and right edges
    if idx_left >= n or idx_right <= 0:
        return half_h
    
    # Compute indices:
    left_start = 0
    left_stop = idx_left  
    mid_start = idx_left  
    mid_stop = idx_right  
    right_stop = n  

    half_h[:] = 0.0
    if 0 <= left_start < left_stop <= n:
        half_h[left_start:left_stop] = 1.0
        
    if mid_start <= mid_stop <= n:
        half_h[mid_start:mid_stop] = p
        
    if right_stop > mid_stop:
        half_h[mid_stop:right_stop] = 1.0
        
    return half_h

def shape_parametric_full_from_a(a):
    """Create full symmetric step function from a parameter."""
    half_h = np.zeros(half_size)
    p = (1 - 2 * a) / (2 * (1 - a))  # desired p
    p = np.clip(p, 0, 1)
    half_h[:] = 0.0
    # Create very simple two-edge structure, then fill middle
    left_idx = int(np.floor(a / dx))
    right_idx = int(np.ceil((1 - a) / dx))
    
    if left_idx > 0 and right_idx < half_size:
        if left_idx > 0:
            half_h[:left_idx] = 1.0
        if right_idx < half_size:
            half_h[right_idx:] = 1.0
        # Middle area
        half_h[left_idx:right_idx] = p
        
    half_h = project_to_sum_box(half_h, target_sum)
    h_full = np.concatenate([half_h, half_h[::-1]])
    return h_full, p

def evaluate_smooth_c5(h):
    """Evaluate smooth C5 for given vector."""
    return compute_c5_smooth(h, beta=50.0)

def evaluate_exact_c5(h):
    """Evaluate exact C5 for given vector."""
    return compute_c5_exact(h)

def parametric_sweep_and_eco_refinement():
    """Try parameter sweep with variation of a."""
    best_c5 = float('inf')
    best_h = None
    best_a = 0.
    
    # Sweep over a from 0.01 to 0.49 with step 0.005 (50 steps to be fast)
    a_range = np.arange(0.01, 0.49, 0.005)
    n_tried = 0
    
    for a in a_range:
        try:
            h_full, p = shape_parametric_full_from_a(a)
            c5 = evaluate_exact_c5(h_full)
            n_tried += 1
            if c5 < best_c5:
                best_c5 = c5
                best_a = a
                best_h = h_full.copy()
        except Exception:
            continue
            
    print(f"Parameter sweep tried {n_tried} points, best found at a={best_a}, C5={best_c5}")
    
    # Refine using L-BFGS on the parameter a (indeed this is key as parametric function is easy to optimize)
    def obj_to_minimize(a):
        h_full, p = shape_parametric_full_from_a(a)
        return compute_c5_exact(h_full)
    
    try:
        res = minimize(obj_to_minimize, best_a, method='L-BFGS-B', bounds=[(0.001,0.49)])
        if res.success:
            a_opt = res.x[0]
            h_full_opt, p_opt = shape_parametric_full_from_a(a_opt)
            c5_opt = evaluate_exact_c5(h_full_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_full_opt
    except Exception:
        pass
    
    return best_h, best_c5

def simulated_annealing(h_start_half, max_iter=200000):
    """Simulated Annealing with exponential cooling."""
    h_full = np.concatenate([h_start_half, h_start_half[::-1]])
    current_c5 = compute_c5_exact(h_full)
    best_c5 = current_c5
    best_h = h_full.copy()
    
    # Temperature parameters
    T = 1e-2
    alpha = 0.9995
    delta_max = 0.05
    
    accept_count = 0
    total_moves = 0
    temp_history = []
    
    # Initial memory for tracking
    prev_accept_ratio = 0.25
    
    for iter in range(max_iter):
        if iter % 20000 == 0:
            print(f"SA iter {iter}, T={T:.2e}, C5={current_c5:.6f}")
        
        # Swap two random components, adjust by delta
        i, j = np.random.randint(0, half_size, size=2)
        if i == j:
            j = (j + 1) % half_size

        delta = np.random.uniform(0, delta_max)
        
        # Check bounds:
        if h_start_half[i] < delta or h_start_half[j] + delta > 1.0:
            continue
            
        # Do the move
        h_new = h_start_half.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # Project to sum and box
        h_new = project_to_sum_box(h_new, target_sum)
        h_full_new = np.concatenate([h_new, h_new[::-1]])
        
        new_c5 = compute_c5_exact(h_full_new)
        
        # Accept or reject
        if new_c5 < current_c5:
            accept = True
        else:
            # Accept with probability
            accept = np.random.rand() < np.exp(-(new_c5 - current_c5)/T)
            
        if accept:
            current_c5 = new_c5
            h_start_half = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_full_new.copy()
            
        total_moves += 1
        if accept:
            accept_count += 1
            
        # Adjust temperature
        if iter % 100 == 0:
            # Adaptive step size
            if total_moves >= 500:
                ratio = accept_count / min(total_moves, 500)
                if ratio > 0.4:
                    delta_max *= 1.1
                elif ratio < 0.1:
                    delta_max *= 0.9
                delta_max = np.clip(delta_max, 1e-4, 0.2)
                # Also -- adjust T
                T = T * alpha
                if T < 1e-6:
                    break
                # Reset counters
                accept_count = 0
                total_moves = 0

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Let's also run in proper stages: sweep + SA + refine
    
    # Stage 1: Parametric sweep and low-dim refinement
    print("Stage 1: Parametric sweep...")
    h_parametric, c5_parametric = parametric_sweep_and_eco_refinement()
    
    print(f"Stage 1 - Parameter sweep yielded C5 = {c5_parametric}")
    
    if c5_parametric <= 0.379:
        h_final = h_parametric
        c5_final = c5_parametric
        print("Found solution under boundary, finishing early.")
        return (h_final.tolist(), float(c5_final), n_points)
        
    if time.time() - start_time > 0.8 * budget_s:
        print("Early exit for BUDGET")
        return (h_parametric.tolist(), float(c5_parametric), n_points)
    
    # Stage 2: SA refinement
    print("Stage 2: Simulated Annealing...")
    h_init_half = h_parametric[:half_size]
    h_sa, c5_sa = simulated_annealing(h_init_half)
    
    c5_final = c5_sa
    h_final = h_sa
    
    if c5_sa < c5_parametric:
        print(f"SA improved to C5 = {c5_sa}")
    else:
        print(f"SA didn't improve, result: {c5_parametric}")
        h_final = h_parametric
        c5_final = c5_parametric
    
    # Final polishing with projected L-BFGS
    if time.time() - start_time < 0.9 * budget_s:
        print("Stage 3: Final L-BFGS polishing...")
        # Also run projected L-BFGS to refine further
        
        # Use a smooth surrogate that is better for optimization
        def smooth_obj(h_half):
            # Ensure half has correct sum
            h = project_to_sum_box(h_half, target_sum)
            h_full = np.concatenate([h, h[::-1]])
            return compute_c5_smooth(h_full, beta=200.0)
            
        def smooth_grad(h_half):
            h = project_to_sum_box(h_half, target_sum)
            h_full = np.concatenate([h, h[::-1]])
            return gradient_smooth(h_full, beta=200.0)[:half_size]
                        
        try:
            # Run L-BFGS-B on half_vector
            res = minimize(
                smooth_obj, 
                h_init_half.copy(), 
                method='L-BFGS-B', 
                bounds=[(0,1) for _ in range(half_size)], 
                jac=smooth_grad,
                options={'gtol': 1e-12, 'ftol': 1e-12, 'maxiter': 500}
            )
            if res.success:
                h_opt = project_to_sum_box(res.x, target_sum)
                h_opt_full = np.concatenate([h_opt, h_opt[::-1]])
                c5_opt = compute_c5_exact(h_opt_full)
                if c5_opt < c5_final:
                    print(f"L-BFGS improved C5 from {c5_final} to {c5_opt}")
                    h_final = h_opt_full
                    c5_final = c5_opt
        except Exception as e:
            print(f"L-BFGS failed with error: {e}")
            
    print(f"Final result: C5 = {c5_final}")
    
    # Double-check that final is correct
    corr = np.correlate(h_final, 1 - h_final, mode="full")
    actual_c5 = dx * np.max(corr)
    print(f"Final verified C5 = {actual_c5}")
    
    return (h_final.tolist(), float(actual_c5), n_points)
