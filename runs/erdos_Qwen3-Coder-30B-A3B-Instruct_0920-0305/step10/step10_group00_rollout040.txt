import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def upscale_h(original_h, n_new):
    """Upscale original_h to n_new points using linear interpolation"""
    n_old = len(original_h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, original_h)

def propose_move(h, max_step, rng):
    """Propose a move that maintains sum constraint and box constraints"""
    n = len(h)
    # try up to few times to find feasible pair
    for _ in range(10):
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # compute feasible interval for delta added to i and subtracted from j
        lower = max(-h[i], h[j] - 1)
        upper = min(1 - h[i], h[j])
        if lower < upper:
            # limit delta magnitude
            lower = max(lower, -max_step)
            upper = min(upper, max_step)
            if lower >= upper:
                continue
            delta = rng.uniform(lower, upper)
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            return new_h
        # try opposite direction: add to j, subtract from i
        lower = max(-h[j], h[i] - 1)
        upper = min(1 - h[j], h[i])
        if lower < upper:
            lower = max(lower, -max_step)
            upper = min(upper, max_step)
            if lower >= upper:
                continue
            delta = rng.uniform(lower, upper)
            new_h = h.copy()
            new_h[j] += delta
            new_h[i] -= delta
            return new_h
    return None

def simulated_annealing(start_h, time_limit, rng):
    """Run simulated annealing for the given time limit"""
    curr_h = start_h.copy()
    curr_val = evaluate_c5(curr_h)
    best_h_local = curr_h.copy()
    best_val_local = curr_val
    
    start = time.time()
    
    T_max = 0.02
    T_min = 0.0001
    
    step_initial = 0.2
    step_final = 0.001
    
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        fraction = elapsed / time_limit
        
        T = T_max * (1 - fraction) + T_min * fraction
        max_step = step_initial * (1 - fraction) + step_final
        
        new_h = propose_move(curr_h, max_step, rng)
        if new_h is None:
            continue
            
        new_val = evaluate_c5(new_h)
        delta = new_val - curr_val
        
        if delta <= 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            curr_h = new_h
            curr_val = new_val
            if new_val < best_val_local:
                best_val_local = new_val
                best_h_local = curr_h.copy()
    
    return best_h_local, best_val_local

def local_refine(h, max_iter, rng):
    """Perform local refinement with decreasing step size"""
    curr_h = h.copy()
    curr_val = evaluate_c5(curr_h)
    
    step_initial = 0.005
    
    for i in range(max_iter):
        elapsed = i / max_iter
        max_step = step_initial * (1 - elapsed) + 1e-6
        
        new_h = propose_move(curr_h, max_step, rng)
        if new_h is None:
            continue
        
        new_val = evaluate_c5(new_h)
        if new_val < curr_val:
            curr_h = new_h
            curr_val = new_val
    
    return curr_h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Generate initial candidates
    candidates = []
    
    # From existing initial_h_values
    try:
        global initial_h_values
        upsampled_h = upscale_h(initial_h_values, n_points)
        upsampled_h = project_to_sum_box(upsampled_h, TARGET_SUM)
        candidates.append(upsampled_h)
    except:
        pass
    
    # Cosine shape
    xs = np.arange(n_points) * dx
    a = 0.5 - 1e-5
    cos_h = 0.5 + a * np.cos(np.pi * xs / 2)
    cos_h = project_to_sum_box(cos_h, TARGET_SUM)
    candidates.append(cos_h)
    
    # Random uniform
    rand_h = np.random.uniform(0, 1, n_points)
    rand_h = project_to_sum_box(rand_h, TARGET_SUM)
    candidates.append(rand_h)
    
    # Binary random
    binary_h = np.zeros(n_points)
    ones_idx = np.random.choice(n_points, size=n_points // 2, replace=False)
    binary_h[ones_idx] = 1.0
    candidates.append(binary_h)
    
    # Evaluate all candidates
    candidate_values = []
    for h in candidates:
        val = evaluate_c5(h)
        candidate_values.append(val)
        if val < best_c5:
            best_c5 = val
            best_h = h.copy()
    
    # Start SA with the best candidate
    sa_time_limit = budget_s * 0.85
    final_h, final_c5 = simulated_annealing(best_h, sa_time_limit, rng)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h.copy()
    
    # Final local refinement
    refine_time_limit = budget_s * 0.15
    start_refine = time.time()
    
    # We use a fixed number of iterations for refinement
    refinement_iter = 50000
    refined_h = local_refine(best_h, refinement_iter, rng)
    refined_c5 = evaluate_c5(refined_h)
    
    if refined_c5 < best_c5:
        best_c5 = refined_c5
        best_h = refined_h.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
