def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Simple projection with sorting-based algorithm
    x = np.array(x)
    sort_idx = np.argsort(x)[::-1]
    x_sorted = x[sort_idx]
    
    # Water-filling approach
    cumsum = np.cumsum(x_sorted)
    n = len(x_sorted)
    idx = -1
    for i in range(n):
        if cumsum[i] + (n - i - 1) * x_sorted[i] >= target_sum:
            idx = i
            break
    
    if idx == -1:
        # All elements are non-positive
        lambda_val = (target_sum - cumsum[-1]) / n
        result = np.clip(x + lambda_val, 0, 1)
    else:
        lambda_val = (target_sum - cumsum[idx]) / (n - idx)
        result = np.zeros_like(x)
        for i in range(idx):
            result[sort_idx[i]] = x_sorted[i]
        for i in range(idx, n):
            result[sort_idx[i]] = np.clip(x_sorted[i] + lambda_val, 0, 1)
    
    return result

def compute_correlations_binary(x):
    """Compute all correlations for binary x - for optimization/hill climbing"""
    n = len(x)
    correlations = np.zeros(n)
    for k in range(n):
        # For shift k, correlation is sum of products at all valid positions
        # More efficient: x[i] * x[i+k] with i+k precomputed
        corr_sum = 0
        for i in range(n - k):
            corr_sum += x[i] * x[i + k]
        correlations[k] = corr_sum
    return correlations

def swap_update_correlations(O, x, a, b):
    """Update correlations after swapping positions a and b"""
    new_O = O.copy()
    
    # Update for shifts where a participates
    for k in range(len(x) - a):
        # Before swap: term a,k was x[a]*x[a+k]
        # After swap: term a,k is x[a]*x[a+k], but now x[a]=0 => contribution becomes 0
        # We reduce the sum by x[a]*x[a+k] 
        if x[a] == 1 and x[a + k] == 1:
            new_O[k] -= 1

    # Update for shifts where b participates  
    for k in range(len(x) - b):
        # Before swap: term b,k was x[b]*x[b+k]
        # After swap: term b,k is x[b]*x[b+k], but now x[b]=1 => contribution becomes 1
        # We increase the sum by x[b]*x[b+k]
        if x[b] == 0 and x[b + k] == 1:
            new_O[k] += 1

    # Also handle the shift k where k = b - a if we're swapping a with b
    # But the above approach should handle all.
    return new_O

    iterations = 0

        # Find positions with 1 that lack partner at k_max 
        candidate_A = []
        for i in range(len(x) - k_max):
            if x[i] == 1 and x[i + k_max] == 0:

        # Find positions with 0 that match partner at k_max  
        candidate_B = []
        for i in range(len(x) - k_max):
            if x[i] == 0 and x[i + k_max] == 1:
                candidate_B.append(i)

        # Early termination
        if best_D <= 0 or len(candidate_A) == 0 or len(candidate_B) == 0:
            break

        if len(candidate_A) > 0 and len(candidate_B) > 0:
            # Try a random swap
            a = np.random.choice(candidate_A)
            b = np.random.choice(candidate_B)
            
            # Create new x with swap
            x_new = x.copy()
            x_new[a] = 0
            x_new[b] = 1
            
            # Compute new correlation

            if new_max < best_D:
                # Accept the swap
                x = x_new
                O = O_new

            elif np.random.rand() < 0.03:  # Accept with probability even if worse to escape local minima
                x = x_new
                O = O_new
                D = D_new
                best_D = new_max
                swaps += 1
                
        k_max = np.argmax(D)
        iterations += 1
    
    # Final check with continuous eval
    h_continuous = x.astype(float)  # Ensure float type for compatibility with other functions
    return h_continuous, evaluate_c5(h_continuous)

def upsample(binary_vec, target_N):
    """Upsample binary vector to target resolution"""
    # Replicate each element to match target resolution
    upsample_factor = target_N // len(binary_vec)
    if upsample_factor == 0:
        upsample_factor = 1
    
    upscaled = np.repeat(binary_vec, upsample_factor)
    if len(upscaled) < target_N:
        # Pad with zeros or ones to match exactly
        pad_size = target_N - len(upscaled)
        upscaled = np.pad(upscaled, (0, pad_size), constant_values=0)
    elif len(upscaled) > target_N:
        upscaled = upscaled[:target_N]
    return upscaled

def differential_evolution_refinement(h_start, target_sum, time_budget, start_time):
    """Refine continuous solution with DE"""
    try:
        def objective(h):
            # Ensure sum constraint by normalizing (fit to be exactly correct)
            h = np.array(h)
            h_norm = h * target_sum / np.sum(h)
            return evaluate_c5(h_norm)
        
        bounds = [(0, 1) for _ in range(len(h_start))]
        
        # Run DE for a limited number of iterations
        res = differential_evolution(
            objective, 
            bounds, 
            maxiter=int(max(100, time_budget * 5)), 
            popsize=20, 
            mutation=(0.5, 1),
            recombination=0.7,
            seed=42,
            disp=False
        )
        
        if res.success:
            h_result = res.x * target_sum / np.sum(res.x)
            return h_result, evaluate_c5(h_result)
    except:
        pass
    
    return h_start, evaluate_c5(h_start)
