import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fft import ifft, fft
import time

# Global Variables
n_points = 200
dx = 2.0 / n_points
half_size = n_points // 2
target_sum = n_points / 4

def autocorr_fft(h):
    """Fast computation of autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(h, L)).real
    return corr

def evaluate_c5(h):
    """Exact C5 calculation using max-over-k"""
    corr = autocorr_fft(h)
    amin = np.min(corr)
    return 1 - dx * amin

def project_to_sum_box(x, target_sum):
    """Project vector to box [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_objective(h, beta=50.0):
    """Smooth approximation to C5 using log-sum-exp for the min over k"""
    corr = autocorr_fft(h)
    # Shift to prevent overflow
    corr_shifted = corr - np.max(corr)
    weights = np.exp(beta * corr_shifted)
    w_sum = np.sum(weights)
    smoothed = 1 - (dx / beta) * np.log(w_sum)
    return smoothed

def smooth_gradient(h, beta=50.0):
    """Gradient of smooth_objective with respect to h"""
    corr = autocorr_fft(h)
    corr_shifted = corr - np.max(corr)
    weights = np.exp(beta * corr_shifted)
    w_sum = np.sum(weights)
    # Compute gradient using convolution trick
    # d/dh_i (sum_k w_k) = sum_k (d w_k / d h_i) = sum_k w_k * beta * d A_k / dh_i
    # dA_k / dh_i = h_{i+k} + h_{i-k}
    # Gradient per-element using 2D convolution equivalent in (h' * h) process (reversed definition)
    # But since we want.grad w.r.t. h_i, we convolve weights with h_rev and then ...
    # We use a more direct method by reusing FFT and mathematical expressions.
    # Actually, just convolve h with w * beta * [1,1] (2 really low-pass for derivative approx)
    # Or compute via standard formula:
    # -dx * sum_k (w_k * (h_{i+l} + h_{i-l}) ) / sum_k w_k (for shift l)
    # But easier to apply chain rule on convolution:

    # Easier approach: derivative of log-sum-exp reference
    # gradient = -dx / w_sum * sum(weights * d(A_k)/dh_i)
    # where dA_k/dh_i = h_{i+k} + h_{i-k}
    # So, sum of weight * (h at right and left shifts)
    # Single convolve that pleased:
    # Efficient to compute gradient using double FFT:
    # We need to compute FFT of [w_k] and FFT of h_rev and then the reverse conv.
    # This is: sum_k w_k h_{i+k} = ...which computes via conv Standard.
    # In fact: cross-correlation of w and h (reverse).
    
    # Manually:
    # For i-th element, gradient = -dx * (sum_k w_k * (h_{i+k} + h_{i-k})) / w_sum
    # We compute (d/dh_i of smooth) = -dx * sum of (w_k * derivative of A_k) / w_sum
    # dA_k/dh_i = h_{i+k} + h_{i-k} 

    # Better solution:
    # Build d_{k,i} = (h_{i+k} + h_{i-k}) used in:  sum_k w_k * d_{k,i} 
    # Compute for k = -L+1 to L-1 where h is assumed zero padded
    # Better: use 2*n-1 sized vectors, standard convolution flipped convention
    # conv(h_rev, w)  yields same.  And it only depends on h_rev not full h.
    # Let us build the vectorized direct gradient expression.

    # Actually avoid duplicating code, adapt it:
    # For size n=200, let's compute gradient of smooth objective directly for one vector
    # Let us compute gradient for (after 2nd term of main idea):
    df = np.zeros_like(h)
    for i in range(len(h)):
        total = 0.
        for k in range(-len(h)+1, len(h)):
            if 0 <= i+k < len(h):
                total += weights[k + len(h)-1] * h[i+k]  # index k in corr vector is -> k+len(h)-1
        # and we need neighbors in other direction? No. A_k = sum h_i h_{i+k}
        # So dA_k/dh_i = h_{i+k} (as k is fixed per index) 
        # but in the grad sum, over k:
        # sum_k w_k * dA_k/dh_i = w_k * h_{i+k}
        # In the same way, same shift index impacts both terms (the linearity)  
        # To be clearer: 
        # A_k = sum h_i h_{i+k}  
        # dA_k/dh_i = h_{i+k} 
        # Using chain rule on log-sum-exp form per h_i:
        # Gibbs' derivative with stationary weights
        # usual (chain rule)
        # d/dh_i (sum exp (beta A_k) ) = sum_k beta exp(beta A_k) [dA_k/dh_i] 
        #              = sum_k beta w_k h_{i+k}  = beta w * h_{+k shift}
        # gradient[i] = -dx * (beta * (sum_k w_k h_{i+k}) ) / sum_k w_k 
        #              = - dx * [sum_k w_k h_{i+k}] / w_sum

        # prepare for conv: we use circulant dimension to match FFT sizes
        # don't write full loop for many super inefficient use of python,
        # instead precompute weights-to-use as array and convolve or use 
        # full fft power directly.
        # Simpler when precompute then better.
        # can do:
        # h_embed = np.zeros(2*n-1); h_embed[n-1:n-1+len(h)] = h
        # w_rev = w[::-1]  (bondary concerned yet?)
        # since h a vector of length 200, we embed in vec of 399 size 
        # w_embed = np.zeros_like(h_embed)
        # for i,... 
        # build both in the same manner; here is the cleanest for both "direct" and convolution:
        
        # Assume precomputed correlation valid for full size.
        # Let's avoid expensive code; use monte carlo derivative imprecise ideas for now.
        
        # Construct simple summation:
        final_term = 0.0
        for k in range(-len(h)+1,len(h)):
            offset = i + k
            if 0 <= offset < len(h):
                final_term += weights[k + len(h) - 1] * h[offset]
        df[i] = final_term / w_sum
    return -dx * df

def low_dim_sweep():
    """Sweep over model parameters a (edge length)"""
    best_a = 0.0
    best_C5 = 1.0
    best_h = None
    
    a_vals = np.linspace(0.01, 0.49, 250)
    for a in a_vals:
        # Compute middle block height p
        if 1 - 2*a <= 0:
            continue
        p = (1 - 2*a) / (2 * (1 - a))
        if p < 0 or p > 1:
            continue
            
        # Construct half-vector h_half
        h_half = np.zeros(half_size)
        h_half[:int(a*dx*half_size)] = 1.0
        h_half[int(a*dx*half_size):int((1-a)*dx*half_size)] = p
        h_half[int((1-a)*dx*half_size):] = 1.0

        # Fix any defects (non integer for mid, etc) from simple float model
        h_half = project_to_sum_box(h_half, target_sum)
        full_h = np.concatenate([h_half, h_half[::-1]])
        c5 = evaluate_c5(full_h)
        
        if c5 < best_C5:
            best_C5 = c5
            best_a = a
            best_h = full_h
            
    return best_h, best_C5

def refine_with_lbfgsb(initial_h):
    """Refine using smooth objective and projected L-BFGS-B"""
    # Use only the half vector and regard symmetry
    h_half = initial_h[:half_size]    
    
    def smooth_obj(x):
        # reconstruct full vec
        h_full = np.concatenate([x, x[::-1]])
        return smooth_objective(h_full, beta=50.0)
    
    def smooth_grad(x):
        h_full = np.concatenate([x, x[::-1]])
        gradient = smooth_gradient(h_full, beta=50.0)
        return gradient[:half_size]
    
    bounds = [(0, 1) for _ in range(half_size)]
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    res = minimize(
        smooth_obj,
        h_half,
        method='L-BFGS-B',
        jac=smooth_grad,
        bounds=bounds,
        constraints=constraints,
        options={'ftol': 1e-12, 'gtol': 1e-12, 'maxiter': 1000}
    )
    
    if res.success:
        best_h_half = res.x
        full_h = np.concatenate([best_h_half, best_h_half[::-1]])
        return full_h
    else:
        # fallback
        return initial_h

def simulated_annealing(initial_h, budget_s):
    """Perform simulated annealing on the full half-vector"""
    h_curr = initial_h[:half_size].copy()
    start_time = time.time()
    
    T = 0.01
    T_min = 1e-6
    alpha = 0.9995
    delta_max = 0.05
    accept_count = 0
    total_moves = 0
    
    best_h = h_curr.copy()
    best_C5 = evaluate_c5(np.concatenate([h_curr, h_curr[::-1]]))
    
    # Convert to full vector for evaluation
    curr_full = np.concatenate([h_curr, h_curr[::-1]])
    curr_C5 = evaluate_c5(curr_full) 
    
    def move(h, delta_max):
        # Pick two indices
        i, j = np.random.choice(half_size, 2, replace=False)
        delta = np.random.uniform(0, delta_max)
        
        # Try to push delta from i to j while maintaining bounds
        if h[i] >= delta and h[j] + delta <= 1:
            h[i] -= delta
            h[j] += delta
            return True
        return False
    
    while T > T_min and time.time() - start_time < budget_s * 0.8:
        # Make a move
        if move(h_curr, delta_max):
            total_moves += 1
            full_h = np.concatenate([h_curr, h_curr[::-1]])
            new_C5 = evaluate_c5(full_h)
            
            # Accept or reject
            if new_C5 < curr_C5:
                curr_C5 = new_C5
                curr_full = full_h
                accept_count += 1
            else:
                if np.random.rand() < np.exp(-(new_C5 - curr_C5) / T):
                    curr_C5 = new_C5
                    curr_full = full_h
                    accept_count += 1
                else:
                    # Revert the move
                    h_curr[i] += delta
                    h_curr[j] -= delta
            
            # Keep the best
            if curr_C5 < best_C5:
                best_C5 = curr_C5
                best_h = h_curr.copy()
                
        # Update temperature and adapt step size
        T *= alpha
        if total_moves % 500 == 0 and total_moves > 0:
            acc_rate = accept_count / 500.0
            if acc_rate > 0.4:
                delta_max *= 1.1
            elif acc_rate < 0.1:
                delta_max *= 0.9
            delta_max = np.clip(delta_max, 1e-4, 0.2)
            accept_count = 0
    
    # Return full vector
    return np.concatenate([best_h, best_h[::-1]]), best_C5

def final_refinement(h_start):
    """Final refinement on the full vector with exact smooth objective"""
    h_half = h_start[:half_size]
    full_h = np.concatenate([h_half, h_half[::-1]])
    
    def obj_exact(h):
        return evaluate_c5(h)
        
    # Recompute to improve proximity to smooth model convergence
    def smooth_obj(x):
        full_vec = np.concatenate([x, x[::-1]])
        return smooth_objective(full_vec, beta=200)
    
    def smooth_grad(x):
        full_vec = np.concatenate([x, x[::-1]])
        gradient = smooth_gradient(full_vec, beta=200)
        return gradient[:half_size]
    
    bounds = [(0, 1) for _ in range(half_size)]
    
    # Optimize with higher β
    res = minimize(
        smooth_obj,
        h_half,
        method='L-BFGS-B',
        jac=smooth_grad,
        bounds=bounds,
        options={'ftol': 1e-12, 'gtol': 1e-12, 'maxiter': 500}
    )

    if res.success:
        final_h_half = res.x
        final_full = np.concatenate([final_h_half, final_h_half[::-1]])
        final_c5 = evaluate_c5(final_full)
        return final_full, final_c5
    else:
        return h_start, evaluate_c5(h_start)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage I: Parameter sweep
    h_start, c5_initial = low_dim_sweep()
    if time.time() - start_time > budget_s * 0.5:
        return (h_start.tolist(), float(c5_initial), n_points)
    
    # Stage II: Local refinement
    h_refined = refine_with_lbfgsb(h_start)
    c5_refined = evaluate_c5(h_refined)
    
    if time.time() - start_time > budget_s * 0.7:
        return (h_refined.tolist(), float(c5_refined), n_points)
    
    # Stage III: Simulated Annealing
    h_sa, c5_sa = simulated_annealing(h_refined, budget_s)
    if time.time() - start_time > budget_s * 0.9:
        return (h_sa.tolist(), float(c5_sa), n_points)
    
    # Stage IV: Final polishing
    h_final, c5_final = final_refinement(h_sa)
    
    return (h_final.tolist(), float(c5_final), n_points)
