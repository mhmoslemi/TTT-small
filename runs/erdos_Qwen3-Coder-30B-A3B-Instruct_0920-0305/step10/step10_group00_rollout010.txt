import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def upscale_h(original_h, n_new):
    """Linearly interpolate original_h to new length n_new"""
    n_old = len(original_h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, original_h)

def propose_move(h, max_step, rng):
    """Propose a move that maintains sum and box constraints"""
    n = len(h)
    # Try up to few times to find feasible pair
    for _ in range(10):
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # compute feasible interval for delta added to i and subtracted from j
        lower = max(-h[i], h[j] - 1)
        upper = min(1 - h[i], h[j])
        if lower < upper:
            # limit delta magnitude
            lower = max(lower, -max_step)
            upper = min(upper, max_step)
            if lower >= upper:
                continue
            delta = rng.uniform(lower, upper)
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            return new_h
        # try opposite direction: add to j, subtract from i
        lower = max(-h[j], h[i] - 1)
        upper = min(1 - h[j], h[i])
        if lower < upper:
            lower = max(lower, -max_step)
            upper = min(upper, max_step)
            if lower >= upper:
                continue
            delta = rng.uniform(lower, upper)
            new_h = h.copy()
            new_h[j] += delta
            new_h[i] -= delta
            return new_h
    return None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    
    # Initialize
    best_c5 = float('inf')
    best_h = None
    
    # Generate initial candidates
    candidates = []
    
    # 1. From provided initial_h_values (upgraded to n_points)
    try:
        global initial_h_values
        upsampled = upscale_h(initial_h_values, n_points)
        upsampled = project_to_sum_box(upsampled, TARGET_SUM)
        candidates.append(upsampled)
    except:
        pass
    
    # 2. Cosine shape
    xs = np.arange(n_points) * dx
    a = 0.5 - 1e-5
    cos_h = 0.5 + a * np.cos(np.pi * xs / 2)
    cos_h = project_to_sum_box(cos_h, TARGET_SUM)
    candidates.append(cos_h)
    
    # 3. Random uniform
    rand_h = np.random.uniform(0, 1, n_points)
    rand_h = project_to_sum_box(rand_h, TARGET_SUM)
    candidates.append(rand_h)
    
    # 4. Binary with exact half ones
    binary_h = np.zeros(n_points)
    ones_idx = np.random.choice(n_points, size=n_points//2, replace=False)
    binary_h[ones_idx] = 1.0
    candidates.append(binary_h)
    
    # Evaluate candidates and find the best one
    for h in candidates:
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Simulated annealing
    curr_h = best_h.copy()
    curr_c5 = best_c5
    
    # SA parameters
    T_max = 0.02
    T_min = 0.0001
    step_initial = 0.1
    step_final = 0.001
    max_iter = 500000
    
    iter_count = 0
    time_limit = budget_s * 0.9
    while time.time() - start_time < time_limit and iter_count < max_iter:
        iter_count += 1
        elapsed = time.time() - start_time
        fraction = elapsed / time_limit
        T = T_max * (1 - fraction) + T_min * fraction
        max_step = step_initial * (1 - fraction) + step_final
        
        new_h = propose_move(curr_h, max_step, rng)
        if new_h is None:
            continue
            
        new_c5 = evaluate_c5(new_h)
        
        if new_c5 < curr_c5:
            curr_h = new_h
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = curr_h.copy()
        else:
            if np.random.random() < np.exp(-(new_c5 - curr_c5) / (T + 1e-12)):
                curr_h = new_h
                curr_c5 = new_c5
    
    # Local refinement
    refine_iter = 50000
    for i in range(refine_iter):
        if time.time() - start_time > time_limit:
            break
            
        elapsed = (time.time() - start_time - time_limit * 0.9) / (time_limit * 0.1)
        max_step = step_initial * (1 - elapsed) + step_final
        
        new_h = propose_move(best_h, max_step, rng)
        if new_h is None:
            continue
            
        new_c5 = evaluate_c5(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
