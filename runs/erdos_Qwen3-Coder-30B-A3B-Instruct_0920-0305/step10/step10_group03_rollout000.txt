import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def construct_better_initial():
    """Construct a better initial solution based on symmetry and structure"""
    h = np.zeros(n_points)
    
    # Create a piecewise constant function with better structure
    # A good starting point is to have a higher concentration in center
    mid = n_points // 2
    width = n_points // 4
    
    # Central peak with gradual decreases
    for i in range(n_points):
        if abs(i - mid) < width:
            h[i] = 0.5 + 0.3 * np.cos(np.pi * (i - mid) / width)
        else:
            h[i] = max(0.1, 0.5 - 0.3 * (abs(i - mid) - width) / (n_points/2 - width))
    
    # Project to ensure correct sum
    return project_to_sum_box(h, TARGET_SUM)

def improved_local_search(h0, iterations=20000):
    """Enhanced local search with multiple perturbation types"""
    current_h = h0.copy()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    for i in range(iterations):
        h_new = current_h.copy()
        
        # Try different perturbation strategies
        if i % 3 == 0:
            # Large perturbation
            indices = np.random.choice(n_points, size=10, replace=False)
            for idx in indices:
                h_new[idx] += np.random.normal(0, 0.05)
        elif i % 3 == 1:
            # Medium perturbation
            indices = np.random.choice(n_points, size=20, replace=False)
            for idx in indices:
                h_new[idx] += np.random.normal(0, 0.02)
        else:
            # Small perturbation
            indices = np.random.choice(n_points, size=50, replace=False)
            for idx in indices:
                h_new[idx] += np.random.normal(0, 0.005)
        
        # Project to valid constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new)
        
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = current_h.copy()
                best_c5 = c5_new
        elif np.random.random() < 0.01:  # Occasionally accept worse solutions
            current_h = h_new.copy()
            current_c5 = c5_new
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Start with better initial construction
    h0 = construct_better_initial()
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Strategy 1: Enhanced local search with multiple perturbation types
    if time.time() - start_time < budget_s * 0.5:
        h_local, c5_local = improved_local_search(h0, 10000)
        if c5_local < best_c5:
            best_h = h_local
            best_c5 = c5_local
    
    # Strategy 2: Random restarts with better initial points
    for i in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
        # Generate better random starting point
        h_random = np.random.rand(n_points)
        h_random = project_to_sum_box(h_random, TARGET_SUM)
        h_rest, c5_rest = improved_local_search(h_random, 3000)
        if c5_rest < best_c5:
            best_h = h_rest
            best_c5 = c5_rest
    
    # Strategy 3: One more enhanced local search
    if time.time() - start_time < budget_s:
        h_final, c5_final = improved_local_search(best_h, 5000)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    
    return (best_h.tolist(), float(best_c5), n_points)
