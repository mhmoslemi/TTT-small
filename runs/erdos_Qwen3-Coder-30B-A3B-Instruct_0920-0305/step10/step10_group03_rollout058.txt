import numpy as np

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_skeleton_search(h0, max_iter=1000):
    """Search using binary skeleton approach"""
    current_h = h0.copy()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    for i in range(max_iter):
        # Generate binary skeleton
        binary_skeleton = (current_h > 0.5).astype(int)
        
        # Perturb bits
        perturbed = binary_skeleton.copy()
        idx = random.randint(0, n_points - 1)
        perturbed[idx] = 1 - perturbed[idx]
        
        # Convert back to step function
        h_new = np.zeros(n_points)
        if np.sum(perturbed) > 0:
            h_new[perturbed.astype(bool)] = 1.0
            
        # Project to sum constraint
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        c5_new = evaluate_c5(h_new)
        
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = current_h.copy()
                best_c5 = c5_new
        elif random.random() < 0.05:
            current_h = h_new.copy()
            current_c5 = c5_new
        
    return best_h, best_c5

def parametric_shape_search(h0, max_iter=5000):
    """Search for optimal step function shape using parametric approach"""
    current_h = h0.copy()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5

        h_new = current_h.copy()
        
        # Select a random segment to adjust
        seg_start = random.randint(0, n_points - 5)

        # Adjust the segment values together
        adjustment = random.uniform(-0.1, 0.1)

            h_new[j] = np.clip(h_new[j] + adjustment, 0, 1)

        h_new = project_to_sum_box(h_new, TARGET_SUM)

def differential_evolution(h0, max_iter=3000):
    """Use differential evolution for global search"""
    # Parameters
    pop_size = 20
    F = 0.8
    CR = 0.9
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_c5(h) for h in population]
    
    # Evolution loop
    for i in range(max_iter):
        # For each individual
        for j in range(pop_size):
            # Select three different individuals
            indices = [k for k in range(pop_size) if k != j]
            a, b, c = [population[k] for k in random.sample(indices, 3)]
            
            # Mutation
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover (Binomial)
            trial = np.zeros(n_points)
            for k in range(n_points):
                if random.random() < CR or k == random.randint(0, n_points - 1):
                    trial[k] = mutant[k]
                else:
                    trial[k] = population[j][k]
            
            # Project to constraint
            trial = project_to_sum_box(trial, TARGET_SUM)
            
            # Selection
            trial_fitness = evaluate_c5(trial)
            if trial_fitness < fitness[j]:
                population[j] = trial
                fitness[j] = trial_fitness
    
    # Return best solution
    best_idx = np.argmin(fitness)
    return population[best_idx], fitness[best_idx]

def local_polish(h0, max_iter=2000):
    """Simple local polish using gradient-free method"""
    current_h = h0.copy()
    current_c5 = evaluate_c5(current_h)
    
    # Small perturbations with acceptance
    for i in range(max_iter):
        h_new = current_h.copy()
        
        # Small random adjustments
        idx = random.randint(0, n_points - 1)
        h_new[idx] = np.clip(h_new[idx] + random.uniform(-0.01, 0.01), 0, 1)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        c5_new = evaluate_c5(h_new)
        
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
    
    return current_h, current_c5
