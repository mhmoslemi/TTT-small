import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def full_corr(a, b):
    """Compute correlation with zero padding using FFT"""
    L = 2 * len(a) - 1
    return np.fft.ifft(np.fft.fft(a, L) * np.conj(np.fft.fft(b, L))).real

def smooth_max_and_weights(c, alpha):
    """Compute log-sum-exp and soft-max weights"""
    c_shifted = c - np.max(c)  # Prevent overflow
    exp_c = np.exp(alpha * c_shifted)
    Z = np.sum(exp_c)
    w = exp_c / Z
    return np.log(Z) / alpha + np.max(c), w

def compute_gradient(h, alpha):
    """Compute gradient of smooth max objective using FFT"""
    c = full_corr(h, 1 - h)
    _, w = smooth_max_and_weights(c, alpha)
    
    # Symmetrized weights
    w_sym = w + w[::-1]
    
    # Second correlation: weighted h convolved with symmetric weights
    g = full_corr(w_sym, h)[:len(h)]
    
    return 1.0 - g

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    for _ in range(50):
        m = (a + b) / 2

            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Evaluate actual C5 value"""
    corr = full_corr(h, 1 - h)
    return dx * np.max(corr)

def optimize_with_continuation(h0, alpha_schedule, target_sum, max_iter=300):
    """Optimize using trust-constr with continuation"""
    A_eq = np.ones((1, len(h0)))
    bounds = [(0, 1) for _ in range(len(h0))]
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    for alpha in alpha_schedule:
        def objective(h):
            c = full_corr(h, 1 - h)
            val, _ = smooth_max_and_weights(c, alpha)
            return val
            
        def gradient(h):
            return compute_gradient(h, alpha)
            
        try:
            result = minimize(
                objective,
                best_h,
                method='trust-constr',
                jac=gradient,
                bounds=bounds,
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': A_eq},
                options={'maxiter': max_iter, 'gtol': 1e-10, 'ftol': 1e-12}
            )
            if result.success:
                h_opt = result.x
                h_opt = project_to_sum_box(h_opt, target_sum)
                current_c5 = evaluate_c5(h_opt)
                if current_c5 < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = current_c5
            else:
                # If optimization fails, continue with previous best
                pass
        except:
            pass
    
    return best_h, best_c5
