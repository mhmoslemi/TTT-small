import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Phase 1: Binary MILP on a smaller grid
    if time.time() - start_time < budget_s * 0.05:
        n_binary = 60
        dx_binary = 2.0 / n_binary
        try:
            # Create binary variables
            x = cp.Variable(n_binary, boolean=True)

            # Approximate the correlation constraints using auxiliary variables
            # This is a huge simplification, just assuming a basic structure
            # Actual MILP would be more involved but this is for illustration
            # For now let's directly solve using cvxpy with a smaller problem
            
            # Create a problem from the binary search, but we'll make it simpler

            h0 = np.zeros(n_binary)
            for i in range(n_binary):
                h0[i] = int(i < n_binary/2)
            
            # Use the initial values if they are good
            h0 = np.array(initial_h_values[:n_binary])
            h0 = project_to_sum_box(h0, TARGET_SUM / 2)
            
            # Upsample to full resolution

            h0_final = h_upsampled[:n_points]
            h0_final = project_to_sum_box(h0_final, TARGET_SUM)
            
            best_h = h0_final.copy()
            best_c5 = evaluate_c5(h0_final)
            
        except Exception as e:
            # Fallback to a random binary solution
            h_random = np.random.rand(n_points)
            h_random = project_to_sum_box(h_random, TARGET_SUM)
            best_h = h_random.copy()
            best_c5 = evaluate_c5(h_random)
    
    # Phase 2: Differential Evolution phase
    if time.time() - start_time < budget_s * 0.9:
        # Start with warm start
        if best_h is None:
            h0 = np.array(initial_h_values)
            h0 = project_to_sum_box(h0, TARGET_SUM)
            best_h = h0.copy()
            best_c5 = evaluate_c5(h0)
        
        # DE parameters
        popsize = 30
        maxiter = 500
        F = 0.8
        CR = 0.9
        population = []
        fitness = []
        
        # Initialize population
        for i in range(popsize):
            if i == 0:
                h = best_h.copy()
            else:
                h = np.random.rand(n_points)
                h = project_to_sum_box(h, TARGET_SUM)
            population.append(h)
            fitness.append(evaluate_c5(h))
        
        # DE iterations
        for iteration in range(maxiter):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            new_population = []
            new_fitness = []
            
            for i in range(popsize):
                # Select three distinct individuals at random
                idxs = [k for k in range(popsize) if k != i]
                a, b, c = [population[k] for k in np.random.choice(idxs, 3, replace=False)]
                
                # Mutation
                mutant = a + F * (b - c)
                mutant = np.clip(mutant, 0, 1)
                
                # Crossover
                trial = np.copy(population[i])
                j_rand = np.random.randint(n_points)
                for j in range(n_points):
                    if np.random.rand() < CR or j == j_rand:
                        trial[j] = mutant[j]
                
                # Project to feasible region
                trial = project_to_sum_box(trial, TARGET_SUM)
                
                # Evaluate
                trial_fitness = evaluate_c5(trial)
                
                # Selection
                if trial_fitness <= fitness[i]:
                    new_population.append(trial)
                    new_fitness.append(trial_fitness)
                else:
                    new_population.append(population[i])
                    new_fitness.append(fitness[i])
            
            population = new_population
            fitness = new_fitness
            
            # Update best solution
            for j, fit in enumerate(fitness):
                if fit < best_c5:
                    best_c5 = fit
                    best_h = population[j].copy()
    
    # Phase 3: If not yet below threshold, do a final deterministic hill climbing
    if time.time() - start_time < budget_s:
        # Run deterministic hill climbing
        current_h = best_h.copy()
        current_c5 = best_c5
        for _ in range(10000):
            if time.time() - start_time > budget_s - 5:
                break
                
            h_new = current_h.copy()
            # Pick random indices to perturb
            indices = np.random.choice(n_points, size=min(20, n_points//10), replace=False)
            for idx in indices:
                # Add small random perturbation
                delta = np.random.normal(0, 0.01)
                h_new[idx] += delta
            
            # Project to constraints
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            c5_new = evaluate_c5(h_new)
            
            if c5_new < current_c5:
                current_h = h_new.copy()
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = current_h.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
