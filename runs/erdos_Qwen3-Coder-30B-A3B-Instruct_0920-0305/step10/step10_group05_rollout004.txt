import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

# Global constants
n_points = 200
dx = 2.0 / n_points
half_size = n_points // 2
target_sum = n_points / 4  # Half of the total area to satisfy sum constraint

def autocorr_fft(h):
    """Compute autocorrelation using FFT."""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value."""
    return dx * (1 - np.min(corr))

def compute_smooth_objective(h, beta=50):
    """Compute smooth surrogate for C5 using log-sum-exp."""
    corr = autocorr_fft(h)
    # Subtract maximum to prevent overflow
    max_corr = np.max(corr)
    weights = np.exp(beta * (1 - (corr - max_corr)))
    log_sum_exp = max_corr + np.log(np.sum(weights)) / beta
    return dx * (1 - log_sum_exp)

def compute_smooth_gradient(h, beta=50):
    """Compute gradient of smooth objective using convolutions."""
    corr = autocorr_fft(h)
    max_corr = np.max(corr)
    weights = np.exp(beta * (1 - (corr - max_corr)))
    weight_sum = np.sum(weights)
    
    # Compute numerator: sum of weights * derivative of correlation
    # Derivative of correlation w.r.t. h_i is h_{i+k} + h_{i-k}
    # We calculate weighted sum of all h_j s.t. they appear in the derivative
    # This can be computed with convolution: convolve weights with reversed h
    
    weights_reversed = weights[::-1]
    L = len(h) * 2 - 1
    num = ifft(fft(weights_reversed, L) * fft(h, L)).real
    
    # Actual gradient for each i
    grad = -dx * num / weight_sum
    
    return grad

def project_to_sum_box(x, target_sum):
    """Project vector x onto [0,1] box with sum = target_sum."""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def build_symmetric_step(a):
    """Build symmetric step function with:
       - two edge blocks of length a (height=1)
       - one center block of length (1-2*a) with height p
    """
    # Compute p from area constraint
    if a >= 0.5:
        p = 0.0  # avoid divide by zero
    else:
        p = (1 - 2 * a) / (2 * (1 - a))
    
    # length of edge block = a*2 = 2*a
    # length of middle block = 2 - 2*a
    half_h = []
    left_edge_len = int(np.floor(2 * a * half_size))
    mid_len = half_size - 2 * left_edge_len
    
    half_h.extend([1.0] * left_edge_len)  # left edge
    half_h.extend([p] * mid_len)          # middle
    half_h.extend([1.0] * left_edge_len)  # right edge
    
    # Adjust for off-by-one due to truncation
    while len(half_h) < half_size:
        half_h.append(1.0)
    for i in range(len(half_h) - half_size):
        half_h.pop()
    
    return np.array(half_h)

def sweep_parameter_scan():
    """Sweep the parameter 'a' to find a good starting point."""
    best_c5 = float('inf')
    best_h = None
    best_a = None
    
    # Sweep a from 0.01 to 0.49 (not too close to boundaries)
    sweep_points = np.arange(0.01, 0.49, 0.002)
    
    for a in sweep_points:
        h = build_symmetric_step(a)
        h = project_to_sum_box(h, target_sum)
        corr = autocorr_fft(h)
        c5 = c5_from_corr(corr, n_points)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_a = a
    
    return best_h, best_c5, best_a

def low_dimensional_optimize(h_initial):
    """Optimize using smooth objective and L-BFGS-B on low dimension."""
    def smooth_func(h):
        return compute_smooth_objective(h, beta=100)
    def smooth_grad(h):
        return compute_smooth_gradient(h, beta=100)
    
    # Use bounded L-BFGS-B with bounds
    bounds = [(0, 1) for _ in range(half_size)]
    result = minimize(smooth_func, h_initial, method='L-BFGS-B', jac=smooth_grad, bounds=bounds, options={'ftol': 1e-12, 'gtol': 1e-12})
    
    if result.success:
        h_opt = project_to_sum_box(result.x, target_sum)
        return h_opt
    else:
        return project_to_sum_box(h_initial, target_sum)

def simulated_annealing(h_initial, budget_s, start_time):
    """Run simulated annealing on the full vector."""
    current_h = h_initial.copy()
    current_c5 = c5_from_corr(autocorr_fft(current_h), n_points)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    T = 0.01
    alpha = 0.9995
    delta_max = 0.05
    accepted_count = 0
    total_count = 0
    
    while T >= 1e-6 and time.time() - start_time < budget_s * 0.85:
        # Pick two indices
        i = np.random.randint(0, half_size)
        j = np.random.randint(0, half_size)
        if i == j:
            continue
            
        # Try a mass transfer move
        delta = np.random.uniform(0, delta_max)
        
        if current_h[i] >= delta and current_h[j] <= 1 - delta:
            new_h = current_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            
            # Project to ensure constraints
            new_h = project_to_sum_box(new_h, target_sum)
            
            # Evaluate
            corr = autocorr_fft(new_h)
            new_c5 = c5_from_corr(corr, n_points)
            
            # Accept or reject
            if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                current_h = new_h
                current_c5 = new_c5
                accepted_count += 1
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
            
            total_count += 1
            # Adaptive delta
            if total_count % 500 == 0:
                acceptance_rate = accepted_count / max(1, total_count)
                if acceptance_rate > 0.4:
                    delta_max = min(0.2, delta_max * 1.1)
                elif acceptance_rate < 0.1:
                    delta_max = max(1e-4, delta_max * 0.9)
                accepted_count = 0
                total_count = 0
        
        T *= alpha
    
    if time.time() - start_time > budget_s * 0.85:
        return best_h, best_c5
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage I: parameter sweep to find initial good solution
    h0, c5_0, best_a = sweep_parameter_scan()
    
    # Stage II: refine with L-BFGS-B
    h1 = low_dimensional_optimize(h0)
    corr = autocorr_fft(h1)
    c5_1 = c5_from_corr(corr, n_points)
    
    # Stage III: SA to escape local minima
    h2, c5_2 = simulated_annealing(h1, budget_s, start_time)
    
    # Stage IV: final polishing with projected L-BFGS
    def smooth_func(h):
        return compute_smooth_objective(h, beta=200)
    def smooth_grad(h):
        return compute_smooth_gradient(h, beta=200)
    
    bounds = [(0, 1) for _ in range(half_size)]
    result = minimize(smooth_func, h2, method='L-BFGS-B', jac=smooth_grad, bounds=bounds, options={'ftol': 1e-12, 'gtol': 1e-12})
    
    if result.success:
        h_final = project_to_sum_box(result.x, target_sum)
    else:
        h_final = h2
    
    # Evaluate final result with exact C5 (max over k)
    # Reconstruct full vector
    h_full = np.concatenate([h_final, h_final[::-1]])
    corr = autocorr_fft(h_full)
    final_c5 = c5_from_corr(corr, n_points)
    
    # Ensure we don't exceed budget
    if time.time() - start_time > budget_s:
        # Return what we have
        pass
    
    return (h_full.tolist(), float(final_c5), n_points)
