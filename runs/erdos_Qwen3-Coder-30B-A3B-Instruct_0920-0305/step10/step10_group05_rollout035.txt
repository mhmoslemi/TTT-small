import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fftpack import ifft, fft
import time

def autocorr_fft(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(h, L)).real
    return corr

def compute_c5_exact(h):
    """Compute exact C5 using max of correlations"""
    corr = autocorr_fft(h)
    # Extract valid correlations (excluding padding)
    valid_corr = corr[len(h)-1:2*len(h)-1]
    min_corr = np.min(valid_corr)
    return 1 - min_corr * (2.0 / len(h))

def smooth_objective(h, beta=50):
    """Smooth approximation using log-sum-exp of correlations"""
    corr = autocorr_fft(h)
    valid_corr = corr[len(h)-1:2*len(h)-1]
    # Subtract max to prevent overflow
    shifted = valid_corr - np.max(valid_corr)
    weights = np.exp(beta * shifted)
    log_sum_exp = np.log(np.sum(weights))
    return 1 - (2.0 / len(h)) * log_sum_exp / beta

def smooth_gradient(h, beta=50):

    # Manual gradient computation

def project_to_sum_box(x, target_sum):
    """Project vector x to sum = target_sum and box constraints [0,1]"""
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    diff = target_sum - current_sum
    if abs(diff) < 1e-12:
        return x
    elif diff > 0:
        # Need to add more mass
        # Simple way: increase each small value slightly (water filling)
        where = np.argsort(x)  # ascending order, pick smallest
        x_copy = x.copy()
        for idx in where:
            if diff <= 0:
                break
            add = min(x_copy[idx] + diff, 1.0) - x_copy[idx]
            x_copy[idx] += add
            diff -= add
        return x_copy
    else:
        # Need to reduce mass
        # First extend with large values to higher priority
        # Then remove excess 
        where = np.argsort(x)[::-1]  # descending
        x_copy = x.copy()
        for idx in where:
            if diff >= 0:
                break
            remove = max(x_copy[idx] + diff, 0) - x_copy[idx]
            x_copy[idx] += remove
            diff -= remove
        return x_copy

def create_parametric_shape(a, m):
    """Create symmetric shape: 2 edges of length 'a', middle block length 'b'"""
    # Total length = 2*a + (2-2*a) = 2 (which specifices the tiling)
    # But we want two edges of length a each, 
    # middle part (2 - 2*a) long, with appropriate height
    # only use first half
    p = (1 - 2*a) / (2*(1-a))
    if a <= 0 or a >= 0.5 or p < 0:
        # fallback to symmetric step shape
        h = np.ones(m)
        h[0:int(m*0.3)] = 0.5
        h[int(m*0.5):] = 0.5
        return h
        
    edge_len = int(a * m)  # floors the result
    midpoint_len = m - 2*edge_len
    if midpoint_len <= 0:
        # Fallback to edge only
        h = np.ones(m)
        h[int(m*0.2):] = 0
        return h
    
    # Regular case
    h = np.ones(m)
    h[0:edge_len] = 1.0
    h[edge_len:edge_len+midpoint_len] = p
    h[edge_len+midpoint_len:] = 1.0
    return h
    
def parametric_sweep(n_points=240, start_a=0.01, end_a=0.49, step=0.002):
    m = n_points // 2
    best_c5 = float('inf')
    best_a = None
    best_h = None
    
    # Set up parameters
    dx = 2.0 / n_points
    target_sum = n_points / 4
    
    a_values = np.arange(start_a, end_a, step)
    for a in a_values:
        h_half = create_parametric_shape(a, m)
        h_full = np.concatenate([h_half, h_half[::-1]])
        h_full = project_to_sum_box(h_full, target_sum * 2)  # full length sum is target_sum * 2
        c5 = compute_c5_exact(h_full)
        if c5 < best_c5:
            best_c5 = c5
            best_a = a
            best_h = h_full
    
    return best_h, best_c5

def apply_local_refinement(best_h, n_points):
    # Run L-BFGS-B on the full vector, possibly to improve 
    m = n_points // 2
    params = best_h[:m]  # Half vector
    target_sum = n_points / 4
    
    def obj(x):
        h_full = np.concatenate([x, x[::-1]])
        return smooth_objective(h_full, beta=50)
    
    def grad(x):
        h_full = np.concatenate([x, x[::-1]])
        return smooth_gradient(h_full, beta=50)[:m]  # only return gradient of top half
    
    bounds = [(0, 1) for _ in range(m)]
    
    # Optimise
    res = minimize(obj, params, method='L-BFGS-B', bounds=bounds, jac=grad, options={'maxiter':1000, 'ftol':1e-12, 'gtol':1e-12})
    
    if res.success:
        refined_params = res.x
        h_full = np.concatenate([refined_params, refined_params[::-1]])
        # Re-project onto feasible region with exact sum
        h_full = project_to_sum_box(h_full, target_sum * 2)
        c5 = compute_c5_exact(h_full)
        return h_full, c5
    else:
        # Fallback to input
        return best_h, compute_c5_exact(best_h)

def simulated_annealing(h0, n_points, budget_s):
    m = n_points // 2
    assert len(h0) == n_points
    
    h_half = h0[:m]
    target_sum = n_points / 4
    
    current_h = h_half.copy()
    current_c5 = compute_c5_exact(np.concatenate([current_h, current_h[::-1]]))
    
    best_h = current_h.copy()
    best_c5 = current_c5
    
    start_time = time.time()
    T = 0.01
    alpha = 0.9995
    max_delta = 0.05
    acceptance_count = 0
    total_trials = 0
    
    while T > 1e-6 and time.time() - start_time < 0.85 * budget_s:
        if total_trials % 1000 == 0 and time.time() - start_time > 0.95 * budget_s:
            break
            
        # Move: transfer some mass from i to j
        i = np.random.randint(0, m)
        j = np.random.randint(0, m)
        delta = np.random.uniform(0, max_delta)
        
        if current_h[i] >= delta and current_h[j] <= 1 - delta:
            new_h = current_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            new_h = project_to_sum_box(np.concatenate([new_h, new_h[::-1]]), target_sum * 2)
            new_c5 = compute_c5_exact(new_h)
            
            if new_c5 < current_c5:
                current_h = new_h[:m].copy()
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = new_c5
                acceptance_count += 1
            else:
                # Accept with probability
                if np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                    current_h = new_h[:m].copy()
                    current_c5 = new_c5
                    acceptance_count += 1
                    
        # Adaptive step size
        total_trials += 1
        if total_trials % 500 == 0 and total_trials > 0:
            ratio = float(acceptance_count) / total_trials
            if ratio > 0.4:
                max_delta *= 1.1
            elif ratio < 0.1:
                max_delta *= 0.9
            max_delta = np.clip(max_delta, 1e-4, 0.2)
            acceptance_count = 0

        T *= alpha
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 4
    
    # Stage 1: Parameteric sweep to find good starting point
    h0, c5_0 = parametric_sweep(n_points)
    if c5_0 < 0.379:
        return (h0.tolist(), c5_0, n_points)
    
    # Stage 2: Local refinement
    h1, c5_1 = apply_local_refinement(h0, n_points)
    if c5_1 < 0.379:
        return (h1.tolist(), c5_1, n_points)
    
    # Stage 3: Simulated Annealing
    rs_h, rs_c5 = simulated_annealing(h1, n_points, budget_s)
    if rs_c5 < 0.379:
        return (np.concatenate([rs_h, rs_h[::-1]]).tolist(), rs_c5, n_points)
    
    # Stage 4: Final refinement with L-BFGS
    final_h, final_c5 = apply_local_refinement(np.concatenate([rs_h, rs_h[::-1]]), n_points)
    
    return (final_h.tolist(), final_c5, n_points)
