import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def perturb_step(h, step_size=0.01):
    """Create a perturbed version of h with adaptive step sizes"""
    h_new = h.copy()
    indices = np.random.choice(n_points, size=min(20, n_points//5), replace=False)
    
    for idx in indices:
        # Adaptive step size based on position
        adaptive_step = step_size * (1.0 - 2.0 * abs(h_new[idx] - 0.5))
        delta = np.random.normal(0, adaptive_step)
        h_new[idx] += delta
    
    return h_new

    step_size = initial_step
    no_improvement_count = 0
    max_no_improvement = 500
    
    for i in range(max_iter):
        # Adapt step size based on performance
        if no_improvement_count > 100:
            step_size = max(min_step, step_size * 0.9)
            no_improvement_count = 0
        
        # Create perturbation
        h_new = perturb_step(current_h, step_size)
        # Project to valid constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new)
        
        # Accept if better
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
            no_improvement_count = 0
            if c5_new < best_c5:
                best_h = current_h.copy()
                best_c5 = c5_new
        else:
            # Occasionally accept worse solutions for escape
            if np.random.random() < 0.03 * (1.0 - current_c5 / 0.5):
                current_h = h_new.copy()
                current_c5 = c5_new
                no_improvement_count += 1
            else:
                no_improvement_count += 1
    
    return best_h, best_c5

def genetic_algorithm(initial_pop_size=20, generations=100, elite_size=5):
    """Genetic algorithm to explore search space"""
    pop = []
    
    # Initialize population
    for _ in range(initial_pop_size):
        h = np.random.rand(n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        pop.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    for gen in range(generations):
        # Evaluate fitness
        fitness = []
        for h in pop:
            c5 = evaluate_c5(h)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        pop = [pop[i] for i in sorted_indices]
        
        if gen < generations - 1:  # Don't mutate last generation
            # Keep elite
            new_pop = pop[:elite_size]
            
            # Crossover and mutation
            while len(new_pop) < initial_pop_size:
                # Tournament selection
                idx1, idx2 = np.random.choice(elite_size, 2, replace=False)
                parent1, parent2 = pop[idx1], pop[idx2]
                
                # Crossover at point
                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                
                # Mutation
                for i in range(n_points):
                    if np.random.random() < 0.05:
                        child[i] += np.random.normal(0, 0.01)
                
                # Project to valid constraints
                child = project_to_sum_box(child, TARGET_SUM)
                new_pop.append(child)
            
            pop = new_pop
    
    return best_h, best_c5
