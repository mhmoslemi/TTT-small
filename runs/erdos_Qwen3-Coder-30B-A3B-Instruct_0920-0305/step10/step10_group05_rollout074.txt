import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

# Global constants
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points
half_size = n_points // 2

def autocorr_fft(h):
    """Compute autocorrelation using FFT."""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def compute_c5_exact(h):
    """Compute exact C5 value using max of autocorrelation."""
    corr = autocorr_fft(h)
    return 1 - dx * np.min(corr)

def compute_c5_smooth(h, beta=50):
    """Smooth approximation of C5 with log-sum-exp for gradient."""
    corr = autocorr_fft(h)
    max_corr = np.max(corr)
    shifted_corr = corr - max_corr  # Prevent overflow in exp
    exp_corr = np.exp(beta * shifted_corr)
    log_sum_exp = max_corr + np.log(np.sum(exp_corr)) / beta
    return 1 - dx * log_sum_exp

def smooth_gradient(h, beta=50):
    """Gradient of smooth C5."""
    corr = autocorr_fft(h)
    max_corr = np.max(corr)
    shifted_corr = corr - max_corr
    exp_corr = np.exp(beta * shifted_corr)
    Z = np.sum(exp_corr)
    # Gradient formula: -dx * sum(exp(beta*A_k) * dA_k/dh_i) / Z
    grad = np.zeros_like(h)
    for i in range(len(h)):
        # Compute dA_k/dh_i for all k
        for k in range(-len(h)+1, len(h)):
            j = i + k
            if 0 <= j < len(h):
                grad[i] += exp_corr[k + len(h) - 1] * h[j] / Z
    return -dx * grad

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    # Use a simple approach: scale to target, then clip
    x_sum = np.sum(x)
    if x_sum <= 0:
        return np.ones_like(x) * (target_sum / len(x))
    
    # Scale to approximate target
    x_scaled = x * (target_sum / x_sum)
    return np.clip(x_scaled, 0, 1)

def generate_h_parametric(a):
    """Generate symmetric h from parametric shape with edge width a."""
    half_vec = np.zeros(half_size)
    # Edge interval width (subdivide into three parts)
    a = max(0.001, min(0.499, a))
    b = 2 - 2 * a
    p = (1 - 2 * a) / (2 * (1 - a))
    if p < 0:
        p = 0.0
    elif p > 1:
        p = 1.0

    # Place the two edge blocks with value 1
    edge_idx = int(np.round(a / dx))
    if edge_idx >= half_size:
        edge_idx = half_size - 1
    half_vec[:edge_idx] = 1
    half_vec[-edge_idx:] = 1
    # Middle block with height p
    mid_start = edge_idx
    mid_end = half_size - edge_idx
    half_vec[mid_start:mid_end] = p

    return half_vec

def sweep_parametric():
    """Sweep parameter a to find best 1D family solution."""
    best_c5 = float('inf')
    best_a = 0.0
    for a in np.arange(0.001, 0.499, 0.002):
        h = generate_h_parametric(a)
        h_full = np.concatenate([h, h[::-1]])
        c5 = compute_c5_exact(h_full)
        if c5 < best_c5:
            best_c5 = c5
            best_a = a
    return best_a, best_c5

def refine_low_dim(best_a):
    """Refine using low-dimensional gradient-based method."""
    def obj(a):
        h = generate_h_parametric(a)
        h_full = np.concatenate([h, h[::-1]])
        return compute_c5_smooth(h_full, beta=50)
    
    def grad(a):
        h = generate_h_parametric(a)
        h_full = np.concatenate([h, h[::-1]])
        return compute_c5_smooth(h_full, beta=50)  # Dummy gradient

    try:
        # Use bounded L-BFGS-B to optimize the parameter
        from scipy.optimize import minimize_scalar
        res = minimize_scalar(obj, bounds=(0.001, 0.499), method='bounded')
        a_opt = res.x
        h = generate_h_parametric(a_opt)
        h_full = np.concatenate([h, h[::-1]])
        c5_opt = compute_c5_exact(h_full)
        return h_full, c5_opt, a_opt
    except Exception:
        # fallback to previous best
        h = generate_h_parametric(best_a)
        h_full = np.concatenate([h, h[::-1]])
        c5 = compute_c5_exact(h_full)
        return h_full, c5, best_a

def simulated_annealing(h0_half):
    """Run simulated annealing on the full half-vector."""
    # Track best solution
    h_best = h0_half.copy()
    c5_best = compute_c5_exact(np.concatenate([h_best, h_best[::-1]]))
    
    # Initialize parameters
    h_curr = h0_half.copy()
    c5_curr = c5_best
    T = 1e-2
    alpha = 0.9995
    delta_max = 0.05
    max_iter = 200000  # Desired iteration count within time budget
    start_time = time.time()
    
    # Acceptance tracking
    accepted = 0
    total_samples = 0
    rate_window = 500

    for i in range(max_iter):
        if time.time() - start_time > 550:
            break
            
        # Random move: select two indices, transfer mass
        idx_a = np.random.randint(0, half_size)
        idx_b = np.random.randint(0, half_size)
        if idx_a == idx_b:
            continue
            
        # Compute delta
        delta = np.random.uniform(0, delta_max)
        if h_curr[idx_a] < delta:
            delta = h_curr[idx_a]
        if h_curr[idx_b] + delta > 1:
            delta = 1 - h_curr[idx_b]
        if delta <= 0:
            continue
            
        # Perform swap
        h_new = h_curr.copy()
        h_new[idx_a] -= delta  
        h_new[idx_b] += delta
        
        # Project to ensure box constraints
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust for sum constraint
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Compute new C5
        h_full_new = np.concatenate([h_new, h_new[::-1]])
        c5_new = compute_c5_exact(h_full_new)
        
        # Acceptance rule
        total_samples += 1
        if c5_new < c5_curr:
            h_curr = h_new.copy()
            c5_curr = c5_new
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new
            accepted += 1
        elif np.random.rand() < np.exp(-(c5_new - c5_curr)/T):
            h_curr = h_new.copy()
            c5_curr = c5_new
            accepted += 1

        # Adjust temperature and delta
        if total_samples % rate_window == 0:
            if accepted > 0.4 * rate_window:
                delta_max = min(0.2, delta_max * 1.1)
            elif accepted < 0.1 * rate_window:
                delta_max = max(1e-4, delta_max * 0.9)
            accepted = 0
        
        T *= alpha
        
        # Periodic save best
        if i % 1000 == 0:
            h_full_best = np.concatenate([h_best, h_best[::-1]])
            c5_best_save = compute_c5_exact(h_full_best)
            if c5_best_save < c5_best:
                c5_best = c5_best_save
                h_best = h_best.copy()
    
    return np.concatenate([h_best, h_best[::-1]]), c5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    global TARGET_SUM
    TARGET_SUM = n_points / 4
    
    # Phase 1: sweep parametric shape
    a_opt, c5_sweep = sweep_parametric()
    h_param = generate_h_parametric(a_opt)
    h_full_sweep = np.concatenate([h_param, h_param[::-1]])
    
    # Phase 2: refine with low-dim optimization
    h_refined, c5_refined, a_final = refine_low_dim(a_opt)
    
    # Phase 3: SMA with full vector
    if c5_refined > 0.3795:
        h_sma, c5_sma = simulated_annealing(h_param)
        best_h = h_sma
        best_c5 = c5_sma
    else:
        best_h = h_refined
        best_c5 = c5_refined
        
    # Final polishing with L-BFGS-B using smooth objective
    def smooth_obj(h):
        h_full = np.concatenate([h, h[::-1]])
        return compute_c5_smooth(h_full, beta=200)
    
    def smooth_grad(h):
        h_full = np.concatenate([h, h[::-1]])
        return smooth_gradient(h_full, beta=200)[:half_size]
    
    h_finished = best_h[:half_size].copy()
    try:
        bounds = [(0, 1) for _ in range(half_size)]
        res = minimize(
            smooth_obj,
            h_finished,
            method='L-BFGS-B',
            bounds=bounds,
            jac=smooth_grad,
            options={'ftol': 1e-12, 'gtol': 1e-12, 'maxiter': 500}
        )
        if res.success:
            h_optimized = res.x
        else:
            h_optimized = h_finished
        h_optimized = project_to_sum_box(h_optimized, TARGET_SUM)
        h_final = np.concatenate([h_optimized, h_optimized[::-1]])
        c5_final = compute_c5_exact(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    except Exception:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
