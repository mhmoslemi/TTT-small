import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def soft_max_surrogate(h, alpha):
    """Compute smooth max surrogate F_alpha(h)"""
    corr = compute_corr_fft(h)
    # Subtract max for numerical stability
    max_c = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_c))
    sum_exp = np.sum(exp_vals)
    return (np.log(sum_exp) + alpha * max_c) / alpha

def gradient_soft_max_surrogate(h, alpha):
    """Compute gradient of smooth max surrogate"""
    corr = compute_corr_fft(h)
    # Numerical stability: subtract max
    max_c = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_c))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    
    # Symmetrized weights
    w_sym = w + w[::-1]
    # Convolution with h
    g = correlate(w_sym, h, mode='valid')
    return 1 - g[:len(h)]

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Exact evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

    restarts = []
    # Binary block start
    h_binary = np.zeros(n_points)
    h_binary[:n_points//2] = 1.0
    h_binary = project_to_sum_box(h_binary, TARGET_SUM)
    restarts.append(h_binary)
    
    # Smooth random start
    h_smooth = np.random.random(n_points)
    h_smooth = project_to_sum_box(h_smooth, TARGET_SUM)
    restarts.append(h_smooth)
    
    # Warm start from initial_h_values if available

            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        h_init = project_to_sum_box(h_init, TARGET_SUM)
        restarts.append(h_init)
    except:

    num_restarts = len(restarts)

                return soft_max_surrogate(h, alpha)

            except:
                pass
    
    if best_h is None:
        # Fallback to initial solution
        try:
            h_default = np.array(initial_h_values)
            h_default = project_to_sum_box(h_default, TARGET_SUM)
            best_h = h_default
            best_c5 = evaluate_c5(h_default)

            best_c5 = evaluate_c5(best_h)
