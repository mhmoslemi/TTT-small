def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def swarm_optimization(h0, iterations=8000):
    """Particle swarm optimization approach"""
    n_particles = 20
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    # Initialize particles
    for _ in range(n_particles):
        h = h0 + np.random.normal(0, 0.01, n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        particles.append(h.copy())
        velocities.append(np.random.normal(0, 0.005, n_points))
        pbest.append(h.copy())
        pbest_vals.append(evaluate_c5(h))
    
    gbest = pbest[np.argmin(pbest_vals)]
    gbest_val = min(pbest_vals)
    
    w = 0.7  # Inertia weight
    c1 = 1.5  # Cognitive parameter
    c2 = 1.5  # Social parameter
    
    for i in range(iterations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for j in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[j] = (w * velocities[j] + 
                           c1 * r1 * (pbest[j] - particles[j]) + 
                           c2 * r2 * (gbest - particles[j]))
            
            # Update position
            particles[j] += velocities[j]
            
            # Apply constraints
            particles[j] = project_to_sum_box(particles[j], TARGET_SUM)
            
            # Evaluate
            current_val = evaluate_c5(particles[j])
            
            # Update personal best
            if current_val < pbest_vals[j]:
                pbest[j] = particles[j].copy()
                pbest_vals[j] = current_val
                
                if current_val < gbest_val:
                    gbest = particles[j].copy()
                    gbest_val = current_val
    
    return gbest, gbest_val
