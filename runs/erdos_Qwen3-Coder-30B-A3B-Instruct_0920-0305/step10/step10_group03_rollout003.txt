# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling algorithm for projection
    x = np.array(x)
    sorted_x = np.sort(x)[::-1]
    cumsum = np.cumsum(sorted_x)
    n = len(x)
    
    # Find the threshold using binary search
    low, high = 0.0, 1.0
    for _ in range(100):
        mid = (low + high) / 2
        # Compute how many elements are above mid
        below_mid = np.sum(sorted_x <= mid)
        above_mid = n - below_mid
        # Compute sum of elements above mid
        sum_above = np.sum(sorted_x[below_mid:]) - mid * above_mid
        # If sum is less than target, need to increase threshold
        if sum_above < target_sum:
            high = mid
        else:
            low = mid
    
    threshold = low
    result = np.clip(x - threshold, 0, 1)
    # Adjust to match exact sum
    current_sum = np.sum(result)
    if current_sum > 0:
        # Scale down all but last component to make sum match target
        scale_factor = target_sum / current_sum
        result = result * scale_factor
        # Clamp to [0,1] again
        result = np.clip(result, 0, 1)
    
    return result

def solve_binary_milp(N, time_limit):
    """Solve binary MILP for optimal step function with N points"""
    try:
        # Binary variables
        x = cp.Variable(N, boolean=True)
        # Objective variable for max overlap
        t = cp.Variable()
        
        # Constraints
        constraints = [cp.sum(x) == N // 2]
        
        # Compute correlation bounds with McCormick envelopes
        for k in range(N):  # For each shift k from 0 to N-1
            if k == 0:
                # For shift 0, sum_i x_i * x_i = sum_i x_i (since x binary)
                # No constraint needed as it's constant N/2
                continue
            else:
                # For each shift k, sum_i x_i * x_{i+k} <= sum_i min(x_i, x_{i+k})
                # Use McCormick envelopes: y[i,k] <= x_i, y[i,k] <= x_{i+k}, 
                # y[i,k] >= x_i + x_{i+k} - 1, y[i,k] >= 0
                # Add constraint sum_y >= N/2 - t
                # But we also want to track y's properly
                # Let's directly write constraint without individual y variables:
                corr_sum = cp.sum(cp.multiply(x[:-k], x[k:]))  # will be sum_i x_i * x_{i+k}
                constraints.append(corr_sum >= N // 2 - t)
        
        # Objective: minimize t
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        # Solve with CBC
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=time_limit)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURATE]:
            return x.value.astype(int)
            
    except Exception as e:
        pass
    
    # Fallback to random solution if MILP failed
    return None

def upsample_binary(binary_vec, target_N):
    """Upsample a binary vector to target_N"""
    n = len(binary_vec)
    # Ensure the factor is integer
    factor = target_N // n
    if factor * n != target_N:
        factor = target_N // n  # Ensure integer division
    # Repeat each element
    upsampled = np.repeat(binary_vec, factor)
    # If we need more elements, pad with 0
    if len(upsampled) < target_N:
        upsampled = np.pad(upsampled, (0, target_N - len(upsampled)), 'constant')
    elif len(upsampled) > target_N:
        upsampled = upsampled[:target_N]
    return upsampled

def binary_local_search(h0, max_iter=50000, time_limit=60):
    """Specialized local search for binary patterns"""
    current_h = np.array(h0, dtype=int)
    current_c5 = evaluate_c5(current_h.astype(float))
    
    best_h = current_h.copy()
    best_c5 = current_c5
    
    start_time = time.time()
    iteration = 0
    while time.time() - start_time < time_limit and iteration < max_iter:
        iteration += 1
        
        # Compute overlaps for all shifts
        corr = np.correlate(current_h, current_h, mode='full')
        # Extract relevant shifts (non-negative)
        overlap = corr[len(current_h)-1:]
        # Update overlaps (N/2 - overlap)
        D = (len(current_h) // 2) - overlap

        k_max = np.argmax(D)

            break
            
        # Find indices where current_h=1 and shift_k gives 0
        # Need to compute overlaps more carefully
        # Indices that are 1 and should be paired but aren't
        ones_indices = np.where(current_h == 1)[0]
        candidates = []
        
        for i in ones_indices:
            if i + k_max < len(current_h) and current_h[i + k_max] == 0:
                candidates.append(i)
        
        if not candidates:
            # No improvement possible by swapping
            break
            
        # Try to find a 0 that can be swapped with a 1 to reduce overlap
        zeros_indices = np.where(current_h == 0)[0]
        swaps = []
        
        for i in candidates:
            if i + k_max < len(current_h) and current_h[i + k_max] == 0:
                # Look for a 0 that matches our shifted 1 (which is missing)
                for j in zeros_indices:
                    if j + k_max < len(current_h) and current_h[j + k_max] == 1:
                        swaps.append((i, j))
        
        if swaps:
            i, j = swaps[0]  # Take first swap
            
            # Try the swap
            temp_h = current_h.copy()
            temp_h[i] = 0
            temp_h[j] = 1
            
            # Check if changed h is valid
            temp_c5 = evaluate_c5(temp_h.astype(float))
            
            if temp_c5 < current_c5:
                current_h = temp_h
                current_c5 = temp_c5
                if temp_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = temp_c5
        else:
            # Random perturbation
            if np.random.rand() < 0.1:
                i = np.random.randint(len(current_h))
                j = np.random.randint(len(current_h))
                # Ensure i != j and avoid same values
                if current_h[i] != current_h[j]:
                    temp_h = current_h.copy()
                    temp_h[i] = 1 - temp_h[i]
                    temp_h[j] = 1 - temp_h[j]
                    
                    temp_c5 = evaluate_c5(temp_h.astype(float))
                    if temp_c5 < current_c5:
                        current_h = temp_h
                        current_c5 = temp_c5
                        if temp_c5 < best_c5:
                            best_h = current_h.copy()
                            best_c5 = temp_c5
            
        if time.time() - start_time > 0.9 * time_limit:
            break
    
    return best_h, best_c5

def differential_evolution_refine(h0, target_sum, time_limit):
    """Refine solution using differential evolution"""
    start_time = time.time()
    
    def objective(h):
        # Project to sum constraint
        h_projected = project_to_sum_box(h, target_sum)
        return evaluate_c5(h_projected)
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(len(h0))]
    
    # Run differential evolution
    result = differential_evolution(
        objective, 
        bounds, 
        maxiter=int(500 * time_limit / 600),  # Adjust based on time limit
        popsize=20,
        strategy='best1bin',
        mutation=(0.5, 1),
        recombination=0.7,
        seed=42,
        disp=False
    )
    
    # Final project to sum constraint
    final_h = project_to_sum_box(result.x, target_sum)
    final_c5 = evaluate_c5(final_h)
    
    return final_h, final_c5
