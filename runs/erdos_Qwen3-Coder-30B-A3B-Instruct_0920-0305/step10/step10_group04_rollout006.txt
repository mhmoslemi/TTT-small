n_points = 240

def fast_corr_eval(x):
    """Fast computation of C5 using correlation of x with its complement."""
    A = correlate(x, x, mode='full')
    L = len(A)
    n = len(x)
    O = np.zeros(L)
    # Precompute prefix sums for column indices
    x_cumsum = np.concatenate(([0], np.cumsum(x)))
    for d in range(-(n-1), n):
        if d >= 0:
            # Sum over i in [0, n-1-d] of x[i]*(1-x[i+d]) = sum x[i] - sum x[i]*x[i+d]
            start = 0
            end = n - d
        else:
            # Sum over i in [0, n+ d-1] of x[i]*(1-x[i+d]) = sum x[i] - sum x[i]*x[i+d]
            start = -d
            end = n
        # x[i] contributions
        if start < len(x_cumsum) and end <= len(x_cumsum):
            prefix_sum = x_cumsum[end] - x_cumsum[start]
            # x[i]*x[i+d] contributions
            if d >= 0:
                prod_sum = A[L//2 + d]
            else:
                prod_sum = A[L//2 + d]
            O[L//2 + d] = prefix_sum - prod_sum
    return dx * np.max(O)

def incrementally_update_corr(x, old_A, i, j):
    """Incrementally update auto-correlation for a swap at indices i and j."""
    n = len(x)
    x_pad = np.zeros(2*n - 1)
    x_pad[n-1 : n-1 + n] = x
    # Determine the contribution of x[i] to A in shifts it takes part in
    if i < n and i + (n - 1) < 2 * n - 1:
        contrib_i = x_pad[i + n - 1 : i + 3 * n - 2] + x_pad[i : i + 2 * n - 1][::-1]
    else:
        contrib_i = np.zeros(2*n - 1)
    contrib_j = x_pad[j + n - 1 : j + 3 * n - 2] + x_pad[j : j + 2 * n - 1][::-1]
    delta = contrib_j - contrib_i
    return old_A + delta

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    result = np.clip(x + (a + b)/2, 0, 1)
    return result
