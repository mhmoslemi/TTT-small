```python
import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

# Global constants
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points

def compute_corr_from_binary(x):
    """Compute C5 using correlation from binary vector"""
    x_rev = x[::-1]
    b = 1 - x
    L = 2 * len(x) - 1
    corr = np.fft.ifft(np.fft.fft(x_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5_binary(x):
    """Evaluate C5 from binary vector x"""
    corr = compute_corr_from_binary(x)
    return dx * np.max(corr)

def compute_corr_from_real(h):
    """Compute C5 using correlation from real vector h"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5_real(h):
    """Evaluate C5 from real vector h"""
    corr = compute_corr_from_real(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_deltas(x, i, j):
    """Compute delta correlation vector when swapping positions i and j in binary vector x"""
    padded = np.concatenate([[0]*(n_points-1), x.tolist(), [0]*(n_points-1)])
    mask = np.array([1]*n_points)  # To avoid changing ones
    mask[i] = 0
    mask[j] = 0
    # Contributions from position i (before swapping):
    delta_i = -(padded[i+n_points-1:i+3*n_points-2] + padded[i:i+2*n_points-1][::-1])
    # Contributions from position j (before swapping):
    delta_j = (padded[j+n_points-1:j+3*n_points-2] + padded[j:j+2*n_points-1][::-1])
    # Full delta vector:
    delta_A = delta_i + delta_j
    return delta_A

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points / 2
    
    # Initialize binary vector with equal numbers of 0s and 1s
    x = np.zeros(n_points, dtype=int)
    x[:n_points//2] = 1
    np.random.shuffle(x)
    
    # Evaluate initial solution
    current_c5 = evaluate_c5_binary(x)
    best_x = x.copy()
    best_c5 = current_c5
    
    # Simulated Annealing parameters
    T0 = 2 * dx
    Tf = 1e-6 * dx
    total_moves = 300000
    cooling_rate = (T0 - Tf) / total_moves
    T = T0
    
    # Tracking variables for stagnation detection
    last_improvement = 0
    stagnation_count = 0
    best_so_far = current_c5
    
    # SA loop
    for move in range(total_moves):
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Choose move type
        if np.random.rand() < 0.6:
            # Targeted swap based on worst shift
            corr = compute_corr_from_binary(x)
            # Compute overlap O_d
            # Using auto-correlation method: O_d = sum(x[i]) - A_d where A_d is auto-correlation of x
            # Alternative way to compute O_d: 
            # O_d = sum over i of x_i * (1 - x_{i+d}) = sum(x_i) - A_d
            # But we need to be careful of boundary conditions.
            # Best to compute it correctly, ignoring boundary overlaps that are insignificant 
            
            # For now, let's estimate using correlation:
            # Algorithm: calculate all O_d
            L = len(x)*2 - 1
            A = np.fft.ifft(np.fft.fft(x[::-1], L) * np.fft.fft(1-x, L)).real  # This is exactly A_d = sum x_i x_{i+d}
            O = np.zeros(L)
            for d in range(L):
                if d < L - n_points + 1:
                    # This is approximate -- use the direct sum 
                    sum_x = np.sum(x)
                    # absorb d window offset behavior manually:
                    # A_{-d} will use sum over terms x_j * x_{j-d} where j < n-|d| to compensate for boundary 
                    # compute from scratch due to hacer lib
                    start = max(0, -d)
                    end = min(n_points, n_points - d)
                    A_d = np.sum(x[start:end] * x[start+d:end+d])
                    O[d] = sum_x - A_d
                else:
                    O[d] = np.inf  # Avoid in calculation
            
            # Find worst shift (positive shift is right_shift, negative is left_shift)
            worst_shift_idx = np.argmax(O)
            worst_shift = worst_shift_idx - n_points + 1  # Translate index to actual shift
            
            # Find indices that would help reduce this worst shift:
            U = []
            V = []
            
            # max d should be in range (n_points-1) to -(n_points-1)
            # it was indeed the max index in -n_points+1 to n_points-1
            # Check both directions
            # For shift d=0 is undefined (no overlap) so start from -n_points+2 to n_points-2  
            # actually, direct translation from index is 
            # shift = idx - (n-1)
            
            # i such that x_i = 1, x_{i+d} = 0
            for i in range(n_points):
                j = i + worst_shift
                if 0 <= j < n_points:
                    if x[i] == 1 and x[j] == 0:
                        U.append(i)
                        
            # j such that x_j = 0, x_{j+d} = 1  
            for j in range(n_points):
                i = j - worst_shift
                if 0 <= i < n_points:
                    if x[j] == 0 and x[i] == 1:
                        V.append(j)
                        
            if U and V:
                i = np.random.choice(U)
                j = np.random.choice(V)
                # try to modify, but we can measure O before after directly
                # as a proxy for exact differencial update, we calculate both (but only record the cost)
                A_old = np.fft.ifft(np.fft.fft(x[::-1], L) * np.fft.fft(1-x, L)).real
                x_new = x.copy()
                x_new[i], x_new[j] = x_new[j], x_new[i]  # swap
                # directly recompute correlation to avoid race conditions / blast
                try:
                    A_new = np.fft.ifft(np.fft.fft(x_new[::-1], L) * np.fft.fft(1-x_new, L)).real
                    # since the formula O_d = n - A_d, we have:
                    corr_new = dx * np.max(A_new)  # This is wrong, need to do the right transformation
                    # Let's be more careful:
                    A_new_d = A_new
                    O_new = np.zeros_like(A_new_d)
                    for d in range(L):
                        if d < L - n_points + 1:
                            start = max(0, -d)
                            end = min(n_points, n_points - d)
                            A_d = np.sum(x_new[start:end] * x_new[start+d:end+d])
                            O_new[d] = n_points/2 - A_d
                    corr_new = dx * np.max(O_new)
                except Exception:
                    # fallback to classic
                    corr_new = evaluate_c5_binary(x_new)
            else:
                # fallback to random swap
                i, j = np.random.choice(n_points, size=2, replace=False)
                x_new = x.copy()
                x_new[i], x_new[j] = x_new[j], x_new[i]
                corr_new = evaluate_c5_binary(x_new)
                if corr_new < current_c5:
                    x = x_new
                    current_c5 = corr_new
                    if corr_new < best_c5:
                        best_x = x.copy()
                        best_c5 = corr_new
                    continue  # skip dynamics for now
        else:
            # Random swap
            i, j = np.random.choice(n_points, size=2, replace=False)
            x_new = x.copy()
            x_new[i], x_new[j] = x_new[j], x_new[i]
            corr_new = evaluate_c5_binary(x_new)
        
        # Acceptance probability
        delta = corr_new - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            x = x_new
            current_c5 = corr_new
            if corr_new < best_c5:
                best_x = x.copy()
                best_c5 = corr_new
                best_so_far = corr_new
                last_improvement = move
        else:
            pass  # Do nothing
            
        T -= cooling_rate
        
        # Stagnation detection
        if move - last_improvement > 20000:
            # Do partial restart
            num_flip = int(0.05 * n_points)
            flip_indices = np.random.choice(n_points, size=num_flip, replace=False)
            for idx in flip_indices:
                x[idx] = 1 - x[idx]
            # Project sum constraint
            x = np.array(x, dtype=float)
            x = project_to_sum_box(x, TARGET_SUM)
            x = np.round(x).astype(int)
            last_improvement = move
        
    # Continue SA to finish
    last_move = move
    
    # Final continuous refinement
    h0 = best_x.astype(float)
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    # First SLSQP refinement
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
    try:
        res = minimize(
            evaluate_c5_real,
            h0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12, 'disp': False}
        )
        if res.success:
            h0 = res.x
        h0 = project_to_sum_box(h0, TARGET_SUM)
    except Exception:
        pass
    
    # Second refinement
    try:
        res = minimize(
            evaluate_c5_real,
            h0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12, 'disp': False}
        )
        if res.success:
            h0 = res.x
    except Exception:
        pass
        
    # Final projection
    h0 = project_to_sum_box(h0, TARGET_SUM)
    final_c5 = evaluate_c5_real(h0)
    
    return (h0.tolist(), float(final_c5), n_points)
```
