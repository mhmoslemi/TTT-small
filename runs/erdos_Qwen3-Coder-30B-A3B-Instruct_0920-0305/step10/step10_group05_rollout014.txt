import numpy as np

def autocorr_fft(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    b = h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_c5_from_autocorr(autocorr, dx):
    """Convert autocorrelation result to C5 value"""
    return 1.0 - dx * np.min(autocorr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_symmetric_h(a, p, n_half):
    """Generate symmetric h vector given edge width `a` and plateau height `p`"""
    # Ensure a and p are valid
    a = max(0.0, min(0.5, a))
    p = max(0.0, min(1.0, p))
    
    # Build half vector
    edge_len = int(np.ceil(a * n_half))
    mid_len = n_half - 2 * edge_len
    if mid_len < 0:
        # Adjust to ensure non-negative middle
        mid_len = 0
        edge_len = (n_half) // 2
    edge_len = max(0, min(edge_len, n_half))
    
    # Direct construction (start with all zeros)
    h_half = np.zeros(n_half)
    h_half[:edge_len] = 1.0
    h_half[-edge_len:] = 1.0
    h_half[edge_len : n_half - edge_len] = p
    
    # Enforce sum constraint exactly
    target_sum = n_half / 2
    current_sum = np.sum(h_half)
    if abs(current_sum - target_sum) > 1e-10:
        h_half = project_to_sum_box(h_half, target_sum)
        
    return h_half

def smooth_objective(h, beta=50):
    """Compute a smooth approximation to C5 using log-sum-exp"""
    # Apply smoothing to the correlation
    full_h = np.concatenate([h, h[::-1]])
    autocorr = autocorr_fft(full_h)
    # Subtract max to prevent overflow
    autocorr_shifted = autocorr - np.max(autocorr)
    weights = np.exp(beta * autocorr_shifted)
    log_sum_exp = np.log(np.sum(weights))
    return 1.0 - (1.0/beta) * log_sum_exp
