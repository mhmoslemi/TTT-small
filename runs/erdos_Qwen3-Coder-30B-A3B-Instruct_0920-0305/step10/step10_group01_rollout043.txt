def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def full_corr(a, b):
    """Compute correlation using FFT with zero padding."""
    L = 2 * len(a) - 1
    return np.fft.ifft(np.fft.fft(a, L) * np.conj(np.fft.fft(b, L))).real

def smooth_max_and_weights(c, alpha):
    """Compute smooth max and softmax weights with numerical stability."""
    # Subtract maximum to prevent overflow
    c_shifted = c - np.max(c)
    exp_c = np.exp(alpha * c_shifted)
    Z = np.sum(exp_c)
    w = exp_c / Z
    smooth_max = np.log(Z) / alpha + np.max(c)
    return smooth_max, w

def compute_gradient(h, alpha):
    """Compute gradient of smooth max with respect to h."""
    c = full_corr(h, 1 - h)
    _, w = smooth_max_and_weights(c, alpha)
    
    # symmetrize weights  
    w_sym = w + w[::-1]
    
    # correlation with symmetrized weights
    g = full_corr(w_sym, h)[:len(h)]
    
    return 1.0 - g

def objective_and_gradient(h, alpha):
    """Return smooth max objective and its gradient."""
    c = full_corr(h, 1 - h)
    obj, _ = smooth_max_and_weights(c, alpha)
    grad = compute_gradient(h, alpha)
    return obj, grad

            return objective_and_gradient(h, alpha)
            
        try:
            res = minimize(
                obj_and_grad,
                current_h,
                method='trust-constr',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - b_eq},
                jac=True,
                options={'gtol': 1e-10, 'ftol': 1e-12}
            )
            if res.success:
                current_h = res.x
            else:
                # If optimization fails, use current_h
                pass
        except Exception:
            # fallback to current_h if optimization fails
            pass
    
    return current_h
