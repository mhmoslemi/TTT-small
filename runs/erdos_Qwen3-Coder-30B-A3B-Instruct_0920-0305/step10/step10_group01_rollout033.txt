import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def full_corr(a, b):
    """Compute full correlation using FFT with zero-padding"""
    L = 2 * len(a) - 1
    return np.fft.ifft(np.fft.fft(a, L) * np.conj(np.fft.fft(b, L))).real

def smooth_max_and_weights(c, alpha):
    """Compute smooth max and corresponding weight vector"""
    c_stable = c - np.max(c)  # Numerical stability
    exp_c = np.exp(alpha * c_stable)
    Z = np.sum(exp_c)
    w = exp_c / Z
    smooth_max = np.log(Z) / alpha + np.max(c)
    return smooth_max, w

def compute_gradient(h, alpha):
    """Compute gradient of smooth max using FFT"""
    c = full_corr(h, 1 - h)
    _, w = smooth_max_and_weights(c, alpha)
    # Symmetrize weights to account for both k and -k
    w_sym = w + w[::-1]
    # Correlation of symmetrized weights with h
    g = full_corr(w_sym, h)
    # Return gradient
    return 1.0 - g[:len(h)]

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Evaluate the true C5 value"""
    c = full_corr(h, 1 - h)
    return dx * np.max(c)

def optimize_with_continuation(h0, alpha_list):
    """Optimize with continuation on alpha values"""
    h = h0.copy()
    for alpha in alpha_list:
        # Define objective function and gradient
        def objective(h):
            c = full_corr(h, 1 - h)
            value, _ = smooth_max_and_weights(c, alpha)
            return value
        
        def gradient(h):
            return compute_gradient(h, alpha)
        
        # Define constraint
        A_eq = np.array([np.ones_like(h)])
        b_eq = np.array([TARGET_SUM])
        
        # Define bounds
        bounds = [(0, 1) for _ in range(len(h))]
        
        # Optimize
        result = minimize(
            objective, 
            h, 
            method='trust-constr',
            jac=gradient,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda x: A_eq @ x - b_eq},
            options={'gtol': 1e-9, 'ftol': 1e-12, 'maxiter': 500}
        )
        
        # Update h
        h = result.x.copy()
        
        # Project back onto feasible set
        h = project_to_sum_box(h, TARGET_SUM)
    
    return h
