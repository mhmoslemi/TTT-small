import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_autocorr_fft(h):
    """Compute auto-correlation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def c5_from_autocorr(autocorr):
    """Convert auto-correlation result to min correlation"""
    return dx * np.min(autocorr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def evaluate_c5_by_autocorr(h):
    """Fast evaluation of C5 via min autocorr"""
    autocorr = compute_autocorr_fft(h)
    return 1 - c5_from_autocorr(autocorr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_piecewise_constant_h(a):
    """Generate h vector for given a parameter"""
    # Compute p for given a
    if a >= 0.5:
        p = 0.0
    else:
        p = (1 - 2 * a) / (2 * (1 - a))
    
    h = np.zeros(n_points)
    edge_length = int(a * n_points)
    
    if edge_length > 0:
        h[:edge_length] = 1
        h[-edge_length:] = 1
        mid_length = n_points - 2 * edge_length
        if mid_length > 0:
            h[edge_length:edge_length+mid_length] = p
    else:
        # Uniform distribution
        h[:] = 0.5
    
    # Ensure sum is correct
    h = h * (TARGET_SUM / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Search for best piecewise constant shape
    best_a = 0.0
    best_c5 = 1.0
    
    # Sweep a parameter
    a_values = np.arange(0.05, 0.45, 0.005)
    
    for a in a_values:
        if time.time() - start_time > budget_s * 0.8:
            break
        try:
            h = generate_piecewise_constant_h(a)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_a = a
        except Exception:
            continue
    
    # Generate final h from best a found
    h_final = generate_piecewise_constant_h(best_a)
    
    # Fine tune with local optimization
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        # Try SLSQP
        res = minimize(
            objective, 
            h_final, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_final = res.x
        h_final = project_to_sum_box(h_final, TARGET_SUM)
        final_c5 = evaluate_c5(h_final)
        
        if final_c5 < best_c5:
            best_c5 = final_c5
    except Exception:
        pass
    
    # Try Simulated Annealing for further improvement
    def simulated_annealing(h_start, max_iter, T0):
        h = h_start.copy()
        c5 = evaluate_c5(h)
        best_h = h.copy()
        best_c5 = c5
        
        T = T0
        
        # Compute initial autocorrelation to use for improvements
        autocorr_start = compute_autocorr_fft(h)
        min_auto_start = np.min(autocorr_start)
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Make a small move: pick two points and adjust
            i1, i2 = np.random.choice(n_points, 2, replace=False)
            delta = np.random.uniform(-0.05, 0.05)
            
            # Ensure adjustment stays within [0, 1]
            if h[i1] + delta >= 0 and h[i1] + delta <= 1 and h[i2] - delta >= 0 and h[i2] - delta <= 1:
                h_new = h.copy()
                h_new[i1] += delta
                h_new[i2] -= delta
                
                # Project back to valid constraints
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                
                c5_new = evaluate_c5(h_new)
                
                if c5_new < c5:
                    h = h_new.copy()
                    c5 = c5_new
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                else:
                    # Accept with probability
                    if np.random.rand() < np.exp(-(c5_new - c5) / T):
                        h = h_new.copy()
                        c5 = c5_new
                        
            T = T0 * (1.0 - i / max_iter)
            
        return best_h, best_c5
    
    # Run simulated annealing with better initial solution
    if time.time() - start_time < budget_s * 0.95:
        try:
            sa_h, sa_c5 = simulated_annealing(h_final, 10000, 0.001)
            if sa_c5 < best_c5:
                best_c5 = sa_c5
                h_final = sa_h
        except Exception:
            pass
    
    return (h_final.tolist(), float(best_c5), n_points)
