TARGET_SUM = None

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_with_methods(h0, methods):
    """Try different optimization methods on the same starting point"""
    best_result = h0.copy()
    best_val = evaluate_c5(h0)
    
    for method in methods:
        try:
            bounds = [(0, 1) for _ in range(len(h0))]
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
            res = minimize(
                evaluate_c5, 
                h0.copy(), 
                method=method, 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h = res.x
                h = project_to_sum_box(h, TARGET_SUM)
                val = evaluate_c5(h)
                if val < best_val:
                    best_result = h
                    best_val = val
        except:
            continue
    
    return best_result

def evolutionary_optimize(h0, population_size, mutation_rate, crossover_rate):
    """Simple evolutionary algorithm"""
    population = [h0.copy()]
    for _ in range(population_size - 1):
        # Add perturbed version
        mutant = h0.copy()
        num_changes = max(1, min(10, len(h0) // 10))
        indices = np.random.choice(len(h0), size=num_changes, replace=False)
        for idx in indices:
            mutant[idx] = np.random.uniform(0, 1)
        mutant = project_to_sum_box(mutant, TARGET_SUM)
        population.append(mutant)
    
    # Evaluate population
    fitness = [evaluate_c5(ind) for ind in population]
    
    # Evolve
    for _ in range(100):
        # Selection
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:population_size//2]
        selected = [population[i] for i in top_indices]
        
        # Create offspring
        new_population = selected.copy()
        while len(new_population) < population_size:
            parent1 = selected[np.random.randint(len(selected))]
            parent2 = selected[np.random.randint(len(selected))]
            
            if np.random.random() < crossover_rate:
                crossover_point = np.random.randint(1, len(h0)-1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.random() < mutation_rate:
                num_mutations = max(1, min(10, len(h0) // 10))
                indices = np.random.choice(len(h0), size=num_mutations, replace=False)
                for idx in indices:
                    child[idx] = np.random.uniform(0, 1)
            
            child = project_to_sum_box(child, TARGET_SUM)
            new_population.append(child)
        
        population = new_population[:population_size]
        fitness = [evaluate_c5(ind) for ind in population]
    
    best_idx = np.argmin(fitness)
    return population[best_idx]

def simulated_annealing_optimize(h0, max_iter, cooling_rate, initial_temp, min_temp):
    """Simulated annealing optimization"""
    current_h = h0.copy()
    current_c5 = evaluate_c5(h0)
    best_h = h0.copy()
    best_c5 = current_c5
    temp = initial_temp
    
    for i in range(max_iter):
        if temp < min_temp:
            break
            
        # Generate neighbor
        neighbor_h = current_h.copy()
        num_changes = max(1, min(5, len(h0) // 20))
        indices = np.random.choice(len(h0), size=num_changes, replace=False)
        for idx in indices:
            delta = np.random.normal(0, 0.01)
            neighbor_h[idx] += delta
        
        neighbor_h = project_to_sum_box(neighbor_h, TARGET_SUM)
        neighbor_c5 = evaluate_c5(neighbor_h)
        
        # Accept or reject
        if neighbor_c5 < current_c5:
            current_h = neighbor_h.copy()
            current_c5 = neighbor_c5
            if neighbor_c5 < best_c5:
                best_h = neighbor_h.copy()
                best_c5 = neighbor_c5
        else:
            # Accept with probability based on temperature
            prob = np.exp(-(neighbor_c5 - current_c5) / temp)
            if np.random.random() < prob:
                current_h = neighbor_h.copy()
                current_c5 = neighbor_c5
        
        temp *= cooling_rate
    
    return best_h
