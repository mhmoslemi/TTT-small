import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def full_corr(a, b):
    """Compute full correlation (zero-padded) using FFT"""
    L = 2 * len(a) - 1
    return np.fft.ifft(np.fft.fft(a, L) * np.conj(np.fft.fft(b, L))).real

def soft_max_stable(x, alpha):
    """Compute softmax with numerical stability"""
    x_stable = x - np.max(x)
    exp_x = np.exp(alpha * x_stable)
    return exp_x / np.sum(exp_x)

def smooth_max_and_weights(c, alpha):
    """Return smooth max and softmax weights for the correlation values"""
    w = soft_max_stable(c, alpha)
    return w

def compute_gradient(h, alpha):
    """Compute gradient of smooth max objective using FFT"""
    c = full_corr(h, 1 - h)
    w = smooth_max_and_weights(c, alpha)
    # Ensure symmetry for weight (for correlation kernel)
    w_sym = w + w[::-1]
    # Correlation of symmetric weights with h
    g = full_corr(w_sym, h)[:len(h)]
    grad = 1.0 - g
    return grad

def smooth_objective_and_grad(h, alpha):
    """Compute smooth max objective and its gradient"""
    c = full_corr(h, 1 - h)
    smooth_c = np.log(np.sum(np.exp(alpha * c))) / alpha
    grad = compute_gradient(h, alpha)
    return smooth_c, grad

def objective_and_gradient(h, alpha):
    """Wrapper for scipy minimize"""
    obj, grad = smooth_objective_and_grad(h, alpha)
    return obj, grad
