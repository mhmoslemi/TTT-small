def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def simulated_annealing_v2(current_h, current_c5, start_time, budget_s):

    max_iter = 10000
    temp_start = 0.5
    temp_end = 0.0001
    alpha = np.exp(np.log(temp_end/temp_start) / max_iter)
    
    h = current_h.copy()
    c5 = current_c5
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        temp = temp_start * (alpha ** i)
        
        # Perturb solution with larger probability to escape local minima
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        # Try larger perturbation
        delta = np.random.normal(0, 0.05)
        h_new[idx] += delta
        
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject based on Metropolis criterion
        if c5_new < c5 or np.random.random() < np.exp(-(c5_new - c5) / temp):
            h = h_new.copy()
            c5 = c5_new
            
    return h, c5

def dynamic_perturbation_strategy(current_h, current_c5, start_time, budget_s):
    """Random perturbation with dynamic step sizes"""
    max_iter = 5000
    h = current_h.copy()
    c5 = current_c5
    
    # Start with larger step size and gradually decrease
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Dynamic step size (decreasing over time)
        step_size = 0.1 * (1 - iter_num / max_iter) + 0.01
        
        # Perturb several points
        h_new = h.copy()
        indices = np.random.choice(n_points, max(5, n_points // 20), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, step_size)
            
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < c5:
            h = h_new.copy()
            c5 = c5_new
            
    return h, c5
