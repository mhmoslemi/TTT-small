import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from cvxpy import Variable, Minimize, Problem, Bool, sum, multiply
import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
N_BINARY = 60  # For MILP skeleton
upsample_factor = n_points // N_BINARY

def compute_corr_fft(h):
    """Fast correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using fast FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_binary_milp():
    """Solve a binary MILP to get a good skeleton"""
    # Decision variables - binary
    x = Variable(N_BINARY, boolean=True)
    # Product variables
    y = Variable((N_BINARY, N_BINARY), boolean=True)
    # Worst case value
    t = Variable(1, nonneg=True)
    
    # Constraints
    constraints = [sum(x) == N_BINARY // 2]
    
    # Bound constraints for y
    for k in range(1, N_BINARY):
        for i in range(N_BINARY - k):
            constraints += [y[i,k] <= x[i]]
            constraints += [y[i,k] <= x[i+k]]
            constraints += [y[i,k] >= x[i] + x[i+k] - 1]
            constraints += [sum(y[:,k]) >= N_BINARY // 2 - t]
    
    # Objective
    prob = Problem(Minimize(t), constraints)
    
    try:
        prob.solve(solver=cp.GLPK_MI, timelimit=30)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return np.round(x.value).astype(int)
    except:
        pass
    
    # Fallback: random binary solution
    return np.random.choice([0, 1], size=N_BINARY, p=[0.5, 0.5])

def upsample_binary(h_binary):
    """Upsample binary solution to full resolution"""
    h_upsampled = np.empty(n_points)
    for i in range(n_points):
        h_upsampled[i] = h_binary[i // upsample_factor]
    return h_upsampled

def swap_search_binary(h_binary, max_swaps=10000):
    """Perform swap search focused on worst shift"""
    # Convert to float for easier handling
    h = h_binary.astype(float)
    n = len(h)
    
    def compute_auto_correlations():
        # Fast correlation using FFT in real domain
        corr = np.correlate(h, h, mode='full')
        # Only right shifts (k = 1 to n-1)
        return corr[n-1:]  # equivalent to rho(k) for k >= 1
    
    def update_correlation_at_swap(i, j, corr_vals):
        # Assuming we're swapping h[i] and h[j]
        # Update the correlation at each shift k where i or j contributes
        # For simplicity in this algorithm, perform exact recomputation
        for k in range(1, n):
            # Handle specific changes due to two positions moving
            if i + k < n:
                corr_vals[k-1] -= h[i] * h[i+k]  # old val
                corr_vals[k-1] += h[j] * h[j+k]  # new val
            if j + k < n:
                corr_vals[k-1] -= h[j] * h[j+k]  # old val
                corr_vals[k-1] += h[i] * h[i+k]  # new val
    
    corr_vals = compute_auto_correlations()
    
    for _ in range(max_swaps):
        # Find worst shift (max(N/2 - rho[k]))
        # Need to be careful about indices
        term = (n // 2) - corr_vals
        k_star = np.argmax(term)
        if term[k_star] <= 0:
            break
        
        A = []
        B = []
        
        for i in range(n - k_star - 1):
            if h[i] == 1 and h[i + k_star] == 0:
                A.append(i)
            if h[i] == 0 and h[i + k_star] == 1:
                B.append(i)
                
        if not A or not B:
            # Try random swap
            idxs = np.random.choice(n, 2, replace=False)
            h[idxs[0]], h[idxs[1]] = h[idxs[1]], h[idxs[0]]
            corr_vals = compute_auto_correlations()
        else:
            # Perform swap from A and B
            a = np.random.choice(A)
            b = np.random.choice(B)
            h[a], h[b] = h[b], h[a]
            corr_vals = compute_auto_correlations()
    
    return h.astype(int)

def simulated_annealing(h_start, max_iter=5000):
    """Simulated annealing refinement"""
    h = h_start.copy()
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    T = 0.1
    alpha = 0.99
    
    for _ in range(max_iter):
        if time.time() > start_time + budget_s - 10:
            break
            
        # Select two indices for update
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Modify a small amount (around 0.05)
        delta = np.random.uniform(-0.05, 0.05)
        
        # Apply change with bounds
        h_i_old = h[i]
        h[i] = np.clip(h[i] + delta, 0, 1)
        h[j] = np.clip(h[j] - delta, 0, 1)
        
        # Efficiently reproject (only modify if needed)
        sum_old = np.sum(h)
        if abs(sum_old - TARGET_SUM) > 1e-10:
            h = project_to_sum_box(h, TARGET_SUM)
        
        new_c5 = evaluate_c5(h)
        delta_e = new_c5 - current_c5
        
        if delta_e < 0 or np.random.rand() < np.exp(-delta_e / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Revert
            h[i] = h_i_old
            h[j] = h[j] + delta
            
        T *= alpha
    
    return best_h, best_c5

def run(seed=42, budget_s=600):
    global TARGET_SUM, n_points, dx, start_time
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Phase A: Binary MILP solving (small N)
    h_binary = solve_binary_milp()
    
    # Phase B: Upsampling
    h0 = upsample_binary(h_binary)
    
    # Phase C: Optional swap search
    h0 = swap_search_binary(h0, 10000).astype(float)
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    # Phase D: DE optimization (full resolution)
    h_de, c5_de = differential_evolution(h0, 300)
    
    # Phase D: SA refinement (if time allows)
    if time.time() < start_time + budget_s - 10:
        h_sa, c5_sa = simulated_annealing(h_de, 2000)
        h_final = h_sa
        c5_final = c5_sa
    else:
        h_final = h_de
        c5_final = c5_de
    
    # Phase E: Hill climbing
    h_hill, c5_hill = hill_climbing(h_final, 1000)
    if c5_hill < c5_final:
        h_final = h_hill
        c5_final = c5_hill
    
    return (h_final.tolist(), float(c5_final), n_points)

def hill_climbing(h_start, max_iter=1000):
    """Simple hill climbing with local search"""
    h = h_start.copy()
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    for _ in range(max_iter):
        # Pick random points to modify
        n_mod = min(n_points // 20 + 1, 10)
        indices = np.random.choice(n_points, n_mod, replace=False)
        # Store the original values
        h_old = h.copy()
        
        # Perturb
        perturb = np.random.normal(0, 0.02, n_mod)
        for i, dp in zip(indices, perturb):
            h[i] = np.clip(h[i] + dp, 0, 1)
        
        # Reproject
        h = project_to_sum_box(h, TARGET_SUM)
        
        new_c5 = evaluate_c5(h)
        
        # Accept if better, sometimes accept worse
        if new_c5 <= current_c5 or np.random.rand() < 0.05:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h = h_old
    
    return best_h, best_c5
