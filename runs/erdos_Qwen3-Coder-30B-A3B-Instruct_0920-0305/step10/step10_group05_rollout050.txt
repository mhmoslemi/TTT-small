import numpy as np
from scipy.signal import correlate
import time
import math

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_autocorr_fft(h):
    """Compute auto-correlation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr) * dx

def c5_from_autocorr(autocorr, n):
    """Convert autocorrelation result to C5 value"""
    min_autocorr = np.min(autocorr) * dx
    return 1 - min_autocorr

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def evaluate_c5_by_autocorr(h):
    """Fast evaluation of C5 using auto-correlation"""
    autocorr = compute_autocorr_fft(h)
    return c5_from_autocorr(autocorr, n_points)

def generate_piecewise_constant_h(a):
    """Generate piecewise constant h with symmetric shape:
    a: edge length with value 1; middle length = 2-2*a with value p
    """
    if a > 0.5:
        a = 0.5
    if a < 0:
        a = 0
    p = (1 - 2 * a) / (2 * (1 - a))
    if p > 1:
        p = 1
    if p < 0:
        p = 0
        
    h = np.zeros(n_points)
    
    # Edge length in terms of points
    edge_len = int(a * n_points / 2)
    if edge_len < 1:
        edge_len = 1
    middle_len = n_points - 2 * edge_len
    
    # Fill the array
    if middle_len > 0:
        h[:edge_len] = 1.0
        h[edge_len:edge_len+middle_len] = p
        h[edge_len+middle_len:] = 1.0
    else:
        # If middle length is too small, make it all edges
        h[:edge_len] = 1.0
        h[edge_len:] = 1.0
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Try scanning a parameter for piecewise constant functions
    a_range = np.linspace(0.01, 0.45, 200)
    for a in a_range:
        if time.time() - start_time > budget_s * 0.9:
            break
        h_candidate = generate_piecewise_constant_h(a)
        h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
        c5_candidate = evaluate_c5(h_candidate)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
    
    # Perform simulated annealing on the best solution found so far
    if best_h is None or best_c5 == float('inf'):
        # Fall back to stochastic approach
        h_init = np.random.rand(n_points)
        h_init = project_to_sum_box(h_init, TARGET_SUM)
        best_h = h_init.copy()
        best_c5 = evaluate_c5(best_h)
    
    # Simulated Annealing with adaptive cooling
    current_h = best_h.copy()
    current_c5 = best_c5
    T = 0.005
    cooling_rate = 0.99995
    max_iter = int(2e5)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Pick two indices to adjust
        i1 = np.random.randint(n_points)
        i2 = np.random.randint(n_points)
        
        # Generate Δh 
        delta = np.random.normal(0, 0.02)
        delta = min(delta, current_h[i1], 1 - current_h[i2])
        
        # Make adjustment
        if abs(delta) > 1e-10:
            new_h = current_h.copy()
            new_h[i1] = max(0, min(1, new_h[i1] - delta))
            new_h[i2] = max(0, min(1, new_h[i2] + delta))
            
            # Ensure sum constraint through projection
            new_h = project_to_sum_box(new_h, TARGET_SUM)
            
            new_c5 = evaluate_c5(new_h)
            
            # Accept or reject based on simulated annealing
            if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                current_h = new_h.copy()
                current_c5 = new_c5
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
        
        T *= cooling_rate
    
    # Final local optimization
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        from scipy.optimize import minimize
        res = minimize(
            objective, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            final_h = res.x
        else:
            final_h = best_h
        final_h = project_to_sum_box(final_h, TARGET_SUM)
        final_c5 = evaluate_c5(final_h)
        
        if final_c5 < best_c5:
            best_h = final_h.copy()
            best_c5 = final_c5
    except Exception:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
