import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.fft import fft, ifft
import time

def project_to_sum_box(u, target_sum):
    """Water-filling projection to sum constraint."""
    def f(lam):
        projected = np.clip(u + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(u + (a + b) / 2, 0, 1)
    return result

def autocorr_fft(h):
    """Compute aperiodic autocorrelation using FFT."""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(h, L)).real
    return corr

def exact_c5(u):
    """Compute exact C5 from half-vector."""
    n = len(u) * 2
    dx = 2.0 / n
    h = np.concatenate([u, u[::-1]])
    A = autocorr_fft(h)
    return 1 - dx * np.min(A)

def lse_c5(u, beta):
    """Smooth surrogate for C5 using log-sum-exp."""
    n = len(u) * 2
    dx = 2.0 / n
    h = np.concatenate([u, u[::-1]])
    A = autocorr_fft(h)
    return 1 - dx * np.log(np.sum(np.exp(beta * A))) / beta

def grad_lse_c5(u, beta):
    """Gradient of LSE surrogate."""
    n = len(u) * 2
    dx = 2.0 / n
    h = np.concatenate([u, u[::-1]])
    A = autocorr_fft(h)
    w = np.exp(beta * A)
    w_sum = np.sum(w)
    
    # Compute gradient of numerator with respect to h[i]
    # d/dh_i [sum_k w_k * (h_{i+k} + h_{i-k})]
    grad_numer = np.zeros(n)
    for k in range(-n+1, n):
        if k >= 0:
            grad_numer += w * np.roll(h, k)
        else:
            grad_numer += w * np.roll(h, k)
            
    return -dx * grad_numer / w_sum

def parametric_to_halfvector(params, n):
    """Convert parameters to half-vector."""
    # params: [a, r, p] for 3-level shape with ramps
    a, r, p = params
    
    m = n // 2
    u = np.zeros(m)
    
    # Zones: [edge, ramp, middle, ramp, edge]
    edge_start = 0
    ramp_start = a
    middle_start = a + r
    middle_end = 1 - (a + r)
    ramp_end = 1 - a
    edge_end = 1
    
    for i in range(m):
        x = i / (m - 1) if m > 1 else 0.5
        if x < a:
            u[i] = 1
        elif x < a + r:
            # Linear ramp: from 1 to p
            t = (x - a) / r
            u[i] = 1 - t * (1 - p)
        elif x <= 1 - (a + r):
            u[i] = p
        elif x < 1 - a:
            # Linear ramp: from p to 1
            t = (x - (1 - (a + r))) / r
            u[i] = p + t * (1 - p)
        else:
            u[i] = 1
    
    return u

def run(seed=42, budget_s=600):

    start_time = time.time()
    
    n = 120
    m = n // 2
    dx = 2.0 / n
    target_sum = n / 4

    best_c5 = float('inf')
    best_u = None
    best_params = None
    
    # Sweep parameter a from 0.05 to 0.45
    a_values = np.linspace(0.05, 0.45, 400)
    for a in a_values:
        p = (1 - 2 * a) / (2 * (1 - a))
        if p < 0 or p > 1:
            continue
        u = np.ones(m)
        # Fill middle section
        mid_start = int(a * m)
        mid_end = int((1 - a) * m)
        u[mid_start:mid_end] = p
        # Enforce mass constraint
        u = project_to_sum_box(u, target_sum)
        c5 = exact_c5(u)
        if c5 < best_c5:
            best_c5 = c5
            best_u = u.copy()
    
    # Phase II: Differential Evolution on parametric family
    if time.time() - start_time > 0.9 * budget_s:
        return (np.concatenate([best_u, best_u[::-1]]).tolist(), float(best_c5), n)
    
    def objective_de(params):
        u = parametric_to_halfvector(params, n)
        return lse_c5(u, 200)
    
    bounds = [(0.05, 0.45), (0.0, 0.3), (0.1, 0.9)]
    res = differential_evolution(
        objective_de, 
        bounds,
        seed=seed,
        maxiter=200,
        popsize=15,
        disp=False,
        tol=1e-6
    )
    if res.success:
        u_seed = parametric_to_halfvector(res.x, n)
        u_seed = project_to_sum_box(u_seed, target_sum)
        c5_seed = exact_c5(u_seed)
        if c5_seed < best_c5:
            best_c5 = c5_seed
            best_u = u_seed.copy()
    
    # Phase III: L-BFGS-B on full discretisation

        best_u
    )
    u_fine = project_to_sum_box(u_fine, n_fine / 4)
    
    def objective_lbfgs(u):
        return lse_c5(u, 200)

            objective_lbfgs,
            u_fine,
            method='L-BFGS-B',

            options={'gtol': 1e-10, 'maxiter': 800}

            u_final = res.x

            u_final = u_fine

            best_c5 = c5_final
            best_u = u_final.copy()

        pass
    
    # Phase IV: Simulated Annealing polishing
    if time.time() - start_time > 0.9 * budget_s:
        return (np.concatenate([best_u, best_u[::-1]]).tolist(), float(best_c5), n)
    
    def accept_move(old_c5, new_c5, T):
        if new_c5 < old_c5:
            return True
        return np.random.rand() < np.exp((old_c5 - new_c5) / T)

        # Accept a move with probability

            idx1, idx2 = np.random.choice(m, 2, replace=False)

            if u_sa[idx1] >= delta and u_sa[idx2] <= 1 - delta:
                u_sa[idx1] -= delta
                u_sa[idx2] += delta
                u_sa = project_to_sum_box(u_sa, target_sum)

                    c5_sa = new_c5
                    if c5_sa < c5_best:
                        c5_best = c5_sa
                        u_best = u_sa.copy()
                else:
                    # Undo
                    u_sa[idx1] += delta
                    u_sa[idx2] -= delta
                    u_sa = project_to_sum_box(u_sa, target_sum)
        else:
            # Add to one, subtract from another
            idx1, idx2 = np.random.choice(m, 2, replace=False)
            delta = np.random.rand() * max_delta
            if u_sa[idx1] >= delta:
