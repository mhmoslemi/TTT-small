# Global variables
TARGET_SUM = None

dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def coordinate_descent(current_h, current_c5, start_time, budget_s):
    """Coordinate descent with adaptive step sizes"""
    h = current_h.copy()
    c5 = current_c5
    step_size = 0.05
    max_iter = 10000
    patience = 500
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Try updating each coordinate
        for idx in range(n_points):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Try small perturbations
            best_h_trial = h.copy()
            best_c5_trial = c5
            
            for pert in [-step_size, step_size]:
                h_trial = h.copy()
                h_trial[idx] += pert
                
                # Project to constraints
                h_trial = project_to_sum_box(h_trial, TARGET_SUM)
                
                c5_trial = evaluate_c5(h_trial)
                
                if c5_trial < best_c5_trial:
                    best_c5_trial = c5_trial
                    best_h_trial = h_trial
                    
            # Accept improvement
            if best_c5_trial < c5:
                h = best_h_trial
                c5 = best_c5_trial
                
        # Adaptive step size reduction
        if i % 100 == 0 and i > 0:
            if abs(c5 - current_c5) < 1e-8:
                step_size *= 0.9
            else:
                step_size = 0.05
        
        if c5 < current_c5:
            current_h = h.copy()
            current_c5 = c5

def random_local_search(current_h, current_c5, start_time, budget_s):

    h = current_h.copy()
    c5 = current_c5
    max_iter = 5000
    restarts = 10
    patience = 200
    
    for restart in range(restarts):
        if time.time() - start_time > budget_s - 5:
            break
            
        local_h = h.copy()
        local_c5 = c5
        iter_count = 0
        
        while iter_count < max_iter:
            if time.time() - start_time > budget_s - 5:
                break
                
            # Random perturbation of a few variables
            h_new = local_h.copy()
            indices = np.random.choice(n_points, size=max(1, n_points//20), replace=False)
            
            for idx in indices:
                perturbation = np.random.normal(0, 0.01)
                h_new[idx] += perturbation

            h_new = project_to_sum_box(h_new, TARGET_SUM)
            c5_new = evaluate_c5(h_new)
            
            if c5_new < local_c5:
                local_h = h_new.copy()
                local_c5 = c5_new
                iter_count = 0  # Reset patience
            else:
                iter_count += 1
                
            if local_c5 < c5:
                h = local_h.copy()
                c5 = local_c5
                
    return h, c5
