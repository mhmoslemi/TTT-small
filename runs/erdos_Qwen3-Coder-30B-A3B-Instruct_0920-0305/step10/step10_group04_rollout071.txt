import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Simulated Annealing parameters
    max_iters = 500000
    temp_start = 0.02
    temp_end = 1e-5
    
    current_h = h0.copy()
    current_c5 = best_c5
    
    # Track best improvement
    last_improvement = 0
    stagnation_count = 0
    max_stagnation = 5000

        if time.time() - start_time > budget_s * 0.85:
            break
            
        # Temperature schedule
        temp = temp_start * (1 - iteration / max_iters) + temp_end
        
        # Try targeted move with some probability
        if np.random.random() < 0.7:
            # Compute correlation to find worst shift
            corr = np.correlate(current_h, 1 - current_h, mode='full')
            max_idx = np.argmax(corr)
            d = max_idx - (n_points - 1)
            
            # Find targeted swap candidates
            if d > 0:
                candidates_i = np.where((np.arange(n_points) + d < n_points) &

                                      (current_h[np.arange(n_points) + d] == 0))[0]

            method='SLSQP',

            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}

            refined_h = res.x
            refined_h = project_to_sum_box(refined_h, TARGET_SUM)
            refined_c5 = evaluate_c5(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
