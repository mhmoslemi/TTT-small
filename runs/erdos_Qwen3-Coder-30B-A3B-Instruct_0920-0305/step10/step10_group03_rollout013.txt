def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_binary_milp(N, time_limit=30):
    """Solve binary MILP for optimal step function"""
    # Create variables
    x = cp.Variable(N, boolean=True)
    t = cp.Variable(1, integer=True)
    
    # Constraints
    constraints = [cp.sum(x) == N // 2]
    
    # Compute shifts and add constraints
    for k in range(N):
        # For shift k, we need elements from i to N-k-1
        if N - k <= 0:
            continue
        # y[i] represents x[i] * x[i+k] for i in 0..N-k-1
        # We represent y as a matrix
        y = cp.Variable(N - k, boolean=True)
        # Add McCormick constraints
        for i in range(N - k):
            constraints.append(y[i] <= x[i])
            constraints.append(y[i] <= x[i + k])
            constraints.append(y[i] >= x[i] + x[i + k] - 1)
        # Add constraint sum y >= N/2 - t
        constraints.append(cp.sum(y) >= N // 2 - t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CBC, maximumSeconds=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value
    except:
        pass
    return None

def upsample_binary(binary_vec, target_N):
    """Upsample binary vector to target_N by repeating each element"""
    upsample_factor = target_N // len(binary_vec)
    if upsample_factor <= 0:
        return binary_vec
    return np.repeat(binary_vec, upsample_factor)

    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)

    best_c5 = current_c5
    
    elapsed = time.time() - start_time
    remaining_time = time_budget - elapsed
    
    # If insufficient time, return current best
    if remaining_time < 5:
        return best_h, best_c5

    iterations = min(5000, int(remaining_time * 500))
    
    for i in range(iterations):
        # Create a small perturbation
        h_new = current_h.copy()
        
        # Select random indices to perturb
        idx1 = np.random.randint(0, len(h_new))
        idx2 = np.random.randint(0, len(h_new))
        
        # Adjust values while maintaining sum constraint
        delta = np.random.uniform(-0.1, 0.1)
        if h_new[idx1] + delta >= 0 and h_new[idx1] + delta <= 1:
            h_new[idx1] += delta
            h_new[idx2] -= delta
            
        # Project to valid constraints
        h_new = project_to_sum_box(h_new, target_sum)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new)
        
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = current_h.copy()
                best_c5 = c5_new
        elif np.random.random() < 0.05:  # Occasionally accept worse solutions
            current_h = h_new.copy()
            current_c5 = c5_new
    
    return best_h, best_c5
