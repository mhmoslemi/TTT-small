import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

# Global constants
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_from_binary(x):
    """Fast correlation of binary vector x with itself."""
    # Auto-correlation using FFT
    A = correlate(x, x, mode='full')
    # Now compute the overlap terms manually
    # O_d = sum_i x_i (1 - x_{i+d}) = sum_i x_i - sum_i x_i x_{i+d}
    # But remember that we also must consider the different indexing for positive and negative lags
    # Let's be more precise:
    # For shift d (d can range from -(n-1) to n-1):
    # O_d = sum_{i=0}^{n-1-d} x_i * (1 - x_{i+d}) 
    #     = (sum_{i=0}^{n-1-d} x_i) - sum_{i=0}^{n-1-d} x_i x_{i+d}
    # Let CumSum = cumsum(x), then we can express it as:
    # sum_{i=0}^{n-1-d} x_i = CumSum[n-1-d]
    # sum_{i=0}^{n-1-d} x_i x_{i+d} = A[n-1+d], but we must be careful due to zero padding.
    # For d positive: A[n-1+d] is the cross-correlation from x to itself at shift d
    # But we want sum_{i=0}^{n-1-d} x_i x_{i+d}, which is A[n-1+d] (this captures elements
    # where indices match in range [0, n-1-d] and [d, n-1])
    # Also A_{-d} should be extracted differently when d is positive, it's the correlation at shift -d
    
    # Alternatively, a clean direct way using A:
    # We know: sum_{i=0}^{n-1-d} x_i x_{i+d} = A[n-1+d] for d ≥ 0.
    # Because A is length 2n-1 and zeros at left and right parts due to inner products
    # In fact, A[n-1] is correlation with zero shift, at A[n-1+d] it's shift d, and at A[n-1-d] it's shift -d.
    # Correlation between x and (1-x) at d is:
    # sum_{i=0}^{n-1-d} x_i * (1 - x_{i+d})  
    # So it's (sum_{i=0}^{n-1-d} x_i) - (sum_{i=0}^{n-1-d} x_i x_{i+d})
    # Consider CumSum = cumsum(x)
    # sum_{i=0}^{k} x_i = CumSum[k] when k < n
    # if k >= n, treat specially or clip.
    
    # One clean vectorized way to compute all O_d at once:
    # Let v = cumsum(x)
    # The number of valid indices for a given shift d (for d ≥ 0) is from 0 to (n-1-d)
    # The number of such valid indices is (n-1-d) + 1 = n - d.
    # But indexing violates the assumption when > n-1. Despite this, we'll use cumulative sums.
    cumsum_x = np.cumsum(x)
    n = len(x)
    
    # Compute all O_d
    O = np.zeros(2 * n - 1)
    for d in range(-(n - 1), n):
        if d >= 0:
            # Valid pairs: [0..n-1-d] and [d..n-1]
            valid_count = n - d
            if valid_count <= 0:
                O[n - 1 + d] = 0.0
                continue
            x_at_d = x[d:d + valid_count]  # fancy slice
            x_here = x[:valid_count]
            cross = np.sum(x_here * x_at_d)
            prefix_sum_here = cumsum_x[valid_count - 1]  # cumsum of x[0..valid_count-1] = x_0 + .. + x_{valid_count-1}
            # O_d = prefix_sum_here - cross
            O[n - 1 + d] = prefix_sum_here - cross
        else:
            # d < 0: Valid pairs: [0..n-1+d] and [n+1+d..n-1] 
            valid_count = n + d
            if valid_count <= 0:
                O[n - 1 + d] = 0.0
                continue
            # Slice x from [n+1+d..n-1]
            x_at_d = x[n + d:]
            x_here = x[:valid_count]
            cross = np.sum(x_here * x_at_d)
            prefix_sum_here = cumsum_x[valid_count - 1]
            O[n - 1 + d] = prefix_sum_here - cross
    
    return O * dx

def compute_corr_fast(x):
    """Faster new computation using more direct approach."""
    # Use correlation with (1-x)
    # Cross-correlation defined using numpy correlate function:
    b = 1 - np.array(x)
    y = correlate(x, b, mode='full')
    return y * dx

def evaluate_c5_fast(h):
    """Evaluate C5 value for a real-valued h vector using correlation."""
    corr = compute_corr_fast(h)
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    toler = 1e-10
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    # Binary search
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < toler:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

# Step functions and helper functions
def compute_delta_auto_correlation(old_x, i, j):
    """Update auto-correlations after swapping indices i and j using slice-based approach."""
    n = len(old_x)

    # Add padding to support slicing w/ negative indices
    padded = np.pad(old_x, (n - 1, n - 1), mode='constant', constant_values=0)
    
    # Effect of turning off the 1 at index i
    old_padded_start = n - 1 + i
    chunk_i = padded[old_padded_start : old_padded_start + 2 * n - 1]
    # Flip the chunk to compute the opposite
    flipped_i = chunk_i[::-1]  # Reverse it

    # Effect of turning on the 0 at index j 
    old_padded_start_j = n - 1 + j
    chunk_j = padded[old_padded_start_j : old_padded_start_j + 2 * n - 1]
    # Again flipped for deltapad
    flipped_j = chunk_j[::-1]  # Reverse it  

    # Delta is contribution of i - contribution of j
    # So delta_a[i+j] = flipped_i[j] - flipped_j[i], so 
    # Let ipos = d + n - 1 (for access) -- within [0, 2n-2], but it’s bit shaky. Let's fix:
    # Item at position k in A_full corresponds to offset (k - (n - 1)) = k - n + 1
    # So i gives shift from (i - n + 1) to i, and j from (j - n + 1) to j
    # Better: treat delta_a = A_new - A_old as final assertion at trace:
    delta = flipped_j - flipped_i
    
    return delta

def swap_indices_binary(x, i, j):
    """Swap elements at indices i and j in binary vector x."""
    x_new = x.copy()
    x_new[i], x_new[j] = x_new[j], x_new[i]
    return x_new

def get_worst_shift(x):
    """Find the shift that maximizes the overlap."""
    O = compute_corr_from_binary(x)
    # Exclude edge encroaching scenarios, look at non-trivial overlaps
    max_idx = np.argmax(O)
    max_shift = max_idx - (len(O) // 2)
    return max_shift

def sample_targeted_swap(x, max_shift):
    """Sample a good swap to reduce max_shift error."""
    n = len(x)
    # Consider two sets:
    # U = {i | x[i] = 1 and x[i+max_shift] = 0}
    # V = {j | x[j] = 0 and x[j+max_shift] = 1}
    
    U = []
    V = []
    
    # When max_shift >= 0, valid i is 0 to n-1-max_shift  
    if max_shift >= 0:
        for i in range(n - max_shift):
            if x[i] == 1 and x[i + max_shift] == 0:
                U.append(i)
        for j in range(n - max_shift):
            if x[j] == 0 and x[j + max_shift] == 1:
                V.append(j)
    else:
        # When max_shift is negative, valid i is -max_shift to n-1  
        for i in range(-max_shift, n):
            if x[i] == 1 and x[i + max_shift] == 0:
                U.append(i)
        for j in range(n + max_shift):
            if x[j] == 0 and x[j + max_shift] == 1:
                V.append(j)

    if U and V:
        # Pick uniformly
        import random
        i = random.choice(U)
        j = random.choice(V)
        return i, j
    else:
        # Fall back to random
        while True:
            i, j = np.random.choice(n, size=2, replace=False)
            if x[i] != x[j]:
                return i, j

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase A: Binary SA
    T0 = 2 * dx
    Tf = 1e-6 * dx
    steps = 300000
    temp_schedule = np.linspace(T0, Tf, steps)
    accepted = 0

    # Initialize binary vector
    init_x = np.zeros(n_points, dtype=int)
    # Fill with exactly n/2 ones
    init_x[:TARGET_SUM] = 1
    np.random.shuffle(init_x)
    x = init_x.copy()
    
    # Evaluate current C5 (will be updated)
    curr_c5 = evaluate_c5_fast(x.astype(float))
    best_c5 = curr_c5
    best_x = x.copy()
    
    # Track time
    cutoff = start_time + 0.70 * budget_s

    # Core Simulated Annealing loop (600s * 0.70 = 420s)
    for step in range(steps):
        if time.time() > cutoff:
            break
        
        if step % 1000 == 0:
            # Log every 1000 steps
            pass
            
        # Decide type of move
        if np.random.random() < 0.6:
            # Targeted move
            max_shift = get_worst_shift(x)
            try:
                i, j = sample_targeted_swap(x, max_shift)
            except:
                i, j = np.random.choice(n_points, size=2, replace=False)
        else:
            # Random swap
            i, j = np.random.choice(n_points, size=2, replace=False)
            while x[i] == x[j]:
                i, j = np.random.choice(n_points, size=2, replace=False)

        # Try the swap
        x_new = swap_indices_binary(x, i, j)
        c5_new = evaluate_c5_fast(x_new.astype(float))
        
        delta = c5_new - curr_c5
        
        if delta < 0 or np.random.random() < np.exp(-delta / temp_schedule[step]):
            x = x_new
            curr_c5 = c5_new
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_x = x.copy()
            accepted += 1

    # Phase B: Continuous refinement
    curr_h = best_x.astype(float)
    curr_h = project_to_sum_box(curr_h, TARGET_SUM)
    
    # First local optimization
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
    try:
        res = minimize(
            evaluate_c5_fast, 
            curr_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1500, 'ftol': 1e-12}
        )
        if res.success:
            curr_h = res.x
            curr_h = project_to_sum_box(curr_h, TARGET_SUM)
    except:
        pass

    # Second local refinement
    try:
        res = minimize(
            evaluate_c5_fast, 
            curr_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12}
        )
        if res.success:
            curr_h = res.x
    except:
        pass

    # Ensure final projection
    curr_h = project_to_sum_box(curr_h, TARGET_SUM)

    final_c5 = evaluate_c5_fast(curr_h)
    return (curr_h.tolist(), final_c5, n_points)
