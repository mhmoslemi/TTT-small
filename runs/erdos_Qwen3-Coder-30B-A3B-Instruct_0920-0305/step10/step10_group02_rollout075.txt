```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from math import gcd

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_binary_sequence(a, b, n):
    """Evaluate a binary sequence generated by linear congruence"""
    h = np.zeros(n, dtype=int)
    for i in range(n):
        if ((i * a + b) % n) < n // 2:
            h[i] = 1
    return h

def find_best_linear_sequence():
    """Find the best binary sequence from linear congruences"""
    best_c5 = float('inf')
    best_h = None
    
    # Find all numbers coprime with n_points
    coprimes = [i for i in range(1, n_points) if gcd(i, n_points) == 1]
    n_coprimes = len(coprimes)
    
    for a in coprimes:
        for b in range(n_points):
            h = evaluate_binary_sequence(a, b, n_points)
            h_float = h.astype(float)
            h_projected = project_to_sum_box(h_float, n_points / 2)
            c5 = evaluate_c5(h_projected)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_projected.copy()
    
    return best_h, best_c5

def smooth_boundaries(h, window_size=3):
    """Smooth boundaries between 0s and 1s"""
    h_smooth = h.copy()
    indices = np.where(np.diff(h) != 0)[0]
    
    # For each boundary, interpolate linearly
    for idx in indices:
        # Determine window
        start = max(0, idx - window_size//2)
        end = min(len(h), idx + window_size//2 + 1)
        # Linear interpolation from 0 to 1 (or 1 to 0) 
        if h[idx] == 1 and h[idx+1] == 0:
            # Going down
            h_smooth[start:end] = np.linspace(1, 0, end-start)
        elif h[idx] == 0 and h[idx+1] == 1:
            # Going up
            h_smooth[start:end] = np.linspace(0, 1, end-start)
    
    return h_smooth

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Start with the best parametric construction
    best_h, best_c5 = find_best_linear_sequence()
    
    # Local search around the best solution
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Try different combinations of local search strategies
    max_iterations = 5000
    for i in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Hybrid strategy: try several local moves
        # Strategy 1: Random swap with small magnitude
        h_new = current_h.copy()
        idx1 = np.random.randint(0, n_points)
        idx2 = np.random.randint(0, n_points)
        
        # Swap values
        h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
        
        # Project back
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Occasionally do a larger perturbation
        if i % 100 == 0 and np.random.random() < 0.3:
            # Larger neighborhood swap
            h_new = current_h.copy()
            idx1 = np.random.randint(0, n_points)
            idx2 = (idx1 + np.random.randint(1, n_points//4)) % n_points
            h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            c5_new = evaluate_c5(h_new)
            
            if c5_new < current_c5:
                current_h = h_new.copy()
                current_c5 = c5_new
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
    
    # Try smoothing and refinement
    try:
        h_smooth = smooth_boundaries(best_h)
        h_smooth = project_to_sum_box(h_smooth, TARGET_SUM)
        c5_smooth = evaluate_c5(h_smooth)
        if c5_smooth < best_c5:
            best_h = h_smooth
            best_c5 = c5_smooth
            
        # Gradient-based refinement
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        res = minimize(
            objective, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        if res.success:
            final_h = res.x
        else:
            final_h = best_h
        final_h = project_to_sum_box(final_h, TARGET_SUM)
        final_c5 = evaluate_c5(final_h)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
            
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
```
