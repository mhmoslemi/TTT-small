import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Minimize, Problem
import math

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Fast correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Extract C5 from correlation"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr)

def project_to_box_sum(x, target_sum):
    """Project onto [0,1] box with fixed sum"""
    x = np.array(x)
    low, high = 0.0, 1.0
    a, b = low - np.max(x), high - np.min(x)
    while abs(a - b) > 1e-10:
        m = (a + b) / 2.0
        proj = np.clip(x + m, low, high)
        if np.sum(proj) > target_sum:
            a = m
        else:
            b = m
    proj = np.clip(x + (a + b) / 2.0, low, high)
    return proj

def generate_parametric_step_functions():
    """Generate representative step function shapes"""
    # Piecewise constant with 2-4 segments
    base_functions = []
    
    # Simple uniform
    h_uniform = np.ones(n_points) * 0.5
    base_functions.append(("uniform", h_uniform.copy()))
    
    # Two-slope
    h_two_slope = np.concatenate([np.ones(n_points//2), np.zeros(n_points//2)])
    base_functions.append(("two_slope", h_two_slope.copy()))
    
    # Truncated exp
    logspace = np.linspace(0, 3, n_points//2)
    exp_part = np.exp(-logspace)
    h_exp = np.concatenate([exp_part, np.zeros(n_points - n_points//2)])
    base_functions.append(("truncated_exp", h_exp.copy()))
    
    return base_functions

def optimize_parametric(h_base, max_iter=1000):
    """Optimize along parametric shape"""
    
    def objective(h):
        return evaluate_c5(h)
    
    # Use gradient descent
    step_size = 0.001
    best_h = h_base.copy()
    best_c5 = evaluate_c5(best_h)
    
    for i in range(max_iter):
        h_current = best_h.copy()
        grad = np.zeros_like(h_current)
        eps = 1e-5
        
        # Numerical gradient
        for j in range(len(h_current)):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = project_to_box_sum(h_plus, TARGET_SUM)
            val_plus = evaluate_c5(h_plus)
            
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = project_to_box_sum(h_minus, TARGET_SUM)
            val_minus = evaluate_c5(h_minus)
            
            grad[j] = (val_plus - val_minus) / (2 * eps)
        
        # Gradient descent update
        h_new = h_current - step_size * grad
        h_new = project_to_box_sum(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        else:
            step_size *= 0.999  # Adaptive decay
    
    return best_h, best_c5

def local_worst_shift_swap(h_current):
    """Perform a local swap that reduces maximum overlap"""
    corr = compute_corr_fft(h_current)
    max_shift_idx = np.argmax(corr)
    max_shift = max_shift_idx - (len(h_current) - 1)
    
    # Find best inversion near max shift
    best_delta = 0
    best_pos = -1
    best_h = h_current.copy()
    
    for i in range(max(0, max_shift), min(len(h_current), max_shift + n_points)):
        # Try flipping a small window
        h_test = h_current.copy()
        range_to_flip = min(20, len(h_current) - i)
        for j in range(range_to_flip):
            h_test[i + j] = 1.0 - h_test[i + j]
        
        # Scale to maintain sum
        h_test = project_to_box_sum(h_test, TARGET_SUM)
        
        c5_test = evaluate_c5(h_test)
        if c5_test < evaluate_c5(best_h):
            best_h = h_test.copy()
            best_delta = c5_test - evaluate_c5(best_h)
    
    return best_h, evaluate_c5(best_h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Generate base functions and optimize each
    base_functions = generate_parametric_step_functions()
    
    best_h = None
    best_c5 = float('inf')
    
    for func_name, h_base in base_functions:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.8:
            break
            
        # Optimize along this shape
        h_opt, c5_opt = optimize_parametric(h_base, max_iter=500)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
    
    # Iterate wors-shift swap phase
    iter_count = 0
    while time.time() - start_time < budget_s * 0.95:
        if iter_count > 100:
            break
        iter_count += 1
        h_new, c5_new = local_worst_shift_swap(best_h)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        else:
            break
    
    # Final polishing with gradient descent
    if time.time() - start_time < budget_s:
        h_final, c5_final = optimize_parametric(best_h, max_iter=500)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    
    return (best_h.tolist(), float(best_c5), n_points)
