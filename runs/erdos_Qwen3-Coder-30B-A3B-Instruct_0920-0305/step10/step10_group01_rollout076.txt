# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def soft_max_grad(h, corr_values, alpha):
    """Compute gradient of soft max surrogate"""
    # Compute w_k = exp(alpha * c_k) / sum(exp(alpha * c_j))
    max_c = np.max(corr_values)
    exp_c = np.exp(alpha * (corr_values - max_c))
    w = exp_c / np.sum(exp_c)
    
    # Compute the symmetrized weight kernel
    w_sym = w + np.concatenate([w[-1:0:-1], w[:1]])
    
    # Compute g_j = sum_k w_k * h_{j+k}
    g_full = correlate(w_sym, h, mode='full')
    g = g_full[len(h)-1:len(h)-1+len(h)]
    
    # Gradient is 1 - g
    grad = np.ones_like(h) - g
    return grad

    # Compute smooth objective
    smooth_obj = np.log(np.sum(exp_c)) / alpha + max_c
    
    # Compute gradient
    w_sym = w + np.concatenate([w[-1:0:-1], w[:1]])
    g_full = correlate(w_sym, h, mode='full')
    g = g_full[len(h)-1:len(h)-1+len(h)]
    grad = np.ones_like(h) - g
    
    return smooth_obj, grad
