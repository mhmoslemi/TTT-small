import numpy as np
from scipy.signal import correlate
import time
import random
from scipy.optimize import minimize

def compute_auto_correlation(x):
    """Compute auto-correlation using FFT"""
    return correlate(x, x, mode='full')

def estimate_c5_from_auto_corr(A, x):
    """Estimate C5 from auto-correlation A and binary vector x"""
    n = len(x)
    # This calculates the overlap as sum(x_i * (1 - x_{i+d}))
    # which equals sum(x_i) - A_d
    # But A_d incorporates sum(x_i * x_{i+d})
    # So overlap for shift d is sum(x_i) - A_d where i ranges from 0 to n-1-d
    
    # The subset was organized to be computed automatically with numpy
    # But we compute the partial sums correctly now 
    cumsum_x = np.concatenate([[0], np.cumsum(x)])
    # We'll compute the overlap term for each shift d
    # For shift d, we want sum_{i=0}^{n-1-d} x_i * (1 - x_{i+d})
    # which is sum(x_i)_{i=0}^{n-1-d} - sum(x_i * x_{i+d})_{i=0}^{n-1-d}
    # which is cumsum_x[n-1-d] - A_d (for positive d)
    
    # We calculate overlaps for all shift offsets
    # Full correlation size is 2*n-1 
    full_len = 2*n - 1
    # Adjust A to size full_len to match correlation outputs
    if len(A) != full_len:
        A = np.pad(A, (0, full_len - len(A)), mode='constant')
    # Now we compute overlaps as sum of x_i - sum of x_i * x_{i+d}
    N = np.sum(x)  # This should be n/2 for binary
    overlap = np.zeros(full_len)
    for i in range(full_len):
        d = i - (n-1)
        if d > 0:
            # For d positive, need to sum over i from 0 to n-1-d,
            # and subtract A[i] which is sum over i from 0 to n-1-d of x_i * x_{i+d}
            # but that's equivalent to sum of x_{i+d} * x_i  
            # -> A[i] = sum_{i=0}^{n-1-d} x_i * x_{i+d} 
            overlap[i] = (cumsum_x[-1] - cumsum_x[n-1-d]) - A[i]
        elif d < 0:
            # For d negative, like sum_{i=0}^{n-1+d} x_i * x_{i+d}
            # by definition A[i] when i = -(d) + n-1 which gives index (n-1) - (-d) = n-1+d 
            # The effective shift d is from positive shift -d in terms of the actual
            # array shift. Maybe just get from it correctly via index
            overlap[i] = cumsum_x[n-1+d] - A[i]
        else:
            # d==0 case  
            overlap[i] = N - A[i]
    
    return overlap.max() * (2.0 / n)

def compute_corr_and_c5(x):
    """Compute C5 evaluation directly from x vector (binary)"""
    n = len(x)
    A = compute_auto_correlation(x)
    # Reconstruct sums via cumulative
    cum_x = np.concatenate([[0], np.cumsum(x)])
    N = np.sum(x)
    
    # Construct optimized overlap array computing overlaps term by term
    max_overlap = -np.inf
    full_len = 2 * n - 1
    A = A[:full_len]  # Ensure same length
    
    # Loop over all possible shifts
    for d in range(-(n-1), n):
        if d >= 0:
            # Need sum_{i=0}^{n-1-d} (x_i * (1 - x_{i+d}))
            # = sum(x_i)_{i=0}^{n-1-d} - sum(x_i * x_{i+d})
            # i from 0 to n-1-d (inclusive)
            # terms are x_i * x_{i+d} which is A[d + n - 1] for correct indexing ?
            # But note that d = 0 -> shift 0 -> A[n-1],
            # so A[0] is shift -(n-1)  
            # A should be indexed from A[n-1] to A[n-1] + (n-1) inclusive = A[full_len-1] 
            start_idx = d + n - 1
            if start_idx < full_len:
                # range from 0 to n-1-d 
                if d == 0:
                    numterms = n
                    overlap = cum_x[-1] - A[start_idx] 
                else:
                    # sum_i=0^{n-1-d} x_i - sum_{j=0}^{n-1-d} x_j * x_{j+d} 
                    # = cumsum_x[n-d] - A[d + n - 1]
                    numterms = n - d
                    overlap = cum_x[n-d] - A[start_idx]
                max_overlap = max(max_overlap, overlap)
            else:
                # d too large or already beyond matrix size, skip
                pass
                
        else:  # d < 0
            # Instead, d = -|d|, we want sum_{i=0}^{n-1+d} x_i * x_{i+d}
            # which equals: sum_{i=0}^{n-1+d} x_i x_{i+(-d)} for d=-k with k>0
            # let's call the part of correlation index as: i+|d| = index within
            # original array, so i = start_idx - |d| 
            # in correlation corr[i] where i relates to i of x[i] and shift d:
            # index pairs are at corr[i + n-1 + d] ???
            # the actual array offset is such that d = 1 means i:d=1 => x_{i} * x_{i+1}
            # and i in correlation corresponds to diff = j - i = d  
            # know A[j] records the sum of products ...
            abs_d = -d
            if abs_d < n:
                # From day 0, shift d = -abs_d means store x[i] * x[i-abs_d]
                # So sum over i from 0 to n-1-abs_d
                # Total sum x[i] over that range is cumsum_x[n-1-abs_d+1], that mismatched
                # So sum of the range is cumsum_x[n-1-abs_d+1]
                # And overlap sum is x_i * x_{i-d}  over i from 0 to n-1-d  
                # OK so start at index n-1 + d = n-1 - abs_d
                start_idx = n-1 + d  # Should be n-1 - abs(d)
                if start_idx >= 0 and start_idx < full_len:
                    # Using overlapping intervals 
                    nums = n - abs_d
                    # shift d is from - (n-1) to n-1
                    # let's build it properly with both end points adjusted
                    # better use cumulative sum approach, move back monotonic easily
                    # use actual actual ranges
                    if d <= 0:
                        if abs_d <= n:
                            start = 0
                            end = n + d  # So for d=-1, end=n-1; d=-2, end=n-2  etc.
                            if end > 0:
                                overlap = cum_x[end] - A[start_idx]  # indices aligned?
                                max_overlap = max(max_overlap, overlap)
            else:
                # abs_d >= n, just skip
                pass                
    
    return max_overlap * (2.0 / n) if max_overlap != -np.inf else np.inf

def compute_delta_corr(x, i, j, A):
    """Compute change in auto correlation due to swapping positions i,j"""
    n = len(x)
    # x[i], x[j] must be different (one is 0, other 1).
    # New pattern after swap: x_i = 0, x_j = 1
    # A_d is sum of x_i * x_{i+d}  
    # Before swap:
    # A_d contributes x_i * x_{i+d} and x_j * x_{j+d}
    # After swap:
    # x_i = 0 so contribution to A_d on x_i is now 0 
    # x_j = 1 so contribution to A_d on x_j is now x_{j+d} 
    # Change in A_d = - x_i * x_{i+d} + x_{j+d}
    
    # Note:
    # A[i] corresponds to shift - (n-1) 
    # So shift d is index (i + n-1 + d)
    # for shift 1 (d=1), index = n-1 + 1 = n
    # for shift 0, index = n-1
    # for shift -1, index = n-2
    
    x_pad = np.concatenate([[0]*(n-1), x, [0]*(n-1)], dtype=np.float64)
    delta = np.zeros(2*n - 1)

    # All that matters is those places where the swaps influence correlations:
    shifts = []
    for d in range(-(n-1), n):
        # Make sure we use the correct correlation index
        corr_index = d + (n-1)  
        if 0 <= corr_index < 2*n - 1:
            shifts.append((d, corr_index))
            
    # We'll modify the delta vector; for each shift d, there is a formula involving update
    # Since update involves considering d major shifts, we build the shifted indices for i and j
    
    # But simpler approach:
    # A_d = sum_{i=0}^{n-1-d} x_i * x_{i+d}  
    # In offset terms this considers x_k * x_{k+d} where k goes from some offset to n-1-d
    # And the change depends on exactly how the contributions change. I think it's:
    # For each d, there are summands that are affected: contributions to A_d involving i and j
    
    # Mutually 
    for d in range(-(n-1), n):
        corr_idx = d + (n-1)
        if not 0 <= corr_idx < 2*n-1:
            continue
            
        # Calculation of terms that change
        # There are only 4 ways that the swap affects the auto-correlation A_d:
        # 1. i in contributing to A_d terms (i + d >= 0) => x_i * x_{i+d}
        # 2. j in contributing to A_d terms, same as above (j + d >= 0) => x_j * x_{j+d} 
        # The A_d is the sum over i from 0 to n-1-d of x_i * x_{i+d}
        
        change = 0.0
        # contributions of i
        if 0 <= i+d < n:
            change -= x[i] * x[i+d]  # need to be zero after swap
            # but we will make x[i]=0, then add x[i+d] if it is valid to become contribution
            change += 0 * x[i+d] if x[i] == 1 else 1 * x[i+d]  # so delta of term
        # analogously for j
        if 0 <= j+d < n:
            change -= x[j] * x[j+d]
            change += x[j+d] * 1 if x[j] == 0 else x[j+d] * 0

        delta[corr_idx] = change

    return delta

def project_to_sum(x, desired_sum):
    """Project x onto the box [0,1] with sum = desired_sum"""
    # Binary search for the right lambda
    a, b = -1.0, 1.0
    tol = 1e-10
    
    # Remove one element and solve for remaining sum = desired_sum - x[k]
    # But do again multiple times or better yet:
    # We can always iterate to bring average to target if the original is not in range [0,1]  
    # But acceleration method is binary search for lambda
    # Placing all elements, add lambda, clamp to [0,1], sum must be desired_sum
    
    def eval_sum(x_plus_lambda):
        return np.sum(np.clip(x_plus_lambda, 0, 1))
    
    lam = (a + b) / 2
    for _ in range(50):
        clamped = np.clip(x + lam, 0, 1)
        current_sum = np.sum(clamped)
        if abs(current_sum - desired_sum) < tol:
            break
        if current_sum > desired_sum:
            b = lam
        else:
            a = lam
        lam = (a + b)/2
        
    return np.clip(x + lam, 0, 1)

def init_binary_h(seed, n=240):
    """Initialize a good binary h vector with approximately n/2 ones"""
    np.random.seed(seed)
    h = np.zeros(n)
    h[:n//2] = 1
    np.random.shuffle(h)
    return h.astype(int)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 240
    dx = 2.0 / n
    desired_sum = n / 2.0

    # Main time allocation  
    time_left = budget_s    
    tSA = 0.7 * time_left  # 70% for SA
    
    # Initialize a binary vector with exactly half 1s
    x = init_binary_h(seed, n)
    
    # Get initial auto_correlation and calculate C5
    A = compute_auto_correlation(x)
    best_c5 = estimate_c5_from_auto_corr(A, x)
    best_x = x.copy()
    
    # For faster correlation calculation, use precomputed indices
    iters = 0
    accepted = 0
    t0 = time.time()
    T = 2. * dx
    T_final = 1e-6 * dx
    N_tot = 300000  # Number of moves to cool down
    cooling_rate = (T - T_final) / N_tot
    # Parameters for fallback 
    stagnation_period = 20000 
    stagnation_threshold = 1e-5
    restart = False
    
    # Record best ever seen during SA
    best_seen = best_c5
    
    while time.time() - start_time < tSA:
        # Decide move type
        if random.random() < 0.6:  # Targeted move
            # Compute current worst shift
            # Recompute from scratch or estimate
            A = compute_auto_correlation(best_x)
            # Compute overlap for each shift and get max
            max_overlap = -np.inf
            try:
                overlap_term = np.zeros(2*n - 1)
                cum_x = np.concatenate([[0], np.cumsum(best_x)])
                for d in range(-(n-1), n):
                    idx = d + (n-1)
                    if 0 <= idx < len(A):
                        if d >= 0:
                            overlap = cum_x[n-d] - A[idx]
                        else:
                            overlap = cum_x[n-1+d] - A[idx]
                        max_overlap = max(max_overlap, overlap)
                worst_shift = None
                for d in range(-(n-1), n):
                    idx = d + (n-1)
                    if 0 <= idx < len(A):
                        if d >= 0:
                            val = cum_x[n-d] - A[idx]
                        else:
                            val = cum_x[n-1+d] - A[idx]
                        if abs(val - max_overlap) < 1e-10:
                            worst_shift = d
                            break
                if worst_shift is not None:
                    # Find sets
                    U = []
                    V = []
                    for i in range(n):
                        if best_x[i] == 1:
                            j = i + worst_shift
                            if 0 <= j < n and best_x[j] == 0:
                                U.append(i)
                        elif best_x[i] == 0:
                            j = i + worst_shift
                            if 0 <= j < n and best_x[j] == 1:
                                V.append(i)
                    if U and V:
                        # Pick one from each set
                        i = random.choice(U)
                        j = random.choice(V)
                        # Try swap
                        trial_x = best_x.copy()
                        trial_x[i], trial_x[j] = trial_x[j], trial_x[i]
                        
                        # Compute delta
                        delta_A = compute_delta_corr(best_x, i, j, A)
                        trial_A = A + delta_A
                        
                        # Evaluate trial
                        trial_c5 = estimate_c5_from_auto_corr(trial_A, trial_x)
                        # Accept if better or with probability exp(-Δ/T)
                        delta = trial_c5 - best_c5
                        if delta < 0 or random.random() < np.exp(-delta / T):
                            best_x = trial_x
                            A = trial_A
                            best_c5 = trial_c5
                            accepted += 1
                            if trial_c5 < best_seen:
                                best_seen = trial_c5
                else:
                    # Fallback: random swap
                    i, j = random.sample(range(n), 2)
                    while best_x[i] == best_x[j]:
                        i, j = random.sample(range(n), 2)
                    trial_x = best_x.copy()
                    trial_x[i], trial_x[j] = trial_x[j], trial_x[i]
                    trial_A = A.copy()
                    # Recompute from scratch for simplicity
                    trial_A = compute_auto_correlation(trial_x)
                    trial_c5 = estimate_c5_from_auto_corr(trial_A, trial_x)
                    delta = trial_c5 - best_c5
                    if delta < 0 or random.random() < np.exp(-delta / T):
                        best_x = trial_x
                        A = trial_A
                        best_c5 = trial_c5
                        accepted += 1
                        if trial_c5 < best_seen:
                            best_seen = trial_c5
                        
            except Exception as e:
                # Fallback on error to simple random
                i, j = random.sample(range(n), 2)
                while best_x[i] == best_x[j]:
                    i, j = random.sample(range(n), 2)
                trial_x = best_x.copy()
                trial_x[i], trial_x[j] = trial_x[j], trial_x[i]
                # Compute with full corr
                trial_A = compute_auto_correlation(trial_x)
                trial_c5 = estimate_c5_from_auto_corr(trial_A, trial_x)
                delta = trial_c5 - best_c5
                if delta < 0 or random.random() < np.exp(-delta / T):
                    best_x = trial_x
                    A = trial_A
                    best_c5 = trial_c5
                    accepted += 1
                    if trial_c5 < best_seen:
                        best_seen = trial_c5

        else:  # Random move
            i, j = random.sample(range(n), 2)
            while best_x[i] == best_x[j]:
                i, j = random.sample(range(n), 2)
            trial_x = best_x.copy()
            trial_x[i], trial_x[j] = trial_x[j], trial_x[i]
            trial_A = A.copy()
            # Recompute from scratch for simplicity
            trial_A = compute_auto_correlation(trial_x)
            trial_c5 = estimate_c5_from_auto_corr(trial_A, trial_x)
            delta = trial_c5 - best_c5
            if delta < 0 or random.random() < np.exp(-delta / T):
                best_x = trial_x
                A = trial_A
                best_c5 = trial_c5
                accepted += 1
                if trial_c5 < best_seen:
                    best_seen = trial_c5
        
        # Cooling schedule
        T = max(T_final, T - cooling_rate)
        iters += 1
        
        # Stagnation restart
        if iters % stagnation_period == 0 and accepted > 0:
            # Restart if no significant improvement
            # Currently we are log_new = None, just estimate
            if accepted < 50:
                pass  # Just leave it as is unless repeats
            else:
                # Attempt partial reinitialization
                print("Restarting from partial re-initialization")
                # Just re-sample target number of 1s
                new_x = np.zeros(n)
                new_x[:n//2] = 1
                np.random.shuffle(new_x)
                A = compute_auto_correlation(new_x)
                new_c5 = estimate_c5_from_auto_corr(A, new_x)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_x = new_x.copy()
                    A = A.copy()
                    best_seen = new_c5
                    
    # Phase B: Local continuous refinement
    
    # Convert binary to continuous
    h0 = best_x.astype(float)
    
    def smooth_h(h_in):
        w = 5
        h_smooth = np.convolve(h_in, np.ones(w)/w, mode='same')
        return h_smooth
    
    # Smooth immediately then project
    h_smooth = smooth_h(h0)
    h_proj = project_to_sum(h_smooth, desired_sum)
    
    # First SLSQP run
    def obj_func(h):
        # use full correlation
        A_h = np.correlate(h, 1-h, mode='full')
        return - np.max(A_h) * dx  # We will minimize -f, hence we negate
    
    bounds = [(0,1) for _ in range(n)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - desired_sum}

    res1 = minimize(obj_func, h_proj, method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12})
    
    if res1.success:
        h_refined = res1.x
        h_refined = project_to_sum(h_refined, desired_sum)
        C5_refined = - obj_func(h_refined)
        if C5_refined < best_c5:
            best_c5 = C5_refined
            best_x = h_refined.copy()
    
    # Second refinement
    h_smooth2 = smooth_h(h_refined)
    h_proj2 = project_to_sum(h_smooth2, desired_sum)
    
    res2 = minimize(obj_func, h_proj2, method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12})
    
    if res2.success:
        h_final = res2.x
        h_final = project_to_sum(h_final, desired_sum)
        C5_final = - obj_func(h_final)
        if C5_final < best_c5:
            best_c5 = C5_final
            best_x = h_final.copy()
    
    # Final exact projection
    best_h = project_to_sum(best_x, desired_sum)
    final_c5 = best_c5
    
    return (best_h.tolist(), float(final_c5), n)
