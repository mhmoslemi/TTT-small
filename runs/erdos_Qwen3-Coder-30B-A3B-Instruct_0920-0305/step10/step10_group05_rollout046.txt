import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from scipy.fft import fft, ifft

def project_to_sum_box(u, target_sum):
    """Project vector u onto box constraints [0,1] with sum = target_sum using water-filling"""
    def f(lam):
        projected = np.clip(u + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    for _ in range(30):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(u + (a + b)/2, 0, 1)
    return result

def autocorr_fft(h):
    """Compute aperiodic autocorrelation via FFT"""
    h_rev = h[::-1]
    n = len(h)
    L = 2 * n - 1
    corr = ifft(fft(h_rev, L) * fft(h, L)).real
    return corr

def exact_c5_half(u):
    """Compute exact C5 from half-vector u"""
    n = len(u)
    h = np.concatenate([u, u[::-1]])
    dx = 2.0 / (2*n)
    A = autocorr_fft(h)
    return 1 - dx * np.min(A)

def lse_c5_half(u, beta):
    """Compute smooth surrogate C5 using log-sum-exp"""
    n = len(u)
    h = np.concatenate([u, u[::-1]])
    dx = 2.0 / (2*n)
    A = autocorr_fft(h)
    return 1 - dx * np.log(np.sum(np.exp(beta * A))) / beta

def grad_lse_c5_half(u, beta):
    """Compute gradient of smooth surrogate C5"""
    n = len(u)
    h = np.concatenate([u, u[::-1]])
    dx = 2.0 / (2*n)
    A = autocorr_fft(h)
    w = np.exp(beta * A)
    w_sum = np.sum(w)
    
    # Compute gradient
    # For each i, compute sum_k w_k * (h_{i+k} + h_{i-k})
    grad = np.zeros_like(u)
    
    # Shift h for both directions
    h_left = np.concatenate([h, [0] * (n-1)])
    h_right = np.concatenate([[0] * (n-1), h])
    
    # Compute matrix form efficiently using FFT
    # matrix element (i,k) = h_{i+k} + h_{i-k} where indices wrap
    grad_terms = np.zeros_like(h)
    for i in range(n):
        for k in range(-n+1, n):
            if 0 <= i + k < len(h) and 0 <= i - k < len(h):
                grad_terms[i] += h[i + k] + h[i - k]
    
    # Get the convolution: 1 sum_k w_k * (h_{i+k}+h_{i-k}) for i from 0 to n-1
    # Or more directly:
    grad_terms = np.zeros_like(h)
    for i in range(n):
        for k in range(-n+1, n):
            idx1 = i + k
            idx2 = i - k
            if 0 <= idx1 < len(h) and 0 <= idx2 < len(h):
                grad_terms[i] += h[idx1] + h[idx2]
    
    # Better approach
    # unfold the convolution: sum_k w_k * (h_{i+k} + h_{i-k}) 
    # This is the same as: w * (h + shift_left(h) + shift_right(h))
    # But h refers to the full array
    # w and h have length 2n-1 = L
    w_pad = np.zeros(2*n-1)
    w_pad[:len(w)] = w / w_sum * dx
    h_pad = np.concatenate([h, [0] * (n-1)])
    h_pad = np.concatenate([[0] * (n-1), h])  # shift right
    
    # Instead, using dummy convolution
    # axis should shift it via convolution
    # build 2n-1 length arrays
    h_full = np.concatenate([h, [0] * (n-1), h])
    w_full = np.concatenate([w, [0] * (n-1)])
    
    # But more directly: 
    # grad[i] = sum_k w[k] * ( h[i+k] + h[i-k] ) for k=-n+1...n-1
    # But shifting h properly in indices
            
    grad_full = np.zeros_like(h_full)
    for i in range(n):
        for k in range(-n+1, n):
            if 0 <= i+k < len(h) and 0 <= i-k < len(h):
                grad_full[i] += w[k + n-1] * (h[i+k] + h[i-k])
    
    # Simpler  
    # assume w is same length as A = 2n-1 
    # compute convolve signal with w:
    # The sparse form is hard to do correctly; use convolution identity
    # easiest: 
    # compute w * h on full size with real convolution
    # and compute contributions from h shifted
    
    # Use periodic reflection: when k goes outside bounds, wrap via periodicity
    grad = np.zeros(n)
    for i in range(n):
        total = 0.0
        for k in range(-n+1, n):
            if 0 <= i+k < len(h) and 0 <= i-k < len(h):
                total += w[k + n-1] * (h[i+k] + h[i-k])
        grad[i] = total * dx / w_sum
        
    return grad[:n]

def parametric_to_halfvector_3level(params, n):
    """3-level symmetric parametrization: a (edge width), h (middle height)"""
    # a: edge width; 0 <= a <= 0.5
    # h: middle height must satisfy 2*a + (2-2*a)*h = 1 => h = (1-2a)/(2*(1-a))
    a, = params
    a = max(0.0, min(0.5, a))
    h = (1.0 - 2.0 * a) / (2.0 * (1.0 - a)) if a < 1.0 else 0.5
    u = np.ones(n // 2)
    m = int(n // 2)
    if a > 0:
        u[:int(a * n // 2)] = 1.0
        u[int(a * n // 2):m - int(a * n // 2)] = h
    else:
        u[0:m] = h
    return u

def parametric_to_halfvector_5level(params, n):
    """5-level parametrization: a (edge), r (ramp), p (middle height)"""
    a, r, = params
    if a < 0: a = 0
    if a > 0.5: a = 0.5
    if r < 0 or r > 0.5 - a: 
        r = max(0, min(0.5 - a, r))
        
    # middle height from area constraint
    if 1 - 2 * a - 2 * r == 0:
        p = 0.5  # fallback
    else:
        p = (1 - 2 * a - 2 * r) / (2 * (1 - a - r))  # can be negative ...
    
    p = max(0, min(1, p))  # bound to [0,1]
    
    u = np.full(n // 2, p)
    if a > 0 or r > 0:
        m = n // 2
        u[:int(a * m)] = 1.0
        if r > 0:
            if int((a + r) * m) < m:
                start_idx = int(a * m)
                end_idx = int((a + r) * m)
                ramp = np.linspace(1.0, p, end_idx - start_idx)
                u[start_idx:end_idx] = ramp
            # symmetric side
            if int((1 - a - r) * m) > 0:
                start_idx = int((1 - a - r) * m)
                end_idx = int((1 - a) * m)
                ramp = np.linspace(p, 1.0, end_idx - start_idx)
                u[start_idx:end_idx] = ramp
    
    return u

def run_de(n=120, params_dim=2):
    """Run global optimization using Differential Evolution"""
    bounds = [(0.05, 0.45), (0.0, 0.3)] if params_dim == 2 else [(0.05, 0.45)]
    
    def obj(params):
        u = parametric_to_halfvector_5level(params, n) if params_dim == 2 else parametric_to_halfvector_3level(params, n)
        return exact_c5_half(u)
    
    res = differential_evolution(obj, bounds, seed=1, maxiter=200, popsize=15*params_dim, tol=1e-6)
    return res.x

def run_lbfgsb(u0, beta=200, maxiter=800):
    """Run smooth L-BFGS-B optimization"""
    def obj(u):
        return lse_c5_half(u, beta)
    
    def grad(u):
        return grad_lse_c5_half(u, beta)
    
    res = minimize(obj, u0, method='L-BFGS-B', bounds=[(0,1)]*len(u0), 
                   jac=grad, options={'maxiter': maxiter, 'gtol': 1e-10})
    
    # Project back to sum constraint
    if res.success:
        u_final = project_to_sum_box(res.x, 0.5 * len(res.x))
        return u_final
    else:
        u_final = project_to_sum_box(res.x, 0.5 * len(res.x))
        return u_final

def run_sa(u0, max_iter=20000):
    """Run simulated annealing polishing"""
    u_best = u0.copy()
    best_c5 = exact_c5_half(u_best)
    T0 = 5e-3
    alpha = 0.9997
    T = T0
    delta_max = 0.05
    accept_count = 0
    
    n = len(u0)
    T_min = 1e-6
    
    for i in range(max_iter):
        # Adaptive temperature
        if i % 1000 == 0 and i > 0:
            if accept_count / 1000 < 0.2:
                delta_max *= 0.8
            elif accept_count / 1000 > 0.4:
                delta_max *= 1.2
            accept_count = 0
            
        # Pick two indices
        i1, i2 = np.random.choice(n, 2, replace=False)
        
        # Propose move
        delta = np.random.uniform(0, delta_max)
        if u_best[i1] >= delta and u_best[i2] <= 1 - delta:
            u_new = u_best.copy()
            u_new[i1] -= delta
            u_new[i2] += delta
            
            # Project to preserve mass
            u_new = project_to_sum_box(u_new, 0.5 * n)
            
            c5_new = exact_c5_half(u_new)
            
            if c5_new < best_c5:
                u_best = u_new.copy()
                best_c5 = c5_new
                accept_count += 1
            elif np.random.uniform() < np.exp(-(c5_new - best_c5) / T):
                u_best = u_new.copy()
                accept_count += 1
                
        T *= alpha
        if T < T_min:
            break
            
    return u_best

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: Parametric sweep
    best_c5 = float('inf')
    best_u = None
    n = 120
    dx = 2.0 / n
    
    # Sweep 3-level family
    param_list = np.linspace(0.05, 0.45, 400)
    for a in param_list:
        u = parametric_to_halfvector_3level([a], n)
        c5 = exact_c5_half(u)
        if c5 < best_c5:
            best_c5 = c5
            best_u = u.copy()
    
    # Phase 2: DE optimization
    if time.time() - start_time < 0.6 * budget_s:
        try:
            params = run_de(n, 2)
            u_de = parametric_to_halfvector_5level(params, n)
            c5_de = exact_c5_half(u_de)
            if c5_de < best_c5:
                best_c5 = c5_de
                best_u = u_de.copy()
        except:
            # Fallback
            pass
    
    # Phase 3: Upscale and refine with L-BFGS
    if time.time() - start_time < 0.9 * budget_s:
        # Upscale
        n_fine = 240
        u_fine = np.interp(np.linspace(0, 1, n_fine // 2), 
                           np.linspace(0, 1, len(best_u)), 
                           best_u)
        
        # Refine with L-BFGS
        u_refined = run_lbfgsb(u_fine, beta=200, maxiter=800)
        c5_refined = exact_c5_half(u_refined)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_u = u_refined.copy()
    
    # Phase 4: SA polishing
    if time.time() - start_time < budget_s:
        try:
            best_u = run_sa(best_u, max_iter=20000)
            best_c5 = exact_c5_half(best_u)
        except:
            pass

    h = np.concatenate([best_u, best_u[::-1]])
    return (h.tolist(), float(best_c5), len(h))
