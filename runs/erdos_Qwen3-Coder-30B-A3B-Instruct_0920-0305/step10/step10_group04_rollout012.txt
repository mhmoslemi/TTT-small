# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_perturbation_search(h, current_c5, start_time, budget_s):
    """Perform local search with adaptive perturbations"""
    current_h = h.copy()
    current_c5 = current_c5
    best_h = current_h.copy()
    best_c5 = current_c5
    
    max_iter = 5000
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Adaptive perturbation size
        pert_size = max(0.001, 0.01 * (1 - i / max_iter))
        
        # Create neighbor
        neighbor_h = current_h.copy()
        num_changes = max(1, min(10, len(current_h) // 10))
        indices = np.random.choice(len(current_h), size=num_changes, replace=False)
        for idx in indices:
            delta = np.random.normal(0, pert_size)
            neighbor_h[idx] += delta
            
        # Project back to constraints
        neighbor_h = project_to_sum_box(neighbor_h, TARGET_SUM)
        neighbor_c5 = evaluate_c5(neighbor_h)
        
        # Accept if better or with some probability
        if neighbor_c5 < current_c5:
            current_h = neighbor_h.copy()
            current_c5 = neighbor_c5
            if neighbor_c5 < best_c5:
                best_h = neighbor_h.copy()
                best_c5 = neighbor_c5
        elif np.random.random() < 0.01:
            current_h = neighbor_h.copy()
            current_c5 = neighbor_c5
    
    return type('Result', (), {'x': best_h, 'success': True})()
