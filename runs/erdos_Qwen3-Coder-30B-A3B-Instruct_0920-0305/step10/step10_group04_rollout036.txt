import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_auto_corr_binary(x):
    """Compute auto-correlation for a binary vector"""
    return correlate(x, x, mode='full')

def compute_overlap_from_auto_corr(A, x):
    """Compute overlap values O_d from auto-correlation A and x"""
    n = len(x)
    O = np.zeros(2*n - 1)
    # For each shift d in [-n+1, n-1]
    # O[d] = sum_{i=0}^{n-1-d} x_i * (1 - x_{i+d})
    #      = sum_{i=0}^{n-1-d} x_i - sum_{i=0}^{n-1-d} x_i * x_{i+d}
    #      = sum_{i=0}^{n-1-d} x_i - A_d
    
    cumsum_x = np.concatenate(([0], np.cumsum(x)))
    for d in range(-(n-1), n):
        if d >= 0:
            # d > 0 corresponds to x[i] and x[i+d]
            start = 0
            end = n - d
        else:
            # d < 0, x[i] and x[i+d] => x[i+d] is in later positions
            start = -d
            end = n
        if end > start:
            # sum_{i=start}^{end-1} x_i = cumsum_x[end] - cumsum_x[start]
            sum_x = cumsum_x[end] - cumsum_x[start]
            # sum_{i=start}^{end-1} x_i * x_{i+d} = A[d + n - 1]
            sum_xxp = A[d + n - 1]
            O[d + n - 1] = sum_x - sum_xxp
        else:
            O[d + n - 1] = 0
    return O

def c5_from_overlap(O):
    """Convert overlap array O to C5 value"""
    return dx * np.max(O)

def compute_c5_fast(x):
    """Compute C5 for binary vector x via fast auto-correlation and overlap computation"""
    A = compute_auto_corr_binary(x)
    O = compute_overlap_from_auto_corr(A, x)
    return c5_from_overlap(O)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_incremental_delta(A, x, i, j):
    """Compute delta-auto-correlation for a swap of x[i],x[j]"""
    n = len(x)
    x_pad = np.concatenate(([0]*(n-1), x.tolist(), [0]*(n-1)))
    
    # For offset d, involve indices d+len(x)-1 and d (though one can be outside)
    # A_d = sum_{k=0}^{n-1-|d|} x_k * x_{k+d}
    # To compute the delta from flipping i and j:
    # i changes from 1 to 0, so it loses x_pad[i+n-1:i+3n-2] + x_pad[i:i+2n-1][::-1] 
    # j changes from 0 to 1, so it gains x_pad[j+n-1:j+3n-2] + x_pad[j:j+2n-1][::-1]
    
    # Shifted for other indexes
    # Note: for d, consider contributions from x[i] when computing x[i+d] where i+d = 0..n-1
    # Beyond that, contributions are 0
    shift_offset = n - 1
    
    # Make delta for k leading into i, and later x_{i+d}, offset part
    idx_lo = i + shift_offset
    idx_hi = i + shift_offset + 2*n - 1
    
    delta_off = - (x_pad[i + shift_offset: i + 3*n - 2] + np.flip(
        x_pad[i: i + 2*n - 1]))
    
    idx_lo2 = j + shift_offset
    idx_hi2 = j + shift_offset + 2*n - 1
    delta_on = (x_pad[j + shift_offset: j + 3*n - 2] + np.flip(
        x_pad[j: j + 2*n - 1]))
    
    return delta_off + delta_on

def get_worst_shift_indicators(x):
    """Get the worst shift d* and exposed/covered sets U,V for targeted moves"""
    n = len(x)
    A = compute_auto_corr_binary(x)
    O = compute_overlap_from_auto_corr(A, x)
    
    d_star = np.argmax(O) - (n - 1)  # Convert index in O to real index d
    worst_shift = d_star
    
    # Build U: exposed indices for d_star (i such that x_i=1, x_{i+d_star}=0)
    U = []
    V = []
    for i in range(n):
        if d_star >= 0:
            j = i + d_star
            if 0 <= j < n and x[i] == 1 and x[j] == 0:
                U.append(i)
        else:
            # d_star is negative, so x_{i+d_star} = x_{i-|d_star|}
            j = i + d_star 
            if 0 <= j < n and x[i] == 1 and x[j] == 0:
                U.append(i)
                
    # Build V: indices such that x_j=0, x_{j+d_star}=1
    for j in range(n):
        if d_star >= 0:
            i = j + d_star
            if 0 <= i < n and x[j] == 0 and x[i] == 1:
                V.append(j)
        else:
            i = j + d_star
            if 0 <= i < n and x[j] == 0 and x[i] == 1:
                V.append(j)
            
    return worst_shift, U, V
