import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from scipy.fft import fft, ifft

def project_to_sum_box(u, target_sum):
    """Water-filling projection to sum constraint."""
    def f(lam):
        projected = np.clip(u + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(u + (a + b)/2, 0, 1)
    return result

def autocorr_fft(h):
    """Compute aperiodic autocorrelation via FFT."""
    n = len(h)
    L = 2 * n - 1
    h_rev = h[::-1]
    A = ifft(fft(h_rev, L) * fft(h, L)).real
    return A

def exact_c5(u):
    """Compute exact C5 value from half-vector."""
    n = len(u) * 2
    dx = 2.0 / n
    h = np.concatenate([u, u[::-1]])
    A = autocorr_fft(h)
    return 1 - dx * np.min(A)

def lse_c5(u, beta):
    """Smooth surrogate of C5 using log-sum-exp."""
    n = len(u) * 2
    dx = 2.0 / n
    h = np.concatenate([u, u[::-1]])
    A = autocorr_fft(h)
    return 1 - dx * np.log(np.sum(np.exp(beta * A))) / beta

def grad_lse_c5(u, beta):
    """Gradient of smooth surrogate of C5."""
    n = len(u) * 2
    dx = 2.0 / n
    h = np.concatenate([u, u[::-1]])
    A = autocorr_fft(h)
    w = np.exp(beta * A)
    W = np.sum(w)
    
    # Compute gradient of weighted sum of convolutions
    grad = np.zeros_like(u)
    for i in range(len(u)):
        # h_{i+k} * h_{i-k} part (stencil: 1,1) on circular domain
        s = 0.0
        for k in range(n):
            j1 = (i + k) % n
            j2 = (i - k) % n
            s += h[j1] * h[j2] * w[k]
        grad[i] = -dx * s / W
    
    # Symmetrize to return gradient for half-vector only
    result = np.zeros_like(u)
    for i in range(len(u)):
        result[i] = grad[i] + grad[len(u)-1-i]
    return result

def parametric_to_halfvector(params, n=120):
    """Construct half-vector from parameters."""
    if len(params) == 2:
        # 3-level case
        a, p = params
        m = n // 2
        u = np.ones(m)
        u[:int(np.round(a * m))] = 1.0
        mid_start = int(np.round(a * m))
        mid_end = m - int(np.round(a * m))
        if mid_end > mid_start:
            u[mid_start:mid_end] = p
        return u
    elif len(params) == 3:
        # 5-level case
        a, r, p = params
        m = n // 2
        u = np.empty(m)
        edge_width = int(np.round(a * m))
        ramp_width = int(np.round(r * m))
        mid_width = m - 2 * edge_width - 2 * ramp_width
        if mid_width < 0:
            mid_width = 0
        
        u[:edge_width] = 1.0
        u[edge_width:edge_width + ramp_width] = np.linspace(1, p, ramp_width, endpoint=False)
        u[edge_width + ramp_width:edge_width + ramp_width + mid_width] = p
        u[edge_width + ramp_width + mid_width:edge_width + ramp_width + mid_width + ramp_width] = np.linspace(p, 1, ramp_width, endpoint=False)
        u[edge_width + ramp_width + mid_width + ramp_width:] = 1.0
        
        # Normalize to unit area
        total_area = np.sum(u) * 2 * (2.0 / n) / 2
        u *= (n/4) / total_area
        return u
    else:
        raise ValueError("Only 2 or 3 parameters supported")

def run_de(bounds, n_points=120, maxiter=200):
    """Run differential evolution on parametric space."""
    def objective(params):
        u = parametric_to_halfvector(params, n_points)
        u = project_to_sum_box(u, n_points / 4)  # Half sum
        return exact_c5(u)
        
    result = differential_evolution(
        objective, 
        bounds, 
        maxiter=maxiter,
        popsize=15 * len(bounds), 
        seed=42, 
        disp=False
    )
    return result.x

def run_lbfgsb(u0, beta=200, maxiter=800):
    """Run projected L-BFGS-B."""
    def objective(u):
        return lse_c5(u, beta)
    
    def grad(u):
        return grad_lse_c5(u, beta)
    
    def project(u):
        return project_to_sum_box(u, len(u) * 2 / 4)
    
    bounds = [(0, 1) for _ in range(len(u0))]
    
    # Initial projection
    u = project(u0)
    for iteration in range(maxiter):
        try:
            result = minimize(
                objective, 
                u, 
                method='L-BFGS-B', 
                jac=grad, 
                bounds=bounds,
                options={'gtol': 1e-10, 'maxiter': 1}
            )
            u = result.x
            u = project(u)
        except:
            break
    return u

def run_sa(u0, max_iter=20000):
    """Run simulated annealing polish."""
    u = u0.copy()
    current_c5 = exact_c5(u)
    T = 5e-3
    alpha = 0.9997
    delta_max = 0.1
    accept_count = 0
    total_moves = 0
    
    start_time = time.time()
    while total_moves < max_iter and time.time() - start_time < 500:
        i, j = np.random.choice(len(u), 2, replace=False)
        delta = np.random.uniform(0, delta_max)
        
        if u[i] >= delta and u[j] <= 1 - delta:
            u_new = u.copy()
            u_new[i] -= delta
            u_new[j] += delta
            new_c5 = exact_c5(u_new)
            
            if new_c5 < current_c5:
                u = u_new
                current_c5 = new_c5
                accept_count += 1
            else:
                prob = np.exp(-(new_c5 - current_c5) / T)
                if np.random.rand() < prob:
                    u = u_new
                    current_c5 = new_c5
                    accept_count += 1
        
        total_moves += 1
        if total_moves % 500 == 0:
            if accept_count / 500.0 < 0.2:
                delta_max *= 0.9
            elif accept_count / 500.0 > 0.4:
                delta_max *= 1.1
            accept_count = 0
            
        T *= alpha
        if T < 1e-6:
            break
    
    return u

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = float('inf')
    best_u = None
    
    # Phase 1: Parametric sweep (3-level shape)
    if time.time() - start_time < budget_s * 0.2:
        n_points = 120
        a_vals = np.linspace(0.05, 0.45, 400)
        best_a = None
        for a in a_vals:
            u = parametric_to_halfvector([a, 0.5], n_points)
            u = project_to_sum_box(u, n_points / 4)
            c5 = exact_c5(u)
            if c5 < best_c5:
                best_c5 = c5
                best_u = u.copy()
                best_a = a
    
    # Phase 2: DE search on enriched parameter space
    if time.time() - start_time < budget_s * 0.5:
        bounds = [(0.05, 0.45), (0, 0.3), (0.1, 1.0)]  # a, r, p
        try:
            params = run_de(bounds, 120, maxiter=150)
            u = parametric_to_halfvector(params, 120)
            u = project_to_sum_box(u, 60)  # Half sum for 120 points
            c5 = exact_c5(u)
            if c5 < best_c5:
                best_c5 = c5
                best_u = u.copy()
        except:
            pass
    
    # Phase 3: Upscale and refine with L-BFGS-B
    if time.time() - start_time < budget_s * 0.8:
        n_fine = 240
        u_fine = np.interp(np.linspace(0, 1, n_fine//2), 
                          np.linspace(0, 1, len(best_u)), 
                          best_u)
        u_fine = project_to_sum_box(u_fine, n_fine / 4)
        u_refined = run_lbfgsb(u_fine, beta=200, maxiter=500)
        if exact_c5(u_refined) < best_c5:
            best_c5 = exact_c5(u_refined)
            best_u = u_refined
    
    # Phase 4: SA polishing
    if time.time() - start_time < budget_s:
        best_u = run_sa(best_u, max_iter=5000)
        best_c5 = exact_c5(best_u)
    
    # Final
    h = np.concatenate([best_u, best_u[::-1]])
    n = len(h)
    dx = 2.0 / n
    return (h.tolist(), float(best_c5), n)
