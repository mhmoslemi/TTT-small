import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import differential_evolution, minimize
import time

def project_to_sum_box(u, target_sum):
    """Water-filling projection to box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(u + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(u + (a + b)/2, 0, 1)
    return result

def autocorr_fft(h):
    """Compute aperiodic autocorrelation using FFT"""
    n = len(h)
    L = 2 * n - 1
    h_rev = h[::-1]
    corr = ifft(fft(h_rev, L) * fft(h, L)).real
    return corr

def exact_c5(u):
    """Compute exact C5 value from half-vector u"""
    n = len(u) * 2
    dx = 2.0 / n
    h = np.concatenate([u, u[::-1]])  # full symmetric vector
    A = autocorr_fft(h)
    C5 = 1 - dx * np.min(A)
    return C5

def lse_c5(u, beta):
    """Smooth surrogate for C5 using log-sum-exp"""
    n = len(u) * 2
    dx = 2.0 / n
    h = np.concatenate([u, u[::-1]])
    A = autocorr_fft(h)
    # Use log-sum-exp with smoothing parameter beta
    lse = np.log(np.sum(np.exp(beta * A))) / beta
    # Correct for the scaling factor - dx * lse
    return 1 - dx * lse

def grad_lse_c5(u, beta):
    """Gradient of smooth surrogate using FFT-based convolution"""
    n = len(u) * 2
    dx = 2.0 / n
    h = np.concatenate([u, u[::-1]])
    
    # Compute the autocorrelation A(k) = sum_i h_i h_{i+k}
    A = autocorr_fft(h)
    w = np.exp(beta * A)
    W_sum = np.sum(w)
    
    # Gradient of smooth surrogate
    # d/dh_i [1 - (dx/beta) * log(sum w_k)]  
    # = -(dx/beta) * sum_k w_k * (h_{i+k} + h_{i-k}) / W_sum
    
    # Compute the numerator efficiently via convolutions using FFT:
    # result[k] = sum_i h_i * h_{i+k}
    # But we want the gradient for each h_i as sum of terms with h_{i+k}
    # This is the conjugate convolution of w with h^rev
    
    h_rev = h[::-1]
    L = 2 * n - 1
    grad = np.zeros_like(h)
    
    # Find contribution from each k
    # dC5/dh_i = -(dx/beta) * (sum_k w_k * h_{i+k}) / W_sum
    # for the right terms, we use FFT convolution:
    # 1. pad w with zeros to length L = 2*n-1
    # 2. pad h_rev to length L
    # 3. convolve these and get result of length L
    # we only want the central part, but handling is easier when computed correctly.
    
    # Direct Fourier approach:
    w_padded = np.zeros(L)
    w_padded[:len(w)] = w
    
    h_rev_padded = np.zeros(L)
    h_rev_padded[:len(h_rev)] = h_rev
    
    # Compute convolution w * h_rev (relevant for k terms in 0,1,...,n-1)  
    # The result should be of length L = 2*n - 1, but we'll take centered splice
    conv_w_hrev = ifft(fft(w_padded) * fft(h_rev_padded)).real
    
    # Repeat the same for the other convolution term (with h shifted backwards)
    # Because we have h_{i-k} contribution, which corresponds to 
    # convolution of w with h_rev shifted back
    
    # More simply: calculate for each i, we want 
    # \sum_k w_k * h_{i+k} = sum_j w_{j-i} h_j for j in 0..n-1  
    # which is convolution of w_rev with h with index shift

    # So compute: full convolution of w (shifted) with h_rev
    # use real indices for bins.
    # the i-th element in the gradient contribution: 
    # we want w_{j-i} * h_j summed over j
    # Same as conv(w_rev, h_rev)[i + n - 1]
    # But really include contribution from h reversals properly...

    # Manual conversion to shift/etc makes it tricky. 
    # Let's just work with standard FFT convolution definition:
    # gradient[k] is sum of w_j * h_{k+j} - all cyclic wraparound assumed  
    # But note A(k) = sum_i h_i h_{i-k}  
    # so the gradient of A(k) wrt h_i is h_{i-k}  

    # The gradient of LSE term for function x -> log(sum w_k * x_k) is:
    # (1/W_sum) * sum_k w_k * grad_x_k  
    # where x = A and grad_A_k = h_{i+k} (but consider leading sign)
    # Actually, correcting the full gradient:
    # \frac{\partial C5}{\partial h_i} = - \frac{dx}{beta} * 
    # (sum_k w_k * (h_{i+k} + h_{i-k})) / W_sum
    
    sign_vec = np.empty(L)  # index matching
    for k in range(L):
        sign_vec[k] = 1 if k < n else -1  # v0 0-indexed for symmetric purposes

    # Instead, do it elementwise:
    # Compute convolution (w * h_rev)^(-1)  
    # that effectively computes sum_k w_k h_{i+k}  
    # and w may also be shifted or whatnot. 
    # Let's go back to doing it directly using the precise definition:
    
    # Direct calculation: for gradient at index i:
    # \sum_k w_k * h_{i-k} + w_k * h_{i+k} and correspond to shifts.
    # Let's get the proper indexing:
    # dC5/dh_i = -dx * (sum_k w_k * (h[i-k] + h[i+k]))  / (beta*W_sum)
    
    # For L=2n-1, we define w and h appropriately:
    h_padded = np.zeros(L)
    h_padded[:n] = h
    w_padded = np.zeros(L)
    w_padded[:len(w)] = w
    
    # Use real FFT symmetry. We compute:
    # (sum_k w_k * h_{i+k}) using convolution of w with shifted h^rev
    # shift with zero padding
    w_trimmed = w[:n] 
    h_rev_trimmed = h_rev[:n]
    
    # General matrix method of convolution would involve warning about symmetry.
    # For simplicity: make sure we allow stride wrapping.
    
    # Use a more tractable approach: compute directly for gradient entries
    # Precompute array of gradient contributions for all indices using convolution 
    conv_result = np.zeros(n)
    for i in range(n):
        # grad contribution from i = sum_k w_k * (h_{i+k} + h_{i-k})
        # for small range only
        grad_term = 0
        for k in range(1-n, n):
            idx1 = i + k 
            idx2 = i - k
            if 0 <= idx1 < n and 0 <= idx2 < n:
                grad_term += w[k + n - 1] * (h[idx1] + h[idx2])
        conv_result[i] = grad_term
        
    grad = -dx / (beta * np.sum(w)) * conv_result
    
    # Only return the half vector part
    return grad[:len(u)]

def parametric_to_halfvector(params, n=120):
    """Build half-vector from 2D parametric family: [a, r]"""
    # a = edge width, r = ramp width, p = middle height  
    # Piecewise function: u(x) = 1 for x in [0, a], 
    #   then ramps to p in [a, a+r], then constant p for [a+r, 1-a-r], 
    #   then ramps down from p to 1 in [1-a-r, 1-a], then 1 for [1-a, 1]
    #   But assumes the mid width = 2 - 2*a - 2*r leads to final p value.
    #   Require area = 1:
    #   area_contribution = a*1 + (r*p) + (1-r)*p = a + p*(r + (1-r)) = a + p 
    #   But also: total = 2*a + (1-2*a)*p = 1  -> p = (1-2a)/(1-a) provided a<1
    #   We then need the second half of the shape to be truncated at 1-a-r
    # But we have so far only implemented pretty general continuous version.
    # Use a variant that is easier to generate.
    
    # HERE: we combine version where 3parts: [a, r] parameters:
    # Edge 1, Ramp, Middle, Ramp, Edge 1 
    a, r = params
    if a < 0 or a > 0.5 or r < 0 or r > 0.5 - a:
        return np.ones(n//2) * 0.5
    
    # The shape: 
    # w = 2*a+2*r, total width for the varying part
    # which is Mr = 1 - w = 1 - 2*(a+r) for middle part  
    # Let's compute height p such that area = 1
    # Exact integral:
    # integral = 2*a*1 + 2*(0.5*(r+p)*r) + (1-2*(a+r)) * p = 1
    #       = 2*a + r*(p+0.5) + (1-2a-2r)*p = 1
    #       = 2*a + r*p + 0.5*r + (1 - 2*a - 2*r)*p = 1
    #       = 2*a + 0.5*r + (1 - 2*a - 2*r + 2 * r) * p = 1  
    #       = 2*a + 0.5*r + p = 1  
    #       => p = 0.5*(1 - 2*a - 0.5*r) = 0.5*(0.5 - 2*a - 0.5*r)

    p = (1 - 2*a - 0.5*r)  # something like this to make area = 1
    # such that p = (1 - 2*a) / (1 - a) ??? Verify precision
    
    # Verify p is reasonable, mostly near 0.5
    if p < 0 or p > 1: p = 1 - 2*a  # fallback for numerical error
    u = np.zeros(n//2)
    dx = 2.0 / n  # width of each subinterval
    
    # indexing partition into piecewise intervals 
    # global: [0, a], [a, a+r], [a+r, 1-a-r], [1-a-r, 1-a], [1-a, 2]
    # mirror: [0, a], [a, a+r], [a+r, 1-a-r], [1-a-r, 1-a], [1-a, 1]
    # will go to central part [a, 1-a] using symmetry
    
    # index convention: i * dx = x  
    # We want to visually map [0, 1] of the SCM domain to [0, n//2]
    edges = np.array([a, a+r, 1-a-r, 1-a])
    indices = np.floor(edges / dx).astype(int)
    indices = np.minimum(indices, n//2)
    
    # fill [0, a]: value 1
    u[:indices[0]] = 1.0
    
    # fill [a, a+r]: ramp down from 1 to p
    xl, xr = indices[0], indices[1]
    if xl + 1 < xr:
        for i in range(xl, xr):
            u[i] = 1.0 - (i - xl) * (1 - p) / (xr - xl)
    
    # fill [a+r, 1-a-r]: constant = p  
    xl, xr = indices[1], indices[2]  
    u[xl:xr] = p
    
    # fill [1-a-r, 1-a]: ramp from p to 1  
    xl, xr = indices[2], indices[3]
    if xl + 1 < xr:
        for i in range(xl, xr):
            u[i] = p + (i - xl) * (1 - p) / (xr - xl)
            
    return u

def run_de(n_points=120, maxiter=200):
    """Sample the parametric family with DE to find good initial solution."""
    def objective(params):
        u = parametric_to_halfvector(params, n_points)
        return exact_c5(u)
    
    bounds = [(0.05, 0.45), (0, 0.3)]
    res = differential_evolution(objective, bounds, maxiter=maxiter, seed=42, disp=False)
    return parametric_to_halfvector(res.x, n_points)

def run_lbfgsb(u0, beta=200, maxiter=800):
    """Projected L-BFGS-B on the fine half-vector."""
    n = len(u0)
    dx = 2.0 / (2 * n)
    
    def objective(u):
        return lse_c5(u, beta)
    
    def grad_objective(u):
        return grad_lse_c5(u, beta)
    
    def project_fn(x):
        return project_to_sum_box(x, target_sum=n/4)
    
    # L-BFGS-B 
    options = {'maxiter': maxiter, 'gtol': 1e-10}
    try:
        result = minimize(objective, u0, method='L-BFGS-B',
                          jac=grad_objective, bounds=[(0,1)] * n,
                          options=options)
        u_opt = result.x
        u_opt = project_fn(u_opt)
    except:
        u_opt = u0
        
    return u_opt

def run_sa(u_best, max_iter=50000):
    """Simulated Annealing polishing"""
    T0 = 5e-3
    alpha = 0.9997
    T_min = 1e-6
    n = len(u_best)
    dx = 2.0 / (2 * n)
    target_sum = n/4
    
    current_u = u_best.copy()
    current_c5 = exact_c5(current_u)
    best_u = current_u.copy()
    best_c5 = current_c5

    T = T0
    delta_max = 0.1
    accept_count = 0
    total_iter = 0

    while T > T_min and total_iter < max_iter:
        if total_iter % 1000 == 0:
            T *= alpha
            
        # Pick two different indices
        i, j = np.random.choice(n, 2, replace=False)
        
        # Propose move
        delta = np.random.uniform(0, delta_max)
        if current_u[i] >= delta and current_u[j] <= 1 - delta:
            new_u = current_u.copy()
            new_u[i] -= delta
            new_u[j] += delta
            
            # Project to sum and bounds
            new_u = project_to_sum_box(new_u, target_sum)
            
            new_c5 = exact_c5(new_u)
            
            if new_c5 < current_c5:
                current_u = new_u
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_u = new_u.copy()
                    best_c5 = new_c5
                accept_count += 1
            else:
                # Accept with probability exponentials
                if np.random.uniform() < np.exp(-(new_c5 - current_c5) / T):
                    current_u = new_u
                    current_c5 = new_c5
                    accept_count += 1
            
            total_iter += 1
            
            # Adjust delta_max to maintain acceptance ratio in 0.2–0.4
            if total_iter % 500 == 0:
                accept_ratio = accept_count / 500.0
                if accept_ratio < 0.2:
                    delta_max *= 0.9
                elif accept_ratio > 0.4:
                    delta_max *= 1.1
                accept_count = 0
                
    return best_u, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Phase 1: Parametric sweep (fast coarse search)
    best_c5 = float('inf')
    best_u = None
    sweep_params = np.linspace(0.05, 0.45, 400)
    for a in sweep_params:
        if time.time() - start_time > budget_s * 0.2:
            break
        u = parametric_to_halfvector([a, 0], n_points)
        c5 = exact_c5(u)
        if c5 < best_c5:
            best_c5 = c5
            best_u = u.copy()

    # Phase 2: DE search in parametric space  
    if time.time() - start_time < budget_s * 0.5:
        try:
            u_de = run_de(n_points, maxiter=200)
            c5_de = exact_c5(u_de)
            if c5_de < best_c5:
                best_c5 = c5_de
                best_u = u_de.copy()
        except:
            pass

    # Phase 3: Up-scale and Refinement with L-BFGS 
    if time.time() - start_time < budget_s * 0.8:
        try:
            # Fine grid with 240 points
            n_fine = 240
            dx_fine = 2.0 / n_fine
            u_fine = np.interp(np.linspace(0, 1, n_fine//2), np.linspace(0, 1, len(best_u)), best_u)
            u_fine = project_to_sum_box(u_fine, n_fine/4)
            u_fine = run_lbfgsb(u_fine, beta=200, maxiter=800)
            c5_fine = exact_c5(u_fine)
            if c5_fine < best_c5:
                best_c5 = c5_fine
                best_u = u_fine.copy()
        except:
            pass

    # Phase 4: SA polishing  
    if time.time() - start_time < budget_s * 0.98:
        try:
            u_final, c5_final = run_sa(best_u, max_iter=10000)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_u = u_final.copy()
        except:
            pass

    # Final full evaluation  
    overall_c5 = exact_c5(best_u)
    
    # Assemble full vector  
    full_h = np.concatenate([best_u, best_u[::-1]])
    
    return (full_h.tolist(), float(overall_c5), n_points)
