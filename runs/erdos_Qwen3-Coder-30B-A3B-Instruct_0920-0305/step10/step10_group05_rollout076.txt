import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.fft import fft, ifft
import time

# Global variables for the problem
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 4

def project_to_sum_box(u, target_sum):
    """Project vector u onto box constraints [0,1] with sum = target_sum using water-filling."""
    def f(lam):
        projected = np.clip(u + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(u + (a + b)/2, 0, 1)
    return result

def autocorr_fft(h):
    """Compute aperiodic autocorrelation using FFT."""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(h, L)).real
    return corr

def exact_c5(u):
    """Compute exact C5 from half-vector u."""
    h = np.concatenate([u, u[::-1]])  # Full vector
    corr = autocorr_fft(h)
    return 1 - dx * np.min(corr)

def lse_c5(u, beta):
    """Compute smooth surrogate for C5 using log-sum-exp."""
    h = np.concatenate([u, u[::-1]])
    corr = autocorr_fft(h)
    return 1 - (dx / beta) * np.log(np.sum(np.exp(beta * corr)))

def grad_lse_c5(u, beta):
    """Compute gradient of smooth surrogate for C5."""
    h = np.concatenate([u, u[::-1]])
    corr = autocorr_fft(h)
    w = np.exp(beta * corr)
    w_sum = np.sum(w)
    
    # Compute convolution of w with reversed h
    w_h_conv = ifft(fft(w, 2*len(h)-1) * fft(h[::-1], 2*len(h)-1)).real[:len(h)]
    
    # Gradient with respect to h
    grad_h = -dx * w_h_conv / w_sum
    
    # Return gradient with respect to u only (half vector)
    return -2 * grad_h[:len(u)]

def parametric_to_halfvector(params, n):
    """Convert parameters to a half-vector."""
    if len(params) == 2:
        a, r = params
        p = (1 - 2 * a) / (2 * (1 - a))
        if a < 0 or a > 0.5 or r < 0 or r > 0.5 - a:
            return None
            
        half = np.ones(n // 2)
        half[:n//4] = 1
        half[n//4:n//2] = p
        return half
        
    elif len(params) == 3:
        a, r, p = params
        if a < 0 or a > 0.5 or r < 0 or r > 0.5 - a or p < 0 or p > 1:
            return None
        half = np.ones(n // 2)
        half[:n//4] = 1
        half[n//4:n//2] = p
        return half
    return None

def run_de():
    """Run differential evolution to find a good starting point."""
    def objective(params):
        u = parametric_to_halfvector(params, n_points)
        if u is None:
            return np.inf
        return exact_c5(u)
    
    bounds = [(0.05, 0.45), (0, 0.3)]
    res = differential_evolution(objective, bounds, seed=42, maxiter=200)
    if not res.success:
        return None
    return res.x

def run_lbfgsb(u0, beta=200):
    """Run L-BFGS-B on the full half-vector."""
    def objective(u):
        return lse_c5(u, beta)
    
    def grad(u):
        return grad_lse_c5(u, beta)
    
    # Project initial guess
    u = project_to_sum_box(u0, TARGET_SUM)
    bounds = [(0, 1) for _ in range(len(u))]
    
    res = minimize(objective, u, method='L-BFGS-B', jac=grad, bounds=bounds, options={'maxiter': 800, 'gtol': 1e-10})
    if res.success:
        u = res.x
    u = project_to_sum_box(u, TARGET_SUM)
    return u

def run_sa(u_best):
    """Run simulated annealing to polish."""
    # Parameters for SA
    T0 = 5e-3
    alpha = 0.9997
    T_min = 1e-6
    max_iter = 10000
    
    u = u_best.copy()
    best_u = u.copy()
    best_c5 = exact_c5(u)
    
    T = T0
    acc_count = 0
    total_count = 0
    delta_max = 0.1
    
    def accept(delta_c5, temperature):
        if delta_c5 < 0:
            return True
        return np.random.rand() < np.exp(-delta_c5 / temperature)
    
    for i in range(max_iter):
        # Stop if time runs out
        if i % 100 == 0 and time.time() - start_time > budget_s * 0.95:
            break
            
        # Pick two distinct indices
        i1, i2 = np.random.choice(len(u), 2, replace=False)
        
        # Proposal step
        delta = np.random.rand() * delta_max
        if u[i1] >= delta and u[i2] <= 1 - delta:
            u_new = u.copy()
            u_new[i1] -= delta
            u_new[i2] += delta
            
            # Adjust for the sum constraint
            u_new = project_to_sum_box(u_new, TARGET_SUM)
            
            old_c5 = exact_c5(u)
            new_c5 = exact_c5(u_new)
            delta_c5 = new_c5 - old_c5
            
            if accept(delta_c5, T):
                u = u_new.copy()
                if new_c5 < best_c5:
                    best_u = u.copy()
                    best_c5 = new_c5
            
            acc_count += 1 if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T) else 0
            total_count += 1
            
            # Update step size
            if total_count % 500 == 0:
                acc_ratio = acc_count / total_count
                if acc_ratio < 0.2:
                    delta_max *= 0.9
                elif acc_ratio > 0.4:
                    delta_max *= 1.1
                acc_count = 0
                total_count = 0
        
        T *= alpha
        if T < T_min:
            break
    
    return best_u, best_c5

def run(seed=42, budget_s=600):
    global start_time
    start_time = time.time()
    
    # Phase 1: Parametric sweep on a simple 3-level shape (fast)
    best_c5 = float('inf')
    best_u = None
    
    # Try a few values of "a"
    for a_val in np.linspace(0.05, 0.4, 20):
        u = np.ones(n_points // 2)
        p = (1 - 2 * a_val) / (2 * (1 - a_val))
        u[:int(n_points * a_val/2)] = 1
        u[int(n_points * a_val/2):] = p
        u = project_to_sum_box(u, TARGET_SUM)
        c5 = exact_c5(u)
        if c5 < best_c5:
            best_c5 = c5
            best_u = u.copy()
    
    # Phase 2: Differential evolution for better initialization
    try:
        de_params = run_de()
        if de_params is not None:
            u_de = parametric_to_halfvector(de_params, n_points)
            if u_de is not None:
                u_de = project_to_sum_box(u_de, TARGET_SUM)
                c5_de = exact_c5(u_de)
                if c5_de < best_c5:
                    best_c5 = c5_de
                    best_u = u_de.copy()
    except Exception:
        pass
    
    # Phase 3: L-BFGS-B refinement with higher resolution
    try:
        u_fine = run_lbfgsb(best_u.copy(), beta=200)
        c5_fine = exact_c5(u_fine)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_u = u_fine.copy()
    except Exception:
        pass
    
    # Phase 4: Simulated Annealing polishing
    try:
        u_sa, c5_sa = run_sa(best_u.copy())
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_u = u_sa.copy()
    except Exception:
        pass
    
    # Make full vector
    h = np.concatenate([best_u, best_u[::-1]])
    return (h.tolist(), float(best_c5), n_points)
