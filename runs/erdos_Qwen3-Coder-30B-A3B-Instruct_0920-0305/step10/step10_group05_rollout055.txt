import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points
half_size = n_points // 2
TARGET_SUM = n_points / 4

def autocorr_fft(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    b = h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_c5_exact(h):
    """Compute exact C5 using max over all shifts"""
    corr = autocorr_fft(h)
    amin = np.min(corr)
    return 1 - dx * amin

def compute_c5_smooth(h, beta=50.0):
    """Compute smooth approximation of C5"""
    corr = autocorr_fft(h)
    max_corr = np.max(corr)
    exp_corr = np.exp(beta * (1 - corr))
    sum_exp = np.sum(exp_corr)
    return 1 - (dx / beta) * np.log(sum_exp)

def smooth_gradient(h, beta=50.0):
    """Compute gradient of smooth C5"""
    corr = autocorr_fft(h)
    max_corr = np.max(corr)
    exp_corr = np.exp(beta * (1 - corr))
    sum_exp = np.sum(exp_corr)
    w = exp_corr / sum_exp
    # Gradient from convolution: sum over k of w_k * (h_{i+k} + h_{i-k})
    gradient = np.zeros_like(h)
    for i in range(len(h)):
        grad_i = 0
        for k in range(len(h)):
            if k < len(h) - i:
                grad_i += w[k] * h[i + k]
            if i >= k:
                grad_i += w[k] * h[i - k]
        gradient[i] = -dx * grad_i
    return gradient

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def build_symmetric_vector(a, p):
    """Build symmetric vector with two edges of height 1 and middle at p"""
    if a < 0 or a > 0.5:
        raise ValueError("a must be between 0 and 0.5")
    if p < 0 or p > 1:
        raise ValueError("p must be between 0 and 1")
    a = max(0, min(0.5, a))
    p = max(0, min(1, p))
    h = np.ones(half_size)
    a_idx = int(np.floor(a / dx))
    if a_idx > 0:
        h[:a_idx] = 1.0
        h[-a_idx:] = 1.0
        mid_len = half_size - 2 * a_idx
        if mid_len > 0:
            h[a_idx:a_idx + mid_len] = p
    else:
        # No edges, only middle plateau
        h[:] = p
    return h

def objective_symmetric(a, p):
    """Objective function for the symmetric construction"""
    h = build_symmetric_vector(a, p)
    h_full = np.concatenate([h, h[::-1]])
    return compute_c5_exact(h_full)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    best_c5 = float('inf')
    best_h = None
    
    # Sweep over parameter a to find a good starting point
    best_a = 0
    best_p = 0
    step_size = 0.002
    a_range = np.arange(0.01, 0.49, step_size)
    
    for a in a_range:
        p = (1 - 2 * a) / (2 * (1 - a))  # Calculated p for a given a
        if p < 0 or p > 1:
            continue
        c5 = objective_symmetric(a, p)
        if c5 < best_c5:
            best_c5 = c5
            best_a = a
            best_p = p
            
    if best_c5 == float('inf'):
        # Initialize with a simple symmetric construction: flat plateau
        h_half = np.full(half_size, 0.5)
        h_full = np.concatenate([h_half, h_half[::-1]])
        c5 = compute_c5_exact(h_full)
        best_c5 = c5
        best_h = h_full
        
    # Now optimize in four dimensions (symmetric case with slope) if needed
    if time.time() - start_time > budget_s * 0.3:
        return (best_h.tolist(), float(best_c5), n_points)
    
    # Local optimization
    try:
        def smooth_objective(x):
            a, p = x
            h = build_symmetric_vector(a, p)
            h_full = np.concatenate([h, h[::-1]])
            return compute_c5_smooth(h_full)
        
        def constraint_eq(x):
            a, p = x
            h = build_symmetric_vector(a, p)
            return np.sum(h) * dx - 1
        
        bounds = [(0.01, 0.49), (0, 1)]
        
        res = minimize(
            smooth_objective,
            [best_a, best_p],
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-12, 'gtol': 1e-12}
        )
        
        if res.success:
            a_opt, p_opt = res.x
            h_opt = build_symmetric_vector(a_opt, p_opt)
            h_full_opt = np.concatenate([h_opt, h_opt[::-1]])
            c5_opt = compute_c5_exact(h_full_opt)
            
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_full_opt
    except Exception:
        pass
    
    if best_h is None:
        # Fallback: simple construction
        h_half = np.full(half_size, 0.5)
        h_full = np.concatenate([h_half, h_half[::-1]])
        best_c5 = compute_c5_exact(h_full)
        best_h = h_full
    
    # Simulated Annealing for global search
    if time.time() - start_time > budget_s * 0.6:
        return (best_h.tolist(), float(best_c5), n_points)
        
    # Initialize SA
    h_current = best_h.copy()
    c5_current = best_c5
    
    # Initialize temperature and step size
    temp = 0.01
    delta_max = 0.05
    
    # Keep track of best solution so far
    best_sa_h = h_current.copy()
    best_sa_c5 = c5_current
    
    # Temperature schedule and adaptation parameters
    alpha = 0.9995
    accept_count = 0
    total_moves = 0
    
    # Perform SA iterations
    while time.time() - start_time < budget_s * 0.95:
        # Pick two indices to swap mass between
        i, j = np.random.randint(0, half_size, 2)
        if i == j:
            continue
            
        # Pick a random transfer amount
        delta = np.random.uniform(0, delta_max)
        
        # Check constraints
        if (h_current[i] >= delta and 
            h_current[j] <= 1 - delta):
            # Transfer delta mass between the two indices
            h_new = h_current.copy()
            h_new[i] -= delta
            h_new[j] += delta
            
            # Reflect to full vector
            h_full_new = np.concatenate([h_new, h_new[::-1]])
            
            # Evaluate new C5
            c5_new = compute_c5_exact(h_full_new)
            
            # Metropolis acceptance step
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
                if c5_new < best_sa_c5:
                    best_sa_h = h_current.copy()
                    best_sa_c5 = c5_new
            else:
                # Accept with probability
                if np.random.rand() < np.exp(-(c5_new - c5_current) / temp):
                    h_current = h_new.copy()
                    c5_current = c5_new
                    
            # Adjust step size based on acceptance ratio
            accept_count += 1
            total_moves += 1
            if total_moves % 500 == 0:
                if accept_count > 500 * 0.4:
                    delta_max = min(0.2, delta_max * 1.1)
                elif accept_count < 500 * 0.1:
                    delta_max = max(1e-4, delta_max * 0.9)
                accept_count = 0
            
            # Cool temperature
            temp *= alpha
            if temp < 1e-6:
                break
    
    if best_sa_c5 < best_c5:
        best_c5 = best_sa_c5
        best_h = np.concatenate([best_sa_h, best_sa_h[::-1]])
    
    # Final refinement with L-BFGS-B on the full vector
    if time.time() - start_time > budget_s * 0.99:
        return (best_h.tolist(), float(best_c5), n_points)
        
    def final_objective(h):
        return compute_c5_smooth(h, beta=200.0)
    
    def final_constraint(h):
        return np.sum(h) * dx - 1
    
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': final_constraint}
        
        res = minimize(
            final_objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'gtol': 1e-12, 'maxiter': 500}
        )
        
        if res.success:
            h_final = res.x
            c5_final = compute_c5_exact(h_final)
            
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final
                
    except Exception:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
