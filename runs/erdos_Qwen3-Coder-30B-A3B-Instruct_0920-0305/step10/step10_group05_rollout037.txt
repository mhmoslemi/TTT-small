```python
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import differential_evolution, minimize
import time

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
HALF_N = n_points // 2

def autocorr_fft(h):
    """Compute aperiodic autocorrelation via FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(h, L)).real
    return corr

def exact_c5(u):
    """Compute exact C5 from half-vector u"""
    h = np.concatenate([u, u[::-1]])  # full vector
    A = autocorr_fft(h)
    return 1 - dx * np.min(A)

def lse_c5(u, beta):
    """Smooth surrogate for C5 using log-sum-exp"""
    h = np.concatenate([u, u[::-1]])
    A = autocorr_fft(h)
    return 1 - (dx / beta) * np.log(np.sum(np.exp(beta * A)))

def grad_lse_c5(u, beta):
    """Gradient of LSE surrogate"""
    h = np.concatenate([u, u[::-1]])
    A = autocorr_fft(h)
    w = np.exp(beta * A)
    w_sum = np.sum(w)
    # Gradient computation using convolution
    h_rev = h[::-1]
    conv = ifft(fft(w, 2*len(h)-1) * fft(h_rev, 2*len(h)-1)).real
    grad = -dx * conv[:len(u)] / w_sum
    return grad

def project_to_sum_box(u, target_sum):
    """Water-filling projection"""
    def f(lam):
        projected = np.clip(u + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(u + (a + b)/2, 0, 1)
    return result

def parametric_to_halfvector(params, n):
    """3-level symmetric shape"""
    a, p = params
    if a < 0 or a > 0.5:
        return None
    u = np.ones(n//2)
    u[:n//4] = 1
    u[n//4:n//2] = p
    return u[:n//2]

def run_de():
    """Differential Evolution on 2D parameter space"""
    def obj(params):
        u = parametric_to_halfvector(params, n_points)
        if u is None:
            return 1e10
        return exact_c5(u)
    
    bounds = [(0.05, 0.45), (0.1, 0.9)]
    result = differential_evolution(obj, bounds, seed=42, maxiter=200, popsize=15)
    return result.x

def run_lbfgsb(u0, beta=200):
    """L-BFGS-B refinement on full discretization"""
    def obj(u):
        return lse_c5(u, beta)
    
    def grad(u):
        return grad_lse_c5(u, beta)
    
    # Use L-BFGS-B with bounds
    res = minimize(obj, u0, method='L-BFGS-B', jac=grad, bounds=[(0,1)]*len(u0), options={'maxiter': 800, 'gtol': 1e-10})
    u_opt = project_to_sum_box(res.x, TARGET_SUM)
    return u_opt, exact_c5(u_opt)

def run_sa(u_start, budget_s):
    """Simulated Annealing polishing"""
    start_time = time.time()
    u = u_start.copy()
    current_c5 = exact_c5(u)
    best_u = u.copy()
    best_c5 = current_c5
    T = 5e-3
    alpha = 0.9997
    delta_max = 0.1
    accept_count = 0
    total_count = 0
    
    while time.time() - start_time < budget_s:
        if T < 1e-6:
            break
            
        i, j = np.random.choice(len(u), 2, replace=False)
        delta = np.random.uniform(0, delta_max)
        
        if u[i] >= delta and u[j] <= 1 - delta:
            u_new = u.copy()
            u_new[i] -= delta
            u_new[j] += delta
            u_new = project_to_sum_box(u_new, TARGET_SUM)
            new_c5 = exact_c5(u_new)
            
            if new_c5 < current_c5:
                u = u_new
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_u = u_new.copy()
                    best_c5 = new_c5
                accept_count += 1
            else:
                if np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                    u = u_new
                    current_c5 = new_c5
                    accept_count += 1
            total_count += 1
            
        # Adjust delta_max to maintain acceptance ratio in [0.2, 0.4]
        if total_count % 500 == 0 and total_count > 0:
            ratio = accept_count / total_count
            if ratio < 0.2:
                delta_max *= 1.1
            elif ratio > 0.4:
                delta_max *= 0.9
            accept_count = 0
            total_count = 0
            
        T *= alpha
        
    return best_u, best_c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: Parametric sweep
    best_params = None
    best_c5_sweep = float('inf')
    for a in np.linspace(0.05, 0.45, 400):
        p = (1 - 2*a) / (2*(1 - a))
        u = parametric_to_halfvector([a, p], n_points)
        if u is not None:
            c5 = exact_c5(u)
            if c5 < best_c5_sweep:
                best_c5_sweep = c5
                best_params = [a, p]
    
    # Phase 2: DE optimization
    remaining_budget = budget_s - (time.time() - start_time)
    if remaining_budget < 10:
        u_best = parametric_to_halfvector(best_params, n_points)
    else:
        try:
            de_params = run_de()
            u_best = parametric_to_halfvector(de_params, n_points)
            if u_best is None:
                u_best = parametric_to_halfvector(best_params, n_points)
            else:
                best_c5_de = exact_c5(u_best)
                if best_c5_de < best_c5_sweep:
                    best_c5_sweep = best_c5_de
        except Exception:
            u_best = parametric_to_halfvector(best_params, n_points)
    
    # Phase 3: Upscaling and L-BFGS-B refinement
    if time.time() - start_time < 0.9 * budget_s:
        try:
            u_best = project_to_sum_box(u_best, TARGET_SUM)
            u_opt, c5_opt = run_lbfgsb(u_best)
            if c5_opt < best_c5_sweep:
                best_c5_sweep = c5_opt
                u_best = u_opt
        except Exception:
            pass
    
    # Phase 4: SA polishing
    remaining_budget = budget_s - (time.time() - start_time)
    if remaining_budget > 5:
        try:
            u_final, c5_final = run_sa(u_best, remaining_budget)
            if c5_final < best_c5_sweep:
                best_c5_sweep = c5_final
                u_best = u_final
        except Exception:
            pass
    
    return (np.concatenate([u_best, u_best[::-1]]).tolist(), float(best_c5_sweep), n_points)
```
