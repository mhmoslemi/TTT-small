import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_autocorr_fft(h):
    """Compute auto-correlation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def min_corr_from_autocorr(auto_corr, n):
    """Convert auto-correlation result to min correlation value"""
    return dx * np.min(auto_corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def evaluate_c5_by_autocorr(h):
    """Evaluate C5 using auto-correlation for verification"""
    auto_corr = compute_autocorr_fft(h)
    min_corr = min_corr_from_autocorr(auto_corr, n_points)
    return 1 - min_corr

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_piecewise_constant_h(a):
    """Generate piecewise constant h with symmetric shape"""
    # a = length of edge regions
    # Edge regions at start and end
    # Middle region has height p
    
    if a <= 0 or a >= 1:
        raise ValueError("a should be between 0 and 1")
    
    p = (1 - 2*a) / (2 * (1 - a))  # solve 2*a*1 + (2-2*a)*p = 1 for p
    
    if p <= 0 or p > 1:
        raise ValueError("Could not compute valid p or out of bounds")
    
    h = np.zeros(n_points)
    
    # Edge regions
    edge_indices = int(a * n_points / 2)
    
    # Left edge
    h[:edge_indices] = 1.0
    
    # Right edge  
    h[-edge_indices:] = 1.0
    
    # Middle region
    middle_start = edge_indices
    middle_end = n_points - edge_indices
    h[middle_start:middle_end] = p
    
    # Ensure sum = TARGET_SUM 
    h = project_to_sum_box(h, TARGET_SUM)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Define a range to search for optimal piecewise constant parameters
    a_values = np.linspace(0.05, 0.45, 100) 
    
    best_c5 = float('inf')
    best_h = None
    best_a = None
    
    # Grid search to find a that gives minimal C5 for piecewise constant h
    for a in a_values:
        if time.time() - start_time > budget_s * 0.7:
            break
        try:
            h = generate_piecewise_constant_h(a)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_a = a
        except Exception:
            continue
    
    # Perturb around the best solution found so far with Simulated Annealing if we still have time
    if time.time() - start_time < budget_s * 0.9:
        # Use the best piecewise constant solution as starting point
        current_h = best_h.copy()
        current_c5 = best_c5

        # Simulated Annealing
        T_start = 0.002
        max_iter = 100000
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s:
                break

            # Pick two indices
            i1 = np.random.randint(0, n_points)
            i2 = np.random.randint(0, n_points)
            
            # Choose random delta
            delta = np.random.uniform(-0.05, 0.05)
            
            # Apply adjustment with bounds checking
            if (current_h[i1] + delta >= 0) and (current_h[i2] - delta >= 0) and \
               (current_h[i1] + delta <= 1) and (current_h[i2] - delta <= 1):
                
                # Update values
                h_new = current_h.copy()
                h_new[i1] += delta
                h_new[i2] -= delta
                
                # Project to sum and box constraints
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                
                # Evaluate
                c5_new = evaluate_c5(h_new)
                
                # Accept or reject
                T = T_start * (1 - i / max_iter)
                if c5_new < current_c5 or np.random.rand() < np.exp(-(c5_new - current_c5) / T):
                    current_h = h_new
                    current_c5 = c5_new
                    
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = current_h.copy()
        return (best_h.tolist(), float(best_c5), n_points)
    else:
        return (best_h.tolist(), float(best_c5), n_points)
