```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_autocorr_fft(h):
    """Compute auto-correlation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def min_autocorr_from_autocorr(auto_corr, n):
    """Convert auto-correlation result to min autocorrelation value"""
    return dx * np.min(auto_corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def evaluate_c5_by_autocorr(h):
    """Evaluate C5 using min autocorrelation"""
    auto_corr = compute_autocorr_fft(h)
    min_auto = min_autocorr_from_autocorr(auto_corr, n_points)
    return 1.0 - min_auto

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_piecewise_h(a):
    """Generate piecewise constant h based on parameter a"""
    # a is the length of each edge region (0 to a and 2-a to 2)
    # middle region is (a to 2-a)
    if a >= 0.5:
        a = 0.4999
    if a <= 0:
        a = 0.0001
    
    # Compute p based on constraint: 2*a + (2-2*a)*p = 1
    p = (1 - 2*a) / (2 * (1 - a))
    if p > 1:
        p = 1.0
    if p < 0:
        p = 0.0
    
    h = np.zeros(n_points)
    
    # Calculate point indices for the regions
    edge_length = int(a * n_points / 2)
    if edge_length < 1:
        edge_length = 1
    mid_length = n_points - 2 * edge_length
    
    # Fill the regions
    h[:edge_length] = 1.0
    h[edge_length:edge_length + mid_length] = p
    h[edge_length + mid_length:] = 1.0
    
    # Ensure sum equals TARGET_SUM
    h = h * TARGET_SUM / np.sum(h)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # First try parametric search
    best_a = 0.1
    best_c5 = 1.0
    
    # Try different a parameters
    a_values = np.linspace(0.02, 0.45, 450)
    
    for a in a_values:
        if time.time() - start_time > budget_s * 0.4:
            break
        try:
            h = generate_piecewise_h(a)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_a = a
                best_c5 = c5
        except Exception:
            continue
    
    # Generate best piecewise h
    h0 = generate_piecewise_h(best_a)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Optimized specific search with better starting points
    # Try random restarts with different initialization strategies
    for restart in range(20):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        h_random = np.random.rand(n_points)
        h_random = project_to_sum_box(h_random, TARGET_SUM)
        
        # Start with better h from our parametric search
        h_start = generate_piecewise_h(np.random.uniform(0.05, 0.35))
        h_start = project_to_sum_box(h_start, TARGET_SUM)
        
        # Try with simulated annealing
        h_sa = simulated_annealing(h_start, 15000, 0.001, start_time, budget_s)
        c5_sa = evaluate_c5(h_sa)
        
        if c5_sa < best_c5:
            best_h = h_sa.copy()
            best_c5 = c5_sa
    
    # Final local optimization refinement with gradient descent
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        # Local optimization with trust-constr 
        res = minimize(
            objective, 
            best_h, 
            method='L-BFGS-B', 
            bounds=bounds, 
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_opt = res.x
        else:
            h_opt = best_h
        h_opt = project_to_sum_box(h_opt, TARGET_SUM)
        c5_opt = evaluate_c5(h_opt)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            
    except Exception:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)

def simulated_annealing(h_start, max_iter, T0, start_time, budget_s):
    """Perform simulated annealing optimization"""
    h_current = h_start.copy()
    c5_current = evaluate_c5(h_current)
    
    T = T0
    alpha = 0.99999
    epsilon_sigma = 0.01
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Choose two random indices different from each other
        i1 = np.random.randint(0, len(h_current))
        i2 = np.random.randint(0, len(h_current))
        while i1 == i2:
            i2 = np.random.randint(0, len(h_current))
        
        # Random move amount
        delta = np.random.normal(0, epsilon_sigma)
        
        # Ensure valid move
        if delta > 0 and h_current[i1] > delta:
            # Move mass from i1 to i2
            h_new = h_current.copy()
            h_new[i1] -= delta
            h_new[i2] += delta
            
            # Ensure constraints
            h_new[i1] = max(0, min(1, h_new[i1]))
            h_new[i2] = max(0, min(1, h_new[i2]))
            
            # Project back to ensure sum constraint is met
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
        elif delta < 0 and h_current[i2] > -delta:
            # Move mass from i2 to i1
            h_new = h_current.copy()
            h_new[i1] += delta  
            h_new[i2] -= delta
            
            # Ensure constraints
            h_new[i1] = max(0, min(1, h_new[i1]))
            h_new[i2] = max(0, min(1, h_new[i2]))
            
            # Project back to ensure sum constraint is met
            h_new = project_to_sum_box(h_new, TARGET_SUM)
        else:
            continue
            
        # Compute new cost
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(c5_new - c5_current) / T):
                h_current = h_new.copy()
                c5_current = c5_new
                
        # Update temperature
        T *= alpha
    
    return h_current
```
