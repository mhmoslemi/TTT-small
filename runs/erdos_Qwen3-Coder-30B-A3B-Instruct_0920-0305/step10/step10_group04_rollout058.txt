import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_overlap_from_binary(x):
    """Compute the overlap using binary correlation, with incremental setup."""
    # Directly compute auto-correlation in double precision
    A = correlate(x, x, mode="full").astype(np.float64)
    # Compute prefix sum of x for fast window summation
    prefix_x = np.concatenate([[0], np.cumsum(x)])
    # Compute window sums with help of prefix
    n = len(x)
    window_sums = np.zeros(2 * n - 1)
    for d in range(-(n-1), n):
        start = max(0, d)
        end = min(n, n + d)
        window_sums[d + n - 1] = prefix_x[end] - prefix_x[start]
    # Compute O_d := (window_sum) - A_d
    O = window_sums - A
    # Return max of O_scaled
    return dx * np.max(O)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_initial_binary():
    """Generate an initial binary vector with exactly n/2 ones."""
    x = np.zeros(n_points, dtype=np.float64)
    x[:n_points//2] = 1.0
    np.random.shuffle(x)
    return x

def get_correlation_delta(x, i, j):
    """Compute delta A for a swap between positions i and j."""
    # Build padded version for index access
    x_padded = np.pad(x, (n_points-1, n_points-1), mode='constant', constant_values=0)
    # Extract slices for both positions
    i_start, i_end = i, i + 2 * n_points - 1
    j_start, j_end = j, j + 2 * n_points - 1
    x_i = x_padded[i_start:i_end]
    x_j = x_padded[j_start:j_end]
    # Reverse slices for negative shifts
    x_i_rev = x_i[::-1]
    x_j_rev = x_j[::-1]
    # Compute the difference vector
    delta_A = - (x_i + x_i_rev) + (x_j + x_j_rev)
    return delta_A

def update_after_swap(x, A, i, j):
    """Correct A to represent updated correlation after a swap at i,j."""
    delta_A = get_correlation_delta(x, i, j)
    return A + delta_A

def find_worst_shift(O_vals):
    """Find the shift index with the maximum overlap."""
    return np.argmax(O_vals)

def targeted_swaps(x, i, j):
    """Swap x[i] and x[j] and update state."""
    x[i], x[j] = x[j], x[i]
    return x

def compute_all_overlaps(x):
    """Compute all O_d for current x."""
    A = correlate(x, x, mode="full").astype(np.float64)
    prefix_x = np.concatenate([[0], np.cumsum(x)])
    n = len(x)
    window_sums = np.zeros(2 * n - 1)
    for d in range(-(n-1), n):
        start = max(0, d)
        end = min(n, n + d)
        window_sums[d + n - 1] = prefix_x[end] - prefix_x[start]
    O = window_sums - A
    return O

def run(seed=42, budget_s=600, **kwargs):
    global n_points, dx, TARGET_SUM
    np.random.seed(seed)
    start_time = time.time()
    
    # Decide how to init, could be from initial_h_values if we want
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        init_vec = np.array(initial_h_values)
        init_vec = project_to_sum_box(init_vec, TARGET_SUM)
    else:
        init_vec = generate_initial_binary()
        
    best_x = init_vec.copy()
    A_now = correlate(best_x, best_x, mode="full").astype(np.float64)
    O_val = compute_all_overlaps(best_x)
    best_C = dx * np.max(O_val)

    # Simulated Annealing
    temp = 2.0 * dx
    T_min = 1e-6 * dx
    alpha = 0.99995  # cooling rate
    steps = 0
    success_count = 0
    recent_improvements = 0
    stagnation_threshold = 20000
    diversity_injection = 0.05
    
    while steps < 1000000 and time.time() - start_time < budget_s - 2:
        if steps > 0 and steps % stagnation_threshold == 0:
            if recent_improvements == 0:
                # Inject diversity
                num_flip = int(n_points * diversity_injection)
                flip_indices = np.random.choice(n_points, num_flip)
                # Flip bits temporarily
                for idx in flip_indices:
                    best_x[idx] = 1 - best_x[idx]
                # Fix the sum constraint
                ones_count_now = np.sum(best_x)
                if ones_count_now != TARGET_SUM:
                    # Find excess 1s and flip them to 0s
                    ones_where = np.where(best_x == 1)[0]
                    zeros_where = np.where(best_x == 0)[0]
                    num_to_flip = abs(ones_count_now - TARGET_SUM)
                    if num_to_flip > 0:
                        # Swap out a 1s with a 0s to fix 
                        if num_to_flip > len(zeros_where):
                            num_to_flip = min(num_to_flip, len(ones_where))
                        idx_flipped = ones_where[:num_to_flip]
                        best_x[idx_flipped] = 0
                        indices_to_flip_back = zeros_where[:num_to_flip]
                        best_x[indices_to_flip_back] = 1
                # Clean up after bias injection
                recent_improvements = 0
                temp = temp * 0.5  # lower temp 
                if temp < T_min:
                    temp = T_min
                continue
            else:
                recent_improvements = 0
            temp = max(T_min, temp * alpha)
            
        # Determine move type and apply swap with O(n) update
        if np.random.rand() < 0.4:
            # Random swap
            i, j = np.random.choice(n_points, 2, replace=False)
            while best_x[i] == best_x[j]:  # make sure diff values
                i, j = np.random.choice(n_points, 2, replace=False)
            A_now = update_after_swap(best_x, A_now, i, j)
        else:
            # Targeted swap based on current worst shift
            O_vals = compute_all_overlaps(best_x)
            d_star = find_worst_shift(O_vals)
            # Get neighbors' Indx Sets
            U = []
            V = []
            offset = d_star + n_points - 1  # full correlation offset
            # All i where x_i=1 and x_{i+d_star}=0 and valid
            for i in range(n_points):
                j = i + d_star
                if 0 <= j < n_points and best_x[i] == 1 and best_x[j] == 0:
                    U.append(i)
                if 0 <= j < n_points and best_x[i] == 0 and best_x[j] == 1:
                    V.append(i)
            if len(U) > 0 and len(V) > 0:
                i = np.random.choice(U)
                j = np.random.choice(V)
                A_now = update_after_swap(best_x, A_now, i, j)
            else:
                # fallback on random
                i, j = np.random.choice(n_points, 2, replace=False)
                while best_x[i] == best_x[j]:
                    i, j = np.random.choice(n_points, 2, replace=False)
                A_now = update_after_swap(best_x, A_now, i, j)
        
        # Recompute over: only has full overlaps from A_now
        O = compute_all_overlaps(best_x)
        new_C = dx * np.max(O)
        delta_C = new_C - best_C
        if delta_C < 0:
            best_C = new_C
            recent_improvements += 1
        elif np.random.rand() < np.exp(-delta_C / temp):
            best_C = new_C
            recent_improvements += 1
        else:
            # revert swap by doing it again (since swapping twice is identity)
            if np.random.rand() < 0.5:
                i, j = np.random.choice(n_points, 2, replace=False)
                while best_x[i] == best_x[j]:
                    i, j = np.random.choice(n_points, 2, replace=False)
                A_now = update_after_swap(best_x, A_now, i, j)
        
        steps += 1
    
    # Final polishing via SLSQP on the best binary point
    h_final = best_x.copy()
    # Project to a valid floating point solution 
    h_final = project_to_sum_box(h_final, TARGET_SUM)
    
    # Final tuning via scipy SLSQP
    try:
        from scipy.optimize import minimize
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
        res = minimize(lambda x: dx * np.max(correlate(x, 1 - x, mode="full")), 
                      h_final,
                      method='SLSQP',
                      bounds=bounds,
                      constraints=cons,
                      options={'maxiter': 1500, 'ftol': 1e-12, 'gtol': 1e-12})
        if res.success:
            h_final = res.x
        h_final = project_to_sum_box(h_final, TARGET_SUM)
    except Exception as e:
        pass
    
    # Optional smoothing
    smoothed = np.convolve(h_final, np.ones(3)/3, mode='same')
    smoothed = project_to_sum_box(smoothed, TARGET_SUM)
    
    # Second optimization pass on smoothed version if beneficial
    try:
        res2 = minimize(lambda x: dx * np.max(correlate(x, 1 - x, mode="full")), 
                       smoothed,
                       method='SLSQP',
                       bounds=[(0, 1) for _ in range(n_points)],
                       constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-12})
        if res2.success:
            smoothed = res2.x
        smoothed = project_to_sum_box(smoothed, TARGET_SUM)
    except Exception as e:
        pass

    # Final bound
    final_c5 = dx * np.max(correlate(smoothed, 1 - smoothed, mode="full"))
    
    return (smoothed.tolist(), float(final_c5), n_points)
