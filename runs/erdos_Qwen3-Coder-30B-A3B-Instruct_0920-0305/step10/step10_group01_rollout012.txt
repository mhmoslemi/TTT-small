import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None

def full_corr(a, b):
    """Compute correlation with zero padding using FFT"""
    L = 2 * len(a) - 1
    return np.fft.ifft(np.fft.fft(a, L) * np.conj(np.fft.fft(b, L))).real

def smooth_max_and_weights(c, alpha):
    """Compute smooth max and corresponding softmax weights"""
    # Subtract max for numerical stability
    c_stable = c - np.max(c)
    exp_c = np.exp(alpha * c_stable)
    Z = np.sum(exp_c)
    w = exp_c / Z
    smooth_max_val = (np.log(Z) + np.max(c)) / alpha
    return smooth_max_val, w

def compute_gradient(h, alpha):
    """Compute gradient of smooth max using FFT"""
    # c_k = sum_i h_i * (1 - h_{i+k})
    c = full_corr(h, 1 - h)
    
    # Softmax weights
    _, w = smooth_max_and_weights(c, alpha)
    
    # Symmetrize weights: w_sym[k] = w[k] + w[-k]
    w_sym = w + w[::-1]
    
    # Correlation of symmetrized weights with h
    g = full_corr(w_sym, h)
    
    # Gradient: 1 - g[:n_points]
    grad = 1.0 - g[:len(h)]
    
    return grad

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using direct correlation"""
    c = full_corr(h, 1 - h)
    return dx * np.max(c)

def objective_and_gradient(h, alpha):
    """Return objective function value and gradient for given alpha"""
    c = full_corr(h, 1 - h)
    obj_val, w = smooth_max_and_weights(c, alpha)
    grad = compute_gradient(h, alpha)
    return obj_val, grad

def continuation_optimize(h0, alpha_list):
    """Run optimization with continuation schedule"""
    h_current = h0.copy()
    for alpha in alpha_list:
        def obj_and_grad(h):
            return objective_and_gradient(h, alpha)
        
        # Define constraints and bounds
        A_eq = np.array([[1.0] * len(h0)])
        b_eq = np.array([TARGET_SUM])
        bounds = [(0, 1) for _ in range(len(h0))]
        
        # Optimize with trust-constr
        res = minimize(
            obj_and_grad,
            h_current,
            method='trust-constr',
            jac=True,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda x: A_eq @ x - b_eq},
            options={'maxiter': 1000, 'verbose': 0}
        )
        
        if res.success:
            h_current = res.x
        else:
            # If optimization fails, try to proceed with current h
            pass
            
        # Project to sum constraint
        h_current = project_to_sum_box(h_current, TARGET_SUM)
    
    return h_current
