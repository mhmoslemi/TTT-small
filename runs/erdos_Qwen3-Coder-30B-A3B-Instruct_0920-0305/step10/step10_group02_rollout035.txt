```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from math import gcd

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_linear_congruential_sequences():
    """Generate candidate sequences using linear congruential method"""
    candidates = []
    coprimes = [i for i in range(1, n_points) if gcd(i, n_points) == 1]
    
    for a in coprimes[:20]:  # Limit to first 20 coprimes for speed
        for b in range(n_points):
            h = np.zeros(n_points)
            for i in range(n_points):
                if ((i * a + b) % n_points) < n_points // 2:
                    h[i] = 1
            candidates.append(h)
    
    return candidates

def evaluate_candidates(candidates):
    """Evaluate all candidate sequences and return the best"""
    best_c5 = float('inf')
    best_h = None
    
    for h in candidates:
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def smooth_boundaries(h, window_size=3):
    """Smooth boundaries by linear interpolation"""
    h_smooth = h.copy()
    boundaries = np.where(np.diff(h) != 0)[0]
    
    for boundary in boundaries:
        start = max(0, boundary - window_size // 2)
        end = min(n_points, boundary + window_size // 2 + 1)
        
        if start > 0 and end < n_points:
            # Linear interpolation from 0 to 1 (or 1 to 0)
            if h[start-1] == 0 and h[end] == 1:
                # Going from 0 to 1
                values = np.linspace(0, 1, end - start)
                h_smooth[start:end] = values
            elif h[start-1] == 1 and h[end] == 0:
                # Going from 1 to 0
                values = np.linspace(1, 0, end - start)
                h_smooth[start:end] = values
    
    # Adjust to maintain sum constraint
    current_sum = np.sum(h_smooth)
    adjustment = (n_points / 2 - current_sum) / n_points
    h_smooth += adjustment
    
    # Clip to [0, 1]
    h_smooth = np.clip(h_smooth, 0, 1)
    return h_smooth

def compute_gradient(h, alpha=10.0):
    """Compute gradient of softmax surrogate for gradient descent"""
    # Compute all overlaps O_k
    corr = compute_corr_fft(h)
    overlaps = corr[:n_points] * dx  # Only the first n_points values
    
    # Compute softmax of overlaps
    max_overlap = np.max(overlaps)
    shifted_overlaps = overlaps - max_overlap
    exp_overlaps = np.exp(alpha * shifted_overlaps)
    weights = exp_overlaps / np.sum(exp_overlaps)
    
    # Compute gradient
    grad = np.zeros_like(h)
    for k in range(n_points):
        if k == 0:
            # Direct overlap
            for i in range(n_points):
                grad[i] += weights[k] * (1 - h[(i+k) % n_points] - h[(i-k) % n_points])
        else:
            # For shift k, compute contributions
            for i in range(n_points):
                grad[i] += weights[k] * (1 - h[(i+k) % n_points] - h[(i-k) % n_points])
    
    return grad

def gradient_descent_refinement(h, max_iter=100):
    """Refine solution using gradient descent on softmax surrogate"""
    h_refined = h.copy()
    best_c5 = evaluate_c5(h_refined)
    
    for iter in range(max_iter):
        grad = compute_gradient(h_refined)
        step_size = 0.001
        
        # Update with gradient descent
        h_new = h_refined - step_size * grad
        
        # Project back to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            h_refined = h_new
            best_c5 = new_c5
        else:
            # If no improvement after several iterations, reduce step size
            if iter > 10 and iter % 5 == 0:
                step_size /= 2
                
    return h_refined, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Generate candidate sequences using linear congruential method
    candidates = generate_linear_congruential_sequences()
    
    best_h, best_c5 = evaluate_candidates(candidates)
    
    # Smoothing step
    if time.time() - start_time < budget_s * 0.8:
        h_smooth = smooth_boundaries(best_h)
        h_smooth = project_to_sum_box(h_smooth, TARGET_SUM)
        smooth_c5 = evaluate_c5(h_smooth)
        if smooth_c5 < best_c5:
            best_h = h_smooth
            best_c5 = smooth_c5
    
    # Gradient-based refinement
    if time.time() - start_time < budget_s * 0.9:
        h_refined, refined_c5 = gradient_descent_refinement(best_h)
        if refined_c5 < best_c5:
            best_h = h_refined
            best_c5 = refined_c5
    
    # Final SLSQP optimization with exact objective
    if time.time() - start_time < budget_s - 10:
        try:
            def objective(h):
                return evaluate_c5(h)
            
            def constraint_eq(h):
                return np.sum(h) - TARGET_SUM
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': constraint_eq}
            
            # Use a rapid variant of SLSQP
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10}
            )
            if res.success:
                best_h = res.x
            best_h = project_to_sum_box(best_h, TARGET_SUM)
            best_c5 = evaluate_c5(best_h)
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
```
