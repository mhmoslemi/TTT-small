import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from scipy.fft import fft, ifft

def project_to_sum_box(u, target_sum):
    """Water-filling projection to enforce sum constraint"""
    def f(lam):
        projected = np.clip(u + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    for _ in range(30):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(u + (a + b)/2, 0, 1)
    return result

def autocorr_fft(h):
    """Compute aperiodic autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.real(ifft(fft(h_rev, L) * fft(h, L)))
    return corr

def exact_c5(u):
    """Compute exact C5 from half-vector u"""
    n = len(u)
    h = np.concatenate([u, u[::-1]])
    dx = 2.0 / (2 * n)
    corr = autocorr_fft(h)
    return 1 - dx * np.min(corr)

def lse_c5(u, beta):
    """Smooth surrogate for C5 using log-sum-exp"""
    n = len(u)
    h = np.concatenate([u, u[::-1]])
    dx = 2.0 / (2 * n)
    corr = autocorr_fft(h)
    return 1 - dx * np.log(np.sum(np.exp(beta * corr))) / beta

def grad_lse_c5(u, beta):
    """Gradient of smooth surrogate using FFT"""
    n = len(u)
    h = np.concatenate([u, u[::-1]])
    dx = 2.0 / (2 * n)
    corr = autocorr_fft(h)
    
    # Compute weights
    w = np.exp(beta * corr)
    Z = np.sum(w)
    
    # Gradient computation using convolution
    w_rev = w[::-1]
    # Forward convolution with h: (h * w_rev)[i] = sum_j h[j] * w_rev[i-j]
    # But we're not computing full convolution, we're computing element-wise
    # So we work with the expression: 
    # d/dh_i lse = - dx * sum_k w_k * (h_{i+k} + h_{i-k}) / Z
    
    # This is a bit tricky with vectors, let's compute it directly:
    grad = np.zeros_like(u)
    for i in range(n):
        val = 0.0
        for k in range(-n+1, n):  # valid indices for correlation
            j1 = i + k
            j2 = i - k
            if 0 <= j1 < 2*n and 0 <= j2 < 2*n:
                val += w[k + n - 1] * (h[j1] + h[j2])
        grad[i] = -dx * val / Z
    
    # Bring back to half-vector (since we computed full gradient of h)
    # For the half-vector gradient, there might be a factor of 2 or adjustment
    return grad

def parametric_to_halfvector(params, n):
    """Build half-vector from parameters:
    params: [a, r, p] where a=half-width, r=ramp width, p=middle height
    In case of 3-level, only a and p
    """
    if len(params) == 2:
        a, p = params
    else:
        a, r, p = params
        
    m = n // 2
    u = np.zeros(m)
    
    if len(params) == 2:
        # 3-level shape: edge (a), middle (2-2a), edge (a) 
        u[:int(np.ceil(a * m))] = 1.0
        u[int(np.ceil(a * m)):int(np.floor((2-a)*m))] = p
        u[int(np.floor((2-a)*m)):] = 1.0
    else:
        # 5-level shape: edge (a), ramp (r), middle (2-2a-2r), ramp (r), edge (a)
        u[:int(np.ceil(a * m))] = 1.0
        start_ramp = int(np.ceil(a * m))
        end_ramp = start_ramp + int(np.ceil(r * m))
        if end_ramp <= int(np.floor((1 - a) * m)):
            ramp_indices = np.arange(start_ramp, end_ramp)
            u[ramp_indices] = 1.0 - (1.0 - p) * (ramp_indices - start_ramp) / (end_ramp - start_ramp)
            mid_start = end_ramp
            mid_end = mid_start + int(np.ceil((2 - 2*a - 2*r) * m))
            if mid_end <= int(np.floor((1 - a) * m)):
                u[mid_start:mid_end] = p
                start_rev = mid_end
                u[start_rev:] = 1.0  # edge
                # Mirror the ramp from the other side
                right_half = u[:len(u)//2]
                u[int(np.floor(a * m))+1:] = np.flipud(right_half[1:])
            else:
                u[mid_start:] = 1.0
        else:
            # fallback to 3-level
            u[int(np.ceil(a * m)):int(np.floor((2-a)*m))] = p
    
    # Normalize to sum constraint
    target_sum = m / 2
    u = project_to_sum_box(u, target_sum)
    return u

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 120
    dx = 2.0 / n_points
    
    # Phase 1: Parameter sweep for simple 3-level shape  
    best_c5 = 1.0
    best_u = None
    
    n_sweep = 400
    a_values = np.linspace(0.05, 0.45, n_sweep)
    for a in a_values:
        p = (1 - 2*a) / (2 * (1 - a))  
        u = parametric_to_halfvector([a, p], n_points//2)
        c5 = exact_c5(u)
        if c5 < best_c5:
            best_c5 = c5
            best_u = u.copy()
    
    # Phase 2: Differential Evolution on parametric family
    if time.time() - start_time > 0.1:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 20:
            return (np.concatenate([best_u, best_u[::-1]]).tolist(), float(best_c5), n_points)
    
    def objective(params):
        u = parametric_to_halfvector(params, n_points//2)
        return exact_c5(u)
    
    bounds = [(0.05, 0.45), (0.0, 0.3)] 
    result = differential_evolution(objective, bounds, seed=seed, maxiter=200, popsize=15)
    u_de = parametric_to_halfvector(result.x, n_points//2)
    c5_de = exact_c5(u_de)
    
    if c5_de < best_c5:
        best_c5 = c5_de
        best_u = u_de.copy()
    
    # Phase 3: Upscaling and L-BFGS-B refinement
    if time.time() - start_time > 0.1:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 100:
            return (np.concatenate([best_u, best_u[::-1]]).tolist(), float(best_c5), n_points)
    
    n_fine = 240
    u_refine = np.interp(np.linspace(0, 1, n_fine//2), np.linspace(0, 1, len(best_u)), best_u)
    u_refine = project_to_sum_box(u_refine, n_fine / 4)
    
    def objective_lse(u):
        return lse_c5(u, beta=200)
    
    def grad_lse(u):
        return grad_lse_c5(u, beta=200)
    
    res = minimize(objective_lse, u_refine, method='L-BFGS-B', 
                   bounds=[(0, 1)] * len(u_refine),
                   jac=grad_lse, options={'maxiter': 800, 'ftol': 1e-12, 'gtol': 1e-10})
    
    u_final = project_to_sum_box(res.x, n_fine / 4)
    final_c5 = exact_c5(u_final)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_u = u_final.copy()
    
    # Phase 4: Simulated Annealing polishing
    if time.time() - start_time > 0.1:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 10:
            return (np.concatenate([best_u, best_u[::-1]]).tolist(), float(best_c5), n_points)
    
    def acceptance_probability(delta, temp):
        if delta < 0:
            return 1.0
        return np.exp(-delta / temp)
    
    current_u = best_u.copy()
    current_c5 = exact_c5(current_u)
    best_u_sa = current_u.copy()
    best_c5_sa = current_c5
    
    temp = 5e-3
    alpha = 0.9997
    max_delta = 0.1
    accept_count = 0
    total_moves = 0
    
    while temp > 1e-6 and time.time() - start_time < budget_s - 10:
        i, j = np.random.choice(len(current_u), 2, replace=False)
        delta = np.random.uniform(0, max_delta)
        
        # Ensure mass transfer is valid
        if current_u[i] >= delta and current_u[j] <= 1 - delta:
            new_u = current_u.copy()
            new_u[i] -= delta
            new_u[j] += delta
            new_u = project_to_sum_box(new_u, n_fine / 4)
            new_c5 = exact_c5(new_u)
            
            delta_c5 = new_c5 - current_c5
            if acceptance_probability(delta_c5, temp) > np.random.random():
                current_u = new_u
                current_c5 = new_c5
                accept_count += 1
                
                if current_c5 < best_c5_sa:
                    best_c5_sa = current_c5
                    best_u_sa = current_u.copy()
            
            total_moves += 1
            if total_moves % 500 == 0:
                if accept_count / total_moves < 0.2:
                    max_delta *= 0.9
                elif accept_count / total_moves > 0.4:
                    max_delta *= 1.1
                accept_count = 0
                total_moves = 0
                
        temp *= alpha
        
    if best_c5_sa < best_c5:
        best_c5 = best_c5_sa
        best_u = best_u_sa
    
    # Final return
    full_h = np.concatenate([best_u, best_u[::-1]])
    return (full_h.tolist(), float(best_c5), n_points)
