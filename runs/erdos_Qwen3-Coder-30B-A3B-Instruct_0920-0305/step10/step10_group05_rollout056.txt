import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import differential_evolution, minimize
import time

# Global variables
n_points = 120
dx = 2.0 / n_points
target_sum = n_points / 4
n_half = n_points // 2

def project_to_sum_box(u, target_sum):
    """Water-filling projection to box constraints with sum = target_sum"""
    def f(lam):
        projected = np.clip(u + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(u + (a + b)/2, 0, 1)
    return result

def autocorr_fft(h):
    """Compute aperiodic autocorrelation via FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(h, L)).real
    return corr

def exact_c5(u):
    """Compute exact C5 from half-vector u"""
    # Reconstruct full vector with symmetry
    h = np.concatenate([u, u[::-1]])
    corr = autocorr_fft(h)
    return 1 - dx * np.min(corr)

def lse_c5(u, beta):
    """Smooth surrogate of C5 using log-sum-exp"""
    # Reconstruct full vector with symmetry
    h = np.concatenate([u, u[::-1]])
    corr = autocorr_fft(h)
    return 1 - (dx / beta) * np.log(np.sum(np.exp(beta * corr)))

def grad_lse_c5(u, beta):
    """Gradient of LSE surrogate of C5"""
    # Reconstruct full vector with symmetry
    h = np.concatenate([u, u[::-1]])
    corr = autocorr_fft(h)
    w = np.exp(beta * corr)
    w_sum = np.sum(w)
    
    # Compute grad = -dx * (conv(w, h_rev) + conv(w, h)) / w_sum
    # where conv is aperiodic convolution
    L = 2 * len(h) - 1
    
    # Compute w * h_rev convolution (reverse)
    h_rev = h[::-1]
    conv1 = ifft(fft(w, L) * fft(h_rev, L)).real
    
    # Compute w * h convolution
    conv2 = ifft(fft(w, L) * fft(h, L)).real
    
    # Total gradient (we only want the first half)
    grad_full = -dx * (conv1 + conv2) / w_sum
    return grad_full[:len(u)]

def parametric_to_halfvector(params, n_half):
    """Construct half-vector from parametric family (3-level: edge, middle, edge)"""
    # params = [a, p] - edge length, middle height
    a = params[0]
    if len(params) == 2:
        # 3-level shape
        p = params[1]
    else:
        # 2-level shape
        p = (1 - 2 * a) / (2 * (1 - a))
        p = np.clip(p, 0, 1)  # Ensure p in [0,1]
    
    u = np.zeros(n_half)
    
    # Build symmetric 3-level shape
    u[:int(a * n_half)] = 1.0
    u[int(a * n_half):int((1-a) * n_half)] = p
    u[int((1-a) * n_half):] = 1.0
    
    return u

def run_de():
    """Run differential evolution on parametrized family"""
    bounds = [(0.05, 0.45), (0.05, 0.95)]
    
    def objective(params):
        u = parametric_to_halfvector(params, n_half)
        u = project_to_sum_box(u, target_sum)
        return exact_c5(u)
    
    result = differential_evolution(objective, bounds, seed=42, 
                                   maxiter=200, popsize=15, 
                                   atol=1e-5)
    return result.x

def run_lbfgsb(u0, beta):
    """Run L-BFGS-B with projected gradients"""
    def obj(u):
        return lse_c5(u, beta)
    
    def grad(u):
        return grad_lse_c5(u, beta)
    
    # Bounds
    bounds = [(0, 1) for _ in range(len(u0))]
    
    # Optimization
    res = minimize(obj, u0, method='L-BFGS-B', jac=grad, bounds=bounds,
                   options={'gtol': 1e-10, 'maxiter': 800})
    
    # Project to feasibility after optimization
    if res.success:
        u_final = res.x
    else:
        u_final = u0
    
    u_final = project_to_sum_box(u_final, target_sum)
    return u_final

def run_sa(u_initial):
    """Simulated annealing polishing"""
    u_best = u_initial.copy()
    c5_best = exact_c5(u_best)
    
    u_current = u_initial.copy()
    c5_current = c5_best
    
    temperature = 5e-3
    alpha = 0.9997
    max_delta = 0.1
    acceptance_ratio = 0.0
    total_moves = 0
    
    start_time = time.time()
    
    while temperature > 1e-6 and (time.time() - start_time) < 600 * 0.15:
        total_moves += 1
        
        # Propose new state
        i, j = np.random.choice(n_half, 2, replace=False)
        delta = np.random.uniform(0, max_delta)
        
        if u_current[i] >= delta and u_current[j] <= 1 - delta:
            u_new = u_current.copy()
            u_new[i] -= delta
            u_new[j] += delta
            
            # Project to sum constraint
            u_new = project_to_sum_box(u_new, target_sum)
            
            c5_new = exact_c5(u_new)
            
            if c5_new < c5_current:
                u_current = u_new
                c5_current = c5_new
                if c5_new < c5_best:
                    u_best = u_new
                    c5_best = c5_new
                acceptance_ratio = 1.0
            else:
                # Accept with probability
                if np.random.uniform() < np.exp(-(c5_new - c5_current)/temperature):
                    u_current = u_new
                    c5_current = c5_new
                    acceptance_ratio = 1.0
                else:
                    acceptance_ratio = 0.0
        
        # Adjust step size
        if total_moves % 500 == 0:
            if acceptance_ratio > 0.4:
                max_delta = min(max_delta * 1.1, 0.5)
            elif acceptance_ratio < 0.2:
                max_delta *= 0.9
                
        # Cooling
        temperature *= alpha
    
    return u_best, c5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: Parametric sweep
    best_c5 = float('inf')
    best_u = None
    
    # Base parameter sweep (3-level)
    a_values = np.linspace(0.05, 0.45, 400)
    for a in a_values:
        u = parametric_to_halfvector([a, 1.0], n_half)
        # Compute appropriate middle height
        p = (1 - 2*a) / (2*(1-a))
        p = np.clip(p, 0, 1)
        u = parametric_to_halfvector([a, p], n_half)
        u = project_to_sum_box(u, target_sum)
        c5 = exact_c5(u)
        if c5 < best_c5:
            best_c5 = c5
            best_u = u.copy()
    
    # Phase 2: DE global search
    if time.time() - start_time < budget_s * 0.3:
        try:
            de_result = run_de()
            u_de = parametric_to_halfvector(de_result, n_half)
            u_de = project_to_sum_box(u_de, target_sum)
            c5_de = exact_c5(u_de)
            if c5_de < best_c5:
                best_c5 = c5_de
                best_u = u_de.copy()
        except:
            pass
    
    # Phase 3: Upscaled refinement with L-BFGS-B
    if time.time() - start_time < budget_s * 0.65:
        # Upscale and refine
        try:
            u_refined = run_lbfgsb(best_u, 200)
            c5_refined = exact_c5(u_refined)
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_u = u_refined.copy()
        except:
            pass
    
    # Phase 4: SA polishing
    if time.time() - start_time < budget_s * 0.85:
        try:
            u_final, c5_final = run_sa(best_u)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_u = u_final.copy()
        except:
            pass
    
    # Final reconstruct full vector
    h = np.concatenate([best_u, best_u[::-1]])
    
    return (h.tolist(), float(best_c5), n_points)
