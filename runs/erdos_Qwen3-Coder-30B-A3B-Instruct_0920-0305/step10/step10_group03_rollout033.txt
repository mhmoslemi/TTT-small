TARGET_SUM = None

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_binary_milp(N_small, time_limit=30):
    """Solve binary integer program for small N"""
    try:
        # Create variables
        x = cp.Variable(N_small, boolean=True)
        t = cp.Variable()
        
        # Constraints
        constraints = [cp.sum(x) == N_small // 2]
        
        # For each shift k, we compute the correlation
        # We need to add constraints for each shift k
        for k in range(N_small):
            # y[i,k] represents x[i] * x[i+k] for i+k < N
            if k == 0:
                continue  # shift 0 is special case, handled by sum constraint
            # For shift k, we have N-k terms in the sum
            y = cp.Variable(N_small - k, boolean=True)
            # Linearization constraints for y[i,k] = x[i] * x[i+k]
            for i in range(N_small - k):
                constraints.extend([
                    y[i] <= x[i],
                    y[i] <= x[i + k],
                    y[i] >= x[i] + x[i + k] - 1
                ])
            # Constraint sum_i y[i,k] >= N/2 - t
            constraints.append(cp.sum(y) >= N_small // 2 - t)
        
        # Objective: minimize t
        prob = cp.Problem(cp.Minimize(t), constraints)
        
        # Solve with time limit
        prob.solve(solver=cp.CBC, max_seconds=time_limit)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value
    except:
        pass
    return None

def upsample_binary(binary_vec, target_N):
    """Upsample binary vector to target_N by replicating each element"""
    upsample_factor = target_N // len(binary_vec)
    if upsample_factor > 0:
        expanded = np.repeat(binary_vec, upsample_factor)
        if len(expanded) > target_N:
            expanded = expanded[:target_N]
        elif len(expanded) < target_N:
            expanded = np.pad(expanded, (0, target_N - len(expanded)), constant_values=0)
        return expanded
    return binary_vec

def differential_evolution_refine(h0, target_sum, time_budget, start_time):
    """Refine solution using differential evolution"""
    def objective(x):
        # Project back to valid constraints
        x_proj = project_to_sum_box(x, target_sum)
        return evaluate_c5(x_proj)
    
    bounds = [(0, 1) for _ in range(len(h0))]
    
    try:
        result = differential_evolution(
            objective,
            bounds,
            maxiter=int(5000 * time_budget / 600),
            popsize=20,
            mutation=(0.5, 1.0),
            recombination=0.7,
            seed=42,
            disp=False
        )
        if result.success:
            return project_to_sum_box(result.x, target_sum)
    except:
        pass
    return h0

def local_hill_climb(h0, target_sum, time_budget, start_time):
    """Perform local hill climbing to improve solution"""
    current_h = h0.copy()
    current_c5 = evaluate_c5(current_h)
    
    max_iter = int(10000 * time_budget / 600)
    
    for i in range(max_iter):
        if time.time() - start_time > time_budget - 1:
            break
            
        # Make small random changes
        h_new = current_h.copy()
        idx1, idx2 = np.random.choice(len(h_new), 2, replace=False)
        
        # Swap small values
        delta = 0.01 * np.random.randn()
        h_new[idx1] = np.clip(h_new[idx1] + delta, 0, 1)
        h_new[idx2] = np.clip(h_new[idx2] - delta, 0, 1)
        
        # Ensure sum constraint
        h_new = project_to_sum_box(h_new, target_sum)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new)
        
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
            
    return current_h
