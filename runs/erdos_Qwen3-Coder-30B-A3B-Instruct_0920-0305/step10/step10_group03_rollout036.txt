import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time
from cvxpy import *
import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_binary_milp():
    """Solve a small binary MILP for initial skeleton"""
    # binary grid size
    N = 60
    dx_small = 2.0 / N
    
    # Decision variables: x[i] = 0 or 1 for i = 0,...,N-1
    x = cp.Variable(N, boolean=True)
    
    # Auxiliary variables y[i,k] for all valid pairs (i,k) with k >= 1
    y = {}
    # Conflict with i+j >= N, or if that would cross boundary, k is valid iff i+k < N
    for i in range(N):
        for k in range(1, N - i):
            y[i, k] = cp.Variable(boolean=True)
    
    # Putting the constraints that define entries of y
    for i in range(N):
        for k in range(1, N - i):
            # y[i,k] <= x[i], y[i,k] <= x[i+k]
            # y[i,k] >= x[i] + x[i+k] - 1
            constraints = [
                y[i, k] <= x[i],
                y[i, k] <= x[i+k],
                y[i, k] >= x[i] + x[i+k] - 1
            ]
    
    # t represents the worst-case uncovered count
    t = cp.Variable(pos=True)
    
    # Our objective: minimize t
    objective = cp.Minimize(t)
    
    # Minimal constraints:
    # 1. sum of x = N/2  
    constraints = [cp.sum(x) == N/2, t >= 0]
    
    # Add CBC constraint for max unalignment over shifts
    for k in range(1, N):
        total = 0
        for i in range(N-k):
            total += y[i, k]
        constraints.append(total >= N/2 - t)
         
    constraints += [y[i,k].value is not None for i,k in y.keys()]
    
    # Definitions
    model = cp.Problem(objective, constraints)
    
    try:
        model.solve(solver=CPLEX, verbose=False, time_limit=30)
        if model.status in ["optimal", "feasible"]:
            x_opt = np.round(x.value).astype(int)
        else:
            # Return best known solution if unable to get optimal
            x_opt = np.zeros(N)
            # Fill small fraction of entries with 1s (e.g. randomly) to avoid trivial zero solution
            ones = np.random.permutation(N)[:N//2]
            x_opt[ones] = 1
    except:
        # Fallback to uniform random solution
        x_opt = np.zeros(N)
        ones = np.random.permutation(N)[:N//2]
        x_opt[ones] = 1
    
    return x_opt, dx_small

def upsample_binary_vector(h_bin, n_points, dx_small):
    """Upsample binary vector to full resolution"""
    s = int(n_points / len(h_bin))
    h_upsampled = np.repeat(h_bin, s)
    return h_upsampled

def swap_search_worst_shift(h_bin, n_points, max_iter=10000):
    """Perform simple swap heuristic to reduce the worst shift"""
    h = h_bin.copy()
    h = project_to_sum_box(h, n_points / 2)
    
    # Precompute auto-correlation counts
    # We only do it for full overlaps (like in real FFT case)
    def compute_auto_correlation(shift):
        return np.sum(h[:-shift] * h[shift:]) if shift < len(h) else 0
    
    # Simulate correlation calculation via exotic loop to ensure consistency
    def get_alignments(h):
        max_k = len(h) - 1
        rho = np.zeros(max_k)
        for k in range(1, max_k):
            total = 0
            for i in range(len(h) - k):
                total += h[i] * h[i+k]
            rho[k] = total
        return rho
    
    rho = get_alignments(h)
    t_worst = np.max(n_points/2 - rho)
    
    for _ in range(max_iter):
        # Find worst shift
        k_star = np.argmax(n_points/2 - rho)
        
        # Identify points that can form beneficial swap
        A = np.where((h[:-k_star] == 1) & (h[k_star:] == 0))[0]  # these in h will become 0 and 0 in shifted
        B = np.where((h[:-k_star] == 0) & (h[k_star:] == 1))[0]  # these in h become 1 and 1 in shifted
        
        if len(A) > 0 and len(B) > 0:
            i = np.random.choice(A)
            j = np.random.choice(B)
            h[i] = 0
            h[j] = 1
            
            # Recalculate auto-correlation efficiently
            def update_rho_update(rho, k):
                # complexity O(N) per shift mutate  
                new_rho = rho.copy()
                if i+k < len(h): new_rho[k] -= (1 if h[i+k] == 1 else 0)
                if j+k < len(h): new_rho[k] += (1 if h[j+k] == 1 else 0)
                return new_rho
            
            for k in range(1, len(h)):
                if i+k < len(h) and j+k < len(h):
                    rho = update_rho_update(rho, k)
                    
            # Update worst-case sparsity  
            t_new = np.max(n_points/2 - rho)
            if t_new <= t_worst:
                t_worst = t_new
            else:
                # Accept randomly if better none
                if np.random.rand() < 0.1:
                    # Revert the previous change for reasons
                    h[i] = 1
                    h[j] = 0
        else:
            # do random swap for diversity - even good is not found 
            i, j = np.random.choice(len(h), size=2, replace=False)
            # Swap
            h[i], h[j] = h[j], h[i]
            
            # Update the most affected k
            k = min(len(h)-i-1, len(h)-j-1, i, j) if len(h) > 0 and i != j else 1
            if k > 0:
                # Recursive now for exactness
                rho = get_alignments(h)
                t_new = np.max(n_points/2 - rho)
                t_worst = t_new
        
        if t_worst == 0: 
            break
    
    return h

def differential_evolution(h_start, n_points, max_iter=5000):
    """Perform Differential Evolution optimization for continuous variables"""
    pop_size = 30
    mutation_factor = 0.8
    crossover_rate = 0.9
    
    def mutate_and_crossover(a, b, c):
        mut = a + mutation_factor * (b - c)
        mut = np.clip(mut, 0, 1)
        trial = np.where(np.random.rand(len(mut)) < crossover_rate, mut, a)
        return trial
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = h_start + np.random.normal(0, 0.02, len(h_start))
        h = project_to_sum_box(h, n_points / 2)
        population.append(h)
    
    # Compute initial fitnesses
    fitnesses = [evaluate_c5(h) for h in population]
    
    bottom_k = 5
    for _ in range(max_iter):
        # Selection without replacement
        selected_indices = np.argsort(fitnesses)[:bottom_k]
        parents = [population[i] for i in selected_indices]
        
        # Evolution - two children per parent pair
        new_population = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                a = parents[i]
                b = parents[(i + 1) % len(parents)]
                c = parents[np.random.randint(len(parents))]
                trial = mutate_and_crossover(a, b, c)
                trial = project_to_sum_box(trial, n_points / 2)
                new_population.append(trial)
                new_population.append(mutate_and_crossover(trial, a, b))
        
        # Evaluation
        new_fitnesses = [evaluate_c5(h) for h in new_population]
        
        # Replace worst performers
        all_pop = population + new_population
        all_fit = fitnesses + new_fitnesses
        combined_sorted = sorted(zip(all_pop, all_fit), key=lambda x: x[1])
        population = [x[0] for x in combined_sorted[:pop_size]]
        fitnesses = [x[1] for x in combined_sorted[:pop_size]]
    
    return population[0]

def simulated_annealing(h_point, n_points, max_iter=5000):
    """Perform simulated annealing for polishing"""
    T = 0.1
    alpha = 0.99
    epsilon = 0.05
    
    def temperature_schedule(t):
        return T * (alpha ** (t // 100))
    
    h_cur = h_point.copy()
    c5_cur = evaluate_c5(h_cur)
    
    for i in range(max_iter):
        # Pick two indices
        idx1, idx2 = np.random.choice(len(h_cur), size=2, replace=False)
        h_new = h_cur.copy()
        
        # Transfer a small amount
        delta = np.random.uniform(-epsilon, epsilon)
        h_new[idx1] += delta  
        h_new[idx2] -= delta
        h_new = np.clip(h_new, 0, 1) 
        
        # Preserve sum
        h_new = project_to_sum_box(h_new, n_points / 2) 
        
        c5_new = evaluate_c5(h_new)
        delta_c5 = c5_new - c5_cur
        if delta_c5 <= 0 or np.random.rand() < np.exp(-delta_c5 / temperature_schedule(i)):
            h_cur = h_new
            c5_cur = c5_new
    
    return h_cur

def hill_climb(h_point, n_points, max_iter=2000):
    """Perform a deterministic hill climbing"""
    h_cur = h_point.copy()
    c5_cur = evaluate_c5(h_cur)
    accepted_improvements = 0
    
    for i in range(max_iter):
        # Select subset of indices
        indices = np.random.choice(len(h_cur), size=max(1, len(h_cur)//20), replace=False)
        h_new = h_cur.copy()
        
        # Perturb selected indices
        for i in indices:
            sigma = 0.01 * (h_cur[i] * (1 - h_cur[i]))
            if sigma > 0:
                delta = np.random.normal(0, sigma)
                h_new[i] += delta
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, n_points / 2)
        
        c5_new = evaluate_c5(h_new)
        if c5_new < c5_cur:
            h_cur = h_new
            c5_cur = c5_new
            accepted_improvements += 1
        elif np.random.rand() < 0.01:  # occasional random jump
            h_cur = h_new
            c5_cur = c5_new
    
    return h_cur

def run(seed=42, budget_s=600, **kwargs):
    """Run optimization pipeline"""
    global TARGET_SUM, n_points, dx
    
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Phase A: Solve binary MILP
    x_opt, dx_small = solve_binary_milp()
    h0 = upsample_binary_vector(x_opt, n_points, dx_small)
    h0 = project_to_sum_box(h0, TARGET_SUM)
    c5_initial = evaluate_c5(h0)
    
    # Phase C: Optional swap search (may not preserve quality of nutsack if not started properly)
    if time.time() - start_time < budget_s * 0.7:
        try:
            h_after_swap = swap_search_worst_shift(h0, n_points, max_iter=5000)
            c5_swap = evaluate_c5(h_after_swap)
            if c5_swap < c5_initial:
                h0 = h_after_swap
                c5_initial = c5_swap
        except:
            pass  # Ignore if swap fails
    
    # Phase D: Differential Evolution
    if time.time() - start_time < budget_s * 0.9:
        try:
            h_de = differential_evolution(h0, n_points, max_iter=2000)
            c5_de = evaluate_c5(h_de)
            if c5_de < c5_initial:
                h0 = h_de
                c5_initial = c5_de
        except:
            pass
    
    # Phase E: Simulated Annealing
    if time.time() - start_time < budget_s * 0.95:
        try:
            h_sa = simulated_annealing(h0, n_points, max_iter=2000)
            c5_sa = evaluate_c5(h_sa)
            if c5_sa < c5_initial:
                h0 = h_sa
                c5_initial = c5_sa
        except:
            pass
    
    # Phase F: Hill Climbing
    try:
        h_final = hill_climb(h0, n_points, max_iter=1000)
        c5_final = evaluate_c5(h_final)
        if c5_final < c5_initial:
            h0 = h_final
            c5_initial = c5_final
    except:
        pass
    
    return (h0.tolist(), float(c5_initial), n_points)
