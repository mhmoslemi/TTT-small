def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr) * (2.0 / len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_modular_sequence(a, b, n):
    """Generate sequence using modular arithmetic"""
    sequence = np.array([(a * i + b) % n < n/2 for i in range(n)])
    return sequence.astype(float)

def compute_soft_threshold_sequence(a, b, t, delta, n):
    """Generate soft threshold sequence"""
    sequence = np.array([(t - ((a * i + b) % n)) / delta for i in range(n)])
    sequence = np.clip(sequence, 0, 1)
    return sequence
