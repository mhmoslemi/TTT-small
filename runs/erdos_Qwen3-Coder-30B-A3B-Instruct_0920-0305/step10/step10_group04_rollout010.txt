import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_c5(h):
    """Fast evaluation of C5 using FFT correlation"""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Start with a random binary pattern with exactly half ones
    h0 = np.zeros(n_points)
    ones_indices = np.random.choice(n_points, size=int(n_points/2), replace=False)
    h0[ones_indices] = 1.0
    
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    # Simulated Annealing with targeted swaps
    max_iterations = 500000
    temperature_start = 0.02
    temperature_end = 1e-5
    
    # Track progress
    last_improvement = 0
    stagnant_count = 0
    stagnant_limit = 5000
    
    current_h = h0.copy()
    current_c5 = best_c5
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Update temperature
        temperature = temperature_start * (1 - iteration / max_iterations) + temperature_end
        
        # Choose to do targeted swap (50% chance) or random swap
        if np.random.random() < 0.5:
            # Random swap
            i, j = np.random.randint(0, n_points, 2)
            while current_h[i] == current_h[j]:
                i, j = np.random.randint(0, n_points, 2)
        else:
            # Targeted swap based on worst shift
            corr = correlate(current_h, 1 - current_h, mode="full")
            max_idx = np.argmax(corr)
            d = max_idx - (n_points - 1)
            
            # Find candidates for swapping
            candidates_i = []
            candidates_j = []
            
            for i in range(n_points):
                j = i + d
                if 0 <= j < n_points:
                    if current_h[i] == 1 and current_h[j] == 0:
                        candidates_i.append(i)
                    elif current_h[i] == 0 and current_h[j] == 1:
                        candidates_j.append(i)
            
            if candidates_i and candidates_j:
                i = np.random.choice(candidates_i)
                j = np.random.choice(candidates_j)
            else:
                # Fall back to random swap if no targeted candidates
                i, j = np.random.randint(0, n_points, 2)
                while current_h[i] == current_h[j]:
                    i, j = np.random.randint(0, n_points, 2)
        
        # Perform swap
        current_h[i], current_h[j] = current_h[j], current_h[i]
        
        # Evaluate new C5
        new_c5 = compute_c5(current_h)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.random() < np.exp(-(new_c5 - current_c5) / temperature):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
                last_improvement = iteration
        else:
            # Revert swap
            current_h[i], current_h[j] = current_h[j], current_h[i]
        
        # Check for stagnation
        if iteration - last_improvement > stagnant_limit:
            # Occasionally restart with a diverse pattern
            if np.random.random() < 0.1:
                h0 = np.zeros(n_points)
                ones_indices = np.random.choice(n_points, size=int(n_points/2), replace=False)
                h0[ones_indices] = 1.0
                current_h = h0.copy()
                current_c5 = compute_c5(current_h)
                last_improvement = iteration
    
    # Final refinement with SLSQP
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
        res = minimize(
            compute_c5, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            refined_h = res.x
            refined_h = project_to_sum_box(refined_h, TARGET_SUM)
            refined_c5 = compute_c5(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    # Ensure the final result has the correct sum
    best_h = project_to_sum_box(best_h, TARGET_SUM)
    best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
