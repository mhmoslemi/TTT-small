import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_autocorr_fft(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def min_autocorr_from_corr(corr, n):
    """Convert correlation result to minimum autocorrelation value"""
    return dx * np.min(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def evaluate_min_autocorr(h):
    """Evaluate minimum autocorrelation"""
    corr = compute_autocorr_fft(h)
    return min_autocorr_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def find_optimal_piecewise_shape():
    """Find optimal piecewise constant shape by scanning parameter a"""
    best_c5 = float('inf')
    best_h = None
    n_scan = 500
    
    # Sweep over parameter 'a' from 0.05 to 0.4
    for i in range(n_scan):
        a = 0.05 + i * (0.35 / (n_scan - 1))
        if a >= 0.5:
            continue
            
        # Compute corresponding p
        p = (1 - 2 * a) / (2 * (1 - a))
        if p < 0 or p > 1:
            continue
            
        # Create piecewise constant function
        h_candidate = np.zeros(n_points)
        
        # Left edge
        left_edge_len = int(a * n_points / 2)
        if left_edge_len > 0:
            h_candidate[:left_edge_len] = 1
            
        # Right edge  
        right_edge_len = int(a * n_points / 2)
        if right_edge_len > 0:
            h_candidate[-right_edge_len:] = 1
            
        # Middle region
        middle_len = n_points - 2 * left_edge_len
        if middle_len > 0:
            h_candidate[left_edge_len:left_edge_len + middle_len] = p
            
        # Normalize to sum = 1
        h_candidate = h_candidate * (1.0 / np.sum(h_candidate))
        
        # Evaluate C5
        c5 = evaluate_c5(h_candidate)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Find initial solution using piecewise shape
    best_h, best_c5 = find_optimal_piecewise_shape()
    
    # Try simulated annealing with this as starting point
    max_iter = int(200000)
    T_start = 0.001
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Run simulated annealing
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbation
        h_new = current_h.copy()
        i1 = np.random.randint(0, n_points)
        i2 = np.random.randint(0, n_points)
        
        # Choose small perturbation amount
        delta = np.random.normal(0, 0.01)
        
        # Apply perturbation
        if h_new[i1] - delta >= 0 and h_new[i2] + delta <= 1:
            h_new[i1] -= delta
            h_new[i2] += delta
            
            # Project to ensure sum constraint and bounds
            h_new = project_to_sum_box(h_new, n_points / 2)
            
            c5_new = evaluate_c5(h_new)
            
            # Accept or reject
            if c5_new < current_c5:
                current_h = h_new.copy()
                current_c5 = c5_new
            else:
                # Accept with probability
                prob = np.exp(- (c5_new - current_c5) / T_start)
                if np.random.random() < prob:
                    current_h = h_new.copy()
                    current_c5 = c5_new
                    
            # Update best solution
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Cool down temperature
        T_start = max(1e-8, T_start * 0.99995)
    
    # Perform local refinement with gradient-based method
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use SLSQP for local refinement
    try:
        res = minimize(
            objective, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_opt = res.x
        else:
            h_opt = best_h
        h_opt = project_to_sum_box(h_opt, n_points / 2)
        c5_opt = evaluate_c5(h_opt)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            
    except Exception:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
