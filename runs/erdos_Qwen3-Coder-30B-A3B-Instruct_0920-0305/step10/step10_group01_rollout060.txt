def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h, L) * np.conj(np.fft.fft(b, L))).real
    return corr

def c5_from_corr(corr, n):

    return dx * np.max(corr)

def evaluate_c5(h):

    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_smooth_max(c, alpha):
    """Compute smooth maximum of correlation values with temperature alpha"""
    c_max = np.max(c)
    exp_term = np.exp(alpha * (c - c_max))
    sum_exp = np.sum(exp_term)
    return c_max + np.log(sum_exp) / alpha

    r = np.fft.ifft(fw_sym * np.conj(fh)).real
