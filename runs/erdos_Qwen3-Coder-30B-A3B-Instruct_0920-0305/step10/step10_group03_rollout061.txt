def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        c5_plus = evaluate_c5(h_plus)
        grad[i] = (c5_plus - base_c5) / eps
    
    return grad

    v = np.zeros_like(h0)
    h = h0.copy()
    
    for t in range(1, max_iter + 1):
        # Compute gradient
        grad = evaluate_gradient(h)
        
        # Update Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        
        # Bias correction
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Update parameters
        h = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project to valid constraints
        h = project_to_sum_box(h, TARGET_SUM)

        if t % 100 == 0:
            current_c5 = evaluate_c5(h)
            if current_c5 < 0.3795:  # Early stopping if we're close to target
                break
    
    return h

def simulated_annealing(h0, max_iter=10000, initial_temp=0.1, cooling_rate=0.95):

    current_h = h0.copy()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    temp = initial_temp
    
    for i in range(max_iter):
        # Create neighbor solution
        h_new = current_h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.02)
        h_new[idx] += delta
        
        # Project to valid constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject
        if c5_new < current_c5 or np.random.random() < np.exp(-(c5_new - current_c5) / temp):

            current_c5 = c5_new
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
        
        # Cool down temperature
        temp *= cooling_rate
        
        # Early stopping
        if best_c5 < 0.3795:
            break
    
    return best_h, best_c5
