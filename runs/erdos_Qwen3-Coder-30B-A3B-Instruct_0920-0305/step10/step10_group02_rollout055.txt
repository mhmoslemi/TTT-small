import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from math import gcd

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_modular_sequence(a, b, n):
    """Generate sequence using modular arithmetic"""
    h = np.zeros(n)
    for i in range(n):
        if ((i * a + b) % n) < n / 2:
            h[i] = 1
    return h

def find_best_modular_candidate(n_points):
    """Find best candidate from modular sequences"""
    best_c5 = float('inf')
    best_h = None
    
    # Find all values coprime to n_points
    coprimes = [i for i in range(1, n_points) if gcd(i, n_points) == 1]
    
    for a in coprimes:
        for b in range(n_points):
            h = generate_modular_sequence(a, b, n_points)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5

def smooth_boundaries(h, window_size=3):
    """Smooth boundaries between 0 and 1"""
    h_smooth = h.copy()
    # Find boundary points
    boundaries = np.where(np.diff(h) != 0)[0]
    
    for boundary in boundaries:
        start_idx = max(0, boundary - window_size//2)
        end_idx = min(len(h), boundary + window_size//2 + 1)
        
        # Linear interpolation
        if h[boundary] == 0:  # From 1 to 0
            ratio = np.linspace(1, 0, end_idx - start_idx)
        else:  # From 0 to 1
            ratio = np.linspace(0, 1, end_idx - start_idx)
        
        h_smooth[start_idx:end_idx] = ratio
    
    return h_smooth

def gradient_descent_refinement(h, target_sum, max_iter=200, eta=0.01):
    """Refine solution using gradient descent"""
    h_current = h.copy()
    
    for iteration in range(max_iter):
        # Compute gradients
        corr = compute_corr_fft(h_current)
        max_corr_idx = np.argmax(corr)
        max_corr_val = corr[max_corr_idx]
        
        # Compute gradient of the maximum correlation
        grad = np.zeros_like(h_current)
        for i in range(len(h_current)):
            # Gradient contribution from the maximum overlap term
            if i + max_corr_idx < len(h_current):
                grad[i] += 1 - h_current[i + max_corr_idx]
            if i - max_corr_idx >= 0:
                grad[i] += 1 - h_current[i - max_corr_idx]
        
        # Apply gradient step
        h_current = h_current - eta * grad
        
        # Project back
        h_current = np.clip(h_current, 0, 1)
        h_current = project_to_sum_box(h_current, target_sum)
    
    return h_current

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Step 1: Find best modular sequence candidate
    best_h, best_c5 = find_best_modular_candidate(n_points)
    
    # Step 2: Local search improvement
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Try local swaps to improve
    for i in range(5000):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Swap two random positions
        idx1 = np.random.randint(0, n_points)
        idx2 = np.random.randint(0, n_points)
        
        h_new = current_h.copy()
        h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
        
        # Project to constraints
        h_new = project_to_sum_box(h_new, target_sum)
        
        c5_new = evaluate_c5(h_new)
        
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
    
    # Step 3: Apply smoothing
    h_smooth = smooth_boundaries(current_h, window_size=3)
    h_smooth = project_to_sum_box(h_smooth, target_sum)
    c5_smooth = evaluate_c5(h_smooth)
    
    # Step 4: Gradient descent refinement
    try:
        h_refined = gradient_descent_refinement(h_smooth, target_sum)
        c5_refined = evaluate_c5(h_refined)
        
        if c5_refined < c5_smooth:
            current_h = h_refined
            current_c5 = c5_refined
    except:
        pass
    
    # Final SLSQP optimization
    try:
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - target_sum
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        # Run with a limited number of iterations
        res = minimize(
            objective, 
            current_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        if res.success:
            current_h = res.x
        current_h = project_to_sum_box(current_h, target_sum)
        current_c5 = evaluate_c5(current_h)
    except:
        pass
    
    return (current_h.tolist(), float(current_c5), n_points)
