```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_correlation_delta(x, i, j, A):
    """
    Compute the delta vector for correlation A when swapping x[i] and x[j].
    """
    n = len(x)
    x_pad = np.concatenate([[0]*(n-1), x.tolist(), [0]*(n-1)])
    
    # Contributions to A_d for turning off bit i
    d_off = np.zeros(2*n-1)
    if 0 <= i < n:
        d_off += x_pad[i+n-1:i+3*n-2] + x_pad[i:i+2*n-1][::-1]
    
    # Contributions to A_d for turning on bit j  
    d_on = np.zeros(2*n-1)
    if 0 <= j < n:
        d_on += x_pad[j+n-1:j+3*n-2] + x_pad[j:j+2*n-1][::-1]
        
    return d_on - d_off

def update_overlap_from_correlation(A, x):
    """Compute overlap from correlation; i.e. max of (sum - A) for proper shifts."""
    n = len(x)
    cumulative_sum = np.concatenate([[0], np.cumsum(x)])
    # For each offset d, compute sum_{i=0}^{n-1-d} x_i
    # We compute this for all d in -n+1..n-1
    O = np.zeros(2*n - 1)
    
    # Sums for positive shifts: d = 0 to n-2
    for d in range(n-1):
        O[n-1+d] = cumulative_sum[n-d] - cumulative_sum[0]
        
    # Sums for negative shifts: d = -n+1 to -1  
    for d in range(1-n, 0):
        O[n-1+d] = cumulative_sum[n] - cumulative_sum[-d]
        
    # Subtract A to get overlap
    O = O - A
    return O

def evaluate_c5_binary(x):
    """Fast evaluation of C5 using correlations for binary vectors."""
    n = len(x)
    A = correlate(x, x, mode='full')
    O = update_overlap_from_correlation(A, x)
    return np.max(O) * (2.0 / n)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 240
    dx = 2.0 / n
    T0 = 2 * dx
    Tf = 1e-6 * dx
    n_accepts = 300000
    T_schedule = np.linspace(T0, Tf, n_accepts)
    
    # Initial binary solution (half 1s and half 0s)
    x = np.zeros(n, dtype=int)
    x[:n//2] = 1
    np.random.shuffle(x)
    
    # Compute initial correlation
    A = correlate(x, x, mode='full')
    O = update_overlap_from_correlation(A, x)
    c5_current = np.max(O) * dx
    c5_best = c5_current
    x_best = x.copy()
    
    accept_count = 0
    
    # Main SA loop
    for i in range(n_accepts):
        if time.time() - start_time > budget_s - 5:
            break
            
        T = T_schedule[i]
        
        # Select move type
        if np.random.rand() < 0.6:
            # Targeted move
            d_best = np.argmax(O)
            if d_best < n:  # Positive shift
                shift = d_best - (n - 1)
            else:  # Negative shift  
                shift = d_best - (n - 1)
                
            # Find candidates for swapping
            U = [i for i in range(n) if 0 <= i + shift < n and x[i] == 1 and x[i+shift] == 0]
            V = [j for j in range(n) if 0 <= j + shift < n and x[j] == 0 and x[j+shift] == 1]
            
            if U and V:
                i_swap = np.random.choice(U)
                j_swap = np.random.choice(V)
            else:
                # Fall back to random
                i_swap = np.random.randint(n)
                j_swap = np.random.randint(n)
                while x[i_swap] == x[j_swap]:
                    j_swap = np.random.randint(n)
        else:
            # Random swap
            i_swap = np.random.randint(n)
            j_swap = np.random.randint(n)
            while x[i_swap] == x[j_swap]:
                j_swap = np.random.randint(n)
    
        # Apply swap
        x[i_swap], x[j_swap] = x[j_swap], x[i_swap]
        
        # Update correlation incrementally
        delta_A = compute_correlation_delta(x, i_swap, j_swap, A)
        A += delta_A
        O = update_overlap_from_correlation(A, x)
        c5_new = np.max(O) * dx
        
        # Accept or reject
        if c5_new < c5_current or np.random.rand() < np.exp(-(c5_new - c5_current) / T):
            c5_current = c5_new
            if c5_current < c5_best:
                c5_best = c5_current
                x_best = x.copy()
            accept_count += 1
        else:
            # Revert swap
            x[i_swap], x[j_swap] = x[j_swap], x[i_swap]
    
    # Convert best binary solution to float
    h = x_best.astype(float)
    
    # Apply smoothing
    h_smoothed = np.convolve(h, np.ones(5)/5, mode='same')
    h = project_to_sum_box(h_smoothed, n / 2)
    
    # Refine with continuous optimization
    bounds = [(0, 1) for _ in range(n)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2}
    
    # First optimization
    try:
        res = minimize(
            lambda x: np.max(update_overlap_from_correlation(
                correlate(x, x, mode='full'), x
            )) * dx,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}
        )
        if res.success:
            h = project_to_sum_box(res.x, n / 2)
    
    except:
        pass
    
    # Second optimization with post-smoothing
    try:
        h_smoothed = np.convolve(h, np.ones(3)/3, mode='same')
        h = project_to_sum_box(h_smoothed, n / 2)
        res = minimize(
            lambda x: np.max(update_overlap_from_correlation(
                correlate(x, x, mode='full'), x
            )) * dx,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}
        )
        if res.success:
            h = project_to_sum_box(res.x, n / 2)
    except:
        pass
        
    # Final projection
    h = project_to_sum_box(h, n / 2)
    
    # Validation
    final_c5 = np.max(update_overlap_from_correlation(
        correlate(h, h, mode='full'), h
    )) * dx
    
    return (h.tolist(), float(final_c5), n)
```
