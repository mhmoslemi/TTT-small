import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_with_differential_evolution(h0, maxiter=100):
    """Use differential evolution for global optimization"""
    def objective(h):
        return evaluate_c5(h)
    
    # Define bounds [0,1] for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Project the initial solution to satisfy the sum constraint
    h0_projected = project_to_sum_box(h0, TARGET_SUM)
    
    try:
        res = differential_evolution(
            objective,

def local_search_improved(h0, iterations=10000):
    """Improved local search with multiple perturbation strategies"""
    current_h = h0.copy()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    for i in range(iterations):
        h_new = current_h.copy()
        
        # Strategy 1: Random small perturbations
        if np.random.random() < 0.7:
            indices = np.random.choice(n_points, size=min(15, n_points//10), replace=False)
            for idx in indices:
                step_size = 0.02 * (1.0 - 2.0 * abs(h_new[idx] - 0.5))
                delta = np.random.normal(0, step_size)
                h_new[idx] += delta
        else:
            # Strategy 2: Large perturbations for exploration
            indices = np.random.choice(n_points, size=min(5, n_points//20), replace=False)
            for idx in indices:
                delta = np.random.uniform(-0.1, 0.1)
                h_new[idx] += delta
        
        # Project to valid constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new)
        
        # Accept if better or occasionally accept worse
        if c5_new < current_c5 or np.random.random() < 0.03:
            current_h = h_new.copy()
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = current_h.copy()
                best_c5 = c5_new
        
        # Occasionally reinitialize
        if i % 500 == 0 and i > 0:
            current_h = best_h.copy()
    
    return best_h, best_c5

        h_refined, c5_refined = local_search_improved(h_random, 8000)
        strategies.append((h_refined, c5_refined))
    
    # Select best among all strategies
    for h_test, c5_test in strategies:
        if c5_test < best_c5:
            best_h = h_test
            best_c5 = c5_test
    
    # Final tuning with local search
    if time.time() - start_time < budget_s:
        best_h, best_c5 = local_search_improved(best_h, 10000)
    
    return (best_h.tolist(), float(best_c5), n_points)
