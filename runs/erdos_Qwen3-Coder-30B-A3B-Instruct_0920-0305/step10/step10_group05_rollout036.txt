import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.fft import fft, ifft
import time

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
half_n = n_points // 2

def project_to_sum_box(u, target_sum):
    """Water-filling projection onto box constraints with sum = target_sum"""
    def f(lam):
        projected = np.clip(u + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(u + (a + b)/2, 0, 1)
    return result

def autocorr_fft(h):
    """Compute full aperiodic autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(h, L)).real
    return corr

def exact_c5(u):
    """Given half-vector u, build full h and return exact C5"""
    h = np.concatenate([u, u[::-1]])
    A = autocorr_fft(h)
    return 1 - dx * np.min(A)

def lse_c5(u, beta):
    """Smooth surrogate for C5 using log-sum-exp"""
    h = np.concatenate([u, u[::-1]])
    A = autocorr_fft(h)
    return 1 - (dx / beta) * np.log(np.sum(np.exp(beta * A)))

def grad_lse_c5(u, beta):
    """Gradient of LSE surrogate"""
    h = np.concatenate([u, u[::-1]])
    A = autocorr_fft(h)
    w = np.exp(beta * A)
    w_sum = np.sum(w)
    
    # Compute gradient of the weighted sum of convolutions
    # First compute h_{i+k} + h_{i-k}, convolved with w
    h_rev = h[::-1]
    # Create expanded arrays for convolution
    h_ext = np.concatenate([h_rev, np.zeros(len(h)-1)])
    w_ext = np.concatenate([w, np.zeros(len(h)-1)])
    # The convolution leads to:
    # grad_i = -dx * sum_k w_k * (h_{i+k} + h_{i-k}) / w_sum
    # but h_{i+k} + h_{i-k} is effectively a convolution of w with the symmetric h
    # Shift h to get h_{i+k}, h_{i-k} contributions
    grad = np.zeros_like(u)
    for i in range(len(u)):
        left = h[i+1:] if i+1 < len(h) else np.zeros(0)
        right = h[:len(h)-i-1] if i+1 < len(h) else np.zeros(0)
        conv = np.sum(w * (np.concatenate([left, np.zeros(i)])) + np.concatenate([np.zeros(i), right]))
        grad[i] = -dx * conv / w_sum
    
    return grad

def parametric_to_halfvector(params, n):
    """Convert parametric representation to half-vector u"""
    # Format: [a, r, p] or [a, r] 
    # a = edge length, r = ramp width, p = middle height
    a = params[0]
    r = params[1] if len(params) > 1 else 0
    p = params[2] if len(params) > 2 else (1.0 - 2*a) / (2.0 * (1.0 - a))
    
    if r > 0.5 - a:
        r = 0.5 - a
    if r < 0:
        r = 0
    if p > 1:
        p = 1
    if p < 0:
        p = 0
        
    u = np.ones(n // 2)
    # Define the piecewise function with ramps
    mid_start = int(a * n // 2)
    mid_end = int((1 - a) * n // 2)
    ramp_end = mid_start + int(r * n // 2) if r > 0 else mid_start
    
    # Set the three regions
    mid_start = int(a * n // 2)
    mid_end = int((1 - a) * n // 2)
    if r > 0:
        ramp_start = mid_start - int(r * n // 2)
        ramp_end = mid_start + int(r * n // 2)
        # Ramp part
        for i in range(ramp_start, mid_start):
            u[i] = 1 + (p - 1) * (i - ramp_start) / (mid_start - ramp_start)
        # Constant part
        u[mid_start:mid_end] = p
        # Ramp part
        for i in range(mid_end, ramp_end):
            u[i] = 1 + (p - 1) * (1 - (i - mid_end) / (ramp_end - mid_end)) 
    else:
        u[mid_start:mid_end] = p
        
    # If there are few extra indices, fill them correctly with clipped values
    u[:mid_start] = 1.0 
    u[mid_end:] = 1.0
    
    return u

def run_de():
    """Run Differential Evolution on 2-D parameter space"""
    def objective(params):
        u = parametric_to_halfvector(params, half_n)
        c5 = exact_c5(u)
        return c5
    
    bounds = [(0.05, 0.45), (0.0, 0.3), (0.0, 1.0)]
    result = differential_evolution(objective, bounds, seed=42, maxiter=200, popsize=15)
    return result.x

def run_lbfgsb(u0, beta=200):
    """Project L-BFGS-B on the full half-vector"""
    def objective(u):
        return lse_c5(u, beta)
    
    def grad(u):
        return grad_lse_c5(u, beta)
    
    # Callback to track best solution
    best_u = [u0.copy()]
    def callback(u):
        u_proj = project_to_sum_box(u, half_n / 2)
        c5 = exact_c5(u_proj)
        if c5 < exact_c5(best_u[0]):
            best_u[0] = u_proj.copy()
    
    # Run optimization with projective bounds
    bounds = [(0, 1) for _ in range(half_n)]
    result = minimize(
        objective,
        u0,
        method='L-BFGS-B',
        jac=grad,
        bounds=bounds,
        callback=callback,
        options={'maxiter': 800, 'gtol': 1e-10}
    )
    
    # Apply final projection to maintain constraints
    u_final = project_to_sum_box(result.x, half_n / 2)
    return u_final

def run_sa(u_best, budget_s):
    """Simulated Annealing polishing"""
    start_time = time.time()
    temp = 5e-3
    alpha = 0.9997
    delta_max = 0.1
    max_accept_ratio = 0.4
    min_accept_ratio = 0.2
    
    # Evaluate exact C5 of initial
    best_c5 = exact_c5(u_best)
    best_u = u_best.copy()
    
    # Mini loop
    iter_count = 0
    accept_count = 0
    total_count = 0
    
    while time.time() - start_time < budget_s:
        # Perturb
        i = np.random.randint(0, half_n)
        j = np.random.randint(0, half_n)
        while i == j:
            j = np.random.randint(0, half_n)
            
        # Transfer mass
        delta = np.random.uniform(0, delta_max)
        if u_best[i] >= delta and u_best[j] <= 1 - delta:
            u_trial = u_best.copy()
            u_trial[i] -= delta
            u_trial[j] += delta
            # Project and compute exact C5
            u_trial = project_to_sum_box(u_trial, half_n / 2)
            c5_trial = exact_c5(u_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_u = u_trial.copy()
            else:
                # Accept with probability
                if np.random.rand() < np.exp(-(c5_trial - best_c5) / temp):
                    best_c5 = c5_trial
                    best_u = u_trial.copy()
                    accept_count += 1
            total_count += 1
            iter_count += 1
        else:
            iter_count += 1
    
        # Adjust temperature
        temp *= alpha
        if temp < 1e-6:
            break
            
        # Adjust delta_max to match acceptance ratio
        if total_count % 500 == 0 and total_count > 0:
            ratio = accept_count / total_count
            if ratio > max_accept_ratio:
                delta_max *= 0.9
            elif ratio < min_accept_ratio:
                delta_max *= 1.1
            accept_count = 0
            total_count = 0
            
    return best_u, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx, half_n
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: Local parametric sweep
    best_c5 = float('inf')
    best_u = None
    a_vals = np.linspace(0.05, 0.45, 400)
    for a in a_vals:
        u = parametric_to_halfvector([a], half_n)
        c5 = exact_c5(u)
        if c5 < best_c5:
            best_c5 = c5
            best_u = u.copy()
    
    # Phase 2: DE search
    try:
        de_params = run_de()
        u_de = parametric_to_halfvector(de_params, half_n)
        c5_de = exact_c5(u_de)
        if c5_de < best_c5:
            best_c5 = c5_de
            best_u = u_de.copy()
    except:
        pass
        
    # Phase 3: Upscale to larger grid and refine with gradient method
    fine_n = 240
    fine_u = np.interp(np.linspace(0, 1, fine_n // 2), np.linspace(0, 1, half_n), best_u)
    fine_u = project_to_sum_box(fine_u, fine_n / 4)
    
    # Refine with L-BFGS-B projected on full grid
    try:
        u_refined = run_lbfgsb(fine_u, beta=200)
        c5_refined = exact_c5(u_refined)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            # Convert back to half-vector of original size  
            best_u = u_refined[:half_n]  # Should already be fine for symmetry
    except:
        pass
    
    # Phase 4: Simulated Annealing
    try:
        sa_u, sa_c5 = run_sa(best_u.copy(), budget_s * 0.15)
        if sa_c5 < best_c5:
            best_c5 = sa_c5
            best_u = sa_u.copy()
    except:
        pass
    
    # Recover full h from symmetric u
    h = np.concatenate([best_u, best_u[::-1]])
    
    return (h.tolist(), float(best_c5), n_points)
