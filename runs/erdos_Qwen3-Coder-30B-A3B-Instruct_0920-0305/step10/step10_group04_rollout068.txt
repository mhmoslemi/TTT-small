import numpy as np

import time

# Global variables
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_overlap_from_binary(x):
    """
    Compute overlap integral from binary indicator vector x.
    """
    # Correlation of x with itself
    x_padded = np.pad(x, (n_points - 1, n_points - 1), mode='constant')
    A = correlate(x, x, mode='full')
    
    # Prefix sums to compute overlapping window sums efficiently
    prefix_x = np.concatenate([[0], np.cumsum(x)])
    
    # For each shift d, compute the overlap
    # O_d = (sum of x_i) - A_d
    overlap_values = []
    for d in range(1 - n_points, n_points):
        if d >= 0:
            # Left shift case: x_i compared to x_{i-d}
            if d < n_points:
                overlap = prefix_x[n_points] - A[n_points - 1 + d]
            else:
                overlap = 0
        else:
            # Right shift case: x_{i+d} compared to x_i
            if -d < n_points:
                overlap = prefix_x[n_points] - A[n_points - 1 + d]
            else:
                overlap = 0
        overlap_values.append(overlap)
    
    # Account for the actual shift length
    return dx * max(overlap_values)

def compute_prefix_sums(x):
    prefix = np.zeros(len(x) + 1)
    for i in range(len(x)):
        prefix[i + 1] = prefix[i] + x[i]
    return prefix

def get_window_sums(prefix_x, d):
    """
    Efficiently compute the sum of x_i for i in range(0, n - |d|) where we are shifting by d,
    for the case when a window of x is overlapping with x shifted by d.
    """
    if d >= 0:
        # Left shift: x_i vs x_{i+d} 
        if d == 0:
            return prefix_x[n_points]  # All ones
        else:
            # Sum of elements in x[0:(n-d)]
            return prefix_x[n_points - d]
    else:
        # Right shift: x_{i+d} vs x_i
        if d == 0:
            return prefix_x[n_points]
        else:
            # Sum of elements in x[(-d):n]
            return prefix_x[n_points] - prefix_x[-d]

def compute_overlaps_from_corr(A, prefix_x):
    """
    Compute all overlap values from correlation A
    """
    # The overlap for shift d is:
    # O_d = (sum elements) - A_d for appropriate d.
    # For simplicity, sum of elements = prefix_x[n_points],
    # and A_d is extracted from A array.
    
    # For n_points = 240, d from -239 to 239.
    # A is of length 2*n-1 = 479, A positions correspond to d = -(n-1) .. n-1
    overlap_values = []
    for d in range(-n_points + 1, n_points):
        idx = n_points - 1 + d
        A_d = A[idx]
        # Compute the sum of x_i where there is a shift
        # Use prefix sums for efficiency
        window_sum = get_window_sums(prefix_x, d)
        overlap = window_sum - A_d
        overlay_values.append(overlap)
        
    return dx * max(overlay_values)

def project_to_sum_box(x, target_sum):
    """
    Project vector x onto box constraints [0,1] with sum = target_sum
    using binary search on Lagrange multiplier
    """
    # Make it floating point to avoid integer overflow in correlation
    x = np.array(x, dtype=float)
    
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    # Anti-overlap bound search
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    for _ in range(50):
        m = (a + b) / 2.0
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    projected = np.clip(x + (a + b)/2.0, 0, 1)
    return projected

def binary_to_float_h(x):

def float_to_binary_h(h):

def update_correlation(A, x, i, j):
    """
    Incremental update of correlation A after swapping x[i] and x[j]
    """
    d_start = -n_points + 1
    d_end   = n_points - 1
    new_A = A.copy()
    
    # Change due to swapping x[i] and x[j]
    for d in range(d_start, d_end + 1):
        # Indices of the elements involved
        i1 = i + d
        j1 = j + d
        # If not within bounds, treat as 0
        if i1 >= 0 and i1 < n_points:
            x_i_d = x[i1]
        else:
            x_i_d = 0
            
        if j1 >= 0 and j1 < n_points:
            x_j_d = x[j1]
        else:
            x_j_d = 0
            
        # store: new_A[d] = old_A[d] - x[i]*x[i+d] + x[j]*x[j+d] - x[j]*x[j+d] + x[i]*x[i+d]
        # More carefully:
        # x[i] now becomes x[j]  
        # x[j] now becomes x[i]
        # x[i+d] and x[j+d] may have been involved
        
        if d < 0:
            # Negative shift case
            idx_i = n_points - 1 + d
            idx_j = n_points - 1 + d
            # val = -x[i] * x[i+d] + x[j] * x[j+d], but we are updating to be -x[j]*x[j+d]+x[i]*x[i+d]
            val = x[j] * x_j_d - x[i] * x_i_d
        elif d == 0:
            # Shift of zero
            idx_i = n_points - 1 + d
            idx_j = n_points - 1 + d
            val = 0  # No change in 0-shift
        else:
            # Positive shift case
            idx_i = n_points - 1 + d
            idx_j = n_points - 1 + d
            # val = -x[i]*x[i+d] + x[j]*x[j+d]
            val = x[j] * x_j_d - x[i] * x_i_d
        new_A[n_points - 1 + d] += val
    
    return new_A

def find_worst_shift(x):
    """
    Based on [0..n-1], find indices i,j that create worst overlap
    Return d^* such that x_i and x_{i+d} have max impact.
    """
    # Correlation
    A = correlate(x, x, mode='full')
    prefix_x = compute_prefix_sums(x)
    
    # Total number of entries
    max_overlap = -float('inf')
    worst_shift = 0
    
    for d in range(1 - n_points, n_points):
        idx = n_points - 1 + d
        A_d = A[idx]
        # Overlap = window_sum - A_d
        window_sum = get_window_sums(prefix_x, d)
        overlap = window_sum - A_d
        if overlap > max_overlap:
            max_overlap = overlap
            worst_shift = d
            
    return worst_shift
