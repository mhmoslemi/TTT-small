import numpy as np
from scipy.signal import correlate
import time

# Global parameters
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_correlation(x):
    """Compute full correlation of binary vector x"""
    return correlate(x, x, mode='full')

def compute_overlaps(A_full, x):
    """Compute overlap values O_d for all shifts d"""
    n = len(x)
    # prefix sum for efficient window sums
    prefix = np.concatenate([[0], np.cumsum(x)])
    # compute overlap for each shift
    overlaps = np.zeros(2*n - 1)
    for d in range(1-n, n):
        start_idx = max(0, d)
        end_idx = min(n, n+d)
        if d >= 0:
            one_count = prefix[end_idx] - prefix[start_idx]
        else:
            one_count = prefix[end_idx] - prefix[start_idx]
        # windows that overlap at shift d
        window_size = n - abs(d)
        # corrected for the fact that we're missing window of size |d|
        overlaps[d+n-1] = one_count - A_full[d+n-1]
    return overlaps

def compute_c5_binary(x):
    """Fast computation of C5 on binary vector using incremental formulas"""
    A = compute_correlation(x)
    overlaps = compute_overlaps(A, x)
    return dx * np.max(overlaps)

def project_to_sum_box(y, target_sum):
    """Project vector y onto [0, 1]^n with sum = target_sum"""
    def f(lam):
        projected = np.clip(y + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(y + (a + b) / 2, 0, 1)
    return result

def random_binary_initialization():
    """Generate random binary vector with exact sum constraint"""
    x = np.zeros(n_points, dtype=np.float64)
    ones_idx = np.random.choice(n_points, size=int(TARGET_SUM), replace=False)
    x[ones_idx] = 1.0
    return x

def evaluate_c5_float(h):
    """Evaluate C5 using FFT on continuous vector"""
    corr = correlate(h, 1-h, mode='full')
    return dx * np.max(corr)

def get_rand_index(x):
    """Find a random index where x[i] is 1 and x[j] is 0"""
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return None, None
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    return i, j

def targeted_swap(x, d_star):
    """Perform targeted swap based on worst shift"""
    uf = (np.arange(n_points) + d_star < n_points) & (np.arange(n_points) + d_star >= 0)
    x_d = x[np.arange(n_points)[uf]]
    x_d_rev = x[np.arange(n_points)[uf][::-1]]
    
    # positions where x[i]=1 and x[i+d_star]=0
    exposed = (x==1) & (x_d==0)
    
    # positions where x[j]=0 and x[j+d_star]=1
    missing = (x==0) & (x_d_rev==1) 
    
    exposed_indices = np.where(exposed)[0]
    missing_indices = np.where(missing)[0]
    
    if len(exposed_indices) == 0 or len(missing_indices) == 0:
        # fallback to random swap
        return get_rand_index(x)
    
    i = np.random.choice(exposed_indices)
    j = np.random.choice(missing_indices)
    return i, j

def get_max_shift(overlaps):
    """Find the worst shift index"""
    return np.argmax(overlaps) - (n_points - 1)

def perform_swap(x, i, j):
    """Perform the swap i <-> j"""
    x[i], x[j] = x[j], x[i]
    return x
