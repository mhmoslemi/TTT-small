import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_overlap_from_binary(x):
    """Compute overlap using correlation via FFT for binary vector."""
    # Ensure x is float
    x = np.array(x, dtype=float)
    # Correlate with reversed vector (for 1 - h(x + k)) effectively
    corr = correlate(x, 1 - x, mode='full')
    max_corr = np.max(corr)
    return dx * max_corr

def compute_overlap_from_binary_fast(x):
    """Fast vectorized computation using prefix sums."""
    x = np.array(x, dtype=float)
    
    # Compute the auto-correlation using FFT
    x_padded = np.pad(x, (n_points-1, n_points-1), mode='constant')
    corr = correlate(x_padded[n_points-1:-n_points+1], x, mode='full')
    max_corr = np.max(corr)
    return dx * max_corr

def project_to_sum_box(x, target_sum):
    """Project vector onto box constraints [0,1] with sum = target_sum"""
    # This is a simpler and more stable projection
    x = np.array(x, dtype=float)
    # Binary search for lambda to satisfy sum constraint
    def f_lambda(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Binary search bounds
    a, b = -1.0, 1.0
    while f_lambda(a) * f_lambda(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(100):
        m = (a + b) / 2
        fm = f_lambda(m)
        if abs(fm) < tol:
            break
        if fm * f_lambda(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT & full correlation"""
    h = np.array(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def make_binary_initial(n):
    """Make an initial binary vector"""
    x = np.zeros(n)
    x[:n//2] = 1
    np.random.shuffle(x)
    return x

def get_worst_shift_pairs(x, d_star):
    """Get set U and V of exposure points for a given shift"""
    U = []
    V = []
    n = len(x)
    if d_star > 0:
        # x[i] and x[i-d_star] for i >= d_star, x[i+d_star] for i < n-d_star
        for i in range(d_star, n):
            if x[i] == 1 and x[i-d_star] == 0:
                U.append(i)
        for j in range(n-d_star):
            if x[j] == 0 and x[j+d_star] == 1:
                V.append(j)
    else:
        d_star = abs(d_star)
        for i in range(d_star):
            if x[i] == 1 and x[i+d_star] == 0:
                U.append(i)
        for j in range(n-d_star):
            if x[j] == 0 and x[j+d_star] == 1:
                V.append(j)
    return U, V

def propose_swap(x, d_star):
    """Generate a swap that targets worst shift."""
    U, V = get_worst_shift_pairs(x, d_star)
    if U and V:
        i = np.random.choice(U)
        j = np.random.choice(V)
        return i, j
    else:
        # fallback to random swap
        i = np.random.choice(np.where(x == 1)[0])
        j = np.random.choice(np.where(x == 0)[0])
        return i, j

def compute_delta_corr(x, i, j, d_star):
    """Compute change in correlation"""
    n = len(x)
    delta_corr = np.zeros(2 * n - 1)
    # We compute delta contribution to A_d for all d, but only relevant ones
    # Numerical issue: if certain d values are out of bounds, will zero out.
    # But we can precompute indices for valid d for i and j and don't need full vector
    for d in range(-n+1, n):
        if 0 <= i+d < n and 0 <= j+d < n:
            delta_corr[d+n-1] = -x[i+d] - x[i-d] + x[j+d] + x[j-d]
        elif i+d < 0 and j+d >= 0:
            delta_corr[d+n-1] = 0 - x[i-d] + x[j+d] + x[j-d]
        elif i+d >= n and j+d < n:
            delta_corr[d+n-1] = -x[i+d] - x[i-d] + x[j+d] + x[j-d]
        elif i+d < 0 and j+d < 0:
            delta_corr[d+n-1] = 0 - x[i-d] + 0 + x[j-d]
        else:  # both d out of range except for the single ones
            delta_corr[d+n-1] = 0
    return delta_corr

def fem_binary_search(sum_val, target_sum, x):
    """Simple method to ensure desired sum"""
    # Too much complexity to require Newton/Secant, just adjust a bit
    diff = sum_val - target_sum
    if abs(diff) < 1e-10:
        return x
    old = x.copy()
    # simple iterative project on deltas
    num_flips = int(abs(diff))
    if num_flips > 0:
        if diff > 0:
            # need to turn 1s into 0s
            pos_indices = np.where(x == 1)[0]
            flip_indices = np.random.choice(pos_indices, size=min(num_flips, len(pos_indices)), replace=False)
            x[flip_indices] = 0
        else:
            # need to turn 0 into 1s
            neg_indices = np.where(x == 0)[0]
            flip_indices = np.random.choice(neg_indices, size=min(num_flips, len(neg_indices)), replace=False)
            x[flip_indices] = 1
    return x

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Step 1: Initialize
    # Start with a random binary vector
    h = make_binary_initial(n_points)
    h = project_to_sum_box(h, TARGET_SUM)
    best_h = h.copy()
    c5 = compute_overlap_from_binary(h)
    best_c5 = c5

    # Simulated Annealing Parameters
    T_initial = 0.01 * dx
    T_final = 1e-6 * dx
    N_anneal = int(300000)
    temp = T_initial
    delta_T = (T_initial - T_final) / N_anneal
    accepted = 0
    stagnation_counter = 0
    stale_best = c5

    # Check and update
    while accepted < N_anneal and (time.time() - start_time < budget_s - 5):
        # Compute current best shift (worst case k)
        corr = correlate(h, 1 - h, mode="full")
        # Get max shift causing maximum overlap (channel response)
        max_shift_idx = np.argmax(corr)
        # Get the actual shift in units of indices
        d_star = max_shift_idx - (len(h) - 1)

        # Propose a move
        if np.random.rand() < 0.6:
            i, j = propose_swap(h, d_star)
        else:
            i = np.random.choice(np.where(h == 1)[0])
            j = np.random.choice(np.where(h == 0)[0])

        # Apply the swap
        new_h = h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]

        # Renormalize to match the sum constraint
        new_h = project_to_sum_box(new_h, TARGET_SUM)
        new_c5 = compute_overlap_from_binary_fast(new_h)

        # Accept or reject
        if new_c5 < c5:
            h = new_h
            c5 = new_c5
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
            accepted += 1
        else:
            # Temporal acceptance probability
            p = np.exp(-(new_c5 - c5) / temp)
            if np.random.rand() < p:
                h = new_h
                c5 = new_c5
                accepted += 1

        # Update temperature
        temp = T_initial - delta_T * accepted

        # Track stagnation for diversity kicks
        if abs(c5 - stale_best) < 1e-5:
            stagnation_counter += 1
            if stagnation_counter >= 20000:
                # Diversity kick: Randomize a fraction of bits
                idxs = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                old_vals = h[idxs].copy()
                h[idxs] = np.random.choice([0.0, 1.0], size=len(idxs))
                h = project_to_sum_box(h, TARGET_SUM)
                c5 = compute_overlap_from_binary_fast(h)
                stale_best = c5
                stagnation_counter = 0
                temp *= 0.5
        else:
            stale_best = c5
            stagnation_counter = 0

    # Now refine with continuous optimization
    h_float = best_h.astype(float)
    h_float = project_to_sum_box(h_float, TARGET_SUM)
    # Use SLSQP to refine final continuous bounds
    try:
        from scipy.optimize import minimize
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
        res = minimize(
            evaluate_c5,
            h_float,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_float = res.x
            h_float = project_to_sum_box(h_float, TARGET_SUM)
            refined_c5 = evaluate_c5(h_float)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = h_float.copy()
    except Exception:
        pass

    # Final evaluation
    best_c5 = evaluate_c5(best_h)

    return (best_h.tolist(), float(best_c5), n_points)
