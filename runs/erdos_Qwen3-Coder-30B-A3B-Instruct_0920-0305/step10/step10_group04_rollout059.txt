import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points

def compute_overlap_from_corr(corr_full, n):
    """Convert correlation result to C5 value"""
    # corr_full contains correlation for all shifts d in range [-n+1, n-1]
    # We want to compute O(d) for d in [0, n-1]
    # O(d) = sum of x_i over the overlap window - A_d
    # where A_d = sum x_i * x_{i+d}
    
    # The actual overlap at shift d is very closely related to:
    # - sum of x_i where i+d is valid in [0,n-1] (call prefix sums for this)
    # - A_d (already computed)
    # But let's recompute accurately
    n2 = n-1
    # prefix sum to compute sum of x_i over [i, i+d] events
    prefix = np.concatenate([[0], np.cumsum(corr_full[n2:])])
    
    # Now compute overlaps O_d for d = 0 to 2n-2
    # But we are only interested in O_d for -n+1 to n-1 
    # We work with left shifts only for now
    # The overlap at shift d (may be negative) is
    # sum_{i=0}^{min(n-1-d,n-1)} x_i * (1 - x_{i+d})
    # = sum_{i=0}^{min(n-1-d,n-1)} x_i - sum_{i=0}^{min(n-1-d,n-1)} x_i * x_{i+d}
    # = sum of x_i over [0, min(n-1-d,n-1)] - A(d) 
    
    # But we also have the mirror cases
    # For right shifts, d > 0, we do:
    # sum_{i=max(0,d)}^{n-1} x_i * (1 - x_{i-d}) = sum_{i=max(0,d)}^{n-1} x_i - A(d)
    
    # In practice since we are computing full correlation, we know:
    # A_{-d} = sum_{i=0}^{n-1-d} x_i * x_{i+d} 
    # The overlap window for shift d is:
    #   indices that satisfy 0 <= i <= n-1 and 0 <= i+d <= n-1
    #   i can range from max(0, -d) to min(n-1, n-1-d)
    #   i can range from max(0, -d) to min(n-1, n-1-d) = min(n-1, n-1-d) = n-1 - max(0,d)  
    #   if d < 0, d=-k, k > 0, shift right. Index range is [0, n-1+k]
    #   if d > 0, shift left: index range is [d, n-1]
    # That makes the overlap calculation 
    # sum(x_{max(0,-d): min(n-1, n-1-d)}) - A_d
    # The key is that A_d = Corr(index n-1+d) element
    
    # But in order to actually compute it without re faffing too much,
    # let's create another folded approach
    A = corr_full[n-1:]
    total_ones = np.sum(corr_full[n-1:])  # this reflects total number of overlaps when computing correlation properly (sum of all ones times ones)
    # Actually let's build indices and compute the value properly
    # Efficient method: 
    # get prefix sum
    prefix = np.concatenate([[0], np.cumsum(corr_full[n-1:])])
    
    max_overlaps = []
    
    # Compute D = 0 to 2*n-2  
    for d in range(2*n-1):
        # If the shift d reflects a shift from 0 to 2*n-2 where d = n-1+i  or d = n-1-i  
        # The total shift index is (d - (n-1)) where d ranges from 0 to 2*(n-1) 
        shift = d - (n-1)
        if shift < 0:
            # Left shift - we want [0, n+shift-1] (i ranges up to n+shift-1)
            # That would be range from 0 to n + shift - 1 inclusive
            # So prefix[ n+shift ] - prefix[0]
            # OR from [s_i, s_f] where s_i = 0, s_f = n+shift-1 
            # Sum from prefix[s_f+1] - prefix[s_i] where s_i = 0, s_f = n+shift-1  => prefix[n+shift] - prefix[0]  
            # # This is wrong. Should be prefix[0..n+shift-1] = prefix[n+shift]
            # Actually prefix[i] means sum from index 0 to i-1  
            # sum from 0 to n+shift-1 = prefix[n+shift]  
            # But if shift is 0 then (i+d) goes up to n-1, so i only goes up to n-1
            # prefix[n+shift] when shift<0 is prefix[n+shift]  
            # But we must be careful about i+d == [0,n-1] intersection
            # i d is in [0, n-1], and i+d in [0,n-1], so:
            # 0 <= i <= n-1 and -d <= i <= n-1
            # max(0, -d) <= i <= n-1  
            # sum from range [max(0, -d), n-1]  
            # =  prefix[n] - prefix[max(0, -d)]
            if shift == 0:
                # This is special case - both range overlaps fully
                continue
            sum_x = prefix[n] - prefix[max(0, -shift)]
            # Overlap = sum over overlap region - A 
            if -shift >= 0 and -shift < n:
                A_d = corr_full[n-1-shift]
                overlap = sum_x - A_d
                max_overlaps.append(overlap)
            else:
                # might happen later but should not in valid range
                continue
        else:
            # Right shift - i+d in [0,n-1], but i in [0,n-1]
            # so max(0, d) <= i <= min(n-1, n-1-d)  
            # = max(0, d) <= i <= n-1-d  
            left = max(0, shift)
            right = n-1-shift
            if left <= right:
                sum_x = prefix[right+1] - prefix[left] 
                A_d = corr_full[n-1+shift]
                overlap = sum_x - A_d
                max_overlaps.append(overlap)
            else:
                # invalid range  
                continue  

    # return dx * max_overlaps
    return max_overlaps

def compute_overlap_exact(h):
    n = len(h)
    # Compute full correlation h(x) * (1 - h(x+k)) 
    # This is done to compute the max over full support  
    one_minus_h = 1 - h  
    # convolution full type: 
    full_cross = correlate(h, one_minus_h, mode='full')
    # The full result has length 2*n-1, with shift indices from -(n-1) to n-1  
    # To convert to overlap values at shift d, we can extract and compute them  
    # In O(n) for each of the shifts
    shifts = range(-(n-1), n)  # 2*n-1 shifts total
    
    # compute for each shift the total overlap
    overlaps = []
    for d in shifts:
        # cross correlate so that i maps to i+d from first array 
        # the contribution from the pair (i, i+d) is h[i]*(1-h[i+d])  
        # sum over all pairs in overlap region
        # when d=0 then i -> i+d always valid, ranges 0..n-1  
        # when d>0 then i -> i+d in [-d,n-1] range  
        # when d<0 then i -> i+d in [0, n-1+d] range 
        
        i_start = max(0, -d)
        i_end = min(n-1, n-1-d) if d > 0 else n-1 
        if i_start <= i_end:
            r = i_end-i_start+1
            overlap = np.sum(h[i_start:i_end+1] * one_minus_h[i_start+d:i_end+d+1]) 
            overlaps.append(overlap)
        else: 
            overlaps.append(0)
    return dx * np.max(overlaps)

def compute_fast_overlaps(h):
    """Calculate overlap efficiently using precomputed logic"""
    n = len(h) 
    # prefix sum of h
    prefix = np.concatenate([[0], np.cumsum(h)])
    
    # Now compute overlaps at each possible shift
    n2 = n-1
    shift_range = range(-n2, n2+1)  # this is 2*n-1 elements
    
    max_val = 0
    full_corr = correlate(h, h, 'full') # compute x * x convolution
    
    # to access i, (i+d) both valid in [0,n-1]:  
    # i in [0, n-1] && i+d in [0, n-1]
    # i in [0, n-1] && i in [-d, n-1-d]
    # i in [max(0, -d), min(n-1, n-1-d)]
    
    for d in shift_range:
        # get range of valid i
        left = max(0, -d)
        right = min(n-1, n-1-d) if d >= 0 else n-1
        if left <= right:
            # valid range for i
            
            # compute sum_{i} h[i] * (1 - h[i+d])  
            sum_both = np.sum(h[left:right+1] * (1 - h[left+d:right+1+d]))
            # but for our purposes compute max over the range of d
            if sum_both > max_val: 
                max_val = sum_both
    
    max_val = (dx * max_val)
    return max_val

def evaluate_c5_binary(x):
    """Fast evaluation of C5 (binary vector) using efficient arithmetic only"""
    n = len(x) 
    prefix_sum = np.concatenate([[0], np.cumsum(x)])  # prefixSum[i] = sum_{j< i} x_j
    
    # we compute correlations in linear time
    # corr[i] = sum_{j=0}^{n-1} x_j * x_{j+i}
    # but we'll compute in direct terms
    
    # For shift d, the (positive shift) region of overlap is
    # such that j and j+d are both in [0,n-1]
    # which is j in [0, n-1-d] and j+d in [0, n-1] 
    # so j in [0, n-1] and j in [0, n-1-d] AND j in [-d, n-1-d]
    # i.e. j in [0, min(n-1, n-1-d)] and j in [max(0, -d), min(n-1, n-1-d)]
    # which gives j in [max(0, -d), min(n-1, n-1-d)]
    
    # Given this for shift d
    # count of ones in window: prefix[n-1-d+1] - prefix[max(0,-d)] 
    #      = sum_{j= max(0,-d)}^{n-1-d} x_j = sum_{j= max(0,-d)}^{n-1-d} x_j  
    # correlation i, i+d: 
    # A(d) = sum_{j=max(0,-d)}^{min(n-1,n-1-d)} x_j * x_{j+d}  
    # overlap (to be minimized):  
    # overlaps sum_{j= max(0,-d)}^{min(n-1,n-1-d)} x_j - A(d) 
    
    max_overlap = 0
    full_correlation = correlate(x, x, 'full')
    
    for d in range(-n+1, n-1+1):
        # the overlapping region
        j_start = max(0, -d)  
        j_end = min(n-1, n-1-d) if d >= 0 else n-1
        if j_start <= j_end:  
            length = j_end - j_start + 1
            # again, first take sum of x_j 
            sum_ones_reg = prefix_sum[j_end+1] - prefix_sum[j_start]
            # and A_d (auto correlation at this shift)
            # index d in full_correlation points to corr [-n+1..n-1]  
            # So shift 0 -> index n-1, shift d -> index n-1+d  
            index = n-1 + d
            A_d = full_correlation[index] 
            # overlap = sum x_j - A_d which we want to minimize  
            overlap = sum_ones_reg - A_d
            if overlap > max_overlap:
                max_overlap = overlap
    return dx * max_overlap

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum
    Using binary search for Lagrange multiplier."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    result = np.clip(x + (a + b) / 2, 0, 1)
    return result

def binary_to_real(b):
    """Binary vector to real vector with same sum"""
    return b.astype(float)

def change_swap(x, i, j):
    """swap bit values at indices i and j in binary vector"""
    x = x.copy()  # avoid side effects
    tmp = x[i]
    x[i] = x[j]
    x[j] = tmp
    return x

def check_consistency(x, target):
    """Ensure that x has target number of 1s"""
    got = np.sum(x)
    if abs(got - target) > 1e-3:
        print(f'WARNING: Projected sum is {got}, expected {target}')
    return True
