TARGET_SUM = None

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_hill_climb(h, target_sum):
    """Local hill climbing approach"""
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    
    for _ in range(100):
        if np.random.random() < 0.7:  # 70% probability of multi-point moves
            # Try changing multiple points
            indices = np.random.choice(len(h), size=max(1, len(h)//20), replace=False)
            new_h = current_h.copy()
            step_size = 0.01
            for idx in indices:
                delta = np.random.choice([-step_size, step_size])
                new_h[idx] += delta
            new_h = project_to_sum_box(new_h, target_sum)
            new_c5 = evaluate_c5(new_h)
            if new_c5 < current_c5:
                current_h = new_h.copy()
                current_c5 = new_c5
        else:
            # Single point changes
            idx = np.random.randint(0, len(h))
            step_size = 0.01
            for delta in [-step_size, step_size]:
                new_h = current_h.copy()
                new_h[idx] += delta
                if 0 <= new_h[idx] <= 1:
                    new_h = project_to_sum_box(new_h, target_sum)
                    new_c5 = evaluate_c5(new_h)
                    if new_c5 < current_c5:
                        current_h = new_h.copy()
                        current_c5 = new_c5
                        break
    
    return current_h, current_c5
