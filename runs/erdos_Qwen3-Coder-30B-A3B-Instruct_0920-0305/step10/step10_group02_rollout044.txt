def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10

        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def get_coprimes(n):
    """Get all numbers coprime with n"""
    return [i for i in range(1, n) if gcd(i, n) == 1]

def generate_linear_congruential_sequence(a, b, n):
    """Generate sequence using linear congruential generator"""
    h = np.zeros(n)
    for i in range(n):
        pos = (a * i + b) % n
        if pos < n // 2:
            h[i] = 1
    return h

def smooth_step_function(h, window_size=3):
    """Apply smoothing to step function"""

    for boundary in boundaries:
        start = max(0, boundary - window_size//2)
        end = min(len(h), boundary + window_size//2 + 1)
        
        # Linear interpolation between adjacent values
        if start > 0 and end < len(h):
            val_before = h[start-1]
            val_after = h[end]
            x = np.linspace(0, 1, end-start)
            h_smooth[start:end] = val_before + x * (val_after - val_before)
    
    # Ensure sum constraint
    current_sum = np.sum(h_smooth)
    delta = (len(h_smooth) / 2) - current_sum
    if abs(delta) > 1e-10:
        h_smooth += delta / len(h_smooth)
    
    # Clip to [0,1]
    h_smooth = np.clip(h_smooth, 0, 1)
    return h_smooth
