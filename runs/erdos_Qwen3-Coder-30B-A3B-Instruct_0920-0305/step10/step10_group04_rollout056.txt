```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_auto_correlation_binary(x):
    """Compute auto-correlation of binary vector using FFT"""
    x_rev = x[::-1]
    L = 2 * len(x) - 1
    return np.fft.ifft(np.fft.fft(x_rev, L) * np.fft.fft(x, L)).real

def compute_overlap_from_auto_correlation(A, x):
    """Compute overlaps from auto-correlation"""
    n = len(x)
    # Compute cumulative sums for prefix sums
    cumsum_x = np.concatenate([[0], np.cumsum(x)])
    # O_d = sum(x_i) - A_d for shift d
    O = np.zeros(2*n - 1)
    for d in range(-(n-1), n):
        if d >= 0:
            # Positive shift: x_i contributes to O_d if x_{i+d} exists
            O[d + n - 1] = (cumsum_x[n - d] - cumsum_x[0]) - A[d + n - 1]
        else:
            # Negative shift: x_{i+d} contributes to O_d if x_i exists
            O[d + n - 1] = (cumsum_x[n] - cumsum_x[-d]) - A[d + n - 1]
    return O

def compute_c5_from_x(x):
    """Compute C5 from binary vector using FFT for auto correlation"""
    A = compute_auto_correlation_binary(x)
    O = compute_overlap_from_auto_correlation(A, x)
    return 2.0 / len(x) * np.max(O)

def compute_delta_A(x, i, j):
    """Compute delta A vector for swapping bits at positions i and j"""
    # Create padded version
    x_padded = np.concatenate([[0]*(len(x)-1), x.tolist(), [0]*(len(x)-1)])
    delta_A = np.zeros(2*len(x)-1)
    # Compute contribution of bit at i being turned off
    contrib_i = (x_padded[i + len(x)-1 : i + 3*len(x)-2] + 
                 x_padded[i : i + 2*len(x)-1][::-1])
    # Compute contribution of bit at j being turned on
    contrib_j = (x_padded[j + len(x)-1 : j + 3*len(x)-2] + 
                 x_padded[j : j + 2*len(x)-1][::-1])
    delta_A = -contrib_i + contrib_j
    return delta_A

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 240
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    # Initialize binary vector
    x = np.zeros(n_points, dtype=int)
    x[:n_points//2] = 1
    np.random.shuffle(x)
    
    # Compute initial correlation and C5
    A = compute_auto_correlation_binary(x)
    O = compute_overlap_from_auto_correlation(A, x)
    current_c5 = dx * np.max(O)
    best_c5 = current_c5
    best_x = x.copy()
    
    # Temperature settings
    T0 = 2 * dx
    Tf = 1e-6 * dx
    n_steps = 300000
    T = T0
    step = 0
    accept_count = 0
    stagnation_window = 20000
    last_improved = 0
    best_c5_history = []
    
    # Simulated annealing loop
    while step < n_steps and time.time() - start_time < budget_s * 0.7:
        if step % stagnation_window == 0 and step > 0:
            # Check for stagnation
            if len(best_c5_history) > 1 and best_c5_history[-1] - best_c5_history[0] < 1e-5:
                # Re-randomize part of the solution
                n_flip = max(1, int(n_points * 0.05))
                idxs = np.random.choice(n_points, size=n_flip, replace=False)
                x[idxs] = 1 - x[idxs]
                # Restore sum constraint
                ones_count = np.sum(x)
                if ones_count != TARGET_SUM:
                    # Find indices where we should flip back
                    to_flip = int(abs(ones_count - TARGET_SUM))
                    indices_off = np.where(x == 1)[0]
                    indices_on = np.where(x == 0)[0]
                    if len(indices_off) >= to_flip and len(indices_on) >= to_flip:
                        flip1 = np.random.choice(indices_off, size=to_flip, replace=False)
                        flip2 = np.random.choice(indices_on, size=to_flip, replace=False)
                        x[flip1] = 0
                        x[flip2] = 1
            best_c5_history = []
            
        # Choose move type
        if np.random.rand() < 0.6:
            # Targeted swap
            O = compute_overlap_from_auto_correlation(A, x)
            d_star = np.argmax(O) - (n_points - 1)
            
            # Find candidates U and V
            U = []
            V = []
            
            if d_star >= 0:
                # Positive shift case
                for i in range(len(x) - d_star):
                    if x[i] == 1 and x[i + d_star] == 0:
                        U.append(i)
                for j in range(d_star, len(x)):
                    if x[j] == 0 and x[j - d_star] == 1:
                        V.append(j)
            else:
                # Negative shift case
                d_abs = -d_star
                for i in range(d_abs, len(x)):
                    if x[i] == 1 and x[i - d_abs] == 0:
                        U.append(i)
                for j in range(len(x) - d_abs):
                    if x[j] == 0 and x[j + d_abs] == 1:
                        V.append(j)
            
            if U and V:
                i = np.random.choice(U)
                j = np.random.choice(V)
                # Compute delta A
                delta_A = compute_delta_A(x, i, j)
                A_new = A + delta_A
                
                # Update overlap and evaluate
                O_new = compute_overlap_from_auto_correlation(A_new, x)
                new_c5 = dx * np.max(O_new)
                
                if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                    # Accept move
                    A = A_new
                    current_c5 = new_c5
                    x[i], x[j] = x[j], x[i]
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_x = x.copy()
                    accept_count += 1
            else:
                # Fall back to random swap
                i, j = np.random.choice(n_points, size=2, replace=False)
                while x[i] == x[j]:
                    i, j = np.random.choice(n_points, size=2, replace=False)
                delta_A = compute_delta_A(x, i, j)
                A_new = A + delta_A
                O_new = compute_overlap_from_auto_correlation(A_new, x)
                new_c5 = dx * np.max(O_new)
                
                if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                    A = A_new
                    current_c5 = new_c5
                    x[i], x[j] = x[j], x[i]
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_x = x.copy()
                    accept_count += 1
        else:
            # Random swap
            i, j = np.random.choice(n_points, size=2, replace=False)
            while x[i] == x[j]:
                i, j = np.random.choice(n_points, size=2, replace=False)
            delta_A = compute_delta_A(x, i, j)
            A_new = A + delta_A
            O_new = compute_overlap_from_auto_correlation(A_new, x)
            new_c5 = dx * np.max(O_new)
            
            if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                A = A_new
                current_c5 = new_c5
                x[i], x[j] = x[j], x[i]
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_x = x.copy()
                accept_count += 1
                
        # Update temperature
        if accept_count > 0:
            T = T0 * (1 - accept_count / n_steps)
            T = max(T, Tf)
            
        step += 1
        if accept_count % 1000 == 0:
            best_c5_history.append(best_c5)
    
    # Convert best binary solution to continuous
    h = best_x.astype(float)
    
    # Apply smoothing before continuous refinement
    kernel = np.ones(5) / 5
    h_smooth = np.convolve(h, kernel, mode='same')
    h_smooth = project_to_sum_box(h_smooth, TARGET_SUM)
    
    # First SLSQP refinement
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
    
    try:
        res = minimize(
            lambda x: dx * np.max(np.correlate(x, 1-x, mode='full')),
            h_smooth,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False}
        )
        if res.success:
            h = res.x
    except:
        pass
    
    # Apply projection to maintain constraints
    h_projected = project_to_sum_box(h, TARGET_SUM)
    
    # Second SLSQP refinement
    try:
        res = minimize(
            lambda x: dx * np.max(np.correlate(x, 1-x, mode='full')),
            h_projected,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False}
        )
        if res.success:
            final_h = res.x
        else:
            final_h = h_projected
    except:
        final_h = h_projected
    
    final_h = project_to_sum_box(final_h, TARGET_SUM)
    final_c5 = dx * np.max(np.correlate(final_h, 1-final_h, mode='full'))
    
    # Evaluate final solution
    test_h = final_h.copy()
    test_c5 = dx * np.max(np.correlate(test_h, 1-test_h, mode='full'))
    
    return (final_h.tolist(), float(test_c5), n_points)
```
