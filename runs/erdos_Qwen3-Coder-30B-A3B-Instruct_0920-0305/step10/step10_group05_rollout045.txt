import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def autocorr_fft(h):
    """Fast autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(h, L)).real
    return corr

def evaluate_c5(h):
    """Evaluate C5 using FFT and find the min of auto-correlation"""
    corr = autocorr_fft(h)
    return 1.0 - np.min(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_symmetric_h(a, p, m):
    """Generate symmetric h vector based on edge size 'a' and middle height 'p'"""
    # Calculate the middle length (in terms of number of points)
    b = 2 - 2 * a
    if b <= 0:
        # Either a is too large or b is getting very small
        b = 0.001  # Prevent division by zero
        p = 0.0
        
    num_points = m * 2
    dx = 2.0 / num_points

    # Create indices for the left half
    indices = np.arange(m)
    h_half = np.zeros(m)
    
    # Left boundary
    left_edge_points = int(np.ceil(a / dx))
    h_half[:left_edge_points] = 1.0
    
    # Middle part
    if left_edge_points < m:
        middle_start = left_edge_points
        middle_end = min(middle_start + int(np.floor(b / dx)), m)
        h_half[middle_start:middle_end] = p

    # Right boundary (mirror)
    h_half = np.concatenate([h_half, h_half[::-1]])
    return h_half

def smooth_objective(h, beta=100):
    """Smooth approximation to 1 - max(auto_corr)"""
    corr = autocorr_fft(h)
    # Shift to center the correlation (remove the offset from first element to real data)
    corr = np.concatenate([corr[-(len(corr)//2 + 1):], corr[:len(corr)//2]])
    # to avoid numerical overflow
    corr_shifted = corr - np.max(corr)
    exp_corr = np.exp(beta * corr_shifted)
    return 1.0 - np.log(np.sum(exp_corr)) / beta

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    m = n_points // 2
    target_sum = 0.5 * n_points * dx  # Since int h dx = 1
    assert abs(target_sum - 0.5) < 1e-10, "Just dot product convenience check"

    # Step 1: Parametric scan of simple symmetric form
    best_local_c5 = float('inf')
    best_local_h = None
    best_local_a = 0.0
    
    # Sweep through approximate parametric form (analytic edge height)
    sweep_steps = 200  # e.g., a from 0.001 to 0.499
    a_list = np.linspace(0.001, 0.499, sweep_steps)
    
    for a in a_list:
        # Compute p from the area constraint
        p = (1.0 - 2.0*a) / (2*(1.0 - a))
        if p < 0:
            continue
        h_half_test = generate_symmetric_h(a, p, m)
        # Ensure constraints are met
        h_test = project_to_sum_box(h_half_test, target_sum)
        c5 = evaluate_c5(h_test)
        if c5 < best_local_c5:
            best_local_c5 = c5
            best_local_h = h_test.copy()
            best_local_a = a
    
    print(f"Parametric sweep best found at a={best_local_a}, C5={best_local_c5}")
    
    # Step 2: Low-dimensional local refinement on four parameters
    def objective(params):
        a, b, p1, p2 = params
        edge = max(0.001, min(0.499, a))
        middle_len = max(0.001, min(2.0, b))
        p_left = max(0.0, min(1.0, p1))
        p_right = max(0.0, min(1.0, p2))
        height = (1.0 - 2*edge - middle_len * (p_left + p_right) / 2) / middle_len

        npoints_left = int(np.round(edge / dx))  # front edge
        npoints_middle = int(np.round(middle_len / dx))  # middle block
        npoints_right = int(np.round(edge / dx))
        
        h_half = np.ones(m)
        h_half[:npoints_left] = 1.0
        h_half[-npoints_right:] = 1.0
        #核心区构造尝试改善
        
        if npoints_middle > 0:
            h_half[npoints_left : npoints_left + npoints_middle] = height
            
        h_half = project_to_sum_box(h_half, target_sum)
        h_full = np.concatenate((h_half, h_half[::-1]))
        return evaluate_c5(h_full)
    
    # try a few combinations of local search
    try:
        x0 = [0.15, 0.5, 0.7, 0.7]  # Initial guess
        bounds = [(0.001, 0.499), (0.001, 2.0), (0.0, 1.0), (0.0, 1.0)]
        res = minimize(objective, x0, bounds=bounds, method='L-BFGS-B', options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12})
        if res.success:
            if res.fun < best_local_c5:
                best_local_c5 = res.fun
                # Reconstruct full h using res.x
                a, b, p1, p2 = res.x
                h_half = np.ones(m)
                npoints_left = int(np.round(a / dx))
                npoints_middle = int(np.round(b / dx))
                height = (1.0 - 2*a - b * (p1 + p2) / 2) / b
                
                h_half[:npoints_left] = 1.0
                h_half[-npoints_left:] = 1.0
                if npoints_middle > 0:
                    h_half[npoints_left : npoints_left + npoints_middle] = height
                h_half = project_to_sum_box(h_half, target_sum)
                best_local_h = np.concatenate((h_half, h_half[::-1]))
    except Exception as e:
        print("L-BFGS failed due to:", str(e))

    # Step 3: Simulated Annealing 
    if time.time() - start_time > budget_s * 0.7:
        return (best_local_h.tolist(), float(best_local_c5), n_points)

    best_SA_h = best_local_h
    best_SA_c5 = best_local_c5
    current_h = best_local_h.copy()
    current_c5 = best_local_c5
    
    # Simulated Annealing parameters
    T0 = 0.01
    alpha = 0.9995
    T = T0
    temp_history = []
    
    # Track step size adaptation
    step_size = 0.05
    accepted_count = 0
    total_count = 0
    
    max_iter = 150000  # effectively controls total runtime

    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Make a random move
        # Randomly select two indices
        i1, i2 = np.random.randint(0, m), np.random.randint(0, m)
        if i1 == i2:
            i2 = (i1 + 1) % m

        # Transfer amount
        delta = min(step_size, 0.5)  # Limit max movement per step
        # Preserve box constraints: ensure we don't take over
        if current_h[i1] >= delta and current_h[i2] <= 1.0 - delta:
            current_h[i1] -= delta
            current_h[i2] += delta
            
            new_c5 = evaluate_c5(current_h)  # round-trip through full vector
            
            # Energy change
            delta_E = new_c5 - current_c5
            
            # Accept or reject using metropolis criterion
            if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
                current_c5 = new_c5
                if current_c5 < best_SA_c5:
                    best_SA_c5 = current_c5
                    best_SA_h = current_h.copy()
                
                accepted_count += 1
                    
            else:
                # rollback the move
                current_h[i1] += delta
                current_h[i2] -= delta

            total_count += 1
        else:
            # Not valid move, skip
            pass

        # Update T
        T = T0 * (alpha ** i)

        # Maintain step size adaptive to acceptance rate
        if i % 500 == 0 and i > 0:
            if total_count > 0:
                acc_rate = accepted_count / total_count
                if acc_rate > 0.4:
                    step_size *= 1.1
                elif acc_rate < 0.1:
                    step_size *= 0.9
                step_size = np.clip(step_size, 1e-4, 0.2)
                accepted_count = 0
                total_count = 0

    # Step 4: Final refinement via L-BFGS in one full iterate
    if time.time() - start_time > budget_s * 0.9:
        # Return best so far
        return (best_SA_h.tolist(), float(best_SA_c5), n_points)

    # Reconstruct full from half (assumed symmetry property preserved)
    x0 = best_SA_h
    bounds = [(0.0, 1.0)] * len(x0)
    
    def smooth_obj(h):
        return smooth_objective(h, beta=200)
    
    def smooth_grad(h):
        return smooth_gradient(h, beta=200)
        
    res = minimize(
        smooth_obj,
        x0,
        method='L-BFGS-B',
        jac=smooth_grad,
        bounds=bounds,
        options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
    )
    
    if res.success:
        final_h = res.x
        final_c5 = evaluate_c5(final_h)
        
        if final_c5 < best_SA_c5:
            best_SA_c5 = final_c5
            best_SA_h = final_h

    print(f"Final result C5 = {best_SA_c5:.5f}")
    return (best_SA_h.tolist(), float(best_SA_c5), n_points)
