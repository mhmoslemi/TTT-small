import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

# Global constants
n_points = 200
dx = 2.0 / n_points
half_size = n_points // 2

def autocorr_fft(h):
    """Compute autocorrelation via FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(h, L)).real
    return corr

def c5_from_autocorr(corr):
    """Compute C5 from autocorrelation"""
    return 1.0 - dx * np.min(corr)

def smooth_c5_objective(h, beta=50.0):
    """Smooth approximation to C5 for gradient computation"""
    corr = autocorr_fft(h)
    max_corr = np.max(corr)
    # Subtract max to prevent overflow
    exp_terms = np.exp(beta * (1.0 - (corr - max_corr)))
    sum_exp = np.sum(exp_terms)
    return 1.0 - (dx / beta) * np.log(sum_exp) - dx * (1.0 - max_corr)

def smooth_c5_gradient(h, beta=50.0):
    """Gradient of smooth C5"""
    corr = autocorr_fft(h)
    max_corr = np.max(corr)
    exp_terms = np.exp(beta * (1.0 - (corr - max_corr)))
    sum_exp = np.sum(exp_terms)
    
    # Gradient of G = -dx * log(sum(exp(beta*(1-A_k)))) / beta
    # dG/dh_i = -dx * sum_k(exp(beta*(1-A_k)) * dA_k/dh_i ) / (beta*sum_exp)
    # where dA_k/dh_i = h_{i+k} + h_{i-k}
    w = exp_terms / sum_exp
    w_rev = w[::-1]
    
    # Compute convolution of weight vector with the original vector
    L = 2 * len(h) - 1
    grad = -dx * ifft(fft(w_rev, L) * fft(h, L)).real
    return grad / beta

def project_to_sum_box(h, target_sum):
    """Project h onto box [0,1] with exact sum = target_sum"""
    # Use water-filling approach to modify values to satisfy sum exactly
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    elif current_sum < target_sum:
        # Need to increase values
        idx = np.argsort(h)
        remaining = target_sum - current_sum
        for i in idx:
            if remaining <= 0:
                break
            increase = min(1.0 - h[i], remaining)
            h[i] += increase
            remaining -= increase
    else:
        # Need to decrease values
        idx = np.argsort(h)[::-1]
        remaining = current_sum - target_sum
        for i in idx:
            if remaining <= 0:
                break
            decrease = min(h[i], remaining)
            h[i] -= decrease
            remaining -= decrease
            
    return h

def parametric_h(a):
    """Create symmetric piecewise constant h based on parameter a"""
    p = (1.0 - 2 * a) / (2.0 * (1.0 - a))
    h = np.full(half_size, p)
    
    # Fill left edge
    left_len = int(a * half_size)
    h[:left_len] = 1.0
    
    # Fill right edge
    right_len = int(a * half_size)
    h[-right_len:] = 1.0
    
    # Handle odd case – ensure sum matches exactly
    h = project_to_sum_box(h, target_sum=n_points / 4)
    return h

def sweep_find_initial():
    """Sweep parameter space to find the best 1D parametric solution"""
    best_c5 = float('inf')
    best_h = None
    best_a = None
    
    # Try different edge lengths from a = 0.01 to a = 0.49 (sweeps 249 values)
    for a in np.linspace(0.01, 0.49, 249):
        h = parametric_h(a)
        c5 = c5_from_autocorr(autocorr_fft(h))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_a = a
            
    return best_h, best_c5, best_a

def refine_with_lbfgsb(h_start):
    """Use L-BFGS-B on half vector to improve parametric solution"""
    def objective(h):
        return smooth_c5_objective(h, beta=100.0)
        
    def gradient(h):
        return smooth_c5_gradient(h, beta=100.0)
    
    result = minimize(
        objective,
        h_start,
        method='L-BFGS-B',
        jac=gradient,
        bounds=[(0, 1) for _ in range(half_size)],
        options={'gtol': 1e-10, 'ftol': 1e-12, 'maxiter': 500}
    )
    
    if result.success:
        h = result.x
        h = project_to_sum_box(h, target_sum=n_points / 4)
        return h
    else:
        return h_start

def simulated_annealing(h_start, max_iter=150000, T0=1e-2, alpha=0.9995):
    """Simulated Annealing on half-vector"""
    current_h = h_start.copy()
    current_h = project_to_sum_box(current_h, target_sum=n_points / 4)
    current_c5 = c5_from_autocorr(autocorr_fft(current_h))
    
    best_h = current_h.copy()
    best_c5 = current_c5
    
    T = T0
    delta_max = 0.05
    acceptance_count = 0
    total_count = 0
    
    for i in range(max_iter):
        if i % 1000 == 0 and i > 0:
            # Periodically check time
            pass
            
        # Get new candidate
        h_new = current_h.copy()
        i1, i2 = np.random.randint(0, half_size, 2)
        # Add delta with small probability of block move
        if np.random.random() < 0.1:
            # Block move
            block_len = min(np.random.randint(1, 10), half_size - i1)
            block_total = np.sum(h_new[i1:i1+block_len])
            h_new[i1:i1+block_len] = 0
            # Distribute evenly
            h_new[i1:i1+block_len] = block_total / block_len
        else:
            # Swap mass move
            delta = np.random.random() * delta_max
            if (h_new[i1] >= delta and 
                h_new[i2] <= 1.0 - delta):
                h_new[i1] -= delta
                h_new[i2] += delta
                h_new = project_to_sum_box(h_new, target_sum=n_points / 4)
        
        c5_new = c5_from_autocorr(autocorr_fft(h_new))
        
        # Acceptance probability
        if c5_new < current_c5:
            accepted = True
        else:
            p = np.exp(-(c5_new - current_c5) / T)
            accepted = np.random.random() < p
        
        if accepted:
            current_h = h_new.copy()
            current_c5 = c5_new
            acceptance_count += 1
        total_count += 1
            
        # Update best
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Adaptive temperature adjustment
        if total_count % 500 == 0 and total_count > 0:
            acceptance_rate = acceptance_count / total_count
            if acceptance_rate > 0.4:
                delta_max = min(delta_max * 1.1, 0.2)
            elif acceptance_rate < 0.1:
                delta_max = max(delta_max * 0.9, 1e-4)
            acceptance_count = 0
            total_count = 0
            
        # Cooling down
        T = T * alpha
        if T < 1e-6:
            break
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    best_h = None
    best_c5 = float('inf')
    
    # Sweep to find initial solution
    if time.time() - start_time > budget_s * 0.1:
        return [], best_c5, 0
    
    h_sweep, c5_sweep, _ = sweep_find_initial()
    if c5_sweep < best_c5:
        best_h = h_sweep.copy()
        best_c5 = c5_sweep
    
    # Refinement with L-BFGS
    if time.time() - start_time > budget_s * 0.3:
        return best_h.tolist(), float(best_c5), n_points
    
    h_refined = refine_with_lbfgsb(h_sweep)
    c5_refined = c5_from_autocorr(autocorr_fft(h_refined))
    if c5_refined < best_c5:
        best_h = h_refined.copy()
        best_c5 = c5_refined
    
    # Simulated Annealing
    if time.time() - start_time > budget_s * 0.6:
        return best_h.tolist(), float(best_c5), n_points
    
    h_sa, c5_sa = simulated_annealing(h_refined, max_iter=100000)
    if c5_sa < best_c5:
        best_h = h_sa.copy()
        best_c5 = c5_sa
    
    # Final L-BFGS on full vector
    if time.time() - start_time > budget_s * 0.9:
        return best_h.tolist(), float(best_c5), n_points
    
    # Reconstruct full vector and do final optimization
    h_full = np.concatenate([best_h, best_h[::-1]])
    c5_exact = c5_from_autocorr(autocorr_fft(h_full))
    
    # Final polish
    def objective_full(h_full):
        corr = autocorr_fft(h_full)
        return 1.0 - dx * np.min(corr)
    
    # Reconstruct the half-vector from optimized full vector and re-run L-BFGS
    h_half = h_full[:half_size]
    h_half_refined = refine_with_lbfgsb(h_half)
    
    h_full_final = np.concatenate([h_half_refined, h_half_refined[::-1]])
    c5_final = c5_from_autocorr(autocorr_fft(h_full_final))
    
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_full_final.copy()
    
    # Return with exact C5 calculation
    final_corr = autocorr_fft(best_h)
    final_c5 = 1.0 - dx * np.min(final_corr)
    
    return best_h.tolist(), float(final_c5), n_points
