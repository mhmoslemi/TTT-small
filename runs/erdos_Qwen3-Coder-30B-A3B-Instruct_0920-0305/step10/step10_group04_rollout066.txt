import numpy as np
from scipy.signal import correlate
import time

# Global constants
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points

def compute_auto_correlation_binary(x):
    """ Compute auto-correlation of binary vector using FFT """
    x_rev = x[::-1]
    L = 2 * len(x) - 1
    corr = np.fft.ifft(np.fft.fft(x_rev, L) * np.fft.fft(x, L)).real
    return corr

def compute_overlap_from_auto_corr(A, x):
    """ Compute overlap values O_d from auto-correlation A """
    n = len(x)
    # Compute prefix sums
    x_cum = np.concatenate([[0], np.cumsum(x)])
    # Compute O_d for all shifts
    O = np.zeros(2*n - 1)
    for d in range(-n+1, n):
        if d >= 0:
            # i = 0 to n-1-d
            i_start = 0
            i_end = n - d
            sum_x = x_cum[i_end] - x_cum[i_start]
            O[d + n - 1] = sum_x - A[d + n - 1]
        else:
            # i = -d to n-1
            i_start = -d
            i_end = n
            sum_x = x_cum[i_end] - x_cum[i_start]
            O[-d + n - 1] = sum_x - A[-d + n - 1]
    return O

def eval_c5_binary(x):
    """ Fast evaluation of C5 for binary vector """
    A = compute_auto_correlation_binary(x)
    O = compute_overlap_from_auto_corr(A, x)
    return dx * np.max(O)

def project_to_sum_box(x, target_sum):
    """ Project vector x onto box constraints [0,1] with sum = target_sum """
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize constraints
    TARGET_SUM = n_points / 2
    
    # Initial binary vector
    x = np.zeros(n_points, dtype=int)
    x[:n_points//2] = 1
    np.random.shuffle(x)
    
    best_x = x.copy()
    best_c5 = eval_c5_binary(x)
    
    # Precompute full auto-correlation for fast incremental updates
    A = compute_auto_correlation_binary(x)
    O = compute_overlap_from_auto_corr(A, x)
    
    # Annealing parameters
    T0 = 2 * dx
    Tf = 1e-6 * dx
    steps_SA = 300000
    T_schedule = np.linspace(T0, Tf, steps_SA)
    
    # Track acceptance rate
    acc_accepted = 0
    acc_total = 0
    
    cut_window = 20000
    last_best = best_c5
    count_no_improvement = 0
    
    # For incremental update
    x_pad = np.zeros(3*n_points - 2)
    x_pad[n_points-1:2*n_points-1] = x
    
    # Main simulated annealing loop
    for step in range(steps_SA):
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        # Temperature
        T = T_schedule[step]
        
        # Choose move type
        if np.random.random() < 0.6:
            # Targeted swap
            d_star = np.argmax(O)
            d = d_star - n_points + 1
            
            # Compute U and V
            U = []
            V = []
            
            # For positive shift d
            if d >= 0:
                for i in range(n_points - d):
                    if x[i] == 1 and x[i+d] == 0:
                        U.append(i)
                    if x[i] == 0 and x[i+d] == 1:
                        V.append(i)
            else:
                d_abs = -d
                for i in range(n_points - d_abs):
                    if x[i] == 1 and x[i+d_abs] == 0:
                        U.append(i)
                    if x[i] == 0 and x[i+d_abs] == 1:
                        V.append(i)
            
            if U and V:
                i = np.random.choice(U)
                j = np.random.choice(V)
            else:
                # Fallback to random swap
                i, j = np.random.choice(n_points, size=2, replace=False)
                while x[i] == x[j]:
                    i, j = np.random.choice(n_points, size=2, replace=False)
        else:
            # Random swap
            i, j = np.random.choice(n_points, size=2, replace=False)
            while x[i] == x[j]:
                i, j = np.random.choice(n_points, size=2, replace=False)
        
        # Save old values
        old_A = A.copy()
        old_O = O.copy()
        
        # Compute contribution of swap to delta_A
        c1 = np.zeros(2*n_points - 1)
        c2 = np.zeros(2*n_points - 1)
        c1[i+n_points-1 : i+3*n_points-2] = x_pad[i+n_points-1:i+3*n_points-2]
        c1[:n_points-1] = c1[:n_points-1][::-1]
        c1 = c1[:2*n_points-1]
        c2[j+n_points-1 : j+3*n_points-2] = x_pad[j+n_points-1:j+3*n_points-2]
        c2[:n_points-1] = c2[:n_points-1][::-1]
        c2 = c2[:2*n_points-1]
        
        # Calculate delta values for the SWAP
        delta_A = - (c1 + c1[::-1]) + (c2 + c2[::-1])
        
        # Update auto-correlations
        A_new = A + delta_A
        
        # Recompute O from new A
        O_new = np.zeros(2*n_points - 1)
        x_cum = np.concatenate([[0], np.cumsum(x)])
        for d in range(-n_points+1, n_points):
            if d >= 0:
                sum_x = x_cum[n-d] - x_cum[0]
                O_new[d + n - 1] = sum_x - A_new[d + n - 1]
            else:
                sum_x = x_cum[n] - x_cum[-d]
                O_new[-d + n - 1] = sum_x - A_new[-d + n - 1]
        
        new_c5 = dx * np.max(O_new)
        
        # Accept or reject
        if new_c5 < best_c5:
            accepted = True
        else:
            if np.random.uniform() < np.exp(-(new_c5 - best_c5) / T):
                accepted = True
            else:
                accepted = False
                
        if accepted:
            # Apply the swap
            x[i], x[j] = x[j], x[i]
            A = A_new.copy()
            O = O_new.copy()
            if new_c5 < best_c5:
                best_x = x.copy()
                best_c5 = new_c5
            acc_accepted += 1
        else:
            # Revert the auto-correlation to before
            A = old_A.copy()
            O = old_O.copy()
            
        acc_total += 1
        
        # Check for stagnation
        if step % cut_window == 0:
            if best_c5 >= last_best - 1e-5:
                count_no_improvement += 1
                if count_no_improvement > 3:
                    # Partial re-randomization
                    indices = np.random.choice(n_points, size=int(0.05*n_points), replace=False)
                    for idx in indices:
                        x[idx] = 1 - x[idx]
                    # Rebalance to sum = TARGET_SUM
                    ones_count = np.sum(x)
                    if ones_count > TARGET_SUM:
                        # Turn some 1s into 0s
                        zeros = np.where(x==0)[0]
                        nums_to_flip = int(ones_count - TARGET_SUM)
                        flip_indices = np.random.choice(zeros, size=nums_to_flip, replace=False)
                        x[flip_indices] = 1
                    elif ones_count < TARGET_SUM:
                        # Turn some 0s into 1s
                        ones = np.where(x==1)[0]
                        nums_to_flip = int(TARGET_SUM - ones_count)
                        flip_indices = np.random.choice(ones, size=nums_to_flip, replace=False)
                        x[flip_indices] = 0
                    # Reassess
                    A = compute_auto_correlation_binary(x)
                    O = compute_overlap_from_auto_corr(A, x)
                    last_best = eval_c5_binary(x)
                    count_no_improvement = 0
            else:
                last_best = best_c5
                count_no_improvement = 0
    
    # Final expensive continuous refinement
    # Convert binary solution to floating point
    h = best_x.astype(float)
    
    # First pass refinement with smoothing and projection
    # Apply smoothing
    kernel = np.ones(5)/5
    h_smooth = np.convolve(h, kernel, mode='same')
    h = h_smooth
    
    # Project to box and sum constraints
    h = project_to_sum_box(h, TARGET_SUM)
    
    # Refine with SLSQP
    from scipy.optimize import minimize
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
    
    res = minimize(
        lambda x: dx * np.max(np.correlate(x, 1-x, mode="full")),
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False}
    )
    
    if res.success:
        h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    
    # Second pass SLSQP with final projection
    res = minimize(
        lambda x: dx * np.max(np.correlate(x, 1-x, mode="full")),
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False}
    )
    
    if res.success:
        h = res.x
    h = project_to_sum_box(h, TARGET_SUM)
    
    # Evaluate final C5
    final_c5 = dx * np.max(np.correlate(h, 1-h, mode="full"))
    
    # Return final values
    return (h.tolist(), float(final_c5), n_points)
