import numpy as np
from scipy.signal import correlate
import time

# Global variables for convenience
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_overlap_from_corr(A_d, n):
    """Compute overlap values for all shifts from auto-correlation"""
    # A_d is the full correlation array (2n-1 elements)
    # For shift d (from -(n-1) to n-1), compute overlap = prefix_sum - A_d
    prefix_sum = np.concatenate([[0], np.cumsum(np.concatenate([[0], A_d[:n-1]]))])
    n_overlaps = len(A_d)
    overlaps = np.zeros(n_overlaps)
    for d in range(-(n-1), n):
        d_idx = d + n - 1
        if d >= 0:
            # Left shift by d: overlap is (sum of 1's in first n-d positions) - A_d
            sum_ones = prefix_sum[n-d] - prefix_sum[0]
            overlaps[d_idx] = sum_ones - A_d[d_idx]
        else:
            # Right shift by |d|: overlap is (sum of 1's in last n+|d| positions) - A_d
            sum_ones = prefix_sum[n] - prefix_sum[-d]
            overlaps[d_idx] = sum_ones - A_d[d_idx]
    return overlaps

def compute_c5_from_correlation(A_d):
    """Compute C5 from auto-correlation data"""
    overlaps = compute_overlap_from_corr(A_d, n_points)
    return dx * np.max(overlaps)

def project_to_sum_box(x, target_sum):
    """Project onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda to get desired sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def initial_binary_h():
    """Create a high-quality initial binary vector"""
    # Start with a similar pattern to initial_h_values if available, but discretized
    try:
        # Try to load initial_h_values
        h_cont = np.array(initial_h_values)  # Not actually using this
        h_cont = np.array(initial_h_values)
        # Convert to binary: [0,1] based on median threshold
        median_val = np.median(h_cont)
        x = (h_cont > median_val).astype(float)
        # Ensure exactly half are ones
        ones_idx = np.where(x == 1)[0]
        zeros_idx = np.where(x == 0)[0]
        if len(ones_idx) < n_points//2:
            # Add ones to reach n_points/2
            num_to_add = n_points//2 - len(ones_idx)
            indices_to_flip = zeros_idx[:num_to_add]
            x[indices_to_flip] = 1
        elif len(ones_idx) > n_points//2:
            num_to_remove = len(ones_idx) - n_points//2
            indices_to_flip = ones_idx[:num_to_remove]
            x[indices_to_flip] = 0
        
        return x.astype(bool)
    except:
        # Fall back to random binary vector
        h = np.random.rand(n_points) > 0.5
        ones_indices = np.where(h)[0]
        if len(ones_indices) < n_points//2:
            to_add = n_points//2 - len(ones_indices)
            # Add random zero indices
            zero_indices = np.where(~h)[0]
            add_indices = np.random.choice(zero_indices, to_add, replace=False)
            h[add_indices] = True
        elif len(ones_indices) > n_points//2:
            to_remove = len(ones_indices) - n_points//2
            remove_indices = np.random.choice(ones_indices, to_remove, replace=False)
            h[remove_indices] = False
        return h

def compute_correlation_incremental(x, old_A, d, i, j):
    """Compute change to correlation due to swapping x[i] and x[j]"""
    # x is boolean; old_A is already computed full correlation
    new_A = old_A.copy()
    n = len(x)
    # Only shift indices around i or j can affect the correlation
    # Since arrays are indexed as [0, ..., 2n-2] corresponding to d = -(n-1) to n-1
    for k in range(2*n-1):
        old_d = k - (n-1)
        if old_d + i < 0 or old_d + i >= n or old_d + j < 0 or old_d + j >= n:
            continue
        # Compute each affected term
        old_term = x[i + old_d] + x[i - old_d] if i + old_d >= 0 and i - old_d < n else 0
        new_term = x[j + old_d] + x[j - old_d] if j + old_d >= 0 and j - old_d < n else 0
        if k < n and old_d >= 0:
            # [2n-1] window, but we are computing shift -d performance
            # d = k - (n-1) == d
            pass
        new_A[k] += new_term - old_term
    return new_A

def binary_to_float_h(x):
    """Convert boolean to float for use in SLSQP"""
    return x.astype(float)

def c5_evaluation_continuous(h):
    """Evaluate C5 on a continuous variable vector"""
    if isinstance(h, bool):
        h = np.array([h] * n_points, dtype=float)
    h_f = project_to_sum_box(np.array(h), TARGET_SUM)
    corr_full = correlate(h_f, 1 - h_f, mode='full')
    max_overlap = np.max(corr_full)
    return dx * max_overlap

def fast_c5_evaluation_binary(x):
    """Fast evaluation of C5 from binary vector x"""
    # Compute correlation in the usual way
    x_np = x.astype(float)
    A_full_d = correlate(x_np, x_np, mode='full')
    return compute_c5_from_correlation(A_full_d)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Binary vector
    x = initial_binary_h()
    old_A = correlate(x.astype(float), x.astype(float), mode='full')
    current_C5 = fast_c5_evaluation_binary(x)
    
    # Keep track of best solution
    best_x = x.copy()
    best_C5 = current_C5
    
    # Cooling schedule parameters
    T0 = 2 * dx
    Tf = 1e-6 * dx
    Ncool = 300000  # Number of accepted moves for cooling
    temp = T0
    cooling_factor = (Tf / T0) ** (1.0 / Ncool)
    
    # Tune based on time
    max_iter = int(5e6)
    accept_count = 0
    last_improvement = 0
    
    # Monitor for stagnation
    improvement_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Identify worst shift & choose move type
        overlaps = compute_overlap_from_corr(old_A, n_points)
        d_max = np.argmax(overlaps) - (n_points - 1)
        
        # Generate swap move: remember indices are zero-indexed
        if np.random.rand() < 0.6:
            # Targeted move for worst shift
            u_indices = []
            v_indices = []
            
            # Check for y_i = 1, y_{i+d_max} = 0
            for i in range(n_points):
                i_shifted = i + d_max
                if 0 <= i_shifted < n_points:
                    if x[i] and not x[i_shifted]:
                        u_indices.append(i)
                    elif not x[i] and x[i_shifted]:
                        v_indices.append(i)
                        
            if len(u_indices) > 0 and len(v_indices) > 0:
                # Pick a random exposed 1 and missing 0
                i = np.random.choice(u_indices)
                j = np.random.choice(v_indices)
                x_new = x.copy()
                x_new[i], x_new[j] = x_new[j], x_new[i]
                # Incremental correlation update
                new_A = compute_correlation_incremental(x, old_A, d_max, i, j)
            else:
                # Fall back to random swap
                i = np.random.randint(n_points)
                j = np.random.randint(n_points)
                while j == i or (x[i] == x[j]):
                    j = np.random.randint(n_points)
                x_new = x.copy()
                x_new[i], x_new[j] = x_new[j], x_new[i]
                new_A = correlate(x_new.astype(float), x_new.astype(float), mode='full')
        else:
            # Pure random swap
            i = np.random.randint(n_points)
            j = np.random.randint(n_points)
            while j == i or (x[i] == x[j]):
                j = np.random.randint(n_points)
            x_new = x.copy()
            x_new[i], x_new[j] = x_new[j], x_new[i]
            new_A = correlate(x_new.astype(float), x_new.astype(float), mode='full')

        new_C5 = compute_c5_from_correlation(new_A)
        
        # Accept or reject based on Metropolis criterion
        if new_C5 < current_C5:
            x = x_new
            old_A = new_A
            current_C5 = new_C5
            accept_count += 1
            
            if new_C5 < best_C5:
                best_x = x.copy()
                best_C5 = new_C5
                improvement_count = 0
        else:
            # Accept with probability exp(-delta/T)
            delta = new_C5 - current_C5
            if np.random.rand() < np.exp(-delta / temp):
                x = x_new
                old_A = new_A
                current_C5 = new_C5
                accept_count += 1
                # Reset if improvement
                improvement_count = 0
            else:
                improvement_count += 1

        # Update temperature and check early termination
        temp *= cooling_factor
        if accept_count % 20000 == 0:
            # Check for stagnation
            if improvement_count > 30000:
                # Inject diversity
                num_flips = int(0.05 * n_points)
                flip_indices = np.random.choice(n_points, num_flips, replace=False)
                x[flip_indices] = ~x[flip_indices]
                # Rebalance to n/2 ones
                ones_idx = np.where(x)[0]
                if len(ones_idx) > n_points//2:
                    to_remove = len(ones_idx) - n_points//2
                    remove_idx = np.random.choice(ones_idx, to_remove, replace=False)
                    x[remove_idx] = False
                elif len(ones_idx) < n_points//2:
                    to_add = n_points//2 - len(ones_idx)
                    zero_idx = np.where(~x)[0]
                    add_idx = np.random.choice(zero_idx, to_add, replace=False)
                    x[add_idx] = True
                # Recompute A after rebalance
                old_A = correlate(x.astype(float), x.astype(float), mode='full')
                current_C5 = compute_c5_from_correlation(old_A)
                # Reset improvement counter
                improvement_count = 0
                # Downcool
                temp = max(Tf, temp * 0.5)

    # Convert to continuous representation for final optimization
    h_final = binary_to_float_h(best_x)

    # Perform SLSQP refinement
    def objective(h_vec):
        h_proj = project_to_sum_box(np.array(h_vec), TARGET_SUM)
        corr_full = correlate(h_proj, 1 - h_proj, mode='full')
        return dx * np.max(corr_full)

    from scipy.optimize import minimize
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
    try:
        res = minimize(objective, h_final, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 1500, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            h_final = project_to_sum_box(h_final, TARGET_SUM)
    except:
        pass

    # Apply smoothing
    h_smooth = np.convolve(h_final, np.ones(5) / 5, mode='same')
    h_final = project_to_sum_box(h_smooth, TARGET_SUM)

    # Final evaluation
    final_C5 = c5_evaluation_continuous(h_final)
    
    return (h_final.tolist(), float(final_C5), n_points)
