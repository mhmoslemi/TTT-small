def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Use sorting method to efficiently project
    sorted_indices = np.argsort(-x)  # Descending order
    cumulative_sum = 0.0
    for i in range(len(x)):
        idx = sorted_indices[i]
        if cumulative_sum + x[idx] <= target_sum:
            cumulative_sum += x[idx]
        else:
            break
    
    # Compute lambda using water-filling approach
    if cumulative_sum >= target_sum:
        lambda_val = 0.0
    else:
        lambda_val = (target_sum - cumulative_sum) / (len(x) - i)
    
    # Apply projection
    result = np.clip(x + lambda_val, 0, 1)
    
    # Adjust to ensure sum constraint
    current_sum = np.sum(result)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Distribute adjustment to elements to make sum exactly target_sum
        # We add diff to elements with smallest values
        sort_idx = np.argsort(result)
        for i in range(len(result)):
            if diff == 0:
                break
            idx = sort_idx[i]
            to_add = min(diff, 1 - result[idx])
            if to_add > 0:
                result[idx] += to_add
                diff -= to_add
        # Ensure robustness by refining
        if abs(np.sum(result) - target_sum) > 1e-10:
            # Manual adjustment for numerical issues
            result = np.clip(result, 0, 1)
            current_sum = np.sum(result)
            diff = target_sum - current_sum
            if abs(diff) > 1e-10:
                # Distribute adjustment
                delta = diff / len(result)
                result += delta
                result = np.clip(result, 0, 1)
    
    return result

def binary_milp(N, time_limit=30):
    """Solve binary integer programming for C5 minimization"""
    try:
        # Create CVXPY problem
        x = cp.Variable(N, boolean=True)
        t = cp.Variable()  # Real variable for the maximum
        
        # Set constraints
        constraints = [cp.sum(x) == N // 2]
        
        # For each shift k, introduce variables representing products
        # and add constraint sum of products >= N/2 - t
        for k in range(1, N):  # Skip shift 0 to reduce size
            # Constraint: sum_{i=0}^{N-1-k} x[i] * x[i+k] >= N/2 - t
            y = cp.Variable(N - k, boolean=True)
            # Product constraints using McCormick envelopes
            for i in range(N - k):
                # y[i] <= x[i]
                constraints.append(y[i] <= x[i])
                # y[i] <= x[i+k]
                constraints.append(y[i] <= x[i+k])
                # y[i] >= x[i] + x[i+k] - 1
                constraints.append(y[i] >= x[i] + x[i+k] - 1)
            # Constraint on sum
            constraints.append(cp.sum(y) >= N // 2 - t)
        
        # Objective: minimize t
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        # Solve with time limit
        # Use solver options that limit time
        try:
            prob.solve(solver=cp.GLPK_MI, time_limit=time_limit)
        except:
            # Fallback if time limit causes error
            prob.solve(solver=cp.GLPK_MI)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value
        else:
            return None
    except:
        return None

def upsample_binary(binary_vec, target_N):
    """Upsample binary vector to target_N by replicating each element"""
    factor = target_N // len(binary_vec)
    if factor > 0:
        return np.repeat(binary_vec, factor).astype(float)
    else:
        # If upsample factor is less than 1, just return the original
        return binary_vec.astype(float)

def swap_update(O, x, a, b):  
    """Update overlap values efficiently after swapping positions a and b"""
    # Create a copy of O
    new_O = O.copy()
    
    # For each shift k, see how overlap changes
    # Only need to consider shifts where a or b contribute to overlap calculation
    
    # Handle shift k where a contributes (a+k is valid)
    for k in range(N - a):
        if x[a] == 1 and x[a + k] == 1:
            new_O[k] -= 1  # Remove old product if a is contributing to overlap at shift k
    
    # Handle shift k where b contributes (b+k is valid)  
    for k in range(N - b):
        if x[b] == 1 and x[b + k] == 1:
            new_O[k] -= 1  # Remove old product if b is contributing to overlap at shift k
    
    # Handle shift k where new pair gives product (a+k is valid, b shift special)
    for k in range(N - a):
        if x[a] == 1 and x[b + k] == 1:
            new_O[k] += 1  # Add new contribution
        elif x[a] == 0 and x[b + k] == 1:
            # Original a was not given a pair contribution, so now it does
            if k < N - b:  # So b+k is valid index
                if x[b] == 0 and x[a + k] == 1:
                    # Originally a was 1 and b+k was 1, this changes
                    # But those should not happen during swapping
                    pass
            # Add contribution from b to a+k pair
            new_O[k] += 1
        # b is now 1 so it becomes 0 at a
        elif x[a] == 1 and x[b + k] == 0:
            # This only changes if there was overlap.
            pass
            
    # Handle shift k where b partner is now a position killed by a becoming 0
    # The other side is already been done
    
    return new_O

def local_improvement_binary(x_initial, max_iterations=10000):
    """Perform local improvement of binary solution"""
    x = x_initial.copy()
    n = len(x)
    
    # Compute initial overlaps
    O = np.zeros(n)
    for k in range(n):
        # For shift k: sum x_i * x_{i+k}
        if k == 0:
            # This is all pairs of same position, which is just x_i * x_i = x_i
            O[k] = sum(x)
        else:
            # Compute for valid range i = 0 to n-k-1
            # This is a little tricky for implementation since we're not using numpy vectorization
            # but doing it step by step so can incorporate complex structures
            sum_x = 0
            for i in range(n - k):
                sum_x += x[i] * x[i + k] 
            O[k] = sum_x
    
    max_iterations = min(max_iterations, 50000)  # Safety
    
    last_improvement = 0
    
    for it in range(max_iterations):
        if it - last_improvement > 1000:  # Early termination if no improvement after 1000 steps
            break
        # Compute uncovered counts
        D = [n//2 - O[k] for k in range(n)]
        
        # Identify worst shift
        max_idx = np.argmax(D)
        max_D = D[max_idx]
        
        if max_D <= 1e-6:  # Must improve!
            break
            
        # Try to find swaps to improve
        a_candidates = []
        b_candidates = []
        
        for i in range(n):
            if x[i] == 1 and i + max_idx < n:
                if x[i + max_idx] == 0:
                    a_candidates.append(i)
            if x[i] == 0 and i + max_idx < n:
                if x[i + max_idx] == 1:
                    b_candidates.append(i)
        
        # If no valid swaps, try random one
        if len(a_candidates) == 0 or len(b_candidates) == 0:
            # Greedy: randomly pick a 1 and a 0
            ones = np.where(x == 1)[0]
            zeros = np.where(x == 0)[0]
            if len(ones) > 0 and len(zeros) > 0:
                aid = np.random.choice(ones)
                bid = np.random.choice(zeros)
                # The swap may have no effect on worst case, but let's proceed
                # We'll update all relevant shifts
                x[aid], x[bid] = x[bid], x[aid]
                # Update overlap vector for all affected positions
                # Recalculate O efficiently
                new_O = np.zeros(n)
                for k in range(n):
                    if k == 0:
                        new_O[k] = sum(x)
                    else:
                        sum_x = 0
                        for i in range(n - k):
                            sum_x += x[i] * x[i + k]
                        new_O[k] = sum_x
                O = new_O
                last_improvement = it
            continue
        
        # Try to move candidates to improve max_D
        a = np.random.choice(a_candidates)
        b = np.random.choice(b_candidates)
        
        # Simulate the swap
        x_new = x.copy()
        x_new[a], x_new[b] = x_new[b], x_new[a]
        
        # Calculate new overlaps
        new_O = np.zeros(n)
        for k in range(n):
            if k == 0:
                new_O[k] = sum(x_new)
            else:
                sum_x = 0
                for i in range(n - k):
                    sum_x += x_new[i] * x_new[i + k]
                new_O[k] = sum_x
                
        new_D = [n//2 - new_O[k] for k in range(n)]
        new_max_D = max(new_D)
        
        if new_max_D <= max_D - 1e-12:
            # Accept the swap
            x = x_new
            O = new_O
            last_improvement = it
        else:
            x = x  # Actually no swap happens
            # With some small probability accept bad move
            if np.random.random() < 0.001:
                x = x_new
                O = new_O
                last_improvement = it
    return x

def differential_evolution_refinement(h_initial, target_sum, time_budget, start_time):
    """Refine initial solution using differential evolution"""
    def objective(h):
        return evaluate_c5(h)
        
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(len(h_initial))]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimize only if we have time for at least a few seconds
    t_limit = max(0.1, time_budget * 0.5)
    
    try:
        res = differential_evolution(
            objective,
            bounds,
            args=(),
            constraints=cons,
            seed=42,
            maxiter=int(10000 * t_limit),
            popsize=10,  # Smaller population for quicker convergence
            mutation=(0.5, 1.0),
            recombination=0.7,
            atol=1e-5,
            callback=lambda x, convergence: time.time() - start_time > time_budget * 0.9,
            disp=False,
        )
        if res.success:
            return project_to_sum_box(res.x, target_sum)
    except:
        pass
    return h_initial
