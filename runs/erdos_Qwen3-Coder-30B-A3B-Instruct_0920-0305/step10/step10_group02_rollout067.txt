def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        idx = np.random.randint(0, n_points)

def variation_optimization(h0, c5_0, start_time, budget_s):
    """Variational optimization using finite differences"""
    current_h = h0.copy()
    current_c5 = c5_0
    max_iter = 1000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = project_to_sum_box(h_plus, TARGET_SUM)
            
            c5_plus = evaluate_c5(h_plus)
            grad[j] = (c5_plus - current_c5) / eps

        lr = 0.001 / (1 + i * 0.001)  # decaying learning rate
        h_new = current_h - lr * grad
        
        # Project and evaluate
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < current_c5:
            current_h = h_new
            current_c5 = c5_new
            
    return current_h, current_c5
