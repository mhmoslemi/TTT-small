import numpy as np

import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5_with_gradient(h):
    """Compute C5 and gradient using finite differences"""
    c5_val = evaluate_c5(h)
    
    # Compute gradient using finite differences
    eps = 1e-6
    grad = np.zeros_like(h)

        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_sum_box(h_plus, TARGET_SUM)
        c5_plus = evaluate_c5(h_plus)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_to_sum_box(h_minus, TARGET_SUM)
        c5_minus = evaluate_c5(h_minus)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return c5_val, grad

def optimize_with_adam(h0, max_iter=10000):
    """Optimize using Adam optimizer"""
    h = h0.copy()
    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.01
    
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for t in range(1, max_iter + 1):
        c5_val, grad = evaluate_c5_with_gradient(h)
        
        if c5_val < best_c5:
            best_h = h.copy()
            best_c5 = c5_val
            
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project back to constraints
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Early stopping if no improvement for a while
        if t > 1000 and abs(best_c5 - evaluate_c5(best_h)) < 1e-8:
            break
            
    return best_h, best_c5

def local_search(h0, iterations=20000):
    """Perform local search with multiple perturbation strategies"""
    current_h = h0.copy()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Multiple perturbation strategies
    strategies = ['random', 'neighbor', 'global']
    
    for i in range(iterations):
        h_new = current_h.copy()
        
        # Choose perturbation strategy
        strategy = np.random.choice(strategies)
        
        if strategy == 'random':
            # Random small perturbations
            indices = np.random.choice(n_points, size=min(15, n_points//8), replace=False)
            for idx in indices:
                delta = np.random.normal(0, 0.005)
                h_new[idx] += delta
        elif strategy == 'neighbor':
            # Perturb neighbors of selected indices
            indices = np.random.choice(n_points, size=min(5, n_points//20), replace=False)
            for idx in indices:
                for neighbor in [idx-1, idx, idx+1]:
                    if 0 <= neighbor < n_points:
                        delta = np.random.normal(0, 0.01)
                        h_new[neighbor] += delta
        else:  # global
            # Perturb many random indices at once
            indices = np.random.choice(n_points, size=min(20, n_points//6), replace=False)
            for idx in indices:
                delta = np.random.normal(0, 0.02)
                h_new[idx] += delta
        
        # Project to valid constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new)
        
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = current_h.copy()
                best_c5 = c5_new
        elif np.random.random() < 0.02:  # Occasionally accept worse solutions
            current_h = h_new.copy()
            current_c5 = c5_new
    
    return best_h, best_c5
