import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import differential_evolution, minimize
import time

def project_to_sum_box(u, target_sum):
    """Water-filling projection to sum constraint"""
    def f(lam):
        projected = np.clip(u + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    # Binary search for lambda
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    return np.clip(u + (a + b)/2, 0, 1)

def autocorr_fft(h):
    """Compute aperiodic autocorrelation using FFT"""
    n = len(h)
    L = 2 * n - 1
    h_rev = h[::-1]
    corr = ifft(fft(h_rev, L) * fft(h, L)).real
    return corr

def exact_c5(u):
    """Compute exact C5 value from half-vector"""
    n = len(u) * 2
    dx = 2.0 / n
    h = np.concatenate([u, u[::-1]])
    A = autocorr_fft(h)
    return 1 - dx * np.min(A)

def lse_c5(u, beta):
    """Smooth surrogate for C5 using log-sum-exp"""
    n = len(u) * 2
    dx = 2.0 / n
    h = np.concatenate([u, u[::-1]])
    A = autocorr_fft(h)
    return 1 - dx * np.log(np.sum(np.exp(beta * A))) / beta

def grad_lse_c5(u, beta):
    """Gradient of smooth surrogate C5"""
    n = len(u) * 2
    dx = 2.0 / n
    h = np.concatenate([u, u[::-1]])
    A = autocorr_fft(h)
    w = np.exp(beta * A)
    sum_w = np.sum(w)
    
    # Compute gradient using convolution
    w_rev = w[::-1]
    conv = ifft(fft(w_rev, 2*n-1) * fft(h, 2*n-1)).real[:n]
    grad = -dx * conv / sum_w
    return grad[:len(u)]

def parametric_to_halfvector(params, n, family='ramp'):
    """
    Build half-vector from parametric representation.
    family: '3level' or 'ramp'
    """
    m = n // 2
    if family == '3level':
        a = params[0]
        # Ensure a is in valid range
        a = max(0.05, min(0.45, a))
        p = (1 - 2*a) / (2 * (1 - a))
        u = np.ones(m)
        u[0:int(a*m)] = 1
        u[int((1-a)*m):] = 1
        u[int(a*m):int((1-a)*m)] = p
    elif family == 'ramp':
        a, r = params[0], params[1]
        a = max(0.05, min(0.4, a))
        r = max(0, min(0.3, r))
        p = (1 - 2*a - 2*r) / (2*(1-a-r))
        u = np.ones(m)
        u[0:int(a*m)] = 1
        u[int((1-a)*m):] = 1
        u[int(a*m):int((a+r)*m)] = np.linspace(1, p, int(r*m))
        u[int((1-a-r)*m):int((1-a)*m)] = np.linspace(p, 1, int(r*m))
        u[int(a*m):int((1-a)*m)] = p
    return u

def run_de(n_points, bounds, maxiter=200):
    """Run differential evolution on parametric space"""
    def objective(params):
        u = parametric_to_halfvector(params, n_points)
        u = project_to_sum_box(u, n_points/4)
        return exact_c5(u)
    
    res = differential_evolution(objective, bounds, maxiter=maxiter, 
                                popsize=15, seed=42, disp=False)
    u_best = parametric_to_halfvector(res.x, n_points)
    u_best = project_to_sum_box(u_best, n_points/4)
    return u_best, res.fun

def run_lbfgsb(u0, beta, maxiter=800):
    """Projected L-BFGS-B optimization"""
    def objective(u):
        return lse_c5(u, beta)
    
    def gradient(u):
        return grad_lse_c5(u, beta)
    
    # Constraints
    bounds = [(0, 1) for _ in range(len(u0))]
    
    # Optimize
    res = minimize(objective, u0, method='L-BFGS-B', jac=gradient,
                   bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
    
    # Final projection
    u_final = project_to_sum_box(res.x, len(u0) * 2 / 4)
    return u_final, exact_c5(u_final)

def run_sa(u0, maxiter=50000):
    """Simulated annealing polishing"""
    current_u = u0.copy()
    current_c5 = exact_c5(current_u)
    
    T = 5e-3
    alpha = 0.9997
    delta_max = 0.05
    accepted = 0
    total = 0
    
    for i in range(maxiter):
        # Temperature update
        if T < 1e-6:
            break
            
        # Move
        i1, i2 = np.random.choice(len(u0), 2, replace=False)
        delta = np.random.uniform(0, delta_max)
        if current_u[i1] >= delta and current_u[i2] <= 1 - delta:
            new_u = current_u.copy()
            new_u[i1] -= delta
            new_u[i2] += delta
            
            # Apply projection to preserve sum
            new_u = project_to_sum_box(new_u, len(u0) * 2 / 4)
            
            new_c5 = exact_c5(new_u)
            total += 1
            
            if new_c5 < current_c5:
                current_u = new_u
                current_c5 = new_c5
                accepted += 1
            else:
                if np.random.uniform() < np.exp(-(new_c5 - current_c5) / T):
                    current_u = new_u
                    current_c5 = new_c5
                    accepted += 1
        
        # Adaptive step size
        if i % 500 == 0 and i > 0:
            ratio = accepted / total
            if ratio < 0.2:
                delta_max *= 0.9
            elif ratio > 0.4:
                delta_max *= 1.1
            accepted = 0
            total = 0
        
        # Temperature decay
        T *= alpha
    
    return current_u, current_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 120
    dx = 2.0 / n_points
    
    # Phase 1: Parametric sweep for initial coarse search
    best_c5 = float('inf')
    best_u = None
    
    # Sweep 3-level parametric space
    a_values = np.linspace(0.05, 0.45, 200)
    for a in a_values:
        if time.time() - start_time > budget_s * 0.1:
            break
        u = parametric_to_halfvector([a], n_points, '3level')
        u = project_to_sum_box(u, n_points/4)
        c5 = exact_c5(u)
        if c5 < best_c5:
            best_c5 = c5
            best_u = u.copy()
    
    # Phase 2: Differential Evolution on enriched space (ramp family)
    bounds = [(0.05, 0.4), (0, 0.3)]
    u_de, c5_de = run_de(n_points, bounds, maxiter=200)
    if c5_de < best_c5:
        best_c5 = c5_de
        best_u = u_de
    
    # Phase 3: High-resolution refinement
    n_fine = 240
    dx_fine = 2.0 / n_fine
    
    # Interpolate to finer grid
    u_interp = np.interp(np.linspace(0, 1, n_fine//2), 
                        np.linspace(0, 1, len(best_u)), best_u)
    u_interp = project_to_sum_box(u_interp, n_fine/4)
    
    # Smooth gradient descent with higher β
    u_refined, c5_refined = run_lbfgsb(u_interp, beta=200, maxiter=500)
    if c5_refined < best_c5:
        best_c5 = c5_refined
        best_u = u_refined
    
    # Phase 4: Local simulated annealing polishing
    u_sa, c5_sa = run_sa(best_u, maxiter=20000)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_u = u_sa
    
    # Final full evaluation 
    h_full = np.concatenate([best_u, best_u[::-1]])
    final_c5 = exact_c5(best_u)
    
    return (h_full.tolist(), float(final_c5), n_points)
