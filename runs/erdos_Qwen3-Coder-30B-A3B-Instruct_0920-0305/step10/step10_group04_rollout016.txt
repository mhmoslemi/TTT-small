import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def fast_evaluate_c5(x):
    """Fast evaluation of C5 using FFT with precomputed A"""
    # Compute auto-correlation of original x
    A = np.correlate(x, x, mode='full')
    # Compute O_d terms for each d
    n = len(x)
    O = np.zeros(2*n - 1)
    for i in range(n):
        if i < n - 1:
            O[i] = np.sum(x[:n-i]) - A[n-1+i]
        else:
            O[i] = np.sum(x[:n-i]) - A[n-1+i]
    
    return c5_from_corr(O, n)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlation_deltas(x, i, j):
    """Compute incremental correlation updates for swapping x[i] and x[j]"""
    n = len(x)
    x_pad = np.concatenate([[0]*(n-1), x.tolist(), [0]*(n-1)])
    delta = np.zeros(2*n-1)
    
    # For position i (turning off)
    if 0 <= i < n:
        # Rightward influence
        if i + 1 < n:
            delta[i+1 : i + 2*n - 1] -= x_pad[i + n - 1 : i + 3*n - 2]
        # Leftward influence  
        if i - 1 >= 0:
            delta[i-1 : i + n - 1] -= x_pad[i : i + 2*n - 1][::-1]
    
    # For position j (turning on)
    if 0 <= j < n:
        # Rightward influence
        if j + 1 < n:
            delta[j+1 : j + 2*n - 1] += x_pad[j + n - 1 : j + 3*n - 2]
        # Leftward influence  
        if j - 1 >= 0:
            delta[j-1 : j + n - 1] += x_pad[j : j + 2*n - 1][::-1]
    
    return delta

def compute_shift_contributions(x):
    """Compute O_d for each d from the given x"""
    n = len(x)
    A = np.correlate(x, x, mode='full')
    cumsum_x = np.cumsum(x)
    O = np.zeros(2*n - 1)
    for d in range(1-n, n):
        # Determine indices based on d
        if d >= 0:
            start_idx = d
            end_idx = n - d
        else:
            start_idx = 0
            end_idx = n + d
        if end_idx > start_idx and end_idx <= n and start_idx < n:
            sum_x_part = cumsum_x[end_idx-1] if start_idx == 0 else cumsum_x[end_idx-1] - cumsum_x[start_idx-1]
            O[d + n - 1] = sum_x_part - A[n - 1 + d]
    return O

def find_worst_shift(x):
    """Find the index and value of the worst shift contribution O_d"""
    O = compute_shift_contributions(x)
    best_idx = np.argmax(O)
    return best_idx - (n_points-1), O[best_idx]

def targeted_swap(x, i, j):
    """Perform a swap and return whether it improved."""
    x_new = x.copy()
    x_new[i], x_new[j] = x_new[j], x_new[i]
    return x_new
