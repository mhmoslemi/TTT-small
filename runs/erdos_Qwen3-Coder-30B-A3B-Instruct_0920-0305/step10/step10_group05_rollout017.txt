import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.fft import fft, ifft
import time

# Global variables
n_points = 120
dx = 2.0 / n_points
n_half = n_points // 2

def project_to_sum_box(u, target_sum):
    """Water-filling projection to [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(u + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    for _ in range(30):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(u + (a + b)/2, 0, 1)
    return result

def autocorr_fft(h):
    """Compute aperiodic autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(h, L)).real
    return corr

def exact_c5(u):
    """Compute exact C5 using FFT and min of autocorrelation"""
    # Reconstruct full vector with symmetry
    h = np.concatenate([u, u[::-1]])
    corr = autocorr_fft(h)
    min_corr = np.min(corr)
    return 1 - dx * min_corr

def lse_c5(u, beta):
    """Smooth surrogate using log-sum-exp"""
    h = np.concatenate([u, u[::-1]])
    corr = autocorr_fft(h)
    return 1 - (dx / beta) * np.log(np.sum(np.exp(beta * corr)))

def grad_lse_c5(u, beta):
    """Gradient of LSE surrogate"""
    h = np.concatenate([u, u[::-1]])
    corr = autocorr_fft(h)
    w = np.exp(beta * corr)
    w_sum = np.sum(w)
    
    # Convolution of weights with h (reverse)
    # This is a bit tricky due to the indexing, compute manually for clarity
    grad = np.zeros_like(u)
    for i in range(len(u)):
        val = 0
        for k in range(len(h)):
            if 0 <= k-i < len(h):
                val += w[k] * h[k-i]
        grad[i] = -dx * val / w_sum
    
    return grad

def parametric_to_halfvector(params, n):
    """Build half-vector from 2D parameters"""
    a, r = params
    if r > 0.5 - a:
        r = 0.5 - a
        
    # Full vector: edge (a) -> ramp (r) -> middle (1-a-2r) -> ramp (r) -> edge (a)
    # Make piecewise constant
    p = (1 - 2*a) / (2 * (1 - a))
    u = np.full(n // 2, p)
    
    # Fill the upper triangles
    u[:int(a*n_half//2)] = 1
    u[int((a + r)*n_half//2):int((1 - a - r)*n_half//2)] = p
    
    if n_half > 0:
        u[int((1 - a - r)*n_half//2):int((1 - a)*n_half//2)] = p
        u[int((1 - a)*n_half//2):int((1 - a + r)*n_half//2)] = 1
    return u

def run_de():
    """Perform DE search on parametric space"""
    bounds = [(0.05, 0.45), (0, 0.3)]
    
    def objective(params):
        u = parametric_to_halfvector(params, n_points)
        u = project_to_sum_box(u, n_points / 4)
        return exact_c5(u)
    
    res = differential_evolution(
        objective,
        bounds,
        seed=42,
        maxiter=200,
        popsize=15,
        disp=False
    )
    return res.x

def run_lbfgsb(initial_u, beta=200):
    """Projected L-BFGS-B optimization"""
    def objective(u):
        return lse_c5(u, beta)
    
    def grad(u):
        return grad_lse_c5(u, beta)
    
    def constraint_eq(u):
        return np.sum(u) - n_points / 4
    
    bounds = [(0, 1) for _ in range(n_half)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective,
            initial_u,
            method='L-BFGS-B',
            bounds=bounds,
            jac=grad,
            constraints=cons,
            options={'ftol': 1e-12, 'gtol': 1e-10}
        )
        if res.success:
            u_opt = res.x
        else:
            u_opt = initial_u
    except:
        u_opt = initial_u
        
    u_opt = project_to_sum_box(u_opt, n_points / 4)
    return u_opt

    """Simulated Annealing polishing"""
    best_u = u_best.copy()
    best_c5 = exact_c5(best_u)
    
    T = 5e-3
    alpha = 0.9997
    max_iter = 50000
    delta_max = 0.05
    
    for i in range(max_iter):
        if i % 500 == 0 and i > 0:
            # Adaptive delta
            if i % 2000 == 0:
                delta_max *= 0.9
            
        # Pick two random indices
        i1, i2 = np.random.choice(n_half, 2, replace=False)
        delta = np.random.uniform(0, delta_max)
        
        # Try to move mass
        if best_u[i1] >= delta and best_u[i2] <= 1 - delta:
            new_u = best_u.copy()
            new_u[i1] -= delta
            new_u[i2] += delta
            
            c5_new = exact_c5(new_u)
            
            if c5_new < best_c5:
                best_u = new_u
                best_c5 = c5_new
            else:
                # Accept with probability
                if np.random.uniform() < np.exp(-(c5_new - best_c5) / T):
                    best_u = new_u
                    best_c5 = c5_new
        
        T *= alpha
        if T < 1e-6:
            break
    
    return best_u, best_c5
