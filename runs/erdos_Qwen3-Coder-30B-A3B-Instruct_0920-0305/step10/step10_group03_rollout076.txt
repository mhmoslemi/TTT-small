import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_milp_skeleton(N=60):
    """Solve binary MILP for skeleton"""
    # Solving a binary integer program to find a good binary starting point
    try:
        # Variables
        x = cp.Variable(N, boolean=True)
        
        # Auxiliary variables for correlation
        y = cp.Variable((N, N), boolean=True)
        t = cp.Variable(1, nonneg=True)
        
        # Constraints
        constraints = [
            cp.sum(x) == N//2,
            cp.sum(y, axis=0) >= N//2 - t,
            y >= x[:, None] + x[None, :] - 1,  # McCormick
            y <= x[:, None],
            y <= x[None, :]
        ]
        
        # Objective
        objective = cp.Minimize(t)
        
        # Formulate and solve
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.GLPK_MI, verbose=False, time_limit=30)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            solution = np.array(x.value).astype(int)
            return solution
        else:
            return None
    except:
        return None
        

def upsample_binary_vector(bin_vec, target_len):
    """Upsample binary vector to target length"""
    s = target_len // len(bin_vec)
    if s == 0:
        s = 1
    upscaled = np.repeat(bin_vec, s)
    if len(upscaled) > target_len:
        upscaled = upscaled[:target_len]
    elif len(upscaled) < target_len:
        upscaled = np.pad(upscaled, (0, target_len - len(upscaled)), mode='constant')
    return upscaled

def local_swap_search(h0, max_swaps=10000):
    """Perform local swap search focused on worst shift"""
    h = h0.copy().astype(float)
    h = project_to_sum_box(h, TARGET_SUM)
    corr = compute_corr_fft(h)
    rho = np.correlate(h, h, mode='full')
    rho = rho[len(h)-1:]
    
    # Find worst shift
    worst_shift = np.argmax(np.array([TARGET_SUM - r for r in rho[1:]])) + 1
    
    for _ in range(max_swaps):
        # Get set A and B for worst shift
        A = []
        B = []
        for i in range(len(h) - worst_shift):
            if h[i] == 1 and h[i + worst_shift] == 0:
                A.append(i)
            elif h[i] == 0 and h[i + worst_shift] == 1:
                B.append(i)
        
        if not A or not B:
            break
        a = np.random.choice(A)
        b = np.random.choice(B)
        
        # Perform swap
        old_rho = rho[worst_shift]
        h[a], h[b] = h[b], h[a]
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Recompute correlation for the specific shift only
        h_corr = corr.copy()
        new_corr = compute_corr_fft(h)
        corr = new_corr
        
        # Update worst case
        new_worst = np.max(corr)
        if new_worst < c5_from_corr(corr, len(h)):
            worst_shift = np.argmax(np.array([TARGET_SUM - r for r in rho[1:]])) + 1
    
    return h

def differential_evolution(h0, budget_s=30):
    """Run DE to refine continuous variables"""
    # Population-based optimization
    pop_size = 30
    pop = []
    for _ in range(pop_size):
        h = h0 + np.random.normal(0, 0.02, len(h0))
        h = project_to_sum_box(h, TARGET_SUM)
        pop.append(h)
    
    for iteration in range(100000):
        if time.time() - start_time > budget_s - 5:
            break
        
        for i, h in enumerate(pop):
            # Mutation
            idxs = [idx for idx in range(pop_size) if idx != i]
            a, b, c = [pop[j] for j in np.random.choice(idxs, 3, replace=False)]
            F = 0.8
            m = a + F * (b - c)
            m = np.clip(m, 0, 1)
            
            # Crossover
            CR = 0.9
            trial = np.copy(h)
            for j in range(len(h)):
                if np.random.rand() < CR:
                    trial[j] = m[j]
            
            # Project
            trial = project_to_sum_box(trial, TARGET_SUM)
            
            # Selection
            if evaluate_c5(trial) <= evaluate_c5(h):
                pop[i] = trial
    
    # Return best
    best = min(pop, key=evaluate_c5)
    return best

def simulated_annealing(h0, budget_s=30):
    """Simulated annealing for polishing"""
    h = h0.copy()
    h = project_to_sum_box(h, TARGET_SUM)
    current_c5 = evaluate_c5(h)
    
    T = 1.0
    alpha = 0.99
    min_T = 1e-5
    
    while time.time() - start_time < budget_s - 5:
        if T < min_T:
            break
            
        # Pick two points to perturb
        i, j = np.random.choice(len(h), 2, replace=False)
        delta = np.random.uniform(-0.05, 0.05)
        new_h = h.copy()
        new_h[i] = np.clip(new_h[i] + delta, 0, 1)
        new_h[j] = np.clip(new_h[j] - delta, 0, 1)
        
        # Force sum constraint
        new_h = project_to_sum_box(new_h, TARGET_SUM)
        
        new_c5 = evaluate_c5(new_h)
        delta_c5 = new_c5 - current_c5
        
        if delta_c5 <= 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = new_h
            current_c5 = new_c5
            
        T *= alpha
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx, start_time
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Phase A: Binary MILP skeleton
    bin_skeleton = None
    if time.time() - start_time < budget_s * 0.05:
        bin_skeleton = binary_milp_skeleton(N=60)
    
    # If binary MILP fails, generate random binary vector as fallback
    if bin_skeleton is None:
        bin_skeleton = np.random.choice([0, 1], size=60, p=[0.5, 0.5])
        while np.sum(bin_skeleton) != 30:
            bin_skeleton = np.random.choice([0, 1], size=60, p=[0.5, 0.5])
    
    # Phase B: Upsampling
    h0 = upsample_binary_vector(bin_skeleton, n_points)
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    # Phase C: Optional local swap search
    if time.time() - start_time < budget_s * 0.1:
        h0 = local_swap_search(h0)
        h0 = project_to_sum_box(h0, TARGET_SUM)
    
    # Phase D: Differential evolution
    h1 = differential_evolution(h0, budget_s * 0.7)
    h1 = project_to_sum_box(h1, TARGET_SUM)
    
    # Phase E: Simulated Annealing
    if time.time() - start_time < budget_s * 0.95:
        h2 = simulated_annealing(h1, budget_s * 0.05)
        h2 = project_to_sum_box(h2, TARGET_SUM)
        best_h = h2 if evaluate_c5(h2) < evaluate_c5(h1) else h1
    else:
        best_h = h1
    
    # Phase F: Final deterministic hill climbing
    final_h = best_h.copy()
    final_c5 = evaluate_c5(final_h)
    
    # Short hill climbing step
    for _ in range(2000):
        if time.time() - start_time > budget_s - 5:
            break
        h_new = final_h.copy()
        # Perturb a subset of indices
        indices = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
        for i in indices:
            delta = np.random.normal(0, 0.01 * h_new[i] * (1 - h_new[i]))
            h_new[i] = np.clip(h_new[i] + delta, 0, 1)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < final_c5:
            final_h = h_new
            final_c5 = new_c5
    
    best_h = final_h
    best_c5 = final_c5
    
    return (best_h.tolist(), float(best_c5), n_points)
