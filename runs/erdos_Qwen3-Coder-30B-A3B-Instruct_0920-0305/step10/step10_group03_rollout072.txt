def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_binary_milp(N_small, time_limit=60):
    """Solve binary MILP for N_small points to find optimal binary pattern"""
    # Create variables: x for binary, y for products, t for objective
    x = cp.Variable(N_small, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(x) == N_small // 2]
    
    # For each shift, we compute the sum of products x[i] * x[i+k]
    # We represent product by y variables and add McCormick constraints
    for k in range(N_small):
        if k == 0:
            continue  # Skip shift 0 as it's constant
        # Define y variables for products at this shift
        y = cp.Variable(N_small - k, boolean=True)
        # Add McCormick constraints
        for i in range(N_small - k):
            constraints.extend([
                y[i] <= x[i],
                y[i] <= x[i+k],
                y[i] >= x[i] + x[i+k] - 1
            ])
        # Add constraint: sum of products >= N/2 - t
        constraints.append(cp.sum(y) >= N_small // 2 - t)
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    problem = cp.Problem(objective, constraints)
    
    try:
        problem.solve(solver=cp.CBC, max_seconds=time_limit)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value
    except:
        pass
    return None

def upsample_binary(binary_vec, target_N):
    """Upsample binary vector to higher resolution"""
    upsample_factor = target_N // len(binary_vec)
    if upsample_factor > 0:
        upsampled = np.repeat(binary_vec, upsample_factor)
        # Pad if needed to exact length
        if len(upsampled) < target_N:
            upsampled = np.pad(upsampled, (0, target_N - len(upsampled)), mode='constant')
        elif len(upsampled) > target_N:
            upsampled = upsampled[:target_N]
        return upsampled
    return binary_vec
