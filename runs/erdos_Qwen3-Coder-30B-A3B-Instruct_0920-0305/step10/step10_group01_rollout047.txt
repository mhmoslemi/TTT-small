import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_smooth_max(h, alpha):
    """Compute smooth max surrogate with temperature alpha"""
    corr = compute_corr_fft(h)
    # Numerically stable soft max
    c_max = np.max(corr)
    exp_corr = np.exp(alpha * (corr - c_max))
    w = exp_corr / np.sum(exp_corr)
    return w, corr

def smooth_objective(h, alpha):
    """Smooth objective function using log-sum-exp"""
    w, corr = compute_smooth_max(h, alpha)
    return np.sum(w * corr) / alpha + c_max / alpha

def smooth_gradient(h, alpha):
    """Analytical gradient of the smooth objective"""
    w, corr = compute_smooth_max(h, alpha)
    # Symmetrize weights for convolution
    w_sym = w + w[::-1]
    # Convolve with h to get gradient contributions
    g = correlate(w_sym, h, mode='valid')
    # The gradient is 1 - g
    grad = np.ones_like(h) - g
    return grad

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_exact_c5(h):
    """Evaluate the exact C5 value using correlation"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

    def binary_start():
        x = np.zeros(n_points)
        x[:n_points//2] = 1.0
        return project_to_sum_box(x, TARGET_SUM)
    
    def smooth_start():
        x = np.random.uniform(0, 1, n_points)
        return project_to_sum_box(x, TARGET_SUM)

            x = np.interp(np.linspace(0, 1, n_points),

        else:
            x = original_h.copy()
        return project_to_sum_box(x, TARGET_SUM)
    
    strategies = [binary_start, smooth_start, warm_start]
    
    # Multi-restart with continuation
    restarts = 0

    # If we didn't find a solution, return the original approach
