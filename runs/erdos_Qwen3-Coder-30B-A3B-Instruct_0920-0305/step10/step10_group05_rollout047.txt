import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 4

def project_to_sum_box(u, target_sum):
    """Project vector u onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(u + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(u + (a + b)/2, 0, 1)
    return result

def autocorr_fft(h):
    """Compute aperiodic autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def exact_c5(u):
    """Compute exact C5 from half-vector u"""
    # Reconstruct full vector with symmetry
    h = np.concatenate([u, u[::-1]])
    corr = autocorr_fft(h)
    return 1 - dx * np.min(corr)

def lse_c5(u, beta):
    """Compute smooth surrogate C5 using log-sum-exp"""
    h = np.concatenate([u, u[::-1]])
    corr = autocorr_fft(h)
    return 1 - (dx / beta) * np.log(np.sum(np.exp(beta * corr)))

def grad_lse_c5(u, beta):
    """Compute gradient of LSE surrogate"""
    h = np.concatenate([u, u[::-1]])
    corr = autocorr_fft(h)
    w = np.exp(beta * corr)
    w_sum = np.sum(w)
    
    # Compute gradient via convolution
    # We convolve w with (h_rev * 2) to get the right contribution
    h_rev = h[::-1]
    grad = np.zeros_like(h)
    for i in range(len(h)):
        for k in range(len(h)):
            if 0 <= i+k < len(h):
                grad[i] += w[k] * h_rev[i+k]
            if 0 <= i-k < len(h):
                grad[i] += w[k] * h_rev[i-k]
    
    # Normalize
    grad /= w_sum
    
    # Return gradient for half-vector only (symmetric)
    return -dx * grad[:len(u)]

def parametric_to_halfvector(params, n):
    """Build half-vector from a 2 or 3 parameter family"""
    m = n // 2
    
    # 3-level: (edge_length, edge_height=1, middle_height)
    if len(params) == 2:
        a, p = params
        u = np.ones(m)
        if a > 0:
            u[:int(np.ceil(a*m))] = 1.0
        if a < 1:
            u[int(np.floor((1-a)*m)):] = p
            
        # Normalize area to match constraint
        current_sum = np.sum(u)
        u *= TARGET_SUM / current_sum
        
    # 5-level with ramps: (edge_length, ramp_width, middle_height)
    elif len(params) == 3:
        a, r, p = params
        # Define the five segments of the shape
        u = np.full(m, p)
        if a > 0:
            u[:int(np.ceil(a*m))] = 1.0
        if r > 0 and a + r < 1:
            # Linear ramp from 1 to p (or p to 1)
            ramp_start = int(np.ceil((a)*m))
            ramp_end = int(np.ceil((a+r)*m))
            ramp = np.linspace(1, p, ramp_end - ramp_start)
            u[ramp_start:ramp_end] = ramp
        if r > 0 and a + r < 1:
            # Linear ramp from p to 1
            ramp_start = int(np.ceil((1-a-r)*m))
            ramp_end = int(np.ceil((1-a)*m))
            ramp = np.linspace(p, 1, ramp_end - ramp_start)
            u[ramp_start:ramp_end] = ramp
            
        # Normalize area to match constraint
        current_sum = np.sum(u)
        u *= TARGET_SUM / current_sum
        
    return u

def run_de():
    """Perform differential evolution on parameterized family"""
    def objective(params):
        u = parametric_to_halfvector(params, n_points)
        return exact_c5(u)
    
    # Parameter bounds: [a, r] or [a, p] where a ∈ [0.05, 0.45], r ∈ [0, 0.3], p ∈ [0.5, 1.0]
    bounds = [(0.05, 0.45), (0.0, 0.3), (0.5, 1.0)]
    
    # Try 2D parameter space first (edge + ramp)
    bounds_2d = [(0.05, 0.45), (0.0, 0.3)]
    result = differential_evolution(objective, bounds_2d, seed=42, popsize=15, maxiter=200)
    u = parametric_to_halfvector(result.x, n_points)
    best_c5 = exact_c5(u)
    best_params = result.x
    
    return best_params, best_c5

def run_lbfgsb(u_start, beta=200):
    """Run L-BFGS-B optimization with projected gradient"""
    def objective(u):
        return lse_c5(u, beta)
    
    def grad_objective(u):
        return grad_lse_c5(u, beta)
    
    # Minimize with bounds and constraints
    # Use scipy.optimize.minimize with L-BFGS-B
    # For projection after gradient updates, do it manually
    u = u_start.copy()
    max_iter = 800
    tolerance = 1e-10
    for i in range(max_iter):
        # Project to feasible region
        u = project_to_sum_box(u, TARGET_SUM)
        g = grad_objective(u)
        # Simple gradient descent step
        step_size = 1e-3
        u_new = u - step_size * g
        u_new = project_to_sum_box(u_new, TARGET_SUM)
        
        # Check for convergence if we're close to optimal
        if np.linalg.norm(g) < tolerance:
            break
            
        u = u_new
        
        # Early stopping check
        if i % 50 == 0:
            current_c5 = exact_c5(u)
            if current_c5 < 0.38:
                break
    
    return u

def run_sa(u_start):
    """Run simulated annealing polishing"""
    u = u_start.copy()
    current_c5 = exact_c5(u)
    
    temperature = 5e-3
    alpha = 0.9997
    max_iter = 10000
    delta_max = 0.1
    acceptance_count = 0
    total_moves = 0
    
    for i in range(max_iter):
        # Acceptance ratio tracking
        if total_moves > 0 and i % 500 == 0:
            if acceptance_count / total_moves < 0.2:
                delta_max *= 1.1
            elif acceptance_count / total_moves > 0.4:
                delta_max *= 0.9
            acceptance_count = 0
            total_moves = 0
        
        # Make a move
        i1, i2 = np.random.choice(len(u), 2, replace=False)
        delta = np.random.uniform(0, delta_max)
        
        # Check constraint
        if u[i1] >= delta and u[i2] <= 1 - delta:
            u_new = u.copy()
            u_new[i1] -= delta
            u_new[i2] += delta
            
            # Project to sum constraint
            u_new = project_to_sum_box(u_new, TARGET_SUM)
            new_c5 = exact_c5(u_new)
            
            # Accept or reject
            if new_c5 <= current_c5:
                u = u_new
                current_c5 = new_c5
                acceptance_count += 1
            else:
                prob = np.exp(-(new_c5 - current_c5) / temperature)
                if np.random.uniform() < prob:
                    u = u_new
                    current_c5 = new_c5
                    acceptance_count += 1
                    
            total_moves += 1
            temperature *= alpha
            if temperature < 1e-6:
                break
    
    return u

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: Parametric sweep on simple 3-level shape
    best_c5 = 1.0
    best_u = None
    
    # Try different parameter combinations
    a_vals = np.linspace(0.05, 0.45, 400)
    for a in a_vals:
        u = parametric_to_halfvector([a, 1.0], n_points)
        c5 = exact_c5(u)
        if c5 < best_c5:
            best_c5 = c5
            best_u = u.copy()
    
    # Phase 2: DE search
    if time.time() - start_time < budget_s * 0.4:
        try:
            params, de_c5 = run_de()
            if de_c5 < best_c5:
                best_c5 = de_c5
                u = parametric_to_halfvector(params, n_points)
                best_u = u.copy()
        except Exception:
            pass
    
    # Phase 3: Upscale and refine with L-BFGS
    if time.time() - start_time < budget_s * 0.8:
        try:
            # Refine on finer grid
            fine_n = 240
            fine_u = project_to_sum_box(best_u, TARGET_SUM)
            hrs_u = run_lbfgsb(fine_u)
            refined_c5 = exact_c5(hrs_u)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_u = hrs_u.copy()
        except Exception:
            pass
    
    # Phase 4: Simulated Annealing polishing
    if time.time() - start_time < budget_s * 0.95:
        try:
            final_u = run_sa(best_u)
            final_c5 = exact_c5(final_u)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_u = final_u.copy()
        except Exception:
            pass
    
    # Construct full vector
    full_h = np.concatenate([best_u, best_u[::-1]])
    
    return (full_h.tolist(), float(best_c5), n_points)
