# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_surrogate(h, alpha=1.0):
    """Compute the smooth max surrogate for the overlap integral"""
    c = compute_corr_fft(h)
    c_max = np.max(c)
    # Stable computation of log-sum-exp
    exp_c = np.exp(alpha * (c - c_max))
    log_sum_exp = c_max + np.log(np.sum(exp_c)) - np.log(len(c))
    return log_sum_exp / alpha

def compute_gradient(h, alpha=1.0):
    """Compute gradient of smooth max surrogate"""
    c = compute_corr_fft(h)
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    weights = exp_c / np.sum(exp_c)
    
    # Symmetrized weights
    w_sym = weights + weights[::-1]
    # The first N entries of correlation between w_sym and h
    g = np.fft.ifft(np.fft.fft(w_sym, 2*len(h)-1) * np.fft.fft(h, 2*len(h)-1)).real[:len(h)]
    
    # Gradient is 1 - g
    grad = 1 - g
    return grad
