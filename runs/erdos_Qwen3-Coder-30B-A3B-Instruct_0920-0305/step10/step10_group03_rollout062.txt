import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import time
import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort indices by descending x values
    sorted_indices = np.argsort(-x)
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_milp(N, time_limit=30):
    """Solve binary MILP for optimal binary pattern"""
    try:
        # Define variables
        x = cp.Variable(N, boolean=True)
        t = cp.Variable()
        
        # Constraints for each shift
        constraints = [cp.sum(x) == N // 2]
        
        # For each shift, compute the overlap vector efficiently
        for k in range(N):
            # Create y variables for this shift
            if k == 0:
                # Special case - shift by 0
                # Actually, correlation at shift 0: sum of squares of x = sum x = N/2
                # We don't need to add constraint for shift 0 as it's automatically satisfied
                pass
            else:
                # Sum_{i=0}^{N-1-k} x_i * x_{i+k} >= N/2 - t
                y_vars = cp.Variable(N - k, boolean=True)
                # Add McCormick constraints for product variables
                for i in range(N - k):
                    # y[i] <= x[i]
                    constraints.append(y_vars[i] <= x[i])
                    # y[i] <= x[i+k]  
                    constraints.append(y_vars[i] <= x[i + k])
                    # y[i] >= x[i] + x[i+k] - 1
                    constraints.append(y_vars[i] >= x[i] + x[i + k] - 1)
                
                # Add the constraint for this shift
                constraints.append(cp.sum(y_vars) >= N // 2 - t)
        
        # Objective
        objective = cp.Minimize(t)
        
        # Create and solve problem
        prob = cp.Problem(objective, constraints)
        
        # Solve with time limit
        prob.solve(solver=cp.GLPK_MI, time_limit=time_limit)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value
    except:
        return None
    
    return None

def upsample_binary(binary_vec, target_N):
    """Upsample binary vector to target_N points via replication"""
    upsample_factor = target_N // len(binary_vec)
    result = np.repeat(binary_vec, upsample_factor)
    return result

def differential_evolution_search(h0, target_sum, time_budget, start_time):
    """Run differential evolution optimization"""
    
    def objective(x):
        # Project to sum constraint
        x_proj = project_to_sum_box(x, target_sum)
        # Evaluate C5
        return evaluate_c5(x_proj)
    
    # Use differential evolution
    bounds = [(0, 1) for _ in range(len(h0))]
    
    result = differential_evolution(
        objective,
        bounds,
        maxiter=max(100, int(time_budget * 10)),
        popsize=20,
        seed=42,
        mutation=(0.5, 1.0),
        recombination=0.7,
        atol=1e-6,
        tol=1e-6,
        callback=lambda x, convergence: time.time() - start_time > time_budget * 0.95
    )
    
    # Final projection and evaluation
    final_x = project_to_sum_box(result.x, target_sum)
    final_c5 = evaluate_c5(final_x)
    
    return final_x, final_c5

def local_hill_climb(h0, target_sum, time_budget, start_time):
    """Perform hill climbing with swaps"""
    current_h = h0.copy()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    max_iter = int(time_budget * 1000)
    
    for i in range(max_iter):
        if time.time() - start_time > time_budget * 0.95:
            break
            
        # Try a random swap
        idx1, idx2 = np.random.choice(len(h0), 2, replace=False)
        
        # Create new vector
        new_h = current_h.copy()
        new_h[idx1], new_h[idx2] = new_h[idx2], new_h[idx1]
        
        # Project to sum constraint
        new_h = project_to_sum_box(new_h, target_sum)
        
        # Evaluate
        new_c5 = evaluate_c5(new_h)
        
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = new_c5
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Try binary optimization approach
    binary_solution = None
    try:
        # Try to solve binary problem with small N for quick solution
        if time.time() - start_time < budget_s * 0.2:
            binary_solution = binary_milp(60, 30)  # Try with 60 points
            
            # If we got a solution
            if binary_solution is not None and np.sum(binary_solution) == 30:
                # Upsample to 120 points
                binary_upsampled = upsample_binary(binary_solution, 120)
                
                # Start with this solution
                h0 = binary_upsampled.astype(float)
                best_c5 = evaluate_c5(h0)
                best_h = h0.copy()
                
                # Run differential evolution on refined points
                if time.time() - start_time < budget_s * 0.7:
                    final_h, final_c5 = differential_evolution_search(
                        h0, TARGET_SUM, budget_s * 0.5, start_time
                    )
                    if final_c5 < best_c5:
                        best_h = final_h
                        best_c5 = final_c5
                
                # Final hill climbing
                if time.time() - start_time < budget_s * 0.95:
                    final_h, final_c5 = local_hill_climb(
                        best_h, TARGET_SUM, budget_s * 0.2, start_time
                    )
                    if final_c5 < best_c5:
                        best_h = final_h
                        best_c5 = final_c5
                
                return (best_h.tolist(), float(best_c5), n_points)
    except:
        pass
    
    # Fallback to random initialization
    h0 = np.random.rand(n_points)
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Try some optimization strategies
    if time.time() - start_time < budget_s * 0.7:
        # Differential evolution on continuous
        final_h, final_c5 = differential_evolution_search(
            h0, TARGET_SUM, budget_s * 0.5, start_time
        )
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    # Local hill climbing
    if time.time() - start_time < budget_s * 0.95:
        final_h, final_c5 = local_hill_climb(
            best_h, TARGET_SUM, budget_s * 0.2, start_time
        )
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    return (best_h.tolist(), float(best_c5), n_points)
