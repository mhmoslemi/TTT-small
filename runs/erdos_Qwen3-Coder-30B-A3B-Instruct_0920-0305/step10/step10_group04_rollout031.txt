import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import math

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Generate initial binary pattern
    h0 = np.zeros(n_points)
    h0[:n_points//2] = 1
    np.random.shuffle(h0)
    
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    # Simulated Annealing
    max_iters = 500000
    temp0 = 0.02
    temp_end = 1e-5
    
    # Precompute list of indices for efficiency
    ones_idx = np.where(h0 == 1)[0]
    zeros_idx = np.where(h0 == 0)[0]
    
    current_h = h0.copy()
    current_c5 = best_c5
    
    # Time management
    time_limit = start_time + budget_s * 0.8
    
    for iteration in range(max_iters):
        if time.time() > time_limit:
            break
            
        # Compute temperature
        T = temp0 * (1 - iteration / max_iters) + temp_end
        
        # Choose two random indices with different values
        i = np.random.randint(n_points)
        j = np.random.randint(n_points)
        while current_h[i] == current_h[j]:
            i = np.random.randint(n_points)
            j = np.random.randint(n_points)
        
        # Store original values for reversion
        orig_i, orig_j = current_h[i], current_h[j]
        
        # Perform swap
        current_h[i], current_h[j] = current_h[j], current_h[i]
        
        # Compute new C5
        new_c5 = compute_c5(current_h)
        
        # Acceptance probability
        delta = new_c5 - current_c5
        if delta <= 0 or np.random.rand() < math.exp(-delta / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
                
            # Update index lists
            if orig_i == 1 and orig_j == 0:
                # i changed from 1 to 0, j changed from 0 to 1
                ones_idx = ones_idx[ones_idx != i]
                zeros_idx = zeros_idx[zeros_idx != j]
                ones_idx = np.append(ones_idx, j)
                zeros_idx = np.append(zeros_idx, i)
        else:
            # Revert swap
            current_h[i], current_h[j] = orig_i, orig_j
    
    # Continuous refinement with SLSQP
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
        
        res = minimize(
            compute_c5, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            refined_h = res.x
            refined_h = project_to_sum_box(refined_h, TARGET_SUM)
            refined_c5 = compute_c5(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
