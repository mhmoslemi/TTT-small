def full_corr(a, b):
    """Compute correlation with zero-padding using FFT"""
    L = 2 * len(a) - 1
    return np.fft.ifft(np.fft.fft(a, L) * np.conj(np.fft.fft(b, L))).real

def softmax(x, axis=None):
    """Numerically stable softmax"""
    x_shifted = x - np.max(x, axis=axis, keepdims=True)
    exp_x = np.exp(x_shifted)
    return exp_x / np.sum(exp_x, axis=axis, keepdims=True)

def smooth_max_and_weights(c, alpha):
    """Compute smooth max and weights"""
    if alpha <= 0:
        return np.max(c), np.zeros_like(c)
    exp_alpha_c = np.exp(alpha * c)
    sum_exp_alpha_c = np.sum(exp_alpha_c)
    w = exp_alpha_c / sum_exp_alpha_c
    smooth_max = np.log(sum_exp_alpha_c) / alpha
    return smooth_max, w

def compute_gradient(h, alpha):
    """Compute gradient of smooth max with respect to h"""
    # Compute correlation c_k = sum_i h_i (1 - h_{i+k})
    c = full_corr(h, 1 - h)
    # Compute softmax weights
    _, w = smooth_max_and_weights(c, alpha)
    # Symmetrize weights
    w_sym = w + w[::-1]
    # Compute correlation of w_sym with h
    g = full_corr(w_sym, h)[:len(h)]
    # Gradient = 1 - g
    return 1.0 - g

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
        
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    grad = compute_gradient(h, alpha)
    return value, grad

def optimize_with_continuation(h0, alpha_list):
    """Optimize given starting point with continuation"""
    h = h0.copy()
    for alpha in alpha_list:
        def obj_and_grad(h):
            return objective_and_gradient(h, alpha)
            
        # Constraints
        A_eq = np.array([np.ones_like(h)])
        b_eq = np.array([TARGET_SUM])
        
        # Bounds
        bounds = [(0, 1) for _ in range(len(h))]
        
        # Optimize
        try:
            res = minimize(
                obj_and_grad,
                h,
                method='trust-constr',
                jac=True,
                bounds=bounds,
                constraints={'type': 'eq', 'fun': lambda x: A_eq @ x - b_eq},
                options={'verbose': 0}
            )
            if res.success:
                h = res.x
            else:
                # fallback to previous good solution
                pass
        except:
            # fallback to previous good solution
            pass
            
        # project to feasible set exactly
        h = project_to_sum_box(h, TARGET_SUM)
    return h
