def propose_move(h, max_step, rng):
    n = len(h)
    # choose i != j
    i = rng.integers(n)
    j = rng.integers(n-1)
    if j >= i:
        j += 1
    # direction: we will add delta to i, subtract from j
    lower = max(-h[i], h[j] - 1)
    upper = min(1 - h[i], h[j])
    if lower >= upper:
        # try reverse direction
        lower = max(-h[j], h[i] - 1)
        upper = min(1 - h[j], h[i])

def propose_move(h, max_step, rng):
    n = len(h)
    # try up to few times to find feasible pair
    for _ in range(10):
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # compute feasible interval for delta added to i and subtracted from j
        lower = max(-h[i], h[j] - 1)
        upper = min(1 - h[i], h[j])
        if lower < upper:
            # limit delta magnitude
            lower = max(lower, -max_step)
            upper = min(upper, max_step)
            if lower >= upper:
                continue
            delta = rng.uniform(lower, upper)
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            return new_h
        # try opposite direction: add to j, subtract from i
        lower = max(-h[j], h[i] - 1)
        upper = min(1 - h[j], h[i])
        if lower < upper:
            lower = max(lower, -max_step)
            upper = min(upper, max_step)
            if lower >= upper:
                continue
            delta = rng.uniform(lower, upper)
            new_h = h.copy()
            new_h[j] += delta
            new_h[i] -= delta
            return new_h
    return None
