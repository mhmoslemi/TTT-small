import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Start with a binary pattern
    h0 = np.zeros(n_points)
    h0[:n_points//2] = 1
    np.random.shuffle(h0)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Simulated annealing
    max_iters = 500000
    T0 = 0.02
    T_end = 1e-5
    
    # Track the best solution found
    current_h = h0.copy()
    current_c5 = best_c5
    
    # Keep track of time
    elapsed = 0
    
    # Main optimization loop
    for iteration in range(max_iters):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Annealing temperature
        T = T0 * (1 - iteration / max_iters) + T_end
        
        # Select two random indices with different values
        i = np.random.randint(n_points)
        j = np.random.randint(n_points)
        while current_h[i] == current_h[j]:
            i = np.random.randint(n_points)
            j = np.random.randint(n_points)
        
        # Swap
        current_h[i], current_h[j] = current_h[j], current_h[i]
        
        # Evaluate new C5
        new_c5 = evaluate_c5(current_h)
        
        # Accept or reject
        delta = new_c5 - current_c5
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = new_c5
        else:
            # Revert swap
            current_h[i], current_h[j] = current_h[j], current_h[i]
    
    # Final SLSQP optimization to refine
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
        res = minimize(
            evaluate_c5, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            refined_h = res.x
            refined_h = project_to_sum_box(refined_h, TARGET_SUM)
            refined_c5 = evaluate_c5(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
