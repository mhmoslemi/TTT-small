import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from math import gcd

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gcd_list(numbers):
    """Compute GCD of a list of numbers"""
    result = numbers[0]
    for i in range(1, len(numbers)):
        result = gcd(result, numbers[i])
    return result

def generate_modular_sequences():
    """Generate candidate sequences using modular arithmetic"""
    # Find coprime numbers with n_points
    coprimes = [i for i in range(1, n_points) if gcd(i, n_points) == 1]
    best_c5 = float('inf')
    best_h = None
    
    for a in coprimes:
        for b in range(n_points):
            # Create binary sequence using linear congruential generator
            h = np.zeros(n_points)
            for i in range(n_points):
                pos = (a * i + b) % n_points
                h[i] = 1 if pos < n_points // 2 else 0
            
            # Evaluate C5
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5

def local_search(h, c5):
    """Perform local search with simulated annealing"""
    current_h = h.copy()
    current_c5 = c5
    best_h = h.copy()
    best_c5 = c5
    
    start_time = time.time()
    max_iter = 10000
    temp = 0.1
    
    for i in range(max_iter):
        if time.time() - start_time > 500:  # Leave some time for final steps
            break
            
        # Try to improve by swapping two points
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        
        # Make a copy and swap
        h_new = current_h.copy()
        h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
        
        # Project to constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        c5_new = evaluate_c5(h_new)
        
        # Accept with probability
        if c5_new < current_c5 or np.random.random() < np.exp(-(c5_new - current_c5) / temp):
            current_h = h_new.copy()
            current_c5 = c5_new
            
            if c5_new < best_c5:
                best_h = current_h.copy()
                best_c5 = c5_new
        
        # Cool down temperature
        temp *= 0.9999
        
    return best_h, best_c5

def smooth_step_function(h):
    """Smooth the step function with linear interpolation at boundaries"""
    # Find all boundaries between 0 and 1
    boundaries = []
    for i in range(n_points - 1):
        if h[i] != h[i+1]:
            boundaries.append(i)
    
    # Apply smoothing to boundaries
    h_smooth = h.copy()
    for i in boundaries:
        # Apply smoothing over 3 points
        if i > 0 and i < n_points - 2:
            # Interpolate from h[i-1] to h[i+1]
            h_smooth[i-1] = 0.5 * (h[i-1] + h[i+1])
            h_smooth[i] = 0.5 * (h[i-1] + h[i+1])
            h_smooth[i+1] = 0.5 * (h[i-1] + h[i+1])
    
    # Adjust sum to target
    current_sum = np.sum(h_smooth)
    delta = (TARGET_SUM - current_sum) / n_points
    h_smooth += delta
    
    # Clip to [0,1]
    h_smooth = np.clip(h_smooth, 0, 1)
    
    return h_smooth

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    
    # Generate initial candidates using modular arithmetic
    h_initial, c5_initial = generate_modular_sequences()
    
    # Use the best initial solution
    best_h = h_initial.copy()
    best_c5 = c5_initial
    
    # Apply local search
    if time.time() - start_time < budget_s - 10:
        h_local, c5_local = local_search(best_h, best_c5)
        if c5_local < best_c5:
            best_h = h_local
            best_c5 = c5_local
    
    # Apply smoothing
    if time.time() - start_time < budget_s - 10:
        h_smooth = smooth_step_function(best_h)
        c5_smooth = evaluate_c5(h_smooth)
        if c5_smooth < best_c5:
            best_h = h_smooth
            best_c5 = c5_smooth
    
    # Final gradient-based refinement
    if time.time() - start_time < budget_s - 10:
        # Create smooth gradient-based refinement
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        # Try different starting points for refinement
        for _ in range(3):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Perturb the solution slightly
            h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
            h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
            
            try:
                res = minimize(
                    objective, 
                    h_perturbed, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=cons,
                    options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10}
                )
                if res.success:
                    h_refined = res.x
                    h_refined = project_to_sum_box(h_refined, TARGET_SUM)
                    c5_refined = evaluate_c5(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                continue
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
