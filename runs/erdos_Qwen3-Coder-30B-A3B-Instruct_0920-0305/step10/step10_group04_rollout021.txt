import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Start with random binary pattern
    h0 = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    # Ensure we have exactly the right number of ones
    ones_count = np.sum(h0)
    if ones_count > TARGET_SUM:
        # Turn some ones into zeros
        indices = np.where(h0 == 1)[0]
        np.random.shuffle(indices)
        for i in indices[:int(ones_count - TARGET_SUM)]:
            h0[i] = 0
    elif ones_count < TARGET_SUM:
        # Turn some zeros into ones
        indices = np.where(h0 == 0)[0]
        np.random.shuffle(indices)
        for i in indices[:int(TARGET_SUM - ones_count)]:
            h0[i] = 1
    
    # Ensure h0 sum is exactly TARGET_SUM (handle numerical precision issues)
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    # Multi-start approach
    for _ in range(5):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Initialize with random binary pattern
        h0 = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
        h0 = project_to_sum_box(h0, TARGET_SUM)
        
        # Ensure exactly TARGET_SUM ones
        ones_count = np.sum(h0)
        if ones_count > TARGET_SUM:
            indices = np.where(h0 == 1)[0]
            np.random.shuffle(indices)
            for i in indices[:int(ones_count - TARGET_SUM)]:
                h0[i] = 0
        elif ones_count < TARGET_SUM:
            indices = np.where(h0 == 0)[0]
            np.random.shuffle(indices)
            for i in indices[:int(TARGET_SUM - ones_count)]:
                h0[i] = 1
        
        h0 = project_to_sum_box(h0, TARGET_SUM)
        
        current_h = h0.copy()
        current_c5 = compute_c5(h0)
        
        # Simulated Annealing
        max_iter = 500000
        T0 = 0.01
        T_end = 1e-6
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s * 0.8:
                break
                
            # Choose two distinct positions
            i = np.random.randint(n_points)
            j = np.random.randint(n_points)
            while current_h[i] == current_h[j]:
                i = np.random.randint(n_points)
                j = np.random.randint(n_points)
            
            # Swap
            current_h[i], current_h[j] = current_h[j], current_h[i]
            
            # Compute new C5
            new_c5 = compute_c5(current_h)
            
            # Acceptance criterion
            T = T0 * (1 - iteration / max_iter) + T_end
            delta = new_c5 - current_c5
            
            if delta <= 0 or np.random.rand() < np.exp(-delta / T):
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
            else:
                # Revert swap
                current_h[i], current_h[j] = current_h[j], current_h[i]
    
    # Local refinement with continuous optimization
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
        res = minimize(
            compute_c5, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            refined_h = res.x
            refined_h = project_to_sum_box(refined_h, TARGET_SUM)
            refined_c5 = compute_c5(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
