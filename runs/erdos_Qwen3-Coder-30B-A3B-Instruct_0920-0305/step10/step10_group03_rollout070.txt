import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_step(h, step_size=0.01):
    """Compute gradient step using finite differences"""
    # Simple finite difference gradient estimation
    eps = 1e-6
    grad = np.zeros_like(h)
    
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_sum_box(h_plus, TARGET_SUM)

        h_minus[i] -= eps

        c5_minus = evaluate_c5(h_minus)

    # Apply gradient step
    h_new = h - step_size * grad
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    return h_new

    best_h = current_h.copy()
    best_c5 = current_c5
    step_size = 0.01
    
    for i in range(max_iter):
        # Adaptive step size decrease
        if i > 0 and i % 100 == 0:
            step_size *= 0.99
        
        h_new = gradient_step(current_h, step_size)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = current_h.copy()
                best_c5 = c5_new
        else:
            # Occasionally accept worsening steps to escape local optima
            if np.random.random() < 0.01:
                current_h = h_new.copy()
                current_c5 = c5_new
                
    return best_h, best_c5

            indices = np.random.choice(n_points, size=min(5, n_points//20), replace=False)

                delta = np.random.normal(0, step_size)
                h_new[idx] += delta
        else:  # global
            # Global perturbation with random direction
            perturbation = np.random.normal(0, 0.01, n_points)
            h_new += perturbation
            
        # Project to valid constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)

        c5_new = evaluate_c5(h_new)
        
        if c5_new < current_c5:

            current_c5 = c5_new

                best_h = current_h.copy()
                best_c5 = c5_new
        elif np.random.random() < 0.03:  # Occasionally accept worse solutions
            current_h = h_new.copy()
            current_c5 = c5_new
    
    return best_h, best_c5
