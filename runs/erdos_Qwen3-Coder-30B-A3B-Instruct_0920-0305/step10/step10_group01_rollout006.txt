from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def smooth_max_surrogate(h, alpha):
    """Compute smooth max surrogate for the correlation values"""
    corr = compute_corr_fft(h)
    # Numerically stable soft-max computation
    corr_max = np.max(corr)
    exp_vals = np.exp(alpha * (corr - corr_max))
    w = exp_vals / np.sum(exp_vals)
    return corr_max + np.log(np.sum(exp_vals)) / alpha

def gradient_smooth_max(h, alpha):
    """Compute the gradient of the smooth max surrogate"""
    corr = compute_corr_fft(h)
    corr_max = np.max(corr)
    exp_vals = np.exp(alpha * (corr - corr_max))
    w = exp_vals / np.sum(exp_vals)
    
    # Symmetrized weights
    w_sym = w + w[::-1]
    # Correlation with h to get g_j
    g = compute_corr_fft(w_sym)[:len(h)]
    
    # Gradient vector
    grad = 1.0 - g
    return grad

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_exact_c5(h):
    """Evaluate the exact C5 using raw correlation"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)
