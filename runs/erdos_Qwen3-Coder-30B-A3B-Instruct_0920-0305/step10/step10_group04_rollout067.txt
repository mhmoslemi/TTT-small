import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    dx = 2.0 / n
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def fast_corr_incremental_update(h, x, i, j):
    """Given binary h, and indices i,j, compute the delta correlation vector."""
    n = len(h)
    # Pre-pad the binary input
    h_pad = np.pad(h, (n-1, n-1), mode='constant', constant_values=0)
    
    delta = np.zeros(2*n - 1)
    
    # Contribution from index i being turned OFF
    # We look at h_pad[i+n-1 : i+3n-2] as the right-side contribution,
    # and reverse view as the left-side contribution.
    left_off = h_pad[i : i+2*n-1][::-1]
    right_off = h_pad[i+n-1 : i+3*n-2]
    delta -= (left_off + right_off)
    
    # Contribution from index j being turned ON
    left_on = h_pad[j : j+2*n-1][::-1] 
    right_on = h_pad[j+n-1 : j+3*n-2]
    delta += (left_on + right_on)
    
    return delta

def compute_total_overlap(x):
    """Compute the total overlap for a binary vector x. Only valid if x is binary."""
    # For binary vector, A_d = sum of products x_i * x_{i+d}
    # O_d = sum x_i - A_d for appropriate range
    
    n = len(x)
    dx = 2.0 / n
    
    # compute A = auto_correlation of x, with mode='full'
    A = correlate(x, x, mode='full')
    
    # Compute the actual overlap term O_d for each shift d
    # for the d'th shift, complement is 1-x
    # O_d = sum of x_i * (1 - x_{i+d})  
    #     = sum of x_i - sum of x_i * x_{i+d} 
    #     = sum of x_i - A_d

    # To compute sum of x_i for prefix:
    x_cumsum = np.concatenate(([0], np.cumsum(x)))
    
    # For each shift d computed from the full correlation, calculate the
    # open sum (with indexes < n) and also account for that d = by base size
    # More directly, we compute the shift d contributions in an optimized way.
    
    # Create auto-correlation indices and precalc prefix sums to derive O_d efficiently.
    O = np.zeros(2 * n - 1)
    # For each d, apply a range based on d:
    # d will be in [-n+1, n-1] with 2*(n-1)+1 values (index 0,1,...,2n-3)
    for d in range(2*n - 1):
        offset = d - (n - 1)
        if offset >= 0:
            # left part: [0, n-1-offset]
            l = 0
            r = n - 1 - offset
            sum_x = x_cumsum[r + 1] - x_cumsum[l]
            sum_x_counts = A[d]
            O[d] = sum_x - sum_x_counts
        else:
            # right part: [-offset, n-1]  
            l = 0 - offset
            r = n - 1
            sum_x = x_cumsum[r + 1] - x_cumsum[l]
            sum_x_counts = A[d]
            O[d] = sum_x - sum_x_counts
    
    return dx * np.max(O)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 240
    dx = 2.0 / n
    
    # Precompute useful constants
    TARGET_SUM = n / 2  # target sum constraint
    
    # Initialize
    best_x = np.zeros(n, dtype=int)
    num_ones = n // 2
    ones_indices = np.random.choice(n, size=num_ones, replace=False)
    best_x[ones_indices] = 1
    
    # Get fast initial correlation evaluation for binary vector
    # This is a key precomputation
    best_h = best_x.astype(float)
    best_C5 = compute_total_overlap(best_x)
    
    # Simulated Annealing loop start
    current_x = best_x.copy()
    current_C5 = best_C5
    current_h = best_h.copy()
    
    # Temperature schedule parameters
    T0 = 2*dx
    T_last = 1e-6 * dx
    total_moves = 300000
    acceptance_window = 10000
    sampled_steps = []
    best_seen = best_C5
    step_count = 0
    
    # Define bound constraints for SLSQP
    bounds = [(0, 1) for _ in range(n)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}

    # Run simulated annealing loop
    for idx in range(total_moves):
        if time.time() - start_time > budget_s - 5:
            break

        # Accept with probability exp(-ΔC5/T)
        # Use targeted or random moves
        if np.random.rand() < 0.6:
            # Select the worst shift
            # Note: for the correlation amplitude evaluation, shortcutting to
            # full correlation and indexing exploiting symmetry helps.
            # But since we have a binary vector x, compute O explicitly more efficiently
            
            # Binary pattern of x.
            A = correlate(current_x, current_x, mode='full')
            
            # Do full computation of O vector (i.e., for shift d give sum of x_i * (1-x_{i+d}) )
            # We compute using cumulative sum of x, and also A_d.
            x_cumsum = np.concatenate(([0], np.cumsum(current_x)))
            O = np.zeros_like(A)
            for d in range(len(A)):
                offset = d - (n - 1)
                if offset >= 0:
                    l = 0
                    r = n - 1 - offset
                    sum_x = x_cumsum[r+1] - x_cumsum[l]
                    A_d = A[d]
                    O[d] = sum_x - A_d
                else:
                    l = -offset
                    r = n - 1
                    sum_x = x_cumsum[r+1] - x_cumsum[l]
                    A_d = A[d]
                    O[d] = sum_x - A_d
                               
            d_star = np.argmax(O)
            worst_d = d_star - (n-1)
            
            # Candidate sets U,V for the worst shift
            U = []  # indices where x_i == 1 and x_{i+worst_d} == 0
            V = []  # indices where x_i == 0 and x_{i+worst_d} == 1
            
            # Forward shift case : x_{i+worst_d} requires computed index to exist (in bounds)
            
            if worst_d >= 0:
                # look for positions where x_i = 1 and x_{i+d} = 0
                for i in range(n - worst_d):
                    if current_x[i] == 1 and current_x[i + worst_d] == 0:
                        U.append(i)
                # look for positions where x_i = 0 and x_{i+d} = 1  
                for j in range(n - worst_d):
                    if current_x[j] == 0 and current_x[j + worst_d] == 1:
                        V.append(j)
            else:  # backwards shift: d < 0
                # x_{i+worst_d} requires x_i to come after, so x_i = 1, x_{i+worst_d} = 0                
                # i + worst_d < i  =>  worst_d < 0
                # valid indices for x_{i+worst_d}: i >= -worst_d = |worst_d|
                # and i + worst_d < n  =>  i < n - worst_d = n + |worst_d|
                # so range is [abs(worst_d), n - abs(worst_d))
                for i in range(-worst_d, n - (-worst_d)):
                    if current_x[i] == 1 and current_x[i + worst_d] == 0:
                        U.append(i)
                for j in range(-worst_d, n - (-worst_d)):
                    if current_x[j] == 0 and current_x[j + worst_d] == 1:
                        V.append(j)
            
            # Perform swap
            if len(U) > 0 and len(V) > 0:
                i = np.random.choice(U)
                j = np.random.choice(V)
                
                # Compute delta correlation vector with incremental update
                delta_corr = fast_corr_incremental_update(current_x, current_h, i, j)
                
                # Compute candidates
                new_A = A + delta_corr
                
                xh = current_x.astype(float)
                xh_cumsum = np.concatenate(([0], np.cumsum(xh)))
                new_O = np.zeros_like(new_A)
                for d in range(len(new_A)):
                    offset = d - (n-1)
                    if offset >= 0:
                        l = 0
                        r = n - 1 - offset  
                        sum_x = xh_cumsum[r+1] - xh_cumsum[l]
                        A_d = new_A[d]
                        new_O[d] = sum_x - A_d
                    else:
                        l = -offset
                        r = n - 1
                        sum_x = xh_cumsum[r+1] - xh_cumsum[l]
                        A_d = new_A[d]
                        new_O[d] = sum_x - A_d
                        
                new_c5 = dx * np.max(new_O)
            else:
                # fallback to random move if no valid weak shift move found
                i, j = np.random.choice(n, size=2, replace=False)
                while current_x[i] == current_x[j]:
                    i, j = np.random.choice(n, size=2, replace=False)
                # apply the swap in both x and h
                current_x[i], current_x[j] = current_x[j], current_x[i]
                current_h[i], current_h[j] = current_h[j], current_h[i]
                new_c5 = compute_total_overlap(current_x)
                
        else:  # random move
            i, j = np.random.choice(n, size=2, replace=False)
            while current_x[i] == current_x[j]:
                i, j = np.random.choice(n, size=2, replace=False)
            # apply the swap in both x and h
            current_x[i], current_x[j] = current_x[j], current_x[i]
            current_h[i], current_h[j] = current_h[j], current_h[i]
            new_c5 = compute_total_overlap(current_x)
        
        # compute temperature
        T = T0 - (T0 - T_last) * (idx / total_moves)
        
        # Accept or reject
        if new_c5 < current_C5:
            current_C5 = new_c5
            current_x = current_x.copy()
            current_h = current_h.copy()
            if new_c5 < best_C5:
                best_C5 = new_c5
                best_x = current_x.copy()
                best_h = current_h.copy()
        elif np.random.random() < np.exp((current_C5 - new_c5) / T):
            current_C5 = new_c5
            current_x = current_x.copy()
            current_h = current_h.copy()

        step_count += 1
        if step_count % acceptance_window == 0:
            if best_C5 < best_seen - 1e-5:
                best_seen = best_C5
            else:
                # Stagnation: perturb the type of best solution
                num_flip = int(0.05 * n)
                flip_indices = np.random.choice(n, num_flip, replace=False)
                # Flip bits 
                old_sum = np.sum(best_x)
                for idx in flip_indices:
                    best_x[idx] = 1 - best_x[idx]
                # Renormalize to correct sum
                new_sum = np.sum(best_x)
                if new_sum < old_sum:
                    # flip some more ones to zero 
                    idxs_to_zero = np.where(best_x == 1)[0]
                    to_change = old_sum - new_sum
                    flip_indices_more = np.random.choice(idxs_to_zero, to_change, replace=False)
                    for idx in flip_indices_more:
                        best_x[idx] = 0
                elif new_sum > old_sum:
                    # flip some zeros to one
                    idxs_to_one = np.where(best_x == 0)[0]
                    to_change = new_sum - old_sum
                    flip_indices_more = np.random.choice(idxs_to_one, to_change, replace=False)
                    for idx in flip_indices_more:
                        best_x[idx] = 1
                best_h = best_x.astype(float)
                best_C5 = compute_total_overlap(best_x)
                # Reset optimization tracker
                best_seen = best_C5

    # Final continuous refinement (SLSQP hybrid)
    # Start with continuous best from binary optimization
    h0 = best_h.copy()
    
    try:
        # First short SLSQP run
        res1 = minimize(
            lambda x: c5_from_corr(compute_corr_fft(x), n),
            h0,
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12, 'disp': False}
        )
        if res1.success:
            refined_h = res1.x
            refined_h = project_to_sum_box(refined_h, TARGET_SUM)
            # Second, smoother run
            # Smoothing filter
            smoothed_h = np.convolve(refined_h, np.ones(5)/5, mode='same')
            refined_h = project_to_sum_box(smoothed_h, TARGET_SUM)
            res2 = minimize(
                lambda x: c5_from_corr(compute_corr_fft(x), n),
                refined_h,
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12, 'disp': False}
            )
            if res2.success:
                final_h = res2.x
                final_h = project_to_sum_box(final_h, TARGET_SUM)
                final_c5 = c5_from_corr(compute_corr_fft(final_h), n)
                if final_c5 < best_C5:
                    best_C5 = final_c5
                    best_h = final_h.copy()
    except Exception as e:
        pass
    
    # Re-evaluate using the validator to make sure everything works
    final_h = best_h
    final_c5 = c5_from_corr(compute_corr_fft(final_h), n)
    
    return (final_h.tolist(), float(final_c5), n)
