import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def full_corr(a, b):
    """Compute full correlation with zero-padding using FFT"""
    L = 2 * len(a) - 1
    return np.fft.ifft(np.fft.fft(a, L) * np.conj(np.fft.fft(b, L))).real

def softmax(x, axis=None):
    """Numerically stable softmax with subtracting max for safety"""
    x_shifted = x - np.max(x, axis=axis, keepdims=True)
    exp_x = np.exp(x_shifted)
    return exp_x / np.sum(exp_x, axis=axis, keepdims=True)

def smooth_max_and_weights(c, alpha):
    """Compute log-sum-exp approximation of max and corresponding weights"""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    # Subtract max for numerical stability
    c_shifted = c - np.max(c)
    exp_c = np.exp(alpha * c_shifted)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    smooth_max = (np.log(sum_exp) + np.max(c)) / alpha
    return smooth_max, w

def gradient_of_smooth_max(h, alpha):
    """Compute gradient of smooth max of correlations"""
    c = full_corr(h, 1 - h)
    _, w = smooth_max_and_weights(c, alpha)
    
    # Symmetrize weights: w_sym[k] = w[k] + w[-k] where w[-k] = w[N-k]
    w_sym = w + w[::-1]
    
    # g = w_sym * h (correlation), but with zero padding in case of odd indices
    g = full_corr(w_sym, h)
    g = g[:len(h)]  # take only the first n_points entries
    g = np.clip(g, 0, 1)  # clip to avoid numerical issues
    
    grad = 1.0 - g
    return grad

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def objective_and_gradient(h, alpha):
    """Evaluate objective function and gradient for trust-constr"""
    c = full_corr(h, 1 - h)
    J, w = smooth_max_and_weights(c, alpha)
    J = J * dx  # scale by dx
    
    # Gradient computation
    w_sym = w + w[::-1]
    g = full_corr(w_sym, h)
    g = g[:len(h)]
    grad = (1.0 - g) * dx
    
    return J, grad

def optimize_with_continuation(initial_h, alpha_schedule):
    """Optimize with continuation schedule"""
    best_h = initial_h.copy()
    best_c5 = float('inf')
    
    for alpha in alpha_schedule:
        # Define problem data
        A_eq = np.array([np.ones(n_points)])
        b_eq = np.array([TARGET_SUM])
        bounds = [(0, 1) for _ in range(n_points)]
        
        def obj_and_grad(h):
            return objective_and_gradient(h, alpha)
        
        result = minimize(
            obj_and_grad,
            best_h.copy(),
            method='trust-constr',
            jac=True,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda x: A_eq @ x - b_eq},
            options={'disp': False}
        )
        
        if result.success:
            h_opt = result.x
        else:
            h_opt = best_h.copy()
            
        # Project to feasible set to maintain constraints
        h_projected = project_to_sum_box(h_opt, TARGET_SUM)
        
        # Evaluate true objective
        c_raw = full_corr(h_projected, 1 - h_projected)
        true_c5 = dx * np.max(c_raw)
        
        if true_c5 < best_c5:
            best_c5 = true_c5
            best_h = h_projected.copy()
    
    return best_h, best_c5
