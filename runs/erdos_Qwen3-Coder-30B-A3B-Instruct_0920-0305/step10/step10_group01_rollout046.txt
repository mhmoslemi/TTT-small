import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def smooth_max_surrogate(h, alpha=1.0):
    """Compute smooth max surrogate F_alpha(h) = (1/alpha) * log(sum(exp(alpha * c_k)))"""
    corr = compute_corr_fft(h)
    corr = corr[:len(h)]  # Only keep valid correlations
    c_k = corr  # These are the correlation values c_k = sum(h_i * (1 - h_{i+k}))
    # numerical stability: subtract max
    c_max = np.max(c_k)
    c_shifted = c_k - c_max
    exp_vals = np.exp(alpha * c_shifted)
    Z = np.sum(exp_vals)
    if Z <= 0:
        return float('inf')
    return (c_max + np.log(Z) / alpha) * dx

def compute_gradient(h, alpha=1.0):
    """Compute gradient of smooth max surrogate using FFT"""
    # c_k = sum_i h_i * (1 - h_{i+k}) = sum_i h_i - sum_i h_i * h_{i+k}
    corr = compute_corr_fft(h)
    c_k = corr[:len(h)]  # Only keep valid correlations
    c_max = np.max(c_k)
    c_shifted = c_k - c_max
    exp_vals = np.exp(alpha * c_shifted)
    Z = np.sum(exp_vals)
    if Z <= 0:
        return np.zeros_like(h)
    w = exp_vals / Z  # Normalized weights
    
    # Weighted correlation
    w_sym = w + w[::-1]
    # Shift w_sym to align with causal indexing
    w_sym_full = np.concatenate([w_sym[len(h)-1:], w_sym[:len(h)]])
    g_full = correlate(w_sym_full, h, mode='full')[:2*len(h)-1]
    g = g_full[len(h)-1:]
    
    # Gradient = 1 - g
    grad = np.ones_like(h) - g
    return grad * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    # Binary search for lambda
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    best_h_exact = None

        initial_h = np.interp(x_new, x_orig, initial_h)

    max_restarts = 100
    restart_count = 0
    
    # Run with multiple restarts
    while time.time() - start_time < budget_s - 5 and restart_count < max_restarts:
        # Choose init strategy at random
        strategy = np.random.choice(['binary', 'smooth', 'warm'])
        if strategy == 'binary':
            # Binary start
            h0 = np.zeros(n_points)
            sorted_idx = np.argsort(np.random.random(n_points))
            h0[sorted_idx[:n_points//2]] = 1.0
            h0 = project_to_sum_box(h0, TARGET_SUM)
        elif strategy == 'smooth':
            # Smooth random start
            h0 = np.random.random(n_points)
            h0 = project_to_sum_box(h0, TARGET_SUM)
        else:  # warm start
            h0 = initial_h.copy()
            h0 = project_to_sum_box(h0, TARGET_SUM)

                break
                
            # Use trust-constr optimizer
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
            
            def objective(h):
                return smooth_max_surrogate(h, alpha)
            
            def grad(h):
                return compute_gradient(h, alpha)

                    objective,
                    current_h,
                    method='trust-constr',
                    bounds=bounds,
                    constraints=cons,
                    jac=grad,

                    corr = compute_corr_fft(h_opt)

            # Refine using smoother
            alpha_schedule_fine = [0.5, 1.0, 2.0, 4.0, 8.0]

                    break
                bounds = [(0, 1) for _ in range(fine_points)]
                cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - fine_points / 2}
                
                def obj(h):
                    return smooth_max_surrogate(h, alpha)
                
                def grad(h):
                    return compute_gradient(h, alpha)
                
                try:
                    res = minimize(
                        obj,
                        h_fine,
                        method='trust-constr',
                        bounds=bounds,
                        constraints=cons,
                        jac=grad,
                        options={'maxiter': 100, 'gtol': 1e-8}
                    )
                    if res.success:
                        h_fine = project_to_sum_box(res.x, fine_points / 2)
                        # Final evaluation
                        corr = compute_corr_fft(h_fine)
                        c5_exact = np.max(corr[:len(h_fine)]) * (2.0 / fine_points)
                        if c5_exact < best_c5:
                            best_c5 = c5_exact
                            best_h = h_fine.copy()
                except Exception:
                    pass
    
    # Ensure we return what we have
    if best_h is None:
        best_h = project_to_sum_box(initial_h, TARGET_SUM)
        
    return (best_h.tolist(), float(best_c5), n_points)
