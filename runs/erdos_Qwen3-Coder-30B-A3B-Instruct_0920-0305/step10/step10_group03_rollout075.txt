# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        # Adaptive perturbation with temperature-like cooling

        # Create perturbation
        h_new = current_h.copy()
        
        # Select a few indices for perturbation (more adaptive)
        num_perturb = max(3, min(20, n_points // 20))

                h_new[idx] += delta
        
        # Project to valid constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new)
        
        # Acceptance probability based on metropolis criterion
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = current_h.copy()
                best_c5 = c5_new
        else:
            # Occasionally accept worse solutions based on temperature
            if np.random.random() < np.exp(-(c5_new - current_c5) / (1e-8 + temperature)):
                current_h = h_new.copy()
                current_c5 = c5_new
    
    return best_h, best_c5

        current_h = h0.copy()
        current_c5 = evaluate_c5(current_h)

                h_minus = h_step.copy()
                
                h_plus[i] = min(1.0, h_step[i] + step_size)

                c5_plus = evaluate_c5(project_to_sum_box(h_plus, TARGET_SUM))

                elif c5_minus < current_c5:
                    h_step[i] = h_minus[i]

            new_c5 = evaluate_c5(new_h)
            
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
            else:
                break  # no improvement, quit

    except ImportError:
        # Fallback to local search if cvxpy not available
        return local_search_improved(h0, max_iter // 2)
