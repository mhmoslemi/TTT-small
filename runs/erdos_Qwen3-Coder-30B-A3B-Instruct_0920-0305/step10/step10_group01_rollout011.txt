import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h, L) * np.conj(np.fft.fft(b, L))).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

            corr = compute_corr_fft(current_h)
            
            # Compute softmax weights
            c_max = np.max(corr)
            exp_term = np.exp(alpha * (corr - c_max))
            w = exp_term / np.sum(exp_term)
            
            # Compute w_sym
            w_sym = w + w[::-1]
            
            # Compute correlation of w_sym with h (this gives the gradient components)
            L = 2 * n_points - 1
            fw_sym = np.fft.fft(w_sym, L)

            r = np.fft.ifft(fw_sym * np.conj(fh)).real
            
            # Gradient: 1 - r[:n_points]

            step_size = 0.05 / (1.0 + iter_count / 500.0)

            if time.time() - start_time >= budget_s:
