def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5_exact(h):
    """Evaluate exact C5 using FFT correlation"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def compute_smooth_max(h, alpha):
    """Compute smooth max surrogate"""
    corr = compute_corr_fft(h)
    max_c = np.max(corr)
    # Numerically stable log-sum-exp
    exp_corr = np.exp(alpha * (corr - max_c))
    return (max_c + np.log(np.sum(exp_corr)) / alpha) * dx

def compute_gradient(h, alpha):
    """Compute gradient of smooth max surrogate"""
    corr = compute_corr_fft(h)
    max_c = np.max(corr)
    exp_corr = np.exp(alpha * (corr - max_c))
    w = exp_corr / np.sum(exp_corr)
    
    # Build symmetrized weights
    w_sym = w + w[::-1]
    # Convolve with h to get g
    g_full = correlate(w_sym, h, mode='full')
    g = g_full[len(h)-1:len(h)+len(h)-1]
    
    return 1 - g

        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

        m = (a + b) / 2

            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
