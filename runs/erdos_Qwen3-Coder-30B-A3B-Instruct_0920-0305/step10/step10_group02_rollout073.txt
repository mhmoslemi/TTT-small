TARGET_SUM = None

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_descent_step(current_h, current_c5, start_time, budget_s):
    """Perform a gradient descent step"""
    if time.time() - start_time > budget_s - 1:
        return current_h, current_c5

    # Simple gradient approximation using finite differences
    h = current_h.copy()
    eps = 1e-6
    c5_ref = evaluate_c5(h)
    
    # Approximate gradient using finite differences
    grad = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_sum_box(h_plus, TARGET_SUM)
        c5_plus = evaluate_c5(h_plus)
        grad[i] = (c5_plus - c5_ref) / eps

    h_new = h - step_size * grad
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    c5_new = evaluate_c5(h_new)
    
    return h_new, c5_new

def hill_climb_step(current_h, current_c5, start_time, budget_s):

    if time.time() - start_time > budget_s - 1:
        return current_h, current_c5

    h = current_h.copy()
    best_h = h.copy()
    best_c5 = current_c5
    
    # Try small perturbations

        h_new = h.copy()

        delta = np.random.uniform(-0.05, 0.05)
        h_new[idx] += delta
        
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
    
    return best_h, best_c5

def simulated_annealing_step(current_h, current_c5, start_time, budget_s):
    """Perform simulated annealing step"""
    if time.time() - start_time > budget_s - 1:
        return current_h, current_c5

    h = current_h.copy()
    c5 = current_c5
    
    # Cool down schedule
    temp = max(0.001, 0.1 * np.exp(-0.0005 * time.time()))
    
    # Perturb a few points
    h_new = h.copy()
    n_perturb = min(20, n_points // 10)
    indices = np.random.choice(n_points, n_perturb, replace=False)
    
    for idx in indices:
        delta = np.random.normal(0, 0.03)
        h_new[idx] += delta
    
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    c5_new = evaluate_c5(h_new)
    
    # Accept with Metropolis criterion
    if c5_new < c5 or np.random.random() < np.exp(-(c5_new - c5) / temp):
        return h_new, c5_new
    else:
        return h, c5
