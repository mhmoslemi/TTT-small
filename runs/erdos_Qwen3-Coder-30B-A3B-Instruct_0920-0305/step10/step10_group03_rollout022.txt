def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_binary_milp(N, time_limit=30):
    """Solve binary MILP for the Erdos problem"""
    try:
        x = cp.Variable(N, boolean=True)
        t = cp.Variable()
        
        # Sum constraint
        constraints = [cp.sum(x) == N // 2]
        
        # For each shift k, compute the overlap
        for k in range(N):
            # We only consider valid indices for shift k
            if k == 0:
                continue
            # Create binary variables for products
            y = cp.Variable(N - k, boolean=True)
            # Add the McCormick constraints
            for i in range(N - k):
                constraints.extend([
                    y[i] <= x[i],
                    y[i] <= x[i + k],
                    y[i] >= x[i] + x[i + k] - 1,
                ])
            # Add constraint: sum of products >= N/2 - t
            constraints.append(cp.sum(y) >= N // 2 - t)
        
        # Objective
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        # Solve with time limit
        prob.solve(solver=cp.GLPK_MI, time_limit=time_limit)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            solution = x.value
            return solution
    except Exception as e:
        return None
    return None

def upsample_binary(binary_vec, target_N):
    """Upsample binary vector to target_N"""
    if len(binary_vec) == 0:
        return []
    # Duplicate each element
    upsampled = np.repeat(binary_vec, target_N // len(binary_vec))
    # If we need extra elements, pad with last value
    if len(upsampled) < target_N:
        upsampled = np.pad(upsampled, (0, target_N - len(upsampled)), constant_values=upsampled[-1])
    elif len(upsampled) > target_N:
        upsampled = upsampled[:target_N]
    return upsampled

def differential_evolution_optimize(h0, target_sum, time_budget, start_time):
    """Optimize using differential evolution"""
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(len(h0))]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Run for limited time
    result = differential_evolution(
        objective, 
        bounds, 
        constraints=cons,
        maxiter=int(1000 * time_budget / 600),
        seed=42,
        popsize=20,
        mutation=(0.5, 1.0),
        recombination=0.7,
        disp=False,
        callback=None
    )
    
    # Project to valid constraints
    if result.success:
        return project_to_sum_box(result.x, target_sum), evaluate_c5(result.x)
    else:
        return h0, evaluate_c5(h0)

def local_hill_climb(h0, target_sum, time_budget, start_time):
    """Perform local hill climbing"""
    current_h = h0.copy()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    max_iter = int(10000 * time_budget / 600)
    for i in range(max_iter):
        if time.time() - start_time > time_budget - 1:
            break
            
        # Try small perturbations
        h_new = current_h.copy()
        idx1 = np.random.randint(len(h_new))
        idx2 = np.random.randint(len(h_new))
        
        # Swap values with small perturbation
        delta = np.random.normal(0, 0.01)
        h_new[idx1] = np.clip(h_new[idx1] + delta, 0, 1)
        h_new[idx2] = np.clip(h_new[idx2] - delta, 0, 1)
        
        # Project to sum constraint
        h_new = project_to_sum_box(h_new, target_sum)
        
        # Evaluate
        c5_new = evaluate_c5(h_new)
        
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        elif np.random.random() < 0.05:  # Occasionally accept worse solutions
            current_h = h_new.copy()
            current_c5 = c5_new
    
    return best_h, best_c5
