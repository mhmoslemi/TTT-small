import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_surrogate(h, alpha):
    """Compute smooth max surrogate with temperature alpha"""
    corr = compute_corr_fft(h)
    # Clamp negative values to prevent numerical issues
    corr = np.maximum(corr, 0.0)
    max_c = np.max(corr)
    corr_shifted = corr - max_c
    exp_corr = np.exp(alpha * corr_shifted)
    Z = np.sum(exp_corr)
    return (max_c + np.log(Z) / alpha)

def smooth_max_gradient(h, alpha):
    """Compute gradient of smooth max surrogate"""
    corr = compute_corr_fft(h)
    max_c = np.max(corr)
    corr_shifted = corr - max_c
    exp_corr = np.exp(alpha * corr_shifted)
    Z = np.sum(exp_corr)
    weights = exp_corr / Z
    
    # Symmetrize weights
    weights_sym = weights + weights[::-1]
    
    # Compute convolved values
    g_full = correlate(weights_sym, h, mode='full')
    g = g_full[len(h)-1:len(h)+len(h)-1]
    
    return 1.0 - g

        else:
            # Random continuous start
            h0 = np.random.uniform(0, 1, n_points)
            h0 = project_to_sum_box(h0, TARGET_SUM)
            
        current_h = h0.copy()
        current_c5 = evaluate_c5(h0)
