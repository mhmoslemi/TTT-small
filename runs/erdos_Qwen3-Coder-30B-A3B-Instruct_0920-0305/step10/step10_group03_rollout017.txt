import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

# Global variables for problem setup
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water-filling"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_binary_milp(n, target_sum):
    """Solve the binary MILP to get a skeleton"""
    # Variables
    x = cp.Variable(n, integer=True)
    y = cp.Variable((n, n), integer=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(x) == target_sum,
        cp.bounds(x, 0, 1)
    ]
    
    # Add correlation inequalities
    for k in range(1, n):
        for i in range(n - k):
            constraints.append(y[i, k] <= x[i])
            constraints.append(y[i, k] <= x[i + k])
            constraints.append(y[i, k] >= x[i] + x[i + k] - 1)
            
    # Worst-case bound
    for k in range(1, n):
        constraints.append(cp.sum(y[:, k]) >= target_sum - t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.GLPK_MI, verbose=False, time_limit=30)
        if prob.status in [cp.OPTIMAL, cp.OPT_FEASIBLE]:
            return x.value
    except:
        pass
    # Fallback to random solution
    return np.random.choice([0, 1], size=n, p=[0.5, 0.5])

def binary_swap_search(h, max_steps=5000):
    """Perform swap search to improve binary solution"""
    h = h.copy()
    # Compute initial correlations
    n = len(h)
    corr = np.zeros(n)
    for k in range(n):
        corr[k] = np.sum(h[:-k] * h[k:]) if k > 0 else np.sum(h * h)
    # Convert to counts (multiply by dx, but we work with integer counts)
    counts = np.round(corr / dx).astype(int)
    
    max_iter = max_steps
    improvement_count = 0
    for _ in range(max_iter):
        # Find worst shift
        worst_k = np.argmax(counts)
        if counts[worst_k] >= n / 2:
            break
        
        # Find candidates for swap
        A = set()
        B = set()
        for i in range(n - worst_k):
            if h[i] == 1 and h[i + worst_k] == 0:
                A.add(i)
            if h[i] == 0 and h[i + worst_k] == 1:
                B.add(i)
        
        if not A or not B:
            # Random swap if no good pairs
            i = np.random.randint(n)
            j = np.random.randint(n)
            if h[i] == 1 and h[j] == 0:
                h[i] = 0
                h[j] = 1
                # Update correlations
                for k in range(1, n):
                    if i + k < n:
                        counts[k] += (h[i + k] - h[i + k] * 2)
                    if j + k < n:
                        counts[k] += (h[j + k] - h[j + k] * 2)
                improvement_count += 1
        else:
            # Best swap
            a = np.random.choice(list(A))
            b = np.random.choice(list(B))
            h[a] = 0
            h[b] = 1
            # Update correlations (simplified)
            for k in range(1, n):
                if a + k < n and b + k < n:
                    counts[k] += (h[a + k] - h[a + k] * 2) + (h[b + k] - h[b + k] * 2)
            improvement_count += 1
    
    return h

def differential_evolution(h0, max_iter=500, pop_size=30):
    """Run differential evolution on full resolution"""
    def evaluate(h):
        return evaluate_c5(h)
    
    n = len(h0)
    # Decode binary if needed
    if h0.dtype in [np.bool_, np.int32, np.int64]:
        h0 = h0.astype(float)
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = h0.copy()
        h += np.random.normal(0, 0.02, n)
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    
    best_h = population[0].copy()
    best_c5 = evaluate(best_h)
    
    tolerance = 1e-8
    for gen in range(max_iter):
        progress = gen / max_iter
        if progress > 0.95 and gen > max_iter // 2:
            break
        
        for i in range(pop_size):
            # Select three distinct individuals
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            F = 0.8
            m = a + F * (b - c)
            m = np.clip(m, 0, 1)
            
            # Crossover
            CR = 0.9
            h = population[i].copy()
            for j in range(n):
                if np.random.rand() < CR or j == np.random.randint(n):
                    h[j] = m[j]
            
            # Re-project and evaluate
            h = project_to_sum_box(h, TARGET_SUM)
            c5 = evaluate(h)
            
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
            
            # Selection
            if c5 < evaluate(population[i]):
                population[i] = h.copy()
    
    return best_h, best_c5

def simulated_annealing(h0, max_iter=2000):
    """Simulated annealing refinement"""
    def evaluate(h):
        return evaluate_c5(h)
    
    T = 0.1
    alpha = 0.99
    h = h0.copy()
    c5 = evaluate(h)
    best_h = h.copy()
    best_c5 = c5
    
    for _ in range(max_iter):
        # Pick two indices
        i = np.random.randint(len(h))
        j = np.random.randint(len(h))
        
        # Make a small change
        delta = np.random.normal(0, 0.05)
        old_i = h[i]
        old_j = h[j]
        
        h[i] += delta
        h[j] -= delta
        
        # Projection to [0,1]
        h[i] = max(0, min(1, h[i]))
        h[j] = max(0, min(1, h[j]))
        
        # Re-project to maintain sum
        h = project_to_sum_box(h, TARGET_SUM)
        
        new_c5 = evaluate(h)
        delta_e = new_c5 - c5
        
        if delta_e < 0 or np.exp(-delta_e / T) > np.random.rand():
            c5 = new_c5
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        else:
            # Rollback
            h[i] = old_i
            h[j] = old_j
        
        T *= alpha
        
        if np.abs(delta_e) < 1e-10:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Set parameters
    TARGET_SUM = n_points / 2
    n_small = 60  # For binary optimization
    
    # Phase A: Solve binary MILP
    h_binary = solve_binary_milp(n_small, n_small // 2)
    h_binary = h_binary.astype(float)
    
    # Phase B: Upsample to full resolution
    h_initial = np.repeat(h_binary, n_points // n_small)
    
    # Phase C: Optional worst-shift swap search
    if time.time() - start_time < budget_s * 0.5:
        h_initial = binary_swap_search(h_initial, 1000)
    
    # Phase D: Differential Evolution
    h_de, c5_de = differential_evolution(h_initial, 300)
    
    # Phase E: Simulated Annealing
    h_sa, c5_sa = simulated_annealing(h_de, 2000)
    
    # Phase F: Hill Climbing
    h_final = h_sa.copy()
    c5_final = c5_sa
    
    # Final hill climbing
    for _ in range(2000):
        if time.time() - start_time > budget_s - 30:
            break
        # Small random perturbations
        h_trial = h_final.copy()
        n_perturb = max(1, n_points // 20)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for idx in indices:
            delta = np.random.normal(0, 0.01)
            h_trial[idx] += delta
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = project_to_sum_box(h_trial, TARGET_SUM)
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < c5_final:
            h_final = h_trial.copy()
            c5_final = c5_trial
    
    # Final evaluation
    final_c5 = evaluate_c5(h_final)
    
    return (h_final.tolist(), float(final_c5), n_points)
