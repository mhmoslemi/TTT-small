import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.fft import fft, ifft
import time

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
n_half = n_points // 2

def project_to_sum_box(u, target_sum):
    """Project u onto box constraints [0,1] with sum = target_sum using water-filling."""
    def f(lam):
        projected = np.clip(u + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(u + (a + b)/2, 0, 1)
    return result

def autocorr_fft(h):
    """Compute aperiodic autocorrelation using FFT."""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(h, L)).real
    return corr

def exact_c5(u):
    """Compute exact C5 from half-vector u."""
    h = np.concatenate([u, u[::-1]])
    corr = autocorr_fft(h)
    return 1 - dx * np.min(corr)

def lse_c5(u, beta):
    """Compute smooth surrogate of C5 using log-sum-exp."""
    h = np.concatenate([u, u[::-1]])
    corr = autocorr_fft(h)
    return 1 - (dx / beta) * np.log(np.sum(np.exp(beta * corr)))

def grad_lse_c5(u, beta):
    """Gradient of smooth surrogate using FFT."""
    h = np.concatenate([u, u[::-1]])
    corr = autocorr_fft(h)
    w = np.exp(beta * corr)
    w_sum = np.sum(w)
    
    # Compute convolution of w with h_rev
    w_h_conv = ifft(fft(w, 2*len(h)-1) * fft(h[::-1], 2*len(h)-1)).real
    grad = -dx * (w_h_conv[:len(u)] + w_h_conv[-len(u):]) / w_sum
    return grad

def parametric_to_halfvector(params, n):
    """Build half-vector from parametric shape."""
    if len(params) == 2:  # 3-level shape: a, p
        a, p = params
        u = np.ones(n // 2)
        u[:int(a * n_half)] = 1.0
        u[int(a * n_half):] = p
        return u
    elif len(params) == 3:  # 5-level shape: a, r, p
        a, r, p = params
        # Build five zones: [0,a), [a,a+r), [a+r,1-a-r), [1-a-r,1-a), [1-a,1)
        u = np.ones(n_half)
        a_idx = int(a * n_half)
        r_idx = int(r * n_half)
        u[a_idx:a_idx+r_idx] = 1.0
        u[a_idx+r_idx:1-a_idx-r_idx] = p
        u[1-a_idx-r_idx:1-a_idx] = 1.0
        return u
    else:
        raise ValueError("Unsupported parameter count")

def run_de():
    """Run differential evolution on parametric space."""
    def objective(params):
        u = parametric_to_halfvector(params, n_half)
        # Adjust target sum for the half-vector
        target_sum = n_half / 2
        u = project_to_sum_box(u, target_sum)
        return exact_c5(u)
    
    bounds = [(0.05, 0.45), (0, 0.3)]  # a in [0.05, 0.45], r in [0, 0.3]
    result = differential_evolution(
        objective, 
        bounds, 
        seed=42, 
        maxiter=200, 
        popsize=15 * len(bounds),
        disp=False
    )
    return result.x

def run_lbfgsb(u0, beta):
    """Run L-BFGS-B with projected gradient on full discretization."""
    def objective(u):
        return lse_c5(u, beta)
    
    def grad(u):
        return grad_lse_c5(u, beta)
    
    # Project initial point to feasible set
    u0 = project_to_sum_box(u0, n_half / 2)
    
    res = minimize(
        objective,
        u0,
        method='L-BFGS-B',
        jac=grad,
        bounds=[(0, 1) for _ in range(len(u0))],
        options={'maxiter': 800, 'gtol': 1e-10, 'ftol': 1e-12}
    )
    
    # Final projection to ensure exact constraint satisfaction
    final_u = project_to_sum_box(res.x, n_half / 2)
    return final_u

def run_sa(u_best):
    """Simulated annealing polishing."""
    current_u = u_best.copy()
    current_c5 = exact_c5(current_u)
    best_u = current_u.copy()
    best_c5 = current_c5
    
    T = 5e-3
    alpha = 0.9997
    delta_max = 0.1
    acc_ratio = 0.0
    total_moves = 0
    accepted_moves = 0
    
    while T > 1e-6:
        i, j = np.random.choice(len(current_u), size=2, replace=False)
        delta = np.random.uniform(0, delta_max)
        
        if current_u[i] >= delta and current_u[j] <= 1 - delta:
            u_new = current_u.copy()
            u_new[i] -= delta
            u_new[j] += delta
            # Project to sum constraint
            u_new = project_to_sum_box(u_new, n_half / 2)
            c5_new = exact_c5(u_new)
            
            if c5_new < current_c5:
                current_u = u_new.copy()
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_u = u_new.copy()
                    best_c5 = c5_new
                accepted_moves += 1
            else:
                p = np.exp(-(c5_new - current_c5) / T)
                if np.random.rand() < p:
                    current_u = u_new.copy()
                    current_c5 = c5_new
                    accepted_moves += 1
            
            total_moves += 1
            if total_moves % 500 == 0:
                acc_ratio = accepted_moves / total_moves
                if acc_ratio < 0.2:
                    delta_max *= 0.95
                elif acc_ratio > 0.4:
                    delta_max *= 1.05
    
    return best_u, best_c5

def run(seed=42, budget_s=600):
    start_time = time.time()
    
    # Phase 1: Parametric sweep (fast)
    best_c5 = float('inf')
    best_u = None
    best_params = None
    
    print("Starting parametric sweep...")
    a_values = np.linspace(0.05, 0.45, 400)
    best_a = 0.2
    best_p = 0.5
    for i, a in enumerate(a_values):
        params = [a, 0.5]
        u = parametric_to_halfvector(params, n_half)
        u = project_to_sum_box(u, n_half / 2)
        c5 = exact_c5(u)
        if c5 < best_c5:
            best_c5 = c5
            best_u = u.copy()
            best_params = params.copy()
    
    print(f"After sweep: C5 = {best_c5:.6f}")
    
    # Phase 2: DE search (200 iterations, ~40% of budget)
    early_term = False
    if time.time() - start_time > budget_s * 0.3:
        early_term = True
    else:
        print("Running DE optimization...")
        de_params = run_de()
        de_u = parametric_to_halfvector(de_params, n_half)
        de_u = project_to_sum_box(de_u, n_half / 2)
        de_c5 = exact_c5(de_u)
        if de_c5 < best_c5:
            best_c5 = de_c5
            best_u = de_u.copy()
            best_params = de_params.copy()
    
    # Phase 3: High resolution refinement
    if time.time() - start_time > budget_s * 0.7 or early_term:
        pass
    else:
        print("Performing high resolution refinement...")
        # Upsample to finer grid
        n_fine = 240
        u_fine = np.interp(np.linspace(0, 1, n_fine // 2),
                          np.linspace(0, 1, n_half),
                          best_u)
        # Run L-BFGS-B
        u_final = run_lbfgsb(u_fine, beta=200)
        final_c5 = exact_c5(u_final)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_u = u_final.copy()
    
    # Phase 4: SA polishing
    if time.time() - start_time > budget_s * 0.9:
        pass
    else:
        print("Running simulated annealing...")
        try:
            sa_u, sa_c5 = run_sa(best_u)
            if sa_c5 < best_c5:
                best_c5 = sa_c5
                best_u = sa_u
        except:
            pass
    
    print(f"Final result: C5 = {best_c5:.6f}")
    h_full = np.concatenate([best_u, best_u[::-1]])
    return (h_full.tolist(), float(best_c5), n_points)
