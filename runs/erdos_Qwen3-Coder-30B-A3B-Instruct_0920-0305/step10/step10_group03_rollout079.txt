<strategy>
1. **Binary MILP for Skeleton Construction**: Use mixed-integer linear programming to find a binary skeleton of the optimal step function. This involves deciding which intervals should be "on" (1) or "off" (0) in a coarse discretization.

2. **Parametric Shape Search**: Identify promising piecewise constant shapes by sweeping parameters like number of steps, full-width-at-half-maximum, etc., in a structured way. Use a coarse grid search over shape parameters, guided by time budgets.

3. **Worst-Shift Local Search with Binary Constraints**: Focus search on moves that target the maximum correlation peak. Introduce specialized moves that swap between 0 and 1 in the binary skeleton to reduce the peak value.

4. **Differential Evolution with Bound Constraints**: Apply a global evolutionary algorithm that operates on continuous relaxations of the final solution. Use bounds on values to ensure feasibility and maintain sum constraint.

5. **Simulated Annealing Polish**: Refine at end with SA that samples around the final solution to make small swaps that reduce the overlap.

6. **Time Management**: Allocate time to:
   - 20% for MILP skeleton
   - 30% for parametric shape search 
   - 25% for worst-shift local search
   - 15% for DE
   - 10% for SA polish
   - Remaining for cleanup and final optimization

7. **Failure Modes**: The binary MILP might get stuck in infeasible regions; we will use iterative refinement. Parametric search may miss shape; we will use multiple sweeps. Local search depends on initial solution; we will try multiple warm starts.

8. **Key Differences**: 
   - Faithful exploration of shape space with multiple algorithmic components.
   - Focused local search with theoretical peak targeting.
   - Use of discrete optimization tools via MILP.

import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import Minimize, Problem
from cvxpy.errors import DCPError

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_milp_skeleton():
    """Use a MILP to generate a binary skeleton for step function"""
    # We discretize to coarser resolution
    n_binary = 20  # Coarse discretization
    x_bin = np.linspace(0, 2, n_binary + 1)
    dx_bin = x_bin[1] - x_bin[0]
    
    # Binary variables
    h_bin = cp.Variable(n_binary, boolean=True)
    
    # Sum constraint
    constraints = [cp.sum(h_bin) == n_binary / 2]
    
    # We want to avoid suspicious correlations by maximizing minimum distance between 1s
    # This is a heuristic, we do not model the actual C5 in MILP
    
    # Solve relaxation
    objective = Minimize(cp.sum(h_bin))
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.GLPK_MI)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            sol = h_bin.value
            # Expand to full resolution
            h_full = np.interp(np.linspace(0, 2, n_points), x_bin[:-1], sol)
            # Interpolate to step function
            h_full = np.array([h_full[i] for i in range(n_points)])
            return h_full
    except:
        pass
    
    # Fallback
    return np.ones(n_points) * 0.5

def parametric_shape_search():
    """Try a variety of parametric forms with step function"""
    # Try many shapes and see if any are promising
    best_c5 = float('inf')
    best_h = None
    
    # Generate shapes with different number of steps
    for n_steps in [5, 10, 15, 20]:
        # Create a step function - each step will have same width
        step_width = 2.0 / n_steps
        heights = np.linspace(0, 1, n_steps + 1)  # 0 to 1
    
        # Try different interlacing
        for i in range(0, n_steps, 2):
            h = np.ones(n_points)
            for j in range(n_steps):
                start_idx = int(j * n_points / n_steps)
                end_idx = int((j + 1) * n_points / n_steps)
                if j % 2 == 0:
                    h[start_idx:end_idx] = heights[j]
                else:
                    h[start_idx:end_idx] = heights[j+1]
            # Adjust to correct mean
            h = project_to_sum_box(h, TARGET_SUM)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h if best_h is not None else np.ones(n_points) * 0.5

def worst_shift_local_search(h0, iterations=5000):
    """Improve solution by focusing on worst correlation peak"""
    current_h = h0.copy()
    best_h = current_h.copy()
    best_c5 = evaluate_c5(current_h)
    
    # Compute current correlation
    corr = compute_corr_fft(current_h)
    max_corr_idx = np.argmax(corr)
    
    for i in range(iterations):
        # Perturb binary positions near worst correlation
        h_new = current_h.copy()
        
        # Try flipping a few values near peak
        start = max(0, max_corr_idx - 50)
        end = min(len(h_new), max_corr_idx + 50)
        
        # Choose indices to flip based on magnitude of peak
        indices = np.random.choice(np.arange(start, end), size=min(5, end - start), replace=False)
        for idx in indices:
            # Toggle small perturbation
            h_new[idx] = 1 - h_new[idx]
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            current_h = h_new.copy()
            best_h = current_h.copy()
            best_c5 = c5_new
            # Recompute correlation
            corr = compute_corr_fft(current_h)
            max_corr_idx = np.argmax(corr)
    
    return best_h, best_c5

def differential_evolution_optimization(h0, maxiter=2000):
    """Global optimization using DE, but respecting constraints"""
    # DE like opt on real-valued solution
    num_points = len(h0)
    bounds = [(0, 1) for _ in range(num_points)]
    
    def objective(x):
        x_proj = project_to_sum_box(x, TARGET_SUM)
        return evaluate_c5(x_proj)
    
    # Scipy DE
    from scipy.optimize import differential_evolution
    try:
        result = differential_evolution(objective, bounds, maxiter=maxiter, seed=42, disp=False)
        if result.success:
            x_opt = project_to_sum_box(result.x, TARGET_SUM)
            return x_opt, evaluate_c5(x_opt)
    except:
        pass
    
    return h0, evaluate_c5(h0)

def simulated_annealing_polish(h0, max_iter=1000):
    """Use SA to improve final solution"""
    current_h = h0.copy()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    temp = 1.0
    cooling_rate = 0.999
    min_temp = 1e-5
    
    for _ in range(max_iter):
        # Neural-style neighbor - small Gaussian perturbations
        h_new = current_h + np.random.normal(0, 0.01, len(current_h))
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject based on temperature
        if c5_new < current_c5 or np.random.rand() < np.exp((current_c5 - c5_new) / temp):
            current_h = h_new.copy()
            current_c5 = c5_new
        
        if c5_new < best_c5:
            best_h = current_h.copy()
            best_c5 = c5_new
            
        temp *= cooling_rate
        if temp < min_temp:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Strategy 1: Try binary MILP skeleton
    h1 = binary_milp_skeleton()
    c1 = evaluate_c5(h1)
    
    # Strategy 2: Try parametric shape search
    h2 = parametric_shape_search()
    c2 = evaluate_c5(h2)
    
    # Final best solution
    if c1 < c2:
        h_best = h1
        c_best = c1
    else:
        h_best = h2
        c_best = c2
    
    # Local improvement
    h_best, c_best = worst_shift_local_search(h_best, 3000)
    
    # DE refinement
    time_left = budget_s - (time.time() - start_time)
    if time_left > 10:
        h_de, c_de = differential_evolution_optimization(h_best, maxiter=int(time_left/4))
        if c_de < c_best:
            h_best = h_de
            c_best = c_de
    
    # SA polish
    h_best, c_best = simulated_annealing_polish(h_best, 1000)
    
    # Final projection to constraints
    h_best = project_to_sum_box(h_best, TARGET_SUM)
    c_best = evaluate_c5(h_best)
    
    return (h_best.tolist(), float(c_best), n_points)
