import numpy as np
from scipy.signal import correlate

import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Use sorting algorithm for simplex projection
    sorted_indices = np.argsort(-x)  # descending order
    cumsum = np.cumsum(x[sorted_indices])
    gamma = (target_sum - cumsum[-1]) / len(x)
    if gamma > 0:
        # Need to adjust
        new_x = x.copy()
        for i in range(len(x)):
            idx = sorted_indices[i]
            if x[idx] <= gamma:
                new_x[idx] = 0
            else:
                new_x[idx] = x[idx] - gamma
                if i < len(x) - 1:
                    gamma = (target_sum - cumsum[i]) / (len(x) - i - 1)
        return new_x
    # If gamma <= 0, we just need to clip to [0,1] and rescale
    return np.clip(x, 0, 1)

def binary_milp(N, time_limit=60):
    """Solve binary MILP for small N"""
    try:
        # Create binary variables
        x = cp.Variable(N, boolean=True)
        t = cp.Variable()
        
        # Constraint: sum of x = N/2
        constraints = [cp.sum(x) == N // 2]
        
        # For each shift k from 0 to N-1
        # We need to compute overlap sum_{i=0}^{N-1-k} x_i * x_{i+k}
        # Add constraint sum y_{i,k} >= N/2 - t for each k
        for k in range(N):
            # For shift k, compute the sum of y_{i,k} for i from 0 to N-1-k
            if k == 0:
                # Shift 0, overlap is sum of x_i * x_i = sum of x_i = N/2
                # So constraint is automatically satisfied
                continue
            else:
                # Create y variables for this shift
                y = cp.Variable(N - k, boolean=True)
                # y_{i,k} <= x[i], y_{i,k} <= x[i+k]
                # y_{i,k} >= x[i] + x[i+k] - 1
                for i in range(N - k):
                    constraints.extend([
                        y[i] <= x[i],
                        y[i] <= x[i + k],
                        y[i] >= x[i] + x[i + k] - 1
                    ])
                # Sum constraint for this shift
                constraints.append(cp.sum(y) >= N // 2 - t)
        
        # Objective: minimize t
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        # Try solving with CBC (more robust than GLPK for small problems)
        # Use a smaller timeout to give results quickly for bigger problems
        prob.solve(solver=cp.CBC, maximumSeconds=time_limit)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value
        else:
            return None
    except:
        return None

def upsample(binary_vec, target_N):
    """Upsample binary vector to target_N by repeating each element"""
    # Repeat each element
    upsampled = np.repeat(binary_vec, target_N // len(binary_vec))
    # Trim to exact length
    upsampled = upsampled[:target_N]
    return upsampled.astype(float)

        x_proj = project_to_sum_box(x, target_sum)
        return evaluate_c5(x_proj)
    
    # Use a custom wrapper to avoid function-call overhead in DE
    res = differential_evolution(
        minimize_func,
        bounds,
        maxiter=int(5000 * (time_budget / 10000)),
        popsize=20,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=42,
        updating='immediate',
        workers=1,
        disp=False
    )
    
    if res.success:
        h_opt = project_to_sum_box(res.x, target_sum)
        return h_opt, evaluate_c5(h_opt)
    else:
        return h_initial, evaluate_c5(h_initial)

def local_hill_climb(h_initial, target_sum, time_budget, start_time):
    """Perform local hill climbing search"""
    current_h = h_initial.copy()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Generate a neighbor by swapping two values
    def generate_neighbor(h, i, j):
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        h_new = project_to_sum_box(h_new, target_sum)
        return h_new
    
    iter_count = 0
    max_iter = int(10000 * (time_budget / 200))
    
    while iter_count < max_iter and time.time() - start_time < time_budget - 1:
        i = np.random.randint(0, len(h_initial))
        j = np.random.randint(0, len(h_initial))
        if i != j:
            h_new = generate_neighbor(h_initial, i, j)
            c5_new = evaluate_c5(h_new)
            if c5_new < current_c5:
                current_h = h_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
            elif np.random.random() < 0.01:  # Occasionally accept worse solutions
                current_h = h_new
                current_c5 = c5_new
        iter_count += 1
    return best_h, best_c5
