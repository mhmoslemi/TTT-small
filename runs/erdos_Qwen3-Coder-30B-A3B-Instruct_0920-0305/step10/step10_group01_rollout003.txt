import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def full_corr(a, b):
    """Compute full correlation with zero padding"""
    L = 2 * len(a) - 1
    return np.fft.ifft(np.fft.fft(a, L) * np.conj(np.fft.fft(b, L))).real

def smooth_max_and_weights(c, alpha):
    """Compute smooth max and soft-max weights with numerical stability"""
    c_shifted = c - np.max(c)
    exp_c = np.exp(alpha * c_shifted)
    w = exp_c / np.sum(exp_c)
    smooth_max = np.log(np.sum(exp_c)) / alpha + np.max(c)
    return smooth_max, w

def compute_gradient(h, alpha):
    """Compute gradient of smooth max objective using FFT"""
    c = full_corr(h, 1 - h)
    _, w = smooth_max_and_weights(c, alpha)
    
    # Symmetrize weights (because correlation is symmetric)
    w_sym = w + w[::-1]
    # Compute the correlation of w_sym with h
    g = full_corr(w_sym, h)
    # Return gradient: 1 - correlation
    return 1.0 - g[:len(h)]

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Compute the true C5 value"""
    c = full_corr(h, 1 - h)
    return dx * np.max(c)
