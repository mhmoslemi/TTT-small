import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def full_corr(a, b):
    """Compute full correlation with zero padding: returns array of length 2*N-1"""
    L = 2 * len(a) - 1
    return np.fft.ifft(np.fft.fft(a, L) * np.conj(np.fft.fft(b, L))).real

def smooth_max_and_weights(c, alpha):
    """Compute smooth max using log-sum-exp with softmax weights"""
    c_shifted = c - np.max(c)  # for numerical stability
    exp_c = np.exp(alpha * c_shifted)
    Z = np.sum(exp_c)
    w = exp_c / Z
    smooth_max = np.log(Z) / alpha + np.max(c)
    return smooth_max, w

def compute_gradient(h, alpha):
    """Compute gradient of smooth max with respect to h"""
    # Full correlation c_k
    c = full_corr(h, 1 - h)
    # Softmax weights w_k
    _, w = smooth_max_and_weights(c, alpha)
    # Symmetrize weights: w_sym_k = w_k + w_{-k}
    w_sym = w + w[::-1]
    # Correlation with h to get gradient (first N entries)
    g = full_corr(w_sym, h)[:len(h)]
    return 1.0 - g

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using full correlation"""
    corr = full_corr(h, 1-h)
    return dx * np.max(corr)

def objective_with_gradient(h, alpha):
    """Objective function with gradient for trust-constr optimizer"""
    c = full_corr(h, 1-h)
    smooth_max, _ = smooth_max_and_weights(c, alpha)
    grad = compute_gradient(h, alpha)
    return smooth_max, grad

    h0 = project_to_sum_box(h0, TARGET_SUM)
    restarts.append(h0.copy())
    
    # 2. Binary block start
    h_binary = np.zeros(n_points)
    h_binary[:int(n_points//2)] = 1
    np.random.shuffle(h_binary)
    h_binary = project_to_sum_box(h_binary, TARGET_SUM)
    restarts.append(h_binary.copy())
    
    # 3. Smooth random start
    h_smooth = np.random.rand(n_points)

    restarts.append(h_smooth.copy())
    
    # Multi-restart loop
    opt_count = 0
    while time.time() - start_time < budget_s - 10 and opt_count < 10:
        # Choose a restart
        h_start = restarts[opt_count % len(restarts)].copy()
        
        # Optimise with continuation on alpha
        current_h = h_start.copy()

            if time.time() - start_time > budget_s - 5:

                val, grad = objective_with_gradient(h, alpha)

                    fun,
                    current_h,
