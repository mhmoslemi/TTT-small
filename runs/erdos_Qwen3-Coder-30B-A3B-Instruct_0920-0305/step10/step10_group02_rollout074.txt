def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10

    return result

def smooth_boundaries(h, window_size=3):
    """Smooth boundaries by linear interpolation"""
    h_smooth = h.copy()
    boundaries = []
    
    # Find boundaries
    for i in range(len(h) - 1):
        if abs(h[i] - h[i+1]) > 0.1:  # Threshold for significant change
            boundaries.append(i)
    
    # Apply smoothing to boundaries
    for bound in boundaries:
        start = max(0, bound - window_size//2)
        end = min(len(h), bound + window_size//2 + 1)
        
        # Create smooth transition
        if h[bound] == 0 and h[bound+1] == 1:
            # 0 -> 1 transition
            for i in range(start, end):
                t = (i - start) / max(1, end - start - 1)
                h_smooth[i] = t * (h[bound+1] - h[bound]) + h[bound]
        elif h[bound] == 1 and h[bound+1] == 0:
            # 1 -> 0 transition
            for i in range(start, end):
                t = (i - start) / max(1, end - start - 1)
                h_smooth[i] = t * (h[bound+1] - h[bound]) + h[bound]
    
    # Ensure sum constraint
    h_smooth = project_to_sum_box(h_smooth, n_points / 2)
    return h_smooth
