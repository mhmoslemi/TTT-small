import numpy as np
from cvxpy import *
import time
from scipy.signal import correlate

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water-filling."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def binary_milp_optimization():
    """Phase A: Use binary integer programming to get a good combinatorial skeleton."""
    N = 60  # Smaller size for MILP
    dx_small = 2.0 / N
    total_sum = N / 2
    
    # Variables: binary x for each point
    x = Variable(N, boolean=True)
    
    # Constraints
    constraints = [sum(x) == total_sum]
    
    # For each shift k, compute correlation values directly with constraints
    # This is efficient because we're doing a hard linearization
    # Add auxiliary variables y[i,k] for x[i]*x[i+k] with convex constraints for binary case
    
    # The most straightforward way: we compute pairs where shift k makes sense
    t = Variable(1, nonneg=True)  # represents max_k (N/2 - sum of x_i * x_{i+k})
    
    # For each k, we have sum_{i} x_i * x_{i+k} <= (N/2 - t) for all k
    max_k = N - 1
    for k in range(1, max_k + 1):
        # Each pair with shift k
        expr = 0
        for i in range(N - k):
            # Constraint for y[i,k] = x[i]*x[i+k]: y[i,k] <= x[i], y[i,k] <= x[i+k], y[i,k] >= x[i]+x[i+k]-1
            # But we don't model y explicitly.
            expr += x[i] * x[i + k]  # These are integer up to checking by cvxpy
        # We know that the complement is bounded by sum = N/2 - these covered
        # See constraints as sum x_i x_{i+k} >= N/2 - t
        constraints.append(sum(x[i] * x[i+k] for i in range(N-k)) >= total_sum - t)
    
    # Objective to minimize t (worst case correlation)
    objective = Minimize(t)
    prob = Problem(objective, constraints)
    
    # Solve with limited time and good solver settings
    try:
        prob.solve(solver=GLPK_MI, time_limit=30)
        if prob.status in ["optimal", "feasible"] and x.value is not None:
            optimal_x = np.round(x.value).astype(int)
            return optimal_x
    except:
        pass
    # Fallback to random binary solution
    return np.random.choice([0, 1], size=N, p=[0.5, 0.5])

def upsample_binary_vector(x_binary, target_length):
    """Phase B: Upsampling the binary vector to final resolution."""
    source_length = len(x_binary)
    upsample_ratio = target_length // source_length
    if upsample_ratio <= 1:
        if len(x_binary) != target_length:
            return x_binary[:target_length] if target_length < len(x_binary) else np.pad(x_binary, (0, target_length - len(x_binary)), 'constant')
        return x_binary
    # Repeat pattern for upsample_ratio
    upsampled = np.tile(x_binary, upsample_ratio)
    return np.array(upsampled[:target_length])

def differential_evolution_refinement(h_initial, max_iter=500):
    """Phase D: Differential Evolution on full resolution."""
    n = len(h_initial)
    # Population size
    P = 30
    # Mutation factor and crossover probability
    F = 0.8
    CR = 0.9
    
    # Initialize population with slight perturbations around h_initial
    pop = []
    for _ in range(P):
        individual = h_initial + np.random.normal(0, 0.02, n)
        project_to_sum_box(individual, TARGET_SUM)
        pop.append(individual)

    best = pop[0]
    best_c5 = evaluate_c5(best)
    
    for it in range(max_iter):
        for i, target in enumerate(pop):
            # Select three distinct donors
            donors = np.random.choice(P, 3, replace=False)
            a, b, c = pop[donors[0]], pop[donors[1]], pop[donors[2]]
            
            # Mutate
            trial = a + F * (b - c)
            
            # Crossover
            mask = np.random.rand(n) < CR
            if not np.any(mask):
                mask[np.random.randint(n)] = True
            trial = np.where(mask, trial, target)
            
            # Project
            trial = project_to_sum_box(trial, TARGET_SUM)
            
            # Evaluate
            trial_c5 = evaluate_c5(trial)
            
            if trial_c5 < evaluate_c5(target):
                pop[i] = trial
                
                if trial_c5 < best_c5:
                    best = trial
                    best_c5 = trial_c5
                    
    return best, best_c5

def deterministic_local_refinement(h_start, iterations=2000):
    """Phase E: Deterministic local refinement using hill climbing."""
    h = h_start.copy()
    old_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = old_c5
    
    for _ in range(iterations):
        # Pick r indices for perturbation (use a small sample)
        indices = np.random.choice(len(h), size=int(len(h)*0.05), replace=False)
        
        # Directly modify those indices
        h_new = h.copy()
        for idx in indices:
            # Perturb with normal distribution proportional to h*(1-h) width
            sd = min(0.05, h[idx] * (1 - h[idx]) * 2)
            delta = np.random.normal(0, sd)
            h_new[idx] += delta
            
        # Re-project after direct changes
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < old_c5:
            h = h_new
            old_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        elif np.random.rand() < 0.01:
            # Occasionally accept worse if random
            h = h_new
            old_c5 = new_c5
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main wrapper."""
    global TARGET_SUM, n_points, dx
    
    np.random.seed(seed)
    start_time = time.time()
    
    # Variable initialization
    TARGET_SUM = n_points / 2
    
    # Phase A: Binary MILP (small size)
    if time.time() - start_time > budget_s * 0.05:
        # If we're already close to budget, skip optimization
        h0_binary = np.random.choice([0, 1], size=60, p=[0.5, 0.5])
    else:
        h0_binary = binary_milp_optimization()
    
    # Phase B: Upsampling to full resolution
    h0_upsampled = upsample_binary_vector(h0_binary, n_points)
    
    # Phase C: Optional swap search
    if time.time() - start_time < budget_s * 0.08:
        h1 = swap_search_on_binary(h0_upsampled, max_iterations=1000)
        h1 = project_to_sum_box(h1, TARGET_SUM)
    else:
        h1 = h0_upsampled
    
    # Phase D: DE refinement
    if time.time() - start_time < budget_s * 0.8:
        h_de, c5_de = differential_evolution_refinement(h1, max_iter=500)
    else:
        h_de, c5_de = h1, evaluate_c5(h1)
    
    # Phase E: Local polishing
    if time.time() - start_time < budget_s * 0.95:
        h_final, c5_final = deterministic_local_refinement(h_de, iterations=2000)
    else:
        h_final, c5_final = h_de, c5_de
    
    # Final check and return
    return (h_final.tolist(), float(c5_final), n_points)
