import numpy as np

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_with_gradient_descent(h0, max_iter=5000):

        return evaluate_c5(h)

        return np.sum(h) - TARGET_SUM

    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )

            return project_to_sum_box(res.x, TARGET_SUM)

        pass
    return h0

                    step_size *= 2.0

                h_new[idx] += delta
        else:
            # Strategy 2: Block perturbation
            block_size = max(1, n_points // 20)
            start_idx = np.random.randint(0, n_points - block_size)
            # Apply same perturbation to a block
            step = np.random.normal(0, 0.02)
            for idx in range(start_idx, min(start_idx + block_size, n_points)):
                h_new[idx] += step
        
        # Project to valid constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new)

            current_c5 = c5_new

                best_h = current_h.copy()
                best_c5 = c5_new
        elif np.random.random() < 0.03:  # Occasionally accept worse solutions
            current_h = h_new.copy()
            current_c5 = c5_new
    
    return best_h, best_c5

    fitness = [evaluate_c5(h) for h in population]

    best_h = population[sorted_indices[0]]
    best_c5 = fitness[sorted_indices[0]]

                child[i] += step
        
        child = project_to_sum_box(child, TARGET_SUM)
        new_population.append(child)
    
    return best_h, best_c5
