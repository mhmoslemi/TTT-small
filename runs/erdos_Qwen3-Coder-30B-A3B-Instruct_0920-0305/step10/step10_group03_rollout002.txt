def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Simple projection via clipping and scaling
    # Sort descending values
    x_sorted = np.sort(x)[::-1]
    
    # Find the threshold using water-filling method
    cumsum = np.cumsum(x_sorted)
    
    # Find the largest k such that (cumsum[k] - target_sum) <= x_sorted[k] * (k + 1)
    k = -1
    for i in range(len(x_sorted)):
        if cumsum[i] - target_sum <= x_sorted[i] * (i + 1):
            k = i
        else:
            break
    
    # Compute the threshold (lambda)
    if k == -1:
        lambda_val = (cumsum[-1] - target_sum) / len(x_sorted)
    else:
        lambda_val = (cumsum[k] - target_sum) / (k + 1)
    
    # Project
    projected = np.clip(x - lambda_val, 0, 1)
    
    # Force correct sum
    current_sum = np.sum(projected)
    if abs(current_sum - target_sum) > 1e-10:
        # Adjust by shifting all values
        delta = (target_sum - current_sum) / len(projected)
        projected = np.clip(projected + delta, 0, 1)
    
    return projected

def upsample_binary(binary_vec: np.ndarray, target_N: int) -> np.ndarray:
    """Upsample binary vector to target_N"""
    # Simple repeat to get target_N
    factor = target_N // len(binary_vec)
    if factor > 0:
        new_h = np.repeat(binary_vec, factor)
        # Pad to exact size if needed
        if len(new_h) < target_N:
            extension = target_N - len(new_h)
            new_h = np.append(new_h, np.zeros(extension))
        return new_h[:target_N]
    else:
        print("Upsampling factor <= 1. Return as is")
        return binary_vec

def differential_evolution(h_initial: np.ndarray, target_sum: float, time_budget: float, start_time: float) -> Tuple[np.ndarray, float]:

    best_h = h_initial.copy()

    current_h = h_initial.copy()
    current_c5 = evaluate_c5(current_h)
    
    # Simulation temperature
    temp = 1.0
    min_temp = 1e-6
    cooling_rate = 0.98
    
    iter_count = 0
    while (time.time() - start_time) < time_budget and iter_count < max_iter:
        iter_count += 1
        # Generate candidate with small random changes
        new_h = current_h.copy()
        
        # Perturb few indices
        perturb_indices = np.random.choice(n, max(1, n//20), replace=False)
        for idx in perturb_indices:
            # Small perturbation
            delta = np.random.normal(0, 0.01)
            new_h[idx] += delta
            
        # Project to constraints
        new_h = project_to_sum_box(new_h, target_sum)
        
        new_c5 = evaluate_c5(new_h)
        
        # Accept with some probability
        if new_c5 < current_c5:
            current_h = new_h.copy()
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        else:
            # Accept with probabilistic criterion
            if np.random.random() < np.exp(-(new_c5 - current_c5) / temp):
                current_h = new_h.copy()
                current_c5 = new_c5
        
        temp *= cooling_rate
        if temp < min_temp:
            temp = min_temp
            
    return best_h, best_c5

def local_hill_climb(h_initial: np.ndarray, target_sum: float, time_budget: float, start_time: float) -> Tuple[np.ndarray, float]:
    """Perform local hill climbing"""
    h = h_initial.copy()
    current_c5 = evaluate_c5(h)
    iter_count = 0
    
    # Each local improvement step
    while (time.time() - start_time) < time_budget:
        iter_count += 1
        # Try individual swaps
        h_new = h.copy()
        i = np.random.randint(0, len(h))
        j = np.random.randint(0, len(h))
        
        # Try to swap values to potentially reduce max overlap
        # Keep same constraints as much as possible
        if i != j:
            # Test swapping values
            val_i = h_new[i]
            val_j = h_new[j]
            h_new[i] = val_j
            h_new[j] = val_i
            
            # Re-project to sum constraint
            h_new = project_to_sum_box(h_new, target_sum)
            
            c5_new = evaluate_c5(h_new)
            
            if c5_new < current_c5:
                h = h_new.copy()
                current_c5 = c5_new
            else:
                # Occasionally accept worsening moves for continues search
                if np.random.random() < 0.05:
                    h = h_new.copy()
                    current_c5 = c5_new
    
    return h, current_c5
