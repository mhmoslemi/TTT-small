# Global variables
TARGET_SUM = None

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

            break

            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def boundary_smoothing(h, window_size=3):
    """Smooth boundaries by interpolating between adjacent values"""
    h_smooth = h.copy()
    
    # Find boundaries
    boundaries = []
    for i in range(len(h_smooth) - 1):
        if h_smooth[i] != h_smooth[i + 1]:
            boundaries.append(i)
    
    # Apply smoothing around boundaries
    for bound in boundaries:
        start = max(0, bound - window_size // 2)
        end = min(len(h_smooth), bound + window_size // 2 + 1)
        
        # Create interpolation
        if h_smooth[bound] == 1 and h_smooth[bound + 1] == 0:
            # 1 to 0 transition
            for i in range(start, end):
                factor = (i - start) / (end - start - 1) if end > start + 1 else 0
                h_smooth[i] = 1 - factor
        elif h_smooth[bound] == 0 and h_smooth[bound + 1] == 1:
            # 0 to 1 transition
            for i in range(start, end):
                factor = (i - start) / (end - start - 1) if end > start + 1 else 0
                h_smooth[i] = factor
    
    # Adjust sum to be correct
    current_sum = np.sum(h_smooth)
    if current_sum > 0:
        h_smooth *= TARGET_SUM / current_sum
    h_smooth = np.clip(h_smooth, 0, 1)
    
    return h_smooth

        alpha = 10.0

        corr = compute_corr_fft(h)
        # Clip to prevent overflow
        corr_clipped = np.clip(corr, -100, 100)
        # Soft maximum approximation with threshold
        return np.log(np.sum(np.exp(alpha * corr_clipped))) / alpha

        # Approximate gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(h)

            val_plus = soft_max_overlaps(h_plus)
            
            grad[i] = (val_plus - base_val) / eps
            
        return grad

        step_size = 0.01
        h_new = h - step_size * grad
