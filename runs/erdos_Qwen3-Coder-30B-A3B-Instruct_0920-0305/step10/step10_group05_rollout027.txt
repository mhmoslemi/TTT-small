import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
from scipy.fft import fft, ifft
import time

def project_to_sum_box(u, target_sum):
    """Project vector u onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(u + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(u + (a + b)/2, 0, 1)
    return result

def autocorr_fft(h):
    """Compute aperiodic autocorrelation using FFT"""
    n = len(h)
    h_rev = h[::-1]
    L = 2 * n - 1
    corr = ifft(fft(h_rev, L) * fft(h, L)).real
    return corr

def exact_c5(u):
    """Compute exact C5 from half-vector u"""
    n = len(u) * 2
    dx = 2.0 / n
    h = np.concatenate([u, u[::-1]])
    corr = autocorr_fft(h)
    min_corr = np.min(corr)
    return 1 - dx * min_corr

def lse_c5(u, beta):
    """Compute smooth surrogate C5 using log-sum-exp"""
    n = len(u) * 2
    dx = 2.0 / n
    h = np.concatenate([u, u[::-1]])
    corr = autocorr_fft(h)
    return 1 - (dx / beta) * np.log(np.sum(np.exp(beta * corr)))

def grad_lse_c5(u, beta):
    """Compute gradient of smooth surrogate C5"""
    n = len(u) * 2
    dx = 2.0 / n
    h = np.concatenate([u, u[::-1]])
    corr = autocorr_fft(h)
    
    w = np.exp(beta * corr)
    w_sum = np.sum(w)
    
    # Compute gradient
    grad = np.zeros_like(u)
    
    # Convolution with aperiodic autocorrelation
    h_rev = h[::-1]
    for i in range(len(u)):
        # i+k indices for the convolution
        conv_val = 0
        for k in range(-len(h)+1, len(h)):
            j = i + k
            if 0 <= j < len(h):
                conv_val += w[k + len(h) - 1] * h_rev[j]
            else:
                conv_val += 0  # Out of bounds
        grad[i] = -dx * conv_val / w_sum
    
    return grad

def parametric_to_halfvector(params, n):
    """Build half-vector from 3 parameters (a, r, p)"""
    a, r, p = params
    
    # Compute middle height to ensure constraint
    # p(a) = (1-2a)/(2*(1-a)) = (1-2a)/(2-2a)
    
    if a < 0 or a > 0.5:
        return None
    
    u = np.zeros(n // 2)
    edge_length = int(np.ceil(a * n / 2))
    ramp_length = int(np.ceil(r * n / 2))
    middle_length = n // 2 - 2 * edge_length - 2 * ramp_length
    
    if middle_length < 0:
        # Adjust for boundary condition
        return None
    
    # Fill edge regions
    u[:edge_length] = 1
    # Fill ramp regions
    if ramp_length > 0:
        u[edge_length:edge_length+ramp_length] = np.linspace(1, p, ramp_length, endpoint=False)
        u[n//2-ramp_length:n//2] = np.linspace(p, 1, ramp_length, endpoint=False)
    # Fill middle region
    u[edge_length+ramp_length:n//2-ramp_length] = p
    
    return u

def run_de(n_points=120, budget_s=600, seed=42):
    """Run differential evolution on parametric space"""
    np.random.seed(seed)
    start_time = time.time()
    
    def de_objective(params):
        u = parametric_to_halfvector(params, n_points)
        if u is None:
            return 1e10
        return exact_c5(u)
    
    # Parameter bounds: [a, r, p] where a in [0,0.5], r in [0,0.3], p derived
    bounds = [(0.05, 0.45), (0.0, 0.3), (0.1, 0.9)]
    
    # Run differential evolution
    result = differential_evolution(
        de_objective,
        bounds,
        maxiter=200,
        popsize=15,
        seed=seed,
        strategy='best1bin',
        tol=1e-6,
        callback=lambda x, convergence: time.time() - start_time > 0.9 * budget_s
    )
    
    return result.x

def run_lbfgsb(u0, beta=200, n_points=240):
    """Run L-BFGS-B on full half-vector"""
    start_time = time.time()
    dx = 2.0 / n_points
    
    # Apply projection to ensure sum constraint
    u = project_to_sum_box(u0, n_points / 4)
    
    def objective(u):
        return lse_c5(u, beta)
    
    def grad(u):
        return grad_lse_c5(u, beta)
    
    options = {'maxiter': 800, 'gtol': 1e-10}
    
    # Run optimization with projected gradients
    def projected_objective(u):
        # Project to feasible set
        u_projected = project_to_sum_box(u, n_points / 4)
        return objective(u_projected)
    
    def projected_grad(u):
        # Project to feasible set
        u_projected = project_to_sum_box(u, n_points / 4)
        return grad(u_projected)
    
    # Optimization using L-BFGS-B
    result = minimize(
        projected_objective,
        u,
        method='L-BFGS-B',
        jac=projected_grad,
        bounds=[(0, 1)] * len(u),
        options=options,
        callback=lambda x: time.time() - start_time > 0.1 * budget_s
    )
    
    # Final projection to ensure sum constraint
    u_opt = project_to_sum_box(result.x, n_points / 4)
    return u_opt

def run_sa(u_best, n_points=240, budget_s=600):
    """Run simulated annealing polishing"""
    start_time = time.time()
    
    def c5_from_h(h):
        corr = autocorr_fft(h)
        dx = 2.0 / n_points
        return 1 - dx * np.min(corr)
    
    # Ensure symmetry on full vector
    h_best = np.concatenate([u_best, u_best[::-1]])
    c5_best = c5_from_h(h_best)
    
    # Simulated annealing
    T = 5e-3
    alpha = 0.9997
    max_delta = 0.05
    acceptance_ratio = 0
    n_accept = 0
    n_total = 0
    
    # Temperature schedule to keep within time budget
    SA_iters = 50000
    for i in range(SA_iters):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Select two indices
        i1, i2 = np.random.choice(len(u_best), 2, replace=False)
        delta = np.random.uniform(0, max_delta)
        
        # Acceptance test
        if u_best[i1] >= delta and u_best[i2] <= 1 - delta:
            # Make move
            u_trial = u_best.copy()
            u_trial[i1] -= delta
            u_trial[i2] += delta
            h_trial = np.concatenate([u_trial, u_trial[::-1]])
            c5_trial = c5_from_h(h_trial)
            
            if c5_trial < c5_best:
                u_best = u_trial.copy()
                c5_best = c5_trial
                acceptance_ratio = (n_accept + 1) / (n_total + 1)
            else:
                # Accept with probability
                if np.random.uniform() < np.exp(-(c5_trial - c5_best) / T):
                    u_best = u_trial.copy()
                    c5_best = c5_trial
                    acceptance_ratio = (n_accept + 1) / (n_total + 1)
                    n_accept += 1
            
            n_total += 1
            
            # Adjust delta
            if n_total % 500 == 0:
                if acceptance_ratio < 0.2:
                    max_delta *= 0.9
                elif acceptance_ratio > 0.4:
                    max_delta *= 1.1
                    
        T *= alpha
    
    return u_best, c5_best
