def full_corr(a, b):
    """Compute correlation using FFT with zero padding"""
    L = 2 * len(a) - 1
    corr = np.fft.ifft(np.fft.fft(a, L) * np.conj(np.fft.fft(b, L))).real
    return corr

def softmax(x, axis=None):
    """Stable softmax with subtraction of max for numerical stability"""
    x_shifted = x - np.max(x, axis=axis, keepdims=True)
    exp_x = np.exp(x_shifted)
    return exp_x / np.sum(exp_x, axis=axis, keepdims=True)

def smooth_max_and_weights(c, alpha):
    """Compute smooth max and corresponding softmax weights"""
    if alpha == 0:
        return np.max(c), np.ones_like(c) / len(c)
    c_scaled = alpha * c
    w = softmax(c_scaled)
    smooth_max = np.sum(w * c) / alpha
    return smooth_max, w

def smooth_obj_and_grad(h, alpha):
    """Compute smooth objective and its gradient"""
    # Compute all correlation values
    corr = full_corr(h, 1 - h)
    
    # Compute smooth max with weights
    _, w = smooth_max_and_weights(corr, alpha)
    
    # Symmetrize weights
    w_sym = w + w[::-1]
    
    # Compute gradient: g = w_sym * h (correlation)
    g = full_corr(w_sym, h)[:len(h)]
    
    # Gradient of smooth objective
    grad = 1.0 - g
    
    # Return smooth max value and gradient
    smooth_max_val, _ = smooth_max_and_weights(corr, alpha)
    return smooth_max_val, grad

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Binary search for lambda
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-12

            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result
