TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort x in descending order to find lambda efficiently
    sorted_indices = np.argsort(-x)
    sorted_x = x[sorted_indices]
    
    # Water-filling approach to find lambda
    cumsum = np.cumsum(sorted_x)
    n = len(sorted_x)
    
    # Find the threshold lambda such that sum(clamp(x - lambda, 0, 1)) = target_sum
    lambda_val = 0.0
    for i in range(n):
        if i > 0:
            # We're adding sorted_x[i] to sum so far
            # sum of previously added elements: cumsum[i-1]
            # Diff of new sum if we set lambda=i
            # lambda = (cumsum[i-1] - target_sum + sorted_x[i]) / (i+1)
            temp_lambda = (cumsum[i-1] - target_sum + sorted_x[i]) / (i+1)
            if temp_lambda <= sorted_x[i]:
                lambda_val = temp_lambda
                break
        else:
            # First iteration, just compute based on sum
            temp_lambda = (sorted_x[i] - target_sum) / 1
            if temp_lambda <= sorted_x[i]:
                lambda_val = temp_lambda
                break
    
    # Apply lambda to get final projected point
    result = np.clip(x - lambda_val, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Full correlation with mode='full'
    # We compute correlation of h with (1-h) using np.correlate
    corr = np.correlate(h, 1-h, mode='full')
    
    # Take maximum correlation value
    max_corr_index = np.argmax(corr)
    max_corr = corr[max_corr_index]
    
    # Return max overlap integral
    return dx * max_corr

def compute_correlation_vector(h):
    """Compute correlation for all shifts efficiently"""
    corr = np.correlate(h, 1-h, mode='full')
    # Only consider positive shifts
    return corr[len(h)-1:]

def binary_milp(N_small, time_limit=30):
    """
    Solve binary integer programming formulation for small N
    Minimize maximum overlap: max_k (dx * (N/2 - correlation(k)))
    """
    # Use cvxpy
    try:
        x = cp.Variable(N_small, boolean=True)
        t = cp.Variable()
        
        # Compute all shift correlations
        constraints = []
        
        # For each shift k from 0 to N-1
        for k in range(N_small):
            if k == 0:
                # Shift 0: correlation is sum_i x_i * x_i = sum_i x_i (because x_i is binary)
                # This is constant = N/2, so we can skip adding constraint for shift 0
                continue
            # Create binary product variables y_i,k for i=0..N-k-1
            y = cp.Variable(N_small - k, boolean=True)
            for i in range(N_small - k):
                # McCormick bounds for y[i] = x[i] * x[i+k]
                constraints += [
                    y[i] <= x[i],
                    y[i] <= x[i+k],
                    y[i] >= x[i] + x[i+k] - 1,
                ]
            # Constraint: sum of products >= N/2 - t
            constraints += [cp.sum(y) >= N_small//2 - t]
        
        # Sum constraint
        constraints += [cp.sum(x) == N_small//2]
        
        # Objective
        objective = cp.Minimize(t)
        
        # Create and solve problem
        prob = cp.Problem(objective, constraints)
        
        # Set solver options
        prob.solve(solver=cp.GLPK_MI, time_limit=time_limit)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value
        else:
            return None
            
    except Exception as e:
        return None

def upsample_binary(binary_vec, target_N):
    """Upsample binary vector to larger N by replicating values"""
    # Determine upsampling factor
    scale = target_N // len(binary_vec)
    if scale <= 0:
        scale = 1
    
    # Repeat each element scale times
    upsampled = np.repeat(binary_vec, scale)
    
    # Trim or pad to target_N
    if len(upsampled) > target_N:
        upsampled = upsampled[:target_N]
    elif len(upsampled) < target_N:
        upsampled = np.pad(upsampled, (0, target_N - len(upsampled)), mode='constant')
        
    return upsampled.astype(float)

def objective_function(h, target_sum):
    """Normalized objective for differential evolution"""
    h_proj = project_to_sum_box(h, target_sum)
    return evaluate_c5(h_proj)

def differential_evolution_optimize(h_initial, target_sum, budget_s, start_time):
    """Run differential evolution with custom objective"""
    
    # Define bounds
    bounds = [(0, 1) for _ in range(len(h_initial))]
    
    def objective(h):
        return objective_function(h, target_sum)
    
    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,
        maxiter=int(budget_s * 50 / len(h_initial)) if len(h_initial) > 0 else 1000,
        popsize=20,
        mutation=(0.5, 1.0),
        recombination=0.7,
        atol=1e-6,
        rtol=1e-6,
        seed=42,
        callback=None
    )
    
    best_h = project_to_sum_box(result.x, target_sum)
    best_c5 = evaluate_c5(best_h)
    
    return best_h, best_c5

def local_improvement_binary(h_init, target_sum, max_iter=10000):

    h = h_init.copy()

    max_corr_index = np.argmax(corr_vec)
    
    # Record best solution so far
    best_h = h.copy()
    best_corr_vec = corr_vec.copy()
    best_max_corr_index = max_corr_index
    
    for iteration in range(max_iter):
        # Recompute correlations
        corr_vec = compute_correlation_vector(h)
        max_corr_index = np.argmax(corr_vec)
        max_corr = corr_vec[max_corr_index]
        
        # If improvement found, break early
        if max_corr <= 1e-10 or iteration > max_iter:
            break
            
        # Find candidates to improve worst correlation
        # For the worst shift k_max, we want to reduce sum of products
        k_max = max_corr_index
        if k_max >= len(h):
            k_max = len(h) - k_max - 1
            if k_max == 0:
                continue
                
        # Try to improve with swaps
        candidates_pos = []
        candidates_neg = []
        
        # Check for positions where a bit pair contributes to max overlap
        for i in range(len(h) - k_max):
            if h[i] == 1 and h[i + k_max] == 0:
                candidates_pos.append(i)
            if h[i] == 0 and h[i + k_max] == 1:
                candidates_neg.append(i)
                
        if candidates_pos and candidates_neg:
            # Try to swap
            i = np.random.choice(candidates_pos)
            j = np.random.choice(candidates_neg)
            
            # Perform swap
            h[i], h[j] = h[j], h[i]
            
            # Check if we improved
            new_corr_vec = compute_correlation_vector(h)
            new_max = new_corr_vec[np.argmax(new_corr_vec)]
            
            # If not better, revert
            if new_max >= max_corr:
                h[i], h[j] = h[j], h[i]
        else:
            # Random swap without constraint checking
            i, j = np.random.choice(len(h), 2, replace=False)
            if h[i] != h[j]:
                h[i], h[j] = h[j], h[i]
        
        # Update best
        if np.max(compute_correlation_vector(h)) < np.max(best_corr_vec):
            best_h = h.copy()
            best_corr_vec = compute_correlation_vector(h)
            best_max_corr_index = np.argmax(best_corr_vec)
    
    # Final projection
    final_h = project_to_sum_box(best_h.astype(float), target_sum)
    final_c5 = evaluate_c5(final_h)
    
    return final_h, final_c5
