def compute_corr_fft(h):
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    return dx * np.max(corr)

def evaluate_c5(h):
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def extract_skeleton(h):
    """Convert h to a skeleton where switch points are identified"""
    # Look for change points
    changes = []
    for i in range(1, n_points):
        if h[i] != h[i-1]:
            changes.append(i)
    changes = [0] + changes + [n_points]
    # Ensure even number of intervals
    if len(changes) % 2 == 1:
        changes.append(n_points)
    return changes
