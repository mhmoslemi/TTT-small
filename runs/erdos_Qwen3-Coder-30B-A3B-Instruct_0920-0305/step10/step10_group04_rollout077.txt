import numpy as np
from scipy.signal import correlate
import time

# Global constants and state
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_from_binary(x):
    """Compute correlation using standard correlate, returns full array for O_d computation"""
    x_rev = x[::-1]
    corr = correlate(x_rev, x, mode="full")
    return corr

def compute_c5_from_corr(corr):
    """Convert auto-correlation to C5 value, assuming x contains n/2 ones"""  
    # Compute the O_d values for all shifts
    n = len(x)
    shifts = np.arange(-(n-1), n)
    O = np.zeros_like(shifts)
    for d in shifts:
        if d >= 0:
            # Positive shift: x_{i+d} gives influence
            O[d] = np.sum(x[:n-d]) - corr[d+n-1]
        else:
            # Negative shift: x_{i-d} gives influence  
            O[d] = np.sum(x[-d:]) - corr[n-1+d]
    return dx * np.max(O)

def compute_c5_incremental_update(x, a, i, j):
    """Compute updated C5 given a swap of x[i] and x[j], using incremental update of correlation.
       x: binary vector
       a: auto-correlation (will be changed)
       i, j: indices to swap
    """
    # pad x for easy slicing
    x_pad = np.concatenate([[0]*(n_points-1), x.tolist(), [0]*(n_points-1)])
    # compute contribution of changes to auto-correlation
    # for each possible shift d
    # compute A_d before flip, then subtract and add
    ΔA = np.zeros(2*n_points - 1, dtype=np.float64)  # for all shifts
    # only indices that change (i and j, and their shifts)
    # flip one location at a time
    for d in range(-(n_points-1), n_points):
        if i+d >= 0 and i+d < n_points:
            ΔA[d + (n_points-1)] -= x[i+d]
        if j+d >= 0 and j+d < n_points:  
            ΔA[d + (n_points-1)] += x[j+d]
            
    # Add those contributions correctly accounting for indexing offsets
    # efficient vectorized slice addition
    mask_i = (i + np.arange(0, 2*n_points - 1) < n_points) & (i + np.arange(0, 2*n_points - 1) >= 0)
    shift_i = i + np.arange(0, 2*n_points - 1) - (n_points - 1)
    mask_j = (j + np.arange(0, 2*n_points - 1) < n_points) & (j + np.arange(0, 2*n_points - 1) >= 0)
    shift_j = j + np.arange(0, 2*n_points - 1) - (n_points - 1)
    # insert the actual increments
    ΔA += x_pad[i+n_points-1:i+3*n_points-2]
    ΔA -= x_pad[j+n_points-1:j+3*n_points-2]

    # recompute O values
    O = np.zeros(2*n_points-1)
    for d in range(-(n_points-1), n_points):
        if d >= 0:
            O[d] = np.sum(x[:n_points-d]) - (a[d + n_points - 1])
        else:
            O[d] = np.sum(x[-d:]) - (a[d + n_points - 1])
    return dx * np.max(O)

def fast_c5_update(x, a, i, j):
    """Very fast computation of C5 after binary swap of positions i,j."""
    # pad x for convolution-style indexing
    pad_x = np.concatenate(([0]*(n_points-1), x, [0]*(n_points-1)))
    scale = 1.0

    # compute the point-wise contribution changes for all possible shifts.
    ΔA = np.zeros(2*n_points - 1, dtype=np.float64)

    # For shift d:
    # contribution of x_i is x[i+d] if valid
    # contribution of x_j is x[j+d] if valid
    # then mark the index range held by the cross product

    # indices involved in shifts near i, j  
    for d in range(-(n_points-1), n_points):
        # entry element at shift d: if present
        if 0 <= i+d < n_points:
            ΔA[d + n_points -1 ] -= pad_x[i+d + n_points - 1]  # contribution at d from i
              
        # entry element at shift d: if present
        if 0 <= j+d < n_points:
            ΔA[d + n_points -1 ] += pad_x[j+d + n_points - 1]  # contribution at d from j

    # now apply the update to auto-correlation
    new_a = a + ΔA  

    # Convert auto-correlation to overlap values O_d
    # O_d = sum_h_i(1 - h_i+d) = sum_h_i - sum_h_i h_i+d
    O = np.zeros_like(new_a)
    for d in range(-(n_points-1), n_points):
        # compute sum_{i=0}^{n-1-d} x_i and sum_{i=0}^{n-1-d} x_i x_{i+d}
        if d >= 0:
            o = np.sum(x[:n_points-d]) - new_a[d + n_points -1]  
        else:  # negative shift
            o = np.sum(x[-d:]) - new_a[d + n_points -1]  
        O[d + n_points - 1] = o
            
    return dx * np.max(O)

def project_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum via binary search"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a)*f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    return np.clip(x + (a + b)/2, 0, 1)

def find_worst_shift_candidates(x):
    """Given binary vector x, find candidates of i, j for targeted swap"""
    # precompute which i can contribute to each d and vice versa
    n = len(x)
    # full auto-correlation:
    corr = compute_corr_from_binary(x)
    # For each shift d, the overlap term is sum(x_i * (1 - x_{i+d}) ) = sum(x_i) - sum(x_i * x_{i+d})
    # Compute O_d = |sum(x_i) - corr[d+N-1]| = |n/2 - corr[d+N-1]| 
    shifts = np.arange(-(n-1), n)
    overlaps = np.zeros_like(shifts, dtype=float)
    for d in shifts:
        if d >= 0:
            overlaps[d] = np.sum(x[:n-d]) - corr[d+n-1]
        else:
            overlaps[d] = np.sum(x[-d:]) - corr[n-1+d]
    
    d_best = np.argmax(overlaps)
    
    # Candidate pick for U and V sets:
    U = []  # positions that flip from 1 to 0 and are 1 at d*.  
    V = []  # positions that flip from 0 to 1 from locations that are 0 at d*
    d = shifts[d_best]
    
    # look forward and backwards for such entries  
    if d >= 0:
        # check if x_i = 1 and x_{i+d} = 0
        for i in range(n - d):
            if x[i] == 1 and x[i+d] == 0:
                U.append(i)
        # check if x_j = 0 and x_{j+d} = 1
        for j in range(d, n):
            if x[j] == 0 and x[j-d] == 1:
                V.append(j)
    else:
        # d < 0 -> i is shifted left (denoted here), so x_{i+|d|}=1 when x_{i}=0
        d_neg = -d
        for i in range(n - d_neg):
            if x[i] == 1 and x[i+d_neg] == 0:
                U.append(i)
        for j in range(d_neg, n):
            if x[j] == 0 and x[j-d_neg] == 1:
                V.append(j)
                
    return U, V, shifts[d_best]

def binary_sa_with_targeted_moves(seed=42, budget_s=600):
    """Binary simulated annealing with targeted moves to improve C5"""
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial binary vector
    x = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
    # Ensure it sums to exactly n_points / 2
    num_ones = np.sum(x)
    if num_ones < TARGET_SUM:
        # Flip 0s to 1s
        flip_indices = np.where(x==0)[0]
        select = np.random.choice(flip_indices, size=int(TARGET_SUM - num_ones), replace=False)
        x[select] = 1
    elif num_ones > TARGET_SUM:
        # Flip 1s to 0s
        flip_indices = np.where(x==1)[0]
        select = np.random.choice(flip_indices, size=int(num_ones - TARGET_SUM), replace=False)
        x[select] = 0
    
    # compute initial correlation
    a = compute_corr_from_binary(x)
    current_c5 = dx * np.max(np.array(np.sum(x[:n]) - a[n_points-1:2*n_points-1]))
    best_c5 = current_c5
    best_x = x.copy()
    
    # cooling schedule
    T0, Tf = 2*dx, 1e-6*dx
    max_moves = 300000
    accepted_moves = 0
    T = T0
    cooling_rate = (T0-Tf)/max_moves
    
    # restart strategy after stagnation
    last_improvement_move = 0
    stuck_tolerance = 20000
    
    # iterate for a long time
    for i in range(500000):
        if time.time() - start_time > budget_s - 5:
            break
            
        if i % 10000 == 0:
            print(f"iter {i}, C5={current_c5:.6f}, T={T:.6f}")
            
        # alternate between targeted swaps and random
        if np.random.random() < 0.6:  
            # targeted swap based on worst shift
            U, V, d = find_worst_shift_candidates(x)
            if U and V:
                i_sw = np.random.choice(U)
                j_sw = np.random.choice(V)
                # precompute updates in the new system
                new_c5 = fast_c5_update(x, a, i_sw, j_sw)
                if new_c5 <= current_c5 or np.random.random() < np.exp(-(new_c5 - current_c5) / T):
                    # accept
                    x[i_sw], x[j_sw] = x[j_sw], x[i_sw]
                    a = compute_corr_from_binary(x)
                    current_c5 = new_c5
                    if current_c5 < best_c5:
                        best_x = x.copy()
                        best_c5 = current_c5
                    accepted_moves += 1
                else:
                    # reject, do nothing since swapped back
                    pass 
        else:
            # random swap
            idx1 = np.random.randint(n_points)
            idx2 = np.random.randint(n_points)
            if idx1 == idx2:
                continue
            if x[idx1] == x[idx2]:  # if both same, no benefit
                continue
            
            # try simple swap
            x[idx1], x[idx2] = x[idx2], x[idx1]
            new_c5 = fast_c5_update(x, a, idx1, idx2)
            if new_c5 <= current_c5 or np.random.random() < np.exp(-(new_c5 - current_c5) / T):
                a = compute_corr_from_binary(x)
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_x = x.copy()
                    best_c5 = current_c5
                accepted_moves += 1
            else:
                x[idx1], x[idx2] = x[idx2], x[idx1]  # undo
                
        # update temperature 
        if accepted_moves < max_moves:
            T = T0 - accepted_moves * cooling_rate
        else:
            # within a range for computing its effect
            pass
                    
        # check for stagnation & re-start
        if accepted_moves - last_improvement_move > stuck_tolerance:
            print("Stagnated for a while, partial reload.")
            x_new = np.copy(best_x)
            # perturb a small fraction
            num_perturb = int(n_points * 0.05)
            indices = np.random.choice(n_points, size=num_perturb, replace=False)
            # flip those bits, maintain sum
            ones_now = np.sum(x_new)
            if ones_now > TARGET_SUM:
                # change some 1s to 0s
                ones_to_change = ones_now - TARGET_SUM
                to_zero = np.random.choice(np.where(x_new == 1)[0], size=int(ones_to_change), replace=False)
                x_new[to_zero] = 0
            else:
                # change some 0s to 1s
                zeros_to_change = TARGET_SUM - ones_now
                to_one = np.random.choice(np.where(x_new == 0)[0], size=int(zeros_to_change), replace=False)
                x_new[to_one] = 1
                
            x = x_new
            a = compute_corr_from_binary(x)
            current_c5 = compute_c5_from_corr(a)  
            last_improvement_move = accepted_moves
            
    return best_x, best_c5

def refine_to_continuous(x_binary, budget_left):
    """Use SLSQP to refine the binary solution to continuous values"""
    # convert to float
    h_float = x_binary.astype(float)
    
    # smoothing to reduce high frequency noise
    # minimal box filter
    h_smooth = np.copy(h_float)
    for i in range(len(h_float)):
        start = max(0, i-1)
        end = min(len(h_float), i+2)
        h_smooth[i] = np.mean(h_float[start:end])
    
    # fix sum constraint
    h_refine = project_sum_box(h_smooth, TARGET_SUM)
    
    # Run SLSQP with bounds and constraints
    from scipy.optimize import minimize
    
    def obj(h):
        # compute overlap integral using standard scipy method
        corr = correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        return max_val * dx
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
    
    try:
        res = minimize(
            obj, 
            h_refine, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12, 'disp': False}
        )
        if res.success:
            h_final = res.x
            h_final = project_sum_box(h_final, TARGET_SUM)
            c5_final = obj(h_final)
            return h_final, c5_final
        else:
            # fallback
            return h_refine, obj(h_refine)
    except Exception as e:
        return h_refine, obj(h_refine)

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    # Binary SA phase
    x_best, c5_best = binary_sa_with_targeted_moves(seed, budget_s * 0.7) 
    
    # compute final answer  
    h_final = x_best.astype(float)
    c5_final = c5_best
    # If we have remaining time, refine with SLSQP 
    if time.time() - start_time < budget_s * 0.95:
        try:
            # Actually not going to re-run full SLSQP here to save time
            # But we'll just execute final projection to ensure integrity
            h_final = project_sum_box(h_final, TARGET_SUM)
            # make sure it reflects actual value
            corr = correlate(h_final, 1 - h_final, mode='full')
            c5_final = dx * np.max(corr)
        except:
            pass
            
    return h_final.tolist(), float(c5_final), n_points
