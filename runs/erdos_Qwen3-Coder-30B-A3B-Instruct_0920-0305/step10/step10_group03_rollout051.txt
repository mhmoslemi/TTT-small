def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        lambda: np.random.normal(0, 0.01, n_points) * (1 - 2 * (np.random.rand(n_points) > 0.5))  # Alternating perturbation

                best_c5 = c5_new
        elif np.random.random() < 0.1:  # Occasionally accept worse solutions
            current_h = h_new.copy()
            current_c5 = c5_new
    
    return best_h, best_c5

def evolutionary_optimization(h0, population_size=20, generations=100):
    """Evolutionary optimization approach"""
    population = []
    fitness = []
    
    # Initialize population
    for _ in range(population_size):
        h = h0 + np.random.normal(0, 0.01, n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
        fitness.append(evaluate_c5(h))
    
    # Evolutionary loop
    for gen in range(generations):
        # Sort by fitness
        idx = np.argsort(fitness)
        population = [population[i] for i in idx]
        fitness = [fitness[i] for i in idx]
        
        # Keep top 50%
        top_population = population[:population_size//2]
        top_fitness = fitness[:population_size//2]
        
        # Create new population
        new_population = top_population.copy()
        
        # Mutate from top individuals
        for _ in range(population_size - len(top_population)):
            parent = np.random.choice(top_population)
            # Add Gaussian noise
            child = parent + np.random.normal(0, 0.005, n_points)
            child = project_to_sum_box(child, TARGET_SUM)
            new_population.append(child)
        
        population = new_population
        fitness = [evaluate_c5(h) for h in population]
    
    # Return best individual
    idx = np.argmin(fitness)
    return population[idx], fitness[idx]
