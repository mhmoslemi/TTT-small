import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from math import gcd

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_beatty_sequence(a, b, N):
    """Generate sequence using linear congruential generator"""
    h = np.zeros(N)
    for i in range(N):
        if ((i * a + b) % N) < N/2:
            h[i] = 1
    return h

def try_soft_threshold(h, t, delta):
    """Apply soft threshold to make continuous"""
    N = len(h)
    softened = np.zeros(N)
    for i in range(N):
        val = (t - ((i * 1 + 0) % N)) / delta
        softened[i] = min(1, max(0, val))
    return softened

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    # Explore parametric families for generating candidates
    best_c5 = float('inf')
    best_h = None

    # Try all coprime values of a
    coprimes = [i for i in range(1, n_points) if gcd(i, n_points) == 1]
    num_a_values = len(coprimes)
    
    # Tweak: Try first few coprimes
    coprimes = coprimes[:min(50, num_a_values)]
    
    # Try all values of b
    for a in coprimes:
        for b in range(n_points):
            if time.time() - start_time > budget_s - 30:
                break
            h = generate_beatty_sequence(a, b, n_points)
            h = project_to_sum_box(h, TARGET_SUM)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        if time.time() - start_time > budget_s - 30:
            break
    
    # If did not find good initial, use random construction
    if best_h is None:
        h0 = np.ones(n_points) * 0.5
        h0 = project_to_sum_box(h0, TARGET_SUM)
        best_h = h0.copy()
        best_c5 = evaluate_c5(h0)
    
    # Now refine with local search and smoothing
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Phase 1: Local hill climbing
    max_iter = 10000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 30:
            break
            
        # Identify current worst lag
        corr = compute_corr_fft(current_h)
        k_worst = np.argmax(corr)
        
        # Heuristic swap move
        h_new = current_h.copy()
        
        # Find indices where we can swap to improve the worst case
        # Look at current h and try to move one 1 to a 0 at the shift k_worst
        max_change = -1
        best_swap = None
        
        # Try to improve worst case by swapping 1->0 at i and 0->1 at i+k_worst if possible
        for i in range(n_points):
            if current_h[i] == 1 and i + k_worst < n_points and current_h[i + k_worst] == 0:
                h_try = h_new.copy()
                h_try[i] = 0
                h_try[i + k_worst] = 1
                c5_try = evaluate_c5(h_try)
                if c5_try < current_c5:
                    if c5_try < current_c5 and c5_try < max_change:
                        max_change = c5_try
                        best_swap = (i, i + k_worst)
            
            if current_h[i] == 0 and i + k_worst < n_points and current_h[i + k_worst] == 1:
                h_try = h_new.copy()
                h_try[i] = 1
                h_try[i + k_worst] = 0
                c5_try = evaluate_c5(h_try)
                if c5_try < current_c5:
                    if c5_try < current_c5 and c5_try < max_change:
                        max_change = c5_try
                        best_swap = (i, i + k_worst)
                        
        # If we found a good improvement, make it
        if best_swap:
            i1, i2 = best_swap
            h_new[i1] = 0
            h_new[i2] = 1
            # Project to constraints
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            current_c5 = evaluate_c5(h_new)
            current_h = h_new.copy()
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            # Random swap with low probability to escape local minima
            if np.random.random() < 0.05:
                indices = np.random.choice(n_points, 2, replace=False)
                i1, i2 = indices[0], indices[1]
                h_new = current_h.copy()
                h_new[i1], h_new[i2] = h_new[i2], h_new[i1]
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                c5_new = evaluate_c5(h_new)
                if c5_new < current_c5:
                    current_c5 = c5_new
                    current_h = h_new.copy()
                    if current_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = current_c5
    
    # Phase 2: Apply smoothing
    smoothed_h = best_h.copy()
    if np.random.random() > 0.5:
        # Apply smoothing and projection
        boundary_indices = []
        for i in range(n_points - 1):
            if smoothed_h[i] != smoothed_h[i + 1]:
                boundary_indices.append(i)
        
        # Add 3-point smoothing around each boundary
        for i in boundary_indices:
            # Only smooth if within bounds
            left = max(0, i - 1)
            right = min(n_points - 1, i + 1)
            for j in range(left, right + 1):
                if j == i:
                    smoothed_h[j] = 0.5
                elif j < i:
                    smoothed_h[j] = 0.5 * (1 - (i - j) / 2.0) if i - j <= 1 else 0.0
                else:
                    smoothed_h[j] = 0.5 * (1 - (j - i) / 2.0) if j - i <= 1 else 0.0

        # Adjust sum
        total = np.sum(smoothed_h)
        if total != TARGET_SUM:
            delta = (TARGET_SUM - total) / n_points
            smoothed_h += delta
            smoothed_h = np.clip(smoothed_h, 0, 1)
    
    # Final optimization with standard scipy optimization
    try:
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        res = minimize(
            objective,
            smoothed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        if res.success:
            final_h = res.x
        else:
            final_h = smoothed_h
    except Exception:
        final_h = smoothed_h
    
    final_h = project_to_sum_box(final_h, TARGET_SUM)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h.tolist(), float(final_c5), n_points)
