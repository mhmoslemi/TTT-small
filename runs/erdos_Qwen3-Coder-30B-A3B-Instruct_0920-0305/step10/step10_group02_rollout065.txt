import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from math import gcd

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_modular_sequences(n_points):
    """Generate sequences using modular arithmetic"""
    # Find coprime numbers with n_points
    coprimes = [i for i in range(1, n_points) if gcd(i, n_points) == 1]
    
    best_h = None
    best_c5 = float('inf')
    
    # Try different a,b parameters
    for a in coprimes[:20]:  # Limit to first 20 coprimes for speed
        for b in range(n_points):
            h = np.zeros(n_points)
            for i in range(n_points):
                if ((i * a + b) % n_points) < n_points // 2:
                    h[i] = 1
            
            # Normalize to sum = n_points / 2
            h = project_to_sum_box(h, n_points / 2)
            
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5

def smooth_boundaries(h, smooth_radius=3):
    """Smooth boundaries to make h continuous"""
    h_smooth = h.copy()
    
    # Find all boundaries (consecutive elements with different values)
    boundaries = []
    for i in range(len(h) - 1):
        if h[i] != h[i + 1]:
            boundaries.append(i)
    
    # Apply smoothing centered at boundaries
    for bound in boundaries:
        start = max(0, bound - smooth_radius)
        end = min(len(h), bound + smooth_radius + 1)
        
        # Create linear interpolation
        if start < bound < end - 1:
            # Interpolate from h[start] to h[end-1]
            if h[start] == 0 and h[end-1] == 1:
                # from 0 to 1
                for i in range(start, end):
                    if i <= bound:
                        h_smooth[i] = (i - start) / (bound - start) if bound > start else 0
                    else:
                        h_smooth[i] = 1 - (i - bound) / (end - 1 - bound) if end - 1 > bound else 1
            elif h[start] == 1 and h[end-1] == 0:
                # from 1 to 0
                for i in range(start, end):
                    if i <= bound:
                        h_smooth[i] = 1 - (i - start) / (bound - start) if bound > start else 1
                    else:
                        h_smooth[i] = (i - bound) / (end - 1 - bound) if end - 1 > bound else 0
    
    # Ensure sum constraint
    h_smooth = project_to_sum_box(h_smooth, len(h) / 2)
    return h_smooth

def gradient_descent_refinement(h, target_sum, max_iter=200, step_size=0.01):
    """Refine solution using gradient descent"""
    h_refined = h.copy()
    current_c5 = evaluate_c5(h_refined)
    
    for _ in range(max_iter):
        # Compute gradients using finite differences
        grad = np.zeros_like(h_refined)
        epsilon = 1e-6
        
        for i in range(len(h_refined)):
            h_plus = h_refined.copy()
            h_plus[i] += epsilon
            h_plus = project_to_sum_box(h_plus, target_sum)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = h_refined.copy()
            h_minus[i] -= epsilon
            h_minus = project_to_sum_box(h_minus, target_sum)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Update
        h_refined = h_refined - step_size * grad
        h_refined = project_to_sum_box(h_refined, target_sum)
        
        new_c5 = evaluate_c5(h_refined)
        if new_c5 < current_c5:
            current_c5 = new_c5
        else:
            # Reduce step size if not improving
            step_size *= 0.9
        
        if step_size < 1e-8:
            break
    
    return h_refined, current_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 120
    dx = 2.0 / n_points
    
    # Generate initial modular sequences
    best_h, best_c5 = generate_modular_sequences(n_points)
    
    # Try a smooth version of the best sequence
    h_smooth = smooth_boundaries(best_h)
    c5_smooth = evaluate_c5(h_smooth)
    
    if c5_smooth < best_c5:
        best_h, best_c5 = h_smooth, c5_smooth
    
    # Run gradient descent refinement 
    if time.time() - start_time < budget_s - 10:
        h_refined, c5_refined = gradient_descent_refinement(best_h, n_points / 2)
        if c5_refined < best_c5:
            best_h, best_c5 = h_refined, c5_refined
    
    # Final optimization using SLSQP
    if time.time() - start_time < budget_s - 5:
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                best_h = res.x
            best_h = project_to_sum_box(best_h, n_points / 2)
            best_c5 = evaluate_c5(best_h)
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
