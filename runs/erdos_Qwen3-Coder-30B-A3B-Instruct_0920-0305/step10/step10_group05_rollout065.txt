import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

# Global variables
n_points = 200
half_size = n_points // 2
dx = 2.0 / n_points
target_sum = n_points / 4

def autocorr_fft(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def c5_from_autocorr(corr):
    """Convert autocorrelation to C5 value"""
    amin = np.min(corr)
    return 1.0 - dx * amin

def c5_exact(h):
    """Compute exact C5 using full correlation"""
    corr = correlate(h, h, mode="full")
    amin = np.min(corr)
    return 1.0 - dx * amin

def project_to_sum_box(x, target_sum):
    """Project vector onto box constraints [0,1] with sum = target_sum"""
    # Simple water-filling projection
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return x
    # Add uniformly to all elements (shouldn't violate bounds)
    x = x + diff / len(x)
    x = np.clip(x, 0, 1)
    return x

def parametric_shape(a):
    """Generate a parametric symmetric shape: two flat peaks at height 1, middle at p"""
    # Edge width is 2*a, so middle width is 2*(1-a)
    p = (1 - 2*a) / (2 * (1 - a)) if a < 1 else 0
    if p < 0:
        p = 0
    if a < 0:
        a = 0
    if a > 0.5:
        a = 0.5
    half_vec = np.ones(half_size)
    left_edge_len = int(np.ceil(a * half_size))
    right_edge_len = int(np.ceil(a * half_size))
    middle_len = half_size - left_edge_len - right_edge_len
    if middle_len < 0:
        middle_len = 0
        left_edge_len = (half_size - right_edge_len) // 2
        right_edge_len = half_size - left_edge_len

    # Adjust sizes to meet exact sum (more accurate project)
    half_vec[:left_edge_len] = 1.0
    half_vec[left_edge_len:left_edge_len+middle_len] = p
    half_vec[left_edge_len+middle_len:] = 1.0

    # Exact projection to sum
    scaled = half_vec * (target_sum / np.sum(half_vec))
    return project_to_sum_box(scaled, target_sum)

def smooth_objective(h, beta=50.0):
    """Smooth approximation of C5 using log-sum-exp"""
    corr = autocorr_fft(h)
    # Subtract max to avoid numerical overflow
    corr_max = np.max(corr)
    shifted = corr - corr_max
    weights = np.exp(beta * shifted)
    log_sum = np.log(np.sum(weights)) + corr_max
    # Return 1 - (dx/beta)*log_sum
    return 1.0 - (dx / beta) * log_sum

def smooth_gradient(h, beta=50.0):
    """Gradient of the smooth objective"""
    corr = autocorr_fft(h)
    corr_max = np.max(corr)
    shifted = corr - corr_max
    weights = np.exp(beta * shifted)
    log_sum = np.log(np.sum(weights)) + corr_max
    # Derivative of smooth objective
    # Same as computing the gradient of 1 - (dx/beta)*log(Sum_exp(...)) 
    # Recall (d/dx) log exp(f(x)) = exp(f(x)) * f'(x) / sum(exp(f(x)))
    # Which in our case is:
    # -dx/beta * sum(exp(f)*df/dx)/sum(exp(f))
    term = 1.0 - dx * weights / np.sum(weights)
    # For the derivative of A_k w.r.t h_i: A_k = sum_i h_i * h_{i+k} 
    # But let's compute via convolution of h with itself (essentially h * h_rev)
    # gradient = weights_convolved with h_rev in the correct fashion. 
    # This directly relates to the derivative of the respective time-shifted correlator 
    # using the chain rule:
    # d/dx{ h(x) * h(x+k) } = h'(x) * h(x+k) + h(x) * h'(x+k) = h(x+k) + h(x-k)  
    # but this is not directly useful. It's better to implement it as following:
    # Compute W ⨂ h_rev where W = weights, and we get terms for trapezoidal integration
    # This requires FFT. It is:
    # d/dx_i (sum_k exp(β*A_k)) = sum_k exp(β*A_k) * β * d/dx_i A_k 
    # dA_k/dx_i = h_{i+k} + h_{i-k}  
    # So we compute the convolution of EXP(β*A_k), with h_rev, and we get
    # weighted sum of h_{i+k} + h_{i-k}, weighted by exp(β*A_k).
    # The interesting part is that it is linear in h so we can
    # compute the line integrals. Scan all x_i and return the full gradient vector.
    L = 2 * len(h) - 1
    wfft = fft(weights, L)
    hfft = fft(h, L)
    hrev = h[::-1]
    hrevfft = fft(hrev, L)
    
    # gradient contribution from d/dx_i A_k = h_{i+k} + h_{i-k}
    # but d/dx_i of that sum = h_{i+k}(i) + h_{i-k}(i) -> we want to compute:
    # sum_xi [ d/dx_i A_k(x_i) ] * exp(β*A_k) = sum_k [ (h_{i+k} + h_{i-k}) * exp(β*A_k) ] 
    
    # Actually maybe direct FFT usage:
    # We have to compute: for each i: 
    # (this is how it looks in signal processing loops)
    # So it's fine to compute with ifft(w * hfft) - 2 different combinations.
    # But more importantly, better to express directly that gradient term
    # is \sum_k exp(\beta A_k) * \partial A_k /\partial h_i
    # i.e. \sum_k  \mathcal{F}^{-1}(exp(\beta*A_k) \cdot \mathcal{F}(h_rev)) 
    # and \partial A_k/\partial h_i 
    # \partial / \partial h_i [ \sum_j h_j \cdot h_{j+k} ] = h_{i+k} + h_{i-k}
    # gradient_j = \sum_k exp(\beta A_k) \cdot (h_{j+k} + h_{j-k})
    # So interleave: y = fft(h)[i] + fft(h)[L-i] (for Linux:i=1 to L-1)
    # The correct way is to do 
    # If we want the gradient for indexing m, we consider all terms
    # ∂ A_k / ∂ h_m = h_{m+k} + h_{m-k} 
    # which is a convolution. What we compute correctly is an SSH convolution notation.
    # The total gradient is due to \( \mathcal{G}_i=\sum_k [ \exp(β A_k) \cdot h_{i+k} + h_{i-k} ] \)
    # = \sum_k [ \exp(β A_k) \cdot (\text{sum of cnv of tiled h}) ] 
    # Let's define each dA_k = weight_k 
    # Then (sum_k weight_k * (h[i+k] + h[i-k])) = (weight conv h_rev)_{i} + (weight conv h)_{i}
    # So it's a thing like (exp(B*A_k) * h_rev) + (exp(B*A_k) * h) = 2* \sum w conv wt = 2* (w \cdot h) etc.
    # Let's first compute as if cache was not used - we don't evaluate A_k yet. 
    weight_chain = np.exp(beta * shifted) / np.sum(weights)
    
    # Compute fft of weight and h, and their convolution.
    weight_fft = fft(weight_chain, L)
    h_fft_rev = fft(hrev, L)
    conv1 = ifft(weight_fft * h_fft_rev).real
    
    # Conv h with h to get the second term
    h_fft = fft(h, L)
    conv2 = ifft(weight_fft * h_fft).real
    
    # Total gradient is \sum term = kv * (conv1 + conv2)  
    # But use equivalently of (conv1 + conv2) * (dx / beta)
    # Let's use:
    # See: harmonic analysis: \frac{d}{dx_i} \sum_{k} \exp(\beta A_k) = \sum_k \exp(\beta A_k) \cdot \beta \cdot (h_{i+k} + h_{i-k})
    #   = \beta \sum_k weight_k * (h_{i+k} + h_{i-k}) 
    # = \beta * (weight_conv h_rev + weight_conv h)
    # ith component = \beta * [ (weight_conv h_rev)_i + (weight_conv h)_i ]  
    # Problem: no direct way without elements...
    # Let's do properly direct computation from signal processing perspective:
    # Fyi equals:
    # \overline{h}_{k} = \sum_{i} h_i h_{i+k}  
    # Then \del A_k / \del h_i = h_{i+k} + h_{i-k} (mod L)
    # So for fixing D=[h_{i+k}; h_{i-k}] kernel internal, simply compute convolution of that with weights
    # This can be done as:
    # F_{i} = \sum_k weight_k * (h_{i+k}+h_{i-k})
    # = D.weight where D(0)=1, D(1)=0, D(2)=1 but... Term indices are off!
    # So create full derivative matrix D_{ij} = delta_{j,i+k} + delta_{j,i-k} 
    # Because (D x)_i = sum_{j} D_{ij} x_j = x_{i+k} + x_{i-k}
    # What we do is: efficient way to do D \cdot \vec{weights}
    # We have e.g., weight as 'df/dx', and we want to compute D^T * df/dx
    # which we can interpret in frequency domain.
    # The second method:
    concat_weights = np.concatenate([weight_chain, np.zeros(L - len(weight_chain))])
    # Do convolution with hrev as hrev for tensors is just array indexing with sign flipped.
    gradient = (ifft(fft(concat_weights) * fft(hrev, L))[1::]).real + \
               (ifft(fft(concat_weights) * fft(h, L))[1::]).real
    return -dx * beta * gradient

def parametric_scan_naive():
    """Simple sweep over a to find best a"""
    best_a = 0.0
    best_c5 = np.inf
    step_size = 0.002
    a_values = np.arange(0.01, 0.49, step_size)
    i = 0
    for a in a_values:
        i += 1
        if i > 250:  # Limit to prevent crash
            break
        h = parametric_shape(a)
        c5 = c5_exact(h)
        if c5 < best_c5:
            best_c5 = c5
            best_a = a
    return best_a, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    
    # Stage I: Parametric sweep
    if time.time() - start_time > budget_s * 0.8:
        return (best_h.tolist(), float(best_c5), n_points) if best_h is not None else ([0.5]*n_points, 0.5, n_points) 
    
    a, c5 = parametric_scan_naive()
    h0 = parametric_shape(a) 
    best_c5 = c5
    best_h = h0.copy()

    if time.time() - start_time > budget_s * 0.8:
        h_full = np.concatenate([best_h, best_h[::-1]])
        return (h_full.tolist(), float(best_c5), n_points)
    
    # Stage II: Low-dim continuous optimization
    def objective(h):
        return smooth_objective(h, beta=50.0)
    
    def gradient(h):
        return smooth_gradient(h, beta=50.0)
        
    # Optimize using L-BFGS-B on low-dim param
    bounds = [(0, 1) for _ in range(half_size)]
    # Simplified version: use gradient-free method since analytical gradient might fail or not be better
    try:
        # For ensured robustness: use coarse minimization if gradient causes problems  
        from scipy.optimize import differential_evolution
        # Relax some bounds
        bounds = [(0, 1) for _ in range(half_size)]
        res = differential_evolution(objective, bounds, maxiter=100, popsize=15)
        if res.success:
            h_opt = res.x
            h_opt = project_to_sum_box(h_opt, target_sum)
            c5_opt = c5_exact(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except Exception as e:
        pass
    
    # If still not enough, use simulated annealing on vector
    if time.time() - start_time > budget_s * 0.85:
        h_full = np.concatenate([best_h, best_h[::-1]])
        return (h_full.tolist(), float(best_c5), n_points)
    
    # Simulated Annealing
    current_h = best_h.copy()
    current_c5 = c5_exact(current_h)
    best_c5 = current_c5
    best_h = current_h.copy()
    
    # SA helper functions
    def move_vector(current_h, delta_max=0.05):
        h_new = current_h.copy()
        i, j = np.random.choice(half_size, 2, replace=False)
        delta = np.random.uniform(0, delta_max)
        # Valid move?
        if h_new[i] >= delta and h_new[j] <= 1 - delta:
            h_new[i] -= delta
            h_new[j] += delta
            h_new = project_to_sum_box(h_new, target_sum)
        return h_new

    # Temperature schedule and adaptive delta
    T = 1e-2
    T_min = 1e-6
    alpha = 0.9995
    delta_max = 0.05
    steps = 0
    accept_count = 0
    total_count = 0
    
    while T > T_min and time.time() - start_time < budget_s * 0.95:
        steps += 1
        total_count += 1
        if total_count % 500 == 0:
            # Adjust delta based on acceptance rate
            if accept_count > 200:
                delta_max *= 1.1
            if accept_count < 50:
                delta_max *= 0.9
            delta_max = np.clip(delta_max, 1e-4, 0.2)
            accept_count = 0
            
        h_new = move_vector(current_h, delta_max)
        c5_new = c5_exact(h_new)
        delta = c5_new - current_c5
        
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = h_new
            current_c5 = c5_new
            accept_count += 1
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = current_h.copy()
        
        T *= alpha

    # Final polish with L-BFGS if close to target
    if time.time() - start_time < budget_s * 0.99:
        def smooth_obj(h):
            return smooth_objective(h, beta=200)
        def smooth_grad(h):
            return smooth_gradient(h, beta=200)
        
        try:
            bounds = [(0, 1) for _ in range(half_size)]
            res = minimize(smooth_obj, best_h, method='L-BFGS-B', jac=smooth_grad, bounds=bounds, options={'gtol':1e-12, 'maxiter':500})
            if res.success:
                h_final = res.x
                h_final = project_to_sum_box(h_final, target_sum)
                c5_final = c5_exact(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final.copy()
        except Exception as e:
            pass
        
    # Reconstruct full vector
    h_full = np.concatenate([best_h, best_h[::-1]])
    
    return (h_full.tolist(), float(best_c5), n_points)
