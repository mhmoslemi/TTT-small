import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from math import gcd

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_linear_congruential_sequence(a, b, n):
    """Generate binary sequence using linear congruential method"""
    seq = np.zeros(n, dtype=int)
    for i in range(n):
        val = (a * i + b) % n
        seq[i] = 1 if val < n // 2 else 0
    return seq

def generate_beatty_sequence(a, n):
    """Generate binary sequence using Beatty sequence method"""
    seq = np.zeros(n, dtype=int)
    for i in range(n):
        val = (i * a) % n
        seq[i] = 1 if val < n // 2 else 0
    return seq

def smooth_step_function(h, window_size=3):
    """Smooth step function by replacing boundaries with linear interpolation"""
    h_smooth = h.copy()
    boundaries = np.where(np.diff(h) != 0)[0]
    
    for b in boundaries:
        start = max(0, b - window_size // 2)
        end = min(len(h), b + window_size // 2 + 1)
        
        # Create linear interpolation between adjacent values
        if start > 0 and end < len(h):
            val1 = h[start-1]
            val2 = h[end]
            x = np.linspace(0, 1, end - start)
            interp_values = val1 + (val2 - val1) * x
            h_smooth[start:end] = interp_values
    
    # Adjust sum to match requirement
    current_sum = np.sum(h_smooth)
    delta = (len(h_smooth) / 2) - current_sum
    h_smooth += delta / len(h_smooth)
    
    return np.clip(h_smooth, 0, 1)

def compute_gradient(h, alpha=10):
    """Compute gradient of softmax surrogate"""
    corr = compute_corr_fft(h)
    corr_values = corr[len(corr)//2:]
    max_corr = np.max(corr_values)
    shifted_corr = corr_values - max_corr
    
    exp_corr = np.exp(alpha * shifted_corr)
    weights = exp_corr / np.sum(exp_corr)
    
    grad = np.zeros_like(h)
    for i in range(len(h)):
        grad_term = 0
        for k in range(len(corr_values)):
            if i + k < len(h):
                grad_term += weights[k] * (-h[i+k])
            if i - k >= 0:
                grad_term += weights[k] * (-h[i-k])
        grad[i] = 1 - grad_term
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate candidate sequences using linear congruential method
    best_c5 = float('inf')
    best_h = None
    best_params = None
    
    # Try different coprime values for the Beatty-like construction
    coprimes = []
    for a in range(1, n_points):
        if gcd(a, n_points) == 1:
            coprimes.append(a)
    
    # Try a sampling of coprimes and offsets
    candidates = []
    for a in coprimes[:20]:
        for b in range(0, min(20, n_points)):
            candidates.append((a, b))
    
    print(f"Generating {len(candidates)} candidates...")
    for i, (a, b) in enumerate(candidates):
        if time.time() - start_time > budget_s - 50:
            break
            
        seq = generate_linear_congruential_sequence(a, b, n_points)
        h = seq.astype(float)
        h = project_to_sum_box(h, target_sum)
        c5 = evaluate_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_params = (a, b)
    
    print(f"Best params: {best_params}, C5: {best_c5}")
    
    # Apply smoothing to the best binary sequence
    if best_h is not None:
        h_smooth = smooth_step_function(best_h)
        h_smooth = project_to_sum_box(h_smooth, target_sum)
        c5_smooth = evaluate_c5(h_smooth)
        
        if c5_smooth < best_c5:
            best_c5 = c5_smooth
            best_h = h_smooth.copy()
    
    # Local search optimization
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Hill climbing - swap-based local search
    max_iterations = 5000
    for i in range(max_iterations):
        if time.time() - start_time > budget_s - 30:
            break
            
        # Find worst lag
        corr = compute_corr_fft(current_h)
        lag_indices = np.arange(len(corr)//2, len(corr))
        overlap_values = corr[lag_indices]
        worst_lag = np.argmax(overlap_values)
        
        # Try random swaps to improve
        if worst_lag < len(current_h):
            # Identify good indices to swap
            i1s = np.where(current_h == 1)[0]
            i0s = np.where(current_h == 0)[0]
            
            if len(i1s) > 0 and len(i0s) > 0:
                # Find indices where swapping would help
                i1 = np.random.choice(i1s)
                i0 = np.random.choice(i0s)
                
                # Make trial swap
                h_trial = current_h.copy()
                h_trial[i1], h_trial[i0] = h_trial[i0], h_trial[i1]
                h_trial = project_to_sum_box(h_trial, target_sum)
                
                c5_trial = evaluate_c5(h_trial)
                
                if c5_trial < current_c5:
                    current_h = h_trial.copy()
                    current_c5 = c5_trial
    
    # Gradient-based refinement of the final result
    try:
        max_iter = 200
        step_size = 0.01
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            grad = compute_gradient(current_h)
            current_h = current_h - step_size * grad
            current_h = project_to_sum_box(current_h, target_sum)
            current_c5 = evaluate_c5(current_h)
            
    except:
        pass
    
    # Final refinement with L-BFGS
    try:
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - target_sum
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        res = minimize(
            objective, 
            current_h, 
            method='L-BFGS-B', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        
        if res.success:
            current_h = res.x
        current_h = project_to_sum_box(current_h, target_sum)
        current_c5 = evaluate_c5(current_h)
        
    except:
        pass
    
    return (current_h.tolist(), float(current_c5), n_points)
