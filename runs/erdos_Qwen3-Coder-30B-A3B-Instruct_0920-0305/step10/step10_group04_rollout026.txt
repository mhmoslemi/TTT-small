import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points
TIME_BUDGET = 600

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def initial_h_values():
    # Import initial values from the global context
    return initial_h_values  # Note: this assumes initial_h_values is accessible globally in environment

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def incremental_update_corr(x, i, j, A):
    """Compute ΔA, the incremental change to correlation after swapping x[i] and x[j]"""
    n = len(x)
    x_pad = np.concatenate(([0]*(n-1), x.tolist(), [0]*(n-1)))
    # contributions from i
    c_i = x_pad[i+n-1:i+3*n-2] + x_pad[i:i+2*n-1][::-1]
    # contributions from j
    c_j = x_pad[j+n-1:j+3*n-2] + x_pad[j:j+2*n-1][::-1]
    delta = c_j - c_i
    return delta

def compute_overlap_vector(A, x, n):
    """Compute the overlap vector O(d) from auto-correlation A and x"""
    cum_x = np.concatenate([[0], np.cumsum(x)]) # cumulative sum of x with leading 0
    O = np.zeros(2*n - 1)
    for d in range(1-n, n):
        if d >= 0:
            O[d+n-1] = cum_x[n-d] - A[d+n-1]
        else:
            O[d+n-1] = cum_x[n] - A[d+n-1] # d < 0, equivalent to sum(x[:n+d])
    return O

def binary_to_real(x, target_sum):
    """Convert binary x (sum = target_sum) to a real vector, projecting if needed"""
    if np.sum(x) - target_sum > 1e-10:
        # Under-project
        real_h = x.astype(float)
        real_h = project_to_sum_box(real_h, target_sum)
    else:
        real_h = x.astype(float)
    return real_h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Step 1: Initialize binary vector
    x = np.zeros(n_points, dtype=int)
    ones_count = int(TARGET_SUM)
    x[:ones_count] = 1
    np.random.shuffle(x)
    
    # Step 2: Initial correlation calculation
    A = correlate(x, x, mode='full')
    O = compute_overlap_vector(A, x, n_points)
    current_c5 = c5_from_corr(O, n_points)
    
    best_c5 = current_c5
    best_x = x.copy()
    
    # Simulated annealing parameters
    T0 = 2 * dx
    Tf = 1e-6 * dx
    steps = 300000
    step_size = max(1, steps // 10)
    temp_schedule = np.linspace(T0, Tf, steps)
    
    # Track history for stagnation detection
    last_improvement = 0
    best_history = []
    
    # Main SA loop
    for t in range(steps):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Choose swap type
        if np.random.random() < 0.6:
            # Targeted move
            d_star = np.argmax(O) - (n_points - 1)
            U = []
            V = []
            if d_star >= 0:
                for i in range(n_points - d_star):
                    if x[i] == 1 and x[i + d_star] == 0:
                        U.append(i)
                    if x[i] == 0 and x[i + d_star] == 1:
                        V.append(i)
            elif d_star < 0:
                for i in range(-d_star, n_points):
                    if x[i] == 1 and x[i - d_star] == 0:
                        U.append(i)
                    if x[i] == 0 and x[i - d_star] == 1:
                        V.append(i)
            
            if U and V:
                i = np.random.choice(U)
                j = np.random.choice(V)
            else:
                # Fall back to random if no good candidates
                i, j = np.random.choice(n_points, size=2, replace=False)
                while x[i] == x[j]:
                    i, j = np.random.choice(n_points, size=2, replace=False)
        else:
            # Random swap
            i, j = np.random.choice(n_points, size=2, replace=False)
            while x[i] == x[j]:
                i, j = np.random.choice(n_points, size=2, replace=False)
        
        # Act on the swap
        old_c5 = current_c5
        x[i], x[j] = x[j], x[i]  # Actual swap
        
        delta = incremental_update_corr(x, i, j, A)
        A += delta
        O = compute_overlap_vector(A, x, n_points)
        current_c5 = c5_from_corr(O, n_points)
        
        T = temp_schedule[t]
        
        # Accept or reject
        if current_c5 < old_c5 or np.random.random() < np.exp(-(current_c5 - old_c5) / T):
            # Accept
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_x = x.copy()
                last_improvement = t
        else:
            # Revert the swap
            x[i], x[j] = x[j], x[i]
            current_c5 = old_c5
        
        # Check for stagnation and inject fresh diversity
        if t - last_improvement > 20000 and t % 10000 == 0:
            # Partial re-randomisation
            n_flip = int(0.05 * n_points)
            indices = np.random.choice(n_points, n_flip, replace=False)
            for idx in indices:
                x[idx] = 1 - x[idx]
            # Adjust to correct sum
            current_sum = np.sum(x)
            if current_sum != TARGET_SUM:
                diff = int(TARGET_SUM - current_sum)
                # Flip back the appropriate number of values
                to_change = np.where(np.random.random(n_points) < abs(diff)/n_points)[0]
                if len(to_change) > abs(diff):
                    to_change = to_change[:abs(diff)]
                for idx in to_change:
                    x[idx] = 1 - x[idx]
        
        # Maintain history for future use
        best_history.append(current_c5)
        if len(best_history) > 1000:
            best_history.pop(0)
    
    # Phase 2: Continuous refinement
    # Convert to real numbers using binary search projection
    best_real = binary_to_real(best_x, TARGET_SUM)
    
    # First SLSQP
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
        res = minimize(
            evaluate_c5,
            best_real,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False}
        )
        if res.success:
            best_real = res.x
            best_real = project_to_sum_box(best_real, TARGET_SUM)
        # Second refinement with smoothing
        window_size = 5
        if window_size <= n_points:
            smoothed = np.convolve(best_real, np.ones(window_size)/window_size, mode='same')
            smoothed = project_to_sum_box(smoothed, TARGET_SUM)
            res = minimize(
                evaluate_c5,
                smoothed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}
            )
            if res.success:
                best_real = res.x
                best_real = project_to_sum_box(best_real, TARGET_SUM)
    except Exception:
        pass
    
    # Final projection to guarantee sum constraint exactly
    final_best_real = project_to_sum_box(best_real, TARGET_SUM)
    
    return (final_best_real.tolist(), float(best_c5), n_points)
