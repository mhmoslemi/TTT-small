def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_smooth_max(h, alpha):
    """Evaluate the smooth max of correlations using log-sum-exp"""
    corr = compute_corr_fft(h)
    corr = corr[:len(corr)//2 + 1]  # Only positive lags
    max_corr = np.max(corr)
    # Numerically stable log-sum-exp
    exp_corr = np.exp(alpha * (corr - max_corr))
    log_sum_exp = max_corr + np.log(np.sum(exp_corr)) - np.log(len(exp_corr))
    return log_sum_exp

def compute_gradient(h, alpha):
    """Compute gradient of smooth max using FFT"""
    corr = compute_corr_fft(h)
    corr = corr[:len(corr)//2 + 1]  # Only positive lags
    max_corr = np.max(corr)
    exp_corr = np.exp(alpha * (corr - max_corr))
    sum_exp = np.sum(exp_corr)
    weights = exp_corr / sum_exp
    # Symmetrize weights
    weights_sym = weights.copy()
    weights_sym = np.concatenate([weights_sym[:-1][::-1], weights_sym])
    # Correlation of symmetrized weights with h
    g_full = np.fft.ifft(np.fft.fft(weights_sym, 2*len(h)-1) * np.fft.fft(h, 2*len(h)-1)).real
    g = g_full[:len(h)]
    grad = 1 - g
    return grad

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    eps = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < eps:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_exact_c5(h):
    """Evaluate the exact C5 value"""
    corr = compute_corr_fft(h)
    max_corr = np.max(corr) * dx
    return max_corr
