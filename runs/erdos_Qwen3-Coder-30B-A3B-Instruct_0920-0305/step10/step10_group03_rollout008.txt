# Linear programming wrapper for discrete constraint handling
def solve_milp_step_function(h_guess, steps=4):
    n_points = len(h_guess)
    dx = 2.0 / n_points
    
    # Create integer variables for step pattern (binary)
    step_vars = cp.Variable(steps, boolean=True)
    
    # Generate possible step positions (evenly spaced)
    step_positions = np.linspace(0, 2, steps + 1)[1:-1]
    step_values = cp.Variable(steps)
    
    # Objective to minimize max correlation element
    # This will be calculated post-solve
    total_var = cp.Variable(1)
    
    # Equality constraint: integral = 1
    integral_constraint = cp.sum(cp.multiply(step_values, np.diff(step_positions))) == 1.0
    
    # Box constraints
    bounds_constraints = [0 <= step_values, step_values <= 1]
    
    # Total constraints
    constraints = bounds_constraints + [integral_constraint]
    
    # Simple dummy objective since we're more interested in feasibility
    objective = cp.Minimize(total_var)
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=CPLEX, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            step_vals = step_values.value
            return step_vals, step_positions
    except Exception as e:
        pass
    return step_vals if 'step_vals' in locals() else None, step_positions

def compute_overlap(h):
    """Compute the overlap using full convolution"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def optimize_constrained_step_function(h_guess, n_steps=4, max_iter=500):
    """Use gradient-based optimization on a constrained step function approach"""
    dx = 2.0 / len(h_guess)

    step_locs = np.linspace(0, 2, n_steps + 1)[1:-1]

        # Construct step function from step values
        h = np.zeros_like(h_guess)
        d = 2.0 / len(h_guess)

            start_idx = int(step_locs[i] / d)
            end_idx = int((step_locs[i] + 1.0) / d) if i < n_steps - 1 else len(h)
            h[start_idx:end_idx] = step_vals[i]
            
        return compute_overlap(h)
    
    def constraint_sum(step_vals):
        h = np.zeros_like(h_guess)
        d = 2.0 / len(h_guess)

            start_idx = int(step_locs[i] / d)
            end_idx = int((step_locs[i] + 1.0) / d) if i < n_steps - 1 else len(h)
            h[start_idx:end_idx] = step_vals[i]
        return np.sum(h) * d - 1.0
    
    bounds = [(0.001, 0.999) for _ in range(n_steps)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        res = minimize(objective, step_heights, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-10})
        if res.success:
            best_vals = res.x
            h_final = np.zeros_like(h_guess)
            d = 2.0 / len(h_guess)
            for i in range(n_steps):
                start_idx = int(step_locs[i] / d)
                end_idx = int((step_locs[i] + 1.0) / d) if i < n_steps - 1 else len(h_final)
                h_final[start_idx:end_idx] = best_vals[i]
            return h_final
    except Exception as e:
        return h_guess
    return h_guess
