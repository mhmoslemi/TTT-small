def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_milp(N, time_limit):
    """Solve binary integer programming for optimal step function"""
    # Create variables
    x = cp.Variable(N, boolean=True)
    t = cp.Variable()
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    
    # Constraints for each shift
    constraints = [cp.sum(x) == N // 2]
    
    # For each shift k, compute the sum of products x[i] * x[i+k]
    # and add constraint that sum >= N/2 - t
    for k in range(N):
        if k == 0:
            # For k=0, x[i]*x[i+k] = x[i]*x[i] = x[i] (since binary)
            # sum_i x[i] = N/2, so this is already satisfied
            continue
        else:
            # Compute sum of products for shift k
            # y[i] = x[i] * x[i+k] for i where i+k < N
            y = cp.Variable(N - k, boolean=True)
            # Constraint bounds for y
            for i in range(N - k):
                # y <= x[i]
                constraints.append(y[i] <= x[i])
                # y <= x[i+k]  
                constraints.append(y[i] <= x[i + k])
                # y >= x[i] + x[i+k] - 1
                constraints.append(y[i] >= x[i] + x[i + k] - 1)
            
            # Constraint for minimum overlap
            constraints.append(cp.sum(y) >= N // 2 - t)
    
    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    try:
        prob.solve(solver=cp.CBC, maximumSeconds=time_limit, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value.astype(int)
    except:
        pass
    
    return None

def upsample_binary(binary_vec, target_N):
    """Upsample binary vector to higher resolution"""
    factor = target_N // len(binary_vec)
    if factor > 1:
        return np.repeat(binary_vec, factor).astype(float)
    return binary_vec.astype(float)

def local_binary_improvement(x, target_sum, time_budget, start_time):
    """Specialized local search for binary patterns"""
    N = len(x)
    max_iter = 100000
    
    # Compute initial overlaps
    def compute_overlaps(x):
        overlaps = []
        for k in range(N):
            overlap = 0
            if k == 0:
                overlap = np.sum(x)  # For shift 0, all positions overlap
            else:
                # Compute overlap at shift k (x[i] * x[i+k])
                for i in range(N - k):
                    overlap += x[i] * x[i + k]
            overlaps.append(overlap)
        return overlaps
    
    def update_overlaps_delta(overlaps, x, a, b):
        # Update overlaps due to swap of positions a and b
        new_overlaps = overlaps.copy()
        # For all shifts k where a or b participate:
        for k in range(N):
            if a + k < N:
                # Update for shift k where a participates in product
                new_overlaps[k] += x[b] * x[a + k] - x[a] * x[a + k]
            if b + k < N:
                # Update for shift k where b participates in product
                new_overlaps[k] += x[a] * x[b + k] - x[b] * x[b + k]
        return new_overlaps
    
    current_overlaps = compute_overlaps(x)
    current_max = np.max(current_overlaps)
    best_x = x.copy()
    best_max = current_max
    
    iter_count = 0
    
    while time.time() - start_time < time_budget and iter_count < max_iter:
        iter_count += 1
        # Find worst shift
        k_max = np.argmax(current_overlaps)
        # Keep track of positions where we can improve
        candidates_a = []
        candidates_b = []
        
        # For shift k_max, find which positions could help
        for i in range(N - k_max):
            if x[i] == 1 and x[i + k_max] == 0:
                candidates_a.append(i)
            if x[i] == 0 and x[i + k_max] == 1:
                candidates_b.append(i)
        
        # Try swapping
        if candidates_a and candidates_b:
            a = np.random.choice(candidates_a)
            b = np.random.choice(candidates_b)
            # Try the swap
            new_x = x.copy()
            new_x[a] = 0
            new_x[b] = 1
            # Compute new overlaps with delta update
            new_overlaps = update_overlaps_delta(current_overlaps, x, a, b)
            new_max = np.max(new_overlaps)
            
            if new_max <= current_max:
                # Accept the swap
                x = new_x
                current_overlaps = new_overlaps
                current_max = new_max
                if new_max < best_max:
                    best_x = x.copy()
                    best_max = new_max
            else:
                # Occasionally accept worse swaps
                if np.random.rand() < 0.05:
                    x = new_x
                    current_overlaps = new_overlaps
                    current_max = new_max
        else:
            # Try random swap if no pairs found
            i = np.random.randint(N)
            j = np.random.randint(N)
            if x[i] != x[j]:
                new_x = x.copy()
                new_x[i] = 1 - x[i]
                new_x[j] = 1 - x[j]
                # Compute new overlaps (slower than delta for big N but OK)
                new_overlaps = compute_overlaps(new_x)
                new_max = np.max(new_overlaps)
                
                if new_max <= current_max:
                    x = new_x
                    current_overlaps = new_overlaps
                    current_max = new_max
                    if new_max < best_max:
                        best_x = x.copy()
                        best_max = new_max
                elif np.random.rand() < 0.05:
                    x = new_x
                    current_overlaps = new_overlaps
                    current_max = new_max
    
    # Final step: extract best solution that satisfies sum constraint
    best_x = project_to_sum_box(best_x, target_sum)
    return best_x, best_max

        return np.sum(h) - target_sum

    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = differential_evolution(
            objective, 
            bounds, 
            constraints=cons,
            seed=42,
            maxiter=100,
            popsize=30,
            mutation=(0.5, 1.0),
            recombination=0.7,
            atol=1e-6,
            x0=h0,
            callback=lambda x, convergence: None
        )
        return project_to_sum_box(res.x, target_sum), evaluate_c5(project_to_sum_box(res.x, target_sum))
    except:
        return h0, evaluate_c5(h0)
