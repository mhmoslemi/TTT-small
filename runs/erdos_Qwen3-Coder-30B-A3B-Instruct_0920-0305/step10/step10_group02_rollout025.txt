def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_modular_sequence(a, b, n):
    """Generate sequence using modular arithmetic"""
    h = np.zeros(n)
    for i in range(n):
        if ((a * i + b) % n) < n / 2:
            h[i] = 1
    return h

def smooth_function(h, window_size=3):
    """Apply smoothing at boundaries"""
    h_smooth = h.copy()
    
    # Find boundary points
    boundaries = []
    for i in range(len(h) - 1):
        if h[i] != h[i+1]:
            boundaries.append(i)
    
    # Apply smoothing
    for b in boundaries:
        start = max(0, b - window_size // 2)
        end = min(len(h), b + window_size // 2 + 1)
        if end - start > 1:
            # Linear interpolation between consecutive points
            if h[start] == 0 and h[end-1] == 1:
                h_smooth[start:end] = np.linspace(0, 1, end - start)
            elif h[start] == 1 and h[end-1] == 0:
                h_smooth[start:end] = np.linspace(1, 0, end - start)
    
    # Adjust sum to target
    current_sum = np.sum(h_smooth)
    delta = (len(h_smooth) / 2) - current_sum
    h_smooth += delta / len(h_smooth)
    
    # Clip to [0,1]
    h_smooth = np.clip(h_smooth, 0, 1)
    return h_smooth
