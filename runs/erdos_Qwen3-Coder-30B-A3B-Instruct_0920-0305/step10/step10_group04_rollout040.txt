import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Start with a random binary initialization
    h0 = np.random.rand(n_points)
    h0 = (h0 > 0.5).astype(float)
    
    # Ensure exactly half ones
    current_sum = np.sum(h0)
    if current_sum > TARGET_SUM:
        # Turn some ones into zeros
        indices = np.where(h0 == 1)[0]
        np.random.shuffle(indices)
        for i in indices[:int(current_sum - TARGET_SUM)]:
            h0[i] = 0
    elif current_sum < TARGET_SUM:
        # Turn some zeros into ones
        indices = np.where(h0 == 0)[0]
        np.random.shuffle(indices)
        for i in indices[:int(TARGET_SUM - current_sum)]:
            h0[i] = 1
    
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    # Simulated Annealing
    max_iterations = 500000
    temp_start = 0.02
    temp_end = 1e-5
    
    current_h = h0.copy()
    current_c5 = best_c5
    
    # Track for early termination
    last_improvement = 0
    stagnation_count = 0
    max_stagnation = 20000
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Update temperature
        temp = temp_start * (1 - iteration / max_iterations) + temp_end
        
        # Try a targeted move with probability 0.7
        if np.random.random() < 0.7:
            # Find the worst shift
            corr = np.correlate(current_h, 1 - current_h, mode="full")
            max_idx = np.argmax(corr)
            d = max_idx - (n_points - 1)
            
            # Find candidates for targeted swaps
            candidates_i = []
            candidates_j = []
            
            for i in range(n_points):
                j = i + d
                if 0 <= j < n_points:
                    if current_h[i] == 1 and current_h[j] == 0:
                        candidates_i.append(i)
                    elif current_h[i] == 0 and current_h[j] == 1:
                        candidates_j.append(i)
            
            if candidates_i and candidates_j:
                i_swap = np.random.choice(candidates_i)
                j_swap = np.random.choice(candidates_j)
                # Perform swap
                current_h[i_swap], current_h[j_swap] = current_h[j_swap], current_h[i_swap]
                new_c5 = compute_c5(current_h)
                
                # Accept or reject
                delta = new_c5 - current_c5
                if delta <= 0 or np.random.random() < np.exp(-delta / temp):
                    current_c5 = new_c5
                    if new_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = new_c5
                        last_improvement = iteration
                else:
                    # Revert swap
                    current_h[j_swap], current_h[i_swap] = current_h[i_swap], current_h[j_swap]
            else:
                # Fallback to random swap
                i = np.random.randint(n_points)
                j = np.random.randint(n_points)
                while current_h[i] == current_h[j]:
                    i = np.random.randint(n_points)
                    j = np.random.randint(n_points)
                
                current_h[i], current_h[j] = current_h[j], current_h[i]
                new_c5 = compute_c5(current_h)
                
                # Accept or reject
                delta = new_c5 - current_c5
                if delta <= 0 or np.random.random() < np.exp(-delta / temp):
                    current_c5 = new_c5
                    if new_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = new_c5
                        last_improvement = iteration
                else:
                    # Revert swap
                    current_h[j], current_h[i] = current_h[i], current_h[j]
        else:
            # Random swap
            i = np.random.randint(n_points)
            j = np.random.randint(n_points)
            while current_h[i] == current_h[j]:
                i = np.random.randint(n_points)
                j = np.random.randint(n_points)
            
            current_h[i], current_h[j] = current_h[j], current_h[i]
            new_c5 = compute_c5(current_h)
            
            # Accept or reject
            delta = new_c5 - current_c5
            if delta <= 0 or np.random.random() < np.exp(-delta / temp):
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = new_c5
                    last_improvement = iteration
            else:
                # Revert swap
                current_h[j], current_h[i] = current_h[i], current_h[j]
        
        # Check for stagnation
        if iteration - last_improvement > max_stagnation:
            # Restart with a new random solution
            h_restart = np.random.rand(n_points)
            h_restart = (h_restart > 0.5).astype(float)
            
            # Ensure exactly half ones  
            current_sum = np.sum(h_restart)
            if current_sum > TARGET_SUM:
                indices = np.where(h_restart == 1)[0]
                np.random.shuffle(indices)
                for i in indices[:int(current_sum - TARGET_SUM)]:
                    h_restart[i] = 0
            elif current_sum < TARGET_SUM:
                indices = np.where(h_restart == 0)[0]
                np.random.shuffle(indices)
                for i in indices[:int(TARGET_SUM - current_sum)]:
                    h_restart[i] = 1
            
            current_h = h_restart
            current_c5 = compute_c5(current_h)
            last_improvement = iteration
    
    # Final refinement with continuous optimization
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
        res = minimize(
            compute_c5, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            refined_h = res.x
            refined_h = project_to_sum_box(refined_h, TARGET_SUM)
            refined_c5 = compute_c5(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
