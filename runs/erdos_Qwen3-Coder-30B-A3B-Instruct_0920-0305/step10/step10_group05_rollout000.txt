```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_autocorr_fft(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def min_autocorr_from_corr(corr, n):
    """Get minimum autocorrelation value"""
    return dx * np.min(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def evaluate_c5_by_autocorr(h):
    """Evaluate C5 using autocorrelation to avoid recomputing correlation"""
    auto_corr = compute_autocorr_fft(h)
    min_auto = np.min(auto_corr)
    return 1.0 - min_auto * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_piecewise_constant(a):
    """Evaluate C5 for piecewise constant symmetric shape"""
    # Parameters
    if a <= 0 or a >= 1:
        return 1.0
    
    # Compute p from constraint: 2*a*1 + (2-2*a)*p = 1
    p = (1 - 2*a) / (2 * (1 - a))
    
    # Check if p is valid
    if p < 0 or p > 1:
        return 1.0
    
    # Generate h
    h = np.zeros(n_points)
    
    # Left edge
    left_edge_idx = int(a * n_points / 2)
    if left_edge_idx > 0:
        h[:left_edge_idx] = 1.0
    
    # Right edge
    right_edge_idx = int((1 - a) * n_points / 2)
    if right_edge_idx > 0:
        h[-right_edge_idx:] = 1.0
    
    # Middle region
    middle_start = left_edge_idx
    middle_end = n_points - right_edge_idx
    if middle_end > middle_start:
        h[middle_start:middle_end] = p
    
    # Normalize to target sum
    h = h * TARGET_SUM / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    # Evaluate
    return evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Try piecewise constant scan for a parameter
    print("Scanning piecewise constant shapes...")
    
    # Precalculate grid of a values to test
    a_values = np.arange(0.01, 0.45, 0.002)
    
    for a in a_values:
        if time.time() - start_time > budget_s * 0.8:
            break
        try:
            c5 = evaluate_piecewise_constant(a)
            if c5 < best_c5:
                best_c5 = c5
                # Generate the actual h vector for this a to return
                if a <= 0 or a >= 1:
                    continue
                p = (1 - 2*a) / (2 * (1 - a))
                if p < 0 or p > 1:
                    continue
                
                h = np.zeros(n_points)
                left_edge_idx = int(a * n_points / 2)
                right_edge_idx = int((1 - a) * n_points / 2)
                if left_edge_idx > 0:
                    h[:left_edge_idx] = 1.0
                if right_edge_idx > 0:
                    h[-right_edge_idx:] = 1.0
                
                middle_start = left_edge_idx
                middle_end = n_points - right_edge_idx
                if middle_end > middle_start:
                    h[middle_start:middle_end] = p
                    
                h = h * TARGET_SUM / np.sum(h)
                h = np.clip(h, 0, 1)
                best_h = h.copy()
        except Exception as e:
            continue
    
    # Now try simulated annealing with best shape as starting point
    print("Starting simulated annealing...")
    max_iter = int(2e5)
    
    if best_h is None:
        # Create initial random h
        h0 = np.random.rand(n_points)
        h0 = project_to_sum_box(h0, TARGET_SUM)
        current_h = h0.copy()
        current_c5 = evaluate_c5(current_h)
    else:
        current_h = best_h.copy()
        current_c5 = evaluate_c5(best_h)
        
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Annealing parameters
    T_start = 0.001
    T = T_start
    cooling_rate = 0.99995
    max_step = 0.05
    
    # Main SA loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Generate new solution by perturbing two random indices
        idx1 = np.random.randint(0, n_points)
        idx2 = np.random.randint(0, n_points)
        
        # Small random delta
        delta = np.random.uniform(-max_step, max_step)
        
        # Ensure we don't go out of bounds with simple perturbation
        if current_h[idx1] + delta >= 0 and current_h[idx1] + delta <= 1 and \
           current_h[idx2] - delta >= 0 and current_h[idx2] - delta <= 1:
            h_new = current_h.copy()
            h_new[idx1] += delta
            h_new[idx2] -= delta
            
            # Project back to constraint
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            c5_new = evaluate_c5(h_new)
            
            # Accept or reject
            if c5_new < current_c5:
                current_h = h_new.copy()
                current_c5 = c5_new
            else:
                # Accept with probability
                if np.random.rand() < np.exp(-(c5_new - current_c5) / T):
                    current_h = h_new.copy()
                    current_c5 = c5_new
            
            # Update best
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Cool down temperature
        T = T_start * (cooling_rate ** i)
    
    # Final refinement with local optimization
    print("Final refinement with local optimization...")
    try:
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        res = minimize(
            objective, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
        else:
            h_refined = best_h
            
        h_refined = project_to_sum_box(h_refined, TARGET_SUM)
        c5_refined = evaluate_c5(h_refined)
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
            
    except Exception:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
```
