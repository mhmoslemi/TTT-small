```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from math import exp, log

# Global constants
n_points = 200
dx = 2.0 / n_points
half_size = n_points // 2
target_sum = n_points / 4

def autocorr_fft(h):
    """Compute autocorrelation via FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def compute_c5_from_correlation(corr):
    """Convert correlation to C5 (lower is better)"""
    amin = np.min(corr)
    return 1 - dx * amin

def project_to_sum_box(x, target_sum):
    """Project vector onto box constraints [0,1] with sum = target_sum"""
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return x
    
    # water filling: add diff to smallest elements
    sorted_indices = np.argsort(x)
    remaining_diff = diff
    x_copy = x.copy()
    
    for i in sorted_indices:
        if remaining_diff > 0:
            max_add = 1 - x_copy[i]
            add = min(remaining_diff, max_add)
            x_copy[i] += add
            remaining_diff -= add
        elif remaining_diff < 0:
            max_sub = x_copy[i]
            sub = min(-remaining_diff, max_sub)
            x_copy[i] -= sub
            remaining_diff += sub
        else:
            break
    return x_copy

def construct_symmetric_h(a, p, n_half):
    """Construct symmetric half-vector given edge length `a` and middle height `p`"""
    # ensure that a is valid for constructing two edges and a middle
    a = max(0., min(a, 0.5))
    # b = 2 - 2*a (reachable edge length), so that mid width = b
    # Area constraint: 2*a*1 + b*p = 1 => 2*a + (2-2*a)*p = 1
    # Solving for p: p = (1-2*a)/(2*(1-a))
    if abs(1 - a) < 1e-10:
        p = 0
    else:
        p = (1 - 2*a) / (2*(1 - a))
    
    p = max(0., min(p, 1))  # Project to [0,1] just in case
    
    half = np.zeros(n_half)
    edge_len = max(0, min(int(a * n_half/1.), n_half//2))  # convert to integer index
    
    # Sawtooth shape: front and back sections at height 1, middle at p
    if edge_len > 0:
        half[:edge_len] = 1
        half[-edge_len:] = 1
        
        # middle region
        mid_start = edge_len
        mid_end = n_half - edge_len
        if mid_end > mid_start:
            half[mid_start:mid_end] = p
    else:
        # if edge_len == 0, make entire thing middle
        half[:] = p
        
    return half

def smooth_objective(h, beta=50):
    """Smooth objective using log-sum-exp of negative autocorrelations"""
    # Compute all autocorrelations via FFT
    corr = autocorr_fft(h)
    # Use log-sum-exp to create smooth max
    max_corr = np.max(corr)
    shifted = corr - max_corr
    weights = np.exp(beta * shifted)
    log_sum_exp = max_corr + log(np.sum(weights)) / beta
    return 1 - dx * log_sum_exp

def smooth_gradient(h, beta=50):
    """Approximate gradient of smooth objective"""
    corr = autocorr_fft(h)
    max_corr = np.max(corr)
    shifted = corr - max_corr
    weights = np.exp(beta * shifted)
    sum_weights = np.sum(weights)
    # Compute gradient through convolution
    # First compute full convolution with weights (non-cyclic, so pad with zeros)
    L = 2 * len(h) - 1
    w_padded = np.zeros(L)
    w_padded[:len(weights)] = weights
    h_rev = np.flip(h)
    h_padded = np.zeros(L)
    h_padded[:len(h)] = h_rev
    grad_convol = np.convolve(w_padded, h_padded, mode='valid')
    # now have divided by sum_weights 
    grad = -dx * (grad_convol / sum_weights)
    return grad

def simulated_annealing(init_h, max_iter=150000, T0=1e-2, alpha=0.9995):
    """Run simulated annealing on the half-vector"""
    current_h = init_h.copy()
    current_c5 = compute_c5_from_correlation(autocorr_fft(current_h))
    best_h = current_h.copy()
    best_c5 = current_c5
    
    T = T0
    accept_count = 0
    total_moves = 0
    delta_step = 0.05
    T_min = 1e-6
    fraction_accept = 0.3  # target acceptance fraction

    start_time = time.time()
    for i in range(max_iter):
        if time.time() - start_time > 570:
            break

        # propose move: selected indices
        k1, k2 = np.random.randint(0, len(current_h), 2)
        delta = np.random.uniform(0, delta_step)
        delta = min(delta, current_h[k1], 1.0 - current_h[k2])

        if delta > 0:
            new_h = current_h.copy()
            new_h[k1] -= delta
            new_h[k2] += delta

            # enfore sum constraint
            new_h = project_to_sum_box(new_h, target_sum)

            # calculate new C5
            new_c5 = compute_c5_from_correlation(autocorr_fft(new_h))

            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
                accept_count += 1
                total_moves += 1
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
            else:
                # Accept with probability
                p_accept = exp(-(new_c5 - current_c5) / T)
                if np.random.rand() < p_accept:
                    current_h = new_h
                    current_c5 = new_c5
                    accept_count += 1
                total_moves += 1

            # adapt step size
            if total_moves % 500 == 0 and total_moves > 0:
                ratio = accept_count / total_moves
                if ratio > fraction_accept:
                    delta_step *= 1.1
                else:
                    delta_step *= 0.9
                delta_step = max(1e-4, min(delta_step, 0.2))
                accept_count = 0
                total_moves = 0

        # decrease temperature
        T = T * alpha
        if T < T_min:
            break

    return best_h, best_c5


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initialize with parametric form
    optimal_a = 0.225  # near known local minimum
    init_h = construct_symmetric_h(optimal_a, 0.5, half_size)
    init_h = project_to_sum_box(init_h, target_sum)

    # Stage I: exhaustive sweep over a
    best_c5 = compute_c5_from_correlation(autocorr_fft(init_h))
    best_h = init_h.copy()
    
    best_a = optimal_a
    sweep_range = np.linspace(0.01, 0.49, 250)
    for a in sweep_range:
        h_temp = construct_symmetric_h(a, 0.5, half_size)
        h_temp = project_to_sum_box(h_temp, target_sum)
        c5_temp = compute_c5_from_correlation(autocorr_fft(h_temp))
        if c5_temp < best_c5:
            best_c5 = c5_temp
            best_h = h_temp.copy()
            best_a = a

    # Stage II: low dimensional optimization with L-BFGS-B
    if time.time() - start_time > budget_s * 0.8:
        return (np.concatenate([best_h, best_h[::-1]]).tolist(), best_c5, n_points)

    def obj(h):
        return smooth_objective(h, beta=50)
    
    def grad(h):
        return smooth_gradient(h, beta=50)
    
    # Optimize the earlier result
    bounds = [(0, 1) for _ in range(len(best_h))]
    res = minimize(obj, best_h, method='L-BFGS-B', jac=grad, bounds=bounds, 
                   options={'ftol': 1e-12, 'gtol': 1e-12, 'maxiter': 500})
    if res.success:
        refined_h = res.x
    else:
        refined_h = best_h
    refined_h = project_to_sum_box(refined_h, target_sum)
    refined_c5 = compute_c5_from_correlation(autocorr_fft(refined_h))
    if refined_c5 < best_c5:
        best_c5 = refined_c5
        best_h = refined_h.copy()

    # Stage III: full SA
    full_h, full_c5 = simulated_annealing(best_h, max_iter=120000)
    if full_c5 < best_c5:
        best_c5 = full_c5
        best_h = full_h.copy()

    # Stage IV: Final polishing with L-BFGS-B
    if time.time() - start_time > budget_s * 0.95:
        return (np.concatenate([best_h, best_h[::-1]]).tolist(), best_c5, n_points)

    # Use a rougher gradient for final polishing
    def obj_final(h):
        return smooth_objective(h, beta=200)
    
    def grad_final(h):
        return smooth_gradient(h, beta=200)
    
    bounds_final = [(0, 1) for _ in range(len(best_h))]
    try:
        res2 = minimize(obj_final, best_h, method='L-BFGS-B', jac=grad_final, bounds=bounds_final,
                        options={'ftol': 1e-12, 'gtol': 1e-12, 'maxiter': 300})
        if res2.success:
            final_h = res2.x
        else:
            final_h = best_h
        final_h = project_to_sum_box(final_h, target_sum)
        final_c5 = compute_c5_from_correlation(autocorr_fft(final_h))
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h.copy()
    except Exception:
        pass

    final_full = np.concatenate([best_h, best_h[::-1]])
    exact_c5 = compute_c5_from_correlation(autocorr_fft(final_full))
    
    return (final_full.tolist(), float(exact_c5), n_points)
```
