def compute_overlap_fast(x):
    """Fast computation of overlap for binary vector using incremental correlation"""
    # Pad the binary vector with zeros
    x_padded = np.pad(x, (n_points-1, n_points-1), mode='constant', constant_values=0)
    # Compute auto-correlation
    A = np.correlate(x_padded, x_padded, mode='full')
    # Compute prefix sums for efficient window overlap counts
    prefix = np.concatenate([[0], np.cumsum(x)])
    # Compute O_d at all offsets using vectorized operations
    max_d = n_points - 1
    O = []
    # More efficient computation: get each O_d incrementally
    for d in range(-max_d, max_d + 1):
        if d >= 0:
            i_start = 0
            i_end = n_points - d
            # ones in range [0, i_end] minus auto-correlations at offset d
            A_d = A[n_points-1+d]
            ones_window = prefix[i_end] - prefix[i_start]
        else:
            i_start = -d
            i_end = n_points
            A_d = A[n_points-1+d]
            ones_window = prefix[i_end] - prefix[i_start]
        O_d = ones_window - A_d
        O.append(O_d)
    return dx * max(O)

def compute_corr_incremental(x, i, j):
    """Compute the change in correlation if we swap bits at indices i and j"""
    n = len(x)
    delta = np.zeros(2 * n - 1)
    # Calculate contribution from position i
    for d in range(-n+1, n):
        if 0 <= i+d < n and 0 <= i-d < n:
            delta[d+n-1] += - (x[i+d] + x[i-d])
    # Calculate contribution from position j
    for d in range(-n+1, n):
        if 0 <= j+d < n and 0 <= j-d < n:
            delta[d+n-1] += (x[j+d] + x[j-d])
    return delta

def project_to_sum_box(x, target_sum):
    """Project vector onto valid binary space with target sum"""
    # Sort indices by value (for binary vectors, this numerically equals sorting by natural order)
    sorted_indices = np.argsort(x)
    # If current sum is now target, we're done
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return np.clip(x, 0, 1)
    
    # Adjust up or down
    if current_sum > target_sum:
        # Reduce the ones that make largest contributing change (reduce small values)
        delta = current_sum - target_sum
        # Find the indices that we can safely reduce while staying >= 0
        to_reduce = sorted_indices[:int(delta)]
        for idx in to_reduce:
            x[idx] = 0
    else:
        # Increase the zeros that would benefit most
        delta = target_sum - current_sum
        to_increase = sorted_indices[-int(delta):]
        for idx in to_increase:
            x[idx] = 1
    return x

def binary_to_real_vector(x_bin):
    """Convert binary vector to real vector using midpoint"""
    return x_bin.astype(float)

def evaluate_c5_continuous(h):
    """Evaluate continuous function using scipy convolution"""
    conv = np.correlate(h, 1-h, mode='full')
    return dx * np.max(conv)
