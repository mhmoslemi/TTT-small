def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Use full correlation to get all shifts
    corr = np.correlate(h, 1-h, mode="full")
    # We want the correlation for shifts from 0 to 2*dx (which is all non-negative shifts)
    # The full correlation is symmetric, we use the results for positive shifts
    # The correlation indexes go from -len+1 to len-1
    # So we take from index len-1 to 2*len-2
    corr_full = corr[len(h)-1:]
    # Convert to C5 bound
    return dx * np.max(corr_full)

def compute_binary_corr(h):
    """Compute correlation efficiently for binary (0/1) vectors"""
    n = len(h)
    # Direct computation with careful handling
    corr = np.zeros(2*n - 1, dtype=float)  # Full correlation
    for i in range(n):
        for j in range(n):
            k = i - j + n - 1  # shift index
            if k >= 0 and k < 2*n - 1:
                corr[k] += h[i] * (1 - h[j])
    return corr

def solve_binary_milp(N, time_limit=60):
    """Solve binary integer program to find optimal 0/1 pattern"""
    try:
        # Variables
        x = cp.Variable(N, boolean=True)
        t = cp.Variable(pos=True)  # This represents integer variable t_int in our formulation
        
        # Constraints
        constraints = [cp.sum(x) == N // 2]

        # For each shift k = 1 to N-1 (as shift 0 is handled already)
        for k in range(1, N):
            y = cp.Variable(N - k, boolean=True)  # y[i] represents x[i]*x[i+k]
            # Linearization constraints for binary products
            for i in range(N - k):
                constraints.append(y[i] <= x[i])
                constraints.append(y[i] <= x[i + k])
                constraints.append(y[i] >= x[i] + x[i + k] - 1)
            
            # At least N/2 - t overlap for this shift
            constraints.append(cp.sum(y) >= N // 2 - t)
            
        # Objective: minimize t (representing max overlap in integer counts)
        prob = cp.Problem(cp.Minimize(t), constraints)
        
        # Solve with time limit
        prob.solve(solver=cp.CBC, max_seconds=time_limit)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value.astype(int)
        else:
            # Return best feasible solution if exists
            return None
            
    except:
        return None

def upsample_binary(binary_vec, target_N):
    """Upsample binary pattern to higher resolution"""
    # Create piecewise constant function by replicating each element
    repeated = np.repeat(binary_vec, target_N // len(binary_vec))
    # Trim or pad to exact length
    if len(repeated) > target_N:
        repeated = repeated[:target_N]
    elif len(repeated) < target_N:
        repeated = np.pad(repeated, (0, target_N - len(repeated)), mode='constant', constant_values=0)
    return repeated.astype(float)

def local_binary_improvement(x, target_sum, time_budget):

        k_max = np.argmax(D)
        
        # Find indices with conflicts
        A = []

        for i in range(n_points - k_max):
            if best_x[i] == 1 and best_x[i + k_max] == 0:
                A.append(i)
                
        # Check indices where we have a 0 but partner is 1 at shift k_max  
        for i in range(n_points - k_max):
            if best_x[i] == 0 and best_x[i + k_max] == 1:
                B.append(i)
                
        if not A or not B:
            # Random move if no moves available
            idx1 = np.random.randint(0, n_points)
            idx2 = np.random.randint(0, n_points)
            if idx1 != idx2 and best_x[idx1] != best_x[idx2]:

                old_val1 = best_x[idx1]
                best_x[idx1] = best_x[idx2]
                best_x[idx2] = old_val1

                corr = compute_binary_corr(best_x)

                # Check if improvement
                new_max_D = np.max(D)
                if new_max_D >= max_D:
                    # Revert if not improving
                    best_x[idx1] = old_val1
                    best_x[idx2] = best_x[idx2]
                else:
                    max_D = new_max_D
            continue
            
        # Try improving swap
        idx_a = np.random.choice(A)
        idx_b = np.random.choice(B)
        
        # Try the swap (attempt to reduce max_D)
        # Temporarily create new state
        x_new = best_x.copy()
        x_new[idx_a] = 0
        x_new[idx_b] = 1
        
        # Quick check by reconstructing only affected overlaps
        temp_corr = np.zeros(2*n_points - 1)
        for i in range(n_points):
            for j in range(n_points):
                k = i - j + n_points - 1
                if k >= 0 and k < 2*n_points - 1:
                    temp_corr[k] += x_new[i] * (1 - x_new[j])
                    
        new_D = np.array([n_points//2 - temp_corr[n_points - 1 + i] for i in range(n_points)])
        new_max_D = np.max(new_D)
        
        if new_max_D < max_D:
            best_x = x_new
            max_D = new_max_D
            corr = temp_corr
            D = new_D
        else:
            # Occasionally accept worse moves for escape
            if np.random.rand() < 0.01:
                # Accept it to escape local minimum
                best_x = x_new
                max_D = new_max_D
                corr = temp_corr
                D = new_D

    return best_x, evaluate_c5(best_x)

def differential_evolution_refine(h, target_sum, time_budget):
    """Refine continuous solution using Differential Evolution"""
    start_time = time.time()
    
    # Objective function
    def objective(h_vals):
        h_proj = project_to_sum_box(h_vals, target_sum)
        return evaluate_c5(h_proj)
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(len(h))]
    
    # Run DE with time limit
    result = differential_evolution(
        objective,
        bounds,
        maxiter=int(time_budget * 500),
        popsize=20,
        seed=42,
        mutation=(0.5, 1),
        recombination=0.7,
        atol=1e-6,
        rtol=1e-6
    )
    
    if result.success:
        h_refined = project_to_sum_box(result.x, target_sum)
        return h_refined, evaluate_c5(h_refined)
    else:
        return h, evaluate_c5(h)
