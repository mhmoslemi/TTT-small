import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_autocorr_fft(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def min_c5_from_autocorr(corr, n):
    """Convert autocorrelation result to minimum C5 value"""
    return dx * np.min(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def evaluate_c5_by_autocorr(h):
    """Fast evaluation of C5 using autocorrelation"""
    autocorr = compute_autocorr_fft(h)
    min_autocorr = np.min(autocorr)
    return 1.0 - (dx * min_autocorr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_piecewise_constant_h(a):
    """Generate piecewise constant h with symmetric shape based on parameter a"""
    # Ensure a is within valid range
    a = max(0.0, min(0.5, a))
    
    # Compute p based on area constraint: 2*a*1 + (2-2*a)*p = 1
    p = (1 - 2*a) / (2 * (1 - a)) if a < 1.0 else 0.5
    
    # Generate h vector
    h = np.zeros(n_points)
    
    # Compute indices for edge regions (length a)
    edge_length = int(a * n_points / 2)
    if edge_length > 0:
        # Left edge
        h[:edge_length] = 1.0
        # Right edge  
        h[-edge_length:] = 1.0
        # Middle region
        if 2*edge_length < n_points:
            h[edge_length:n_points-edge_length] = p
    else:
        # If a is very small, make middle region be the whole thing
        h[:] = p
        
    return h

def find_best_piecewise_h():
    """Find best piecewise constant h by scanning parameter a"""
    best_c5 = float('inf')
    best_h = None
    best_a = 0.0
    
    # Scan a in reasonable range
    a_values = np.linspace(0.05, 0.45, 500)
    
    for a in a_values:
        h = generate_piecewise_constant_h(a)
        # Ensure sum is exactly correct
        h = project_to_sum_box(h, n_points / 2)
        c5 = evaluate_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_a = a
    
    return best_h, best_c5, best_a

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    
    # First try finding best piecewise constant h by scanning parameter
    try:
        piecewise_h, piecewise_c5, best_a = find_best_piecewise_h()
        if piecewise_c5 < best_c5:
            best_h = piecewise_h
            best_c5 = piecewise_c5
            print(f"Piecewise constant h found with C5 = {piecewise_c5:.6f}")
    except Exception as e:
        print(f"Error in piecewise search: {e}")
    
    # Then run simulated annealing on various n_points
    n_points_choices = [200]
    
    for n in n_points_choices:
        if time.time() - start_time > budget_s * 0.8:
            break
            
        n_points = n
        dx = 2.0 / n_points
        TARGET_SUM = n_points / 2
        
        # Try with initial piecewise solution as starting point
        if best_h is not None:
            h = best_h.copy()
        else:
            # Generate initial h using piecewise choice
            h = generate_piecewise_constant_h(0.2)
            h = project_to_sum_box(h, TARGET_SUM)
        
        # Simulated Annealing
        current_h = h.copy()
        current_c5 = evaluate_c5(current_h)
        best_h_sa = current_h.copy()
        best_c5_sa = current_c5
        
        # Temperature schedule
        T_start = 0.0005
        max_iter = 100000
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Random move: pick two indices, move small amount
            i1 = np.random.randint(0, n)
            i2 = np.random.randint(0, n)
            
            # Small move
            delta = np.random.normal(0, 0.01) 
            
            # Ensure we stay within [0,1] and maintain sum constraint
            if delta > 0 and current_h[i1] >= delta:
                new_h = current_h.copy()
                new_h[i1] -= delta
                new_h[i2] += delta
                
                # Project back to valid constraints
                new_h = project_to_sum_box(new_h, TARGET_SUM)
                
                new_c5 = evaluate_c5(new_h)
                
                # Accept with probability
                if new_c5 < current_c5:
                    current_h = new_h.copy()
                    current_c5 = new_c5
                    if current_c5 < best_c5_sa:
                        best_h_sa = current_h.copy()
                        best_c5_sa = current_c5
                else:
                    # Accept with probability based on temperature
                    if np.random.rand() < np.exp(-(new_c5 - current_c5) / T_start):
                        current_h = new_h.copy()
                        current_c5 = new_c5
                        
        if best_c5_sa < best_c5:
            best_h = best_h_sa
            best_c5 = best_c5_sa
    
    # Final refinement with local search
    if time.time() - start_time < budget_s * 0.98:
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                best_h, 
                method='L-BFGS-B', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_opt = res.x
            else:
                h_opt = best_h
            h_opt = project_to_sum_box(h_opt, TARGET_SUM)
            c5_opt = evaluate_c5(h_opt)
            
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                
        except Exception:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
