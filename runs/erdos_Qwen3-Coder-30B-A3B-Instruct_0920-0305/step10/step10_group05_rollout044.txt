import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

# Global constants
n_points = 200
dx = 2.0 / n_points
half_size = n_points // 2
target_sum = n_points / 4

def autocorr_fft(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def compute_c5_exact(h):
    """Compute exact C5 value using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return 1 - dx * np.min(corr)

def project_to_sum_box(h, target_sum):
    """Project h to box [0,1]^n with sum = target_sum"""
    # Simple projection using clipping and scaling
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    elif current_sum > target_sum:
        # Need to decrease some values
        diff = current_sum - target_sum
        # Sort indices in descending order of value
        indices = np.argsort(h)[::-1]
        cumsum = 0.0
        for i in indices:
            if cumsum + (h[i] - (1 - 1e-10)) <= diff:
                cumsum += h[i] - (1 - 1e-10)
                h[i] = 1 - 1e-10
            else:
                decrease = min(diff - cumsum, h[i])
                h[i] -= decrease
                cumsum += decrease
                break
    else:
        # Need to increase some values
        diff = target_sum - current_sum
        indices = np.argsort(h)
        cumsum = 0.0
        for i in indices:
            if cumsum + (1 - 1e-10 - h[i]) <= diff:
                cumsum += 1 - 1e-10 - h[i]
                h[i] = 1 - 1e-10
            else:
                increase = min(diff - cumsum, 1 - 1e-10 - h[i])
                h[i] += increase
                cumsum += increase
                break
    return h

def parametric_h_half(a):
    """Generate a half-vector from parameter a"""
    # Edge size a (per side)
    if a >= 0.5:
        a = 0.4999
    # Middle size = 2 - 2*a (symmetric case)
    b = 2 - 2 * a
    # Compute p from area constraint
    p = (1 - 2 * a) / (2 * (1 - a))
    if p > 1:
        p = 1
    elif p < 0:
        p = 0
    # Create points
    h_half = np.zeros(half_size)
    # Total indices
    indices_left = int(a / dx)
    indices_right = int(a / dx)
    indices_middle = half_size - 2 * indices_left
    if indices_middle < 0:
        indices_middle = 0
    left = indices_left
    right = half_size - indices_right
    middle_start = left
    middle_end = right
    # Fill with constants
    if left > 0:
        h_half[:left] = 1.0
    if middle_end > middle_start:
        h_half[middle_start:middle_end] = p
    if right < half_size:
        h_half[right:] = 1.0
    # Normalize
    h_half = project_to_sum_box(h_half, target_sum)
    return h_half

def smooth_objective(h, beta=50.0):
    """Smooth approximation to -C5 using log-sum-exp"""
    corr = autocorr_fft(h)
    # Ensure we don't have overflow
    max_corr = np.max(corr)
    corr_shifted = corr - max_corr
    # Weighted sum with exponential
    exp_terms = np.exp(beta * corr_shifted)
    log_sum = np.log(np.sum(exp_terms)) + max_corr
    c5_smooth = 1.0 - dx * (log_sum / beta)
    return c5_smooth

def smooth_gradient(h, beta=50.0):
    """Gradient of the smooth objective"""
    corr = autocorr_fft(h)
    max_corr = np.max(corr)
    corr_shifted = corr - max_corr
    exp_terms = np.exp(beta * corr_shifted)
    Z = np.sum(exp_terms)
    # Filter h to get inverse autocorrelation
    # Manually compute the gradient as a convolution
    # Soft gradient via convolution.
    # But first compute these terms correctly.
    w = exp_terms / Z
    # The gradient for each term w_k * h_{i+k} is w_k
    # So for i we have sum_{k} w_k * h_{i+k} + sum_{k} w_k * h_{i-k}
    # which makes it symmetric: 
    # G[i] = sum_{k} w_k ( h[i+k] + h[i-k] ) 
    # For padding being empty? traditional convolution uses zero-pad
    res = np.zeros_like(h)
    # This is trickier in Python; let's compute manually with quick manual loop
    # We convolve w with h_reverse (or h along 0) and then add with h and sum over kernel
    # This avoids going through FFT.
    # Slightly indirect but we do the FFT-core idea manually
    L = len(h)
    conv_result = np.zeros_like(h)
    for i in range(L):
        s = 0
        for k in range(-L+1, L):
            if 0 <= i+k < L:
                s += w[k+L-1] * h[i+k]
        conv_result[i] = s
    return -dx * (conv_result) / Z * beta

    best_c5 = float('inf')

    # Phase 1: Sweep over parameterized shapes
    best_param = None
    best_c5_sweep = float('inf')
    param_range = np.arange(0.01, 0.49, 0.001)
    for a in param_range:
        h_half = parametric_h_half(a)
        c5 = compute_c5_exact(h_half)
        if c5 < best_c5_sweep:
            best_c5_sweep = c5
            best_param = a
        if time_check():
            break

    # Phase 2: Use the best parameters from sweep, perform local L-BFGS on a parametric form
    h_init = parametric_h_half(best_param)
    # Refine if more time allows
    # For low-dim, exact solution is easier, so let's do an easy update
    # We modify internal structure with one step of L-BFGS
    def obj(h):
        return smooth_objective(h, beta=100.0)

    def grad(h):
        return smooth_gradient(h, beta=100.0)

    bounds = [(0, 1) for _ in range(half_size)]

    try:
        res = minimize(obj, h_init, method='L-BFGS-B', jac=grad, bounds=bounds, 
                      options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12})
        if res.success:
            h_refined = res.x
        else:
            h_refined = h_init
    except Exception as e:
        h_refined = h_init

    h_refined = project_to_sum_box(h_refined, target_sum)

        best_c5 = c5_refined

            delta_e = c5_new - c5_current
            if np.random.rand() < np.exp(-delta_e / T):
                c5_current = c5_new
                h_current = h_new
                if c5_current < best_c5:
