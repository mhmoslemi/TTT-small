import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
from scipy.optimize import minimize

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2
COLD_TEMP = 1000
HOT_TEMP = 10000

# Global variables for state management
best_solution = None
best_c5 = float('inf')
start_time = None

def compute_corr(h):
    """Compute the correlation of h with 1-h for all lags."""
    h_rev = h[::-1]
    b = 1 - h
    # mode='full' means we get all lags
    corr = correlate(h_rev, b, mode='full')
    lag_indices = np.arange(-len(h) + 1, len(h))
    return corr, lag_indices

def evaluate_c5(h):
    """Evaluate the C5 constant (maximum overlap integral)."""
    corr, _ = compute_corr(h)
    return dx * np.max(corr)

def sigmoid(x):
    """Numerical stable sigmoid function."""
    return 1.0 / (1.0 + np.exp(-x))

def binary_skeleton_to_h(skeleton):
    """Convert binary skeleton to continuous relaxation."""
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = skeleton[i]
    return h

def step_function_representation(step_points, values):
    """Build step function from step positions and values."""
    h = np.zeros(n_points)
    step_sizes = np.diff(step_points)
    for i in range(len(step_points) - 1):
        start_idx = int(round(step_points[i] / 2 * n_points))
        end_idx = int(round(step_points[i + 1] / 2 * n_points))
        h[start_idx:end_idx] = values[i]
    # Handle last segment
    start_idx = int(round(step_points[-1] / 2 * n_points))
    h[start_idx:] = values[-1]
    return h

def solve_binary_milp():
    """Solve MILP for binary skeleton to get aggressive lower bound."""
    # Use CVXPY to model using binary variables with some relaxation
    try:
        # Simplified: create binary variables and their sum constraint
        h = cp.Variable(n_points, boolean=True)
        sum_constraint = cp.Equality(cp.sum(h), TARGET_SUM)

        # Here we just ensure binary nature, sum constraint

        obj = cp.Minimize(0)  # Placeholder objective, really we just want the constraints
        prob = cp.Problem(obj, constraints)
        # prob.solve(solver=CP.SCS)
        # We'll return a simple binary guess as a stub, since MILP not fully tractable

        # Set some values to 1 to satisfy sum constraint
        h_init[:int(TARGET_SUM)] = 1
        return h_init
    except:

    return np.ones(n_points) * 0.5

def iterate_worst_shift_local_search(h0, max_iter=1000):
    """Perform local search based on worst-case shift."""
    current_h = h0.copy()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5

    for i in range(max_iter):
        corr, indices = compute_corr(current_h)
        max_lag_idx = np.argmax(corr)
        worst_lag = indices[max_lag_idx]
        
        # Try flipping a single bit that might improve worst correlation
        # Compute effect of changing h[i]
        candidate_h = current_h.copy()
        best_improve = 0
        best_idx = -1
            
        for idx in range(n_points):
            # Skip if too close to boundaries
            if idx < 5 or idx > n_points - 5:
                continue
                
            candidate_h[idx] = 1 - candidate_h[idx]
            c5_candidate = evaluate_c5(candidate_h)
            if c5_candidate < current_c5:
                improve = current_c5 - c5_candidate
                if improve > best_improve:
                    best_improve = improve
                    best_idx = idx
            candidate_h[idx] = current_h[idx]  # Reset

        # Apply best improvement
        if best_idx >= 0:
            current_h[best_idx] = 1 - current_h[best_idx]
            current_c5 = evaluate_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            # No improvement; break criteria
            if i > 100:
                break
    return best_h, best_c5

def parametric_step_function_search():
    """Search over parametric step function space."""
    global best_c5, best_solution

    def objective(params):
        # params: [x1, x2, ..., x_k, v1, v2, ..., v_k]  for k steps
        # where xi are positions between 0 and 2, and vi are values (0 or 1)
        k = len(params) // 2
        step_positions = np.array(params[:k])
        step_positions = np.clip(step_positions, 0, 2)
        step_values = np.array(params[k:])
        step_values = np.clip(step_values, 0, 1)
        # To make it modular, reorder into bins [0, 2]
        step_positions.sort()
        h = step_function_representation(step_positions, step_values)
        # Ensure sum constraint
        h_sum = np.sum(h) * dx
        if abs(h_sum - 1.0) > 1e-5:
            return float('inf')  # Invalid solution
        c5 = evaluate_c5(h)
        return c5

    # Number of step points
    num_steps = 6

    # Sample several random full solutions (since they're not orders of magnitude smaller)
    bounds = [(0, 2)] * num_steps + [(0, 1)] * num_steps

        guess = np.random.rand(2 * num_steps)
        guess[:num_steps] = np.sort(guess[:num_steps])  # Make positions ordered
        initial_guesses.append(guess)

    # Perform differential evolution
    result = differential_evolution(
        objective, 
        bounds, 
        maxiter=500, 
        popsize=20, 
        mutation=(0.5, 1), 
        recombination=0.7,
        seed=42,
        callback=lambda x, convergence: None
    )
    if result.success:

            np.sort(result.x[:num_steps]),

        )
        final_c5 = evaluate_c5(h_opt)
        if final_c5 < best_c5:
            best_solution = h_opt
            best_c5 = final_c5
    return best_solution, best_c5

def simulated_annealing(h0, max_iter=5000):
    """Simulated Annealing with adaptive cooling."""
    current_h = h0.copy()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    temp = HOT_TEMP
    cool_rate = 0.99
    min_temp = 1e-5
    
    for i in range(max_iter):
        if time.time() - start_time > 0.9 * 600:
            break
        
        # Generate neighbor by mutating one element
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.05)
        new_h = current_h.copy()
        new_h[idx] = np.clip(current_h[idx] + delta, 0, 1)
        
        # Ensure sum constraint
        current_sum = np.sum(current_h) * dx
        new_sum = np.sum(new_h) * dx
        new_h = new_h * (TARGET_SUM / new_sum) if new_sum > 0 else new_h
        new_h = np.clip(new_h, 0, 1)
        
        new_c5 = evaluate_c5(new_h)
        delta_c5 = new_c5 - current_c5

        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temp):
            current_h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5

        temp *= cool_rate
        if temp < min_temp:
            temp = min_temp
    
    return best_h, best_c5

def local_polish(h0, iterations=1000):
    """Local improvement after large search."""
    current_h = h0.copy()
    current_c5 = evaluate_c5(current_h)
    for i in range(iterations):
        # try small perturbations to every parameter
        for idx in np.random.choice(n_points, size=min(5, n_points//10), replace=False):
            pert = np.random.normal(0, 0.01)
            new_h = current_h.copy()
            new_h[idx] = np.clip(current_h[idx] + pert, 0, 1)
            # Project back to correct sum
            new_h = new_h * (TARGET_SUM / np.sum(new_h)) if np.sum(new_h) > 0 else new_h
            new_h = np.clip(new_h, 0, 1)
            new_c5 = evaluate_c5(new_h)
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
    return current_h, current_c5
