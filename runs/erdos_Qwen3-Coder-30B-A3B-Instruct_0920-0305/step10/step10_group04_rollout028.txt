import numpy as np
from scipy.signal import correlate
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_overlap_from_corr(corr, n):
    """Compute overlap from correlation using NumPy fast path"""
    dx = 2.0 / n
    return dx * np.max(corr)

def compute_correlation_fast(binary_vec):
    """Fast computation of auto-correlation starting from binary vector"""
    # Take correlation with reversed vector to compute correlations for all shifts
    # Full mode correlation
    corr = np.correlate(binary_vec, binary_vec, mode='full')  # Returns 2*n-1 elements
    return corr

def compute_window_ones(binary_vec, max_shift):
    """Compute sum of ones in all overlapping windows"""
    n = len(binary_vec)
    prefix_sum = np.append([0], np.cumsum(binary_vec))
    total_ones = prefix_sum[n]
    window_ones = []
    for d in range(-max_shift, max_shift+1):
        if d >= 0:
            ones = prefix_sum[n-d] - prefix_sum[0]
        else:
            ones = prefix_sum[n] - prefix_sum[-d]
        window_ones.append(ones)
    return np.array(window_ones)

def update_correlation_incremental(old_corr, old_binary_vec, i, j):
    """Update correlation after swapping binary bits at positions i and j"""
    n = len(old_binary_vec)
    delta = np.zeros(2*n - 1)
    for d in range(-n+1, n):
        if i+d >= 0 and i+d < n:
            delta[d + n - 1] += -old_binary_vec[i+d]
        if j+d >= 0 and j+d < n:
            delta[d + n - 1] += old_binary_vec[j+d]
        if i-d >= 0 and i-d < n:
            delta[d + n - 1] += -old_binary_vec[i-d]
        if j-d >= 0 and j-d < n:
            delta[d + n - 1] += old_binary_vec[j-d]
    new_corr = old_corr + delta
    return new_corr
