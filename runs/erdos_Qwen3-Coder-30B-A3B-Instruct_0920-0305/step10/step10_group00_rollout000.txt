import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT with real FFT for efficiency"""
    n = len(h)
    L = 2 * n - 1
    H_rev = np.fft.rfft(h[::-1], L)
    B = np.fft.rfft(1 - h, L)
    corr = np.fft.irfft(H_rev * B, L)
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def upscale_h(original_h, n_new):
    """Upscale initial_h_values to required resolution"""
    n_old = len(original_h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, original_h)

def propose_move(h, max_step, rng):
    """Propose a move that maintains sum constraint and box constraints"""
    n = len(h)
    # try up to few times to find feasible pair
    for _ in range(10):
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # compute feasible interval for delta added to i and subtracted from j
        lower = max(-h[i], h[j] - 1)
        upper = min(1 - h[i], h[j])
        if lower < upper:
            # limit delta magnitude
            lower = max(lower, -max_step)
            upper = min(upper, max_step)
            if lower >= upper:
                continue
            delta = rng.uniform(lower, upper)
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            return new_h
        # try opposite direction: add to j, subtract from i
        lower = max(-h[j], h[i] - 1)
        upper = min(1 - h[j], h[i])
        if lower < upper:
            lower = max(lower, -max_step)
            upper = min(upper, max_step)
            if lower >= upper:
                continue
            delta = rng.uniform(lower, upper)
            new_h = h.copy()
            new_h[j] += delta
            new_h[i] -= delta
            return new_h
    return None

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Generate initial candidates
    candidates = []
    
    # From provided initial values
    try:
        global initial_h_values
        upsampled_h = upscale_h(initial_h_values, n_points)
        upsampled_h = project_to_sum_box(upsampled_h, TARGET_SUM)
        candidates.append(upsampled_h)
    except:
        pass
    
    # Cosine shape
    xs = np.arange(n_points) * dx
    a = 0.5 - 1e-5
    cos_h = 0.5 + a * np.cos(np.pi * xs / 2)
    cos_h = project_to_sum_box(cos_h, TARGET_SUM)
    candidates.append(cos_h)
    
    # Random uniform
    rand_h = np.random.uniform(0, 1, n_points)
    rand_h = project_to_sum_box(rand_h, TARGET_SUM)
    candidates.append(rand_h)
    
    # Binary with exact half ones
    binary_h = np.zeros(n_points)
    ones_idx = np.random.choice(n_points, size=n_points // 2, replace=False)
    binary_h[ones_idx] = 1.0
    candidates.append(binary_h)
    
    # Find best initial candidate
    best_h = None
    best_c5 = float('inf')
    
    for candidate in candidates:
        c5 = evaluate_c5(candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate.copy()
    
    # Simulated Annealing
    curr_h = best_h.copy()
    curr_c5 = best_c5
    best_h_local = curr_h.copy()
    best_c5_local = curr_c5
    
    rng = np.random.default_rng(seed)
    
    total_time_limit = budget_s * 0.9
    start_sa = time.time()
    
    while time.time() - start_time < total_time_limit:
        elapsed = time.time() - start_sa
        fraction = elapsed / total_time_limit
        T = max(0.0001, 0.02 * (1 - fraction))
        max_step = max(0.001, 0.2 * (1 - fraction))
        
        new_h = propose_move(curr_h, max_step, rng)
        if new_h is None:
            continue
            
        new_c5 = evaluate_c5(new_h)
        delta = new_c5 - curr_c5
        
        if delta <= 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            curr_h = new_h
            curr_c5 = new_c5
            
            if new_c5 < best_c5_local:
                best_c5_local = new_c5
                best_h_local = curr_h.copy()
    
    # Local refinement
    refine_start = time.time()
    refine_time_limit = budget_s * 0.1
    
    curr_h = best_h_local.copy()
    curr_c5 = best_c5_local
    
    max_step_refine = 0.005
    
    for i in range(50000):
        if time.time() - refine_start > refine_time_limit:
            break
            
        elapsed_refine = time.time() - refine_start
        fraction = elapsed_refine / refine_time_limit
        max_step = max_step_refine * (1 - fraction) + 1e-6
        
        new_h = propose_move(curr_h, max_step, rng)
        if new_h is None:
            continue
            
        new_c5 = evaluate_c5(new_h)
        
        if new_c5 < curr_c5:
            curr_h = new_h
            curr_c5 = new_c5
            
            if new_c5 < best_c5_local:
                best_c5_local = new_c5
                best_h_local = curr_h.copy()
    
    return (best_h_local.tolist(), float(best_c5_local), n_points)
