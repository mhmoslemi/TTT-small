def evaluate_c5(h):
    """Compute C5 using FFT correlation"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water filling"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_milp_optimization():
    """Solve a binary MILP to get a good skeleton"""
    # Only solve small problem
    N = MILP_N
    dx_small = 2.0 / N
    x = Variable(N, boolean=True)
    t = Variable()
    
    # Constraints
    constrs = [cp_sum(x) == N / 2, t >= 0]
    
    #y_vars = {}
    y_vars = {}
    # For each shift
    for k in range(1, N):
        # Create auxiliary variables
        for i in range(N - k):
            var_key = (i, k)
            y_vars[var_key] = Variable(boolean=True)
            
    # Add constraints
    for k in range(1, N):
        for i in range(N - k):
            var_key = (i, k)
            y_vars[var_key] = Variable(boolean=True)
            constrs.extend([
                y_vars[var_key] <= x[i],
                y_vars[var_key] <= x[i + k],
                y_vars[var_key] >= x[i] + x[i + k] - 1,
            ])
        # Add constraint: sum of products is at least (N/2 - t)
        constrs.append(cp_sum([y_vars[(i,k)] for i in range(N - k)]) >= N / 2 - t)
        
    # Objective
    prob = Problem(Minimize(t), constrs)
    prob.solve(solver=cp.GLPK_MI, time_limit=MILP_TIMELIMIT, verbose=False)
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        solution = np.round(np.array([var.value for var in x])).astype(int)
        return solution
    else:
        return None

def upsample_binary_vector(bin_vec, factor):
    """Upsample binary vector to final resolution"""
    return np.repeat(bin_vec, factor)

def worst_shift_swap_search(h_vec, n_iter=10000):
    """Perform swaps to improve worst-case shift"""
    h_new = h_vec.copy()
    # Compute correlation
    corr = np.correlate(h_new, h_new, mode="full")
    corr = np.roll(corr, len(corr)//2)
    corr = corr[len(corr)//2:]  # Only positive shifts
    
    worst_shift = np.argmax(corr)
    it = 0
    while it < n_iter:
        # Find nodes that improve worst shift
        A = [i for i in range(len(h_new)-worst_shift) if h_new[i] == 1 and h_new[i+worst_shift] == 0]
        B = [j for j in range(len(h_new)-worst_shift) if h_new[j] == 0 and h_new[j+worst_shift] == 1]
        if A and B:
            a = np.random.choice(A)
            b = np.random.choice(B)
            h_new[a], h_new[b] = 0, 1
        else:
            # Random swap
            i1, i2 = np.random.randint(0, len(h_new), 2)
            h_new[i1], h_new[i2] = h_new[i2], h_new[i1]
        it += 1
        
        # Recompute correlation
        corr = np.correlate(h_new, h_new, mode="full")
        corr = np.roll(corr, len(corr)//2)
        corr = corr[len(corr)//2:]
        worst_shift = np.argmax(corr)
    
    return h_new

    T = 0.1
    alpha = 0.99
    max_iter = 100000
    
    for i in range(max_iter):
        if time.time() - start_time > TIME_BUDGET - 5:
            break
            
        # Generate neighbor
        h_new = h_current.copy()
        ind1, ind2 = np.random.randint(0, len(h0), 2)
        delta = np.random.uniform(-0.05, 0.05)
        
        h_new[ind1] = max(0, min(1, h_new[ind1] + delta))
        h_new[ind2] = max(0, min(1, h_new[ind2] - delta))
        
        # Re-project to preserve sum
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        c_new = evaluate_c5(h_new)
        delta_e = c_new - c_current
        
        if delta_e < 0 or np.random.rand() < np.exp(-delta_e / T):

            c_current = c_new
            
            if c_current < c_best:
                h_best = h_current.copy()
                c_best = c_current
                
        T *= alpha
    
    return h_best, c_best
