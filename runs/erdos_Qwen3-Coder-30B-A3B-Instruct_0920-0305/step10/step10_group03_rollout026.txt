import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from cvxpy import *
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
MILP_N = 60  # Binary problem size

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water-filling"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_binary_milp(N):
    """Solve the binary integer programming problem for small N"""
    x = Variable(N, boolean=True)
    t = Variable(pos=True)
    
    # Mass constraint
    constraints = [sum(x) == N/2]
    
    # For each shift k = 1,...,N-1, add correlation constraints
    for k in range(1, N):
        # y[i,k] = x[i] * x[i+k] for i = 0,...,N-1-k
        for i in range(N - k):
            y = Variable(boolean=True)
            # McCormick constraints
            constraints += [y <= x[i], y <= x[i+k], y >= x[i] + x[i+k] - 1]
            # Worst-case bound constraint: sum y[i,k] >= N/2 - t
            constraints += [sum([y for i in range(N-k)]) >= N/2 - t]
    
    # Minimize t
    prob = Problem(Minimize(t), constraints)
    
    try:
        prob.solve(solver=CPLEX, time_limit=30)
        return np.array([x.value[i] for i in range(N)])
    except:
        # Fallback: return a heuristic binary solution
        return np.ones(N) * 0.5

def swap_improve_binary(x, max_iter=10000):
    """Improve a binary solution by swapping poor pairs"""
    # Convert to integer for easier handling
    x = x.astype(int)
    
    # Precompute all correlations
    def compute_auto_corr(x):
        L = len(x)
        corr = np.zeros(L)
        for k in range(L):
            corr[k] = sum(x[i]*x[i+k] for i in range(L-k))
        return corr
    
    corr = compute_auto_corr(x)
    worst_k = np.argmax(np.array([len(x)/2 - c for c in corr]))
    
    for iteration in range(max_iter):
        if worst_k == 0:
            break
            
        A = [i for i in range(len(x)-worst_k) if x[i] == 1 and x[i+worst_k] == 0]
        B = [j for j in range(worst_k, len(x)) if x[j] == 0 and x[j-worst_k] == 1]
        
        if not A or not B:
            # No more improving swaps, try random
            if np.random.random() < 0.1:
                i, j = np.random.randint(len(x)), np.random.randint(len(x))
                if i + worst_k < len(x):
                    if x[i] == 1 and x[i+worst_k] == 0:
                        if x[j] == 0 and x[j-worst_k] == 1:
                            # Swap x[i] and x[j] 
                            x[i], x[j] = 0, 1
                            corr = compute_auto_corr(x)
                            worst_k = np.argmax(np.array([len(x)/2 - c for c in corr]))
            else:
                break
        else:
            # Pick a random improving pair
            a = np.random.choice(A)
            b = np.random.choice(B)
            x[a], x[b] = 0, 1  # swap
            corr = compute_auto_corr(x)
            worst_k = np.argmax(np.array([len(x)/2 - c for c in corr]))
            
    return x.astype(float)

def differential_evolution(h0, max_iter=2000):
    """Differential evolution optimization for continuous values with projection"""
    # Parameters
    P = 30  # Population size
    F = 0.8 # Mutation factor
    CR = 0.9 # Crossover probability
    
    def project_h(individ):
        return project_to_sum_box(individ, TARGET_SUM)
    
    # Initialize population
    population = [h0.copy()]
    for i in range(1, P):
        pert = np.random.normal(0, 0.02, len(h0))
        new_indiv = project_h(h0 + pert)
        population.append(new_indiv)
    
    # Evaluate initial population
    fitnesses = [evaluate_c5(indiv) for indiv in population]
    
    # Evolution loop

        for i in range(P):
            # Choose three distinct individuals
            candidates = [j for j in range(P) if j != i]
            a, b, c = np.random.choice(candidates, 3, replace=False)
            
            # Mutation and crossover
            mutant = population[a] + F * (population[b] - population[c])
            trial = np.where(np.random.random(len(h0)) < CR, mutant, population[i])
            
            # Projection
            trial = project_h(trial)
            
            # Selection
            fit_trial = evaluate_c5(trial)
            if fit_trial <= fitnesses[i]:
                population[i] = trial
                fitnesses[i] = fit_trial
    
    # Return best individual
    best_idx = np.argmin(fitnesses)
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx, start_time
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Phase A: Binary MILP
    print("Starting binary MILP...")
    x_binary = solve_binary_milp(MILP_N)
    
    # Phase B: Upsample and improve binary
    print("Upsampling...")
    h_upsampled = np.repeat(x_binary, n_points // MILP_N)
    h_upsampled = h_upsampled.astype(float)
    
    # Phase C: Swap improve if needed
    print("Swapping...")
    h_upsampled = swap_improve_binary(h_upsampled, 10000)
    
    # Phase D: Differential evolution
    print("Differential evolution...")
    h_final = differential_evolution(h_upsampled, 2000)
    
    # Phase E: Final polishing
    print("Final polishing...")
    current_c5 = evaluate_c5(h_final)
    
    # Simple hill climbing
    for _ in range(2000):
        if time.time() - start_time > budget_s - 10:
            break
            
        h_new = h_final.copy()
        indices = np.random.choice(len(h_final), size=int(0.05 * len(h_final)), replace=False)
        for i in indices:
            # Small perturbation
            delta = np.random.normal(0, 0.02)
            h_new[i] += delta
            
        # Project
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < current_c5:
            h_final = h_new
            current_c5 = new_c5
            
    # Final check
    final_c5 = evaluate_c5(h_final)
    
    print(f"Final C5: {final_c5}")
    return (h_final.tolist(), float(final_c5), n_points)
