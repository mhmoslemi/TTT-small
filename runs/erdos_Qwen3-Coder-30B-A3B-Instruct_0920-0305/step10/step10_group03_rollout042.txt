def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_milp(N, time_limit=30):
    """Solve binary integer program for optimal step function"""
    try:
        # Create variables
        x = cp.Variable(N, boolean=True)
        t = cp.Variable()
        
        # Constraints
        constraints = [cp.sum(x) == N // 2]
        
        # For each shift k from 1 to N-1
        for k in range(1, N):
            # Create y variables for products
            y = cp.Variable(N - k, boolean=True)
            # Add constraints for y = x[i] * x[i+k]
            for i in range(N - k):
                constraints.append(y[i] <= x[i])
                constraints.append(y[i] <= x[i + k])
                constraints.append(y[i] >= x[i] + x[i + k] - 1)
            # Add constraint that sum of y >= N/2 - t
            constraints.append(cp.sum(y) >= N // 2 - t)
        
        # Objective: minimize t
        objective = cp.Minimize(t)
        
        # Create problem and solve
        prob = cp.Problem(objective, constraints)
        
        # Use GLPK_MI solver which is free
        prob.solve(solver=cp.GLPK_MI, time_limit=time_limit)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value
        else:
            return None
            
    except:
        return None

def upsample_binary(binary_vec, target_N):
    """Upsample binary vector to target N points"""
    upsample_factor = target_N // len(binary_vec)
    upsampled = np.repeat(binary_vec, upsample_factor)
    # Adjust length if needed
    if len(upsampled) > target_N:
        upsampled = upsampled[:target_N]
    elif len(upsampled) < target_N:
        upsampled = np.pad(upsampled, (0, target_N - len(upsampled)), 'constant')
    
    return upsampled

def local_improvement_binary(x, max_iter=10000):
