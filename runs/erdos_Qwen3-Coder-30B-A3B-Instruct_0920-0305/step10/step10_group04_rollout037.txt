import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constant
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points

def compute_auto_correlation(x):
    """Compute auto-correlation of binary vector x using FFT."""
    x_padded = np.pad(x, (0, 0), 'constant')
    auto_corr = correlate(x_padded, x_padded, mode='full')
    return auto_corr

def compute_overlaps(auto_corr, x):
    """Compute overlap values O_d for all shifts d."""
    n = len(x)
    overlaps = np.zeros(2*n - 1)
    cumsum_x = np.concatenate([[0], np.cumsum(x)])

    # For each shift offset d from -(n-1) to n-1
    for d in range(-(n-1), n):
        if d >= 0:
            # O_d = sum_{i=0}^{n-1-d} x_i * (1 - x_{i+d})
            #     = sum_{i=0}^{n-1-d} x_i - sum_{i=0}^{n-1-d} x_i * x_{i+d}
            #     = cumsum_x[n-1-d] - auto_corr[n-1+d]
            overlaps[d + n - 1] = cumsum_x[n - d] - auto_corr[n - 1 + d]
        else:
            # d < 0: it's symmetric via index shift
            overlaps[d + n - 1] = cumsum_x[n - d] - auto_corr[n - 1 + d]
    return overlaps

def evaluate_c5_binary(x):
    """Fast evaluation of C5 for binary vector x."""
    auto_corr = compute_auto_correlation(x)
    overlaps = compute_overlaps(auto_corr, x)
    return dx * np.max(overlaps)

def flip_bits(x, i, j):
    """Swap x[i] and x[j] in the binary vector x."""
    x[i], x[j] = x[j], x[i]

def evaluate_c5_continuous(h):
    """Evaluate C5 for continuous vector h using FFT-based correlation."""
    corr = correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using binary search."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_delta_correlation(x, i, j):
    """Compute the change in correlation vector when flipping bits i and j."""
    n = len(x)
    x_pad = np.pad(x, (n-1, n-1), 'constant')
    delta_A = np.zeros(2*n - 1)
    # For i being turned off:
    if 0 <= i < n:
        idx_left = i
        idx_right = i + 2*(n-1)
        # Add to delta_A the contribution of x[i] on all shifts it affects
        delta_A[idx_left : idx_right + 1] -= x_pad[idx_left + n -1 : idx_right + n]
        delta_A[idx_left : idx_right + 1] -= x_pad[idx_left : idx_right + 1][::-1]
    # For j being turned on:
    if 0 <= j < n:
        idx_left = j
        idx_right = j + 2*(n-1)
        # Add to delta_A the contribution of x[j] on all shifts it affects
        delta_A[idx_left : idx_right + 1] += x_pad[idx_left + n -1 : idx_right + n]
        delta_A[idx_left : idx_right + 1] += x_pad[idx_left : idx_right + 1][::-1]
    return delta_A

def find_worst_shift(x, auto_corr):
    """Find the shift that gives the maximum overlap."""
    overlaps = compute_overlaps(auto_corr, x)
    i = np.argmax(overlaps)
    return i - len(overlaps) // 2

def get_candidates_for_shift(x, d):
    """Get sets U and V for the given shift d."""
    n = len(x)
    U = []  # positions i where x[i]=1, x[i+d]=0 and 0≤i+d<n
    V = []  # positions j where x[j]=0, x[j+d]=1 and 0≤j+d<n

    d_pos = -d
    if d > 0:
        # x[i] * (1 - x[i+d]) for d>0
        for i in range(n - d):
            if x[i] > 0 and x[i+d] == 0:
                U.append(i)
        for j in range(n - d):
            if x[j] == 0 and x[j+d] > 0:
                V.append(j)
    else:
        # d <= 0
        for i in range(n + d):
            if x[i] > 0 and x[i+d] == 0:
                U.append(i)
        for j in range(n + d):
            if x[j] == 0 and x[j+d] > 0:
                V.append(j)
    return U, V

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Set target sum
    TARGET_SUM = n_points / 2
    
    # Initial binary vector - start with a random pattern that's half ones
    x = np.zeros(n_points, dtype=int)
    ones_indices = np.random.choice(n_points, size=n_points // 2, replace=False)
    x[ones_indices] = 1
    
    # Fast evaluation using auto-correlation
    auto_corr = compute_auto_correlation(x) 
    current_c5 = evaluate_c5_binary(x)
    
    # Track best solution
    best_x = x.copy()
    best_c5 = current_c5

    # Simulated Annealing parameters
    T0 = 2 * dx
    Tf = 1e-6 * dx
    total_moves = 300000
    T = T0
    accept_count = 0  # for cooling schedule
    
    # Parameters for convergence check
    stagnation_window = 20000
    last_improve = 0
    last_best_c5 = current_c5
    
    # Main Simulated Annealing loop
    for move in range(total_moves):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Decide move type
        if np.random.random() < 0.6:
            # Targeted move: try to improve worst shift
            d_star = find_worst_shift(x, auto_corr)
            U, V = get_candidates_for_shift(x, d_star)
            if U and V:
                i = np.random.choice(U)
                j = np.random.choice(V)
                # Perform the swap
                flip_bits(x, i, j)
                delta_A = compute_delta_correlation(x, i, j)
                new_auto_corr = auto_corr + delta_A
                overlaps = compute_overlaps(new_auto_corr, x)
                new_c5 = dx * np.max(overlaps)
                
                # Accept based on SA
                if new_c5 < current_c5 or np.random.random() < np.exp(-(new_c5 - current_c5) / T):
                    auto_corr = new_auto_corr
                    current_c5 = new_c5
                else:
                    # Revert swap
                    flip_bits(x, i, j)
            else:
                # Fall back to random swap
                i, j = np.random.choice(n_points, 2, replace=False)
                flip_bits(x, i, j)
                delta_A = compute_delta_correlation(x, i, j)
                new_auto_corr = auto_corr + delta_A
                overlaps = compute_overlaps(new_auto_corr, x)
                new_c5 = dx * np.max(overlaps)
                
                if new_c5 < current_c5 or np.random.random() < np.exp(-(new_c5 - current_c5) / T):
                    auto_corr = new_auto_corr
                    current_c5 = new_c5
                else:
                    flip_bits(x, i, j)
        else:
            # Random swap
            i, j = np.random.choice(n_points, 2, replace=False)
            flip_bits(x, i, j)
            delta_A = compute_delta_correlation(x, i, j)
            new_auto_corr = auto_corr + delta_A
            overlaps = compute_overlaps(new_auto_corr, x)
            new_c5 = dx * np.max(overlaps)
            
            if new_c5 < current_c5 or np.random.random() < np.exp(-(new_c5 - current_c5) / T):
                auto_corr = new_auto_corr
                current_c5 = new_c5
            else:
                flip_bits(x, i, j)
    
        # Update temperature
        accept_count += 1
        if accept_count > 0:
            T = T0 * (1.0 - (accept_count / total_moves))
            T = max(T, Tf)
        
        # Check for stagnation
        if accept_count % stagnation_window == 0:
            if abs(current_c5 - last_best_c5) < 1e-5:
                # Partially randomize the solution to escape flat spots
                to_flip = 0.05 * n_points
                flips = np.random.choice(n_points, size=int(to_flip), replace=False)
                for idx in flips:
                    x[idx] = 1 - x[idx]
                # Rebalance sum to maintain n/2 ones using a simple swapping method
                ones_count = np.sum(x)
                missing = int(0.5 * n_points) - ones_count
                if missing > 0:
                    zero_indices = np.where(x == 0)[0]
                    index_shuffle = np.random.permutation(zero_indices)
                    for i in range(missing):
                        x[index_shuffle[i]] = 1
                elif missing < 0:
                    one_indices = np.where(x == 1)[0]
                    index_shuffle = np.random.permutation(one_indices)
                    for i in range(-missing):
                        x[index_shuffle[i]] = 0
                # Reindex correlations after large change
                auto_corr = compute_auto_correlation(x)
                current_c5 = evaluate_c5_binary(x)
            last_best_c5 = current_c5  # Store the current best value
    
        # Update best if necessary
        if current_c5 < best_c5:
            best_x = x.copy()
            best_c5 = current_c5
            
    # Continuous refinement phase (final 20% of time)
    h_values = best_x.astype(float)
    h_values = project_to_sum_box(h_values, TARGET_SUM)  # Final projection ensuring sum = n/2
    
    # First SLSQP
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
        res = minimize(
            evaluate_c5_continuous,
            h_values,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False}
        )
        if res.success:
            h_values = project_to_sum_box(res.x, TARGET_SUM)
        # Second round of SLSQP as a safety measure
        try:
            res2 = minimize(
                evaluate_c5_continuous,
                h_values,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-12, 'disp': False}
            )
            if res2.success:
                h_values = project_to_sum_box(res2.x, TARGET_SUM)
        except:
            pass
    except:
        pass
        
    # Final validation with evaluate_erdos_solution
    final_c5 = evaluate_erdos_solution(h_values)
    
    return (h_values.tolist(), float(final_c5), n_points)
