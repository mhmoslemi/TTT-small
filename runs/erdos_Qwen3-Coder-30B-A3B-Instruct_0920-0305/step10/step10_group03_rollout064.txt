import numpy as np

# Global variables
TARGET_SUM = None

dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

            start = max(0, idx - 3)

                delta = np.random.normal(0, step_size * 0.5)
                h_new[j] += delta
        else:  # Global perturbation
            for idx in range(n_points):

                    delta = np.random.normal(0, step_size)
                    h_new[idx] += delta

        h_new = project_to_sum_box(h_new, TARGET_SUM)

        c5_new = evaluate_c5(h_new)
        
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        elif np.random.random() < 0.01:  # Occasionally accept worse solutions
            current_h = h_new.copy()
            current_c5 = c5_new
    
    return best_h, best_c5

def optimize_with_cma_es(h0, max_iter=5000):
    """CMA-ES-like optimization approach with adaptive sampling"""
    current_h = h0.copy()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Simple evolution strategy with population
    population_size = 20
    population = []
    
    # Initialize population around current solution
    for i in range(population_size):
        h_pert = current_h.copy()
        # Add Gaussian noise with decreasing variance
        sigma = max(0.01, 0.1 * (1.0 - i / population_size))
        noise = np.random.normal(0, sigma, n_points)
        h_pert += noise
        h_pert = project_to_sum_box(h_pert, TARGET_SUM)
        population.append((h_pert, evaluate_c5(h_pert)))
    
    for iter_num in range(max_iter):
        # Sort population by fitness
        population.sort(key=lambda x: x[1])
        
        # Keep top 30% and generate new individuals
        elite_size = population_size // 3
        elite = population[:elite_size]
        
        # Generate new population
        new_population = elite.copy()
        
        # Crossover and mutation
        for _ in range(population_size - elite_size):
            # Select two parents
            parent1, parent2 = np.random.choice(elite, size=2, replace=False)
            
            # Crossover (blend)
            alpha = np.random.random()
            h_child = alpha * parent1[0] + (1 - alpha) * parent2[0]
            
            # Mutation
            if np.random.random() < 0.3:
                sigma = max(0.005, 0.05 * (1.0 - iter_num / max_iter))
                noise = np.random.normal(0, sigma, n_points)
                h_child += noise
            
            h_child = project_to_sum_box(h_child, TARGET_SUM)
            c5_child = evaluate_c5(h_child)
            new_population.append((h_child, c5_child))
        
        population = new_population
        
        # Update best
        for h_test, c5_test in population:
            if c5_test < best_c5:
                best_h = h_test.copy()
                best_c5 = c5_test
    
    return best_h, best_c5
