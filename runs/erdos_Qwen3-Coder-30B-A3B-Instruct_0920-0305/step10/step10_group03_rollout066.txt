```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water-filling"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_binary_milp(N=60):
    """Solve the binary integer program for a coarse grid and return binary solution"""
    # Build the model manually
    x = cp.Variable(N, boolean=True)
    t = cp.Variable(1, pos=True)

    # Constraints  
    constraints = [cp.sum(x) == N/2]
    constraints.append(t >= 0)
    
    # Build correlations for all shifts
    for k in range(1, N):
        y = cp.Variable(N-k, boolean=True)
        # McCormick constraints for y[i] = x[i]*x[i+k]
        for i in range(N-k):
            constraints.append(y[i] <= x[i])
            constraints.append(y[i] <= x[i+k])
            constraints.append(y[i] >= x[i] + x[i+k] - 1)
        
        constraints.append(cp.sum(y) >= N/2 - t)
    
    # Solve the model
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=CPLEX, verbose=False, timelimit=30)
        if prob.status in [OPTIMAL, FEASIBLE]:
            return np.round(x.value).astype(int)
    except:
        pass
    # Fallback: use a simple greedy construction
    x_fallback = np.zeros(N)
    ones_idx = np.random.choice(N, size=int(N/2), replace=False)
    x_fallback[ones_idx] = 1
    return x_fallback

def upsample_binary(x_coarse, n_points):
    """Upsample a coarse binary vector to full resolution"""
    s = n_points // len(x_coarse)
    h_upsampled = np.repeat(x_coarse, s)
    # Pad or truncate to exact length
    if len(h_upsampled) > n_points:
        h_upsampled = h_upsampled[:n_points]
    elif len(h_upsampled) < n_points:
        h_upsampled = np.pad(h_upsampled, (0, n_points - len(h_upsampled)), 'constant')
    return h_upsampled

def worst_shift_swap_search(h0, max_iter=10000):
    """Perform swap search focused on the worst shift to reduce max correlation"""
    h = h0.copy()
    n = len(h)
    # Precompute correlation
    corr = compute_corr_fft(h)
    max_corr = np.max(corr)
    worst_shift = np.argmax(corr)
    
    swaps = 0
    for _ in range(max_iter):
        if swaps >= 10000:
            break
        # Find worst shift
        corr = compute_corr_fft(h)
        worst_shift = np.argmax(corr)
        
        # Find indices where we can swap (i, i+worst_shift)
        A = []  # i such that h[i] = 1 and h[i+worst_shift] = 0
        B = []  # j such that h[j] = 0 and h[j+worst_shift] = 1
        
        for i in range(n - worst_shift):
            # Check if this swap is valid
            if h[i] == 1 and h[i + worst_shift] == 0:
                A.append(i)
            if h[i] == 0 and h[i + worst_shift] == 1:
                B.append(i)
        
        if not A or not B:
            # Random swap to escape local minima
            i1 = np.random.randint(len(h))
            i2 = np.random.randint(len(h))
            if i1 != i2:
                h[i1], h[i2] = h[i2], h[i1] 
            swaps += 1
            continue
        
        # Pick a random valid swap
        i = np.random.choice(A)
        j = np.random.choice(B)
        h[i], h[j] = h[j], h[i]
        swaps += 1
    
    return h

def differential_evolution(h0, max_iter=2000):
    """Run differential evolution to refine the solution"""
    def objective(h):
        return evaluate_c5(h)
    
    # Start with a random population
    n_pop = 30
    population = []
    for _ in range(n_pop):
        trial = h0 + np.random.normal(0, 0.02, len(h0))
        trial = project_to_sum_box(trial, TARGET_SUM)
        population.append(trial)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    
    for iter_count in range(max_iter):
        if iter_count % 50 == 0 and iter_count > 0:
            # Early termination if not improving enough
            pass
        
        for i in range(n_pop):
            # Select three distinct donors
            idxs = np.random.choice(n_pop, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutate
            F = 0.8
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover  
            CR = 0.9
            j_rand = np.random.randint(len(h0))
            trial = []
            for j in range(len(h0)):
                if np.random.random() < CR or j == j_rand:
                    trial.append(mutant[j])
                else:
                    trial.append(population[i][j])
            trial = np.array(trial)
            
            # Project
            trial = project_to_sum_box(trial, TARGET_SUM)
            
            # Selection
            trial_c5 = evaluate_c5(trial)
            if trial_c5 < objective(population[i]):
                population[i] = trial
                
            if trial_c5 < best_c5:
                best_h = trial.copy()
                best_c5 = trial_c5

    return best_h, best_c5

def simulated_annealing(h0, max_iter=2000):
    """Run simulated annealing refinement"""
    h = h0.copy()
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    T = 0.1
    alpha = 0.99
    min_T = 1e-6
    
    for i in range(max_iter):
        if T < min_T:
            break
        
        # Pick random indices to adjust
        i1 = np.random.randint(len(h))
        i2 = np.random.randint(len(h))
        
        if i1 == i2:
            continue
            
        # Save old state
        old_h = h.copy()
        old_c5 = current_c5
        
        # Perturb
        delta = np.random.normal(0, 0.05)
        h[i1] += delta
        h[i2] -= delta
        
        # Clamp
        h = np.clip(h, 0, 1)
        
        # Project back to sum constraint
        h = project_to_sum_box(h, TARGET_SUM)
        
        new_c5 = evaluate_c5(h)
        delta_e = new_c5 - old_c5
        
        if delta_e < 0 or np.random.random() < np.exp(-delta_e / T):
            current_c5 = new_c5
        else:
            h = old_h
            current_c5 = old_c5
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        T *= alpha
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Phase A: Solve MILP on small grid
    h_binary = solve_binary_milp()
    h_upsampled = upsample_binary(h_binary, n_points)
    
    # Apply local swap search (optional)
    if time.time() - start_time < budget_s * 0.2:
        h_upsampled = worst_shift_swap_search(h_upsampled, 10000)
    
    # Phase B: Differential Evolution (main optimization phase)
    if time.time() - start_time < budget_s * 0.8:
        h_de, c5_de = differential_evolution(h_upsampled, 1000)
        h_final = h_de
        c5_final = c5_de
    else:
        h_final = h_upsampled
        c5_final = evaluate_c5(h_upsampled)
    
    # Phase C: Simulated Annealing refinement
    if time.time() - start_time < budget_s:
        h_sa, c5_sa = simulated_annealing(h_final, 500)
        if c5_sa < c5_final:
            h_final = h_sa
            c5_final = c5_sa
    
    # Phase D: Final polish with hill climbing
    if time.time() - start_time < budget_s * 0.95:
        h_hc = h_final.copy()
        best_h = h_hc.copy()
        best_c5 = c5_final
        
        for _ in range(1000):
            # Perturb a small fraction of points
            num_perturb = max(1, n_points // 20)
            indices = np.random.choice(n_points, size=num_perturb, replace=False)
            
            h_new = h_hc.copy()
            for i in indices:
                delta = np.random.normal(0, 0.02) * (h_new[i] * (1 - h_new[i]))
                h_new[i] += delta
                h_new[i] = max(0, min(1, h_new[i]))
            
            # Project back
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            c5_new = evaluate_c5(h_new)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                
        h_final = best_h
        c5_final = best_c5
    
    # Final check and return
    c5_final = evaluate_c5(h_final)
    return (h_final.tolist(), float(c5_final), n_points)
```
