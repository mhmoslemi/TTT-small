import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_autocorr_fft(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def c5_from_autocorr(autocorr, n):
    """Convert autocorrelation result to C5 value"""
    min_autocorr = np.min(autocorr)
    return 1 - min_autocorr * dx

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def evaluate_c5_by_autocorr(h):
    """Fast evaluation of C5 using autocorrelation"""
    autocorr = compute_autocorr_fft(h)
    return c5_from_autocorr(autocorr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_piecewise_constant_h(a):
    """Generate piecewise constant h with symmetric structure"""
    # a: length of edge regions [0,a] and [2-a,2] with h=1
    # middle region [a, 2-a] with height p
    p = (1 - 2*a) / (2*(1-a))  # derived from area constraint
    if p < 0 or p > 1:
        # fallback to uniform
        p = 0.5
        
    h = np.zeros(n_points)
    
    # Calculate indices
    edge_length = int(a * n_points / 2)
    if edge_length == 0:
        edge_length = 1
    middle_length = n_points - 2 * edge_length
    if middle_length <= 0:
        # fallback
        middle_length = 1
        edge_length = (n_points - 1) // 2
    
    # Fill edges with 1
    h[:edge_length] = 1
    h[-edge_length:] = 1
    
    # Fill middle with p
    if middle_length > 0:
        h[edge_length:edge_length+middle_length] = p
    
    # Ensure sum matches
    actual_sum = np.sum(h) * dx
    scaling = TARGET_SUM / actual_sum
    h = h * scaling
    
    return h

def optimized_initial_h():
    """Generate an optimized initial h using piecewise constant approximation"""
    best_c5 = float('inf')
    best_h = None
    
    # Try various values of a between 0.05 and 0.45
    for a in np.linspace(0.05, 0.45, 20):
        h = generate_piecewise_constant_h(a)
        h = project_to_sum_box(h, TARGET_SUM)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h if best_h is not None else np.full(n_points, 0.5)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Generate initial solution using piecewise constant strategy
    best_h = optimized_initial_h()
    best_c5 = evaluate_c5(best_h)
    
    # Use Simulated Annealing with carefully designed moves
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Simulated Annealing parameters
    max_iter = 200000
    T_start = 0.001
    
    # Precompute h_rev for correlation
    h_rev = current_h[::-1]
    
    # Adaptive step sizes
    delta_max = 0.1
    cooling_rate = 0.9999
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random coordinate selection with probability bias towards improving moves
        # Simple move: randomly select 2 indices and adjust values
        i1 = np.random.randint(0, n_points)
        i2 = np.random.randint(0, n_points)
        
        # Ensure not the same index
        if i1 == i2:
            continue
            
        # Try small random adjustment
        delta = np.random.uniform(-delta_max, delta_max)
        delta1 = min(delta, current_h[i1], 1 - current_h[i2])
        delta2 = -delta1
            
        # Create new state  
        h_new = current_h.copy()
        h_new[i1] = max(0, min(1, h_new[i1] + delta1))
        h_new[i2] = max(0, min(1, h_new[i2] + delta2))
        
        # Project to maintain sum constraint
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = current_h.copy()
                best_c5 = c5_new
        else:
            # Accept with probability
            T = T_start * (cooling_rate ** i)
            if T > 1e-8:
                if np.random.rand() < np.exp(-(c5_new - current_c5) / T):
                    current_h = h_new.copy()
                    current_c5 = c5_new
    
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        res = minimize(
            objective, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
        else:
            h_refined = best_h
        h_refined = project_to_sum_box(h_refined, TARGET_SUM)
        c5_refined = evaluate_c5(h_refined)
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
            
    except Exception:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
