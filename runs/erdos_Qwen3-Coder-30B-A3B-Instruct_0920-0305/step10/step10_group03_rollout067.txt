def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

            constraints = [sum(x) == n_bin / 2, t >= 0]
            
            # Correlation constraints for each shift
            for k in range(1, n_bin):
                # For all valid i where i+k < n_bin
                y = [x[i] * x[i+k] for i in range(n_bin - k)]
                constraints.append(sum(y) >= n_bin/2 - t)
            
            # Objective
            objective = Minimize(t)
            prob = Problem(objective, constraints)
            
            # Solve with a time limit
            prob.solve(solver=GLPK_MI, time_limit=30)

                x_star = x.value
                h_bin = x_star
            else:
                # If the MILP did not solve, use the initial vector
                h_bin = h0[:n_bin]
                
        except:
            h_bin = h0[:60]
    else:
        h_bin = h0[:60]
    
    # Phase B: Upsampling
    if time.time() - start_time < 45:
        upsample_factor = n_points // len(h_bin)
        h_upsampled = np.repeat(h_bin, upsample_factor)
        if len(h_upsampled) < n_points:
            h_upsampled = np.pad(h_upsampled, (0, n_points - len(h_upsampled)), 'constant')
        elif len(h_upsampled) > n_points:
            h_upsampled = h_upsampled[:n_points]
        h_start = h_upsampled
        h_start = project_to_sum_box(h_start, TARGET_SUM)
    else:
        h_start = h0.copy()
    
    best_h = h_start.copy()

    # Phase D: Differential Evolution with DE/rand/1/bin strategy
    if time.time() - start_time < 500:
        try:
            # Population-based optimization
            pop_size = 30
            population = []
            for _ in range(pop_size):
                individual = np.random.normal(h_start, 0.02)
                individual = project_to_sum_box(individual, TARGET_SUM)
                population.append(individual)
            
            # Evaluate initial population
            fitness = [evaluate_c5(ind) for ind in population]

            max_generations = 2000

                for i in range(pop_size):
                    # Select three distinct donors
                    donors = list(range(pop_size))

                    # Mutation
                    F = 0.8

                    if trial_c5 <= fitness[i]:
