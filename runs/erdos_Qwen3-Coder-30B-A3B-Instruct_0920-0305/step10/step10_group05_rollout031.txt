import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_autocorr_fft(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def c5_from_autocorr(autocorr):
    """Convert autocorrelation result to C5 value"""
    min_autocorr = np.min(autocorr)
    return 1 - dx * min_autocorr

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def evaluate_c5_by_autocorr(h):
    """Evaluate C5 using autocorrelation for potential speedup"""
    autocorr = compute_autocorr_fft(h)
    return c5_from_autocorr(autocorr)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Try parameterized piecewise constant approach to get good initial solution
    best_a = 0.2
    best_c5 = float('inf')
    best_h = None
    
    # Scan a values for symmetric piecewise constant function
    a_values = np.linspace(0.05, 0.45, 200)
    
    for a in a_values:
        if time.time() - start_time > budget_s * 0.2:
            break
        # Compute p for the middle region
        p = (1 - 2 * a) / (2 * (1 - a))
        if p < 0 or p > 1:
            continue
        # Create the piecewise constant h
        h = np.zeros(n_points)
        edge_length = int(a * n_points / 2)
        mid_length = n_points - 2 * edge_length
        # Set edge regions to 1
        h[:edge_length] = 1
        h[-edge_length:] = 1
        # Set middle region to p
        h[edge_length:edge_length + mid_length] = p
        # Ensure sum = 1
        h = h * (n_points / 2) / np.sum(h)
        # Make sure h is in [0,1]
        h = np.clip(h, 0, 1)
        
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # If we haven't found good solution yet, proceed with simulated annealing
    if best_h is None or best_c5 > 0.4:
        # Initialize with best from parametric sweep or random
        if best_h is None:
            h0 = np.random.rand(n_points)
            h0 = project_to_sum_box(h0, n_points / 2)
        else:
            h0 = best_h
        
        # Simulated Annealing
        max_iter = 200000
        T_start = 0.001
        T = T_start
        decay_rate = 0.9999
        log_T = np.log(T_start)
        
        current_h = h0.copy()
        current_c5 = evaluate_c5(current_h)
        best_h = current_h.copy()
        best_c5 = current_c5
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
            
            # Random move: pick two indices and move mass
            i1 = np.random.randint(0, n_points)
            i2 = np.random.randint(0, n_points)
            
            # Choose delta from a normal distribution
            delta = np.random.normal(0, 0.01)
            
            # Ensure we can make the move
            if delta > 0:
                if current_h[i1] >= delta and current_h[i2] <= 1 - delta:
                    h_new = current_h.copy()
                    h_new[i1] -= delta
                    h_new[i2] += delta
                    # Project to maintain constraints
                    h_new = project_to_sum_box(h_new, n_points / 2)
                else:
                    continue
            else:
                if current_h[i2] >= -delta and current_h[i1] <= 1 + delta:
                    h_new = current_h.copy()
                    h_new[i1] -= delta
                    h_new[i2] += delta
                    h_new = project_to_sum_box(h_new, n_points / 2)
                else:
                    continue
            
            c5_new = evaluate_c5(h_new)
            
            # Accept or reject
            if c5_new < current_c5:
                current_h = h_new.copy()
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_h = current_h.copy()
                    best_c5 = c5_new
            else:
                # Accept with probability based on temperature
                if np.random.rand() < np.exp(-(c5_new - current_c5) / T):
                    current_h = h_new.copy()
                    current_c5 = c5_new
            
            # Update temperature
            if i % 1000 == 0:
                log_T -= 0.0001
                T = np.exp(log_T)
        
        # Fine-tune using local optimization
        if time.time() - start_time < budget_s * 0.98:
            def objective(h):
                return evaluate_c5(h)
            
            def constraint_eq(h):
                return np.sum(h) - TARGET_SUM
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': constraint_eq}
            
            try:
                res = minimize(
                    objective, 
                    best_h, 
                    method='L-BFGS-B', 
                    bounds=bounds, 
                    constraints=cons,
                    options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
                )
                if res.success:
                    h_opt = res.x
                else:
                    h_opt = best_h
                h_opt = project_to_sum_box(h_opt, TARGET_SUM)  
                c5_opt = evaluate_c5(h_opt)
                
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
            except Exception:
                pass
    
    return (best_h.tolist(), float(best_c5), n_points)
