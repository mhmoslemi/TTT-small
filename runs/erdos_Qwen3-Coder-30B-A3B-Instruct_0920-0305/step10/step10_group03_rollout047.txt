import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_milp_skeleton(N=60):
    """Solve a binary MILP to get a good combinatorial starting point"""
    x = cp.Variable(N, boolean=True)
    t = cp.Variable()
    
    # Constraints for correlation computation
    y = {}
    constraints = []
    
    # Constraint: mass
    constraints.append(cp.sum(x) == N/2)
    
    # Constraint: for each k, no more than (N/2 - t) ones overlap
    for k in range(1, N):
        y[k] = cp.Variable(N-k, boolean=True)
        # y[i] = x[i]*x[i+k]
        for i in range(N-k):
            constraints.extend([
                y[k][i] <= x[i],
                y[k][i] <= x[i+k],
                y[k][i] >= x[i] + x[i+k] - 1
            ])
        constraints.append(cp.sum(y[k]) >= N/2 - t)
        
    # Set objective: minimize t
    objective = cp.Minimize(t)
    
    # Solve the MILP
    problem = cp.Problem(objective, constraints)
    try:
        problem.solve(solver=CPLEX, time_limit=30)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_sol = x.value
            return x_sol
    except:
        pass
    # Fallback to random binary
    return np.random.randint(0, 2, N)

def upsample_binary_vector(x_binary, target_length):
    """Upsample binary vector to target_length"""
    n = len(x_binary)
    upsample_factor = target_length // n
    if upsample_factor < 1:
        upsample_factor = 1
        
    x_upsampled = np.repeat(x_binary, upsample_factor)
    # Pad if needed
    if len(x_upsampled) < target_length:
        x_upsampled = np.pad(x_upsampled, (0, target_length-len(x_upsampled)), 'constant', constant_values=0)
    elif len(x_upsampled) > target_length:
        x_upsampled = x_upsampled[:target_length]
    
    return x_upsampled

def swap_search_on_binary(h_binary, max_swaps=10000):
    """Perform iterative swap search to reduce worst shift in the binary problem"""
    h = h_binary.copy()
    n = len(h)
    
    # Precompute correlations for all shifts
    def compute_correlation(h_h):
        corr = []
        for k in range(1, n):
            corr_k = 0
            for i in range(n - k):
                corr_k += h_h[i] * h_h[i + k]
            corr.append(corr_k)
        return np.array(corr)
    
    rho = compute_correlation(h)
    max_corr = np.max(rho)
    min_corr = np.min(rho)
    
    # Find worst shift
    k_star = np.argmin(rho)
    
    # Iterative swap hold
    for swap in range(max_swaps):
        # Find elements that can be swapped to improve worst shift
        A = []
        B = []
        k = k_star + 1  # because range 1..n-1 corresponds to indices 0..n-2
        for i in range(n - k):
            if h[i] == 1 and h[i + k] == 0: 
                A.append(i)
            if h[i] == 0 and h[i + k] == 1:
                B.append(i)
        
        # If no swap brings improvement, do a random one
        if len(A) > 0 and len(B) > 0:
            a = np.random.choice(A)
            b = np.random.choice(B)
            h[a], h[b] = h[b], h[a]  # swap
            rho = compute_correlation(h)  # update
            k_star = np.argmin(rho)
        elif np.random.rand() < 0.1:
            # Random swap to escape local minima
            i, j = np.random.randint(0, n, 2)
            h[i], h[j] = h[j], h[i]
            rho = compute_correlation(h)
            k_star = np.argmin(rho)
    
    return h

# Optimizations using DE and simulated annealing
def differential_evolution_optimization(h0, n_points, budget_s):
    """Differential evolution optimization on continuous variables"""
    def objective(h):
        return evaluate_c5(h)
    
    popsize = 30
    maxiter = 10000
    N = n_points
    F = 0.8
    CR = 0.9
    
    # Initialize population
    population = []
    for _ in range(popsize):
        h = h0 + np.random.normal(0, 0.02, N)
        h = np.clip(h, 0, 1)
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    
    # Evaluate initial population
    fitness = [objective(ind) for ind in population]
    best_idx = np.argmin(fitness)
    best_individual = population[best_idx]
    best_fitness = fitness[best_idx]
    
    # Time tracking
    start_time = time.time()
    
    for iteration in range(maxiter):
        if time.time() - start_time > budget_s - 5:
            break
            
        for i in range(popsize):
            # Select three distinct individuals
            idxs = [j for j in range(popsize) if j != i]
            a, b, c = np.random.choice(idxs, 3, replace=False)
            
            # Mutation
            mutant = population[a] + F * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            trial = []
            for j in range(N):
                if np.random.rand() < CR or j == np.random.randint(0, N):
                    trial.append(mutant[j])
                else:
                    trial.append(population[i][j])
            trial = np.array(trial)
            
            # Project
            trial = project_to_sum_box(trial, TARGET_SUM)
            
            # Selection
            trial_fitness = objective(trial)
            if trial_fitness <= fitness[i]:
                population[i] = trial
                fitness[i] = trial_fitness
                if trial_fitness < best_fitness:
                    best_individual = trial.copy()
                    best_fitness = trial_fitness
    
    return best_individual, best_fitness

def simulated_annealing_optimization(h0, budget_s):
    """Simulated Annealing optimization"""
    h_current = h0.copy()
    c5_current = evaluate_c5(h_current)
    h_best = h_current.copy()
    c5_best = c5_current
    
    start_time = time.time()
    T = 1.0
    alpha = 0.99
    max_iter = 10000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Pick two indices
        i1, i2 = np.random.randint(0, len(h_current), 2)
        while i1 == i2:
            i2 = np.random.randint(0, len(h_current))
            
        # Transfer small amount
        delta = np.random.uniform(-0.05, 0.05)
        if h_current[i1] + delta < 0:
            delta = -h_current[i1]
        elif h_current[i1] + delta > 1:
            delta = 1 - h_current[i1]
            
        h_trial = h_current.copy()
        h_trial[i1] += delta
        h_trial[i2] -= delta
        
        # Ensure boundaries
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = project_to_sum_box(h_trial, TARGET_SUM)
        
        c5_trial = evaluate_c5(h_trial)
        
        # Accept or reject
        if c5_trial < c5_current:
            h_current = h_trial
            c5_current = c5_trial
            if c5_trial < c5_best:
                h_best = h_trial.copy()
                c5_best = c5_trial
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(c5_trial - c5_current) / T):
                h_current = h_trial
                c5_current = c5_trial
                
        T *= alpha
    
    return h_best, c5_best

def hill_climbing_optimization(h0, budget_s):
    """Deterministic hill climbing"""
    h_current = h0.copy()
    c5_current = evaluate_c5(h_current)
    h_best = h_current.copy()
    c5_best = c5_current
    
    start_time = time.time()
    max_iter = 2000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Pick small subset of indices
        n_select = max(1, int(0.05 * len(h_current)))
        indices = np.random.choice(len(h_current), n_select, replace=False)
        
        # Propose perturbations (with varying std for different values)
        h_trial = h_current.copy()
        for idx in indices:
            std = 0.1 * h_current[idx] * (1 - h_current[idx])
            delta = np.random.normal(0, std)
            h_trial[idx] += delta
            
        # Clip and project
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = project_to_sum_box(h_trial, TARGET_SUM)
        
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < c5_current:
            h_current = h_trial
            c5_current = c5_trial
            if c5_trial < c5_best:
                h_best = h_trial.copy()
                c5_best = c5_trial
    
    return h_best, c5_best
