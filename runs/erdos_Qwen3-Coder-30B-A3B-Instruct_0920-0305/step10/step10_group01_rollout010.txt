import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h, L) * np.conj(np.fft.fft(b, L))).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max(c, alpha):
    """Compute smooth maximum with numerical stability"""
    c_max = np.max(c)
    exp_term = np.exp(alpha * (c - c_max))
    sum_exp = np.sum(exp_term)
    return c_max + np.log(sum_exp) / alpha

def gradient_step(h, alpha, step_size):
    """Perform one gradient step with smoothing parameter alpha"""
    # compute correlation raw c (full)
    c = compute_corr_fft(h)
    
    # compute softmax weights with numerical stability
    c_max = np.max(c)
    exp_term = np.exp(alpha * (c - c_max))
    w = exp_term / np.sum(exp_term)
    
    # compute w_sym = w + w[::-1] (symmetric weights)
    w_sym = w + w[::-1]
    
    # compute correlation between w_sym and h (full)
    L = 2 * len(h) - 1
    fw_sym = np.fft.fft(w_sym, L)
    fh = np.fft.fft(h, L)
    corr_w_sym_h = np.fft.ifft(fw_sym * np.conj(fh)).real
    
    # gradient = 1 - corr_w_sym_h[:n]
    gradient = 1 - corr_w_sym_h[:len(h)]
    
    # update h
    h_new = h - step_size * gradient
    
    # project back to feasible set
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Use the provided initial solution as starting point if available
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h0 = np.array(initial_h_values)
    else:
        # Generate a random binary-like initialization
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
        np.random.shuffle(h0)
    
    h0 = project_to_sum_box(h0, TARGET_SUM)
    current_h = h0.copy()
    current_c5 = evaluate_c5(h0)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()

    # Smoothing parameters
    alpha_max = 50.0
    alpha_schedule = np.linspace(0.5, alpha_max, 50)
    
    # Inner optimization loop
    max_inner_iter = 5000
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_inner_iter:
        # Find current alpha based on iteration count
        alpha_idx = min(iter_count // 100, len(alpha_schedule) - 1)
        alpha = alpha_schedule[alpha_idx]
        
        # Step size scheduler
        step_size = 0.05 / (1 + iter_count / 2000)
        
        # Gradient step
        current_h = gradient_step(current_h, alpha, step_size)
        current_c5 = evaluate_c5(current_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
        
        iter_count += 1
    
    # Final evaluation of best solution
    if best_h is None:
        best_h = h0.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
