import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

# Global variables
n_points = 200
dx = 2.0 / n_points
half_size = n_points // 2
target_sum = n_points / 4

def autocorr_fft(h):
    """Compute autocorrelation using FFT."""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def compute_c5_exact(h):
    """Compute exact C5 value using numpy correlate for max."""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return 1.0 - dx * max_corr

def compute_c5_smooth(h, beta=50.0):
    """Compute smooth approximation to C5."""
    corr = autocorr_fft(h)
    corr = 1.0 - corr  # because we want h(x)(1-h(x+k))
    max_corr = np.max(corr)
    # use log-sum-exp smoothing
    # Actually, we want to maximize the minimum, so the C5 = 1 - min(A)
    # so smooth = 1 - (1/beta)*log(sum(exp(beta*min(A))))
    # but that is equivalent to computing smooth max of the negative
    # let's recompute the actual minimal correlation
    min_corr = np.min(corr)
    return 1.0 - dx * min_corr

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    try:
        while f(a) * f(b) > 0:
            a -= 0.1
            b += 0.1
    except:
        a, b = -0.5, 0.5
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def parametric_shape(a):
    """Generate symmetric shape with two edges at height 1 and middle plateau."""
    # a = length of each edge block (in [0,0.5])
    h = np.zeros(half_size)
    edge_len = int(np.round(a / dx))
    edge_len = min(edge_len, half_size // 2)
    if edge_len > 0:
        h[:edge_len] = 1.0
        h[-edge_len:] = 1.0
    
    # compute the plateau height
    remaining = half_size - 2 * edge_len
    if remaining > 0:
        # area constraint: 2*edge_len*1 + remaining*p = target_sum  
        p = (target_sum - 2 * edge_len * 1.0) / remaining
        if p < 0:
            p = 0
        h[edge_len:edge_len+remaining] = p
    
    return h

def parametric_sweep():
    """Sweep parameterized shapes for coarse search."""
    best_c5 = float('inf')
    best_h = None

    step = 0.002
    a_range = np.arange(0.01, 0.49, step)

    for a in a_range:
        h = parametric_shape(a)
        h = project_to_sum_box(h, target_sum)
        c5 = compute_c5_exact(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    return best_h, best_c5

def smooth_gradient(h, beta=50.0):
    """Compute gradient of smooth surrogate for C5."""
    # Compute the autocorrelation
    corr = autocorr_fft(h)
    # The gradient of 1 - (1/beta)*log(sum(exp(beta * (1 - corr))))  
    # is (1/beta) * (sum(exp(beta * (1 - corr)))^{-1}) * exp(beta * (1 - corr)) * (-beta)
    # = - exp(beta * (1 - corr)) / sum(exp(beta * (1 - corr)))

    exp_term = np.exp(beta * (1 - corr))
    weights = exp_term / np.sum(exp_term)

    # gradient of correlation wrt h_i
    grad = np.zeros_like(h)
    # For each i we compute the derivative of sum_k exp(beta * A_k)
    # where A_k = sum_j h_j * h_{j+k}
    for i in range(len(h)):
        # grad[i] = - (1/beta) * sum_k weights[k] * dh_k/dh_i  
        # dh_k/dh_i can be computed based on h_{i+k} or h_{i-k}
        temp = np.zeros_like(h)
        temp[i] = 1.0
        aux_corr = autocorr_fft(temp)
        # now we differentiate thing by terms i, k -> i+k, i-k
        # This gives us the derivative
        temp_grad = np.zeros_like(h)
        for j in range(len(h)):
            temp_grad[j] = 0.0
            if (j - i) >= 0 and (j - i) < len(h):
                temp_grad[j] += h[i]
            if i - (len(h) - j) >= 0 and i - (len(h) - j) < len(h):  
                temp_grad[j] += h[i]
        # But this backward direction is tricky.
        # Instead we compute by cycling through all k values and applying chain rule
        # let's just use a numeric method here to be robust, at a slight cost:
        # compute at finite difference
        # super elegant method: we need sum_k weights[k] * \partial A_k/\partial h_i
        # = \sum_{j} (h_{i+k} + h_{i-k}) * weights[k] where we assume h_{...} = 0 if out of bounds
        # For symmetry we can do very efficiently:
        grad_part_fwd = np.zeros_like(h)
        grad_part_bwd = np.zeros_like(h)

        for j in range(len(h)):
            k = (j - i) % len(h)  # shift
            if k < len(corr):
                grad_part_fwd[j] += corr[k]  # this is - dx weight*k * h_{i - k} at shift k
        # but it's simpler to use energy pattern linearity:
        # we are looking at d/dh_i of sum_k w_k A_k
        #         = sum_k w_k dA_k/dh_i = sum_k w_k (h_{i+k} + h_{i-k})
        grad_line = np.zeros_like(h)
        for j in range(len(h)):
            # h_{i+k} for k = j - i
            if (j - i) >= 0:
                grad_line[j] += h[i + (j - i)] if i + (j - i) < len(h) else 0
            if (i - j) > 0:
                grad_line[j] += h[i - (j - i)] if i - (j - i) < len(h) else 0

        # Actually, in the window we have the tile it right
        # Let's first compute it directly based on convolution
        # y = fft( h_rev ) * fft( h ) gives us the auto-correlation
        # So conv(np.flip(h), h) with wraparound is: conv(h_rev, h)
        # Gradient of y_k w.r.t h_i is:
        #   d / dh_i sum_j h_rev[j] h[j + k] when k is fixed 
        #   If j = i-s, then h_rev[i-s] h[i-s+k] 
        #   If j = i-k, then h_rev[i-k] h[i-k+k] 
        # Complexity avoids.

        # Instead, just brute-force calculation using ifft.
        # Use that in convolution derivative it's smoother to sum for all indices...
        # Just use central finite differences for now, or derive it properly using FFT and shift.

        # Simpler to make finite-diff computation stable:
        grad[i] = 0
        # Actually just use negated form of finite dif with 1e-5 step
        for j in range(len(h)):
            if h[j] > 0:
                # Compute gradient procedure:
                delta = 1e-5
                h_temp = h.copy()
                h_temp[j] += 2.0 * delta
                c5_up = compute_c5_smooth(h_temp, beta)
                h_temp = h.copy()
                h_temp[j] -= 2.0 * delta
                c5_down = compute_c5_smooth(h_temp, beta)
                grad[i] += (c5_up - c5_down) / (4.0 * delta)

    return -grad / beta

def lbfgsb_optimize_on_parametric(h0, maxiter=200):
    """Perform L-BFGS-B optimization starting from parametric."""
    def smooth_obj(h):
        return compute_c5_smooth(h, beta=50)  

    bounds = [(0, 1) for _ in range(half_size)]
    res = minimize(
        smooth_obj,
        h0,
        method="L-BFGS-B",
        bounds=bounds,
        options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-12}
    )
    if res.success:
        h_opt = res.x
    else:
        h_opt = h0
    h_opt = project_to_sum_box(h_opt, target_sum)
    return h_opt

def simulated_annealing(h_start, max_iter=150000, initial_T=0.01, cooling_rate=0.9995):
    """Perform simulated annealing on the half vector."""
    current_h = h_start.copy()
    current_c5 = compute_c5_exact(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    T = initial_T
    delta_max = 0.05
    acc_count = 0
    acc_total = 0

    for i in range(max_iter):
        # Time limit
        if time.time() - start_time > budget_s * 0.85:
            break

        # New candidate by swapping mass
        j, k = np.random.randint(0, half_size, 2)
        delta = np.random.uniform(0, delta_max) 

        # Try to move from j to k
        # Ensure it doesn't go out of bounds
        if current_h[j] >= delta and current_h[k] <= 1 - delta:
            h_new = current_h.copy()
            h_new[j] -= delta
            h_new[k] += delta

            h_new = project_to_sum_box(h_new, target_sum)
            c5_new = compute_c5_exact(h_new)

            # Accept criteria
            if c5_new < current_c5:
                current_h = h_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                acc_count += 1
                acc_total += 1
            else:
                # Accept with probability
                if np.random.rand() < np.exp(-(c5_new - current_c5) / T):
                    current_h = h_new
                    current_c5 = c5_new
                    acc_count += 1
                    acc_total += 1

        # Adaptive step size based on acceptance rate
        if (i + 1) % 500 == 0 and i > 0:
            if acc_total > 0:
                acc_rate = acc_count / acc_total
                if acc_rate > 0.4:
                    delta_max *= 1.1
                elif acc_rate < 0.1:
                    delta_max *= 0.9
                delta_max = np.clip(delta_max, 1e-4, 0.2)
                acc_count = 0
                acc_total = 0
            
        T *= cooling_rate

        # Periodic storage
        if i % 1000 == 0:
            pass  # just iteration bookkeeping

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global start_time
    start_time = time.time()
    np.random.seed(seed)

    h_best = None
    c5_best = float('inf')

    # Stage I: parametric sweep
    h0, c50 = parametric_sweep()
    if c50 < c5_best:
        h_best = h0.copy()
        c5_best = c50

    # Stage II: Optimization with L-BFGS-B
    h_opt = lbfgsb_optimize_on_parametric(h0)
    c5_opt = compute_c5_exact(h_opt)
    if c5_opt < c5_best:
        h_best = h_opt
        c5_best = c5_opt

    # Stage III: Simulated Annealing
    h_sa, c5_sa = simulated_annealing(h_opt)
    if c5_sa < c5_best:
        h_best = h_sa
        c5_best = c5_sa

    # Stage IV: Final polishing with L-BFGS-B
    h_final = lbfgsb_optimize_on_parametric(h_sa)
    c5_final = compute_c5_exact(h_final)
    if c5_final < c5_best:
        h_best = h_final
        c5_best = c5_final

    # Ensure we properly reflect to full vector
    h_full = np.concatenate([h_best, h_best[::-1]])
    h_full = project_to_sum_box(h_full, target_sum * 2)

    # Final call to compute exact C5 value
    exact_c5 = compute_c5_exact(h_full)

    return (h_full.tolist(), float(exact_c5), n_points)
