import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_smoothing(h, alpha):
    """Evaluate smooth surrogate F_alpha(h) with temperature alpha"""
    corr = compute_corr_fft(h)
    # Clip to prevent overflow
    corr = np.clip(corr, -100, 100)
    exp_corr = np.exp(alpha * corr)
    F_alpha = np.log(np.sum(exp_corr)) / alpha
    return F_alpha

def compute_gradient(h, alpha):
    """Compute gradient of smooth surrogate using FFT"""
    corr = compute_corr_fft(h)
    corr = np.clip(corr, -100, 100)
    exp_corr = np.exp(alpha * corr)
    Z = np.sum(exp_corr)
    w = exp_corr / Z  # Softmax weights
    # Symmetrize weights
    w_sym = w + np.roll(w, len(w)//2)[::-1]
    # Correlate with h to get g
    g_full = compute_corr_fft(w_sym)
    g = g_full[:len(h)]  # First N entries
    grad = 1 - g
    return grad

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result
