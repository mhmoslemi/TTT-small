import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT for faster evaluation"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def upscale_h(original_h, n_new):
    """Upscale h from old length to new length using linear interpolation"""
    n_old = len(original_h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, original_h)

def propose_move(h, max_step, rng):
    """Propose a move that maintains sum and box constraints"""
    n = len(h)
    # try up to few times to find feasible pair
    for _ in range(10):
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # compute feasible interval for delta added to i and subtracted from j
        lower = max(-h[i], h[j] - 1)
        upper = min(1 - h[i], h[j])
        if lower < upper:
            # limit delta magnitude
            lower = max(lower, -max_step)
            upper = min(upper, max_step)
            if lower >= upper:
                continue
            delta = rng.uniform(lower, upper)
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            return new_h
        # try opposite direction: add to j, subtract from i
        lower = max(-h[j], h[i] - 1)
        upper = min(1 - h[j], h[i])
        if lower < upper:
            lower = max(lower, -max_step)
            upper = min(upper, max_step)
            if lower >= upper:
                continue
            delta = rng.uniform(lower, upper)
            new_h = h.copy()
            new_h[j] += delta
            new_h[i] -= delta
            return new_h
    return None

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Generate initial candidate solutions
    initial_candidates = []
    
    # From the provided initial solution
    try:
        global initial_h_values
        upsampled_h = upscale_h(np.array(initial_h_values), n_points)
        upsampled_h = project_to_sum_box(upsampled_h, TARGET_SUM)
        initial_candidates.append(upsampled_h)
    except:
        pass
    
    # Cosine shape
    xs = np.arange(n_points) * dx
    a = 0.5 - 1e-5
    cos_h = 0.5 + a * np.cos(np.pi * xs / 2)
    cos_h = project_to_sum_box(cos_h, TARGET_SUM)
    initial_candidates.append(cos_h)
    
    # Random uniform
    rand_h = rng.uniform(0, 1, n_points)
    rand_h = project_to_sum_box(rand_h, TARGET_SUM)
    initial_candidates.append(rand_h)
    
    # Binary random
    binary_h = np.zeros(n_points)
    ones_idx = rng.choice(n_points, size=n_points // 2, replace=False)
    binary_h[ones_idx] = 1.0
    initial_candidates.append(binary_h)
    
    # Evaluate all initial candidates and select the best
    for candidate in initial_candidates:
        val = evaluate_c5(candidate)
        if val < best_c5:
            best_c5 = val
            best_h = candidate.copy()
    
    # Simulated Annealing
    curr_h = best_h.copy()
    curr_val = best_c5
    
    # SA parameters
    total_time = budget_s * 0.9
    T0 = 0.02
    T_min = 0.0001
    max_step_initial = 0.2
    max_step_final = 0.001
    max_iter = 500000
    
    iter_count = 0
    while time.time() - start_time < total_time and iter_count < max_iter:
        iter_count += 1
        
        # Temperature and step size
        elapsed = time.time() - start_time
        fraction = elapsed / total_time
        T = T0 * (1 - fraction) + T_min * fraction
        max_step = max_step_initial * (1 - fraction) + max_step_final
        
        # Propose move
        new_h = propose_move(curr_h, max_step, rng)
        if new_h is None:
            continue
            
        # Evaluate new solution
        new_val = evaluate_c5(new_h)
        
        # Accept or reject
        delta = new_val - curr_val
        if delta <= 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            curr_h = new_h
            curr_val = new_val
            if new_val < best_c5:
                best_c5 = new_val
                best_h = curr_h.copy()
    
    # Final deterministic local search
    curr_h = best_h.copy()
    curr_val = best_c5
    
    max_step_refine = 0.005
    refine_iter = 50000
    
    for i in range(refine_iter):
        if time.time() - start_time > budget_s * 0.98:
            break
        elapsed = time.time() - start_time
        fraction = elapsed / (budget_s * 0.98)
        max_step = max_step_refine * (1 - fraction) + 1e-6
        new_h = propose_move(curr_h, max_step, rng)
        if new_h is None:
            continue
        new_val = evaluate_c5(new_h)
        if new_val < curr_val:
            curr_h = new_h
            curr_val = new_val
            if new_val < best_c5:
                best_c5 = new_val
                best_h = curr_h.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
