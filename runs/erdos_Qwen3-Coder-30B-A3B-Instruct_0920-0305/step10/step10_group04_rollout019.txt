import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points

def compute_overlap_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def compute_binary_overlap(x):
    """Fast evaluation of overlap for binary vector using correlation"""
    # Compute correlation of x with itself (auto-correlation)
    A = correlate(x, x, mode="full")
    n = len(x)
    
    # Get the middle part of the correlation which corresponds to valid shifts
    center = len(A) // 2
    A = A[center - n + 1:center + n - 1]
    
    # Compute overlap for different shifts
    # For shift d: overlap = sum_i x_i * (1 - x_{i+d})
    # This is equivalent to: sum_i x_i - sum_i x_i x_{i+d}
    # But sum_i x_i = n/2, so overlap = n/2 - A_d
    overlap = np.full(len(A), n/2, dtype=float)
    for i in range(len(A)):
        if i < n:
            # Compute overlap for shift i
            try:
                overlap[i] -= A[i]
            except IndexError:
                pass
    
    # Handle wraps around
    # We need to compute for both positive and negative shifts properly
    # In our case we only analyze shift d in [-n+1,n-1], so we'll have explicit negative shifts
    # Computing overlaps for all possible shifts using prefix sum
    prefix = np.cumsum(x)
    
    # For a shift d, the overlap is computed as the total number of 1s in one window minus the number of matches
    shifts = np.arange(-(n-1), n)
    overlap_values = np.zeros(len(shifts))
    
    # For this implementation, using efficient loop-based calculation
    for i, d in enumerate(shifts):
        if d >= 0:
            # right shift case
            start = d
            end = n
            if start < n:
                window_sum = prefix[end-1] - (prefix[start-1] if start > 0 else 0)
                count = (x[start:end]).sum()
                overlap_values[i] = count - A[n-1+i]
            else:
                overlap_values[i] = 0
        else:
            # left shift case  
            start = 0
            end = n + d
            if end > 0:
                window_sum = prefix[end-1]
                count = (x[start:end]).sum()
                overlap_values[i] = count - A[n-1+i]
            else:
                overlap_values[i] = 0
    
    # Return max of all shifts  
    return dx * np.max(overlap_values)

def compute_binary_overlap_new(x):
    """Fast overlapping computation using prefix sum approach"""
    # Compute prefix sums 
    prefix = np.concatenate([[0], np.cumsum(x)])
    n = len(x)
    
    # For shift d, we want sum over i of x_i * (1 - x_{i+d})
    # This equals sum x_i - sum x_i * x_{i+d} 
    # The second part is same as A_d (auto-correlation, but we count overlapping pairs)
    A = correlate(x, x, mode="full")
    center = len(A) // 2 
    A = A[center - n + 1:center + n - 1]
    
    # Simplifying: sum of x_i at i, shifted by d, where we count the sum of products of x_i * x_{i+d}
    # That is the correlation A_d by definition
    # Overlap for shift d = sum_{i} x_i - sum_{i} x_i * x_{i+d}
    # But also it is the count of ones in the window positions plus then adjust by the overlap
    
    # We use an analytical approach:
    # For any shift d:
    # start, end valid indices for checking: 
    # if d >= 0, the left shift is from 0 to n-d, values at left side
    # if d < 0, then the right shift is from -d to n, the right side
    # This directly corresponds:
    # overlap = sum(x) - product match
    
    # Manual computation:
    # Let's compute per shift for both directional bias
    max_overlap = 0
    for d in range(-(n-1), n):
        # compute overlap for shift d
        if d >= 0:
            # x[i+d] becomes x[i] where i runs from 0 to n-d-1
            # Overlap = (Sum of x_i) - (Sum of x_i * x_{i+d})
            if d >= n:
                overlap = 0
            else:
                # overlap = sum(x)-sum(x(i) * x(i+d)); if x(1+2*j)*x(3+2*j) 
                # pair product overlap is when non-shifts are sequential positions
                total = np.sum(x)
                # count overlapping 1's
                total_match = 0
                for i in range(n-d):
                    total_match += (x[i]*x[i+d])
                overlap = total - total_match
            # store the max overlap
        else:
            d = abs(d)
            # equivalent to shift -d which means right-shift by d
            if d >= n:
                overlap = 0
            else:
                # Iterate manually: compute overlap for shifts d  
                total = np.sum(x)
                total_match = 0
                for i in range(n-d):
                    total_match += (x[i]*x[i+d])
                overlap = total - total_match
                
        if overlap > max_overlap:
            max_overlap = overlap
    
    return dx * max_overlap

def compute_overlap_complex(x):
    """Use prefix sum approach more correctly for all overlaps, without using 
       naive trigonometric reduction."""
    # x is a binary vector of length n
    # the overlap at shift d is:
    # sum_{i} x[i] (1 - x[i+d])
    # Which means: sum_{i} x[i] - sum_{i} x[i] * x[i+d]
    # First sum is simply sum_{i} x[i] = count
    n = len(x)
    
    # Compute prefix sum  
    prefix = np.concatenate([[0], np.cumsum(x)])
    
    # Total number of 1s in the vector
    total_ones = np.sum(x)
    
    # For computing overlap at shift d:
    # Overlap = total_ones - dot product from x at i to n-d
    # of x and x_slided
    
    # But avoid iteration by computing all correlations and overlaps  
    max_overlap = 0
    
    # Use a sliding window analysis for all possible shifts:
    for d in range(-(n-1), n):
        if d >= 0:
            # Slide x to the right by d
            start = 0  
            end = n - d
            window_slice = x[start:end]
            matched_slice = x[d:end+d]  
            match = np.sum(window_slice * matched_slice)
            overlap = total_ones - match
        else:
            # d < 0, slide x to the left
            d_abs = abs(d)
            start = d_abs
            end = n
            window_slice = x[start:end]
            matched_slice = x[:n-d_abs]
            match = np.sum(window_slice * matched_slice)
            overlap = total_ones - match
            
        if overlap > max_overlap:
            max_overlap = overlap
            
    return dx * max_overlap

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using binary search"""
    # Simple solution: just clamp
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def swap_indices(arr, i, j):
    """Swap elements in an array at positions i, j, updates sum constraint"""
    temp = arr[i]
    arr[i] = arr[j]
    arr[j] = temp
    return arr

def get_worst_shift_indices(x, current_shift=None):
    """Determine indices that lead to the worst overlap at some shift,
       and handle natural wrap in overlap"""
    n = len(x)
    # First compute all overlaps efficiently 
    # Focus on exact overlap for large overlaps
    worst_shift = 0
    max_overlap = 0
    
    # Use fast sliding window method
    prefix = np.concatenate([[0], np.cumsum(x)])
    total_ones = np.sum(x)
    
    for d in range(-(n-1), n):
        if d >= 0:
            # this shift is looking at advancing i into i+d 
            # overlap is [1 to n-d, 0 to n-d], count 1s and matches: numerically do prefix 
            start = 0  
            end = n - d
            if end == 0:
                overlap = total_ones
            else:
                window_count = prefix[end] - prefix[start]
                matched = 0
                # Count flanking overlaps exactly
                overlap = 0
                for i in range(end):
                    matched += x[i] * x[i + d]
                overlap = window_count - matched
        else:
            # d is negative
            abs_d = abs(d)
            start = 0  
            end = n - abs_d
            if end == 0:
                overlap = total_ones  # or zero
            else:
                matched = 0
                # same approach
                for i in range(end):
                    matched += x[i] * x[i - d]
                window_count = prefix[end] - prefix[start]
                overlap = window_count - matched
        if overlap > max_overlap:
            max_overlap = overlap
            worst_shift = d
    return worst_shift

def binary_to_real(x_binary):
    """Convert binary vector to a real-valued approximation"""
    return x_binary.astype(np.float64)

def compute_c5_real(h):
    """Compute C5 for real-valued function"""
    corr = correlate(h, 1-h, mode="full") * dx
    return np.max(corr)

def evaluate_c5_fast(x):
    """Fast evaluation of C5 for a binary vector"""
    # Want to compute max_i sum_j=0^{n-1} x_j * (1 - x_{j+i})
    # which is min_i sum_j=0^{n-1} x_j x_{j+i} + total sum - (sum_i 1 - overlap_i)
    # patched version using method of sliding aggregates
    n = len(x)
    
    # slide through shifts and directly compute
    prefix = np.concatenate([[0], np.cumsum(x)])
    total_ones = np.sum(x)
    max_overlap = 0
    
    for shift in range(-(n-1), n):
        if shift >= 0:
            # right shift  
            overlap = total_ones - np.sum(x[:n-shift] * x[shift:])
        else:
            # left shift  
            overlap = total_ones - np.sum(x[-shift:] * x[:-shift])
        if overlap > max_overlap:
            max_overlap = overlap
    
    return dx * max_overlap

    n_iters = 300000
    T = T0
    accept_count = 0
    stagnation_counter = 0
    last_improvement = 0
    
    # Begin SA loop
    for iteration in range(n_iters):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Determine type of move
        if np.random.rand() < 0.4: # random swap
            i, j = np.random.choice(n_points, 2, replace=False)
            while current_binary[i] == current_binary[j]:
                i, j = np.random.choice(n_points, 2, replace=False)
        else:
            # Find current worst shift
            worst_shift = get_worst_shift_indices(current_binary)
            
            # If worst_shift is not well defined, fall back
            try:  # checks if they're empty
                exposed = np.where((current_binary == True) & 
                                   (np.roll(current_binary, -worst_shift) == False))[0]
                missing = np.where((current_binary == False) &

                if len(exposed) > 0 and len(missing) > 0:
                    i = np.random.choice(exposed)
                    j = np.random.choice(missing)
                else:
                    i, j = np.random.choice(n_points, 2, replace=False)
                    while current_binary[i] == current_binary[j]:
                        i, j = np.random.choice(n_points, 2, replace=False)
            except:
                # fallback to random
                i, j = np.random.choice(n_points, 2, replace=False)
                while current_binary[i] == current_binary[j]:
                    i, j = np.random.choice(n_points, 2, replace=False)
        
        # Execute swap in binary vector
        current_binary[i], current_binary[j] = current_binary[j], current_binary[i]
        
        # Recompute C5 
        old_c5 = current_c5
        current_c5 = evaluate_c5_fast(current_binary.astype(np.int8))

        if current_c5 < old_c5:
            accept_count += 1
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_binary = current_binary.copy()

    try:
        res = minimize(obj_real, best_real, method='SLSQP', bounds=bounds, constraints=cons,

            refined_real = project_to_sum_box(refined_real, TARGET_SUM)

        if res.success:

            best_c5 = compute_c5_real(best_real)
