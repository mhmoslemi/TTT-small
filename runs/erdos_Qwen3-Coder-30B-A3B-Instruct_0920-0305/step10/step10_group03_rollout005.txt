import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_with_gradient_descent(h0, max_iter=5000):

        return evaluate_c5(h)

        return np.sum(h) - TARGET_SUM
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-14}
        )
        if res.success:
            return project_to_sum_box(res.x, TARGET_SUM)
    except:
        pass
    return h0

        for idx in indices:
            # Larger step size with probability, smaller step size otherwise
            if np.random.random() < 0.7:
                step_size = 0.03 * (1.0 - 2.0 * abs(h_new[idx] - 0.5))
            else:
                step_size = 0.005 * (1.0 - 2.0 * abs(h_new[idx] - 0.5))
            delta = np.random.normal(0, step_size)
            h_new[idx] += delta
            
        # Project to valid constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new)
        
        # Simulated annealing acceptance criterion
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = current_h.copy()
                best_c5 = c5_new
        else:
            # Accept worse solution with decreasing probability
            if np.random.random() < np.exp(-((c5_new - current_c5) / (1e-8 + current_c5 * 0.01))):
                current_h = h_new.copy()
                current_c5 = c5_new
    
    return best_h, best_c5

def alternating_optimization(h0, max_iter=500):
    """Try alternating optimization between subsets of variables"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Optimize using groups of indices
        groups = [np.arange(j, n_points, 6) for j in range(6)]
        
        for group in groups:
            h_test = h.copy()
            # This is a temporary change for group optimization
            h_group = h_test[group]
            h_group = np.clip(h_group + np.random.normal(0, 0.02, len(h_group)), 0, 1)
            h_test[group] = h_group
            
            h_test = project_to_sum_box(h_test, TARGET_SUM)
            c5_test = evaluate_c5(h_test)
            
            if c5_test < best_c5:
                h = h_test.copy()
                best_c5 = c5_test
        
        # Occasionally make global change
        if i % 10 == 0 and np.random.random() < 0.3:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = project_to_sum_box(h, TARGET_SUM)
            best_c5 = evaluate_c5(h)
    
    return h, best_c5
