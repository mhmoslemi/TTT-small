from scipy.optimize import differential_evolution, minimize
import time

# Global constants
TARGET_SUM = None

dx = 2.0 / n_points
half_n = n_points // 2

def project_to_sum_box(u, target_sum):
    """Water-filling projection onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(u + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(u + (a + b)/2, 0, 1)
    return result

def autocorr_fft(h):
    """Compute aperiodic autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(h, L)).real
    return corr

def exact_c5(u):
    """Compute exact C5 from half-vector u"""
    h = np.concatenate([u, u[::-1]])
    corr = autocorr_fft(h)
    return 1 - dx * np.min(corr)

def lse_c5(u, beta):
    """Smooth surrogate for C5 using log-sum-exp"""
    h = np.concatenate([u, u[::-1]])
    corr = autocorr_fft(h)
    return 1 - (dx / beta) * np.log(np.sum(np.exp(beta * corr)))

def grad_lse_c5(u, beta):
    """Gradient of the LSE surrogate"""
    h = np.concatenate([u, u[::-1]])
    corr = autocorr_fft(h)
    w = np.exp(beta * corr)
    w_sum = np.sum(w)
    
    # Convolution of w with reversed h
    conv = ifft(fft(w, 2*len(h)-1) * fft(h[::-1], 2*len(h)-1)).real
    
    grad = -dx * conv[:len(u)] / w_sum
    return grad

def parametric_to_halfvector(params, n):
    """Create half-vector from parametric family:
    params[0] = edge length a (0.05 to 0.45)
    params[1] = ramp width r (0 to 0.3)
    """
    a, r = params[0], params[1]
    p = (1 - 2*a) / (2 * (1 - a))
    
    u = np.zeros(n // 2)
    # Edge level
    u[:int(a * n / 2)] = 1.0
    # Ramp
    if r > 0:
        ramp_len = int(r * n / 2)
        ramp_indices = np.arange(int(a * n / 2), int((a + r) * n / 2))
        ramp_vals = np.linspace(1.0, p, len(ramp_indices))
        u[ramp_indices] = ramp_vals
    # Middle
    middle_start = int((a + r) * n / 2)
    middle_end = int((2 - a - r) * n / 2)
    if middle_end > middle_start:
        u[middle_start:middle_end] = p
    # Reverse ramp
    if r > 0:
        ramp_end = int((2 - a) * n / 2)
        ramp_start = int((2 - a - r) * n / 2)
        if ramp_end > ramp_start:
            ramp_indices = np.arange(ramp_start, ramp_end)
            ramp_vals = np.linspace(p, 1.0, len(ramp_indices))
            u[ramp_indices] = ramp_vals
    # Reverse edge
    u[-int(a * n / 2):] = 1.0

    # Ensure target sum
    target_sum = n / 4
    u = project_to_sum_box(u, target_sum)
    return u

def run_de():
    """Perform differential evolution on parametric space"""
    def objective(params):
        u = parametric_to_halfvector(params, n_points)
        return exact_c5(u)
    
    bounds = [(0.05, 0.45), (0, 0.3)]
    res = differential_evolution(
        objective, 
        bounds, 
        seed=42, 
        maxiter=200,
        popsize=15,
        disp=False
    )
    return parametric_to_halfvector(res.x, n_points)

def run_lbfgsb(u0, beta=200):
    """Run projected L-BFGS-B on full discretization"""
    def objective(u):
        return lse_c5(u, beta)
    
    def grad(u):
        return grad_lse_c5(u, beta)
    
    def project(u):
        return project_to_sum_box(u, n_points / 4)
    
    # Initial projection
    u = project(u0)
    
    # Minimize with bounds and projection
    result = minimize(
        objective,
        u,
        method='L-BFGS-B',
        jac=grad,
        bounds=[(0, 1)] * len(u),
        options={'gtol': 1e-10, 'maxiter': 800}
    )
    
    # Final projection
    u_final = project(result.x)
    return u_final

    # Start with high temperature
    T = 5e-3
    alpha = 0.9997
    
    # Track acceptance ratio
    accept_count = 0
    total_moves = 0
    target_accept_ratio = 0.3
    
    # Max delta for mass transfer
    delta_max = 0.05
    
    while time.time() - start_time < budget_s * 0.95:
        if T < 1e-6:
            break
            
        # Pick two indices
        i, j = np.random.choice(len(current_u), 2, replace=False)
        
        # Try to transfer mass
        delta = np.random.uniform(0, delta_max)
        if current_u[i] >= delta and current_u[j] <= 1 - delta:
            # Make move
            new_u = current_u.copy()
            new_u[i] -= delta
            new_u[j] += delta
            
            # Project
            new_u = project_to_sum_box(new_u, n_points / 4)
            
            # Compute new C5
            new_c5 = exact_c5(new_u)
            
            # Accept or reject
            if new_c5 < current_c5:
                current_u = new_u
                current_c5 = new_c5
                accept_count += 1
            else:
                if np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                    current_u = new_u
                    current_c5 = new_c5
                    accept_count += 1
            
            total_moves += 1
            
            # Adjust delta_max based on acceptance ratio
            if total_moves % 500 == 0:
                if total_moves > 0:
                    accept_ratio = accept_count / total_moves
                    if accept_ratio < target_accept_ratio - 0.05:
                        delta_max *= 0.9
                    elif accept_ratio > target_accept_ratio + 0.05:
                        delta_max *= 1.1
                    accept_count = 0
                    total_moves = 0
        
        T *= alpha
        
    return current_u
