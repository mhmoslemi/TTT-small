import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp
from scipy.fft import ifft, fft

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_binary_milp(N=60):
    """Solve small binary MILP to get combinatorial skeleton"""
    x = Variable(N, boolean=True)
    t = Variable(pos=True)
    
    # Constraints: sum = N/2
    constraints = [sum(x) == N/2]
    
    # Add auxiliary variables and constraints for correlation
    for k in range(1, N):
        # Create auxiliary variables for y[i,k] = x[i]*x[i+k] (where i+k < N)
        for i in range(N-k):
            y = Variable(boolean=True)
            constraints.extend([
                y <= x[i],
                y <= x[i+k],
                y >= x[i] + x[i+k] - 1
            ])
            constraints.append(sum([y for i in range(N-k)]) >= N/2 - t)
    
    # Objective to minimize worst-case penalty
    objective = Minimize(t)
    prob = Problem(objective, constraints)
    
    # Try to solve with GLPK_MI
    try:
        prob.solve(solver=GLPK_MI, verbose=False, max_iters=5000)
        if prob.status in [OPTIMAL, FEASIBLE]:
            return np.array([float(v.value) for v in x]).round().astype(int)
    except:
        pass
    # Fallback to random
    return np.random.choice([0,1], size=N, p=[0.5,0.5])

def upsample_binary(x, s):
    """Upsample binary vector by integer factor s"""
    h = np.repeat(x, s)
    return h

def local_swap_search(h, max_iter=10000):
    """Worst-shift focused swap search"""
    h = h.copy().astype(int)
    n = len(h)
    # Precompute correlations (full auto-correlation)
    corr = compute_corr_fft(h)
    worst_k = np.argmin(corr)
    
    for i in range(max_iter):
        # Check if we're approaching target
        if corr[worst_k] > n/2 - 0.01:
            break
            
        # Find A and B sets
        A = []
        B = []
        for j in range(n - worst_k):
            if h[j] == 1 and h[j + worst_k] == 0:
                A.append(j)
            if h[j] == 0 and h[j + worst_k] == 1:
                B.append(j)
        
        if A and B:
            # Choose random pair with decreasing probability
            ai = np.random.choice(A)
            bi = np.random.choice(B)
            # Perform swap
            h[ai] = 0
            h[bi] = 1
            
            # Update correlation incrementally
            corr = update_corr_fft(h, corr, ai, bi, worst_k)
            worst_k = np.argmin(corr)
        else:
            # Accept unlikely swap for escape
            if np.random.rand() < 0.01:
                idx1 = np.random.randint(0, n)
                idx2 = np.random.randint(0, n)
                if ((h[idx1] == 1 and h[idx2] == 0) or (h[idx1] == 0 and h[idx2] == 1)):
                    h[idx1] = 1 - h[idx1]
                    h[idx2] = 1 - h[idx2]
                
                # Re-compute correlation
                corr = compute_corr_fft(h)
                worst_k = np.argmin(corr)

    return h

def compute_corr_fft(h):
    """Compute auto-correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def update_corr_fft(h, corr_old, i, j, k):
    """Update correlation efficiently after one bit swap"""
    diff = 0.0
    if i < len(h) - k and j < len(h) - k:
        if h[i] == 1 and h[j] == 0:
            # Remove overlap for bit at i+k
            if i+k < len(h) and h[i+k] == 1:
                diff += 1
            # Add overlap for bit at j+k
            if j+k < len(h) and h[j+k] == 1:
                diff -= 1
        elif h[i] == 0 and h[j] == 1:
            # Remove overlap for bit at j+k
            if j+k < len(h) and h[j+k] == 1:
                diff -= 1
            # Add overlap for bit at i+k
            if i+k < len(h) and h[i+k] == 1:
                diff += 1
    
    # Approximate change in correlation counts
    corr_new = corr_old.copy()
    # Correct worst shift
    if (i+k < len(h)):
        corr_new[k] -= diff
    if (j+k < len(h)):
        corr_new[k] += diff
    return corr_new

def de_optimize(h0, n_points, max_iter=2000):
    """Differential evolution optimization"""
    # DE parameters
    npop = 30
    F = 0.8
    CR = 0.9
    
    # Initialize population
    population = []
    for _ in range(npop):
        h = h0.copy()
        # Add small noise to avoid stagnation
        noise = np.random.normal(0, 0.02, n_points)
        h += noise
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    
    # DE main loop
    for i in range(max_iter):
        for j in range(npop):
            # Select three distinct indices
            idxs = list(range(npop))
            idxs.remove(j)
            a, b, c = np.random.choice(idxs, 3, replace=False)
            
            # Mutation
            mutant = population[a] + F * (population[b] - population[c])
            
            # Crossover
            trial = np.copy(population[j])
            for k in range(n_points):
                if np.random.random() < CR or k == np.random.randint(n_points):
                    trial[k] = mutant[k]
            
            # Project and evaluate
            trial = project_to_sum_box(trial, TARGET_SUM)
            c5_old = evaluate_c5(population[j])
            c5_new = evaluate_c5(trial)
            
            # Selection
            if c5_new <= c5_old:
                population[j] = trial
            
            # Check early termination

                break
                
        if time.time() - start_time > budget_s - 5:
            break
                
    # Return best
    best_h = min(population, key=evaluate_c5)
    return best_h

def hill_climb(h0, max_iter=2000):
    """Simple hill climbing"""
    h_current = h0.copy()
    c5_current = evaluate_c5(h_current)
    best_h = h_current.copy()
    best_c5 = c5_current
    
    for i in range(max_iter):
        # Perturb a subset of points
        n_perturb = max(1, int(n_points * 0.05))  # 5% of indices
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        
        h_new = h_current.copy()
        for idx in indices:
            # Add small Gaussian noise with adaptive scaling
            sigma = min(0.1, 0.5 * h_current[idx] * (1 - h_current[idx]))
            delta = np.random.normal(0, sigma)
            h_new[idx] += delta
            if h_new[idx] < 0:
                h_new[idx] = 0
            elif h_new[idx] > 1:
                h_new[idx] = 1
            
        # Project
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        c5_new = evaluate_c5(h_new)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        else:
            # Occasionally accept worse moves to escape local minima
            if np.random.rand() < 0.01:
                h_current = h_new
                c5_current = c5_new
                
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx, start_time
    np.random.seed(seed)
    start_time = time.time()
    
    # Reserve time for final processing
    budget_s -= 5
    
    # Initialize
    TARGET_SUM = n_points / 2
    dx = 2.0 / n_points
    
    # Phase A: Binary MILP
    h_initial = solve_binary_milp(N=60)
    
    # Phase B: Upsample
    h_upsampled = upsample_binary(h_initial, n_points // len(h_initial))
    
    # Phase C: Local swap search (if binary form has large correlation)
    if time.time() - start_time < budget_s * 0.8:
        h_swap = local_swap_search(h_upsampled.copy(), max_iter=5000)
    else:
        h_swap = h_upsampled
    
    # Phase D: Differential evolution
    if time.time() - start_time < budget_s * 0.9:
        h_de = de_optimize(h_swap, n_points, max_iter=1000)
    else:
        h_de = h_swap
    
    # Phase E: Hill climbing
    h_final, best_c5 = hill_climb(h_de, max_iter=1000)
    
    # Final clean up: re-project to sum constraint
    h_final = project_to_sum_box(h_final, TARGET_SUM)
    final_c5 = evaluate_c5(h_final)
    
    return (h_final.tolist(), float(final_c5), n_points)
