import numpy as np
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points
MAX_ITER_ANNEAL = 300000
SCHEDULE_STEPS = 300000
SAMPLES_TO_RESTART = 20000
RESTART_THRESHOLD = 1e-5
LAMBDA_SEARCH_ITER = 50

def compute_correlations(x):
    """Compute full correlation of binary vector x with itself"""
    n = len(x)
    padded_x = np.pad(x, (n-1, n-1), mode='constant', constant_values=0)
    A_d = np.correlate(padded_x, padded_x, mode='full')
    A_d = A_d[n-1:3*n-2]  # Extract the correlation from both sides
    return A_d

def compute_overlap(window_sums, A_d):
    """Compute full overlap for a binary vector given window sums and correlation"""
    n = len(window_sums)
    # For shift d, overlap = sum of x_i * x_{i+d} for all valid i
    # overlap_{-d} = window_sum_{-d} - A_{-d}
    # overlap_{d} = window_sum_{d} - A_{d}
    # We compute overlap for all shifts d in [-n+1,n-1]
    # But actually, we must extract F_d based on full correlation
    # which is computed such that for shift d, the window sum is from 0 to n-1-d
    # For positive d, d's overlap is sum of x_i * x_{i+d} that are both in [0, n-1-d]

    # Compute window sums for all shifts d
    overlaps = np.zeros(2*n - 1)
    for d in range(-(n-1), n):
        # d from -n+1 to n-1
        idx = d + (n-1)
        # My binary vector is a 1-dimensional array x[0], x[1], ..., x[n-1]
        # For shift d, valid positions are [0, n-1-d] (where d>=0)
        # Wait, let me recompute correctly
        # if d >= 0, then x_i * x_{i+d} for i in [0, n-1-d]
        # if d < 0, then x_i * x_{i+d} for i in [n-1+d, n-1]
        # Purely a 1D indexing problem
        if d >= 0:
            window_len = n - d
            # Compute sum of x_i * x_{i+d} over i in 0, 1, ..., n-1-d
            overlap_d = window_sums[d] - A_d[n-1 + d]
        else:
            # d < 0, so shift of d (negative) means we shift to the left
            window_len = n + d
            i_start = 0
            j_start = -d # = |d|
            # x_i * x_{i+d} = x_i * x_{j}: i from 0 to window_len-1, j from d to d + window_len-1
            # i from 0 to window_len-1, j from |d| to |d| + window_len-1
            overlap_d = window_sums[-d] - A_d[n-1 + d] # Part of A that has coefficient dependence on x_{i+d}
        
        overlaps[idx] = overlap_d
    
    return overlaps

def compute_overlap_window_sums(x):
    """Compute window sums using prefix sum technique"""
    n = len(x)
    prefix = np.zeros(n+1)
    prefix[1:] = np.cumsum(x)
    # For shift d, window is [0, n-1-d], for d>=0
    # For shift d, window is [0+d, n-1] if d<0 => [i_start, i_start-n+1], i_start=-d
    window_sums = np.zeros(2*n - 1)
    
    # loop over shifts
    for d in range(-(n-1), n):
        idx = d + (n-1)
        if d >= 0:
            window_len = n - d
            s = prefix[window_len] - prefix[0]  # Sum over i from 0 to window_len-1
            window_sums[idx] = s
        else:
            # d < 0
            i_start = -d  # = |d|
            window_len = n + d
            s = prefix[n] - prefix[i_start]  # Sum over j in [d, n-1] which becomes [j_start, n-1]
            window_sums[idx] = s
    return window_sums

def compute_c5_binary(x):
    """Compute C5 from binary vector x"""
    n = len(x)
    # Prefix sum
    prefix = np.zeros(n+1)
    prefix[1:] = np.cumsum(x)
    
    # Window sums
    window_sums = np.zeros(2*n - 1)
    for d in range(-(n-1), n):
        idx = d + (n-1)
        if d >= 0:
            window_len = n - d
            window_sums[idx] = prefix[window_len] - prefix[0]
        else:
            i_start = -d
            window_len = n + d
            window_sums[idx] = prefix[n] - prefix[i_start]
    
    # Now compute auto correlation
    A_d = compute_correlations(x)
    
    # now compute overlaps 
    overlaps = window_sums - A_d # at all shifts d in [-n+1,n-1]
    c5 = dx * np.max(overlaps)
    return c5

def project_onto_feasible(y):
    """Project a real vector y onto the feasible set {h: 0 <= h <= 1, sum(h)=n/2}"""
    n = len(y)
    target_sum = n / 2
    
    # Binary search for lambda
    def f(lam):
        projected = np.clip(y + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    for _ in range(LAMBDA_SEARCH_ITER):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(y + (a + b)/2, 0, 1)
    return result

def compute_c5_continuous(h):
    """Compute C5 using correlation"""
    c = np.correlate(h, 1-h, mode="full")
    return dx * np.max(c)

def get_worst_shift_candidates(x, A_d, window_sums):
    """Get the worst shift from the current state"""
    n = len(x)
    overlaps = window_sums - A_d
    worst_ind = np.argmax(overlaps)
    worst_shift = worst_ind - (n - 1)
    return worst_shift

def binary_to_float(x):
    """Convert binary vector to float vector, preserving sum"""
    return x.astype(float)

def take_targeted_swap(x, d_star):
    """Perform a targeted swap to reduce the worst shift"""
    # Find indices of exposed 1s and missing 0s
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    n = len(x)
    
    # The sets U and V
    U = []  # indices where x[i]=1 and x[i+d_star]=0
    V = []  # indices where x[j]=0 and x[j+d_star]=1
    
    for i in ones:
        j = i + d_star
        if 0 <= j < n:
            if x[j] == 0:
                U.append(i)
    
    for j in zeros:
        i = j - d_star
        if 0 <= i < n:
            if x[i] == 1:
                V.append(j)
    
    if U and V:
        i = np.random.choice(U)
        j = np.random.choice(V)
        x[i], x[j] = x[j], x[i]
        return True
    else:
        # if cannot do targeted, do random swap
        return False

def take_random_swap(x):
    """Perform a random swap"""
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    
    if len(ones) > 0 and len(zeros) > 0:
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        x[i], x[j] = x[j], x[i]
        return True
    else:
        return False

    if current_sum > n/2:

        ones_indices = np.where(x0 == 1)[0]
        flip_indices = np.random.choice(ones_indices, size=needed_flips, replace=False)
        x0[flip_indices] = 0
    elif current_sum < n/2:
        # Need to flip needed_flips zeros to ones
        zeros_indices = np.where(x0 == 0)[0]
        flip_indices = np.random.choice(zeros_indices, size=needed_flips, replace=False)
        x0[flip_indices] = 1
    
    # Initial best
    best_x = x0.copy()
    best_c5 = compute_c5_binary(best_x)
    
    # Initialize real-valued version
    best_h = binary_to_float(best_x)
    
    # Thermal annealing phase
    temp_initial = 2 * dx
    temp_final = 1e-6 * dx
    temp_schedule = np.linspace(temp_initial, temp_final, SCHEDULE_STEPS)
    current_x = best_x.copy()
    current_h = best_h.copy()
    current_c5 = best_c5
    
    accepted_count = 0
    last_improvement = 0
    restart_count = 0
    
    for iteration in range(MAX_ITER_ANNEAL):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5:
            break
        
        T = temp_schedule[min(iteration, len(temp_schedule) - 1)]
        do_targeted = np.random.rand() < 0.6
        
        if do_targeted:
            # Find worst shift
            A_d = compute_correlations(current_x)
            window_sums = compute_overlap_window_sums(current_x)
            d_star = get_worst_shift_candidates(current_x, A_d, window_sums)
            
            # perform targeted swap
            changed = take_targeted_swap(current_x, d_star)
            if not changed: # fallback to random
                take_random_swap(current_x)
        else:
            # Random swap
            take_random_swap(current_x)
        
        # Must recompute due to significant changes
        new_c5 = compute_c5_binary(current_x)
        
        # Accept if better
        if new_c5 < current_c5:
            current_c5 = new_c5
            current_x = current_x.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_x = current_x.copy()
                last_improvement = iteration
        else:
            # Maybe accept with probability exp(-Δ/T)
            delta = new_c5 - current_c5
            if np.random.rand() < np.exp(-delta / T):
                current_c5 = new_c5
                current_x = current_x.copy()
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_x = current_x.copy()
                    last_improvement = iteration
            else:
                # revert the swap
                pass
        
        # Stagnation check and restart
        if iteration % SAMPLES_TO_RESTART == 0:
            if iteration > 0 and abs(best_c5 - last_improvement) < RESTART_THRESHOLD:
                # Inject diversity
                indices_to_flip = np.random.choice(n, size=int(0.05*n), replace=False)
                for i in indices_to_flip:
                    current_x[i] = 1 - current_x[i]
                
                # restore exact sum
                current_sum = np.sum(current_x)
                needed = int(abs(current_sum - n/2))
                if current_sum > n/2:
                    flip_indices = np.random.choice(np.where(current_x == 1)[0], size=needed, replace=False)
                    current_x[flip_indices] = 0
                elif current_sum < n/2:
                    flip_indices = np.random.choice(np.where(current_x == 0)[0], size=needed, replace=False)
                    current_x[flip_indices] = 1
                
                # record next restart
                restart_count += 1
                # reduce temperature
                T = max(T * 0.5, temp_final * 10)
                
        accepted_count += 1
    
    # Convert best solution to floating point
    h0_float = binary_to_float(best_x)
    h_refined = h0_float.copy()
    
    # Continuous refinement with SLSQP
    # Objective function for continuous problem
    def obj_func(h):
        return compute_c5_continuous(h)
    
    bounds = [(0, 1) for _ in range(n)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
    
    res = minimize(obj_func, h_refined, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 1500, 'ftol': 1e-12, 'gtol': 1e-12})
    
    if res.success:
        h_refined = res.x
        h_refined = project_onto_feasible(h_refined)
    
    # Optionally apply smoothing
    averaged_h = np.convolve(h_refined, np.ones(3)/3, mode='same')
    averaged_h = project_onto_feasible(averaged_h)
    
    refineres2 = minimize(obj_func, averaged_h, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-12})
    
    if refineres2.success:
        final_h = refineres2.x
        final_h = project_onto_feasible(final_h)
    else:
        final_h = averaged_h
    
    final_c5 = compute_c5_continuous(final_h)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h.copy()
    
    # Final check using exact routine
    final_c5 = compute_c5_continuous(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
