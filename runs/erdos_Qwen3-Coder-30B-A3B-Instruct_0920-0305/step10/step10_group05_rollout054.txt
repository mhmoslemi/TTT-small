import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def autocorr_fft(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    b = h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5_exact(h):
    """Compute exact C5 using max-over-k"""
    corr = autocorr_fft(h)
    return 1 - np.min(corr)

def evaluate_c5_smooth(h, beta=50.0):
    """Compute smooth approximation to C5 using log-sum-exp"""
    corr = autocorr_fft(h)
    max_corr = np.max(corr)
    # Center for numerical stability
    corr_centered = corr - max_corr
    exp_corr = np.exp(beta * corr_centered)
    log_sum_exp = np.log(np.sum(exp_corr)) - np.log(len(h))
    return 1 - (1.0 / beta) * log_sum_exp

def smooth_gradient(h, beta=50.0):
    """Compute gradient of smooth C5"""
    corr = autocorr_fft(h)
    max_corr = np.max(corr)
    corr_centered = corr - max_corr
    exp_corr = np.exp(beta * corr_centered)
    Z = np.sum(exp_corr)
    # Gradient using convolution trick
    # d/dh_i (log sum exp) = (sum of exp * gradient) / Z
    # For log-sum-exp, gradient = exp * (1/Z) * d/dh_i (sum of terms)
    # Which corresponds to derivative of autocorr term with respect to h_i
    # The derivative of A_k = h_i * h_{i+k} into h_i is h_{i+k}
    # But we need more careful calculation:
    # But we are integrating [h(x)(1-h(x+k))]
    # So the derivative with respect to h_i is (1 - h_i - h_{i+k} - h_{i-k} for
    # k<0 and k>0; here the factor is h_{i+k} in the case of the autocorr
    # But let's do it via observed algebra...
    # Derivative is just convolution of exp\corr with appropriate sign:
    # Let W = exp(beta * (corr - max_corr)), all terms bounded by 1 -> exp(0) = 1
    # Let w = W / Z -> sum(w) = 1
    # G_i = sum_k w_k * (h_{k+i} + h_{k-i}) 
    # So gradient should be sum_k w_k * (h_{k+i} + h_{k-i})  but we are looking at
    # \int h(x)(1 - h(x+k)) = \sum_i h_i (1 - h_{i+k})
    # so the gradient w.r.t. h_i is:
    # (1 - h_{i+k} - h_{i-k}) * (the incentive term w_k * ...)
    # rather than 
    # If we denote A(i) = \sum_k h_k h_{k+i} 
    # Then d A(i) / d h_j = h_{j+i} + h_{j-i}
    # Therefore, \delta C5 / \delta h_i = -dx \sum_k w_k * (h_{i+k} + h_{i-k})
    # Note that we have a sign issue in the original objective: 
    # we are minimizing max_k \int h(x) (1 - h(x+k))
    # So, let us take diff of 1 - \int h(x) (1 - h(x+k)) = \int h(x) h(x+k) - 1
    # So we want gradient of \int h(x) h(x+k) (not 1-h)
    # That's the same: d/dh_i \sum_k h_i h_{i+k} = h_{k+i} + h_{k-i} where k is index of shift
    # And then the objective is just 1 - \min_k (\int h(x) h(x+k)) 
    # so gradient = -dx * sum_k w_k * (h_{i+k} + h_{i-k})
    # But we need to compute the convolution of w with h in both directions
    
    # To compute derivative of sum_k w_k * h_{i+k} 
    # does that have the correct model?
    # Let's compute: w * h as a convolution and w * h_REV 
    # For w_k * (h_{j+i} + h_{j-i}) -> P_k is indicator 1 if k==(i+j) else 0, 
    # so that would be sum_k w_k * (h_{k} + h_{k-2*(j-i)}) 
    # etc. Let's be more careful...
    # Let us compute: sum_k w_k * h_{k+i}  
    # That convolution is FFT (w, L) * FFT (h^[i], L)
    # So that's like FFT(w) * FFT(h)(k) * exp(i*2pi*k*i/L) -> i*k
    # So far may be okay...
    # The correct interpretation: 
    # Let W = [w_(-n+1), ..., w_0, ..., w_(n-1)]
    # and H = [h_0,..., h_{n-1}] we want convolution of W and H (circular if needed)
    # For k=0, account for the term w_k * h_{i+k}, 
    # that is sum_k w_k h_{i+k}  
    # If i=0
    # Let's make a direct computation
    
    # Let us simplify this and write it in convolution form
    # we compute \int h(x) h(x+k) which leads to \sum_i h_i h_{i+k}
    # Therefore to get gradient of (1 - \min) wrt h_i, consider  
    # -1 * d/dh_i \sum_k w_k \sum_j h_j h_{j+k} = -1 * sum_k w_k * (h_{k+i} + h_{i-k})
    
    # Take a simpler path:
    # We can write: h * h,
    # Then: (h * h)[j] = sum_k h_k h_{j+k} for every j  
    # Then the derivative of this wrt h_i:
    # d/dh_i (h_k h_{k+j}) = 0 if i != j and k, 1 if i=k and therefore h_k, 1 if i=k+j and h_k etc  
    # So we drop the case where k+j = i 
    # Derivative = h_{k+i} + h_{k-i}  
    # But we want to sum over k and w_k  
    # let F = \sum_k w_k h_{k+i} = conv(w, h)[i]  
    # let G = \sum_k w_k h_{k-i} = conv(w_rev, h)[i]  
    # yield gradient_i = -dx (F[i] + G[i])
    # But make sure to pad appropriately
    
    # without loss of generality, use 'full' convolution which will create 2n-1 elements
    conv_f = np.convolve(exp_corr, h, mode='full')[:len(h)]
    conv_g = np.convolve(exp_corr[::-1], h, mode='full')[:len(h)]
    
    grad = -np.ones_like(h) * 1.0  # placeholder
    # ensure same length as h, not assuming length is L
    grad = -(conv_f + conv_g) / Z * 1.0
    return grad

def project_to_sum_box(h, target_sum):
    """Project vector h onto sum = target_sum and box [0,1]"""
    # Simple projection approach
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # linearly scale to sum to target_sum
    scale = target_sum / current_sum
    h = h * scale
    return np.clip(h, 0, 1)

def parametric_h(a):
    """Construct symmetric shape: two edge blocks of height 1, central plateau of height p"""
    n = 200
    m = n // 2
    h_half = np.zeros(m)
    # area: 2 * a * 1 + (2-2*a) * p = 1 => p = (1-2*a)/(2*(1-a))
    if a >= 0.5 or a <= 0:
        print(f"Warning: a={a} out of allowed range [0.01, 0.49]")
        return h_half
    p = (1.0 - 2.0*a) / (2.0 * (1.0 - a))
    left = int(np.floor(a * m))
    right = int(np.ceil(a * m))
    
    if a >= 0.5:
        # if too wide, ensure not > 1
        h_half[:] = 1.0
        # but this is not valid for the rest, we truncate later.
        # simplify handling
        h_half = np.ones(m)
        h_half[left:right] = p
        return h_half
    else:
        h_half[:] = 0.0
        h_half[:left] = 1.0
        h_half[left:right] = p
        h_half[right:] = 1.0
        return h_half

def parametric_sweep():
    """Coarse sweep over a, check C5"""
    n = 200
    m = n // 2
    target_sum = n / 4.0  # sum of half vector
    best_c5 = 1e10
    best_a = 0
    start = time.time()
    for a in np.arange(0.01, 0.49, 0.002):
        if time.time() - start > 500:
            break
        h = parametric_h(a)
        h = project_to_sum_box(h, target_sum)
        
        # Convert to full for FFT
        h_full = np.concatenate([h, h[::-1]])
        c5 = evaluate_c5_exact(h_full)
        if c5 < best_c5:
            best_c5 = c5
            best_a = a
    return best_a, best_c5

def optimize_low_dimensional():
    """Use bounded L-BFGS with smooth surrogate on 1D parametric"""
    n = 200
    m = n // 2
    target_sum = n / 4.0

    def objective(a):
        # Ensure a is valid
        if a <= 0 or a >= 0.5:
            return 1
        # Construct h_half from a
        h_half = parametric_h(a)
        h_half = project_to_sum_box(h_half, target_sum)
        h_full = np.concatenate([h_half, h_half[::-1]])
        return evaluate_c5_smooth(h_full, beta=50.0)

    # minimize smooth objective with upper bound
    # out of range returns large value
    res = minimize(objective, x0=0.2, bounds=[(0.001, 0.49)], method='L-BFGS-B',
                   options={'ftol': 1e-10, 'gtol': 1e-10})
    best_a = res.x[0]
    h_full = np.concatenate([parametric_h(best_a), parametric_h(best_a)[::-1]])
    h_full = project_to_sum_box(h_full, 1.0)
    final_c5 = evaluate_c5_exact(h_full)
    return h_full, final_c5

def simulated_annealing(h0_half, n_iters=150000):
    """Apply simulated annealing"""
    # h0 is first half, full vector built as h + h_rev
    n = 200
    target_sum = n / 4.0  # sum constraint for half vector
    dt = 1.0 / n  # pixel width
    
    # Initialize full from half
    h_full = np.concatenate([h0_half, h0_half[::-1]])
    current_c5 = evaluate_c5_exact(h_full)
    best_h = h_full.copy()
    best_c5 = current_c5
    T = 1e-2
    T_min = 1e-6
    alpha = 0.9995
    
    accept_count = 0
    total_moves = 0
    delta_max = 0.05
    
    start = time.time()
    for iter in range(n_iters):
        if time.time() - start > 550:
            break
            
        # Pick two random indices
        i, j = np.random.randint(0, len(h0_half), size=2)
        if i == j:
            continue
        delta = np.random.uniform(0, delta_max)
        
        # Test move
        h_new_half = h0_half.copy()
        if h_new_half[i] >= delta and h_new_half[j] <= 1 - delta:
            h_new_half[i] -= delta
            h_new_half[j] += delta
            h_new_full = np.concatenate([h_new_half, h_new_half[::-1]])
            h_new_full = project_to_sum_box(h_new_full, 1.0) # full sum constraint
            new_c5 = evaluate_c5_exact(h_new_full)
            if new_c5 < current_c5:
                h0_half = h_new_half
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = h_new_full.copy()
                    best_c5 = current_c5
                # accept always when improving
                accept_count += 1
                total_moves += 1
            else:
                # Metropolis acceptance
                if np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                    h0_half = h_new_half
                    current_c5 = new_c5
                    accept_count += 1
                total_moves += 1
        else:
            total_moves += 1
        # Adapt step size
        if iter > 0 and iter % 500 == 0:
            if accept_count > 0.4 * 500:
                delta_max = min(delta_max * 1.1, 0.2)
            elif accept_count < 0.1 * 500:
                delta_max = max(delta_max * 0.9, 1e-4)
            accept_count = 0
        # Cool down
        T = max(T * alpha, T_min)
    
    return best_h, best_c5
