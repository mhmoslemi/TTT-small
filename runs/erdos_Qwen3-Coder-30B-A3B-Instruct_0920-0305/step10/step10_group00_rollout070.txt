import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation using FFT"""
    L = 2 * len(h) - 1
    H_rev = np.fft.rfft(h[::-1], L)
    B = np.fft.rfft(1 - h, L)
    corr = np.fft.irfft(H_rev * B, L)
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def upscale_h(original_h, n_new):
    """Upscale original_h to n_new points using linear interpolation"""
    n_old = len(original_h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, original_h)

def propose_move(h, max_step, rng):
    """Propose a move that maintains sum constraint and bounds"""
    n = len(h)
    # try up to few times to find feasible pair
    for _ in range(10):
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # compute feasible interval for delta added to i and subtracted from j
        lower = max(-h[i], h[j] - 1)
        upper = min(1 - h[i], h[j])
        if lower < upper:
            # limit delta magnitude
            lower = max(lower, -max_step)
            upper = min(upper, max_step)
            if lower >= upper:
                continue
            delta = rng.uniform(lower, upper)
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            return new_h
        # try opposite direction: add to j, subtract from i
        lower = max(-h[j], h[i] - 1)
        upper = min(1 - h[j], h[i])
        if lower < upper:
            lower = max(lower, -max_step)
            upper = min(upper, max_step)
            if lower >= upper:
                continue
            delta = rng.uniform(lower, upper)
            new_h = h.copy()
            new_h[j] += delta
            new_h[i] -= delta
            return new_h
    return None

def run(seed=42, budget_s=600, **kwargs):
    global n_points, dx, TARGET_SUM
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize candidates
    candidates = []
    
    # From provided initial solution
    try:
        global initial_h_values
        upsampled = upscale_h(initial_h_values, n_points)
        upsampled = project_to_sum_box(upsampled, TARGET_SUM)
        candidates.append(upsampled)
    except:
        pass
    
    # Cosine shape
    xs = np.arange(n_points) * dx
    a = 0.5 - 1e-5
    cos_h = 0.5 + a * np.cos(np.pi * xs / 2)
    cos_h = project_to_sum_box(cos_h, TARGET_SUM)
    candidates.append(cos_h)
    
    # Random uniform
    rand_h = np.random.uniform(0, 1, n_points)
    rand_h = project_to_sum_box(rand_h, TARGET_SUM)
    candidates.append(rand_h)
    
    # Binary random with exact half
    binary_h = np.zeros(n_points)
    ones_idx = np.random.choice(n_points, size=n_points//2, replace=False)
    binary_h[ones_idx] = 1.0
    candidates.append(binary_h)
    
    # Evaluate all candidates
    best_val = float('inf')
    best_h = None
    
    for candidate in candidates:
        val = evaluate_c5(candidate)
        if val < best_val:
            best_val = val
            best_h = candidate.copy()
    
    # Simulated Annealing
    curr_h = best_h.copy()
    curr_val = best_val
    best_h_local = curr_h.copy()
    best_val_local = curr_val
    
    rng = np.random.default_rng(seed)
    total_allowed = budget_s * 0.9
    max_step_initial = 0.2
    max_step_final = 0.001
    
    while time.time() - start_time < total_allowed:
        elapsed = time.time() - start_time
        fraction = elapsed / total_allowed
        T = 0.02 * (1 - fraction) + 0.0001 * fraction
        max_step = max_step_initial * (1 - fraction) + max_step_final
        
        new_h = propose_move(curr_h, max_step, rng)
        if new_h is None:
            continue
            
        new_val = evaluate_c5(new_h)
        
        if new_val < curr_val or rng.random() < np.exp(-(new_val - curr_val) / (T + 1e-12)):
            curr_h = new_h
            curr_val = new_val
            
            if new_val < best_val_local:
                best_val_local = new_val
                best_h_local = curr_h.copy()
    
    # Final local refinement
    curr_h = best_h_local.copy()
    curr_val = best_val_local
    max_step_refine = 0.005
    
    refine_iters = 50000
    for i in range(refine_iters):
        if time.time() - start_time > budget_s * 0.99:
            break
            
        elapsed = i / refine_iters
        max_step = max_step_refine * (1 - elapsed) + 1e-6
        
        new_h = propose_move(curr_h, max_step, rng)
        if new_h is None:
            continue
            
        new_val = evaluate_c5(new_h)
        
        if new_val < curr_val:
            curr_h = new_h
            curr_val = new_val
            
            if new_val < best_val_local:
                best_val_local = new_val
                best_h_local = curr_h.copy()
    
    return (best_h_local.tolist(), float(best_val_local), n_points)
