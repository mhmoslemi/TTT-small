import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 240  # Increased grid size for higher resolution
dx = 2.0 / n_points

def compute_overlap_from_binary(x):
    """Given a binary vector x, compute the correlation overlap.
    Returns the maximum overlap C5."""
    # Compute the auto-correlation
    x_pad = np.pad(x, (n_points-1, n_points-1), mode='constant', constant_values=0)
    A = correlate(x, x, mode='full')
    # Compute overlaps for all shifts
    # Total sum of ones
    ones_count = np.sum(x)
    # Compute prefix sum for fast window sums
    prefix = np.concatenate([[0], np.cumsum(x)])
    # Compute all overlap values (centered at 0, excludes the automatic padding part)
    max_overlap = 0.0
    for d in range(-(n_points-1), n_points):
        if d >= 0:
            # Left shift: x_i * x_{i+d} 
            # Slot for the correlation term
            start = n_points-1 + d
            end = n_points-1 + n_points - d
            window_ones = prefix[end] - prefix[start]
            # Actual overlap: (ones_count - A[d])
            total_ones = ones_count
            overlap = total_ones - A[n_points-1+d]
        else:
            # Right shift
            start = n_points-1 + d
            end = n_points-1 + n_points 
            window_ones = prefix[end] - prefix[start]
            # Compute as before but make index correctly
            overlap = ones_count - A[n_points-1+d]
        # Store if maximum
        if overlap > max_overlap:
            max_overlap = overlap
    return dx * max_overlap

def compute_overlap_from_binary_fast(x):
    """Fast computation using prefix sums over all shifts"""
    n = len(x)
    # Prefix sum to compute window sums
    prefix = np.concatenate([[0], np.cumsum(x)])
    # Total number of ones
    ones_count = np.sum(x)
    # Full auto-correlation
    A = correlate(x, x, mode='full')
    # Compute overlaps and store the max
    max_overlap = 0.0
    total_elements = 2 * n - 1
    for d in range(-(n - 1), n):
        if d >= 0:
            # x[i] * x[i + d]
            start = d
            end = n - d
            window_ones = prefix[end] - prefix[start]  # This is simply the number of 1s in the window
        else:
            # x[i] * x[i - d]
            start = -d
            end = n + d
            window_ones = prefix[end] - prefix[start]
        # Overlap = ones_count - A[d]  
        if d >= 0:
            a = A[n - 1 + d]
        else:
            a = A[n - 1 + d]
        overlap = ones_count - a
        if overlap > max_overlap:
            max_overlap = overlap
    return dx * max_overlap

def compute_shifted_window_sums(x):
    """Efficiently compute the shifted window sums (sum of 1-jumps)"""
    prefix = np.concatenate([[0], np.cumsum(x)])
    n = len(x)
    sums = []
    for d in range(-(n-1), n):
        if d >= 0:
            start = d
            end = n - d
        else:
            # Right shift
            start = -d
            end = n + d
        sums.append(prefix[end] - prefix[start])
    return np.array(sums)

def compute_change_in_auto_correlation(x, i, j):
    """Given a binary vector x, g1ive the change in A[d] if we swap positions i and j"""
    n = len(x)
    # Find the change in shift values involving i and j
    # The change in dB[k] = - (x[i+k] + x[i-k]) + (x[j+k] + x[j-k])
    delta_A = np.zeros(2 * n - 1)
    # For each shift from -n+1 to n-1, compute change
    for d in range(-(n-1), n):
        # x[i+d] and x[i-d], vise versa for j
        left_i = i - d
        right_i = i + d
        left_j = j - d
        right_j = j + d
        # Bounds checking
        val_i = x[left_i] if 0 <= left_i < n else 0
        val_i_r = x[right_i] if 0 <= right_i < n else 0
        val_j = x[left_j] if 0 <= left_j < n else 0
        val_j_r = x[right_j] if 0 <= right_j < n else 0
        delta_A[d+n-1] = -(val_i + val_i_r) + (val_j + val_j_r)
    return delta_A

def get_worst_shift_indices(x):
    """Get the indices that form U V sets for worst shift"""
    A = correlate(x, x, mode='full')
    n = len(x)
    prefix = np.concatenate([[0], np.cumsum(x)])
    ones_count = np.sum(x)
    # Find the shift that maximizes overlap = ones_count - A[d]
    max_overlap = 0.0
    best_d = 0
    # Compute for all shifts
    for d in range(-(n-1), n):
        if d >= 0:
            start = d
            end = n - d
        else:
            start = -d
            end = n + d
        window_ones = prefix[end] - prefix[start]
        overlap = ones_count - A[n-1+d]
        if overlap > max_overlap:
            max_overlap = overlap
            best_d = d
    # Compute U and V sets
    U = []
    V = []
    for i in range(n):
        j = i + best_d
        if 0 <= j < n:
            if x[i] == 1 and x[j] == 0:
                U.append(i)
            elif x[i] == 0 and x[j] == 1:
                V.append(i)
    return best_d, U, V

def project_vector_to_sum_box(x, target_sum):
    """Project vector x onto [0,1]^n such that sum=x"""
    # This is a simple method for large numbers of 1s
    # We binary search on lambda to project
    def compute_sum_with_lambda(lam):
        proj = np.clip(x + lam, 0, 1)
        return np.sum(proj)
    def f(lam):
        return compute_sum_with_lambda(lam) - target_sum
    # Binary search for lambda
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    return np.clip(x + (a + b)/2, 0, 1)


def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialization 
    TARGET_SUM = n_points / 2
    total_steps = 300000
    n_accept = 0
    T0 = 2*dx
    Tf = 1e-6*dx
    # Start with a binary vector set correctly
    initial_binary = np.zeros(n_points, dtype=int)
    initial_binary[:int(n_points/2)] = 1
    np.random.shuffle(initial_binary)
    best_x = initial_binary.copy()
    best_C = compute_overlap_from_binary_fast(best_x)
    
    # Initialize temperature schedule
    T = T0
    cooling_rate = (T0 - Tf) / total_steps
    
    # Track stagnation
    stagnant_count = 0
    stagnation_threshold = 20000
    last_best = best_C
    last_best_iteration = 0
    
    # Simulated Annealing main loop
    for i in range(total_steps):
        if time.time() - start_time > budget_s - 5:
            break
        # Swap two locations
        # Option 1: Targeted swap based on worst shift
        # Option 2: Random swap
        if np.random.rand() < 0.6:
            # Try targeted swap
            d, U, V = get_worst_shift_indices(best_x)
            if U and V:
                i1 = np.random.choice(U)
                i2 = np.random.choice(V)
                # Create a new binary vector
                new_x = best_x.copy()
                new_x[i1] = 0
                new_x[i2] = 1
                # Recompute the change in overlap using precomputed method
                old_A = correlate(best_x, best_x, mode='full')
                new_A = correlate(new_x, new_x, mode='full')
                change_A = new_A - old_A
                # Correlation partial changes are I don't compute
                C_new = compute_overlap_from_binary_fast(new_x)
            else:
                # Fall back to simple swap
                idx1 = np.random.randint(0, n_points)
                idx2 = np.random.randint(0, n_points)
                while idx1 == idx2:
                    idx2 = np.random.randint(0, n_points)
                new_x = best_x.copy()
                new_x[idx1], new_x[idx2] = new_x[idx2], new_x[idx1]
                C_new = compute_overlap_from_binary_fast(new_x)
        else:
            # Pure random swap
            idx1 = np.random.randint(n_points)
            idx2 = np.random.randint(n_points)
            while idx1 == idx2:
                idx2 = np.random.randint(n_points)
            new_x = best_x.copy()
            new_x[idx1], new_x[idx2] = new_x[idx2], new_x[idx1]
            C_new = compute_overlap_from_binary_fast(new_x)
        
        # Accept or reject
        delta = C_new - best_C
        if delta <= 0:
            # Accept unconditionally
            best_x = new_x.copy()
            best_C = C_new
        else:
            # Accept with probability exp(-delta/T)
            if np.random.rand() < np.exp(-delta / T):
                best_x = new_x.copy()
                best_C = C_new
        
        n_accept += 1
        # Update temperature
        if n_accept > 0:
            T = max(Tf, T0 - cooling_rate * n_accept)
        
        # Control stagnation
        if i - last_best_iteration > stagnation_threshold:
            if abs(best_C - last_best) < 1e-5:
                # Inject diversity
                # Randomly flip some bits
                n_flip = int(n_points * 0.05)
                flip_indices = np.random.choice(n_points, n_flip, replace=False)
                for idx in flip_indices:
                    best_x[idx] = 1 - best_x[idx]
                # Project to exactly the right sum
                best_x = project_vector_to_sum_box(best_x.astype(float), TARGET_SUM).astype(int)
                last_best = best_C
                last_best_iteration = i        
                T = T * 0.5  # Reset temperature
                stagnant_count += 1
            else:
                last_best = best_C
                last_best_iteration = i

    # Final continuous refinement (Project back to floating point)
    h_float = best_x.astype(float)
    h_float = project_vector_to_sum_box(h_float, TARGET_SUM)
    
    # Run SLSQP on the continuous problem for fine-tuning
    from scipy.optimize import minimize
    
    def obj(h):
        h_ = h.copy()
        corr = correlate(h_, 1-h_, mode='full')
        return -dx * np.max(corr)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
    res = minimize(
        obj,
        h_float,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1500, 'ftol': 1e-12, 'gtol': 1e-12}
    )
    if res.success:
        refined_h = res.x
    else:
        refined_h = h_float
    
    # Ensure sum constraint still satisfied exactly 
    refined_h = project_vector_to_sum_box(refined_h, TARGET_SUM)
    
    # Apply moving average smoothing
    from scipy.ndimage import uniform_filter1d
    h_smooth = uniform_filter1d(refined_h, size=3)
    h_smooth = project_vector_to_sum_box(h_smooth, TARGET_SUM)
    
    # Final adjustment with SLSQP
    res2 = minimize(
        obj,
        h_smooth,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-12}
    )
    if res2.success:
        h_final = res2.x
    else:
        h_final = h_smooth

    h_final = project_vector_to_sum_box(h_final, TARGET_SUM)
    
    # Compute final bound
    final_corr = correlate(h_final, 1 - h_final, mode='full')
    final_c5 = dx * np.max(final_corr)
    
    return (h_final.tolist(), float(final_c5), n_points)
