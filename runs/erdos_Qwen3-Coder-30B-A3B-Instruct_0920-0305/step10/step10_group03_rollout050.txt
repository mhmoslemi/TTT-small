import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}

        # Use L-BFGS-B as it respects bounds better
        res = minimize(

def genetic_algorithm(start_h, generations=200, population_size=50):
    """Genetic algorithm approach"""
    def mutate_individual(individual, mutation_rate=0.1):
        mutated = individual.copy()
        for i in range(len(mutated)):
            if np.random.random() < mutation_rate:
                # Add small random change
                delta = np.random.normal(0, 0.02)
                mutated[i] += delta
        return project_to_sum_box(mutated, TARGET_SUM)
    
    def crossover(parent1, parent2):
        # Single-point crossover
        crossover_point = np.random.randint(1, len(parent1)-1)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        return project_to_sum_box(child1, TARGET_SUM), project_to_sum_box(child2, TARGET_SUM)
    
    # Initialize population
    population = [start_h.copy()]
    for _ in range(population_size - 1):
        individual = np.random.rand(n_points)
        individual = project_to_sum_box(individual, TARGET_SUM)
        population.append(individual)
    
    best_solution = start_h.copy()
    best_fitness = evaluate_c5(start_h)
    
    for generation in range(generations):
        # Evaluate fitness
        fitness_scores = [evaluate_c5(ind) for ind in population]
        
        # Keep best individuals for next generation
        sorted_indices = np.argsort(fitness_scores)
        top_indices = sorted_indices[:population_size//2]
        new_population = [population[i] for i in top_indices]
        
        # Add some diversity by combining parents
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = population[np.random.choice(top_indices)]
            parent2 = population[np.random.choice(top_indices)]
            child1, child2 = crossover(parent1, parent2)
            new_population.extend([child1, child2])
        
        # Apply mutation
        for i in range(len(new_population)):
            new_population[i] = mutate_individual(new_population[i])
        
        # Update best solution
        current_fitness = [evaluate_c5(ind) for ind in new_population]
        best_idx = np.argmin(current_fitness)
        if current_fitness[best_idx] < best_fitness:
            best_solution = new_population[best_idx].copy()
            best_fitness = current_fitness[best_idx]
            
        population = new_population[:population_size]
        
        if generation % 20 == 0:
            print(f"Gen {generation}: Best C5 = {best_fitness:.6f}")
    
    return best_solution, best_fitness
