# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_autocorr_fft(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def c5_from_autocorr(autocorr):
    """Convert autocorrelation result to C5 value"""
    min_autocorr = np.min(autocorr)
    return 1 - min_autocorr * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # central region from (a, 2-a) has height p
    if a > 0.5:
        a = 0.5
    
    # Compute height p for central region to satisfy sum constraint
    p = (1.0 - 2.0 * a) / (2.0 * (1.0 - a)) if a < 1.0 else 0.0
    
    # Ensure p is in [0, 1]
    p = max(0.0, min(1.0, p))
    
    h = np.zeros(n_points)
    
    # Create indices for the three regions
    edge_length = int(a * n_points / 2)
    if edge_length > 0:
        # Left edge
        h[:edge_length] = 1.0
        # Right edge
        h[-edge_length:] = 1.0
        # Middle
        middle_length = n_points - 2 * edge_length
        if middle_length > 0:
            h[edge_length:n_points-edge_length] = p
    else:
        # Uniform distribution
        h[:] = 0.5
    
    # Scale to ensure sum = TARGET_SUM
    current_sum = np.sum(h) * dx
    if current_sum > 0:
        h = h * (TARGET_SUM / current_sum)
    
    # Ensure all values are in [0, 1]
    h = np.clip(h, 0, 1)
    
    return h

def refine_with_simulated_annealing(h_start, c5_start, start_time, max_time):

    max_iter = int(1e5)
    T = T0
    cooling_rate = 0.99999
    
    for i in range(max_iter):
