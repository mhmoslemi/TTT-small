import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from cvxpy import *
import time

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
MILP_N = 60  # Binary problem size
dx_small = 2.0 / MILP_N

# Helper functions
def compute_corr_fft(h):
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    return dx * np.max(corr)

def evaluate_c5(h):
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    # Water-filling algorithm for projection
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_binary_skeleton(n):
    """Generate a random binary vector with prescribed sum"""
    vec = np.zeros(n)
    ones = np.random.choice(n, size=n//2, replace=False)
    vec[ones] = 1
    return vec

def upsample_vector(h_coarse, factor):
    """Upsample a coarse vector to finer resolution"""
    h_fine = np.repeat(h_coarse, factor)
    return h_fine[:n_points]

def solve_binary_milp():
    """Solve binary MILP to get a good skeleton"""
    x = Variable(MILP_N, boolean=True)
    t = Variable()
    constraints = [sum(x) == MILP_N // 2]
    
    # Add constraints for correlations
    for k in range(1, MILP_N):
        for i in range(MILP_N - k):
            y = Variable(boolean=True)
            constraints.append(y <= x[i])
            constraints.append(y <= x[i + k])
            constraints.append(y >= x[i] + x[i + k] - 1)
            constraints.append(sum([y for i in range(MILP_N - k)]) >= (MILP_N // 2) - t)

    objective = Minimize(t)
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=GLPK_MI, verbose=False, time_limit=30)
        if prob.status in [OPTIMAL, FEASIBLE]:
            result = x.value
            return result
    except:
        pass
    return generate_binary_skeleton(MILP_N)

def get_worst_shift_indices(x, k):
    A = set()
    B = set()
    for i in range(len(x) - k):
        if x[i] == 1 and x[i+k] == 0:
            A.add(i)
        elif x[i] == 0 and x[i+k] == 1:
            B.add(i)
    return list(A), list(B)

def do_swap_search(x, max_swaps=10000):
    rho = np.zeros(len(x))
    for k in range(1, len(x)):
        rho[k] = sum(x[i]*x[i+k] for i in range(len(x)-k))
    best_x = x.copy()
    best_rho = rho.copy()
    
    for _ in range(max_swaps):
        worst_k = np.argmax([(len(x)//2) - rho[k] for k in range(1, len(x))])
        A, B = get_worst_shift_indices(x, worst_k)
        if A and B:
            i = np.random.choice(A)
            j = np.random.choice(B)
            x[i], x[j] = x[j], x[i]
            # Update rho incrementally
            for k in range(1, len(x)):
                if i+k < len(x): rho[k] += x[i]*x[i+k] - x[i]*x[i+k] 
                if i+k < len(x): rho[k] += x[j]*x[j+k] - x[j]*x[j+k]
        else:
            if np.random.random() < 0.01:
                i, j = np.random.randint(0, len(x)), np.random.randint(0, len(x))
                if i != j:
                    x[i], x[j] = x[j], x[i]

    return x

def run_de(pop_size=30, max_iter=100):
    """Differential evolution"""
    pop = []
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        pop.append(h)
    
    best_h = pop[0].copy()
    best_c5 = evaluate_c5(best_h)
    
    for _ in range(max_iter):
        for h in pop:
            # Mutation
            a, b, c = np.random.choice(pop, 3, replace=False)
            F, CR = 0.8, 0.9
            m = a + F*(b - c)
            trial = np.where(np.random.rand(n_points) < CR, m, h)
            
            # Crossover + infeasible handling
            trial = project_to_sum_box(trial, TARGET_SUM)
            
            # Selection
            trial_c5 = evaluate_c5(trial)
            if trial_c5 < evaluate_c5(h):
                h[:] = trial[:]
            if trial_c5 < best_c5:
                best_h[:] = trial[:]
                best_c5 = trial_c5
    
    return best_h, best_c5

def run_sa(h_start, max_iter=2000):
    h = h_start.copy()
    c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    temp = 0.1
    alpha = 0.99
    
    for _ in range(max_iter):
        i, j = np.random.randint(0, n_points, 2)
        delta = np.random.normal(0, 0.05)
        old = h[i]
        new = h[i] + delta
        if new < 0:
            new = 0
        elif new > 1:
            new = 1
        h[i] = new
        
        # Update sum through projection
        h = project_to_sum_box(h, TARGET_SUM)
        
        new_c5 = evaluate_c5(h)
        delta_c5 = new_c5 - c5
        if delta_c5 < 0 or np.random.random() < np.exp(-delta_c5/temp):
            c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        else:
            h[i] = old
        temp *= alpha
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Solve MILP on smaller grid
    if time.time() - start_time < budget_s * 0.05:
        x_small = solve_binary_milp()
        x_small = x_small.astype(int)
        h0 = upsample_vector(x_small, n_points // MILP_N)
        h0 = project_to_sum_box(h0, TARGET_SUM)
    else:
        h0 = np.ones(n_points) * 0.5
        h0 = project_to_sum_box(h0, TARGET_SUM)
    
    # Mixed-upsampled swap search
    if time.time() - start_time < budget_s * 0.1:
        h0 = do_swap_search(h0, max_swaps=2000)
        h0 = project_to_sum_box(h0, TARGET_SUM)
    
    # Run DE
    if time.time() - start_time < budget_s * 0.8:
        h_de, c5_de = run_de(pop_size=30, max_iter=300)
        h0 = h_de.copy()
        best_c5 = c5_de
    else:
        best_c5 = evaluate_c5(h0)
    
    # SA polish
    if time.time() - start_time < budget_s * 0.95:
        h_sa, c5_sa = run_sa(h0, max_iter=1000)
        if c5_sa < best_c5:
            h0 = h_sa.copy()
            best_c5 = c5_sa
    
    # Final polishing
    if time.time() - start_time < budget_s:
        # Simple hill climbing using residuals
        h_best = h0.copy()
        c5_best = evaluate_c5(h0)
        for _ in range(1000):
            h_new = h_best.copy()
            indices = np.random.choice(n_points, size=10, replace=False)
            for i in indices:
                delta = np.random.normal(0, 0.02) * h_new[i] * (1 - h_new[i])
                h_new[i] += delta
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            c5_new = evaluate_c5(h_new)
            if c5_new < c5_best:
                h_best[:] = h_new[:]
                c5_best = c5_new
        h0 = h_best.copy()
        best_c5 = c5_best
    
    return (h0.tolist(), float(best_c5), n_points)
