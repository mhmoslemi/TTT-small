import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Initialize with random binary pattern
    h0 = np.zeros(n_points)
    ones_indices = np.random.choice(n_points, int(TARGET_SUM), replace=False)
    h0[ones_indices] = 1
    
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    # Simulated Annealing
    temp0 = 0.02
    temp_end = 1e-5
    max_iterations = 500000
    cooling_rate = (temp0 - temp_end) / max_iterations
    
    # Track best in current run to avoid stagnation
    current_h = h0.copy()
    current_c5 = best_c5
    stagnation_count = 0
    max_stagnation = 5000
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Cooling schedule
        temperature = max(temp_end, temp0 - iteration * cooling_rate)
        
        # Try a random swap
        i, j = np.random.randint(0, n_points, 2)
        while current_h[i] == current_h[j]:
            i, j = np.random.randint(0, n_points, 2)
            
        # Swap
        current_h[i], current_h[j] = current_h[j], current_h[i]
        
        # Compute new C5
        new_c5 = compute_c5(current_h)
        
        # Accept or reject
        delta = new_c5 - current_c5
        if delta <= 0 or np.random.rand() < np.exp(-delta / temperature):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            # Revert swap
            current_h[i], current_h[j] = current_h[j], current_h[i]
            
        # Occasionally restart to escape local minima
        if stagnation_count > max_stagnation:
            current_h = h0.copy()
            current_c5 = compute_c5(current_h)
            stagnation_count = 0
    
    # Final refinement with SLSQP
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
        res = minimize(
            compute_c5, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        if res.success:
            refined_h = res.x
            refined_h = project_to_sum_box(refined_h, TARGET_SUM)
            refined_c5 = compute_c5(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
