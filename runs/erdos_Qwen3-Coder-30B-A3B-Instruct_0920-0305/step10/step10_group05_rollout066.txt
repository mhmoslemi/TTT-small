import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

# Global parameters
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
n_half = n_points // 2

def project_to_sum_box(u, target_sum):
    """Water-filling projection to enforce sum constraint"""
    def f(lam):
        projected = np.clip(u + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(u + (a + b)/2, 0, 1)
    return result

def autocorr_fft(h):
    """Compute aperiodic autocorrelation using FFT"""
    h_rev = h[::-1]
    b = h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def exact_c5(u):
    """Compute exact C5 from half-vector u"""
    h = np.concatenate([u, u[::-1]])
    A = autocorr_fft(h)
    return 1 - dx * np.min(A)

def lse_c5(u, beta):
    """Compute LSE surrogate for C5"""
    h = np.concatenate([u, u[::-1]])
    A = autocorr_fft(h)
    return 1 - dx * np.log(np.sum(np.exp(beta * A))) / beta

def grad_lse_c5(u, beta):
    """Compute gradient of LSE surrogate for C5"""
    h = np.concatenate([u, u[::-1]])
    A = autocorr_fft(h)
    w = np.exp(beta * A)
    w_sum = np.sum(w)
    # Convolution of w with reversed h
    w_conv_h_rev = np.fft.irfft(np.fft.rfft(w) * np.fft.rfft(h[::-1], len(h)), len(h))
    grad = -dx * w_conv_h_rev / w_sum
    return grad[:n_half]

def parametric_to_halfvector(params, n):
    """Build half-vector from parametric family: 3-level shape"""
    a, p = params  # a: edge length, p: middle height
    u = np.ones(n//2)
    idx = int(np.round(a * n / 2))
    u[idx:] = p
    return u

def run_de():
    """Run differential evolution on parametric family"""
    def obj(params):
        u = parametric_to_halfvector(params, n_half)
        # Ensure mass constraint
        u = project_to_sum_box(u, n_half / 2)
        return exact_c5(u)
    
    bounds = [(0.05, 0.45), (0.1, 0.8)]
    res = differential_evolution(
        obj, bounds, seed=42, maxiter=100, popsize=15,
        disp=False, polish=False, workers=1
    )
    return res.x

def run_lbfgsb(u0, beta=200):
    """Run projected L-BFGS-B optimization"""
    def obj(u):
        return lse_c5(u, beta)
    
    def grad(u):
        return grad_lse_c5(u, beta)
    
    u = u0.copy()
    for _ in range(800):
        # Projected gradient step
        g = grad(u)
        u_new = u - 0.01 * g
        u_new = project_to_sum_box(u_new, n_half / 2)
        # Check convergence
        if np.linalg.norm(u - u_new) < 1e-10:
            break
        u = u_new
        
    return u

def run_sa(u0, max_iter=20000):
    """Run simulated annealing polishing"""
    u = u0.copy()
    best_u = u.copy()
    best_c5 = exact_c5(best_u)
    
    T0 = 5e-3
    T = T0
    alpha = 0.9997
    accept_ratio = 0.0
    accept_count = 0
    total_count = 0
    
    for i in range(max_iter):
        if i % 500 == 0:
            if accept_ratio > 0.4:
                T *= 0.99
            elif accept_ratio < 0.2:
                T *= 1.01
            
            accept_ratio = accept_count / max(1, total_count)
            accept_count = 0
            total_count = 0
            
        if T < 1e-6:
            break
            
        # Pick two distinct indices
        i1 = np.random.randint(0, n_half)
        i2 = np.random.randint(0, n_half)
        while i1 == i2:
            i2 = np.random.randint(0, n_half)
        
        # Get max delta
        delta_max = min(u[i1], 1 - u[i2])
        if delta_max < 1e-6:
            continue
            
        delta = np.random.uniform(0, delta_max)
        
        # Perform move
        u_new = u.copy()
        u_new[i1] -= delta
        u_new[i2] += delta
        
        # Accept or reject
        c5_old = exact_c5(u)
        c5_new = exact_c5(u_new)
        
        total_count += 1
        if c5_new < c5_old:
            u = u_new
            accept_count += 1
            if c5_new < best_c5:
                best_u = u_new.copy()
                best_c5 = c5_new
        else:
            if np.random.rand() < np.exp(-(c5_new - c5_old) / T):
                u = u_new
                accept_count += 1
    
    return best_u, best_c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    global TARGET_SUM, n_points, dx, n_half
    TARGET_SUM = n_points / 2
    n_half = n_points // 2
    
    # Phase 1: Parametric sweep
    best_c5 = float('inf')
    best_u = None
    
    # Simple 1D sweep (ignore previous approach due to its nature)
    for a in np.linspace(0.05, 0.45, 40):
        p = (1 - 2 * a) / (2 * (1 - a))
        if p > 1 or p < 0:
            continue
        u = parametric_to_halfvector([a, p], n_half)
        u = project_to_sum_box(u, n_half / 2)
        c5 = exact_c5(u)
        if c5 < best_c5:
            best_c5 = c5
            best_u = u.copy()
    
    # Phase 2: Differential Evolution
    if time.time() - start_time < budget_s * 0.5:
        try:
            params = run_de()
            u = parametric_to_halfvector(params, n_half)
            u = project_to_sum_box(u, n_half / 2)
            c5 = exact_c5(u)
            if c5 < best_c5:
                best_c5 = c5
                best_u = u.copy()
        except:
            pass
    
    # Phase 3: L-BFGS with fine grid
    if time.time() - start_time < budget_s * 0.8:
        try:
            # Interpolate to a finer grid
            u = run_lbfgsb(best_u, beta=200)
            u = project_to_sum_box(u, n_half / 2)
            c5 = exact_c5(u)
            if c5 < best_c5:
                best_c5 = c5
                best_u = u.copy()
        except:
            pass
    
    # Phase 4: SA polishing
    if time.time() - start_time < budget_s:
        try:
            u, c5 = run_sa(best_u)
            if c5 < best_c5:
                best_c5 = c5
                best_u = u.copy()
        except:
            pass
    
    # Build full vector
    h = np.concatenate([best_u, best_u[::-1]])
    return (h.tolist(), float(best_c5), n_points)
