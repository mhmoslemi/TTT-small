import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_autocorr_fft(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def min_autocorr_from_corr(corr, n):
    """Get minimum autocorrelation from correlation result"""
    return dx * np.min(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def evaluate_c5_by_autocorr(h):
    """Fast evaluation of C5 using autocorrelation"""
    autocorr = compute_autocorr_fft(h)
    min_auto = np.min(autocorr)
    return 1 - min_auto * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_piecewise_constant_h(a):
    """Generate piecewise constant h with symmetric shape."""
    # Given a: edge length of interval with height 1
    # Then middle length = 2 - 2*a
    # Using constraint 2*a*1 + (2-2*a)*p = 1
    # Solving for p: p = (1-2*a)/(2*(1-a))
    
    if a >= 0.5:
        a = 0.499
    p = (1 - 2*a) / (2 * (1 - a))
    
    h = np.zeros(n_points)
    
    # Determine indices for the edge regions
    edge_len = int(a * n_points / 2)
    
    # Left edge
    h[:edge_len] = 1.0
    # Right edge
    h[-edge_len:] = 1.0
    
    # Middle region
    middle_len = n_points - 2*edge_len
    if middle_len > 0:
        h[edge_len:n_points-edge_len] = p
    
    # Ensure correct sum
    h = project_to_sum_box(h, n_points / 2)
    
    return h

def find_optimal_piecewise_constant():
    """Find optimal piecewise constant shape by scanning values of a."""
    best_c5 = float('inf')
    best_a = 0
    best_h = None
    
    # Scan a from 0.01 to 0.49 in steps
    for a in np.arange(0.01, 0.49, 0.005):
        h = generate_piecewise_constant_h(a)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_a = a
            best_h = h.copy()
    
    return best_h, best_c5, best_a

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Try to find better initial solution using piecewise constant approximation
    best_h, best_c5, _ = find_optimal_piecewise_constant()
    
    # Try simulated annealing to further improve
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Simulated Annealing parameters
    max_iter = 200000
    T_start = 0.001
    T = T_start
    cooling_rate = 0.9999
    
    # Track acceptance rate
    accept_count = 0
    total_moves = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Randomly pick two indices
        i1 = np.random.randint(n_points)
        i2 = np.random.randint(n_points)
        
        # Random step size
        step_size = np.random.uniform(0.001, 0.05)
        
        # Compute the adjustment amount ensuring h stays in [0,1]
        delta = np.random.choice([-1, 1]) * step_size
        delta1 = -delta
        
        # Adjust values
        h_new = current_h.copy()
        h_new[i1] = np.clip(h_new[i1] + delta1, 0, 1)
        h_new[i2] = np.clip(h_new[i2] + delta, 0, 1)
        
        # Project to correct sum
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Compute new C5
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject based on simulated annealing
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
            accept_count += 1
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(c5_new - current_c5) / T):
                current_h = h_new.copy()
                current_c5 = c5_new
                accept_count += 1
        
        # Update temperature
        T = T_start * (cooling_rate ** i)
        
        # Occasionally do a local optimization step
        if i % 5000 == 0 and i > 0:
            # Use L-BFGS-B to refine the solution locally
            try:
                def objective(h):
                    return evaluate_c5(h)
                
                def constraint_eq(h):
                    return np.sum(h) - TARGET_SUM
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons = {'type': 'eq', 'fun': constraint_eq}
                
                from scipy.optimize import minimize
                res = minimize(
                    objective, 
                    current_h, 
                    method='L-BFGS-B', 
                    bounds=bounds, 
                    constraints=cons,
                    options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
                )
                if res.success:
                    h_opt = res.x
                else:
                    h_opt = current_h
                h_opt = project_to_sum_box(h_opt, TARGET_SUM)
                c5_opt = evaluate_c5(h_opt)
                
                if c5_opt < current_c5:
                    current_h = h_opt.copy()
                    current_c5 = c5_opt
                    
            except Exception:
                pass
    
    # Final optimization
    try:
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        from scipy.optimize import minimize
        res = minimize(
            objective, 
            current_h, 
            method='L-BFGS-B', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_final = res.x
        else:
            h_final = current_h
        h_final = project_to_sum_box(h_final, TARGET_SUM)
        c5_final = evaluate_c5(h_final)
        
        if c5_final < current_c5:
            current_h = h_final.copy()
            current_c5 = c5_final
            
    except Exception:
        pass
    
    return (current_h.tolist(), float(current_c5), n_points)
