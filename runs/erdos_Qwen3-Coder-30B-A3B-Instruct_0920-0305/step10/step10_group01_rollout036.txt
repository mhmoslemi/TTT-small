import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_smooth_max(h, alpha):
    """Evaluate smooth max surrogate"""
    corr = compute_corr_fft(h)
    # Ensure numerical stability
    corr_max = np.max(corr)
    exp_corr = np.exp(alpha * (corr - corr_max))
    sum_exp = np.sum(exp_corr)
    if sum_exp == 0:
        return float('inf')
    smooth_max = corr_max + np.log(sum_exp) / alpha
    return smooth_max

def compute_gradient(h, alpha):
    """Compute gradient of smooth max surrogate"""
    corr = compute_corr_fft(h)
    # Numerical stability
    corr_max = np.max(corr)
    exp_corr = np.exp(alpha * (corr - corr_max))
    sum_exp = np.sum(exp_corr)
    if sum_exp == 0:
        return np.zeros_like(h)
    weights = exp_corr / sum_exp
    
    # Create symmetric weights for correlation
    weights_sym = weights + np.flip(weights)
    # Zero-pad to match correlation length
    weights_sym = np.pad(weights_sym, (0, len(h) - 1), mode='constant')
    
    # Full correlation of weights_sym and h
    g_full = np.fft.ifft(np.fft.fft(weights_sym) * np.fft.fft(h)).real
    g = g_full[:len(h)]
    grad = 1 - g
    return grad

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    x = np.clip(x, 0, 1)
    
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result
