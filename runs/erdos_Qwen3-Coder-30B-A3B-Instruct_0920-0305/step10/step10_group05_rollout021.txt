import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_autocorr_fft(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def evaluate_c5_autocorr(h):
    """Evaluate C5 using autocorrelation"""
    auto_corr = compute_autocorr_fft(h)
    min_auto = np.min(auto_corr)
    return 1 - dx * min_auto

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_piecewise_constant_h(a):
    """Generate piecewise constant h with symmetric shape"""
    # Calculate p from the constraint: 2*a*1 + (2-2*a)*p = 1
    p = (1 - 2*a) / (2 - 2*a)
    
    # Create h with symmetric piecewise constant shape
    h = np.zeros(n_points)
    
    # Convert a to number of points
    edge_points = int(np.round(a * n_points / 2))
    
    # Fill edge regions
    h[:edge_points] = 1
    h[-edge_points:] = 1
    
    # Fill middle region
    middle_start = edge_points
    middle_end = n_points - edge_points
    h[middle_start:middle_end] = p
    
    # Adjust to match exact sum
    current_sum = np.sum(h) * dx
    if abs(current_sum - 1.0) > 1e-10:
        # Adjust by scaling
        scale_factor = 1.0 / current_sum
        h *= scale_factor
        
    return h

def optimize_piecewise_constant():
    """Find optimal parameter a for piecewise constant h"""
    best_a = 0.0
    best_c5 = float('inf')
    
    # Search in reasonable range of a
    search_range = np.arange(0.01, 0.45, 0.001)
    
    for a in search_range:
        try:
            h = generate_piecewise_constant_h(a)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_a = a
        except Exception:
            continue
    
    return best_a, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Try optimization for piecewise constant shape first
    try:
        best_a, best_c5_pc = optimize_piecewise_constant()
        h = generate_piecewise_constant_h(best_a)
        h = project_to_sum_box(h, TARGET_SUM)
        current_c5 = evaluate_c5(h)
        best_h = h.copy()
        best_c5 = current_c5
    except Exception:
        # Fallback to random initialization
        h = np.random.rand(n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        best_h = h.copy()
        best_c5 = evaluate_c5(h)
    
    # Simulated Annealing parameters
    T_start = 0.001
    max_iter = 200000
    step_sigma = 0.02
    
    # Initialize for SA
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Run SA
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random move
        i1 = np.random.randint(0, n_points)
        i2 = np.random.randint(0, n_points)
        
        # Ensure both indices are different
        if i1 == i2:
            continue
            
        # Choose random delta
        epsilon = np.random.uniform(0, step_sigma)
        
        # Make sure we don't violate bounds
        if current_h[i1] >= epsilon and current_h[i2] <= 1 - epsilon:
            # Perform move
            h_new = current_h.copy()
            h_new[i1] -= epsilon
            h_new[i2] += epsilon
            
            # Project to maintain sum and bounds
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            # Evaluate new solution
            c5_new = evaluate_c5(h_new)
            
            # Annealing acceptance probability
            T = T_start * (1 - i / max_iter)
            if c5_new < current_c5 or np.random.random() < np.exp(-(c5_new - current_c5) / max(T, 1e-10)):
                current_h = h_new.copy()
                current_c5 = c5_new
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
    
    # Final refinement with gradient descent
    try:
        from scipy.optimize import minimize
        
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        res = minimize(objective, 
                      best_h, 
                      method='SLSQP', 
                      bounds=bounds, 
                      constraints=cons,
                      options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12})
        
        if res.success:
            h_opt = res.x
        else:
            h_opt = best_h
        h_opt = project_to_sum_box(h_opt, TARGET_SUM)
        c5_opt = evaluate_c5(h_opt)
        
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            
    except Exception:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
