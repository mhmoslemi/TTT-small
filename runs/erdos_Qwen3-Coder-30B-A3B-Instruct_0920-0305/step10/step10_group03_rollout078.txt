def compute_overlap(h):
    """Compute overlap integral using FFT approach"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_box(h_values):
    """Project values to [0,1]"""
    return np.clip(h_values, 0, 1)

def project_to_sum(h_values, target_sum):
    """Project to sum = target_sum using clipping and rescaling"""
    h_proj = project_to_box(h_values)
    current_sum = np.sum(h_proj)
    if abs(current_sum - target_sum) < 1e-10:
        return h_proj
    scale_factor = target_sum / current_sum
    h_scaled = h_proj * scale_factor
    return project_to_box(h_scaled)

def sim_anneal_polish(h0, max_iter=2000):
    """Simulated annealing polish step"""
    current_h = h0.copy()

    best_h = current_h.copy()
    best_c5 = current_c5
    T = 1.0
    T_min = 1e-8
    alpha = 0.99

        # Perturb
