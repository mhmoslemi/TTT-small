# Global variables
TARGET_SUM = None

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        h_new = project_to_sum_box(h_new, TARGET_SUM)

        c5_new = evaluate_c5(h_new)

                best_h = current_h.copy()
                best_c5 = c5_new

            current_h = h_new.copy()
            current_c5 = c5_new
    
    return best_h, best_c5

def evolutionary_strategy(h0, pop_size=20, generations=50):
    """Evolutionary approach with mutation and selection"""
    population = [h0.copy()]
    for _ in range(pop_size - 1):
        # Generate random individuals
        h_random = np.random.rand(n_points)
        h_random = project_to_sum_box(h_random, TARGET_SUM)
        population.append(h_random)
    
    for gen in range(generations):
        # Evaluate fitness
        fitness = [evaluate_c5(ind) for ind in population]
        
        # Selection: keep top 50%
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:pop_size//2]
        population = [population[i] for i in top_indices]
        
        # Generate new population through mutation
        new_population = population.copy()
        for _ in range(pop_size - len(population)):
            # Select parent
            parent = population[np.random.randint(len(population))]
            # Mutate
            h_mut = parent.copy()
            indices = np.random.choice(n_points, size=n_points//10, replace=False)
            for idx in indices:
                h_mut[idx] += np.random.normal(0, 0.02)
            h_mut = project_to_sum_box(h_mut, TARGET_SUM)
            new_population.append(h_mut)
        
        population = new_population
    
    # Return best individual
    fitness = [evaluate_c5(ind) for ind in population]
    best_idx = np.argmin(fitness)
    return population[best_idx], fitness[best_idx]
