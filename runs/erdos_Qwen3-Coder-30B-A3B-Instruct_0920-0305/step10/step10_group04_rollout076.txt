import numpy as np
from scipy.signal import correlate
import time
import math

# Global variables
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_binary_auto_correlation(x):
    """Compute auto-correlation for binary vector"""
    # Using FFT to compute correlation with itself
    L = 2 * len(x) - 1
    x_rev = x[::-1]
    corr = np.fft.ifft(np.fft.fft(x_rev, L) * np.fft.fft(x, L)).real
    return corr

def compute_overlap_from_auto_corr(A, x):
    """Compute overlap from auto-correlation"""
    n = len(x)
    O = np.zeros(2 * n - 1)
    # For each shift d
    for d in range(-(n-1), n):
        if d >= 0:
            # Shift x to the right by d
            sum_x_left = np.sum(x[:n-d])
        else:
            # Shift x to the left by |d|
            sum_x_left = np.sum(x[-d:])
        
        O[d + n - 1] = sum_x_left - A[d + n - 1]
    
    return O

def c5_from_correlation(O):
    """Convert overlap array to C5 value"""
    return dx * np.max(O)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_to_float_projection(x_binary):
    """Convert binary to float vector with projected sum constraint"""
    h = x_binary.astype(float)
    return project_to_sum_box(h, TARGET_SUM)

def compute_delta_correlation(x, i, j):
    """Compute delta correlation vector for swapping x[i] and x[j]"""
    n = len(x)
    pad_x = np.pad(x, (n-1, n-1), mode='constant', constant_values=0)
    delta_A = np.zeros(2*n - 1)
    
    # Contributions from i being turned off
    if 0 <= i + (n-1) < 3*n-2 and 0 <= i < n:
        delta_A += pad_x[i+n-1:i+3*n-2] + pad_x[i:i+2*n-1][::-1]
    
    # Contributions from j being turned on
    if 0 <= j + (n-1) < 3*n-2 and 0 <= j < n:
        delta_A -= pad_x[j+n-1:j+3*n-2] + pad_x[j:j+2*n-1][::-1]
    
    return delta_A

def find_worst_shift_positions(x, A, d_star):
    """Find positions that can be swapped to reduce worst shift d_star"""
    n = len(x)
    U = []  # Exposed positions for shift d_star
    V = []  # Positions that can cover for d_star
    
    d_star_index = d_star + n - 1
    
    if d_star >= 0:
        for i in range(n - d_star):
            if x[i] == 1 and x[i + d_star] == 0:
                U.append(i)
            if x[i] == 0 and x[i + d_star] == 1:
                V.append(i)
    else:
        d_abs = -d_star
        for i in range(n - d_abs):
            if x[i + d_abs] == 0 and x[i] == 1:
                U.append(i)
            if x[i + d_abs] == 1 and x[i] == 0:
                V.append(i)
    
    return U, V

def perturb_binary_vector(x, percentage):
    """Perturb binary vector by randomly flipping some bits"""
    n = len(x)
    num_flips = int(percentage * n)
    indices = np.random.choice(n, size=num_flips, replace=False)
    x_new = x.copy()
    for idx in indices:
        x_new[idx] = 1 - x_new[idx]
    # Ensure exact counts
    ones_count = np.sum(x_new)
    if ones_count > TARGET_SUM:
        # Flip enough 1s to 0s
        flip_indices = np.where(x_new == 1)[0]
        to_flip = int(ones_count - TARGET_SUM)
        for idx in np.random.choice(flip_indices, size=to_flip, replace=False):
            x_new[idx] = 0
    elif ones_count < TARGET_SUM:
        # Flip enough 0s to 1s
        flip_indices = np.where(x_new == 0)[0]
        to_flip = int(TARGET_SUM - ones_count)
        for idx in np.random.choice(flip_indices, size=to_flip, replace=False):
            x_new[idx] = 1
    return x_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase A: Simulated Annealing with Binary Vectors
    # Initialize binary vector
    initial_x = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
    # Ensure we have exactly target sum
    ones_count = np.sum(initial_x)
    if ones_count > TARGET_SUM:
        flip_indices = np.where(initial_x == 1)[0]
        to_flip = int(ones_count - TARGET_SUM)
        for idx in np.random.choice(flip_indices, size=to_flip, replace=False):
            initial_x[idx] = 0
    elif ones_count < TARGET_SUM:
        flip_indices = np.where(initial_x == 0)[0]
        to_flip = int(TARGET_SUM - ones_count)
        for idx in np.random.choice(flip_indices, size=to_flip, replace=False):
            initial_x[idx] = 1
    
    current_x = initial_x.copy()
    A = compute_binary_auto_correlation(current_x)
    O = compute_overlap_from_auto_corr(A, current_x)
    current_c5 = c5_from_correlation(O)
    
    best_x = current_x.copy()
    best_c5 = current_c5
    
    # SA parameters
    T0 = 2.0 * dx
    Tf = 1e-6 * dx
    num_moves = 300000
    temp_schedule = [T0 + (Tf - T0) * i / num_moves for i in range(num_moves)]
    
    # Track progress
    last_improvement = 0
    
    for iteration in range(num_moves):
        if time.time() - start_time > budget_s - 5:
            break
            
        T = temp_schedule[iteration]
        
        # Try a move
        if np.random.rand() < 0.6:
            # Targeted move
            d_star = np.argmax(O) - n_points + 1
            U, V = find_worst_shift_positions(current_x, A, d_star)
            if U and V:
                i = np.random.choice(U)
                j = np.random.choice(V)
                # Apply the swap
                current_x[i], current_x[j] = current_x[j], current_x[i]
                # Update correlation incrementally
                delta_A = compute_delta_correlation(current_x, i, j)
                A += delta_A
                # Update overlap
                O = compute_overlap_from_auto_corr(A, current_x)
                new_c5 = c5_from_correlation(O)
            else:
                # Fall back to random swap
                i = np.random.randint(0, n_points)
                j = np.random.randint(0, n_points)
                while current_x[i] == current_x[j]:
                    j = np.random.randint(0, n_points)
                # Apply the swap
                current_x[i], current_x[j] = current_x[j], current_x[i]
                # Update correlation incrementally
                delta_A = compute_delta_correlation(current_x, i, j)
                A += delta_A
                # Update overlap
                O = compute_overlap_from_auto_corr(A, current_x)
                new_c5 = c5_from_correlation(O)
        else:
            # Random swap
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            while current_x[i] == current_x[j]:
                j = np.random.randint(0, n_points)
            # Apply the swap
            current_x[i], current_x[j] = current_x[j], current_x[i]
            # Update correlation incrementally
            delta_A = compute_delta_correlation(current_x, i, j)
            A += delta_A
            # Update overlap
            O = compute_overlap_from_auto_corr(A, current_x)
            new_c5 = c5_from_correlation(O)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            current_c5 = new_c5
        else:
            # Revert swap
            current_x[i], current_x[j] = current_x[j], current_x[i]
            # Reapply the same correlation update
            A -= delta_A
            O = compute_overlap_from_auto_corr(A, current_x)
        
        # Update best solution
        if current_c5 < best_c5:
            best_x = current_x.copy()
            best_c5 = current_c5
            last_improvement = iteration
        
        # Stagnation detection
        if (iteration - last_improvement) > 20000:
            current_x = perturb_binary_vector(best_x, 0.05)
            A = compute_binary_auto_correlation(current_x)
            O = compute_overlap_from_auto_corr(A, current_x)
            current_c5 = c5_from_correlation(O)
            last_improvement = iteration
    
    # Phase B: Continuous refinement
    h = binary_to_float_projection(best_x)
    
    # First try with SLSQP
    try:
        from scipy.optimize import minimize
        def objective(h_vals):
            return c5_from_correlation(compute_overlap_from_auto_corr(
                compute_binary_auto_correlation(np.clip(h_vals, 0, 1)), 
                np.clip(h_vals, 0, 1)
            ))
        
        bounds = [(0, 1)] * n_points
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False}
        )
        if res.success:
            h = project_to_sum_box(res.x, TARGET_SUM)
        else:
            h = project_to_sum_box(h, TARGET_SUM)
    except:
        h = project_to_sum_box(h, TARGET_SUM)
    
    # Smooth and refine again
    def smooth_array(arr, width=3):
        """Apply moving average smoothing"""
        padded = np.pad(arr, width//2, mode='edge')
        return np.convolve(padded, np.ones(width)/width, mode='valid')[:len(arr)]
    
    h_smooth = smooth_array(h, 3)
    h = project_to_sum_box(h_smooth, TARGET_SUM)
    
    # Final refinement with SLSQP
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'disp': False}
        )
        if res.success:
            h = project_to_sum_box(res.x, TARGET_SUM)
    except:
        pass
    
    # Final check
    h = project_to_sum_box(h, TARGET_SUM)
    
    # Evaluate final solution
    final_A = compute_binary_auto_correlation(h)
    final_O = compute_overlap_from_auto_corr(final_A, h)
    final_c5 = c5_from_correlation(final_O)
    
    return (h.tolist(), float(final_c5), n_points)
