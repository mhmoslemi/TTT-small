def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)

def project_to_sum_box(x, target_sum):

        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10

        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break

            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_surrogate(h, alpha):
    """Compute smooth max surrogate F_alpha(h) = (1/alpha) * log(sum(exp(alpha * c_k)))"""

    # Numerically stable computation
    exp_terms = np.exp(alpha * (c_k - max_c))
    sum_exp = np.sum(exp_terms)
    return (1.0 / alpha) * (np.log(sum_exp) + max_c)

    # Compute softmax weights

    w_sym = w_k + w_k[::-1]
    
    # Compute correlation with symmetrized weights
    g_full = compute_corr_fft(w_sym)
    g = g_full[:len(h)]
    
    # Gradient is 1 - g
    grad = 1.0 - g
    return grad
