def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def full_corr(a, b):
    """Compute correlation using FFT with zero-padding, return full correlation"""
    L = 2 * len(a) - 1
    return np.fft.ifft(np.fft.fft(a, L) * np.fft.fft(b, L).conj()).real

def smooth_max_and_weights(c, alpha):
    """Compute smooth max and weights using log-sum-exp trick"""
    c_shifted = c - np.max(c)
    exp_c = np.exp(alpha * c_shifted)
    exp_sum = np.sum(exp_c)
    weights = exp_c / exp_sum
    smooth_max = (np.log(exp_sum) + np.max(c)) / alpha
    return smooth_max, weights

def compute_gradient(h, alpha):
    """Compute gradient of smooth max objective using FFT"""
    c = full_corr(h, 1 - h)
    _, w = smooth_max_and_weights(c, alpha)
    
    # Symmetrize weights
    w_sym = w + w[::-1]
    
    # Correlation of symmetrized weights with h
    g = full_corr(w_sym, h)
    
    # Extract first N entries and compute gradient
    g = g[:len(h)]
    grad = 1.0 - g
    
    return grad

def smooth_objective(h, alpha):
    """Evaluate smooth objective"""
    c = full_corr(h, 1 - h)
    val, _ = smooth_max_and_weights(c, alpha)
    return val

def objective_with_grad(h, alpha):
    """Return both objective value and gradient"""
    c = full_corr(h, 1 - h)
    val, w = smooth_max_and_weights(c, alpha)
    
    # Symmetrize weights
    w_sym = w + w[::-1]
    
    # Correlation of symmetrized weights with h
    g = full_corr(w_sym, h)
    g = g[:len(h)]
    grad = 1.0 - g
    
    return val, grad
