def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def swap_update(O, x, a, b):
    """Update correlation counts after swapping x[a] and x[b]"""
    new_O = O.copy()
    
    # For each shift k where a or b appears in a correlation term
    for k in range(len(x)):
        # Contribution from x[a] with x[a+k] (if exists)
        if a + k < len(x) and a + k >= 0:
            if x[a] == 1 and x[a+k] == 1:
                new_O[k] -= 1
            elif x[a] == 1 and x[a+k] == 0:
                new_O[k] += 1
            elif x[a] == 0 and x[a+k] == 1:
                new_O[k] -= 1
                
        # Contribution from x[b] with x[b+k] (if exists)
        if b + k < len(x) and b + k >= 0:
            if x[b] == 1 and x[b+k] == 1:
                new_O[k] -= 1
            elif x[b] == 1 and x[b+k] == 0:
                new_O[k] += 1
            elif x[b] == 0 and x[b+k] == 1:
                new_O[k] -= 1
                
        # Contribution from x[a+k] with x[a] (if exists)
        if a - k >= 0 and a - k < len(x):
            if x[a-k] == 1 and x[a] == 1:
                new_O[k] -= 1
            elif x[a-k] == 1 and x[a] == 0:
                new_O[k] += 1
            elif x[a-k] == 0 and x[a] == 1:
                new_O[k] -= 1
                
        # Contribution from x[b+k] with x[b] (if exists)
        if b - k >= 0 and b - k < len(x):
            if x[b-k] == 1 and x[b] == 1:
                new_O[k] -= 1
            elif x[b-k] == 1 and x[b] == 0:
                new_O[k] += 1
            elif x[b-k] == 0 and x[b] == 1:
                new_O[k] -= 1
    
    return new_O

def swap_improvement_local_search(h0, max_iter=10000):
    """Improve binary pattern by swapping bits to reduce max overlap"""
    current_h = h0.copy().astype(float)
    n = len(current_h)
    
    # Compute initial overlaps
    O = np.zeros(n)
    for k in range(n):
        O[k] = sum(current_h[i] * current_h[i+k] for i in range(n-k))
    
    D = 0.5 * n - O  # Uncovered count for each shift
    best_D = np.max(D)
    best_h = current_h.copy()
    best_D_max = best_D
    
    # Record current best
    best_c5 = 1 - best_D_max / (0.5 * n)  # Max correlation -> Min C5
    
    # Iterative improvement by swapping bits
    for i in range(max_iter):
        D_max = np.max(D)
        if D_max <= 0.5:  # Already optimal
            break
            
        # Find shifts causing most uncovered
        k_max = np.argmax(D)
        k_best = k_max
        
        # Find positions where swap will help
        candidates_a = []
        candidates_b = []
        
        # For shift k_max
        for a in range(n):
            if current_h[a] == 1 and (a + k_max < n) and current_h[a + k_max] == 0:
                candidates_a.append(a)
            if (a + k_max < n) and current_h[a] == 0 and current_h[a + k_max] == 1:
                candidates_b.append(a)
                
        # If no candidates, try all valid pairs
        if not candidates_a or not candidates_b:
            # Random swap
            a = np.random.randint(0, n)
            b = np.random.randint(0, n)
            if current_h[a] == current_h[b]:
                continue
        else:
            # Try to swap a good pair
            a = np.random.choice(candidates_a)
            b = np.random.choice(candidates_b)
            
        # Try the swap
        new_h = current_h.copy()
        new_h[a] = 1 - new_h[a]
        new_h[b] = 1 - new_h[b]
        
        # Check sum constraint
        if abs(np.sum(new_h) - 0.5 * n) > 1e-6:
            continue
            
        # Compute new overlaps efficiently
        new_O = swap_update(O, current_h, a, b)
        new_D = 0.5 * n - new_O
        new_D_max = np.max(new_D)
        
        if new_D_max < D_max:
            current_h = new_h
            O = new_O
            D = new_D
            D_max = new_D_max
            # Update best if improved
            current_c5 = 1 - D_max / (0.5 * n)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_D_max = D_max
                best_c5 = current_c5
    
    return best_h, 1 - best_D_max / (0.5 * n)

def upsample_binary(binary_vec, target_N):
    """Upsample binary pattern to higher resolution"""
    # Replicate each element
    upsampled = np.repeat(binary_vec, target_N // len(binary_vec))
    # Pad if necessary
    if len(upsampled) < target_N:
        upsampled = np.pad(upsampled, (0, target_N - len(upsampled)), constant_values=0)
    elif len(upsampled) > target_N:
        upsampled = upsampled[:target_N]
    return upsampled

def differential_evolution_optimization(h_initial, target_sum, time_budget, start_time):
    """Run differential evolution optimization"""
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(len(h_initial))]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Wrap objective to project onto sum constraint inside
    def constrained_objective(h):
        h_projected = project_to_sum_box(h, target_sum)
        return objective(h_projected)
    
    # Run DE with timeout
    try:
        res = differential_evolution(
            constrained_objective, 
            bounds, 
            constraints=cons,
            maxiter=1000,  # Limit iterations to prevent long run
            popsize=20,
            seed=42,
           Updating='deferred'
        )
        if res.success:
            final_h = project_to_sum_box(res.x, target_sum)
            return final_h, evaluate_c5(final_h)
    except:
        pass
    
    return h_initial, evaluate_c5(h_initial)
