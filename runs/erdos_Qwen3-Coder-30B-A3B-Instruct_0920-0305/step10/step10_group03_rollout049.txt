# Global variables
TARGET_SUM = None

dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def worst_shift_swap(h, iterations, budget_s_start):

                delta = np.clip(np.random.normal(0, 0.02), -0.1, 0.1)
                h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
        
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < current_c5:

            current_c5 = c5_new

                best_c5 = current_c5
        elif np.random.random() < 0.02:  # Occasionally accept worse solutions
            current_h = h_new.copy()
            current_c5 = c5_new
    
    return best_h, best_c5

def parametric_step_function_search(h0, budget_s):
    """Search for step function parameters that yield lower C5"""
    from scipy.optimize import minimize
    
    # Define parametric representation using fewer variables for step function
    # Parameters: step locations and heights (n_steps = 5, say)
    def step_function_from_params(params, n_points):
        # Convert parameters to step function
        n_steps = 5
        # params = [x1, x2, ..., x5, h1, h2, ..., h5]
        step_locs = params[:n_steps] * 2.0  # map from [0,1] to [0,2]
        step_heights = params[n_steps:]  # height in [0, 1]
        step_heights = np.clip(step_heights, 0, 1)
        
        h = np.zeros(n_points)
        dx = 2.0 / n_points
        for i in range(n_points):
            x = i * dx
            # Find appropriate step height for this x
            idx = np.searchsorted(step_locs, x, side='right') - 1
            if idx >= 0:
                h[i] = step_heights[idx]
        
        return h
    
    def obj_func(params):
        h = step_function_from_params(params, n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        return evaluate_c5(h)
    
    # Initial guess
    init_params = np.random.rand(10)  # 5 step locations, 5 heights
    
    def cons_func(params):
        h = step_function_from_params(params, n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': cons_func}
    
    try:
        res = minimize(obj_func, init_params, method='SLSQP', bounds=[(0,1)]*10, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12})
        if res.success:
            h_opt = step_function_from_params(res.x, n_points)
            h_opt = project_to_sum_box(h_opt, TARGET_SUM)
            return h_opt, evaluate_c5(h_opt)
    except:
        pass
    
    return h0, evaluate_c5(h0)

def differential_evolution_search(h0, budget_s):
    """Use Differential Evolution optimization on the full space"""
    start_time = time.time()
    
    def objective(h):
        h_proj = project_to_sum_box(h, TARGET_SUM)
        return evaluate_c5(h_proj)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = differential_evolution(objective, bounds, maxiter=500, popsize=15, 
                                    mutation=(0.5, 1), recombination=0.7,
                                    seed=42, disp=False, polish=True)
        if res.success:
            h_opt = project_to_sum_box(res.x, TARGET_SUM)
            return h_opt, evaluate_c5(h_opt)
    except:
        pass
    
    return h0, evaluate_c5(h0)
