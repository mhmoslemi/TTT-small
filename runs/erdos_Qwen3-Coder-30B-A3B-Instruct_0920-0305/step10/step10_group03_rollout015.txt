# Global variables
TARGET_SUM = None

dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_with_cvxpy(h0):
    """Use CVXPY for optimization"""
    try:
        import cvxpy as cp
        h = cp.Variable(n_points)
        objective = cp.Minimize(cp.max(cp.conv(h, 1 - h, 'full')))
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == TARGET_SUM
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            return project_to_sum_box(h.value, TARGET_SUM)
    except:
        pass
    return h0

        lambda x, idx: x + np.random.normal(0, 0.01, 1)[0],
        lambda x, idx: x + np.random.uniform(-0.03, 0.03),
        lambda x, idx: np.clip(x + np.random.normal(0, 0.02, len(x)), 0, 1)
    ]
    
    for i in range(max_iter):
        h_new = current_h.copy()
        
        # Perform different types of perturbations
        if np.random.random() < 0.7:
            # Single point perturbations
            indices = np.random.choice(n_points, size=min(15, n_points//8), replace=False)
            for idx in indices:
                h_new[idx] = strategies[np.random.randint(0, len(strategies))](h_new, idx)
        else:
            # Global perturbation
            h_new = strategies[-1](h_new, 0)
        
        # Project to valid constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new)
        
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = current_h.copy()
                best_c5 = c5_new
        elif np.random.random() < 0.03:  # Occasionally accept worse solutions
            current_h = h_new.copy()
            current_c5 = c5_new
    
    return best_h, best_c5

        return evaluate_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]

    try:

            h0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 3000, 'ftol': 1e-14, 'gtol': 1e-14}
        )
        if res.success:
            return project_to_sum_box(res.x, TARGET_SUM)
    except:
        pass
    return h0
