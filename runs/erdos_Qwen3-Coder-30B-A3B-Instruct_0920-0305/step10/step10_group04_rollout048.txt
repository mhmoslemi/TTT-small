import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points
INITIAL_BINARY = None

def compute_correlation_and_overlap(x):
    """Fast computation of A_d and O_d for binary x"""
    # Compute auto-correlation (full mode)
    x_padded = np.pad(x, (n_points-1, n_points-1), mode='constant', constant_values=0)
    A = correlate(x, x, mode='full')
    # Compute prefix sum for efficient window sum
    x_cumsum = np.concatenate([[0], np.cumsum(x)])
    # Compute O_d = sum of x in window minus A_d
    O = np.zeros(2*n_points - 1)
    for d in range(-n_points+1, n_points):
        if d >= 0:
            u = x_cumsum[n_points] - x_cumsum[d]
            v = x_cumsum[n_points-d] - x_cumsum[0]
        else:
            u = x_cumsum[n_points+d] - x_cumsum[0]
            v = x_cumsum[n_points] - x_cumsum[-d]
        O[d+n_points-1] = u - A[d+n_points-1]
    return A, O

def evaluate_c5_binary(x):
    """Evaluate C5 for a binary vector"""
    _, O = compute_correlation_and_overlap(x)
    return dx * np.max(O)

def evaluate_c5_continuous(h):
    """Evaluate C5 for a continuous vector using exact method"""
    corr = correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector onto [0,1] box with sum = target_sum using binary search on lambda"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    return np.clip(x + (a + b)/2, 0, 1)

def generate_initial_binary():
    """Generate binary vectors from initial_h_values if available"""
    try:
        init_values = np.array(initial_h_values)
        binary = (init_values > 0.5).astype(int)
        # Adjust to sum exactly n/2
        total_ones = np.sum(binary)
        if total_ones < n_points // 2:
            indices = np.where(binary == 0)[0]
            to_flip = indices[:n_points//2 - total_ones]
            binary[to_flip] = 1
        elif total_ones > n_points // 2:
            indices = np.where(binary == 1)[0]
            to_flip = indices[:total_ones - n_points//2]
            binary[to_flip] = 0
        return binary
    except:
        return np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])

def fast_update_A(x, old_A, i, j):
    """Incremental update of auto-correlation after swapping bits at positions i and j"""
    d_start = -(n_points - 1)
    new_A = old_A.copy()
    
    # For both positions i and j, compute the delta in all A_d values
    for d in range(-n_points+1, n_points):
        delta = 0
        # The x_i + x_{i-d} terms
        if 0 <= i+d < n_points:
            delta -= x[i+d]
        if 0 <= i-d < n_points:
            delta -= x[i-d]
        # The x_j + x_{j-d} terms  
        if 0 <= j+d < n_points:
            delta += x[j+d]
        if 0 <= j-d < n_points:
            delta += x[j-d]
        # Apply delta to the corresponding element in A
        new_A[d + n_points - 1] += delta
    
    return new_A

def target_swap(x, d_star, A):
    """Generate a targeted swap that improves the worst overlap shift"""
    # Sets for exposed 1s and missing 0s at shift d_star
    U = []; V = []
    for i in range(n_points):
        if x[i] == 1:
            if 0 <= i + d_star < n_points and x[i + d_star] == 0:
                U.append(i)
            if 0 <= i - d_star < n_points and x[i - d_star] == 1:
                V.append(i)
    
    if U and V:
        i = np.random.choice(U)
        j = np.random.choice(V)
        return i, j, True
    return None, None, False

def simulated_annealing_binary(start_binary, T0=2*dx, Tf=1e-6*dx, max_steps=300000):
    """Perform simulated annealing on binary vector with targeted swaps"""
    current_x = start_binary.copy()
    current_A, _ = compute_correlation_and_overlap(current_x)
    current_C5 = evaluate_c5_binary(current_x)
    best_x = current_x.copy()
    best_C5 = current_C5
    T = T0
    step = 0
    
    temperature_schedule = np.linspace(T0, Tf, max_steps)
    
    while step < max_steps:
        if step % 20000 == 0:
            # Check stagnation
            step_time = time.time()
            if step_time - start_time > budget_s - 5:
                break
                
        # Choose move type
        if np.random.random() < 0.4:
            # Random swap
            i, j = np.random.choice(np.arange(n_points), 2, replace=False)
            while current_x[i] != 1 or current_x[j] != 0:
                i, j = np.random.choice(np.arange(n_points), 2, replace=False)
        else:
            # Targeted swap
            _, O = compute_correlation_and_overlap(current_x)
            d_star = np.argmax(O) - (n_points - 1)
            i, j, success = target_swap(current_x, d_star, current_A)
            if not success:
                # Fallback to random swap
                i, j = np.random.choice(np.arange(n_points), 2, replace=False)
                while current_x[i] != 1 or current_x[j] != 0:
                    i, j = np.random.choice(np.arange(n_points), 2, replace=False)
        
        # Perform the swap
        new_x = current_x.copy()
        new_x[i], new_x[j] = new_x[j], new_x[i]
        
        # Update A in-place
        new_A = fast_update_A(current_x, current_A, i, j)
        
        new_C5 = dx * np.max(O)  # Placeholder - we need recompute O from new_A
        # Compute new overlap values correctly
        _, new_O = compute_correlation_and_overlap(new_x)
        new_C5 = dx * np.max(new_O)
        
        # Accept or reject
        delta = new_C5 - current_C5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            current_x = new_x
            current_A = new_A
            current_C5 = new_C5
            if current_C5 < best_C5:
                best_x = current_x.copy()
                best_C5 = current_C5
        
        if step % 5000 == 0:
            T = temperature_schedule[step]
        
        step += 1
        
    # After SA, convert to continuous for fine tuning
    h_continuous = best_x.astype(float)
    h_continuous = project_to_sum_box(h_continuous, TARGET_SUM)
    
    return h_continuous, best_C5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx, INITIAL_BINARY
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    
    # Generate initial binary vector
    INITIAL_BINARY = generate_initial_binary()
    
    # Convert to continuous with exact projection
    h_initial = INITIAL_BINARY.astype(float)
    h_initial = project_to_sum_box(h_initial, TARGET_SUM)
    
    # Perform simulated annealing on binary space
    final_h, final_c5 = simulated_annealing_binary(INITIAL_BINARY)
    
    # Final continuous refinement
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
        res = minimize(
            evaluate_c5_continuous,
            final_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            final_h = res.x
            final_h = project_to_sum_box(final_h, TARGET_SUM)
            final_c5 = evaluate_c5_continuous(final_h)
    except:
        pass
    
    # Optionally smooth and re-optimize
    try:
        # Low-pass smoothing
        kernel = np.ones(3) / 3
        smooth_h = np.convolve(final_h, kernel, mode='same')
        smooth_h = project_to_sum_box(smooth_h, TARGET_SUM)
        smooth_h = np.clip(smooth_h, 0, 1)
        res = minimize(
            evaluate_c5_continuous,
            smooth_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            final_h = res.x
            final_h = project_to_sum_box(final_h, TARGET_SUM)
            final_c5 = evaluate_c5_continuous(final_h)
    except:
        pass
    
    return (final_h.tolist(), float(final_c5), n_points)
