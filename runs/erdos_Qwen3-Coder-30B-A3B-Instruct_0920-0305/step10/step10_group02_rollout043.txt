import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def genetic_algorithm(current_h, current_c5, start_time, budget_s):
    """Generator algorithm for global optimization"""
    population_size = 50
    generations = 200
    mutation_rate = 0.1
    elite_count = 5
    
    population = []
    fitness = []
    
    # Initialize population
    for _ in range(population_size):
        h = current_h + np.random.normal(0, 0.02, n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
        fitness.append(evaluate_c5(h))
    
    # Evolution loop
    for generation in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        new_population = population[:elite_count]
        
        # Generate offspring
        for _ in range(population_size - elite_count):
            # Tournament selection
            parent1_idx = np.random.randint(0, population_size//2)
            parent2_idx = np.random.randint(0, population_size//2)
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover (uniform)
            mask = np.random.random(n_points) > 0.5
            child = np.where(mask, parent1, parent2)
            
            # Mutation
            if np.random.random() < mutation_rate:
                mutation_idx = np.random.randint(0, n_points)
                child[mutation_idx] += np.random.normal(0, 0.03)
            
            child = project_to_sum_box(child, TARGET_SUM)
            new_population.append(child)
            
        population = new_population
        fitness = [evaluate_c5(h) for h in population]
        
        # Update best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < current_c5:
            current_h = population[best_idx].copy()
            current_c5 = fitness[best_idx]
    
    return current_h, current_c5

def local_refinement(current_h, start_time, budget_s):
    """Improve solution using multiple local search strategies"""
    h = current_h.copy()
    c5 = evaluate_c5(h)
    
    # Try different approaches
    strategies = [
        lambda x: gradient_descent_step(x, start_time, budget_s),
        lambda x: local_perturbation_search(x, start_time, budget_s),
        lambda x: simulated_annealing_step(x, start_time, budget_s)
    ]
    
    for strategy in strategies:
        if time.time() - start_time > budget_s - 10:
            break
        try:
            new_h, new_c5 = strategy(h)
            if new_c5 < c5:
                h = new_h
                c5 = new_c5
        except:
            continue
            
    return h, c5

def gradient_descent_step(h, start_time, budget_s):

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Try to optimize with more aggressive settings
    try:
        res = minimize(
            objective,

def local_perturbation_search(h, start_time, budget_s):
    """Search for improvements by small local perturbations"""
    h_best = h.copy()
    c5_best = evaluate_c5(h)
    
    iterations = 1000
    for i in range(iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        h_new = h_best.copy()
        # Perturb one component at a time
        for _ in range(5):
            idx = np.random.randint(0, n_points)
            h_new[idx] += np.random.normal(0, 0.01)
            
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return h_best, c5_best

def simulated_annealing_step(h, start_time, budget_s):
    """Run simulated annealing with better cooling schedule"""
    temp_start = 0.05
    temp_end = 0.0001
    cooling_rate = 0.999
    max_iter = 2000
    
    h_current = h.copy()
    c5_current = evaluate_c5(h_current)
    
    temp = temp_start
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Perturb solution
        h_new = h_current.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.02)
        h_new[idx] += delta
        
        # Project to valid range
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject based on Metropolis criterion
        if c5_new < c5_current or np.random.random() < np.exp(-(c5_new - c5_current) / temp):
            h_current = h_new.copy()
            c5_current = c5_new
            
        temp = max(temp_end, temp * cooling_rate)
    
    return h_current, c5_current
