import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import math

# Global constants        
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_auto_correlation(x):
    """Compute auto-correlation of binary vector x using FFT"""
    # x is a boolean or int array
    return correlate(x, x, mode='full')

def compute_overlap_from_auto(auto_corr, x):
    """Compute overlap values from auto-correlation"""
    n = len(x)
    # The auto-correlation length is 2*n - 1
    # We need to compute O_d = sum_{i=0}^{n-1-d} x_i * (1 - x_{i+d})  
    # Which is (sum_{i=0}^{n-1-d} x_i) - A_d where A_d is the auto-correlation
    # We compute cumulative sum for prefix sums
    cum_sum = np.concatenate([[0], np.cumsum(x)])
    d_vals = np.arange(-n+1, n)
    overlaps = np.zeros_like(d_vals)
    for i, d in enumerate(d_vals):
        if d >= 0:
            # O_d = (sum_{0}^{n-1-d} x_i) - A_d 
            # The sum is from index 0 to n-1-d -> from cum_sum[n-d] to cum_sum[n] (0-indexed)
            # But cum_sum[i] gives sum from 0 to i-1.
            # So cum_sum[n] - cum_sum[n-d-1] = sum from n-d-1+1 = n-d to n-1, that's the reverse.
            # Actually cum_sum[n-1-d+1] - cum_sum[0] = sum_{0}^{n-1-d}
            if (n-1-d) >= 0:
                prefix_sum = cum_sum[n-1-d+1] - cum_sum[0]
            else:
                prefix_sum = 0
        else:
            # For negative d, we flip directions
            # O_d = sum_{0}^{n-1+d} x_i - A_d
            if (n-1+d) >= 0:
                prefix_sum = cum_sum[n-1+d+1] - cum_sum[0]
            else:
                prefix_sum = 0
        overlaps[i] = prefix_sum - auto_corr[n-1+d]
    return overlaps

def compute_c5_fast(x):
    """Fast computation of C5 for a binary vector"""
    auto_corr = compute_auto_correlation(x)
    overlaps = compute_overlap_from_auto(auto_corr, x)
    return dx * np.max(overlaps)

def update_auto_corr_delta(auto_corr, x, i, j):
    """Incrementally update auto-correlation after swapping x[i] and x[j] 
       Returns the delta vector to be added to auto_corr"""
    n = len(x)
    # First pad x to allow indexing with various shifts  
    x_padded = np.zeros(3*n - 2)
    x_padded[n-1 : 2*n-1] = x
    
    # Delta for index i (when it turns off)
    delta_i = np.zeros(2*n - 1)
    for d in range(-n+1, n):
        if 0 <= i+d < n:
            delta_i[d+n-1] = x_padded[i + n - 1 + d]
        else:
            delta_i[d+n-1] = 0
    
    # Delta for index j (when it turns on)  
    delta_j = np.zeros(2*n - 1)
    for d in range(-n+1, n):
        if 0 <= j+d < n:
            delta_j[d+n-1] = x_padded[j + n - 1 + d]
        else:
            delta_j[d+n-1] = 0
            
    # The reverse side - contribution of x_i and x_j at negative shifts
    # Exploit symmetry: For shift -d, the contribution from i is x_{i+d}, 
    # for shift d, x_{i-d} (but for index in negative position)
    # So it's done in reverse way in loop
    # Actually start from index 0, but be super careful
    delta_i_rev = np.zeros(2*n - 1) 
    for d in range(-n+1, n):
        # For delta_i, contribution of x_i to shift d is x_{i+d}
        if not (0 <= i+d < n):
            delta_i_rev[d+n-1] = 0
        else:
            delta_i_rev[d+n-1] = x_padded[i + n - 1 + d]
            
    delta_j_rev = np.zeros(2*n - 1)
    for d in range(-n+1, n):
        if not (0 <= j+d < n):
            delta_j_rev[d+n-1] = 0
        else:
            delta_j_rev[d+n-1] = x_padded[j + n - 1 + d]
    
    # Combine since i goes off (negative) and j goes on (positive)  
    return -(delta_i + delta_i_rev) + (delta_j + delta_j_rev)

    
def swap_bits_binary(x, i, j):
    """Swap bits at positions."""
    x[i], x[j] = x[j], x[i]

def compute_worst_shift(x, auto_corr):
    """Returns the shift index that maximizes overlap"""
    overlaps = compute_overlap_from_auto(auto_corr, x)
    return np.argmax(overlaps)


def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase A: Binary Simulated Annealing 
    T_init = 2 * dx
    T_final = 1e-6 * dx
    total_moves = 300000
    accepted_moves = 0
    
    # Create initial binary vector - half ones, half zeros  
    x = np.zeros(n_points, dtype=int)
    x[:n_points//2] = 1
    np.random.shuffle(x)
    
    # Evaluate initial solution
    auto_corr = compute_auto_correlation(x)
    current_c5 = compute_c5_fast(x)
    best_c5 = current_c5
    best_x = x.copy()
    
    # Temperature schedule - linear cooling
    delta_T = (T_init - T_final) / total_moves
    
    # Main annealing loop
    for move in range(total_moves):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Decide move type
        if np.random.rand() < 0.6:
            # Targeted swap - based on worst shift
            worst_shift = compute_worst_shift(x, auto_corr)
            
            # Get potential candidate positions for swap
            U = []
            V = []
            n = len(x)
            for i in range(n):
                j = i + worst_shift
                if 0 <= j < n:
                    if x[i] == 1 and x[j] == 0:
                        U.append(i)
                    elif x[i] == 0 and x[j] == 1:
                        V.append(i)
            if len(U) > 0 and len(V) > 0:
                # Pick one pair uniformly
                i = np.random.choice(U)
                j = np.random.choice(V)
                # Perform the swap with delta update  
                delta = update_auto_corr_delta(auto_corr, x, i, j)
                auto_corr += delta 
                swap_bits_binary(x, i, j)
                # Compute new C5
                auto_corr_cum = compute_auto_correlation(x)
                overlaps_new = compute_overlap_from_auto(auto_corr_cum, x)
                new_c5 = dx * np.max(overlaps_new)
                if new_c5 < current_c5:
                    current_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_x = x.copy()
                else:
                    # Probabilistic acceptance
                    if np.random.rand() < np.exp(- (new_c5 - current_c5) / T_init):
                        current_c5 = new_c5
                    else:
                        # Revert the swap
                        swap_bits_binary(x, i, j)
                        auto_corr -= delta
            else:
                # Fall back to random swap if no good candidates
                i, j = np.random.choice(n_points, size=2, replace=False)
                if x[i] == x[j]:
                    continue  # same values, no change after swap
                    
                # Save old value
                old_auto_corr = auto_corr.copy()
                delta = update_auto_corr_delta(auto_corr, x, i, j)
                auto_corr += delta 
                swap_bits_binary(x, i, j)
                
                # Compute new C5
                auto_corr_cum = compute_auto_correlation(x)
                overlaps_new = compute_overlap_from_auto(auto_corr_cum, x)
                new_c5 = dx * np.max(overlaps_new)
                if new_c5 < current_c5:
                    current_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_x = x.copy()
                else:
                    # Probabilistic acceptance
                    if np.random.rand() < np.exp(- (new_c5 - current_c5) / T_init):
                        current_c5 = new_c5
                    else:
                        # Revert the swap
                        swap_bits_binary(x, i, j)
                        auto_corr = old_auto_corr
        else:
            # Random swap
            i, j = np.random.choice(n_points, size=2, replace=False)
            if x[i] == x[j]:
                continue  # same values, no change after swap
                
            # Save old value
            old_auto_corr = auto_corr.copy()
            delta = update_auto_corr_delta(auto_corr, x, i, j)
            auto_corr += delta 
            swap_bits_binary(x, i, j)
            
            # Compute new C5
            auto_corr_cum = compute_auto_correlation(x)
            overlaps_new = compute_overlap_from_auto(auto_corr_cum, x)
            new_c5 = dx * np.max(overlaps_new)
            if new_c5 < current_c5:
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_x = x.copy()
            else:
                # Probabilistic acceptance
                if np.random.rand() < np.exp(- (new_c5 - current_c5) / T_init):
                    current_c5 = new_c5
                else:
                    # Revert the swap
                    swap_bits_binary(x, i, j)
                    auto_corr = old_auto_corr
    
        # Cool down temperature  
        T_init -= delta_T
    
    # Phase B: Continuous refinement 
    # Convert binary solution to continuous with smoothing
    h_scaled = best_x.astype(float)
    # Smooth a bit
    kernel = np.ones(5) / 5
    h_smooth = np.convolve(h_scaled, kernel, mode='same')
    h_final = project_to_sum_box(h_smooth, TARGET_SUM)
    
    # Run first SLSQP 
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
    res = minimize(
        lambda h: dx * np.max(np.correlate(h, 1 - h, mode='full')),
        h_final,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False}
    )
    if res.success:
        h_final = res.x.copy()
        h_final = project_to_sum_box(h_final, TARGET_SUM)
        
        # Second run in case of plateaus  
        res2 = minimize(
            lambda h: dx * np.max(np.correlate(h, 1 - h, mode='full')),
            h_final,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'disp': False}
        )
        if res2.success:
            h_final = res2.x.copy()
    
    # Final projection
    h_final = project_to_sum_box(h_final, TARGET_SUM)
    c5_bound = dx * np.max(np.correlate(h_final, 1 - h_final, mode='full'))
    
    return (h_final.tolist(), float(c5_bound), n_points)
