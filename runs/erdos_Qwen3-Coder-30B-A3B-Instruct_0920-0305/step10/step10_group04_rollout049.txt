import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def compute_overlap_from_binary(x):
    """Compute overlap using binary vector and incremental rearrangement."""
    n = len(x)
    dx = 2.0 / n
    
    # Compute auto-correlation
    A = correlate(x, x, mode='full')
    
    # Prefix sum for window sums
    prefix = np.concatenate([[0], np.cumsum(x)])
    
    # Compute overlaps for all shifts
    overlaps = []
    for d in range(-n+1, n):
        # Calculate overlap at this shift
        if d >= 0:
            window_sum = prefix[n-d] - prefix[0]
        else:
            window_sum = prefix[n] - prefix[-d]
        if d <= 0:
            A_d = A[n-1+d]
        else:
            A_d = A[n-1+d]
        overlap = window_sum - A_d
        overlaps.append(overlap)
    
    # Return maximum overlap scaled by dx
    return dx * max(overlaps)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search to find lambda
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    return np.clip(x + (a + b)/2, 0, 1)

def find_worst_shift_indices(x, d_star):
    """Find indices for targeted swap."""
    n = len(x)
    U = []
    V = []
    
    # All positions where x_i=1 and x_{i+d_star}=0
    start = max(0, -d_star)
    end = min(n, n - d_star)
    for i in range(start, end):
        if x[i] == 1 and x[i + d_star] == 0:
            U.append(i)
    
    # All positions where x_j=0 and x_{j+d_star}=1
    for j in range(start, end):
        if x[j] == 0 and x[j + d_star] == 1:
            V.append(j)
    
    return U, V

def objective_function(h):
    """Objective function to minimize: C5"""
    # This assumes h is a valid continuous representation
    n = len(h)
    dx = 2.0 / n
    
    # Compute auto-correlation
    h_rev = np.flip(h)
    b = 1 - h
    corr = correlate(h_rev, b, mode='full')
    
    # Maximize overlapping integral
    C5 = dx * np.max(corr)
    return C5
