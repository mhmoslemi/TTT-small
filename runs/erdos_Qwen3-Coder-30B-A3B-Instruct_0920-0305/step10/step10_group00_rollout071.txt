import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def upscale_h(original_h, n_new):
    """Resample original_h to n_new points using linear interpolation"""
    n_old = len(original_h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, original_h)

def propose_move(h, max_step, rng):
    """Propose a move that maintains sum constraint and box constraints."""
    n = len(h)
    # try up to few times to find feasible pair
    for _ in range(10):
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # compute feasible interval for delta added to i and subtracted from j
        lower = max(-h[i], h[j] - 1)
        upper = min(1 - h[i], h[j])
        if lower < upper:
            # limit delta magnitude
            lower = max(lower, -max_step)
            upper = min(upper, max_step)
            if lower >= upper:
                continue
            delta = rng.uniform(lower, upper)
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            return new_h
        # try opposite direction: add to j, subtract from i
        lower = max(-h[j], h[i] - 1)
        upper = min(1 - h[j], h[i])
        if lower < upper:
            lower = max(lower, -max_step)
            upper = min(upper, max_step)
            if lower >= upper:
                continue
            delta = rng.uniform(lower, upper)
            new_h = h.copy()
            new_h[j] += delta
            new_h[i] -= delta
            return new_h
    return None

def simulated_annealing(start_h, time_budget, rng):
    """Run simulated annealing with time budget"""
    curr_h = start_h.copy()
    curr_val = evaluate_c5(curr_h)
    best_h_local = curr_h.copy()
    best_val_local = curr_val
    
    start = time.time()
    total = time_budget
    T0 = 0.02
    T_min = 0.0001
    max_step_initial = 0.2
    max_step_final = 0.001
    
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1 - elapsed/total) + T_min * (elapsed/total)
        max_step = max_step_initial * max(0.0, 1 - elapsed/total) + max_step_final * (elapsed/total)
        
        new_h = propose_move(curr_h, max_step, rng)
        if new_h is None:
            continue
            
        new_val = evaluate_c5(new_h)
        delta = new_val - curr_val
        
        if delta <= 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            curr_h = new_h
            curr_val = new_val
            if new_val < best_val_local:
                best_val_local = new_val
                best_h_local = curr_h.copy()
                
    return best_h_local, best_val_local

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    best_c5 = float('inf')
    best_h = None
    rng = np.random.default_rng(seed)
    
    # Generate initial solutions
    init_options = []
    
    # From provided initial_h_values
    try:
        global initial_h_values
        init_h = np.array(initial_h_values)
        if len(init_h) != n_points:
            init_h = upscale_h(init_h, n_points)
        init_h = project_to_sum_box(init_h, TARGET_SUM)
        init_options.append(init_h)
    except:
        pass

    # Cosine shape
    xs = np.arange(n_points) * dx
    a = 0.5 - 1e-5
    cos_h = 0.5 + a * np.cos(np.pi * xs / 2)
    cos_h = project_to_sum_box(cos_h, TARGET_SUM)
    init_options.append(cos_h)
    
    # Random uniform
    rand_h = np.random.uniform(0, 1, n_points)
    rand_h = project_to_sum_box(rand_h, TARGET_SUM)
    init_options.append(rand_h)
    
    # Binary random
    binary_h = np.zeros(n_points)
    ones_idx = np.random.choice(n_points, size=n_points//2, replace=False)
    binary_h[ones_idx] = 1.0
    init_options.append(binary_h)
    
    # Find best initial solution
    best_init_h = None
    best_init_val = float('inf')
    for h in init_options:
        val = evaluate_c5(h)
        if val < best_init_val:
            best_init_val = val
            best_init_h = h
    
    # Run simulated annealing with best initial
    sa_time = budget_s * 0.9
    sa_h, sa_val = simulated_annealing(best_init_h, sa_time, rng)
    
    # Run local refinement
    refine_h, refine_val = local_refine(sa_h, 50000, rng)
    
    # Return best solution

    final_val = refine_val if refine_val < sa_val else sa_val
    
    return (final_h.tolist(), float(final_val), n_points)
