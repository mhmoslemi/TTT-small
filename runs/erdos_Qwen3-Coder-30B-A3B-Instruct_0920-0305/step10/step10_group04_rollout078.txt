def compute_overlap_from_binary(x):

    # For shift d: sum of x[i] where x[i] overlaps with x[i+d] so that
    # the number of 1s in the overlapping window = (partial sum up to i+1) - (partial sum up to i-d+1)
    # i.e., O_d = (prefix[i+1] - prefix[i-d]) 
    # But recurrence with precomputed A is simpler

    # This is trickier with the '1 - h(x+k)' part; need a roll

    max_overlap = np.max(overlaps) * dx  
    return max_overlap

def compute_overlap_exact(h):
    """Compute exact overlap using provided correlate function"""
    corr = correlate(h, 1-h, mode="full")
    return dx * np.max(corr)

def binary_to_float(binary):
    """Convert binary vector to float vector with correct sum"""
    h = binary.astype(float)  
    # Project to correct sum
    lambda_val = find_lambda(h)
    h = np.clip(h + lambda_val, 0, 1)  
    return h

def find_lambda(y):
    """Find lambda for projection"""
    def residual(lam):
        proj = np.clip(y + lam, 0, 1)
        return np.sum(proj) - TARGET_SUM
    
    # Binary search over lambda
    a, b = -1.0, 1.0
    for _ in range(100):
        m = (a + b) / 2
        res = residual(m)
        if abs(res) < 1e-10:
            break
        if res * residual(a) < 0:
            b = m
        else:
            a = m
    return (a + b)/2

def project_to_sum_box(x, target_sum):
    """Project to [0,1] range with fixed sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    # Expand search range until sign changes
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search for lambda
    tol = 1e-10
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        new_c5 = compute_overlap_from_binary(binary_used)
        old_c5 = current_c5
        if new_c5 < old_c5:
            # Always accept
            current_c5 = new_c5
        else:
            # Accept with temperature-based probability
            delta = new_c5 - old_c5
            prob = np.exp(-delta / (temp_start * (1.0 - (accepted_moves / cooling_iters))) )
            if np.random.random() < prob:
                current_c5 = new_c5
        
        # May increase the counter for cooldown
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = binary_to_float(binary_used)
            
        # Check stagnation
        if iter_count % 20000 == 0:
            # Flood restart: flipping a portion of bits
            if abs(current_c5 - old_c5) < 1e-5:
                frac_flip = 0.03  # 3% of bits
                num_to_flip = int(frac_flip * n_points)
                old_fixed =(binary_used.copy())
                to_flip = np.random.choice(n_points, size=num_to_flip, replace=False)
                # Flip bits
                for p in to_flip:
                    binary_used[p] = 1 - binary_used[p]
                # Now adjust balance manually 
                # Count how many ones are off
                diff = int(TARGET_SUM - np.sum(binary_used)) 
                if diff > 0:
                    # Units to be set from 0s  
                    zero_idx = np.where(binary_used == 0)[0]
                    if len(zero_idx) >= diff:
                        to_set = np.random.choice(zero_idx, size=diff, replace=False)
                        for x in to_set:
                            binary_used[x] = 1
                elif diff < 0: 
                    one_idx = np.where(binary_used == 1)[0]
                    if len(one_idx) >= -diff:
                        to_unset = np.random.choice(one_idx, size=-diff, replace=False)
                        for x in to_unset:
                            binary_used[x] = 0
                new_c5 = compute_overlap_from_binary(binary_used)  
                current_c5 = new_c5
                
        accepted_moves += 1
    
    # Phase B: Continuous optimization
    # Refine using SLSQP  
    # Overwrite h with polymorphism  
    h = best_h.copy()
    # Project h to sum and box
    h = project_to_sum_box(h, TARGET_SUM) 
    # Fine-tune with local search
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
        from scipy.optimize import minimize
        res = minimize(
            compute_overlap_exact, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            refined_h = res.x
            refined_h = project_to_sum_box(refined_h, TARGET_SUM)
            refined_c5 = compute_overlap_exact(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception as e:
        pass
        
    # Potentially apply smoothing
    try:
        from scipy.ndimage import uniform_filter1d  
        smoothed = uniform_filter1d(best_h, size=5, mode='reflect')
        reprojected = project_to_sum_box(smoothed, TARGET_SUM)
        smoothed_c5 = compute_overlap_exact(reprojected)
        if smoothed_c5 < best_c5:
            best_h = reprojected
            best_c5 = smoothed_c5
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
