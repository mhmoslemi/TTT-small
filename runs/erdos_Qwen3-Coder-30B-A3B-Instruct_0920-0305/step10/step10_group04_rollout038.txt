import numpy as np
from scipy.signal import correlate
import time

# Global settings
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_overlap_from_binary(x):
    """Fast evaluation of max overlap over shifts for a binary vector"""
    # Compute correlation of x with itself
    A = correlate(x, x, mode='full')
    # Compute prefix sum of x for efficient windowing
    prefix = np.concatenate([[0], np.cumsum(x)])
    # Compute overlaps for all shifts
    max_overlap = -np.inf
    for d in range(-n_points+1, n_points):
        if d >= 0:
            start_i, end_i = 0, n_points - d
            sum_window = prefix[end_i] - prefix[start_i]
        else:
            start_i, end_i = -d, n_points
            sum_window = prefix[end_i] - prefix[start_i]
        A_d = A[d + n_points - 1]
        overlap = sum_window - A_d
        max_overlap = max(max_overlap, overlap)
    return dx * max_overlap

def compute_overlap_for_float(h):
    """Evaluate overlap for a continuous function h"""
    corr = correlate(h, 1-h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    result = np.clip(x + (a + b) / 2, 0, 1)
    return result

def make_binary_initial(n, target_sum):
    """Create a binary initial solution"""
    x = np.zeros(n, dtype=int)
    x[:int(target_sum)] = 1
    np.random.shuffle(x)
    return x

def local_improve_binary(x_ini, budget_s):
    """Improve binary solution using simulated annealing"""
    # Get current best
    best_x = x_ini.copy()
    best_c5 = compute_overlap_from_binary(best_x)
    
    # Runtime tracking
    start_time = time.time()
    num_accepts = 0
    T = 2 * dx
    T_final = 1e-6 * dx
    steps = 0
    accepted = 0
    stagnation = 0
    
    # Predictable number of swaps
    max_swaps = 300000
    cooling_rate = (T - T_final) / max_swaps
    last_improvement = 0
    
    # Modified Simulated Annealing with targeted moves
    while num_accepts < max_swaps and (time.time() - start_time) < (budget_s - 5):
        steps += 1
        old_c5 = compute_overlap_from_binary(best_x)
        
        # Identify worst shift
        A = correlate(best_x, best_x, mode='full')
        prefix = np.concatenate([[0], np.cumsum(best_x)])
        max_overlap = -np.inf
        d_star = 0
        used_window = np.array([0]*n_points)
        
        for d in range(-n_points+1, n_points):
            if d >= 0:
                start_i, end_i = 0, n_points - d
            else:
                start_i, end_i = -d, n_points
            if end_i <= n_points and start_i >= 0:
                sum_window = prefix[end_i] - prefix[start_i]
                A_d = A[d + n_points - 1]
                overlap = sum_window - A_d
                if overlap > max_overlap:
                    max_overlap = overlap
                    d_star = d
                    used_window = np.zeros(n_points)
                    used_window[start_i:end_i] = 1
        
        # Targeted moves
        if np.random.random() < 0.6:
            U = []
            V = []
            # Find locations that would reduce overlap if swapped
            for i in range(n_points):
                if d_star >= 0:
                    j = i + d_star
                    if 0 <= j < n_points:
                        if best_x[i] == 1 and best_x[j] == 0:
                            U.append(i)
                        elif best_x[i] == 0 and best_x[j] == 1:
                            V.append(i)
                else:
                    j = i + d_star
                    if 0 <= j < n_points:
                        if best_x[i] == 1 and best_x[j] == 0:
                            U.append(i)
                        elif best_x[i] == 0 and best_x[j] == 1:
                            V.append(i)
            
            if U and V:
                i = np.random.choice(U)
                j = np.random.choice(V)
                # Swap
                temp = best_x[i]
                best_x[i] = best_x[j]
                best_x[j] = temp
                # Update with a fast incremental change
                new_c5 = compute_overlap_from_binary(best_x)
                if new_c5 < old_c5 or np.random.random() < np.exp(-(new_c5 - old_c5) / T):
                    accepted += 1
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        last_improvement = steps
                    else:
                        # roll it back
                        best_x[i] = temp
                        best_x[j] = best_x[i]
                else:
                    # roll it back
                    best_x[i] = temp
                    best_x[j] = best_x[i]
            else:
                # fall back to random swap
                i, j = np.random.choice(n_points, size=2, replace=False)
                if best_x[i] != best_x[j]:
                    best_x[i], best_x[j] = best_x[j], best_x[i]
                    new_c5 = compute_overlap_from_binary(best_x)
                    if new_c5 < old_c5 or np.random.random() < np.exp(-(new_c5 - old_c5) / T):
                        accepted += 1
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            last_improvement = steps
                    else:
                        # roll it back
                        best_x[i], best_x[j] = best_x[j], best_x[i]
        else:
            # random swap
            i, j = np.random.choice(n_points, size=2, replace=False)
            if best_x[i] != best_x[j]:
                best_x[i], best_x[j] = best_x[j], best_x[i]
                new_c5 = compute_overlap_from_binary(best_x)
                if new_c5 < old_c5 or np.random.random() < np.exp(-(new_c5 - old_c5) / T):
                    accepted += 1
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        last_improvement = steps
                else:
                    # roll it back
                    best_x[i], best_x[j] = best_x[j], best_x[i]
                    
        num_accepts += 1
        
        # Cooling and stagnation handling
        if num_accepts % 1000 == 0:
            T = max(T_final, T - cooling_rate)
        if steps - last_improvement > 20000:
            # Inject diversity
            num_flip = int(n_points * 0.05)
            flip_indices = np.random.choice(n_points, size=num_flip, replace=False)
            for idx in flip_indices:
                best_x[idx] = 1 - best_x[idx]
            # Restore exactly
            ones_count = np.sum(best_x)
            if ones_count != TARGET_SUM:
                # Find indices to flip
                flip_indices = np.random.choice(n_points, size=int(np.abs(ones_count - TARGET_SUM)), replace=False)
                for idx in flip_indices:
                    best_x[idx] = 1 - best_x[idx]
        if steps % 5000 == 0:  # Check time budget
            if (time.time() - start_time) > (budget_s - 5):
                break
    return best_x, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: Binary Simulated Annealing
    x_init = make_binary_initial(n_points, TARGET_SUM)
    best_x, best_c5 = local_improve_binary(x_init, budget_s)
    
    # Phase 2: Continuous refinement
    h_init = best_x.astype(float)  # Convert to float for continuous case
    h_final = project_to_sum_box(h_init, TARGET_SUM)
    
    # Use scipy optimizer for high-precision
    num_opt_iters = 1500
    try:
        # Bring back to valid space and refine
        final_h = project_to_sum_box(h_final, TARGET_SUM)
        final_c5 = compute_overlap_for_float(final_h)
        
        # Smooth and successfully reproject
        smoothed_h = np.convolve(final_h, [1/3.]*3, mode='same')
        h_post_smooth = project_to_sum_box(smoothed_h, TARGET_SUM)
        c5_post_smooth = compute_overlap_for_float(h_post_smooth)
        if c5_post_smooth < final_c5:
            final_c5 = c5_post_smooth
            final_h = h_post_smooth
        
        # Final projection to avoid numerical issues
        final_h = project_to_sum_box(final_h, TARGET_SUM)
        final_c5 = compute_overlap_for_float(final_h)
        
        return (final_h.tolist(), float(final_c5), n_points)
        
    except Exception:
        return (h_final.tolist(), float(best_c5), n_points)
