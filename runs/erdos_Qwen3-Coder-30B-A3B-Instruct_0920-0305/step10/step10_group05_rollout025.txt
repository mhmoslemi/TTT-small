# Global constants
n_points = 200
dx = 2.0 / n_points
half_size = n_points // 2
target_sum = n_points / 4

def autocorr_fft(h):
    """Fast computation of autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(h, L)).real
    return corr

def compute_c5_autocorr(corr, dx):
    """Convert autocorrelation to C5 value"""
    return 1 - dx * np.min(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    x = np.array(x)
    sum_x = np.sum(x)
    if sum_x == target_sum:
        return x.copy()
    elif sum_x < target_sum:
        # Need to increase values
        diff = target_sum - sum_x
        # This is a hack for correctness. A proper projection is non-trivial
        # but the current code mostly works since we're close to valid
        # We'll just set the values to ensure sum is correct
        x = np.clip(x + diff / len(x), 0, 1)
    else:
        # Need to decrease values
        diff = sum_x - target_sum
        # This is also approximate, but sufficient for now
        x = np.clip(x - diff / len(x), 0, 1)
    # Ensure it's exactly target sum
    x_sum = np.sum(x)
    if abs(x_sum - target_sum) > 1e-10:
        x = x * target_sum / x_sum
    return x

def parametric_function(x, a, p, n_points):
    """Generate piecewise constant function based on parameters a and p"""
    # Take only first half
    half = x[:half_size]
    half_func = np.zeros_like(half)
    # Edge blocks (two of them on each side)
    edge_len = int(a * half_size)
    if edge_len < 1:
        edge_len = 1
    plate_len = half_size - 2 * edge_len
    if plate_len < 0:
        plate_len = 0
    if edge_len > 0:
        half_func[:edge_len] = 1.0
        half_func[-edge_len:] = 1.0
    if plate_len > 0:
        half_func[edge_len:half_size-edge_len] = p
    # Ensure it sums to target_sum
    current_sum = np.sum(half_func)
    if current_sum > 0:
        half_func *= target_sum / current_sum
    return half_func

def smooth_objective(h, beta=50.0):
    """Smooth approximation of C5 to get gradient"""
    L = len(h)
    h_rev = h[::-1]
    L_full = 2 * L - 1
    # Compute A_k (autocorrelation values)
    corr = ifft(fft(h_rev, L_full) * fft(h, L_full)).real
    # Use log-sum-exp for smoothing the minimum
    corr_shifted = corr - np.max(corr)  # Prevent overflow
    weights = np.exp(beta * corr_shifted)
    # sum of exponentials
    sum_weights = np.sum(weights)
    # smooth max
    smooth_max = np.log(sum_weights) / beta + np.max(corr)
    return 1 - dx * smooth_max

def smooth_gradient(h, beta=50.0):
    """Gradient of smooth objective w.r.t. h"""
    L = len(h)
    h_rev = h[::-1]
    L_full = 2 * L - 1
    corr = ifft(fft(h_rev, L_full) * fft(h, L_full)).real
    
    corr_shifted = corr - np.max(corr)
    weights = np.exp(beta * corr_shifted)
    sum_weights = np.sum(weights)
    # This is a numerical derivation of gradient.
    # Symbolically: d_obj/dh_k = -dx * (sum w_i * dA_k/dh_k)/sum w_i
    # Where dA_k/dh_i = h_{i+k} + h_{i-k} for i around k (zero padded)
    
    # Convolution (numerical derivative formulation):
    # Compute sum of weighted dA_k/dh_q terms
    # Compute all the derivatives of A_k at each point h_q
    # But we recompute manually using current h to avoid matrix of size n^2
    
    # simpler: compute gradient directly from sum of exp(weights) * weight* 
    # assumes A_k is a function of all h_i ->
    # approx gradient = -dx / (sum w * h) * (weight elements that give dA_k/dh_q respectively)
    # Used approach: avg with everything fixed, compute full derivative tensor of A_k terms and weights
    
    # Since the gradient for each index is:
    # sum_k w_k * h_{i+k} + h_{i-k}
    # we need to compute sum over k of w_k*h_{i+k} + w_k*h_{i-k}
    
    # Consider convolution 
    g = np.zeros_like(h)
    for i in range(len(h)):
        for k in range(len(h)):
            # h_{i+k} + h_{i-k}
            if (i+k < len(h)) and (i-k >= 0):
                g[i] += weights[k] * (h[i+k] + h[i-k])
            elif (i+k < len(h)):
                g[i] += weights[k] * h[i+k]
            elif (i-k >= 0):
                g[i] += weights[k] * h[i-k]
    # Just compute direct based on correlation values
    # because A_i = sum_j h_j h_{i+j} (circular modulo length)
    # we get dA_i / d h_j = h_{i+j} (and for the -j term too)
    # so gradient is sum_k w_k * (h_{k+i} + h_{k-i}) = 
    # This can be computed using convolution 
    # Let's reuse a more standard approach: 
    # weighted autocorrelation of h with itself (or convolution of different shifted copies)

    # Let's project gradient directly 
    g = np.zeros_like(h) 
    # Compute via piece wise derivative identity
    # Let's compute for all k = 0..L
    # The weight tensor is w*delta_{i-j} where the derivative of A_k is 
    # h_{i+k} + h_{i-k} but only if indices are valid. Recomputing standard gradient:
    # if i = 0: gradient w.r.t h_0 = sum_k w_k*(h_k + h_{-k}) 
    # But since we zero pad, the second component is 0 for k>0. But also it's shifted
    # Using standard autocorrelation interpretation:
    # dA_k/dh_i = h_{i+k} + h_{i-k} 
    # So g_i = sum_k w_k * (h_{i+k} + h_{i-k}) = 2*(h*h)_k 
    # where the autocorrelation (loop in code) is 0-based shifted forward but note
    # Real argument: Compute convolution of h and weights to get value at each index.
    # We use convolution theorem for gradient.

    # Naive: manually derive best approximator
    # Do a 2D stretch of convolution matrix for kernel.  
    # Instead compute via point-wise derivative rules or through explicit indexerutils
    
    # Linear approach, don't be too precise, use estimate gradient form of direct convolution.
    # formulation using explicit and applying same logic correctly
    # A_k = sum_i h_i h_{i+k}.  dA_k/dh_i = h_{i+k}  
    # We compute such value for k = 0..L-1 and for each i compute the gradient of that. 
    # So for case of each i:
    # grad = sum_k w_k * dA_k/dh_i 
    # So grad_i = sum_k w_k * (h_{i+k} + h_{i-k} terms only valid ones)  

    # For data interpretation of arrays
    # h identified as h_new = h_rev
    # Let's go from the other direction:
    # compute weight*dA_k/dh_i for every i, 
    # Sum over k
    # Derivative matrix d(A_k)/d(h_i) -> equivalent to matrix multiplication  
    #    dA_k/dh_i = h_{i+k} + h_{i-k}
    # -> This is indexed by two variables k, i -> overall gradient vector.  
    # But let's take a shortcut and embed the desired gradient as a linear convolution trick
    
    # compute step-by-step: 
    # start with proper positional derivatives
    # g_i for each i = sum_k w_k *(h_{i+k} + h_{i-k})  
    # Linearize this with FFT. Note mirroring rules.
    
    # Work around it with the simple approach: 
    # clone weights -> no need to do a funky convolution. 
    # Instead use back-computation  
    # Just per each i sum over k weights_k * h_{i+k} + weights_k * h_{i-k} 
    # This also involves boundary conditions.
    # Let's take a common ground approach: We can apply convolution using as simple cycles
    # To match input size. Accurate computation with periodic padding
    # let’s avoid strange library complications, just do loops and create vectors
        
    # Redefine convolution approach:
    # The gradients is weighted sum where:
    # sig_k = sum_i intra[weight_i * h_{i+k} + weight_i * h_{i-k}]
    # so the derivative = [h_conv_weight + h_conv_reversed_weight]  
    # Convolution it correctly with boundary checks
    # We'll use computed correl maybe just directly 
    # In a small setting with many operations, let’s just parameterize-zeros assumption
    
    g = np.zeros_like(h)
    for i in range(len(h)):
        grad_term = 0
        for k in range(len(h)):
            if i+k < len(h):
                grad_term += weights[k] * h[i+k] 
            if i-k >= 0:
                grad_term += weights[k] * h[i-k]
        g[i] = grad_term
    g = g / sum_weights  # Normalize
    g = -dx * g
    return g

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # ============ INITIALIZE VECTOR =============
    h_full = np.copy(initial_h_values)
    h_full = np.clip(h_full, 0, 1)
    if abs(np.sum(h_full)*dx - 1) > 1e-10:
        h_full = h_full * (1./np.sum(h_full))  

    # Partition vector as symmetric
    h_half = h_full[:half_size]

    # ============= PLOT SCANNING ===============
    # Try parametric shapes: 
    best_h_half = h_half.copy()
    best_c5 = compute_c5_autocorr(autocorr_fft(np.concatenate([h_half,h_half[::-1]])), dx)

    # Sweep a to find good shape but in few steps with clean math
    steps = np.arange(0.01, 0.49, 0.005)
    best_a = 0.0
    best_p = 1.0
    found_best = False

    for a in steps:
        p = (1.0 - 2*a) / (2.0 * (1-a))
        # Generate smooth h_half using linear interpolation over piecewise constant
        if (abs(p) > 1e6): continue
        # use F array to sample estimates of the shape
        new_h = parametric_function(np.ones(half_size), a, p, half_size)
        temp_h = new_h.copy()
        temp_h = project_to_sum_box(temp_h, target_sum)
        full_vector = np.concatenate([temp_h, temp_h[::-1]])
        corr = autocorr_fft(full_vector)
        c5 = compute_c5_autocorr(corr, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h_half = temp_h.copy()
            best_a = a
            best_p = p
            found_best = True

    # ============= REFINE WITH L-BFGS ============
    def obj_model(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = autocorr_fft(h_full)
        return compute_c5_autocorr(corr, dx)
 
    def constraint_eq(h):
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(half_size)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Do multiple restarts for L-BFGS
    for _ in range(3):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        h_init = project_to_sum_box(best_h_half, target_sum) 
        try:
            res = minimize(
                obj_model, 
                h_init, 
                method='L-BFGS-B', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_refined = res.x.copy()
            else: 
                h_refined = h_init
        except Exception:
            h_refined = h_init
            
        h_refined = project_to_sum_box(h_refined, target_sum)
        h_full_refined = np.concatenate([h_refined, h_refined[::-1]])  
        corr = autocorr_fft(h_full_refined)
        c5_refined = compute_c5_autocorr(corr, dx)
        if c5_refined < best_c5:
            best_c5 = c5_refined  
            best_h_half = h_refined.copy()
            
    # ============= SIMULATED ANNEALING =============
    if time.time() - start_time < budget_s * 0.95:
        def sa_move(h, delta):
            i = np.random.randint(0, half_size)
            j = np.random.randint(0, half_size)
            if i == j:
                return h.copy()
            # Choose a move amount
            d = np.random.uniform(0, delta)
            if h[i] - d >= 0 and h[j] + d <= 1:
                h_new = h.copy()
                h_new[i] -= d
                h_new[j] += d
                return h_new
            return h.copy()
            
        current_h = best_h_half.copy()
        current_c5 = compute_c5_autocorr(autocorr_fft(np.concatenate([current_h, current_h[::-1]])), dx)
        
        # Initial settings
        temp = 0.01
        delta_max = 0.05
        accept_ratio = 0.2  # We'll track acceptance rate
        accept_count = 0
        total_count = 0
        
        for iteration in range(100000):
            if time.time() - start_time > budget_s - 10:
                break
                
            new_h = sa_move(current_h, delta_max)
            new_h = project_to_sum_box(new_h, target_sum)
            new_full = np.concatenate([new_h, new_h[::-1]])
            new_c5 = compute_c5_autocorr(autocorr_fft(new_full), dx)

            if new_c5 < current_c5:
                current_h = new_h.copy()
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h_half = current_h.copy()
            else:
                # Metropolis acceptance
                p = np.exp((current_c5 - new_c5)/temp)
                if np.random.rand() < p:
                    current_h = new_h.copy()
                    current_c5 = new_c5
                    accept_count += 1
                    
            temp *= 0.9995
            if temp < 1e-6:
                break
                
            total_count += 1
            if total_count % 1000 == 0:
                if total_count > 0:
                    accept_ratio = accept_count / total_count
                if accept_ratio > 0.4:
                    delta_max *= 1.1
                elif accept_ratio < 0.1:
                    delta_max *= 0.9
                delta_max = np.clip(delta_max, 1e-4, 0.2)
                
    # ============= FINAL L-BFGS =============
    if time.time() - start_time < budget_s - 5:
        def obj_smooth(h):
            # Smoothing to get gradient
            h_full = np.concatenate([h, h[::-1]])
            corr = autocorr_fft(h_full)
            # Return simple smooth objective
            return 1 - dx * np.min(corr)
            
        bounds = [(0, 1) for _ in range(half_size)]
        try:
            res = minimize(
                obj_smooth, 
                best_h_half, 
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_final = res.x.copy()
            else: 
                h_final = best_h_half
        except Exception:
            h_final = best_h_half
            
        h_final = project_to_sum_box(h_final, target_sum)
        h_full_final = np.concatenate([h_final, h_final[::-1]])
        corr = autocorr_fft(h_full_final)
        best_c5 = compute_c5_autocorr(corr, dx)


    # Reconstruct full function
    final_h_half = best_h_half
    final_h_full = np.concatenate([final_h_half, final_h_half[::-1]])

    return (final_h_full.tolist(), float(best_c5), n_points)
