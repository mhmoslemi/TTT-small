def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_binary_milp(N, time_limit=60):
    """Solve binary MILP for optimal pattern"""
    # Create variables
    x = cp.Variable(N, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(x) == N // 2]
    
    # For each shift k, compute sum of products
    for k in range(N):
        if k == 0:
            # For shift 0, it's always N/2 (as it would be sum of x_i^2 = sum of x_i)
            # Since x is binary, we can skip or handle separately
            continue
        else:
            # Compute sum of products x[i] * x[i+k] for valid pairs
            y = cp.Variable(N - k, boolean=True)
            # Add constraints for each product
            for i in range(N - k):
                constraints.extend([
                    y[i] <= x[i],
                    y[i] <= x[i + k],
                    y[i] >= x[i] + x[i + k] - 1
                ])
            # Add constraint that sum of y must be at least N/2 - t
            constraints.append(cp.sum(y) >= N // 2 - t)
    
    # Objective: minimize t
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    # Try solving with GLPK_MI
    try:
        prob.solve(solver=cp.GLPK_MI, time_limit=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value
    except:
        pass
    
    return None

def upsample_binary(binary_vec, target_N):
    """Upsample binary vector to target_N"""
    upsample_factor = target_N // len(binary_vec)
    if upsample_factor <= 0:
        return binary_vec
    upsampled = np.repeat(binary_vec, upsample_factor)
    # Truncate or pad to exact target_N
    if len(upsampled) > target_N:
        upsampled = upsampled[:target_N]
    elif len(upsampled) < target_N:
        upsampled = np.pad(upsampled, (0, target_N - len(upsampled)), 'constant')
    return upsampled

def local_improvement_binary(x, time_budget, start_time):

    # Compute initial correlation

        overlaps = np.zeros(N)

            overlap = 0

                overlap += x[i] * x[i + k]
            overlaps[k] = overlap
        return overlaps
    
    overlaps = compute_overlap()
    max_d = N // 2 - np.min(overlaps)
    
    # Keep track of best solution
    best_x = x.copy()
    best_d = max_d
    
    # Iteratively improve
    while time.time() - start_time < time_budget * 0.8:
        # Find worst shift k_max
        k_max = np.argmin(overlaps)
        if overlaps[k_max] >= N // 2 - 1e-6:  # No improvement possible
            break
            
        # Find candidates for swapping
        candidates_i = []
        candidates_j = []
        
        for i in range(N - k_max):
            if x[i] == 1 and x[i + k_max] == 0:
                candidates_i.append(i)
                
        for j in range(N - k_max):
            if x[j] == 0 and x[j + k_max] == 1:
                candidates_j.append(j)
                
        if not candidates_i or not candidates_j:
            break
            
        # Try random swap
        i = np.random.choice(candidates_i)
        j = np.random.choice(candidates_j)
        
        # Make swap
        new_x = x.copy()
        new_x[i] = 0
        new_x[j] = 1
        
        # Compute new overlaps
        new_overlaps = np.zeros(N)
        for k in range(N):
            overlap = 0
            for l in range(N - k):
                overlap += new_x[l] * new_x[l + k]
            new_overlaps[k] = overlap
            
        new_d = N // 2 - np.min(new_overlaps)
        
        if new_d <= best_d:
            x = new_x
            overlaps = new_overlaps
            best_x = x.copy()
            best_d = new_d
            
        # A few random swaps to escape local minima
        for _ in range(3):
            if time.time() - start_time > time_budget * 0.8:
                break
            candidates_i = []
            candidates_j = []
            for i in range(N - k_max):
                if x[i] == 1 and x[i + k_max] == 0:
                    candidates_i.append(i)
            for j in range(N - k_max):
                if x[j] == 0 and x[j + k_max] == 1:
                    candidates_j.append(j)
            if candidates_i and candidates_j:
                i = np.random.choice(candidates_i)
                j = np.random.choice(candidates_j)
                new_x = x.copy()
                new_x[i] = 0
                new_x[j] = 1
                new_overlaps = np.zeros(N)
                for k in range(N):
                    overlap = 0
                    for l in range(N - k):
                        overlap += new_x[l] * new_x[l + k]
                    new_overlaps[k] = overlap
                new_d = N // 2 - np.min(new_overlaps)
                if new_d <= best_d:
                    x = new_x
                    overlaps = new_overlaps
                    best_x = x.copy()
                    best_d = new_d
    
    return best_x

def differential_evolution_refinement(h_initial, time_budget, start_time):
    """Use differential evolution to refine continuous solution"""
    def objective(h):
        return evaluate_c5(h)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(len(h_initial))]
    
    # Ensure sum constraint by scaling
    target_sum = len(h_initial) / 2.0
    
    def constraint_eq(h):
        return np.sum(h) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = differential_evolution(
            objective, 
            bounds,
            constraints=cons,
            maxiter=int(time_budget * 10),
            popsize=20,
            seed=42,
            mutation=(0.5, 1.0),
            recombination=0.7,
            atol=1e-8,
            rtol=1e-8
        )
        return project_to_sum_box(res.x, target_sum)
    except:
        return h_initial
