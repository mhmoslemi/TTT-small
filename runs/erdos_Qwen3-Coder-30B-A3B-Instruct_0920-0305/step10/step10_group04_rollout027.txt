```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_vectorized_binary(x):
    """Vectorized computation of correlation using FFT for binary vector x"""
    x_rev = x[::-1]
    L = 2 * len(x) - 1
    corr = np.fft.ifft(np.fft.fft(x_rev, L) * np.fft.fft(x, L)).real
    return corr

def c5_from_corr_vectorized(corr):
    """Given full correlation vector, return max value scaled by dx"""
    return dx * np.max(corr)

def c5_eval_binary(x):
    """Fast evaluation of C₅ for binary vector x"""
    if not isinstance(x, np.ndarray):
        x = np.array(x)
    corr = compute_corr_vectorized_binary(x)
    return c5_from_corr_vectorized(corr)

def compute_correlation_update_vectorized(x, i, j):
    """Vectorized computation of change in correlation vector if we swap x[i] and x[j]"""
    x_pad = np.concatenate(([0]*(n_points-1), x.tolist(), [0]*(n_points-1)))
    if i < j:
        # x[i] gets 0, x[j] gets 1
        off_contrib = -x_pad[i + n_points - 1:i + 3 * n_points - 2] - x_pad[i:n_points + i - 1][::-1]
        on_contrib = x_pad[j + n_points - 1:j + 3 * n_points - 2] + x_pad[j:n_points + j - 1][::-1]
    else:
        # x[j] gets 0, x[i] gets 1
        off_contrib = -x_pad[j + n_points - 1:j + 3 * n_points - 2] - x_pad[j:n_points + j - 1][::-1]
        on_contrib = x_pad[i + n_points - 1:i + 3 * n_points - 2] + x_pad[i:n_points + i - 1][::-1]
    delta_corr = off_contrib + on_contrib
    return delta_corr

def project_to_box_sum(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_restraint_refinement(x):
    """Apply local SLSQP refinement to the continuous version of x"""
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        return -dx * np.max(corr)
        
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - TARGET_SUM}
    
    try:
        res = minimize(objective, x, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False})
        if res.success:
            h_refined = res.x
            h_refined = project_to_box_sum(h_refined, TARGET_SUM)
            return h_refined
    except:
        pass
    return x

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def eval_h(h):
        return c5_eval_binary(h)
    
    # Initialize binary vector with balanced ones and zeros
    x_init = np.zeros(n_points, dtype=int)
    x_init[:n_points//2] = 1
    np.random.shuffle(x_init)
    
    best_x = x_init.copy()
    best_c5 = eval_h(best_x)
    
    # Simulated Annealing parameters
    T0 = 2 * dx
    Tf = 1e-6 * dx
    niters = 300000  # About 420 seconds, assuming ~1k swaps/second
    temp_schedule = np.linspace(T0, Tf, niters)
    
    # Initialize temperature counter
    tidx = 0
    
    # SA loop
    accepted_moves = 0
    stagnation_count = 0
    best_so_far = best_c5
    S = 20000  # Re-randomization interval
    rho = 0.05  # Fraction of bits to flip
    
    for it in range(niters):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Decide move type
        if np.random.random() < 0.6:
            # Targeted move
            corr_full = compute_corr_vectorized_binary(best_x)
            d_star = np.argmax(corr_full)
            d_star = d_star - (n_points - 1)  # Convert index back to offset
            
            # Build candidate sets U and V
            U = []
            V = []
            if d_star >= 0:
                for i in range(n_points - d_star):
                    if best_x[i] == 1 and best_x[i + d_star] == 0:
                        U.append(i)
                for j in range(n_points - d_star):
                    if best_x[j] == 0 and best_x[j + d_star] == 1:
                        V.append(j)
            else:
                for i in range(-d_star, n_points):
                    if best_x[i] == 1 and best_x[i + d_star] == 0:
                        U.append(i)
                for j in range(-d_star, n_points):
                    if best_x[j] == 0 and best_x[j + d_star] == 1:
                        V.append(j)
            
            if U and V:
                i = np.random.choice(U)
                j = np.random.choice(V)
            else:
                # Fallback to random swap
                i, j = np.random.choice(n_points, 2, replace=False)
                while best_x[i] == best_x[j]:
                    i, j = np.random.choice(n_points, 2, replace=False)
        else:
            # Random swap
            i, j = np.random.choice(n_points, 2, replace=False)
            while best_x[i] == best_x[j]:
                i, j = np.random.choice(n_points, 2, replace=False)
        
        # Propose the move
        x_trial = best_x.copy()
        x_trial[i], x_trial[j] = x_trial[j], x_trial[i]
        
        # Compute change in correlation updates
        delta_corr = compute_correlation_update_vectorized(best_x, i, j)
        corr_trial = corr_full + delta_corr
        trial_c5 = c5_from_corr_vectorized(corr_trial)
        
        # Accept or reject
        delta = trial_c5 - best_c5
        T = temp_schedule[tidx]
        tidx += 1
        
        if delta <= 0 or np.random.random() < np.exp(-delta / T):
            best_c5 = trial_c5
            best_x = x_trial.copy()
            accepted_moves += 1
            if trial_c5 < best_so_far:
                best_so_far = trial_c5
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            # Undo the move (already done since we copy on entry)
            pass
        
        # Check for stagnation and re-randomize
        if accepted_moves % S == 0 and stagnation_count > 0.1 * S:
            # Re-randomize fraction rho of bits
            num_flip = int(rho * n_points)
            flip_indices = np.random.choice(n_points, num_flip, replace=False)
            
            # Flip bits
            for idx in flip_indices:
                best_x[idx] = 1 - best_x[idx]
                
            # Ensure exact sum
            ones_count = np.sum(best_x)
            if ones_count != n_points // 2:
                to_flip = int(np.abs(ones_count - n_points // 2))
                # Find zeros and flip them
                zeros = np.where(best_x == 0)[0]
                if len(zeros) >= to_flip:
                    flip_indices = np.random.choice(zeros, to_flip, replace=False)
                    for idx in flip_indices:
                        best_x[idx] = 1
                else:
                    # Find ones to flip
                    ones = np.where(best_x == 1)[0]
                    flip_indices = np.random.choice(ones, to_flip, replace=False)
                    for idx in flip_indices:
                        best_x[idx] = 0
                        
            stagnation_count = 0
    
    # Convert to float for refinement
    h_float = best_x.astype(float)
    
    # Local refinement 1
    h_float = local_restraint_refinement(h_float)
    
    # Smooth slightly
    h_smooth = np.convolve(h_float, np.ones(3)/3, mode='same')
    # Re-project
    h_smooth = project_to_box_sum(h_smooth, TARGET_SUM)
    # Local refinement 2
    h_smooth = local_restraint_refinement(h_smooth)
    
    # Final projection to ensure sum constraint exactly
    h_final = project_to_box_sum(h_smooth, TARGET_SUM)
    
    # Evaluate final bound
    final_c5 = eval_h(h_final)
    
    return (h_final.tolist(), float(final_c5), n_points)
```
