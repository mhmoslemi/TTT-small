import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_initial_solution():
    """Create a more structured initial solution with strategic peaks"""
    h = np.zeros(n_points)
    
    # Place two main peaks to create strong suppression
    peak1_pos = n_points // 4
    peak2_pos = 3 * n_points // 4
    
    # Peak width parameter
    width = n_points // 10
    
    # Add first peak
    for i in range(max(0, peak1_pos - width//2), min(n_points, peak1_pos + width//2)):
        h[i] += 1.0 * np.exp(-0.5 * ((i - peak1_pos) / (width/4))**2)
    
    # Add second peak
    for i in range(max(0, peak2_pos - width//2), min(n_points, peak2_pos + width//2)):
        h[i] += 1.0 * np.exp(-0.5 * ((i - peak2_pos) / (width/4))**2)
    
    # Normalize to correct sum
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def local_search(h, max_iter=1000):
    """Perform local search with strategic moves"""
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for _ in range(max_iter):
        # Make a series of small changes
        h_new = best_h.copy()
        
        # Choose a random region to modify
        region_start = random.randint(0, n_points - 10)
        region_end = min(region_start + 10, n_points)
        
        # Modify within region
        for i in range(region_start, region_end):
            if random.random() < 0.5:
                delta = random.uniform(-0.05, 0.05)
                h_new[i] += delta
                h_new[i] = max(0, min(1, h_new[i]))
        
        # Project to constraint
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
    return best_h, best_c5

def evolutionary_operators(h1, h2):
    """Crossover and mutation operators"""
    # Uniform crossover
    child = np.where(np.random.rand(len(h1)) > 0.5, h1, h2)
    
    # Mutation
    for i in range(len(child)):
        if random.random() < 0.1:
            child[i] += random.gauss(0, 0.02)
            child[i] = max(0, min(1, child[i]))
            
    return project_to_sum_box(child, TARGET_SUM)

def evolutionary_search(h0, budget_s):
    """Evolutionary algorithm that will efficiently optimize the solution"""
    start_time = time.time()
    population_size = 20
    num_generations = 100
    best_solution = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Initialize population
    population = [h0.copy()]
    population_c5 = [best_c5]
    
    for _ in range(population_size - 1):
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = project_to_sum_box(h_rand, TARGET_SUM)
        population.append(h_rand)
        population_c5.append(evaluate_c5(h_rand))
    
    for gen in range(num_generations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Sort by fitness
        sorted_idx = np.argsort(population_c5)
        population = [population[i] for i in sorted_idx]
        population_c5 = [population_c5[i] for i in sorted_idx]
        
        if population_c5[0] < best_c5:
            best_solution = population[0].copy()
            best_c5 = population_c5[0]
        
        # Create new generation
        new_population = [population[0]]  # Elitism
        new_population_c5 = [population_c5[0]]
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection (select top 5)
            idx1 = sorted_idx[random.randint(0, 4)]
            idx2 = sorted_idx[random.randint(0, 4)]
            
            child = evolutionary_operators(population[idx1], population[idx2])
            new_population.append(child)
            new_population_c5.append(evaluate_c5(child))
        
        population = new_population
        population_c5 = new_population_c5
    
    return best_solution, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    h0 = generate_initial_solution()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # First phase: evolutionary search
    h_evolved, c5_evolved = evolutionary_search(h0, budget_s * 0.6)
    if c5_evolved < best_c5:
        best_h = h_evolved.copy()
        best_c5 = c5_evolved
    
    # Second phase: Local refinement with modified parameters
    h_refined, c5_refined = local_search(best_h, int(1000 * (1 - time.time() / (start_time + budget_s))))
    if c5_refined < best_c5:
        best_h = h_refined.copy()
        best_c5 = c5_refined
        
    # Third phase: Additional local search
    for _ in range(50):
        if time.time() > start_time + budget_s:
            break
        h_test = best_h + np.random.normal(0, 0.005, n_points)
        h_test = project_to_sum_box(h_test, TARGET_SUM)
        c5_test = evaluate_c5(h_test)
        if c5_test < best_c5:
            best_h = h_test.copy()
            best_c5 = c5_test
    
    return (best_h.tolist(), float(best_c5), n_points)
