import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_binary_milp(n, m):
    """Solve binary MILP to find optimal 0/1 pattern for max min autocorrelation"""
    # Variables
    y = cp.Variable(n, boolean=True)
    w = cp.Variable((n, n), integer=False)  # aux vars for products
    t = cp.Variable(1)
    
    # Constraints
    constraints = []
    constraints.append(cp.sum(y) == m)
    
    # Product constraints (w[k,i] = y[i]*y[i+k])
    for k in range(1, n):
        for i in range(n - k):
            constraints.extend([
                w[i, k] <= y[i],
                w[i, k] <= y[i+k],
                w[i, k] >= y[i] + y[i+k] - 1,
                w[i, k] >= 0
            ])
    
    # Min autocorrelation constraint
    for k in range(1, n):
        constraints.append(cp.sum(w[:, k]) >= t)
    
    # Objective
    objective = cp.Maximize(t)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.CBC, max_seconds=0.38 * 600)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            y_solution = y.value.round().astype(int)
            # Compute achieved min autocorrelation
            t_value = t.value
            return y_solution, t_value
    except:
        pass
    return None, None

def construct_from_binary(y):
    """Convert binary solution to continuous step function"""
    return y.astype(float)

def compute_subgradient(h, k_star):
    """Compute subgradient for the active shift k_star"""
    g = np.zeros_like(h)
    n = len(h)
    for i in range(n):
        left = h[i - k_star] if i - k_star >= 0 else 0
        right = h[i + k_star] if i + k_star < n else 0
        g[i] = 1 - left - right
    return g

def project_sum_box(h, target_sum):
    """Project onto sum constraint"""
    h = np.clip(h, 0, 1)
    diff = np.sum(h) - target_sum
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling projection
    def sth(x):
        return np.sum(np.clip(h + x, 0, 1)) - target_sum
    
    a, b = -1, 1
    for _ in range(50):
        mid = (a + b) / 2
        if sth(mid) * sth(a) < 0:
            b = mid
        else:
            a = mid
    
    return np.clip(h + (a + b)/2, 0, 1)

def deterministic_subgradient_descent(h0, max_iter=5000, timeout_sec=0.35 * 600):
    """Subgradient descent based on worst-shift derivative"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    for t in range(max_iter):
        if time.time() - start_time > timeout_sec:
            break
            
        # Compute correlation and find worst shift
        corr = compute_corr_fft(h)
        k_star = np.argmax(corr)
        
        # Compute subgradient
        g = compute_subgradient(h, k_star)
        
        # Update with diminishing step size
        alpha = 0.2 * dx / np.sqrt(t + 1)
        h_new = h - alpha * g
        
        # Project back
        h_new = project_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new
        
    return best_h, best_c5

def random_swap_move(h, target_sum):
    """Perform a random swap move to escape local minima"""
    h_new = h.copy()
    i, j = np.random.choice(len(h), size=2, replace=False)
    
    # Limit delta to maintain feasibility
    delta = np.random.uniform(-0.02, 0.02)
    h_new[i] += delta
    h_new[j] -= delta
    
    # Ensure feasibility
    h_new = np.clip(h_new, 0, 1)
    h_new = project_sum_box(h_new, target_sum)
    
    return h_new

def stochastic_refinement(h0, target_sum, timeout_sec=0.2 * 600):
    """Stochastic hill climbing with simulated annealing"""
    h_current = h0.copy()
    c5_current = evaluate_c5(h_current)
    best_h = h_current.copy()
    best_c5 = c5_current
    
    start_time = time.time()
    T = 0.1
    for iter in range(20000):
        if time.time() - start_time > timeout_sec:
            break
            
        # Random swap
        h_new = random_swap_move(h_current, target_sum)
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        else:
            # Simulated annealing acceptance
            delta = c5_new - c5_current
            if np.random.rand() < np.exp(-delta / T):
                h_current = h_new
                c5_current = c5_new
                
        # Cool down
        T = max(0.001, T * 0.9999)
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Global setup
    TARGET_SUM = n_points / 2
    best_h = None
    best_c5 = float('inf')
    
    # Phase 1: Binary MILP
    phase1_end = start_time + 0.4 * budget_s
    y_binary, t_opt = solve_binary_milp(n_points, TARGET_SUM)
    
    if y_binary is not None:
        h_initial = construct_from_binary(y_binary)
        c5_initial = dx * (TARGET_SUM - t_opt)
        if c5_initial < best_c5:
            best_h = h_initial.copy()
            best_c5 = c5_initial
    
    # Fallback to random if MILP failed
    if best_h is None:
        h_initial = np.random.uniform(0, 1, n_points)
        h_initial = project_to_sum_box(h_initial, TARGET_SUM)
        best_h = h_initial.copy()
        best_c5 = evaluate_c5(best_h)
    
    # Phase 2: Deterministic subgradient descent
    phase2_end = start_time + 0.75 * budget_s
    h_refined, c5_refined = deterministic_subgradient_descent(
        best_h, timeout_sec=phase2_end - time.time()
    )
    if c5_refined < best_c5:
        best_h = h_refined.copy()
        best_c5 = c5_refined
    
    # Phase 3: Stochastic refinement
    phase3_end = start_time + budget_s
    h_final, c5_final = stochastic_refinement(
        best_h, TARGET_SUM, timeout_sec=phase3_end - time.time()
    )
    if c5_final < best_c5:
        best_h = h_final.copy()
        best_c5 = c5_final
    
    return (best_h.tolist(), float(best_c5), n_points)
