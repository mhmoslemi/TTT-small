import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_with_gradients(h0, max_iter=1000):
    """Optimize using gradient-based method with projection"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def improve_solution_iterative(h0, max_iter=1000):

    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Small tweaks to few elements
        idxs = np.random.choice(n_points, size=min(15, n_points//4), replace=False)
        for idx in idxs:
            old_val = h[idx]
            # Try multiple small changes
            for delta in [-0.02, -0.01, 0.01, 0.02]:
                h_new = h.copy()
                h_new[idx] += delta
                
                if h_new[idx] < 0 or h_new[idx] > 1:
                    continue
                    
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                new_c5 = evaluate_c5(h_new)
                
                if new_c5 < best_c5:
                    h = h_new.copy()
                    best_c5 = new_c5
                    break
            else:
                continue
            break
        else:
            # If no improvement, perturb slightly
            if np.random.random() < 0.3:
                idx = np.random.randint(0, n_points)
                delta = np.random.uniform(-0.01, 0.01)
                h[idx] += delta
                h = project_to_sum_box(h, TARGET_SUM)
                best_c5 = evaluate_c5(h)
    
    return h

def insert_peaks(h):
    """Add two sharp peaks to make better overlap suppression"""
    # Locate areas with low values and add two strong spikes
    thresholds = [0.2, 0.5, 0.8]
    for thres in thresholds:
        mask = (h < thres)
        if np.sum(mask) > 10:

            indices = np.where(mask)[0]

                h[idx] = 1.0
                # Smooth out nearby values by reducing them slightly
                for offset in [-1, 0, 1]:
                    if 0 <= idx + offset < len(h):
                        h[idx + offset] = max(0, h[idx + offset] - 0.2)
            break
    
    return project_to_sum_box(h, TARGET_SUM)

def genetic_algorithm_search():
    """Use a simple genetic algorithm to explore the space"""
    population_size = 30
    generations = 50
    mutation_rate = 0.1
    
    # Initialize population with diverse samples
    population = []
    for _ in range(population_size):
        h = np.random.uniform(0, 1, n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    
    for gen in range(generations):
        # Evaluate fitness
        fitness = [evaluate_c5(h) for h in population]
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        
        # Keep top 50%
        new_population = population[:population_size//2]
        
        # Create new offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = population[np.random.randint(0, population_size//2)]
            parent2 = population[np.random.randint(0, population_size//2)]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points-1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if np.random.random() < mutation_rate:
                    child[i] += np.random.normal(0, 0.02)
            
            child = project_to_sum_box(child, TARGET_SUM)
            new_population.append(child)
        
        population = new_population
    
    # Return best individual
    fitness = [evaluate_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    return population[best_idx]
