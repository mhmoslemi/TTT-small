TARGET_SUM = None

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def update_correlation(c, h, i, delta_i, j, delta_j):
    """Update correlation efficiently after changing h[i] and h[j]"""
    n = len(h)
    
    # Compute updates for i
    if delta_i != 0:
        # Add contribution from b[i+s] terms
        c_slice_start = n - i - 1
        c_slice_end = 2 * n - i - 1
        c[c_slice_start:c_slice_end] += delta_i * h[::-1][i:2*n-1-i]
        
        # Subtract contribution from a[i-s] terms  
        c_slice_start = i
        c_slice_end = i + n
        c[c_slice_start:c_slice_end] -= delta_i * h[::-1][n-1::-1][n-1-i:n-1-i+n]

    # Compute updates for j
    if delta_j != 0:
        # Add contribution from b[j+s] terms
        c_slice_start = n - j - 1
        c_slice_end = 2 * n - j - 1
        c[c_slice_start:c_slice_end] += delta_j * h[::-1][j:2*n-1-j]
        
        # Subtract contribution from a[j-s] terms
        c_slice_start = j
        c_slice_end = j + n
        c[c_slice_start:c_slice_end] -= delta_j * h[::-1][n-1::-1][n-1-j:n-1-j+n]
    
    # Add cross terms from delta_i * delta_j
    if delta_i != 0 and delta_j != 0:
        # delta_i * delta_j contributes to shifts: j - i and i - j
        shift_ij = j - i + n - 1  # c index for shift j - i
        shift_ji = i - j + n - 1  # c index for shift i - j
        c[shift_ij] += delta_i * delta_j
        c[shift_ji] += delta_i * delta_j
