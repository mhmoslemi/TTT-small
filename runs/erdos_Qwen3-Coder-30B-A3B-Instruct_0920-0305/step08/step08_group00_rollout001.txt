import numpy as np
from scipy.signal import correlate
import time

def compute_corr_int(h):
    """Compute correlation using FFT and return integer values"""
    h_rev = h[::-1].astype(np.float64)
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    # round to nearest integer (small float errors)
    corr_int = np.rint(corr).astype(int)
    return corr_int

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_int(h)
    return np.max(corr) * 2.0 / len(h)

def simulated_annealing(n_points, time_budget):

        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
            
        i_one = np.random.choice(ones_idx)
        i_zero = np.random.choice(zeros_idx)
        
        # Swap

        cand_h[i_one] = 0

                break
                
            i_one = np.random.choice(ones_idx)
            i_zero = np.random.choice(zeros_idx)
            swaps.append((i_one, i_zero))
            
        improved = False
        for i_one, i_zero in swaps:
            cand_h = best_h.copy()
            cand_h[i_one] = 0

            if cand_max < best_max:

                best_max = cand_max
                improved = True
                no_improve = 0
                break

            no_improve += 1

                break
