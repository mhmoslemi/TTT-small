import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import math

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def random_binary_solution():
    """Generate a random binary solution with exactly TARGET_SUM ones"""
    h = np.zeros(n_points, dtype=float)
    ones_indices = np.random.choice(n_points, size=int(TARGET_SUM), replace=False)
    h[ones_indices] = 1.0
    return h

def swap_bits(h, i, j):
    """Swap bits at positions i and j and return new array"""
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new

def get_ones_zeros_indices(h):
    """Get indices of ones and zeros"""
    ones_indices = np.where(h == 1)[0]
    zeros_indices = np.where(h == 0)[0]
    return ones_indices, zeros_indices

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = None
    best_c5 = float('inf')
    
    # Simulated Annealing parameters
    initial_temp = 0.01
    final_temp = 0.0001
    max_iter = 100000
    cooling_rate = 0.999
    
    # Track improvement
    last_improvement = 0
    improvement_threshold = 5000
    
    # Reserve some time for final refinement
    remaining_budget = budget_s * 0.95
    
    restart_count = 0
    while time.time() - start_time < remaining_budget:
        # Generate random binary solution
        h = random_binary_solution()
        current_c5 = evaluate_c5(h)
        
        # Get indices for sampling
        ones_indices, zeros_indices = get_ones_zeros_indices(h)
        
        temp = initial_temp
        iteration = 0
        
        while iteration < max_iter and time.time() - start_time < remaining_budget:
            # Sample a 1 index and a 0 index to swap
            if len(ones_indices) > 0 and len(zeros_indices) > 0:
                i = np.random.choice(ones_indices)
                j = np.random.choice(zeros_indices)
                
                # Perform swap
                h_new = swap_bits(h, i, j)
                new_c5 = evaluate_c5(h_new)
                
                # Accept or reject
                if new_c5 < current_c5:
                    h = h_new
                    current_c5 = new_c5
                    ones_indices, zeros_indices = get_ones_zeros_indices(h)
                    
                    if new_c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = new_c5
                        last_improvement = iteration
                else:
                    # Accept with probability
                    if np.random.random() < math.exp((current_c5 - new_c5) / temp):
                        h = h_new
                        current_c5 = new_c5
                        ones_indices, zeros_indices = get_ones_zeros_indices(h)
            
            # Cool down
            temp *= cooling_rate
            iteration += 1
            
            # Check if we're stuck
            if iteration - last_improvement > improvement_threshold:
                break
        
        restart_count += 1
    
    # Final refinement with local search
    if best_h is not None:
        # Perform a few rounds of local search on best solution
        ones_indices, zeros_indices = get_ones_zeros_indices(best_h)
        for _ in range(1000):
            if len(ones_indices) > 0 and len(zeros_indices) > 0:
                i = np.random.choice(ones_indices)
                j = np.random.choice(zeros_indices)
                h_new = swap_bits(best_h, i, j)
                new_c5 = evaluate_c5(h_new)
                
                if new_c5 < best_c5:
                    best_h = h_new
                    best_c5 = new_c5
                    ones_indices, zeros_indices = get_ones_zeros_indices(best_h)
    
    # Return result
    if best_h is None:
        # Fallback to initial solution if nothing found
        best_h = random_binary_solution()
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
