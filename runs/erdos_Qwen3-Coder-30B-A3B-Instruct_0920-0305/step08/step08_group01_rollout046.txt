import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def build_grid_from_intervals(intervals, n):
    """Build binary grid from intervals on [0,2] with n points."""
    dx = 2.0 / n
    h = np.zeros(n)
    for s, e in intervals:
        start_idx = int(np.floor(s / dx))
        end_idx = int(np.ceil(e / dx))
        start_idx = max(0, start_idx)
        end_idx = min(n, end_idx)
        h[start_idx:end_idx] = 1
    return h

def evaluate_c5_fft(h):
    """Fast C5 computation using FFT."""
    dx = 2.0 / len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def decode_params(z, m, n):
    """Decode z into intervals such that total length = 1 and all in [0,2]."""
    epsilon = 1e-4
    l = epsilon + (1 - 2*epsilon) * z[:m]
    l = l / np.sum(l)  # normalize to sum to 1
    g = z[m:]  # gaps
    
    # Adjust the last gap so that the entire intervals fit in [0,2]
    cum_pos = [0]
    for i in range(m):
        cum_pos.append(cum_pos[-1] + l[i] + (g[i] if i < len(g) else 0))
    
    # Scale gaps if needed to fit exactly
    total_len = cum_pos[-1]
    if total_len > 2:
        scale = 2 / total_len
        l *= scale
        g *= scale
        cum_pos = [0]
        for i in range(m):
            cum_pos.append(cum_pos[-1] + l[i] + (g[i] if i < len(g) else 0))
    
    # Final adjustment if needed
    last_pos = cum_pos[-1]
    if last_pos < 2:
        # Add remaining space to last gap
        diff = 2 - last_pos
        if m > 0:
            g[-1] += diff
    
    intervals = []
    pos = 0
    for i in range(m):
        s = pos
        e = s + l[i]
        intervals.append((s, e))
        pos = e + (g[i] if i < len(g) else 0)
        
    return intervals

def objective(z, m, n):
    """Objective function for optimization."""
    intervals = decode_params(z, m, n)
    h = build_grid_from_intervals(intervals, n)
    c5 = evaluate_c5_fft(h)
    return c5

def local_refinement(intervals, n, max_iter=200):
    """Refine intervals by small shifts that improve max correlation."""
    dx = 2.0 / n
    best_intervals = intervals[:]
    best_c5 = evaluate_c5_fft(build_grid_from_intervals(best_intervals, n))
    
    for iteration in range(max_iter):
        improved = False
        for i in range(len(best_intervals)):
            s, e = best_intervals[i]
            length = e - s
            
            # Try moving the left edge
            for delta in [-dx/2, dx/2]:
                new_s = s + delta
                if new_s >= 0:
                    new_intervals = best_intervals[:]
                    new_intervals[i] = (new_s, new_s + length)
                    new_c5 = evaluate_c5_fft(build_grid_from_intervals(new_intervals, n))
                    if new_c5 < best_c5:
                        best_intervals = new_intervals[:]
                        best_c5 = new_c5
                        improved = True
                        
            # Try moving the right edge
            for delta in [-dx/2, dx/2]:
                new_e = e + delta
                if new_e <= 2:
                    new_intervals = best_intervals[:]
                    new_intervals[i] = (s, new_e)
                    new_c5 = evaluate_c5_fft(build_grid_from_intervals(new_intervals, n))
                    if new_c5 < best_c5:
                        best_intervals = new_intervals[:]
                        best_c5 = new_c5
                        improved = True
                        
        if not improved:
            break
        
    return best_intervals

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 180
    m = 5
    
    # Phase 1: Generate initial population from random intervals
    initial_pop = []
    for _ in range(20):
        z = np.random.rand(2 * m)
        initial_pop.append(z)
    
    # Phase 2: Differential Evolution Optimization
    bounds = [(0, 1)] * (2 * m)
    de_result = differential_evolution(
        objective, bounds, args=(m, n),
        popsize=20,
        maxiter=int(100 + budget_s * 0.6 / 360 * 100),
        mutation=0.8,
        recombination=0.9,
        seed=seed,
        callback=lambda x, convergence: None,
        disp=False
    )
    
    # Decode the best solution found by DE
    best_intervals = decode_params(de_result.x, m, n)
    
    # Phase 3: Local refinement
    best_intervals = local_refinement(best_intervals, n)
    best_c5 = evaluate_c5_fft(build_grid_from_intervals(best_intervals, n))
    
    # Phase 4: Fallback if needed
    if best_c5 >= 0.38092:
        best_c5 = 0.38092
        # Use random search for a potentially better solution
        best_h = None
        best_c5_fallback = 1.0
        for _ in range(5000):
            z = np.random.rand(2 * m)
            intervals = decode_params(z, m, n)
            h = build_grid_from_intervals(intervals, n)
            c5 = evaluate_c5_fft(h)
            if c5 < best_c5_fallback:
                best_c5_fallback = c5
                best_h = h.copy()
        if best_h is not None:
            best_c5 = best_c5_fallback
    
    # Final evaluation using the official correlation method
    final_h = build_grid_from_intervals(best_intervals, n)
    c5_final = evaluate_c5_fft(final_h)
    
    # Convert to the required format
    h_values = final_h.tolist()
    return (h_values, float(c5_final), n)
