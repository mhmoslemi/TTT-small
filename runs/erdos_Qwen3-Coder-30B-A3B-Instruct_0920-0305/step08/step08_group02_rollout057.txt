import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(ifft(fft(h_rev, L) * fft(b, L)))
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using bisection"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_milp_search(h0, N, M, budget_s):
    """Find optimal binary solution using MILP"""
    start = time.time()
    try:
        # Binary variables
        y = cp.Variable(N, boolean=True)
        # Auxiliary variables
        w = cp.Variable((N-1, N), nonneg=True)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(y) == M,
            t >= 0,
            t <= M
        ]
        
        # Product linearisation constraints
        for k in range(N-1):
            for i in range(N-k):
                constraints.extend([
                    w[k, i] <= y[i],
                    w[k, i] <= y[i+k],
                    w[k, i] >= y[i] + y[i+k] - 1,
                    w[k, i] >= 0
                ])
        
        # Minimum autocorrelation constraints
        for k in range(N-1):
            constraints.append(cp.sum(w[k, :N-k]) >= t)
        
        # Objective
        objective = cp.Maximize(t)
        
        # Solve with timeout
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, max_seconds=budget_s*0.9)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            y_solution = y.value
            t_opt = t.value
            h_opt = y_solution.astype(float)
            return h_opt, (M - t_opt)
        
    except Exception:
        pass
    return None, None

def greedy_binary_construction(N, M):
    """Create feasible binary initial solution with good autocorrelation"""
    y = np.zeros(N, dtype=int)
    pos = 0
    while np.sum(y) < M and pos < N:
        if np.sum(y) + 1 <= M:
            y[pos] = 1
        pos += 2  # Place 1s every 2 positions
    return y.astype(float)

def compute_gradient(h, k_star):
    """Compute sub-gradient of the worst shift"""
    n = len(h)
    g = np.zeros(n)
    for i in range(n):
        g[i] = 1 - h[(i+k_star) % n] - h[(i-k_star) % n]
    return g

def deterministic_descent(h0, M, max_time, base_step=0.01):
    """Deterministic sub-gradient descent with diminishing steps"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start = time.time()
    t = 1
    while time.time() - start < max_time:
        # Compute worst shift
        corr = compute_corr_fft(h)
        k_star = np.argmax(corr)
        
        g = compute_gradient(h, k_star)
        
        # Take step
        alpha = base_step / np.sqrt(t)
        h_new = h - alpha * g
        h_new = project_to_sum_box(h_new, M)
        
        # Evaluate
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new
        t += 1
        
        if t % 100 == 0 and time.time() - start > max_time * 0.95:
            break
    
    return best_h, best_c5

def simulated_annealing(h0, M, max_time):
    """Stochastic refinement using 2-index swaps and simulated annealing"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start = time.time()
    T_start = 5 * 0.01  # dx = 2/N = 0.01667 for N=120, using 0.01 as an approximation
    
    while time.time() - start < max_time:
        # Generate perturbation
        i, j = np.random.choice(len(h), size=2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        
        # Apply perturbation
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, M)
        
        # Evaluate
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            best_h = h.copy()
        else:
            T = T_start * (1 - (time.time() - start) / max_time)
            if T > 0 and np.random.random() < np.exp(-(c5_new - best_c5) / T):
                h = h_new.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    N = 120
    M = N // 2
    dx = 2.0 / N
    h0 = np.zeros(N)
    
    # Phase 1: Binary MILP or greedy construction
    binary_milp_time = 0.4 * budget_s
    
    h_bin, c5_bin = binary_milp_search(h0, N, M, binary_milp_time)
    
    if h_bin is None:
        h_bin = greedy_binary_construction(N, M)
        
    # Phase 2: Deterministic descent
    descent_time = 0.35 * budget_s
    h_cont, c5_cont = deterministic_descent(h_bin, M, descent_time, dx * 0.5)
    
    # Phase 3: Stochastic refinement
    sa_time = budget_s - binary_milp_time - descent_time
    
    h_final, c5_final = simulated_annealing(h_cont, M, sa_time)
    
    return (h_final.tolist(), float(c5_final), N)
