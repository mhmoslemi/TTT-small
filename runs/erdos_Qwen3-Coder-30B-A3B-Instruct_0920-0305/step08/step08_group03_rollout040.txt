import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def update_correlation(c, h, i, delta, a_indices, b_indices):
    """Efficiently update correlation when changing h[i] by delta"""
    # For i, we need to contributing shifts to correlation array
    # This handles one elements contributions to all shifts where overlaps happen
    # Since i gets delta, and we're computing h[i] * (1 - h[i+s]) changes, 
    # we need to find which correlations this affects
    
    # We need to track which elements will be used in correlations
    n = len(h)
    # Using the full correlation mode, shifts range from -(n-1) to n-1
    # But we don't need to compute exact positions every time, just focus on keys
    
    # We add contribution to shifts based on h[i] + delta 
    # The correlation i+s part: contributed to shift s is (i+s) in valid range [0,n-1]
    # shift s = i + delta such that continuous position valid
    # Let's update shift contributions explicitly
    
    # For n_points=256, this is faster than FFT every time
    # We'll handle updates per shift location
    
    # For strided access with slices:
    # delta_a contributes to c[n-i-1 : 2n-i-1] by +delta * b
    # delta_b contributes to c[i : i+n] by -delta * a_rev
    # But we modify a value and then compute how corrs change
    
    # Store relevant slices of updated arrays
    # For every element, explicit manual updating:
    # For element i, its contribution to c[shift] depends on its position in convolution:
    # c[i+s] += delta * b[i+s] 
    # c[i-s] -= delta * a[i-s] 
    
    # s ranges from (-i) to (n-1-i) for b effect
    # s ranges from (i-(n-1)) to i for a effect with b_j 1-delta contribution 
    # Since it's full convolution, we're just changing internal summation terms
    # Delta for modified h[i]:  
    # correlation effect with i + s evolution
    
    # We'll manually loop over shifts for efficiency, but efficiently:
    # But starting with how correlation changes from one change
    
    # Instead, we'll compute the direct correlation effects:
    # A change in h[i] by delta changes c[s] by delta * b[i+s] (forward) 
    # and its -= delta * a[i-s] (backward), when indices valid
    
    delta_i = delta
    # We compute contributions to correlation c from delta change in h[i]
    # and the corresponding product contributions from changes in (1-h) 
    # which means change in b[i]
    
    # These are "from actual convolution" fomula
    # delta contribution to c_r from element i with change:
    # Index r = 0 to 2n-2, shift = r - (n-1) 
    for shift_idx in range(2 * n - 1):
        shift = shift_idx - (n - 1)
        corr_idx = shift_idx
        # Forward contribution: if h[i] * b[i+s] contributes to c[shift_idx] 
        # For correlation c[s] (shift s = r - n + 1) in correlation array, 
        # i contributes to the term when i + shift is valid i+shift in [0, n-1]
        # That is shift in [ -i, n-1-i]
        # Therefore, at shift index: r = i + shift + n -1 
        # but i+shift is within [0, n-1] requires shift in [ -i, n-1-i]
        # i+shift <= n-1 => shift <= n-1-i  
        # i+shift >= 0 => shift >= -i
        # i ranges both forward and backward
        # To compute effects on effective correlation indices, 
        if -i <= shift <= n-1-i: # See if this shift allows contribution from i 
            old_val = c[corr_idx]
            # Subtract old effect and add new effect
            c[corr_idx] += delta_i * b[i+shift]  # add delta * (1-h[i+shift]) 
            # updated change should consider channels already considered, 
            # computing for h[i] 

    # Similar for reverse effect: if we're at shift from a[i-s] contribution
    # The coefficient a[i-s] contributes to c[shift] 
    # that is i-s ranges from 0 to n-1, so s ranges from i-(n-1) to i
    # shift indexed by corr_idx = s + n - 1 => s = corr_idx - n + 1
    for shift_idx in range(2 * n - 1):
        shift = shift_idx - (n - 1)
        if i - (n - 1) <= shift <= i:
            # contribution from change in h[i] affects via a[i-s] for s = shift, 
            # i-s = i-shift: this term should contribute to c[shift_idx]
            # Previous was: c[shift_idx] += delta_i * b[i+shift]
            # This is for coefficient a[i-s] in product a_j * b_j
            # Actually now we contribute: -delta_i * a[i-shift]
            # i-shift is valid: 0 <= i-shift <= n-1 => i-shift is in [0,n-1]
            # This means shift ranges from i-(n-1) to i = [i-n+1, i]   
            # This is valid.  
            # but I'm computing wrong sign behavior. Compensation should be  direct.
            # Get the index for value we actually compute change to average the mixerd:
            corr_idx = shift_idx
            c[corr_idx] -= delta_i * a_indices[i-shift]
    
    return c

                h[j] = h_old[j]

                b[j] = 1 - h[j]

        temperature *= cooling_rate
                
    return (best_h.tolist(), float(best_c5), n_points)
