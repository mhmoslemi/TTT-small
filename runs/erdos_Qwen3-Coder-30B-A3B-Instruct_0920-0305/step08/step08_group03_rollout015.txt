def generate_critical_shifts(m=5):
    """Generate all critical shifts for a piecewise constant function with m intervals"""
    # For m=5, we have 5 intervals. Total length 2.
    # Critical shifts are all |s_i - s_j| where s_i are cumulative start points
    # Also include boundaries 0, 2 and mirrored ones
    shifts = set()
    for i in range(m+1):  # interval endpoints (including 0 and 2)
        s_i = i * 2.0 / m
        for j in range(m+1):
            s_j = j * 2.0 / m
            shifts.add(abs(s_i - s_j))
    # Add endpoints
    shifts.add(0.0)
    shifts.add(2.0)
    return sorted(list(shifts))

def compute_overlap_integral(k, lengths, heights):
    """Compute overlap integral J(k) for a given shift k"""
    # lengths: m-1 values (last is determined)
    # heights: m values
    # k: shift value in [0,2]
    m = len(lengths) + 1
    # Recompute interval boundaries
    s = [0.0]
    for i in range(len(lengths)):
        s.append(s[-1] + lengths[i])
    s.append(2.0)
    # Compute linear term: int_0^{2-k} h(x) dx
    linear_term = 0.0
    for i in range(m):
        if s[i] >= 2 - k:
            break
        overlap_start = max(s[i], 0.0)
        overlap_end = min(s[i+1], 2.0 - k)
        if overlap_end > overlap_start:
            linear_term += heights[i] * (overlap_end - overlap_start)
    
    # Compute quadratic term: sum_{i,j} a_i a_j * overlap_length
    quadratic_term = 0.0
    for i in range(m):
        for j in range(m):
            # Interval i: [s[i], s[i+1])
            # Interval j (shifted by k): [s[j] + k, s[j+1] + k]
            overlap_start = max(s[i], s[j] + k)
            overlap_end = min(s[i+1], s[j+1] + k)
            if overlap_end > overlap_start:
                overlap_length = overlap_end - overlap_start
                quadratic_term += heights[i] * heights[j] * overlap_length
    
    return linear_term - quadratic_term

def compute_c5_from_params(lengths, heights):
    """Evaluate the maximum overlap integral over all critical shifts"""
    global critical_shifts
    if critical_shifts is None:
        critical_shifts = generate_critical_shifts(len(lengths) + 1)
    # Recompute the last length from the constraint
    total_length = sum(lengths)
    if abs(total_length - 2.0) > 1e-10:
        # Normalize to sum to 2
        lengths = [x * 2.0 / total_length for x in lengths]
    last_length = 2.0 - sum(lengths)
    
    # Recompute the last height from the mass constraint
    mass = sum(h * l for h, l in zip(heights, lengths)) + heights[-1] * last_length
    if abs(mass - 1.0) > 1e-10:
        # Scale heights to satisfy mass constraint
        scale = 1.0 / mass
        heights = [h * scale for h in heights]
    # Simplified approach: we'll manually compute the objective without directly normalizing here,
    # since the constraint is used in the optimization solver.
    
    # Evaluate J(k) for all k in critical_shifts and return the maximum
    if not critical_shifts:
        critical_shifts = generate_critical_shifts(len(lengths) + 1)
    values = [compute_overlap_integral(k, lengths, heights) for k in critical_shifts]
    return max(values)

def project_to_sum_box(x, target_sum, lower=0.0, upper=1.0):
    """Project vector x onto box constraints with sum = target_sum"""
    # Simple algorithm to project onto consistent constraints
    x = np.array(x)
    x = np.clip(x, lower, upper)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x
    factor = target_sum / current_sum
    x = factor * x
    return np.clip(x, lower, upper)
