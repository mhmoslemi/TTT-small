import numpy as np
from scipy.signal import correlate
import time


def project_to_capped_simplex(x, target_sum):
    """Project x onto the capped simplex: 0 <= x[i] <= 1, sum(x) = target_sum"""
    # Clip to [0,1]
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    
    if abs(current_sum - target_sum) < 1e-12:
        return x

    # Binary search for lambda to make sum equal to target
    a, b = -1.0, 1.0
    def eval_lambda(lam):
        y = np.clip(x + lam, 0, 1)
        return np.sum(y) - target_sum

    for _ in range(50):
        lam = (a + b) / 2
        val = eval_lambda(lam)
        if abs(val) < 1e-12:
            break
        if val * eval_lambda(a) < 0:
            b = lam
        else:
            a = lam

    lam_final = (a + b) / 2
    result = np.clip(x + lam_final, 0, 1)
    return result


def compute_correlation_incremental(h, b, c, i, j, delta):
    """Update correlation c after moving delta from i to j."""
    n = len(h)
    h_rev = h[::-1]
    
    # Update contribution from b (at index i)
    s1 = (n - 1) - i
    c[s1:(s1 + n)] += delta * b[i:(i + n) if i + n <= n else slice(i, n)] if i < n else 0
    
    # Update contribution from h_rev (at index i)
    c[i:(i + n)] -= delta * h_rev[0:(n - i) if i < n else 0] if i < n else 0
    
    # Same for index j
    s2 = (n - 1) - j
    c[s2:(s2 + n)] -= delta * b[j:(j + n) if j + n <= n else slice(j, n)] if j < n else 0
    
    # Update contribution from h_rev (at index j)
    c[j:(j + n)] += delta * h_rev[0:(n - j) if j < n else 0] if j < n else 0

    # Add quadratic correction (delta^2 term at shifts (j-i) and (i-j))
    # These have indices (j-i)+(n-1) and (i-j)+(n-1)
    shift1 = (j - i) + (n - 1)  # corresponds to shift = j - i
    shift2 = (i - j) + (n - 1)  # corresponds to shift = i - j
    if 0 <= shift1 < len(c):
        c[shift1] += delta * delta
    if 0 <= shift2 < len(c):
        c[shift2] += delta * delta

    return c

    eval_every = 500
    step_every = 100
    temp_every = 1000

    # Iteration loop
    while time.time() < max_time:
        # Propose move: transfer small amount from index i to index j
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        
        while i == j or h[i] < delta_step or h[j] > 1 - delta_step:
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)

        effective_delta = min(delta_step, h[i], 1 - h[j])
        h_i_old = h[i]
        h_j_old = h[j]
        
        # Apply the change in h
        h[i] -= effective_delta
        h[j] += effective_delta
        b[i] = 1 - h[i]
        b[j] = 1 - h[j]

        # Accept based on simulated annealing criterion
        if new_obj < current_obj or np.random.rand() < np.exp((current_obj - new_obj) / temperature):
            # Accept
            current_obj = new_obj
            if current_obj < best_value:

                no_improvement_count = 0
        else:

            h[i] = h_i_old
            h[j] = h_j_old
            b[i] = 1 - h[i]
            b[j] = 1 - h[j]

        # Update temperature every few steps
        temp_count += 1
        if temp_count >= temp_every:

            temp_count = 0
            
        # Adapt delta based on no improvement

        if no_improvement_count >= max_no_improvement:

                no_improvement_count = 0

            delta_step *= 0.95
            if delta_step < 0.0005:
                delta_step = 0.0005

    return (best_h.tolist(), float(best_value), n)
