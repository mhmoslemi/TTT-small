def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def piecewise_const_from_params(params, n_points):

    # Compute widths (Dirichlet-style)
    widths = np.exp(log_widths)
    widths = widths * 2.0 / np.sum(widths)
    
    # Compute heights (sigmoid)
    heights = 1 / (1 + np.exp(-heights_raw))
    heights = np.clip(heights, 0, 1)
    
    # Compute block boundaries
    cum_widths = np.concatenate([[0], np.cumsum(widths)])
    
    # Create discretized function

    block_indices = np.searchsorted(cum_widths, positions, side='right') - 1
    block_indices = np.clip(block_indices, 0, K-1)
    h = heights[block_indices]
    
    return h
