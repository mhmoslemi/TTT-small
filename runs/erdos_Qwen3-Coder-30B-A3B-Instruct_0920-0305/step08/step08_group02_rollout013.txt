import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_sum_box(v, M):
    """Projects vector v onto the set {x | sum(x)=M, 0 <= x <= 1}"""
    # First, clip to [0,1]
    v = np.clip(v, 0, 1)
    # Then, project onto sum=M using bisection
    a, b = -1.0, 1.0
    while True:
        lam = (a + b) / 2
        candidate = np.clip(v + lam, 0, 1)
        sum_candidate = np.sum(candidate)
        if abs(sum_candidate - M) < 1e-10:
            break
        if sum_candidate > M:
            a = lam
        else:
            b = lam
    return np.clip(v + (a + b) / 2, 0, 1)

def evaluate_c5(h):
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def solve_milp_binary(N, M):
    """Solve the binary MILP to maximize worst-case autocorrelation"""
    # Variables
    y = cp.Variable(N, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(y) == M]
    
    w_vars = []
    shift_sums = []
    
    for k in range(1, N):
        # w_{k,i} = y_i * y_{i+k} for i in range(N-k)
        w_k = cp.Variable(N - k)
        w_vars.append(w_k)
        
        # Linearization constraints
        constraints += [w_k <= y[:N-k]]
        constraints += [w_k <= y[k:]]
        constraints += [w_k >= y[:N-k] + y[k:] - 1]
        constraints += [w_k >= 0]
        
        shift_sum = cp.sum(w_k)
        shift_sums.append(shift_sum)
        constraints += [shift_sum >= t]
    
    # Objective
    prob = cp.Problem(cp.Maximize(t), constraints)
    
    # Try CBC first, then GLPK_MI, then ECOS_BB
    try:
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=190)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, verbose=False, max_seconds=190)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS_BB, verbose=False, max_seconds=190)
            except Exception:
                return None
    
    if prob.status in ["optimal", "optimal_inaccurate"]:
        y_sol = np.round(y.value).astype(int)
        return y_sol
    
    return None

def subgradient_descent(h0, M, max_iter=20000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for it in range(max_iter):
        corr = correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr) - (n_points - 1)
        
        # Compute gradient for the worst shift k_max
        grad = np.ones(n_points)
        if k_max >= 0:
            grad[:n_points - k_max] -= h[k_max:]
        else:
            k_abs = -k_max
            grad[k_abs:] -= h[:n_points - k_abs]
        
        if k_max > 0:
            grad[k_max:] -= h[:n_points - k_max]
        elif k_max < 0:
            k_abs = -k_max
            grad[:n_points - k_abs] -= h[k_abs:]
        
        # Gradient step
        alpha = 0.5 / np.sqrt(1 + it)
        h_new = h - alpha * grad
        
        # Project back
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, M)
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        
        h = h_new
        
        if it % 5000 == 0:
            pass  # Optionally print something like progress
    
    return best_h, best_c5

def random_perturbation(h0, M, max_iter=2000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Pick two indices at random that are not at bounds
        idx1 = np.random.randint(0, len(h))
        idx2 = np.random.randint(0, len(h))
        
        # Small change
        delta = np.random.uniform(-0.02, 0.02)
        h_new = h.copy()
        h_new[idx1] += delta
        h_new[idx2] -= delta
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, M)
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        elif np.random.rand() < 0.01:  # Accept worse solution sometimes
            h = h_new
        else:
            pass
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points / 2
    
    # Phase 1: Try MILP optimization
    milp_time = 0.4 * budget_s
    best_c5 = float('inf')
    best_h = None
    
    # Solve MILP for binary pattern
    milp_start = time.time()
    y_sol = solve_milp_binary(n_points, TARGET_SUM)
    milp_time = time.time() - milp_start
    
    if y_sol is not None:
        h0 = y_sol.astype(float)
        # Evaluate initial C5
        c5_init = evaluate_c5(h0)
        best_h = h0.copy()
        best_c5 = c5_init
    else:
        # Fallback to random initialization
        h0 = np.random.rand(n_points)
        h0 = project_sum_box(h0, TARGET_SUM)
        c5_init = evaluate_c5(h0)
        best_h = h0.copy()
        best_c5 = c5_init
        
    # Phase 2: Subgradient descent on continuous adaptation
    time_left = budget_s - milp_time
    sg_time = 0.35 * budget_s
    
    # Run subgradient descent
    sg_start = time.time()
    if best_h is not None:
        h_sgd, c5_sgd = subgradient_descent(best_h, TARGET_SUM, max_iter=10000)
        if c5_sgd < best_c5:
            best_c5 = c5_sgd
            best_h = h_sgd
    
    sg_time = time.time() - sg_start
        
    # Phase 3: Random refinement
    remaining_time = budget_s - milp_time - sg_time
    if remaining_time > 20:
        rp_start = time.time()
        h_rp, c5_rp = random_perturbation(best_h.copy(), TARGET_SUM, max_iter=1000)
        if c5_rp < best_c5:
            best_c5 = c5_rp
            best_h = h_rp
        
        rp_time = time.time() - rp_start
    
    return (best_h.tolist(), float(best_c5), n_points)
