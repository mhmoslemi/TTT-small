import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_sum_box(h, target_sum):
    """Project vector h onto [0,1] box with sum = target_sum"""
    def objective(lam):
        projected = np.clip(h + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while objective(a) * objective(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = objective(m)
        if abs(fm) < tol:
            break
        if fm * objective(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def binary_milp(N, M, max_seconds=200):
    """Solve binary MILP version of the problem"""
    # define binary variables
    y = cp.Variable(N, boolean=True)
    # auxiliary continuous variables for bilinear terms
    w = cp.Variable((N-1, N), nonneg=True)  
    # constraint on sum  
    constraints = [cp.sum(y) == M]
    
    # Product constraints: w[k,i] = y[i] * y[i+k]
    for k in range(N-1):
        for i in range(N-k):
            constraints += [
                w[k,i] <= y[i],
                w[k,i] <= y[i+k],
                w[k,i] >= y[i] + y[i+k] - 1,
                w[k,i] >= 0,
            ]
    
    # Lower bound on autocorrelations
    t = cp.Variable()
    for k in range(N-1):
        constraints += [cp.sum(w[k,:N-k]) >= t]
        
    # upper bound on t
    constraints += [t >= 0, t <= M]
    # objective
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CBC, max_seconds=max_seconds)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_EOF]:
            y_sol = y.value.astype(int)
            t_sol = t.value
            return y_sol, t_sol
        else:
            return None, None
    except:
        return None, None

def construct_h_step_function(y):
    """Convert binary solution back to h_step function"""
    return y.astype(float)

def compute_grad(active_shift, h):
    """Compute sub-gradient for the active shift"""
    grad = np.zeros_like(h)
    n = len(h)
    for i in range(n):
        left = h[(i - active_shift) % n] if active_shift > 0 else 0
        right = h[(i + active_shift) % n] if active_shift < n-1 else 0
        grad[i] = 1 - left - right
    return grad

def subgradient_descent(h0, max_time_s, alpha0=0.5*dx):
    """Deterministic sub-gradient descent with diminishing steps"""
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h)
    start_time = time.time()
    
    iter_count = 0
    while time.time() - start_time < max_time_s:
        # Find worst shift
        corr = np.correlate(h, 1-h, mode="full")
        active_shift = np.argmax(corr) - (len(h)-1)
        
        # Compute gradient
        grad = compute_grad(active_shift, h)
        
        # Update
        alpha = alpha0 / np.sqrt(iter_count + 1)
        h_new = h - alpha * grad
        
        # Project
        h_new = project_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new.copy()
        iter_count += 1
    return best_h, best_c5

def local_search(h0, max_time_s):
    """Simple local search using simulated annealing like swaps"""
    h = h0.copy()
    current_c5 = evaluate_c5(h)
    start_time = time.time()
    T = 0.05
    decay_rate = 0.9999
    n = len(h)
    
    while time.time() - start_time < max_time_s:
        # Pick two random indices
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
            
        # Try swap move
        delta = np.random.uniform(-0.02, 0.02)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Project and recompute
        h_new = project_sum_box(h_new, TARGET_SUM)
        new_c5 = evaluate_c5(h_new)
        
        # Accept or reject based on Metropolis criterion
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            h = h_new.copy()
            current_c5 = new_c5
            
        T *= decay_rate
        
    return h, current_c5

def initial_feasible_solution():
    """Construct a reasonable feasible starting solution"""
    h = np.ones(n_points) * 0.5
    h = project_sum_box(h, TARGET_SUM)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points / 2
    start_time = time.time()
    
    # Phase 1: Try MILP for binary version
    try:
        y_sol, t_opt = binary_milp(n_points, TARGET_SUM, max_seconds=budget_s * 0.4)
        if y_sol is not None and t_opt is not None:
            h_start = construct_h_step_function(y_sol)
            current_c5 = dx * (TARGET_SUM - t_opt)
        else:
            h_start = initial_feasible_solution()
            current_c5 = evaluate_c5(h_start)
    except:
        # fallback to random solution
        h_start = initial_feasible_solution()
        current_c5 = evaluate_c5(h_start)

    # Phase 2: Apply sub_gradient descent
    remaining_time = budget_s * 0.4  # 40% of total
    h_refined, c5_refined = subgradient_descent(h_start, remaining_time)
    if c5_refined < current_c5:
        current_c5 = c5_refined
        h_start = h_refined
    
    # Phase 3: Final local search with stochastic refinement
    final_time = budget_s - (time.time() - start_time)
    if final_time > 1:
        h_final, c5_final = local_search(h_start, final_time)
        if c5_final < current_c5:
            current_c5 = c5_final
            h_start = h_final
    
    return (h_start.tolist(), float(current_c5), n_points)
