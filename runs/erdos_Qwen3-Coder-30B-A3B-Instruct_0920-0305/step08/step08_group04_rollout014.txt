def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using waterfilling"""
    # Sort the vector in descending order
    sorted_indices = np.argsort(x)[::-1]
    sorted_x = x[sorted_indices]
    
    # Use waterfilling algorithm
    cumsum = np.cumsum(sorted_x)
    n = len(x)
    
    # Find the threshold
    threshold = 0.0
    for i in range(n):
        if (cumsum[i] + (n - i - 1)) <= target_sum:
            threshold = (target_sum - cumsum[i]) / (n - i - 1)
        else:
            threshold = (target_sum - cumsum[i] + sorted_x[i]) / (n - i)
            break
    
    # Apply threshold
    projected = np.clip(x, 0, 1)
    if np.sum(projected) > target_sum:
        # Adjust using the threshold
        for i in range(n):
            if projected[i] > threshold:
                projected[i] = threshold
    elif np.sum(projected) < target_sum:
        # Apply threshold to values that are below threshold
        for i in range(n):
            if projected[i] < threshold:
                projected[i] = threshold
        # Now adjust up to target sum
        diff = target_sum - np.sum(projected)
        if diff > 0:
            # Find the values that can be increased
            sorted_indices_rev = np.argsort(projected)
            for i in range(n):
                if diff <= 0:
                    break
                idx = sorted_indices_rev[i]
                available = 1.0 - projected[idx]
                increase = min(available, diff)
                projected[idx] += increase
                diff -= increase
    else:
        # Already matches, return as is
        pass
    
    # Ensure mass constraint is exactly met
    current_sum = np.sum(projected)
    if abs(current_sum - target_sum) > 1e-12:
        # Adjust for numerical error
        weighted_sum = np.sum(projected * np.arange(n))
        adjustment = (target_sum - current_sum) / n
        projected = np.clip(projected + adjustment, 0, 1)
    
    return projected

def rasterise_and_project(theta):
    """Convert parameters to a step function and project onto sum-constrained box"""
    # Parse the parameters: first K widths (with exp), then K heights (with sigmoid)
    w_params = theta[:n_blocks]
    a_params = theta[n_blocks:]
    
    # Compute widths
    u = np.exp(np.clip(w_params, -15, 15))  # Prevent overflow
    w = 2.0 * u / np.sum(u)
    
    # Compute heights
    a = 1.0 / (1.0 + np.exp(-a_params))
    
    # Create the step function
    h = np.zeros(n_points)
    cumulative = 0
    for i in range(n_blocks):
        start_idx = int(cumulative / dx)
        end_idx = int((cumulative + w[i]) / dx)
        h[start_idx:end_idx] = a[i]
        cumulative += w[i]
    
    # Apply projection to handle potential floating point issues
    h_projected = project_to_sum_box(h, TARGET_SUM)
    return h_projected

    h = rasterise_and_project(θ)
