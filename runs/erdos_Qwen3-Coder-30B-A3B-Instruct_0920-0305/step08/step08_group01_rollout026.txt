import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

# Global constants
n_points = 180
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def build_grid_from_intervals(intervals, n):
    """Build a binary grid from interval endpoints."""
    h = np.zeros(n)
    for s, e in intervals:
        # Find grid points whose centers fall within [s, e]
        grid_centers = np.linspace(dx/2, 2 - dx/2, n)
        mask = (grid_centers >= s) & (grid_centers < e)
        h[mask] = 1
    return h

def evaluate_c5_fft(h):
    """Fast evaluation of C5 using FFT correlation."""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def decode_params(z, m):
    """Convert parameter vector z into interval positions and lengths."""
    # Ensure a reasonable range for lengths
    epsilon = 1e-4
    l = epsilon + (1 - 2*epsilon) * z[:m]
    # Normalize lengths to sum to 1
    l = l / np.sum(l)
    # Gaps
    g = z[m:]
    # Compute intervals
    intervals = []
    s = 0
    for i in range(m):
        e = s + l[i]
        intervals.append((s, e))
        if i < m - 1:
            s = e + g[i]
        else:
            s = e
    # Make sure final end doesn't exceed 2
    final_end = intervals[-1][1]
    if final_end > 2:
        # Scale all gaps
        scale = (2 - np.sum(l)) / np.sum(g)
        for i in range(m - 1):
            g[i] *= scale
        intervals[-1] = (intervals[-1][0], 2.0)
    else:
        # Pad gaps if needed to ensure final point is at exactly 2
        total_length = np.sum(l) + np.sum(g)
        if total_length < 2:
            g[-1] = 2 - total_length
    return intervals

def objective(z, m):
    """Objective function for optimization."""
    intervals = decode_params(z, m)
    h = build_grid_from_intervals(intervals, n_points)
    return evaluate_c5_fft(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # hyperparameters
    m = 5
    popsize = 15 * 2 * m  # set population size for DE
    
    # Phase 1: generate a small random initial population
    initial_pop = []
    best_c5 = float('inf')
    best_z = None
    for _ in range(10):
        z = np.random.rand(2 * m)
        c5 = objective(z, m)
        if c5 < best_c5:
            best_c5 = c5
            best_z = z.copy()
        initial_pop.append(z)
    
    # bounds for differential evolution
    bounds = [(0, 1) for _ in range(2 * m)]
    
    # Phase 2: Differential Evolution
    de_start = time.time()
    de_budget = 0.6 * budget_s
    maxiter = int(de_budget / 0.001) // popsize  # estimate iters
    maxiter = min(maxiter, 200)  # cap at 200
    
    # differential evolution with custom callback
    def callback(xk, convergence):
        # compute C5 at current best
        z = xk.copy()
        c5 = objective(z, m)
        global global_best_z, global_best_c5
        if c5 < global_best_c5:
            global_best_c5 = c5
            global_best_z = z.copy()
    
    global global_best_z, global_best_c5
    global_best_c5 = float('inf')
    global_best_z = None
    
    try:
        res = differential_evolution(
            lambda z: objective(z, m),
            bounds,
            strategy='best1bin',
            maxiter=maxiter,
            popsize=popsize,
            mutation=(0.5, 1.0),
            recombination=0.9,
            atol=1e-7,
            callback=callback,
            seed=seed
        )
        
        if global_best_c5 < float('inf'):
            final_z = global_best_z.copy()
        else:
            final_z = res.x
    except:
        # fallback to best random start
        final_z = best_z.copy()
    
    # Phase 3: Local refinement
    refine_end = start_time + 0.8 * budget_s
    intervals = decode_params(final_z, m)
    h = build_grid_from_intervals(intervals, n_points)
    final_c5 = evaluate_c5_fft(h)
    
    # perform local refinement
    while time.time() < refine_end:
        improved = False
        for i in range(m):
            # Try shifting left edge of interval i
            s, e = intervals[i]
            new_s = s - dx/2
            if new_s >= 0 and i == 0 or new_s >= intervals[i-1][1]:
                new_intervals = intervals.copy()
                new_intervals[i] = (new_s, e)
                # Check if constraint still valid
                if i < m - 1 and new_s + (e - s) > new_intervals[i+1][0]:
                    continue
                h_new = build_grid_from_intervals(new_intervals, n_points)
                c5_new = evaluate_c5_fft(h_new)
                if c5_new < final_c5:
                    intervals = new_intervals
                    final_c5 = c5_new
                    improved = True
                    break
            
            # Try shifting right edge of interval i
            new_e = e + dx/2
            if new_e <= 2 and (i == m - 1 or new_e <= intervals[i+1][0]):
                new_intervals = intervals.copy()
                new_intervals[i] = (s, new_e)
                h_new = build_grid_from_intervals(new_intervals, n_points)
                c5_new = evaluate_c5_fft(h_new)
                if c5_new < final_c5:
                    intervals = new_intervals
                    final_c5 = c5_new
                    improved = True
                    break
        if not improved:
            break
    
    # Final grid construction and verification
    h_final = build_grid_from_intervals(intervals, n_points)
    # Test with exact evaluation as per grader
    corr = correlate(h_final, 1 - h_final, mode='full')
    final_c5_exact = dx * np.max(corr)
    
    # fallback if needed
    if final_c5_exact >= 0.38092:
        # Do a quick random sweep
        sweep_best_c5 = float('inf')
        sweep_best_h = None
        for _ in range(5000):
            z = np.random.rand(2 * m)
            c5 = objective(z, m)
            if c5 < sweep_best_c5:
                sweep_best_c5 = c5
                sweep_best_z = z.copy()
                
        if sweep_best_c5 < final_c5_exact:
            intervals = decode_params(sweep_best_z, m)
            h_final = build_grid_from_intervals(intervals, n_points)
            corr = correlate(h_final, 1 - h_final, mode='full')
            final_c5_exact = dx * np.max(corr)
    
    return (h_final.tolist(), float(final_c5_exact), n_points)
