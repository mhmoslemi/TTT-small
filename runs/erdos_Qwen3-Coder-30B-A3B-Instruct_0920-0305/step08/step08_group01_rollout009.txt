import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    # Use correlate with mode='full' to compute all shifts
    corr = correlate(h, 1 - h, mode='full')
    # The maximum overlap is at the center of the correlation
    # But we want the maximum value over all shifts
    max_corr = np.max(corr)
    return max_corr * (2.0 / len(h))

def project_to_box_and_sum(h, target_sum):
    """Project onto box constraints [0,1] and adjust sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Adjust by scaling and clipping
    scale_factor = target_sum / current_sum
    h = h * scale_factor
    h = np.clip(h, 0, 1)
    return h

def optimize_with_evolutionary(h0, population_size=50, generations=100, 
                              mutation_rate=0.1, elite_fraction=0.2):
    """Evolutionary algorithm approach"""
    n_points = len(h0)
    target_sum = n_points / 2
    population = [h0.copy() for _ in range(population_size)]
    
    for gen in range(generations):
        # Evaluate fitness (lower C5 is better)
        fitness = []
        for h in population:
            c5 = evaluate_c5(h)
            fitness.append((c5, h))
        
        # Sort by fitness (ascending)
        fitness.sort(key=lambda x: x[0])
        best_c5 = fitness[0][0]
        
        # Keep elite
        elite_count = int(elite_fraction * population_size)
        new_population = [h for _, h in fitness[:elite_count]]
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = fitness[random.randint(0, population_size//2)][1]
            parent2 = fitness[random.randint(0, population_size//2)][1]
            
            # Crossover (uniform)
            child = np.zeros(n_points)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] += random.gauss(0, 0.05)
            
            # Project to valid space
            child = project_to_box_and_sum(child, target_sum)
            new_population.append(child)
        
        population = new_population
    
    # Return best solution
    fitness = [(evaluate_c5(h), h) for h in population]
    fitness.sort(key=lambda x: x[0])
    return fitness[0][1]

def create_sturmian_like_initialization():
    """Create initial solution inspired by Sturmian sequences"""
    n = 120
    # Milder approach that still makes effective use of the interval
    x = np.linspace(0, 2, n)
    
    # Construct piece-wise linear for more structured behavior
    h = np.zeros(n)
    
    # We'll build two peaks, easier to see as local maxima
    # First peak at around x=0.5
    peak1 = 0.5
    width1 = 0.3
    for i in range(n):
        dist = abs(x[i] - peak1)
        h[i] = max(0, 1 - dist / width1)
    
    # Second peak at around x=1.5
    peak2 = 1.5
    width2 = 0.3
    for i in range(n):
        dist = abs(x[i] - peak2)
        h[i] += max(0, 1 - dist / width2)
    
    # Ensure sum constraint
    h = project_to_box_and_sum(h, n / 2)
    
    # Add regularization and smoothing
    for _ in range(3):
        h = np.convolve(h, np.ones(5)/5, mode='same')
        h = project_to_box_and_sum(h, n / 2)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    
    # Parity check to ensure only one solution uses each method
    for run_num in range(10):  # Decompose into multiple runs
        if time.time() - start_time > budget_s - 3:
            break
            
        if run_num == 0:
            # Start with Sturmian-like construction
            h = create_sturmian_like_initialization()
        else:
            # Random initialization
            h = np.random.uniform(0, 1, 120)
            h = project_to_box_and_sum(h, 60)
        
        # Refine via evolutionary algorithm
        h = optimize_with_evolutionary(h, 
                                     population_size=30 + run_num * 5, 
                                     generations=20 + run_num * 5,
                                     mutation_rate=0.1 - run_num * 0.01)
        
        # Evaluate final result
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return (best_h.tolist(), float(best_c5), 120)
