import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Minimize, Problem
import math

def compute_corr_fft(h):
    """Compute the correlation using FFT."""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value."""
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    best_h = None
    best_c5 = float('inf')
    
    # Phase 1: MILP binary optimization
    MILP_TIME = 0.4 * budget_s
    MILP_START = time.time()
    
    try:
        # Define the MILP problem using CVXPY
        y = cp.Variable(n_points, boolean=True)
        t = cp.Variable()
        
        # Define w_{k,i} variables for product constraints
        w = {}
        for k in range(1, n_points):
            for i in range(n_points - k):
                w[(k, i)] = cp.Variable()
        
        constraints = [
            cp.sum(y) == TARGET_SUM,
            t <= TARGET_SUM
        ]
        
        # Product constraints for w_{k,i} = y_i * y_{i+k}
        for k in range(1, n_points):
            for i in range(n_points - k):
                w[(k, i)] <= y[i]
                w[(k, i)] <= y[i + k]
                w[(k, i)] >= y[i] + y[i + k] - 1
                w[(k, i)] >= 0
                
                # Constraint that this corresponds to a real autocorrelation
                constraints.append(w[(k, i)] <= 1)
        
        # Minimum autocorrelation constraint for each shift k
        for k in range(1, n_points):
            constraints.append(cp.sum([w[(k, i)] for i in range(n_points - k)]) >= t)
        
        # Set up the optimization problem
        prob = Problem(Minimize(-t), constraints)
        
        # Set time limit for solver
        cp.settings.SOLUTION_TIME = MILP_TIME - (time.time() - MILP_START) - 0.1
        
        # Solve the problem
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=MILP_TIME - (time.time() - MILP_START) - 0.1)
        
        # Extract solution
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            y_sol = y.value
            if y_sol is not None:
                h0 = y_sol.astype(float)
                c5 = c5_from_corr(compute_corr_fft(h0), n_points) * dx
                if c5 < best_c5:
                    best_h = h0.copy()
                    best_c5 = c5
    except Exception as e:
        pass  # Fall back to other methods
    
    # If no binary solution, generate a random one
    if best_h is None:
        h0 = np.random.uniform(0, 1, n_points)
        h0 = project_to_sum_box(h0, TARGET_SUM)
        c5 = c5_from_corr(compute_corr_fft(h0), n_points) * dx
        best_h = h0.copy()
        best_c5 = c5
    
    # Phase 2: Subgradient descent refinement
    SGD_TIME = 0.35 * budget_s
    sgd_start = time.time()
    h = best_h.copy()
    current_c5 = best_c5
    
    def compute_subgradient(h, k):
        """Compute the subgradient of the k-th correlation."""
        g = np.zeros_like(h)
        # Define the correlation at shift k
        shifted_h = np.roll(h, k)
        g = 1 - shifted_h - np.roll(h, -k)
        return g
    
    alpha = dx * 0.5  # Initial step size
    while time.time() < sgd_start + SGD_TIME:
        # Compute correlation for all shifts
        corr = compute_corr_fft(h)
        k_star = np.argmax(corr)
        g = compute_subgradient(h, k_star)
        h_new = h - alpha * g
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        new_c5 = c5_from_corr(compute_corr_fft(h_new), n_points) * dx
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        alpha = alpha * 0.95  # Reduce step size
    
    # Phase 3: Random Stochastic Refinement
    STOCH_TIME = budget_s - (MILP_TIME + SGD_TIME)
    stoch_start = time.time()
    
    def try_swap(h_test, i, j, delta):
        """Try a small delta swap between i and j."""
        h_new = h_test.copy()
        old_diff = h_new[i] - h_new[j]
        h_new[i] = h_new[i] + delta
        h_new[j] = h_new[j] - delta
        h_new = np.clip(h_new, 0, 1)
        # Re-project to fix sum constraint
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        return h_new
    
    h = best_h.copy()
    current_c5 = best_c5
    iterations = 0
    max_iter = max(1000, int(10*STOCH_TIME))
    
    while time.time() < stoch_start + STOCH_TIME and iterations < max_iter:
        # Swap two random indices with small change
        i, j = np.random.randint(0, n_points, 2)
        delta = np.random.uniform(-0.02, 0.02)
        h_new = try_swap(h, i, j, delta)
        new_c5 = c5_from_corr(compute_corr_fft(h_new), n_points) * dx
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        iterations += 1
    
    return (best_h.tolist(), float(best_c5), n_points)
