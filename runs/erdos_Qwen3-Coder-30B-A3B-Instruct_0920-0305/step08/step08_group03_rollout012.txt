import numpy as np
from scipy.signal import correlate
import time

def project_to_capped_simplex(x, target_sum):
    """Project vector x onto the capped simplex: 0 <= x[i] <= 1, sum(x) = target_sum"""
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-12:
        return x
    
    # Binary search for lambda such that sum(clamp(x + lambda, 0, 1)) = target_sum
    a, b = -1.0, 1.0
    for _ in range(100):
        m = (a + b) / 2
        projected = np.clip(x + m, 0, 1)
        diff = np.sum(projected) - target_sum
        if abs(diff) < 1e-12:
            break
        if diff < 0:
            a = m
        else:
            b = m
    
    return np.clip(x + (a + b)/2, 0, 1)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return np.max(corr)

def smart_correlation_update(c, h, b, i, j, delta):
    """Incrementally update the correlation c with mass transfer from i to j"""
    n = len(h)
    # Get the original h and b value from indices i and j
    h_i_old = h[i]
    h_j_old = h[j]
    b_i_old = b[i]
    b_j_old = b[j]
    
    # Update h and b
    h[i] = h_i_old - delta
    h[j] = h_j_old + delta
    b[i] = 1.0 - h[i]
    b[j] = 1.0 - h[j]
    
    # Update correlations
    # Contribution from index i (h changed)
    i_start = max(0, i - (n - 1))
    i_end = min(n, i + 1)
    # Align with full mode corr index
    corr_start = (n - 1) - i
    # Add as vectorized slice
    if corr_start + i_end - i_start <= len(c):
        c[corr_start + i_start : corr_start + i_end] += delta * b[i_start:i_end]
        
    # Contribution from index j (h changed)
    j_start = max(0, j - (n - 1))
    j_end = min(n, j + 1)
    corr_start = (n - 1) - j
    if corr_start + j_end - j_start <= len(c):
        c[corr_start + j_start : corr_start + j_end] -= delta * b[j_start:j_end]
        
    # Cross term for shift i -> j: corr at shift (j-i) and (i-j) both change by delta^2
    # i shifts to the left (by 0 to n-1), so it can contribute to corr of shift -s to 0
    # j shifts to the right, so it can contribute to corr of shift 0 to s
    # key shift is (j - i) with negative index in corr space, corr_index = (n-1) + (j-i)
    corr_idx1 = (n - 1) + (j - i)
    corr_idx2 = (n - 1) + (i - j)
    if 0 <= corr_idx1 < len(c):
        c[corr_idx1] += delta * delta
    if 0 <= corr_idx2 < len(c):
        c[corr_idx2] += delta * delta
    
    # And double-check that we didn't violate box constraints.
    # Also recompute resulting correlation exactly after steps, after some time or moves.
    return c, h, b

    b = 1.0 - h
    c = correlate(h, b, mode='full')
    Cmax = np.max(c)
    C5_current = Cmax * dx

    # Annealing params
    T = 0.01   # Temperature
    alpha_T = 0.995  # Cooling rate
    delta = 0.02  # Step size
    min_delta = 5e-4
    max_iters = 1000000  # Total steps
    K_eval = 500
    K_step = 1000
    K_temp = 5000
    K_noimp = 2000

    best_h = h.copy()
    best_C5 = C5_current
    current_h = h.copy()
    current_b = b.copy()
    current_c = c.copy()
    current_Cmax = Cmax
    no_improvement_iter = 0
    total_steps = 0

    accept_count = 0
    total_accepted = 0

    while time.time() - start_time <= budget_s and total_steps < max_iters:
        total_steps += 1
        
        # Selection of move indices
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        while i == j:
            j = np.random.randint(0, n)
        
        # Bound check
        h_i = current_h[i]
        h_j = current_h[j]
        if h_i < delta or h_j > 1 - delta:
            continue
            
        # Compute max delta to not violate bounds
        max_delta = min(h_i, 1 - h_j)
        delta_i = min(delta, max_delta)
        
        # Save current state
        h_save = current_h.copy()
        b_save = current_b.copy()
        c_save = current_c.copy()
        cmax_save = current_Cmax

            current_c, current_h, current_b, i, j, -delta_i
        )
        current_Cmax = np.max(current_c)
        new_C5 = current_Cmax * dx
        delta_C5 = new_C5 - C5_current
        
        # Accept based on simulated annealing
        if delta_C5 < 0 or np.random.rand() < np.exp(-delta_C5 / T):
            # Accept the move
            C5_current = new_C5
            accept_count += 1
            
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = current_h.copy()

            no_improvement_iter = 0

            current_h[:] = h_save
            current_b[:] = b_save
            current_c[:] = c_save
            current_Cmax = cmax_save
            
        # Adapt step size
        if total_steps % K_step == 0:
            if delta > min_delta:
                delta *= 0.9
                
        # Cool temperature periodically
        if total_steps % K_temp == 0:
            T *= alpha_T
            
        # Reset temperature if stagnated
        if no_improvement_iter > K_noimp:
            T = 0.01
            
        # Re-evaluate correlation exactly every K_eval steps
        if total_steps % K_eval == 0:
            # Recompute full correlation
            c_exact = correlate(current_h, 1 - current_h, mode='full')
            current_Cmax = np.max(c_exact)
            current_c = c_exact
            C5_current = current_Cmax * dx
            no_improvement_iter += 1

        # Occasionally randomly restart (by chance)
        if total_steps % 50000 == 0:
            # Start fresh from current point to avoid getting stuck
            # or replicate the best if already quite good
            if np.random.rand() < 0.1:

                b = 1.0 - h
                c = correlate(h, b, mode='full')
                current_c = c.copy()
                current_Cmax = np.max(c)
                C5_current = current_Cmax * dx
                current_h = h.copy()
                current_b = b.copy()
                total_accepted = 0
                accept_count = 0

    return (best_h.tolist(), float(best_C5), n)
