import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *
import cvxpy as cp

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
MAX_ITERS = 5000
TIME_BUDGET = 600

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - np.array(h)
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def build_milp_model(x, t, y, shifts):
    """Build MIP model for the problem"""
    constraints = []
    
    # Mass constraint
    mass_cons = cp.sum(x) == TARGET_SUM
    constraints.append(mass_cons)
    
    # Linearization of products y[i,s] = x[i]*x[i+s]
    for s in shifts:
        for i in range(len(x)):
            if i + s >= 0 and i + s < len(x):
                constraints.append(y[i, s] <= x[i])
                constraints.append(y[i, s] <= x[i + s])
                constraints.append(y[i, s] >= x[i] + x[i + s] - 1)
    
    # Epigraph constraints
    for s in shifts:
        cons = cp.sum([y[i, s] for i in range(len(x)) if i + s < len(x)]) >= TARGET_SUM - t
        constraints.append(cons)
        
    # Non-negativity
    constraints.append(t >= 0)
    
    return constraints

def run_milp_row_generation():
    """Use row generation to solve the MILP problem"""
    start_time = time.time()
    
    # Initialize discrete variables
    x = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Dictionary to store auxiliary variables
    y = {}
    initial_shifts = list(range(1, 6))  # Start with small shifts
    
    # Initialize auxiliary variables
    for s in initial_shifts:
        for i in range(len(x)):
            if i + s < len(x):
                y[(i, s)] = cp.Variable(boolean=True)
    
    # Set initial constraints
    constraints = build_milp_model(x, t, y, initial_shifts)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create problem
    prob = cp.Problem(objective, constraints)
    
    # Row generation loop
    active_shifts = initial_shifts.copy()
    best_h = None
    best_c5 = float('inf')
    
    while True:
        if time.time() - start_time > 0.7 * TIME_BUDGET:
            break
            
        try:
            # Solve the current MILP
            prob.solve(solver=CPLEX, max_seconds=30, verbosity=0)
            
            if prob.status != cp.OPTIMAL:
                break
                
            # Obtain current binary solution
            x_sol = x.value.astype(int)
            t_sol = t.value
            
            # Compute actual C5 using FFT on full correlation
            h_float = x_sol.astype(float)
            actual_c5 = evaluate_c5(h_float)
            
            if actual_c5 < best_c5:
                best_h = h_float.copy()
                best_c5 = actual_c5
                
            # Compute all correlations and check for violations
            corr_fft = compute_corr_fft(h_float)
            
            # Find the maximum overlap
            max_shift = np.argmax(corr_fft)
            max_val = corr_fft[max_shift]
            
            # If the model is accurate, we're at optimality
            if abs(max_val - t_sol * dx) < 1e-8:
                break
            else:
                # Add the most violated shift
                if max_shift < n_points:
                    new_shift = max_shift
                else:
                    new_shift = max_shift - 2 * n_points + 2
                
                if new_shift not in active_shifts:
                    active_shifts.append(new_shift)
                    
                    # Add new auxiliary variable
                    for i in range(len(x)):
                        if i + new_shift < len(x):
                            y[(i, new_shift)] = cp.Variable(boolean=True)
                    
                    # Update constraints for the new shift
                    new_cons = build_milp_model(x, t, y, [new_shift])
                    
                    # Add them to the problem
                    prob.constraints.extend(new_cons)
                    
        except Exception:
            break
    
    return best_h, best_c5

def refine_with_continuous_optimization(h_binary, c5_binary):
    """Refine binary solution with smooth optimization"""
    start_time = time.time()
    
    # Convert to continuous variables
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    mass_cons = cp.sum(h) == TARGET_SUM
    bounds_cons = [h >= 0, h <= 1]
    
    # Objective - smooth approximation of max
    gamma = 100.0
    inv_gamma = 1.0 / gamma
    
    # Compute all correlation terms
    corr_terms = []
    for s in range(-n_points + 1, n_points):
        term = 0
        for i in range(len(h)):
            if i + s >= 0 and i + s < len(h):
                term += h[i] * (1 - h[i + s])
        corr_terms.append(term)
    
    # Smooth max over all shifts
    smooth_term = cp.sum(cp.exp(gamma * (cp.hstack(corr_terms) - t)))
    smooth_obj = t + inv_gamma * cp.log(smooth_term)
    
    # Objective function
    objective = cp.Minimize(smooth_obj)
    
    # Constraints
    constraints = [mass_cons] + bounds_cons
    
    # Initialize with current solution
    h0 = h_binary.copy()
    
    try:
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=SCS, max_iters=5000, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_opt = h.value
            c5_opt = t.value * dx
            return h_opt.tolist(), c5_opt
    except:
        pass
    
    return h_binary.tolist(), c5_binary

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set global parameters
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points / 2
    dx = 2.0 / n_points
    
    # Start with coarse MILP solution
    h_sol, c5_sol = run_milp_row_generation()

    if h_sol is None or c5_sol == float('inf'):

        h0 = h0 * TARGET_SUM / np.sum(h0)
        
        # Simple local search
        best_h = h0.copy()
        best_c5 = evaluate_c5(h0)
        
        for i in range(2000):
            if time.time() - start_time > budget_s * 0.9:
                break
                
            idx = np.random.randint(0, n_points)
            delta = np.random.uniform(-0.02, 0.02)
            h_new = best_h.copy()
            h_new[idx] += delta
            
            if h_new[idx] < 0:
                h_new[idx] = 0
            elif h_new[idx] > 1:
                h_new[idx] = 1
            
            # Project sum constraint
            sum_diff = TARGET_SUM - np.sum(h_new)
            if abs(sum_diff) > 1e-6:
                # Distribute to other elements
                h_new += sum_diff / (n_points - 1)
                h_new = np.clip(h_new, 0, 1)
                # Ensure the sum is correct
                h_new = h_new * TARGET_SUM / np.sum(h_new)
            
            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                
        h_sol, c5_sol = best_h.tolist(), best_c5
    
    # Now attempt refinement with continuous optimization
    if c5_sol < float('inf') and time.time() - start_time < budget_s * 0.95:
        h_sol, c5_sol = refine_with_continuous_optimization(np.array(h_sol), c5_sol)
    
    # Final verification
    c5_eval = evaluate_c5(np.array(h_sol))
    
    return (h_sol, float(c5_eval), n_points)
