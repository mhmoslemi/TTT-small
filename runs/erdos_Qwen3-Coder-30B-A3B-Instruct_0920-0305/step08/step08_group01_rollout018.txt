import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def initialize_with_pattern():
    """Initialize with a pattern that tries to spread out the function to minimize overlap"""
    h = np.zeros(n_points)
    
    # Create a periodic-like pattern with more concentrated mass
    for i in range(n_points):
        x = i * dx
        if 0 <= x < 0.5:
            h[i] = 0.6
        elif 0.5 <= x < 1.0:
            h[i] = 0.8
        elif 1.0 <= x < 1.5:
            h[i] = 0.4
        else:  # 1.5 <= x <= 2.0
            h[i] = 0.2
    
    # Scale to meet constraint
    current_sum = np.sum(h)
    scale_factor = TARGET_SUM / current_sum
    h = h * scale_factor
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    return h

def simulate_annealing(h0, max_iter=5000):
    """Use simulated annealing to improve solution"""
    h_current = h0.copy()
    c5_current = evaluate_c5(h_current)
    
    # Annealing parameters
    temp = 1.0
    min_temp = 1e-8
    alpha = 0.995
    
    for i in range(max_iter):
        # Create neighbor solution
        h_new = h_current.copy()
        
        # Randomly select indices to modify
        indices = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
        
        for idx in indices:
            # Perturb the value
            delta = np.random.normal(0, 0.05)  # Gaussian perturbation
            h_new[idx] += delta
            # Project to bounds
            h_new[idx] = max(0, min(1, h_new[idx]))
        
        # Project back to sum constraint
        h_new = project_to_sum_constraint(h_new)
        
        # Calculate energy change
        c5_new = evaluate_c5(h_new)
        delta = c5_new - c5_current
        
        # Accept or reject
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h_current = h_new
            c5_current = c5_new
            
        # Decrease temperature
        temp = max(min_temp, temp * alpha)
        
        if i % 500 == 0:
            # Periodically print progress
            pass
            
    return h_current, c5_current

def project_to_sum_constraint(h):
    """Project h to satisfy the sum constraint"""
    current_sum = np.sum(h)
    if abs(current_sum - TARGET_SUM) < 1e-10:
        return h
        
    # Simple adjustment to correct sum
    adjustment = (TARGET_SUM - current_sum) / n_points
    
    # Adjust while respecting bounds
    h = np.clip(h + adjustment, 0, 1)
    
    # Rescale to exactly meet the constraint using the method of Lagrange multipliers
    # This is needed to ensure the exactly correct sum
    # A different approach: just adjust the last few elements
    total_adjust = TARGET_SUM - np.sum(h)
    h[-1] += total_adjust
    
    # Ensure all elements are within bounds
    h = np.clip(h, 0, 1)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = initialize_with_pattern()
    best_c5 = evaluate_c5(best_h)
    
    # Simulated annealing approach
    max_iter = 5000
    current_time = time.time()
    end_time = start_time + budget_s
    
    while current_time < end_time:
        # Run simulation annealing
        h, c5 = simulate_annealing(best_h.copy(), max_iter=1000)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Add more stochastic refinement
        for _ in range(10):
            if current_time > end_time:
                break
                
            # Try different random initializations
            h_trial = np.random.uniform(0, 1, n_points)
            h_trial = project_to_sum_constraint(h_trial)
            
            # Local refinement
            for _ in range(50):
                idx = np.random.randint(0, n_points)
                if np.random.random() < 0.5:
                    h_trial[idx] = min(1, h_trial[idx] + np.random.normal(0, 0.03))
                else:
                    h_trial[idx] = max(0, h_trial[idx] - np.random.normal(0, 0.03))
                    
                h_trial[idx] = max(0, min(1, h_trial[idx]))
            
            h_trial = project_to_sum_constraint(h_trial)
            new_c5 = evaluate_c5(h_trial)
            
            if new_c5 < best_c5:
                best_h = h_trial.copy()
                best_c5 = new_c5
                
        current_time = time.time()
        
    # Final check and refinement
    final_h = best_h.copy()
    final_c5 = evaluate_c5(final_h)
    
    return (final_h.tolist(), float(final_c5), n_points)
