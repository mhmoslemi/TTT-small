import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from scipy.linalg import qr
import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_sdp_relaxation(n):
    """Solve SDP relaxation to find a near-optimal solution"""
    # Create decision variables
    X = cp.Variable((n, n), symmetric=True)
    mu = cp.Variable(n)
    lambda_var = cp.Variable()
    
    # Constraints
    constraints = []
    for s in range(1, (n+1)//2):
        # Trace of R_s * X
        sum_term = cp.sum(cp.diag(X, offset=s))
        constraints.append(sum_term >= lambda_var)
    
    for i in range(n):
        constraints.append(X[i, i] == mu[i])
    constraints.append(cp.sum(mu) == n / 2)
    constraints.append(0 <= mu)
    constraints.append(mu <= 1)
    constraints.append(X >> 0)  # PSD constraint
    
    # Symmetry breaking: fix first entry to 1 to prevent translation symmetry
    constraints.append(X[0, 0] == 1)
    # Adjust sum since one entry is fixed
    constraints.append(cp.sum(mu[1:]) == (n/2 - 1))
    
    # Objective: maximize lambda
    objective = cp.Maximize(lambda_var)
    prob = cp.Problem(objective, constraints)
    
    # Solve the problem
    try:
        prob.solve(solver='SCS', verbose=False, max_iters=2000, eps=1e-4)
        if prob.status == 'optimal':
            return mu.value, X.value
        else:
            return None, None
    except:
        return None, None

def random_rounding(mu, X, n, num_rounds=500):
    """Apply random hyperplane rounding with different vectors to generate binary vectors"""
    best_x = None
    best_c5 = float('inf')
    
    # SVD of X to get row vectors v_i
    U, s, Vt = np.linalg.svd(X)
    
    for _ in range(num_rounds):
        # Generate random Gaussian vector
        g = np.random.randn(U.shape[1])
        
        # Compute scores (v_i . g)
        scores = np.dot(U, g)
        
        # Choose the (n/2)th largest score as threshold
        sorted_scores = np.sort(scores)[::-1]
        threshold = sorted_scores[n//2 - 1] if n//2 > 0 else sorted_scores[-1]
        
        # Create binary vector
        x = (scores >= threshold).astype(float)
        
        # Ensure exactly n/2 ones
        ones = np.where(x == 1)[0]
        zeros = np.where(x == 0)[0]
        
        if len(ones) > n//2:
            # Remove some 1s
            to_remove = len(ones) - n//2
            remove_indices = np.random.choice(ones, to_remove, replace=False)
            x[remove_indices] = 0
        elif len(ones) < n//2:
            # Add some 0s
            to_add = n//2 - len(ones)
            add_indices = np.random.choice(zeros, to_add, replace=False)
            x[add_indices] = 1
            
        # Evaluate C5
        c5 = evaluate_c5(x)
        if c5 < best_c5:
            best_c5 = c5
            best_x = x.copy()
    
    return best_x, best_c5

def shift_aware_local_search(x, max_iterations=5000):
    """Perform local search using shift-aware swaps"""
    def eval_c5_with_fixup(x):
        corr = compute_corr_fft(x)
        return c5_from_corr(corr, n_points)
    
    current_c5 = eval_c5_with_fixup(x)
    best_x = x.copy()
    best_c5 = current_c5
    
    # Precompute correlation array
    corr = compute_corr_fft(x)
    
    for iteration in range(max_iterations):
        # Find worst shift
        s_star = np.argmax(corr)
        
        # Get indices contributing to worst shift
        bad_indices = []
        good_indices = []
        
        # Manually compute valid shift indices for s_star
        for i in range(len(x) - s_star):
            if x[i] == 1 and x[i + s_star] == 0:
                bad_indices.append(i)
            elif x[i] == 0 and x[i + s_star] == 1:
                good_indices.append(i)
        
        if not bad_indices or not good_indices:
            break
            
        # Select random swaps from Bad and Good sets
        i = np.random.choice(bad_indices)
        j = np.random.choice(good_indices)
        
        # Perform swap
        x_new = x.copy()
        x_new[i], x_new[j] = x_new[j], x_new[i]
        
        new_c5 = eval_c5_with_fixup(x_new)
        if new_c5 < current_c5:
            x = x_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_x = x.copy()
                best_c5 = current_c5
        else:
            # Occasionally do neutral swaps to escape local minima
            if np.random.random() < 0.1:
                x = x_new
                current_c5 = new_c5
                
    return best_x, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n = n_points
    TARGET_SUM = n / 2
    dx = 2.0 / n
    
    best_c5 = float('inf')
    best_x = np.zeros(n)
    
    # Phase 1: SDP Solver with rounding and refinement
    print("Starting SDP solve...")
    sdp_time = 0.0
    attempted_rounding = 0
    total_rounding_time = 0.0
    
    while sdp_time < 150 and time.time() < start_time + budget_s:
        start_sdp = time.time()
        mu, X = solve_sdp_relaxation(n)
        
        if mu is None:
            break
            
        sdp_time += time.time() - start_sdp
        print(f"SDP completed in {sdp_time:.2f}s")
        
        # Store closest integer approximations
        mu_approx = np.clip(mu, 0, 1)
        mu_rounded = np.round(mu_approx)
        mu_projected = project_to_sum_box(mu_rounded, TARGET_SUM)
        
        print(f"SDP solution mu approximate sum={np.sum(mu_approx):.3f}, projected={np.sum(mu_projected):.3f}")
        
        # Rounding with diverse samples
        print("\nStarting random rounding...")
        start_round = time.time()
        x_rounded, c5_rounded = random_rounding(mu, X, n)
        total_rounding_time += time.time() - start_round
        attempted_rounding += 1
        
        if x_rounded is not None and c5_rounded < best_c5:
            best_c5 = c5_rounded
            best_x = x_rounded.copy()
            print(f"Updated best C5 to {best_c5:.6f}")
            
        # Refinement step on the best rounded vector
        if x_rounded is not None:
            refined_x, refined_c5 = shift_aware_local_search(x_rounded.copy(), max_iterations=1000)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_x = refined_x.copy()
                print(f"Refined best C5 to {best_c5:.6f}")
                
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Optional upscale if time remains
    if time.time() < start_time + budget_s - 50 and n_points < 200:
        print("Performing optional upscale...")
        n_upscale = 180
        dx_upscale = 2.0 / n_upscale
        mu_upscale = np.zeros(n_upscale)
        x_upscale = np.zeros(n_upscale)
        
        # Interpolate best_x to 180 points
        x_coarse = best_x
        x_upscale = np.interp(np.linspace(0, 2, n_upscale), np.linspace(0, 2, n_points), x_coarse)
        # Make binary (threshold to 0.5)
        x_upscale = (x_upscale > 0.5).astype(float)
        # Project to correct sum
        x_upscale = project_to_sum_box(x_upscale, n_upscale / 2)
        
        # Evaluate the coarse upscaled version before local search
        initial_c5_upscaled = evaluate_c5(x_upscale)
        print(f"Upscaled C5 before local search: {initial_c5_upscaled:.6f}")
        
        # Local refinement on upscale
        refined_upscale_x, refined_upscale_c5 = shift_aware_local_search(x_upscale, max_iterations=1000)
        if refined_upscale_c5 < best_c5:
            best_c5 = refined_upscale_c5
            best_x = refined_upscale_x.copy()
            print(f"Final C5 improved to {best_c5:.6f} via upscale")
    
    # Final verification
    final_c5 = evaluate_c5(best_x)
    if final_c5 < best_c5 and np.isfinite(final_c5):
        best_c5 = final_c5
    
    print(f"Final best C5 = {best_c5:.6f}")
    return (best_x.tolist(), float(best_c5), n_points)
