import numpy as np

def compute_corr_fft(h):
    h_rev = h[::-1]
    b = 1 - np.array(h)
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    return np.max(corr)

def evaluate_c5(h, dx):
    corr = compute_corr_fft(h)
    return dx * c5_from_corr(corr)
