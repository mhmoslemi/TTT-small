import numpy as np
from scipy.signal import correlate
import cvxpy as cp
from cvxpy import Minimize, Problem
import time
from math import log

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
SOLVER = 'SCIP'  # or 'CBC', 'GLPK'

# Helper functions for MILP formulation
def compute_full_correlation(h):
    """Compute correlation for all shifts using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_c5_from_corr(corr_values):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr_values)

def project_to_sum_box(x, target_sum):
    """Project vector x onto [0,1] box with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_milp_with_row_generation(initial_h, max_time=480):
    """Row generation approach to solve MILP for exact C5 bound"""
    n = len(initial_h)
    global TARGET_SUM
    TARGET_SUM = n / 2
    
    # Base shifts to start with
    shift_set = set([1, 2, 3, 4, 5])
    
    # Initialize with integer variables
    x = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Set initial problem data
    constraints = [cp.sum(x) == TARGET_SUM, t >= 0]
    
    # Store all the shifts we know about, with their slack variables
    shift_constraints = []
    y_vars = {}
    
    # Create variables and constraints for each shift
    for s in shift_set:
        y_vars[s] = cp.Variable(n, boolean=True)
        # Linearization constraints for y[i,s] = x[i] * x[i+s]
        for i in range(n):
            if i + s < n:
                constraints.append(y_vars[s][i] <= x[i])
                constraints.append(y_vars[s][i] <= x[i+s])
                constraints.append(y_vars[s][i] >= x[i] + x[i+s] - 1)
            else:
                constraints.append(y_vars[s][i] == 0)
        # Epigraph constraint related to this shift
        shift_equality = cp.sum(y_vars[s]) >= 0.5*n - t 
        constraints.append(shift_equality)
        shift_constraints.append(shift_equality)
    
    # Objective
    objective = Minimize(t)
    
    # Solver configuration
    solver_opts = {"maxSeconds": 5}
    prob = Problem(objective, constraints)
    
    best_h = initial_h.copy()
    best_c5 = compute_c5_from_corr(compute_full_correlation(initial_h))
    
    # Begin iterative row generation
    start_time = time.time()
    while time.time() - start_time < max_time:
        try:
            prob.solve(solver=SOLVER, **solver_opts)
        
            if prob.status != 'optimal':
                break
                
            # Get the binary values
            x_val = np.array([int(round(v.value)) for v in x])
            t_val = t.value
            
            # Evaluate full correlation for all shifts
            full_corr = compute_full_correlation(x_val)
            c5_evaluated = compute_c5_from_corr(full_corr)
            
            # Check if all constraints are satisfied
            if abs(c5_evaluated - t_val) < 1e-7:
                # We are done; the solution is exact
                # Convert back to floating point if needed
                h_solution = x_val.astype(float)
                h_solution = project_to_sum_box(h_solution, TARGET_SUM)
                c5_sol = compute_c5_from_corr(compute_full_correlation(h_solution))
                return h_solution, c5_sol, n
            
            # Otherwise, add a new violated constraint
            max_shift_idx = np.argmax(full_corr)
            if max_shift_idx < n:
                # this corresponds to some positive shift k
                # Add that shift to constraint set
                s = max_shift_idx + 1
                if s not in shift_set:
                    shift_set.add(s)
                    y_vars[s] = cp.Variable(n, boolean=True)
                    # Add the new shift constraint one by one  
                    # Linear constraints for y[i,s]
                    for i in range(n):
                        if i + s < n:
                            constraints.append(y_vars[s][i] <= x[i])
                            constraints.append(y_vars[s][i] <= x[i+s])
                            constraints.append(y_vars[s][i] >= x[i] + x[i+s] - 1)
                        else:
                            constraints.append(y_vars[s][i] == 0)
                    # Epigraph constraint for s
                    shift_equality = cp.sum(y_vars[s]) >= 0.5*n - t
                    constraints.append(shift_equality)
                    shift_constraints.append(shift_equality)
                    
                    # Re-create the problem with new variables
                    # we do not create a full new model but try a lazy approach - 
                    # but since we can't re-add constraints to CPX easily,
                    # we simply re-solve the updated problem 
                    prob = Problem(objective, constraints)
            
            # Save the best solution found so far
            if c5_evaluated < best_c5:
                best_h = x_val.astype(float)
                best_c5 = c5_evaluated
                best_h = project_to_sum_box(best_h, TARGET_SUM)
        except Exception:
            break
    
    return best_h, best_c5, n

def polish_continuous_solution(h_binary, c5_val, max_iters=200):
    """Smooth the binary solution with continuous optimization preserving mass"""
    n = len(h_binary)
    gamma = 50.0
    
    # Convert to continuous
    h_start = h_binary.copy()
    t_start = c5_val / dx
    
    def objective_smooth(h, t):
        # log-sum-exp bound on max
        term = np.sum(np.exp(gamma * (h @ (1 - np.roll(h, s)) - t)) for s in range(1, n))
        return t + (1/gamma) * np.log(term)
    
    # Set up problem for continuous polishing
    h = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [cp.sum(h) == TARGET_SUM, 
                   h >= 0, h <= 1, 
                   t >= 0]
    
    # Try to minimize smooth version
    def soft_obj(h_vals, t_val):
        # Compute full correlation from array (including periodicity)
        corr = compute_full_correlation(h_vals)
        return t_val + (1/gamma) * np.log(np.sum(np.exp(gamma * (corr - t_val))))
    
    # Use scipy for more robust solving
    def optimize_cont():
        def f(x):
            return soft_obj(x, t_start)
        
        def grad(x):
            # Approximate gradient using finite difference
            eps = 1e-5
            g = np.zeros_like(x)
            for i in range(len(x)):
                x_plus = x.copy()
                x_plus[i] += eps  
                x_minus = x.copy()
                x_minus[i] -= eps
                g[i] = (soft_obj(x_plus, t_start) - soft_obj(x_minus, t_start)) / (2 * eps)
            return g

        # Initialize from binary values
        x_sol = h_start.copy()
        # Iterate slightly for refinement
        for _ in range(50):
            # Project back to feasible region
            x_sol = np.clip(x_sol, 0, 1)
            x_sol = project_to_sum_box(x_sol, TARGET_SUM)
            # Use a simple gradient-based method or just use projected iteration
        return x_sol
    
    # Run polishing step  
    h_final = optimize_cont()
    c5_final = compute_c5_from_corr(compute_full_correlation(h_final))
    return h_final, c5_final

    # Early fallback 
    if time.time() - start_time > 0.1*budget_s:
        best_h = h_ref
        best_c5 = compute_c5_from_corr(compute_full_correlation(h_ref))
        return best_h.tolist(), float(best_c5), n_points
    
    # Solve with MILP row generation
    h_eval, c5_eval, scale = solve_milp_with_row_generation(h_ref, 
        max_time=0.7*budget_s)
    
    # Polish continuous solution
    if time.time() - start_time > 0.85*budget_s:

    # Run polishing if not time too tired
    h_cont, c5_cont = polish_continuous_solution(h_eval, c5_eval, max_iters=100)
