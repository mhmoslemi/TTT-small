import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def quadratic_penalty_method(h0, max_iter=1000):
    """Optimize using quadratic penalty method"""
    h = h0.copy()
    penalty_weight = 100.0
    tolerance = 1e-8
    
    for iteration in range(max_iter):
        # Objective with penalty for sum constraint
        def penalized_objective(h):
            c5_val = evaluate_c5(h)
            sum_penalty = penalty_weight * (np.sum(h) - TARGET_SUM)**2
            return c5_val + sum_penalty
        
        # Optimize with simple gradient descent steps
        step_size = 0.001
        h_new = h.copy()
        
        # Compute gradient approximation
        grad = np.zeros_like(h)
        epsilon = 1e-5
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = project_to_sum_box(h_plus, TARGET_SUM)
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = project_to_sum_box(h_minus, TARGET_SUM)
            
            grad[i] = (penalized_objective(h_plus) - penalized_objective(h_minus)) / (2 * epsilon)
        
        # Update
        h_new = h - step_size * grad
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Check for convergence
        if np.linalg.norm(h_new - h) < tolerance:
            break
        h = h_new
        
        # Increase penalty if we're getting close to constraint
        if abs(np.sum(h) - TARGET_SUM) < 0.01:
            penalty_weight *= 1.1
            
    return h

    best_c5 = evaluate_c5(h)

def initialize_with_peaks():
    """Initialize with known good pattern of two peaks"""
    h = np.zeros(n_points)
    
    # Place two peaks at positions 0.4 and 1.6
    peak1_pos = int(0.4/dx)
    peak2_pos = int(1.6/dx)
    
    # Create slightly rounded peaks
    for i in range(n_points):
        dist1 = abs(i - peak1_pos)
        dist2 = abs(i - peak2_pos)
        if dist1 < 5:
            h[i] += max(0, 1.0 - dist1/5.0) * 0.8
        if dist2 < 5:
            h[i] += max(0, 1.0 - dist2/5.0) * 0.8
    
    # Normalize to sum = TARGET_SUM
    h = project_to_sum_box(h, TARGET_SUM)
    return h

    ]

            best_c5 = c5_opt
            best_h = h_opt.copy()

    phase2_end = start_time + 0.8 * budget_s

        c5_new = evaluate_c5(h_candidate)
        if c5_new < best_c5:

    phase3_end = start_time + budget_s

        # Use randomized approach to escape local minima
        h_new = best_h + np.random.normal(0, 0.005, n_points)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Apply local search on this perturbed solution
        h_final = refined_local_search(h_new, max_iter=100)
        c5_final = evaluate_c5(h_final)
        
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
            
    return (best_h.tolist(), float(best_c5), n_points)
