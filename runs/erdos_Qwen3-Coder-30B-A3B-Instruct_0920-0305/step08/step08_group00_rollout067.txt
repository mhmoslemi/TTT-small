def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_descent_step(h, step_size=0.01):
    """Perform one gradient descent step using finite differences"""
    h = h.copy()
    current_c5 = evaluate_c5(h)
    
    # Compute gradient numerically
    grad = np.zeros_like(h)
    eps = 1e-6
    
    for i in range(len(h)):
        h_plus = h.copy()

        grad[i] = (c5_plus - current_c5) / eps
    
    # Update with gradient ascent to minimize
    h_new = h - step_size * grad
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    
    return h_new

def stretched_gaussian(x, center, width, height):
    """Create a stretched Gaussian function"""
    return height * np.exp(-0.5 * ((x - center) / width) ** 2)
