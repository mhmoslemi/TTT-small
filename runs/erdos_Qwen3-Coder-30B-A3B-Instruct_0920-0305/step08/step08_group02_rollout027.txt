```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from scipy.fft import fft, ifft

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water-filling"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_binary_milp():
    """Solve binary MILP for the Erdős problem"""
    N = n_points
    M = N // 2
    
    # Create binary variables
    y = cp.Variable(N, boolean=True)
    
    # Auxiliary variables for products
    w = {}
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(y) == M, t >= 0, t <= M]
    
    # Product constraints
    for k in range(1, N):
        for i in range(N - k):
            w_key = (k, i)
            w[w_key] = cp.Variable()
            constraints.extend([
                w[w_key] <= y[i],
                w[w_key] <= y[i + k],
                w[w_key] >= y[i] + y[i + k] - 1,
                w[w_key] >= 0
            ])
    
    # Minimum autocorrelation constraints
    for k in range(1, N):
        constraints.append(cp.sum([w[(k, i)] for i in range(N - k)]) >= t)
    
    # Solve
    prob = cp.Problem(cp.Maximize(t), constraints)
    
    try:
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=0.4 * 600)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            return y.value
    except:
        pass
    
    # Fallback to GLPK or ECOS_BB
    try:
        prob.solve(solver=cp.GLPK_MI, verbose=False, max_seconds=0.4 * 600)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            return y.value
    except:
        pass
    
    try:
        prob.solve(solver=cp.ECOS_BB, verbose=False, max_seconds=0.4 * 600)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            return y.value
    except:
        pass
    
    return None

def greedy_binary_construction():
    """Create a greedy binary solution"""
    N = n_points
    M = N // 2
    y = np.zeros(N, dtype=int)
    
    # Greedily place 1s to maximize minimum autocorrelation
    placed = 0
    i = 0
    while placed < M:
        if y[i] == 0:
            y[i] = 1
            placed += 1
        i = (i + 2) % N  # Place roughly every 2 positions
    
    return y

def compute_subgradient(h, k_star):
    """Compute the subgradient of the worst shift"""
    g = np.zeros_like(h)
    if k_star < len(h):
        g = 1 - np.roll(h, k_star) - np.roll(h, -k_star)
    else:
        # Handle 2D case for large k
        g = 1 - np.concatenate([h[k_star:], h[:k_star]]) - np.concatenate([h[-k_star:], h[:-k_star]])
    return g

def deterministic_subgradient_descent(h0, max_seconds):
    """Perform deterministic sub-gradient descent"""
    start_time = time.time()
    
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    alpha0 = 0.5 * dx
    t = 1
    
    while time.time() - start_time < max_seconds:
        # Compute current correlation
        corr = compute_corr_fft(h)
        k_star = np.argmax(corr)
        
        # Compute subgradient
        g = compute_subgradient(h, k_star)
        
        # Take gradient step
        h_new = h - (alpha0 / np.sqrt(t)) * g
        
        # Project onto box and simplex
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        
        h = h_new.copy()
        t += 1
    
    return best_h, best_c5

def stochastic_refinement(h0, max_seconds):
    """Perform stochastic swap moves and simulated annealing"""
    start_time = time.time()
    
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    T_start = 5 * dx
    T_end = 0
    initial_temp = T_start
    
    # Simulated annealing parameters
    max_iter = 100000
    k = 0
    
    while k < max_iter and time.time() - start_time < max_seconds:
        # Generate two random indices
        i, j = np.random.choice(n_points, size=2, replace=False)
        
        # Generate small random perturbation
        delta = np.random.uniform(-0.02, 0.02)
        
        # Update values
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project onto simplex with sum constraint
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        new_c5 = evaluate_c5(h_new)
        
        # Accept or reject
        if new_c5 < best_c5:
            h = h_new.copy()
            best_h = h_new.copy()
            best_c5 = new_c5
        else:
            # Accept with probability based on temperature
            if k > 0:
                T = T_start * (1 - k / max_iter)
                if np.random.rand() < np.exp(-(new_c5 - best_c5) / T):
                    h = h_new.copy()
        
        k += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    
    # Phase 1: Binary optimization (MILP or greedy)
    print("Phase 1: Binary optimization...")
    binary_solution = solve_binary_milp()
    
    if binary_solution is not None:
        h0 = binary_solution.astype(float)
        # Evaluate the binary solution
        c5_binary = evaluate_c5(h0)
        print(f"Binary MILP solution: C5 = {c5_binary}")
    else:
        print("MILP failed, using greedy construction...")
        binary_solution = greedy_binary_construction()
        h0 = binary_solution.astype(float)
        c5_binary = evaluate_c5(h0)
        print(f"Greedy binary solution: C5 = {c5_binary}")
    
    # Phase 2: Continuous refinement using sub-gradient descent
    print("Phase 2: Continuous refinement...")
    phase2_time = 0.35 * budget_s
    h_contin, c5_contin = deterministic_subgradient_descent(h0, phase2_time)
    print(f"Sub-gradient refinement: C5 = {c5_contin}")
    
    # Phase 3: Stochastic fine-tuning
    print("Phase 3: Stochastic refinement...")
    phase3_time = budget_s - (0.4 * budget_s + phase2_time)
    h_final, c5_final = stochastic_refinement(h_contin, phase3_time)
    
    # Return the best solution found
    final_h = h_final.copy()
    final_c5 = evaluate_c5(final_h)
    
    print("Final result:", final_c5)
    return (final_h.tolist(), float(final_c5), n_points)
```
