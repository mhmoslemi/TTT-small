def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def perturb_solution(h, strength=0.02):
    """Perturb solution slightly"""
    h_new = h + np.random.normal(0, strength, n_points)

        # Gradient step
        h_new = h - learning_rate * grad_approx
        
        # Project back to constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)

            h = h_new.copy()

            # Reduce learning rate if no improvement
            learning_rate *= 0.95
            
        if i % 100 == 0:
            # Occasionally recompute gradient to check convergence
            c5_new = evaluate_c5(h)
            if abs(c5_current - c5_new) < tolerance:
                break
                
    return h
