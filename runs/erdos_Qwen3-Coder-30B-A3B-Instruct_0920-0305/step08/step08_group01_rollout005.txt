# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_shift_matrices(n):
    """Compute shift matrices R_s as sparse matrices"""
    R_list = []
    for s in range(1, (n+1)//2):  # Only half since s and n-s are symmetric
        R_s = np.zeros((n, n))
        for i in range(n - s):
            R_s[i, i + s] = 1
        R_list.append(R_s)
    return R_list

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def project_to_sum(h, target_sum):
    """Project to sum constraint"""
    current_sum = np.sum(h)
    if np.abs(current_sum - target_sum) < 1e-10:
        return h
    delta = target_sum - current_sum
    if delta > 0:
        indices = np.argsort(h)
        for i in range(int(delta)):
            h[indices[i]] += 1.0
    else:
        indices = np.argsort(h)[::-1]
        for i in range(int(-delta)):
            h[indices[i]] -= 1.0
    return np.clip(h, 0, 1)

def round_hyperplane(V, target_sum):
    """Randomized hyperplane rounding with exact sum constraint"""
    g = np.random.randn(V.shape[1])
    scores = V @ g
    # Sort and find threshold
    sorted_indices = np.argsort(scores)[::-1]
    threshold_idx = int(target_sum)
    if threshold_idx < len(scores):
        threshold = scores[sorted_indices[threshold_idx]]
    else:
        threshold = np.min(scores) - 1
    x = (scores >= threshold).astype(float)
    # Adjust if necessary to match target sum
    current_sum = np.sum(x)
    diff = int(current_sum - target_sum)
    if diff > 0:
        # Need to flip some 1s to 0s
        flip_indices = sorted_indices[:diff]
        for idx in flip_indices:
            x[idx] = 0
    elif diff < 0:
        # Need to flip some 0s to 1s
        flip_indices = sorted_indices[-diff:]
        for idx in flip_indices:
            x[idx] = 1
    return x

def solve_sdp(n_points, R_list):
    """Solve the SDP relaxation using CVXPY"""
    X = Semidef(n_points, name='X')
    mu = Variable(n_points, name='mu')
    lam = Variable(name='lam')
    
    # Constraints for the structure
    constraints = [
        mu == diag(X),
        0 <= mu,
        mu <= 1,
        sum(mu) == TARGET_SUM,
        X >> 0
    ]
    
    # Add shift constraints
    for i, R_s in enumerate(R_list):
        # trace(R_s @ X) >= lam
        constraints.append(trace(R_s @ X) >= lam)
    
    objective = Maximize(lam)
    prob = Problem(objective, constraints)
    
    # Solve with time limit
    prob.solve(solver=SCS, max_iters=1000, verbose=False)
    
    if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
        return prob.value, X.value, mu.value
    else:
        # Fallback to Nan
        return np.nan, None, None

def local_search_binary(h, max_iter=10000):
    """Swap-based local search to reduce the worst overlap"""
    h = h.copy()
    current_corr = correlate(h, 1-h, mode="full")
    worst_shift = np.argmax(current_corr)
    max_overlap = current_corr[worst_shift]
    
    # Create sets of contributing indices
    Bad = set()
    Good = set()
    
    for i in range(len(h)):
        if h[i] == 1 and i + worst_shift < len(h) and h[i + worst_shift] == 0:
            Bad.add(i)
        if h[i] == 0 and i + worst_shift < len(h) and h[i + worst_shift] == 1:
            Good.add(i)
    
    for _ in range(max_iter):
        if not Bad or not Good:
            break
        # Pick a random element from Bad and Good
        i = np.random.choice(list(Bad))
        j = np.random.choice(list(Good))
        
        # Swap
        h[i], h[j] = h[j], h[i]
        
        # Update local sets
        Bad.discard(i)
        Good.discard(j)
        if h[i] == 1 and i + worst_shift < len(h) and h[i + worst_shift] == 0:
            Bad.add(i)
        if h[j] == 0 and j + worst_shift < len(h) and h[j + worst_shift] == 1:
            Good.add(j)
        if h[i] == 0 and i + worst_shift < len(h) and h[i + worst_shift] == 1:
            Good.add(i)
        if h[j] == 1 and j + worst_shift < len(h) and h[j + worst_shift] == 0:
            Bad.add(j)
            
        # Recalculate the worst overlap
        current_corr = correlate(h, 1-h, mode="full")
        new_overlap = np.max(current_corr)
        if new_overlap < max_overlap:
            max_overlap = new_overlap
            # Update worst_shift based on potential shift
            worst_shift = np.argmax(current_corr)
            # Need refresh Bad, Good sets
            Bad.clear()
            Good.clear()
            for k in range(len(h)):
                if h[k] == 1 and k + worst_shift < len(h) and h[k + worst_shift] == 0:
                    Bad.add(k)
                if h[k] == 0 and k + worst_shift < len(h) and h[k + worst_shift] == 1:
                    Good.add(k)

    return h, evaluate_c5(h)

def upscale_pattern(pattern_coarse, fine_n):
    """Upscale pattern"""
    r = fine_n // len(pattern_coarse)
    pattern_fine = np.repeat(pattern_coarse, r)
    pattern_fine = pattern_fine[:fine_n]
    return pattern_fine
