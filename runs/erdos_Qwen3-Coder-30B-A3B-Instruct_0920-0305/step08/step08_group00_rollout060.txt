```python
import numpy as np
from scipy.signal import correlate
import time

# Global constants
n_points = 200
dx = 2.0 / n_points

def compute_corr_int(h):
    """Compute correlation using FFT and return integer values"""
    h_rev = h[::-1].astype(np.float64)
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    # round to nearest integer to handle fft floating point errors
    corr_int = np.rint(corr).astype(int)
    return corr_int

def compute_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr_int = compute_corr_int(h)
    return dx * np.max(corr_int)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Generate initial binary pattern with exactly n/2 ones
    h0 = np.zeros(n_points, dtype=int)
    h0[:n_points//2] = 1
    np.random.shuffle(h0)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Simulated Annealing
    T0 = 1.0
    T = T0
    cooling_rate = 0.9995
    min_T = 1e-6
    max_iter = 100000
    
    for i in range(max_iter):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Generate candidate by swapping a random 1 and 0
        ones_indices = np.where(best_h == 1)[0]
        zeros_indices = np.where(best_h == 0)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            # Avoid edge case where all are 1 or all 0
            break
            
        i_one = np.random.choice(ones_indices)
        i_zero = np.random.choice(zeros_indices)
        
        # Create candidate
        candidate_h = best_h.copy()
        candidate_h[i_one] = 0
        candidate_h[i_zero] = 1
        
        # Compute new C5
        candidate_c5 = compute_c5(candidate_h)
        delta = candidate_c5 - best_c5
        
        # Accept or reject
        if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, min_T)):
            best_h = candidate_h
            best_c5 = candidate_c5
            
        # Cool temperature
        T = max(T * cooling_rate, min_T)
    
    # Final local refinement
    if time.time() - start_time < 0.95 * budget_s:
        # Try hill climbing to improve further
        patience = 1000
        no_improve = 0
        local_h = best_h.copy()
        local_c5 = best_c5
        
        while no_improve < patience and time.time() - start_time < 0.95 * budget_s:
            # Generate random swaps
            swaps_tried = 0
            best_swap = None
            best_swap_c5 = local_c5
            
            while swaps_tried < 50 and time.time() - start_time < 0.95 * budget_s:
                i_one = np.random.choice(np.where(local_h == 1)[0])
                i_zero = np.random.choice(np.where(local_h == 0)[0])
                
                test_h = local_h.copy()
                test_h[i_one] = 0
                test_h[i_zero] = 1
                
                test_c5 = compute_c5(test_h)
                swaps_tried += 1
                
                if test_c5 < best_swap_c5:
                    best_swap_c5 = test_c5
                    best_swap = (i_one, i_zero)
            
            if best_swap is not None:
                i_one, i_zero = best_swap
                local_h[i_one] = 0
                local_h[i_zero] = 1
                local_c5 = best_swap_c5
                no_improve = 0
            else:
                no_improve += 1
                
        best_h = local_h
        best_c5 = local_c5
    
    # Convert to float list for return
    return (best_h.astype(float).tolist(), float(best_c5), n_points)
```
