```python
import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

# Global parameters
TARGET_SUM = None
n_points = 180
dx = 2.0 / n_points
m_intervals = 5  # Number of intervals, should be <= 5 for k=5
n_eval = 180

def decode_intervals(z):
    """Decode z to interval parameters (lengths and gaps)"""
    # z in [0,1]^(2*m)
    # First m values are lengths, next m are gaps
    l = np.array(z[:m_intervals])
    g = np.array(z[m_intervals:])
    # Map to proper ranges
    l = 1e-4 + (1 - 2e-4) * l  # enforce l > 0
    g = g  # No bound needed for gaps
    
    # Total length should be 1
    l = l * (1.0 / np.sum(l))
    
    # Compute cumulative positions
    s = np.zeros(m_intervals)
    e = np.zeros(m_intervals)
    s[0] = 0
    e[0] = l[0]
    for i in range(1, m_intervals):
        s[i] = e[i-1] + g[i-1]
        e[i] = s[i] + l[i]
    
    # If we exceed the domain [0,2], scale the gaps
    if e[-1] > 2:
        scale = 2 / e[-1]
        l *= scale
        g *= scale
        e = s + l
    
    return s, e

def build_binary_from_intervals(s, e, n):
    """Build binary vector from interval endpoints"""
    h_bin = np.zeros(n)
    x = np.linspace(0, 2, n, endpoint=False) + dx / 2  # grid centers
    for i in range(len(s)):
        mask = (x >= s[i]) & (x <= e[i])
        h_bin[mask] = 1
    return h_bin

def compute_c5_fft(h):
    """Fast C5 computation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def objective(z):
    s, e = decode_intervals(z)
    h_bin = build_binary_from_intervals(s, e, n_eval)
    c5 = compute_c5_fft(h_bin)
    return c5

def local_refinement(theta_de, num_iter=1000):
    """Perform deterministic local refinement around a solution"""
    s, e = decode_intervals(theta_de)
    h_bin = build_binary_from_intervals(s, e, n_eval)
    best_c5 = compute_c5_fft(h_bin)
    best_s, best_e = s.copy(), e.copy()
    
    for iteration in range(num_iter):
        improved = False
        # Try to move each interval boundary
        for i in range(len(s)):
            # Try left boundary
            s_l = s[i] - dx / 4
            if i == 0:
                s_l = max(0, s_l)
            else:
                s_l = max(s[i-1] + dx, s_l)
            
            if s_l < e[i] - dx:
                e_l = e[i]
                s_trial = s.copy()
                s_trial[i] = s_l
                s_trial, e_trial = s_trial, e.copy()
                # Recompute e coordinates
                for j in range(1, len(s_trial)):
                    s_trial[j] = e_trial[j-1] + (s_trial[j] - e_trial[j-1]) if s_trial[j] > e_trial[j-1] else e_trial[j-1]
                    e_trial[j] = s_trial[j] + (e_trial[j] - s_trial[j])
                h_bin = build_binary_from_intervals(s_trial, e_trial, n_eval)
                c5_trial = compute_c5_fft(h_bin)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_s, best_e = s_trial.copy(), e_trial.copy()
                    improved = True

            # Try right boundary
            e_r = e[i] + dx / 4
            if i == (len(e) - 1):
                e_r = min(2, e_r)
            else:
                e_r = min(s[i+1] - dx, e_r)
                
            if e_r > s[i] + dx:
                s_r = s[i]
                s_trial = s.copy()
                e_trial = e.copy()
                e_trial[i] = e_r
                h_bin = build_binary_from_intervals(s_trial, e_trial, n_eval)
                c5_trial = compute_c5_fft(h_bin)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_s, best_e = s_trial.copy(), e_trial.copy()
                    improved = True
                    
        if not improved:
            break
            
    return best_s, best_e, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial population of random intervals
    def setup_initial_population(size):
        pops = []
        for i in range(size):
            z = np.random.rand(2 * m_intervals)
            pops.append(z)
        return pops
    
    bounds = [(0, 1) for _ in range(2 * m_intervals)]
    initial_pop = setup_initial_population(10)

    # run differential evolution
    de_start = time.time()
    res = differential_evolution(
        objective,
        bounds,
        maxiter=budget_s * 0.6 / 0.05,  # approx 1000 iterations
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        callback=lambda x, convergence: None,
        disp=False
    )
    de_time = time.time() - de_start
    
    # Refine the best solution
    refined_s, refined_e, refined_c5 = local_refinement(res.x, num_iter=500)
    best_c5 = refined_c5
    final_s, final_e = refined_s, refined_e
    
    # Use fallback if needed
    if best_c5 > 0.38092 or de_time < 0.1:
        # Generate random intervals and refine
        def random_intervals():
            z = np.random.rand(2 * m_intervals)
            s, e = decode_intervals(z)
            return s, e
        
        best_s_fallback = np.array([0]*m_intervals)
        best_e_fallback = np.array([0]*m_intervals)
        best_c5_fallback = float('inf')
        
        for _ in range(5000):
            s, e = random_intervals()
            h_bin = build_binary_from_intervals(s, e, n_eval)
            c5 = compute_c5_fft(h_bin)
            if c5 < best_c5_fallback:
                best_c5_fallback = c5
                best_s_fallback = s.copy()
                best_e_fallback = e.copy()
        
        # Refine the best from randomness
        s_refined, e_refined, c5_refined = local_refinement(np.concatenate([best_s_fallback, best_e_fallback]), num_iter=1000)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            final_s, final_e = s_refined, e_refined
    
    # Reconstruct the vector h
    h_final = build_binary_from_intervals(final_s, final_e, n_points)
    
    # Final verification using np.correlate
    corr = correlate(h_final, 1-h_final, mode='full')
    final_c5 = np.max(corr) * dx
    return (h_final.tolist(), float(final_c5), n_points)
```
