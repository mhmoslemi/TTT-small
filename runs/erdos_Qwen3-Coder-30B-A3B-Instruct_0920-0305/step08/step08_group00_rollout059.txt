```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    dx = 2.0 / len(h)
    corr = np.fft.ifft(np.fft.fft(h[::-1], 2*len(h)-1) * np.fft.fft(1-h, 2*len(h)-1)).real
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    n_points = 120
    dx = 2.0 / n_points
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with the provided construction
    h0 = np.array(initial_h_values)
    h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), h0)
    h0 = np.clip(h0, 0, 1)
    h0 = h0 / np.sum(h0) * (n_points / 2)  # Re-scale to have sum = n_points/2
    h0 = np.round(h0).astype(int)  # Force binary
    
    # Use CVXPY to create a MILP
    # Variables h_i = 0 or 1
    h = cp.Variable(n_points, integer=True)
    
    # Objective: minimize C
    C = cp.Variable()
    
    # All constraints for overlap sums <= C
    constraints = []
    for k in range(-n_points+1, n_points):
        # correlation sum for shift k
        if k >= 0:
            # h_i * (1 - h_{i+k}) for i = 0..n-k-1
            expr = 0
            for i in range(n_points - k):
                expr += h[i] * (1 - h[i+k])
        else:
            # h_{i-k} * (1 - h_i) for i = 0..n+k-1
            expr = 0
            for i in range(n_points + k):
                expr += h[i-k] * (1 - h[i])
                
        constraints.append(expr <= C)
    
    # Sum constraint
    constraints.append(cp.sum(h) == n_points / 2)
    
    # Bounds
    for i in range(n_points):
        constraints.append(h[i] >= 0)
        constraints.append(h[i] <= 1)

    # Objective function
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    try:
        # Try solving with CBC as a first attempt, or others if not available
        prob.solve(solver=cp.CBC, verbose=False, maxSeconds=budget_s*0.6)
        
        # Check if we have a solution
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = np.array([float(x.value) for x in h])
        else:
            h_sol = None
    except Exception as e:
        h_sol = None
        
    # Fallback if MILP fails: use rounding of a continuous relaxation
    if h_sol is None or np.any((h_sol < 0) | (h_sol > 1)):
        try:
            # Solve the continuous relaxation
            h_cont = cp.Variable(n_points)
            constraints_cont = []
            
            # Sum constraint
            constraints_cont.append(cp.sum(h_cont) == n_points / 2)
            
            # Bounds
            for i in range(n_points):
                constraints_cont.append(h_cont[i] >= 0)
                constraints_cont.append(h_cont[i] <= 1)
                
            # Objective: same but directly penalized rather than using auxiliary variable
            # Use this trick: penalize non-binary values: add a small term like 1e-6 * sum(h*(1-h))
            # But we'll still solve a similar problem with max correlation via an LP relaxation
            objective_cont = cp.Minimize(C)
            prob_cont = cp.Problem(objective_cont, constraints_cont)
            prob_cont.solve(solver=cp.ECOS, verbose=False)
            
            if prob_cont.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = np.array([float(x.value) for x in h_cont])
                # Round to binary
                h_sol = np.round(h_sol)
                # Ensure sum = n/2 by setting largest entries to 1
                idxs = np.argsort(-h_sol)
                ones = int(n_points / 2)
                h_sol.fill(0)
                h_sol[idxs[:ones]] = 1
            else:
                h_sol = np.round(h0).astype(int)
        except:
            h_sol = np.round(h0).astype(int)
    
    # Final refinement with local search
    best_h = h_sol.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Swap one 1 and one 0 if it improves
    if time.time() < start_time + budget_s * 0.95:
        for i in range(10000):  # Enough iterations to find good swaps
            if time.time() > start_time + budget_s * 0.95:
                break
            
            i_try = np.random.randint(0, n_points)
            j_try = np.random.randint(0, n_points)
            
            if best_h[i_try] == 1 and best_h[j_try] == 0:
                # Try swapping
                h_temp = best_h.copy()
                h_temp[i_try] = 0
                h_temp[j_try] = 1
                
                if np.sum(h_temp) == n_points / 2:
                    c5_new = evaluate_c5(h_temp)
                    if c5_new < best_c5:
                        best_h = h_temp
                        best_c5 = c5_new
    
    return (best_h.tolist(), float(best_c5), n_points)
```
