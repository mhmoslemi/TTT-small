def compute_critical_shifts(m, lengths):
    """Compute all critical shift values"""
    s = np.concatenate([[0], np.cumsum(lengths)])
    shifts = set()
    for i in range(m):
        for j in range(m):
            shifts.add(abs(s[i] - s[j]))
            shifts.add(abs((2 - s[i]) - (2 - s[j])))
            shifts.add(s[i])
            shifts.add(2 - s[j])
    return sorted(list(shifts))

def J(k, lengths, heights):
    """Compute integral of h(x)(1-h(x+k)) for a given shift k"""
    s = np.concatenate([[0], np.cumsum(lengths)])
    m = len(lengths)
    
    # Linear term: integral of h(x) over [0, 2-k]
    linear_term = 0.0
    end = 2 - k
    pos = 0
    for i in range(m):
        start = s[i]
        if start >= end:
            break
        end_i = s[i+1]
        if end_i <= end:
            linear_term += heights[i] * (end_i - start)
        else:
            linear_term += heights[i] * (end - start)
            break
    
    # Product term: sum over all pairs of intervals
    product_term = 0.0
    for i in range(m):
        start_i = s[i]
        end_i = s[i+1]
        if start_i >= 2 - k:
            break
        for j in range(m):
            start_j = s[j]
            end_j = s[j+1]
            if start_j >= 2 or start_j >= end_i:
                continue
            # Compute overlap length
            overlap_start = max(start_i, start_j + k)
            overlap_end = min(end_i, end_j + k)
            if overlap_start < overlap_end:
                overlap_len = overlap_end - overlap_start
                product_term += heights[i] * heights[j] * overlap_len
    
    return linear_term - product_term

def evaluate_c5_from_params(lengths, heights):
    """Evaluate C5 using critical shifts"""
    m = len(lengths)
    critical_shifts = compute_critical_shifts(m, lengths)
    
    # For simplicity, evaluate exactly at critical shifts
    values = [J(k, lengths, heights) for k in critical_shifts]
    return max(values)
