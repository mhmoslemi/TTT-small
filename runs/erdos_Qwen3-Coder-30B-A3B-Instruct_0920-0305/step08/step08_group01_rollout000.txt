import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def compute_c5_integer(h):
    """Return integer max correlation count"""
    corr = compute_corr_fft(h)
    return int(np.max(corr) / dx)

def random_binary_solution():
    """Generate a random binary solution with exactly target_sum ones"""
    h = np.zeros(n_points, dtype=float)
    ones_indices = np.random.choice(n_points, size=TARGET_SUM, replace=False)
    h[ones_indices] = 1.0
    return h

def simulated_annealing_binary(max_iter=20000, temp_start=0.01, temp_end=0.0001):
    """Simulated Annealing for binary patterns"""
    # Initial solution
    h = random_binary_solution()
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Temperature schedule
    temp = temp_start
    decay_rate = (temp_end / temp_start) ** (1.0 / max_iter)
    
    # Keep track of time
    start_time = time.time()
    
    for iteration in range(max_iter):
        # Check budget
        if time.time() - start_time > 580:
            break
            
        # Swap two random bits: one from 1s and one from 0s
        ones_indices = np.where(h == 1)[0]
        zeros_indices = np.where(h == 0)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            continue
            
        i = np.random.choice(ones_indices)
        j = np.random.choice(zeros_indices)
        
        # Perform the swap
        h[i] = 0
        h[j] = 1
        
        # Compute new C5
        new_c5 = evaluate_c5(h)
        
        # Accept or reject
        if new_c5 < current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Accept with probability
            delta = new_c5 - current_c5
            if np.random.random() < np.exp(-delta / temp):
                current_c5 = new_c5
            else:
                # Revert swap
                h[i] = 1
                h[j] = 0
        
        # Update temperature
        temp *= decay_rate
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points // 2
    best_c5 = float('inf')
    best_h = None
    
    # Run multiple restarts with SA
    restarts = 0
    while time.time() - start_time < budget_s * 0.95:
        restarts += 1
        h, c5 = simulated_annealing_binary(max_iter=10000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final optimization using gradient-based method if budget allows
    if time.time() - start_time < budget_s * 0.99:
        # Simple gradient-based refinement
        from scipy.optimize import minimize
        def objective(h):
            return evaluate_c5(h)
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
        
        try:
            res = minimize(
                objective,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_c5(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Ensure final solution is available
    return (best_h.tolist(), float(best_c5), n_points)
