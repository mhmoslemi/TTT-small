import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = np.correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def project_sum_box(v, target_sum):
    """Project vector v onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(v + (a + b)/2, 0, 1)
    return result

def solve_milp_binary(N, M):
    """
    Attempt to solve MILP for binary case (y_i ∈ {0,1}) to maximize min overlap.
    """
    try:
        y = cp.Variable(N, boolean=True)
        t = cp.Variable()
        constraints = [cp.sum(y) == M]

        # We only add constraints for k=1,...,N-1; k=0 is automatically satisfied
        for k in range(1, N):
            if k >= N: continue
            w = cp.Variable(N - k)
            constraints += [w <= y[:N-k]]
            constraints += [w <= y[k:]]
            constraints += [w >= y[:N-k] + y[k:] - 1]
            constraints += [w >= 0]
            constraints += [cp.sum(w) >= t]

        prob = cp.Problem(cp.Maximize(t), constraints)
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=200)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return y.value
        else:
            return None
    except:
        return None

def subgradient_descent(h0, max_iter=20000):
    """Run sub-gradient descent, improving from h0 by moving in direction that minimizes maximum overlap."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for t in range(1, max_iter+1):
        # Find worst shift k_max
        corr = np.correlate(h, 1-h, mode="full")
        k_max = np.argmax(corr) - (len(h)-1)
        
        # Compute sub gradient for shift k_max
        g = np.ones(len(h))
        if k_max >= 0:
            if k_max < len(h):
                g[:len(h)-k_max] -= h[k_max:]
            else:
                g[:] = 0.0
        else:
            k = -k_max
            if k < len(h):
                g[k:] -= h[:len(h)-k]
            else:
                g[:] = 0.0
                
        if k_max > 0:
            if k_max < len(h):
                g[k_max:] -= h[:len(h)-k_max]
            else:
                g[:] = 0.0
        elif k_max < 0:
            k = -k_max
            if k < len(h):
                g[:len(h)-k] -= h[k:]
            else:
                g[:] = 0.0

        # Gradient step
        alpha = 0.1 / np.sqrt(t)
        h_new = h - alpha * g
        
        # Projection
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, TARGET_SUM)
        
        # Accept improvement
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            h = h_new.copy()
            best_h = h.copy()
            best_c5 = new_c5
        else:
            h = h_new.copy()
            
        if t % 1000 == 0:
            print(f"SGD iter {t}, C5 = {best_c5:.6f}")

    return best_h, best_c5

def stochastic_refinement(h0, max_iter=5000):
    """Quick refinement using simulated annealing / random swaps."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    temp_schedule = 1.0 / (1.0 + np.arange(max_iter) * 1e-4)
    
    for i in range(max_iter):
        # Random swap
        idx1, idx2 = np.random.choice(len(h), size=2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        
        h_new = h.copy()
        h_new[idx1] += delta
        h_new[idx2] -= delta
        
        # Correct norms
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, TARGET_SUM)
        
        # Evaluate improvement
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            h = h_new.copy()
            best_h = h.copy()
            best_c5 = new_c5
        else:
            # Accept with probability based on simulated annealing
            prob = np.exp(-(new_c5 - best_c5) / temp_schedule[i])
            if np.random.rand() < prob:
                h = h_new.copy()

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points / 2
    final_h = None
    final_c5 = float('inf')
    
    # Phase 1: Solve MILP for binary optimum if possible
    print("Solving MILP binary...")
    sol = solve_milp_binary(n_points, TARGET_SUM)
    if sol is not None:
        # Convert to float h
        h0 = sol.astype(float)
        c5_initial = evaluate_c5(h0)
        print(f"Initial binary C5 = {c5_initial:.6f}")
        final_h = h0
        final_c5 = c5_initial
    else:
        h0 = np.random.uniform(0, 1, n_points)
        h0 = project_sum_box(h0, TARGET_SUM)
        final_h = h0
        final_c5 = evaluate_c5(h0)
        print(f"Initial random C5 = {final_c5:.6f}")

    # Phase 2: Run sub-gradient descent
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        print("Running SGD...")
        h_opt, c5_opt = subgradient_descent(final_h, max_iter=min(20000, max(2000, int(remaining_time*200))))
        if c5_opt < final_c5:
            final_h = h_opt
            final_c5 = c5_opt
            print(f"SGD improved C5 to = {final_c5:.6f}")

    # Phase 3: Stochastic refinement to escape local minima
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        print("Running stochastic refinement...")
        h_refined, c5_refined = stochastic_refinement(final_h, max_iter=min(5000, int(remaining_time*500)))
        if c5_refined < final_c5:
            final_h = h_refined
            final_c5 = c5_refined
            print(f"Stochastic refined C5 to = {final_c5:.6f}")

    return (final_h.tolist(), float(final_c5), n_points)
