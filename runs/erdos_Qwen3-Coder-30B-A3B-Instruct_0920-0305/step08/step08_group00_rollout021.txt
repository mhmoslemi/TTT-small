```python
import numpy as np
from scipy.signal import correlate
import time
import math

# Global variables
n_points = 200  # Using 200 points for higher resolution
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def compute_corr_fft(h):
    """Compute correlation using FFT - returns integer counts"""
    h_rev = h[::-1].astype(np.float64)
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.rint(corr).astype(int)

def evaluate_c5(h):
    """Evaluate C5 using FFT correlation"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with random binary pattern
    h = np.zeros(n_points, dtype=int)
    h[:TARGET_SUM] = 1
    np.random.shuffle(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Simulated Annealing parameters
    T0 = 1.0  # Initial temperature, should be comparable to delta_c5
    T = T0
    cooling_rate = 0.9995
    min_temperature = 1e-6
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Temperature schedule based on expected changes
    max_iter = 100000  # Max iterations
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s:
        # Generate random swap
        ones_indices = np.where(h == 1)[0]
        zeros_indices = np.where(h == 0)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            break
            
        i_one = np.random.choice(ones_indices)
        i_zero = np.random.choice(zeros_indices)
        
        # Create candidate solution
        h_new = h.copy()
        h_new[i_one] = 0
        h_new[i_zero] = 1
        
        # Compute new correlation
        corr_new = compute_corr_fft(h_new)
        c5_new = dx * np.max(corr_new)
        
        # Compute change in cost (difference in max correlation)
        delta = c5_new - best_c5
        
        # Accept based on Metropolis criterion
        if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, 1e-10)):
            h = h_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        
        # Update temperature
        T = max(T * cooling_rate, min_temperature)
        iter_count += 1
        
        # Occasionally reheat if stuck
        if iter_count % 1000 == 0:
            T = min(T * 1.1, T0)
    
    # Perform final local hill climbing
    local_iter = 10000
    patience = 100
    
    h_current = best_h.copy()
    c5_current = best_c5
    
    no_improve = 0
    for _ in range(local_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
        # Generate random swaps  
        swaps = []
        for _ in range(20):
            ones_indices = np.where(h_current == 1)[0]
            zeros_indices = np.where(h_current == 0)[0]
            if len(ones_indices) == 0 or len(zeros_indices) == 0:
                continue
            i_one = np.random.choice(ones_indices)
            i_zero = np.random.choice(zeros_indices)
            swaps.append((i_one, i_zero))
        
        improved = False
        for i_one, i_zero in swaps:
            h_test = h_current.copy()
            h_test[i_one] = 0
            h_test[i_zero] = 1
            
            corr_test = compute_corr_fft(h_test)
            c5_test = dx * np.max(corr_test)
            
            if c5_test < c5_current:
                h_current = h_test
                c5_current = c5_test
                improved = True
                no_improve = 0
                break
                
        if not improved:
            no_improve += 1
            if no_improve >= patience:
                break
    
    # Final evaluation
    final_c5 = evaluate_c5(h_current)
    
    return (h_current.astype(float).tolist(), float(final_c5), n_points)
```
