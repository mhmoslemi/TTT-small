def get_interval_bounds_and_lengths(h):
    """Break h into pieces at the discontinuities, return interval bounds and lengths."""
    x = np.linspace(0., 2., len(h) + 1)
    # Compute breakpoints where h changes
    jump_points = [0.0]  # Always start at 0
    for i in range(len(h) - 1):
        # Near values are equal
        if abs(h[i] - h[i+1]) > 1e-10:
            jump_points.append(x[i+1])
    jump_points.append(2.0)
    intervals = [(jump_points[i], jump_points[i+1]) for i in range(len(jump_points)-1)]
    return intervals, jump_points

def piecewise_overlap(h, k, intervals):
    """Compute overlap integral for a given k and step function h."""
    if k <= 0:
        return 0.0
    if k >= 2:
        return 0.0
    
    # Integration limits
    lower_limit = 0.0
    upper_limit = 2.0 - k

    # Start with total mass = 1:
    total_integrand = 0.0
    total_mass = 0.0

    for i, (start, end) in enumerate(intervals):
        # Overlap region of [start, end] and [start+k, end+k], projected on [0, 2-k]
        overlap_start = max(start, k)
        overlap_end = min(end, 2.0 - k + k)  # equal to min(end, 2-k)
        overlap_start = max(overlap_start, 0.0)
        overlap_end = min(overlap_end, 2.0 - k)
        
        if overlap_start >= overlap_end:
            continue
        
        # Compute max of h over [overlap_start, overlap_end]
        height = h[i]
        integral_val = height * (overlap_end - overlap_start)
        total_integrand += integral_val

        # Also compute integral of h(x+k) over [start+k, end+k],
        # but only where the region is within [0, 2-k]
        start_shifted = start + k
        end_shifted = end + k
        if h[i] == 0:
            continue
        try:
            # When shifted interval intersects [0, 2-k]:
            overlap_interval_start = max(start_shifted, 0)
            overlap_interval_end = min(end_shifted, 2 - k)
            if overlap_interval_start < overlap_interval_end:
                # This is the area between the functions
                area = h[i]**2 * (overlap_interval_end - overlap_interval_start)
                # Area under h(x) * h(x + k)
                total_mass += area
        except Exception:
            pass
    return total_integrand - total_mass

def eval_c5_from_step_function(h):
    """Compute C5 = max_{k≥0} ∫ h(x)(1 - h(x+k)) dx."""
    intervals, _ = get_interval_bounds_and_lengths(h)
    k_vals = np.linspace(0.0, 2.0, 200)  # Sample critical shifts
    overlaps = []
    for k in k_vals:
        over = piecewise_overlap(h, k, intervals)
        overlaps.append(over)
    return max(overlaps)
