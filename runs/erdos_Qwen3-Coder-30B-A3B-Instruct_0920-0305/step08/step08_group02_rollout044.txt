import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate
from scipy.optimize import minimize

# Set seed for reproducibility
np.random.seed(42)

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Evaluate C5 using FFT correlation"""
    corr = compute_corr_fft(h)
    dx = 2.0 / len(h)
    return dx * np.max(corr)

def project_sum_box(h, target_sum):
    """Project onto sum constraint and box constraints"""
    def objective(lam):
        proj = np.clip(h + lam, 0, 1)
        return np.sum(proj) - target_sum

    a, b = -1.0, 1.0
    while objective(a) * objective(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = objective(m)
        if abs(fm) < tol:
            break
        if fm * objective(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def build_binary_milp(N, M):
    """Construct a binary MILP for max-min autocorrelation"""
    # Variables
    y = cp.Variable(N, boolean=True)
    w = cp.Variable((N-1, N), nonneg=True)  # w[k][i] = y[i] * y[i+k]
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.sum(y) == M
    ]

    # Product constraints
    for k in range(1, N):
        for i in range(N - k):
            constraints += [
                w[k-1, i] <= y[i],
                w[k-1, i] <= y[i+k],
                w[k-1, i] >= y[i] + y[i+k] - 1,
                w[k-1, i] >= 0
            ]

    # Minimum autocorrelation constraint
    for k in range(1, N):
        constraints += [
            cp.sum(w[k-1, :N-k]) >= t
        ]

    # Objective
    objective = cp.Maximize(t)

    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CBC, max_seconds=210, verbose=False)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            return y.value
    except:
        pass

    try:
        prob.solve(solver=cp.GLPK_MI, max_seconds=210, verbose=False)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            return y.value
    except:
        pass

    return None

def subgradient_descent(h0, max_iter=5000):
    """Apply projected subgradient descent to refine solution"""
    h = h0.copy()
    dx = 2.0 / len(h)
    
    for i in range(max_iter):
        # Compute current correlations
        corr = compute_corr_fft(h)
        k_star = np.argmax(corr)
        
        # Compute subgradient
        g = np.zeros_like(h)
        if k_star < len(h):
            g += (1 - np.roll(h, k_star))
        if k_star > 0:
            g += (1 - np.roll(h, -k_star))

        # Step size (diminishing)
        alpha = 0.5 * dx / np.sqrt(i + 1)

        # Update
        h_new = h - alpha * g
        h_new = project_sum_box(h_new, len(h) / 2)

        # Evaluate improvement
        old_c5 = evaluate_c5(h)
        new_c5 = evaluate_c5(h_new)

        if new_c5 < old_c5:
            h = h_new.copy()
        else:
            # Sometimes do a small perturbation if no improvement
            if np.random.rand() < 0.05:
                h = h_new.copy()
    return h

def stochastic_refinement(h0, max_time=10):
    """Perform lightweight stochastic refinement"""
    h = h0.copy()
    start_time = time.time()
    dx = 2.0 / len(h)
    
    while time.time() - start_time < max_time:
        # Swap move
        i, j = np.random.choice(len(h), size=2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        h[i] += delta
        h[j] -= delta
        
        # Clip
        h = np.clip(h, 0, 1)
        # Project
        h = project_sum_box(h, len(h) / 2)
        # Evaluate
        new_c5 = evaluate_c5(h)
        old_c5 = evaluate_c5(h0)
        
        if new_c5 < old_c5:
            h0 = h.copy()
        else:
            # Accept with probability based on temperature
            if np.random.rand() < 0.1:
                h0 = h.copy()
                
    return h0

def initial_binary_construction(N):
    """Construct initial binary solution for the problem"""
    # Start with uniform distribution then refine greedily
    h = np.zeros(N)
    # Randomly place some ones
    indices = np.random.choice(N, size=N//4, replace=False)
    h[indices] = 1
    # Project to maintain sum
    h = project_sum_box(h, N/2)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    N = 120
    dx = 2.0 / N
    M = N / 2
    best_h = None
    best_c5 = float('inf')
    
    # Phase 1: Binary MILP solving (if possible)
    phase1_end = start_time + 0.4 * budget_s
    binary_sol = None
    if time.time() < phase1_end:
        binary_sol = build_binary_milp(N, M)
        if binary_sol is not None:
            h0 = binary_sol.astype(float)
            c5 = evaluate_c5(h0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h0.copy()

    # Phase 2: Subgradient descent refinement
    phase2_end = start_time + 0.75 * budget_s
    if time.time() < phase2_end and best_h is None:
        h0 = initial_binary_construction(N)
        best_h = h0.copy()
        best_c5 = evaluate_c5(h0)
        
    if time.time() < phase2_end and best_h is not None:
        h_refined = subgradient_descent(best_h, max_iter=2000)
        c5_refined = evaluate_c5(h_refined)
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined

    # Phase 3: Stochastic refinement
    phase3_end = start_time + budget_s
    if time.time() < phase3_end and best_h is not None:
        h_final = stochastic_refinement(best_h.copy(), max_time=10)
        c5_final = evaluate_c5(h_final)
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final

    # Fallback: if we have nothing, generate a safe solution
    if best_h is None:
        h0 = initial_binary_construction(N)
        best_h = h0.copy()
        best_c5 = evaluate_c5(best_h)

    return (best_h.tolist(), float(best_c5), N)
