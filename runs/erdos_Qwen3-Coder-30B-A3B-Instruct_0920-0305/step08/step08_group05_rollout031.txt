import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def neighbor_sum(h, lag):
    """Compute neighbor sum for a given lag"""
    lag_abs = abs(lag)
    if lag_abs == 0:
        return 2 * h
    else:
        w = np.zeros_like(h)
        if lag > 0:
            w[:n_points-lag_abs] += h[lag_abs:]
            w[lag_abs:] += h[:-lag_abs]
        else:
            w[lag_abs:] += h[:-lag_abs]
            w[:-lag_abs] += h[lag_abs:]
        return w

    TARGET_SUM = n_points / 2
    
    step_size = 0.05
    no_improve_count = 0
    iteration = 0

        # Compute correlation and find the worst lag
        corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
        idx_max = np.argmax(corr)

        # Compute neighbor sum for the worst lag
        w = neighbor_sum(best_h, lag_max)
        
        # Find candidate indices for transfer
        eps = 1e-8
        low_candidates = np.where(best_h > eps)[0]
        high_candidates = np.where(best_h < 1 - eps)[0]
        
        if len(low_candidates) == 0 or len(high_candidates) == 0:
            break
            
        # Get indices with minimum and maximum w values among candidates
        low_idx = low_candidates[np.argmin(w[low_candidates])]
        high_idx = high_candidates[np.argmax(w[high_candidates])]
        
        # If indices are the same, skip this iteration
        if low_idx == high_idx:
            continue
            
        # Compute transfer amount
        delta = min(step_size, best_h[low_idx], 1 - best_h[high_idx])
        
        if delta < 1e-12:
            continue

        h_candidate[low_idx] -= delta

            no_improve_count = 0

            no_improve_count += 1

            step_size = max(step_size, 1e-6)
            no_improve_count = 0

                    best_c5 = c5_final
        except:
            pass

    return (best_h.tolist(), float(best_c5), n_points)
