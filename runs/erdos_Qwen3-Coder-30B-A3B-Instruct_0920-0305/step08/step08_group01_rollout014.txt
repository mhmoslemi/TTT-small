import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time
from scipy.linalg import eigh

def compute_shift_matrices(n):
    """Compute shift matrices R_s for all s = 1 to floor(n/2)"""
    shift_matrices = []
    for s in range(1, (n // 2) + 1):
        R_s = np.zeros((n, n))
        for i in range(n - s):
            R_s[i, i + s] = 1
        shift_matrices.append(R_s)
    return shift_matrices

def solve_sdp(n, shift_matrices, max_time=100):
    """Solve the SDP relaxation"""
    # Variables
    X = cp.Variable((n, n), symmetric=True)
    mu = cp.Variable(n)
    lam = cp.Variable()

    # Constraints
    constraints = [
        X >> 0,  # PSD
        cp.diag(X) == mu,
        0 <= mu,
        mu <= 1,
        cp.sum(mu) == n / 2,
        mu[0] == 1  # set x_0 = 1 to avoid global translation symmetries
    ]

    # Add shift constraints
    for R_s in shift_matrices:
        constraints.append(cp.trace(R_s @ X) >= lam)

    # Objective
    objective = cp.Maximize(lam)
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000, time_limit=max_time)
        return X.value, mu.value, lam.value
    except Exception:
        return None, None, None

def round_solution(X, mu, n):
    """Round the SDP solution to a binary vector using hyperplane rounding"""
    if X is None or mu is None:
        return None
        
    # Cholesky decomposition or eigenvalue decomposition
    try:
        U, V = eigh(X)
        V = V[:, U > 1e-8]  # Keep only non-negative eigenvalues
        V = V @ np.diag(np.sqrt(U[U > 1e-8]))
    except:
        # Fallback on singular value decomposition
        try:
            U, s, Vt = np.linalg.svd(X)
            V = U[:, :len(s)][np.newaxis, :] * np.sqrt(s)
        except:
            return None
            
    best_x = None
    best_c5 = float('inf')
    
    # Try multiple rounds
    for _ in range(500):
        g = np.random.randn(V.shape[1])
        scores = V @ g
        order = np.argsort(-scores)
        x = np.zeros(n)
        x[order[:n//2]] = 1
        
        # Maximize overlap under binary constraints, so that we don't adjust
        # the entire solution but just time to get best ones
        c = max(correlate(x, 1 - x, mode='full'))
        if c < best_c5:
            best_c5 = c
            best_x = x
            
    return best_x

def swap_improvement(x, max_iter=5000):
    """Input binary vector x. Return better binary vector by local swaps."""
    n = len(x)
    x = x.copy()
    
    # Precompute the maximum correlation and its index
    corr = correlate(x, 1 - x, mode='full')
    max_corr_idx = np.argmax(corr)
    max_corr_val = corr[max_corr_idx]
    
    for iteration in range(max_iter):
        # Recompute correlations to find worst shift
        corr = correlate(x, 1 - x, mode='full')
        s_max = np.argmax(corr)
        max_val = corr[s_max]
        
        # If we've hit the target which is around 0.3807*2
        if max_val < 0.76:
            break
            
        # Find contributing 1's and 0's to the worst shift
        bad_positions = []
        good_positions = []
        
        # Allowed shifts only go up to n-1, but full correlation is of length 2n-1
        s = s_max - n + 1  # index in range [-n+1, n-1]
        
        if s >= 0:
            # x_i * (1 - x_{i+s}) is being counted in shift s
            # Check for 1's at i that have 0 at i+s
            # i ranges from 0 to n-1-s inclusive
            for i in range(n - s):
                if x[i] == 1 and x[i + s] == 0:
                    bad_positions.append(i)
            # i ranges from s to n-1 inclusive to find good 0's
            for i in range(s, n):
                if x[i] == 0 and x[i - s] == 1:
                    good_positions.append(i)
        else:
            # s is negative
            s_rev = -s
            for i in range(s_rev, n):
                if x[i] == 1 and x[i - s_rev] == 0:
                    bad_positions.append(i)
            for i in range(n - s_rev):
                if x[i] == 0 and x[i + s_rev] == 1:
                    good_positions.append(i)

        if not bad_positions or not good_positions:
            break

        # Pick a random bad 1 and random good 0 to swap
        bad = np.random.choice(bad_positions)
        good = np.random.choice(good_positions)
        
        # Perform the swap
        x[bad] = 0
        x[good] = 1
        
        # Check current overlap to decide if better
        new_corr = correlate(x, 1 - x, mode='full')
        new_max = np.max(new_corr)
        
        # If the swap hurt, revert it
        if new_max > max_val:
            x[bad] = 1
            x[good] = 0
            # Backoff or just continue, but we still try random moves
            continue
        else:
            max_val = new_max

    return x
