import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize
import math

def compute_correlation(h):
    """Compute overlap integral using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_c5(h):
    """Fast evaluation of C5"""
    corr = compute_correlation(h)
    return np.max(corr) * (2.0 / len(h))

def project_sum_box(x, target_sum):
    """Project onto sum constraint [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_binary_milp(N, M):
    """Solve binary MILP to maximize minimum autocorrelation"""
    # Variables: y_i binary 0/1, w_{k,i} auxiliary, t lower bound
    y = cp.Variable(N, boolean=True)
    w = cp.Variable((N-1, N), nonneg=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.sum(y) == M,
    ]

    # Linearization constraints for w_{k,i} = y_i * y_{i+k}
    for k in range(1, N):
        for i in range(N - k):
            constraints.extend([
                w[k-1, i] <= y[i],
                w[k-1, i] <= y[i + k],
                w[k-1, i] >= y[i] + y[i + k] - 1,
                w[k-1, i] >= 0
            ])

    # Min autocorrelation constraints
    for k in range(1, N):
        constraints.append(cp.sum(w[k-1, :]) >= t)

    # Objective
    objective = cp.Maximize(t)

    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CBC, max_seconds=0.35)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMALLY_SOLVED:
            y_val = y.value
            t_val = t.value
            return y_val, t_val
    except Exception as e:
        pass
    return None, None

def build_feasible_h(y, N):
    """Build continuous h from binary y"""
    h = y.astype(float)
    return h

def subgradient_descent(h0, max_iter, step_size):    
    """Deterministic subgradient descent"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n = len(h)
    step = step_size

    for i in range(max_iter):
        # Calculate correlation
        corr = compute_correlation(h)
        # Find shift that gives max correlation
        k_star = np.argmax(corr)
        
        # Compute gradient for worst shift
        grad = np.zeros(n)
        for j in range(n):
            if j + k_star < n:
                grad[j] += -h[j + k_star]
            if j >= k_star:
                grad[j] += -h[j - k_star]
            grad[j] += 1
            
        # Update
        h_new = h - step * grad
        h_new = project_sum_box(h_new, n/2)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new
        if i % 20 == 0:
            step *= 0.995
    
    return best_h, best_c5

def stochastic_refinement(h0, budget_s):
    """Simple stochastic refinement using swaps"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    # Repeated swap moves
    while time.time() - start_time < budget_s:
        # Try to swap two values
        i, j = np.random.choice(len(h), 2, replace=False)
        delta = np.random.uniform(-0.03, 0.03)
        h_new = h.copy()
        h_new[i] = np.clip(h_new[i] + delta, 0, 1)
        h_new[j] = np.clip(h_new[j] - delta, 0, 1)
        h_new = project_sum_box(h_new, len(h)/2)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            best_h = h.copy()
        elif np.random.random() < 0.1:
            h = h_new.copy()

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    N = 120
    M = N / 2
    dx = 2.0 / N
    best_c5 = float('inf')
    best_h = None
    
    # Phase 1: Binary MILP (try to maximize minimum autocorrelation)
    h_milp = None
    if time.time() - start_time < 0.4 * budget_s:
        y_opt, t_opt = solve_binary_milp(N, M)
        if y_opt is not None:
            h_milp = build_feasible_h(y_opt, N)
            c5_milp = dx * (M - t_opt)
            if c5_milp < best_c5:
                best_c5 = c5_milp
                best_h = h_milp.copy()
    
    # Phase 2: Deterministic subgradient from MILP (or random fallback)
    if time.time() - start_time > 0.7 * budget_s:
        if best_h is None:
            best_h = np.random.uniform(0, 1, N)
            best_h = project_sum_box(best_h, M)
        return (best_h.tolist(), best_c5, N)
        
    # Use MILP result if available, fallback to random
    h0 = h_milp if h_milp is not None else np.random.uniform(0, 1, N)
    h0 = project_sum_box(h0, M)
    
    # Run deterministic optimization
    remaining_time = 0.35 * budget_s - (time.time() - start_time)
    if remaining_time > 0:
        max_iter = min(1000, max(50, int(remaining_time * 10)))
        step_size = 0.05 * dx
        h_opt, c5_opt = subgradient_descent(h0, max_iter, step_size)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 3: Stochastic refinement
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.01 and best_h is not None:
        h_final, c5_final = stochastic_refinement(best_h.copy(), remaining_time)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    if best_h is None:
        best_h = np.random.uniform(0, 1, N)
        best_h = project_sum_box(best_h, M)
        best_c5 = compute_c5(best_h)

    return (best_h.tolist(), float(best_c5), N)
