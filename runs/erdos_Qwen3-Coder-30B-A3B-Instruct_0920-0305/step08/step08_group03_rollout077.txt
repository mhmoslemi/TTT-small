import numpy as np

# Global constants
TARGET_SUM = 1.0
n_points = 120
dx = 2.0 / n_points

def compute_overlap_for_shift(h_vals, k):
    """Compute the integral over x of h(x)(1 - h(x + k)) for given shift k."""
    # h_vals: array of length n_points, interpreted as piecewise constant
    # This uses approx integrals. More accurate: use continuous version.
    # But we'll use discrete inner product over sampled points for simplicity.
    # k is in [0, 2]
    if k < 0 or k > 2:
        return 0.0
    # Find valid domain for integration: [0, 2-k]
    valid_len = int(np.ceil((2 - k) / dx))
    if valid_len <= 0:
        return 0.0
    
    h = np.array(h_vals)
    # Shifted h
    h_k = np.concatenate([h, [0] * int(np.ceil(k / dx))])[:len(h)]
    integrand = h[:valid_len] * (1 - h_k[:valid_len])
    return dx * np.sum(integrand)

def compute_c5(h_vals):
    """Compute the maximum overlap over all valid k in [0, 2]."""
    max_overlap = 0.0
    # Sample k from 0 to 2 with a sufficient resolution
    n_k_samples = 200  # adjust depending on speed/accuracy goals
    ks = np.linspace(0, 2, n_k_samples)
    for k in ks:
        overlap = compute_overlap_for_shift(h_vals, k)
        max_overlap = max(max_overlap, overlap)
    return max_overlap

def piecewise_to_h(lengths, heights):
    """Convert piecewise lengths and heights to full h array."""
    # lengths = [l1, l2, ..., l_m], heights = [h1, ..., h_m]
    n = n_points
    dx = 2.0 / n
    h = np.zeros(n)
    pos = 0
    for i in range(len(lengths)):
        l = lengths[i]
        h_i = heights[i]
        steps = int(np.round(l / dx))
        if pos + steps > n:
            steps = n - pos
        h[pos:pos+steps] = h_i
        pos += steps
        if pos >= n:
            break
    return h

def interpolate_and_preserve_sum(h_vals, n_points_new=120):
    """Interpolate h_vals from old grid to new grid (of size n_points_new),
    adjusting values so that ∫ h = 1."""
    h_orig = np.array(h_vals)
    dx_orig = 2.0 / len(h_orig)
    dx_new = 2.0 / n_points_new
    x_orig = np.linspace(0, 2, len(h_orig))
    x_new = np.linspace(0, 2, n_points_new)
    h_new = np.interp(x_new, x_orig, h_orig)
    # Rescale to preserve integral
    integral_old = np.sum(h_orig) * dx_orig
    integral_new = np.sum(h_new) * dx_new
    h_new *= (TARGET_SUM / integral_new)
    return h_new

# Define parametric version with low number of intervals
def initialize_with_few_intervals(m=6):
    # Initialize heights to 1 and uniform lengths
    # Set some restrictions to generate starting reasonable config
    lengths = [2.0 / m] * m
    heights = [1.0 / (m//2 + 1)] * (m//2) + [1.0] * (m - m//2)
    # Normalize heights such that sum(heights * lengths) = 1
    total_mass = sum(h * l for h, l in zip(heights, lengths))
    heights = [h * TARGET_SUM / total_mass for h in heights]
    return lengths, heights

def step_function_from_params(params, num_intervals=5):
    """params = [l1, l2, ..., l(num_intervals-1), h1, h2, ..., h(num_intervals-1)]
    
    Final interval length and height computed based on constraints."""
    half = num_intervals - 1
    lengths = params[:half]
    heights = params[half:]
    total_length = sum(lengths)
    assert total_length <= 2, "Over the interval"
    lengths.append(2.0 - total_length)
    
    # Compute final height
    mass_contributed = sum(h * l for h, l in zip(heights, lengths[:-1]))
    final_h = (TARGET_SUM - mass_contributed) / lengths[-1]
    assert 0 <= final_h <= 1, f"Height constraint violated {final_h}"
    heights.append(final_h)
    return lengths, heights

def calculate_c5_wrapper(params):
    """Returns the maximum overlap for given piecewise params."""
    lengths, heights = step_function_from_params(params)
    h_full = piecewise_to_h(lengths, heights)
    c5 = compute_c5(h_full)
    return c5
