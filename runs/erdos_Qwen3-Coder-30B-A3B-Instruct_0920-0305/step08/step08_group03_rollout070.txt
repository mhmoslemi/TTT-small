import time

TARGET_SUM = None
n_points = 256  # Increased resolution for better optimization

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def update_correlation(c, h, b, i, delta):
    """Incrementally update correlation vector when changing h[i] by delta"""
    # Update the correlation values using efficient vectorized operations
    n = len(h)
    
    # b slice contribution: shift from -i to n-1-i 
    if i < n:
        b_slice = b[i:]
        c_slice_start = n - i - 1
        c_slice_end = 2*n - i - 1
        c[c_slice_start:c_slice_end] += delta * b_slice
    
    # a slice contribution: shift from i-(n-1) to i
    if i >= 0:
        # a indices from i down to 0
        a_slice = h[:i+1][::-1]  # reversed slice
        c_slice_start = i
        c_slice_end = i + len(a_slice)
        c[c_slice_start:c_slice_end] -= delta * a_slice
    
    return c
