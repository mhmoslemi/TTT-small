import time

# Global constants
n_points = 200
dx = 2.0 / n_points

def compute_corr_int(h):
    """Compute correlation of h and (1-h) using FFT, return integer values"""
    h_rev = h[::-1].astype(np.float64)
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    # round to nearest integer (small float errors)
    corr_int = np.rint(corr).astype(int)
    return corr_int

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_int(h)
    return dx * np.max(corr)

        zeros_idx = np.where(current_h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break

        i_zero = np.random.choice(zeros_idx)
        
        # Create candidate by swapping
        cand_h = current_h.copy()
        cand_h[i_one] = 0
        cand_h[i_zero] = 1

                best_max = cand_max

        iteration += 1
        if iteration % 1000 == 0 and iteration > 0:
            # Decrease temperature more aggressively if not improving recently
            T = max(0.0001, T * 0.999)
    
    return best_h, best_max

                i_one = np.random.choice(ones_idx)
                i_zero = np.random.choice(zeros_idx)

        for i_one, i_zero in swaps:

            no_improve += 1
    
    return best_h, best_max
