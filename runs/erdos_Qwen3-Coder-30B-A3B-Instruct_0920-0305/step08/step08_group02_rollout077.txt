import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.fft import fft, ifft

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def water_fill_projection(h, target_sum):
    """Project to sum constraint and box constraints using water filling."""
    # Binary search on lambda
    def sum_at_lambda(lam):
        return np.sum(np.clip(h + lam, 0, 1))
    
    lower, upper = -1.0, 1.0
    for _ in range(50):
        mid = (lower + upper) / 2
        val = sum_at_lambda(mid)
        if val < target_sum:
            lower = mid
        else:
            upper = mid
    lam = (lower + upper) / 2
    return np.clip(h + lam, 0, 1)

def compute_subgradient(h, k_star):
    """Compute subgradient for worst shift k_star"""
    n = len(h)
    g = np.zeros_like(h)
    for i in range(n):
        g[i] = 1 - h[(i + k_star) % n] - h[(i - k_star) % n]
    return g

def binary_milp_construction():
    """Use MILP to find the best binary construction."""
    n = n_points
    M = n // 2
    # Create variables, all binary, and continuous auxiliary
    y = cp.Variable(n, boolean=True)
    w = cp.Variable((n-1, n), nonneg=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(y) == M,
        t >= 0,
        t <= M,
    ]
    
    # Product linearization constraints: w[k,i] = y[i]*y[i+k]  
    for k in range(1, n):
        for i in range(n-k):
            constraints += [
                w[k-1, i] <= y[i],
                w[k-1, i] <= y[i+k],
                w[k-1, i] >= y[i] + y[i+k] - 1,
                w[k-1, i] >= 0
            ]
    
    # Minimum autocorrelation constraint
    for k in range(1, n):
        constraints += [cp.sum(w[k-1, :n-k]) >= t]
    
    # Objective: maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    start_time = time.time()
    try:
        prob.solve(solver=cp.CBC, max_seconds=0.4 * 600)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INFEASIBLE]:
            y_opt = y.value
            if y_opt is not None:
                return np.round(y_opt).astype(int)  # Return binary solution
    except:
        pass
    
    return None  # Fallback to greedy

def greedy_binary_construction():
    """Create binary starting solution using greedy spacing."""
    n = n_points
    M = n // 2
    # Greedy algorithm that places 1s as far apart as possible while maintaining ~M ones
    y = np.zeros(n, dtype=int)
    count = 0
    
    # Try to place 1s every 2 positions but adjust to exact count
    indices = list(range(0, n, 2))  # Default spacing
    for i in range(len(indices)):
        if count >= M:
            break
        if indices[i] < n:
            y[indices[i]] = 1
            count += 1
    
    # Adjust to exactly M
    if count < M:
        for i in range(n):
            if y[i] == 0 and count < M:
                y[i] = 1
                count += 1
    elif count > M:
        count = 0
        for i in range(n-1, -1, -1):
            if y[i] == 1 and count > M:
                y[i] = 0
                count -= 1
    
    return y

def determine_worst_shift(h):
    """Find shift k that maximizes overlap integral F_k(h)."""
    corr = compute_corr_fft(h)
    k_star = np.argmax(corr)
    return k_star

def sub_gradient_descent(h0, max_time):
    """Perform deterministic sub-gradient descent."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    k_star = determine_worst_shift(h)
    alpha = 0.5 * dx  # Initial step size
    t = 1  # Iteration counter
    
    while time.time() - start_time < max_time:
        # Compute gradient
        g = compute_subgradient(h, k_star)
        # Update step
        h_new = h - (alpha / np.sqrt(t)) * g
        # Project
        h_new = water_fill_projection(h_new, TARGET_SUM)
        
        current_c5 = evaluate_c5(h_new)
        if current_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = current_c5
        
        h = h_new
        k_star = determine_worst_shift(h)
        t += 1
    
    return best_h, best_c5

def swap_move_annealing(h0, total_time):
    """Use simulated annealing to escape local minima."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    temp = 5.0 * dx
    cool_rate = temp / (total_time * 0.2)
    
    while time.time() - start_time < total_time:
        # Pick two random indices
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        
        # Perturb
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        h_new = np.clip(h_new, 0, 1)
        h_new = water_fill_projection(h_new, TARGET_SUM)
        
        # Compute new C5
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            h = h_new.copy()
            best_h = h_new.copy()
            best_c5 = new_c5
        else:
            # Accept with probability
            if temp > 0:
                p = np.exp(-(new_c5 - best_c5) / temp)
                if np.random.rand() < p:
                    h = h_new.copy()
        
        temp -= cool_rate
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Phase 1: Binary optimization using MILP
    h_binary = binary_milp_construction()
    if h_binary is None:
        h_binary = greedy_binary_construction()
    
    h0 = h_binary.astype(float)
    c5_initial = evaluate_c5(h0)
    
    # Phase 2: Continuous refinement via sub-gradient
    sg_time = 0.35 * budget_s
    h_refined, c5_refined = sub_gradient_descent(h0, sg_time)
    
    # Phase 3: Stochastic refinement by simulated annealing
    remaining_time = budget_s - sg_time
    h_final, c5_final = swap_move_annealing(h_refined, remaining_time)
    
    return (h_final.tolist(), float(c5_final), n_points)
