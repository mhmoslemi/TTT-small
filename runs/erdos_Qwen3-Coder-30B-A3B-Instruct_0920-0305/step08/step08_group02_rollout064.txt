import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Minimize, Variable, Problem, sum as cp_sum, norm
import math

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(h, target_sum):
    """Project vector h onto box constraints [0,1] with sum = target_sum"""
    h = np.array(h)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Water-filling projection onto the sum constraint
    def sum_proj(lambda_val):
        return np.sum(np.clip(h + lambda_val, 0, 1)) - target_sum
    
    a, b = -1, 1
    for _ in range(50):
        mid = (a + b) / 2
        if sum_proj(mid) > 0:
            b = mid
        else:
            a = mid
        if abs(sum_proj(mid)) < 1e-8:
            break
    
    lambda_opt = (a + b) / 2
    h = np.clip(h + lambda_opt, 0, 1)
    return h

def solve_binary_milp(N, M, max_seconds=20):
    """Solve binary version using MILP to maximize minimum autocorrelation"""
    # Ensure sufficient size for problem
    if N < 4:
        return None, float("-inf")
    
    # Variables
    y = Variable(N, boolean=True)
    w = Variable((N, N), nonneg=True)
    t = Variable()
    
    # Constraints
    constraints = [
        cp_sum(y) == M,
        t >= 0
    ]

    # Autocorrelation constraints
    for k in range(1, N):
        # Only consider valid indices
        for i in range(N - k):
            # Linearisation of w[i,k] = y[i] * y[i+k]
            constraints.extend([
                w[i, k] <= y[i],
                w[i, k] <= y[i+k],
                w[i, k] >= y[i] + y[i+k] - 1,
                w[i, k] >= 0
            ])
    
    # Minimum autocorrelation constraint
    for k in range(1, N):
        constraints.append(cp_sum(w[:, k]) >= t)

    # Objective
    problem = Problem(Maximize(t), constraints)

    try:
        problem.solve(solver=cp.CBC, max_seconds=max_seconds)
        if problem.status in ["optimal", "optimal_inaccurate"]:
            y_val = np.round(y.value).astype(int)
            # Return value of t and the binary solution
            return y_val, t.value
        else:
            return None, float("-inf")
    except Exception:
        return None, float("-inf")

def get_binary_initial(N, M):
    """Get a good initial binary solution"""
    # Solve MILP for the binary version
    y, t_opt = solve_binary_milp(N, M, 20)
    if y is not None:
        return y.astype(float)
    else:
        # Fallback to a simple sum and symmetry
        y = np.zeros(N)
        y[:M] = 1
        np.random.shuffle(y)
        return y

def subgradient_descent_step(h, step_size):
    """Perform a single gradient descent step ignoring line search"""
    n = len(h)
    # Get the active shift
    corr = compute_corr_fft(h)
    worst_k = np.argmax(corr)
    # Compute subgradient for worst shift
    g = np.zeros(n)
    if worst_k == 0:
        g = np.ones_like(h)
    else:
        # Indices for shift k
        forward_indices = list(range(worst_k, n))
        backward_indices = list(range(0, n - worst_k))
        # Gradient is 1 - h[shifted backward] - h[shifted forward]
        g[: -worst_k]   -= h[worst_k:]
        g[worst_k:]     -= h[: -worst_k]
        g += np.ones_like(h)
    
    # Update h with gradient step
    h_new = h - step_size * g
    # Project back onto constraints
    h_new = project_to_sum_box(h_new, len(h) / 2)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    N = 120
    dx = 2.0 / N
    total_sum = N / 2

    # Phase 1: Solve MILP for binary initial value
    phase1_end = start_time + 0.4 * budget_s
    y_binary = get_binary_initial(N, total_sum)
    if y_binary is None or np.sum(y_binary) != total_sum:
        y_binary = np.random.choice([0, 1], size=N, p=[0.5, 0.5])
        y_binary = project_to_sum_box(y_binary, total_sum)  

    h_current = y_binary.copy()
    best_c5 = evaluate_c5(h_current)  

    # Phase 2: Deterministic subgradient descent
    phase2_end = start_time + 0.75 * budget_s
    step_size = 0.1 * dx  
    iteration = 0
    best_iter = 0
    best_h = h_current.copy()

    while time.time() < phase2_end:
        iteration += 1
        if iteration % 50 == 0:
            step_size *= 0.95  # Decaying step size
        h_new = subgradient_descent_step(h_current, step_size) 
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_iter = iteration
        h_current = h_new.copy()

    # Phase 3: Stochastic refinement
    phase3_end = start_time + budget_s
    h_refine = best_h.copy()
    best_final_c5 = best_c5

    while time.time() < phase3_end:
        idx1, idx2 = np.random.randint(0, N, size=2)
        if idx1 == idx2:
            continue
        old_val1, old_val2 = h_refine[idx1], h_refine[idx2]
        delta = np.random.uniform(-0.02, 0.02)
        h_temp = h_refine.copy()
        h_temp[idx1] = old_val1 + delta
        h_temp[idx2] = old_val2 - delta
        h_temp = project_to_sum_box(h_temp, total_sum)
        new_c5 = evaluate_c5(h_temp)
        if new_c5 < best_final_c5:
            h_refine = h_temp.copy()
            best_final_c5 = new_c5

    return (h_refine.tolist(), float(best_final_c5), N)
