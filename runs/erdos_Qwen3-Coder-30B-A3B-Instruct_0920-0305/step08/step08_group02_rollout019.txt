import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Strategy 1: Coarse grid search with different patterns  
    pattern_types = [
        # Basic step function with peaks
        lambda: np.array([0.0]*30 + [1.0]*60 + [0.0]*30),
        # Sine wave modulated
        lambda: np.clip(0.5 + 0.5 * np.sin(np.linspace(0, 4*np.pi, n_points)), 0, 1),
        # Two sharp peaks
        lambda: np.array([0.0]*40 + [1.0]*20 + [0.0]*20 + [1.0]*20 + [0.0]*20),
        # Smooth ramp up and down
        lambda: np.clip(np.linspace(0, 1, n_points//2) + np.linspace(1, 0, n_points//2), 0, 1),
    ]
    
    # Evaluate coarse patterns
    for i, pattern_fn in enumerate(pattern_types):
        h = pattern_fn()
        h = project_to_sum_box(h, TARGET_SUM)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Strategy 2: Local improvement using simulated annealing approach
    current_h = best_h.copy()
    current_c5 = best_c5
    temp = 1.0
    cooling_rate = 0.999
    min_temp = 1e-6
    
    sa_iter = 0
    while time.time() - start_time < 0.8 * budget_s:
        sa_iter += 1
        
        # Perturb current solution
        h_new = current_h.copy()
        # Make small random changes to several positions
        indices = np.random.choice(n_points, size=min(15, n_points//8), replace=False)
        for idx in indices:
            delta = np.random.uniform(-0.05, 0.05)
            h_new[idx] += delta
        
        # Project back to feasible space
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        new_c5 = evaluate_c5(h_new)
        
        # Accept or reject based on temperature
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / temp):
            current_h = h_new.copy()
            current_c5 = new_c5
            
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
        
        # Cool down temperature
        temp *= cooling_rate
        if temp < min_temp:
            break
    
    # Strategy 3: Fine tune with LBFGS optimization
    from scipy.optimize import minimize
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    # Try several random starting points for optimization
    for _ in range(15):
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Start with current best or similar
        h_start = best_h + np.random.normal(0, 0.01, n_points)
        h_start = project_to_sum_box(h_start, TARGET_SUM)
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                h_start, 
                method='L-BFGS-B', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-10}
            )
            if res.success:
                h_opt = res.x
                h_opt = project_to_sum_box(h_opt, TARGET_SUM)
                c5_opt = evaluate_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            continue
    
    # Final thorough local search
    if time.time() - start_time < budget_s - 1.0:
        # More aggressive local search
        for _ in range(500):
            if time.time() - start_time > budget_s - 0.5:
                break
            h_test = best_h.copy()
            # Modify a few random positions
            indices = np.random.choice(n_points, size=10, replace=False)
            for idx in indices:
                h_test[idx] = np.random.uniform(0, 1)
            h_test = project_to_sum_box(h_test, TARGET_SUM)
            c5_test = evaluate_c5(h_test)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
