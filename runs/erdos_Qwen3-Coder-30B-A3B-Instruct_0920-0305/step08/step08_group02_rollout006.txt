import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water filling"""
    def project_with_lambda(lam):
        return np.clip(x + lam, 0, 1)
    
    def sum_with_lambda(lam):
        return np.sum(project_with_lambda(lam))
    
    a, b = -1.0, 1.0
    while sum_with_lambda(a) > target_sum or sum_with_lambda(b) < target_sum:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = sum_with_lambda(m)
        if abs(fm - target_sum) < tol:
            break
        if fm > target_sum:
            b = m
        else:
            a = m
    
    return project_with_lambda((a + b) / 2)

def greedy_binary_initialization(n, target_sum):
    """Create a greedy binary initial solution with good autocorrelation"""
    y = np.zeros(n, dtype=int)
    positions = []
    
    # Place 1s as far apart as possible while maintaining sum
    for i in range(0, n, 2):
        if len(positions) < target_sum:
            y[i] = 1
            positions.append(i)
    
    # Fill gaps
    for i in range(len(positions) - 1):
        gap = positions[i + 1] - positions[i]
        if gap > 2:
            # Try placing a 1 in the middle gap or earlier
            for j in range(positions[i] + 1, positions[i + 1]):
                if len(positions) < target_sum and np.sum(y) < target_sum:
                    y[j] = 1
                    positions.append(j)
                if len(positions) >= target_sum:
                    break
    
    # Ensure exact sum
    while np.sum(y) < target_sum:
        for i in range(n):
            if y[i] == 0:
                y[i] = 1
                break
    
    return y.astype(float)

def solve_binary_milp(n, target_sum, max_time):
    """Solve the binary MILP for the best autocorrelation"""
    try:
        # Variables
        y = cp.Variable(n, boolean=True)
        w = cp.Variable((n, n), integer=True)
        t = cp.Variable()

        # Constraints
        constraints = [
            cp.sum(y) == target_sum,
            t >= 0,
            t <= target_sum
        ]

        # Linearization of w[k,i] = y[i]*y[i+k]
        for k in range(1, n):
            for i in range(n - k):
                constraints += [
                    w[i, k] <= y[i],
                    w[i, k] <= y[i + k],
                    w[i, k] >= y[i] + y[i + k] - 1,
                    w[i, k] >= 0
                ]

        # Objectives
        for k in range(1, n):
            constraints += [
                cp.sum(w[:, k]) >= t
            ]

        # Objective
        objective = cp.Maximize(t)

        # Solve
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, time_limit=max_time)
        result = y.value
        if result is not None:
            return result.astype(float)

    except Exception as e:
        pass

    return None

def subgradient_descent_phase(h0, max_time):
    """Perform deterministic sub-gradient descent"""
    h = h0.copy()
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    alpha0 = 0.5 * (2.0 / len(h))
    t = 1
    while time.time() - start_time < max_time:
        # Compute correlation and find worst shift
        corr = compute_corr_fft(h)
        k_star = np.argmax(corr)
        
        # Compute sub-gradient
        g = np.zeros_like(h)
        if k_star < len(h):
            # g[j] = 1 - h[j+k_star] - h[j-k_star]
            for j in range(len(h)):
                left_val = h[j - k_star] if j - k_star >= 0 else 0
                right_val = h[j + k_star] if j + k_star < len(h) else 0
                g[j] = 1 - left_val - right_val
        
        # Take step with diminishing learning rate
        alpha = alpha0 / np.sqrt(t)
        h_new = h - alpha * g
        
        # Project
        h_new = project_to_sum_box(h_new, len(h)/2)
        
        # Evaluate and update if improvement
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        h = h_new
        t += 1

    return best_h, best_c5

def swap_move_annealing(h0, max_time):
    """Perform stochastic simulations for fine tuning"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    T_start = 5 * (2.0 / len(h))
    
    while time.time() - start_time < max_time:
        # Annealing temperature
        elapsed = time.time() - start_time
        T = max(T_start * (1 - elapsed / max_time), 0)
        
        # Select two indices
        i, j = np.random.choice(len(h), 2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        
        # Make swap
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Clip and project
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, len(h)/2)
        
        # Evaluate
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            best_h = h.copy()
        elif T > 0 and np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            h = h_new.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    
    start_time = time.time()
    n_points = 120
    TARGET_SUM = n_points / 2
    dx = 2.0 / n_points
    
    # Phase 1: Binary MILP or Greedy initialization
    h0 = None
    phase1_time = 0.4 * budget_s
    try:
        result = solve_binary_milp(n_points, TARGET_SUM, phase1_time)
        if result is not None:
            h0 = result
        else:
            h0 = greedy_binary_initialization(n_points, TARGET_SUM)
    except Exception:
        # Fall back to greedy if MILP fails for any reason
        h0 = greedy_binary_initialization(n_points, TARGET_SUM)
    
    # Phase 2: Sub-gradient descent
    phase2_time = 0.35 * budget_s
    h1, c5_1 = subgradient_descent_phase(h0, phase2_time)
    
    # Phase 3: Swap-simulated annealing
    phase3_time = budget_s - phase1_time - phase2_time
    h_final, c5_final = swap_move_annealing(h1, phase3_time)
    
    return (h_final.tolist(), float(c5_final), n_points)
