import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import differential_evolution
import math

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def create_initial_solution():
    """Create a smart initial solution using a known good pattern"""
    # Start with a simple piecewise constant construction
    h = np.zeros(n_points)
    
    # Divide into 4 regions
    regions = 4
    segments = n_points // regions
    
    # Fill alternating regions
    for i in range(regions):
        start_idx = i * segments
        end_idx = start_idx + segments if i < regions - 1 else n_points
        if i % 2 == 0:
            h[start_idx:end_idx] = 1.0
        else:
            h[start_idx:end_idx] = 0.0
    
    # Adjust to match target sum
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def create_guided_initial():
    """Create a more sophisticated initial solution using a greedy strategy"""
    # Start with a smooth random function and then optimize
    h = np.random.uniform(0, 1, n_points)
    h = project_to_sum_box(h, TARGET_SUM)
    
    # Apply a simple heuristic: try to place concentration near center
    center = n_points // 2
    for i in range(n_points):
        dist_from_center = abs(i - center)
        # Reduce values near boundaries slightly
        if dist_from_center < n_points // 4:
            h[i] *= 1.0 - 0.3 * (dist_from_center / (n_points // 4))
    
    h = project_to_sum_box(h, TARGET_SUM)
    
    # Use a simple greedy approach to concentrate mass in sections
    weights = np.abs(np.linspace(-1, 1, n_points))
    weights = (weights - np.min(weights)) / (np.max(weights) - np.min(weights)) 
    weights = 1.0 - weights  # More at center
    
    for i in range(n_points):
        h[i] = min(1.0, h[i] + weights[i] * 0.2)
    
    h = project_to_sum_box(h, TARGET_SUM)
    
    return h

def fitness_function(h):
    """Fitness function to minimize (negative of C5 since we minimize)"""
    return -evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points / 2
    
    # Use DeNovo genetic algorithm approach
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_func(h):
        return np.sum(h) - TARGET_SUM  # Should be zero
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Use differential evolution with custom bounds
    result = differential_evolution(
        fitness_function,
        bounds,
        maxiter=500,
        popsize=20,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        disp=False,
        callback=lambda x, convergence: None,
        strategy='best1bin'
    )
    
    best_h = project_to_sum_box(result.x, TARGET_SUM)
    final_c5 = evaluate_c5(best_h)
    
    # Do additional local refinement using gradient-free method
    # Try simple local perturbations
    best_c5_refined = final_c5
    best_h_refined = best_h.copy()
    
    for _ in range(200):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Make small random change
        h_new = best_h_refined.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.02, 0.02)
        h_new[idx] += delta
        
        # Project to constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5_refined:
            best_c5_refined = new_c5
            best_h_refined = h_new.copy()
    
    # Final polishing with a deeper local search step  
    best_h_final = best_h_refined.copy()
    best_c5_final = best_c5_refined
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
            
        # Slightly more aggressive optimization
        h_new = best_h_final + np.random.normal(0, 0.005, n_points)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5_final:
            best_c5_final = new_c5
            best_h_final = h_new.copy()
    
    return (best_h_final.tolist(), float(best_c5_final), n_points)
