# Global constants
n_points = 300
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(ifft(fft(h_rev, L) * fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling projection
    sorted_indices = np.argsort(x)[::-1]
    sorted_x = x[sorted_indices]
    
    # Binary search for threshold
    low, high = 0.0, 1.0
    while high - low > 1e-10:
        mid = (low + high) / 2
        # Compute sum of clipped values
        clipped = np.clip(sorted_x - mid, 0, 1)
        if np.sum(clipped) > target_sum:
            low = mid
        else:
            high = mid
    
    threshold = (low + high) / 2
    projected = np.clip(x - threshold, 0, 1)
    
    # Adjust for potential sum difference
    current_sum = np.sum(projected)
    if abs(current_sum - target_sum) > 1e-10:
        # Add difference equally to lowest entries
        diff = target_sum - current_sum
        if diff > 0:
            idxs = np.argsort(projected)
            for i in range(int(diff/dx)):
                if i >= len(idxs):
                    break
                idx = idxs[i]
                if projected[idx] < 1.0:
                    projected[idx] = min(1.0, projected[idx] + diff/(i+1))
        elif diff < 0:
            idxs = np.argsort(projected)[::-1]
            for i in range(int(-diff/dx)):
                if i >= len(idxs):
                    break
                idx = idxs[i]
                if projected[idx] > 0.0:
                    projected[idx] = max(0.0, projected[idx] + diff/(i+1))
    
    return projected

def rasterise_and_project(θ):
    """Convert parameter vector to h and project to proper constraints"""
    K = 8
    # Widths (exponential)
    u = np.exp(np.clip(θ[:K], -15, 15))
    widths = 2 * u / np.sum(u)
    
    # Heights (sigmoid)
    heights = 1 / (1 + np.exp(-θ[K:]))
    
    # Build h
    h = np.zeros(n_points)
    cum_width = 0
    for i in range(K):
        start = int(cum_width / dx)
        end = int((cum_width + widths[i]) / dx)
        if end > start:
            h[start:end] = heights[i]
        cum_width += widths[i]
    
    # Project to sum and box
    h = project_to_sum_box(h, TARGET_SUM)
    return h
