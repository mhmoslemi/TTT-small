import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import time

# Global variables
n_points = 300
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT with zero padding
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water-filling"""
    # Sort the values in descending order to get the water filling algorithm
    sorted_indices = np.argsort(x)[::-1]
    sorted_x = x[sorted_indices]
    
    # Water filling algorithm to find the threshold lambda
    cumsum = np.cumsum(sorted_x)
    n = len(sorted_x)
    threshold_idx = -1
    
    # Find where the sum of values above threshold is less than target_sum
    for i in range(n):
        if cumsum[i] <= target_sum:
            threshold_idx = i
        else:
            break
    
    # If total sum is less than target, adjust all values up to target
    if threshold_idx == -1:
        # All values are too small, increase uniformly
        multiplier = target_sum / np.sum(x)
        result = x * multiplier
    else:
        # Some values are above threshold and some are below
        # Value at threshold idx and above should be set to threshold value
        if threshold_idx == n - 1:
            # All values in sorted_x are too small, but increase the last ones
            threshold = (target_sum - cumsum[threshold_idx-1]) / (threshold_idx - 1)
            result = np.array([min(sorted_x[i], threshold) for i in range(n)])
        else:
            # Calculate the threshold
            threshold = (target_sum - cumsum[threshold_idx]) / (n - threshold_idx - 1)
            # Set values above threshold to threshold
            result = sorted_x.copy()
            for i in range(threshold_idx+1, n):
                result[i] = threshold
            # Project the previous values to also satisfy constraints
            sum_less = np.sum(result[:threshold_idx+1])
            if sum_less > target_sum:
                # Need to downscale the values
                multiplier = target_sum / sum_less
                for i in range(threshold_idx+1):
                    result[i] *= multiplier
            elif sum_less < target_sum:
                # Need to increase values
                diff = target_sum - sum_less
                diff_per = diff / (threshold_idx+1)
                for i in range(threshold_idx+1):
                    result[i] += diff_per
        
        # Reorder to original indices
        final_result = np.zeros_like(result)
        final_result[sorted_indices] = result
        result = final_result
    
    return np.clip(result, 0, 1)

def rasterise_and_project(θ):
    """Convert parameters to a step function h with proper constraints"""
    K = 8  # Number of blocks
    widths_raw = np.exp(θ[:K])
    widths = 2 * widths_raw / np.sum(widths_raw)
    heights = 1 / (1 + np.exp(-θ[K:]))
    
    h = np.zeros(n_points)
    x_grid = np.linspace(0, 2, n_points, endpoint=False) + dx/2
    
    cum_widths = np.insert(np.cumsum(widths), 0, 0)
    
    for i in range(n_points):
        block_idx = np.searchsorted(cum_widths, x_grid[i], side='right') - 1
        h[i] = heights[block_idx]
    
    # Project to sum constraint
    h_projected = project_to_sum_box(h, TARGET_SUM)
    return h_projected

def obj(θ):
    """Objective function for DE"""
    h = rasterise_and_project(θ)
    c5 = evaluate_c5(h)
    return c5

def compute_gradient(h, active_shifts):
    """Compute sub-gradient of max(k) L(k)"""
    n = len(h)
    g = np.zeros(n)
    for s in active_shifts:
        if s >= 0:
            g[:n-s] += -dx * h[s:]
            g[s:] += -dx * h[:n-s]
        else:
            t = -s
            g[t:] += -dx * h[:n-t]
            g[:n-t] += -dx * h[t:]
    return g / len(active_shifts)

        h_try = h_cur - eta * d

            eta = min(eta_max, eta * eta_increase)
        else:
            eta = max(eta_min, eta * eta_decrease)
            no_improve_count += 1
            
        # Stochastic mass transfer
        if iteration % swap_freq == 0 or no_improve_count > no_improve_limit:
            swap_count += 1
            # Find indices where we can move mass
            idx_pos = np.where(h_cur > 0)[0]
            idx_neg = np.where(h_cur < 1)[0]
            
            if len(idx_pos) > 0 and len(idx_neg) > 0:
                i_pos = np.random.choice(idx_pos)
                i_neg = np.random.choice(idx_neg)
                
                delta = min(0.02, h_cur[i_neg]) * np.random.random()
                h_temp = h_cur.copy()
                h_temp[i_pos] += delta
                h_temp[i_neg] -= delta
                
                # Project onto constraints
                h_temp = project_to_sum_box(h_temp, TARGET_SUM)
                new_c5 = evaluate_c5(h_temp)
                
                if new_c5 < c5_cur + 1e-4:
                    h_cur = h_temp
                    c5_cur = new_c5
                    if new_c5 < best_c5:
                        best_h = h_cur.copy()
                        best_c5 = new_c5
                    no_improve_count = 0
                    eta = min(eta_max, eta * eta_increase)
    
    return (best_h.tolist(), float(best_c5), n_points)
