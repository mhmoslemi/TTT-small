import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import time

# Global variables
TARGET_SUM = None
n_points = 300
dx = 2.0 / n_points
n_coarse = 150
dx_coarse = 2.0 / n_coarse

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water-filling"""
    # Sort in descending order to compute the water filling threshold
    sorted_indices = np.argsort(x)[::-1]
    sorted_x = x[sorted_indices]
    
    # Compute cumulative sum of sorted x and check where to insert threshold
    cumsum = np.cumsum(sorted_x)
    n = len(sorted_x)
    
    # Find threshold lambda such that sum of clipped values equals target
    # This is based on the solution to the water-filling problem
    # Compute an initial guess
    if np.sum(sorted_x) <= target_sum:
        # All values can be kept as is
        return np.clip(x, 0, 1)
    
    # Find appropriate threshold
    threshold = 0.0
    sum_so_far = 0.0
    for i in range(n):
        # If we set all values from index i onwards to 1.0, we check if that's too much
        # or too little
        sum_with_ones = sum_so_far + (n - i) * 1.0
        if sum_with_ones > target_sum:
            # The threshold is between this value and the previous one
            threshold = (target_sum - sum_so_far) / (n - i)
            break
        sum_so_far += sorted_x[i]
    else:
        threshold = 0.0  # it should not be reached with valid parameters

    # Apply the threshold
    result = np.clip(x, 0, 1)
    if threshold > 0:
        # Re-adjust using the water filling rule
        y = np.sort(x)[::-1]  # sorted in descending order
        cumsum = np.cumsum(y)
        i = next((i for i, s in enumerate(cumsum) if s > target_sum), len(y))
        if i > 0:
            cut = (target_sum - cumsum[i-1]) / (len(y) - i)
        else:
            cut = target_sum / len(y)
        result = np.clip(y - cut, 0, 1)[::-1]  # reverse back to original order
        if i < len(y):
            result[sorted_indices[:i]] = 1.0

    return result

def rasterise_and_project(θ):
    """Convert parametric representation to h with proper sum and box constraint"""
    K = 8
    theta_widths = θ[:K]
    theta_heights = θ[K:] # Shape: [K]
    
    u = np.exp(np.clip(theta_widths, -15, 15))
    w = 2.0 * u / np.sum(u)
    
    a = 1.0 / (1.0 + np.exp(-theta_heights))
    a = np.clip(a, 1e-6, 1.0 - 1e-6)  # avoid exact 0 & 1 for numerical stability
    
    # create piecewise constant h
    h = np.zeros(n_points)
    cum_width = 0
    for i in range(K):
        start_idx = int(cum_width / dx)
        end_idx = int((cum_width + w[i]) / dx)
        if end_idx > n_points:
            end_idx = n_points
        h[start_idx:end_idx] = a[i]
        cum_width += w[i]

    # Project to box and sum    
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def objective(θ):
    h = rasterise_and_project(θ)
    return evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed) 
    global TARGET_SUM, n_points, dx
    start_time = time.time()
    
    # Set target and allocation
    TARGET_SUM = n_points / 2
    n_coarse = 150
    dx_coarse = 2.0 / n_coarse

    # Initial guess from existing construction, interpolate if needed

    if len(h0) != n_points:

        x_new = np.linspace(0, 2, n_points)
        h0 = np.interp(x_new, x_orig, h0)
    h0 = project_to_sum_box(h0, TARGET_SUM)
    initial_c5 = evaluate_c5(h0)

    # Step 1: Coarse DE search with K=8 blocks
    def coarse_rasterise_and_project(θ):
        K = 8
        theta_widths = θ[:K]
        theta_heights = θ[K:] # Shape: [K]
        
        u = np.exp(np.clip(theta_widths, -15, 15))
        w = 2.0 * u / np.sum(u)
        
        a = 1.0 / (1.0 + np.exp(-theta_heights))
        a = np.clip(a, 1e-6, 1.0 - 1e-6)
        
        h_coarse = np.zeros(n_coarse)
        cum_width = 0
        for i in range(K):
            start_idx = int(cum_width / dx_coarse)
            end_idx = int((cum_width + w[i]) / dx_coarse)
            if end_idx > n_coarse:
                end_idx = n_coarse
            h_coarse[start_idx:end_idx] = a[i]
            cum_width += w[i]

        # Project to box and sum
        h_coarse = project_to_sum_box(h_coarse, n_coarse / 2)
        # Interpolate to high resolution
        h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), h_coarse)
        h_fine = project_to_sum_box(h_fine, TARGET_SUM)
        
        return h_fine

    def coarse_objective(θ):
        h = coarse_rasterise_and_project(θ)
        return evaluate_c5(h)

    bounds = [(-5, 5)] * 16 # 8 widths, 8 heights
    de_result = differential_evolution(
        coarse_objective,
        bounds,
        popsize=12,
        maxiter=120,
        seed=seed,
        disp=False,
        updating='deferred'  # To avoid some types of convergence issues
    )

    best_h_coarse = coarse_rasterise_and_project(de_result.x)
    best_c5_coarse = de_result.fun
    
    # Step 2: Local refinement on coarse grid
    # Use projected sub-gradient descent
    
    def compute_subgradient(h, corr, active_shifts, n):
        # Compute the combined gradient from all active shifts
        g = np.zeros(n)
        for shift in active_shifts:
            if shift >= 0:
                g[:n-shift] -= dx * h[shift:]
                g[shift:] -= dx * h[:n-shift]
            else:
                t = -shift
                g[t:] -= dx * h[:n-t]
                g[:n-t] -= dx * h[t:]
        g /= len(active_shifts)  # averge over active shifts
        return -g  # We want to descend

    def projected_descent(h, max_iter=500):
        h_current = h.copy()
        c5_current = evaluate_c5(h_current)
        
        best_h = h_current.copy()
        best_c5 = c5_current
        
        eta = 0.5  # initial learning rate
        eta_min = 1e-4
        eta_max = 2.0
        
        no_improvement_count = 0
        last_improved = 0
        
        for i in range(max_iter):
            # Check time limit periodically
            if time.time() - start_time > budget_s - 2:
                return best_h, best_c5
            
            # compute correlation
            h_rev = h_current[::-1]
            b = 1 - h_current
            L = 2 * len(h_current) - 1
            corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
            corr *= dx
            
            max_val = corr.max()
            active = np.where(np.abs(corr - max_val) < 1e-12)[0]
            shifts = active - (len(h_current) - 1)
            
            g = compute_subgradient(h_current, corr, shifts, len(h_current))
            
            # Step
            h_new_raw = h_current - eta * g
            h_new = project_to_sum_box(h_new_raw, TARGET_SUM)
            
            # evaluate
            c5_new = evaluate_c5(h_new)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                no_improvement_count = 0
                last_improved = i
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            else:
                no_improvement_count += 1
                # adaptive learning rate
                eta *= 0.7
                eta = max(eta, eta_min)
            
        return best_h, best_c5
    
    # Start with coarse solution, and refine
    h_init = best_h_coarse
    h_refined, c5_refined = projected_descent(h_init, max_iter=500)
    
    # Step 3: Upsample to fine resolution
    h_fine_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), 
                               h_refined[:n_coarse])
    # Project to correct sum and box
    h_fine_initial = project_to_sum_box(h_fine_initial, TARGET_SUM)
    
    # Step 4: Full resolution refinement with PSGD and swaps
    
    def full_refine(h_init):
        h_current = h_init.copy()
        c5_current = evaluate_c5(h_current)
        
        best_h = h_current.copy()
        best_c5 = c5_current
        
        # Prepare training
        eta = 0.5
        eta_min = 1e-4
        eta_max = 2.0
        no_improvement_count = 0
        swap_freq = 30
        swap_stationary = 200
        last_improved = 0
        
        iters = 0
        while time.time() - start_time < budget_s - 2:
            # compute full correlation to get active shifts
            h_rev = h_current[::-1]
            b = 1 - h_current
            L = 2 * len(h_current) - 1
            corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
            corr *= dx
            
            max_val = corr.max()
            active = np.where(np.abs(corr - max_val) < 1e-12)[0]
            shifts = active - (len(h_current) - 1)
            
            # Compute subgradient
            g = compute_subgradient(h_current, corr, shifts, len(h_current))
            
            # Step
            h_new_raw = h_current - eta * g
            h_new = project_to_sum_box(h_new_raw, TARGET_SUM)
            
            c5_new = evaluate_c5(h_new)
            
            # Accept or reject the step
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                no_improvement_count = 0
                last_improved = iters
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            else:
                no_improvement_count += 1
                # adaptively reduce step size
                eta *= 0.7
                eta = max(eta, eta_min)
                
            # Periodic stochastic mass transfer to escape plateaus
            if (iters % swap_freq == 0) or (no_improvement_count > swap_stationary):
                # pick two indices
                indices = np.where((h_current > 0) & (h_current < 1))[0]
                if len(indices) >= 2:
                    i_plus = np.random.choice(indices)
                    i_minus = np.random.choice(indices)
                    while i_plus == i_minus:
                        i_minus = np.random.choice(indices)
                    
                    # swap a random amount
                    delta = min(0.02, h_current[i_minus]) * np.random.rand()
                    h_new = h_current.copy()
                    h_new[i_plus] += delta
                    h_new[i_minus] -= delta
                    h_new = project_to_sum_box(h_new, TARGET_SUM)
                    c5_new_swap = evaluate_c5(h_new)
                    if c5_new_swap < best_c5 + 1e-4:
                        h_current = h_new
                        c5_current = c5_new_swap
                        if c5_current < best_c5:
                            best_h = h_current.copy()
                            best_c5 = c5_current
                            
            
            iters += 1
        return best_h, best_c5
    
    final_h, final_c5 = full_refine(h_fine_initial)
    
    return (final_h.tolist(), float(final_c5), n_points)
