```python
import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

# Global parameters
N = 120
dx = 2.0 / N
TARGET_SUM = N / 2
MAX_SECONDS = 600

def evaluate_c5(h):
    """FFT-based evaluation of C5"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project onto both box constraints and sum constraint"""
    x = np.clip(x, 0, 1)
    diff = target_sum - np.sum(x)
    if abs(diff) < 1e-10:
        return x
    # Water-filling projection onto sum constraint
    # Try to distribute the difference
    sorted_indices = np.argsort(x)
    x_proj = x.copy()
    i = 0
    while abs(diff) > 1e-10 and i < len(x_proj):
        if diff > 0:
            # Try to raise some value to 1
            idx = sorted_indices[i]
            if x_proj[idx] < 1:
                max_inc = min(diff, 1 - x_proj[idx])
                x_proj[idx] += max_inc
                diff -= max_inc
            else:
                i += 1
        else:
            # Try to lower some value to 0
            idx = sorted_indices[::-1][i]
            if x_proj[idx] > 0:
                max_dec = min(-diff, x_proj[idx])
                x_proj[idx] -= max_dec
                diff += max_dec
            else:
                i += 1
    return x_proj

def build_binary_milp_solution():
    """Build a high-quality 0/1 solution using MILP."""
    start_time = time.time()
    try:
        # Binary variables
        y = cp.Variable(N, boolean=True)
        # Auxiliary variables for product constraints
        w = {}
        t = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(y) == TARGET_SUM,
            t >= 0,
        ]
        
        # Product constraints for all shifts and valid indices
        for k in range(1, N):
            for i in range(N - k):
                w_ki = cp.Variable()
                w[(k, i)] = w_ki
                constraints += [
                    w_ki <= y[i],
                    w_ki <= y[i + k],
                    w_ki >= y[i] + y[i + k] - 1,
                    w_ki >= 0
                ]
                # Constraint: sum of w_ki >= t
                constraints += [cp.sum([w[(k, i)] for i in range(N - k)]) >= t]
        
        # Solve
        objective = cp.Maximize(t)
        prob = cp.Problem(objective, constraints)
        
        timeout = MAX_SECONDS * 0.4 - (time.time() - start_time)
        prob.solve(solver=cp.CBC, maximum_seconds=timeout)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            y_sol = np.round(y.value).astype(float)
            # Ensure correct sum
            y_sol = project_to_sum_box(y_sol, TARGET_SUM)
            return y_sol
    except Exception:
        pass
    return None

def compute_worst_shift_grad(h):
    """Compute subgradient for the worst-shift F_k at current h"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    k_star = np.argmax(corr)
    # Create gradient g_j = 1 - h_{j+k_star} - h_{j-k_star}
    g = np.zeros_like(h)
    if k_star < len(h):
        # left shift
        for i in range(k_star, len(h)):
            g[i] += 1 - h[i - k_star]
        # right shift
        for i in range(len(h) - k_star):
            g[i] += 1 - h[i + k_star]
    else:
        # Handle circular correlation fallback -- shouldn't happen in usual domain
        for i in range(len(h)):
            for j in range(len(h)):
                k = (j - i) % len(h)
                if k in [k_star, len(h)-k_star]:
                    g[i] += 1 - h[j]
    return g

def subgradient_descent(initial_h, max_time):
    """Perform subgradient descent on the worst shift."""
    start_time = time.time()
    h = initial_h.copy()  # Start at integer solution or random
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    alpha0 = 0.5 * dx
    iter_count = 0
    while time.time() < start_time + max_time:
        iter_count += 1
        grad = compute_worst_shift_grad(h)
        alpha = alpha0 / np.sqrt(iter_count)
        h -= alpha * grad
        h = project_to_sum_box(h, TARGET_SUM)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def stochastic_refinement(initial_h, max_time):
    """Use swaps in positions to refine solution."""
    start_time = time.time()
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    T = 1.0  # Initial temperature
    alpha = 0.995  # Cooling rate
    iter_count = 0
    
    while time.time() < start_time + max_time:
        iter_count += 1
        # Pick two different indices
        i = np.random.randint(0, len(h))
        j = np.random.randint(0, len(h))
        if i == j:
            continue
        delta = np.random.uniform(-0.02, 0.02)
        # Create new solution
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # Ensure values remain in [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Project sum constraint
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        # Evaluate
        c5_new = evaluate_c5(h_new)
        delta_c5 = c5_new - best_c5
        
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new.copy()
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        T *= alpha  
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: Binary MILP solution
    h = build_binary_milp_solution()
    if h is None:
        # Fallback: random feasible
        h = np.random.uniform(0, 1, N)
        h = project_to_sum_box(h, TARGET_SUM)

    # Phase 2: Deterministic subgradient descent
    subgrad_time = budget_s * 0.35
    h, c5 = subgradient_descent(h, subgrad_time)
    
    # Phase 3: Stochastic refinement with simulated annealing
    refine_time = budget_s - (time.time() - start_time) - 0.1
    if refine_time > 0:
        h, c5 = stochastic_refinement(h, refine_time)
    
    return (h.tolist(), float(c5), N)
```
