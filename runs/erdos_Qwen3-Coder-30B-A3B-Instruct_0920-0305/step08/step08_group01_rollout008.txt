```python
import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import differential_evolution
import random

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def adaptive_initialization():
    """Initialize with a combination of structured and random patterns"""
    # Start with a smooth baseline
    x = np.linspace(0, 2, n_points)
    baseline = 0.5 + 0.3 * np.sin(2 * np.pi * x)
    
    # Add some peaks to suppress overlap
    peaks = []
    for i in range(3):
        center = random.uniform(0.2, 1.8)
        width = random.uniform(0.1, 0.3)
        height = random.uniform(0.6, 1.0)
        peaks.append(height * np.exp(-0.5 * ((x - center) / width)**2))
    
    base = np.clip(baseline + sum(peaks), 0, 1)
    
    # Smear with random noise
    noise = np.random.normal(0, 0.05, n_points)
    h = np.clip(base + noise, 0, 1)
    
    # Project to correct sum
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    
    # Use evolutionary algorithm approach
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    # Start with adaptive initialization
    h0 = adaptive_initialization()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Use differential evolution with custom strategy
    de_bounds = [(0, 1) for _ in range(n_points)]
    
    # Run differential evolution for a portion of the budget
    de_time = 0.7 * budget_s
    de_start = time.time()
    
    # Custom callback to track the best solution
    def callback(x, convergence):
        nonlocal best_h, best_c5
        h = project_to_sum_box(x, TARGET_SUM)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    try:
        result = differential_evolution(
            objective, 
            de_bounds, 
            maxiter=int(1000), 
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            callback=callback,
            disp=False,
            tol=1e-12
        )
        
        if result.success:
            final_h = project_to_sum_box(result.x, TARGET_SUM)
            final_c5 = evaluate_c5(final_h)
            if final_c5 < best_c5:
                best_h = final_h.copy()
                best_c5 = final_c5
    except:
        pass
    
    # Local refinement of the best solution
    best_h = project_to_sum_box(best_h, TARGET_SUM)
    best_c5 = evaluate_c5(best_h)
    
    # Fine-tune with local search
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
            
        # Small random perturbations
        h_new = best_h + np.random.normal(0, 0.005, n_points)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
    
    # Final refinement 
    while time.time() - start_time < budget_s:
        # Try more aggressive search in promising regions
        idxs = np.random.choice(n_points, size=min(10, n_points//4), replace=False)
        improved = False
        
        for idx in idxs:
            h_new = best_h.copy()
            
            # Try different perturbations
            for delta in [-0.03, -0.02, -0.01, 0.01, 0.02, 0.03]:
                h_new[idx] += delta
                if h_new[idx] < 0 or h_new[idx] > 1:
                    continue
                    
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                c5_new = evaluate_c5(h_new)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    improved = True
                    break
            if improved:
                break
        else:
            # Random small perturbations if nothing improved
            if random.random() < 0.7:
                idx = random.randint(0, n_points - 1)
                delta = random.uniform(-0.01, 0.01)
                best_h[idx] += delta
                best_h = project_to_sum_box(best_h, TARGET_SUM)
                best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
```
