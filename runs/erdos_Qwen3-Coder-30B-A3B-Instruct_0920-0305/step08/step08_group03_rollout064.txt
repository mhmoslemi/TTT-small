def get_critical_shifts():
    # For 5 intervals, we'll compute all pairs of shifts
    # This is a conservative but precise estimate
    return np.linspace(0, 2, 101)  # 101 points in [0,2] should be enough

def evaluate_c5_fast(h):
    """Fast evaluation using precomputed critical shifts"""
    # Specialized for 5 interval step function
    h = np.array(h)
    n = len(h)
    L = 2.0
    dx = L / n
    
    # Compute all the relevant shifts
    K = get_critical_shifts()
    
    # For each k in K, compute the integral
    max_value = -1e30
    for k in K:
        integral = 0.0
        for i in range(n):
            x = i * dx
            x_plus_k = x + k
            if x_plus_k <= L:
                h_x = h[i]
                # Find h(x + k)
                idx_k = int(x_plus_k / dx)
                if idx_k >= n:
                    h_x_plus_k = 0
                else:
                    h_x_plus_k = h[idx_k]
                integral += h_x * (1 - h_x_plus_k)
        max_value = max(max_value, integral)
    
    return max_value

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Simple projection using clipping and rescaling
    x = np.array(x)
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x
    # Rescale to match target sum
    return x * target_sum / current_sum

        pos += lengths[i]

        linear_part = 0.0
        product_part = 0.0
        
        # Compute total mass up to point (2 - k) 
        cumulative = 0.0
        mass_end = L_total - k
        for s_start, s_end, a in intervals:
            if s_start >= mass_end:
                break
            s_start_cap = max(s_start, 0)
            s_end_cap = min(s_end, mass_end)
            if s_end_cap > s_start_cap:
                linear_part += a * (s_end_cap - s_start_cap)
        
        # Compute product term
        for i, (s_start1, s_end1, a1) in enumerate(intervals):
            for j, (s_start2, s_end2, a2) in enumerate(intervals):
                if s_end1 <= k + s_start2 or s_end2 <= k + s_start1:
                    continue  # No overlap 
                s_start_over = max(s_start1, k + s_start2)
                s_end_over = min(s_end1, k + s_end2)
                if s_end_over > s_start_over:
                    product_part += a1 * a2 * (s_end_over - s_start_over)
        
        return linear_part - product_part
    
    # Compute max over critical shifts
    max_val = -1e30
    for k in K_crit:
        val = J(k)
        max_val = max(max_val, val)
        
    return max_val
