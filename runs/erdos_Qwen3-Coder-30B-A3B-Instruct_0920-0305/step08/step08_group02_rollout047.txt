import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import Minimize, Variable, sum as cp_sum, max as cp_max, Problem

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_binary_milp(N, M):
    """Solve binary MILP to get the best 0/1 solution"""
    try:
        y = Variable(N, boolean=True)
        w = {}
        t = Variable()
        
        # Constraints
        constraints = [
            cp_sum(y) == M,
            t >= 0,
            t <= M,
        ]
        
        # Product linearisation and minimum autocorrelation constraints
        for k in range(1, N):
            for i in range(N - k):
                if (k, i) not in w:
                    w[(k, i)] = Variable()
            
            # Linear constraint for all (k,i) pairs
            for i in range(N - k):
                constraints.append(w[(k, i)] <= y[i])
                constraints.append(w[(k, i)] <= y[i + k])
                constraints.append(w[(k, i)] >= y[i] + y[i + k] - 1)
                constraints.append(w[(k, i)] >= 0)
                constraints.append(cp_sum([w[(k, i)] for i in range(N - k)]) >= t)
        
        objective = Maximize(t)
        prob = Problem(objective, constraints)
        
        # Try different solvers
        solvers = ["CBC", "GLPK_MI", "ECOS_BB"]
        for solver in solvers:
            try:
                prob.solve(solver=solver, max_seconds=0.4)
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    y_val = np.array(y.value).round().astype(int)
                    return y_val
            except:
                continue
                
    except Exception as e:
        pass
    
    return None

def greedy_binary_initialization(N, M):
    """Create a good binary starting solution using a greedy heuristic"""
    y = np.zeros(N, dtype=int)
    
    # Place a 1 in positions around intervals of 2*N/M = 2*N/120 = 1/30 of the domain
    # Strategy: place 1's sparsely and greedily
    placements = []
    pos = 0
    while len(placements) < M:
        # Place at position closest to i * 2 * (N/M) (they are evenly spaced)
        if pos >= N:
            # Add remaining if needed
            if len(placements) < M:
                placements += [np.random.randint(0, N)] * (M - len(placements))
            break
        placements.append(pos)
        pos += 2  # Slightly bigger step
    
    # Add to vector
    for idx in placements[:M]:
        y[idx] = 1
    
    return y

def compute_subgradient(h, k_star):
    """Calculate subgradient for a given k_star"""
    N = len(h)
    g = np.zeros(N)
    for i in range(N):
        g[i] = 1.0 - h[(i + k_star) % N] - h[(i - k_star) % N]
    return g

def project_to_sum_box_fast(h, target_sum):
    """Fast projection function"""
    # Clip first, then adjust using binary search for sum constraint
    h = np.clip(h, 0, 1)
    
    def f(lam):
        projected = np.clip(h + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search for lambda
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b) / 2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Set parameters
    TARGET_SUM = n_points / 2
    
    # Phase 1: Binary MILP or greedy solution
    milp_time = 0.4 * budget_s
    h0 = None
    
    # Try to solve MILP for binary case
    if h0 is None:
        try:
            y_binary = solve_binary_milp(n_points, TARGET_SUM)
            if y_binary is not None:
                h0 = y_binary.astype(float)
                milp_time_taken = time.time() - start_time
            else:
                h0 = greedy_binary_initialization(n_points, TARGET_SUM)
        except Exception as e:
            # fallback to greedy
            h0 = greedy_binary_initialization(n_points, TARGET_SUM)
    
    if h0 is None:
        h0 = np.random.uniform(0, 1, n_points)
        h0 = project_to_sum_box(h0, TARGET_SUM)
    
    # Ensure it's feasible
    h = project_to_sum_box(h0, TARGET_SUM)
    
    # Phase 2: Deterministic sub-gradient descent
    sg_time = 0.35 * budget_s
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    # Compute active shift
    corr = compute_corr_fft(h)
    k_star = np.argmax(corr)
    
    # Deterministic gradient descent
    alpha = 0.5 * dx  # Initial step size
    t = 0
    max_iterations = 100000
    iteration = 0
    while (time.time() - start_time) < start_time + sg_time and iteration < max_iterations:
        iteration += 1
        # Compute gradient
        g = compute_subgradient(h, k_star)
        
        # Take step
        h_new = h - (alpha / np.sqrt(iteration + 1)) * g
        h_new = project_to_sum_box_fast(h_new, TARGET_SUM)
        
        # Update best if necessary
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new.copy()
        corr = compute_corr_fft(h)
        k_star = np.argmax(corr)
    
    # Phase 3: Stochastic annealing
    sa_time = budget_s - sg_time - milp_time
    t_start = 5 * dx
    T = t_start
    t = 0
    
    while time.time() < start_time + budget_s:
        # Make 2-index swap move
        i, j = np.random.choice(n_points, size=2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Clip and project
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box_fast(h_new, TARGET_SUM)
        
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            c5_new = best_c5
        else:
            # Metropolis criterion
            if np.random.random() < np.exp(-(c5_new - best_c5) / T):
                h = h_new.copy()
                
        # Update temperature
        T = max(0.0, t_start * (1 - (time.time() - (start_time + sg_time + milp_time)) / sa_time))
        
        # Update best if necessary
        c5_new = evaluate_c5(h)
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
    
    return (best_h.tolist(), float(best_c5), n_points)
