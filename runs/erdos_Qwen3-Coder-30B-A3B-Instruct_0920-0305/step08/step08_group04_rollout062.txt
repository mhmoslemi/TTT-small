import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    np.random.seed(seed)
    start_time = time.time()
    
    # Multi-start optimization with different heuristics
    best_h = None
    best_c5 = float('inf')
    
    # Strategy 1: Original starting point from initial_h_values

    x_orig = np.linspace(0, 2, len(h0))
    x_new = np.linspace(0, 2, n_points)
    h0 = np.interp(x_new, x_orig, h0)
    h0 = project_to_sum_box(h0, n_points/2)
    
    # Strategy 2: Symmetric piecewise constant
    h1 = np.zeros(n_points)
    # Divide into 4 segments with alternating high/low values
    seg_size = n_points // 4
    for i in range(4):
        start = i * seg_size
        end = (i+1) * seg_size if i < 3 else n_points
        if i % 2 == 0:
            h1[start:end] = 0.7
        else:
            h1[start:end] = 0.3
    h1 = project_to_sum_box(h1, n_points/2)
    
    # Strategy 3: Smooth with Gaussian-like peaks
    h2 = np.zeros(n_points)
    centers = [n_points//4, 3*n_points//4]
    widths = [n_points//8, n_points//8]
    for i in range(n_points):
        val = 0
        for center, width in zip(centers, widths):
            x = (i - center) / width
            val += 0.6 * np.exp(-0.5 * x**2)
        h2[i] = val
    h2 = project_to_sum_box(h2, n_points/2)
    
    # Strategy 4: Sparse high value in middle
    h3 = np.zeros(n_points)
    mid_start = n_points//2 - n_points//8
    mid_end = n_points//2 + n_points//8
    h3[mid_start:mid_end] = 1.0
    h3 = project_to_sum_box(h3, n_points/2)
    
    # Strategy 5: Raised cosine shape
    h4 = np.zeros(n_points)
    x = np.linspace(0, 2, n_points)
    h4 = 0.5 * (1 + np.cos(np.pi * x))
    h4 = project_to_sum_box(h4, n_points/2)
    
    # Strategy 6: Mixed binary
    h5 = np.zeros(n_points)
    # Place 1s in alternating pattern but with more 0s
    for i in range(0, n_points, 4):
        if i < n_points:
            h5[i] = 0.9
    h5 = project_to_sum_box(h5, n_points/2)
    
    initial_guesses = [h0, h1, h2, h3, h4, h5]
    
    # Use COBYLA method which is more stable for constrained optimization
    for i, h_start in enumerate(initial_guesses):
        if time.time() - start_time > budget_s:
            break
            
        h_current = h_start.copy()
        bounds = [(0, 1) for _ in range(n_points)]

            return evaluate_c5(h)

            return np.sum(h) - TARGET_SUM

    tabu_size = 10

            break

        neighbor = best_h + np.random.normal(0, 0.005, n_points)

            # Add to tabu list
            tabu_list.append(neighbor_tuple)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        else:
            # Occasionally accept worse solutions to escape local minima
            if np.random.random() < 0.05:
                best_h = neighbor.copy()
                best_c5 = c5_neighbor
                tabu_list.append(neighbor_tuple)
                if len(tabu_list) > tabu_size:
                    tabu_list.pop(0)
    
    # Final check
    if best_h is not None:
        best_c5 = evaluate_c5(best_h)

    if best_h is None:
        # Fallback to initial construction if nothing was found
        best_h = h0
        best_c5 = evaluate_c5(h0)
    
    return (best_h.tolist(), float(best_c5), n_points)
