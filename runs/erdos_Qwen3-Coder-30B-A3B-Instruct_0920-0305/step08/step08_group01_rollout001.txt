import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def random_binary_solution():
    """Generate a random binary solution with exactly TARGET_SUM ones"""
    h = np.zeros(n_points, dtype=float)
    ones_indices = np.random.choice(n_points, size=int(TARGET_SUM), replace=False)
    h[ones_indices] = 1.0
    return h

def simulated_annealing_binary(max_iter=20000, initial_temp=0.015, final_temp=0.0001):
    """Simulated annealing for binary patterns"""
    # Initialize
    current_h = random_binary_solution()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Temperature schedule
    start_time = time.time()
    elapsed = 0
    iteration = 0
    
    # Track improvements
    last_improvement = 0
    patience = 1000
    
    while iteration < max_iter and elapsed < 550:
        # Update temperature
        fraction = elapsed / 550.0
        temp = initial_temp * (final_temp / initial_temp) ** fraction
        
        # Perform swap move
        ones_indices = np.where(current_h == 1)[0]
        zeros_indices = np.where(current_h == 0)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            break
            
        i = np.random.choice(ones_indices)
        j = np.random.choice(zeros_indices)
        
        # Swap values
        new_h = current_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        
        # Compute new C5
        new_c5 = evaluate_c5(new_h)
        
        # Accept or reject the move
        if new_c5 < current_c5:
            # Accept improvement
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
                last_improvement = iteration
        else:
            # Accept with probability
            delta = current_c5 - new_c5
            if np.random.random() < np.exp(delta / temp):
                current_h = new_h
                current_c5 = new_c5
                
        iteration += 1
        elapsed = time.time() - start_time
        
        # Early exit if no improvement in patience iterations
        if iteration - last_improvement > patience:
            break
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points / 2
    
    best_h = None
    best_c5 = float('inf')
    
    # Run multiple restarts
    restarts = 0
    while time.time() - start_time < budget_s * 0.95:
        h, c5 = simulated_annealing_binary(max_iter=20000)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
        restarts += 1
    
    # Final refinement with gradient-based optimization
    if best_h is not None:
        # Try gradient-based refinement
        try:
            from scipy.optimize import minimize
            
            def objective(h):
                return evaluate_c5(h)
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
            
            # Use SLSQP with bounds and constraint
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            
            if res.success:
                refined_h = res.x
                refined_c5 = evaluate_c5(refined_h)
                if refined_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
                    
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
