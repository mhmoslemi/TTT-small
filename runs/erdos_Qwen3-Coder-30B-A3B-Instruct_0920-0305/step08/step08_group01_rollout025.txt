import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
from scipy.linalg import cholesky

def compute_shift_matrices(n):
    """Compute shift matrices R_s (only for s=1,...,n//2)"""
    R_list = []
    for s in range(1, n//2 + 1):
        R = np.zeros((n, n))
        for i in range(n - s):
            R[i, i+s] = 1
        R_list.append(R)
    return R_list

def evaluate_c5(h):
    """Evaluate the C5 bound for step function h"""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / len(h))

def solve_sdp(n, R_list, symmetry_break=True):
    """Solve the SDP relaxation"""
    # Variables
    X = cp.Variable((n, n), symmetric=True)
    mu = cp.Variable(n)
    lambda_var = cp.Variable()

    # Constraints
    constraints = [X >> 0]  # PSD constraint
    for s, R in enumerate(R_list):
        constraints.append(cp.trace(R @ X) >= lambda_var)
    
    if symmetry_break:
        # Fix x_0 = 1, so μ_0 = 1
        constraints.append(mu[0] == 1)
        constraints.append(cp.sum(mu[1:]) == (n - 1) / 2)
    else:
        constraints.append(cp.sum(mu) == n / 2)
    
    for i in range(n):
        constraints.append(mu[i] <= 1)
        constraints.append(mu[i] >= 0)
        constraints.append(X[i, i] == mu[i])
    
    # Objective
    objective = cp.Maximize(lambda_var)
    prob = cp.Problem(objective, constraints)
    
    # Solve
    prob.solve(solver=cp.SCS, verbose=False, max_iters=1000, eps=1e-4)
    
    if prob.status == cp.OPTIMAL:
        return mu.value, X.value
    else:
        # Fallback to naive initialization
        return None, None

def round_sdp_solution(mu, V, n):
    """Round the relaxed SDP solution to binary vector"""
    num_rounds = 500
    best_h = None
    best_c5 = float('inf')
    
    for _ in range(num_rounds):
        # Sample random Gaussian vector
        g = np.random.randn(V.shape[1])
        # Compute scores
        scores = V @ g
        # Create binary vector by sorting scores
        idxs = np.argsort(scores)[::-1]
        h_bin = np.zeros(n)
        h_bin[idxs[:n//2]] = 1
        
        # Evaluate
        c5 = evaluate_c5(h_bin)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_bin.copy()
    
    return best_h, best_c5

def local_refinement(h, max_iter=10000):
    """Do local shift-aware swaps to refine the solution"""
    c = correlate(h, 1 - h, mode="full")
    kmax = np.argmax(c)
    worst_c = c[kmax]
    
    # Get the sets contributing to worst shift
    n = len(h)
    Bad = []
    Good = []
    for i in range(n - kmax):
        if h[i] == 1 and h[i+kmax] == 0:
            Bad.append(i)
        if h[i] == 0 and h[i+kmax] == 1:
            Good.append(i)
    
    # Perform swaps
    h_new = h.copy()
    for _ in range(max_iter):
        if len(Bad) == 0 or len(Good) == 0:
            break
        
        # Pick random bad and good
        i = np.random.choice(Bad)
        j = np.random.choice(Good)
        
        # Perform swap
        h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # Update sets
        if h_new[i] == 0:
            Bad.remove(i)
        elif h_new[i] == 1:
            Good.append(i)
        if h_new[j] == 1:
            Good.remove(j)
        elif h_new[j] == 0:
            Bad.append(j)
        
        # Check for improvement
        c_new = correlate(h_new, 1 - h_new, mode="full")
        if c_new[kmax] < worst_c:
            # Accept the change
            h = h_new.copy()
            worst_c = c_new[kmax]
        else:
            # Restore
            h_new[i], h_new[j] = h_new[j], h_new[i]
            # Rebuild bad/good sets
            Bad = []
            Good = []
            for idx in range(n - kmax):
                if h[idx] == 1 and h[idx+kmax] == 0:
                    Bad.append(idx)
                if h[idx] == 0 and h[idx+kmax] == 1:
                    Good.append(idx)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 120
    dx = 2.0 / n_points
    
    # Build shift matrices
    R_list = compute_shift_matrices(n_points)
    
    # Solve SDP
    mu, X = solve_sdp(n_points, R_list)
    if mu is None or X is None:
        # Fallback to a simple construction
        h_best = np.ones(n_points//2)
        h_best = np.concatenate((h_best, np.zeros(n_points - n_points//2)))
        c5 = evaluate_c5(h_best)
        return (h_best.tolist(), float(c5), n_points)
    
    # Factorize
    try:
        V = cholesky(X, lower=True)
    except:
        # If Cholesky fails, try eigenvalue decomposition
        try:
            _, V = np.linalg.eigh(X)
            V = V[:, ::-1]  # Reverse for higher eigenvalues first
        except:
            # Fallback to identity initialization
            V = np.eye(n_points)
    
    # Round
    h, c5 = round_sdp_solution(mu, V, n_points)
    
    # Refine locally
    h_refined = local_refinement(h)
    c5_refined = evaluate_c5(h_refined)
    
    if c5_refined < c5:
        h = h_refined
        c5 = c5_refined
    
    # If still not enough time, try an upscale
    if time.time() - start_time < 0.8 * budget_s:
        # Try a finer version
        n_upscale = 180
        R_upscale = compute_shift_matrices(n_upscale)
        mu_upscale = np.interp(np.linspace(0, n_points-1, n_upscale), np.arange(n_points), mu)
        X_upscale = np.zeros((n_upscale, n_upscale))
        # Initialize with a coarse expanded version
        h_coarse = np.zeros(n_points)
        h_coarse[:n_points//2] = 1
        h_expanded = np.tile(h_coarse, n_upscale // n_points)
        mu_upscale = np.interp(np.linspace(0, n_points-1, n_upscale), np.arange(n_points), mu)
        # Re-solve SDP with expanded setup (only if not too slow)
        mu_u, X_u = solve_sdp(n_upscale, R_upscale)
        if mu_u is not None and X_u is not None:
            try:
                V_u = cholesky(X_u, lower=True)
            except:
                try:
                    _, V_u = np.linalg.eigh(X_u)
                    V_u = V_u[:, ::-1]
                except:
                    V_u = np.eye(n_upscale)
            h_up, c5_up = round_sdp_solution(mu_u, V_u, n_upscale)
            if c5_up < c5:
                h = h_up
                c5 = c5_up
    
    return (h.tolist(), float(c5), n_points)
