import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):

    # Initialize
    TARGET_SUM = n_points / 2
    h0 = adaptive_sampling_initialization()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Genetic algorithm approach
    population_size = 20
    num_generations = 100
    mutation_rate = 0.1
    elite_size = 4
    
    # Create initial population
    population = []
    for _ in range(population_size):
        h = adaptive_sampling_initialization()
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_c5(h) for h in population]
    
    for generation in range(num_generations):
        if time.time() - start_time > budget_s:
            break
            
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        new_population = sorted_population[:elite_size]
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1_idx = sorted_indices[np.random.randint(0, elite_size)]
            parent2_idx = sorted_indices[np.random.randint(0, elite_size)]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points-1)
            child = np.concatenate([population[parent1_idx][:crossover_point],

            # Mutation
            if np.random.random() < mutation_rate:

                child[mutate_idx] = np.random.random()

            new_population.append(child)

                best_h[idx] += delta
                best_h = project_to_sum_box(best_h, TARGET_SUM)
                best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
