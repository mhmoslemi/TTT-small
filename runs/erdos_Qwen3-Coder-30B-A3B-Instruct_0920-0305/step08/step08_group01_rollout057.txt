def build_grid_from_intervals(intervals, n_points):
    """Convert intervals to binary grid of size n_points"""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    # Compute grid centers
    centers = np.arange(0.5*dx, 2.0, dx)
    for s, e in intervals:
        mask = (centers >= s) & (centers < e)
        h[mask] = 1.0
    return h

def c5_from_corr_fft(h):
    """Evaluate C5 using FFT-based correlation"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr)

def decode_params(z, m=5):
    """Convert z vector to interval parameters in [0,2] with total length = 1"""
    # Ensure z values are valid [0,1] range
    z = np.clip(z, 0, 1)
    l = np.array([0.0001 + (1-0.0002)*z[i] for i in range(m)])
    g = np.array([z[m+i] for i in range(m-1)])  # g_{m-1} only
    l = l * (1 / np.sum(l))  # Renormalize
    # Compute cumulative positions
    s = [0.0]
    for i in range(m):
        s.append(s[-1] + l[i] + (g[i] if i < len(g) else 0))
    # Adjust to fit within [0,2]
    if s[-1] > 2:
        scale = 2.0 / s[-1]
        l = l * scale
        g = g * scale
        s = [val * scale for val in s]
    # Compute intervals
    intervals = [(s[i], s[i+1]) for i in range(m)]
    return intervals

def objective(z):
    """Evaluate the objective function (C5) for given interval parameters z."""
    try:
        intervals = decode_params(z)
        h = build_grid_from_intervals(intervals, n_points)
        return c5_from_corr_fft(h) * dx
    except:
        return 1e10  # Return large value for invalid inputs

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Parameters
    m = 5  # Number of intervals
    n_points = 180
    dx = 2.0 / n_points
    time_limit = start_time + budget_s
    target_sum = n_points / 2  # Sum constraint for h

    # Initialize with random trials to seed DE
    best_c5 = 1e10
    best_intervals = None

    for trial in range(50):
        if time.time() > time_limit:
            break
        z = np.random.rand(2*m)
        try:
            c5 = objective(z)
            if c5 < best_c5:
                best_c5 = c5
                best_intervals = decode_params(z)
        except:
            continue

    # Run differential evolution
    bounds = [(0, 1) for _ in range(2*m)]
    de_maxiter = int(10 * (budget_s * 0.6) / 10) if budget_s > 10 else 100  # Limited by time
    try:
        res = differential_evolution(
            objective,
            bounds,
            maxiter=de_maxiter,
            popsize=15,
            strategy='best1bin',
            disp=False,
            seed=seed,
        )
        if not res.success:
            raise ValueError("DE failed")
        best_z = res.x
        intervals = decode_params(best_z)
        h = build_grid_from_intervals(intervals, n_points)
        current_best_c5 = c5_from_corr_fft(h) * dx
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_intervals = intervals
    except:
        pass

    # Local refinement
    if best_intervals is not None:
        h = build_grid_from_intervals(best_intervals, n_points)
        prev_c5 = best_c5
        for _ in range(200):  # Limited iterations
            if time.time() > time_limit:
                break
            scores_before = np.zeros(len(best_intervals))
            for i, (s, e) in enumerate(best_intervals):
                scores_before[i] = max(
                    c5_from_corr_fft(
                        build_grid_from_intervals(
                            [(s - dx/4, e) if i == 0 else (s, e) for j, (s, e) in enumerate(best_intervals) if j != i] +
                            [(s + dx/4, e) if i == 0 else (s, e) for j, (s, e) in enumerate(best_intervals) if j != i]
                            + ([(s, e - dx/4)] if i < len(best_intervals)-1 else [])
                            + ([(s, e + dx/4)] if i > 0 else [])
                        )
                    ) * dx,
                    c5_from_corr_fft(
                        build_grid_from_intervals(
                            [(s, e - dx/4) if i == len(best_intervals)-1 else (s, e) for j, (s, e) in enumerate(best_intervals) if j != i] +
                            [(s, e + dx/4) if i == len(best_intervals)-1 else (s, e) for j, (s, e) in enumerate(best_intervals) if j != i]
                            + ([(s - dx/4, e)] if i < len(best_intervals)-1 else [])
                            + ([(s + dx/4, e)] if i > 0 else [])
                        )
                    ) * dx
                )

            changed = False
            for i in range(len(best_intervals)):
                if scores_before[i] < prev_c5 - 1e-10:
                    prev_c5 = scores_before[i]
                    best_intervals[i] = (best_intervals[i][0] - dx/4, best_intervals[i][1])
                    if best_intervals[i][0] < 0:
                        best_intervals[i] = (0, best_intervals[i][1])
                    changed = True

            for i in range(len(best_intervals)-1, -1, -1):
                if scores_before[i] < prev_c5 - 1e-10:
                    prev_c5 = scores_before[i]
                    best_intervals[i] = (best_intervals[i][0], best_intervals[i][1] + dx/4)
                    if best_intervals[i][1] > 2.0:
                        best_intervals[i] = (best_intervals[i][0], 2.0)
                    changed = True
            if not changed:
                break

    # Final evaluation
    try:
        if best_intervals:
            h_final = build_grid_from_intervals(best_intervals, n_points)
            final_c5 = c5_from_corr_fft(h_final) * dx
            if final_c5 < best_c5:
                best_c5 = final_c5
        else:
            # Fallback to initial
            h_final = np.array(initial_h_values)
            final_c5 = c5_from_corr_fft(h_final) * dx
            best_c5 = final_c5
    except:
        # Fallback to initial
        h_final = np.array(initial_h_values)
        final_c5 = c5_from_corr_fft(h_final) * dx
        best_c5 = final_c5

    # Final verification with np.correlate
    try:
        c_original = correlate(h_final, 1 - h_final, mode="full")
        final_c5_original = np.max(c_original) * dx
        best_c5 = final_c5_original
    except:
        pass

    return (h_final.tolist(), float(best_c5), n_points)
