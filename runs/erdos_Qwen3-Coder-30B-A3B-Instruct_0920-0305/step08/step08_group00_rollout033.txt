import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def sigmoid(x, k=10, x0=0.5):
    """Sigmoid function for smooth transitions"""
    return 1 / (1 + np.exp(-k*(x-x0)))

def create_smooth_initial():
    """Create a smooth initial guess with sigmoidal transitions"""
    x = np.linspace(0, 2, n_points)
    # Create a smooth step function with sigmoid transitions
    h = sigmoid(x, k=20, x0=0.5) * (1 - sigmoid(x, k=20, x0=1.5))
    h = h / np.sum(h) * TARGET_SUM
    return h

def create_peaked_initial():
    """Create a solution with sharp peaks in key regions"""
    h = np.zeros(n_points)
    
    # Add a central peak
    center = n_points // 2
    h[center-5:center+6] = 1.0
    
    # Add a second peak on the right
    right_center = 3*n_points//4
    h[right_center-3:right_center+4] = 0.8
    
    # Adjust to meet constraints
    h = h / np.sum(h) * TARGET_SUM
    return h

def generate_multi_peak():
    """Generate a multi-peak solution that might improve overlap"""
    h = np.zeros(n_points)
    peaks = [n_points//4, n_points//2, 3*n_points//4]
    heights = [1.0, 0.9, 0.8]
    
    for i, (peak_pos, height) in enumerate(zip(peaks, heights)):
        start = max(0, peak_pos - 8)
        end = min(n_points, peak_pos + 9)
        h[start:end] = np.maximum(h[start:end], height * np.exp(-((np.arange(start, end) - peak_pos)**2) / 16))
    
    # Normalize
    h = h / np.sum(h) * TARGET_SUM
    return h

def adaptive_init():
    """Create initial solution using multiple strategies"""
    # We'll try several approaches and pick best
    strategies = [
        create_smooth_initial,
        create_peaked_initial,
        generate_multi_peak,
        lambda: np.full(n_points, 1.0/n_points)  # Uniform
    ]
    
    best_c5 = float('inf')
    best_h = None
    
    for strategy in strategies:
        h = strategy()
        h = project_to_sum_box(h, TARGET_SUM)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h

def improve_with_local_search(h, max_iter=1000):
    """Improve a solution with local search"""
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for iteration in range(max_iter):
        # Try small mutations
        h_trial = best_h.copy()
        for _ in range(5):  # Make 5 small changes
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.02)
            h_trial[idx] += delta
            
        # Project back to constraints
        h_trial = project_to_sum_box(h_trial, TARGET_SUM)
        new_c5 = evaluate_c5(h_trial)
        
        if new_c5 < best_c5:
            best_h = h_trial.copy()
            best_c5 = new_c5
            
    return best_h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    
    # Use our improved initialization
    h0 = adaptive_init()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Try local search improvement for a bit
    search_time = 0.5 * budget_s
    while time.time() - start_time < search_time:
        h_improved = improve_with_local_search(best_h, max_iter=100)
        c5_improved = evaluate_c5(h_improved)
        if c5_improved < best_c5:
            best_h = h_improved.copy()
            best_c5 = c5_improved
    
    # More extensive local search
    search_time = 0.8 * budget_s
    while time.time() - start_time < search_time:
        # Random tweak
        h_tweak = best_h + np.random.normal(0, 0.01, n_points)
        h_tweak = project_to_sum_box(h_tweak, TARGET_SUM)
        c5_tweak = evaluate_c5(h_tweak)
        
        if c5_tweak < best_c5:
            best_h = h_tweak.copy()
            best_c5 = c5_tweak
            
        # Local search in small neighborhood
        h_local = improve_with_local_search(best_h, max_iter=50)
        c5_local = evaluate_c5(h_local)
        if c5_local < best_c5:
            best_h = h_local.copy()
            best_c5 = c5_local
    
    # Final refinement with a conservative approach
    final_time = budget_s
    while time.time() - start_time < final_time:
        # Mutation with large steps
        h_mutation = best_h + np.random.normal(0, 0.03, n_points)
        h_mutation = project_to_sum_box(h_mutation, TARGET_SUM)
        c5_mutation = evaluate_c5(h_mutation)
        
        if c5_mutation < best_c5:
            best_h = h_mutation.copy()
            best_c5 = c5_mutation
            
        # Set a few random points to 0 or 1
        h_binary = best_h.copy()
        for i in range(10):
            idx = np.random.randint(0, n_points)
            if np.random.random() < 0.5:
                h_binary[idx] = 1.0
            else:
                h_binary[idx] = 0.0
        h_binary = project_to_sum_box(h_binary, TARGET_SUM)
        c5_binary = evaluate_c5(h_binary)
        if c5_binary < best_c5:
            best_h = h_binary.copy()
            best_c5 = c5_binary
    
    return (best_h.tolist(), float(best_c5), n_points)
