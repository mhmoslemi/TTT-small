import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 200
dx = 2.0 / n_points

def compute_corr_int(h):
    """Compute correlation using FFT and return integer values"""
    h_rev = h[::-1].astype(np.float64)
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    # round to nearest integer (small float errors)
    corr_int = np.rint(corr).astype(int)
    return corr_int

def compute_c5(h):
    """Compute C5 value using integer correlation"""
    corr_int = compute_corr_int(h)
    return dx * np.max(corr_int)

def generate_initial_binary():
    """Generate initial random binary vector with exactly n/2 ones"""
    h = np.zeros(n_points, dtype=int)
    h[:n_points//2] = 1
    np.random.shuffle(h)
    return h

def simulated_annealing(n_points, time_budget):
    """Simulated annealing for binary pattern optimization"""
    start_time = time.time()
    
    # Generate initial binary pattern
    h = generate_initial_binary()
    best_h = h.copy()
    best_corr_int = compute_corr_int(h)
    best_max = best_corr_int.max()
    current_h = h.copy()
    current_max = best_max
    
    # Temperature schedule
    T = 5.0
    cooling = 0.9995
    
    iteration = 0
    while time.time() - start_time < time_budget:
        iteration += 1
        
        # Generate swap: pick one 1 and one 0 at random
        ones_idx = np.where(current_h == 1)[0]
        zeros_idx = np.where(current_h == 0)[0]
        i_one = np.random.choice(ones_idx)
        i_zero = np.random.choice(zeros_idx)
        
        # Swap
        cand_h = current_h.copy()
        cand_h[i_one] = 0
        cand_h[i_zero] = 1
        
        # Compute new correlation and max
        cand_corr_int = compute_corr_int(cand_h)
        cand_max = cand_corr_int.max()
        delta = cand_max - current_max
        
        # Accept or reject
        if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, 1e-6)):
            current_h = cand_h
            current_max = cand_max
            
            # Update best solution
            if cand_max < best_max:
                best_h = cand_h.copy()
                best_max = cand_max
                
        # Cool down
        T *= cooling
        
        # Check if we should stop (very cold)
        if T < 1e-8:
            break
    
    return best_h, best_max

def local_improvement(h, max_iter=1000, patience=200):
    """Local hill climbing improvement"""
    best_h = h.copy()
    best_max = compute_corr_int(best_h).max()
    no_improve = 0
    
    for _ in range(max_iter):
        # Generate candidate swaps
        swaps = []
        for _ in range(50):
            i_one = np.random.choice(np.where(best_h == 1)[0])
            i_zero = np.random.choice(np.where(best_h == 0)[0])
            swaps.append((i_one, i_zero))
            
        improved = False
        for i_one, i_zero in swaps:
            cand_h = best_h.copy()
            cand_h[i_one] = 0
            cand_h[i_zero] = 1
            cand_max = compute_corr_int(cand_h).max()
            if cand_max < best_max:
                best_h = cand_h
                best_max = cand_max
                improved = True
                break
                
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= patience:
                break
                
    return best_h, best_max

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Run multiple annealing runs with different seeds
    best_h = None
    best_c5 = float('inf')
    
    # First phase: multiple simulated annealing runs
    num_runs = 5
    run_time = budget_s * 0.7
    run_start = time.time()
    
    for run_id in range(num_runs):
        if time.time() - start_time > budget_s:
            break
            
        # Create new seed for this run
        seed_run = seed + run_id * 1000
        np.random.seed(seed_run)
        
        # Run simulated annealing
        h, max_corr = simulated_annealing(n_points, run_time / num_runs)
        
        # Run local improvement
        h, max_corr = local_improvement(h, 500, 100)
        
        c5 = dx * max_corr
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
    # Final optimization: more local refinement
    if best_h is not None:
        best_h, final_max = local_improvement(best_h, 1000, 200)
        best_c5 = dx * final_max
    
    # Convert best_h to float list as required by return format
    if best_h is not None:
        h_values = best_h.astype(float).tolist()
    else:
        # Fallback: generate a basic pattern
        h_values = [1.0] * (n_points // 2) + [0.0] * (n_points // 2)
        np.random.shuffle(h_values)
    
    return (h_values, best_c5, n_points)
