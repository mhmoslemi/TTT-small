import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
from scipy.fft import fft, ifft

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def waterfilling_projection(h, target_sum):
    """Projection onto simplex with box constraints using bisection"""
    def f(lam):
        # Compute sum of clamp(h + lam, 0, 1)
        clamped = np.clip(h + lam, 0, 1)
        return np.sum(clamped) - target_sum
    
    # Binary search for lambda
    a, b = -1.0, 1.0
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def compute_autocorrelation(h, shift):
    """Compute autocorrelation for a specific shift"""
    return np.sum(h[:-shift] * h[shift:])

def compute_worst_shift(h):
    """Find the shift that gives maximum overlap"""
    n = len(h)
    max_overlap = -1
    worst_k = 0

    # For each possible shift
    for k in range(1, n):
        # Compute overlap for this shift using correlation
        corr_val = np.sum(h[:-k] * h[k:])
        overlap = np.sum(h) - corr_val
        if overlap > max_overlap:
            max_overlap = overlap
            worst_k = k
    
    return worst_k

def gradient_step(h, k_star, alpha):
    """Take a gradient step with subgradient"""
    n = len(h)
    
    # Construct the subgradient g
    g = np.zeros(n)
    # For shift k_star, g_i = 1 - h_{i+k*} - h_{i-k*}
    for i in range(n):
        h_left = h[i - k_star] if i >= k_star else 0
        h_right = h[i + k_star] if i + k_star < n else 0
        g[i] = 1 - h_left - h_right
    
    # Take step
    h_new = h - alpha * g
    
    # Project
    h_new = waterfilling_projection(h_new, TARGET_SUM)
    
    return h_new

def subgradient_descent(h_initial, max_time_s):
    """Deterministic sub-gradient descent"""
    start_time = time.time()
    
    h = h_initial.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Use decreasing step size
    alpha = 0.5 * dx
    t = 1
    
    while time.time() - start_time < max_time_s:
        # Find worst shift
        k_star = compute_worst_shift(h)
        
        # Compute step
        h_new = gradient_step(h, k_star, alpha / np.sqrt(t))
        
        # Evaluate
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new
        t += 1
        
        # Early termination if no improvement for long time
        if t % 500 == 0:
            if evaluate_c5(h) == best_c5:
                break
    
    return best_h, best_c5

def generate_binary_start():
    """Create initial binary solution via greedy spacing"""
    h = np.zeros(n_points)
    count = 0
    step_size = 2  # approximately N/M = 2
    for i in range(0, n_points, step_size):
        if count < TARGET_SUM:
            h[i] = 1
            count += 1
        else:
            break
    return h

def solve_binary_milp():
    """Solve binary MILP optimization"""
    # Variables
    y = cp.Variable(n_points, boolean=True)
    t = cp.Variable()

    # Objective
    objective = cp.Maximize(t)

    # Constraints
    constraints = [cp.sum(y) == TARGET_SUM, t <= 0]

    # Add product constraints for each shift
    for k in range(1, n_points):
        # Linearize y_i * y_{i+k}
        w = cp.Variable(n_points - k, nonneg=True)
        constraints.extend([
            w <= y[:-k],
            w <= y[k:],
            w >= y[:-k] + y[k:] - 1,
            w >= 0
        ])
        # Add to the t constraint
        constraints.append(cp.sum(w) >= t)

    # Solve
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers
    solvers = [cp.CBC, cp.GLPK_MI, cp.ECOS_BB]
    for solver in solvers:
        try:
            prob.solve(solver=solver, max_seconds=0.4)
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_UNKNOWN]:
                return np.array(y.value, dtype=float)
        except:
            continue
    return None

def swap_move(h, delta):
    """Perform 2-index swap move for stochastic refinement"""
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    
    # Calculate new values
    h_i_new = h[i] + delta
    h_j_new = h[j] - delta
    
    # Clip
    h_i_new = max(0, min(1, h_i_new))
    h_j_new = max(0, min(1, h_j_new))
    
    # Get new h
    h_new = h.copy()
    h_new[i] = h_i_new
    h_new[j] = h_j_new
    
    # Project
    h_new = waterfilling_projection(h_new, TARGET_SUM)
    
    return h_new

def simulated_annealing_refinement(h_initial, max_time_s):
    """Stochastic refinement via simulated annealing"""
    start_time = time.time()
    
    h = h_initial.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    T_start = 5 * dx
    t = 0
    max_iter = 100000
    
    while time.time() - start_time < max_time_s and t < max_iter:
        # Swap move
        delta = np.random.uniform(-0.02, 0.02)
        h_new = swap_move(h, delta)
        
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            h = h_new
            best_h = h_new.copy()
            best_c5 = c5_new
        else:
            # Accept with probability
            T = T_start * (1 - t / max_iter)
            if T > 0:
                prob = np.exp(-(c5_new - best_c5) / T)
                if np.random.random() < prob:
                    h = h_new
        
        t += 1
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Phase 1: Binary MILP or greedy construction
    h_initial = None
    phase1_end = start_time + 0.4 * budget_s
    
    try:
        # Try solving the binary MILP first
        binary_sol = solve_binary_milp()
        if binary_sol is not None:
            h_initial = binary_sol
            c5_initial = evaluate_c5(h_initial)
        else:
            # Fallback to greedy
            h_initial = generate_binary_start()
            h_initial = project_to_sum_box(h_initial, TARGET_SUM)
            c5_initial = evaluate_c5(h_initial)
    except:
        # Fallback to greedy
        h_initial = generate_binary_start()
        h_initial = project_to_sum_box(h_initial, TARGET_SUM)
        c5_initial = evaluate_c5(h_initial)
    
    # Phase 2: Sub-gradient descent
    phase2_end = start_time + 0.75 * budget_s
    h_refined, c5_refined = subgradient_descent(h_initial, phase2_end - start_time)
    
    # Phase 3: Stochastic refinement
    h_final, c5_final = simulated_annealing_refinement(h_refined, budget_s - (time.time() - start_time))
    
    return (h_final.tolist(), float(c5_final), n_points)
