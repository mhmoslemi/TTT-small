from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def evaluate_c5_corr(h):
    """Evaluation of C5 using np.correlate for consistency with target evaluation"""
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def neighbor_sum(h, lag):
    """Compute neighbor sum for given lag"""
    lag_abs = abs(lag)
    if lag_abs == 0:
        return 2 * h
    else:
        w = np.zeros_like(h)
        if lag > 0:
            w[:n_points-lag_abs] += h[lag_abs:]
            w[lag_abs:] += h[:n_points-lag_abs]
        else:
            w[lag_abs:] += h[:-lag_abs]
            w[:-lag_abs] += h[lag_abs:]
        return w

def mass_transfer_step(h, step_size, epsilon=1e-6):
    """Perform one mass transfer step"""
    # Compute correlation
    corr = np.correlate(h, 1-h, mode="full")
    idx_max = np.argmax(corr)
    lag_max = idx_max - (n_points - 1)
    
    # Compute neighbor sum for the worst lag
    w = neighbor_sum(h, lag_max)
    
    # Find good indices for transfer
    low_mask = h > epsilon
    high_mask = h < 1 - epsilon
    
    low_indices = np.where(low_mask)[0]
    high_indices = np.where(high_mask)[0]
    
    if len(low_indices) == 0 or len(high_indices) == 0:
        return h, False
    
    # Find indices with min and max w values among valid indices
    low_idx = low_indices[np.argmin(w[low_indices])]
    high_idx = high_indices[np.argmax(w[high_indices])]
    
    if low_idx == high_idx:
        return h, False
    
    # Compute transfer amount
    delta = min(step_size, h[low_idx], 1 - h[high_idx])
    
    if delta < 1e-12:
        return h, False
    
    # Apply transfer
    h_new = h.copy()
    h_new[low_idx] -= delta
    h_new[high_idx] += delta
    
    return h_new, True

            h_final = project_to_sum_box(res.x, TARGET_SUM)
            c5_final = evaluate_c5_corr(h_final)
