import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def build_grid_from_intervals(intervals, n_points):
    """Build binary grid h from interval endpoints [s1, e1, s2, e2, ...]"""
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    x_grid = np.linspace(0, 2, n_points, endpoint=False) + dx/2
    
    for i in range(0, len(intervals), 2):
        s, e = intervals[i], intervals[i+1]
        mask = (x_grid >= s) & (x_grid < e)
        h[mask] = 1.0
    return h

def evaluate_c5_fft(h):
    """Compute C5 using FFT as before"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr)

def decode_params(z, m, n_points):
    """Convert parameter vector z to interval endpoints with mass=1 and fit [0,2]"""
    # z is of length 2*m
    l = np.clip(z[:m], 1e-6, 1-1e-6)   # lengths
    g = np.clip(z[m:], 0, 100)         # gaps
    
    # Renormalize lengths to sum to 1
    l = l / np.sum(l)
    # Compute interval positions
    s = np.zeros(m)
    e = np.zeros(m)
    s[0] = 0
    e[0] = s[0] + l[0]
    for i in range(1, m):
        s[i] = e[i-1] + g[i-1]
        e[i] = s[i] + l[i]
    
    # If end exceeds 2, scale gaps down
    final_end = e[-1]
    if final_end > 2:
        scale = 2 / final_end
        g = g * scale
        e = s + l
        for i in range(1, m):
            s[i] = e[i-1] + g[i-1]
            e[i] = s[i] + l[i]
    
    # Return [s1,e1,s2,e2,...]
    intervals = np.empty(2*m)
    intervals[0::2] = s
    intervals[1::2] = e
    return intervals

def objective(z, m, n_points=180):
    intervals = decode_params(z, m, n_points)
    h = build_grid_from_intervals(intervals, n_points)
    c5 = evaluate_c5_fft(h)
    return c5

def local_refinement(intervals, n_points=180, max_iter=200):
    """Refine intervals by local edge moves"""
    intervals = intervals.copy()
    h = build_grid_from_intervals(intervals, n_points)
    dx = 2.0 / n_points
    current_c5 = evaluate_c5_fft(h)
    
    for _ in range(max_iter):
        improved = False
        # Try moving each edge
        for i in range(0, len(intervals), 2):
            # Try moving left edge left
            s_new = max(0, intervals[i] - dx/2)
            if s_new < intervals[i+1] - dx:
                intervals[i] = s_new
                h_new = build_grid_from_intervals(intervals, n_points)
                c5_new = evaluate_c5_fft(h_new)
                if c5_new < current_c5 - 1e-10:
                    current_c5 = c5_new
                    improved = True
                else:
                    intervals[i] = intervals[i]  # revert
            # Try moving left edge right
            s_new = min(2-dx, intervals[i] + dx/2)
            if s_new < intervals[i+1] - dx:
                intervals[i] = s_new
                h_new = build_grid_from_intervals(intervals, n_points)
                c5_new = evaluate_c5_fft(h_new)
                if c5_new < current_c5 - 1e-10:
                    current_c5 = c5_new
                    improved = True
                else:
                    intervals[i] = intervals[i]
            # Try moving right edge left
            e_new = max(intervals[i] + dx, intervals[i+1] - dx/2)
            if e_new > intervals[i] + dx:
                intervals[i+1] = e_new
                h_new = build_grid_from_intervals(intervals, n_points)
                c5_new = evaluate_c5_fft(h_new)
                if c5_new < current_c5 - 1e-10:
                    current_c5 = c5_new
                    improved = True
                else:
                    intervals[i+1] = intervals[i+1]  # revert
            # Try moving right edge right
            e_new = min(2, intervals[i+1] + dx/2)
            if e_new > intervals[i] + dx:
                intervals[i+1] = e_new
                h_new = build_grid_from_intervals(intervals, n_points)
                c5_new = evaluate_c5_fft(h_new)
                if c5_new < current_c5 - 1e-10:
                    current_c5 = c5_new
                    improved = True
                else:
                    intervals[i+1] = intervals[i+1]
        if not improved:
            break
    return intervals

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Settings
    n_points = 180
    m = 5  # number of intervals
    bounds = [(0, 1) for _ in range(2*m)]
    max_iter = 100
    popsize = 15 * 2 * m
    # Trial runs to get a sense of the best value
    best_c5 = float('inf')
    best_intervals = None
    
    # Phase 1: Random initialization
    num_init = 20
    for _ in range(num_init):
        z = np.random.rand(2*m)
        c5 = objective(z, m, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_intervals = decode_params(z, m, n_points)
    
    # Phase 2: Differential Evolution
    def obj_wrapper(z):
        return objective(z, m, n_points)
        
    phase2_end = start_time + 0.7 * budget_s
    
    de_result = differential_evolution(
        obj_wrapper, 
        bounds, 
        seed=seed,
        maxiter=max_iter,
        popsize=popsize,
        mutation=(0.5, 1.0),
        recombination=0.7,
        atol=1e-6,
        rtol=1e-6,
        callback=lambda x, f, context: None,
        disp=False
    )
    
    if de_result.success:
        z_best = de_result.x
        intervals = decode_params(z_best, m, n_points)
        c5 = objective(z_best, m, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_intervals = intervals
    
    # Phase 3: Local refinement on best solution
    if best_intervals is not None:
        refined = local_refinement(best_intervals, n_points)
        h_refined = build_grid_from_intervals(refined, n_points)
        c5_refined = evaluate_c5_fft(h_refined)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_intervals = refined
    
    # Phase 4: Fallback if necessary
    if time.time() < start_time + 0.9 * budget_s:
        # Random sampling and refine
        num_random = 5000
        for _ in range(num_random):
            z = np.random.rand(2*m)
            c5 = objective(z, m, n_points)
            if c5 < best_c5:
                best_c5 = c5
                intervals = decode_params(z, m, n_points)
                refined = local_refinement(intervals, n_points)
                h_refined = build_grid_from_intervals(refined, n_points)
                c5_refined = evaluate_c5_fft(h_refined)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_intervals = refined
    
    # Final verification using reference correlate
    if best_intervals is not None:
        h_final = build_grid_from_intervals(best_intervals, n_points)
        c5_ref = np.max(correlate(h_final, 1 - h_final, mode='full')) * (2.0 / n_points)
        if abs(c5_ref - best_c5) > 1e-6:
            best_c5 = c5_ref
    
    # Convert to final h
    h_final = build_grid_from_intervals(best_intervals, 120) if best_intervals is not None else np.zeros(120)
    
    return (h_final.tolist(), float(best_c5), n_points)
