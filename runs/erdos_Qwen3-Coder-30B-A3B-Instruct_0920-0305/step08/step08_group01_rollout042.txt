import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Use the pre-defined initial solution as starting point
    h0 = np.array(initial_h_values)
    h0 = np.clip(h0, 0, 1)
    
    # Warm start from the initial solution and refine using MILP approach
    def solve_milp_with_row_generation():
        # Variables
        x = cp.Variable(n_points, boolean=True)
        t = cp.Variable()
        
        # Initialize shifts to consider
        shifts = [1, 2, 3, 4, 5]
        
        # Constraints
        constraints = [cp.sum(x) == TARGET_SUM, t >= 0]
        
        # Build initial model
        y_vars = {}
        for s in shifts:
            for i in range(n_points):
                if i + s < n_points:
                    key = (i, s)
                    y_vars[key] = cp.Variable(boolean=True)
        
        # Add linearization constraints
        for s in shifts:
            for i in range(n_points):
                if i + s < n_points:
                    key = (i, s)
                    y = y_vars[key]
                    constraints.extend([
                        y <= x[i],
                        y <= x[i + s],
                        y >= x[i] + x[i + s] - 1
                    ])
        
        # Epigraph constraints
        for s in shifts:
            expr = 0
            for i in range(n_points):
                if i + s < n_points:
                    key = (i, s)
                    expr += y_vars[key]
            constraints.append(expr >= TARGET_SUM - t)
        
        # Objective
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        # Variables for efficient computation of full correlation
        eval_shifts = list(range(-n_points + 1, n_points))
        eval_shifts = [s for s in eval_shifts if abs(s) <= 5]
        shifts_considered = set(shifts)
        
        # Initialize with refined solution
        x.value = h0.copy()
        x.value = np.round(x.value).astype(int)
        if abs(np.sum(x.value) - TARGET_SUM) > 1e-6:
            # Project to sum constraint
            current_sum = np.sum(x.value)
            diff = int(TARGET_SUM - current_sum)
            if diff > 0:
                # Set some zeros to ones
                indices = np.where(x.value == 0)[0]
                for i in range(diff):
                    if i < len(indices):
                        x.value[indices[i]] = 1
            elif diff < 0:
                # Set some ones to zeros
                indices = np.where(x.value == 1)[0]
                for i in range(-diff):
                    if i < len(indices):
                        x.value[indices[i]] = 0
        
        best_h = h0.copy()
        best_c5 = evaluate_c5(best_h)
        
        # Iteratively add violated constraints
        while True:
            if time.time() - start_time > 0.9 * budget_s:
                break
                
            # Solve current MILP
            try:
                prob.solve(solver=CP.ECOS_BB, verbose=False, max_iter=10000)
            except:
                break
            
            if prob.status != cp.OPTIMAL:
                break
                
            # Get current binary solution
            x_val = np.round(x.value).astype(int)
            t_val = t.value
            
            # Evaluate true overlaps for all shifts using FFT
            x_float = x_val.astype(float)
            corr = compute_corr_fft(x_float)
            max_corr = np.max(corr)
            max_shifts = np.argmax(corr)
            
            # Get all valid shifts
            valid_shifts = list(range(-n_points + 1, n_points))
            max_shift_found = max_shifts - n_points + 1
            
            # Check if we have a new best solution
            if max_corr < best_c5:
                best_h = x_float.copy()
                best_c5 = max_corr
            
            # If there is a violation, add new constraint
            if max_corr > t_val + 1e-8:
                if abs(max_shift_found) not in shifts_considered:
                    shift_to_add = abs(max_shift_found)
                    # Add the shift to consider
                    shifts.append(shift_to_add)
                    shifts_considered.add(shift_to_add)
                    
                    # Add new variables and constraints
                    for i in range(n_points):
                        if i + shift_to_add < n_points:
                            key = (i, shift_to_add)
                            y_vars[key] = cp.Variable(boolean=True)
                            y = y_vars[key]
                            constraints.extend([
                                y <= x[i],
                                y <= x[i + shift_to_add],
                                y >= x[i] + x[i + shift_to_add] - 1
                            ])
                            
                    # Add epigraph constraint
                    expr = 0
                    for i in range(n_points):
                        if i + shift_to_add < n_points:
                            key = (i, shift_to_add)
                            expr += y_vars[key]
                    constraints.append(expr >= TARGET_SUM - t)
            else:
                # Converged, break
                break
                
        return best_h, best_c5
    
    def polish_solution(h):
        """Use smooth optimization technique to polish the binary solution"""
        # Convert to continuous
        h_continuous = h.copy()
        
        def smooth_overlap(h_vec, tau, gamma=100):
            """Smooth approximation to the max-over-shifts"""
            corr_vals = compute_corr_fft(h_vec)
            return tau + (1/gamma) * np.log(np.sum(np.exp(gamma * (corr_vals - tau))))
        
        def objective(x):
            return smooth_overlap(x, tau=evaluate_c5(h_continuous))
            
        # Optimize the continuous version with slight modifications
        constraints = [
            cp.sum(x) == TARGET_SUM
        ]
        
        # Use SLSQP to get a smooth upper bound
        x = cp.Variable(n_points)
        result = minimize(
            objective,
            h_continuous,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM},
            options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        
        if result.success:
            h_refined = result.x
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
        return h_continuous, evaluate_c5(h_continuous)
    
    # Execute main algorithm
    best_h, best_c5 = solve_milp_with_row_generation()
    if best_c5 > 0.38:  # if not competitive return the initial
        best_h, best_c5 = polish_solution(best_h)
    else:
        # Even better circuitous route with extra polishing
        h_polished, c5_polished = polish_solution(best_h)
        if c5_polished < best_c5:
            best_h, best_c5 = h_polished, c5_polished
            
    # Finalize with better direct polishing if time allows
    if time.time() - start_time < 0.95 * budget_s:
        # Do final direct polishing of best solution
        def obj(h):
            # Return part of the negative smoothed version of the overlap
            return -evaluate_c5(h)
        
        # Fine tune the final result
        res = minimize(
            obj,
            best_h,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM},
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        
        if res.success:
            refined_h = res.x
            refined_c5 = evaluate_c5(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5

    return (best_h.tolist(), float(best_c5), n_points)
