import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from math import sqrt

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
MAX_ITER = 5000  # For sub-gradient descent

def evaluate_c5(h):
    """Fast evaluation using FFT of h*(1-h)"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

# Step 1: Try solving a binary MILP to get a good 0/1 pattern
def solve_binary_milp(n):
    """Solve a MILP to get a good binary pattern"""
    # Binary variables for y_i (0/1)
    y = cp.Variable(n, boolean=True)
    M = n // 2  # Target sum
    
    # Auxiliary variables w_{k,i} 
    w = {}
    constraints = [cp.sum(y) == M]
    
    # Manually construct the constraints for linearization of products
    # Record the set of (k,i) pairs
    for k in range(1, n):
        for i in range(n - k):
            # Create a new variable for the product y_i * y_{i+k}
            if (k, i) not in w:
                w[(k, i)] = cp.Variable(1, boolean=False)
                # Linearization constraints
                constraints.append(w[(k, i)] <= y[i])
                constraints.append(w[(k, i)] <= y[i + k])
                constraints.append(w[(k, i)] >= y[i] + y[i + k] - 1)
                constraints.append(w[(k, i)] >= 0)
            
    # Formulate the minimum auto-correlation constraint
    t = cp.Variable(1)
    for k in range(1, n):
        for i in range(n - k):
            constraints.append(w[(k, i)] >= t)
    
    # Objective: maximize t
    prob = cp.Problem(cp.Maximize(t), constraints)
    
    try:
        prob.solve(solver=cp.CBC, max_seconds=0.35 * 600)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_EOF]:
            return y.value
    except:
        pass  # Fall back to random initialization

    # Return random binary solution if MILP fails
    return np.random.rand(n) > 0.5

# Step 2: Refine binary solution using sub-gradient descent
def project_to_box_and_sum(x, target_sum):
    """Project vector x onto [0,1] and sum = target_sum using water-filling"""
    x = np.clip(x, 0, 1)
    
    def total(x, lam):
        return np.sum(np.clip(x + lam, 0, 1)) - target_sum
    
    # Binary search for lambda
    a, b = -1, 1
    for _ in range(50):
        m = (a + b) / 2
        if total(x, m) < 0:
            a = m
        else:
            b = m
    
    lam = (a + b) / 2
    x = np.clip(x + lam, 0, 1)
    return x

def subgradient_descent_step(h, k_best, step_size):
    """Perform one sub-gradient step"""
    # Compute the gradient for the worst shift
    g = np.zeros_like(h)
    for i in range(len(h)):
        if i + k_best < len(h):
            g[i] += -h[i + k_best]
        if i - k_best >= 0:
            g[i] += -h[i - k_best]
        g[i] += 1
    
    # Update
    h_new = h - step_size * g
    h_new = project_to_box_and_sum(h_new, TARGET_SUM)
    return h_new

# Step 3: Stochastic refinement
def stochastic_refinement(h, max_iter=1000, T_start=0.1, T_end=0.001):
    """Simple simulated annealing variant with swaps"""
    T = T_start
    T_step = (T_start - T_end) / max_iter
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    for i in range(max_iter):
        T = max(T_end, T - T_step)
        
        # Pick two indices, not at extreme values
        i1 = np.random.randint(0, len(h))
        i2 = np.random.randint(0, len(h))
        
        # Only swap if both are not at bounds
        if 0 < h[i1] < 1 and 0 < h[i2] < 1:
            delta = np.random.uniform(-0.02, 0.02)
            h_new = current_h.copy()
            h_new[i1] += delta
            h_new[i2] -= delta
            
            # Clip to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Project to sum constraint
            h_new = project_to_box_and_sum(h_new, TARGET_SUM)
            
            new_c5 = evaluate_c5(h_new)
            
            # Accept or reject
            if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                current_h = h_new
                current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points / 2
    
    # Phase 1: Try to solve the binary MILP
    h_binary = solve_binary_milp(n_points)
    h_start = h_binary.astype(float)
    best_c5 = evaluate_c5(h_start)
    best_h = h_start.copy()
    
    # Phase 2: Sub-gradient descent using the binary solution
    elapsed = time.time() - start_time
    time_for_sg = 0.35 * budget_s - elapsed
    
    if time_for_sg > 0.01:
        iters = min(MAX_ITER, int(time_for_sg * 20))  # Roughly 20 iter/sec
        step_size = 0.5 * dx
        
        # Find the worst shift first
        corr = np.correlate(h_start, 1 - h_start, mode='full')
        k_best = np.argmax(corr) - (len(h_start) - 1)
        corr_k = corr[np.argmax(corr)]
        
        # Run for a few iterations or until time runs out
        for i in range(iters):
            if time.time() - start_time > 0.7 * budget_s:
                break
            h_start = subgradient_descent_step(h_start, k_best, step_size)
            new_c5 = evaluate_c5(h_start)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_start.copy()
                
            # Re-evaluate the worst shift
            corr = np.correlate(h_start, 1 - h_start, mode='full')
            k_best = np.argmax(corr) - (len(h_start) - 1)
    
    # Phase 3: Random stochastic refinement
    elapsed = time.time() - start_time
    if elapsed < budget_s - 0.1:
        h_final, c5_final = stochastic_refinement(best_h, max_iter=300)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    
    # Return
    return (best_h.tolist(), float(best_c5), n_points)
