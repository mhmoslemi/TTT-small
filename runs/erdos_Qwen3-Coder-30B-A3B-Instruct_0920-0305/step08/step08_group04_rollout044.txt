import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import time

# Global variables
n_points = 300
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling projection
    sorted_indices = np.argsort(-x)  # descending order
    x_sorted = x[sorted_indices]
    
    # Find the threshold lambda
    cumulative_sum = np.cumsum(x_sorted)
    n = len(x_sorted)
    threshold = (cumulative_sum - target_sum) / np.arange(1, n + 1)
    lambda_val = np.max(np.concatenate(([0], threshold)))
    
    result = np.clip(x - lambda_val, 0, 1)
    return result

def rasterise_and_project(θ):
    """Convert parameters θ to a step function h with correct sum and bounds"""
    # θ is 16-dimensional: 8 widths + 8 heights
    widths_raw = np.exp(np.clip(θ[:8], -15, 15))  # Prevent overflow
    heights_raw = 1 / (1 + np.exp(-θ[8:]))
    
    # Normalise widths
    widths = 2 * widths_raw / np.sum(widths_raw)
    
    # Create the piecewise constant function
    h = np.zeros(n_points)
    cum_widths = np.cumsum(widths)
    cum_widths = np.insert(cum_widths, 0, 0)
    
    for i in range(8):
        start_idx = int(cum_widths[i] / 2 * n_points)
        end_idx = int(cum_widths[i+1] / 2 * n_points)
        h[start_idx:end_idx] = heights_raw[i]
    
    # Project to sum and bounds
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def obj(θ):
    """Objective function for differential evolution"""
    h = rasterise_and_project(θ)
    c5 = evaluate_c5(h)
    return c5
