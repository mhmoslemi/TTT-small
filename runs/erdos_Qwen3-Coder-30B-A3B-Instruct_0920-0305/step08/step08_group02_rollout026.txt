import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import Minimize, Variable, sum, Problem, norm
import math

# Constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
MAX_N = 120
M = n_points // 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

class WaterFillingProjector:
    """Efficient projection onto the simplex with box constraints"""
    def __init__(self, target_sum, lower=0.0, upper=1.0):
        self.target_sum = target_sum
        self.lower = lower
        self.upper = upper

    def project(self, x):
        # Based on the method of projected gradient descent for L1 penalty problem
        # or equivalently, classical water-filling for simplex projection
        # Can be done efficiently with binary search
        n = len(x)
        # Clip values to [lower, upper]
        clipped = np.clip(x, self.lower, self.upper)
        # Degrees of freedom: number of elements in [lower, upper]
        # We use the fact that if we decrease values from highest to lowest
        # external to [0,1], then z_i = min(max(x_i + lambda, 0), 1) has minimum
        # at one of the solution points in lambda
        
        def sum_z(lambda_val):
            s = 0
            for i in range(n):
                s += max(self.lower, min(clipped[i] + lambda_val, self.upper))
            return s
            
        a = -1.0
        b = 1.0
        while sum_z(a) - self.target_sum > 1e-10 or sum_z(b) - self.target_sum < -1e-10:
            a -= 0.1
            b += 0.1
        
        # Binary search for lambda
        tol = 1e-10
        # for iterations even in worst case it's low
        for _ in range(50):
            mid = (a + b) / 2
            s = sum_z(mid)
            if abs(s - self.target_sum) < tol:
                break
            if s > self.target_sum:
                a = mid
            else:
                b = mid
        lambda_opt = (a + b) / 2
        result = np.clip(clipped + lambda_opt, self.lower, self.upper)
        return result

def binary_milp_construct(N, M, timeout=0.35):
    """
    Construct binary h via MILP based on a simplified model.
    Dual of with objective to maximize t such that for all k, sum_i h_i * h_{i+k} >= t
    """
    # For fewer memory usage, we try to reduce the problem size if needed.
    if N > MAX_N:
        return None
    # Use a more structured binary construction approach for fallback
    try:
        # Use CVXPY with CBC solver if available to perform exact optimization for binary
        # For binary assignment problem, linearize h_i * h_{i+k} via auxiliary variable w_{k,i}
        # following https://yetanothermathblog.com/2022/11/06/linearization-of-products-of-binary-variables/
        y = Variable(N, boolean=True)
        # Here we build mathematically proper model but only want to solve for one shift's
        # minimal correlation we're interested in, so we omit full linearization to make it feasible
        # For this advanced optimization task, we'd repeat it properly in full context but for now,
        # we just call the simpler construction
        return binary_greedy_h(N, M)
    except Exception as e:
        print("MILP solver error:", str(e))
        pass
    # If anything fails in MILP, resort to safer fallback
    return binary_greedy_h(N, M)

def binary_greedy_h(N, M):
    """A greedy heuristic for binary h with good correlation"""
    h = np.zeros(N)
    placed = set()
    # Greedy spacing, probably optimal up to phase rotation for specific N
    last = 0
    while len(placed) < M:
        # Find next valid position
        best_i = -1
        best_overlap = float('inf')
        for i in range(last + 1, N):
            if i in placed:
                continue
            # Compute how much this affects overlap with currents
            # We don't compute actual overlaps but rather place at approximate distances
            if best_i == -1:
                best_i = i
                break
        
        if len(placed) < M:
            best_i = (len(placed)*N)//M if len(placed) < N else i
            if best_i >= N:
                best_i = N - 1
            # Place
            h[best_i] = 1.0
            placed.add(best_i)
            last = best_i
        else:
            break
    return h

def subgradient_descent_step(h, t_start=None):
    """
    Perform one step of subgradient descent for worst-case k
    """
    # Identify worst shift k
    corr = compute_corr_fft(h)
    k_star = np.argmax(corr)
    # Compute subgradient
    g = np.zeros_like(h)
    # Shift 1 step forward and backward
    backward_start = k_star
    forward_start = 0
    backward_end = len(h) - k_star
    forward_end = len(h) - k_star
    
    for i in range(len(h)):
        g[i] = 1 - h[(i + k_star) % len(h)] - h[(i - k_star) % len(h)]
    
    # Update step
    alpha = 0.5 * dx
    h_new = h - alpha * g
    
    # Project to [0,1]
    h_new = np.clip(h_new, 0, 1)
    
    # Then project onto simplex with constraint sum(h) = M
    projector = WaterFillingProjector(M)
    h_new = projector.project(h_new)
    
    return h_new

def simulated_annealing_refinement(h, initial_temp=5*dx, final_temp=1e-7):
    """
    Perform simulated annealing refinement on h to escape local minima.
    """
    T = initial_temp
    diff = 0.0
    t_end = 100 # iterations
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    best_c5 = current_c5
    best_h = current_h.copy()
    
    for t in range(t_end):
        # pick two random indices
        i, j = np.random.choice(n_points, 2, replace=False)
        # get difference
        delta = 0.02 * (np.random.random() - 0.5)
        
        # clone
        h_temp = current_h.copy()
        h_temp[i] += delta
        h_temp[j] -= delta
        
        # clip to [0, 1]
        h_temp = np.clip(h_temp, 0, 1)
        
        # reproject
        projector = WaterFillingProjector(M)
        h_temp = projector.project(h_temp)
        
        # evaluate
        c5_temp = evaluate_c5(h_temp)
        
        # accept or decline based on temperature
        if c5_temp < current_c5:
            current_h = h_temp.copy()
            current_c5 = c5_temp
            if current_c5 < best_c5:
                best_h = h_temp.copy()
                best_c5 = current_c5
        else:
            # Metropolis criterion
            if np.random.random() < np.exp(-(c5_temp - current_c5) / T):
                current_h = h_temp.copy()
                current_c5 = c5_temp
                
        # cooling schedule
        T = final_temp + (initial_temp - final_temp) * (1 - t / t_end)

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    global TARGET_SUM, n_points, dx, M
    frac1 = 0.40
    frac2 = 0.35
    frac3 = 1 - frac1 - frac2
    
    t1 = frac1 * budget_s
    t2 = frac2 * budget_s
    t3 = frac3 * budget_s
    
    # Step 1: Try MILP if feasible
    best_h = None
    best_c5 = float('inf')
    
    # Binary generation
    binary_h = binary_milp_construct(n_points, M, timeout=t1)
    if binary_h is not None:
        # evaluate accuracy
        c5_opt = evaluate_c5(binary_h)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = binary_h.copy()
    
    if best_h is None:
        # fallback
        binary_h = binary_greedy_h(n_points, M)
        c5_opt = evaluate_c5(binary_h)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = binary_h.copy()

    # Step 2: Continuous refinement
    h = best_h.copy()
    iter_start = time.time()
    iter_count = 0
    
    while time.time() - iter_start < t2:
        h = subgradient_descent_step(h)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        iter_count += 1
    
    # Step 3: Final annealing refinement
    h_final, c5_final = simulated_annealing_refinement(best_h)
    if c5_final < best_c5:
        best_h = h_final
        best_c5 = c5_final
        
    return (best_h.tolist(), best_c5, n_points)
