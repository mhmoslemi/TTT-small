import numpy as np

dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def step_function_overlap_eval(h_vals, lengths, num_intervals):
    """
    Compute C5 for a piecewise constant h(x) via exact formulas
    """
    # h(x) is piecewise constant with intervals [0,l1), [l1,l1+l2), ... 
    # Values are given by h_vals[0], h_vals[1], ... 
    # Lengths are given by lengths[0], lengths[1], ...
    # At the end, what remains of lengths must be exact to make the total length = 2
    
    a = np.array(h_vals)
    l = np.array(lengths)
    # Total length 
    total_length = np.sum(l)
    # We need to ensure we close exactly to 2 (due to rounding)
    l[-1] = 2 - np.sum(l[:-1])
    
    # Compute the support interval edges
    s = [0.0]
    for i in range(len(l)):
        s.append(s[-1] + l[i])

    # Compute all possible critical shifts for the max
    # Simplification using the fact that the function is piecewise 
    # and only the overlaps matter for the integral
    # The maximum overlap occurs over a few specific points
    critical_shifts = []
    # Critical shifts from interval boundaries
    for i in range(len(s)):
        for j in range(len(s)):
            if s[i] <= s[j] and s[j] <= s[i] + 2:
                critical_shifts.append(s[j] - s[i])
    # Also consider each interval edge shift
    for i in range(len(s)):
        critical_shifts.append(s[i])
    # Remove duplicates and make sure it's within [0,2]
    critical_shifts = sorted(set(c for c in critical_shifts if 0 <= c <= 2)) # [0, 2]

    max_overlap = 0.0

    def get_len_constraint(k):
        # Compute how much of each interval contributes when we shift by k
        total = 0
        for i in range(len(a)):
            start = s[i]
            end = s[i+1]
            overlap_start = max(0, start)
            overlap_end = min(end, 2 - k)
            if overlap_end > overlap_start:
                total += a[i] * (overlap_end - overlap_start)
        return total

    def get_quad_constraint(k):
        # For each pair of intervals, compute overlap at shift k
        total = 0.0
        for i in range(len(a)):
            for j in range(len(a)):
                start_i = s[i]
                end_i = s[i+1] 
                start_j = s[j] 
                end_j = s[j+1]
                overlap_start = max(start_i, start_j + k)
                overlap_end = min(end_i, end_j + k)
                if overlap_start < overlap_end:
                    total += a[i] * a[j] * (overlap_end - overlap_start)
        return total

    # Walk over all critical shifts and find maximum
    for k in critical_shifts:
        linear_term = get_len_constraint(k)
        quad_term = get_quad_constraint(k)
        overlap = linear_term - quad_term
        if overlap > max_overlap:
            max_overlap = overlap

    return max_overlap
