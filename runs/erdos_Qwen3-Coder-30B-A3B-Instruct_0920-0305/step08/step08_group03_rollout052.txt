import numpy as np
import time

# Global constants
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points
MAX_ITER = 1000000
K_EVAL = 500
K_STEP = 100
K_TEMP = 2000
T_INIT = 0.01
ALPHA_TEMP = 0.995
DELTA_INIT = 0.02
DELTA_MIN = 5e-4
ACCEPTANCE_TARGET = 0.3
M_NOIMP = 2000

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda to satisfy sum(clamp(x + lam, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    EPS = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < EPS:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlation(h, b):
    """Compute correlation using FFT"""
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h[::-1], L) * np.fft.fft(b, L)))
    return corr

def update_correlation_incremental(c, h, b, i, j, delta):
    """Update correlation incrementally when transferring delta from i to j"""
    # Compute contribution due to change at i
    # From h[i], b[i] contributes to corr at shifts s where 0 <= i+s < n
    # That is, s between -i and n-1-i. Corresponds to index s + (n-1) = shift + (n-1) 
    # So index starts at -i + (n-1) = n-1-i, and ends at n-1 + (n-1) = 2*n-2
    if i < (n_points - 1):
        s1_start = max(0, i + 1 - (n_points - 1))
        c[s1_start : s1_start + (n_points - 1 - i)] -= delta * h[:n_points - 1 - i] 
    if i > 0:
        s2_start = max(0, (n_points - 1) - i)
        c[s2_start : s2_start + i] += delta * b[1 : 1 + i] 

    # Contribution due to b changing (loss of h[i])
    if i < (n_points - 1):
        offset = (n_points - 1) - i
        end = min(offset + (n_points - 1), len(c))
        c[offset : end] -= delta * h[:end - offset]
    if i > 0:
        offset = i + (n_points - 1)
        c[:(n_points - 1)] -= delta * b[1 : 1 + (n_points - 1 - i)] 

    # Key correction: if there was a non-zero contribution at indices
    # where both i and j took part (this only when shift is 0 or ±(j-i))
    # We capture the term +delta**2 at shifts (j - i) and (i - j)
    idx_diff = (j - i) + (n_points - 1)
    if 0 <= idx_diff < len(c):
        c[idx_diff] += delta * delta
    idx_neg_diff = (i - j) + (n_points - 1)
    if 0 <= idx_neg_diff < len(c):
        c[idx_neg_diff] += delta * delta

    return c

    b = 1 - h
    c = compute_correlation(h, b)
    Cmax = np.max(c)
    best_value = dx * Cmax
    best_h = h.copy()

    # Parameters
    T = T_INIT
    delta = DELTA_INIT
    last_improvement = 0
    acc_count = 0
    acc_sum = 0.0
    i_iter = 0

    step_count = 0
    
    # Main optimization loop
    while i_iter < MAX_ITER and time.time() - start_time < budget_s:
        i_iter += 1
        if i_iter % K_EVAL == 0:
            # Re-compute c from scratch every K_EVAL iters to prevent drift
            c = compute_correlation(h, b)
            Cmax = np.max(c)
            current_value = dx * Cmax
            if current_value < best_value:
                best_value = current_value
                best_h = h.copy()
        if i_iter % K_STEP == 0 and i_iter > 0:
            if delta > DELTA_MIN:
                delta *= 0.9
                delta = max(delta, DELTA_MIN)
        if i_iter % K_TEMP == 0 and i_iter > 0:
            T *= ALPHA_TEMP
            # Reset if no improvement for a long time
            if i_iter - last_improvement > M_NOIMP:
                T = T_INIT
                last_improvement = i_iter

        # Pick two random indices
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        if i == j:
            continue
            
        # Compute max allowed delta
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta < 1e-12:
            continue
        delta_prop = min(delta, 0.9 * max_delta)
        if delta_prop < 1e-12:
            continue
            
        # Store current state for rollback
        h_old = h.copy()
        b_old = b.copy()
        c_old = c.copy()
        Cmax_old = Cmax
        
        # Perform the transfer
        h[i] -= delta_prop
        h[j] += delta_prop
        b[i] = 1.0 - h[i]
        b[j] = 1.0 - h[j]
        
        # Project to feasible set
        h = project_to_sum_box(h, TARGET_SUM)
        b = 1 - h
        
        # Rebuild c from scratch for now (until update is solid)
        # Notice: Later, we may implement the incrementally efficient update here
        # But for robustness we recompute until fully tested.
        c = compute_correlation(h, b)
        Cmax = np.max(c)
        current_value = dx * Cmax
        obj_change = current_value - best_value
        
        # Accept based on Metropolis rule
        if current_value < best_value:
            best_value = current_value
            best_h = h.copy()
            last_improvement = i_iter
            acc_sum += 1.0
        else:
            p_accept = np.exp(-obj_change / T)
            if np.random.rand() < p_accept:
                acc_sum += 1.0
                last_improvement = i_iter
            else:
                # Rollback
                h[:] = h_old
                b[:] = b_old
                c[:] = c_old
                Cmax = Cmax_old
                current_value = dx * Cmax

        # Update acceptance rate
        acc_count += 1
        if i_iter % 1000 == 0:
            if acc_count > 0:
                acc_rate = acc_sum / acc_count
                # Adjust delta based on acceptance rate if needed (optional)
                if acc_rate > ACCEPTANCE_TARGET * 1.2 and delta < 1.0:
                    delta *= 1.05
                elif acc_rate < ACCEPTANCE_TARGET * 0.8 and delta > DELTA_MIN:
                    delta *= 0.95
        
        step_count += 1

    # Final verification
    c = compute_correlation(best_h, 1 - best_h)
    final_Cmax = np.max(c)
    final_value = dx * final_Cmax
    if final_value < best_value:
        best_value = final_value
        best_h = best_h.copy()
    
    return (best_h.tolist(), float(best_value), n_points)
