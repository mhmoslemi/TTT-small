import numpy as np

from scipy.optimize import differential_evolution
import time

# Global variables
TARGET_SUM = None
n_points = 300
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def piecewise_const_from_params(params, n_points):
    """Convert parameters to piecewise constant function"""
    K = 8
    log_widths = params[:K]
    heights_raw = params[K:]
    
    # Compute widths using softmax
    widths = np.exp(log_widths)
    widths = widths / np.sum(widths) * 2.0  # Scale to total width of 2
    
    # Compute heights using sigmoid
    heights = 1 / (1 + np.exp(-heights_raw))
    
    # Clip heights to [0,1]
    heights = np.clip(heights, 0, 1)
    
    # Create piecewise constant function
    h = np.zeros(n_points)
    bin_edges = np.concatenate([[0], np.cumsum(widths)])
    
    # Assign heights to bins
    positions = np.linspace(0, 2, n_points, endpoint=False) + dx/2
    bin_indices = np.digitize(positions, bin_edges) - 1
    bin_indices = np.clip(bin_indices, 0, K-1)
    
    for i in range(K):
        mask = bin_indices == i
        h[mask] = heights[i]
    
    return h

def de_objective(params):
    """Objective function for differential evolution"""
    h = piecewise_const_from_params(params, n_points)
    h = project_to_sum_box(h, TARGET_SUM)
    c5 = evaluate_c5(h)
    return c5

def compute_corr(h):
    """Compute correlation array for gradient computation"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return corr

                delta[:n_points-shift] += current_h[shift:]
                # Backward neighbor contribution
                delta[shift:] += current_h[:n_points-shift]
            else:

                shift_abs = -shift
                delta[shift_abs:] += current_h[:n_points-shift_abs]
                delta[:n_points-shift_abs] += current_h[shift_abs:]
        
        # Subtract 1 for each position, accounting for all active shifts
        delta -= len(active_indices)
        
        # Step in negative gradient direction
        new_h = current_h + step_size * delta

        # Gradually reduce step size if no improvement for many steps

            step_size *= 0.995
    
    return (best_h.tolist(), float(best_c5), n_points)
