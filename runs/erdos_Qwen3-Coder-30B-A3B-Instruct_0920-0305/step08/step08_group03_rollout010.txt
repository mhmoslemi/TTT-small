import numpy as np

import time
import random

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def update_correlation(c, h, i, delta):
    """Incrementally update correlation when moving delta mass from i to j"""
    n = len(h)
    # When changing h[i] by delta, update c
    # For b = 1 - h, we also change b[i] by -delta
    
    # Contributions from b[i]
    # For each shift s where i+s is valid (0 <= i+s <= n-1)
    # That is s from -i to n-1-i
    s_start = max(-i, -(n-1))
    s_end = min(n-1-i, n-1)
    for s in range(s_start, s_end+1):
        if 0 <= i+s <= n-1:
            c[s + n - 1] += delta * (1 - h[i+s])
    
    # Contributions from a[i] (which is h[i])
    # For each shift s where i-s is valid (0 <= i-s <= n-1)
    # That is s from 0 to i
    s_start = 0
    s_end = i
    for s in range(s_start, s_end+1):
        if 0 <= i-s <= n-1:
            c[s + n - 1] -= delta * h[i-s]
