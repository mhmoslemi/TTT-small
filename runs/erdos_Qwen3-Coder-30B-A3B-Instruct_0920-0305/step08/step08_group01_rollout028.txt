import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_search_improve(h, num_iter=1000):
    """Refine solution using neighborhood search with adaptive step sizes"""
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    
    # Adaptive step sizes
    step_size = 0.02
    no_improve_count = 0
    
    for iteration in range(num_iter):
        # Random neighborhood change
        idx = random.randint(0, n_points - 1)
        delta = random.gauss(0, step_size)
        
        # Try to modify this element
        new_h = current_h.copy()
        new_h[idx] += delta
        
        if new_h[idx] < 0:
            new_h[idx] = 0
        elif new_h[idx] > 1:
            new_h[idx] = 1
            
        new_h = project_to_sum_box(new_h, TARGET_SUM)
        new_c5 = evaluate_c5(new_h)
        
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Reduce step size if stuck
        if no_improve_count > 50:
            step_size *= 0.9
            no_improve_count = 0
            
    return current_h, current_c5

def pattern_based_refine(h):
    """Apply pattern-based refinement that attempts to suppress overlaps systematically"""
    h_refined = h.copy()
    
    # Identify regions with high values and try to "flatten" them
    # while adding suppression in the overlap regions
    for i in range(0, n_points, 10):
        # Target a window
        start = max(0, i - 5)
        end = min(n_points, i + 5)
        
        # If this region has a peak
        if np.max(h_refined[start:end]) >= 0.8:
            # Temporarily flatten the peak in that region
            peak_idx = np.argmax(h_refined[start:end])
            peak_idx += start
            
            # Reduce the peak
            h_refined[peak_idx] *= 0.8
            
            # Try to add some suppression near the peak

            for offset in [-1, 1, -2, 2]:
                idx = peak_idx + offset
                if 0 <= idx < n_points:

    # Ensure correct sum
    h_refined = project_to_sum_box(h_refined, TARGET_SUM)
    return h_refined
