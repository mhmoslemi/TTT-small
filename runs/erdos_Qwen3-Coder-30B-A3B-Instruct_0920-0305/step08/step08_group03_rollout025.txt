import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = 1.0
n_points = 120
dx = 2.0 / n_points

def compute_overlap_integral(h, k, intervals, endpoints):
    """Compute overlap integral for a fixed shift k analytically"""
    # Find the integral of h over [0, 2-k]
    linear_integral = 0.0
    # Find all intervals overlapping [0, 2-k]
    total_length = 2.0 - k
    pos = 0.0
    for i, (start, end, height) in enumerate(intervals):
        if pos >= total_length:
            break
        # overlap between [pos, pos + (end-start)] and [0, total_length]
        overlap_start = max(pos, 0.0)
        overlap_end = min(pos + (end - start), total_length)
        if overlap_end > overlap_start:
            linear_integral += height * (overlap_end - overlap_start)
        pos += (end - start)
    
    # Compute the product integral
    product_integral = 0.0
    pos1 = 0.0
    for i, (start1, end1, height1) in enumerate(intervals):
        pos2 = 0.0
        for j, (start2, end2, height2) in enumerate(intervals):
            overlap_start = max(pos1, pos2 + k)
            overlap_end = min(pos1 + (end1 - start1), pos2 + (end2 - start2))
            if overlap_end > overlap_start:
                overlap_length = overlap_end - overlap_start
                product_integral += height1 * height2 * overlap_length
            pos2 += (end2 - start2)
        pos1 += (end1 - start1)
    
    return linear_integral - product_integral

def compute_critical_shifts(endpoints):
    """Generate all critical shift values"""
    shift_set = set()
    # Add base shifts and mirrored ones
    for i in range(len(endpoints)):
        for j in range(len(endpoints)):
            shift = abs(endpoints[i] - endpoints[j])
            if shift <= 2.0:
                shift_set.add(shift)
            # Handle mirrored sides: at bounds
            if endpoints[i] <= 2.0:
                shift_set.add(2.0 - endpoints[i])
            if endpoints[j] <= 2.0:
                shift_set.add(2.0 - endpoints[j])
    return sorted(shift_set)

def compute_c5_from_params(lengths, heights, critical_shifts):
    """Compute max overlap for given piecewise function parameters"""
    # Reconstruct intervals and endpoints
    endpoints = [0.0]
    for l in lengths:
        endpoints.append(endpoints[-1] + l)
    
    intervals = []
    for i in range(len(lengths)):
        start = endpoints[i]
        end = endpoints[i+1]
        height = heights[i]
        intervals.append((start, end, height))
    
    all_overlaps = []
    for k in critical_shifts:
        j_val = compute_overlap_integral(None, k, intervals, endpoints)
        all_overlaps.append(j_val)
    
    return max(all_overlaps)

def construct_initial_solution():
    """Construct a reasonable initial solution using known patterns"""
    # Start with simplest pattern, e.g. a 5-step structure
    # Like: [0, 0.4] at height 1, [0.4, 1.0] at 0.5, [1.0, 1.6] at 1, [1.6, 2.0] at 0
    
    # Split [0,2] into 5 segments
    lengths = [0.4, 0.6, 0.6, 0.4, 0.0]
    heights = [1.0, 0.5, 1.0, 0.0, 0.0]
    
    # Normalize to sum 1 (multiply by correct factor to satisfy mass constraint)
    total_mass = sum(h * l for h, l in zip(heights, lengths))
    factor = TARGET_SUM / total_mass
    heights = [h * factor for h in heights]
    
    # Ensure lengths sum to 2
    lengths = np.array(lengths)
    lengths = lengths * 2.0 / np.sum(lengths)
    
    return lengths.tolist(), heights

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = n_points
    
    # Define m (number of intervals)
    m = 5
    
    # Generate list of critical shifts
    base_endpoints = np.linspace(0, 2, m+1)
    critical_shifts = compute_critical_shifts(base_endpoints)

    # Try different initial configurations
    initial_configs = []
    
    # Configuration 1: Binary-like structure: 1,0,1,0,1
    init_h = [1.0, 0.0, 1.0, 0.0, 1.0]
    init_l = [0.4, 0.4, 0.4, 0.4, 0.4] # if they were evenly spaced
    # Normalize lengths to sum to 2
    norms = sum(init_l)
    init_l = [x * 2 / norms for x in init_l]
    # Adjust heights to enforce mass constraint
    sum_mass = sum(h * l for h, l in zip(init_h, init_l))
    scale = TARGET_SUM / sum_mass
    init_h = [h * scale for h in init_h]
    initial_configs.append((init_l, init_h))
    
    # Configuration 2: 0,1,1,0 sequence
    init_h2 = [0.0, 1.0, 1.0, 0.0, 0.0]
    init_l2 = [0.6, 0.6, 0.6, 0.6, 0.6]
    norms2 = sum(init_l2)
    init_l2 = [x * 2 / norms2 for x in init_l2]
    sum_mass2 = sum(h * l for h, l in zip(init_h2, init_l2))
    scale2 = TARGET_SUM / sum_mass2
    init_h2 = [h * scale2 for h in init_h2]
    initial_configs.append((init_l2, init_h2))

    # Optimize using gradient-free optimizers and grid search
    from scipy.optimize import differential_evolution
    from scipy.optimize import minimize
    import itertools

    def expand_grid(directions, levels_per_dim):
        # Uniform sampling over simplex using barycentric subdivision
        grid_points = []
        # Determine number of points in each dir
        total_points = levels_per_dim ** (len(directions) - 1) # simplex dims are len-1
        for _ in range(1000): # Sample default
            weights = np.random.uniform(0, 1, len(directions))
            weights /= weights.sum()
            weights = weights[:-1]  # Last is inferred for simplex
            grid_points.append(weights)
        return grid_points[:1000]
    
    # Reconstruct h from results later
    def reconstruct_h_func(lengths, heights):
        endpoints = [0.0]
        for l in lengths:
            endpoints.append(endpoints[-1] + l)
        h_recon = []
        for i in range(len(lengths)):
            h_val = heights[i]
            start = endpoints[i]
            end = endpoints[i+1]
            num_points = max(1, int((end - start) / (2.0 / n_points)))
            h_recon.extend([h_val] * num_points)
        return h_recon

    # Defer optimization until later to avoid repeated setup
    
    def objective(params):
        lengths = list(params[:m - 1])
        last_length = 2.0 - sum(lengths)
        lengths.append(last_length)
        
        heights = list(params[m-1:])
        last_height = (TARGET_SUM - sum(h * l for h, l in zip(heights[:-1], lengths[:-1]))) / lengths[-1] if lengths[-1] > 0 else 0.0
        heights.append(last_height)
        
        # Ensure bounds
        for i in range(len(lengths)):
            if lengths[i] < 1e-10:
                lengths[i] = 1e-10
        for i in range(len(heights)):
            if heights[i] < 0: heights[i] = 0
            elif heights[i] > 1: heights[i] = 1
        
        return compute_c5_from_params(lengths, heights, critical_shifts)

    # Use DE + local refinement
    bounds = [(1e-6, 2.0)] * (m - 1) + [(0, 1.0)] * m
    best_res = None
    best_val = float('inf')
    
    for n_restart in range(10):
        # Fom a new starting point
        init = np.array([np.random.uniform(*b) for b in bounds])
        
        try:
            res = differential_evolution(objective, bounds, maxiter=1000, popsize=15, seed=seed+n_restart)
            if res.fun < best_val:
                best_val = res.fun
                best_res = res
                print(f"DE improvement: {best_val:.6f}")
        except:
            continue
    
        # Try local refinement from DE solution
        if best_res is not None:
            try:
                x0 = np.array(best_res.x)
                res2 = minimize(objective, x0, method='Nelder-Mead', options={'maxiter': 1000, 'xtol': 1e-8, 'ftol': 1e-8})
                if res2.fun < best_val:
                    best_val = res2.fun
                    print(f"Local refinement improvement: {best_val:.6f}")
            except:
                continue
    
    # Convert final to full step function representation
    if best_res is not None:
        lengths = list(best_res.x[:m - 1])
        last_length = 2.0 - sum(lengths)
        lengths.append(last_length)
        
        heights = list(best_res.x[m-1:])
        last_height = (TARGET_SUM - sum(h * l for h, l in zip(heights[:-1], lengths[:-1]))) / lengths[-1] if lengths[-1] > 0 else 0.0
        heights.append(last_height)
        
        # Clamp values
        lengths = np.clip(lengths, 1e-6, 2.0)
        heights = np.clip(heights, 0, 1.0)
        
        # Ensure sum constraints
        total_mass = sum(h * l for h, l in zip(heights, lengths))
        scale_factor = TARGET_SUM / total_mass
        heights = [h * scale_factor for h in heights]
        
        # Reconstruct vector
        h_final = reconstruct_h_func(lengths, heights)
        
        # Final check
        h_array = np.array(h_final)
        c5_final = np.max(np.correlate(h_array, 1 - h_array, mode='full')) * dx
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
            
    # Perform quick local search if there's still a better option
    if best_h is None:
        lengths, heights = construct_initial_solution()
        init_length = lengths
        init_heights = heights

        h_dump = reconstruct_h_func(init_length, init_heights)
        c5_initial = np.max(np.correlate(np.array(h_dump), 1 - np.array(h_dump), mode='full')) * dx

        # Do simple grid search + local
        best_c5 = c5_initial
        best_h = h_dump

    # Run a short refinement pass (incremental)
    if best_h is not None and time.time() - start_time < budget_s - 10:
        # Use SLSQP or similar refinement on best found
        h_array = np.array(best_h)
        
        def eval_c5(h):
            return np.max(np.correlate(h, 1 - h, mode='full')) * dx

        def constraint_sum(h):
            return np.sum(h) * dx - TARGET_SUM

        bounds = [(0, 1)] * len(h_array)
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(eval_c5, h_array, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8})
            if res.success and eval_c5(res.x) < best_c5:
                best_h = res.x.tolist()
                best_c5 = eval_c5(res.x)
        except:
            pass
    
    return (best_h if best_h else [0.5]*n_points, float(best_c5), n_points)
