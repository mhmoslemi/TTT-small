import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import time

# Global variables
TARGET_SUM = None
n_points = 300
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def piecewise_const_from_params(params, n_points):
    """Convert parameters to piecewise constant function"""
    K = 8
    widths = np.exp(params[:K])
    widths = widths * 2.0 / np.sum(widths)
    
    heights = 1 / (1 + np.exp(-params[K:]))
    heights = np.clip(heights, 0, 1)
    
    # Compute cumulative boundaries
    boundaries = np.concatenate([[0], np.cumsum(widths)])
    
    # Create positions
    positions = np.linspace(0, 2, n_points, endpoint=False) + dx/2
    
    # Determine which block each position belongs to
    block_indices = np.searchsorted(boundaries, positions, side='right') - 1
    block_indices = np.clip(block_indices, 0, K-1)
    
    # Create h vector
    h = heights[block_indices]
    
    return h

def de_objective(params):
    """Objective function for differential evolution"""
    h = piecewise_const_from_params(params, n_points)
    h = project_to_sum_box(h, TARGET_SUM)
    c5 = evaluate_c5(h)
    return c5

                best_c5 = c5_candidate
                best_h = h_candidate.copy()

    current_h = best_h.copy()
    
    # Time limit for local refinement
    local_time_limit = 500

    while time.time() - start_time < budget_s - 10:
        if time.time() - start_time > local_time_limit:
            break

        active_lags = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        
        # Compute gradient step
        delta = np.zeros(n_points)
        step_size = 0.05
        
        for lag in active_lags:
            shift = lag - (n_points - 1)
            
            if shift >= 0:
                # Forward neighbor
                delta[:-shift] += current_h[shift:]
                # Backward neighbor
                delta[shift:] += current_h[:-shift]
            else:
                s_abs = -shift
                # Forward neighbor (i + shift = i - s_abs)
                delta[s_abs:] += current_h[:-s_abs]
                # Backward neighbor (i - shift = i + s_abs)
                delta[:-s_abs] += current_h[s_abs:]
        
        # Subtract constant term for all positions (since we're computing sum of neighbors)
        delta -= len(active_lags)
        
        # Apply gradient descent step
        new_h = current_h + step_size * delta
        
        # Project onto constraints
        new_h = np.clip(new_h, 0, 1)
        new_h = project_to_sum_box(new_h, TARGET_SUM)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(new_h)
        
        # Accept if improvement

            best_h = new_h.copy()
            current_h = new_h.copy()
        elif new_c5 < evaluate_c5(current_h):
            current_h = new_h.copy()
        
        iteration += 1
    
    # Final check
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
