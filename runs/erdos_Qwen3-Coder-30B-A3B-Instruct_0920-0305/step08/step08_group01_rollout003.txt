import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from cvxpy import *

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Warm start from initial_h_values
    h_ref = np.array(initial_h_values)
    x_ref = np.linspace(0, 2, len(h_ref))
    x_new = np.linspace(0, 2, n_points)
    h0 = np.interp(x_new, x_ref, h_ref)
    
    # Add some random perturbation
    h0 = h0 + np.random.normal(0, 0.01, n_points)
    h0 = np.clip(h0, 0, 1)
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Phase 1: Multiple random starts
    for run_id in range(50):
        if time.time() > start_time + 0.6 * budget_s:
            break
            
        # Different initialization strategies
        strategies = [
            lambda: np.random.uniform(0, 1, n_points),
            lambda: np.random.choice([0, 1], size=n_points, p=[0.35, 0.65]),
            lambda: np.clip(np.random.normal(0.5, 0.15, n_points), 0, 1),
            lambda: np.clip(np.random.normal(0.7, 0.2, n_points), 0, 1),
            lambda: np.clip(np.random.normal(0.3, 0.2, n_points), 0, 1),
        ]
        h_start = strategies[run_id % len(strategies)]()
        h_start = project_to_sum_box(h_start, TARGET_SUM)
        
        # Local optimization
        try:
            # Simple step method
            h_test = h_start.copy()
            for _ in range(500):
                idx = np.random.randint(0, n_points)
                old_val = h_test[idx]
                delta = np.random.uniform(-0.02, 0.02)
                h_test[idx] += delta
                h_test[idx] = np.clip(h_test[idx], 0, 1)
                h_test = project_to_sum_box(h_test, TARGET_SUM)
                new_c5 = evaluate_c5(h_test)
                if new_c5 < best_c5:
                    best_h = h_test.copy()
                    best_c5 = new_c5
                else:
                    # Revert
                    h_test[idx] = old_val
        except:
            pass

    # Phase 2: Fine-tune the best solution
    h_final = best_h.copy()
    for _ in range(1000):
        if time.time() > start_time + budget_s:
            break
        idx = np.random.randint(0, n_points)
        old_val = h_final[idx]
        steps = [-0.01, -0.005, 0.005, 0.01]
        best_delta = 0.0
        best_val = evaluate_c5(h_final)
        for delta in steps:
            h_test = h_final.copy()
            h_test[idx] += delta
            if h_test[idx] < 0 or h_test[idx] > 1:
                continue
            h_test = project_to_sum_box(h_test, TARGET_SUM)
            new_c5 = evaluate_c5(h_test)
            if new_c5 < best_val:
                best_val = new_c5
                best_delta = delta
        if best_delta != 0.0:
            h_final[idx] += best_delta
            h_final = project_to_sum_box(h_final, TARGET_SUM)
            best_c5 = best_val

    return (best_h.tolist(), float(best_c5), n_points)
