        h_rev = h[::-1]

        L = 2 * len(h) - 1
        corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
        return corr

            return np.sum(projected) - target_sum

            a -= 0.1

                break

            else:
                a = m

        result = np.clip(x + (a + b)/2, 0, 1)

                                # y[i,s] >= x[i] + x[i+s] - 1
                                y_constraints.append(y_vars[(i, s_max)] >= x[i] + x[i + s_max] - 1)
