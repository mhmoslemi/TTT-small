import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import time

# Global variables
n_points = 300
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling projection for sum constraint with bounds [0,1]
    sorted_x = np.sort(x)[::-1]
    cumsum = np.cumsum(sorted_x)
    # Find the threshold lambda
    n_exceed = np.sum(cumsum > target_sum)
    if n_exceed > 0:
        lambda_val = (cumsum[n_exceed-1] - target_sum) / n_exceed
        projected = np.clip(x - lambda_val, 0, 1)
    else:
        lambda_val = (cumsum[-1] - target_sum) / len(x)
        projected = np.clip(x - lambda_val, 0, 1)
    return projected

def rasterise_and_project(params):
    """Convert box parameters to a h vector and project to sum constraint"""
    # Parameters are: [w1, w2, ..., wK, a1, a2, ..., aK] where K=8
    K = 8
    widths = params[:K]
    heights = params[K:]
    
    # Enforce positivity and normalize widths to total length 2
    u = np.exp(np.clip(widths, -15, 15))  # Clamp to avoid overflow
    w = 2 * u / np.sum(u)  # Normalized widths
    
    # Convert heights to [0,1] via sigmoid
    a = 1 / (1 + np.exp(-heights))
    
    # Create h vector by rasterizing blocks
    h = np.zeros(n_points)
    cum_widths = np.concatenate([[0], np.cumsum(w)])
    for i in range(K):
        start_idx = int(cum_widths[i] / dx)
        end_idx = int(cum_widths[i+1] / dx)
        h[start_idx:end_idx] = a[i]
    
    # Apply the exact projection
    return project_to_sum_box(h, TARGET_SUM)

def obj(θ):
    """Objective function for Differential Evolution"""
    h = rasterise_and_project(θ)
    c5 = evaluate_c5(h)
    return c5

def compute_subgradient(h, corr, active_shifts, n):
    """Compute subgradient of C5 at given h"""
    g = np.zeros(n)
    n_active = len(active_shifts)
    
    for s in active_shifts:
        if s >= 0:
            # Add contribution from h_{i+s}
            g[:n-s] += -dx * h[s:]
            # Add contribution from h_{i-s} 
            g[s:] += -dx * h[:n-s]
        else:
            t = -s
            # Add contribution from h_{i+s}
            g[t:] += -dx * h[:n-t]
            # Add contribution from h_{i-s}
            g[:n-t] += -dx * h[t:]
    
    g /= n_active
    return -g  # negative gradient for descent

def psgd_refinement(h_init, max_iter=5000, time_budget=350):
    """Projected sub-gradient descent with mass-transfer moves"""
    h = h_init.copy()
    c5_best = evaluate_c5(h)
    h_best = h.copy()
    
    # Time tracking
    start_time = time.time()
    
    # Initial correlation computation
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    max_val = corr.max()
    active = np.where(corr >= max_val - 1e-12)[0]
    active_shifts = active - (n_points - 1)
    
    # Step size control
    eta = 0.5
    eta_min = 1e-4
    eta_max = 2.0
    no_improvement_count = 0
    T_swap = 30
    N_noimp = 200
    
    for i in range(max_iter):
        if time.time() - start_time > time_budget:
            break
            
        # Compute subgradient
        g = compute_subgradient(h, corr, active_shifts, n_points)
        
        # Descent step
        h_candidate = h - eta * g
        h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
        c5_candidate = evaluate_c5(h_candidate)
        
        if c5_candidate < c5_best:
            h = h_candidate
            c5_best = c5_candidate
            h_best = h.copy()
            no_improvement_count = 0
            # Increase step size
            eta = min(eta * 1.05, eta_max)
        else:
            no_improvement_count += 1
            # Decrease step size
            eta = max(eta * 0.7, eta_min)
        
        # Update correlation and active set
        h_rev = h[::-1]
        b = 1 - h
        corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
        max_val = corr.max()
        active = np.where(corr >= max_val - 1e-12)[0]
        active_shifts = active - (n_points - 1)
        
        # Mass transfer swaps
        if i % T_swap == 0 or no_improvement_count > N_noimp:
            i_plus = np.random.choice(n_points)
            i_minus = np.random.choice(n_points)
            if h[i_plus] < 1 and h[i_minus] > 0:
                delta = min(0.02, h[i_minus]) * np.random.rand()
                h[i_plus] += delta
                h[i_minus] -= delta
                h = project_to_sum_box(h, TARGET_SUM)
                # Recompute correlation after swap
                h_rev = h[::-1]
                b = 1 - h
                corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
                max_val = corr.max()
                active = np.where(corr >= max_val - 1e-12)[0]
                active_shifts = active - (n_points - 1)
        
    return h_best, c5_best
