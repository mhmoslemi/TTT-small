import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import math

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def random_binary_solution():
    """Generate a random binary solution with exactly TARGET_SUM ones"""
    h = np.zeros(n_points, dtype=float)
    ones_indices = np.random.choice(n_points, size=int(TARGET_SUM), replace=False)
    h[ones_indices] = 1.0
    return h

def simulated_annealing_binary():
    """Run simulated annealing on binary patterns"""
    # Initialize
    h = random_binary_solution()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Annealing parameters
    initial_temp = 0.015
    final_temp = 0.0001
    max_iter = 200000
    
    start_time = time.time()
    elapsed = 0
    
    # Keep track of recent improvements
    last_improvement = 0
    no_improvement_count = 0
    
    # Get initial indices
    ones_indices = np.where(h > 0.5)[0]
    zeros_indices = np.where(h < 0.5)[0]
    
    # Main loop
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > 550:  # Leave 50 seconds for final optimization
            break
            
        # Temperature schedule (exponential decay)
        temp = initial_temp * math.pow(final_temp / initial_temp, elapsed / 550)
        
        # Sample i from ones and j from zeros
        i_idx = np.random.choice(len(ones_indices))
        j_idx = np.random.choice(len(zeros_indices))
        
        i_bit = ones_indices[i_idx]
        j_bit = zeros_indices[j_idx]
        
        # Create new solution by swapping bits
        h_new = h.copy()
        h_new[i_bit] = 0.0
        h_new[j_bit] = 1.0
        
        # Recompute correlation and C5
        new_c5 = evaluate_c5(h_new)
        
        # Accept or reject the new solution
        if new_c5 < best_c5:
            # Update best
            best_h = h_new.copy()
            best_c5 = new_c5
            last_improvement = i
            no_improvement_count = 0
        else:
            # Accept with probability
            diff = best_c5 - new_c5  # positive difference
            if np.random.rand() < np.exp(diff / temp):
                h = h_new
                ones_indices = np.where(h > 0.5)[0]
                zeros_indices = np.where(h < 0.5)[0]
                no_improvement_count += 1
            else:
                no_improvement_count += 1
                
        # Update indices for next iteration
        if i_bit in ones_indices:
            # Already updated
            pass
        else:
            ones_indices = np.where(h > 0.5)[0]
            
        if j_bit in zeros_indices:
            # Already updated
            pass
        else:
            zeros_indices = np.where(h < 0.5)[0]
            
        # Occasionally restart if no improvement for long
        if no_improvement_count > 10000 and i > 20000:
            h = random_binary_solution()
            h_new = h.copy()
            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                last_improvement = i
                no_improvement_count = 0

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    
    # Parameters
    TARGET_SUM = n_points // 2
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Run multiple restarts to find better solutions
    restarts = 8
    for restart in range(restarts):
        if time.time() - start_time > budget_s - 10:  # Leave time for final refinement
            break
            
        # Get binary solution from simulated annealing
        h, c5 = simulated_annealing_binary()
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Final local refinement - gradient-based approach
    if best_h is not None:
        # Clip to [0,1] bounds
        best_h = np.clip(best_h, 0, 1)
        
        # Small refinement step
        def objective(h):
            return evaluate_c5(h)
        
        # Use the best solution as starting point for gradient computation
        refined_h = best_h.copy()
        initial_c5 = evaluate_c5(refined_h)
        
        # Run a simple gradient-based local search
        for _ in range(2000):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Small random perturbations
            perturb_idx = np.random.choice(n_points, size=min(5, n_points//5), replace=False)
            h_test = refined_h.copy()
            
            for idx in perturb_idx:
                # Add small random change respecting bounds
                delta = np.random.uniform(-0.05, 0.05)
                new_val = h_test[idx] + delta
                h_test[idx] = np.clip(new_val, 0, 1)
                
            new_c5 = evaluate_c5(h_test)
            
            if new_c5 < initial_c5:
                refined_h = h_test
                initial_c5 = new_c5
        
        best_h = refined_h
        best_c5 = evaluate_c5(best_h)
            
    return (best_h.tolist(), float(best_c5), n_points)
