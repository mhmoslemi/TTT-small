```python
import numpy as np
from scipy.optimize import differential_evolution
import time

# The discrete version of h as piecewise constant
# With m intervals

# Global constants
m = 5  # intervals, better to use 5 for low-dim
TARGET_SUM = 1.0  # total mass, 1.0
dx = 2.0 / 120  # sample width for evaluation

# We actually optimize over the m-1 lengths and m-1 heights
# that specify the piecewise constant function, i.e., 2*(m-1) = 8 variables.
# The last height and last length are derived from the constraints.

def precompute_intervals_and_shifts():
    """Precomputes critical shifts and interval boundaries for efficiency."""
    critical_shifts = []
    for i in range(1000):
        k = i * 0.002
        critical_shifts.append(k)
    return np.array(critical_shifts)

def compute_overlap_for_shift(h_vals, k):
    """Exact computation for overlap integral J(k) = ∫₀^{2‑k} h(x)·(1‑h(x+k)) dx."""
    # construct h step-wise using the interval lengths and heights
    # h_vals is [l_1, ..., l_{m-1}, a_1, ..., a_{m-1}] 
    l = h_vals[:m-1]
    a = h_vals[m-1:]
    l_sum = 2.0 - np.sum(l)
    a_m = (1.0 - np.sum(a * l)) / l_sum
    a_full = np.append(a, a_m)
    l_full = np.append(l, l_sum)
    
    # compute step locations
    s = np.insert(np.cumsum(l_full), 0, 0)

    # Compute overlap of intervals I_p with I_q shifted by k
    # dist between intervals (indexing by p,q)
    total = 0.0
    # Integration interval [0, 2-k]
    for i, interval in enumerate(l_full):
        s_prev = s[i]
        s_curr = s[i+1]
        if s_curr <= k:
            continue
        # interval intersects [0, 2-k] at [max(0, s_prev), min(2-k, s_curr)]
        start = max(0, s_prev)
        end = min(2-k, s_curr)
        if start >= end:
            continue
        # geometry of overlap with shifted interval
        width = end - start
        height = a_full[i]
        total += height * width - height * height * width
    return total

def compute_all_overlaps(h_vals, shift_points):
    """Compute the overlap integral for many shifts."""
    # Use just a subset of shifts for speed in evaluation phase
    overlaps = []
    for k in shift_points:
        overlaps.append(compute_overlap_for_shift(h_vals, k))
    return np.array(overlaps)

def compute_c5_fast(h_vals):
    """Fast computation using precomputed critical shifts."""
    # For fast evaluation of C5, we calculate a set of sample shifts
    shift_points = precompute_intervals_and_shifts()
    overlaps = compute_all_overlaps(h_vals, shift_points)
    return np.max(overlaps)

def goal_function(h_vals):
    """Objective function to minimize."""
    return compute_c5_fast(h_vals)

def objective_with_scaling(h_vals):
    """Wrapper for scipy.optimize to properly scale the variables."""
    # h_vals is [l1, ..., l_{m-1}, a1, ..., a_{m-1}]
    # All must be in [0, inf) and sum to 2 and 1 respectively
    return goal_function(h_vals)

def project_to_box_constraints(h_vals):
    """Project h_vals to maintain bounds on elements."""
    l = h_vals[:m-1]
    a = h_vals[m-1:]
    # Force l components to be nonnegative
    l = np.maximum(l, 0.0)
    # adjust last l component
    l_sum = np.sum(l)
    l_m = max(0.0, 2.0 - l_sum)
    l = np.append(l, l_m)
    # adjust a components
    a = np.clip(a, 0.0, 1.0)
    a_m = (1.0 - np.sum(a * l[:-1])) / l[-1] if l[-1] > 1e-12 else 0.0
    a_m = max(0.0, min(1.0, a_m))
    a = np.append(a, a_m)
    return np.append(l[:-1], a)

def run(seed=42, budget_s=600, **kwargs):
    """Main function to find C5."""
    np.random.seed(seed)
    start_time = time.time()
    
    n = 2 * (m-1)
    bounds = [(0.0, 2.0)] * (m-1) + [(0.0, 1.0)] * (m-1)

    # Initial guess from previous solution to start near it
    initial_guess = np.array([0.3, 0.3, 0.3, 0.3, 0.2, 0.4, 0.6, 0.45])
    initial_guess = project_to_box_constraints(initial_guess)
    
    best_h = initial_guess
    best_c5 = compute_c5_fast(best_h)
    
    # Perform evolutionary global search with local refinement
    try:
        # Use differential evolution for robust search
        res = differential_evolution(
            objective_with_scaling,
            bounds,
            maxiter=2000,
            popsize=20,
            mutation=(0.5, 1.0),
            recombination=0.7,
            atol=1e-10,
            rtol=1e-10,
            seed=seed
        )
        if res.success:
            refined = project_to_box_constraints(res.x)
            c5 = compute_c5_fast(refined)
            if c5 < best_c5:
                best_c5 = c5
                best_h = refined
    except Exception as e:
        pass

    # Local refinement - SLSQP starting from best point
    try:
        from scipy.optimize import minimize
        res = minimize(
            objective_with_scaling,
            best_h,
            method='SLSQP',
            bounds=bounds,
            options={'maxiter': 500, 'ftol': 1e-12}
        )
        if res.success:
            refined = project_to_box_constraints(res.x)
            c5 = compute_c5_fast(refined)
            if c5 < best_c5:
                best_c5 = c5
                best_h = refined
    except Exception as e:
        pass

    # Additional improvement attempts with random refinements
    max_attempts = 1000
    attempts = 0
    while time.time() - start_time < budget_s and attempts < max_attempts:
        attempts += 1
        # perturb best solution slightly
        perturbed = best_h + np.random.normal(0, 0.01, n)
        perturbed = project_to_box_constraints(perturbed)
        c5 = compute_c5_fast(perturbed)
        if c5 < best_c5:
            best_c5 = c5
            best_h = perturbed

    # output results
    final_l = best_h[:m-1]
    final_a = best_h[m-1:]
    l_sum = 2.0 - np.sum(final_l)
    final_a_m = (1.0 - np.sum(final_a * final_l)) / l_sum
    final_a = np.append(final_a, final_a_m)
    final_l = np.append(final_l, l_sum)

    # generate final h values with 120 sample points
    hvals = np.zeros(120)
    s = np.insert(np.cumsum(final_l), 0, 0)
    dxs = 2.0 / 120
    for i in range(120):
        x = i * dxs
        j = np.searchsorted(s, x, side='right') - 1
        if j < 0:
            j = 0
        hvals[i] = final_a[j]

    # return h values, C5 bound, and number of points
    return hvals.tolist(), float(best_c5), 120
```
