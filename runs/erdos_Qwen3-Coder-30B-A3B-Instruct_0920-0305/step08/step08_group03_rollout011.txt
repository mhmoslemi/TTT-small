import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None

dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def update_correlation(c, a, b, i, delta, j, delta_j):
    """Update correlation efficiently by computing only changed contributions"""
    # Update correlation for changes at indices i and j
    # Compute contributions from first index i
    # For shifts where b[:, i+s] contributes
    if i >= 0 and i < len(a):
        shift_start = -i
        shift_end = len(a) - 1 - i
        for s in range(shift_start, shift_end + 1):
            if 0 <= i + s < len(a):
                c_idx = s + len(a) - 1
                c[c_idx] += delta * b[i + s]
    
    # For shifts where a[:, i-s] contributes (but we subtract)
    if i >= 0 and i < len(a):
        shift_start = i - (len(a) - 1)
        shift_end = i
        for s in range(shift_start, shift_end + 1):
            if 0 <= i - s < len(a):
                c_idx = s + len(a) - 1
                c[c_idx] -= delta * a[i - s]
               
    # Same for j index
    if j >= 0 and j < len(a):
        shift_start = -j
        shift_end = len(a) - 1 - j
        for s in range(shift_start, shift_end + 1):
            if 0 <= j + s < len(a):
                c_idx = s + len(a) - 1
                c[c_idx] += delta_j * b[j + s]
    
    # For shifts where a[:, j-s] contributes (but we subtract)
    if j >= 0 and j < len(a):
        shift_start = j - (len(a) - 1)
        shift_end = j
        for s in range(shift_start, shift_end + 1):
            if 0 <= j - s < len(a):
                c_idx = s + len(a) - 1
                c[c_idx] -= delta_j * a[j - s]
    
    # Add quadratic terms: delta * delta_j contribution to specific shift locations
    if i != j:
        # Delta terms contribute only at two specific indices
        shift1 = j - i + len(a) - 1  # corresponds to shift (j - i)
        shift2 = i - j + len(a) - 1  # corresponds to shift (i - j)
        if 0 <= shift1 < len(c):
            c[shift1] += delta * delta_j
        if 0 <= shift2 < len(c):
            c[shift2] += delta * delta_j
