import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_search(h0, max_iter=500):
    """Local search with adaptive step sizes"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    step_sizes = [0.05, 0.02, 0.01]
    for i in range(max_iter):
        # Try different step sizes
        for step in step_sizes:
            # Try changing a few elements
            idxs = np.random.choice(n_points, size=min(10, n_points//5), replace=False)
            for idx in idxs:
                # Try both increasing and decreasing
                for delta in [-step, step]:
                    h_new = h.copy()
                    h_new[idx] += delta
                    if h_new[idx] < 0 or h_new[idx] > 1:
                        continue
                    h_new = project_to_sum_box(h_new, TARGET_SUM)
                    new_c5 = evaluate_c5(h_new)
                    if new_c5 < best_c5:
                        h = h_new.copy()
                        best_c5 = new_c5
                        break
            else:
                continue
            break
        else:
            # If no improvement, try a global perturbation
            if np.random.random() < 0.5:
                idx = np.random.randint(0, n_points)
                delta = np.random.uniform(-0.02, 0.02)
                h[idx] += delta
                h = project_to_sum_box(h, TARGET_SUM)
                best_c5 = evaluate_c5(h)
    
    return h

def path_following_optimization(h0):
    """Use path following with line search"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    # Compute gradient estimate from finite differences
    eps = 1e-4
    for _ in range(100):
        # Generate random direction
        direction = np.random.randn(n_points)
        # Normalize
        direction = direction / np.linalg.norm(direction)
        
        # Try step sizes
        step_sizes = [0.01, 0.005, 0.001]
        for step in step_sizes:
            h_new = h + step * direction
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5:
                h = h_new.copy()
                best_c5 = new_c5
                break
    
    return h

def initial_solution():
    """Create a smart initial solution"""
    # Composite of functions that work well
    h = np.zeros(n_points)
    
    # Multiple peaks at strategic points
    peaks = [0.2, 0.5, 0.8]
    for p in peaks:
        idx = int(p * n_points)
        # Add a bump
        for i in range(30):
            if idx - 15 + i < 0 or idx - 15 + i >= n_points:
                continue
            h[idx - 15 + i] = max(h[idx - 15 + i], 0.5 * np.exp(-((i - 15) ** 2) / 50.0))
    
    # Smooth the result
    h = np.convolve(h, np.ones(10)/10, mode='same')
    
    # Make sure sum is correct
    h = project_to_sum_box(h, TARGET_SUM)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = initial_solution()
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: Multiple restarts with different methods
    for run_id in range(20):
        if time.time() > start_time + 0.3 * budget_s:
            break
            
        # Different initialization strategies
        if run_id < 5:
            h_start = initial_solution()
        elif run_id < 10:
            h_start = np.random.uniform(0, 1, n_points)
        elif run_id < 15:
            h_start = np.clip(np.random.normal(0.5, 0.2, n_points), 0, 1)
        else:
            h_start = np.random.choice([0, 0.5, 1], size=n_points, p=[0.3, 0.4, 0.3])
        
        h_start = project_to_sum_box(h_start, TARGET_SUM)
        
        # Multiple local search iterations
        h_opt = local_search(h_start, max_iter=300)
        c5_opt = evaluate_c5(h_opt)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt

    # Phase 2: Path following optimization
    if time.time() < start_time + 0.7 * budget_s:
        h_path = path_following_optimization(best_h.copy())
        c5_path = evaluate_c5(h_path)
        if c5_path < best_c5:
            best_h = h_path.copy()
            best_c5 = c5_path

    # Phase 3: Final iterations of local search with larger steps
    while time.time() < start_time + budget_s:
        h_interim = local_search(best_h.copy(), max_iter=200)
        c5_interim = evaluate_c5(h_interim)
        if c5_interim < best_c5:
            best_h = h_interim.copy()
            best_c5 = c5_interim

    return (best_h.tolist(), float(best_c5), n_points)
