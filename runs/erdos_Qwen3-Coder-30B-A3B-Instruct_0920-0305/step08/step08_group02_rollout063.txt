import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_sum_box(v, target_sum):
    # Project vector v onto the hyperplane sum(v) = target_sum with box constraints [0,1]
    # using "water filling" algorithm
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-10:
        return v

    # Bisection on Lagrange multiplier lambda
    a, b = -1.0, 1.0
    tol = 1e-10
    max_iter = 50

    for _ in range(max_iter):
        lam = (a + b) / 2
        new_v = np.clip(v + lam, 0, 1)
        new_sum = np.sum(new_v)
        if abs(new_sum - target_sum) < tol:
            break
        if new_sum > target_sum:
            a = lam
        else:
            b = lam

    return np.clip(v + (a + b) / 2, 0, 1)

def solve_milp(N, M, solver=cp.CBC):
    """Solve the MILP for the best binary pattern that maximizes minimum autocorrelation"""
    try:
        y = cp.Variable(N, boolean=True)
        t = cp.Variable()

        constraints = [cp.sum(y) == M]

        # Group w variables by shift k
        w_vars = []
        shift_sums = []

        for k in range(1, N):
            w_k = cp.Variable(N - k)
            w_vars.append(w_k)
            shift_sums.append(cp.sum(w_k))
            constraints += [w_k <= y[:N-k]]
            constraints += [w_k <= y[k:]]
            constraints += [w_k >= y[:N-k] + y[k:] - 1]
            constraints += [w_k >= 0]

        constraints += [s >= t for s in shift_sums]
        constraints += [t >= 0, t <= M]

        prob = cp.Problem(cp.Maximize(t), constraints)
        prob.solve(solver=solver, verbose=False, max_seconds=200)

        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            sol = y.value
            if sol is not None:
                return sol
    except:
        pass
    return None

def evaluate_c5(h):
    corr = correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def gradient_descent(h0, M, max_iter=10000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # find worst shift
        corr = correlate(h, 1-h, mode="full")
        k_max = np.argmax(corr) - (n_points - 1)
        
        # compute gradient
        g = np.ones(n_points)
        if k_max >= 0:
            g[:n_points-k_max] -= h[k_max:]
        else:
            k = -k_max
            g[k:] -= h[:n_points-k]
            
        if k_max > 0:
            g[k_max:] -= h[:n_points-k_max]
        elif k_max < 0:
            k = -k_max
            g[:n_points-k] -= h[k:]
            
        # gradient descent step
        alpha = 0.5 / np.sqrt(i + 1)
        h_new = h - alpha * g
        
        # project
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, M)
        
        # accept if better
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        h = h_new
        
    return best_h, best_c5

def stochastic_refinement(h0, M, max_iter=5000):
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # swap two values
        i1 = np.random.randint(0, n_points)
        i2 = np.random.randint(0, n_points)
        
        # avoid identical indices
        while i1 == i2:
            i2 = np.random.randint(0, n_points)
            
        # delta
        delta = np.random.uniform(-0.02, 0.02)
        
        # avoid bounds
        if h[i1] + delta < 0 or h[i1] + delta > 1 or h[i2] - delta < 0 or h[i2] - delta > 1:
            continue
            
        # try move
        h_new = h.copy()
        h_new[i1] += delta
        h_new[i2] -= delta
        
        # project
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, M)
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            
    return h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points / 2
    
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    
    # Phase 1: MILP for binary pattern
    milp_end = start_time + 0.4 * budget_s
    
    sol = solve_milp(n_points, TARGET_SUM, solver=cp.CBC)
    if sol is not None:
        h0 = sol.astype(float)
        c5_0 = evaluate_c5(h0)
        if c5_0 < best_c5:
            best_h = h0.copy()
            best_c5 = c5_0
    else:
        # fallback to random initialization
        h0 = np.random.uniform(0, 1, n_points)
        h0 = project_sum_box(h0, TARGET_SUM)
        c5_0 = evaluate_c5(h0)
        if c5_0 < best_c5:
            best_h = h0.copy()
            best_c5 = c5_0
    
    # Phase 2: Subgradient descent
    descent_end = start_time + 0.75 * budget_s
    if time.time() < descent_end:
        h_descent, c5_descent = gradient_descent(best_h, TARGET_SUM, max_iter=20000)
        if c5_descent < best_c5:
            best_h = h_descent.copy()
            best_c5 = c5_descent
    
    # Phase 3: Stochastic refinement
    final_end = start_time + budget_s
    if time.time() < final_end:
        h_refine, c5_refine = stochastic_refinement(best_h, TARGET_SUM, max_iter=5000)
        if c5_refine < best_c5:
            best_h = h_refine.copy()
            best_c5 = c5_refine

    return (best_h.tolist(), float(best_c5), n_points)
