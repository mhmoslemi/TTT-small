import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_with_gradients(h0, max_iter=2000):
    """Optimize using gradient-based method with projection"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def local_search(h0, max_iter=2000):
    """Improved local search with better neighborhood exploration"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    for i in range(max_iter):
        # Try to improve with small perturbations
        idx = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.02, 0.02)
        
        # Try different sizes of perturbations
        perturbations = [-0.03, -0.02, -0.01, 0.01, 0.02, 0.03]
        for delta in perturbations:
            h_new = h.copy()
            h_new[idx] += delta
            if h_new[idx] < 0 or h_new[idx] > 1:
                continue
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5:
                h = h_new.copy()
                best_c5 = new_c5
                best_h = h.copy()
                break

        # Occasionally try larger changes
        if np.random.random() < 0.1:
            # Select several points to tweak
            indices = np.random.choice(n_points, size=min(5, n_points//10), replace=False)
            h_new = h.copy()
            for idx in indices:
                delta = np.random.uniform(-0.03, 0.03)
                h_new[idx] += delta
                h_new[idx] = np.clip(h_new[idx], 0, 1)
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5:
                h = h_new.copy()
                best_c5 = new_c5
                best_h = h.copy()

    return best_h

def refined_initialization():
    """Create a more refined initial solution"""
    # Start with a piecewise constant function
    h = np.zeros(n_points)
    
    # Divide into 4 segments
    segments = 4
    segment_size = n_points // segments
    for i in range(segments):
        start_idx = i * segment_size
        end_idx = (i+1) * segment_size if i < segments-1 else n_points
        if i % 2 == 0:
            h[start_idx:end_idx] = 1.0
        else:
            h[start_idx:end_idx] = 0.0
    
    # Ensure correct sum
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def peak_insertion_and_optimization(h0, max_iter=1000):
    """Insert peaks in low regions and optimize"""
    h = h0.copy()
    # First do local search 
    h = local_search(h, max_iter // 2)
    
    # Identify low value regions and enhance them
    low_regions = np.where(h < 0.3)[0]
    if len(low_regions) > 0:
        for _ in range(3):
            idx = np.random.choice(low_regions)
            h[idx] = 1.0
            # Reduce neighbors
            for offset in [-2, -1, 0, 1, 2]:
                neighbor_idx = idx + offset
                if 0 <= neighbor_idx < n_points:
                    h[neighbor_idx] = max(0, h[neighbor_idx] - 0.15)
    
    h = project_to_sum_box(h, TARGET_SUM)
    h = local_search(h, max_iter // 2)
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = refined_initialization()
    best_c5 = evaluate_c5(best_h)
    
    # Run longer optimization
    remaining_time = budget_s
    while remaining_time > 10: 
        # Try several different optimization strategies
        strategies = [
            lambda x: local_search(x, max_iter=500),
            lambda x: optimize_with_gradients(x, max_iter=200),
            lambda x: peak_insertion_and_optimization(x, max_iter=300)
        ]
        
        for strategy in strategies:
            if remaining_time < 10:
                break
            start_iter = time.time()
            new_h = strategy(best_h.copy())
            new_c5 = evaluate_c5(new_h)
            
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
                
            # Update time budget
            elapsed = time.time() - start_iter
            remaining_time -= elapsed
            
            if remaining_time < 10:
                break
                
        # Random perturbation
        if remaining_time > 10:
            h_pert = best_h + np.random.normal(0, 0.005, n_points)
            h_pert = project_to_sum_box(h_pert, TARGET_SUM)
            h_pert = local_search(h_pert, max_iter=100)
            c5_pert = evaluate_c5(h_pert)
            if c5_pert < best_c5:
                best_h = h_pert.copy()
                best_c5 = c5_pert
                
        elapsed = time.time() - start_time
        remaining_time = budget_s - elapsed
        
    return (best_h.tolist(), float(best_c5), n_points)
