import numpy as np

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Use FFT to compute correlation efficiently"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Special case: sum already perfect
    if abs(np.sum(x) - target_sum) < 1e-12:
        return np.clip(x, 0, 1)

    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Binary search bracketing
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    # Binary search on lambda
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_binary_milp():
    """Use MILP to solve for binary h that maximizes min autocorrelation"""
    try:
        # Variables
        y = cp.Variable(n_points, boolean=True)
        w = cp.Variable((n_points, n_points), nonneg=True)
        t = cp.Variable()

        # Constraints
        constraints = [
            cp.sum(y) == TARGET_SUM,
            t <= TARGET_SUM  # Tighter LB for t
        ]

        # Product constraints for all shifts
        for k in range(1, n_points):
            for i in range(n_points - k):
                # w[i,k] = y[i] * y[i+k]
                constraints += [
                    w[i,k] <= y[i],
                    w[i,k] <= y[i+k],
                    w[i,k] >= y[i] + y[i+k] - 1,
                    w[i,k] >= 0
                ]
                # min autocorrelation constraint
                constraints += [cp.sum(w[:,k]) >= t]

        # Objective
        objective = cp.Maximize(t)

        # Solve
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, max_seconds=0.38 * 600)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            y_solution = y.value
            return y_solution
        else:
            return None
    except:
        return None

def subgradient_descent(h0, max_iter=500):
    """Deterministic subgradient descent on the worst-case shift"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    # Schedule
    alpha0 = 0.5 * dx

    for iteration in range(max_iter):
        # Compute correlation
        corr = compute_corr_fft(h)
        k_star = np.argmax(corr)
        F_k_star = corr[k_star]

        # Compute gradient g_j = 1 - h_{j+k*} - h_{j-k*}
        g = np.zeros(n_points)
        for i in range(n_points):
            if i + k_star < n_points:
                g[i] += -h[i+k_star]
            if i - k_star >= 0:
                g[i] += -h[i-k_star]
            g[i] += 1

        # Update
        alpha = alpha0 / np.sqrt(iteration + 1)
        h_new = h - alpha * g
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)

        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new

        if iteration % 20 == 0 and iteration > 0:
            if evaluate_c5(h) >= best_c5:
                break

    return best_h, best_c5

def local_search(h0, budget_s):
    """Perform hill climbing after subgradient descent"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    end_time = time.time() + budget_s * 0.2
    iterations = 0
    
    while time.time() < end_time:
        iterations += 1
        if iterations > 500:
            break
            
        # Try random swap moves
        i, j = np.random.randint(0, n_points, size=2)
        if i == j:
            continue
            
        # Choose small perturbation
        delta = np.random.uniform(-0.02, 0.02)
        h_new = h.copy()
        h_new[i] = np.clip(h[i] + delta, 0, 1)
        h_new[j] = np.clip(h[j] - delta, 0, 1)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            
    return h, best_c5
