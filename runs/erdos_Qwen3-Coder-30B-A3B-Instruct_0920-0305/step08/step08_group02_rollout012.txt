import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_sum_box(v, M):
    """Project vector v onto sum = M and box [0,1] constraints using water-filling"""
    v = np.clip(v, 0, 1)
    diff = M - np.sum(v)
    if abs(diff) < 1e-12:
        return v
    if diff > 0:
        # Need to increase values
        # Sort decreasing and increase stepwise
        idxs = np.argsort(v)[::-1]
        for i in idxs:
            if diff <= 0:
                break
            can_add = 1 - v[i]
            add = min(can_add, diff)
            v[i] += add
            diff -= add
    else:
        # Need to decrease values
        idxs = np.argsort(v)
        for i in idxs:
            if diff >= 0:
                break
            can_subtract = v[i]
            subtract = min(can_subtract, -diff)
            v[i] -= subtract
            diff += subtract
    return v

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def solve_milp_binary(N, M, max_time_s):
    """Solve binary MILP to maximize minimum autocorrelation"""
    # Variables
    y = cp.Variable(N, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(y) == M]
    
    # Product linearization and link to t
    for k in range(1, N):
        if k > N // 2:
            # Skip for large shifts to reduce time
            break
        # w_k[i] represents y[i] * y[i+k]
        w_k = cp.Variable(N - k)
        constraints += [w_k <= y[:N-k]]
        constraints += [w_k <= y[k:]]
        constraints += [w_k >= y[:N-k] + y[k:] - 1]
        constraints += [w_k >= 0]
        # Accumulate sum for shift k
        sum_w_k = cp.sum(w_k)
        constraints += [sum_w_k >= t]

    # Objective
    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, time_limit=max_time_s, verbose=False)
    except:
        try:
            prob.solve(solver=cp.GLPK_MI, time_limit=max_time_s, verbose=False)
        except:
            try:
                prob.solve(solver=cp.ECOS_BB, time_limit=max_time_s, verbose=False)
            except:
                return None

    if prob.status == cp.OPTIMAL or prob.status == cp.UNBOUNDED:
        y_opt = np.round(y.value).astype(int)
        return y_opt
    else:
        return None

def subgradient_descent(h0, max_iter=20000, initial_c5=1.0):
    """Apply subgradient descent using the worst shift"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    # Precompute 1 - h for efficiency
    h_minus = 1 - h

    for t in range(1, max_iter + 1):
        # 1. Compute correlation
        corr = np.correlate(h, h_minus, mode="full")
        k_max = np.argmax(corr) - (n_points - 1)
        max_val = corr[np.argmax(corr)]

        # 2. Compute the gradient of F_kmax=h*(1-h_shift)
        g = np.ones(n_points)
        if k_max >= 0:
            # For positive shift: h[i] - h[i+k] - h[i-k]
            g[:n_points - k_max] -= h[k_max:]
        else:
            k_abs = -k_max
            g[k_abs:] -= h[:n_points - k_abs]

        # Invert to find descent direction
        g = -g
        
        # 3. Step size with decreasing schedule
        eta = 0.01 / np.sqrt(t)
        h_new = h + eta * g
        
        # 4. Project back
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, TARGET_SUM)
        
        # 5. Accept
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c5

def stochastic_refinement(h0, max_iter=5000):
    """Simple stochastic refinement stepping away from local minima"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for _ in range(max_iter):
        # Random indices
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        
        # Adjust h[i], h[j] to preserve sum
        if h[i] + delta < 0 or h[i] + delta > 1:
            continue
        if h[j] - delta < 0 or h[j] - delta > 1:
            continue

        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    global TARGET_SUM, dx, n_points
    TARGET_SUM = n_points / 2
    dx = 2.0 / n_points
    
    # Phase 1: Binary MILP approx
    try:
        time_left = budget_s * 0.4
        y_opt = solve_milp_binary(n_points, TARGET_SUM, time_left)
        if y_opt is not None:
            h0 = y_opt.astype(float)
        else:
            # Fallback to random
            h0 = np.random.uniform(0, 1, n_points)
            h0 = project_sum_box(h0, TARGET_SUM)
    except:
        # Fallback to random initialization
        h0 = np.random.uniform(0, 1, n_points)
        h0 = project_sum_box(h0, TARGET_SUM)
    
    best_h = h0.copy()
    c5 = evaluate_c5(best_h)
    
    # Phase 2: Deterministic subgradient search
    time_left = budget_s * 0.35
    time_budget = time_left - 0.5  # overhead safety
    remaining = max(3000, int(1e4 * time_budget / budget_s))
    
    h_refined, c5_refined = subgradient_descent(h0, max_iter=remaining)
    if c5_refined < c5:
        best_h = h_refined.copy()
        c5 = c5_refined

    # Phase 3: Stochastic refinement 
    time_left = budget_s * 0.25
    time_budget = time_left - 0.5
    remaining = max(1000, int(2000 * time_budget / budget_s))
    
    h_final, c5_final = stochastic_refinement(best_h, max_iter=remaining)
    if c5_final < c5:
        best_h = h_final.copy()
        c5 = c5_final
    
    return (best_h.tolist(), float(c5), n_points)
