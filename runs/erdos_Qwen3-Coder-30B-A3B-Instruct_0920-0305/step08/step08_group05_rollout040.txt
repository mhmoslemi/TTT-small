import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def neighbor_sum(h, lag):
    """Compute neighbor sum w_i = h[i+lag] + h[i-lag] for valid indices"""
    lag_abs = abs(lag)
    if lag_abs >= len(h):
        return np.zeros_like(h)
    
    w = np.zeros_like(h)
    if lag >= 0:
        w[:len(h)-lag_abs] += h[lag_abs:]
        w[lag_abs:] += h[:len(h)-lag_abs]
    else:
        w[lag_abs:] += h[:-lag_abs]
        w[:len(h)-lag_abs] += h[lag_abs:]
    return w

    TARGET_SUM = n_points / 2

    step = 0.05
    no_improve_count = 0
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 1.0:
        # Compute correlation and find worst lag
        corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
        idx_max = np.argmax(corr)
        lag_max = idx_max - (n_points - 1)
        
        # Compute neighbor sum for worst lag
        w = neighbor_sum(best_h, lag_max)
        
        # Find indices with low and high w values
        eps = 1e-8
        low_indices = np.where(best_h > eps)[0]
        high_indices = np.where(best_h < 1 - eps)[0]
        
        if len(low_indices) == 0 or len(high_indices) == 0:
            break
            
        # Find low index with minimum w and high index with maximum w
        low_idx = low_indices[np.argmin(w[low_indices])]
        high_idx = high_indices[np.argmax(w[high_indices])]
        
        if low_idx == high_idx:
            # Try alternative approach if same index
            # Find next low and high indices
            sorted_w = np.argsort(w)
            if len(sorted_w) > 1:
                # Find first valid low that is not high_idx
                for i in sorted_w:
                    if best_h[i] > eps and i != high_idx:
                        low_idx = i
                        break
                # Find first valid high that is not low_idx
                for i in reversed(sorted_w):
                    if best_h[i] < 1 - eps and i != low_idx:
                        high_idx = i
                        break
        
        # Compute transfer amount
        delta = min(step, best_h[low_idx], 1 - best_h[high_idx])
        
        if delta < 1e-10:
            no_improve_count += 1
            if no_improve_count > 100:
                step *= 0.5
                step = max(step, 1e-5)
                no_improve_count = 0
            continue

        h_new = best_h.copy()
        h_new[low_idx] -= delta
        h_new[high_idx] += delta

    cons = {'type': 'eq', 'fun': constraint_eq}

            objective, 
            best_h.copy(), 
            method='SLSQP',

            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}

            c5_candidate = evaluate_c5(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
