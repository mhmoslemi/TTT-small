```python
import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

# Global constants
n_points = 180
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def build_grid(intervals, n):
    """Build binary grid from list of intervals [(start, end)]"""
    h = np.zeros(n)
    grid_centers = np.linspace(dx/2, 2-dx/2, n)
    for s, e in intervals:
        mask = (grid_centers >= s) & (grid_centers < e)
        h[mask] = 1
    return h

def compute_c5_fft(h):
    """Compute C5 using FFT-based correlation"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def decode_params(z, m):
    """Decode interval parameters from z ∈ [0,1]^{2m} to intervals"""
    epsilon = 1e-4
    l = epsilon + (1 - 2*epsilon) * z[:m]
    g = z[m:]  # gaps
    
    # Normalize lengths to satisfy sum = 1
    l = l * (1.0 / np.sum(l))
    
    # Compute positions
    s = np.zeros(m)
    e = np.zeros(m)
    s[0] = 0
    e[0] = s[0] + l[0]
    
    for i in range(1, m):
        s[i] = e[i-1] + g[i-1]
        e[i] = s[i] + l[i]
    
    # If end extends beyond 2, uniformly scale final gaps
    if e[-1] > 2:
        scale_factor = (2.0 - e[-1]) / np.sum(g)
        g = g * scale_factor
        s = np.zeros(m)
        e = np.zeros(m)
        s[0] = 0
        e[0] = s[0] + l[0]
        for i in range(1, m):
            s[i] = e[i-1] + g[i-1]
            e[i] = s[i] + l[i]
    
    intervals = [(s[i], e[i]) for i in range(m)]
    return intervals

def objective(z, m=5):
    """Objective function for optimization"""
    intervals = decode_params(z, m)
    h = build_grid(intervals, n_points)
    return compute_c5_fft(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Validate parameters
    m = 5  # Number of intervals
    dim = 2 * m
    
    # Phase 1: Random sampling to seed DE with good candidates
    best_c5 = float('inf')
    best_z = None
    
    for _ in range(20):
        z = np.random.rand(dim)
        c5 = objective(z, m)
        if c5 < best_c5:
            best_c5 = c5
            best_z = z.copy()
    
    # Phase 2: Differential evolution
    # Set bounds for z [0,1]^dim
    bounds = [(0, 1) for _ in range(dim)]
    maxiter = int(100 + 500 * (0.7 * budget_s / 600))  # varies based on budget
    popsize = max(15, dim * 2)
    
    def callback(xk, convergence):
        nonlocal best_z, best_c5
        c5 = objective(xk, m)
        if c5 < best_c5:
            best_c5 = c5
            best_z = xk.copy()
        return False  # Continue optimization
    
    try:
        result = differential_evolution(
            lambda z: objective(z, m),
            bounds,
            seed=seed,
            maxiter=maxiter,
            popsize=popsize,
            mutation=(0.5, 1.0),
            recombination=0.9,
            callback=callback,
            disp=False
        )
        if result.success and objective(result.x, m) < best_c5:
            best_c5 = objective(result.x, m)
            best_z = result.x.copy()
    except:
        pass  # Fall back to random if DE fails
    
    # Phase 3: Local refinement with interval boundary moves
    if best_z is not None:
        # Convert to intervals
        intervals = decode_params(best_z, m)
        h = build_grid(intervals, n_points)
        best_c5 = compute_c5_fft(h)
        
        # Adaptive local improvement loop
        time_remaining = budget_s - (time.time() - start_time)
        max_local_iter = int(time_remaining * 500)  # heuristic based on speed
        local_iter = 0
        
        improved = True
        while improved and local_iter < max_local_iter:
            improved = False
            local_iter += 1
            for i in range(m):
                # Try moving each edge of each interval
                s, e = intervals[i]
                l_original = e - s
                
                # Try moving left
                s_new = s - dx/2
                if s_new >= 0:
                    intervals_try = intervals.copy()
                    intervals_try[i] = (s_new, e)
                    h_try = build_grid(intervals_try, n_points)
                    c5_try = compute_c5_fft(h_try)
                    if c5_try < best_c5 - 1e-12:
                        best_c5 = c5_try
                        intervals = intervals_try
                        improved = True
                        break
                        
                # Try moving right
                e_new = e + dx/2
                if e_new <= 2:
                    intervals_try = intervals.copy()
                    intervals_try[i] = (s, e_new)
                    h_try = build_grid(intervals_try, n_points)
                    c5_try = compute_c5_fft(h_try)
                    if c5_try < best_c5 - 1e-12:
                        best_c5 = c5_try
                        intervals = intervals_try
                        improved = True
                        break
                        
                # Try shrinking left edge
                s_new = s + dx/2
                if s_new < e and l_original > dx:
                    intervals_try = intervals.copy()
                    intervals_try[i] = (s_new, e)
                    h_try = build_grid(intervals_try, n_points)
                    c5_try = compute_c5_fft(h_try)
                    if c5_try < best_c5 - 1e-12:
                        best_c5 = c5_try
                        intervals = intervals_try
                        improved = True
                        break
                        
                # Try shrinking right edge
                e_new = e - dx/2
                if e_new > s and l_original > dx:
                    intervals_try = intervals.copy()
                    intervals_try[i] = (s, e_new)
                    h_try = build_grid(intervals_try, n_points)
                    c5_try = compute_c5_fft(h_try)
                    if c5_try < best_c5 - 1e-12:
                        best_c5 = c5_try
                        intervals = intervals_try
                        improved = True
                        break

    # Final phase: fallback to random if needed
    time_remaining = budget_s - (time.time() - start_time)
    if time_remaining < 10.0:
        # For fallback, generate a random sample and improve it
        best_z_fallback = np.random.rand(dim)
        s = objective(best_z_fallback, m)
        if s < best_c5:
            best_c5 = s
            best_z = best_z_fallback
    
    # Final reconstruction
    final_intervals = decode_params(best_z, m)
    h_final = build_grid(final_intervals, n_points)
    c5_final = compute_c5_fft(h_final)
    
    # Ensure we return the value computed by exact correlation
    exact_corr = correlate(h_final, 1 - h_final, mode="full")
    c5_exact = dx * np.max(exact_corr)
    
    return (h_final.tolist(), float(c5_exact), n_points)
```
