import numpy as np

def compute_shift_matrices(n):
    """Construct binary shift matrices R_s"""
    R = {}
    for s in range(1, (n+1)//2):
        R_s = np.zeros((n, n))
        for i in range(n-s):
            R_s[i, i+s] = 1
        R[s] = R_s
    return R

def evaluate_c5_fft(h):
    """Fast evaluation of C5 using FFT"""
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * 2.0 / len(h)

def solve_sdp(n, shift_matrices, use_mosek=False):
    """Solve the SDP relaxation"""
    # Variables
    X = cp.Variable((n, n), symmetric=True)
    mu = cp.Variable(n)
    lam = cp.Variable()
    
    # Constraints
    constraints = [X >> 0,  # PSD
                   cp.diag(X) == mu,
                   0 <= mu, mu <= 1,
                   cp.sum(mu) == n/2,
                   X[0,0] == 1]  # Symmetry breaking
    
    # For each shift, add constraint
    for s, R_s in shift_matrices.items():
        constraints.append(cp.trace(R_s @ X) >= lam)
    
    # Objective
    objective = cp.Maximize(lam)
    prob = cp.Problem(objective, constraints)
    
    # Solver setup
    solver = MOSEK if use_mosek else SCS
    max_time = 100 if use_mosek else 30  # 100s for MOSEK, 30s for SCS
    
    try:
        prob.solve(solver=solver, verbose=False, time_limit=max_time)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return X.value, mu.value, lam.value
        else:
            print(f"SDP solver status: {prob.status}")
            return None, None, None
    except Exception as e:
        print(f"Error solving SDP: {e}")
        return None, None, None

def random_rounding(V, n):
    """Hyperplane rounding to binary vector"""
    g = np.random.randn(V.shape[1])
    scores = V @ g
    # Sort scores and choose threshold to get exactly n/2 ones
    sorted_indices = np.argsort(scores)[::-1]
    threshold_idx = n // 2
    threshold = scores[sorted_indices[threshold_idx]]
    
    x = np.zeros(n)
    x[scores >= threshold] = 1
    return x

def shift_specific_swap(x, shift_s):
    """Swap one 1 from a bad position to a good position for the worst shift"""
    n = len(x)
    # Find positions contributing to shift s
    bad = [i for i in range(n-shift_s) if x[i] == 1 and x[i+shift_s] == 0]
    good = [i for i in range(n-shift_s) if x[i] == 0 and x[i+shift_s] == 1]
    
    if not bad or not good:
        return x.copy()
        
    i = np.random.choice(bad)
    j = np.random.choice(good)
    
    x_new = x.copy()
    x_new[i] = 0
    x_new[j] = 1
    return x_new

def polish_with_swaps(x_init, max_iter=5000):
    """Refine solution with shift-aware swaps"""
    x_best = x_init.copy()
    best_c5 = evaluate_c5_fft(x_best)
    
    # Precompute correlation array
    corr = np.correlate(x_best, 1-x_best, mode='full')
    max_shift = np.argmax(corr)
    
    for i in range(max_iter):
        # Identify shifting issues
        x_new = shift_specific_swap(x_best, max_shift)
        new_c5 = evaluate_c5_fft(x_new)
        
        if new_c5 <= best_c5:
            x_best = x_new
            best_c5 = new_c5
            corr = np.correlate(x_best, 1 - x_best, mode='full')
            max_shift = np.argmax(corr)
            
    return x_best, best_c5
