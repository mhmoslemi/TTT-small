import numpy as np
from scipy.signal import correlate

import time
from scipy.optimize import minimize

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
MAX_ITER = 1000
TIME_FRACTION_FOR_MILP = 0.7
TIME_FRACTION_FOR_POLISHING = 0.2
TIME_FRACTION_FOR_FALLBACK = 0.1

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def evaluate_c5_exact(h):
    """Exact discrete evaluation of C5"""
    n = len(h)
    max_overlap = 0
    for s in range(-n + 1, n):
        overlap = 0.0
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                overlap += h[i] * (1 - h[j])
        max_overlap = max(max_overlap, overlap)
    return dx * max_overlap

def solve_milp_row_generation(h0, timeout_s):
    """Exact MILP solving with row generation"""
    start_time = time.time()
    
    # Initialize shifts to consider
    shifts = [1, 2, 3, 4, 5]
    total_time = 0
    
    # Setup continuous variables for the current iteration
    x = Variable(n_points, boolean=True)
    t = Variable()
    
    # Prepare the constraints list
    constraints = [sum(x) == TARGET_SUM, t >= 0]
    
    # Add initial constraints for shifts
    y = {}
    for s in shifts:
        y[s] = [Variable(boolean=True) for i in range(n_points)]
        for i in range(n_points):
            j = i + s
            if 0 <= j < n_points:
                constraints.extend([
                    y[s][i] <= x[i],
                    y[s][i] <= x[j],
                    y[s][i] >= x[i] + x[j] - 1
                ])
            else:
                constraints.append(y[s][i] == 0)
    
    # Add epigraph constraints
    for s in shifts:
        constraints.append(sum(y[s]) >= TARGET_SUM - t)
    
    # Objective
    objective = Minimize(t)
    
    # Build and solve model
    prob = Problem(objective, constraints)
    prob.solve(solver=GUROBI, verbose=False)
    
    # Get current solution
    x_val = np.array([v.value for v in x]).round()
    
    # Evaluate the true overlaps
    corr_exact = compute_corr_fft(x_val)
    max_c5_val = np.max(corr_exact)
    
    # Add more shifts if needed
    iteration = 0
    while (time.time() - start_time < timeout_s * 0.7 and 
           max_c5_val > t.value + 1e-8 and 
           iteration < 100):
        iteration += 1
        
        # Add the shift that causes maximum overlap
        max_idx = np.argmax(corr_exact)
        shift = max_idx - (len(corr_exact) // 2)
        
        assert shift not in shifts, "Shift already in consideration"
        shifts.append(shift)
        
        # Add new variables for this shift
        s = shift
        y[s] = [Variable(boolean=True) for i in range(n_points)]
        for i in range(n_points):
            j = i + s
            if 0 <= j < n_points:
                constraints.extend([
                    y[s][i] <= x[i],
                    y[s][i] <= x[j],
                    y[s][i] >= x[i] + x[j] - 1
                ])
            else:
                constraints.append(y[s][i] == 0)
        
        # Add epigraph constraint
        constraints.append(sum(y[s]) >= TARGET_SUM - t)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=GUROBI, verbose=False)
            x_val = np.array([v.value for v in x]).round()
            corr_exact = compute_corr_fft(x_val)
            max_c5_val = np.max(corr_exact)
        except:
            break
    
    return x_val, t.value

def smooth_epigraph_optimization(h0, t_val, gamma=100):
    """Use smooth approximation to refine continuous solution"""
    h = h0.copy()
    target_sum = TARGET_SUM
    n = len(h)
    
    def smooth_max_overshift(h, t, gamma):
        """Smooth upper bound of max over shifts"""
        total_overlaps = []
        for s in range(-n + 1, n):
            overlap = 0.0
            for i in range(n):
                j = i + s
                if 0 <= j < n:
                    overlap += h[i] * (1 - h[j])
            total_overlaps.append(overlap)
        
        return t + (1 / gamma) * np.log(np.sum(np.exp(gamma * (np.array(total_overlaps) - t))))
    
    def objective(h_vars, t_var):
        h = h_vars.reshape(-1)
        return smooth_max_overshift(h, t_var, gamma)
    
    # Reshape for optimizer
    h_vars = h.copy()
    t_var = t_val
    
    # Constraint: sum of h equals target
    def mass_constraint(h_vars):
        return np.sum(h_vars) - target_sum
    
    # Optimization using SLSQP
    res = minimize(
        lambda x: objective(x, t_var),
        h_vars,
        method='SLSQP',
        bounds=[(0, 1)] * len(h_vars),
        constraints={'type': 'eq', 'fun': mass_constraint},
        options={'maxiter': 500}
    )
    
    if res.success:
        h_final = np.clip(res.x, 0, 1)
    else:
        h_final = h
    
    return h_final

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    
    TARGET_SUM = n_points / 2
    
    # Phase 1: Exact MILP with row generation
    try:
        x_milp, t_milp = solve_milp_row_generation(None, budget_s * 0.7)
        if x_milp is not None:
            # Ensure correct mass
            x_milp = np.clip(x_milp, 0, 1)
            current_sum = np.sum(x_milp)
            if abs(current_sum - TARGET_SUM) > 1e-6:
                ratio = TARGET_SUM / current_sum
                x_milp = np.clip(x_milp * ratio, 0, 1)
            
            c5_milp = evaluate_c5(x_milp)
            if c5_milp < best_c5:
                best_c5 = c5_milp
                best_h = x_milp.copy()
    except:
        pass
    
    # Phase 2: Continuous polishing if any binary solution was obtained
    if best_h is not None and (time.time() - start_time) < budget_s * 0.9:
        polished_h = smooth_epigraph_optimization(best_h, best_c5/dx)
        polished_c5 = evaluate_c5(polished_h)
        if polished_c5 < best_c5:
            best_c5 = polished_c5
            best_h = polished_h.copy()
            
    # Fallback: improve based on heuristic if no better solution was found
    if best_h is None:
        h0 = generate_initial_solution()
        c5_start = evaluate_c5(h0)
        if c5_start < best_c5:
            best_c5 = c5_start
            best_h = h0.copy()
        
        # Apply local search to initial
        for i in range(2000):
            if time.time() - start_time > budget_s * 0.9:
                break
            idx = np.random.randint(0, n_points)
            delta = np.random.uniform(-0.02, 0.02)
            h_new = best_h.copy()
            h_new[idx] += delta
            h_new = np.clip(h_new, 0, 1)
            # Renormalize
            current_sum = np.sum(h_new)
            h_new = h_new * (TARGET_SUM / current_sum)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    # Final check for correctness
    best_h = np.clip(best_h, 0, 1)
    current_sum = np.sum(best_h)
    if abs(current_sum - TARGET_SUM) > 1e-6:
        best_h = best_h * (TARGET_SUM / current_sum)
    
    return (best_h.tolist(), float(best_c5), n_points)
