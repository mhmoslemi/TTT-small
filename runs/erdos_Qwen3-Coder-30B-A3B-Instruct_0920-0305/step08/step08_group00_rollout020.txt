import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 200
dx = 2.0 / n_points

def compute_corr_int(h):
    """Compute correlation using FFT and return integer counts"""
    h_rev = h[::-1].astype(np.float64)
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    # round to nearest integer (small float errors)
    corr_int = np.rint(corr).astype(int)
    return corr_int

def c5_from_corr(corr_int):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr_int)

def eval_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr_int = compute_corr_int(h)
    return c5_from_corr(corr_int)

def generate_random_binary(n, ones_count):
    """Generate random binary vector with exact number of ones"""
    h = np.zeros(n, dtype=int)
    h[:ones_count] = 1
    np.random.shuffle(h)
    return h

def simulated_annealing(n, time_budget, seed=42):
    """Run simulated annealing with temperature cooling"""
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial binary pattern
    h = generate_random_binary(n, n//2)
    best_h = h.copy()
    best_corr = compute_corr_int(h)
    best_max = best_corr.max()
    current_h = h.copy()
    current_max = best_max

    # Annealing parameters
    T = 5.0  # initial temperature
    cooling_rate = 0.9995
    min_T = 1e-6
    
    iterations = 0
    while time.time() - start_time < time_budget:
        # Generate swap
        ones_idx = np.where(current_h == 1)[0]
        zeros_idx = np.where(current_h == 0)[0]
        if len(ones_idx) > 0 and len(zeros_idx) > 0:
            i_one = np.random.choice(ones_idx)
            i_zero = np.random.choice(zeros_idx)
            # swap
            cand_h = current_h.copy()
            cand_h[i_one] = 0
            cand_h[i_zero] = 1

            cand_corr = compute_corr_int(cand_h)
            cand_max = cand_corr.max()
            delta = cand_max - current_max
            
            if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, min_T)):
                # accept
                current_h = cand_h
                current_max = cand_max
                
                if cand_max < best_max:
                    best_h = cand_h.copy()
                    best_max = cand_max
                    
        T *= cooling_rate
        iterations += 1
        # Prevent oscillation at low temp
        if T < min_T:
            T = min_T
            
    return best_h, best_max

def hill_climb(h, max_iter=5000, patience=1000):
    """Perform hill climbing to refine solution"""
    best_h = h.copy()
    best_max = compute_corr_int(best_h).max()
    no_improve = 0
    
    for _ in range(max_iter):
        # Generate random swaps
        improved = False
        for _ in range(50):  # try up to 50 swaps per iteration
            i_one = np.random.choice(np.where(best_h == 1)[0])
            i_zero = np.random.choice(np.where(best_h == 0)[0])
            
            cand_h = best_h.copy()
            cand_h[i_one] = 0
            cand_h[i_zero] = 1
            
            cand_max = compute_corr_int(cand_h).max()
            if cand_max < best_max:
                best_h = cand_h
                best_max = cand_max
                improved = True
                break
                
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= patience:
                break
                
    return best_h, best_max

def run(seed=42, budget_s=600, **kwargs):
    global n_points, dx
    start_time = time.time()
    
    # Initial solution using simulated annealing
    time_remaining = budget_s * 0.8
    h0, c5_0 = simulated_annealing(n_points, time_remaining, seed)
    
    # Refinement via hill climbing
    h1, c5_1 = hill_climb(h0, max_iter=2000, patience=500)
    
    # Final check for better solution
    time_remaining = budget_s - (time.time() - start_time)
    if time_remaining > 10:
        h2, c5_2 = simulated_annealing(n_points, time_remaining * 0.8, seed + 1)
        h3, c5_3 = hill_climb(h2, max_iter=1000, patience=250)
        
        if c5_3 < c5_1:
            best_h = h3
            best_c5 = c5_3
        else:
            best_h = h1
            best_c5 = c5_1
    else:
        best_h = h1
        best_c5 = c5_1
        
    return (best_h.tolist(), float(best_c5), n_points)
