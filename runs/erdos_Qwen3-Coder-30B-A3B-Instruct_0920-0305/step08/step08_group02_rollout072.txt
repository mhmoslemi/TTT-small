import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

def project_sum_box(v, M):
    """Project vector v onto box [0,1] with sum = M using waterfilling."""
    v = np.clip(v, 0, 1)
    diff = M - np.sum(v)
    if abs(diff) < 1e-10:
        return v
    # bisection to find lambda
    a, b = -1.0, 1.0
    while True:
        lam = (a + b) / 2
        v_new = np.clip(v + lam, 0, 1)
        s = np.sum(v_new)
        if abs(s - M) < 1e-10:
            break
        if s > M:
            a = lam
        else:
            b = lam
    return np.clip(v + lam, 0, 1)

def solve_milp_binary(N, M):
    """Solve MILP to find binary y that maximizes minimal autocorrelation."""
    # Variables
    y = cp.Variable(N, boolean=True)
    t = cp.Variable()
    
    # The binary constraints
    constraints = [cp.sum(y) == M]

    # Product variables w_k,i
    w_vars = {}
    shift_sums = []
    for k in range(1, N):
        indices = range(N - k)
        w_k = cp.Variable(N - k)
        w_vars[k] = w_k
        
        # Linearized constraints for w_k[i] = y[i]*y[i+k]
        for i in indices:
            constraints.append(w_k[i] <= y[i])
            constraints.append(w_k[i] <= y[i+k])
            constraints.append(w_k[i] >= y[i] + y[i+k] - 1)
            constraints.append(w_k[i] >= 0)
            
        shift_sum = cp.sum(w_k)
        shift_sums.append(shift_sum)
        constraints.append(shift_sum >= t)

    # Objective
    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=200)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return y.value
    except Exception:
        pass
    # fallback to GLPK_MI if CBC not available or fails
    try:
        prob.solve(solver=cp.GLPK_MI, verbose=False, max_seconds=200)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return y.value
    except Exception:
        pass
    return None

def evaluate_c5(h):
    """Fast evaluation of C5 via FFT"""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / len(h))

def subgradient_descent(h0, M, max_iter=20000, step_size=1e-3):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for t in range(1, max_iter + 1):
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr) - (len(h) - 1)
        
        g = np.ones_like(h)
        if k_max >= 0:
            if k_max < len(h):
                g[:len(h) - k_max] -= h[k_max:]
        else:
            k = -k_max
            if k < len(h):
                g[k:] -= h[:len(h) - k]
                
        # Symmetric pieces
        if k_max > 0:
            if k_max < len(h):
                g[k_max:] -= h[:len(h) - k_max]
        elif k_max < 0:
            k = -k_max
            if k < len(h):
                g[:len(h) - k] -= h[k:]
                
        # SGD step
        h_new = h - step_size * g
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, M)
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        h = h_new.copy()

    return best_h, best_c5

def stochastic_refinement(h0, M, max_iter=5000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    for i in range(max_iter):
        idx1, idx2 = np.random.choice(len(h), size=2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        h_new = h.copy()
        h_new[idx1] += delta
        h_new[idx2] -= delta
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, M)
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        h = h_new.copy()
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 120
    dx = 2.0 / n_points
    M = n_points // 2
    target_c5 = 0.3807
    
    h_best = None
    c5_best = float('inf')
    
    # Phase 1: MILP
    milp_start = time.time()
    y_opt = solve_milp_binary(n_points, M)
    if y_opt is not None:
        h0 = y_opt.astype(float)
        c50 = evaluate_c5(h0)
        if c50 < c5_best:
            c5_best = c50
            h_best = h0.copy()
    
    # Phase 2: Subgradient descent
    sg_duration = 0.35 * budget_s
    sg_start = time.time()
    if h_best is not None:
        h_after_sg, c5_sg = subgradient_descent(h_best, M, max_iter=20000)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_after_sg.copy()
    
    # Phase 3: Simulated Annealing
    sa_duration = budget_s - (time.time() - start_time)
    if sa_duration > 10:
        h_after_sa, c5_sa = stochastic_refinement(h_best, M, max_iter=5000)
        if c5_sa < c5_best:
            c5_best = c5_sa
            h_best = h_after_sa.copy()
    
    if h_best is None:
        # fallback if no good solution found so far
        h_fallback = np.random.uniform(0, 1, n_points)
        h_fallback = project_sum_box(h_fallback, M)
        c5_fallback = evaluate_c5(h_fallback)
        c5_best = c5_fallback
        h_best = h_fallback.copy()
    
    return (h_best.tolist(), float(c5_best), n_points)
