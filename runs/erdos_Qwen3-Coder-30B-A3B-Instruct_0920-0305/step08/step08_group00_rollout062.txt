import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Create a better starting solution using a carefully chosen pattern
    # Using a symmetric pattern with strong peaks at strategic locations
    h0 = np.zeros(n_points)
    
    # Create a symmetric pattern with peaks at 0.3 and 1.7 to better handle overlaps
    peak_positions = [0.3, 1.7] 
    peak_width = 0.1
    peak_height = 1.0
    
    for i in range(n_points):
        x = i * dx
        for pos in peak_positions:
            if abs(x - pos) < peak_width:
                # Gaussian-like peak
                h0[i] += peak_height * np.exp(-0.5 * ((x - pos) / peak_width)**2)
    
    # Normalize to sum = TARGET_SUM
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    # Further refine using a heuristic approach
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Phase 1: Local optimization using gradient descent
    for iteration in range(200):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Random perturbations with projection
        h_test = best_h + np.random.normal(0, 0.01, n_points)
        h_test = np.clip(h_test, 0, 1)
        h_test = project_to_sum_box(h_test, TARGET_SUM)
        
        c5_test = evaluate_c5(h_test)
        if c5_test < best_c5:
            best_h = h_test.copy()
            best_c5 = c5_test
    
    # Phase 2: Pattern-based refinement
    for iteration in range(100):
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Try to refine by adjusting two key regions
        # Make a few optimized local changes
        indices_to_adjust = np.random.choice(n_points, size=10, replace=False)
        h_candidate = best_h.copy()
        
        for idx in indices_to_adjust:
            # Try small changes in both directions
            best_change = 0
            best_val = h_candidate[idx]
            
            for delta in [-0.02, -0.01, 0.01, 0.02]:
                new_val = h_candidate[idx] + delta
                if 0 <= new_val <= 1:
                    h_candidate[idx] = new_val
                    h_projected = project_to_sum_box(h_candidate, TARGET_SUM)
                    new_c5 = evaluate_c5(h_projected)
                    
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h_projected.copy()
                        best_val = new_val
                        
            h_candidate[idx] = best_val  # Reset for next iteration
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
