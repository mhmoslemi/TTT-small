from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

            # We use the standard MILP encoding for products:
            # y[i] <= h[i]
            # y[i] <= h[i+k]
            # y[i] >= h[i] + h[i+k] - 1

        objective = cp.Minimize(C)
