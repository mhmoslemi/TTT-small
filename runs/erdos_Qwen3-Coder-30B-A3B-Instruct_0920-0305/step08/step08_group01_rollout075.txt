import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import SCS, MOSEK
from scipy.linalg import eigh

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
H_FRACTIONAL = None
BEST_H = None
BEST_C5 = float('inf')

def compute_shift_matrices(n):
    """Compute shift matrices for all valid shifts"""
    Rs = []
    for s in range(1, n//2 + 1):
        R = np.zeros((n,n))
        for i in range(n-s):
            R[i, i+s] = 1
        Rs.append(R)
    return Rs

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    corr = np.correlate(h_rev, b, mode="full")
    return dx * np.max(corr)

def project_to_sum(h, target_sum):
    """Project h to satisfy sum constraint"""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Adjust values to meet sum constraint while keeping within [0,1]
    diff = target_sum - current_sum
    if diff > 0:
        # Increase some values
        idxs = np.argsort(h)
        h_copy = h.copy()
        for i in range(len(idxs)-1, -1, -1):
            idx = idxs[i]
            if diff <= 0:
                break
            delta = min(diff, 1 - h_copy[idx])
            h_copy[idx] += delta
            diff -= delta
        return h_copy
    else:
        # Decrease some values
        idxs = np.argsort(h, kind='stable')
        h_copy = h.copy()
        for i in range(len(idxs)):
            idx = idxs[i]
            if diff >= 0:
                break
            delta = min(-diff, h_copy[idx])
            h_copy[idx] -= delta
            diff += delta
        return h_copy

def solve_sdp(n, Rs):
    """Solve the SDP relaxation to get fractional solution"""
    # Variables for PSD matrix and helper vectors
    X = cp.Variable((n, n), symmetric=True)
    mu = cp.Variable(n)
    lambda_var = cp.Variable()

    # Constraints
    constraints = [
        X >> 0,  # Positive semidefinite
        cp.diag(X) == mu,  # Diagonal elements are mu
        mu >= 0,  # Box constraints
        mu <= 1,  # Box constraints
        cp.sum(mu) == n/2,  # Sum constraint
    ]

    # Add overlap constraints
    for i, R in enumerate(Rs):
        trace_constraint = cp.trace(R @ X) >= lambda_var
        constraints.append(trace_constraint)

    # Objective: maximize lambda
    objective = cp.Maximize(lambda_var)
    prob = cp.Problem(objective, constraints)
    
    # Solve with timeout
    prob.solve(solver=SCS, max_iters=1000, eps=1e-4, verbose=False)
    
    if prob.status == cp.OPTIMAL:
        return X.value, mu.value, prob.value
    else:
        return None, None, None

def random_rounding(X_opt, mu, n):
    """Random hyperplane rounding of the SDP solution"""
    global H_FRACTIONAL
    H_FRACTIONAL = X_opt
    try:
        # Using Cholesky factorization
        U = np.linalg.cholesky(X_opt)
        rows = U.T  # Transpose to get rows as vectors
    except np.linalg.LinAlgError:
        # Fallback to eigendecomposition if Cholesky fails
        vals, vecs = eigh(X_opt)
        vals = np.maximum(vals, 0)  # Ensure non-negative
        sqrt_vals = np.sqrt(vals)
        U = vecs * sqrt_vals.reshape(1, -1)
        rows = U.T

    candidates = []
    for _ in range(500):  # Try 500 rounds
        g = np.random.normal(0, 1, U.shape[0])
        scores = np.dot(rows, g)
        # Keep the top n/2 indices
        idxs = np.argpartition(-scores, n//2)[:n//2]
        binary = np.zeros(n)
        binary[idxs] = 1
        binary = project_to_sum(binary, n/2)
        c5 = evaluate_c5(binary)
        candidates.append((binary.copy(), c5))
        
    if candidates:
        best = min(candidates, key=lambda x: x[1])
        return best[0], best[1]
    else:
        return mu, float('inf')

def local_refinement(h_initial):
    """Perform fast swap-based local refinement"""
    h = h_initial.copy()
    current_corr = correlate(h[::-1], 1-h, mode="full")
    max_shift = np.argmax(current_corr)
    max_val = current_corr[max_shift]
    
    ITERATIONS = 5000
    
    for _ in range(ITERATIONS):
        # Find positions causing max overlap
        # The overlap is computed for shift max_shift
        bad_indices = np.where((h[:-max_shift] == 1) & (h[max_shift:] == 0))[0]
        good_indices = np.where((h[:-max_shift] == 0) & (h[max_shift:] == 1))[0]
        
        if len(bad_indices) == 0 or len(good_indices) == 0:
            break
            
        # Pick a random position from bad set and good set
        idx_bad = np.random.choice(bad_indices)
        idx_good = np.random.choice(good_indices)
        
        # Swap them
        h[idx_bad], h[idx_good] = h[idx_good], h[idx_bad]

        # Recompute correlation after update
        current_corr = correlate(h[::-1], 1-h, mode="full")
        max_shift = np.argmax(current_corr)
        max_val = current_corr[max_shift]
        
    return h, evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx, BEST_H, BEST_C5
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up initial values
    TARGET_SUM = n_points / 2
    Rs = compute_shift_matrices(n_points)
    
    # Solve SDP problem
    X_opt, mu_opt, obj_opt = solve_sdp(n_points, Rs)
    if X_opt is None:
        raise RuntimeError("SDP solver failed")
    
    # Randomized rounding
    h_rounded, c5_rounded = random_rounding(X_opt, mu_opt, n_points)
    if c5_rounded < BEST_C5:
        BEST_H = h_rounded.copy()
        BEST_C5 = c5_rounded
    
    # Local refinement
    h_refined, c5_refined = local_refinement(h_rounded)
    if c5_refined < BEST_C5:
        BEST_H = h_refined.copy()
        BEST_C5 = c5_refined
    
    return (BEST_H.tolist(), float(BEST_C5), n_points)
