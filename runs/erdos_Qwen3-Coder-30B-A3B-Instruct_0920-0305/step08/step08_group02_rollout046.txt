```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

# Global constants
N = 120
dx = 2.0 / N
TARGET_SUM = N / 2
MAX_SECONDS_MILP = 0.4 * 600
MAX_SECONDS_GRAD = 0.35 * 600
MAX_SECONDS_SWEEP = 600 - MAX_SECONDS_MILP - MAX_SECONDS_GRAD

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, N)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using bisection"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_milp():
    """Solve MILP for binary h to find optimal autocorrelation pattern"""
    # Set up binary variables y_i (h_i) for i = 0,...,N-1
    y = cp.Variable(N, boolean=True)
    t = cp.Variable()

    # Constraints: sum y_i = M
    constraints = [cp.sum(y) == TARGET_SUM]

    # Add auxiliary variables w_{k,i} for each k=1..N-1 and i=0..N-k-1
    w_vars = {}
    for k in range(1, N):
        w_vars[k] = cp.Variable(N - k, nonneg=True)
    
    # Linearization constraints for w_vars
    for k in range(1, N):
        for i in range(N - k):
            constraints.extend([
                w_vars[k][i] <= y[i],
                w_vars[k][i] <= y[i+k],
                w_vars[k][i] >= y[i] + y[i+k] - 1,
                w_vars[k][i] >= 0
            ])
    
    # Constraint that each correlation is at least t
    for k in range(1, N):
        constraints.append(cp.sum(w_vars[k]) >= t)
    
    # Bounds on t
    constraints.extend([0 <= t, t <= TARGET_SUM])
    
    # Objective: maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CBC, max_seconds=MAX_SECONDS_MILP)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURATE]:
            return y.value
    except:
        pass
    return None

def greedy_binary_initialization():
    """Construct a good binary initial vector using spacing heuristic"""
    h = np.zeros(N)
    # Place 1s roughly every 2 positions while trying to minimize correlations
    placed = 0
    i = 0
    while placed < TARGET_SUM:
        if i < N:
            h[i] = 1
            placed += 1
            i += 2  # Skip next position for spacing
        else:
            # If we have not placed enough, put them in the remaining spots
            for j in range(N):
                if h[j] == 0:
                    h[j] = 1
                    placed += 1
                    if placed >= TARGET_SUM:
                        break
            break
    return h

def find_worst_shift(h):
    """Find the shift k that gives maximum overlap"""
    corr = compute_corr_fft(h)
    # The overlap at shift k is M - sum of h_i * h_{i+k}
    # We want to minimize this, so we maximize sum h_i * h_{i+k}
    # Hence we maximize the correlation
    return np.argmax(corr)

def subgradient_descent(h0):
    """Perform subgradient descent on the objective"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_grad = time.time()
    t = 1
    alpha = 0.5 * dx  # Initial step size
    
    while time.time() - start_grad < MAX_SECONDS_GRAD:
        k_star = find_worst_shift(h)
        g = np.zeros(N)
        
        for i in range(N):
            # h_{i+k_star} may go out of bounds, clip appropriately
            h_left = h[i - k_star] if i - k_star >= 0 else 0
            h_right = h[i + k_star] if i + k_star < N else 0
            g[i] = 1 - h_left - h_right
        
        h_new = h - (alpha / np.sqrt(t)) * g
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        if evaluate_c5(h_new) < best_c5:
            best_h = h_new.copy()
            best_c5 = evaluate_c5(h_new)
        
        h = h_new
        t += 1
        
        if t % 200 == 0:
            if evaluate_c5(h) < best_c5:
                best_h = h.copy()
                best_c5 = evaluate_c5(h)
    
    return best_h, best_c5

def simulated_annealing(h0, c5_0):
    """Perform simulated annealing on the best solution from gradient descent"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_0
    
    start_sa = time.time()
    temp_start = 5 * dx
    num_swaps = 0
    
    while time.time() - start_sa < MAX_SECONDS_SWEEP:
        i, j = np.random.choice(N, 2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Clamp
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        delta_c5 = new_c5 - best_c5
        
        # Annealing probability
        t = temp_start * (1 - (time.time() - start_sa) / MAX_SECONDS_SWEEP)
        if t > 0:
            prob = np.exp(-delta_c5 / t)
            if np.random.rand() < prob:
                h = h_new.copy()
                if new_c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = new_c5
        else:
            if new_c5 < best_c5:
                h = h_new.copy()
                best_h = h_new.copy()
                best_c5 = new_c5
        
        num_swaps += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Phase 1: Binary MILP or greedy initialization
    binary_solution = binary_milp()
    if binary_solution is not None:
        h0 = binary_solution
        h0 = h0.astype(float)
    else:
        h0 = greedy_binary_initialization()
    
    # Phase 2: Subgradient descent
    best_h, best_c5 = subgradient_descent(h0)
    
    # Phase 3: Simulated annealing
    final_h, final_c5 = simulated_annealing(best_h, best_c5)
    
    # Return best found solution
    return (final_h.tolist(), float(final_c5), N)
```
