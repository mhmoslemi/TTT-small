import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

# Global constants
N_POINTS = 120
DX = 2.0 / N_POINTS
TARGET_SUM = N_POINTS // 2

def project_sum_box(v, M):
    """Project vector v onto box constraints [0,1] with sum = M"""
    v = np.clip(v, 0, 1)
    diff = M - np.sum(v)
    if abs(diff) < 1e-10:
        return v
    
    # Binary search for the correct Lagrange multiplier
    a, b = -1.0, 1.0
    tol = 1e-10
    for _ in range(50):
        lam = (a + b) / 2
        proj = np.clip(v + lam, 0, 1)
        sum_proj = np.sum(proj)
        if abs(sum_proj - M) < tol:
            break
        if sum_proj < M:
            a = lam
        else:
            b = lam
    return np.clip(v + (a + b)/2, 0, 1)

def solve_milp_binary(N, M):
    """Solve the binary MILP to maximize minimum autocorrelation"""
    # Define variables
    y = cp.Variable(N, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(y) == M]
    
    # Add auxiliary variables and constraints for each shift
    w_vars = {}
    shift_sums = []
    for k in range(1, N):
        w_k = cp.Variable(N - k)
        # Create binary product constraints
        for i in range(N - k):
            constraints += [w_k[i] <= y[i]]
            constraints += [w_k[i] <= y[i + k]]
            constraints += [w_k[i] >= y[i] + y[i + k] - 1]
            constraints += [w_k[i] >= 0]
            # Special case: if they're both 0, w_k[i] should be 0
        shift_sum = cp.sum(w_k)
        shift_sums.append(shift_sum)
        constraints += [shift_sum >= t]

    # Set objective
    problem = cp.Problem(cp.Maximize(t), constraints)
    
    try:
        problem.solve(solver=cp.CBC, verbose=False, max_seconds=200)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return y.value
    except Exception:
        pass
    
    # Try fallback solvers if CBC fails
    solvers_to_try = [cp.GLPK_MI, cp.ECOS_BB]
    for solver in solvers_to_try:
        try:
            problem.solve(solver=solver, verbose=False, max_seconds=200)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return y.value
        except Exception:
            continue
    
    return None

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = correlate(h, 1 - h, mode="full") * DX
    return np.max(corr)

def subgradient_descent(h0, T_max=20000):
    """Subgradient descent using explicit gradient of worst shift"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    for t in range(1, T_max):
        # 1. Compute correlation, identify k_max (worst shift)
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr) - (N_POINTS - 1)

        # 2. Build sub-gradient
        g = np.ones(N_POINTS)
        if k_max >= 0:
            g[:N_POINTS - k_max] -= h[k_max:]
        else:
            k = -k_max
            g[k:] -= h[:N_POINTS - k]
        if k_max > 0:
            g[k_max:] -= h[:N_POINTS - k_max]
        elif k_max < 0:
            k = -k_max
            g[:N_POINTS - k] -= h[k:]
        
        # 3. Gradient step (move opposite to gradient)
        alpha = 1.0 / np.sqrt(t)
        h_new = h - alpha * g
        
        # 4. Project back onto feasible polytope
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, TARGET_SUM)
        
        # 5. Accept if improvement
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        h = h_new

    return best_h, best_c5

def stochastic_refinement(h0, budget_s):
    """Lightweight simulated annealing to escape flat spots"""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    temp = 0.1
    decay_rate = 0.99
    
    while time.time() - start < budget_s:
        # Pick two random indices
        i, j = np.random.randint(0, N_POINTS, size=2)
        if i == j:
            continue
            
        # Attempt a small change
        delta = np.random.uniform(-0.02, 0.02)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        # Ensure the change keeps sum within bounds
        h_new = project_sum_box(h_new, TARGET_SUM)
        new_c5 = evaluate_c5(h_new)
        
        # Accept or reject
        if new_c5 < best_c5:
            h = h_new.copy()
            best_h = h_new.copy()
            best_c5 = new_c5
        else:
            # Accept with probability based on temperature
            if np.random.random() < np.exp(-(new_c5 - best_c5) / temp):
                h = h_new.copy()
        
        temp *= decay_rate
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Phase 1: Solve MILP for binary solution if possible
    h_milp = None
    if budget_s >= 200:
        milp_result = solve_milp_binary(N_POINTS, TARGET_SUM)
        if milp_result is not None:
            h_milp = milp_result.astype(float)

    # Phase 2: Subgradient descent from MILP or random
    remaining_time = budget_s - (time.time() - start_time)
    if h_milp is not None:
        h_final, c5_final = subgradient_descent(h_milp, int(remaining_time * 0.7))
    else:
        # Fallback: random initial feasible point
        h_init = np.random.uniform(0, 1, N_POINTS)
        h_init = project_sum_box(h_init, TARGET_SUM)
        h_final, c5_final = subgradient_descent(h_init, int(remaining_time * 0.7))

    # Phase 3: Final stochastic refinement
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        h_final, c5_final = stochastic_refinement(h_final, remaining_time)
    
    return (h_final.tolist(), float(c5_final), N_POINTS)
