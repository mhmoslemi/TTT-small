import numpy as np

from scipy.optimize import differential_evolution
import time

# Global variables
TARGET_SUM = None
n_points = 300
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def compute_corr(h):
    """Compute full correlation array"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return corr * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def piecewise_constant_from_params(params, n_points):
    """Convert parameters to h array (piecewise constant)"""
    K = 8  # number of blocks
    widths = np.exp(params[:K])
    widths = widths / np.sum(widths) * 2.0  # normalize to sum to 2
    heights = 1.0 / (1.0 + np.exp(-params[K:]))  # sigmoid
    heights = np.clip(heights, 0, 1)
    
    # Create cumulative widths
    cum_widths = np.concatenate([[0], np.cumsum(widths)])
    
    # Create h array
    h = np.zeros(n_points)
    positions = np.linspace(0, 2, n_points, endpoint=False) + dx/2
    block_indices = np.searchsorted(cum_widths, positions, side='right') - 1
    block_indices = np.clip(block_indices, 0, K-1)
    h = heights[block_indices]
    
    return h

def de_objective(params):
    """Objective function for differential evolution"""
    h = piecewise_constant_from_params(params, n_points)
    h = project_to_sum_box(h, TARGET_SUM)
    c5 = evaluate_c5(h)
    return c5

        for lag in active_lags:

            no_improvement_count = 0

            step_size *= 0.9
            no_improvement_count = 0

                pert_h[idx2] += eps

                if pert_c5 < best_c5:
                    best_c5 = pert_c5
                    best_h = pert_h.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
