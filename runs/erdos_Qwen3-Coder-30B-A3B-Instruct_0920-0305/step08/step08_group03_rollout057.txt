def evaluate_c5_piecewise(h_vals, intervals):
    """Evaluate C5 using exact piecewise integration"""
    # intervals: [(start, end, height), ...]
    # h_vals: [length_1, ..., length_{m-1}, height_1, ..., height_{m-1}]
    m = len(intervals)
    dx = 2.0 / sum(intervals[i][1] - intervals[i][0] for i in range(m))
    total_length = sum(intervals[i][1] - intervals[i][0] for i in range(m))
    assert abs(total_length - 2.0) < 1e-10

    # Set up cumulative intervals
    cum_len = [0.0]
    for start, end, _ in intervals:
        cum_len.append(cum_len[-1] + (end - start))
    
    def get_h_at(x):
        # Find which interval x belongs to
        for i, (start, end, height) in enumerate(intervals):
            if start <= x < end:
                return height
        if abs(x - 2.0) < 1e-10:
            return intervals[-1][2]  # Assume last interval continues to 2.0
        return 0.0
    
    # Evaluate max over k
    max_val = 0.0
    k_vals = np.linspace(0.0, 2.0, 1001)  # fine grid
    for k in k_vals:
        integral = 0.0
        # x ranges from 0 to 2-k
        for a, b, h in intervals:
            x_start = max(0.0, a)
            x_end = min(2.0 - k, b)
            if x_end > x_start:
                integral += h * (x_end - x_start)
        # subtract ∫ h(x)h(x+k) dx
        integral2 = 0.0
        for i in range(m):
            for j in range(m):
                a1, b1, h1 = intervals[i]
                a2, b2, h2 = intervals[j]
                # overlap of [a1,b1] with [a2+k, b2+k]
                start_overlap = max(a1, a2 + k)
                end_overlap = min(b1, b2 + k)
                if start_overlap < end_overlap:
                    integral2 += h1 * h2 * (end_overlap - start_overlap)
        val = integral - integral2
        if val > max_val:
            max_val = val
    return max_val

def generate_piecewise_intervals(h_vals):
    """Generate interval list from h_vals [length_1, ..., length_{m-1}, height_1, ..., height_{m-1}]"""
    m = (len(h_vals) // 2) + 1
    lengths = h_vals[:m-1]
    heights = h_vals[m-1:]
    
    # Normalize so that sum of lengths = 2
    total_len = sum(lengths)
    if total_len > 0:
        lengths = [l * 2.0 / total_len for l in lengths]
    
    # Compute cumulative positions
    positions = [0.0]
    for l in lengths:
        positions.append(positions[-1] + l)
        
    # Create intervals
    intervals = []
    for i in range(m-1):
        intervals.append((positions[i], positions[i+1], heights[i]))
    
    # Set final height based on the constraint
    # integral_over_intervals = sum(height * length) = 1 (constraint)
    # So sum_{i=0..m-1}heights[i]*lengths[i] = 1
    # But we have m-1 independent lengths and heights
    # Final height is determined by:
    # sum_{i=0..m-2} heights[i]*lengths[i] + height_{m-1}*lengths[m-1] = 1
    # => height_{m-1} = (1 - sum_{i=0..m-2})/lengths[m-1] if lengths[m-1] != 0
    
    # In our case, m == 6 so we have 5 lengths and 5 heights
    if m >= 2:
        remaining_area = 1.0 - sum(h * l for h, l in zip(heights[:-1], lengths))
        final_length = 2.0 - sum(lengths)
        if final_length > 1e-10:
            final_height = remaining_area / final_length
            intervals.append((positions[-1], 2.0, final_height))
        else:
            intervals.append((positions[-1], 2.0, 0.0))
    else:
        intervals.append((0.0, 2.0, 0.5))  # fallback
    
    return intervals

def project_to_sum_constraint(h_vals):
    """Project to the unit mass constraint ∫ h = 1"""
    intervals = generate_piecewise_intervals(h_vals)
    # Compute total area
    area = 0.0
    for start, end, height in intervals:
        area += height * (end - start)
    if area > 0:
        scale_factor = 1.0 / area
        # apply to heights without changing lengths
        scaled_intervals = [(start, end, height * scale_factor) for start, end, height in intervals]
        return scaled_intervals
    return intervals

def construct_initial_guess():

    # Try something like two flat regions with small bump
    h_vals = np.array([0.3, 0.4, 0.4, 0.3, 0.3, 0.6, 0.4, 0.4, 0.6])
    return h_vals
