import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import time

# Global variables
TARGET_SUM = None
n_points = 300
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def piecewise_const_from_params(params, n_points):
    """Convert parameters to piecewise constant function"""
    K = 8
    log_widths = params[:K]
    heights_raw = params[K:]
    
    # Convert to widths (ensure sum = 2)
    widths = np.exp(log_widths)
    widths = widths * 2.0 / np.sum(widths)
    
    # Convert to heights (clip to [0,1])
    heights = 1.0 / (1.0 + np.exp(-heights_raw))
    heights = np.clip(heights, 0, 1)
    
    # Build piecewise constant array
    cumulative_widths = np.concatenate([[0], np.cumsum(widths)])
    bin_edges = cumulative_widths
    
    positions = np.linspace(0, 2, n_points, endpoint=False) + dx/2
    bin_indices = np.digitize(positions, bin_edges) - 1
    bin_indices = np.clip(bin_indices, 0, K-1)
    
    h = heights[bin_indices]
    
    return h

def de_objective(params):
    """Objective function for differential evolution"""
    h = piecewise_const_from_params(params, n_points)
    h = project_to_sum_box(h, TARGET_SUM)
    c5 = evaluate_c5(h)
    return c5

        # Sample parameters

                best_c5 = c5
                best_h = h.copy()
        except:
            continue
    
    # Use the best from random search to continue in DE
    try:
        # Differential evolution optimization
        bounds = [(-5, 5) for _ in range(16)]  # 8 log-widths + 8 heights
        
        result = differential_evolution(

    max_iter = 5000

        # Compute full correlation array
        corr = np.real(np.fft.ifft(np.fft.fft(current_h[::-1]) * np.fft.fft(1 - current_h)))

        active_indices = np.where(np.abs(corr - max_corr) < 1e-10)[0]
        
        # Compute gradient for active indices
        delta = np.zeros(n_points)

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            
        current_h = new_h
        iteration += 1
        
        # Occasionally reduce step size
        if iteration % 500 == 0:
            step_size *= 0.9

    # Final refinement with simulated annealing-like random moves
    iteration = 0
    max_iter = 2000
    
    while iteration < max_iter and time.time() - start_time < budget_s:
        # Random pair transfer
        i1, i2 = np.random.choice(n_points, 2, replace=False)
        if current_h[i1] > 0 and current_h[i2] < 1:
            epsilon = np.random.rand() * 0.02
            new_h = current_h.copy()
            new_h[i1] -= epsilon
            new_h[i2] += epsilon
            new_h = np.clip(new_h, 0, 1)
            new_h = project_to_sum_box(new_h, TARGET_SUM)
            
            new_c5 = evaluate_c5(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
            elif np.random.rand() < 0.1:  # Accept worse with probability
                current_h = new_h
            else:
                current_h = new_h
        iteration += 1
        
    return (best_h.tolist(), float(best_c5), n_points)
