import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_corr_int(h):
    """Compute integer correlation counts using FFT"""
    corr = compute_corr_fft(h)
    # round to nearest integer (small float errors)
    corr_int = np.rint(corr).astype(int)
    return corr_int

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize binary pattern with half ones
    h = np.zeros(n_points, dtype=int)
    h[:n_points//2] = 1
    np.random.shuffle(h)
    
    # Simulated Annealing parameters
    T0 = 5.0
    T = T0
    cooling = 0.9995
    best_h = h.copy()
    best_corr = compute_corr_int(best_h)
    best_max = best_corr.max()
    
    # Annealing loop
    while time.time() - start_time < 0.8 * budget_s:
        # Generate random swap
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if len(ones_idx) > 0 and len(zeros_idx) > 0:
            i_one = np.random.choice(ones_idx)
            i_zero = np.random.choice(zeros_idx)
            
            # Perform swap
            new_h = h.copy()
            new_h[i_one] = 0
            new_h[i_zero] = 1
            
            # Compute new correlation
            new_corr = compute_corr_int(new_h)
            new_max = new_corr.max()
            delta = new_max - best_max
            
            # Accept with probability based on temperature
            if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, 1e-10)):
                h = new_h
                best_max = new_max
                best_h = h.copy()
                best_corr = new_corr
                
        T *= cooling
    
    # Local hill climbing refinement
    max_iter = 10000
    patience = 1000
    
    for _ in range(max_iter):
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Generate random swaps
        swaps = []
        for _ in range(20):
            ones_idx = np.where(best_h == 1)[0]
            zeros_idx = np.where(best_h == 0)[0]
            if len(ones_idx) > 0 and len(zeros_idx) > 0:
                i_one = np.random.choice(ones_idx)
                i_zero = np.random.choice(zeros_idx)
                swaps.append((i_one, i_zero))
        
        improved = False
        for i_one, i_zero in swaps:
            cand_h = best_h.copy()
            cand_h[i_one] = 0
            cand_h[i_zero] = 1
            cand_corr = compute_corr_int(cand_h)
            cand_max = cand_corr.max()
            
            if cand_max < best_max:
                best_h = cand_h
                best_max = cand_max
                best_corr = cand_corr
                improved = True
                break
                
        if not improved:
            patience -= 1
        else:
            patience = 1000
            
        if patience <= 0:
            break
    
    # Final evaluation
    final_corr = compute_corr_int(best_h)
    final_c5 = dx * np.max(final_corr)
    
    return (best_h.tolist(), float(final_c5), n_points)
