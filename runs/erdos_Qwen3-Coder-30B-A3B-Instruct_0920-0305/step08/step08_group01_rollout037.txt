import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

# Global constants
n_points = 180
dx = 2.0 / n_points
TARGET_SUM = n_points / 2
MAX_M = 7
MIN_M = 5

def build_binary_from_intervals(intervals, n):
    """Convert interval endpoints to binary step function on grid"""
    h = np.zeros(n)
    grid_centers = np.linspace(dx/2, 2-dx/2, n)
    for s, e in intervals:
        mask = (grid_centers >= s) & (grid_centers < e)
        h[mask] = 1
    return h

def evaluate_c5_fft(h):
    """Fast C5 evaluation using FFT correlation"""
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.irfft(np.fft.rfft(h_rev) * np.fft.rfft(b))
    return dx * np.max(corr.real)

def decode_intervals(z, m):
    """Decode flat vector z to interval endpoints"""
    l = np.abs(z[:m]) + 1e-6  # lengths
    g = np.abs(z[m:])        # gaps
    
    # Normalise lengths to sum to 1
    l = l * (1.0 / np.sum(l))
    
    # Compute interval positions
    s = np.concatenate([[0], np.cumsum(l[:-1] + g[:-1])])
    e = s + l
    
    # If intervals exceed domain [0,2], rescale gaps
    total_length = e[-1]
    if total_length > 2:
        # Compress gaps
        scale = (2 - np.sum(l)) / np.sum(g)
        g = g * scale
        e = s + l
        e[-1] = 2  # Force endpoint to exactly 2
    
    # Create interval pairs
    intervals = [(s[i], e[i]) for i in range(m)]
    return intervals

def objective(z):
    """Objective function to minimize C5"""
    m = MIN_M  # Start with 5 intervals
    intervals = decode_intervals(z, m)
    h = build_binary_from_intervals(intervals, n_points)
    return evaluate_c5_fft(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_params = None
    
    # Phase 1: Quick random sampling for initial population
    phase1_end = start_time + 0.1 * budget_s
    for _ in range(100):
        if time.time() > phase1_end:
            break
        m = MIN_M
        z = np.random.rand(2 * m)
        try:
            c5 = objective(z)
            if c5 < best_c5:
                best_c5 = c5
                best_params = z.copy()
        except:
            continue
    
    # Phase 2: Differential evolution
    phase2_end = start_time + 0.7 * budget_s
    m = MIN_M
    bounds = [(0, 1) for _ in range(2 * m)]
    
    def callback(x, convergence):
        nonlocal best_c5, best_params
        try:
            c5 = objective(x)
            if c5 < best_c5:
                best_c5 = c5
                best_params = x.copy()
        except:
            pass
    
    try:
        result = differential_evolution(
            objective,
            bounds,
            maxiter=int(10000 / (2 * m)),
            popsize=20,
            callback=callback,
            seed=seed,
            disp=False
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_params = result.x.copy()
    except:
        pass
    
    # If best_params not found, use fallback 
    if best_params is None:
        m = MIN_M
        best_params = np.random.rand(2 * m)
        
    # Phase 3: Local refinement
    phase3_end = start_time + 0.9 * budget_s
    intervals = decode_intervals(best_params, m)
    h = build_binary_from_intervals(intervals, n_points)
    best_c5 = evaluate_c5_fft(h)
    
    # Refinement with discrete step moves
    def compute_worst_shift(h):
        c = correlate(h, 1 - h, mode="full")
        return np.argmax(c)
        
    def try_interval_move(s, e, delta_s, delta_e, intervals):
        """Try moving one interval and return new intervals"""
        new_intervals = intervals.copy()
        i = intervals.index((s, e))
        s_new = max(0, s + delta_s)
        e_new = max(s_new + dx, e + delta_e)
        if e_new <= 2:
            new_intervals[i] = (s_new, e_new)
            return new_intervals
        return None
    
    # Simple local search using interval adjustments
    max_refinements = 500
    for _ in range(max_refinements):
        if time.time() > phase3_end:
            break
        # Try moving each interval slightly
        improved = False
        for i, (s, e) in enumerate(intervals):
            new_intervals = try_interval_move(s, e, -dx/4, dx/4, intervals)
            if new_intervals:
                h_new = build_binary_from_intervals(new_intervals, n_points)
                c5_new = evaluate_c5_fft(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    intervals = new_intervals
                    improved = True
                    break
        
        if not improved:
            break
    
    # Final verification with exact correlate
    h_final = build_binary_from_intervals(intervals, n_points)
    final_c5 = evaluate_c5_fft(h_final)
    
    # Fallback if needed
    fallback_c5 = float('inf')
    if final_c5 >= best_c5:
        fallback_c5 = final_c5
    else:
        fallback_c5 = best_c5
    
    # Try more intervals if needed
    if fallback_c5 >= 0.381:
        m = MAX_M
        bounds = [(0, 1) for _ in range(2 * m)]
        try:
            result = differential_evolution(
                objective,
                bounds,
                maxiter=int(5000 / (2 * m)),
                popsize=15,
                seed=seed,
                disp=False
            )
            if result.success:
                intervals_m = decode_intervals(result.x, m)
                h_m = build_binary_from_intervals(intervals_m, n_points)
                c5_m = evaluate_c5_fft(h_m)
                if c5_m < fallback_c5:
                    fallback_c5 = c5_m
                    intervals = intervals_m
        except:
            pass
    
    # Final adjustment and evaluation
    h_final = build_binary_from_intervals(intervals, n_points)
    final_c5 = evaluate_c5_fft(h_final)
    
    # Ensure we return the best one
    return (h_final.tolist(), float(final_c5), n_points)
