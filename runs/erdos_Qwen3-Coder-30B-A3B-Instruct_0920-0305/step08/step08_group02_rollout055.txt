```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, SUM
import math

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_sum_box(h, target_sum):
    """Project h onto box constraints [0,1] with sum = target_sum using water-filling"""
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    
    if abs(current_sum - target_sum) < 1e-10:
        return h_clipped
    
    # Binary search for lambda
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        proj_sum = np.sum(np.clip(h_clipped + m, 0, 1))
        if abs(proj_sum - target_sum) < 1e-10:
            break
        elif proj_sum < target_sum:
            a = m
        else:
            b = m
    
    lam = (a + b) / 2
    result = np.clip(h_clipped + lam, 0, 1)
    return result

def compute_subgradient(h, k_star):
    """Compute subgradient of max_k F_k at h with active shift k_star"""
    g = np.zeros_like(h)
    for i in range(len(h)):
        g[i] = 1
        if i + k_star < len(h):
            g[i] -= h[i + k_star]
        if i - k_star >= 0:
            g[i] -= h[i - k_star]
    return g

def subgradient_step(h, alpha, k_star):
    """Perform one subgradient step"""
    g = compute_subgradient(h, k_star)
    h_new = h - alpha * g
    return project_sum_box(h_new, TARGET_SUM)

def determine_active_shift(h):
    """Determine shift that maximizes F_k = M - sum(h_i * h_{i+k})"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    # The shift k corresponds to index k in the correlation result
    k_star = np.argmax(corr[len(h)-1:])
    # Return the actual shift (not offset by len(h)-1)
    return k_star

def solve_binary_milp():
    """Solve for binary y valued sequence that maximizes minimum autocorrelation"""
    M = int(n_points // 2)
    
    # Binary variables
    y = Variable(n_points, boolean=True)
    
    # Auxiliary variables for products
    w = []
    for k in range(1, n_points):
        row = []
        for i in range(n_points - k):
            w_i = Variable(1, name=f"w_{k}_{i}")
            row.append(w_i)
        w.append(row)

    # Objective: maximize t
    t = Variable(1, name="t")
    
    # Constraints
    constraints = [
        cp.sum(y) == M,
        t >= 0,
        t <= M
    ]
    
    # Each w must be y_i * y_{i+k}
    for k in range(1, n_points):
        for i in range(n_points - k):
            # w_{k,i} <= y_i
            constraints.append(w[k-1][i] <= y[i])
            # w_{k,i} <= y_{i+k}
            constraints.append(w[k-1][i] <= y[i + k])
            # w_{k,i} >= y_i + y_{i+k} - 1
            constraints.append(w[k-1][i] >= y[i] + y[i + k] - 1)
            # w_{k,i} >= 0
            constraints.append(w[k-1][i] >= 0)
            
            # Enforce w_{k,i} = y_i * y_{i+k} (the product constraint)
            # We use the constraints that already ensure this
            # The feasibility of t will be handled by the objective to select a tight bound
    
    # Set minimum correlation constraints
    for k in range(1, n_points):
        constraint = cp.sum(w[k-1]) >= t
        constraints.append(constraint)
    
    # Solve the problem
    objective = Maximize(t)
    prob = Problem(objective, constraints)
    
    try:
        # Try with different solvers
        solvers = [cp.CBC, cp.GLPK_MI]
        result = None
        for solver in solvers:
            try:
                result = prob.solve(solver=solver, max_seconds=0.35 * 600 - 0.1)
                if result is not None and result > 1e-6:
                    break
            except Exception:
                continue
        if result is None:
            return None
        binary_solution = np.round(y.value).astype(int)
        return binary_solution
    except Exception:
        return None

def greedy_binary_construction():
    """Construct a good binary pattern greedily"""
    M = n_points // 2
    y = np.zeros(n_points, dtype=int)  
    indices = []
    used = np.zeros(n_points, dtype=bool)
    
    # Start with pattern 1 0 1 0 1 0 ...
    for i in range(n_points):
        if i % 2 == 0:
            indices.append(i)
            y[i] = 1
    used[indices] = True
    
    # Try mutation to increase minimum autocorrelation
    for _ in range(M - len(indices)):
        best_i = -1
        best_increase = -1e10
        for i in range(n_points):
            if not used[i]:
                # Check if turning this on increases the minimum size
                new_y = y.copy()
                new_y[i] = 1
                temp_sum = np.sum(new_y)
                
                if temp_sum <= M:
                    # Compute lowest correlation
                    h = new_y.astype(float)
                    h_rev = h[::-1]
                    b = 1 - h
                    L = 2 * len(h) - 1
                    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
                    min_corr = np.min(corr[len(h)-1:])
                    
                    if min_corr > best_increase:
                        best_increase = min_corr
                        best_i = i
                        
        if best_i != -1:
            y[best_i] = 1
            used[best_i] = True
            indices.append(best_i)
        
    return y

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()

    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Phase 1: Binary MILP approach (if feasible)
    elapsed = time.time() - start_time
    if elapsed < 0.4 * budget_s:
        y_sol = solve_binary_milp()
        if y_sol is not None:
            h_binary = y_sol.astype(float)
            # Use this for initial solution
            h = h_binary.copy()
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 2: If no binary solution, fall back to greedy or default
    if best_h is None:
        h = greedy_binary_construction().astype(float)
        h = project_sum_box(h, TARGET_SUM)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 3: Subgradient method
    elapsed = time.time() - start_time
    phase_end_time = start_time + 0.35 * budget_s
    if elapsed < 0.35 * budget_s:
        alpha = 0.5 * dx  # Initial step size
        iteration = 0
        while time.time() < phase_end_time:
            k_star = determine_active_shift(best_h)
            h_new = subgradient_step(best_h, alpha, k_star)
            c5_new = evaluate_c5(h_new)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            
            iteration += 1
            if iteration % 50 == 0:
                # Decrease step size
                alpha *= 0.95
                
    # Phase 4: Final stochastic refinement
    phase_end_time = start_time + budget_s
    if time.time() < phase_end_time:
        while time.time() < phase_end_time:
            # Random swap between two elements with non-boundary values
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            if 0 < i < n_points - 1 and 0 < j < n_points - 1:
                delta = np.random.uniform(-0.02, 0.02)
                # Ensure change does not violate constraints too much
                if 0 <= best_h[i] + delta <= 1 and 0 <= best_h[j] - delta <= 1:
                    h_trial = best_h.copy()
                    h_trial[i] += delta
                    h_trial[j] -= delta
                    h_trial = project_sum_box(h_trial, TARGET_SUM)
                    
                    c5_trial = evaluate_c5(h_trial)
                    if c5_trial < best_c5:
                        best_h = h_trial.copy()
                        best_c5 = c5_trial
    
    return (best_h.tolist(), float(best_c5), n_points)
```
