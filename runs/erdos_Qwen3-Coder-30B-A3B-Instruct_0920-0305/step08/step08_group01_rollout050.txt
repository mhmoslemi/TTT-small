import numpy as np
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def random_binary_solution():
    """Create a random binary solution with exactly TARGET_SUM ones"""
    h = np.zeros(n_points, dtype=float)
    ones_indices = np.random.choice(n_points, size=TARGET_SUM, replace=False)
    h[ones_indices] = 1.0
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    TARGET_SUM = n_points // 2
    
    # Set up temperature schedule
    initial_temp = 0.015
    final_temp = 0.0001
    max_iter = 200000
    
    # Track best solution
    best_h = random_binary_solution()
    best_c5 = compute_c5(best_h)
    
    # Main optimization loop
    iteration = 0
    last_improvement = 0
    temp = initial_temp
    
    while iteration < max_iter and (time.time() - start_time) < budget_s * 0.95:
        # Generate a random swap
        ones_indices = np.where(best_h == 1)[0]
        zeros_indices = np.where(best_h == 0)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            # Restart if we don't have both 0s and 1s
            best_h = random_binary_solution()
            best_c5 = compute_c5(best_h)
            last_improvement = iteration
            continue
            
        i = np.random.choice(ones_indices)
        j = np.random.choice(zeros_indices)
        
        # Create candidate solution by swapping values
        candidate_h = best_h.copy()
        candidate_h[i] = 0
        candidate_h[j] = 1
        
        # Compute C5 for candidate
        candidate_c5 = compute_c5(candidate_h)
        
        # Accept or reject based on temperature
        if candidate_c5 < best_c5:
            best_h = candidate_h
            best_c5 = candidate_c5
            last_improvement = iteration
        else:
            # Accept with some probability based on temperature
            delta = candidate_c5 - best_c5
            acceptance_prob = np.exp(-delta / temp)
            if np.random.random() < acceptance_prob:
                best_h = candidate_h
                best_c5 = candidate_c5
                last_improvement = iteration
        
        iteration += 1
        
        # Update temperature
        progress = (time.time() - start_time) / budget_s
        temp = initial_temp * (final_temp / initial_temp) ** progress
        
        # Occasionally restart if no improvement for a long time
        if iteration - last_improvement > 5000 and iteration > 10000:
            best_h = random_binary_solution()
            best_c5 = compute_c5(best_h)
            last_improvement = iteration
    
    return (best_h.tolist(), float(best_c5), n_points)
