```python
import numpy as np
from scipy.signal import correlate
import time
import math

# Global variables
n_points = 200
dx = 2.0 / n_points

def compute_corr_int(h):
    """Compute correlation using FFT and return integer array"""
    h_rev = h[::-1].astype(np.float64)
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    # round to nearest integer (small float errors)
    corr_int = np.rint(corr).astype(int)
    return corr_int

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_int(h)
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Generate initial binary pattern: random balanced
    h = np.zeros(n_points, dtype=int)
    h[:n_points//2] = 1
    np.random.shuffle(h)
    
    # Find best solution among multiple random seeds
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Run simulated annealing for different random seeds
    for seed_iter in range(5):
        np.random.seed(seed + seed_iter)
        
        # Generate new random binary pattern
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
        np.random.shuffle(h)
        
        # Annealing parameters
        T = 1.0
        cooling = 0.9995
        max_iter = 100000
        iter_count = 0
        
        current_h = h.copy()
        current_c5 = evaluate_c5(current_h)
        
        # Track best during this seed run
        seed_best_h = current_h.copy()
        seed_best_c5 = current_c5
        
        while iter_count < max_iter and (time.time() - start_time) < budget_s * 0.6:
            # Generate random swap
            ones_idx = np.where(current_h == 1)[0]
            zeros_idx = np.where(current_h == 0)[0]
            
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
                
            i_one = np.random.choice(ones_idx)
            i_zero = np.random.choice(zeros_idx)
            
            # Create candidate
            cand_h = current_h.copy()
            cand_h[i_one] = 0
            cand_h[i_zero] = 1
            
            # Evaluate candidate
            cand_c5 = evaluate_c5(cand_h)
            
            # Accept with simulated annealing probability
            delta = cand_c5 - current_c5
            if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, 1e-10)):
                current_h = cand_h
                current_c5 = cand_c5
                
                # Update seed best if improved
                if cand_c5 < seed_best_c5:
                    seed_best_h = cand_h.copy()
                    seed_best_c5 = cand_c5
            
            # Update global best if improved
            if seed_best_c5 < best_c5:
                best_h = seed_best_h.copy()
                best_c5 = seed_best_c5
            
            # Cooling schedule
            T *= cooling
            iter_count += 1
    
    # Refinement: hill climbing on best solution
    if (time.time() - start_time) < budget_s * 0.9:
        refine_h = best_h.copy()
        refine_max_iter = 50000
        patience = 1000
        
        for _ in range(refine_max_iter):
            if (time.time() - start_time) >= budget_s * 0.9:
                break
                
            # Evaluate current
            current_corr = compute_corr_int(refine_h)
            current_max = current_corr.max()
            
            # Try many random swaps
            improved = False
            for _ in range(50):
                # Generate random swap
                ones_idx = np.where(refine_h == 1)[0]
                zeros_idx = np.where(refine_h == 0)[0]
                
                if len(ones_idx) == 0 or len(zeros_idx) == 0:
                    break
                    
                i_one = np.random.choice(ones_idx)
                i_zero = np.random.choice(zeros_idx)
                
                # Create candidate
                cand_h = refine_h.copy()
                cand_h[i_one] = 0
                cand_h[i_zero] = 1
                
                # Evaluate candidate
                cand_corr = compute_corr_int(cand_h)
                cand_max = cand_corr.max()
                
                if cand_max < current_max:
                    refine_h = cand_h
                    improved = True
                    break
            
            if not improved:
                patience -= 1
                if patience <= 0:
                    break
            else:
                patience = 1000
    
    # Final evaluation
    final_h = refine_h if 'refine_h' in locals() else best_h
    final_c5 = evaluate_c5(final_h)
    
    return (final_h.tolist(), float(final_c5), n_points)
```
