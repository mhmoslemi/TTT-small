import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
from scipy.linalg import cholesky
import math

def compute_shift_matrices(n):
    """Compute shift matrices R_s as sparse diagonal matrices."""
    R = []
    for s in range(1, (n//2) + 1):
        # Create a sparse diagonal matrix for shift s
        # R_s[i,i+s] = 1
        row = np.arange(n - s)
        col = np.arange(s, n)
        data = np.ones_like(row, dtype=int)
        R_s = np.zeros((n, n))
        R_s[row, col] = data
        R.append(R_s)
    return R

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    dx = 2.0 / len(h)
    corr_full = correlate(h, 1 - h, mode="full")
    corr = corr_full[len(h)-1:]  # Take positive shifts
    return dx * np.max(corr)

def solve_sdp(n, shift_matrices, seed=42):
    """Solve SDP relaxation program to get an approximate solution."""
    np.random.seed(seed)
    
    # Variables
    X = cp.Variable((n, n), symmetric=True)
    mu = cp.Variable(n)
    lam = cp.Variable()
    
    # Constraints
    constraints = [
        X >> 0,  # semidefinite
        cp.diag(X) == mu,
        mu >= 0,
        mu <= 1,
        cp.sum(mu) == n / 2,
        # Symmetry breaking: fix first coordinate to 1 (for derivative of the shift upper bound)
        mu[0] == 1.0
    ]

    # Add overlap constraints for each shift
    for s, R in enumerate(shift_matrices):
        constraints.append(cp.trace(R @ X) >= lam)

    # Objective
    objective = cp.Maximize(lam)
    prob = cp.Problem(objective, constraints)

    try:
        # Solve with timeout
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000, eps=1e-4)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            X_opt = X.value
            mu_opt = mu.value
            lam_opt = lam.value
            return X_opt, mu_opt, lam_opt
        else:
            return None, None, None
    except:
        return None, None, None

def hyperplane_round(X, n, n_rounds=500):
    """Round the relaxed solution using random hyperplane method."""
    if X is None:
        return None, float('inf')
    
    # Cholesky factorization of X
    try:
        L = cholesky(X, lower=True)
        # rows of L are v_i
        rows = L.T  # Each row is v_i
    except:
        return None, float('inf')
    
    best_h = None
    best_c5 = float('inf')
    
    for _ in range(n_rounds):
        # generate random Gaussian vector
        g = np.random.randn(rows.shape[1])
        # compute scores
        scores = np.dot(rows, g)
        # rank and threshold to get exactly n/2 ones
        sorted_indices = np.argsort(-scores)  # indices of highest scores first
        h = np.zeros(n)
        h[sorted_indices[:n//2]] = 1.0
        
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def local_refinement(x, max_iter=5000, target_c5=None, verbose=False):
    """Refine a binary vector by swapping 1s and 0s to reduce max correlation."""
    dx = 2.0 / len(x)
    start_time = time.time()
    
    # Compute initial full correlation array
    corr_full = correlate(x, 1 - x, mode="full")
    corr = corr_full[len(x)-1:]  # Positive shifts only
    max_shift = np.argmax(corr)
    max_val = corr[max_shift]
    
    if verbose:
        print(f"Initial max overlap at shift {max_shift}: {max_val * dx}")
    
    for it in range(max_iter):
        now = time.time()
        if now - start_time > 100:  # 100 seconds for refinement
            break
            
        # Identify indices contributing to bad shift
        bad_indices = set()
        good_indices = set()
        
        for i in range(len(x)):
            if x[i] == 1 and i + max_shift < len(x) and x[i + max_shift] == 0:
                bad_indices.add(i)
            if x[i] == 0 and i + max_shift < len(x) and x[i + max_shift] == 1:
                good_indices.add(i)
        
        if not bad_indices or not good_indices:
            break
            
        # Pick a bad 1 and a good 0 at random
        i_bad = np.random.choice(list(bad_indices))
        j_good = np.random.choice(list(good_indices))
        
        # Do the swap
        x[i_bad], x[j_good] = x[j_good], x[i_bad]
        
        # Recompute correlation for affected positions
        corr_full = correlate(x, 1 - x, mode="full")
        corr = corr_full[len(x)-1:]
        new_max_shift = np.argmax(corr)
        new_max_val = corr[new_max_shift]
        
        if new_max_val <= max_val:
            max_shift = new_max_shift
            max_val = new_max_val
        else:
            # revert swap if it got worse
            x[i_bad], x[j_good] = x[j_good], x[i_bad]
    
    final_c5 = evaluate_c5(x)
    if verbose:
        print(f"After local refinement: {final_c5}")
        
    return x, final_c5
