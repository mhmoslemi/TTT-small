def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def update_correlation_c(c, h, i, delta_i, j, delta_j):
    """Update correlation efficiently after changing h[i] and h[j]"""
    # For shift s in range -(n-1) to n-1, c[s + (n-1)] = sum(h[k] * (1 - h[k+s]))
    # where both indices are in bounds (0 to n-1)
    n = len(h)
    
    # Add contributions from h[i] change
    if i >= 0 and i < n:
        # b contribution: h[i] * (1 - h[i+s]) 
        # for s where i+s is valid (0 <= i+s <= n-1)
        s_start = -i
        s_end = n-1-i
        if s_start <= s_end:
            c_idx_start = s_start + (n-1)
            c_idx_end = s_end + (n-1) + 1
            # Get the valid indices in b
            b_slice = 1 - h[np.arange(i+s_start, i+s_end+1)]
            c[c_idx_start:c_idx_end] += delta_i * b_slice
            
        # a contribution: h[i-s] * h[i]  
        # for s where i-s is valid (0 <= i-s <= n-1) 
        s_start = i-(n-1)
        s_end = i
        if s_start <= s_end:
            c_idx_start = s_start + (n-1)
            c_idx_end = s_end + (n-1) + 1
            # Get the reversed slice of a
            a_slice = h[np.arange(i-s_end, i-s_start+1)] # reversed
            c[c_idx_start:c_idx_end] -= delta_i * a_slice
            
    # Add contributions from h[j] change        
    if j >= 0 and j < n:
        # b contribution
        s_start = -j
        s_end = n-1-j
        if s_start <= s_end:
            c_idx_start = s_start + (n-1)
            c_idx_end = s_end + (n-1) + 1
            # Get the valid indices in b
            b_slice = 1 - h[np.arange(j+s_start, j+s_end+1)]
            c[c_idx_start:c_idx_end] += delta_j * b_slice
            
        # a contribution  
        s_start = j-(n-1)
        s_end = j
        if s_start <= s_end:
            c_idx_start = s_start + (n-1)
            c_idx_end = s_end + (n-1) + 1
            # Get the reversed slice of a
            a_slice = h[np.arange(j-s_end, j-s_start+1)] # reversed
            c[c_idx_start:c_idx_end] -= delta_j * a_slice
            
    return c
