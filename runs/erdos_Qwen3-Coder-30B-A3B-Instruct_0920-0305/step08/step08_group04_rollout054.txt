TARGET_SUM = None

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(ifft(fft(h_rev, L) * fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # water-filling projection
    # sort x in descending order
    sorted_x = np.sort(x)[::-1]
    # find the point where cumulative sum exceeds target_sum
    cumsum = np.cumsum(sorted_x)
    # Find largest index where cumsum <= target_sum
    idx = np.argmax(cumsum >= target_sum)
    if cumsum[idx] < target_sum:
        idx = len(cumsum)
    # The threshold value
    if idx == 0:
        threshold = (target_sum - 0) / len(x)
    else:
        threshold = (target_sum - cumsum[idx-1]) / (len(x) - idx)
    # project
    projected = np.clip(x - threshold, 0, 1)
    return projected

def rasterise_and_project(θ):

    widths = np.exp(θ[:n_block])

    # Normalize widths to sum to 2
    widths = 2 * widths / np.sum(widths)
    
    # Create cumulative widths
    cum_widths = np.concatenate([[0], np.cumsum(widths)])
    
    # Randomly sample from initial guess for exact matching
    h = np.zeros(n_points)

        start_idx = int(cum_widths[i] / 2 * n_points)

        if i == n_block - 1:
            end_idx = n_points
        h[start_idx:end_idx] = heights[i]
    
    # Project to sum constraint
    h = project_to_sum_box(h, TARGET_SUM)
    return h
