```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 120
    dx = 2.0 / n
    target_sum = n // 2  # exactly n/2 ones

    # MILP formulation
    def solve_milp(n, target_sum):
        # Variables
        h = cp.Variable(n, boolean=True)
        C = cp.Variable()  # upper bound on max mismatch

        # Constraints
        constraints = [cp.sum(h) == target_sum]
        constraints.append(C >= 0)

        # For each shift k, compute the overlap and add constraint
        for k in range(-n+1, n):
            # indices for current shift
            if k >= 0:
                i_indices = range(n - k)
                j_indices = range(k, n)
            else:
                i_indices = range(-k, n)
                j_indices = range(n + k)
            
            # mismatch sum
            mismatch_sum = 0
            for i, j in zip(i_indices, j_indices):
                mismatch_sum += h[i] * (1 - h[j])
            constraints.append(mismatch_sum <= C)

        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        # Try to solve with CBC or SCIP
        try:
            prob.solve(solver=CPLEX, verbose=False)  # Try CPLEX first
        except:
            try:
                prob.solve(solver=SCIP, verbose=False)  # fallback to SCIP
            except:
                try:
                    prob.solve(solver=CBC, verbose=False)  # fallback to CBC
                except:
                    # If all solvers fail, return None to fallback to relaxation
                    return None, None

        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h_val = h.value
            c_val = C.value
            return h_val, c_val
        else:
            return None, None

    # If MILP doesn't finish or fail, do relaxation and rounding
    def solve_relaxation_and_round(n, target_sum):
        # Relax binary constraints
        h = cp.Variable(n, continuous=True)
        C = cp.Variable()

        # Constraints
        constraints = [cp.sum(h) == target_sum]
        constraints.append(0 <= h)
        constraints.append(h <= 1)
        constraints.append(C >= 0)

        # For each shift k, add constraint
        for k in range(-n+1, n):
            if k >= 0:
                i_indices = list(range(n - k))
                j_indices = list(range(k, n))
            else:
                i_indices = list(range(-k, n))
                j_indices = list(range(n + k))
            
            mismatch_sum = 0
            for i, j in zip(i_indices, j_indices):
                mismatch_sum += h[i] * (1 - h[j])
            
            constraints.append(mismatch_sum <= C)

        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=CPLEX, verbose=False)
        except:
            try:
                prob.solve(solver=SCIP, verbose=False)
            except:
                try:
                    prob.solve(solver=CBC, verbose=False)
                except:
                    return None, None

        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h_val = h.value
            c_val = C.value
            # Rounding step
            idxs = np.argsort(h_val)[::-1]  # descending order
            h_rounded = np.zeros(n)
            h_rounded[idxs[:target_sum]] = 1
            return h_rounded, c_val
        else:
            return None, None

    # Fast local search for small post-processing
    def local_improve(h, max_iter=500):
        def evaluate_c5(h):
            h_rev = h[::-1]
            b = 1 - h
            corr = np.correlate(h_rev, b, mode="full")
            return dx * np.max(corr)
        
        best_c5 = evaluate_c5(h)
        h_best = h.copy()
        
        for iter in range(max_iter):
            if time.time() - start_time > 0.9 * budget_s:
                break
            # Try swapping a 1 and a 0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            if len(ones) == 0 or len(zeros) == 0:
                break

            i_swap = np.random.choice(ones)
            j_swap = np.random.choice(zeros)
            
            h_new = h.copy()
            h_new[i_swap] = 0
            h_new[j_swap] = 1
            
            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5:
                h_best = h_new.copy()
                best_c5 = new_c5
                
        return h_best, best_c5

    # Main Flow
    best_c5 = float('inf')
    best_h = None
    remaining_time = budget_s - (time.time() - start_time)

    if remaining_time > 10:
        h_milp, c_milp = solve_milp(n, target_sum)
        if h_milp is not None:
            best_h = h_milp
            best_c5 = c_milp * dx
        else:
            h_lp, c_lp = solve_relaxation_and_round(n, target_sum)
            if h_lp is not None:
                best_h, best_c5 = local_improve(h_lp, max_iter=300)

    # If no MILP solution found and not enough time for relaxation, fallback to method
    if best_h is None or best_c5 == float('inf'):
        h_fallback = np.zeros(n)
        # Simple construction: place 1s and 0s in a way that minimizes overlap
        # Try one that makes few long runs of 1s
        for i in range(0, n, 3):
            if i < n:
                h_fallback[i] = 1
        h_fallback = h_fallback / np.sum(h_fallback) * target_sum  # normalize to exact sum
        h_fallback = np.round(h_fallback).astype(int)
        if np.sum(h_fallback) != target_sum:
            # adjust to match exactly
            diff = target_sum - np.sum(h_fallback)
            if diff > 0:
                indices = np.where(h_fallback == 0)[0]
                for i in range(diff):
                    if i < len(indices):
                        h_fallback[indices[i]] = 1
            elif diff < 0:
                indices = np.where(h_fallback == 1)[0]
                for i in range(-diff):
                    if i < len(indices):
                        h_fallback[indices[i]] = 0
        # keep only integer values
        h_fallback = h_fallback.astype(int)
        best_h = h_fallback
        best_c5 = 0.5  # this is an upper bound; focus on trying valid solutions

    # Final fast check using FFT of best solution
    def fast_evaluate_c5(h):
        h_rev = h[::-1]
        b = 1 - h
        corr = np.correlate(h_rev, b, mode="full")
        return dx * np.max(corr)

    final_c5 = fast_evaluate_c5(best_h)

    # If not enough time left, just use what we have
    return (best_h.tolist(), float(final_c5), n)
```
