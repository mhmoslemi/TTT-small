import numpy as np
from scipy.signal import correlate
import time

def project_to_box_sum(h, target_sum):
    """Project h onto [0,1]^n with sum = target_sum using capped simplex projection."""
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if abs(current_sum - target_sum) < 1e-12:
        return h_clipped
    
    # Binary search for lambda
    a, b = -1.0, 1.0
    for _ in range(50):
        mid = (a + b) / 2
        h_test = np.clip(h + mid, 0, 1)
        test_sum = np.sum(h_test)
        if test_sum > target_sum:
            b = mid
        else:
            a = mid
    
    return np.clip(h + (a + b) / 2, 0, 1)

def compute_correlation_incr(h, b, c, i, j, delta, h_rev):
    """Efficiently update correlation array c for a mass transfer move."""
    # b-part update for h[i] change
    if 0 <= i < len(h):
        start_idx = (len(h) - 1) - i
        end_idx = start_idx + len(h)
        c[start_idx:end_idx] += delta * b[i]
        
    # h-part update for h[j] change
    if 0 <= j < len(h):
        start_idx = j
        end_idx = j + len(h_rev)
        c[start_idx:end_idx] -= delta * h_rev[len(h_rev) - 1 - j::-1]
    
    # Add quadratic correction term (difficult to apply exactly but not needed)
    # add small amount for consistency
    return c

def eval_c5_fast(h, dx):
    """Fast evaluation of C5 using direct correlation."""
    b = 1 - h
    corr = correlate(h, b, mode='full')
    return dx * np.max(corr)
