import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_corr_int(h):
    """Compute correlation as integer values"""
    corr = compute_corr_fft(h)
    return np.rint(corr).astype(int)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def hill_climb(h, max_iter=1000, patience=100):
    """Perform hill climbing to improve solution"""
    best_h = h.copy()
    best_max = compute_corr_int(best_h).max()
    no_improve = 0
    
    for _ in range(max_iter):
        # Generate several random swaps
        swaps = []
        for _ in range(10):
            i_one = np.random.choice(np.where(best_h == 1)[0])
            i_zero = np.random.choice(np.where(best_h == 0)[0])
            swaps.append((i_one, i_zero))
        
        improved = False
        for i_one, i_zero in swaps:
            cand_h = best_h.copy()
            cand_h[i_one] = 0
            cand_h[i_zero] = 1
            cand_max = compute_corr_int(cand_h).max()
            if cand_max < best_max:
                best_h = cand_h
                best_max = cand_max
                improved = True
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= patience:
                break
    
    return best_h, best_max

def simulated_annealing_binary(n_points, time_budget):
    """Simulated annealing with binary variables for improved convergence"""
    start_time = time.time()
    
    # Initial binary pattern
    h = np.zeros(n_points, dtype=int)
    h[:n_points//2] = 1
    np.random.shuffle(h)
    
    best_h = h.copy()
    best_max = compute_corr_int(h).max()
    current_h = h.copy()
    current_max = best_max
    
    # Annealing parameters
    T = 5.0
    cooling_rate = 0.9995
    
    iter_count = 0
    while time.time() - start_time < time_budget:
        # Generate candidate by swapping one 1 and one 0
        ones_idx = np.where(current_h == 1)[0]
        zeros_idx = np.where(current_h == 0)[0]
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
            
        i_one = np.random.choice(ones_idx)
        i_zero = np.random.choice(zeros_idx)
        
        # Create candidate
        cand_h = current_h.copy()
        cand_h[i_one] = 0
        cand_h[i_zero] = 1
        
        # Compute new maximum correlation
        cand_max = compute_corr_int(cand_h).max()
        delta = cand_max - current_max
        
        # Accept or reject based on simulated annealing
        if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, 1e-8)):
            current_h = cand_h
            current_max = cand_max
            
            # Update best if improved
            if cand_max < best_max:
                best_h = cand_h.copy()
                best_max = cand_max
        
        T *= cooling_rate
        iter_count += 1
        
        if iter_count % 1000 == 0:
            # Occasionally adjust temperature
            T = max(T * cooling_rate, 0.001)
    
    return best_h, best_max
