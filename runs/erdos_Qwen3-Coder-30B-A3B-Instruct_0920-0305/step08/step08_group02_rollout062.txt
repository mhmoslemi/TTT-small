import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_sum_box(v, M):
    """Project vector v onto box constraints [0,1] with sum = M"""
    def f(lam):
        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - M

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(v + (a + b)/2, 0, 1)
    return result

def objective_c5(h):
    """Compute C5 using FFT correlation"""
    correlation = correlate(h, 1 - h, mode="full")
    return dx * np.max(correlation)

def solve_milp_binary(N, M, max_seconds=None):
    """Solve MILP for binary h that maximizes minimal autocorrelation"""
    # Variables
    y = cp.Variable(N, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(y) == M]
    
    # Linearize y_i * y_{i+k} for all valid k and i
    w_vars = []  # store w_k vectors
    
    for k in range(1, N):
        # Ensure indices don't exceed N
        indices = range(N - k)
        w_k = cp.Variable(len(indices), nonneg=True)
        w_vars.append(w_k)
        
        # Constraint linking w_k[i] with y[i], y[i + k]
        for i in indices:
            constraints.append(w_k[i] <= y[i])
            constraints.append(w_k[i] <= y[i+k])
            constraints.append(w_k[i] >= y[i] + y[i+k] - 1)
            constraints.append(w_k[i] >= 0)
        
        # Constraint
        constraints.append(cp.sum(w_k) >= t)
    
    # Objective
    prob = cp.Problem(cp.Maximize(t), constraints)
    
    # Try to solve
    try:
        prob.solve(solver=cp.CBC, max_seconds=max_seconds)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURATE]:
            y_sol = y.value
            return y_sol
    except Exception as e:
        pass
    
    # Fallback to other solver
    try:
        prob.solve(solver=cp.GLPK_MI, max_seconds=max_seconds)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURATE]:
            y_sol = y.value
            return y_sol
    except Exception as e:
        pass
    
    # Final fallback
    try:
        prob.solve(solver=cp.ECOS_BB, max_seconds=max_seconds)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURATE]:
            y_sol = y.value
            return y_sol
    except Exception as e:
        pass
    
    return None

def stochastic_refinement(h0, max_iter=2000, budget_s=600, start_time=None):
    """Perform simulated annealing with small random perturbations"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = objective_c5(h)
    
    temp_schedule = lambda t: 1.0 / (1.0 + t / 100.0)
    
    for i in range(max_iter):
        if start_time and time.time() - start_time > budget_s * 0.1:
            break
            
        # Pick two indices to swap
        idx1, idx2 = np.random.choice(len(h), size=2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        h_new = h.copy()
        h_new[idx1] += delta
        h_new[idx2] -= delta
        
        # Clip values back to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to maintain sum constraint
        h_new = project_sum_box(h_new, TARGET_SUM)
        
        new_c5 = objective_c5(h_new)
        
        # Accept if improvement or with probability
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            best_h = h_new.copy()
        else:
            # Accept with simulated annealing probability
            delta_e = new_c5 - best_c5
            temp = temp_schedule(i)
            if np.random.random() < np.exp(-delta_e / temp):
                h = h_new.copy()
    
    return best_h, best_c5

def subgradient_descent(h0, max_iter=20000, budget_s=600, start_time=None):
    """Perform subgradient descent with step size decay"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = objective_c5(h)
    
    # If we had a deterministic descent
    for i in range(max_iter):
        if start_time and time.time() - start_time > budget_s * 0.35:
            break
        
        # Compute correlation and identify k_max (worst shift)
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr) - (len(h) - 1)  # shift index
        
        # Compute gradient for the worst shift k_max
        g = np.ones(len(h))
        
        if k_max >= 0:
            # Subtract h_{i+k_max} for i = 0 to N-k_max-1
            g[:len(h)-k_max] -= h[k_max:]
        else:
            # For negative shifts, subtract h_{i-k_max} = h_{i+|k_max|}
            k = -k_max
            g[k:] -= h[:len(h)-k]
        
        # Handle symmetric influence of both directions
        if k_max > 0:
            g[k_max:] -= h[:len(h)-k_max]
        elif k_max < 0:
            k = -k_max
            g[:len(h)-k] -= h[k:]
        
        # Step size
        alpha = 0.01 / np.sqrt(i + 1)  # Diminishing step size
        
        # Update
        h_new = h - alpha * g
        
        # Project back
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, TARGET_SUM)
        
        new_c5 = objective_c5(h_new)
        
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            best_h = h_new.copy()
        else:
            h = h_new.copy()

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    best_h = None
    best_c5 = float('inf')
    
    # Phase 1: Try to solve MILP
    milp_time = 0.4 * budget_s
    y_sol = solve_milp_binary(n_points, TARGET_SUM, max_seconds=milp_time)
    
    if y_sol is not None:
        h0 = y_sol.astype(float)
        c5_start = objective_c5(h0)
        if c5_start < best_c5:
            best_h = h0.copy()
            best_c5 = c5_start
    else:
        # Fall back to a random feasible initialization if MILP fails
        h0 = np.random.uniform(0, 1, n_points)
        h0 = project_sum_box(h0, TARGET_SUM)
        c5_start = objective_c5(h0)
        if c5_start < best_c5:
            best_h = h0.copy()
            best_c5 = c5_start
    
    # Phase 2: Subgradient descent
    if best_h is not None:
        h_refined, c5_refined = subgradient_descent(best_h, max_iter=20000, budget_s=budget_s, start_time=start_time)
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined

    # Phase 3: Stochastic refinement (smaller portion of time)
    if best_h is not None:
        h_final, c5_final = stochastic_refinement(best_h, max_iter=2000, budget_s=budget_s, start_time=start_time)
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final

    return (best_h.tolist(), float(best_c5), n_points)
