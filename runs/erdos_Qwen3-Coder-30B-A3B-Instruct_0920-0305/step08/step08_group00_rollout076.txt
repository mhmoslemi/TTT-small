import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def random_initialization():
    """Random initialization"""
    h = np.random.uniform(0, 1, n_points)

def hill_climb(h0, max_iter=1000):
    """Simple hill climbing approach"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Try small perturbations
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.01)
        
        h_new = h.copy()
        h_new[idx] += delta
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
    
    return h

def genetic_algorithm(seed_h, n_generations=50, pop_size=30):
    """Genetic algorithm approach"""
    np.random.seed(42)
    
    # Initialize population
    population = [seed_h.copy()]
    for _ in range(pop_size - 1):
        h = seed_h + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    
    for gen in range(n_generations):
        # Evaluate fitness
        fitness = [evaluate_c5(h) for h in population]
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        
        if fitness[sorted_indices[0]] < 0.3809:
            return sorted_pop[0]
            
        # Selection: keep top 30%
        elite = sorted_pop[:pop_size // 3]
        
        # Crossover and mutation
        new_pop = elite.copy()
        while len(new_pop) < pop_size:
            # Select two parents
            parent1 = elite[np.random.randint(0, len(elite))]
            parent2 = elite[np.random.randint(0, len(elite))]
            
            # Crossover
            crossover_point = np.random.randint(0, n_points)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if np.random.random() < 0.05:
                    child[i] += np.random.normal(0, 0.01)
                    child[i] = np.clip(child[i], 0, 1)
            
            child = project_to_sum_box(child, TARGET_SUM)
            new_pop.append(child)
        
        population = new_pop
    
    # Return best individual
    fitness = [evaluate_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    return population[best_idx]
