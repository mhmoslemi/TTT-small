import numpy as np
import time

# Global constants
n_points = 256
dx = 2.0 / n_points
TARGET_SUM = n_points / 2
TIME_LIMIT = 600.0

def project_to_box_sum(h, target_sum):
    """Project h onto the feasible set: 0 <= h[i] <= 1 and sum(h) = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Binary search for lambda
    def f(lam):
        proj = np.clip(h + lam, 0, 1)
        return np.sum(proj) - target_sum

    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    lam = (a + b) / 2
    return np.clip(h + lam, 0, 1)

def compute_correlation(h):
    """Compute full correlation using FFT, return correlation array of length 2*n-1"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return corr

def update_correlation_single(corr, h, b, i, delta):
    """Update correlation in-place after changing h[i] by delta"""
    # b part: corr[ -i : -i + n ] += delta * b
    # h part: corr[ i : i + n ] -= delta * h_rev 
    # Apply slice operations
    if i < 0:
        corr[-i : -i + n] += delta * b[0 : n + i]
    elif i + n > len(corr):
        corr[i : i + n] -= delta * h[: (len(corr) - i)][::-1]
    else:
        corr[i : i + n] -= delta * h[:n][::-1]

def fast_evaluate_c5(h, corr=None):
    """Fast evaluation of C5 using precomputed correlation"""
    if corr is None:
        corr = compute_correlation(h)
    return dx * np.max(corr)
