import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

# Global configuration
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def project_sum_box(v, M):
    """Project vector v onto box [0, 1] with sum = M"""
    v = np.clip(v, 0, 1)
    # Use water-filling algorithm to enforce sum
    def fun(lam):
        new_v = np.clip(v + lam, 0, 1)
        return np.sum(new_v) - M

    # Binary search for lambda
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = fun(m)
        if abs(fm) < 1e-10:
            break
        if fm * fun(a) < 0:
            b = m
        else:
            a = m
    lam = (a + b) / 2
    result = np.clip(v + lam, 0, 1)
    return result

def solve_milp_binary(N, M, timeout_s):
    """
    Solve binary MILP to maximize minimum overlap.
    Returns binary vector y, or None if failed.
    """
    # Variables
    y = cp.Variable(N, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(y) == M]

    # Linearization of y_i * y_{i+k} for all shifts k
    for k in range(1, N):
        indices = range(N - k)
        # Variables for products w_{k,i} = y_i * y_{i+k}
        w_k = cp.Variable(len(indices))
        # Add constraints
        for i in indices:
            constraints += [
                w_k[i] <= y[i],
                w_k[i] <= y[i + k],
                w_k[i] >= y[i] + y[i + k] - 1,
                w_k[i] >= 0
            ]
        # Ensure min overlap constraint
        constraints += [cp.sum(w_k) >= t]

    # Objective
    prob = cp.Problem(cp.Maximize(t), constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.CBC, max_seconds=timeout_s)
    except Exception as e:
        # Fallback to GLPK if CBC fails
        try:
            prob.solve(solver=cp.GLPK_MI, max_seconds=timeout_s)
        except Exception:
            return None
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_EOF]:
        return y.value
    return None

def subgradient_descent(h_start, max_iter=20000):
    """Refine solution using subgradient descent on worst shift."""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for t in range(1, max_iter + 1):
        # Identify worst shift k
        corr = correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr) - (n_points - 1)  # Convert to shift index

        # Compute gradient: g_j = 1 - h_{j+k} - h_{j-k}
        g = np.ones(n_points)
        if k_max > 0:
            # subtract h_{j+k}
            g[k_max:] -= h[:-k_max]
        elif k_max < 0:
            k = -k_max
            g[:k] -= h[k:]
        if k_max >= 0:
            g[:-k_max] -= h[k_max:]
        elif k_max < 0:
            k = -k_max
            g[k:] -= h[:-k]

        alpha = 1e-3 / np.sqrt(t)
        h_new = h - alpha * g
        h_new = project_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
        
    return best_h, best_c5

def stochastic_refinement(h_start, timeout_s):
    """Perform simulated annealing refinement."""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    start_time = time.time()
    
    temp = 0.1  # Initial temperature
    decay_rate = 0.999
    min_temp = 1e-6
    
    while time.time() - start_time < timeout_s / 2.0:
        # Pick two indices
        i, j = np.random.choice(n_points, size=2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        
        # Adjust h
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Clip and project
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, TARGET_SUM)
        
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5)/temp):
            h = h_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                
        temp *= decay_rate
        if temp < min_temp:
            temp = min_temp
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    TARGET_SUM = n_points / 2

    # Phase 1: Try solving the MILP (40% of time)
    milp_timeout = 0.4 * budget_s
    y_solution = None
    h_milp = None
    
    try:
        y_solution = solve_milp_binary(n_points, TARGET_SUM, milp_timeout)
        if y_solution is not None:
            h_milp = y_solution.astype(float)
        else:
            # fallback to random start
            h_milp = np.random.uniform(0.1, 0.9, n_points)
            h_milp = project_sum_box(h_milp, TARGET_SUM)
    except Exception:
        # fallback to random start
        h_milp = np.random.uniform(0.1, 0.9, n_points)
        h_milp = project_sum_box(h_milp, TARGET_SUM)

    # Phase 2: Subgradient descent (35% of remaining time)
    remaining_time = budget_s - (milp_timeout + 0.1)
    if remaining_time > 1.0:
        h_final, c5_best = subgradient_descent(h_milp, max_iter=int(remaining_time * 300))
    else:
        h_final = h_milp
        c5_best = evaluate_c5(h_milp)
    
    # Phase 3: Stochastic refinement
    if time.time() - start_time < budget_s - 1.0:
        random_h = np.random.uniform(0.1, 0.9, n_points)
        random_h = project_sum_box(random_h, TARGET_SUM)
        h_refined, c5_refined = stochastic_refinement(random_h, budget_s - (time.time() - start_time))
        if c5_refined < c5_best:
            h_final, c5_best = h_refined, c5_refined
    
    return (h_final.tolist(), float(c5_best), n_points)
