import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_sum_box(v, M):
    """Project vector v onto sum = M and box [0, 1] using water-filling method."""
    v = np.clip(v, 0, 1)
    v_sum = np.sum(v)
    if abs(v_sum - M) < 1e-10:
        return v
    # Use bisection to find Lagrange multiplier lambda
    a, b = -1.0, 1.0
    def f(lam):
        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - M
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    lam = (a + b) / 2
    result = np.clip(v + lam, 0, 1)
    return result

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def solve_milp_binary(N, M, max_seconds):
    """Solve MILP for binary sequence to maximize minimum autocorrelation."""
    # Variables
    y = cp.Variable(N, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(y) == M]
    
    # Add auxiliary variables and constraints for each shift k
    w_vars = []
    shift_sums = []
    
    for k in range(1, N):
        w_k = cp.Variable(N - k)
        w_vars.append(w_k)
        # Linearization for product
        constraints.append(w_k <= y[:N-k])
        constraints.append(w_k <= y[k:])
        constraints.append(w_k >= y[:N-k] + y[k:] - 1)
        constraints.append(w_k >= 0)
        shift_sum = cp.sum(w_k)
        shift_sums.append(shift_sum)
        constraints.append(shift_sum >= t)
    
    # Objective
    prob = cp.Problem(cp.Maximize(t), constraints)
    
    try:
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=max_seconds)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return y.value
        else:
            return None
    except Exception:
        return None

def subgradient_descent(h0, M, max_iter=20000):
    """Subgradient descent which moves in direction of minimal overlap."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for t in range(1, max_iter + 1):
        # Compute correlation
        corr = compute_corr_fft(h)
        k_max = np.argmax(corr) - (len(h) - 1)  # Convert full index back to shift
        
        # Compute the gradient of F_k_max = M - sum h_i h_{i+k_max}
        g = np.ones_like(h)
        if k_max >= 0:
            if k_max < len(h):
                g[:-k_max] -= h[k_max:]
        else:
            k_abs = -k_max
            if k_abs < len(h):
                g[k_abs:] -= h[:-k_abs]

        # Gradient step size
        alpha = 0.1 / np.sqrt(t)
        
        # Step
        h_new = h - alpha * g
        
        # Project
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, M)
        
        # Evaluate
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        
        h = h_new
    
    return best_h, best_c5

def stochastic_refinement(h0, M, remaining_time):
    """Random perturbations with simulated annealing"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    start_time = time.time()
    
    while time.time() < start_time + remaining_time:
        # Swap two elements
        i, j = np.random.choice(len(h), size=2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Clip to bounds
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, M)
        
        new_c5 = evaluate_c5(h_new)
        
        # Accept with probability based on current time (cooling)
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
        else:
            # Accept with low probability (simulated annealing)
            time_ratio = (time.time() - start_time) / remaining_time
            prob = np.exp(-(new_c5 - best_c5) / (0.01 * (1 - time_ratio + 1e-10)))
            if np.random.random() < prob:
                h = h_new.copy()
    
    return h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    global TARGET_SUM, n_points, dx
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    # Allocate time
    milp_seconds = 0.4 * budget_s
    descent_seconds = 0.35 * budget_s
    refine_seconds = budget_s - (milp_seconds + descent_seconds)
    
    # Phase 1: MILP
    y_binary = None
    try:
        y_binary = solve_milp_binary(n_points, TARGET_SUM, max_seconds=milp_seconds-1)
    except Exception:
        pass

    # Phase 2: Sub-Gradient Descent
    if y_binary is None:
        # Fallback: Random initialization
        h_start = np.random.uniform(0, 1, n_points)
        h_start = project_sum_box(h_start, TARGET_SUM)
    else:
        h_start = y_binary.astype(float)
    
    if time.time() - start_time < descent_seconds:
        h_final, final_c5 = subgradient_descent(h_start, TARGET_SUM, 20000)
    else:
        h_final = h_start.copy()
        final_c5 = evaluate_c5(h_start)
    
    # Phase 3: Stochastic Refinement
    if time.time() - start_time < refine_seconds:
        h_final, final_c5 = stochastic_refinement(h_final, TARGET_SUM, refine_seconds)
    
    return (h_final.tolist(), float(final_c5), n_points)
