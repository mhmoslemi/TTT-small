import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return np.max(corr)

        # For each shift k, we compute the mismatch sum. But to do MILP, we should
        # model this differently. We want to introduce variables w_{i,k} = h_i * (1-h_{i+k})
        # And define C = max_k sum_i w_{i,k} as the bound to minimize.
        
        # Let's also enforce constraints that 0 <= w_{i,k} <= 1
        # Must compute w_{i,k} = h_i * (1 - h_{i+k})
        # This can be modeled via binary constraints:
        # y_{i,k} = h_i * h_{i+k}
        # w_{i,k} = h_i - y_{i,k}
        # So:
        # y_{i,k} >= h_i + h_{i+k} - 1
        # y_{i,k} <= h_i
        # y_{i,k} <= h_{i+k}
        # and w_{i,k} = h_i - y_{i,k}
        
        # We'll formulate the constraint using Variables for each shifted pair:
        # y[k] = [y_{i,k}] (when we look at i+k mod n)
        # This seems laborious to express in cvxpy syntax.

        # Better approach:
        # Introduce a scalar C, and for each k, enforce:
        # sum_i h_i * (1 - h_{i+k}) <= C
        C = cp.Variable()
        for k in range(-n+1, n):
            # Purpose: For each k, create sum_{i} h_i * (1 - h_{i+k}) <= C
            # That is: sum_{i} h_i - sum_{i} h_i * h_{i+k} <= C
            # Let p_{i,k} = h_i * h_{i+k}, then: sum_i h_i - sum_i p_{i,k} <= C
            # But we don't have access to p_{i,k}. Let's reconsider.
            # Set: w_{i,k} = h_i * (1 - h_{i+k})
            # So: w_{i,k} <= h_i
            # w_{i,k} <= 1 - h_{i+k}
            # w_{i,k} >= h_i + 1 - h_{i+k} - 1 = h_i - h_{i+k}
            # But also, w_{i,k} >= 0
            # A better formulation for w_{i,k} = h_i - y_{i,k}, where y_{i,k} = h_i * h_{i+k}
            # sum_i w_{i,k} <= C for each k
            # Let's write down the model more carefully and compute queries...
            # Build the constraint to evaluate:
            # Note: we're dropping the FFT path for stability.
            
            expr = 0.0
            for i in range(n):
                j = (i + k) % n
                if k < 0:
                    # j = i + k (can be negative)
                    j = i + k
                    if j < 0:
                        j += n
                else:
                    # j = i + k
                    if j >= n:
                        j -= n
                expr += h[i] * (1 - h[j])

        try:

                return h_solved
        except:

        # Reformulate to LP, then solve and round.

        constraints += h_bounds

                h_rounded = [0.0] * n
                for i in range(n//2):
