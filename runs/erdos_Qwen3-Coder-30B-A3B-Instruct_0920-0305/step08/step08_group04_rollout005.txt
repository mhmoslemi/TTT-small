import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import time

# Global constants
n_points = 300
dx = 2.0 / n_points
TARGET_SUM = n_points / 2
n_coarse = 150

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling projection using sorting
    x_sorted = np.sort(x)[::-1]
    cumsum = np.cumsum(x_sorted)
    # Find the threshold
    threshold_idx = np.argmax(cumsum > target_sum)
    if threshold_idx == 0:
        # All elements can be at their maximum
        if cumsum[-1] <= target_sum:
            # All can be 1.0
            projection = np.ones_like(x)
        else:
            # Some need to be reduced
            lambda_val = (cumsum[-1] - target_sum) / len(x_sorted)
            projection = np.clip(x_sorted - lambda_val, 0, 1)
    else:
        # Compute lambda such that sum(projected) = target_sum
        # First find how many are at their maximum
        # Then find lambda for remaining terms
        if threshold_idx == len(x_sorted):
            # All elements should be scaled to meet target_sum
            lambda_val = (cumsum[-1] - target_sum) / len(x_sorted)
            projection = np.clip(x_sorted - lambda_val, 0, 1)
        else:
            # Threshold_idx-1 elements are at max 1, remaining are at lambda
            # At threshold_idx, cumsum[threshold_idx-1] > target_sum
            # And cumsum[threshold_idx-2] <= target_sum
            # So last threshold_idx-1 elements are set to 1.0
            # The remaining elements are set to a common value
        
            total_sum_of_ones = threshold_idx
            remaining_sum = target_sum - total_sum_of_ones
            
            # We want to make first (n - threshold_idx) elements 
            # equal to remaining_sum/(n - threshold_idx)
            lambda_val = remaining_sum / (len(x_sorted) - threshold_idx)
            
            projection = np.ones_like(x_sorted)
            projection[threshold_idx:] = np.clip(lambda_val, 0, 1)
    
    # Sort back to original order
    projection = projection[np.argsort(-x)]
    return projection

def rasterise_and_project(θ):
    """Convert a parametrized vector to a proper step function satisfying constraints"""
    K = 8
    # Split the parameters into widths and heights
    widths = np.exp(θ[:K])
    widths = 2 * widths / np.sum(widths)
    heights = 1.0 / (1.0 + np.exp(-θ[K:]))
    
    # Roughly determine where each width applies
    cum_widths = np.concatenate(([0], np.cumsum(widths)))
    
    # Create piecewise constant function
    h = np.zeros(n_points)
    
    # Instead of block assignment, we do assignment based on where x falls
    xgrid = np.linspace(dx/2, 2-dx/2, n_points)
    
    for i in range(K):
        start_idx = int(cum_widths[i] / 2 * n_points)
        end_idx = int(cum_widths[i+1] / 2 * n_points)
        h[start_idx:end_idx] = heights[i]
    
    # Ensure the heights are in range and sum constraint is met
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def obj(θ):
    h = rasterise_and_project(θ)
    c5 = evaluate_c5(h)
    return c5

    initial_h = np.array(initial_h_values)
    x_orig = np.linspace(0, 2, len(initial_h))
    x_coarse = np.linspace(0, 2, n_coarse)
    h_interpolated = np.interp(x_coarse, x_orig, initial_h)
    h_interpolated = project_to_sum_box(h_interpolated, n_coarse / 2)
    
    # Now check if the current fine resolution version is better since

        c5 = evaluate_c5(h)

    return rasterise_and_project(res.x)

def run_psgd(h_start, max_iter=5000):
    """Run projected sub-gradient descent from an initial point"""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Initialize time tracking
    start_time = time.time()
    
    # For mass transfer
    n_init = len(h)
    T_swap = 30
    N_noimp = 200
    no_improvement_count = 0
    
    for it in range(max_iter):
        if time.time() - start_time > 450:
            break
        # Compute correlation to find active shifts
        corr = np.real(np.fft.ifft(np.fft.fft(h[::-1]) * np.fft.fft(1 - h))) * dx
        max_val = corr.max()
        active = np.where(corr >= max_val - 1e-12)[0]
        shifts = active - (n_points - 1)
        
        # Assemble gradient vector
        g = np.zeros(n_points)
        for s in shifts:
            if s >= 0:
                # g[i] += -dx * h[i+s] - dx * h[i-s]
                g[:n_points-s] += -dx * h[s:]
                g[s:] += -dx * h[:n_points-s]
            else:
                # s < 0
                t = -s
                g[t:] += -dx * h[:n_points-t]
                g[:n_points-t] += -dx * h[t:]
        
        # Average
        g /= len(shifts)
        
        # Descent direction
        d = -g
        
        # Adaptive step size
        eta = 0.5 if 'eta' not in locals() else eta
        eta = max(1e-4, min(eta, 2.0))
        
        # Trial step
        h_trial = h + eta*d
        h_trial = project_to_sum_box(h_trial, TARGET_SUM)
        c5_trial = evaluate_c5(h_trial)
        
        if c5_trial < best_c5:
            h = h_trial
            best_h = h_trial.copy()
            best_c5 = c5_trial
            no_improvement_count = 0
            # Increase step size
            eta = min(2.0, 1.05 * eta)
        else:
            # Decrease step size
            eta = max(1e-4, 0.7 * eta)
            no_improvement_count += 1
        
        # Periodic mass transfer (random subset swaps)
        if it % T_swap == 0 or no_improvement_count > N_noimp:
            # Find valid indices
            valid_high = np.where(h > 0)[0]
            valid_low = np.where(h < 1)[0]
            
            if len(valid_high) > 0 and len(valid_low) > 0:
                i1 = np.random.choice(valid_high)
                i2 = np.random.choice(valid_low)
                
                transfer_amount = min(0.02, h[i1]) * np.random.rand()
                h[i1] = max(0, h[i1] - transfer_amount)
                h[i2] = min(1, h[i2] + transfer_amount)
                
                # Re-project
                h = project_to_sum_box(h, TARGET_SUM)
                c5_new = evaluate_c5(h)
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
                    
    return best_h, best_c5

    h_initial = np.array(initial_h_values)

    h_interpolated = project_to_sum_box(h_interpolated, TARGET_SUM)
    c5_initial = evaluate_c5(h_interpolated)
    
    # Global search on the coarser grid
    h_coarse = coarse_global_search()
    h_coarse_refined, c5_coarse = run_psgd(h_coarse, max_iter=500)
    
    # Upsample to fine grid
    x_coarse = np.linspace(dx/2, 2-dx/2, n_coarse)
    x_fine = np.linspace(dx/2, 2-dx/2, n_points)
    h_upsampled = np.interp(x_fine, x_coarse, h_coarse_refined)
    h_fine_start = project_to_sum_box(h_upsampled, TARGET_SUM)
    
    # Run local optimization on the fine grid
    h_best, c5_best = run_psgd(h_fine_start, max_iter=8000)
    
    return (h_best.tolist(), float(c5_best), n_points)
