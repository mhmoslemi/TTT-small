```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Generate initial solution using a piecewise constant approach
    h_init = np.zeros(n_points)
    
    # Create a more structured approach with different regions
    # Place values in a few key regions to avoid sharp transitions
    segments = 5
    segment_size = n_points // segments
    
    for i in range(segments):
        start = i * segment_size
        end = (i+1) * segment_size if i < segments-1 else n_points
        if i % 2 == 0:
            h_init[start:end] = 0.7
        else:
            h_init[start:end] = 0.3
    
    # Ensure sum matches target
    h_init = project_to_sum_box(h_init, n_points/2)
    
    best_h = h_init.copy()
    best_c5 = evaluate_c5(h_init)
    TARGET_SUM = n_points / 2

    # Try several optimization approaches
    # Approach 1: Gradient-based optimization with constraints
    current_h = h_init.copy()
    
    # Local search with simulated annealing inspired approach
    T = 1.0
    cooling_rate = 0.999
    min_temperature = 1e-6
    
    while time.time() - start_time < budget_s * 0.8 and T > min_temperature:
        # Make a small random perturbation
        h_candidate = current_h + np.random.normal(0, 0.01, n_points)
        h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
        
        # Evaluate
        c5_candidate = evaluate_c5(h_candidate)
        
        # Accept with probability based on simulated annealing
        if c5_candidate < best_c5 or np.random.rand() < np.exp(-(c5_candidate - best_c5) / T):
            current_h = h_candidate.copy()
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = current_h.copy()
        
        T *= cooling_rate
    
    # Refine with optimization from best solution
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            best_h.copy(), 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12, 'disp': False}
        )
        if res.success:
            h_final = project_to_sum_box(res.x, TARGET_SUM)
            c5_final = evaluate_c5(h_final)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
    except:
        pass
    
    # Final local search with random restarts
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
            
        # Initialize with some random perturbations
        h_local = best_h + np.random.normal(0, 0.005, n_points)
        h_local = project_to_sum_box(h_local, TARGET_SUM)
        
        try:
            res = minimize(
                objective, 
                h_local.copy(), 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12, 'disp': False}
            )
            if res.success:
                h_candidate = project_to_sum_box(res.x, TARGET_SUM)
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
        except:
            continue

    return (best_h.tolist(), float(best_c5), n_points)
```
