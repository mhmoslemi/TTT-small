```python
import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

# Global parameters
n_points = 180  # Grid resolution
dx = 2.0 / n_points
TARGET_SUM = 1.0  # Total mass constraint
m_intervals = 5   # Number of intervals (to be tuned)
MAX_EVALUATIONS = 5000  # Total allowed function evaluations
epsilon = 1e-4  # To prevent zero-length intervals

def decode_intervals(z):
    """Convert parameter vector z to interval positions and lengths."""
    # z is a vector of length 2*m_intervals, corresponding to (l1,...,lm, g1,...,gm)
    # We map z[i] in [0,1] to l_i in [epsilon, 1-epsilon] 
    # and g_i in [0, 1] scaled so that total goes to 2
    l = epsilon + (1 - 2*epsilon) * z[:m_intervals]
    l = l / np.sum(l)  # Normalize lengths to sum to 1
    
    # Compute positions
    s = np.zeros(m_intervals)
    e = np.zeros(m_intervals)
    s[0] = 0
    e[0] = s[0] + l[0]
    
    for i in range(1, m_intervals):
        s[i] = e[i-1] + z[m_intervals + i - 1]  # g_{i-1} in [0,1] scaled
        e[i] = s[i] + l[i]
    
    # Ensure last endpoint does not exceed 2
    total_length = e[-1]
    if total_length > 2:
        # Scale gaps to fit
        scale_factor = 2 / total_length
        s = s * scale_factor
        e = e * scale_factor
        # Adjust the lengths accordingly (but we keep original l)
        
    # Add last gap to bring end to exactly 2
    if total_length < 2:
        gap_needed = 2 - total_length
        # Add proportionally to last gap
        l_last = l[-1]
        s_last = s[-1]
        e_last = e[-1]
        s[-1] = s_last
        e[-1] = 2.0
        l[-1] = e[-1] - s[-1]
        if l[-1] < epsilon:
            l[-1] = epsilon
            s[-1] = e[-1] - l[-1]
    
    return s, e, l

def build_grid_from_intervals(s, e, n):
    """Build binary step function from interval endpoints."""
    h = np.zeros(n)
    grid_points = np.linspace(0, 2, n, endpoint=False) + dx/2
    
    for i in range(len(s)):
        mask = (grid_points >= s[i]) & (grid_points < e[i])
        h[mask] = 1
    
    return h

def evaluate_c5_fft(h):
    """Fast evaluation of C5 using FFT correlation."""
    # Note: This exact routine is also used in extern grading
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def objective(z):
    """Objective function: minimize C5."""
    try:
        s, e, l = decode_intervals(z)
        h = build_grid_from_intervals(s, e, n_points)
        c5 = evaluate_c5_fft(h)
    except:
        c5 = 1e10  # Invalid solution
    return c5

def local_refinement(best_z):
    """Perform gradient-free local search on interval positions."""
    max_iters = 3000
    best_c5 = objective(best_z)
    best_params = best_z.copy()
    
    # Get current intervals from params
    s, e, l = decode_intervals(best_params)
    
    # Try moving each edge slightly
    for _ in range(max_iters):
        improved = False
        for idx in range(m_intervals):
            # Try shifting left edge
            if s[idx] > 0:
                z_new = best_params.copy()
                z_new[idx] -= 0.001
                try:
                    c5_new = objective(z_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_params = z_new.copy()
                        improved = True
                except:
                    pass
            # Try shifting right edge  
            if e[idx] < 2:
                z_new = best_params.copy()
                z_new[idx + m_intervals] += 0.001  # g index
                try:
                    c5_new = objective(z_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_params = z_new.copy()
                        improved = True
                except:
                    pass
        if not improved:
            break  # No improvement in this run
            
    return best_params, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    remaining_budget = budget_s
    
    # Phase 1: Random initialization
    phase1_end = start_time + 0.1 * budget_s
    best_random = float('inf')
    best_random_z = None
    for _ in range(500):  # Initial random sampling to get start point
        if time.time() > phase1_end:
            break
        z = np.random.rand(2 * m_intervals)
        c5 = objective(z)
        if c5 < best_random:
            best_random = c5
            best_random_z = z.copy()
            
    # Phase 2: Differential evolution for global search
    phase2_end = start_time + 0.7 * budget_s
    
    # Set up bounds for DE
    bounds = [(0, 1)] * (2 * m_intervals)
    
    try:
        result = differential_evolution(
            objective,
            bounds,
            maxiter=10000,
            popsize=30,
            mutation=(0.5, 1.0),
            recombination=0.7,
            seed=seed,
            callback=lambda x, convergence: None,
            workers=1,
            updating='immediate'
        )
        
        if not result.success:
            best_z = best_random_z
        else:
            best_z = result.x

        # Rerun the best non-random solution with full refinement
        best_z_refined, final_c5 = local_refinement(best_z)
        if final_c5 < best_c5:
            best_c5 = final_c5
            s, e, l = decode_intervals(best_z_refined)
            best_h = build_grid_from_intervals(s, e, n_points)
    except:
        # Fallback to best random solution  
        if best_random_z is not None:
            s, e, l = decode_intervals(best_random_z)
            best_h = build_grid_from_intervals(s, e, n_points)
            best_c5 = evaluate_c5_fft(best_h)

    # Phase 3: Final local refinement
    phase3_end = start_time + 0.9 * budget_s
    if best_h is not None:
        final_z, refined_c5 = local_refinement(best_z_refined)
        final_s, final_e, final_l = decode_intervals(final_z)
        best_h = build_grid_from_intervals(final_s, final_e, n_points)
        best_c5 = evaluate_c5_fft(best_h)

    # Phase 4: Fallback check
    if best_c5 >= 0.38092:
        # Generate many random trials 
        best_rand_c5 = float('inf')
        best_rand_h = None
        for _ in range(3000):  # Enough to sample good ones
            z = np.random.rand(2 * m_intervals)
            try:
                s, e, l = decode_intervals(z)
                h = build_grid_from_intervals(s, e, n_points)
                c5 = evaluate_c5_fft(h)
                if c5 < best_rand_c5:
                    best_rand_c5 = c5
                    best_rand_h = h
            except:
                continue
                
        if best_rand_c5 < best_c5:
            best_c5 = best_rand_c5
            best_h = best_rand_h
            
    # Final verification with np.correlate
    if best_h is not None:
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = dx * np.max(corr.real)
        best_c5 = final_c5

    return (best_h.tolist(), float(best_c5), n_points)
```
