def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Reverse h for correlation
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr)

            # Add constraint: m_k <= C
