import time

# Global variables
n_points = 200
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Evaluate C5 using correlation"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def evaluate_c5_fft(h):
    """Fast evaluation using FFT"""
    b = 1 - h
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def neighbor_sum(h, lag):
    """Compute neighbor sum w_i = h[i+lag] + h[i-lag] for valid indices"""
    lag_abs = abs(lag)
    if lag_abs >= len(h):
        return np.zeros_like(h)
    
    w = np.zeros_like(h)
    if lag >= 0:
        w[:len(h)-lag_abs] += h[lag_abs:]
        w[lag_abs:] += h[:len(h)-lag_abs]
    else:
        w[lag_abs:] += h[:-lag_abs]
        w[:-lag_abs] += h[lag_abs:]
    return w

def mass_transfer_step(h, step_size, eps=1e-6):
    """Perform one mass transfer step to improve C5"""
    # Compute full correlation
    corr = np.correlate(h, 1 - h, mode='full')
    idx_max = np.argmax(corr)
    lag_max = idx_max - (len(h) - 1)
    
    # Compute neighbor sum for the active lag
    w = neighbor_sum(h, lag_max)
    
    # Find indices with low and high w values that satisfy bounds
    low_indices = np.where(h > eps)[0]
    high_indices = np.where(h < 1 - eps)[0]
    
    if len(low_indices) == 0 or len(high_indices) == 0:
        return h, False
    
    # Find low w index with highest h value (needs to be > epsilon)
    low_candidates = low_indices[np.argsort(w[low_indices])]
    low_idx = None
    for idx in low_candidates:
        if h[idx] > eps:
            low_idx = idx
            break
            
    # Find high w index with lowest h value (needs to be < 1-epsilon)
    high_candidates = high_indices[np.argsort(-w[high_indices])]
    high_idx = None
    for idx in high_candidates:
        if h[idx] < 1 - eps:
            high_idx = idx
            break
    
    if low_idx is None or high_idx is None:
        return h, False
    
    # Compute transfer amount
    delta = min(step_size, h[low_idx], 1 - h[high_idx])
    if delta < 1e-12:
        return h, False
    
    # Perform transfer
    h_new = h.copy()
    h_new[low_idx] -= delta
    h_new[high_idx] += delta
    
    return h_new, True
