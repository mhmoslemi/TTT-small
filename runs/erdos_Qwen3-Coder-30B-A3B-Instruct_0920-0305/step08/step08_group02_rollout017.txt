import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from scipy.fft import fft, ifft

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def clip_and_project(h, target_sum):
    """Project vector h onto box constraints [0,1] with sum = target_sum using water-filling"""
    h = np.clip(h, 0, 1)
    def project_sum(lam):
        proj = np.clip(h + lam, 0, 1)
        return np.sum(proj) - target_sum
        
    # Bisection search for lambda
    a, b = -1.0, 1.0
    for _ in range(50):
        mid = (a + b) / 2
        if abs(project_sum(mid)) < 1e-10:
            break
        if project_sum(mid) * project_sum(a) < 0:
            b = mid
        else:
            a = mid
    lam = (a + b) / 2
    result = np.clip(h + lam, 0, 1)
    return result

def binary_milp_initialization():
    """Use MILP to find binary configuration with good autocorrelation"""
    global n_points, dx
    
    try:
        # Create binary variables
        y = cp.Variable(n_points, boolean=True)
        # Auxiliary variables for product linearization
        w = {}
        t = cp.Variable()

        # Maximum autocorrelation constraint
        constraints = [cp.sum(y) == n_points // 2, t >= 0]
        # Product linearization constraints
        for k in range(1, n_points):
            w[k] = cp.Variable(n_points - k, continuous=True)
            for i in range(n_points - k):
                constraints.append(w[k][i] <= y[i])
                constraints.append(w[k][i] <= y[i + k])
                constraints.append(w[k][i] >= y[i] + y[i + k] - 1)
                constraints.append(w[k][i] >= 0)
        # Minimum autocorrelation constraints
        for k in range(1, n_points):
            constraints.append(cp.sum(w[k]) >= t)
        
        # Objective
        obj = cp.Maximize(t)
        prob = cp.Problem(obj, constraints)
        
        # Solve with time limit
        prob.solve(solver=cp.CBC, max_seconds=0.4 * 600)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            y_opt = y.value
            return y_opt
        else:
            raise RuntimeError("MILP failed")
    except:
        return None

def greedy_binary_initialization():
    """Greedy initialization for binary configuration"""
    y = np.zeros(n_points, dtype=int)
    placed = 0
    # Place ones with spacing to reduce overlap
    spacing = n_points // (n_points // 2)
    for i in range(0, n_points, spacing):
        if placed < n_points // 2:
            y[i] = 1
            placed += 1
        if placed >= n_points // 2:
            break
    return y.astype(float)

def compute_subgradient(h, worst_shift):
    """Compute subgradient for the worst shift"""
    g = np.zeros_like(h)
    if worst_shift < len(h):
        g += 1.0
        g[:len(h)-worst_shift] -= h[worst_shift:]
        g[worst_shift:] -= h[:-worst_shift]
    else:
        g += 1.0
        g[:len(h)-worst_shift] -= h[worst_shift:]
        g[worst_shift:] -= h[:-worst_shift]
        # For the wrap-around (the last few elements)
        g = g - np.roll(h, worst_shift)
    return g

def subgradient_descent(h0, max_time):
    """Perform deterministic subgradient descent"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    t = 1
    alpha = 0.5 * dx
    
    while True:
        # Evaluate on current solution
        corr = compute_corr_fft(h)
        worst_shift = np.argmax(corr)
        # Compute subgradient
        g = compute_subgradient(h, worst_shift)
        # Update step
        h_new = h - (alpha / np.sqrt(t)) * g
        # Project
        h_new = clip_and_project(h_new, TARGET_SUM)
        # Evaluate new solution
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
        h = h_new
        t += 1
        
        if time.time() - start_time > max_time:
            break
            
    return best_h, best_c5

def swap_move(h, temp):
    """Perform a 2-index swap move with Metropolis acceptance"""
    h_new = h.copy()
    i, j = np.random.choice(n_points, 2, replace=False)
    delta = np.random.uniform(-0.02, 0.02)
    h_new[i] += delta
    h_new[j] -= delta
    h_new = clip_and_project(h_new, TARGET_SUM)
    c5_new = evaluate_c5(h_new)
    c5_old = evaluate_c5(h)
    delta_c5 = c5_new - c5_old
    
    if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temp):
        return h_new, c5_new
    else:
        return h, c5_old

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    h_best = None
    c5_best = float('inf')
    
    # Phase 1: Binary optimization (MILP or greedy)
    phase1_end = start_time + 0.4 * budget_s
    try:
        h_binary = binary_milp_initialization()
        if h_binary is None:
            h_binary = greedy_binary_initialization()
    except:
        h_binary = greedy_binary_initialization()
        
    h_binary = clip_and_project(h_binary, TARGET_SUM)
    c5_initial = evaluate_c5(h_binary)
    if c5_initial < c5_best:
        h_best = h_binary.copy()
        c5_best = c5_initial
        
    # Phase 2: Subgradient descent
    phase2_end = start_time + 0.75 * budget_s
    h_subgrad, c5_subgrad = subgradient_descent(h_binary, max_time=phase2_end - start_time)
    if c5_subgrad < c5_best:
        h_best = h_subgrad.copy()
        c5_best = c5_subgrad
        
    # Phase 3: Stochastic refinement
    phase3_end = start_time + budget_s
    temp = 5 * dx
    temp_decay = (temp - 0) / (phase3_end - phase2_end)
    
    while time.time() < phase3_end:
        h_new, c5_new = swap_move(h_best, temp)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
        temp = max(0, temp - temp_decay)
        
    return (h_best.tolist(), float(c5_best), n_points)
