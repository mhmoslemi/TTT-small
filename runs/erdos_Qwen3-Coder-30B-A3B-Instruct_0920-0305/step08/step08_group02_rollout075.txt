import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize
from cvxpy import CBC, GLPK_MI, ECOS_BB

def compute_corr_fft(h):
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    return np.max(corr)

def project_sum_box(h, target_sum):
    def f(lam):
        projected = np.clip(h + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def binary_milp_solution(N, M, max_time):
    # Define variables - y[i] are binary, w[k,i] are continuous
    y = cp.Variable(N, boolean=True)
    w = cp.Variable((N, N), nonneg=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.sum(y) == M,
        t <= M
    ]

    # Product constraints for all valid shifts
    for k in range(1, N):
        for i in range(N - k):
            # w[k,i] is constrained to be y[i]*y[i+k]
            constraints.extend([
                w[i, k] <= y[i],
                w[i, k] <= y[i + k],
                w[i, k] >= y[i] + y[i + k] - 1,
                w[i, k] >= 0
            ])
            # The sum for this shift must be at least t
            constraints.append(cp.sum(w[:, k]) >= t)

    # Objective
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)

    # Try different solvers in order
    solver = None
    try:
        prob.solve(solver=CBC, max_seconds=max_time - 0.1)
        solver = CBC
    except:
        try:
            prob.solve(solver=GLPK_MI, max_seconds=max_time - 0.1)
            solver = GLPK_MI
        except:
            try:
                prob.solve(solver=ECOS_BB, max_seconds=max_time - 0.1)
                solver = ECOS_BB
            except:
                return None, None

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        y_sol = y.value.astype(int)
        t_sol = t.value
        return y_sol, t_sol
    return None, None

def evaluate_c5(h):
    # Use direct correlation evaluation like in the initial code
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def gradient_step(h, k_star, dx):
    # Compute the subgradient for the active shift k_star
    g = np.zeros_like(h)
    for i in range(len(h)):
        left_idx = i - k_star
        right_idx = i + k_star
        if left_idx >= 0:
            g[i] += h[left_idx]
        if right_idx < len(h):
            g[i] += h[right_idx]
        g[i] = 1 - g[i]
    return g

def subgradient_descent(h_start, steps, dx, find_t_max=True):
    h = h_start.copy()
    best_c5 = evaluate_c5(h)

    for t in range(steps):
        # Find the shift k* that maximizes F_k = M - sum h_i * h_{i+k}
        corr = compute_corr_fft(h)
        M = len(h)/2.0
        F = M - corr[corr.size//2:]  # Only positive shifts
        k_star = np.argmax(F)
        g = gradient_step(h, k_star, dx)
        # Take a step
        alpha = 0.5 * dx / np.sqrt(t+1)
        h_new = h - alpha * g
        h_new = project_sum_box(h_new, len(h)/2)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
        else:
            # With probability, accept worse step to escape plateaus
            import math
            if math.exp(-(c5_new - best_c5)/0.01) > np.random.random():
                h = h_new.copy()
                best_c5 = c5_new
    return h, best_c5

def swap_move(h, target_sum, dx):
    # Do a swap between two positions
    i, j = np.random.choice(len(h), 2, replace=False)
    delta = np.random.uniform(-0.02, 0.02)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    # Ensure bounds
    h_new = np.clip(h_new, 0, 1)
    h_new = project_sum_box(h_new, target_sum)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    N = 120
    dx = 2.0 / N
    M = N / 2.0

    # Phase 1: Binary MILP optimization
    h = np.zeros(N)
    best_c5 = float('inf')
    y_best = None
    t_opt = None
    try:
        # Remaining time for MILP
        remaining = budget_s - (time.time() - start_time)
        if remaining > 1:
            y_best, t_opt = binary_milp_solution(N, M, remaining)
            if y_best is not None:
                h = y_best.astype(float)
                best_c5 = dx * (M - t_opt)
    except:
        pass

    # Phase 2: Subgradient descent refinement
    if best_c5 >= 1.0 or y_best is None:
        # Start with a random feasible point if MILP failed
        h = np.random.uniform(0, 1, N)
        h = project_sum_box(h, M)

    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        h_refined, c5_refined = subgradient_descent(h, max(500, int(0.3 * remaining_time)), dx)
        if c5_refined < best_c5:
            h = h_refined.copy()
            best_c5 = c5_refined

    # Phase 3: Stochastic fine-tuning
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5:
        t_start = time.time()
        while time.time() - t_start < remaining_time - 2:
            h_new = swap_move(h, M, dx)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                h = h_new.copy()
                best_c5 = c5_new

    return (h.tolist(), float(best_c5), N)
