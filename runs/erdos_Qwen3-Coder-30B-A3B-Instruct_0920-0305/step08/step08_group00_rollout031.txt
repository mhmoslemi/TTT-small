import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_corr_int(h):
    """Compute integer correlation counts using FFT"""
    corr = compute_corr_fft(h)
    return np.round(corr).astype(int)

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def simulated_annealing(n_points, time_budget):
    """Simulated annealing optimization for binary pattern"""
    start_time = time.time()
    
    # Generate initial binary pattern
    h = np.zeros(n_points, dtype=int)
    h[:n_points//2] = 1
    np.random.shuffle(h)
    
    best_h = h.copy()
    best_corr_int = compute_corr_int(best_h)
    best_max = best_corr_int.max()
    current_h = h.copy()
    current_max = best_max
    
    # Temperature schedule
    T0 = 2.0
    T = T0
    cooling = 0.9995
    
    # Main annealing loop
    while time.time() - start_time < time_budget:
        # Generate random swap
        ones_idx = np.where(current_h == 1)[0]
        zeros_idx = np.where(current_h == 0)[0]
        if len(ones_idx) > 0 and len(zeros_idx) > 0:
            i_one = np.random.choice(ones_idx)
            i_zero = np.random.choice(zeros_idx)
            
            # Create candidate
            cand_h = current_h.copy()
            cand_h[i_one] = 0
            cand_h[i_zero] = 1
            
            # Evaluate
            cand_corr_int = compute_corr_int(cand_h)
            cand_max = cand_corr_int.max()
            delta = cand_max - current_max
            
            # Accept or reject
            if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, 1e-6)):
                current_h = cand_h
                current_max = cand_max
                
                # Update best if improved
                if cand_max < best_max:
                    best_h = cand_h.copy()
                    best_max = cand_max
                    
        T *= cooling
    
    return best_h, best_max

def hill_climb(h, max_iter=1000, patience=100):
    """Local hill climbing improvement"""
    best_h = h.copy()
    best_max = compute_corr_int(best_h).max()
    no_improve = 0
    
    for _ in range(max_iter):
        # Generate random swaps
        swaps = []
        for _ in range(20):
            if len(np.where(best_h==1)[0]) > 0 and len(np.where(best_h==0)[0]) > 0:
                i_one = np.random.choice(np.where(best_h==1)[0])
                i_zero = np.random.choice(np.where(best_h==0)[0])
                swaps.append((i_one, i_zero))
        
        improved = False
        for i_one, i_zero in swaps:
            cand_h = best_h.copy()
            cand_h[i_one] = 0
            cand_h[i_zero] = 1
            cand_max = compute_corr_int(cand_h).max()
            if cand_max < best_max:
                best_h = cand_h
                best_max = cand_max
                improved = True
                break
                
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= patience:
                break
                
    return best_h, best_max

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points / 2
    dx = 2.0 / n_points
    
    # Phase 1: Multiple simulated annealing runs
    best_h = None
    best_c5 = float('inf')
    
    # Run multiple seeds to avoid local optima
    for i in range(20):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Set different random seed for each iteration
        np.random.seed(seed + i)
        
        # Run temperature-based simulated annealing
        time_left = budget_s * 0.8 - (time.time() - start_time)
        if time_left <= 1:
            break
            
        h, corr_max = simulated_annealing(n_points, time_left * 0.8)
        
        # Evaluate C5
        c5 = c5_from_corr(compute_corr_fft(h), n_points)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
    if best_h is None:
        # Fallback to random pattern
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
        np.random.shuffle(h)
        best_h = h
        best_c5 = evaluate_c5(h)
    
    # Phase 2: Final local refinement
    time_left = budget_s - (time.time() - start_time)
    if time_left > 1:
        refined_h, refined_c5 = hill_climb(best_h, max_iter=500, patience=50)
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
