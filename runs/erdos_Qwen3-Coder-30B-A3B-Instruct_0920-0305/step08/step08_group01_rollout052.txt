import numpy as np

def evaluate_erdos_solution(h):
    """Evaluate C5 for a discretized h"""
    n = len(h)
    dx = 2.0 / n
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    C5 = dx * np.max(corr)
    return C5

def compute_correlations(h):
    """Compute all correlations efficiently"""
    n = len(h)
    dx = 2.0 / n
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def generate_shifts(n):
    """Generate all valid shifts for correlation"""
    return list(range(-(n-1), n))

def generate_milp_model(n, shifts):
    """Generate MILP model for the problem"""
    x = Variable(n, boolean=True)
    t = Variable()
    
    # Mass constraint
    constraints = [sum(x) == n / 2, t >= 0]
    
    # Linearization of products
    y_vars = {}
    for s in shifts:
        y_vars[s] = {}
        for i in range(n):
            if 0 <= i + s < n:
                y = Variable(boolean=True)
                y_vars[s][i] = y
                constraints.extend([
                    y <= x[i],
                    y <= x[i+s],
                    y >= x[i] + x[i+s] - 1
                ])
    
    # Epigraph constraints
    for s in shifts:
        expr = sum(y_vars[s][i] for i in range(n) if 0 <= i + s < n)
        expr = expr - n/2 + t
        constraints.append(expr >= 0)
    
    objective = Minimize(t)
    problem = Problem(objective, constraints)
    return problem, x, t, y_vars

def solve_milp(problem, time_limit=30):
    """Solve the MILP with time limit"""
    try:
        problem.solve(solver=CPLEX, time_limit=time_limit)
        return True
    except Exception:
        try:
            problem.solve(solver=GLPK_MI, time_limit=time_limit)
            return True
        except Exception:
            try:
                problem.solve(solver=SCIP, time_limit=time_limit)
                return True
            except Exception:
                return False

def refine_shifts(problem, x, shifts, n, time_limit=5):
    """Refine shifts by finding worst correlation"""
    try:
        x_val = x.value
        if x_val is None:
            return shifts
        corr = compute_correlations(x_val)
        max_corr_idx = np.argmax(corr)
        max_corr = corr[max_corr_idx]
        
        # Find the actual shift that caused this
        s = max_corr_idx - (n - 1)  # Shift is indexed from -n+1 to n-1
        if s not in shifts:
            shifts.append(s)
        return shifts
    except:
        return shifts

            best_c5 = c5_val
            
        # Add more shifts

            max_corr_idx = np.argmax(corr)

                c5_val = dx * max_corr

                    best_c5 = c5_val
            else:

    except Exception:
        pass
        
    return best_h, best_c5

def smooth_solution(h, c5_val):
    """Smooth binary solution with continuous optimization"""
    n = len(h)
    dx = 2.0 / n
    
    # Use smooth epigraph approach
    gamma = 100.0
    warm_start = np.copy(h)
    
    def smooth_objective(h_vec, t_val):
        # Compute all correlations
        corr = compute_correlations(h_vec)
        max_corr = np.max(corr)
        # Smooth approximation to max
        sum_exp = np.sum(np.exp(gamma * (corr - t_val)))
        return t_val + (1.0 / gamma) * np.log(sum_exp)
    
    # Initialize with binary values as continuous
    h_init = h.copy()
    # constrain sum to 1 (n/2 * dx = 1)
    h_init = h_init * (n / (2.0 * sum(h_init)))
    
    def objective(h_vec):
        return smooth_objective(h_vec, c5_val)
        
    # Use SLSQP for smooth optimization
    try:
        result = minimize(objective, h_init, method='SLSQP',
                         options={'maxiter': 100, 'ftol': 1e-6},
                         constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n/2})
        if result.success:
            return result.x
    except:
        return h_init
    
    return h_init
