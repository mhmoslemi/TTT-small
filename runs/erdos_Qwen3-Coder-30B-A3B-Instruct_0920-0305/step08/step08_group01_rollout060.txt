```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def random_binary_solution():
    """Generate a random binary solution with exactly TARGET_SUM ones"""
    h = np.zeros(n_points, dtype=np.float64)
    ones_indices = np.random.choice(n_points, size=TARGET_SUM, replace=False)
    h[ones_indices] = 1.0
    return h

def simulated_annealing_binary():
    """Run simulated annealing on binary patterns"""
    # Initial solution
    h = random_binary_solution()
    current_c5 = evaluate_c5(h)
    
    best_h = h.copy()
    best_c5 = current_c5
    
    # Temperature schedule
    initial_temp = 0.01
    final_temp = 0.0001
    max_iterations = 150000
    
    start_time = time.time()
    elapsed = 0
    
    iteration = 0
    while iteration < max_iterations and elapsed < 580:
        elapsed = time.time() - start_time
        if elapsed > 580:
            break
            
        # Generate neighbor by swapping one 1 and one 0
        ones_indices = np.where(h == 1)[0]
        zeros_indices = np.where(h == 0)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            # Reinitialize if constraint violated
            h = random_binary_solution()
            current_c5 = evaluate_c5(h)
            iteration += 1
            continue
            
        i = np.random.choice(ones_indices)
        j = np.random.choice(zeros_indices)
        
        # Create new solution by swapping
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        
        new_c5 = evaluate_c5(h_new)
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
        else:
            # Accept with probability based on temperature
            delta = new_c5 - current_c5
            temperature = initial_temp * (final_temp / initial_temp) ** (elapsed / 580)
            if temperature > 0 and np.random.random() < np.exp(-delta / temperature):
                h = h_new
                current_c5 = new_c5
                
        iteration += 1
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    
    # Set parameters
    n_points = 200
    dx = 2.0 / n_points
    TARGET_SUM = n_points // 2
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    
    # Multiple restarts
    restarts = 0
    while time.time() - start_time < budget_s - 5:
        h, c5 = simulated_annealing_binary()
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        restarts += 1
        
    return (best_h.tolist(), float(best_c5), n_points)
```
