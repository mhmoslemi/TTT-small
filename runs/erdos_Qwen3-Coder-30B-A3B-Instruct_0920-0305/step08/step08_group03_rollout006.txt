# Function to get max overlap value for piecewise constant h
def compute_c5_piecewise(lengths, heights):
    """Compute C5 for piecewise constant function defined by lengths and heights."""
    # Total number of intervals
    m = len(lengths)
    # Verify sum of lengths
    assert abs(sum(lengths) - 2.0) < 1e-10, "Sum of lengths must be 2"
    # Verify mass constraint
    mass = sum(h * l for h, l in zip(heights, lengths))
    assert abs(mass - 1.0) < 1e-10, "Mass constraint is violated"
    
    # Assume sorted lengths, compute breaks
    breaks = [0.0] + list(itertools.accumulate(lengths))
    
    # Critical shifts are at boundaries; let's do some tensile checks
    # Compute all possible critical values (shift values which may be too close)
    # So we evaluate at shifts where overlaps change
    # Within one interval's length, the overlap changes based on:
    # When interval p overlaps the shifted q
    
    # So we compute for shift values

    # Compute all intervals' contributions for various shifts
    # Here's an approximate evaluation
    # First interpolate any sample from literature to match our parameters
    # For now, not so expensive:
    # Get candidate shifts from lengths themselves...
    n_evals = 100
    k_vals = np.linspace(0, 2, n_evals)
    
    max_val = -1.0
    
    for k in k_vals:
        # For shift k, compute h(x)*h(x+k) integral
        contrib = 0.0
        for i in range(m):
            a = heights[i]
            l = lengths[i]
            start = breaks[i]
            end = breaks[i+1]
            if start < 2 - k:
                # Effective interval within [0,2-k]
                eff_start = max(start, 0)
                eff_end = min(end, 2 - k)
                if eff_end > eff_start:
                    # Basic hand-shifted part
                    shift_len = eff_end - eff_start
                    contrib += a * shift_len

        # Now get combined h(x)(1-h(x+k)))
        # Which is integral of h(x) - h(x)*h(x+k)
        # = integral h(x) - integral h(x)*h(x+k)
        linear_term = contrib

        # This is problematic because intersection depends on both interval structures
        # Let's evaluate more accurately - compute correlation explicitly on discrete samples
        # For now, we'll define N samples inverse to and evaluate integral approximations
        
        # Actually we want the integral of h(x)(1-h(x+k))
        # Which equals integral h(x) - integral h(x)*h(x+k) = integral h(x) - conv(h,h)(k)
        # But that's not quite right; let's simulate that using FFT variant
        
        # Return a better simuator
    
    # Or if we want closed form:
    # The integral over 0 to 2-k of h(x)*(1-h(x+k))= 
    # integral h(x) - integral h(x) h(x+k) = M - assoc_sum of appropriate overlap...
    # Let's try a best-effort and return a composition method of discrete approx here for m=5
    return _compute_c5_approx(lengths, heights)

def _compute_c5_approx(lengths, heights):
    """Compute approximate C5 using sampled values"""
    # Whether we post-process or build in advance, let's use FFT to quickly get it
    n_points = 1000
    dx = 2.0 / n_points

    # Build h function from lengths and heights
    h = np.zeros(n_points)
    accum_len = 0
    for i, (a, l) in enumerate(zip(heights, lengths)):
        start_idx, end_idx = int(accum_len / 2 * n_points), int((accum_len + l) / 2 * n_points)
        h[start_idx:end_idx] = a
        accum_len += l

    # Compute correlation of h with 1-h using discrete method
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    
    return max_corr
