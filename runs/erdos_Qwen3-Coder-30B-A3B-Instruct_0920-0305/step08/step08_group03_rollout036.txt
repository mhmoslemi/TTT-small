def compute_overlap_from_intervals(lengths, heights, k):
    """
    Compute ∫₀^{2‑k} h(x)(1-h(x+k)) dx using interval representation.
    """
    total_len = sum(lengths)
    assert abs(total_len - 2.0) < 1e-10, "Total length should be 2."

    # Compute contribution to linear term, J = ∫₀^{2‑k} h(x) dx
    linear_term = 0.0
    # Compute contribution to product term, ∫₀^{2‑k} h(x)h(x+k) dx
    product_term = 0.0

    # Distance of left end of each interval from 0
    left_ends = [0.0]
    for i in range(len(lengths)):
        left_ends.append(left_ends[-1] + lengths[i])

    # For k, find intersection with all intervals
    # Linear term integral over [0, 2-k]
    end_point = 2.0 - k
    if end_point <= 0:
        return 0.0
    # Find all intervals that intersect [0, end_point]
    # This is a careful geometric computation
    for i in range(len(lengths)):
        start = left_ends[i]
        end = left_ends[i+1]
        overlap_start = max(0.0, start)
        overlap_end = min(end_point, end)
        if overlap_end > overlap_start:
            segment_len = overlap_end - overlap_start
            linear_term += heights[i] * segment_len

    if k >= 2.0:
        return 0.0  # No overlap at all

    # Product term ∫₀^{2‑k} h(x)h(x+k) dx
    # Create a list of (expr_start, expr_end, weight1, weight2)
    # The full supported area of h(x+hk) in x is [0, 2-k]
    overlap_exprs = list()
    for i in range(len(lengths)):
        # h(x) on I_i = [left_ends[i], left_ends[i+1])
        # h(x+k) overlaps with [0, 2-k] with itself shifted left by k
        # e.g. on x = [0, 2-k], evaluate h(x) h(x+k)
        h_start = left_ends[i]
        h_end = left_ends[i+1]
        # h(x+k) is backed up by k units
        hk_start = h_start + k
        hk_end = h_end + k

        # Both h(x) (on [h_start, h_end]) and h(x+k) (on [hk_start, hk_end])
        # are supported on [0, 2-k] => (h_start, h_end) intersect [0, 2-k] and
        # (hk_start, hk_end) intersect [0, 2-k]

        # Find intersection: the interval where both h and h(x+k) are defined.
        int_left = max(h_start, 0.0)
        int_right = min(h_end, 2.0 - k)

        if int_left < int_right:
            # Now we can compute the overlap with h(x+k)
            # Interval overlap with [0, 2-k]
            # and the shift interval with [0, 2-k]
            # h(x+k) = h(x) on [k, 2-k]
            # So that h(x+k) has support on [k, 2-k] == [0+k, 2-k]
            overlap_left = max(h_start, k)
            overlap_right = min(h_end, 2.0 - k)
            if overlap_left < overlap_right:
                overlap_len = overlap_right - overlap_left
                product_term += heights[i] * heights[i] * overlap_len

    # For cross-terms, we need to enumerate pairs
    # But efficiently we can also compute correctly

    # To be representative, we compute full overlap list
    # Create list of support intervals (with overhangs) and
    # figure out overlaps as region pairs

    # Simplified loop over all pairs of intervals
    product_term = 0.0
    for i in range(len(lengths)):
        for j in range(len(lengths)):
            h1_start = left_ends[i]
            h1_end = left_ends[i+1]
            h2_start = left_ends[j] + k
            h2_end = left_ends[j+1] + k

            inter_start = max(h1_start, h2_start)
            inter_end = min(h1_end, h2_end)

            if inter_start < inter_end:
                inter_len = inter_end - inter_start
                product_term += heights[i] * heights[j] * inter_len

    # The contribution is: (J(k) = ∫₀^{2‑k} h(x) dx) - (P(k) = ∫₀^{2‑k} h(x)h(x+k) dx)
    # = linear_term - product_term
    return linear_term - product_term

def evaluate_c5_from_intervals(lengths, heights):
    """
    Compute full C5 = max_k of the integral over all shifts.
    """
    # Determine critical points
    # These must include: 0, 2 - sum(lengths), and all transition points
    critical_k = [0.0]
    left_ends = [0.0]
    for i in range(len(lengths)):
        left_ends.append(left_ends[-1] + lengths[i])
    for i in range(len(lengths)):
        for j in range(len(lengths)):
            # Overlap kernels: shifts by differences of interval boundaries
            if left_ends[i] > 0:
                critical_k.append(left_ends[i])
            if left_ends[i+1] < 2:
                critical_k.append(left_ends[i+1])
            if left_ends[j] > 0:
                critical_k.append(left_ends[j] + 2.0 - 2.0)
            if left_ends[j+1] < 2:
                critical_k.append(left_ends[j+1] + 2.0 - 2.0)

    # Make unique
    critical_k = list(set(critical_k))
    critical_k = [x for x in critical_k if 0 <= x <= 2]
    critical_k.sort()

    # We check k on a dense enough set to capture maximums
    k_grid = np.linspace(0, 2, 1000)  # Increasing density near critical points
    # And also include the exact points
    k_grid_full = sorted(set(list(k_grid) + critical_k))

    max_overlap = float('-inf')
    for k in k_grid_full:
        overlap = compute_overlap_from_intervals(lengths, heights, k)
        max_overlap = max(max_overlap, overlap)
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    np.set_printoptions(precision=10)
    start_time = time.time()
    
    # Use the provided initial function to construct a piecewise constant one
    h0 = initial_h_from_file()
    dx0 = 2.0 / len(h0)
    
    # Convert to 5 interval structure: conservative choice
    m = 5
    num_intervals = m
    # Approximate the h0 function as a piecewise constant with num_intervals.
    # Use k-means style bucketing or even simpler - split into intervals.
    
    # Simple: split x evenly and average the function values
    segment_len = 2.0 / num_intervals
    lengths = [segment_len] * num_intervals
    heights = []

    # Compute average over each segment
    for i in range(num_intervals):
        start_idx = int(np.floor(i * len(h0) / num_intervals))
        end_idx = int(np.ceil((i+1) * len(h0) / num_intervals))
        avg_h = np.mean(h0[start_idx:end_idx])
        heights.append(avg_h)

    # Fix normalization
    total_mass = sum(h * l for h, l in zip(heights, lengths))
    scale = 1.0 / total_mass
    heights = [h * scale for h in heights]

    # Optimization loop
    best_lengths = lengths[:]
    best_heights = heights[:]
    best_c5 = evaluate_c5_from_intervals(lengths, heights)

    # Local improvement loop
    max_iterations = 10000
    iteration = 0

    # Gradient descent style updates to the parameters
    # We will now do limited coordinate descent and global sampling

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        # Test neighboring values
        improvements = []
        # Slightly change one length and height
        for i in range(num_intervals):
            delta = 0.01
            for change_type in ['length', 'height_h']:
                new_lengths = lengths[:]
                new_heights = heights[:]
                
                if change_type == 'length' and i < num_intervals - 1:
                    # Only move last length
                    if i < num_intervals - 1:
                        new_lengths[i] = max(0.001, lengths[i] + delta * (1 if np.random.random() > 0.5 else -1))
                        new_lengths[num_intervals-1] = 2.0 - sum(new_lengths[:num_intervals-1])
                    else:
                        new_lengths[i] = max(0.001, lengths[i] + delta * (1 if np.random.random() > 0.5 else -1))
                        new_lengths[i] -= 0.1 if new_lengths[i] > lengths[i] else -0.1
                        if new_lengths[i] < 0.001:
                            new_lengths[i] = 0.001
                    if new_lengths[i] < 0.001:
                        new_lengths[i] = 0.001
                elif change_type == 'height_h' and i < num_intervals - 1:
                    new_heights[i] = max(0.001, min(0.999, heights[i] + delta * (1 if np.random.random() > 0.5 else -1)))
                else:
                    continue

                # Normalize
                total_mass = sum(h * l for h, l in zip(new_heights, new_lengths))
                scale = 1.0 / total_mass
                new_heights = [h * scale for h in new_heights]
                try:
                    c5 = evaluate_c5_from_intervals(new_lengths, new_heights)
                    improvements.append((c5, new_lengths, new_heights, i, change_type))
                except:
                    pass
        
        # Pick best improvement
        if improvements:
            min_c5, new_lengths, new_heights, i, change_type = min(improvements, key=lambda x: x[0])
            if min_c5 < best_c5:
                best_c5 = min_c5
                best_lengths = new_lengths
                best_heights = new_heights

        # Periodically wiggle
        if iteration % 50 == 0:
            new_lengths = lengths[:]
            new_heights = heights[:]
            for i in range(num_intervals):
                if i < num_intervals - 1:
                    new_lengths[i] = max(0.001, lengths[i] * (1 + np.random.normal(0, 0.02)))
                if i < num_intervals - 1:
                    new_heights[i] = max(0.01, min(0.99, heights[i] * (1 + np.random.normal(0, 0.02))))
            total_mass = sum(h * l for h, l in zip(new_heights, new_lengths))
            scale = 1.0 / total_mass
            new_heights = [h * scale for h in new_heights]
            try:
                c5 = evaluate_c5_from_intervals(new_lengths, new_heights)
                if c5 < best_c5:
                    best_c5 = c5
                    best_lengths = new_lengths
                    best_heights = new_heights
            except:
                pass

    # Convert result back to the discretized vector for final output
    h_disc = np.zeros(n_points)
    segment_len = 2.0 / n_points
    cum_len = 0.0

    for i, (h, l) in enumerate(zip(best_heights, best_lengths)):
        if i == len(best_heights) - 1:
            # Final segment
            final_len = 2.0 - cum_len
            num_points = int(round(final_len / segment_len))
            if num_points > 0:
                h_disc[int(cum_len/segment_len):int(cum_len/segment_len + num_points)] = h
            cum_len += final_len
            break
        else:
            num_points = int(round(l / segment_len))
            if num_points > 0:
                h_disc[int(cum_len/segment_len):int(cum_len/segment_len + num_points)] = h
            cum_len += l

    # Normalize to correct sum
    total_norm = np.sum(h_disc)
    h_disc = h_disc * (1.0 / total_norm)

    # Refresh to make sure the sum is exactly 1
    final_sum = np.sum(h_disc)
    h_disc = h_disc * (1.0 / final_sum) * (n_points/2)  # we want the sum to be n_points/2

    return (h_disc.tolist(), float(best_c5), n_points)
