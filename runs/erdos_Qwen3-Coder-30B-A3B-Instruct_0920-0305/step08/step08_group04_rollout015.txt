import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import time

# Global variables
n_points = 300
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling projection: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    sorted_x = np.sort(x)[::-1]
    n = len(x)
    
    # Find the threshold index
    cumsum = np.cumsum(sorted_x)
    threshold_idx = -1
    for i in range(n):
        if cumsum[i] + (n - i - 1) * sorted_x[i] >= target_sum:
            threshold_idx = i
            break
    
    if threshold_idx == -1:
        # All elements are 0
        return np.zeros_like(x)
    
    # Compute lambda
    lambda_val = (target_sum - cumsum[threshold_idx - 1]) / (n - threshold_idx) if threshold_idx > 0 else (target_sum - cumsum[threshold_idx]) / (n - threshold_idx)
    
    projected = np.clip(x + lambda_val, 0, 1)
    return projected

def rasterise_and_project(theta):
    """Convert parametric representation to h vector and project to sum constraint"""
    K = 8  # number of blocks
    # Split theta into widths and heights
    widths_raw = theta[:K]
    heights_raw = theta[K:]
    
    # Convert widths to actual widths
    u = np.exp(np.clip(widths_raw, -15, 15))  # Clamp to prevent overflow
    w = 2 * u / np.sum(u)  # Normalize to total length 2

    # Convert heights to actual heights using sigmoid
    a = 1 / (1 + np.exp(-heights_raw))
    
    # Create the h vector
    h = np.zeros(n_points)
    cumulative_width = 0
    for i in range(K):
        start_idx = int(round(cumulative_width / dx))
        end_idx = int(round((cumulative_width + w[i]) / dx))
        if end_idx > start_idx:  # Ensure we don't have zero-width blocks
            # Fill block with height a[i]
            h[start_idx:end_idx] = a[i]
        cumulative_width += w[i]
    
    # Ensure exact mass constraint
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def obj(θ):
    """Objective function for differential evolution"""
    h = rasterise_and_project(θ)
    c5 = evaluate_c5(h)
    return c5

def get_active_shifts(corr, tol=1e-12):
    """Find active shifts that achieve max correlation"""
    max_val = corr.max()
    active = np.where(corr >= max_val - tol)[0]
    shifts = active - (len(corr) // 2)
    return shifts

def compute_subgradient(h, corr):
    """Compute subgradient of C5"""
    # Get active shifts
    shifts = get_active_shifts(corr)
    
    # Initialize gradient
    g = np.zeros_like(h)
    n = len(h)
    
    # Compute average gradient over all active shifts
    for s in shifts:
        g += -dx  # constant term
        if s >= 0:
            # h[i] coefficient: -h[i+s]
            g[:n-s] -= dx * h[s:]
            # h[i+s] coefficient: -h[i]
            g[s:] -= dx * h[:n-s]
        else:
            t = -s
            # h[i] coefficient: -h[i-s]
            g[t:] -= dx * h[:n-t]
            # h[i-t] coefficient: -h[i]
            g[:n-t] -= dx * h[t:]
    
    # Average over active set
    if len(shifts) > 0:
        g /= len(shifts)
    
    # Return descent direction
    return -g

def adaptive_step_size(η, improvement):
    """Update step size based on whether there was an improvement"""
    if improvement:
        return min(η * 1.05, 2.0)  # Increase step size
    else:
        return max(η * 0.7, 1e-4)  # Decrease step size

def mass_transfer_swap(h, rng):
    """Perform a stochastic mass transfer to escape plateaus"""
    # Find indices where transfer is possible
    indices_nonzero = np.where(h > 0)[0]
    indices_less_1 = np.where(h < 1)[0]
    
    # Pick two indices
    if len(indices_nonzero) == 0 or len(indices_less_1) == 0:
        return h
    
    i_plus = indices_less_1[rng.choice(len(indices_less_1))]
    i_minus = indices_nonzero[rng.choice(len(indices_nonzero))]
    
    # Transfer a small amount
    transfer_amount = min(0.02, h[i_minus]) * rng.random()
    h[i_plus] += transfer_amount
    h[i_minus] -= transfer_amount
    
    # Re-project to maintain constraints
    h = project_to_sum_box(h, TARGET_SUM)
    return h
