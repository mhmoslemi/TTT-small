import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

# Global constants
TARGET_SUM = None
N_POINTS = 180
DX = 2.0 / N_POINTS
N_INTERVALS = 5
MAX_EVALS = 8000

def decode_intervals(z, m=N_INTERVALS):
    """Convert parameter vector z to interval endpoints and lengths."""
    # Ensure we are working with the right dimensional space
    if len(z) != 2 * m:
        raise ValueError("z should have 2*m elements")
    
    # Map to interval lengths and gaps ensuring positivity
    epsilon = 1e-4
    lengths = epsilon + (1 - 2 * epsilon) * z[:m]
    lengths = lengths / np.sum(lengths)  # Normalize to sum to 1
    
    gaps = z[m:]  # These are scaled by remaining space
    gaps = np.maximum(0, gaps * (2.0 - np.sum(lengths)))  # Scale gaps so final point is at most 2
    
    # Compute cumulative position
    positions = [0.0]
    for i in range(m):
        positions.append(positions[-1] + lengths[i] + gaps[i])
    
    # Check if we exceed boundary
    if positions[-1] > 2:
        # Rescale gaps to fit exactly
        additional_space = 2 - positions[-1]
        scale_factor = additional_space / sum(gaps) if sum(gaps) > 0 else 1.0
        gaps = gaps * scale_factor
        positions = [0.0]
        for i in range(m):
            positions.append(positions[-1] + lengths[i] + gaps[i])
    
    # Extract start and end positions
    starts = np.array(positions[:-1])
    ends = np.array(positions[1:])
    
    return starts, ends

def build_grid_from_intervals(starts, ends, n=N_POINTS):
    """Build binary grid h based on interval start/end positions."""
    h = np.zeros(n)
    grid_centers = np.linspace(DX / 2, 2 - DX / 2, n)  # Centers of bins
    
    # For each grid center, check if it falls into any interval
    for idx, center in enumerate(grid_centers):
        for start, end in zip(starts, ends):
            if start <= center <= end:
                h[idx] = 1.0
                break
    return h

def compute_c5_fft(h):
    """Fast computation of C5 using FFT."""
    # Correlate h with (1 - h) using FFT
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return DX * np.max(corr)

def objective(z):
    """Objective function that takes interval parameters and returns C5 value."""
    try:
        starts, ends = decode_intervals(z)
        h = build_grid_from_intervals(starts, ends)
        return compute_c5_fft(h)
    except Exception:
        return np.inf  # penalty for infeasible solution

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up the parameters
    dimensions = 2 * N_INTERVALS
    bounds = [(0, 1)] * dimensions
    
    # Phase 1: Generate initial population from random samples
    initial_population = []
    for _ in range(10):
        z0 = np.random.rand(dimensions)
        if np.sum(z0[:N_INTERVALS]) > 0:  # avoid trivial case
            initial_population.append(z0)
    
    # Phase 2: Differential evolution optimization
    de_start = time.time()
    result = differential_evolution(
        func=objective,
        bounds=bounds,
        maxiter=int(200 * budget_s / 600),
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        callback=lambda x, convergence: None,
        disp=False
    )
    
    # Extract optimized parameters
    best_z = result.x
    best_c5 = result.fun
    
    # Ensure we get a valid solution by refining the best one found
    starts, ends = decode_intervals(best_z)
    h_grid = build_grid_from_intervals(starts, ends)
    best_c5 = compute_c5_fft(h_grid)
    
    # Phase 3: Local refinement of the best solution
    # Use Tabu search to refine boundaries (small steps near optimal)
    cur_z = best_z.copy()
    local_iterations = 0
    max_local = int(100 * budget_s / 600)
    
    while time.time() - start_time < 0.9 * budget_s and local_iterations < max_local:
        local_iterations += 1
        # Try to improve by moving one boundary
        best_improved = False
        for i in range(N_INTERVALS):
            # Try moving start of interval i
            test_z = cur_z.copy()
            step_size = DX / 4
            test_z[i] += step_size
            if test_z[i] > 1.0:
                test_z[i] = 1.0
            new_c5 = objective(test_z)
            if new_c5 < best_c5:
                best_z = test_z
                best_c5 = new_c5
                best_improved = True
                break
            
            # Try moving end of interval i (using the gap parameter)
            test_z = cur_z.copy()
            test_z[i + N_INTERVALS] -= step_size
            if test_z[i + N_INTERVALS] < 0:
                test_z[i + N_INTERVALS] = 0
            new_c5 = objective(test_z)
            if new_c5 < best_c5:
                best_z = test_z
                best_c5 = new_c5
                best_improved = True
                break
        
        if not best_improved:
            break

    final_starts, final_ends = decode_intervals(best_z)
    final_h = build_grid_from_intervals(final_starts, final_ends)
    final_c5 = compute_c5_fft(final_h)

    # Phase 4: Final verification with exact correlation if needed
    # Use np.correlate directly for final verification (same method as grader)
    final_h_exact = build_grid_from_intervals(final_starts, final_ends, n=120)
    if len(final_h_exact) < 120:
        final_h_exact = np.pad(final_h_exact, (0, 120 - len(final_h_exact)), 'constant')
    elif len(final_h_exact) > 120:
        final_h_exact = final_h_exact[:120]
    corr_exact = correlate(final_h_exact, 1 - final_h_exact, mode='full')
    final_c5_exact = DX * np.max(corr_exact)
    
    # Return detailed result
    return (final_h_exact.tolist(), float(final_c5_exact), 120)
