```python
import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import differential_evolution
import math

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_initial_solution():
    """Generate a more structured initial solution"""
    # Start with piecewise constant function
    # Divide interval [0,2] into 4 segments and 
    # assign values to maximize overlap suppression
    
    h = np.zeros(n_points)
    
    # Create better initial solution with more structure
    segment_size = n_points // 4
    for i in range(4):
        start_idx = i * segment_size
        end_idx = (i + 1) * segment_size if i < 3 else n_points
        if i % 2 == 0:
            h[start_idx:end_idx] = 1.0
        else:
            h[start_idx:end_idx] = 0.0
    
    return project_to_sum_box(h, TARGET_SUM)

def generate_population(num_individuals):
    """Generate a population of diverse solutions"""
    population = []
    for _ in range(num_individuals):
        # Mix of different approaches
        r = np.random.random()
        if r < 0.3:
            # Random solution
            h = np.random.uniform(0, 1, n_points)
        elif r < 0.6:
            # Step function with random steps
            h = np.zeros(n_points)
            for i in range(0, n_points, 5):
                if i < n_points:
                    h[i:] = np.random.random()
                    break
        else:
            # Use the initial solution with noise
            h = generate_initial_solution() + np.random.normal(0, 0.05, n_points)
        
        h = np.clip(h, 0, 1)
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    
    return population

def mutate_individual(h, strength=0.05):
    """Mutation operator"""
    h_mut = h.copy()
    # Select random indices to mutate
    n_mutate = max(1, int(0.1 * n_points))
    indices = np.random.choice(n_points, n_mutate, replace=False)
    
    for idx in indices:
        # Add small random change
        delta = np.random.normal(0, strength)
        h_mut[idx] += delta
        # Apply boundary clipping and project to sum constraint
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = project_to_sum_box(h_mut, TARGET_SUM)
    
    return h_mut

def crossover_parents(h1, h2):
    """Crossover operator"""
    # Uniform crossover
    mask = np.random.random(n_points) > 0.5
    h_child = np.where(mask, h1, h2)
    h_child = project_to_sum_box(h_child, TARGET_SUM)
    return h_child

def evolutionary_optimization():
    """Use differential evolution for global optimization"""
    
    # Define bounds for each variable [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        # Make sure it satisfies the constraint
        h_projected = project_to_sum_box(h, TARGET_SUM)
        return evaluate_c5(h_projected)
    
    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,
        maxiter=100,
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=42,
        tol=1e-8,
        workers=1,
        updating='immediate',
        disp=True
    )
    
    if result.success:
        h_final = project_to_sum_box(result.x, TARGET_SUM)
        return h_final
    else:
        # Fallback to local refinement if DE fails
        h_initial = generate_initial_solution()
        return h_initial

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up problem parameters
    TARGET_SUM = n_points / 2
    best_h = None
    best_c5 = float('inf')
    
    # Try multiple strategies
    
    # Strategy 1: Differential Evolution
    try:
        h1 = evolutionary_optimization()
        c5_1 = evaluate_c5(h1)
        if c5_1 < best_c5:
            best_h = h1.copy()
            best_c5 = c5_1
    except:
        pass
        
    # Strategy 2: Generate several candidates using different approaches and select best
    candidates = []
    for i in range(10):
        # Random initialization
        h_random = np.random.uniform(0, 1, n_points)
        h_random = project_to_sum_box(h_random, TARGET_SUM)
        
        # Structured initialization
        h_struct = generate_initial_solution()
        
        # Add noise to initial solution
        h_noisy = generate_initial_solution() + np.random.normal(0, 0.05, n_points)
        h_noisy = np.clip(h_noisy, 0, 1)
        h_noisy = project_to_sum_box(h_noisy, TARGET_SUM)
        
        candidates.extend([h_random, h_struct, h_noisy])
    
    # Evaluate all candidates
    for h in candidates:
        c5_val = evaluate_c5(h)
        if c5_val < best_c5:
            best_h = h.copy()
            best_c5 = c5_val
    
    # Local refinement with gradient-free optimization
    if best_h is not None:
        # If time permits, do one final optimization 
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            def objective(h):
                h_projected = project_to_sum_box(h, TARGET_SUM)
                return evaluate_c5(h_projected)
            
            # Use L-BFGS-B for fast local improvement
            from scipy.optimize import minimize
            result = minimize(
                objective,
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-12}
            )
            
            h_final = project_to_sum_box(result.x, TARGET_SUM)
            c5_final = evaluate_c5(h_final)
            
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
        except:
            pass

    # Final safeguard
    if best_h is None:
        best_h = generate_initial_solution()
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
```
