import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_corr_int(h):
    """Compute integer correlation counts from FFT"""
    corr = compute_corr_fft(h)
    return np.rint(corr).astype(int)

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def initialize_binary_pattern(n):
    """Initialize a random binary pattern with exactly n/2 ones"""
    h = np.zeros(n, dtype=int)
    h[:n//2] = 1
    np.random.shuffle(h)
    return h

def simulated_annealing_binary(initial_h, time_budget):
    """Simulated Annealing for binary pattern optimization"""
    start_time = time.time()
    current_h = initial_h.copy()
    current_corr_int = compute_corr_int(current_h)
    current_max = current_corr_int.max()
    
    best_h = current_h.copy()
    best_max = current_max
    
    # Temperature control
    T = 2.0  # Initial temperature (mismatches scale)
    cooling_rate = 0.9995
    
    iterations = 0
    while time.time() - start_time < time_budget:
        iterations += 1
        # Generate a random swap
        ones_idx = np.where(current_h == 1)[0]
        zeros_idx = np.where(current_h == 0)[0]
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
            
        i_one = np.random.choice(ones_idx)
        i_zero = np.random.choice(zeros_idx)
        
        # Create candidate by swapping
        cand_h = current_h.copy()
        cand_h[i_one] = 0
        cand_h[i_zero] = 1
        
        # Compute new correlation
        cand_corr_int = compute_corr_int(cand_h)
        cand_max = cand_corr_int.max()
        
        # Acceptance probability
        delta = cand_max - current_max
        if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, 1e-6)):
            current_h = cand_h
            current_max = cand_max
            
            if cand_max < best_max:
                best_h = cand_h.copy()
                best_max = cand_max
                
        # Cool down
        T *= cooling_rate
        
        # Occasionally reheat
        if iterations % 1000 == 0 and T < 0.1:
            T = 0.1
            
    return best_h, best_max

def deterministic_hill_climb(h, max_iter=5000, patience=1000):
    """Deterministic hill climbing with random swaps"""
    best_h = h.copy()
    best_corr_int = compute_corr_int(best_h)
    best_max = best_corr_int.max()
    no_improve = 0
    
    for _ in range(max_iter):
        # Generate multiple random swaps
        improved = False
        for _ in range(20):  # Try 20 swaps
            ones_idx = np.where(best_h == 1)[0]
            zeros_idx = np.where(best_h == 0)[0]
            
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
                
            i_one = np.random.choice(ones_idx)
            i_zero = np.random.choice(zeros_idx)
            
            # Create candidate
            cand_h = best_h.copy()
            cand_h[i_one] = 0
            cand_h[i_zero] = 1
            
            # Compute new correlation
            cand_corr_int = compute_corr_int(cand_h)
            cand_max = cand_corr_int.max()
            
            if cand_max < best_max:
                best_h = cand_h
                best_max = cand_max
                best_corr_int = cand_corr_int
                improved = True
                no_improve = 0
                break
                
        if not improved:
            no_improve += 1
            if no_improve >= patience:
                break
                
    return best_h, best_max

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Multiple random starts for robustness
    num_starts = 5
    for start_iter in range(num_starts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Generate initial binary pattern
        h0 = initialize_binary_pattern(n_points)
        
        # Annealing to escape local optima
        anneal_budget = 0.7 * budget_s / num_starts
        h_annealed, max_corr = simulated_annealing_binary(h0, anneal_budget)
        
        # Local refinement
        h_refined, max_corr_refined = deterministic_hill_climb(h_annealed, 1000, 500)
        
        # Evaluate final
        current_c5 = dx * max_corr_refined
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_refined.copy()
            
    # Final polishing if time allows
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        h_final, max_corr_final = deterministic_hill_climb(best_h, 2000, 1000)
        best_c5 = dx * max_corr_final
        best_h = h_final
    
    return (best_h.tolist(), float(best_c5), n_points)
