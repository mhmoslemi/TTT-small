import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_sum_box(v, target_sum):
    """Project vector v onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(v + (a + b)/2, 0, 1)
    return result

def solve_milp_binary(N, M, timeout=100):
    """Solve binary MILP to maximize minimum autocorrelation"""
    try:
        y = cp.Variable(N, boolean=True)
        t = cp.Variable()
        
        # Cardinality constraint
        constraints = [cp.sum(y) == M]
        
        # Linearization for each shift
        for k in range(1, N):
            # Variables for product y_i * y_{i+k}
            w = cp.Variable(N - k)
            # Add constraints to enforce w[i] = y[i]*y[i+k]
            constraints.extend([
                w <= y[:N-k],
                w <= y[k:],
                w >= y[:N-k] + y[k:] - 1,
                w >= 0
            ])
            constraints.append(cp.sum(w) >= t)
        
        # Bound t
        constraints.append(t >= 0)
        constraints.append(t <= M)
        
        # Problem
        prob = cp.Problem(cp.Maximize(t), constraints)
        
        # Solve with timeout
        prob.solve(solver=cp.CBC, time_limit=timeout, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            y_sol = np.round(y.value).astype(int)
            return y_sol
    except:
        pass
    return None

def subgradient_descent(h0, max_iter=20000):
    """Subgradient descent with projection"""
    h = h0.copy()
    M = len(h) // 2
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for t in range(1, max_iter + 1):
        # Compute correlation
        corr = correlate(h, 1-h, mode='full')
        k_max = np.argmax(corr) - (len(h) - 1)
        
        # Compute subgradient
        g = np.ones(len(h))
        if k_max >= 0:
            g[:len(h)-k_max] -= h[k_max:]
        else:
            k = -k_max
            g[k:] -= h[:len(h)-k]
            
        if k_max > 0:
            g[k_max:] -= h[:len(h)-k_max]
        elif k_max < 0:
            k = -k_max
            g[:len(h)-k] -= h[k:]
            
        # Gradient step
        alpha = 0.01 / np.sqrt(t)
        h_new = h - alpha * g
        
        # Project
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, M)
        
        # Evaluate
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        h = h_new
        
    return best_h, best_c5

def stochastic_refinement(h0, max_iter=3000, timeout=100):
    """Lightweight stochastic refinement using simulated annealing"""
    start_time = time.time()
    h = h0.copy()
    M = len(h) // 2
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    t = 0
    while time.time() - start_time < timeout and t < max_iter:
        # Random indices to modify
        i, j = np.random.choice(len(h), size=2, replace=False)
        h1 = h.copy()
        delta = np.random.uniform(-0.02, 0.02)
        
        # Ensure valid bounds
        new_val_i = h[i] + delta
        new_val_j = h[j] - delta
        
        if 0 <= new_val_i <= 1 and 0 <= new_val_j <= 1:
            h1[i] = new_val_i
            h1[j] = new_val_j
            
            # Project back
            h1 = project_sum_box(h1, M)
            
            new_c5 = evaluate_c5(h1)
            if new_c5 < best_c5:
                best_h = h1.copy()
                best_c5 = new_c5
            h = h1
        
        t += 1
    return best_h, best_c5

def evaluate_c5(h):
    """Evaluate C5 using FFT correlation"""
    corr = correlate(h, 1-h, mode='full')
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    
    start_time = time.time()
    TARGET_SUM = n_points // 2
    
    # Phase 1: Solve Binary MILP
    solution = None
    phase1_budget = 0.4 * budget_s
    phase1_end = start_time + phase1_budget
    
    try:
        solution = solve_milp_binary(n_points, TARGET_SUM, timeout=phase1_budget * 0.9)
    except Exception as e:
        pass
    
    h = None
    total_c5 = float('inf')
    
    if solution is not None:
        h = solution.astype(float)
        total_c5 = evaluate_c5(h)
    else:
        # Fallback to random initialization
        h = np.random.uniform(0, 1, n_points)
        h = project_sum_box(h, TARGET_SUM)
        total_c5 = evaluate_c5(h)
    
    # Phase 2: Deterministic gradient descent
    phase2_budget = 0.35 * budget_s
    phase2_end = start_time + phase1_budget + phase2_budget
    
    if time.time() < phase2_end:
        h_refined, c5_refined = subgradient_descent(h, max_iter=10000)
        if c5_refined < total_c5:
            h = h_refined
            total_c5 = c5_refined
    
    # Phase 3: Stochastic refinement
    phase3_budget = budget_s - (phase1_budget + phase2_budget)
    if time.time() < start_time + budget_s:
        h_final, c5_final = stochastic_refinement(h, timeout=phase3_budget, max_iter=2000)
        if c5_final < total_c5:
            h = h_final
            total_c5 = c5_final
    
    return (h.tolist(), float(total_c5), n_points)
