import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, boolean, inv_pos

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Fast computation of correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Evaluate the overlap integral C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_sum_box(v, M):
    """Project vector v onto the feasible set: sum = M, 0 <= v <= 1 """
    # First clip to bounds
    v = np.clip(v, 0, 1)
    # Now adjust to meet sum = M using water-filling approach
    v_sum = np.sum(v)
    if abs(v_sum - M) < 1e-10:
        return v
    elif v_sum > M:
        # Need to reduce values
        diff = v_sum - M
        # Sort in increasing order to reduce least values first
        sorted_indices = np.argsort(v)
        v = v.copy()
        for idx in reversed(sorted_indices):
            if diff <= 0:
                break
            reduce = min(v[idx], diff)
            v[idx] -= reduce
            diff -= reduce
    else:
        # Need to increase values
        diff = M - v_sum
        sorted_indices = np.argsort(v)
        v = v.copy()
        for idx in sorted_indices:
            if diff <= 0:
                break
            increase = min(1.0 - v[idx], diff)
            v[idx] += increase
            diff -= increase
    return v

def solve_milp_binary(N, target_sum):
    """Solve the MILP for finding binary h with maximum min autocorrelation"""
    # Variable definitions
    y = Variable(N, boolean=True)
    t = Variable()
    
    # Constraints
    constraints = [cp_sum(y) == target_sum]
    
    # Add linearization constraints for all shift k
    w_vars = []
    shift_sums = []
    
    for k in range(1, N):
        indices = range(N - k)
        w_k = Variable(len(indices), nonneg=True)
        w_vars.append(w_k)
        
        # Linearization constraints: w_k[i] = y[i] * y[i+k]
        for i in indices:
            constraints.append(w_k[i] <= y[i])
            constraints.append(w_k[i] <= y[i + k])
            constraints.append(w_k[i] >= y[i] + y[i + k] - 1)
        
        shift_sum = cp_sum(w_k)
        shift_sums.append(shift_sum)
        constraints.append(shift_sum >= t)
    
    # Bounds on t
    constraints.append(t >= 0)
    constraints.append(t <= target_sum)
    
    # Objective
    prob = Problem(Maximize(t), constraints)
    
    # Solve with timeout
    try:
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=120)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            y_opt = y.value
            if y_opt is not None:
                return y_opt
    except:
        pass
    return None

def subgradient_descent(h0, max_iter=10000):
    """Apply a subgradient descent with step-size decay"""
    h = h0.copy()
    M = TARGET_SUM
    
    for t in range(1, max_iter + 1):
        # Compute correlation and get the worst shift
        corr = compute_corr_fft(h)
        k_max = np.argmax(corr) - (n_points - 1)
        
        # Compute gradient w.r.t. h for shift k_max
        g = np.ones(n_points)
        if k_max >= 0:
            if k_max > 0:
                g[k_max:] -= h[:-k_max]
            if k_max < n_points - 1:
                g[:n_points-k_max] -= h[k_max:]
        else:
            k = -k_max
            if k < n_points:
                g[:n_points-k] -= h[k:]
            if k > 0:
                g[k:] -= h[:n_points-k]

        # Step size schedule
        alpha = 0.5 / np.sqrt(t)
        
        # Update h
        h_new = h - alpha * g
        
        # Project back
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, M)
        
        # Accept improvement
        if evaluate_c5(h_new) < evaluate_c5(h):
            h = h_new
    return h

def stochastic_refinement(h0, max_iter=5000):
    """Simple simulated annealing refinement"""
    h = h0.copy()
    M = TARGET_SUM
    
    for t in range(max_iter):
        i, j = np.random.choice(n_points, size=2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        
        if 0 <= h[i] + delta <= 1 and 0 <= h[j] - delta <= 1:
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
            
            h_new = project_sum_box(h_new, M)
            
            new_c5 = evaluate_c5(h_new)
            old_c5 = evaluate_c5(h)
            
            if new_c5 < old_c5:
                h = h_new
            else:
                # Accept with probability that decays with time
                prob = np.exp(-(new_c5 - old_c5) / (0.02 + 0.001 * t))
                if np.random.random() < prob:
                    h = h_new
    return h

    noise_level = 0.01
    h_perturbed = h_interp + np.random.normal(0, noise_level, n_points)
    h_perturbed = np.clip(h_perturbed, 0, 1)
    h_perturbed = project_sum_box(h_perturbed, TARGET_SUM)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    best_h = None
    best_c5 = float('inf')

    # Phase 1: Try to solve the MILP for binary h
    milp_time_start = time.time()
    y_opt = solve_milp_binary(n_points, TARGET_SUM)

    if y_opt is not None:
        h_milp = y_opt
        c5_milp = evaluate_c5(h_milp)
        if c5_milp < best_c5:
            best_c5 = c5_milp
            best_h = h_milp.copy()
    else:
        # Fallback to adaptive initialization
        h_milp = adaptive_initialization()
        best_h = h_milp.copy()
        best_c5 = evaluate_c5(best_h)

    remaining_time = budget_s - (time.time() - start_time)

    # Phase 2: Subgradient descent from best h
    if remaining_time > 30:
        h_sgd = subgradient_descent(best_h, max_iter=5000)
        c5_sgd = evaluate_c5(h_sgd)
        if c5_sgd < best_c5:
            best_c5 = c5_sgd
            best_h = h_sgd.copy()

    # Phase 3: Stochastic refinement
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        h_sa = stochastic_refinement(best_h, max_iter=3000)
        c5_sa = evaluate_c5(h_sa)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    return (best_h.tolist(), float(best_c5), n_points)
