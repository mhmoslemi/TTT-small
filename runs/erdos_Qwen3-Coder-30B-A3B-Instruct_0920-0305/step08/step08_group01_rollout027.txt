import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def build_binary_from_intervals(intervals, n):
    """Convert interval endpoints to binary vector of length n on [0,2]"""
    h = np.zeros(n)
    dx = 2.0 / n
    for s, e in intervals:
        # Get grid indices that fall inside [s, e]
        start_idx = max(0, int(np.ceil(s / dx)))
        end_idx = min(n, int(np.floor(e / dx)) + 1)
        # Set the grid cells that are fully or partially inside [s,e] to 1
        for i in range(start_idx, end_idx):
            # Compute intersection of [i*dx, (i+1)*dx] with [s,e]
            left = max(i * dx, s)
            right = min((i + 1) * dx, e)
            if left < right:
                h[i] = 1.0
    return h

def compute_c5_fast(h):
    """Evaluate C5 using FFT correlation"""
    # FFT correlation using scipy.signal.correlate
    corr = correlate(h, 1 - h, mode='full')
    dx = 2.0 / len(h)
    return np.max(corr) * dx

def decode_params(z, m, n):
    """Decode interval parameters into interval list and binary vector"""
    # z[0:m] are lengths, z[m:2m] are gaps
    # clamp lengths to avoid zero
    eps = 1e-6
    l = eps + (1 - 2 * eps) * z[:m]
    g = z[m:2*m]  # gaps between intervals
    
    # Normalize lengths to sum to 1
    l = l / np.sum(l)
    
    # Build interval positions
    intervals = []
    s = 0
    for i in range(m):
        e = s + l[i]
        intervals.append((s, e))
        if i < m - 1:
            s = e + g[i]
        else:
            s = e  # no final gap needed
    
    # Now we must ensure total interval ends within [0,2]
    end_pos = s  # last end position
    if end_pos > 2.0:
        # Compress all gaps proportionally if needed
        total_gap_len = sum(g)
        factor = (2.0 - np.sum(l)) / total_gap_len if total_gap_len > 0 else 1.0
        g = np.array(g) * factor
        # Recompute positions
        intervals = []
        s = 0
        for i in range(m):
            e = s + l[i]
            intervals.append((s, e))
            if i < m - 1:
                s = e + g[i]
            else:
                s = e
    
    # Convenience: return binary vector on grid
    h = build_binary_from_intervals(intervals, n)
    return intervals, h

def objective(z, m, n):
    """Objective function for DE minimization"""
    try:
        _, h = decode_params(z, m, n)
        c5 = compute_c5_fast(h)
        return c5
    except Exception:
        # Return large value on error to penalize
        return 1e10
