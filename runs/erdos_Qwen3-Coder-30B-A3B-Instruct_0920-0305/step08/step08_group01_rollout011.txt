```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def random_binary_solution():
    """Generate a random binary solution with exactly TARGET_SUM ones"""
    h = np.zeros(n_points, dtype=float)
    ones_indices = np.random.choice(n_points, size=int(TARGET_SUM), replace=False)
    h[ones_indices] = 1.0
    return h

def accept_probability(old_c5, new_c5, temp):
    """Calculate acceptance probability for simulated annealing"""
    if new_c5 < old_c5:
        return 1.0
    else:
        return np.exp((old_c5 - new_c5) / temp)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = None
    best_c5 = float('inf')
    
    # Simulated Annealing with restarts
    max_iter_per_restart = 50000
    temperature_schedule = lambda t: 0.01 * (0.999 ** t)
    
    # Outer loop with restarts
    restart_count = 0
    while time.time() - start_time < budget_s * 0.95:
        restart_count += 1
        # Generate new random binary solution
        h = random_binary_solution()
        current_c5 = evaluate_c5(h)
        
        # Track best in this restart
        local_best_c5 = current_c5
        local_best_h = h.copy()
        
        # Simulated Annealing inner loop
        for iteration in range(max_iter_per_restart):
            if time.time() - start_time > budget_s:
                break
                
            # Choose two random indices: one with 1, one with 0
            ones_indices = np.where(h == 1)[0]
            zeros_indices = np.where(h == 0)[0]
            
            if len(ones_indices) == 0 or len(zeros_indices) == 0:
                continue
                
            i = np.random.choice(ones_indices)
            j = np.random.choice(zeros_indices)
            
            # Perform swap
            h_copy = h.copy()
            h_copy[i] = 0
            h_copy[j] = 1
            
            # Compute new C5
            new_c5 = evaluate_c5(h_copy)
            
            # Accept with probability
            temp = temperature_schedule(iteration)
            if accept_probability(current_c5, new_c5, temp):
                h = h_copy
                current_c5 = new_c5
                
                # Update local best
                if new_c5 < local_best_c5:
                    local_best_c5 = new_c5
                    local_best_h = h_copy.copy()
            
            # Update global best
            if local_best_c5 < best_c5:
                best_c5 = local_best_c5
                best_h = local_best_h.copy()
                
        # Early termination if no improvement in this restart
        if local_best_c5 >= current_c5 * 0.9999 and restart_count > 5:
            break
    
    if best_h is None:
        # Fallback to initial solution
        best_h = random_binary_solution()
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
```
