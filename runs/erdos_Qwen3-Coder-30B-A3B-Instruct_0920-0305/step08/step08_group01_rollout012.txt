```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = 120
    dx = 2.0 / best_n
    
    # Helper function for FFT-based evaluation
    def evaluate_c5(h):
        h_rev = h[::-1]
        b = 1 - h
        L = 2 * len(h) - 1
        corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
        return dx * np.max(corr)
    
    # MILP formulation solver
    def solve_milp_with_row_generation(n, max_time=0.7 * budget_s):
        # Variables
        x = cp.Variable(n, boolean=True)
        t = cp.Variable(pos=True)
        
        # Mass constraint
        mass_constraint = cp.sum(x) == n // 2
        
        # Storage for constraints and variables
        constraints = [mass_constraint]
        y_vars = {}
        s_active = set()
        
        # Initial small set of shifts
        initial_shifts = [1, 2, 3, 4, 5]
        for s in initial_shifts:
            s_active.add(s)
        
        def add_shift_constraint(s):
            if s in y_vars:
                return
            y = cp.Variable(n, boolean=True)  # for shift s
            y_vars[s] = y
            # Add constraints for variable y
            for i in range(n):
                j = i + s
                if j < n:
                    constraints.append(y[i] <= x[i])
                    constraints.append(y[i] <= x[j])
                    constraints.append(y[i] >= x[i] + x[j] - 1)
        
        # Add all initial shifts
        for s in s_active:
            add_shift_constraint(s)
        
        # Objective
        objective = cp.Minimize(t)
        
        # Solve loop with cutting plane
        iteration = 0
        while time.time() - start_time < max_time and iteration < 500:
            # Add epigraph constraints
            new_constraints = []
            for s in s_active:
                if s in y_vars and not np.any(y_vars[s].value is None):
                    new_constraints.append(cp.sum(y_vars[s]) >= n//2 - t)
            constraints.extend(new_constraints)
            
            # Create and solve problem
            prob = cp.Problem(objective, constraints)
            try:
                prob.solve(solver=CPLEX, max_seconds=5)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    x_val = np.round(x.value).astype(int)
                    if np.sum(x_val) == n//2:
                        # Recompute true C5 by FFT
                        h_test = x_val.astype(float)
                        c5_test = evaluate_c5(h_test)
                        # Check if new max is significant violating shift
                        # Re-evaluate for all deviations
                        shift_range = np.arange(-n+1, n)
                        worst_overlaps = []
                        for shift in shift_range:
                            if shift == 0:
                                continue
                            c_val = np.sum(h_test * np.roll(1-h_test, shift))
                            worst_overlaps.append(c_val)
                        max_shift_overlap = np.max(worst_overlaps) / dx if worst_overlaps else 0
                        if abs(max_shift_overlap - t.value) < 1e-8:
                            return x_val, c5_test
                        else:
                            # Add the violating shift
                            violating_shift = shift_range[np.argmax(worst_overlaps)]
                            s_active.add(violating_shift)
                            add_shift_constraint(violating_shift)
                else:
                    break
            except Exception as e:
                # Fallback to default from previous solution
                break
            
            iteration += 1
        
        # Default fallback if not converged
        x_val = np.zeros(n, dtype=int)
        # Initialize with approximately half ones
        half = n // 2
        x_val[:half] = 1
        np.random.shuffle(x_val)
        c5 = evaluate_c5(x_val.astype(float))
        return x_val, c5
    
    # Phase 1: Solve with coarse grid
    final_h_bin, c5_coarse = solve_milp_with_row_generation(best_n, 0.7 * budget_s)
    h_coarse = final_h_bin.astype(float)
    
    # Phase 2: Polish the solution with continuous optimization
    def smooth_objective(h, t, gamma=100):
        # Compute overlaps using FFT
        h_rev = h[::-1]
        b = 1 - h
        L = 2 * len(h) - 1
        corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
        return t + (1/gamma) * np.log(np.sum(np.exp(gamma * (corr - t))))
    
    def smooth_constraint(h):
        return np.sum(h) - best_n/2
    
    # Use scipy.optimize for polishing
    from scipy.optimize import minimize
    
    # Initialize with coarse solution
    h_init = h_coarse.copy()
    t_init = c5_coarse / dx
    
    bounds = [(0, 1) for _ in range(best_n)]
    cons = {'type': 'eq', 'fun': smooth_constraint}
    
    def obj_func(h):
        return smooth_objective(h, t_init)
    
    try:
        res = minimize(obj_func, h_init, bounds=bounds, constraints=cons, 
                      method='SLSQP', options={'maxiter': 200, 'ftol': 1e-12})
        if res.success:
            polished_h = np.clip(res.x, 0, 1)
            polished_c5 = evaluate_c5(polished_h)
            if polished_c5 < best_c5:
                best_c5 = polished_c5
                best_h = polished_h
        else:
            # Still retain best from previous
            best_h = h_coarse
            best_c5 = evaluate_c5(h_coarse)
    except Exception:
        # Fallback to converted binary
        best_h = h_coarse
        best_c5 = evaluate_c5(h_coarse)
    
    # Final validation
    test_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), float(test_c5), best_n)
```
