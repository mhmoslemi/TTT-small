import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_initial_population(size):
    """Generate diverse initial solutions"""
    population = []
    for _ in range(size):
        # Mix of different strategies
        strategy = np.random.choice([0, 1, 2, 3])
        if strategy == 0:
            # Random uniform
            h = np.random.uniform(0, 1, n_points)
        elif strategy == 1:
            # Piecewise constant
            h = np.zeros(n_points)
            segments = np.random.randint(2, 8)
            for i in range(segments):
                start = int(i * n_points / segments)
                end = int((i + 1) * n_points / segments)
                h[start:end] = np.random.uniform(0, 1)
        elif strategy == 2:
            # Gaussian bumps
            h = np.zeros(n_points)
            n_peaks = np.random.randint(3, 8)
            for _ in range(n_peaks):
                center = np.random.randint(0, n_points)
                width = np.random.randint(10, 30)
                for i in range(max(0, center - width), min(n_points, center + width)):
                    h[i] += np.exp(-((i - center) ** 2) / (2 * width ** 2))
        else:
            # Sine waves
            h = np.zeros(n_points)
            n_cycles = np.random.randint(2, 6)
            for i in range(n_points):
                h[i] = 0.5 + 0.5 * np.sin(2 * np.pi * n_cycles * i / n_points)
        
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    return population

def crossover(parent1, parent2):
    """Single-point crossover"""
    crossover_point = np.random.randint(1, n_points - 1)
    child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    return child

def mutate(h, mutation_rate=0.1):
    """Mutation with different operators"""
    h_mut = h.copy()
    for i in range(n_points):
        if np.random.random() < mutation_rate:
            # Mix of mutation types
            mut_type = np.random.choice([0, 1, 2])
            if mut_type == 0:
                # Add small random noise
                h_mut[i] += np.random.normal(0, 0.05)
            elif mut_type == 1:
                # Replace with random value
                h_mut[i] = np.random.uniform(0, 1)
            else:
                # Flip (but keep within constraints)
                h_mut[i] = 1 - h_mut[i]
    h_mut = project_to_sum_box(h_mut, TARGET_SUM)
    return h_mut

def evolutionary_optimization():
    """Use evolutionary algorithm on the problem"""
    pop_size = 30
    generations = 100
    elite_size = 5
    
    # Generate initial population
    population = generate_initial_population(pop_size)
    fitness_scores = [evaluate_c5(h) for h in population]
    
    # Evolution loop
    for gen in range(generations):
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Keep elite
        elite = population[:elite_size]
        
        # Generate new population
        new_population = elite.copy()
        while len(new_population) < pop_size:
            # Tournament selection
            tournament_size = 3
            parent1_idx = np.random.choice(sorted_indices[:pop_size//2])
            parent2_idx = np.random.choice(sorted_indices[:pop_size//2])
            parent1, parent2 = population[parent1_idx], population[parent2_idx]
            
            # Crossover
            child = crossover(parent1, parent2)
            # Mutation
            child = mutate(child)
            
            new_population.append(child)
        
        population = new_population
        fitness_scores = [evaluate_c5(h) for h in population]
        
        # Early stopping if improvement is small
        if gen > 10 and abs(fitness_scores[0] - fitness_scores[-1]) < 1e-6:
            break
    
    # Return best solution
    best_idx = np.argmin(fitness_scores)
    return population[best_idx], fitness_scores[best_idx]

def refine_solution(h_initial):
    """Refine solution with local search to polish final solution"""
    h_best = h_initial.copy()
    c5_best = evaluate_c5(h_best)
    
    # Local hill climbing in small regions
    for _ in range(200):
        # Choose random region
        region_size = np.random.randint(5, 20)
        start_idx = np.random.randint(0, n_points - region_size)
        
        # Try small perturbations
        for i in range(start_idx, start_idx + region_size):
            if np.random.random() < 0.3:  # 30% chance to modify
                old_val = h_best[i]
                # Try few increments/decrements
                for delta in [-0.05, -0.02, 0.02, 0.05]:
                    h_new = h_best.copy()
                    h_new[i] += delta
                    if 0 <= h_new[i] <= 1:
                        h_new = project_to_sum_box(h_new, TARGET_SUM)
                        c5_new = evaluate_c5(h_new)
                        if c5_new < c5_best:
                            h_best = h_new
                            c5_best = c5_new
                            break
                else:
                    # Undo if no improvement found
                    h_best[i] = old_val
    
    return h_best, c5_best

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Run evolutionary optimization
    h_evolutionary, c5_evolutionary = evolutionary_optimization()
    
    # Attempt to refine
    h_final, c5_final = refine_solution(h_evolutionary)
    
    return (h_final.tolist(), float(c5_final), n_points)
