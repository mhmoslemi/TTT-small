import numpy as np
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def build_critical_shifts(splits):
    """Given interval splits, generate critical shift values."""
    shifts = set()
    for i in range(len(splits)):
        for j in range(len(splits)):
            shifts.add(abs(splits[i] - splits[j]))
            shifts.add(abs((2 - splits[i]) - (2 - splits[j])))
    shifts.add(0.0)
    shifts.add(2.0)
    return sorted(list(shifts))

def integrate_piecewise(h_values, splits):
    """Compute int_0^{2-k} h(x) dx for each k."""
    integral_vals = []
    for k in range(len(h_values)):
        total = 0.0
        current_pos = 0.0
        for i, (h, split) in enumerate(zip(h_values, splits[:-1])):
            if current_pos >= 2 - k:
                break
            interval_start, interval_end = split, splits[i+1]
            # Compute overlap with [0, 2-k]
            overlap_start = max(current_pos, 0.0)
            overlap_end = min(interval_end, 2.0 - k)
            if overlap_start < overlap_end:
                total += h * (overlap_end - overlap_start)
            current_pos = interval_end
        integral_vals.append(total)
    return integral_vals

def compute_overlap_integrals(L, A, K_crit):
    """Compute each J(k;L,A) for given parameters"""
    # Compute splits from lengths
    splits = [0.0] + [sum(L[:i+1]) for i in range(len(L))]
    
    # For each k, compute the integral
    J = []
    for k in K_crit:
        # Linear term = int_0^{2-k} h(x) dx
        linear_term = 0.0
        current_pos = 0.0
        for i, h in enumerate(A):
            if current_pos >= 2 - k:
                break
            interval_start, interval_end = splits[i], splits[i+1]
            overlap_start = max(current_pos, 0.0)
            overlap_end = min(interval_end, 2.0 - k)
            if overlap_start < overlap_end:
                linear_term += h * (overlap_end - overlap_start)
            current_pos = interval_end

        # Product term: sum_{i,j} a_i a_j overlap_length
        product_term = 0.0
        for i, h_i in enumerate(A):
            for j, h_j in enumerate(A):
                interval_start_i, interval_end_i = splits[i], splits[i+1]
                interval_start_j, interval_end_j = splits[j], splits[j+1]
                # Compute overlap of [interval_start_i, interval_end_i] with [interval_start_j + k, interval_end_j + k]
                ov_start = max(interval_start_i, interval_start_j + k)
                ov_end = min(interval_end_i, interval_end_j + k)
                if ov_start < ov_end:
                    product_term += h_i * h_j * (ov_end - ov_start)

        J.append(linear_term - product_term)
    
    return J

def run_optimization(m, seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    # Generate critical shifts
    # We'll represent a generic 5-step function in sensible way
    splits_needed = m + 1
    
    def compute_c5_from_params(lengths, heights):
        # Infer the last length from sum constraint
        L = list(lengths) + [2.0 - np.sum(lengths)]
        A = list(heights)
        # Compute the critical shift values
        splits = [0.0] + [sum(L[:i+1]) for i in range(len(L)-1)] 
        K_crit = build_critical_shifts(splits)
        # Compute all J(k)
        J = compute_overlap_integrals(L, A, K_crit)

    best_c5 = float('inf')
    best_result = None
    time_limit_hit = False

        np.random.seed(seed + restarts)

            J = compute_overlap_integrals(list(L_guess) + [2.0 - np.sum(L_guess)], list(A_guess), K_crit)
            c5_score = max(J)

    if best_result is not None and not time_limit_hit:

            ll = params[:m-1]

            ll = ll * 2.0 / np.sum(ll)

            L = list(ll) + [2.0 - np.sum(ll)]

            ll = ll * 2.0 / np.sum(ll)
