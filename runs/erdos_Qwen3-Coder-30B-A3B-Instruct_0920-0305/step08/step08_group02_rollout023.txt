```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_sum_box(v, M):
    """Project vector v onto sum=M and box [0,1] constraints"""
    v = np.clip(v, 0, 1)
    # Water-filling to adjust the sum to exactly M
    # Use bisection on lambda to find correct value
    def objective(lam):
        # Ensure values within [0,1]
        new_vals = np.clip(v + lam, 0, 1)
        return np.sum(new_vals) - M

    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = objective(m)
        if abs(fm) < 1e-10:
            break
        if fm * objective(a) < 0:
            b = m
        else:
            a = m
    result = np.clip(v + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def solve_milp_binary(N):
    """Solve MILP for binary y vector to maximize minimal autocorrelation"""
    # Create variables
    y = cp.Variable(N, boolean=True)
    t = cp.Variable()
    M = N // 2

    # Cost and constraints
    constraints = [cp.sum(y) == M]  # sum of ones = M

    # For each shift k, define variables w_k for overlap terms
    W = {}  # dict mapping k to w_k variable
    Sum_terms = []

    for k in range(1, N):  # k from 1 to N-1
        # indices for valid positions: i from 0 to N-k-1
        w_k = cp.Variable(N - k, nonneg=True)
        W[k] = w_k

        # Add constraints linking w_k with y[i] and y[i+k]
        constraints.append(w_k <= y[:N - k])
        constraints.append(w_k <= y[k:])
        # w_k >= y[i] + y[i+k] - 1
        constraints.append(w_k >= y[:N - k] + y[k:] - 1)
        # w_k >= 0 (already implied by nonneg=True)

        sum_k = cp.sum(w_k)
        Sum_terms.append(sum_k)
        constraints.append(sum_k >= t)

    # Define objective
    prob = cp.Problem(cp.Maximize(t), constraints)

    # Try CBC solver first, then fallback
    solver_options = {'maxSeconds': 100}
    try:
        prob.solve(solver=cp.CBC, **solver_options)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_EOF:
            return True, y.value
    except:
        pass

    try:
        prob.solve(solver=cp.GLPK_MI, **solver_options)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_EOF:
            return True, y.value
    except:
        pass

    return False, None

def subgradient_descent(h0, max_iter=20000):
    """Sub-gradient descent with projection"""
    h = h0.copy()
    M = len(h) // 2
    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    for t in range(1, max_iter + 1):
        # Compute correlation
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr) - (len(h) - 1)

        # Compute sub-gradient for shift k_max
        g = np.ones_like(h)
        if k_max > 0:
            g[k_max:] -= h[:-k_max]
            g[:k_max] -= h[-k_max:]
        elif k_max < 0:
            k = -k_max
            g[:k] -= h[-k:]
            g[k:] -= h[:-k]
        else:
            # k_max == 0: gradient is just -1 added to all elements
            g *= -1

        # Gradient step
        alpha = 0.1 / np.sqrt(t)
        h_new = h - alpha * g
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, M)

        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

        h = h_new

        # Early stopping if improvement very small
        if c5_new < 0.375:
            break

    return best_h, best_c5

def stochastic_refinement(h0, budget_s):
    """Run short stochastic search to escape plateaus"""
    h = h0.copy()
    M = len(h) // 2
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    start_time = time.time()
    remaining = budget_s

    while remaining > 0.1:
        # Choose two random indices, avoid boundaries
        i, j = np.random.choice(len(h) - 1, 2, replace=False)
        if h[i] == 0 or h[i] == 1 or h[j] == 0 or h[j] == 1:
            continue

        # Small random perturbation
        delta = np.random.uniform(-0.01, 0.01)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, M)

        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

        h = h_new
        remaining = budget_s - (time.time() - start_time)
    return best_h, best_c5

def adaptive_initialization():
    """Create an initial good solution for both phases"""
    # Start from provided initial_h_values if available
    try:
        import initial_h_values
        h_ref = np.array(initial_h_values)
        x_ref = np.linspace(0, 2, len(h_ref))
        x_new = np.linspace(0, 2, n_points)
        h_interp = np.interp(x_new, x_ref, h_ref)
    except:
        h_interp = np.random.uniform(0, 1, n_points)
    h_interp = project_sum_box(h_interp, n_points // 2)
    return h_interp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initialize global constants
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points // 2
    dx = 2.0 / n_points

    # Phase 1: Try to solve current binary MILP if possible (fast)
    milp_solved, y_opt = solve_milp_binary(n_points)
    best_h = None
    best_c5 = float('inf')

    if milp_solved:
        h_opt = y_opt.astype(float)
        c5_opt = evaluate_c5(h_opt)
        best_h = h_opt
        best_c5 = c5_opt
    else:
        # Fallback to random initialization
        best_h = adaptive_initialization()
        best_c5 = evaluate_c5(best_h)

    # Phase 2: Deterministic sub-gradient descent
    time_remaining = budget_s - (time.time() - start_time)
    if time_remaining > 10:
        h_from_milp, c5_from_milp = subgradient_descent(best_h.copy(), max_iter=min(10000, int(time_remaining * 200)))
        if c5_from_milp < best_c5:
            best_h = h_from_milp
            best_c5 = c5_from_milp

    # Phase 3: Stochastic refinement
    time_remaining = budget_s - (time.time() - start_time)
    if time_remaining > 10:
        h_refined, c5_refined = stochastic_refinement(best_h.copy(), time_remaining)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined

    return (best_h.tolist(), float(best_c5), n_points)
```
