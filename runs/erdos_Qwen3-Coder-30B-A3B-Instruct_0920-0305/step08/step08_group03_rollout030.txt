def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def update_correlation(c, h, b, i, delta, j, delta_j):

    # For i: change in h[i] => change in b[i] = -delta
    # b contribution: delta * b[i+s] to c[s + n-1] for s from -i to n-1-i
    # a contribution: -delta * a[i-s] to c[s + n-1] for s from i-(n-1) to i
    n = len(h)
    
    # Handle b contribution for i
    s_min = max(-i, -(n-1))
    s_max = min(n-1-i, n-1)
    
    if s_min <= s_max:

        s_end = s_max + n
        c[s_start:s_end] += delta * b[i+s_min:i+s_max+1]
    
    # Handle a contribution for i
    s_min_a = max(i-(n-1), 0)
    s_max_a = i
    if s_min_a <= s_max_a:
        s_start_a = s_min_a + n - 1
        s_end_a = s_max_a + n
        # Need a[i-s] contribution in reverse order since indices go from i down to 0
        c[s_start_a:s_end_a] -= delta * h[i-s_max_a:i-s_min_a+1][::-1]
    
    # For j: change in h[j] => change in b[j] = -delta_j
    # b contribution: delta_j * b[j+s] to c[s + n-1] for s from -j to n-1-j
    # a contribution: -delta_j * a[j-s] to c[s + n-1] for s from j-(n-1) to j
    s_min = max(-j, -(n-1))
    s_max = min(n-1-j, n-1)
    if s_min <= s_max:
        s_start = s_min + n - 1
        s_end = s_max + n
        c[s_start:s_end] += delta_j * b[j+s_min:j+s_max+1]

    s_min_a = max(j-(n-1), 0)
    s_max_a = j
