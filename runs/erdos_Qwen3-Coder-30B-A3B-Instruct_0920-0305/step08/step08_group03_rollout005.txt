import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import math

# Global variable
TARGET_SUM = 1.0
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def build_step_function(lengths, heights, n_points):
    """Construct a step function from lengths and heights"""
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    idx_start = 0
    for i in range(len(lengths)):
        end_idx = int(np.round(lengths[i] / dx)) + idx_start
        if end_idx > n_points:
            end_idx = n_points
        h[idx_start:end_idx] = heights[i]
        idx_start = end_idx
    return h

def compute_J(k, lengths, heights):
    """Compute J(k) analytically"""
    dx = 2.0 / len(lengths)
    # Compute cumulative positions
    pos = [0.0]
    for l in lengths:
        pos.append(pos[-1] + l)
    
    # Linear term: integral of h(x) over [0, 2-k]
    linear_part = 0.0
    for i in range(len(lengths)):
        start = pos[i]
        end = pos[i+1]
        if start >= 2 - k:
            break
        if end <= 2 - k:
            linear_part += heights[i] * (end - start)
        else:
            linear_part += heights[i] * ((2 - k) - start)
    
    # Product term: sum of a_p * a_q * overlap_length
    product_part = 0.0
    for i in range(len(lengths)):
        for j in range(len(lengths)):
            start1 = pos[i]
            end1 = pos[i+1]
            start2 = pos[j]
            end2 = pos[j+1]
            
            # overlap between intervals [start1, end1] and [start2+k, end2+k]
            overlap_start = max(start1, start2 + k)
            overlap_end = min(end1, end2 + k)
            
            if overlap_start < overlap_end:
                overlap_length = overlap_end - overlap_start
                product_part += heights[i] * heights[j] * overlap_length
    
    return linear_part - product_part

def compute_max_J(lengths, heights):
    """Compute the maximum J(k) over all critical k values"""
    # Generate critical shifts
    critical_shifts = []
    pos = [0.0]
    for l in lengths:
        pos.append(pos[-1] + l)
    
    for i in range(len(pos)-1):
        for j in range(len(pos)-1):
            # Mechanically compute all alignments
            k1 = abs(pos[i] - pos[j])
            k2 = abs(pos[i] - pos[j] - 2)
            k3 = pos[i] 
            k4 = 2 - pos[j]
            
            for k in [k1, k2, k3, k4]:
                if 0 <= k <= 2:
                    critical_shifts.append(k)
    
    critical_shifts = list(set(critical_shifts))
    critical_shifts.sort()
    
    max_val = -float('inf')
    for k in critical_shifts:
        val = compute_J(k, lengths, heights)
        if val > max_val:
            max_val = val
    
    return max_val

def optimize_step_function():
    # Optimize a 5-step function
    # Decision variables: 4 lengths + 4 heights (last is derived) plus 1 aux variable
    # But to reduce complexity:
    # We directly optimize 5 interval lengths (sum=2) + 5 heights (0,1) 
    
    # Trial with parametric form
    def objective(vars):
        # Still use all 10 variables
        lengths = vars[:5]
        heights = vars[5:]
        
        # Ensure lengths are POSITIVE
        if np.any(lengths <= 0):
            return 1e10
        if np.sum(lengths) != 2.0:
            lengths = lengths * 2.0 / np.sum(lengths)
        
        # Ensure heights are in [0,1]
        heights = np.clip(heights, 0, 1)
        # Mass constraint: sum a_i l_i = 1
        mass = np.sum(np.array(heights) * np.array(lengths))
        if abs(mass - 1.0) > 1e-8:
            return 1e10
        
        # Now calculate max J
        max_J = compute_max_J(lengths, heights)
        return max_J
    
    def constraint_sum_length(vars):
        return np.sum(vars[:5]) - 2.0
    
    def constraint_mass(vars):
        lengths = vars[:5]
        heights = vars[5:]
        mass = np.sum(np.array(heights) * np.array(lengths))
        return mass - 1.0
    
    def constraint_bounds(vars):
        # The bounds are incorporated in scipy.optimize.minimize
        lengths = vars[:5]
        heights = vars[5:]
        return np.concatenate([lengths, heights, 1 - heights])
    
    initial_guess = np.array([0.5, 0.4, 0.3, 0.3, 0.5] + [0.8, 0.4, 0.2, 0.6, 0.3])
    bounds = [(1e-6, 2.0)]*5 + [(0.0, 1.0)]*5
    
    # Try multiple initial solvers
    best_value = float('inf')
    best_vars = None
    
    # Random restarts
    n_restarts = 10
    for _ in range(n_restarts):
        # randomly perturbed initial guess
        # Let lengths sum to 2
        lengths = np.random.dirichlet([1.0]*5) * 2
        heights = np.random.random(5)
        init_guess = np.concatenate([lengths, heights])
        try:
            res = minimize(objective, init_guess, 
                          method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': constraint_sum_length}, 
                                       {'type': 'eq', 'fun': constraint_mass}],
                          options={'maxiter': 1000})
            if res.success and res.fun < best_value:
                best_value = res.fun
                best_vars = res.x
        except:
            pass
    
    if best_vars is None:
        # fallback to simple trial-and-error
        best_guess = np.array([0.4, 0.4, 0.4, 0.4, 0.4] + [0.5, 0.5, 0.5, 0.5, 0.5])
        lengths = best_guess[:5]
        heights = best_guess[5:]
        # try to directly perturb
        best_val = compute_max_J(lengths, heights)
        best_vars = best_guess.copy()
    else:
        # Final validation
        lengths = best_vars[:5]
        heights = best_vars[5:]
    
    return list(lengths), list(heights), best_value

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None
    best_n_points = n_points
    
    # Run with 5 interval pieces
    lengths, heights, c5_val = optimize_step_function()
    
    # construct the full step function and compute final C5 value
    h = build_step_function(lengths, heights, n_points)
    h = project_to_sum_box(h, TARGET_SUM)

    # compute exact final C5
    final_c5 = evaluate_c5(h)
    
    # Re-optimize locally if needed
    improvement_attempts = 0
    max_improvement = 5000
    attempt_timeout = budget_s * 0.9
    
    while time.time() - start_time < attempt_timeout and improvement_attempts < max_improvement:
        improvement_attempts += 1
        # Add small perturbation
        h_new = h + np.random.normal(0, 0.001, n_points)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        if c5_new < final_c5:
            final_c5 = c5_new
            h = h_new
        # Occasionally do some DE search hybrid
        if improvement_attempts % 200 == 0 or improvement_attempts < 100:
            try:
                from scipy.optimize import differential_evolution
                bounds = [(0, 1)] * n_points
                def objective_h(h_hyp):
                    return evaluate_c5(h_hyp)
                cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
                res_de = differential_evolution(
                    objective_h,
                    bounds,
                    constraints=cons,
                    maxiter=30,
                    popsize=10
                )
                if res_de.success:
                    h_candidate = project_to_sum_box(res_de.x, TARGET_SUM)
                    c5_candidate = evaluate_c5(h_candidate)
                    if c5_candidate < final_c5:
                        final_c5 = c5_candidate
                        h = h_candidate
            except:
                pass

    return h.tolist(), float(final_c5), n_points
