import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from scipy.fft import fft, ifft

# Global constants
N = 120
dx = 2.0 / N
TARGET_SUM = N / 2
MAX_TIME = 600

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, N)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using bisection"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    # Find interval containing zero
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search for the root
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_milp_construction():
    """Solve binary MILP for optimal step function"""
    try:
        # 0.4 * budget_s = 240s
        cpu_time = time.time()
        
        # Create CVXPY variables
        y = cp.Variable(N, boolean=True)
        w = cp.Variable((N-1, N), continuous=True)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(y) == TARGET_SUM,
            t >= 0, 
            t <= TARGET_SUM
        ]
        
        # Add auxiliary variable constraints
        for k in range(N-1):
            for i in range(N-k):
                constraints.extend([
                    w[k,i] <= y[i],
                    w[k,i] <= y[i + k],
                    w[k,i] >= y[i] + y[i + k] - 1,
                    w[k,i] >= 0
                ])
        
        # Add min autocorrelation constraints
        for k in range(N-1):
            constraints.append(cp.sum(w[k,:]) >= t)
        
        # Objective
        objective = cp.Maximize(t)
        
        # Problem
        prob = cp.Problem(objective, constraints)
        
        # Solve
        prob.solve(solver=cp.CBC, max_seconds=240)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURACY]:
            y_sol = y.value
            h = y_sol
            # Get value of t
            t_val = t.value
            # C5 for this binary case
            c5 = dx * (TARGET_SUM - t_val)
            return h, c5
        else:
            return None, None
    except:
        return None, None

def greedy_binary_construction():
    """Create a good binary solution with greedy spacing"""
    # Create an initial binary vector that tries to balance spacing
    h = np.zeros(N, dtype=int)
    count = 0
    i = 0
    while count < TARGET_SUM and i < N:
        h[i] = 1
        count += 1
        i += 2  # Place 1 every 2nd position
    
    # If we didn't place enough, fill with more
    if count < TARGET_SUM:
        i = 1
        while count < TARGET_SUM and i < N:
            if h[i] == 0:
                h[i] = 1
                count += 1
            i += 2
    
    return h.astype(float), 1.0  # Placeholder for now

def sub_gradient_descent(h0, max_time):
    """Perform sub-gradient descent with diminishing step sizes"""
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    t = 1
    
    # Find initial worst correlation
    corr = compute_corr_fft(h)
    k_star = np.argmax(corr)
    
    while time.time() - start_time < max_time:
        # Compute sub-gradient for the worst shift
        g = np.zeros(N)
        if k_star < N:
            g[:] = 1.0
            g[:-k_star] -= h[k_star:]
            g[k_star:] -= h[:-k_star]
        else:
            g[:] = 1.0
            g[:N-k_star] -= h[k_star:] 
            g[N-k_star:] -= h[:k_star] 
            
        # Step size with diminishing rate
        alpha = 0.5 * dx / np.sqrt(t)
        
        # Update
        h_new = h - alpha * g
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        
        # Update for next iteration
        h = h_new
        
        # Re-compute the active shift
        corr = compute_corr_fft(h)
        k_star = np.argmax(corr)
        
        t += 1
        if t % 100 == 0:
            if time.time() - start_time > max_time:
                break
    
    return best_h, best_c5

def swap_move_annealing(h0, max_time):
    """Perform simulated annealing with 2-index swaps"""
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    temp = 5 * dx
    
    # Use fixed number of iterations for steady-state progression
    max_iter = 20000
    iter_count = 0
    
    while time.time() - start_time < max_time and iter_count < max_iter:
        # Randomly select two indices
        i, j = np.random.choice(N, size=2, replace=False)
        
        # Small perturbation delta
        delta = np.random.uniform(-0.02, 0.02)
        
        # Create new candidate
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct sum
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new value
        new_c5 = evaluate_c5(h_new)
        
        # Accept or reject
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            best_h = h_new.copy()
        else:
            # Accept with probability based on temperature
            prob = np.exp(-(new_c5 - best_c5) / temp)
            if np.random.random() < prob:
                h = h_new.copy()
        
        # Decrease temperature
        temp *= 0.9999  # Slight cooling
        iter_count += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    
    # Phase 1: Binary optimization (MILP or greedy)
    t1_start = time.time()
    
    # Try to solve MILP first
    h_binary, c5_binary = binary_milp_construction()
    
    if c5_binary is None:  # Failed or timed-out
        h_binary, c5_binary = greedy_binary_construction()
    
    # Adjust to full precision
    if h_binary is not None:
        h_binary = np.array(h_binary)  # Most likely already float, but just to be sure
        # Step size for initial continuous refinement is relatively small
        h_continuous = h_binary.copy()
        # Apply projection for validation (if needed)
        h_continuous = project_to_sum_box(h_continuous, TARGET_SUM)
        c5_continuous = evaluate_c5(h_continuous)
        if c5_continuous < best_c5:
            best_c5 = c5_continuous
            best_h = h_continuous.copy()
    
    # Phase 2: Continuous refinement with sub-gradient descent
    remaining_time = min(0.35 * budget_s, budget_s - (time.time() - start_time))
    
    h_refined, c5_refined = sub_gradient_descent(h_continuous, remaining_time)
    
    if c5_refined < best_c5:
        best_c5 = c5_refined
        best_h = h_refined.copy()
    
    # Phase 3: Final simulated annealing
    remaining_time = budget_s - (time.time() - start_time)
    
    h_final, c5_final = swap_move_annealing(h_refined, remaining_time)
    
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final.copy()
    
    return (best_h.tolist(), float(best_c5), N)
