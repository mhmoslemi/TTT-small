import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def evaluate_c5_discrete(h, dx):
    """Evaluate the discrete C5 with correct dx scaling"""
    corr = np.correlate(h, 1-h, mode="full")
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try multiple grid sizes
    grid_sizes = [120, 180, 200]
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    for n_points in grid_sizes:
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        dx = 2.0 / n_points
        
        # Use 120 grid solution to initialize
        if n_points == 120:
            h_current = np.array(initial_h_values)
        else:
            # Interpolate or create a pattern
            h_coarse = np.array(initial_h_values)
            h_current = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_coarse)), h_coarse)
        
        # Refine with local optimization
        h_current = project_to_sum_box(h_current, n_points / 2)
        
        try:
            # Simple iteration approach with better local search
            for i in range(500):
                if time.time() - start_time > 0.8 * budget_s:
                    break
                
                # Random perturbation
                h_new = h_current + np.random.normal(0, 0.005, n_points)
                h_new = project_to_sum_box(h_new, n_points / 2)
                
                # Evaluate
                c5_new = evaluate_c5_discrete(h_new, dx)
                
                if c5_new < best_c5 and c5_new < 0.381:  # Early stopping criterion
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n_points
                    
                if c5_new < 0.3805:  # Aggressive early stopping
                    break
                    
                # Random small improvement steps
                for _ in range(10):
                    idx = np.random.randint(0, n_points)
                    delta = np.random.uniform(-0.01, 0.01)
                    h_new = h_current.copy()
                    h_new[idx] += delta
                    
                    if 0 <= h_new[idx] <= 1:
                        h_new = project_to_sum_box(h_new, n_points / 2)
                        c5_new = evaluate_c5_discrete(h_new, dx)
                        if c5_new < best_c5 and c5_new < 0.381:
                            best_c5 = c5_new
                            best_h = h_new.copy()
                            best_n = n_points
                            
                h_current = h_new
                
        except Exception as e:
            continue
            
    # If no good solution found, use last attempt
    if best_h is None:
        h_current = np.array(initial_h_values)
        dx = 2.0 / 120
        best_c5 = evaluate_c5_discrete(h_current, dx)
        best_h = h_current
        best_n = 120
        
    # Run final optimization on the best result
    if time.time() - start_time < 0.95 * budget_s:
        try:
            h_final = best_h
            dx = 2.0 / best_n
            
            # Smooth optimization
            def objective(h_flat):
                return evaluate_c5_discrete(h_flat, dx)
                
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - best_n / 2}
            bounds = [(0, 1) for _ in range(best_n)]
            
            res = minimize(objective, h_final, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = res.x
                
            # Final evaluation
            final_c5 = evaluate_c5_discrete(h_final, dx)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = h_final
                
        except Exception as e:
            pass
            
    return (best_h.tolist(), float(best_c5), best_n)
