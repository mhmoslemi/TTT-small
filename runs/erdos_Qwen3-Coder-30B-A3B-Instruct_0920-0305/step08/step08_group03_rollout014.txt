import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt

# Global variables
n_points = 256
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_overlap_at_shift(k, lengths, heights):
    """Compute the overlap integral for a specific shift k using piecewise formula"""
    total = 0.0
    cumulative_start = 0.0
    total_mass = 0.0
    for i, (l, h) in enumerate(zip(lengths, heights)):
        start_i = cumulative_start
        end_i = cumulative_start + l
        total_mass += h * l
        cumulative_start = end_i
        
        # Find where the shifted interval [start_i+k, end_i+k] intersects [0,2-k]
        shifted_start = start_i + k
        shifted_end = end_i + k
        
        # Check which portion is within [0, 2-k]
        intersect_start = max(0., shifted_start)
        intersect_end = min(2-k, shifted_end)
        
        if intersect_end > intersect_start:
            overlap_length = intersect_end - intersect_start
            # Add contribution: h_i * integral(1 - h_j) over this interval
            # But we need to integrate over all other intervals that overlap
            integral_term = 0.0
            cumulative_start_j = 0.0
            for j, (l2, h2) in enumerate(zip(lengths, heights)):
                start_j = cumulative_start_j
                end_j = cumulative_start_j + l2
                cumulative_start_j = end_j
                
                # Find overlap of intervals i and j
                overlap_start = max(start_i, start_j + k)
                overlap_end = min(end_i, end_j + k)
                if overlap_end > overlap_start:
                    # Overlap length in the shifted domain
                    overlap_length_ij = overlap_end - overlap_start
                    integral_term += h * h2 * overlap_length_ij
            
            total += integral_term
    
    # Linear term 
    linear_term = 0.0
    cumulative_start = 0.0
    for i, (l, h) in enumerate(zip(lengths, heights)):
        start = cumulative_start
        end = cumulative_start + l
        cumulative_start = end
        
        # H(x) where x in [0, 2-k]
        if start >= 2-k:
            break
        if end <= 2-k:
            linear_term += h * l
        else:
            # Partial overlap
            linear_term += h * (2-k - start) 
            
    # Integral of h(x)(1 - h(x+k)) = integral(h(x)) - integral(h(x) h(x+k))
    return linear_term - total

def parametric_c5_objective(params):
    """Objective function for NLP solver"""
    m = len(params) // 2
    lengths = params[:m-1]
    heights = params[m:] 
    
    # Compute last values
    total_length = sum(lengths)
    if total_length >= 2.0:
        lengths = [l*1.999/total_length for l in lengths]
    lengths.append(2.0 - sum(lengths))
    
    total_mass = sum(h*l for h,l in zip(heights, lengths))
    if total_mass > 0:
        heights = [h*1.0/total_mass for h in heights]
    heights.append((1.0 - sum(h*l for h,l in zip(heights[:-1], lengths[:-1]))) / lengths[-1])
    
    # Make sure heights are in [0,1]
    heights = [max(0, min(1, h)) for h in heights]
    
    critical_k = []
    cumulative = [0.0]
    for l in lengths[:-1]:
        cumulative.append(cumulative[-1] + l)
        
    # Generate all critical shifts
    for i in range(len(cumulative)):
        for j in range(len(cumulative)):
            abs_diff = abs(cumulative[i] - cumulative[j])
            if abs_diff <= 2 and abs_diff >= 0:
                critical_k.append(abs_diff)
            if abs_diff != 0 and abs_diff <= 2:
                # asymmetry from 2 - cumulative
                critical_k.append(2 - abs_diff)
    
    # Also include boundary points and 0, 2
    critical_k.extend([0.0, 2.0])
    critical_k = list(set(critical_k))
    
    # Find maximum overlap
    max_overlap = -float('inf')
    for k in critical_k:
        val = compute_overlap_at_shift(k, lengths, heights)
        if val > max_overlap:
            max_overlap = val
            
    return max_overlap

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # We look for a better solution by starting from a better parametric class
    # A 5-segment function with pattern 0-h-1-0 (asymmetric)
    # 5 segments, hence 4 lengths + 5 heights = 9 parameters
    
    best_c5 = float('inf')
    best_params = None

    # Try different combinations; prefer 5 intervals
    for _ in range(100):
        # Sample from a proposal distribution (simple random)
        lengths = np.random.uniform(0.1, 0.6, 4)  # Not too long
        heights = np.random.uniform(0.1, 0.9, 5)  # Make sure they are not extreme

        # Force sum to 2 and adjust last one
        total_length = sum(lengths)
        lengths = [l * 2.0 / total_length for l in lengths]
        lengths.append(2.0 - sum(lengths[:-1]))

        # Add tiny amount to avoid zero
        heights = np.clip(heights, 0.01, 0.99)

        # Adjust first and last heights to be 0, and maybe middle 1
        heights[0] = 0.0
        heights[-1] = 0.0
        heights[2] = max(0.1, min(0.9, heights[2]))
        
        # Compute weighted average sum to satisfy mass constraint
        weighted_sum = sum(h * l for h, l in zip(heights, lengths))
        heights = [h * (1. / weighted_sum) for h in heights] 
        
        params_init = list(lengths) + list(heights) 
        
        try:
            # Use scipy minimize with different methods
            cons = [{'type': 'eq', 'fun': lambda x: sum(x[:4]) - 2.0}]
            bnds = [(0.001,1.0) for _ in range(9)]
            res = minimize(
                parametric_c5_objective,
                params_init,
                method='SLSQP',
                bounds=bnds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            
            if res.success:
                c5 = res.fun
                if c5 < best_c5:
                    best_c5 = c5
                    best_params = res.x.copy()
        except:
            pass
    
    if best_params is None:
        # fallback to discretized approach with higher resolution
        # Create initial very low quality solution
        h = np.ones(n_points) * 0.5
        best_h = project_to_sum_box(h, TARGET_SUM)
        best_c5 = evaluate_c5(best_h)
        
        return (best_h.tolist(), float(best_c5), n_points)
    
    m = len(best_params) // 2
    
    # Convert back to proper model
    lengths = best_params[:m-1]
    heights = best_params[m:] 
    
    # Scale and normalize
    total_length = sum(lengths)
    if total_length > 0:
        lengths = [l * 2.0 / total_length for l in lengths]
    lengths.append(2.0 - sum(lengths[:-1]))
    
    total_mass = sum(h*l for h,l in zip(heights, lengths))
    if total_mass > 0:
        heights = [h * 1.0 / total_mass for h in heights]
    heights.append((1.0 - sum(h*l for h,l in zip(heights[:-1], lengths[:-1]))) / lengths[-1])
    
    heights = [max(0, min(1, h)) for h in heights]
    
    # Create the final step function
    cumulative = [0.0]
    for l in lengths[:-1]:
        cumulative.append(cumulative[-1] + l)
    cumulative.append(2.0)

    # Create the high resolution step function by replicating the constant heights
    h = []
    for i, (l_start, l_end) in enumerate(zip(cumulative[:-1], cumulative[1:])):
        h.extend([heights[i]] * int(round(l_end - l_start, 2) * n_points / 2.0))
    # Ensure exactly n_points
    h = h[:n_points]
    if len(h) < n_points:
        h.extend([heights[-1]] * (n_points - len(h)))
        
    # final polishing with local increment method
    h = np.array(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Simple incremental local search to improve the solution
    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        # quick local search step
        h_trial = h.copy()
        i = np.random.choice(n_points)
        # slightly perturb one point
        h_trial[i] += np.random.normal(0, 0.01)
        h_trial = project_to_sum_box(h_trial, TARGET_SUM)
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
        h = h_trial.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
