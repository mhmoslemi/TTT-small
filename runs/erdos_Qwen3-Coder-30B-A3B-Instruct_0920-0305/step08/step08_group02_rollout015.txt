import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import Minimize, Variable, sum, Problem, norm, multiply
import math

# Global constants
MAX_N = 120
dx = 2.0 / MAX_N
TARGET_SUM = MAX_N / 2

def compute_overlap_correlation(h):
    """Compute overlap integral using FFT for speed"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_overlap_correlation(h)
    return dx * np.max(corr)

def project_to_box_and_sum(x, target_sum):
    """Project x onto [0,1] box and sum constraint using bisection"""
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    
    if abs(current_sum - target_sum) < 1e-10:
        return x
    
    # Water-filling projection
    def s_lam(lam):
        return np.sum(np.clip(x + lam, 0, 1))
    
    a, b = -1.0, 1.0
    tol = 1e-10
    for _ in range(100):  # Bisection iterations
        m = (a + b) / 2
        s_m = s_lam(m)
        if abs(s_m - target_sum) < tol:
            break
        if s_m > target_sum:
            a = m
        else:
            b = m
    lam = (a + b) / 2
    
    return np.clip(x + lam, 0, 1)

def solve_binary_milp(N, M):
    """Solve binary MILP version to maximize minimum autocorrelation"""
    try:
        # Use CVXPY with CBC solver for binary optimization
        y = Variable(N, boolean=True)
        w = Variable((N, N), nonneg=True)  # auxiliary variables for products
        t = Variable()
        
        # Constraints
        constraints = [
            sum(y) == M,
            t >= 0,
        ]
        
        # Product constraints w_{k,i} = y_i * y_{i+k} (relaxation)
        for k in range(1, N):
            for i in range(N - k):
                constraints += [
                    w[i, k] <= y[i],
                    w[i, k] <= y[i + k],
                    w[i, k] >= y[i] + y[i + k] - 1,
                    w[i, k] >= 0
                ]
        
        # Auto-correlations at least t
        for k in range(1, N):
            constraints += [
                sum(w[:, k]) >= t
            ]
        
        # Objective
        problem = Problem(Maximize(t), constraints)
        
        # Solve with timeout
        problem.solve(solver=cp.CBC, max_seconds=0.35*600)
        
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return np.round(y.value).astype(int)
        
    except Exception as e:
        pass
    
    return None

def construct_better_initial_h():
    """Construct a better starting point from the binary MILP or heuristic"""
    global MAX_N, dx, TARGET_SUM
    binary_h = solve_binary_milp(MAX_N, TARGET_SUM)
    
    if binary_h is not None:
        # Convert binary to continuous function by smoothing
        h = binary_h.astype(float)
        # Add some smoothing for better numerical gradients later
        h = np.clip(h + np.random.normal(0, 0.02, MAX_N), 0, 1)
        h = project_to_box_and_sum(h, TARGET_SUM)
        return h
    
    # Fallback random initialisation
    h = np.random.uniform(0, 1, MAX_N)
    h = project_to_box_and_sum(h, TARGET_SUM)
    return h

def subgradient_descent_step(h, step_size):
    """Perform one step of subgradient descent to minimize max overlap"""
    # Find active shift k*
    corr = compute_overlap_correlation(h)
    k_star = np.argmax(corr)
    F_star = corr[k_star]

    # Compute subgradient (direct gradient of worst shift)
    g = np.zeros_like(h)
    n = len(h)
    
    # For shift k*, compute contributions from neighbors
    if k_star == 0:
        g = np.ones_like(h)
    else:
        # Derivative of F_k* with respect to h_j is 1 - h_{j + k*} - h_{j - k*}
        for j in range(n):
            # j - k* and j + k* can go out of bounds, use 0 where out of range
            term1 = 0 if (j + k_star >= n) else h[j + k_star] 
            term2 = 0 if (j - k_star < 0) else h[j - k_star]
            g[j] = 1.0 - term1 - term2
    
    # Update with step size
    new_h = h - step_size * g
    
    # Project back
    new_h = project_to_box_and_sum(new_h, TARGET_SUM)
    return new_h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None
    
    # Phase 1: Optimized binary MILP (very limited time)
    t1 = start_time + 0.4 * budget_s
    try:
        binary_start = solve_binary_milp(MAX_N, TARGET_SUM)
        if binary_start is not None:
            h = binary_start.astype(float)
            h = project_to_box_and_sum(h, TARGET_SUM)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    except:
        pass
    
    # If we don't have a good binary start, go to heuristic
    if best_h is None:
        best_h = construct_better_initial_h()
        best_c5 = evaluate_c5(best_h)
    
    # Phase 2: Sub-gradient descent using analytical gradients
    t2 = start_time + 0.75 * budget_s
    step_size = 0.02 * dx
    
    for it in range(50000):  # Large number of iter, time will be checked
        if time.time() > t2:
            break
        
        h = subgradient_descent_step(best_h, step_size)

        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Adaptive step size
        step_size = max(0.00001, step_size * 0.9999)
    
    # Phase 3: Random refinement with trimming to maintain quality
    t3 = start_time + budget_s
    if t3 > time.time():
        iter_count = 0
        max_iter = 2000
        while time.time() < t3 and iter_count < max_iter:
            # Small random change
            idx1, idx2 = np.random.choice(MAX_N, 2, replace=False)
            delta = np.random.uniform(-0.02, 0.02)
            h_try = best_h.copy()
            h_try[idx1] += delta
            h_try[idx2] -= delta
            # Clip
            h_try = np.clip(h_try, 0, 1)
            # Project back
            h_try = project_to_box_and_sum(h_try, TARGET_SUM)
            
            c5_try = evaluate_c5(h_try)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
            
            iter_count += 1
    
    return (best_h.tolist(), float(best_c5), MAX_N)
