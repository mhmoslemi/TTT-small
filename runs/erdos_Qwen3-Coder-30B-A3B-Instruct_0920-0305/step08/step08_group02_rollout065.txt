import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize

# Global constants
N = 120
dx = 2.0 / N
TARGET_SUM = N / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_milp_solver():
    """Solve the binary max-min autocorrelation problem via MILP"""
    # Create problem variables: binary y_i and continuous auxiliaries w_k,i
    y = cp.Variable(N, boolean=True)
    w = cp.Variable((N, N), integer=False)
    t = cp.Variable()

    # Define constraints
    constraints = [
        cp.sum(y) == TARGET_SUM,
        t >= 0
    ]
    
    # Add piecewise constraints for w_{k,i} <= y_i * y_{i+k}
    for k in range(1, N):
        for i in range(N - k):
            constraints.extend([
                w[k, i] <= y[i],
                w[k, i] <= y[i + k],
                w[k, i] >= y[i] + y[i + k] - 1,
                w[k, i] >= 0
            ])
    
    # Add minimum autocorrelation constraint
    for k in range(1, N):
        constraints.append(cp.sum(w[k, :N-k]) >= t)
    
    # Create and solve the problem
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)

    start_time = time.time()
    try:
        prob.solve(solver=cp.CBC, max_seconds=0.35 * 600)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, max_seconds=0.35 * 600)
        except Exception:
            return None

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        y_sol = y.value
        t_opt = t.value
        return y_sol, t_opt
    return None

def compute_gradient(h, worst_k):
    """Compute subgradient for the worst shift"""
    grad = np.zeros_like(h)
    for i in range(len(h)):
        if i + worst_k < len(h):
            grad[i] += -h[i + worst_k]
        if i - worst_k >= 0:
            grad[i] += -h[i - worst_k]
        grad[i] += 1
    return grad

def subgradient_descent(h0, max_iter=1000):
    """Perform subgradient descent to refine solution"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_from_corr(compute_corr_fft(h))

    for t in range(max_iter):
        # Compute worst shift
        corr = compute_corr_fft(h)
        worst_k = np.argmax(corr)
        # Compute gradient
        g = compute_gradient(h, worst_k)

        # Step size schedule
        alpha = 0.5 * dx / np.sqrt(t + 1)

        # Update with step
        h_new = h - alpha * g
        # Project back
        h_new = project_to_sum_box(h_new, TARGET_SUM)

        # Evaluate
        c5_new = c5_from_corr(compute_corr_fft(h_new))
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new.copy()

    return best_h, best_c5

def greedy_binary_initialization():
    """Create an initial binary solution via greedy construction"""
    y = np.zeros(N, dtype=int)
    used = set()
    min_autocorr = float('inf')
    
    # Initialize with some basic structure
    # For now, simplest approach: place 1s evenly
    for i in range(0, N, 3):
        if len(used) < TARGET_SUM and i not in used:
            y[i] = 1
            used.add(i)
    
    # Now swap 1s to improve minimum autocorrelation 
    # Always working with binary values, just start differently  
    # Assuming starts with ~balanced 1s and 0s, limit this to 100 operations
    for _ in range(100):
        if len(used) < TARGET_SUM:
            # Add at best position
            best_idx = -1
            best_score = float('inf')
            for i in range(N):
                if i not in used:
                    tmp_y = y.copy()
                    tmp_y[i] = 1
                    score = compute_min_autocorr(tmp_y)
                    if score < best_score:
                        best_score = score
                        best_idx = i
            if best_idx != -1:
                y[best_idx] = 1
                used.add(best_idx)
        else:
            break

    return y

def compute_min_autocorr(y):
    """Compute the minimum aperiodic autocorrelation of a binary sequence"""
    # For binary sequenced only, directly compute
    h = np.array(y).astype(float)
    corr = np.correlate(h, h, mode='full')
    # Exclude self correlation (which is N, so Max_k everywhere)
    best = np.min(corr[N:])
    return best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: Try binary MILP
    final_h = None
    sol = binary_milp_solver()
    if sol is not None:
        y, t_opt = sol
        h = y.astype(float)
        c5_initial = dx * (TARGET_SUM - t_opt)
        # Refine using subgradient
        refined_h, refined_c5 = subgradient_descent(h, max_iter=500)
        final_h = refined_h
        final_c5 = refined_c5
    else:
        # Relax: Use greedy binary construction
        y = greedy_binary_initialization()
        h = y.astype(float)
        h = project_to_sum_box(h, TARGET_SUM)
        refined_h, final_c5 = subgradient_descent(h, max_iter=1000)
        final_h = refined_h

    # Phase 3: Randomized local search 
    # Use 30% of remaining time
    remain_time = start_time + 0.7 * budget_s - time.time()
    if remain_time > 0.01:
        for _ in range(500):
            if time.time() - start_time > 0.9 * budget_s:
                break
            # Small swaps of values
            idx1, idx2 = np.random.choice(N, size=2, replace=False)
            if not (final_h[idx1] == 0 or final_h[idx1] == 1 or final_h[idx2] == 0 or final_h[idx2] == 1):
                delta = np.random.uniform(-0.02, 0.02)
                h_new = final_h.copy()
                h_new[idx1] += delta
                h_new[idx2] -= delta
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                c5_new = c5_from_corr(compute_corr_fft(h_new))
                if c5_new < final_c5:
                    final_h = h_new.copy()
                    final_c5 = c5_new

    return (final_h.tolist(), float(final_c5), N)
