import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def random_binary_solution():
    """Generate random binary solution with exactly TARGET_SUM ones"""
    h = np.zeros(n_points, dtype=float)
    indices = np.random.choice(n_points, size=int(TARGET_SUM), replace=False)
    h[indices] = 1.0
    return h

def swap_bits(h, i, j):
    """Swap bits at positions i and j"""
    h[i], h[j] = h[j], h[i]

def count_mismatches(h, shift):
    """Count mismatches at given shift"""
    mismatch_count = 0
    for i in range(len(h) - shift):
        if h[i] == 1 and h[i + shift] == 0:
            mismatch_count += 1
    return mismatch_count

def get_worst_shift(h):
    """Find the shift that causes maximum mismatches"""
    max_count = 0
    worst_shift = 0
    for shift in range(1, len(h)):
        count = count_mismatches(h, shift)
        if count > max_count:
            max_count = count
            worst_shift = shift
    return worst_shift, max_count

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    max_iter = 50000
    
    # Best solution tracking
    best_h = None
    best_c5 = float('inf')
    
    # Simulated Annealing parameters
    initial_temp = 0.01
    final_temp = 0.0001
    cooling_rate = 0.9995
    
    # Main loop with restarts
    iteration = 0
    while time.time() - start_time < budget_s * 0.95:
        # Generate new random binary solution
        h = random_binary_solution()
        current_c5 = evaluate_c5(h)
        
        # Track current best in this restart
        current_best_c5 = current_c5
        current_best_h = h.copy()
        
        # Simulated Annealing for this restart
        temp = initial_temp
        no_improvement_count = 0
        max_no_improvement = 2000  # Safety break
        
        while iteration < max_iter and no_improvement_count < max_no_improvement:
            # Generate candidate by swapping two bits
            ones_indices = np.where(h == 1)[0]
            zeros_indices = np.where(h == 0)[0]
            
            if len(ones_indices) == 0 or len(zeros_indices) == 0:
                break
                
            i = np.random.choice(ones_indices)
            j = np.random.choice(zeros_indices)
            
            # Make the swap
            h[i], h[j] = h[j], h[i]
            
            # Compute new C5
            new_c5 = evaluate_c5(h)
            
            # Accept or reject the change
            delta = new_c5 - current_c5
            
            if delta < 0 or np.random.random() < np.exp(-delta / temp):
                # Accept the change
                current_c5 = new_c5
                if new_c5 < current_best_c5:
                    current_best_c5 = new_c5
                    current_best_h = h.copy()
                
                no_improvement_count = 0
            else:
                # Revert the swap
                h[i], h[j] = h[j], h[i]
                no_improvement_count += 1
            
            # Update temperature
            temp *= cooling_rate
            
            # Update iteration counter
            iteration += 1
            
            # Check if we should exit due to time
            if time.time() - start_time > budget_s * 0.95:
                break
        
        # Update global best if we found a better restart
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()
    
    # Final refinement with gradient-based optimization
    if best_h is not None:
        # Use the best solution found
        final_h = best_h.copy()
        final_c5 = evaluate_c5(final_h)
        
        # Some additional local search on the best solution
        for _ in range(500):
            if time.time() - start_time > budget_s * 0.98:
                break
            ones_indices = np.where(final_h == 1)[0]
            zeros_indices = np.where(final_h == 0)[0]
            if len(ones_indices) == 0 or len(zeros_indices) == 0:
                break
            i = np.random.choice(ones_indices)
            j = np.random.choice(zeros_indices)
            # Try swapping
            final_h[i], final_h[j] = final_h[j], final_h[i]
            new_c5 = evaluate_c5(final_h)
            if new_c5 >= final_c5:
                # Revert if not better
                final_h[i], final_h[j] = final_h[j], final_h[i]
            else:
                final_c5 = new_c5
        
        return (final_h.tolist(), float(final_c5), n_points)
    
    # If no solution found, return default
    return ([0.0] * n_points, 0.5, n_points)
