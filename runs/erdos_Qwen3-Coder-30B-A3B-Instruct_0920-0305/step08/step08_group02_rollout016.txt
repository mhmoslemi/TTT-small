import numpy as np

# Global variables
TARGET_SUM = None

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Projection onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def water_filling_projection(h, target_sum):
    """Project h onto the set {x : 0 ≤ x ≤ 1, sum(x) = target_sum} using water-filling"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Project onto sum constraint using bisection
    def sum_with_lambda(lam):
        return np.sum(np.clip(h + lam, 0, 1)) - target_sum
    
    a, b = -1.0, 1.0
    # Make sure we have a bracket
    while sum_with_lambda(a) * sum_with_lambda(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    for _ in range(30):  # max iterations
        mid = (a + b) / 2
        f_mid = sum_with_lambda(mid)
        if abs(f_mid) < 1e-10:
            break
        if f_mid * sum_with_lambda(a) < 0:
            b = mid
        else:
            a = mid
    
    lam = (a + b) / 2
    return np.clip(h + lam, 0, 1)

def solve_binary_milp(n, m):
    """Solve the binary MILP to find optimal 0/1 pattern"""
    try:
        # Variables
        y = cp.Variable(n, boolean=True)
        # Ensure sum = m
        constraints = [cp.sum(y) == m]
        
        # Generate correlation constraints
        # Use sub-sampled shifts for efficiency
        aux_vars = {}
        corr_constraints = []
        for k in range(1, n):
            aux_vars[k] = cp.Variable(n-k, pos=True)
            for i in range(n-k):
                # Linearize y[i] * y[i+k] = aux_vars[k][i]
                constraints += [
                    aux_vars[k][i] <= y[i],
                    aux_vars[k][i] <= y[i+k],
                    aux_vars[k][i] >= y[i] + y[i+k] - 1,
                    aux_vars[k][i] >= 0
                ]
            corr_constraints.append(cp.sum(aux_vars[k]) >= cp.Constant(m))
        
        # Objective
        objective = cp.Maximize(cp.Constant(0))
        
        # Problem setup
        prob = cp.Problem(objective, constraints + corr_constraints) 
        
        # Solve
        prob.solve(solver=cp.CBC, max_seconds=0.4 * 600)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            y_val = y.value
            if y_val is not None:
                return np.array(y_val, dtype=int)
    except:
        pass
    
    return None

def greedy_binary_initialization():
    """Create a greedy binary initialization with good autocorrelation"""
    h = np.zeros(n_points)
    # Place 1s with spacing that maximizes minimal autocorrelation
    # Start with a simple spacing
    spacing = int(n_points / (n_points / 2))  # approximately 2
    
    for i in range(0, n_points, spacing):
        if i < n_points:
            h[i] = 1.0
    
    # Shuffle a few bits to balance gap
    indices = np.where(h == 1)[0]
    if len(indices) > 2:
        # Perturb one 1 to another position if beneficial
        # This is approximated as a very fast heuristic
        pass
    
    # Project to desired sum
    total = np.sum(h)
    if total != n_points / 2:
        # Find indices to flip
        if total > n_points / 2:
            # Need to flip some 1s to 0s
            to_zero = int(total - n_points / 2)
            flip_indices = np.argsort(h)[:to_zero]
            h[flip_indices] = 0
        else:
            # Need to flip some 0s to 1s
            to_one = int(n_points / 2 - total)
            flip_indices = np.argsort(h)[::-1][:to_one]
            h[flip_indices] = 1
    
    return h

def sub_gradient_descent(h0):
    """Perform sub-gradient descent for improving h0"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Step size parameters
    alpha0 = 0.5 * dx 
    max_iter = 10000
    
    # Iterate for a fixed number of steps or until time expires
    start_time = time.time()
    for t in range(1, max_iter + 1):
        # Compute all correlations
        corr = compute_corr_fft(h)
        k_star = np.argmax(corr)
        
        # Compute sub-gradient: g[j] = 1 - h[j+k_star] - h[j-k_star]
        g = np.zeros_like(h)
        if k_star >= 0 and k_star < len(h):
            # Shift h to the left by k_star and add with right shift
            h_shifted_l = np.roll(h, -k_star)
            h_shifted_r = np.roll(h, k_star)
            g = 1 - h_shifted_l - h_shifted_r
            
        # Gradient step
        h_new = h - (alpha0 / np.sqrt(t)) * g
        
        # Project to feasible set
        h_new = water_filling_projection(h_new, TARGET_SUM)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new)
        
        # Update best solution if improved
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        
        # Update current for next iteration
        h = h_new.copy()
        
        # Check time limit
        if time.time() - start_time > 0.35 * 600:
            break
    
    return best_h, best_c5

def simulated_annealing(h0, max_time):
    """Perform simulated annealing to escape plateaus"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    T = 5 * dx
    T_final = 0
    
    while time.time() - start_time < max_time:
        # Try a swap move
        i, j = np.random.choice(n_points, size=2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        
        # Make move
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Re-project to sum constraint
        h_new = water_filling_projection(h_new, TARGET_SUM)
        
        # Compute new C5
        new_c5 = evaluate_c5(h_new)
        
        # Accept or reject
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h.copy()
        else:
            # Accept probabilistically
            delta_c5 = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta_c5 / T):
                h = h_new
            # Reduce temperature
            T = T_final + (T - T_final) * 0.9999
            
    return best_h, best_c5
