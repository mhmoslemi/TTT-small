    def solve_milp_with_row_generation(n):
        # The feasible region is: h[i] ∈ [0,1], sum(h) = n/2
        # and we want to minimize max_s (sum h[i] * (1 - h[i+s]))
        # Let's define variables
        x = cp.Variable(n, integer=True)
        t = cp.Variable()
        # Constraints
        mass_constraint = cp.sum(x) == n//2

        def create_y_vars_and_constraints():
            # Precompute valid shifts for all i
            y_vars = []
            y_cons = []
            
            # For each shift s, in range [-n+1, n-1]
            for s in shifts:
                # For each valid i
                for i in range(n):
                    if i + s >= 0 and i + s < n:
                        y_name = f'y_{i}_{s}'
                        y[y_name] = cp.Variable(boolean=True)
                        y_vars.append(y[y_name])
                        
                        # Linearization of x[i] * x[i+s] for binary variables
                        y_cons.append(y[y_name] <= x[i])
                        y_cons.append(y[y_name] <= x[i+s])
                        y_cons.append(y[y_name] >= x[i] + x[i+s] - 1)
                
            return y_vars, y_cons

            epigraph_cons = []

                expr = 0

                        y_name = f'y_{i}_{s}'
                        expr += y[y_name]
                # max_s Σ_i x_i·(1-x_{i+s}) = max_s (n/2 - Σ_i y_{i,s}) 
                # This is rewritten as: Σ_i y_{i,s} ≥ n/2 - t
                epigraph_cons.append(expr >= n/2 - t)
            return epigraph_cons

            prob = cp.Problem(cp.Minimize(t), constraints + y_cons + epigraph_cons)

                max_corr = np.max(corr)
                # Convert back to C5 value
                C5_val = max_corr * dx
                
                # Check if this is a violation
                if C5_val > t_val + 1e-8:
                    violated_s = np.argmax(corr) - (len(corr) // 2)
                    if violated_s not in shifts:
                        shifts.append(violated_s)
                        if len(shifts) > 1000:  # safety limit
                            break
                else:  # converged

                # Handle solver failure gracefully
                break
