import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smart_initialization():
    """Create a more intelligent initial solution based on symmetry and known good patterns"""
    # Start with a symmetric piecewise constant function
    h = np.ones(n_points) * 0.5
    
    # Create an initial good pattern by adding some structure
    # The idea is to have high values in some regions and low in others
    # to create good overlap suppression
    
    # Select some intervals to set to 1.0 and others to 0.0 (or near 0)
    interval_size = n_points // 5  # Make 5 intervals
    
    for i in range(5):
        start = i * interval_size
        end = start + interval_size
        if i % 2 == 0:
            h[start:end] = 1.0  # Set to 1
        else:
            h[start:end] = 0.0  # Set to 0
    
    # Make sure sum is correct
    h = project_to_sum_box(h, TARGET_SUM)
    
    # Add slight perturbations to avoid local minima
    noise = np.random.normal(0, 0.01, n_points)
    h = np.clip(h + noise, 0, 1)
    h = project_to_sum_box(h, TARGET_SUM)
    
    return h

def gradient_based_optimization(h0, max_iter=1000):
    """Optimize using scipy's minimize with bounds and constraints"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def local_search(h0, max_iter=1000):
    """Perform local search with careful mutations"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Try changing a few elements
        change_count = 0
        indices_to_change = np.random.choice(n_points, size=min(5, n_points//10), replace=False)
        
        for idx in indices_to_change:
            old_val = h[idx]
            
            # Try small perturbations
            for delta in [-0.03, -0.02, -0.01, 0.01, 0.02, 0.03]:
                h_new = h.copy()
                h_new[idx] += delta
                
                if h_new[idx] < 0 or h_new[idx] > 1:
                    continue
                    
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                new_c5 = evaluate_c5(h_new)
                
                if new_c5 < best_c5:
                    h = h_new.copy()
                    best_c5 = new_c5
                    change_count += 1
                    break
            else:
                continue
            break
        
        # If we didn't make progress, perturb slightly
        if change_count == 0 and np.random.random() < 0.3:
            idx = np.random.randint(0, n_points)
            delta = np.random.uniform(-0.01, 0.01)
            h[idx] += delta
            h = project_to_sum_box(h, TARGET_SUM)
            best_c5 = evaluate_c5(h)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = smart_initialization()
    best_c5 = evaluate_c5(best_h)
    
    # Main optimization loop with different strategies
    iterations_done = 0
    
    while time.time() < start_time + budget_s:
        # Phase 1: Gradient-based optimization with diverse restarts
        for _ in range(5):
            if time.time() > start_time + budget_s:
                break
                
            h_start = smart_initialization()
            h_opt = gradient_based_optimization(h_start, max_iter=200)
            c5_opt = evaluate_c5(h_opt)
            
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                
        # Phase 2: Local search to further improve
        h_improved = local_search(best_h.copy(), max_iter=300)
        c5_improved = evaluate_c5(h_improved)
        
        if c5_improved < best_c5:
            best_h = h_improved.copy()
            best_c5 = c5_improved
            
        # Phase 3: Add some diversity to prevent overfitting
        if np.random.random() < 0.5:
            # Start with random noisy initialization
            h_rand = np.random.uniform(0, 1, n_points)
            h_rand = project_to_sum_box(h_rand, TARGET_SUM)
            h_opt_rand = gradient_based_optimization(h_rand, max_iter=150)
            c5_rand = evaluate_c5(h_opt_rand)
            
            if c5_rand < best_c5:
                best_h = h_opt_rand.copy()
                best_c5 = c5_rand
                
        iterations_done += 1
        
        if iterations_done > 100:
            # Early exit condition if no more improvement
            break
    
    return (best_h.tolist(), float(best_c5), n_points)
