def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def get_interval_bounds(lens, starts):
    """Given interval lengths, return interval start and end points."""
    bounds = []
    s = 0
    for l in lens:
        bounds.append((s, s + l))
        s += l
    return bounds

def compute_overlap_for_shift(a_vals, lens, k):
    """Compute the overlap integral for shift k using piecewise constant."""
    n = len(a_vals)
    total = 0.0
    
    # For linear term: J(k) = ∫₀^{2‑k} h(x) dx
    linear_total = 0.0
    
    # For quadratic term: ∫₀^{2‑k} h(x)·h(x+k) dx
    quadratic_total = 0.0
    
    # 1. Linear term integration
    for i in range(n):
        s1, s2 = get_interval_bounds(lens, [0]*n)[i]
        s1 = max(s1, 0)
        s2 = min(s2, 2)
        
        # overlap with [0, 2-k]
        overlap_start = max(s1, 0)
        overlap_end = min(s2, 2 - k)
        if overlap_end > overlap_start:
            length = overlap_end - overlap_start
            linear_total += a_vals[i] * length
        else:
            length = 0
            
        # 2. Quadratic term integration
        for j in range(n):
            s1_j, s2_j = get_interval_bounds(lens, [0]*n)[j]
            s1_j = max(s1_j, 0)
            s2_j = min(s2_j, 2)

            # overlap of [s1_i, s2_i] with [s1_j+k, s2_j+k]
            overlap_start_quadratic = max(s1_i, s1_j + k)
            overlap_end_quadratic = min(s2_i, s2_j + k)
            
            if overlap_end_quadratic > overlap_start_quadratic:
                length_quadratic = overlap_end_quadratic - overlap_start_quadratic
                quadratic_total += a_vals[i] * a_vals[j] * length_quadratic
            else:
                length_quadratic = 0

    # Return the value of the integral
    return linear_total - quadratic_total

def compute_all_critical_shifts(lens):
    """Return list of critical shifts for which overlap might actually be zero."""
    # Critical values are potential breakpoints in the formula,
    # ranging from 0 to 2 (plus boundary points from each interval).
    # In this case, there are at most m*(m+1) relevant shifts.
    shifts = set()
    shifts.add(0)  # initial shift
    shifts.add(2)  # final shift
    
    # Add interval boundaries to shifts list for completeness
    total_length = sum(lens)
    current = 0
    for i in range(len(lens)):
        current += lens[i]
        shifts.add(current)
        
    return sorted(list(shifts))

def compute_c5_piecewise(a_vals, lens):
    """Compute C5 for the piecewise constant function defined by heights and lengths."""
    critical_shifts = compute_all_critical_shifts(lens)
    max_overlap = -float('inf')
    
    # Compute overlap integral over each critical shift and track maximum
    for k in critical_shifts:
        overlap = compute_overlap_for_shift(a_vals, lens, k)
        if overlap > max_overlap:
            max_overlap = overlap
            
    return max_overlap

def generate_candidate_piecewise(m, indices=None):
    """Generate a candidate piecewise constant solution."""
    # Generate initial candidate with small number of intervals
    if indices is None:
        # Create intervals with heights 0, 0.5, and 1.
        a_vals = [0.0, 0.5, 1.0]
        len_vals = [0.5, 1.0, 0.5]
        
        # normalize to length 2, and ensure mass equals 1
        total_len = sum(len_vals)
        len_vals = [l/total_len*2 for l in len_vals]
        
        # adjust a_vals to satisfy mass constraint
        total_mass = sum(a_val * l for a_val, l in zip(a_vals, len_vals))
        a_vals = [a * (1.0/total_mass) for a in a_vals]
        return a_vals, len_vals

def generate_initial_guesses(num_guesses=50, m=5):
    """Generate several initial guesses."""
    initial_guesses = []
    for i in range(num_guesses):
        # Start with two halves at heights 0 and 1, maybe with one in between
        # variants: [0,1], [1,0], [0,0.5,1], ...
        
        # pick random number of intervals and set them to uniform
        lst = []
        starting_pos = 0
        for j in range(m-1):
            # create a random length for interval j+1, constrained so that sum
            # equals 2
            remaining = 2 - starting_pos
            if j < m - 2:
                width = np.random.uniform(0.01, 0.7 * remaining)
            else:
                width = remaining
            lst.append(width)
            starting_pos += width
        lens = lst + [2 - starting_pos]

        # heights
        a_vals = np.random.uniform(0, 1, m)
        # adjust to satisfy mass = 1
        mass = sum(a * l for a, l in zip(a_vals, lens))
        a_vals = [a * (1.0 / mass) for a in a_vals]
        initial_guesses.append((a_vals, lens))

    return initial_guesses

def is_feasible(a_vals, lens):
    """Check if the piecewise function is feasible."""

    eps = 1e-10
    total_mass = sum(a * l for a, l in zip(a_vals, lens))
    if abs(total_mass - 1.0) > eps:
        return False

    if any(a < 0 or a > 1 for a in a_vals):
        return False
    
    if any(l <= 0 for l in lens):
        return False

    return True

def project_to_feasible(a_vals, lens):
    """Make a_vals and lens feasible."""
    total_mass = sum(a * l for a, l in zip(a_vals, lens))
    a_vals = [a / total_mass for a in a_vals]
    
    # make lens positive
    lens = [max(1e-10, l) for l in lens]
    
    # normalize lens to sum to 2
    total_len = sum(lens)
    lens = [l * (2.0 / total_len) for l in lens]
    
    # ensure initial parameter validation
    a_vals = [max(0, min(1, a)) for a in a_vals]

    return a_vals, lens
    
def refine_solution(a_vals, lens, tar_len=2):

    h_fine_new = np.zeros(n_points)
    lens_fine = [l * n_points / 2 for l in lens]
    
    # distribute lengths over fine grid
    idx_start = 0
    for a, l in zip(a_vals, lens):
        l_idx = int(round(l * n_points / 2))
        h_fine_new[idx_start:idx_start+l_idx] = a
        idx_start += l_idx
        
    # Normalize on discrete space
    total_mass = sum(h_fine_new)
    h_fine_new = h_fine_new * (2 / total_mass)
    
    return h_fine_new
