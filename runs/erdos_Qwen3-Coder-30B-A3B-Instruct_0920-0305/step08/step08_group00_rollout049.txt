def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1-h, mode="full")
    return dx * np.max(corr)

def solve_milp(n):
    """Solve the MILP formulation to get an optimal or near-optimal h"""
    # Variables
    h = cp.Variable(n, boolean=True)
    C = cp.Variable()

    constraints = [
        cp.sum(h) == n // 2,  # exactly half ones
        C >= 0
    ]

    # This represents w_{i,k} = h_i * (1 - h_{i+k})
    # and m_k = sum w_{i,k} <= C
    
    for k in range(-n+1, n):
        if k >= 0:
            # shifts k to the right: h_i * (1 - h_{i+k})
            # i from 0 to n-1-k
            start = 0
            end = n - k
        else:

            # i from -k to n-1
            start = -k
            end = n
        
        if k >= 0:
            indices = list(range(start, end))
        else:
            indices = list(range(start, end))
        
        # For shift k, sum_m (h_i * (1 - h_{i+k})) <= C
        if indices:
            # Use binary variables y_{i,k} = h_i * h_{i+k}
            # Add auxiliary constraint for each i
            for i in indices:
                if k >= 0:
                    j = i + k
                else:
                    j = i + k
                if 0 <= j < n:
                    y = cp.Variable(boolean=True)
                    constraints.append(y <= h[i])
                    constraints.append(y <= h[j])

                    constraints.append(y >= h[i] + h[j] - 1)

            constraints.append(total <= C)

    # We are still not modeling properly how to use "max" in the objective. 
    # So, we instead setup very many variables and fit the constraints that C >= W_k.
    # A more robust way to do this is:
    
    # Let's instead compute all possible correlation sums and bound each via linear constraints.
    # i,k take following values:
    # k in {-n+1...n-1}
    # for each k: i in [0, n-1] such that (i+k) is valid (i+k < n and i+k >= 0)

    # Recovery based on efficient constraint build:
    # Note: we need to find match bounds for each shift
    # So define: for each k, compute mismatch > (h_i * (1 - h_{i+k})), 
    # using bivariate binary products.

    M = []

        # Loop over i=0..n-1-k or len indices
        if k >= 0:  # i+k is valid when i <= n-1-k
            valid_indices = list(range(n-k))  # i from 0 to n-1-k
        else:
            valid_indices = list(range(-k, n))  # i from -k to n-1
        expr = 0
        for i in valid_indices:
            j = i + k
            if 0 <= j < n:
                # Replace h_i * (1 - h_j) with linearized form
                y = cp.Variable(boolean=True)  # y == h_i AND h_j
                constraints.append(y <= h[i])
                constraints.append(y <= h[j])
                constraints.append(y >= h[i] + h[j] - 1)
                expr += h[i] - y

        constraints.append(expr <= C)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a = -1.0
    b = 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    result = np.clip(x + (a + b)/2, 0, 1)
    return result
