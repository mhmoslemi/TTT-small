import numpy as np
from scipy.signal import correlate
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    c_shift = (j - i) + (len(h) - 1)
    c[c_shift] += delta_i * b[i] - delta_j * h[i]

    # H part: slide from j to j+n
    start_h = max(0, j)
    end_h = min(len(h), j + len(h))

        c[start_h:end_h] -= delta_j * h[j + (len(h)-1):j + (len(h)-1) + (end_h - start_h)][::-1]

    return c

        if total_moves % K_step == 0:

                no_improvement_count = 0
            delta = max(delta * alpha_delta, min_delta)
    
    return (best_h.tolist(), float(best_value), n)
