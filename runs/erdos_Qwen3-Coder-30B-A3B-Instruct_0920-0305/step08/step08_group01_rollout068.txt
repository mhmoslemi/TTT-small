TARGET_SUM = None

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def mutate_solution(h, rate=0.1, amp=0.1):
    """Mutate solution by adding Gaussian noise to random positions"""
    h_new = h.copy()
    indices = np.random.choice(n_points, size=int(n_points * rate), replace=False)
    for idx in indices:
        noise = np.random.normal(0, amp)
        h_new[idx] += noise
    return project_to_sum_box(h_new, TARGET_SUM)

def crossover(parent1, parent2, rate=0.3):
    """Crossover two solutions"""
    h_new = parent1.copy()

    for idx in indices:
        h_new[idx] = parent2[idx]
