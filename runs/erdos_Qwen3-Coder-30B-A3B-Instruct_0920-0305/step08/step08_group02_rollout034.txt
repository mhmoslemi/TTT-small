import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from scipy.optimize import Bounds

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    #Phase 1: Use MILP to get an integer solution
    print("Starting MILP phase...")
    h_milp = None
    best_c5_milp = float("inf")
    try:
        # Binary variable for the MILP
        y = cp.Variable(n_points, boolean=True)
        # Auxiliary variables to capture y[i]*y[i+k] products (we only care about non-zero shifts)
        w = {k: cp.Variable((n_points - k,), nonneg=True) for k in range(1, n_points)}
        # Objective - maximize a lower bound t on autocorrelations
        t = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(y) == TARGET_SUM,
            t <= TARGET_SUM  # lower bound on total autocorrelation starting value
        ]
        
        # Add constraints for w variables
        for k in range(1, n_points):
            for i in range(n_points - k):
                # w_{k,i} = y_i * y_{i+k}
                constraints += [
                    w[k][i] <= y[i],
                    w[k][i] <= y[i+k],
                    w[k][i] >= y[i] + y[i+k] - 1,
                    w[k][i] >= 0
                ]
        
        # Add constraints that all autocorrelations are above t
        for k in range(1, n_points):
            constraints.append(cp.sum(w[k]) >= t)
        
        # Objective
        objective = cp.Maximize(t)
        prob = cp.Problem(objective, constraints)
        
        # Solve with limited time
        time_left = budget_s * 0.4
        prob.solve(solver=cp.CBC, max_seconds=time_left)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            y_sol = y.value
            # Recover the continuous value from binary
            h_milp = np.array(y_sol)
            best_c5_milp = evaluate_c5(h_milp)
            print(f"MILP approach gave C5 = {best_c5_milp}")
        else:
            print("MILP did not solve optimally, using random start")
    except Exception as e:
        print(f"Error in MILP: {e}")
        print("Using random initial warm-up")
    
    if h_milp is None:
        h_milp = np.random.uniform(0, 1, n_points)
        h_milp = project_to_sum_box(h_milp, TARGET_SUM)
        best_c5_milp = evaluate_c5(h_milp)
        print(f"Random start gave C5 = {best_c5_milp}")

    # Phase 2: Do continuous refinement with exact gradient
    print("Starting gradient descent...")
    h_refined = h_milp.copy()
    best_h = h_refined.copy()
    best_c5 = best_c5_milp
    time_left = budget_s * 0.35
    
    # Compute subgradient for the worst shift
    def compute_subgradient(h):
        # Evaluate all autocorrelations
        corr = compute_corr_fft(h)
        max_ind = np.argmax(corr)
        k_worst = max_ind - (n_points - 1)  # Transform index to shift
        if k_worst < 0:
            k_worst = k_worst + n_points
        
        # Compute gradient for current k_worst
        g = np.zeros_like(h)
        for i in range(n_points):
            # Term for h_{i+k}
            if i + k_worst < n_points:
                g[i] += -h[i + k_worst]
            # Term for h_{i-k}
            if i - k_worst >= 0:
                g[i] += -h[i - k_worst]
            # Term for h_i
            g[i] += 1
        
        return g, k_worst
    
    # Gradient descent loop
    step_size = 0.5 * dx
    for i in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
        g, _ = compute_subgradient(h_refined)
        h_new = h_refined - step_size * g
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        h_refined = h_new.copy()
        # Adaptive step size
        step_size *= 0.9995
        if i % 1000 == 0:
            print(f"After {i} iterations: C5 = {best_c5}")
    
    # Final improvement with local search
    print("Final local search...")
    h_final = best_h.copy()
    best_c5_final = best_c5
    
    # Perturb and try small variations
    trial_count = 0
    while time.time() - start_time < budget_s - 1 and trial_count < 1000:
        # Small random changes
        idx_changes = np.random.choice(n_points, size=min(n_points//10, 10), replace=False)
        h_trial = h_final.copy()
        for idx in idx_changes:
            # Make small change
            h_trial[idx] += np.random.normal(0, 0.01)
        h_trial = project_to_sum_box(h_trial, TARGET_SUM)
        new_c5 = evaluate_c5(h_trial)
        if new_c5 < best_c5_final:
            h_final = h_trial.copy()
            best_c5_final = new_c5
        trial_count += 1
    
    print(f"Final result C5 = {best_c5_final}")
    return (h_final.tolist(), float(best_c5_final), n_points)
