def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def solve_milp_round(x_init, n, max_shifts=20):
    """Solve a MILP using row-generation approach to find exact C5 value"""
    # Create variable for h values (binary)
    x = Variable(n, boolean=True)
    t = Variable(1, nonneg=True)  # Epigraph variable
    
    # Mass constraint
    constraints = [cp_sum(x) == n // 2]
    
    # Generate small initial set of shifts to start with
    shifts = list(range(1, min(max_shifts, n)))
    y_vars = {}
    
    # Initialize all auxiliary variables y_{i,s} and their constraints
    for s in shifts:
        for i in range(n):
            if i + s < n:
                var = Variable(1, boolean=True)
                y_vars[(i, s)] = var
                constraints.extend([
                    var <= x[i],
                    var <= x[i + s],
                    var >= x[i] + x[i + s] - 1
                ])
    
    # Epigraph constraints for current shifts
    for s in shifts:
        expr = 0
        for i in range(n):
            if i + s < n:
                expr += y_vars[(i, s)]
        constraints.append(expr >= n // 2 - t)
    
    # Objective 
    objective = Minimize(t)
    prob = Problem(objective, constraints)
    
    # Solve and check for violations

    prob.solve(solver=cp.CBC, verbose=False, max_seconds=120)
    
    x_sol = np.round(x.value).astype(int)
    t_sol = t.value
    
    if t_sol is None:
        return x_init, np.inf
    
    # Now evaluate actually full correlations to ensure no violations

    max_corr = np.max(corr_full[:n])  # Since our shifts should cover all meanings
    
    # Check if this is the actual upper bound or not
    if np.abs(max_corr - t_sol) > 1e-6 and time.time() - start_eval < 10:  # Refine if needed
        additional_shifts = []

            if corr_full[i - 1] > t_sol + 1e-6:
                additional_shifts.append(i - 1)

        if additional_shifts:
            s_max = additional_shifts[0]
            for s in additional_shifts:
                if corr_full[s] > corr_full[s_max]:
                    s_max = s
            
            shifts.append(s_max)
            # Add all new auxiliary products
            for i in range(n):
                if i + s_max < n:
                    var = Variable(1, boolean=True)
                    y_vars[(i, s_max)] = var
                    constraints.extend([
                        var <= x[i],
                        var <= x[i + s_max],
                        var >= x[i] + x[i + s_max] - 1
                    ])
            
            # Add epigraph constraint for this shift
            expr = 0
            for i in range(n):
                if i + s_max < n:
                    expr += y_vars[(i, s_max)]
            constraints.append(expr >= n // 2 - t)

            prob.solve(solver=cp.CBC, verbose=False, max_seconds=300)
            x_sol = np.round(x.value).astype(int)
            t_sol = t.value
    
    return x_sol, t_sol

        return t + (1.0 / gamma) * np.log(np.sum(np.exp(gamma * (corr - t))))

    def objective(x):
        return smooth_max(x, t_val) * dx

    def constraint_sum(x):
        return np.sum(x) - n // 2
    
    bounds = [(0, 1) for _ in range(n)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    res = minimize(objective, x_start, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-8})
    
    if res.success:
        return res.x
    else:
        return x_start
