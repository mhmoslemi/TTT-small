import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def compute_J(k, L, A, K_crit):
    """Compute J(k; L, A) analytically - overlap integral at shift k"""
    m = len(L)
    s = [0] + list(np.cumsum(L))
    total_mass = 0
    product_term = 0
    # Linear term: ∫ h(x) dx over [0, 2-k]
    for i in range(m):
        if s[i] >= 2 - k:
            break
        left = max(s[i], 0)
        right = min(s[i+1], 2-k)
        if right > left:
            total_mass += A[i] * (right - left)
    
    # Product term: ∑ a_p a_q ω_{pq}(k)
    for p in range(m):
        for q in range(m):
            left_p = s[p]
            right_p = s[p+1]
            left_q = s[q] + k
            right_q = s[q+1] + k
            overlap_left = max(left_p, left_q)
            overlap_right = min(right_p, right_q)
            if overlap_right > overlap_left:
                product_term += A[p] * A[q] * (overlap_right - overlap_left)
    
    return total_mass - product_term

def generate_critical_shifts(L):
    """Generate critical shifts for which the overlap function changes"""
    s = [0] + list(np.cumsum(L))
    critical = set()
    for i in range(len(s)):
        for j in range(len(s)):
            critical.add(abs(s[i] - s[j]))
            if i < len(s) - 1 and j < len(s) - 1:
                critical.add(2 - s[i] - (2 - s[j]))
                critical.add(s[i])
                critical.add(2 - s[j])
    return sorted(list(critical))

def optimize_step_function():
    """Optimize step function with a fixed number of intervals"""
    m = 5  # Five intervals
    bounds = [(1e-6, 2.0) for _ in range(m-1)] + [(0, 1.0) for _ in range(m-1)]
    
    def objective(x):
        L = list(x[:m-1])
        L.append(2.0 - sum(L))
        A = list(x[m-1:])
        # Compute the last height from the mass constraint
        if sum(A[i] * L[i] for i in range(m-1)) > 1:
            A[-1] = 0.0
        else:
            A[-1] = (1.0 - sum(A[i] * L[i] for i in range(m-1))) / L[-1]
        A[-1] = max(0.0, min(1.0, A[-1]))
        critical_shifts = generate_critical_shifts(L)
        values = [compute_J(k, L, A, critical_shifts) for k in critical_shifts]
        return max(values) if values else 0
    
    # Initial guess: uniform lengths and alternating heights
    initial_lengths = [2.0/m] * (m-1)
    initial_heights = [1.0 if i % 2 == 0 else 0.0 for i in range(m-1)]
    initial_guess = initial_lengths + initial_heights
    
    res = minimize(objective, initial_guess, method='L-BFGSB', bounds=bounds, options={'maxiter': 500})
    if not res.success:
        print("Optimization did not converge")
        return None, float('inf')
    
    L_opt = list(res.x[:m-1])
    L_opt.append(2.0 - sum(L_opt))
    A_opt = list(res.x[m-1:])
    last_area = 1.0 - sum(A_opt[i] * L_opt[i] for i in range(m-1))
    A_opt.append(max(0.0, min(1.0, last_area / L_opt[-1])))
    
    critical_shifts = generate_critical_shifts(L_opt)
    final_c5 = max(compute_J(k, L_opt, A_opt, critical_shifts) for k in critical_shifts)
    return L_opt, A_opt, final_c5

def construct_h_from_step_params(L, A):
    """Construct high-resolution h from step parameters"""
    m = len(L)
    h = np.zeros(n_points)
    s = [0] + list(np.cumsum(L))
    dx = 2.0 / n_points
    for i in range(n_points):
        x = i * dx
        for j in range(m):
            if s[j] <= x < s[j+1]:
                h[i] = A[j]
                break
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Multi-start to avoid local optima
    best_c5 = float('inf')
    best_h = None
    num_restarts = 10
    for _ in range(num_restarts):
        L_opt, A_opt, c5 = optimize_step_function()
        if c5 < best_c5:
            best_c5 = c5
            h = construct_h_from_step_params(L_opt, A_opt)
            best_h = h.copy()
        if time.time() - start_time > budget_s * 0.95:
            break
    
    # Polish the result with a few iterations of local refinement if needed
    if best_h is not None and best_c5 > 0.379:
        h_polished = best_h.copy()
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            idx = np.random.randint(0, n_points)
            h_prev = h_polished[idx]
            new_val = np.clip(h_prev + np.random.normal(0, 0.005), 0, 1)
            h_changed = h_polished.copy()
            h_changed[idx] = new_val
            # Adjust the sum to be 1
            current_sum = np.sum(h_changed) * dx
            if abs(current_sum - 1.0) > 1e-10:
                # Adjust all values proportionally to fix the sum
                factor = 1.0 / current_sum
                h_changed *= factor
            proposed_c5 = evaluate_c5(h_changed)
            if proposed_c5 < best_c5:
                best_c5 = proposed_c5
                best_h = h_changed.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
