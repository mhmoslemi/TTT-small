import numpy as np
from scipy.signal import correlate
import time
import random

# Global constants
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_corr_int(h):
    """Compute integer correlation counts"""
    corr = compute_corr_fft(h)
    corr_int = np.rint(corr).astype(int)
    return corr_int

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr_int = compute_corr_int(h)
    return dx * np.max(corr_int)

def simulated_annealing(n_points, time_budget):
    """Perform simulated annealing to minimize C5"""
    start_time = time.time()
    
    # Initialize binary vector with exactly n/2 ones
    h = np.zeros(n_points, dtype=int)
    h[:n_points//2] = 1
    np.random.shuffle(h)
    
    best_h = h.copy()
    best_corr = compute_corr_int(best_h)
    best_max = best_corr.max()
    
    current_h = h.copy()
    current_corr = best_corr.copy()
    current_max = best_max
    
    # Temperature schedule
    T = 1.0
    cooling_rate = 0.9995
    
    iterations = 0
    while time.time() - start_time < time_budget:
        # Generate candidate by swapping two random positions
        ones_idx = np.where(current_h == 1)[0]
        zeros_idx = np.where(current_h == 0)[0]
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
            
        i_one = random.choice(ones_idx)
        i_zero = random.choice(zeros_idx)
        
        # Create candidate
        cand_h = current_h.copy()
        cand_h[i_one] = 0
        cand_h[i_zero] = 1
        
        # Compute new correlation
        cand_corr = compute_corr_int(cand_h)
        cand_max = cand_corr.max()
        
        # Accept or reject
        delta = cand_max - current_max
        if delta <= 0 or random.random() < np.exp(-delta / max(T, 1e-10)):
            current_h = cand_h
            current_corr = cand_corr
            current_max = cand_max
            
            # Update best
            if cand_max < best_max:
                best_h = cand_h.copy()
                best_corr = cand_corr.copy()
                best_max = cand_max
                
        # Cool down
        T *= cooling_rate
        iterations += 1
        
        # Occasionally reheat
        if iterations % 1000 == 0:
            T = max(T * 0.99, 0.001)
            
    return best_h, best_max

def hill_climb(h, max_iter=10000, patience=1000):
    """Local hill climbing to refine solution"""
    best_h = h.copy()
    best_corr = compute_corr_int(best_h)
    best_max = best_corr.max()
    
    no_improve = 0
    for _ in range(max_iter):
        # Generate random swaps
        improvements = []
        for _ in range(50):
            ones_idx = np.where(best_h == 1)[0]
            zeros_idx = np.where(best_h == 0)[0]
            
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
                
            i_one = random.choice(ones_idx)
            i_zero = random.choice(zeros_idx)
            
            # Create candidate
            cand_h = best_h.copy()
            cand_h[i_one] = 0
            cand_h[i_zero] = 1
            
            # Compute correlation
            cand_corr = compute_corr_int(cand_h)
            cand_max = cand_corr.max()
            
            if cand_max < best_max:
                improvements.append((cand_h, cand_max, i_one, i_zero))
                
        if improvements:
            # Pick the best improvement
            improvements.sort(key=lambda x: x[1])
            best_improvement = improvements[0]
            best_h = best_improvement[0]
            best_max = best_improvement[1]
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= patience:
                break
                
    return best_h, best_max

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points / 2
    n_points = 200
    dx = 2.0 / n_points
    
    best_h = None
    best_c5 = float('inf')
    
    # Multi-start approach with different initializations
    for run_id in range(10):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Different initialization strategies
        if run_id == 0:
            # Random balanced
            h = np.zeros(n_points, dtype=int)
            h[:n_points//2] = 1
            np.random.shuffle(h)
        elif run_id == 1:
            # Alternating pattern (0,1,0,1...)
            h = np.array([i % 2 for i in range(n_points)])
        elif run_id == 2:
            # Blocks pattern - ones in middle
            h = np.zeros(n_points, dtype=int)
            start_idx = n_points // 4
            end_idx = 3 * n_points // 4
            h[start_idx:end_idx] = 1
        else:
            # Random with different seed
            np.random.seed(seed + run_id)
            h = np.zeros(n_points, dtype=int)
            h[:n_points//2] = 1
            np.random.shuffle(h)
            
        # Simulated annealing
        h_anneal, max_corr = simulated_annealing(n_points, budget_s * 0.1)
        
        # Local hill climbing
        h_refined, max_corr_refined = hill_climb(h_anneal, max_iter=5000, patience=500)
        
        if max_corr_refined < best_c5:
            best_c5 = max_corr_refined
            best_h = h_refined.copy()
    
    # Final optimization
    final_h, final_max = hill_climb(best_h, max_iter=10000, patience=1000)
    if final_max < best_c5:
        best_c5 = final_max
        best_h = final_h.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
