def compute_overlap_intervals(a, l, k):
    """Compute overlap integral for a fixed shift k"""
    # All intervals [s_{j-1}, s_j)
    s = [0]
    for j in range(len(l)):
        s.append(s[-1] + l[j])
    
    # Linear term: ∫₀^{2-k} h(x) dx
    linear = 0.0
    for j in range(len(l)):
        start = max(s[j], 0)
        end = min(s[j+1], 2 - k)
        if end > start:
            linear += a[j] * (end - start)
    
    # Quadratic term: ∫₀^{2-k} h(x)h(x+k) dx
    quadratic = 0.0
    for p in range(len(l)):
        for q in range(len(l)):
            start = max(s[p], s[q] + k)
            end = min(s[p+1], s[q+1] + k)
            if end > start:
                quadratic += a[p] * a[q] * (end - start)
    
    return linear - quadratic

def evaluate_c5_parametric(a, l):
    """Evaluate max overlap for given piecewise constant function h(x)"""
    # Total length is 2; we have m intervals
    s = [0]
    for j in range(len(l)):
        s.append(s[-1] + l[j])
    
    # Compute critical shifts for varying k in [0, 2]
    # At each boundary and also when overlap changes
    critical_k = []
    for j in range(len(l) + 1):  
        critical_k.append(s[j])
        if j > 0 and j < len(l):
            critical_k.append(s[j] - 0.0001)  # small perturbation to ensure metric
            critical_k.append(s[j] + 0.0001)
    
    # some additional numerical critical points
    critical_k += [k/100 for k in range(1, 200)]
    critical_k = list(set(critical_k))
    critical_k = [k for k in critical_k if 0 <= k <= 2]
    
    # Evaluate J(k) for all critical k values
    values = []
    for k in critical_k:
        if k <= 2:  # prevent out-of-bounds issues
            Jk = compute_overlap_intervals(a, l, k)
            values.append(Jk)
            
    return max(values)

def project_to_sum(a, l, target_sum):
    """Project a, l to satisfy sum(a*l) = target_sum"""
    total_mass = sum(a[j]*l[j] for j in range(len(a)))
    if abs(total_mass - target_sum) < 1e-10:
        return a, l
    scale = target_sum / total_mass
    a_new = [a[j]*scale for j in range(len(a))]
    return a_new, l
