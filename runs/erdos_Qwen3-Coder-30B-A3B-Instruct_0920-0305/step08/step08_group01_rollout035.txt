import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.linalg import cholesky
from scipy.optimize import minimize

# Global variable
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def build_shift_matrices(n):
    """Build shift matrices for all valid shifts (1 to n//2)"""
    R_list = []
    for s in range(1, n//2 + 1):
        R_s = np.zeros((n, n))
        for i in range(n - s):
            R_s[i, i + s] = 1
        R_list.append(R_s)
    return R_list

def solve_sdp(n, R_list, max_time=100):
    """Solve the SDP relaxation"""
    # Variables
    X = cp.Variable((n, n), symmetric=True)
    mu = cp.Variable(n)
    lam = cp.Variable()

    # Constraints
    constraints = [
        X >> 0,  # PSD
        cp.diag(X) == mu,  # diag(X) = mu
        0 <= mu,
        mu <= 1,  # bounds on mu
        cp.sum(mu) == TARGET_SUM,  # mass constraint
    ]

    # Add shift constraints
    for R_s in R_list:
        constraints.append(cp.trace(R_s @ X) >= lam)

    # Objective
    objective = cp.Maximize(lam)
    problem = cp.Problem(objective, constraints)

    # Solve
    try:
        problem.solve(solver=cp.SCS, max_iters=5000, time_limit=max_time)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return X.value, mu.value, lam.value
        else:
            print(f"SDP solve status: {problem.status}")
            return None, None, None
    except Exception as e:
        print(f"SDP solve error: {e}")
        return None, None, None

def randomized_rounding(V, n):
    """Hyperplane rounding approach with thresholding"""
    num_rounds = 500

    best_x = None
    best_c5 = float('inf')

    for _ in range(num_rounds):
        g = np.random.randn(V.shape[1])
        scores = V @ g
        # Set top half to 1, others to 0
        sorted_indices = np.argsort(-scores)
        x = np.zeros(n)
        x[sorted_indices[:n//2]] = 1

        # Check mass constraint
        if np.abs(np.sum(x) - TARGET_SUM) > 1e-8:
            # Adjust if necessary
            ones = np.where(x == 1)[0]
            zeros = np.where(x == 0)[0]
            diff = int(np.sum(x) - TARGET_SUM)
            if diff > 0:
                # Need to lose some 1s
                for i in range(diff):
                    if i < len(ones):
                        x[ones[i]] = 0
            else:
                # Need to gain some 1s
                for i in range(-diff):
                    if i < len(zeros):
                        x[zeros[i]] = 1

        c5_val = evaluate_c5(x)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_x = x.copy()

    return best_x, best_c5

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    h = np.asarray(h)
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def hill_climb_local(x, max_iter=5000):
    """Incremental local search focusing on worst shift"""
    best_x = x.copy()
    best_c5 = evaluate_c5(best_x)
    corr = correlate(best_x, 1 - best_x, mode="full")

    for iteration in range(max_iter):
        s_star = np.argmax(corr)
        bad_indices = np.where((best_x == 1) & (np.roll(best_x, -s_star) == 0))[0]
        good_indices = np.where((best_x == 0) & (np.roll(best_x, -s_star) == 1))[0]

        if len(bad_indices) > 0 and len(good_indices) > 0:
            i = np.random.choice(bad_indices)
            j = np.random.choice(good_indices)

            # Do the swap
            new_x = best_x.copy()
            new_x[i] = 0
            new_x[j] = 1

            new_c5 = evaluate_c5(new_x)
            if new_c5 <= best_c5:
                best_x = new_x
                best_c5 = new_c5
                corr = correlate(best_x, 1 - best_x, mode="full")
        else:
            # try random swap occasionally
            if np.random.rand() < 0.1:
                i, j = np.random.choice(n_points, 2, replace=False)
                new_x = best_x.copy()
                new_x[i], new_x[j] = new_x[j], new_x[i]
                new_c5 = evaluate_c5(new_x)
                if new_c5 <= best_c5:
                    best_x = new_x
                    best_c5 = new_c5
                    corr = correlate(best_x, 1 - best_x, mode="full")

    return best_x, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None

    try:
        R_list = build_shift_matrices(n_points)
        remaining_budget = budget_s * 0.8  #留给SDP及回合两个阶段折扣

        # Phase 1: SDP solve and rounding
        solve_start = time.time()
        X_star, mu_star, lam_star = solve_sdp(n_points, R_list, max_time=remaining_budget*0.7)  # 70%的剩余时间用于SDP

        if X_star is not None:
            # Factorization and rounding
            V = cholesky(X_star, lower=True)  # Lower triangle
            x_rounded, c5_rounded = randomized_rounding(V, n_points)
            if x_rounded is not None and c5_rounded < best_c5:
                best_h = x_rounded
                best_c5 = c5_rounded

        # Phase 2: Local improvement
        if best_h is not None:
            refine_start = time.time()
            refined_h, refined_c5 = hill_climb_local(best_h, max_iter=5000)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h

        # Clean up final result
        if best_h is not None:
            return (best_h.tolist(), float(best_c5), n_points)
        else:
            # Fallback or error
            return ([0.5]*n_points, 1.0, n_points)

    except Exception as e:
        print("Error encountered", e)
        # Return fallback
        h_fallback = np.full(n_points, 0.5).tolist()
        return (h_fallback, 1.0, n_points)
