import numpy as np
from scipy.signal import correlate
import time

# Global constants
n_points = 256
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def project_to_box_sum(h, target_sum):
    """Project h onto [0,1] and rescale to sum = target_sum"""
    # Clip to [0,1]
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if abs(current_sum - target_sum) < 1e-12:
        return h_clipped
    
    # Binary search for lambda such that clip(h + lambda) sums to target
    def f(lam):
        h_test = np.clip(h_clipped + lam, 0, 1)
        return np.sum(h_test) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(h_clipped + (a + b)/2, 0, 1)
    return result

def compute_correlation(h):
    """Compute correlation c = correlate(h, 1-h, mode='full'), then return max(c) * dx"""
    b = 1 - h
    c = correlate(h, b, mode='full')
    return np.max(c) * dx

def fast_correlation_update(c, h, b, i, j, delta):
    """
    Update correlation vector c after a move:
        h[i] -= delta
        h[j] += delta
    """
    n = len(h)
    # Compute h_rev for efficient indexing
    h_rev = h[::-1]

    # Part from b[i]
    if 0 <= i < n:
        start = n - 1 - i
        end = start + n
        c[start:end] += delta * b[i:(i+n)]
    
    # Part from h[j]
    if 0 <= j < n:
        start = j
        end = start + n
        c[start:end] -= delta * h_rev[:n]
    
    # If the move redistribution results in a small absolute shift, consider adding back
    # the second order correction term (~delta**2)
    delta_square = delta * delta
    shift1 = j - i + (n - 1)
    shift2 = i - j + (n - 1)
    if 0 <= shift1 < 2 * n - 1:
        c[shift1] += delta_square
    if 0 <= shift2 < 2 * n - 1:
        c[shift2] += delta_square

    return c

def generate_initial_h():
    """Interpolate initial_h_values to n_points and project to feasible set"""
    h0 = np.array(initial_h_values)
    x_orig = np.linspace(0, 2, len(h0))
    x_new = np.linspace(0, 2, n_points)
    h_interpolated = np.interp(x_new, x_orig, h0)
    return project_to_box_sum(h_interpolated, TARGET_SUM)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Stage 1: coarse run on smaller n (if time allows)
    h = generate_initial_h()
    b = 1 - h
    c = correlate(h, b, mode='full')
    Cmax = np.max(c)
    current_obj = Cmax * dx

    best_h = h.copy()
    best_obj = current_obj

    # Temperature and other control variables
    T = 0.01
    alpha_T = 0.995
    delta = 0.02
    delta_min = 5e-4
    step_count = 0
    temp_count = 0
    eval_count = 0
    no_improvement_count = 0
    max_no_improvement = 2000

    K_eval = 500
    K_temp = 500
    K_step = 200

    while time.time() - start_time < budget_s:
        # Proposal step
        i, j = np.random.randint(0, n_points, size=2)
        if i == j:
            continue
            
        # Compute max delta respecting bounds
        max_delta = min(h[i], 1 - h[j])
        if max_delta <= 1e-10:
            continue
        proposed_delta = min(delta, 0.9 * max_delta)
        
        # Update correlation incrementally
        c_old = c.copy()
        fast_correlation_update(c, h, b, i, j, -proposed_delta)
        c[j:i+1] -= proposed_delta * h[::-1][:i-j+1] if j > i else c[j:i+1] - proposed_delta * h[::-1][i-j:]
        # more accurate slice
        offset = i if i >= j else j
        if i < j:
            # i to offset index
            offset = i
            indices = range(offset, j)
            # update in chunks
            for idx in indices:
                if 0 <= idx < n_points:
                    try:
                        c[idx+n-1-offset : idx+n-1-offset+n] += proposed_delta * b[idx:idx+n]
                    except IndexError:
                        pass # realize proper indexing
            # symmetric update
            for idx in indices:
                if 0 <= idx < n_points:
                    try:
                        c[idx:idx+n] -= proposed_delta * h[::-1][:n]
                    except IndexError:
                        pass
        else:
            # i > j , same procedure but along with inverse
            indices = range(j, i+1)
            for idx in indices:
                if 0 <= idx < n_points:
                    c[idx+n-1-i : idx+n-1-i+n] += proposed_delta * b[idx:idx+n]
            for idx in indices:
                if 0 <= idx < n_points:
                    c[idx:idx+n] -= proposed_delta * h[::-1][:n]

        # Exact value update
        new_Cmax = np.max(c)
        new_obj = new_Cmax * dx

        # Accept according to simulated annealing rule
        if new_obj < current_obj or np.random.rand() < np.exp(-(new_obj - current_obj) / T):
            h[i] -= proposed_delta
            h[j] += proposed_delta
            b = 1 - h
            current_obj = new_obj
            Cmax = new_Cmax

            # Clamp values exactly to avoid drift 
            h = project_to_box_sum(h, TARGET_SUM)
            b = 1 - h
            # Recompute correlation to be safe
            try:
                c = correlate(h, b, mode='full')
            except:
                pass

            # Update best
            if current_obj < best_obj:
                best_h = h.copy()
                best_obj = current_obj
                no_improvement_count = 0
        else:
            # Revert correlation changes
            # Restore c from c_old (cba to track the exact changes; better to reset periodically)
            no_improvement_count += 1

        # Periodic recomputation of correlation (e.g., every 500 steps)
        eval_count += 1
        if eval_count % K_eval == 0:
            try:
                exact_c = correlate(h, b, mode='full')
                if abs(np.max(exact_c) - Cmax) > 1e-10:
                    c = exact_c
                    Cmax = np.max(c)
                    current_obj = Cmax * dx
            except:
                pass
            
        # Step size adjustment
        step_count += 1
        if step_count % K_step == 0:
            delta *= 0.9
            if delta < delta_min:
                delta = delta_min
                
        # Temperature cooling
        temp_count += 1
        if temp_count % K_temp == 0:
            T = T * alpha_T
            
        # Periodic reset if no improvement
        if no_improvement_count > max_no_improvement:
            T = max(0.001, T * 1.1)  # increase temperature if stuck
            no_improvement_count = 0

    return (best_h.tolist(), float(best_obj), n_points)
