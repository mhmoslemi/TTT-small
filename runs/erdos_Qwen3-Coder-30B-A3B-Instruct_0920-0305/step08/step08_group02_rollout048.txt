import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def initialize_with_refined_patterns():
    """Create initial solution with refined step function patterns"""
    h = np.zeros(n_points)
    
    # Add multiple step functions with different sizes and positions
    step_positions = [0.2, 0.4, 0.6, 0.8]
    step_heights = [0.8, 0.7, 0.6, 0.5]
    
    for pos, height in zip(step_positions, step_heights):
        start_idx = int(pos * n_points / 2)
        end_idx = int((pos + 0.2) * n_points / 2)
        if start_idx < n_points and end_idx > 0:
            h[max(0, start_idx):min(n_points, end_idx)] = height
    
    # Add some randomization to prevent local minima
    h += np.random.normal(0, 0.02, n_points)
    h = np.clip(h, 0, 1)
    
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def local_improvement_step(h, n_iter=100):
    """Apply local improvements to current solution"""
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    for _ in range(n_iter):
        # Create small perturbation
        h_new = best_h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.01)
        h_new[idx] += delta
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
    
    return best_h

def genetic_local_search(h0, generations=50, population_size=20):
    """Genetic algorithm approach with local search"""
    population = [h0.copy()]
    for _ in range(population_size - 1):
        h_mut = h0 + np.random.normal(0, 0.03, n_points)
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = project_to_sum_box(h_mut, TARGET_SUM)
        population.append(h_mut)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    
    for gen in range(generations):
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_c5(h)
            fitness.append((c5, h))
        
        fitness.sort(key=lambda x: x[0])
        if fitness[0][0] < best_c5:
            best_h = fitness[0][1].copy()
            best_c5 = fitness[0][0]
        
        # Create new population
        new_population = [fitness[0][1], fitness[1][1]]  # Keep top 2
        
        # Crossover and mutation
        for _ in range(population_size - 2):
            parent1 = fitness[np.random.randint(0, population_size//2)][1]
            parent2 = fitness[np.random.randint(0, population_size//2)][1]
            
            # Uniform crossover
            mask = np.random.random(n_points) > 0.5
            child = np.where(mask, parent1, parent2)
            
            # Add mutation
            mutate_idx = np.random.randint(0, n_points)
            child[mutate_idx] += np.random.normal(0, 0.02)
            
            child = np.clip(child, 0, 1)
            child = project_to_sum_box(child, TARGET_SUM)
            
            new_population.append(child)
        
        population = new_population
    
    return best_h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    
    # Phase 1: Initialize with multiple patterns
    initial_h = initialize_with_refined_patterns()
    
    # Phase 2: Apply genetic local search for more diverse search
    best_h = genetic_local_search(initial_h, generations=30, population_size=15)
    best_c5 = evaluate_c5(best_h)
    
    # Phase 3: Local refinements
    time_limit = start_time + budget_s
    
    while time.time() < time_limit:
        # Random local search
        refined = local_improvement_step(best_h, n_iter=50)
        new_c5 = evaluate_c5(refined)
        
        if new_c5 < best_c5:
            best_h = refined
            best_c5 = new_c5
        
        # Occasionally restart with new pattern
        if np.random.random() < 0.15:
            new_start = initialize_with_refined_patterns()
            best_h = new_start
        
        # Break if not improving for too long
        if time.time() > start_time + 0.9 * budget_s:
            break
    
    # Final verification
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
