import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def project_sum_box(v, M):
    """Project vector v onto sum = M with box constraints [0,1]"""
    v = np.clip(v, 0, 1)
    diff = M - np.sum(v)
    if abs(diff) < 1e-10:
        return v
    # Water-filling projection
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    n = len(v_sorted)
    threshold = 0.0
    for i in range(n):
        if cumsum[i] + (i+1)*v_sorted[i] < M:
            continue
        threshold = (M - cumsum[i]) / (i+1)
        break
    else:
        # All values are under the required threshold?
        return v
    # Apply thresholding
    result = np.maximum(v - threshold, 0)
    # Adjust to sum exactly
    adjustment = (M - np.sum(result)) / len(result)
    result += adjustment
    return np.clip(result, 0, 1)

def solve_milp(N, M, timeout_s):
    """Solve MILP for binary pattern that maximizes min autocorrelation"""
    # Variables
    y = cp.Variable(N, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(y) == M]
    
    # Autocorrelation constraints
    # w_{i,k} = y_i * y_{i+k} for all valid i, k pairs
    w_vars = {}
    shift_sums = {}
    
    for k in range(1, N):
        indices = range(N - k)
        if len(indices) == 0:
            continue
        w_k = cp.Variable(len(indices), continuous=True)
        w_vars[k] = w_k
        
        # Linearization: w_k[i] <= y[i], w_k[i] <= y[i+k]
        # w_k[i] >= y[i] + y[i+k] - 1
        # w_k[i] >= 0
        
        constraints.extend([
            w_k <= y[indices],
            w_k <= y[k:(N-k)],
            w_k >= y[indices] + y[k:(N-k)] - 1,
            w_k >= 0
        ])
        
        shift_sums[k] = cp.sum(w_k)
        constraints.append(shift_sums[k] >= t)
    
    # Objective
    prob = cp.Problem(cp.Maximize(t), constraints)
    
    try:
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=timeout_s)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURATE]:
            y_sol = y.value.round().astype(int).tolist()
            t_val = t.value
            return y_sol, t_val
    except Exception:
        pass
    return None, None

def compute_worst_shift(h):
    """Compute worst shift from current h: argmax_k correlate(h, 1-h)"""
    corr = correlate(h, 1-h, mode='full') 
    k_max = np.argmax(corr) - (len(h) - 1)
    return k_max

def subgradient_descent(h0, target_sum, max_iter, seed=None):
    """Deterministic subgradient descent with projection"""
    if seed is not None:
        np.random.seed(seed)
    h = h0.copy()
    best_h = h.copy()
    best_C5 = float('inf')
    
    for t in range(max_iter):
        k_max = compute_worst_shift(h)
        # Subgradient of F_kmax = sum(h) - sum(h_i * h_{i+k_max})
        g = np.ones_like(h)
        k = abs(k_max)
        if k_max >= 0:
            # k_max >= 0 means shift h[j] -> h[j + k_max]
            if k_max > 0:
                g[:len(h)-k_max] -= h[k_max:]
                g[k_max:] -= h[:len(h)-k_max]
        else:
            # Negative shift h[j] -> h[j - k_max]
            if k_max < 0:
                g[k:] -= h[:len(h)-k] 
                g[:k]   -= h[k:]
        
        alpha = 1.0 / np.sqrt(t+1)
        h_new = h - alpha * g
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, target_sum)
        
        if np.sum((h_new - h)**2) < 1e-10:
            # Optimization converged, break
            break
            
        # Compute new C5
        new_C5 = np.max(correlate(h_new, 1-h_new, mode='full')) * (2.0/len(h_new))
        
        if new_C5 < best_C5:
            best_h = h_new.copy()
            best_C5 = new_C5
            
        h = h_new
    
    return best_h, best_C5

def stochastic_refinement(h0, target_sum, remaining_seconds):
    """Simple simulated annealing after deterministic descent"""
    h = h0.copy()
    best_h = h.copy()
    best_C5 = np.max(correlate(h, 1-h, mode='full')) * (2.0/len(h))
    
    start_t = time.time()
    max_iter = 100000
    
    for i in range(max_iter):
        if time.time() - start_t > remaining_seconds:
            break
            
        # Pick two random indices
        idx1, idx2 = np.random.choice(len(h), 2, replace=False)
        
        # Try small move
        delta = np.random.uniform(-0.02, 0.02)
        
        h_new = h.copy()
        h_new[idx1] += delta
        h_new[idx2] -= delta
        
        # Clip to bounds
        h_new = np.clip(h_new, 0, 1)
        # Projection onto sum
        h_new = project_sum_box(h_new, target_sum)
        
        C5_new = np.max(correlate(h_new, 1-h_new, mode='full')) * (2.0/len(h_new))
        
        # Accept if better, or probabilistically
        if C5_new < best_C5:
            h = h_new.copy()
            best_h = h_new.copy()
            best_C5 = C5_new
        else:
            # Simulated annealing acceptance probability
            diff = C5_new - best_C5
            if diff > 0:
                temp = 1.0 - (time.time() - start_t) / remaining_seconds
                if temp > 0 and np.random.random() < np.exp(-diff / temp):
                    h = h_new.copy()
    
    return best_h, best_C5

def evaluate_c5(h):
    """Fast evaluation using FFT: the C5 metric"""
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    N = 120
    dx = 2.0 / N
    target_sum = N // 2
    
    # Time allocation
    milp_time = min(0.4 * budget_s, 200)  # Max 200 seconds for MILP
    descent_time = 0.35 * budget_s
    refine_time = budget_s - milp_time - descent_time
    
    # Step 1: Solve binary MILP for initial solution
    y_sol, t_opt = None, None
    init_h = None
    
    try:
        y_sol, t_opt = solve_milp(N, target_sum, milp_time)
        if y_sol is not None:
            init_h = np.array(y_sol, dtype=float)
    except Exception:
        pass
    
    if init_h is None:
        # Fallback: start with a random binary feasible solution
        init_h = np.zeros(N)
        ones = np.random.choice(N, target_sum, replace=False)
        init_h[ones] = 1
        # Randomize slightly to avoid perfect binary pattern
        for i in range(N):
            if init_h[i] == 0 and np.random.random() < 0.03:
                init_h[i] = 1
            elif init_h[i] == 1 and np.random.random() < 0.03:
                init_h[i] = 0
    
    # Project to desired sum exactly
    init_h = project_sum_box(init_h, target_sum)
    
    # Step 2: Subgradient descent to improve
    max_descent_iter = int(descent_time / (0.0001 if N <= 100 else 0.0005))
    improved_h, improved_C5 = subgradient_descent(
        init_h.copy(), target_sum, max_descent_iter, seed=seed*31
    )
    
    # Step 3: Stochastic refinement with simulated annealing
    if refine_time > 10:
        final_h, final_C5 = stochastic_refinement(
            improved_h, target_sum, refine_time
        )
    else:
        final_h, final_C5 = improved_h.copy(), improved_C5
    
    return (final_h.tolist(), float(final_C5), int(N))
