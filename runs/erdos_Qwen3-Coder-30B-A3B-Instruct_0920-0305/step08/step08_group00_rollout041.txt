def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_corr_int(h):
    """Compute integer correlation counts (fast)"""
    h_rev = h[::-1].astype(np.float64)
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    # round to nearest integer (small float errors)
    corr_int = np.rint(corr).astype(int)
    return corr_int

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def evaluate_c5_int(h):
    """Fast evaluation of C5 using integer correlation"""
    corr_int = compute_corr_int(h)
    return dx * np.max(corr_int)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_random_binary(n, target_sum):
    """Generate a random binary vector with exact target sum"""
    h = np.zeros(n, dtype=int)
    h[:int(target_sum)] = 1
    np.random.shuffle(h)
    return h

def simulated_annealing_binary(n_points, time_budget_s, seed=42):
    """Simulated annealing with binary vectors"""
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    h = generate_random_binary(n_points, n_points / 2)
    best_h = h.copy()
    best_corr = compute_corr_int(best_h)
    best_max = best_corr.max()
    current_h = h.copy()
    current_max = best_max
    
    # Annealing parameters
    T = 1.0
    cooling = 0.9995
    min_T = 1e-6
    
    iter_count = 0
    while time.time() - start_time < time_budget_s:
        iter_count += 1
        
        # Generate a random swap
        ones_idx = np.where(current_h == 1)[0]
        zeros_idx = np.where(current_h == 0)[0]
        
        if len(ones_idx) > 0 and len(zeros_idx) > 0:
            i_one = np.random.choice(ones_idx)
            i_zero = np.random.choice(zeros_idx)
            
            # Make candidate
            cand_h = current_h.copy()
            cand_h[i_one] = 0
            cand_h[i_zero] = 1
            
            # Compute new correlation
            cand_corr = compute_corr_int(cand_h)
            cand_max = cand_corr.max()
            
            # Accept or reject
            delta = cand_max - current_max
            if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, min_T)):
                current_h = cand_h
                current_max = cand_max
                
                if cand_max < best_max:
                    best_h = cand_h.copy()
                    best_max = cand_max
                    best_corr = cand_corr.copy()
        
        # Cooling schedule
        T *= cooling
        
        # Occasionally refresh temperature to avoid getting stuck
        if iter_count % 1000 == 0:
            T = max(T * 0.99, min_T)
    
    return best_h, best_max

def improve_solution_hill_climb(h, time_budget_s, seed=42):
    """Improvement by hill climbing local search"""
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = h.copy()
    best_corr = compute_corr_int(best_h)
    best_max = best_corr.max()
    
    patience = 5000
    no_improve = 0
    
    while time.time() - start_time < time_budget_s and no_improve < patience:
        # Generate random swaps and keep best improvement
        best_swap_improvement = False
        best_swap = None
        
        for _ in range(100):
            ones_idx = np.where(best_h == 1)[0]
            zeros_idx = np.where(best_h == 0)[0]
            
            if len(ones_idx) > 0 and len(zeros_idx) > 0:
                i_one = np.random.choice(ones_idx)
                i_zero = np.random.choice(zeros_idx)
                
                # Test swap
                cand_h = best_h.copy()
                cand_h[i_one] = 0
                cand_h[i_zero] = 1
                
                cand_corr = compute_corr_int(cand_h)
                cand_max = cand_corr.max()
                
                if cand_max < best_max:
                    best_swap = (i_one, i_zero)
                    best_max = cand_max
                    best_corr = cand_corr
                    best_swap_improvement = True
                    break  # Take first improvement
        
        if best_swap_improvement:
            # Apply best improvement
            i_one, i_zero = best_swap
            best_h[i_one] = 0
            best_h[i_zero] = 1
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_max
