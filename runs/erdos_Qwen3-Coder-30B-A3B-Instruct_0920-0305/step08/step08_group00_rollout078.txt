def evaluate_c5_fast(h, dx):
    """Evaluate C5 using FFT for speed."""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def construct_h_from_pattern(pattern, n):
    """Construct h by interpolating pattern of length n/2 to n points."""
    h = np.zeros(n)
    # assume pattern is values 0 or 1, length n//2
    for i in range(len(pattern)):
        idx = int(i * n / len(pattern))
        h[idx] = pattern[i]
    h = project_to_sum_box(h, n / 2)
    return h
