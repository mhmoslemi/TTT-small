import numpy as np
from scipy.signal import correlate
import time

def project_to_capped_simplex(x, target_sum):
    """Projects vector x onto the capped simplex {x : 0 <= x[i] <= 1, sum(x) = target_sum}"""
    x_clipped = np.clip(x, 0, 1)
    current_sum = np.sum(x_clipped)
    if abs(current_sum - target_sum) < 1e-12:
        return x_clipped
    # Binary search for lambda
    a, b = -1.0, 1.0
    for _ in range(50):
        lam = (a + b) / 2
        projected = np.clip(x + lam, 0, 1)
        s = np.sum(projected)
        if abs(s - target_sum) < 1e-12:
            return projected
        if s < target_sum:
            a = lam
        else:
            b = lam
    return np.clip(x + (a + b)/2, 0, 1)

def fast_correlation_update(c, h, b, i, j, delta):
    """Update correlation vector c for a mass transfer move"""
    n = len(h)
    # b-part: add to c indices corresponding to shift (j - i)
    left = (n - 1) - i
    right = left + n
    if left < right:
        c[left:right] += delta * b[j:(j+n)]
    # h-part: subtract from c indices corresponding to shift (i - j)  
    left = i
    right = i + n
    if left < right:
        c[left:right] -= delta * h[::-1][j:(j+n)]
    return c

    # Variables
    h = h_initial.copy()
    b = 1 - h
    c = correlate(h, b, mode='full')
    Cmax = np.max(c)
    Obj = Cmax * dx
    
    # Track best solution
    best_h = h.copy()
    best_value = Obj
    
    # Annealing parameters
    T = 0.01
    alpha_T = 0.995
    delta = 0.02
    delta_min = 5e-4
    alpha_delta = 0.9
    eval_freq = 500
    step_freq = 100
    temp_freq = 500
    no_improvement_patience = 2000
    
    # Main loop
    iteration = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s:
        iteration += 1
        # Sample indices
        i, j = np.random.choice(n, size=2, replace=False)
        
        # Compute allowable delta to respect box constraints
        delta_max = min(h[i], 1 - h[j])
        curr_delta = min(delta, 0.9 * delta_max)
        
        # Update h and b
        h[i] -= curr_delta
        h[j] += curr_delta
        b[i] = 1.0 - h[i]
        b[j] = 1.0 - h[j]
        
        # Fast update of correlation matrix c
        fast_correlation_update(c, h, b, i, j, curr_delta)
        # Update Cmax
        Cmax = np.max(c)
        Obj_new = Cmax * dx
        
        # Accept/reject based on simulated annealing
        if Obj_new < Obj or np.random.rand() < np.exp((Obj - Obj_new) / T):
            Obj = Obj_new
            if Obj < best_value:
                best_h[:] = h
                best_value = Obj
                no_improvement_count = 0
        else:
            # Revert changes
            h[i] += curr_delta
            h[j] -= curr_delta
            b[i] = 1.0 - h[i]
            b[j] = 1.0 - h[j]
            # Recompute correlation matrix for consistency (sometimes)
            if iteration % eval_freq == 0:
                c = correlate(h, 1 - h, mode='full')
                Cmax = np.max(c)
                Obj = Cmax * dx
        
        # Adjust learning rate
        if iteration % step_freq == 0 and delta > delta_min:
            delta *= alpha_delta
        
        # Cool temperature
        if iteration % temp_freq == 0:
            T *= alpha_T
        
        # Periodically recompute exact c if needed (safety check)
        if iteration % eval_freq == 0:
            c_exact = correlate(h, 1 - h, mode='full')
            if abs(np.max(c_exact) - Cmax) > 1e-10:
                c = c_exact

                Obj = Cmax * dx
            # Safety check on projection
            h = project_to_capped_simplex(h, TARGET)
            b = 1 - h    
        
        # Reset temperature if no improvement
        no_improvement_count += 1
        if no_improvement_count > no_improvement_patience:
            T = 0.01
            no_improvement_count = 0

    return (best_h.tolist(), float(best_value), n)
