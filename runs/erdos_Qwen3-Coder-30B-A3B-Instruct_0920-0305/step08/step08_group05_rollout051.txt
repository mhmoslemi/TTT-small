def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def neighbor_sum(h, lag):
    """Compute neighbor sum for given lag"""
    lag_abs = abs(lag)
    if lag_abs == 0:
        return 2 * h
    else:
        w = np.zeros_like(h)
        if lag > 0:
            w[:n_points-lag_abs] += h[lag_abs:]
            w[lag_abs:] += h[:n_points-lag_abs]
        else:
            w[lag_abs:] += h[:-lag_abs]
            w[:n_points-lag_abs] += h[lag_abs:]
        return w

def mass_transfer_step(h, step_size):
    """Perform one iteration of mass transfer"""
    # Compute full correlation to find worst lag
    corr = np.correlate(h, 1 - h, mode='full') * dx
    idx_max = np.argmax(corr)
    lag_max = idx_max - (n_points - 1)
    
    # Compute neighbor sum for the worst lag
    w = neighbor_sum(h, lag_max)
    
    # Find indices with low and high w values
    eps = 1e-8
    low_candidates = np.where(h > eps)[0]
    high_candidates = np.where(h < 1 - eps)[0]
    
    if len(low_candidates) == 0 or len(high_candidates) == 0:
        return h, False
    
    # Find the best low and high indices
    low_idx = low_candidates[np.argmin(w[low_candidates])]
    high_idx = high_candidates[np.argmax(w[high_candidates])]
    
    # Avoid same index
    if low_idx == high_idx:
        return h, False
        
    # Compute transfer amount
    delta = min(step_size, h[low_idx], 1 - h[high_idx])
    
    if delta < 1e-12:
        return h, False
        
    # Perform transfer
    h_new = h.copy()
    h_new[low_idx] -= delta
    h_new[high_idx] += delta
    
    return h_new, True

def refine_h(h_init, max_iter, step_initial):
    """Refine h using mass transfer steps"""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    step = step_initial
    no_improve = 0
    
    for i in range(max_iter):
        h_new, improved = mass_transfer_step(h, step)
        
        if improved:
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new.copy()
                best_c5 = c5_new
                no_improve = 0
            h = h_new
            no_improve = 0
        else:
            no_improve += 1
            
        if no_improve > 100:
            step *= 0.5
            step = max(step, 1e-6)
            no_improve = 0

    return best_h, best_c5
