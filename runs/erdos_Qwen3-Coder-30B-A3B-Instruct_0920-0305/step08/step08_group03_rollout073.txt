import time

# Global parameters
n_points = 256
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def project_to_box_sum(h, target_sum):
    """Project vector h onto [0,1] with sum = target_sum using capped simplex projection."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    
    # Binary search for lambda such that sum(clamp(h + lambda, 0, 1)) == target_sum
    def f(lam):
        projected = np.clip(h + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    return np.clip(h + (a + b)/2, 0, 1)

def compute_correlation_incremental(h, b, c, i, j, delta):
    """Update correlation incrementally after transferring delta mass from i to j."""
    if i == j:
        return c  # No change
    
    # Create reversed h for efficient slice update
    h_rev = h[::-1]
    
    # Update the slice from b (move h[i] -> b[i])
    if 0 <= i < len(h):
        start_idx = (n_points - 1) - i
        c[start_idx:start_idx + n_points] += delta * b[i:i + n_points]
    
    # Update the slice from h_rev (move b[j] -> h_rev[j])
    if 0 <= j < len(h):
        start_idx = j
        c[start_idx:start_idx + n_points] -= delta * h_rev[j:j + n_points]
    
    # Add quadratic correction (accounting for alignment)
    s1 = j - i  # shift for correlation
    s2 = i - j
    if 0 <= s1 + n_points - 1 < 2 * n_points - 1:
        c[s1 + n_points - 1] += delta * delta
    if 0 <= s2 + n_points - 1 < 2 * n_points - 1:
        c[s2 + n_points - 1] += delta * delta
        
    return c
