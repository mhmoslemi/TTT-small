def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5_analytic(piecewise_h, lengths):
    """Exact evaluation of C5 using piecewise structure"""
    m = len(piecewise_h)
    total_length = sum(lengths)
    total_mass = sum(piecewise_h[i] * lengths[i] for i in range(m))
    
    # Ensure mass is satisfied
    if abs(total_mass - 1.0) > 1e-10:
        # Scale piecewise_h to match mass constraint
        scale = 1.0 / total_mass
        piecewise_h = [p * scale for p in piecewise_h]
    
    # Compute positions of intervals
    positions = [0.0]
    for l in lengths:
        positions.append(positions[-1] + l)
    
    # Compute critical shifts we care about
    shifts = []
    for i in range(m):
        shift_l = positions[i]
        shift_r = positions[i+1]
        shifts.append(shift_l)
        shifts.append(shift_r)
        for j in range(i+1, m):
            shift_l = positions[j] - positions[i]
            shifts.append(shift_l)
            shift_r = positions[i+1] - positions[j]
            shifts.append(shift_r)
    
    # Only consider nonnegative shifts up to 2
    shifts = sorted(set([s for s in shifts if 0 <= s <= 2]))
    
    # For each shift, compute J(k)
    max_overlap = 0
    for shift_k in shifts:
        # Compute integral ∫₀^{2-k} h(x)(1-h(x+k)) dx
        total = 0
        k = shift_k
        for i in range(m):
            pos_start = positions[i]
            pos_end = positions[i+1]
            if pos_start >= 2 - k:
                break
            # length of overlap with domain [0, 2-k] 
            overlap_start = max(pos_start, 0)
            overlap_end = min(pos_end, 2 - k)
            if overlap_start < overlap_end:
                overlap_len = overlap_end - overlap_start
                # compute product term: h(x) * (1 - h(x+k))
                # for interval [pos_start, pos_end]:
                # h(x) = piecewise_h[i] on this interval
                # h(x+k) = piecewise_h[j] for j such that pos_start+k in [positions[j], positions[j+1])
                k_start = pos_start + k  # where shifted x starts
                k_end = pos_end + k      # where shifted x ends
                # Find which interval k_start is in
                j = 0
                while j < m and positions[j+1] <= k_start:
                    j += 1
                if j < m and k_start < positions[j+1]:
                    shifted_h = piecewise_h[j]
                else:
                    shifted_h = 0
                total += piecewise_h[i] * (1 - shifted_h) * overlap_len
        max_overlap = max(max_overlap, total)
    return max_overlap

def try_parametric_search(n_intervals=5, budget_s=600):
    best_c5 = float('inf')
    best_params = None
    start_time = time.time()

    # Simplified parametrization:
    # Pieces with lengths and heights
    # lengths are the lengths of intervals, need to sum to 2
    # heights are the values on those intervals, need to sum to total_mass (scaled to 1)

    # Since parametric optimization is low-dimensional, we can be exhaustive 
    n_dim = 2 * (n_intervals - 1)
    if n_dim > 10: return None
    
    def objective(params):
        # params: [lengths[0],...,lengths[m-2], heights[0],...,heights[m-2]]
        lengths = params[:n_intervals-1]
        heights = params[n_intervals-1:]
        # Ensure lengths sum up to 2, and heights are in [0,1]
        lengths = [max(0.001, l) for l in lengths]
        remaining_length = 2 - sum(lengths)
        if remaining_length <= 0:
            return float('inf')
        lengths.append(remaining_length)
        heights = [max(0.0, min(1.0, h)) for h in heights]
        
        # Compute final height given constraint
        try:
            total_mass = sum(h * l for h, l in zip(heights, lengths[:-1]))
            final_h = (1 - total_mass) / lengths[-1] if lengths[-1] > 0 else 0
            final_h = max(0.0, min(1.0, final_h))
            heights.append(final_h)
        except:
            return float('inf')
            
        c5 = evaluate_c5_analytic(heights, lengths) / dx
        
        # Regularization: penalize uneven distribution
        avg_height = sum(heights) / len(heights)
        penalty = sum((h - avg_height)**2 for h in heights) * 1e-3
        return c5 + penalty

    bounds = [(0.001, 2.0) for _ in range(n_intervals-1)] + [(0.0, 1.0) for _ in range(n_intervals-1)]
    
    # Try several strategies to prevent falling into poor local minima
    best_obj = float('inf')
    best_params_final = None

    # Start with a simple grid refinement attempt first  
    if n_intervals <= 6:
        for attempt in range(100):
            if time.time() - start_time > budget_s - 1:
                break
                
            # sample some parameters (more uniform sampling)
            params = np.random.uniform(0.001, 2.0, n_intervals-1)
            # and heights in [0,1]
            heights = np.random.uniform(0.0, 1.0, n_intervals-1)
            try:  
                obj = objective(list(params) + list(heights))
                if obj < best_obj:
                    best_obj = obj
                    best_params_final = list(params) + list(heights)
            except:
                continue

    # Perform for longer with deterministic local search after grid
    if best_params_final is not None:
        try:
            res = minimize(objective, best_params_final, method='Nelder-Mead', 
                           options={'maxiter': 2000, 'disp': False})
            if res.success and res.fun < best_obj:
                best_obj = res.fun
                best_params_final = res.x.tolist()
        except:
            pass

    # Then run some global search methods to get a better value
    try:
        # Use differential evolution to see if it goes better
        res = differential_evolution(objective, bounds, maxiter=500, 
                                     popsize=15, strategy='best1bin', 
                                     atol=1e-6, rtol=1e-6, seed=42)
        if res.success and res.fun < best_obj:
            best_obj = res.fun
            best_params_final = res.x.tolist()
    except:
        pass

    if best_params_final is not None:
        lengths = best_params_final[:n_intervals-1]
        remaining_length = 2 - sum(lengths)
        lengths.append(remaining_length)  
        heights = best_params_final[n_intervals-1:]
        try:
            total_mass = sum(h * l for h, l in zip(heights, lengths[:-1]))
            final_h = (1 - total_mass) / lengths[-1] if lengths[-1] > 0 else 0
            final_h = max(0.0, min(1.0, final_h))
            heights.append(final_h)
        except:
            return None
        
        c5_result = evaluate_c5_analytic(heights, lengths) / dx
        return (lengths, heights, c5_result)
    return None

def run(seed=42, budget_s=600, **kwargs):

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    params = try_parametric_search(n_intervals=5, budget_s=budget_s)
    if params is not None:
        lengths, heights, c5 = params
        n_intervals = len(heights)
        
        # Construct a dense step function
        h_dense = []
        pos = 0
        for i in range(n_intervals):
            h_dense.extend([heights[i]] * int(lengths[i] / dx))  # round down to integer grid points
            
        # Now scale so that sum(h_dense) * dx = 1
        total = sum(h_dense) * dx
        h_dense = [x * (1.0 / total) for x in h_dense]
        
        # Adjust to exactly match n_points
        if len(h_dense) > n_points:
            h_dense = h_dense[:n_points]
        elif len(h_dense) < n_points:
            # Pad with some value, like last value
            pad_val = h_dense[-1] if len(h_dense) > 0 else 0.5
            h_dense.extend([pad_val] * (n_points - len(h_dense)))
            
        # Compute actual C5 value
        try:
            c5_actual = evaluate_c5(h_dense)
            if c5_actual < best_c5:
                best_c5 = c5_actual
                best_h = h_dense
                best_n = n_points
        except:
            pass

    # If parametric approach doesn't work, fallback to some continuous method
    if best_h is None:
        # Scale previous h and refine it a bit
        h0 = initial_guess_120()
        h0_scaled = project_to_sum_box(h0, n_points / 2)
        best_c5 = evaluate_c5(h0_scaled)
        best_h = list(h0_scaled)
        best_n = n_points

    # Return best result found
    return (best_h, float(best_c5), best_n)
