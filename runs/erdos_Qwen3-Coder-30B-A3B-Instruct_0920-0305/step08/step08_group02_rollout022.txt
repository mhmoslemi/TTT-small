import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def project_sum_box(v, target_sum):
    """Project vector v onto sum = target_sum with box constraints [0, 1]"""
    # Algorithm: Water-filling style projection
    # Given v, find lambda such that sum(clamp(v + lambda, 0, 1)) == target_sum
    # This is faster using bisection

    # Helper for computing sum with current lambda
    def f(lam):
        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Bisection
    a, b = -1.0, 1.0
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    lam = (a + b) / 2
    return np.clip(v + lam, 0, 1)

def evaluate_c5(h):
    """Fast C5 evaluation using FFT (used in adaptive sampling, local search)"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def solve_milp_binary(N, target_sum):
    """
    Solve binary MILP for maximum minimal autocorrelation.
    Uses cvxpy with CBC or fallbacks.
    If failed, return None.
    """
    try:
        # Variables
        y = cp.Variable(N, boolean=True)
        t = cp.Variable()
        
        # Constraints
        constraints = [cp.sum(y) == target_sum]
        
        # W variables for products over shifts
        w_vars = {}
        shift_cons = []
        
        for k in range(1, N):
            w_k = cp.Variable(N - k)
            w_vars[k] = w_k
            # Fix w_{k,i} = y_i * y_{i+k}
            constraints += [w_k <= y[:N-k], w_k <= y[k:], 
                            w_k >= y[:N-k] + y[k:] - 1, w_k >= 0]
            shift_cons.append(cp.sum(w_k) >= t)
        
        constraints += shift_cons
        
        # Bounding t
        constraints += [t >= 0, t <= target_sum]

        # Objective
        prob = cp.Problem(cp.Maximize(t), constraints)
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=190)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            # Extract binary solution
            y_sol = np.array([1 if yi.value >= 0.5 else 0 for yi in y])
            return y_sol
    except:
        pass
    return None

def subgradient_descent_step(h, k_max, max_iter):
    """Single subgradient descent step that reduces the worst shift"""
    g = np.ones_like(h)
    N = len(h)
    if k_max >= 0:
        # Subtract h_{j+k}
        g[:N - k_max] -= h[k_max:]
    else:
        k = -k_max
        g[k:] -= h[:N - k] 

    if k_max > 0:
        # Subtract h_{j-k}
        g[k_max:] -= h[:N - k_max]
    elif k_max < 0:
        k = -k_max
        g[:N - k] -= h[k:]

    # Gradient step
    alpha = 0.02 / np.sqrt(max_iter + 1)
    h_new = h - alpha * g
    
    # Project back
    h_new = np.clip(h_new, 0, 1)
    h_new = project_sum_box(h_new, TARGET_SUM)
    
    return h_new

def subgradient_descent(h0, num_iter=15000):
    """Improve using deterministic sub-gradient descent"""
    h = h0.copy()
    best = h.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(num_iter):
        corr = correlate(h, 1 - h, mode='full')
        k_best = np.argmax(corr) - (n_points - 1)
        
        h_new = subgradient_descent_step(h, k_best, i+1)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best = h_new.copy()
            best_c5 = c5_new
            
        h = h_new
        
    return best, best_c5

def stochastic_refinement(h0, max_iter=8000, time_budget_s=200):
    """Light simulated annealing like perturbations"""
    h = h0.copy()
    best = h.copy()
    best_c5 = evaluate_c5(h)
    
    start = time.time()
    for i in range(max_iter):
        now = time.time()
        if now - start > time_budget_s:
            break
            
        # Swap small delta between two random positions
        i1 = np.random.randint(0, n_points)
        i2 = np.random.randint(0, n_points)
        
        delta = np.random.uniform(-0.02, 0.02)
        
        h_temp = h.copy()
        h_temp[i1] += delta
        h_temp[i2] -= delta
        
        # Project
        h_temp = np.clip(h_temp, 0, 1)
        h_temp = project_sum_box(h_temp, TARGET_SUM)
        
        c5 = evaluate_c5(h_temp)
        
        if c5 < best_c5:
            h = h_temp.copy()
            best_c5 = c5
            best = h.copy()
        else:
            # Accept with decreasing probability
            p = np.exp(-(c5 - best_c5) / (0.01 * (1.0 + i / max_iter)))
            if np.random.random() < p:
                h = h_temp.copy()
                
    return best, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Time allocation
    milp_time = 0.4 * budget_s
    descent_time = 0.35 * budget_s
    anneal_time = budget_s - (milp_time + descent_time)
    
    # Phase 1: MILP for binary solution or fallback
    h = np.ones(n_points) * 0.5  # Default fallback
    best_c5 = evaluate_c5(h)
    
    # Try to solve MILP
    milp_solution = solve_milp_binary(n_points, TARGET_SUM)
    
    if milp_solution is not None:
        h = milp_solution.astype(float)
        best_c5 = evaluate_c5(h)
        
    # Phase 2: Aggressive subgradient descent 
    h_descent, c5_descent = subgradient_descent(h, int(descent_time * 2000 / 600))
    
    if c5_descent < best_c5:
        h = h_descent
        best_c5 = c5_descent
        
    # Phase 3: Stochastic refinement on best
    h_anneal, c5_anneal = stochastic_refinement(h, int(anneal_time * 2000 / 600), anneal_time)
    
    if c5_anneal < best_c5:
        h = h_anneal
        best_c5 = c5_anneal
        
    return (h.tolist(), float(best_c5), n_points)
