def build_critical_shifts(m):
    """Generate critical shift values for m intervals."""
    # Start with all combinations of boundaries (including mirrored)
    shifts = set()
    
    # Add 0 and 2 to ensure coverage
    shifts.add(0.0)
    shifts.add(2.0)
    
    # Add all |s_p - s_q| for i < j
    for i in range(m):
        for j in range(i+1, m):
            shift = abs(i - j)
            shifts.add(shift)
            shifts.add(2.0 - shift)
            
    # Add all |(2 - s_p) - (2 - s_q)| = |s_p - s_q| (already covered)
    # Also add endpoints (0 and 2) explicitly
    shifts.add(0.0)
    shifts.add(2.0)
    
    return sorted(list(shifts))

def compute_J(k, L, A):
    """Compute J(k;L,A) = integral of h(x)(1-h(x+k)) dx for shift k."""
    # Compute linear term: integral of h(x) from 0 to 2-k
    linear_term = 0.0
    overlap_sum = 0.0
    
    # We need to compute the overlap contribution
    # For simplicity, we will handle both parts analytically 
    
    # Create interval boundaries
    s = np.concatenate([[0], np.cumsum(L)])
    
    # Compute linear term (all of h(x) up to 2-k)
    x_end = max(0, 2 - k)
    linear_term = 0.0
    
    for i in range(len(L)):
        start = s[i]
        end = s[i+1]
        if start >= x_end:
            break
        if end <= x_end:
            linear_term += A[i] * (end - start)
        else:
            linear_term += A[i] * (x_end - start)
            break

    # Now compute product term (h(x) h(x+k))
    for i in range(len(L)):
        for j in range(len(L)):
            start1 = s[i]
            end1 = s[i+1]
            start2 = s[j] + k
            end2 = s[j+1] + k
            
            # Compute overlap
            overlap_left = max(start1, start2)
            overlap_right = min(end1, end2)
            
            if overlap_left < overlap_right:
                overlap_length = overlap_right - overlap_left
                overlap_sum += A[i] * A[j] * overlap_length
    
    # Return J = linear - product
    return linear_term - overlap_sum

def compute_C(L, A):
    """Compute maximum of J(k; L, A) over all critical shifts."""
    global K_CRIT
    if K_CRIT is None:
        # Compute critical shifts with m=5 intervals
        K_CRIT = build_critical_shifts(5)
    
    max_val = -float('inf')
    for k in K_CRIT:
        val = compute_J(k, L, A)
        if val > max_val:
            max_val = val
    return max_val

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradients(k, L, A):

            start2 = s[j] + k

            overlap_right = min(end1, end2)
            
            if overlap_left < overlap_right:
                overlap_length = overlap_right - overlap_left
                
                # Partial derivatives with respect to L

                if j >= 0:

                dJ_dA[i] -= A[j] * overlap_length
                dJ_dA[j] -= A[i] * overlap_length
    
    return dJ_dL, dJ_dA
