import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import time

# Global variables for the problem
n_points = 300
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT with proper zero-padding
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water-filling"""
    # Sort the values to compute threshold
    sorted_x = np.sort(x)[::-1]  # descending order
    
    # Compute cumulative sums from the top
    cumsum_sorted = np.cumsum(sorted_x)
    n = len(sorted_x)
    
    # Find the threshold index
    # For a value p to be the new value floor for all elements at or above it,
    # the sum of all values above it equals (target_sum - sum of values below it)
    # We use a binary search on the cumulative sums
    i = 0
    while i < n and cumsum_sorted[i] > target_sum - (n - i - 1) * sorted_x[i]:
        i += 1
    
    # The threshold is the value where the condition fails
    # i might be 0 (all elements get capped at 1) or n (none capped)
    if i == 0:
        threshold = 1
    elif i == n:
        threshold = 0
        # Adjust for all elements getting allocated
        if np.sum(sorted_x) < target_sum:
            # This shouldn't happen if x start <=1, but let's be safe
            # Solution on simplex with target sum
            threshold = 1 if np.sum(sorted_x) > target_sum else sorted_x[-1]
    else:
        threshold = (target_sum - cumsum_sorted[i-1]) / (n - i)
    
    # Apply thresholding
    result = np.clip(x, 0, threshold)
    
    # If we under-allocated, rescale to hit target_sum
    actual_sum = np.sum(result)
    if actual_sum > 0:
        result = result * (target_sum / actual_sum)
    
    # Ensure proper bounds
    result = np.clip(result, 0, 1)
    return result

def rasterise_and_project(theta):
    """Convert parameter vector to step function and project"""
    K = 8  # Number of blocks
    # Split theta into widths and heights
    widths = np.exp(theta[:K])
    widths = 2 * widths / np.sum(widths)  # Ensure total width = 2
    
    heights = 1 / (1 + np.exp(-theta[K:]))
    heights = np.clip(heights, 1e-6, 1 - 1e-6)  # Avoid boundary issues
    
    # Create block structure
    block_widths = np.floor(widths / dx).astype(int)
    block_widths[0] = max(1, block_widths[0])  # Ensure at least one point in first block
    
    # Take care of overflow edge cases
    if np.sum(block_widths) < n_points:
        # Add extra points to last block
        block_widths[-1] += n_points - np.sum(block_widths)
    else:
        # Trim
        block_widths[0] += np.sum(block_widths) - n_points
    
    # Ensure all widths are non-negative and sum to n_points
    if np.sum(block_widths) != n_points:
        block_widths[-1] -= np.sum(block_widths) - n_points
    
    h = np.zeros(n_points)
    pos = 0
    for i in range(K):
        end = min(pos + block_widths[i], n_points)
        if end > pos:
            h[pos:end] = heights[i]
            pos = end
        if pos >= n_points:
            break
    
    # Project to sum and box
    h = project_to_sum_box(h, TARGET_SUM)
    return h

    c5 = evaluate_c5(h)
    return c5
