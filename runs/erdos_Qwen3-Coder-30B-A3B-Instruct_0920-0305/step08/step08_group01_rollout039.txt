def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    # Use scipy's correlate with 'full' mode for convolution-like operation
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum and scale by dx
    return dx * np.max(corr)

def project_to_box_and_sum(x, target_sum):
    """Project to [0,1] box constraints and adjust sum to target"""
    # First clip to [0,1]
    x = np.clip(x, 0, 1)
    
    # Then adjust sum using simple projection method
    current_sum = np.sum(x)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return x
    
    # Simple adjustment: add diff evenly to all points (will be clipped)
    x += diff / len(x)
    x = np.clip(x, 0, 1)
    
    return x
