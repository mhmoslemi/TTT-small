import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import time

# Global variables
TARGET_SUM = None
n_points = 300
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def piecewise_const_from_params(params, n_points):
    """Convert piecewise constant parameters to h array"""
    # params = [log_w1, ..., log_wK, h1, ..., hK]  
    K = len(params) // 2
    log_widths = params[:K]
    heights_raw = params[K:]
    
    # Compute widths from log widths
    widths = np.exp(log_widths)
    widths = 2 * widths / np.sum(widths)  # normalize to length 2
    
    # Compute heights from raw values (logistic sigmoid)
    heights = 1.0 / (1.0 + np.exp(-heights_raw))
    
    # Clip to [0, 1]
    heights = np.clip(heights, 0, 1)
    
    # Create piecewise constant h
    h = np.zeros(n_points)
    bin_edges = np.concatenate([[0], np.cumsum(widths)])
    
    # Ensure bin_edges are within [0, 2]
    bin_edges = np.clip(bin_edges, 0, 2)
    
    # Fix endpoint
    bin_edges[-1] = 2.0
    
    # For each point, assign height based on which block it belongs to
    positions = np.linspace(0, 2, n_points, endpoint=False) + dx/2
    block_indices = np.searchsorted(bin_edges, positions, side='right') - 1
    block_indices = np.clip(block_indices, 0, K-1)
    
    # Assign heights for each point
    for i in range(n_points):
        h[i] = heights[block_indices[i]]
    
    return h

def de_objective(params):

    h = piecewise_const_from_params(params, n_points)
    h = project_to_sum_box(h, TARGET_SUM)
    c5 = evaluate_c5(h)
    return c5

                    best_h = new_h.copy()
