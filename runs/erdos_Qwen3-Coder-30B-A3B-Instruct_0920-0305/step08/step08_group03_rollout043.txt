import numpy as np
from scipy.signal import correlate
import time

# Global constants
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def get_correlation_update(h, b, c, i, j, delta, h_rev):
    """Compute updated correlation vector after a mass transfer move."""
    c_new = c.copy()
    # Update b-part for i
    if 0 <= i < len(h):
        start_b = (n_points - 1) - i
        end_b = start_b + n_points
        if start_b < len(c_new) and end_b <= len(c_new):
            c_new[start_b:end_b] += delta * b[i:i+n_points]  # vectorized slice
    # Update h-part for j
    if 0 <= j < len(h):
        start_h = j
        end_h = j + n_points
        if start_h < len(c_new) and end_h <= len(c_new):
            c_new[start_h:end_h] -= delta * h_rev  # vectorized slice
    return c_new

    # Project onto feasible set
    h = project_to_sum_box(h_interpolated, n_points / 2)
    b = 1 - h
    h_rev = h[::-1]
    
    # Full correlation using FFT - we'll use this for comparison and exact resets
    c = correlate(h, b, mode='full')
    Cmax = np.max(c)
    C5 = dx * Cmax
    
    best_h = h.copy()
    best_c5 = C5
    
    # Annealing parameters
    T = 0.01  # Initial temperature
    alpha_T = 0.995 # Temperature decay
    delta_initial = 0.02
    delta = delta_initial
    delta_min = 5e-4
    T_step = 500
    T_temp = 2000
    K_eval = 1000
    K_step = 2000
    K_temp = 20000
    
    no_improve_count = 0 # Control temperature decay
    
    # Main loop
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Try to make a move (mass transfer)
        i, j = np.random.randint(0, n_points, 2)
        # Ensure we don't violate the box constraints
        max_delta = min(h[i], 1 - h[j])
        if max_delta < 0.001:
            continue
        delta = min(delta, 0.9 * max_delta)
        
        old_Cmax = Cmax
        old_h = h.copy()
        
        # Apply delta move
        delta_i = -delta
        delta_j = +delta
        h[i] += delta_i
        h[j] += delta_j
        b[i] -= delta_i  # because b = 1 - h
        b[j] -= delta_j
        
        # Update correlations incrementally
        c = get_correlation_update(h, b, c, i, j, delta_i, h_rev)
        c = get_correlation_update(h, b, c, j, i, delta_j, h_rev)
        
        # Quadratic correction: added terms due to δ²
        s1 = j - i + (n_points - 1)
        s2 = i - j + (n_points - 1)
        if 0 <= s1 < len(c) and 0 <= s2 < len(c):
            c[s1] += delta ** 2
            c[s2] += delta ** 2
        
        Cmax = np.max(c)
        new_C5 = dx * Cmax
        
        # Decide to accept or reject
        if new_C5 < old_C5 or np.random.rand() < np.exp(-(new_C5 - old_C5) / T):
            # Accept the move
            old_C5 = new_C5
            no_improve_count = 0

            h = old_h.copy()

            c = correlate(h, b, mode='full')

            new_C5 = dx * Cmax

        if iteration % K_step == 0 and delta > delta_min:

            T *= alpha_T
        
        # Frequently recompute Cmax exactly to avoid drift
        if iteration % K_eval == 0:
            c = correlate(h, 1 - h, mode='full')
            Cmax = np.max(c)
            new_C5 = dx * Cmax
            if abs(new_C5 - old_C5) > 1e-12:
                old_C5 = new_C5
            if new_C5 < best_c5:
                best_c5 = new_C5
                best_h = h.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
