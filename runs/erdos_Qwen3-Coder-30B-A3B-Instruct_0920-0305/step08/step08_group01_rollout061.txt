import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def random_binary_solution():
    """Generate a random binary solution with exactly TARGET_SUM ones"""
    h = np.zeros(n_points, dtype=float)
    ones_indices = np.random.choice(n_points, size=TARGET_SUM, replace=False)
    h[ones_indices] = 1.0
    return h

def swap_bits(h, i, j):
    """Swap bits at positions i and j"""
    h[i], h[j] = h[j], h[i]

def simulated_annealing(seed=42, budget_s=600):
    """Run simulated annealing optimization"""
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    best_h = random_binary_solution()
    best_c5 = evaluate_c5(best_h)
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Temperature schedule
    initial_temp = 0.01
    final_temp = 0.0001
    max_iterations = 200000
    temp_decay = (final_temp / initial_temp) ** (1.0 / max_iterations)
    
    # Tracking
    iterations = 0
    last_improvement = 0
    
    while iterations < max_iterations and time.time() - start_time < budget_s * 0.95:
        # Update temperature
        temp = initial_temp * (temp_decay ** iterations)
        
        # Sample i from ones and j from zeros
        ones_indices = np.where(current_h == 1)[0]
        zeros_indices = np.where(current_h == 0)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            break
            
        i = np.random.choice(ones_indices)
        j = np.random.choice(zeros_indices)
        
        # Swap bits
        swap_bits(current_h, i, j)
        new_c5 = evaluate_c5(current_h)
        
        # Accept or reject
        delta = current_c5 - new_c5
        
        if delta > 0 or np.random.random() < np.exp(delta / temp):
            # Accept the change
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
                last_improvement = iterations
        else:
            # Revert the swap
            swap_bits(current_h, i, j)
            
        iterations += 1
        
        # Periodically check for stagnation
        if iterations - last_improvement > 5000:
            # Random restart
            current_h = random_binary_solution()
            current_c5 = evaluate_c5(current_h)
            last_improvement = iterations
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx

    # Run final optimization with gradient descent if time allows
    if time.time() - start_time < budget_s * 0.99:
        # Try final refinement

            return evaluate_c5(h)
            
        h_refined = best_h.copy()
        
        # Use a few iterations of local search with bounds
        ones_indices = np.where(h_refined == 1)[0]
        zeros_indices = np.where(h_refined == 0)[0]
        
        # Try small perturbations to improve solution
        for _ in range(1000):
            if len(ones_indices) > 0 and len(zeros_indices) > 0:
                i = np.random.choice(ones_indices)
                j = np.random.choice(zeros_indices)
                h_test = h_refined.copy()
                h_test[i] = 0
                h_test[j] = 1
                new_c5 = evaluate_c5(h_test)
                if new_c5 < evaluate_c5(h_refined):
                    h_refined = h_test.copy()
                    ones_indices = np.where(h_refined == 1)[0]
                    zeros_indices = np.where(h_refined == 0)[0]
            else:
                break
                
        final_c5 = evaluate_c5(h_refined)
        if final_c5 < best_c5:
            best_h = h_refined.copy()
            best_c5 = final_c5
    
    return (best_h.tolist(), float(best_c5), n_points)
