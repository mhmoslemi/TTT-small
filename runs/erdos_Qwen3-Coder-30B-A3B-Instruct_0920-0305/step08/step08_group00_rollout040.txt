import numpy as np
from scipy.signal import correlate
from scipy import optimize
import time
import math

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_int(h):
    """Compute correlation using FFT and return integer counts"""
    h_rev = h[::-1].astype(np.float64)
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    # round to nearest integer (small float errors)
    corr_int = np.rint(corr).astype(int)
    return corr_int

def compute_c5(h):
    """Compute C5 value from integer correlation"""
    corr = compute_corr_int(h)
    return dx * np.max(corr)

def evaluate_c5(h):
    """Run-time evaluation of C5 using FFT"""
    return compute_c5(h)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def adaptive_sampling_initialization():

def insert_peaks(h):
    """Add two sharp peaks to make better overlap suppression"""
    h_copy = h.copy()
    # Locate areas with low values and add two strong spikes
    thresholds = [0.2, 0.5, 0.8]
    for thres in thresholds:
        mask = (h_copy < thres)
        if np.sum(mask) > 10:
            # Set some low values to 1
            indices = np.where(mask)[0]
            for _ in range(min(2, len(indices)//3)):
                idx = np.random.choice(indices)
                h_copy[idx] = 1.0
                # Smooth out nearby values by reducing them slightly
                for offset in [-1, 0, 1]:
                    if 0 <= idx + offset < len(h_copy):
                        h_copy[idx + offset] = max(0, h_copy[idx + offset] - 0.2)
            break
    
    return project_to_sum_box(h_copy, TARGET_SUM)

def simulated_annealing(n_points, time_budget, start_time):
    """Run simulated annealing to minimize max overlap"""
    # Generate initial binary pattern
    h = np.zeros(n_points, dtype=int)
    h[:n_points//2] = 1
    np.random.shuffle(h)
    best_h = h.copy()
    current_h = h.copy()
    
    corr_best = compute_corr_int(best_h)
    best_max = corr_best.max()
    
    corr_current = corr_best.copy()
    current_max = best_max

    # Set cooling schedule
    T0 = 5.0  # Initial temperature
    T = T0
    cooling = 0.9995
    min_T = 1e-4  # Minimum temperature

    iterations = 0
    while time.time() - start_time < time_budget:
        # Generate swap
        ones_idx = np.where(current_h == 1)[0]
        zeros_idx = np.where(current_h == 0)[0]
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
            
        i_one = np.random.choice(ones_idx)
        i_zero = np.random.choice(zeros_idx)
        
        # Swap
        cand_h = current_h.copy()
        cand_h[i_one] = 0
        cand_h[i_zero] = 1

        cand_corr = compute_corr_int(cand_h)
        cand_max = cand_corr.max()
        delta = cand_max - current_max
        
        if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, 1e-6)):
            current_h = cand_h
            current_max = cand_max
            corr_current = cand_corr
            
            # Update best
            if cand_max < best_max:
                best_h = cand_h.copy()
                best_max = cand_max
                corr_best = cand_corr
        
        T = max(T * cooling, min_T)
        iterations += 1
        
        if iterations % 1000 == 0:
            # Occasionally re-evaluate if we're improving everywhere
            pass
    
    return best_h, best_max

def hill_climb(h, max_iter=10000, patience=500):
    """Local search to improve solution by swapping individual elements"""
    best_h = h.copy()
    corr_best = compute_corr_int(best_h)
    best_max = corr_best.max()
    
    no_improve = 0
    for _ in range(max_iter):
        # Generate random swaps
        improved = False
        for _ in range(100):  # multiple candidate swaps
            i_one = np.random.choice(np.where(best_h == 1)[0])
            i_zero = np.random.choice(np.where(best_h == 0)[0])
            
            cand_h = best_h.copy()
            cand_h[i_one] = 0
            cand_h[i_zero] = 1
            
            cand_corr = compute_corr_int(cand_h)
            cand_max = cand_corr.max()
            
            if cand_max < best_max:
                best_h = cand_h
                best_max = cand_max
                corr_best = cand_corr
                improved = True
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
            if no_improve >= patience:
                break
    
    return best_h, best_max
