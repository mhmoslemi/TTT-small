import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time
import cvxpy as cp
from scipy.linalg import cholesky
from numpy.random import multivariate_normal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def compute_corr_fft(h):
        h_rev = h[::-1]
        b = 1 - h
        L = 2 * len(h) - 1
        corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
        return corr

    def c5_from_corr(corr):
        return np.max(corr)

    def evaluate_c5(h):
        corr = compute_corr_fft(h)
        return c5_from_corr(corr)

    def project_to_sum_box(x, target_sum):
        def f(lam):
            projected = np.clip(x + lam, 0, 1)
            return np.sum(projected) - target_sum

        a, b = -1.0, 1.0
        while f(a) * f(b) > 0:
            a -= 0.1
            b += 0.1

        tol = 1e-10
        for _ in range(50):
            m = (a + b) / 2
            fm = f(m)
            if abs(fm) < tol:
                break
            if fm * f(a) < 0:
                b = m
            else:
                a = m

        result = np.clip(x + (a + b)/2, 0, 1)
        return result

    def compute_shift_matrix(n, s):
        R = np.zeros((n, n))
        for i in range(n - s):
            R[i, i + s] = 1
        return R

    def solve_sdp(n_points, dx):
        print(f"Solving SDP for n_points={n_points}")
        # Variables
        X = Semidef(n_points, name="X")
        mu = Variable(n_points, name="mu")
        lambda_var = Variable(name="lambda_var")

        # Constraints
        constraints = [X >> 0]
        for i in range(n_points):
            constraints.append(X[i, i] == mu[i])

        # Adding constraints for mu bounds
        for i in range(n_points):
            constraints.append(mu[i] >= 0)
            constraints.append(mu[i] <= 1)

        # Mass constraint
        constraints.append(sum(mu) == n_points / 2)

        # Shift constraints
        for s in range(1, (n_points//2) + 1):
            R_s = compute_shift_matrix(n_points, s)
            constraints.append(trace(R_s @ X) >= lambda_var)

        # Objective
        objective = Maximize(lambda_var)

        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=SCS, verbose=False, max_iters=10000, eps=1e-4, time_limit_sec=120)
            if prob.status in [OPTIMAL, APPROX_IMA]:
                mu_sol = mu.value  # Get diagonal of X*
                return mu_sol, prob.value
        except Exception as e:
            print(f"SDP solver failed: {e}")
        return None, None

    def round_and_evaluate(x_star, n_points, dx):
        num_rounds = 500
        best_c5 = float('inf')
        best_h = None
        for _ in range(num_rounds):
            V = cholesky(np.eye(n_points), lower=True)
            try:
                # Generate random Gaussian vector
                g = np.random.normal(0, 1, n_points)
                scores = [np.dot(V[i], g) for i in range(n_points)]
                sorted_indices = np.argsort(scores)[::-1]
                selected_indices = set(sorted_indices[:n_points // 2])
                h = np.zeros(n_points)
                for i in range(n_points):
                    h[i] = 1 if i in selected_indices else 0
                h = project_to_sum_box(h, n_points / 2)
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            except Exception:
                pass
        return best_h, best_c5

    def improve_by_swap_search(h_in, n_points, dx):
        best_h = h_in.copy()
        best_c5 = evaluate_c5(best_h)
        max_iter = 10000
        for _ in range(max_iter):
            if time.time() - start_time > 0.9 * budget_s:
                break
            corr = compute_corr_fft(best_h)
            s_max = np.argmax(corr)
            Bad = []
            Good = []
            for i in range(n_points - s_max):
                if best_h[i] == 1 and best_h[i + s_max] == 0:
                    Bad.append(i)
                if best_h[i] == 0 and best_h[i + s_max] == 1:
                    Good.append(i)
            if not Bad or not Good:
                break
            i = np.random.choice(Bad)
            j = np.random.choice(Good)
            h_new = best_h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
            c5_new = evaluate_c5(h_new)
            if c5_new <= best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        return best_h, best_c5

    n_points = 120
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None

    # First Phase: Solve SDP
    mu_star, lambda_star = solve_sdp(n_points, dx)
    if mu_star is not None:
        # Round solution
        h_rounded, c5_rounded = round_and_evaluate(mu_star, n_points, dx)
        if h_rounded is not None and c5_rounded < best_c5:
            best_h = h_rounded.copy()
            best_c5 = c5_rounded

    # Second Phase: Local search
    if best_h is not None:
        h_fine, c5_fine = improve_by_swap_search(best_h, n_points, dx)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()

    # Optional upscale
    if time.time() - start_time < 0.85 * budget_s:
        n_upscale = 180
        dx_upscale = 2.0 / n_upscale
        h_upscale = np.zeros(n_upscale)
        for i in range(n_points):
            start_idx = i * (n_upscale // n_points)
            end_idx = min(start_idx + (n_upscale // n_points), n_upscale)
            h_upscale[start_idx:end_idx] = best_h[i]
        h_upscale = project_to_sum_box(h_upscale, n_upscale / 2)

        # Solve SDP on upscale
        mu_star_up, _ = solve_sdp(n_upscale, dx_upscale)
        if mu_star_up is not None:
            # Round solution
            h_rounded, c5_rounded = round_and_evaluate(mu_star_up, n_upscale, dx_upscale)
            if h_rounded is not None and c5_rounded < best_c5:
                best_h = h_rounded.copy()
                best_c5 = c5_rounded

        # Improve the upscaled solution
        if best_h is not None:
            h_fine_up, c5_fine_up = improve_by_swap_search(best_h, n_upscale, dx_upscale)
            if c5_fine_up < best_c5:
                best_c5 = c5_fine_up
                best_h = h_fine_up.copy()

    # Final check
    if best_h is None:
        # Fallback to initial value
        best_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.ones(n_points) * 0.5
        best_h = project_to_sum_box(best_h, n_points / 2)
        best_c5 = evaluate_c5(best_h)

    return (best_h.tolist(), float(best_c5), n_points)
