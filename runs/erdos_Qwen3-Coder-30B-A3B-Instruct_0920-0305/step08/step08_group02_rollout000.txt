import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_with_gradients(h0, max_iter=1000):
    """Optimize using gradient-based method with projection"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def local_search_improvement(h0, max_iter=1000):

    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Randomly sample a subset of elements to modify
        idxs = np.random.choice(n_points, size=min(20, n_points//4), replace=False)
        improved = False
        
        for idx in idxs:
            old_val = h[idx]
            # Try multiple small changes
            for delta in [-0.03, -0.02, -0.01, 0.01, 0.02, 0.03]:
                h_new = h.copy()
                h_new[idx] += delta
                
                if h_new[idx] < 0 or h_new[idx] > 1:
                    continue
                
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                new_c5 = evaluate_c5(h_new)
                
                if new_c5 < best_c5:
                    h = h_new.copy()
                    best_c5 = new_c5
                    improved = True
                    break
            if improved:
                break
        
        # If no improvement in this step, make small random changes
        if not improved and np.random.random() < 0.3:
            idx = np.random.randint(0, n_points)
            delta = np.random.uniform(-0.015, 0.015)
            h[idx] += delta
            h = project_to_sum_box(h, TARGET_SUM)
            best_c5 = evaluate_c5(h)
    
    return h

    h = np.zeros(n_points)
    
    # Place two prominent peaks
    peak1_pos = n_points // 3
    peak2_pos = 2 * n_points // 3
    
    # Make peaks tall and narrow
    peak_width = n_points // 20
    for i in range(max(0, peak1_pos - peak_width), min(n_points, peak1_pos + peak_width)):
        h[i] = max(h[i], 1.0 - abs(i - peak1_pos) / peak_width)
    
    for i in range(max(0, peak2_pos - peak_width), min(n_points, peak2_pos + peak_width)):
        h[i] = max(h[i], 1.0 - abs(i - peak2_pos) / peak_width)
    
    # Smooth the edges
    h = np.clip(h, 0, 1)
    
    # Normalize to proper sum
    h = project_to_sum_box(h, TARGET_SUM)
    
    # Fine tune with local search
    h = local_search_improvement(h, max_iter=300)
    return h
