```python
import numpy as np
import time

# Global variables
n_points = 200
TARGET_SUM = n_points // 2
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def random_binary_solution():
    """Generate random binary solution with exactly TARGET_SUM ones"""
    h = np.zeros(n_points, dtype=float)
    ones_indices = np.random.choice(n_points, size=TARGET_SUM, replace=False)
    h[ones_indices] = 1.0
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    
    # Simulated Annealing with restarts
    max_iter = 50000
    initial_temp = 0.015
    final_temp = 0.0001
    time_fraction = 0.95
    
    iteration = 0
    while time.time() - start_time < budget_s * time_fraction:
        h = random_binary_solution()
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Simulated Annealing loop for this restart
        temp = initial_temp
        for i in range(max_iter):
            if time.time() - start_time >= budget_s * time_fraction:
                break
                
            # Choose two random positions
            ones_indices = np.where(h == 1)[0]
            zeros_indices = np.where(h == 0)[0]
            
            if len(ones_indices) == 0 or len(zeros_indices) == 0:
                break
                
            i_pos = np.random.choice(ones_indices)
            j_pos = np.random.choice(zeros_indices)
            
            # Swap the positions
            h[i_pos] = 0
            h[j_pos] = 1
            
            new_c5 = evaluate_c5(h)
            
            # Accept or reject
            delta = current_c5 - new_c5
            
            if delta > 0 or np.random.random() < np.exp(delta / temp):
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = new_c5
            else:
                # Revert the change
                h[i_pos] = 1
                h[j_pos] = 0
                
            # Cooling schedule
            temp = initial_temp * (final_temp / initial_temp) ** (i / max_iter)
            iteration += 1
            
    # Final refinement with gradient-based optimization
    if best_h is not None and time.time() - start_time < budget_s * 0.99:
        # Attempt to refine using local search
        h_refined = best_h.copy()
        c5_refined = evaluate_c5(h_refined)
        
        # Do limited local search
        for _ in range(1000):
            if time.time() - start_time >= budget_s * 0.99:
                break
                
            ones_indices = np.where(h_refined == 1)[0]
            zeros_indices = np.where(h_refined == 0)[0]
            if len(ones_indices) == 0 or len(zeros_indices) == 0:
                break
                
            # Try swapping pairs to improve C5
            i_pos = np.random.choice(ones_indices)
            j_pos = np.random.choice(zeros_indices)
            
            # Swap
            h_refined[i_pos] = 0
            h_refined[j_pos] = 1
            
            new_c5 = evaluate_c5(h_refined)
            
            if new_c5 < c5_refined:
                c5_refined = new_c5
                if new_c5 < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = new_c5
            else:
                # Revert
                h_refined[i_pos] = 1
                h_refined[j_pos] = 0
                
    return (best_h.tolist(), float(best_c5), n_points)
```
