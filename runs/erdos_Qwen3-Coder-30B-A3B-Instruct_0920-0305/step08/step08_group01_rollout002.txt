```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_binary_solution(h_bin):
    """Evaluate a binary solution (from MILP)"""
    h = np.array(h_bin, dtype=float)
    h = project_to_sum_box(h, TARGET_SUM)
    return evaluate_c5(h)

def build_milp_for_shifts(shifts, n=None):
    """Build a MILP model with given shifts for approximating the exact solution"""
    if n is None:
        n = n_points
        
    # Variables
    x = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Objective
    objective = cp.Minimize(t)
    
    # Mass constraint
    mass_constraint = cp.Equality(cp.sum(x), TARGET_SUM)
    
    # Linearization constraints for shifts
    constraints = [mass_constraint, cp.Ge(t, 0)]
    
    # Add constraints for each shift s for which we have products
    for s in shifts:
        # Define binary product variables y[i] = x[i]*x[i+s] if valid
        y_vars = []
        for i in range(n):
            if i + s >= 0 and i + s < n:
                y = cp.Variable(boolean=True)
                y_vars.append((i, y))
                
                constraints.append(cp.Le(y, x[i]))
                constraints.append(cp.Le(y, x[i+s]))
                constraints.append(cp.Ge(y, x[i] + x[i+s] - 1))
                
        # Epigraph constraint: sum(y) >= n/2 - t
        if y_vars:
            sum_y = cp.sum([y for _, y in y_vars])
            constraints.append(cp.Ge(sum_y, TARGET_SUM - t))
    
    # Create problem and solve
    prob = cp.Problem(objective, constraints)
    
    return prob, x, t

def solve_milp_with_row_generation(time_budget_s):
    """Solve using row generation for MILP with exact shifts"""
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    
    # Initial shifts to consider
    active_shifts = list(range(1, 6))  # Start with few small shifts
    
    # Generate initial solution
    h_init = np.array(initial_h_values)
    h_init = h_init / np.sum(h_init) * TARGET_SUM
    
    current_h = np.clip(h_init, 0, 1)
    # Convert to binary for initial feasibility checking
    h_binary = (current_h > 0.5).astype(int)
    current_h = project_to_sum_box(h_binary, TARGET_SUM)
    
    # Run row generation loop
    iteration = 0
    while time.time() - start_time < time_budget_s * 0.7:
        iteration += 1
        if time.time() - start_time > time_budget_s * 0.7:
            break
            
        # Create Milp with current shifts
        try:
            prob, x_var, t_var = build_milp_for_shifts(active_shifts)
            prob.solve(solver=cp.CBC, verbose=False, maximumSeconds=5)
            
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                t_val = t_var.value
                x_best = x_var.value
                
                # Calculate actual C5 from all shifts with valid indices
                h_eval = x_best.astype(float)
                corr = compute_corr_fft(h_eval)
                measured_c5 = np.max(corr) * dx
                
                if measured_c5 < best_c5:
                    best_c5 = measured_c5
                    best_h = h_eval.copy()
                    
                # Identify worst shift not included in active set
                indices = np.arange(len(corr))
                idx_max = np.argmax(corr)
                s_max = idx_max - (len(corr) - 1) // 2
                
                # Check if we must maintain more about this shift 
                if not s_max in active_shifts:
                    active_shifts.append(s_max)
                    if len(active_shifts) > 20:  # Limit growth in active set
                        active_shifts = active_shifts[-10:]  # Keep newest ones
            else:
                break
                
        except:
            break
                
        # Small chance of break to test continuation
        if iteration > 100:  # Prevent too many iterations
            break
            
    return best_h, best_c5

def refine_with_constrained_optimization(h_binary, c5_old):
    """Use continuous optimization to slightly improve binary solution"""
    h_initial = np.array(h_binary)
    h_continuous = h_initial.astype(float)
    
    # Smooth epigraph function
    def smooth_max_overlap(h, tau, gamma=100):
        """Smooth sup over all shifts with parameter gamma"""
        # Calculate all k-shifting: take fast convolution
        corr = compute_corr_fft(h)  # This method encodes sixel-style folds
        return tau + (1.0 / gamma) * np.log(np.sum(np.exp(gamma * (corr - tau))))
    
    def objective(tau_h):
        h, tau = tau_h[:-1], tau_h[-1]
        h = np.clip(h, 0, 1)
        return smooth_max_overlap(h, tau)  
    
    def constraint_eq(x):
        return np.sum(x[:-1]) - TARGET_SUM
    
    initial_guess = np.append(h_continuous, c5_old)
    
    try:
        res = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            constraints={'type': 'eq', 'fun': constraint_eq},
            bounds=[(0, 1) for _ in range(n_points)] + [(0, None)],
            options={'ftol': 1e-6, 'gtol': 1e-6},
            maxiter=200
        )
        if res.success:
            h_opt = np.clip(res.x[:-1], 0, 1)
            best_tau = res.x[-1]
            final_c5 = evaluate_c5(h_opt)
            return h_opt, final_c5
    except:
        pass
    
    return h_continuous, c5_old

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    
    # Phase 1: MILP row generation to get an exact binary optimum
    milp_start = time.time()
    
    h_best, c5_best = solve_milp_with_row_generation(budget_s * 0.7)
    
    if h_best is None:
        # Fallback to existing code if MILP fails
        h_best = np.array(initial_h_values)
        h_best = np.clip(h_best, 0, 1)
        h_best = project_to_sum_box(h_best, TARGET_SUM)
        c5_best = evaluate_c5(h_best)
    
    # Phase 2: polishment of solution
    polish_start = time.time()
    if (time.time() - polish_start) < budget_s * 0.2 and c5_best < 0.5:
        h_best, c5_best = refine_with_constrained_optimization(h_best, c5_best)  
    
    # Final evaluation to make sure it's the actual C5 bound
    c5_final = evaluate_c5(h_best)
    
    return (h_best.tolist(), float(c5_final), n_points)
```
