# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_with_gradients(h0, max_iter=2000):
    """Optimize using gradient-based method with projection"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

                    h_new = h.copy()
                    h_new[idx] += delta
                    
                    if h_new[idx] < 0 or h_new[idx] > 1:
                        continue
                        
                    h_new = project_to_sum_box(h_new, TARGET_SUM)
                    new_c5 = evaluate_c5(h_new)
                    
                    if new_c5 < best_c5:
                        h = h_new.copy()
                        best_c5 = new_c5

                    continue

            # Try a global perturbation
            if np.random.random() < 0.5:
                # Randomly add some noise
                h_noise = h + np.random.normal(0, 0.005, n_points)
                h_noise = project_to_sum_box(h_noise, TARGET_SUM)
                new_c5 = evaluate_c5(h_noise)
                if new_c5 < best_c5:
                    h = h_noise.copy()
                    best_c5 = new_c5
                    
    return h

def insert_peaks(h):
    """Add two sharp peaks to make better overlap suppression"""
    h = h.copy()
    
    # Locate low areas to inject peaks
    mask = (h < 0.3)
    if np.sum(mask) > 10:
        indices = np.where(mask)[0]
        # Inject two high peaks
        for _ in range(2):
            if len(indices) > 0:
                idx = np.random.choice(indices)
                h[idx] = 1.0
                # Slightly decrease neighboring values
                for offset in [-2, -1, 0, 1, 2]:
                    if 0 <= idx + offset < len(h):
                        h[idx + offset] = max(0, h[idx + offset] - 0.3)
        
        h = project_to_sum_box(h, TARGET_SUM)
    
    return h

def genetic_algorithm_approach(h0, generations=100, pop_size=30):
    """Use a basic genetic algorithm to explore the solution space"""
    population = [h0.copy() for _ in range(pop_size)]
    for i in range(pop_size):
        # Add some diversity
        population[i] += np.random.normal(0, 0.01, n_points)
        population[i] = project_to_sum_box(population[i], TARGET_SUM)
    
    best_individual = h0.copy()
    best_fitness = evaluate_c5(h0)
    
    for gen in range(generations):
        # Evaluate fitness
        fitness = [evaluate_c5(ind) for ind in population]
        
        # Keep the best
        elite_idx = np.argmin(fitness)
        if fitness[elite_idx] < best_fitness:
            best_fitness = fitness[elite_idx]
            best_individual = population[elite_idx].copy()
        
        # Generate next generation
        new_population = [population[elite_idx].copy()]  # Elitism
        
        # Tournament selection followed by crossover and mutation
        for _ in range(pop_size - 1):
            # Tournament selection
            idx1, idx2 = np.random.choice(pop_size, 2, replace=False)
            if fitness[idx1] < fitness[idx2]:
                parent1 = population[idx1]
            else:
                parent1 = population[idx2]
                
            # Mutation
            child = parent1 + np.random.normal(0, 0.005, n_points)
            child = project_to_sum_box(child, TARGET_SUM)
            new_population.append(child)
            
        population = new_population
        
    return best_individual, best_fitness
