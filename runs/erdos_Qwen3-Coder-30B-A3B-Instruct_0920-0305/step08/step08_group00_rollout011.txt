import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_corr_int(h):
    """Compute correlation of h and (1-h) as integer values"""
    corr = compute_corr_fft(h)
    # Return integer part, rounded to nearest
    return np.rint(corr).astype(int)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = None
    best_c5 = float('inf')
    
    # Helper function: simulated annealing with binary patterns
    def simulated_annealing_binary(time_budget):
        # Generate initial binary pattern with exactly n_points/2 ones
        h = np.zeros(n_points, dtype=int)
        h[:int(n_points/2)] = 1
        np.random.shuffle(h)
        
        best_h = h.copy()
        best_corr = compute_corr_int(best_h)
        best_max = best_corr.max()
        
        T0 = 5.0
        T = T0
        cooling = 0.9995
        no_improve = 0
        max_no_improve = 1000
        
        while time.time() - start_time < time_budget and no_improve < max_no_improve:
            # Generate candidate by swapping a random 1 and 0
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
            
            i_one = np.random.choice(ones_idx)
            i_zero = np.random.choice(zeros_idx)
            
            # Create candidate
            cand_h = h.copy()
            cand_h[i_one] = 0
            cand_h[i_zero] = 1
            
            # Compute correlation
            cand_corr = compute_corr_int(cand_h)
            cand_max = cand_corr.max()
            
            delta = cand_max - best_max
            
            # Accept with simulated annealing probability
            if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, 1e-6)):
                h = cand_h
                best_max = cand_max
                best_corr = cand_corr
                if cand_max < best_c5:
                    best_h = h.copy()
                    best_c5 = cand_max
                no_improve = 0
            else:
                no_improve += 1
                
            T *= cooling
            
        return best_h, best_c5
    
    # Hill climbing to refine solution
    def hill_climb(h, max_iter=1000, patience=100):
        best_h = h.copy()
        best_max = compute_corr_int(best_h).max()
        no_improve = 0
        
        for _ in range(max_iter):
            # Try random swaps
            swaps = []
            for _ in range(20):
                ones_idx = np.where(best_h == 1)[0]
                zeros_idx = np.where(best_h == 0)[0]
                if len(ones_idx) == 0 or len(zeros_idx) == 0:
                    break
                i_one = np.random.choice(ones_idx)
                i_zero = np.random.choice(zeros_idx)
                swaps.append((i_one, i_zero))
                
            improved = False
            for i_one, i_zero in swaps:
                cand_h = best_h.copy()
                cand_h[i_one] = 0
                cand_h[i_zero] = 1
                cand_max = compute_corr_int(cand_h).max()
                
                if cand_max < best_max:
                    best_h = cand_h
                    best_max = cand_max
                    improved = True
                    break
                    
            if improved:
                no_improve = 0
            else:
                no_improve += 1
                if no_improve >= patience:
                    break
                    
        return best_h, best_max
    
    # Run multiple annealing runs with different seeds
    time_left = budget_s - (time.time() - start_time)
    
    # Run initial optimization
    if time_left > 10:
        candidates = []
        for i in range(5):
            np.random.seed(seed + i)
            h, c5 = simulated_annealing_binary(time_left * 0.8)
            if h is not None and c5 < best_c5:
                best_h = h
                best_c5 = c5
                candidates.append((h, c5))
        
        # Fine tune with hill climbing
        if best_h is not None:
            refined_h, refined_c5 = hill_climb(best_h, max_iter=500, patience=50)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    
    # Final refinement
    if best_h is not None:
        # Instead of using breakup, we use the provided method directly
        # Just return as the problem requires floating point values
        result_h = [float(x) for x in best_h]
        result_c5 = dx * best_c5
        return (result_h, float(result_c5), n_points)
    
    # If no solution found, return default (but should not happen)
    h_default = np.zeros(n_points)
    h_default[:n_points//2] = 1
    return ([float(x) for x in h_default], 0.5, n_points)
