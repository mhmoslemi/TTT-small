```python
import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
from scipy.linalg import eigh
import math

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = np.correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def compute_shift_matrix(n, s):
    """Construct binary shift matrix R_s for shift s"""
    R = np.zeros((n, n))
    for i in range(n - s):
        R[i, i + s] = 1
    return R

def solve_sdp(n, mu0_fixed=True, target_sum=None):
    """Solve the SDP relaxation for the Erdős problem"""
    # Create variables
    X = cp.Variable((n, n), symmetric=True)
    mu = cp.Variable(n)
    lambda_var = cp.Variable()

    # Constraints
    constraints = [X >> 0]  # PSD constraint
    if mu0_fixed:
        constraints.append(mu[0] == 1.0)  
    else:
        constraints.append(mu[0] == 0.5) 
        
    constraints.append(cp.sum(mu) == target_sum)
    constraints.append(mu >= 0)
    constraints.append(mu <= 1)
    constraints.append(cp.diag(X) == mu)
    
    # Add shift constraints
    for s in range(1, n // 2 + 1):
        R_s = compute_shift_matrix(n, s)
        # trace(R_s @ X) >= lambda_var
        constraints.append(cp.trace(R_s @ X) >= lambda_var)

    # Objective
    objective = cp.Maximize(lambda_var)
    
    # Solve the problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=2000, eps=1e-4)
        return X.value, mu.value, lambda_var.value
    except Exception as e:
        print("SDP failure:", e)
        return None, None, None

def random_hyperplane_rounding(V, target_sum):
    """Performs randomized hyperplane rounding of the SDP solution"""
    # V is such that X ≈ V@V.T
    # Sample a random vector
    g = np.random.randn(V.shape[1])
    scores = V @ g
    # Use the top target_sum elements
    indices = np.argpartition(-scores, target_sum)
    x = np.zeros(V.shape[0])
    x[indices[:target_sum]] = 1
    return x

def project_to_sum(x, target_sum):
    """Project onto sum constraint"""
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-5:
        return x
    # Simple approach: keep top elements by value
    indices = np.argsort(-x)
    # Make sure we don't add 1s more than we remove 0s (and vice versa)
    # Just reorder to match the sum
    to_set_to_zero = int(current_sum - target_sum)
    if to_set_to_zero > 0:
        x[indices[:to_set_to_zero]] = 0
    elif to_set_to_zero < 0:
        x[indices[to_set_to_zero:]] = 1
    return x

def local_improvement(x, max_iter=10000):
    """Perform local search to improve x by swapping at worst shift"""
    # Precompute full correlation
    corr = correlate(x, 1 - x, mode="full")
    worst_shift = np.argmax(corr)
    # Convert to actual shift index (centered)
    shift = worst_shift - (len(x) - 1)
    
    # Find indices where x_i=1 and x_{i+shift}=0 (contributes to worst shift)
    bad_indices = []
    for i in range(len(x) - abs(shift)):
        if shift > 0:
            if x[i] == 1 and x[i + shift] == 0:
                bad_indices.append(i)
        else:
            if x[i] == 1 and x[i + shift] == 0:
                bad_indices.append(i)
    
    # Find indices where x_j=0 and x_{j+shift}=1 (i.e., swapping helps)
    good_indices = []
    for j in range(len(x) - abs(shift)):
        if shift > 0:
            if x[j] == 0 and x[j + shift] == 1:
                good_indices.append(j)
        else:
            if x[j] == 0 and x[j + shift] == 1:
                good_indices.append(j)
    
    unless_changed = 0
    i = 0
    while i < max_iter:
        if not bad_indices or not good_indices:
            break
        bad_idx = np.random.choice(bad_indices)
        good_idx = np.random.choice(good_indices)
        new_x = x.copy()
        new_x[bad_idx] = 0
        new_x[good_idx] = 1
        
        # Recompute just the specific correlations affected by the change
        old_corr = corr.copy()
        # Update corr for affected positions
        corr = correlate(new_x, 1 - new_x, mode="full")
        new_worst = np.argmax(corr)
        if corr[new_worst] <= old_corr[worst_shift]:
            x = new_x
            # Update affected lists
            # We mustえ더 확실히 상태를 갱신해야 한다...
            # 나중에 개선할 예정이지만 여기서 탐색이 아직은 통계적으로 과도것?
            worst_shift = new_worst
            shift = worst_shift - (len(x) - 1)
        i += 1
        if i % 100 == 0:
            unless_changed += 1
            if unless_changed > 1000:
                break
    return x

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    global n_points, dx, TARGET_SUM
    TARGET_SUM = n_points / 2
    
    best_c5 = float('inf')
    best_h = None
    # Phase 1: SDP solve with warm start
    # Try to solve
    remaining = budget_s - (time.time() - start_time)
    if remaining < 10:
        return ([0.5]*n_points, 0.5, n_points)  # Fallback
    
    print("Solving SDP...")
    X_sol, mu_sol, lambda_val = solve_sdp(n_points, mu0_fixed=True, target_sum=TARGET_SUM)
    if X_sol is None:
        return ([0.5]*n_points, 0.5, n_points)
    
    print("Rounding...")
    # Extract row vectors from V such that X ≈ V @ V.T
    # Check if it's rank one (not too dense)
    try:
        V = eigh(X_sol, eigvals_only=False)
        V = V[:, -1]  # Take the eigenvector associated with largest eigenvalue
        V = np.outer(V, V)
        
        best_c5 = float('inf')
        best_x = None
        
        # Do multiple random rounds
        max_rounds = 500
        for i in range(max_rounds):
            if time.time() - start_time > budget_s - 30:
                break
            x_rounded = random_hyperplane_rounding(V, TARGET_SUM)
            x_rounded = project_to_sum(x_rounded, TARGET_SUM)
            c5 = evaluate_c5(x_rounded)
            if c5 < best_c5:
                best_c5 = c5
                best_x = x_rounded.copy()
                
        if best_x is not None:
            print("Improving by local search...")
            # Local improvement
            improved_x = local_improvement(best_x.copy(), 5000)
            improved_c5 = evaluate_c5(improved_x)
            if improved_c5 < best_c5:
                best_c5 = improved_c5
                best_x = improved_x
                
        if best_c5 == float('inf'):
            return ([0.5]*n_points, 0.5, n_points)
            
        return (best_x.tolist(), float(best_c5), n_points)
                
    except Exception as e:
        print("Error in rounding:", e)
        return ([0.5]*n_points, 0.5, n_points)
```
