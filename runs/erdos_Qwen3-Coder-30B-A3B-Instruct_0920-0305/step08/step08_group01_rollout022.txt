```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

# Bypassing local variables for code clarity
# Global constant
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2.0

def compute_corr_fft(h):
    """FFT-based correlation computation"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value using dx"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT and skip the max() to avoid huge memory"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector onto box constraints [0,1] with sum = target_sum using binary search"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')

    # Some reasonable shifts to begin with
    initial_shifts = list(range(1, 6))  # s = 1,2,3,4,5
    initial_shifts.extend([-x for x in initial_shifts])  # include negative shifts
    initial_shifts = list(set(initial_shifts))  # dedupe

    already_added = set()
    current_shifts = initial_shifts.copy()
    acoustic_test_theory = []
    
    # MILP variables
    x = cp.Variable(n_points, boolean=True)
    t = cp.Variable()  # epigraph variable
    y = {}
    y_constr = []

    # Setup solver
    mip_solver = 'GUROBI' if 'GUROBI' in cp.installed_solvers() else 'CBC'
    
    # SIMD optimization
    def build_model(shifts):
        # Rebuild the epigraph part
        model = cp.Problem(cp.Minimize(t), [
            cp.sum(x) == TARGET_SUM,
            t >= 0
        ])
        return model

    def solve_milp(shifts, model=None):
        # Re-build model with fresh constraints
        if model is None:
            model = cp.Problem(cp.Minimize(t), [
                cp.sum(x) == TARGET_SUM,
                t >= 0
            ])
        
        # Clear previous y entries
        for key in list(y.keys()):
            y.pop(key, None)
        y_constr.clear()
        
        # Add fresh constraint terms
        products = {}
        for s in shifts:
            for i in range(n_points):
                if i + s >= 0 and i + s < n_points:
                    # This is how to index into x with a shift
                    key = (i, s)
                    y[key] = cp.Variable(boolean=True)
                    y_constr.append(y[key] <= x[i])
                    y_constr.append(y[key] <= x[i + s])
                    y_constr.append(y[key] >= x[i] + x[i + s] - 1)
                    if (i, s) not in products:
                        products[(i, s)] = y[key]
                else:
                    pass  # ignore out-of-bounds
        
        # Add epigraph constraints
        for s in shifts:
            lhs = cp.sum([y[(i, s)] for i in range(n_points) if i+s >= 0 and i+s < n_points])
            y_constr.append(lhs >= TARGET_SUM - t)

        model.constraints.extend(y_constr)
        try:
            model.solve(solver=mip_solver, verbose=False, maxSeconds=10)
            return x.value, t.value
        except Exception:
            try:
                model.solve(solver='CBC', verbose=False)
                return x.value, t.value
            except:
                return None, None

    # Main solution loop with row generation
    # Early termination trick: assume best at ~0.4 
    MAX_ITERATIONS = 100
    MAX_TIME_S = 0.7 * budget_s
    it = 0
    while sum(1 for t in range(n_points) if t in current_shifts) < 30 and time.time() - start_time < MAX_TIME_S:
        model = build_model(current_shifts)
        x_val, t_val = solve_milp(current_shifts, model)

        if x_val is None:
            break
            
        # Evaluate the actual best
        better_h = np.round(x_val)
        current_c5 = evaluate_c5(better_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = better_h.copy()
            
        # Compute all correlations to see which one breaches the bound
        corr = compute_corr_fft(better_h)
        max_idx = np.argmax(corr)
        max_corr_val = corr[max_idx]
        
        # Check how large the difference: this strategy tries to add
        # the most violating shift if possible
        if max_corr_val > t_val + 1e-8:
            new_s = max_idx - (len(better_h) - 1)  # or use direct index?
            # Find match of this shift in indices
            if new_s not in current_shifts:
                current_shifts.append(new_s)
                # debug print
                # print(f"\tAdding shift {new_s} (max: {max_corr_val} > t={t_val})")
                
        it += 1

    # Fallback
    if best_h is None:
        best_h = np.ones(n_points) * 0.5  # default behavior
        best_c5 = evaluate_c5(best_h)

    # Polish final solution
    FINAL_OPT_TIME = 0.2 * budget_s
    TIMES_POLISH = 50
    polishing_h = best_h.copy()
    
    # define objective function
    def epigraph_smooth(h, t, gamma=100):
        total = 0
        corr = compute_corr_fft(h)
        for i in range(len(corr)):
            total += np.exp(gamma * (corr[i] - t))
        return t + np.log(total) / gamma

    def obj_func(h_and_t):
        h = h_and_t[:n_points]
        t = h_and_t[n_points]
        return epigraph_smooth(h, t)
    
    def grad_func(h_and_t):
        # Simple approximate gradient
        h = h_and_t[:n_points]
        t = h_and_t[n_points]
        corr = compute_corr_fft(h)
        g_h = np.zeros(n_points)
        g_t = 1.0
        log_term = 0
        shift_term = 0
        # Approximate gradient based on shifting idea
        d = np.ones(n_points) * 0.001
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += d[i]
            h_plus = project_to_sum_box(h_plus, TARGET_SUM)
            corr_plus = compute_corr_fft(h_plus)
            grad = (corr_plus - corr) / d[i]
            g_h[i] = np.sum(grad)
        return np.append(g_h, g_t)

    # Initial linear construction
    init_h = best_h.copy()
    init_t = best_c5/dx
    if all(abs(x) < 1e-6 for x in init_h):  # fallback logic
        init_h = np.ones(n_points) * 0.5
    init_val = np.append(init_h, init_t)
    
    # Run a few rounds of_optimization
    start_polish = time.time()
    for j in range(TIMES_POLISH):
        if time.time() - start_polish > FINAL_OPT_TIME:
            break
        try:
            # Smooth optimization approach
            res = minimize(
                obj_func,
                init_val,
                method='SLSQP',
                bounds=[(0, 1)] * n_points + [(0, None)],
                jac=grad_func,
                options={'maxiter': 1, 'ftol': 1e-10}
            )
            if res.success:
                init_val = res.x
                polished_h = np.clip(init_val[:n_points], 0, 1)
                polished_h = project_to_sum_box(polished_h, TARGET_SUM)
                polished_c5 = evaluate_c5(polished_h)
                if polished_c5 < best_c5:
                    best_c5 = polished_c5
                    best_h = polished_h.copy()
            else:
                break
        except:
            break
            

    return (best_h.tolist(), float(best_c5), n_points)
```
