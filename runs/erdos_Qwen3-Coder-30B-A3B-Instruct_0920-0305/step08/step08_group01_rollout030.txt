import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_c5(h):
    """Compute C5 value from correlation"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def random_binary_solution():
    """Generate a random binary solution with exactly TARGET_SUM ones"""
    h = np.zeros(n_points, dtype=float)
    ones_indices = np.random.choice(n_points, size=int(TARGET_SUM), replace=False)
    h[ones_indices] = 1.0
    return h

def simulate_annealing():
    """Run simulated annealing on binary patterns"""
    # Initialize best solution
    best_h = random_binary_solution()
    best_c5 = compute_c5(best_h)
    
    # Annealing parameters
    initial_temp = 0.01
    final_temp = 0.0001
    max_iter = 100000
    
    start_time = time.time()
    for i in range(max_iter):
        if time.time() - start_time > 550:  # Leave some time for final refinement
            break
            
        # Temperature schedule
        temp = initial_temp * (final_temp / initial_temp) ** (i / max_iter)
        
        # Generate neighbor by swapping a 1 and 0
        ones_indices = np.where(best_h == 1)[0]
        zeros_indices = np.where(best_h == 0)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            continue
            
        i_swap = np.random.choice(ones_indices)
        j_swap = np.random.choice(zeros_indices)
        
        # Create new solution
        new_h = best_h.copy()
        new_h[i_swap] = 0
        new_h[j_swap] = 1
        
        # Compute new C5
        new_c5 = compute_c5(new_h)
        
        # Accept or reject
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
        else:
            # Accept with probability
            delta = new_c5 - best_c5
            if np.random.random() < np.exp(-delta / temp):
                best_h = new_h
                best_c5 = new_c5
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    TARGET_SUM = n_points / 2
    best_h = None
    best_c5 = float('inf')
    
    # Run multiple restarts
    num_restarts = 10
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s - 10:
            break
            
        h, c5 = simulate_annealing()
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Final optimization step
    if best_h is not None:
        # Perform a simple gradient descent refinement
        try:
            # Since we have a binary solution, refine by smoothing edges
            # This is a rough gradient descent - not exactly proper
            for _ in range(1000):
                if time.time() - start_time > budget_s - 5:
                    break
                h = best_h.copy()
                
                # Small random changes
                idx = np.random.randint(0, n_points)
                h[idx] = 1 - h[idx]  # Flip bit
                h = np.clip(h, 0, 1)
                
                # Renormalize to maintain sum constraint
                current_sum = np.sum(h)
                if current_sum != TARGET_SUM:
                    # Find indices to adjust
                    if current_sum > TARGET_SUM:
                        # Too many ones - reduce
                        indices_to_reduce = np.where(h == 1)[0]
                        if len(indices_to_reduce) > 0:
                            idx_reduce = np.random.choice(indices_to_reduce)
                            h[idx_reduce] = 0
                    else:
                        # Too few ones - increase
                        indices_to_increase = np.where(h == 0)[0]
                        if len(indices_to_increase) > 0:
                            idx_increase = np.random.choice(indices_to_increase)
                            h[idx_increase] = 1
                
                new_c5 = compute_c5(h)
                if new_c5 < best_c5:
                    best_h = h
                    best_c5 = new_c5
                    
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
