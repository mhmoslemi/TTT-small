# Global variables
TARGET_SUM = None

dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Use projection to move to nearest point satisfying sum constraint
    x = np.array(x)
    current_sum = np.sum(x)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return np.clip(x, 0, 1)
    
    # Sort indices by value
    sorted_indices = np.argsort(x)
    
    # Adjust from smallest to largest
    for i in range(len(x)):
        idx = sorted_indices[i]
        if diff > 0:
            # Need to increase, can't exceed 1
            delta = min(diff, 1.0 - x[idx])
            x[idx] += delta
            diff -= delta
            if diff <= 0:
                break
        else:
            # Need to decrease, can't go below 0
            delta = min(-diff, x[idx])
            x[idx] -= delta
            diff += delta
            if diff >= 0:
                break
    
    return np.clip(x, 0, 1)

def artificial_neural_network_like_search(h0):
    """Use an MLP-inspired approach with adaptive learning"""
    best_h = h0.copy()
    best_value = evaluate_h_solution(best_h)
    
    # Perturb the current best with adaptive learning rate
    for iter_i in range(300):
        # Sample a directional vector (gaussian noise)
        direction = np.random.normal(0, 0.02, n_points)
        # Scale by learning rate that decreases over time
        lr = 0.01 * (1 - iter_i/300)
        
        # Try various steps
        step_sizes = [lr * 0.5, lr, lr * 2.0]
        for step in step_sizes:
            h_candidate = best_h + step * direction
            value = evaluate_h_solution(h_candidate)
            
            if value < best_value:
                best_h = h_candidate.copy()
                best_value = value
                break
    
    return best_h

def genetic_algorithm_search():
    """Run a simple genetic algorithm with mutation and crossover"""
    population_size = 30
    generations = 100
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h = np.random.uniform(0, 1, n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    
    for gen in range(generations):

        fitness = [evaluate_h_solution(h) for h in population]
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep best
        new_population = population[:population_size//3]
        
        # Create offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1_idx = np.random.choice(len(population[:population_size//2]))
            parent2_idx = np.random.choice(len(population[:population_size//2]))
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover (uniform)
            mask = np.random.rand(n_points) > 0.5
            child = np.where(mask, parent1, parent2)
