import time

# Global constants
TARGET_SUM = None
n_points = 300
dx = 2.0 / n_points
n_coarse = 150

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Use water-filling projection
    # Sort elements (descending)
    sorted_indices = np.argsort(x)[::-1]
    sorted_x = x[sorted_indices]
    
    # Find the threshold lambda
    cumsum = np.cumsum(sorted_x)
    threshold = (target_sum - cumsum[-1]) / len(sorted_x)
    
    if threshold >= sorted_x[-1]:
        # All elements can be raised to 1
        threshold = sorted_x[-1]
    
    # Check if threshold is reasonable
    for i in range(len(sorted_x) - 1, -1, -1):
        if cumsum[i] <= target_sum:
            threshold = (target_sum - cumsum[i]) / (len(sorted_x) - i)
            break
    
    # Cap at 1 and make sure sum = target_sum
    result = np.clip(x + threshold, 0, 1)
    
    # Adjust exactly to sum
    current_sum = np.sum(result)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Distribute difference to small values
        if diff > 0:
            # Need larger values
            sorted_then = np.argsort(result)
            for i in range(len(sorted_then) - 1, -1, -1):
                idx = sorted_then[i]
                available = 1.0 - result[idx]
                if available > 1e-10:
                    add = min(diff, available)
                    result[idx] += add
                    diff -= add
                    if abs(diff) < 1e-10:
                        break
        else:
            # Need smaller values
            sorted_then = np.argsort(result)[::-1]
            for i in range(len(sorted_then)):
                idx = sorted_then[i]
                available = result[idx]
                if available > 1e-10:
                    take = min(-diff, available)
                    result[idx] -= take
                    diff += take
                    if abs(diff) < 1e-10:
                        break
    
    return result

def rasterise_and_project(θ):
    """Rasterise a parameter vector θ to h with sum constraint"""
    K = 8  # number of blocks
    # Extract widths and heights
    widths = np.exp(θ[:K])
    widths /= np.sum(widths) / 2  # normalise to width 2
    heights = 1 / (1 + np.exp(-θ[K:]))
    
    # Rasterize
    h = np.zeros(n_points)
    pos = 0
    for i in range(K):
        end_pos = int(pos + np.round(widths[i] / dx))
        if end_pos >= n_points:
            end_pos = n_points
        h[pos:end_pos] = heights[i]
        pos = end_pos
        if pos >= n_points:
            break
    
    # Project to exact constraints
    h = project_to_sum_box(h, TARGET_SUM)
    return h

    c5 = evaluate_c5(h)
    return c5
