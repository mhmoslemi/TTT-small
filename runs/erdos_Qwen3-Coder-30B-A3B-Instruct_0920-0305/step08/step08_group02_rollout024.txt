```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using bisection"""
    def project_with_lambda(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = project_with_lambda(m)
        if abs(fm) < 1e-10:
            break
        if fm * project_with_lambda(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_autocorrelation(h, k):
    """Evaluate autocorrelation at shift k"""
    corr = np.correlate(h, h, mode='full')
    return corr[len(h) - 1 + k]

def solve_binary_milp(N, M, max_seconds=200):
    """Solve binary MILP to maximize minimum autocorrelation"""
    if N > 120:
        N = 120
    
    try:
        # Create variables
        y = cp.Variable(N, boolean=True)
        w = cp.Variable((N-1, N), nonneg=True)
        t = cp.Variable(1, nonneg=True)
        
        constraints = [
            cp.sum(y) == M,
            t >= 0
        ]
        
        # Add product constraints
        for k in range(1, N-1):
            for i in range(N-k):
                constraints.extend([
                    w[k-1, i] <= y[i],
                    w[k-1, i] <= y[i+k],
                    w[k-1, i] >= y[i] + y[i+k] - 1,
                    w[k-1, i] >= 0
                ])
        
        # Minimum autocorrelation constraints
        for k in range(1, N-1):
            constraints.append(cp.sum(w[k-1, :]) >= t)
        
        # Objective
        objective = cp.Maximize(t)
        prob = cp.Problem(objective, constraints)
        
        # Solve using CBC
        prob.solve(solver=cp.CBC, max_seconds=max_seconds, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return y.value.astype(int)
        
    except Exception as e:
        print(f"MILP solver error: {e}")
    
    # Fallback strategy if MILP fails
    return None

def construct_greedy_binary(N, M):
    """Construct binary sequence via greedy max-min autocorrelation approach"""
    y = np.zeros(N, dtype=int)
    used_indices = set()
    
    # Try to place M ones optimally
    for _ in range(M):
        best_score = -1
        best_i = -1
        for i in range(N):
            if i in used_indices:
                continue
                
            # Temporarily set y[i] = 1 and compute min autocorrelation
            y_copy = y.copy()
            y_copy[i] = 1
            min_ac = float('inf')
            
            for k in range(1, N):
                ac = evaluate_autocorrelation(y_copy, k)
                if ac < min_ac:
                    min_ac = ac
            
            if min_ac > best_score:
                best_score = min_ac
                best_i = i
                
        if best_i >= 0:
            y[best_i] = 1
            used_indices.add(best_i)
            
    return y

def compute_subgradient(h, k_star):
    """Compute subgradient w.r.t. the worst shift k_star"""
    N = len(h)
    g = np.zeros(N)
    
    for i in range(N):
        g[i] = 1.0
        if i + k_star < N:
            g[i] -= h[i + k_star]
        if i - k_star >= 0:
            g[i] -= h[i - k_star]
    
    return g

def subgradient_descent(h0, max_seconds=200):
    """Perform deterministic subgradient descent"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    t = 1
    
    while time.time() - start_time < max_seconds:
        # Compute worst shift
        corr = compute_corr_fft(h)
        k_star = np.argmax(corr)
        if k_star >= len(corr) // 2:
            k_star = k_star - len(corr) + 1
        
        # Compute subgradient
        g = compute_subgradient(h, k_star)
        
        # Gradient step with diminishing step size
        alpha = 0.5 * dx / np.sqrt(t)
        h_new = h - alpha * g
        
        # Project back to feasible set
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new point
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        h = h_new
        t += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points / 2
    
    # Phase 1: Binary MILP (for up to 0.4 * budget_s)
    h_best = None
    c5_best = float('inf')
    
    phase1_end = start_time + 0.4 * budget_s
    y_binary = solve_binary_milp(n_points, TARGET_SUM, max_seconds=0.4 * budget_s - 1)
    
    if y_binary is not None:
        h_binary = y_binary.astype(float)
        h_best = h_binary
        c5_best = evaluate_c5(h_binary)
    else:
        # Fallback: try greedy binary approach
        y_greedy = construct_greedy_binary(n_points, TARGET_SUM)
        h_greedy = y_greedy.astype(float)
        h_best = h_greedy
        c5_best = evaluate_c5(h_greedy)
    
    # Phase 2: Deterministic Subgradient Descent (0.35 * budget_s)
    phase2_end = start_time + 0.75 * budget_s
    h_refined, c5_refined = subgradient_descent(h_best, max_seconds=phase2_end - time.time())
    
    if c5_refined < c5_best:
        h_best = h_refined
        c5_best = c5_refined
    
    # Phase 3: Stochastic local search (remaining time)
    phase3_end = start_time + budget_s
    while time.time() < phase3_end:
        # Swap move with random indices
        i, j = np.random.choice(n_points, size=2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        
        h_new = h_best.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Clamp values
        h_new = np.clip(h_new, 0, 1)
        # Re-project on sum constraint
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        c5_new = evaluate_c5(h_new)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
    
    return (h_best.tolist(), float(c5_best), n_points)
```
