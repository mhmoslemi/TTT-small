import numpy as np
from scipy.signal import correlate
import time

# Global constants
n_points = 256
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def project_to_box_sum(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Clip first
    x_clipped = np.clip(x, 0, 1)
    current_sum = np.sum(x_clipped)
    
    if abs(current_sum - target_sum) < 1e-10:
        return x_clipped
    
    # Binary search for lambda to get exact sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation using FFT"""
    b = 1 - h
    corr = correlate(h, b, mode='full')
    return dx * np.max(corr)

    """Update correlation c after transferring delta mass from i to j"""

    n = len(h)
    if 0 <= i < n and 0 <= j < n:
        # b-part: contribution from changing h[i]
        s_start = max(0, (n-1) - i)
        s_end = min(n, (n-1) - i + n)
        c[s_start:s_end] += delta * b[i + np.arange(s_end - s_start)]
        
        # h-part: contribution from changing b[j] 
        s_start = max(0, j)
        s_end = min(n, j + n)
        c[s_start:s_end] -= delta * h[np.arange(s_end - s_start, 0, -1) - 1]
