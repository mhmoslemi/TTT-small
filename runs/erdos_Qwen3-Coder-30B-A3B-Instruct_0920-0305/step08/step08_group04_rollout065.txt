import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import time

# Fast evaluation of C5 using FFT
def evaluate_c5(h):
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return np.max(corr)

# Water-filling projection onto sum and box constraints
def project_to_sum_box(x, target_sum):
    # Sort values in descending order
    sorted_indices = np.argsort(x)[::-1]
    sorted_vals = x[sorted_indices]
    
    # Initialize cumulative sum
    cumsum = 0.0
    # Find the threshold lambda
    lambda_val = 0.0
    for i in range(len(sorted_vals)):
        cumsum += sorted_vals[i]
        # Check if we should stop
        if cumsum > target_sum:
            # Adjust lambda to satisfy constraint
            lambda_val = np.clip((cumsum - target_sum) / (i + 1), 0.0, 1.0)
            break
    else:
        # All could be 1.0 or less, but might not sum up to target
        if cumsum < target_sum:
            lambda_val = np.clip((target_sum - cumsum) / len(sorted_vals), 0.0, 1.0)
    
    # Apply thresholding
    result = np.maximum(np.minimum(x - lambda_val, 1.0), 0.0)
    
    # Adjust to enforce exact sum constraint
    current_sum = np.sum(result)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Distribute the difference among smallest elements
        sort_indices = np.argsort(result)
        for idx in sort_indices:
            if diff <= 1e-10:
                break
            add = min(diff, 1.0 - result[idx])
            result[idx] += add
            diff -= add
    
    return result

# Linear interpolation from coarse to fine grid
def interpolate_h(h_coarse, n_fine):
    n_coarse = len(h_coarse)
    coarse_x = np.linspace(0, 2, n_coarse)
    fine_x = np.linspace(0, 2, n_fine)
    h_fine = np.interp(fine_x, coarse_x, h_coarse)
    return h_fine

# Parametric construction: Block function with K blocks
def create_block_function(θ, n_blocks=8, n_grid=300):
    # Extract widths and heights
    widths = np.exp(θ[:n_blocks])
    widths = 2 * widths / np.sum(widths)
    heights = 1 / (1 + np.exp(-θ[n_blocks:]))
    
    # Build step function
    h = np.zeros(n_grid)
    dx = 2.0 / n_grid
    cumulative_width = 0
    for i in range(n_blocks):
        start_idx = int(cumulative_width / dx)
        end_idx = int((cumulative_width + widths[i]) / dx)
        end_idx = min(end_idx, n_grid)
        if start_idx < end_idx:
            h[start_idx:end_idx] = heights[i]
        cumulative_width += widths[i]
    
    return h

# Objective for differential evolution
def de_objective(θ):
    h = create_block_function(θ, n_blocks=8, n_grid=300)
    h_projected = project_to_sum_box(h, 150)
    return evaluate_c5(h_projected)

# Sub-gradient implementation for PSGD
def compute_subgradient(h, corr):
    max_val = corr.max()
    active = np.where(corr >= max_val - 1e-12)[0]
    shifts = active - (len(h) - 1)
    
    g = np.zeros(len(h))
    for s in shifts:
        g += -1.0  # constant term
        if s >= 0:
            g[:len(h)-s] += -h[s:]  # -h_{i+s}
            g[s:] += -h[:len(h)-s]  # -h_{i-s}
        else:
            t = -s
            g[t:] += -h[:len(h)-t]
            g[:len(h)-t] += -h[t:]
    
    return g / len(shifts)  # Average over active shifts
