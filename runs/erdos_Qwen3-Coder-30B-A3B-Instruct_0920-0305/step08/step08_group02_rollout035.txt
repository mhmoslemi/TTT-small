import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize
from cvxpy import Variable, Minimize, Problem, sum, multiply, index
import math

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_correlation(h):
    """Compute correlation using FFT for efficiency"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_correlation(h)
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    x = np.array(x)
    # Bisection on lambda to satisfy sum constraint
    def sum_with_lambda(lam):
        y = np.clip(x + lam, 0, 1)
        return np.sum(y) - target_sum
    
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = sum_with_lambda(m)
        if abs(fm) < 1e-10:
            break
        if fm * sum_with_lambda(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_milp_initialization():
    """Solve a MILP for binary version to get a high-quality starting point"""
    try:
        # Set up MILP problem using CVXPY
        N = n_points
        M = N // 2
        y = Variable(N, boolean=True)
        
        # Initialize w_{k,i} to be auxiliary variables for product y_i * y_{i+k}
        w = {}
        t = Variable()
        constraints = [sum(y) == M, t >= 0]
        
        # For each shift k from 1 to N-1
        for k in range(1, N):
            for i in range(N - k):
                w[(k, i)] = Variable()
                constraints.append(w[(k, i)] <= y[i])
                constraints.append(w[(k, i)] <= y[i+k])
                constraints.append(w[(k, i)] >= y[i] + y[i+k] - 1)
                constraints.append(w[(k, i)] >= 0)
        
        # Minimum autocorrelation constraint: sum of w for each k >= t
        for k in range(1, N):
            expr = sum([w[(k, i)] for i in range(N-k)])
            constraints.append(expr >= t)
        
        prob = Problem(Maximize(t), constraints)
        prob.solve(solver=cp.CBC, max_seconds=0.35*n_points)

        if prob.status == cp.OPTIMAL:
            ySol = [int(round(val.value)) for val in y]
            h0 = np.array(ySol)
            return h0
        else:
            return None
    except Exception:
        return None

def subgradient_descent(h_init, max_iter=10000):
    """Sub-gradient descent to refine continuous solution"""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    # Step size
    alpha_0 = 0.1 * dx
    for t in range(max_iter):
        # Compute all correlations
        corr = compute_correlation(h)
        k_star = np.argmax(corr)
        # Compute subgradient for the worst shift
        g = np.zeros_like(h)
        for i in range(len(h)):
            if i - k_star >= 0:
                g[i] += -h[i - k_star]
            if i + k_star < len(h):
                g[i] += -h[i + k_star]
            g[i] += 1

        # Update with step size decreasing as 1/sqrt(t+1)
        alpha = alpha_0 / (1 + math.sqrt(t))
        h_candidate = h - alpha * g
        h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
        
        c5_candidate = evaluate_c5(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
        
        h = h_candidate.copy()
        
        if t > 500 and abs(c5_candidate - best_c5) < 1e-8:
            break
            
    return best_h, best_c5

def lightweight_stochastic_refinement(h_init, max_iter=500):
    """Lightweight simulated annealing for further improvement"""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_temp = 0.05
    for t in range(max_iter):
        temp = start_temp * (1 - t / max_iter)
        i, j = np.random.choice(len(h), 2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        elif np.random.random() < np.exp(-(c5_new - best_c5) / (temp + 1e-10)):
            h = h_new.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    
    # Phase 1: Binary MILP
    h_init = binary_milp_initialization()
    
    if h_init is None:
        # Fallback to other reasonable initialization
        h_init = 0.5 + 0.3 * np.random.randn(n_points)
        h_init = project_to_sum_box(h_init, TARGET_SUM)

    # Phase 2: Continuous refinement via subgradient descent
    h_refined, c5_refined = subgradient_descent(h_init, int(0.35 * budget_s))
    
    # Phase 3: Stochastic improvement
    h_final, c5_final = lightweight_stochastic_refinement(h_refined, int(0.25 * budget_s))
    
    best_h = h_final
    best_c5 = c5_final
    
    # Allow for a final optimization phase with limited time
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1.0:
        # Try remaining time on more optimization
        h_final2, c5_final2 = subgradient_descent(h_final, int(remaining_time*0.2))
        if c5_final2 < best_c5:
            best_h = h_final2
            best_c5 = c5_final2

    return (best_h.tolist(), float(best_c5), n_points)
