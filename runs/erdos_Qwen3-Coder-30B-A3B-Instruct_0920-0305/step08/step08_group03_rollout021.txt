import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def update_correlation(c, h, i, delta_i, j, delta_j):
    """Incrementally update the correlation array when mass is moved from i to j"""
    # No change for same index
    if i == j:
        return c.copy()
    
    # We need to compute the change to the correlation array c due to changes in h[i] and h[j]
    # We will calculate changes for all shifts s
    new_c = c.copy()
    
    # delta_i is change to h[i], so delta_j = -delta_i (mass conservation)
    # For index i:
    # Contribution from b[i+s] when i+s valid
    if 0 <= i < len(h):
        # s range: from -i to (n-1-i) inclusive
        start_s = -i
        end_s = len(h) - 1 - i
        for s in range(start_s, end_s + 1):
            c_idx = s + len(h) - 1
            if 0 <= i + s < len(h):
                new_c[c_idx] += delta_i * (1 - h[i + s])
    
    # Contribution from a[i-s] when i-s valid
    if 0 <= i < len(h):
        # s range: from (i-(n-1)) to i inclusive
        start_s = i - (len(h) - 1)
        end_s = i
        for s in range(start_s, end_s + 1):
            c_idx = s + len(h) - 1
            if 0 <= i - s < len(h):
                new_c[c_idx] -= delta_i * h[i - s]
    
    # For index j:
    # Contribution from b[j+s] when j+s valid
    if 0 <= j < len(h):
        # s range: from -j to (n-1-j) inclusive
        start_s = -j
        end_s = len(h) - 1 - j
        for s in range(start_s, end_s + 1):
            c_idx = s + len(h) - 1
            if 0 <= j + s < len(h):
                new_c[c_idx] += delta_j * (1 - h[j + s])
    
    # Contribution from a[j-s] when j-s valid
    if 0 <= j < len(h):
        # s range: from (j-(n-1)) to j inclusive
        start_s = j - (len(h) - 1)
        end_s = j
        for s in range(start_s, end_s + 1):
            c_idx = s + len(h) - 1
            if 0 <= j - s < len(h):
                new_c[c_idx] -= delta_j * h[j - s]
    
    # Add cross terms: delta_i * (-delta_i) = -delta_i^2 
    # at positions where they apply
    delta_i_squared = delta_i * delta_i
    
    if 0<=j-i<len(h):
        c_idx1 = (j - i) + len(h) - 1
        if 0 <= c_idx1 < 2*len(h)-1:
            new_c[c_idx1] -= delta_i_squared
            
    if 0<=i-j<len(h):
        c_idx2 = (i - j) + len(h) - 1
        if 0 <= c_idx2 < 2*len(h)-1:
            new_c[c_idx2] -= delta_i_squared
    
    return new_c

    best_h = h_interpolated.copy()
    best_c5 = evaluate_c5(h_interpolated)
    TARGET_SUM = n_points / 2

    # Initial correlation computation
    corr = np.correlate(h_interpolated, 1-h_interpolated, mode="full")
    best_corr = corr.copy()

    # Simulated annealing with local refinement
    temp_start = 0.2
    temp_min = 1e-6
    temp_decay = 0.995
    max_iter = 30000
    
    iteration = 0
    temp = temp_start
    
    while time.time() - start_time < budget_s and iteration < max_iter:
        iteration += 1
        
        # Annealing schedule
        if temp < temp_min:
            temp = temp_min
            
        # Choose two random indices
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Propose move
        delta = np.random.uniform(-0.01, 0.01)
        delta_i = delta
        delta_j = -delta
        
        # Apply changes to h
        h_new = best_h.copy()
        h_new[i] += delta_i
        h_new[j] += delta_j
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct sum
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Compute new correlation using incremental update if possible; otherwise FFT
        corr_new = update_correlation(best_corr, best_h, i, delta_i, j, delta_j)
        
        # Evaluate new C5
        new_c5 = dx * np.max(corr_new)
        
        # Accept or reject the move
        if new_c5 < best_c5:
            # Accept immediately
            best_h = h_new.copy()
            best_c5 = new_c5
            best_corr = corr_new.copy()
        elif np.random.rand() < np.exp(-(new_c5 - best_c5) / (temp + 1e-15)):
            # Accept with probability
            best_h = h_new.copy()
            best_c5 = new_c5
            best_corr = corr_new.copy()
            
        # Decrease temperature
        temp *= temp_decay

    return (best_h.tolist(), float(best_c5), n_points)
