import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from math import ceil

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def water_fill_projection(h, target_sum):
    """Exact projection onto {h in [0,1]^N : sum(h) = target_sum}"""
    h = np.array(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        candidate = np.clip(h + mid, 0, 1)
        sum_val = np.sum(candidate)
        if abs(sum_val - target_sum) < 1e-10:
            return candidate
        if sum_val > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0, 1)

def binary_milp_search(N, M, max_seconds):
    """Solve discrete binary MILP for best autocorrelation structure"""
    if N > 120:
        return None, None
    try:
        # Variables
        y = cp.Variable(N, boolean=True)
        w = cp.Variable((N-1, N), nonneg=True)
        t = cp.Variable()

        # Constraints
        constraints = [cp.sum(y) == M]
        for k in range(N - 1):
            for i in range(N - k):
                constraints.extend([
                    w[k, i] <= y[i],
                    w[k, i] <= y[i + k],
                    w[k, i] >= y[i] + y[i + k] - 1,
                    w[k, i] >= 0
                ])
        for k in range(N - 1):
            constraints.append(cp.sum(w[k, :]) >= t)
        constraints.extend([0 <= t, t <= M])
        # Symmetry breaking
        constraints.append(y[0] == 1)
        
        # Objective
        objective = cp.Maximize(t)
        prob = cp.Problem(objective, constraints)
        
        # Solve with timeout
        prob.solve(solver=cp.CBC, max_seconds=max_seconds)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            y_sol = y.value
            return y_sol, prob.value
    except Exception:
        pass
    # Fallback on greedy construction if MILP fails
    return None, None

def greedy_binary_construction(n_points, M):
    """Generate a binary vector with good spacing properties"""
    h = np.zeros(n_points)
    idx = 0
    for i in range(M):
        h[idx] = 1
        idx += 2  # Try placing 1s every 2 positions
        if idx >= n_points:
            idx = i % 2  # Start over with different offset
    h = h.astype(float)
    return h

def subgradient_descent(h0, max_seconds):
    """Perform deterministic subgradient descent on continuous problem"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    # Step size for sub-gradient descent
    alpha0 = 0.5 * dx
    
    # Iterate until timeout
    iteration = 0
    while time.time() - start_time < max_seconds:
        # Compute correlations
        corr = compute_corr_fft(h)
        kstar = np.argmax(corr)
        # Build subgradient
        if kstar < n_points:
            h_plus_k = np.roll(h, -kstar)
            h_minus_k = np.roll(h, kstar)
        else:
            h_plus_k = np.roll(h, -(kstar - n_points))
            h_minus_k = np.roll(h, kstar - n_points)
        g = 1 - h_plus_k - h_minus_k
        
        # Update step
        alpha = alpha0 / np.sqrt(iteration + 1)
        h_new = h - alpha * g
        
        # Project
        h_new = water_fill_projection(h_new, TARGET_SUM)
        
        # Evaluate
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new.copy()
        iteration += 1
        
        if iteration % 100 == 0 and time.time() - start_time > max_seconds:
            break
            
    return best_h, best_c5

def swap_move_annealing(h0, max_seconds):
    """Stochastic swap moves to escape plateaus"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    T_start = 5 * dx
    
    iteration = 0
    while time.time() - start_time < max_seconds:
        # Pick two random indices
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        if i == j:
            continue
        
        # Small perturbation
        delta = np.random.uniform(-0.02, 0.02)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Clip and re-project
        h_new = np.clip(h_new, 0, 1)
        h_new = water_fill_projection(h_new, TARGET_SUM)
        
        # Evaluate
        c5_new = evaluate_c5(h_new)
        
        # Accept based on Metropolis criterion
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            best_h = h_new.copy()
        else:
            T = T_start * (1 - (time.time() - start_time) / max_seconds)
            if T > 0:
                p_accept = np.exp(-(c5_new - best_c5) / T)
                if np.random.rand() < p_accept:
                    h = h_new.copy()
            
        iteration += 1
        
        if iteration % 100 == 0 and time.time() - start_time > max_seconds:
            break
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    TARGET_SUM = n_points / 2

    # Global question: Base solution on MILP or greedy binary?
    # We'll allow some backoff for solver availability
    
    # Compute phases
    phase1_time = 0.4 * budget_s
    phase2_time = 0.35 * budget_s
    phase3_time = budget_s - phase1_time - phase2_time
    
    # Phase 1: Find good binary structure (MILP or greedy)
    binary_sol, best_autocorr_value = binary_milp_search(n_points, TARGET_SUM, phase1_time)
    if binary_sol is None:
        h0 = greedy_binary_construction(n_points, TARGET_SUM)
    else:
        h0 = binary_sol.astype(float)
        # Ensure target sum
        h0 = project_to_sum_box(h0, TARGET_SUM)
    
    # Phase 2: Subgradient descent on continuous variables
    h1, c5_phase2 = subgradient_descent(h0, phase2_time)
    
    # Phase 3: Stochastic swap annealing
    h_final, c5_final = swap_move_annealing(h1, phase3_time)
    
    return (h_final.tolist(), float(c5_final), n_points)
