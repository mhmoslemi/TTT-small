import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_with_gradients(h0, max_iter=1000):
    """Optimize using gradient-based method with projection"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def local_search(h0, max_iter=1000):
    """Local search with neighborhood exploration"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Make small random changes
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.005)
        h_new = h.copy()
        h_new[idx] += delta
        
        if h_new[idx] < 0 or h_new[idx] > 1:
            continue
            
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            
    return h

def multi_scale_sampling():
    """Better initialization using multi-scale approach"""
    h = np.ones(n_points) * 0.5
    
    # Create localized regions of 0 and 1
    regions = [
        (0.1, 0.3, 1.0),   # High region
        (0.4, 0.6, 0.0),   # Low region  
        (0.7, 0.9, 1.0),   # High region
    ]
    
    for start, end, val in regions:
        idx_start = int(start * n_points)
        idx_end = int(end * n_points)
        h[idx_start:idx_end] = val
    
    # Add small noise
    h += np.random.normal(0, 0.02, n_points)
    h = np.clip(h, 0, 1)
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def improve_with_genetic(h0, pop_size=20, generations=50):
    """Improve solution using simple genetic algorithm"""
    population = [h0.copy()]
    
    for _ in range(pop_size - 1):
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = project_to_sum_box(h_rand, TARGET_SUM)
        population.append(h_rand)
    
    for gen in range(generations):
        fitness = [evaluate_c5(h) for h in population]
        # Select top half
        sorted_indices = np.argsort(fitness)[:pop_size//2]
        population = [population[i] for i in sorted_indices]
        
        # Create new generation
        new_pop = population.copy()
        for _ in range(pop_size - len(population)):
            # Crossover of two parents
            parent1 = population[np.random.randint(0, len(population))]
            parent2 = population[np.random.randint(0, len(population))]
            
            # Uniform crossover
            mask = np.random.random(n_points) > 0.5
            child = np.where(mask, parent1, parent2)
            
            # Mutation
            mut_idx = np.random.randint(0, n_points)
            child[mut_idx] = np.clip(child[mut_idx] + np.random.normal(0, 0.02), 0, 1)
            
            child = project_to_sum_box(child, TARGET_SUM)
            new_pop.append(child)
            
        population = new_pop
    
    # Return best from final population
    fitness = [evaluate_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    return population[best_idx]
