import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

# Predefined constant
TARGET_SUM = 1.0  # For the integrated constraint ∫₀² h(x) dx = 1
n_points = 180    # Grid resolution for FFT-based correlation
dx = 2.0 / n_points
epsilon = 1e-4  # To avoid zero interval lengths

def decode_interval_params(z, m):
    """Convert a vector z ∈ [0,1]^{2m} to a valid interval configuration θ"""
    l = epsilon + (1 - 2*epsilon) * z[:m]
    l = l / np.sum(l)  # Ensure sum = 1
    g = z[m:]  # Gaps; not normalized yet

    s = [0.0]
    for i in range(m):
        s.append(s[-1] + l[i] + g[i] if i < m-1 else s[-1] + l[i])

    # Ensure we stay within [0,2]
    total_length = s[-1]
    if total_length > 2.0:
        # Scale gaps
        scale = (2.0 - np.sum(l)) / np.sum(g)
        g_new = g * scale
        s_new = [0.0]
        for i in range(m):
            s_new.append(s_new[-1] + l[i] + g_new[i] if i < m-1 else s_new[-1] + l[i])
        s = s_new
    else:
        # Add extra gap at the end
        extra_gap = 2.0 - s[-1]
        s[-1] = 2.0  # Final point fixed at 2.0

    return np.array(s)

def build_grid_from_intervals(intervals, n):
    """Convert interval endpoints to binary grid representation"""
    h = np.zeros(n)
    step = 2.0 / n
    for i in range(n):
        x = (i + 0.5) * step  # Centre of grid cell
        for j in range(len(intervals)-1):
            if intervals[j] <= x < intervals[j+1]:
                h[i] = 1.0
                break
    return h

def evaluate_c5_fft(h):
    """Fast evaluation of C5 using FFT correlation"""
    b = 1 - h
    c = correlate(h, b, mode='full')
    return dx * np.max(c)

def objective(z):
    """Objective function for differential evolution"""
    intervals = decode_interval_params(z, m)
    h = build_grid_from_intervals(intervals, n_points)
    return evaluate_c5_fft(h)

def local_refinement_initial(theta, max_iter=500):
    """Perform coordinate-descent like refinement on interval endpoints"""
    best_theta = theta.copy()
    best_c5 = evaluate_c5_fft(build_grid_from_intervals(best_theta, n_points))
    improved = True
    iter_count = 0
    while improved and iter_count < max_iter:
        improved = False
        # Iterate through intervals and their edges
        intervals = decode_interval_params(best_theta, m)
        for i in range(1, m):  # Only modify interior edges
            # Try moving left edge
            theta_trial = best_theta.copy()
            theta_trial[i-1] = max(epsilon, theta_trial[i-1] - dx/4)
            intervals_new = decode_interval_params(theta_trial, m)
            h_new = build_grid_from_intervals(intervals_new, n_points)
            c5_new = evaluate_c5_fft(h_new)
            if c5_new < best_c5:
                best_theta = theta_trial.copy()
                best_c5 = c5_new
                improved = True
                break

            # Try moving right edge
            theta_trial = best_theta.copy()
            theta_trial[i + m-1] = max(0, theta_trial[i + m-1] - dx/4)
            intervals_new = decode_interval_params(theta_trial, m)
            h_new = build_grid_from_intervals(intervals_new, n_points)
            c5_new = evaluate_c5_fft(h_new)
            if c5_new < best_c5:
                best_theta = theta_trial.copy()
                best_c5 = c5_new
                improved = True
                break
        iter_count += 1
    return best_theta, best_c5

def fallback_random_search(best_c5):
    """Perform fallback random sampling if DE didn't help"""
    best_h = None
    best_c5_found = best_c5
    for _ in range(5000):
        z = np.random.rand(2 * m)
        intervals = decode_interval_params(z, m)
        h = build_grid_from_intervals(intervals, n_points)
        c5 = evaluate_c5_fft(h)
        if c5 < best_c5_found:
            best_c5_found = c5
            best_h = h.copy()
    if best_h is not None:
        return best_h, best_c5_found
    return None, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    # Initialization
    global m
    m = 5  # Start with a small number of intervals
    start_time = time.time()
    bounds = [(0., 1.)] * (2 * m)
    
    # Phase 1: Random seeding for DE initial population
    best_sol = None
    best_c5 = float('inf')
    for _ in range(10):
        z = np.random.rand(2 * m)
        intervals = decode_interval_params(z, m)
        h = build_grid_from_intervals(intervals, n_points)
        c5 = evaluate_c5_fft(h)
        if c5 < best_c5:
            best_c5 = c5
            best_sol = z.copy()
    
    # Phase 2: Differential Evolution optimization
    de_end_time = start_time + budget_s * 0.7
    res = differential_evolution(
        objective,
        bounds,
        seed=seed,
        maxiter=300,
        popsize=20,
        disp=False,
        callback=lambda x, convergence: None,
        mutations=(0.5, 1.0),
        recombination=0.9
    )
    
    if not res.success or res.fun >= best_c5:
        # Use random seed to start with
        z_start = best_sol
    else:
        z_start = res.x
        best_c5 = res.fun

    # Decode to interval configuration and construct binary
    intervals = decode_interval_params(z_start, m)
    h = build_grid_from_intervals(intervals, n_points)
    c5 = evaluate_c5_fft(h)
    
    if c5 < best_c5:
        best_c5 = c5
        best_sol = z_start.copy()
    
    # Phase 3: Local refinement after DE
    if time.time() < de_end_time + budget_s * 0.15:
        refined_theta, refined_c5 = local_refinement_initial(best_sol)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_sol = refined_theta

    # Convert final theta to binary grid
    intervals_final = decode_interval_params(best_sol, m)
    h_final = build_grid_from_intervals(intervals_final, n_points)
    
    # Verifying with original np.correlate
    c5_final = evaluate_c5_fft(h_final)
    h_eval = h_final.tolist()
    
    # Phase 4: Fallback if needed
    if time.time() < start_time + budget_s:
        if c5_final >= 0.38092:
            h_fallback, c5_fallback = fallback_random_search(c5_final)
            if h_fallback is not None:
                h_eval = h_fallback.tolist()
                c5_final = c5_fallback

    return (h_eval, float(c5_final), n_points)
