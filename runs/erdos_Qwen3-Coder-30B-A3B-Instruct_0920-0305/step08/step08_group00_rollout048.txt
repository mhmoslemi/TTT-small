import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Minimize, Problem, Variable, Binary, sum as cp_sum, maximum as cp_max

def compute_corr_fft(h):
    """Compute correlation using FFT (same as before)"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    lb = 0
    ub = 1
    
    h_vars = Variable(n_points, boolean=True)
    
    # Add the sum constraint
    constraints = [cp_sum(h_vars) == TARGET_SUM]

    shifts = list(range(-n_points + 1, n_points))
    # Keep unique shifts (should be 2*n_points - 1)
    assert len(shifts) == 2 * n_points - 1
    
    # Define auxiliary variables for each shift
    # w_{i,k} = h_i * (1 - h_{i+k})
    # Linearization: w_{i,k} <= h_i
    #              w_{i,k} <= 1 - h_{i+k}
    #              w_{i,k} >= h_i + (1 - h_{i+k}) - 1 = h_i - h_{i+k}
    # But we'll compute the entire constraint directly:
    # w_{i,k} = h_i * y_{i,k} where y_{i,k} = 1-h_{i+k}
    # => w_{i,k} = h_i * (1-h_{i+k}) = h_i - h_i*h_{i+k}
    
    # More direct model:
    # For each shift k, let m_k be the mismatch count.
    # We use auxiliary binary variables y_{i,k} = h_i * h_{i+k}
    # Then w_{i,k} = h_i - y_{i,k} (if dealing with different indices)
    max_c = Variable()
    constraints.append(max_c >= 0)
    
    # Since we require max of m_k, we need constraints for all shifts
    # But we compute all misalignments and add max constraint later.
    # Vectors for k = 0 to 2*n_points-2 (or equivalently k in -n+1 to n-1)
    # The constraint we add is:
    # For k in neg range to pos range:  sum h[i]*(1-h[i+k]) <= C
    # One tricky part is that shifting is out-of-bounds for some indices.
    
    # Instead, we will enforce the bound by bounding on a
    # correctly indexed linear combination
    
    # Instead, model this with a more direct approach:
    # Fix C as a variable, add constraints that sum(h[i]* (1 - h[i+k])) <= C for all k
    # Maximize C, but using constraints that correspond to L2 norm bug prevents this
    # Better:
    # Introduce auxiliary variables for all overlaps
    # Maximize a dummy variable and add constraint
    # Use a totally new MILP formulation at the top
    
    # Let's simplify the approach by using a regular Python + cvxpy MILP model to find the best with integer programming
    
    
    # Constraint integrity:
    # Sort of redundant but still useful:
    for k in range(-n_points + 1, n_points):
        l = k + n_points - 1  # index into the correlation output
        # Due to sliding window, we do a list of expressions
        # Let's treat calculations for each shift
        # More safe approach:
        # For shift k, build the valid set of (i, i+k) pairs
        corr_expr = 0
        if k >= 0:
            # shift right, so h_{i+k} is defined for i in [0, n_points-k-1]
            for i in range(n_points - k):
                corr_expr += h_vars[i] * (1 - h_vars[i + k])
        else:
            # shift left, h_{i+k} is defined for i in [-k, n_points-1]
            for i in range(-k, n_points):
                corr_expr += h_vars[i] * (1 - h_vars[i + k])
        
        # To model max_over_shifts: add constraint max_{k} corr_expr_k <= max_c
        # So add: corr_expr_k <= max_c  for all k
        # Alternatively, we just form one big linear problem
        # We manually enforce: sum(h[i] * (1 - h[i+k]) ) <= max_c for all k

        # MODEL INTRODUCED: Setup for max_k
        constraints.append(corr_expr <= max_c)

        idxs = np.argsort(-h_relaxed)
        h_round = np.zeros(n_points)

            if best_h[i1] < best_h[i2]:  # swapping 0 to 1 and 1 to 0

                i_one = i2
            else:
                i_zero = i2
                i_one = i1
            if best_h[i_one] == 1 and best_h[i_zero] == 0:
                h_trial = best_h.copy()

                h_trial[i_one] = 0

                sum_ok = np.sum(h_trial)
                if sum_ok > TARGET_SUM:
                    idxs = np.where(h_trial == 1)[0]
                    for j in range(int(sum_ok - TARGET_SUM)):
                        if j < len(idxs):
                            h_trial[idxs[j]] = 0
                elif sum_ok < TARGET_SUM:
                    idxs = np.where(h_trial == 0)[0]
                    for j in range(int(TARGET_SUM - sum_ok)):
                        if j < len(idxs):
                            h_trial[idxs[j]] = 1
                c5_trial = c5_from_corr(compute_corr_fft(h_trial), n_points)
                if c5_trial < best_c5:
                    best_h = h_trial
                    best_c5 = c5_trial
        # Final return values
        return (best_h.tolist(), float(best_c5), n_points)
    else:

        # Create a good starting guess
        h_initial = np.zeros(n_points)
        # Simple region-based alternating pattern
        # Divide into segments and toggle ones and zeros
        segment_size = n_points // 4
        ones_positions = []
        for i in range(4):
            start = i * segment_size
            end = start + segment_size
            if i % 2 == 0:
                # Make it '1'
                for j in range(start, end):
                    if len(ones_positions) < TARGET_SUM:
                        ones_positions.append(j)
        for i in ones_positions:
            if i < n_points:
                h_initial[i] = 1
        # Local refinement
        h_current = h_initial.copy()
        best_c5 = evaluate_c5(h_current)
        remaining_time = budget_s - (time.time() - start_time)
        # For polishing, we use a fast method
        while time.time() - start_time < budget_s * 0.95:
            # Small random perturbations
            for i in range(10):
                i1 = np.random.randint(0, n_points)
                i2 = np.random.randint(0, n_points)
                # Swap 1-0 only if it increases sum-of-ones match
                if h_current[i1] == 1 and h_current[i2] == 0:
                    h_new = h_current.copy()
                    h_new[i1] = 0
                    h_new[i2] = 1
                    c5_new = evaluate_c5(h_new)
                    if c5_new < best_c5:
                        h_current = h_new
                        best_c5 = c5_new
        return (h_current.tolist(), float(best_c5), n_points)
