```python
import numpy as np
from scipy.signal import correlate
import time

# Global constants
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def random_binary_solution():
    """Generate random binary solution with exact target sum"""
    h = np.zeros(n_points, dtype=float)
    ones_indices = np.random.choice(n_points, size=int(TARGET_SUM), replace=False)
    h[ones_indices] = 1.0
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = None
    best_c5 = float('inf')
    
    # Simulated Annealing parameters
    initial_temp = 0.015
    final_temp = 0.0001
    max_iter = 200000
    time_per_iter = 0.00005  # Estimate of time per iteration
    
    # Loop with time budget tracking
    iter_count = 0
    while time.time() - start_time < budget_s * 0.95:
        # Generate random binary solution
        h = random_binary_solution()
        current_c5 = evaluate_c5(h)
        
        # Simulated Annealing for this restart
        temp = initial_temp
        iter_since_improvement = 0
        max_iter_per_restart = min(max_iter, int((budget_s * 0.95 - (time.time() - start_time)) / time_per_iter))
        
        for i in range(max_iter_per_restart):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Find current ones and zeros
            ones_indices = np.where(h == 1)[0]
            zeros_indices = np.where(h == 0)[0]
            
            # Sample positions to swap
            if len(ones_indices) > 0 and len(zeros_indices) > 0:
                i_swap = np.random.choice(ones_indices)
                j_swap = np.random.choice(zeros_indices)
                
                # Perform swap
                h_new = h.copy()
                h_new[i_swap] = 0
                h_new[j_swap] = 1
                
                # Calculate new correlation
                new_c5 = evaluate_c5(h_new)
                
                # Accept or reject
                if new_c5 < current_c5:
                    h = h_new
                    current_c5 = new_c5
                    iter_since_improvement = 0
                    if new_c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = new_c5
                else:
                    iter_since_improvement += 1
                    # Accept with probability based on temperature
                    if np.random.random() < np.exp(-(new_c5 - current_c5) / temp):
                        h = h_new
                        current_c5 = new_c5
                        iter_since_improvement = 0
                        if new_c5 < best_c5:
                            best_h = h.copy()
                            best_c5 = new_c5
                            
            # Decrease temperature
            temp = initial_temp * (final_temp / initial_temp) ** (i / max_iter_per_restart)
            
            iter_count += 1
            if iter_since_improvement > 5000:
                break
    
    # If no good solution found, use the random binary initialization
    if best_h is None:
        h = random_binary_solution()
        best_c5 = evaluate_c5(h)
        best_h = h
    
    return (best_h.tolist(), float(best_c5), n_points)
```
