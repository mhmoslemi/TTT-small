import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_improvement(h0, max_iter=1000):
    """Local search improvement"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    for i in range(max_iter):
        # Try small random changes to a few points
        num_changes = min(10, n_points // 3)
        indices = np.random.choice(n_points, size=num_changes, replace=False)
        
        for idx in indices:
            old_val = h[idx]
            # Try several small changes
            changes = [-0.02, -0.01, 0.01, 0.02]
            np.random.shuffle(changes)
            
            for delta in changes:
                h_new = h.copy()
                h_new[idx] += delta
                if h_new[idx] < 0 or h_new[idx] > 1:
                    continue
                    
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                new_c5 = evaluate_c5(h_new)
                
                if new_c5 < best_c5:
                    h = h_new.copy()
                    best_c5 = new_c5
                    best_h = h.copy()
                    break
            else:
                continue
            break
        else:
            # If no improvement, try a different approach
            if np.random.random() < 0.3:
                # Random noise perturbation
                h_noise = h + np.random.normal(0, 0.005, n_points)
                h_noise = project_to_sum_box(h_noise, TARGET_SUM)

                    best_h = h.copy()

def pattern_based_initialization():
    """Create an initialization based on analytical patterns that minimize overlap"""
    # Initial guess: create a function with 2 peaks, one near 0.5 and other near 1.5
    h = np.zeros(n_points)

    peak1_pos = int(0.4 * n_points)
    peak2_pos = int(1.6 * n_points)
    
    # Peak with width 
    width = int(0.1 * n_points)
    
    # Fill in peak 1
    start1 = max(0, peak1_pos - width)
    end1 = min(n_points, peak1_pos + width)
    h[start1:end1] = np.clip(np.linspace(0, 1, end1-start1), 0, 1)
    h[peak1_pos] = 1.0
    h[peak1_pos-1] = h[peak1_pos+1] = 0.5
    
    # Fill in peak 2
    start2 = max(0, peak2_pos - width)
    end2 = min(n_points, peak2_pos + width)
    h[start2:end2] = np.clip(np.linspace(0, 1, end2-start2), 0, 1)
    h[peak2_pos] = 1.0
    h[peak2_pos-1] = h[peak2_pos+1] = 0.5
    
    # Normalize to correct sum  
    h = project_to_sum_box(h, TARGET_SUM)
    return h

            h_perturbed = best_h + np.random.normal(0, 0.01, n_points)

                best_h = h_fine.copy()
                best_c5 = c5_fine

    final_time = time.time()

        if np.random.random() < 0.5:

        best_h = adaptive_sampling_initialization()
