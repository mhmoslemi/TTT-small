import numpy as np
from scipy.signal import correlate

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    result = np.clip(x + (a + b) / 2, 0, 1)
    return result

def get_correlation_incremental(h, b, c, i, j, delta):
    # Get the current correlations
    c_new = c.copy()
    
    # Single index changes affecting correlation matrix
    # For changes at h[i] -> h[i] - delta, h[j] -> h[j] + delta
    
    # Part due to changing h[i] (b[i+s] contributes)
    if 0 <= i < len(h):
        start_idx = max(0, (n_points - 1) - i)
        end_idx = min((n_points - 1) + (n_points - 1 - i), len(c_new))
        if start_idx < end_idx:
            c_new[start_idx:end_idx] += delta * b[i:(i + (end_idx - start_idx))]
    
    # Part due to changing b[i] (h[i-s] contributes, but we use reversed h)
    h_rev = h[::-1]
    if 0 <= i < len(h):
        start_idx = max(0, i)
        end_idx = min(len(c_new), i + len(h_rev))
        if start_idx < end_idx:
            c_new[start_idx:end_idx] -= delta * h_rev[(start_idx - i):(end_idx - i)]
    
    # Similarly for j (since b[j] changes)
    # b[j] -> b[j] + delta
    if 0 <= j < len(h):
        start_idx = max(0, (n_points - 1) - j)
        end_idx = min((n_points - 1) + (n_points - 1 - j), len(c_new))
        if start_idx < end_idx:
            c_new[start_idx:end_idx] -= delta * b[j:(j + (end_idx - start_idx))]
    
    # h[j] -> h[j] + delta
    if 0 <= j < len(h):
        start_idx = max(0, j)
        end_idx = min(len(c_new), j + len(h_rev))
        if start_idx < end_idx:
            c_new[start_idx:end_idx] += delta * h_rev[(start_idx - j):(end_idx - j)]
    
    # Add the quadratic term approximated for small delta
    # This is essentially the negative cross term between indexes i and j
    # It comes up at shifts +/- (j-i)
    shift_ij = j - i + (n_points - 1)
    if 0 <= shift_ij < len(c_new):
        c_new[shift_ij] += delta**2
    
    shift_ji = i - j + (n_points - 1)
    if 0 <= shift_ji < len(c_new):
        c_new[shift_ji] += delta**2
    
    return c_new

def evaluate_c5(h):
    """Evaluate C5 using FFT-based correlation for full array"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return np.max(corr) * dx

def get_next_delta(current_delta, recent_acceptance):
    """Adaptively adjust delta based on acceptance rate"""
    if recent_acceptance > DELTA_ADJUST_RATIO:
        return min(current_delta * 1.1, 0.05)
    else:
        return max(current_delta * 0.9, DELTA_MIN)

    h = project_to_sum_box(h_init, n_points / 2)
    
    b = 1 - h
    c = correlate(h, b, mode='full')
    Cmax = np.max(c)
    current_obj = Cmax * dx
    
    best_h = h.copy()
    best_obj = current_obj
    
    # Parameters
    delta = DELTA_INIT
    temperature = T0
    last_improvement = 0
    neg_acceptance = 0
    pos_acceptance = 0
    acceptance_count = 0
    
    iteration = 0
    np.random.seed(seed)
    
    while time.time() - start_time < budget_s and iteration < MAX_ITER:
        iteration += 1
        
        # Get indices
        i, j = np.random.randint(0, n_points, 2)
        if i == j: continue
        
        # Compute maximal allowed delta
        delta_max = min(h[i], 1 - h[j])
        delta_to_use = min(delta, delta_max * 0.9)
        
        # Perform move: transfer delta from i to j
        h[i] -= delta_to_use
        h[j] += delta_to_use
        
        # Compute new correlation data
        c = get_correlation_incremental(h, b, c, i, j, delta_to_use)
        Cmax_new = np.max(c)
        obj_new = Cmax_new * dx
        
        # Determine acceptance probability
        if obj_new < current_obj:
            accept = True
        else:
            prob = np.exp(-(obj_new - current_obj) / temperature)
            accept = np.random.rand() < prob
        
        # Apply or reject the change
        if accept:
            current_obj = obj_new
            b = 1 - h
            if obj_new < best_obj:
                best_h = h.copy()
                best_obj = obj_new
            last_improvement = iteration
            
            # Track acceptance rates in window
            pos_acceptance += 1
        else:
            # back out the change
            h[i] += delta_to_use
            h[j] -= delta_to_use
            
            neg_acceptance += 1
            
        # Update counters
        acceptance_count += 1
    
        # Periodic corrections and bookkeeping
        if iteration % K_EVAL == 0:
            # Recompute using full correlation for safety
            c_full = correlate(h, 1 - h, mode='full')
            max_correl = np.max(c_full)
            c = c_full.copy()
            if abs(max_correl * dx - current_obj) > 1e-12:
                current_obj = max_correl * dx
        
        # Step size adjustment
        if iteration % K_STEP == 0:
            recent_ratio = float(pos_acceptance) / (pos_acceptance + neg_acceptance)
            delta = get_next_delta(delta, recent_ratio)
            pos_acceptance = 0
            neg_acceptance = 0
        
        # Cooling temperature
        if iteration % K_TEMP == 0:
            temperature *= ALPHA_TEMP
            
            # Refresher if no improvement in long time
            if iteration - last_improvement > 2000:
                b = 1 - h
                c = correlate(h, b, mode='full')
                Cmax = np.max(c)
                current_obj = Cmax * dx
                last_improvement = iteration
    
    return (best_h.tolist(), float(best_obj), n_points)
