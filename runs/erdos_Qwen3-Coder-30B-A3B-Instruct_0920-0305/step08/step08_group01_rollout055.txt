import numpy as np

import time
import cvxpy as cp

def compute_shift_matrices(n):
    """Precompute shift matrices for optimization constraints."""
    R = {}
    for s in range(1, n//2 + 1):
        R[s] = np.zeros((n, n))
        for i in range(n - s):
            R[s][i, i + s] = 1
    return R

        constraints.append(mu[i] <= 1)
