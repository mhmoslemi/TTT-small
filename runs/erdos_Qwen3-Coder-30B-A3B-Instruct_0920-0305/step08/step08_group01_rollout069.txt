import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def construct_patterned_solution():
    """Construct a solution using a patterned approach with specific shapes"""
    h = np.zeros(n_points)
    
    # Create piecewise constant function with specific pattern
    # Start with some structure that helps reduce overlap
    segments = [
        (0.0, 0.2, 0.8),   # High region
        (0.2, 0.4, 0.2),   # Low region  
        (0.4, 0.6, 0.8),   # High region
        (0.6, 0.8, 0.2),   # Low region
        (0.8, 1.0, 0.8),   # High region
    ]
    
    for start, end, val in segments:
        start_idx = int(start * n_points)
        end_idx = int(end * n_points)
        h[start_idx:end_idx] = val
    
    # Ensure sum constraint
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def hill_climb_with_mutation(h, max_iter=500):
    """Improved hill climbing with mutation"""
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    for iteration in range(max_iter):
        # Create a neighbor by small random changes  
        neighbor = current_h.copy()
        # Pick a few random indices to mutate
        indices = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
        
        for idx in indices:
            # Small random perturbation
            delta = np.random.uniform(-0.05, 0.05)
            neighbor[idx] += delta
            # Keep within bounds
            neighbor[idx] = np.clip(neighbor[idx], 0, 1)
        
        # Project onto feasible set
        neighbor = project_to_sum_box(neighbor, TARGET_SUM)
        
        # Evaluate neighbor
        neighbor_c5 = evaluate_c5(neighbor)
        
        if neighbor_c5 < current_c5:  # Accept better solution
            current_h = neighbor
            current_c5 = neighbor_c5
            
            if neighbor_c5 < best_c5:
                best_h = neighbor
                best_c5 = neighbor_c5
        elif np.random.random() < 0.05:  # Accept worse solution with small probability
            current_h = neighbor
            current_c5 = neighbor_c5
            
    return best_h, best_c5

def refine_solution_advanced(h, max_iter=200):
    """Advanced refinement using adaptive local search"""
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    
    for iteration in range(max_iter):
        # Try multiple small changes to different indices
        modified = False
        for _ in range(5):  # Try several modifications
            if np.random.random() < 0.7:  # Most of the time do a move
                # Pick random index and adjust it
                idx = np.random.randint(0, n_points)
                # Try several small values
                deltas = [-0.02, -0.01, 0.01, 0.02]
                np.random.shuffle(deltas)
                
                for delta in deltas:
                    new_h = current_h.copy()
                    new_h[idx] += delta
                    new_h[idx] = np.clip(new_h[idx], 0, 1)
                    
                    new_h = project_to_sum_box(new_h, TARGET_SUM)
                    new_c5 = evaluate_c5(new_h)
                    
                    if new_c5 < current_c5:
                        current_h = new_h
                        current_c5 = new_c5
                        modified = True
                        break
        
        if not modified:
            # If no change, make one random full perturbation
            if np.random.random() < 0.2:
                idx = np.random.randint(0, n_points)
                current_h[idx] = np.random.random()
                current_h = project_to_sum_box(current_h, TARGET_SUM)
                current_c5 = evaluate_c5(current_h)
    
    return current_h, current_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Phase 1: Construct patterned solutions
    for i in range(10):
        if time.time() - start_time > budget_s * 0.4:
            break
        h = construct_patterned_solution()
        h, c5 = hill_climb_with_mutation(h, max_iter=200)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 2: Refine best solution
    if best_h is not None:
        best_h, best_c5 = refine_solution_advanced(best_h, max_iter=250)
    
    # Phase 3: Additional random starts with local search
    for i in range(20):
        if time.time() - start_time > budget_s * 0.8:
            break
        
        # Random initial solution
        h = np.random.uniform(0, 1, n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Local optimization
        h, c5 = hill_climb_with_mutation(h, max_iter=150)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 4: Final adjustment
    if best_h is not None:
        best_h, best_c5 = refine_solution_advanced(best_h, max_iter=100)
    
    return (best_h.tolist(), float(best_c5), n_points)
