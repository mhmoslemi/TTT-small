def evaluate_c5(h_array):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h_array[::-1]
    b = 1 - h_array
    L = 2 * len(h_array) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def piecewise_to_array(ℓ, a):
    """Convert piecewise representation to full array (upsample)"""
    m = len(ℓ)
    h = np.zeros(n_points)
    start = 0
    for i in range(m):
        end = start + int(np.round(ℓ[i] / 2.0 * n_points))
        h[start:end] = a[i]
        start = end
    return h

def parameterize_from_piecewise(ℓ, a, k):
    """Evaluate overlapping integral for shift k using exact geometric computation"""
    # Compute shift k contribution
    # overlapping intervals: for each pair of intervals I_p and I_q shifted by k
    # compute if there's overlap, and by how much

    total_len = 0.0
    total_prod = 0.0

    # cumulative positions
    s = [0.0]
    for i in range(len(ℓ)):
        s.append(s[-1] + ℓ[i])
    
    for p in range(len(ℓ)):
        for q in range(len(ℓ)):
            # interval p: [s[p], s[p+1])
            # interval q with shift k: [s[q] + k, s[q+1] + k]
            left = max(s[p], s[q] + k)
            right = min(s[p+1], s[q+1] + k)
            if right > left:
                overlap_len = right - left
                total_len += overlap_len
                total_prod += a[p] * a[q] * overlap_len
    return total_len - total_prod  # This is ∫ h(x) (1 - h(x+k)) dx
