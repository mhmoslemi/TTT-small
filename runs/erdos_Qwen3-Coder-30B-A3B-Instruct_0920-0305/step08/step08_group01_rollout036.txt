import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

# Global variables
n_points = 180
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def decode_params(z, m):
    """Convert parameter vector z to interval positions"""
    # Rescale z to [eps, 1-eps] for lengths, [0,1] for gaps
    eps = 1e-4
    l = eps + (1 - 2*eps) * z[:m]
    l = l / np.sum(l)  # Normalise to unit mass
    g = z[m:]  # Gaps
    
    # Place intervals sequentially
    s = np.zeros(m)
    e = np.zeros(m)
    s[0] = 0.0
    e[0] = s[0] + l[0]
    for i in range(1, m):
        s[i] = e[i-1] + g[i-1]
        e[i] = s[i] + l[i]
    
    # If final interval extends beyond 2, scale gaps down
    total_length = e[-1]
    if total_length > 2:
        scale_factor = 2.0 / total_length
        l = l * scale_factor
        g = g * scale_factor
        s = np.zeros(m)
        e = np.zeros(m)
        s[0] = 0
        e[0] = l[0]
        for i in range(1, m):
            s[i] = e[i-1] + g[i-1]
            e[i] = s[i] + l[i]
    return s, e

def build_grid_from_intervals(s, e, n):
    """Build binary grid h from interval positions"""
    h = np.zeros(n)
    grid_centers = np.arange(0.5, n) * dx
    
    for i in range(len(s)):
        mask = (grid_centers >= s[i]) & (grid_centers < e[i])
        h[mask] = 1
    
    return h

def compute_c5_fft(h):
    """Fast C5 computation using FFT correlation"""
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def objective(z, m):
    """Objective function for DE optimization"""
    s, e = decode_params(z, m)
    h = build_grid_from_intervals(s, e, n_points)
    c5 = compute_c5_fft(h)
    return c5

def local_refinement(s, e, max_iter=200):
    """Refine interval edges to improve worst-case overlap"""
    best_s, best_e = s.copy(), e.copy()
    h = build_grid_from_intervals(best_s, best_e, n_points)
    best_c5 = compute_c5_fft(h)
    
    iter_count = 0
    improved = True
    while improved and iter_count < max_iter:
        improved = False
        iter_count += 1
        for i in range(len(s)):
            # Try moving left edge
            for delta in [-dx/2, dx/2]:
                s_new = best_s.copy()
                s_new[i] += delta
                e_new = best_e.copy()
                e_new[i] += delta
                # Enforce constraints
                if s_new[i] < 0 or e_new[i] > 2 or s_new[i] >= e_new[i]:
                    continue
                # Ensure ordering with neighbors
                if i > 0 and s_new[i] < best_e[i-1]:
                    continue
                if i < len(s)-1 and s_new[i] >= e_new[i]:
                    continue
                h_new = build_grid_from_intervals(s_new, e_new, n_points)
                c5_new = compute_c5_fft(h_new)
                if c5_new < best_c5:
                    best_s, best_e = s_new.copy(), e_new.copy()
                    best_c5 = c5_new
                    improved = True
                    break
            if improved:
                break
            # Try moving right edge
            for delta in [-dx/2, dx/2]:
                s_new = best_s.copy()
                e_new = best_e.copy()
                e_new[i] += delta
                if e_new[i] <= s_new[i] or e_new[i] > 2:
                    continue
                # Enforce ordering with neighbors
                if i < len(s)-1 and e_new[i] > s_new[i+1]:
                    continue
                h_new = build_grid_from_intervals(s_new, e_new, n_points)
                c5_new = compute_c5_fft(h_new)
                if c5_new < best_c5:
                    best_s, best_e = s_new.copy(), e_new.copy()
                    best_c5 = c5_new
                    improved = True
                    break
            if improved:
                break
    return best_s, best_e, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial random sampling to start DE population
    m = 5  # Number of intervals
    initial_population = []
    best_c5 = float('inf')
    best_z = None
    
    # Generate some initial samples
    for _ in range(20):
        z = np.random.rand(2 * m)
        c5 = objective(z, m)
        if c5 < best_c5:
            best_c5 = c5
            best_z = z.copy()
        initial_population.append(z)
    
    # Differential evolution
    bounds = [(0, 1) for _ in range(2 * m)]
    de_budget = 0.6 * budget_s
    end_time = start_time + de_budget
    best_de_c5 = float('inf')
    best_de_z = None
    
    def de_callback(x, convergence):
        nonlocal best_de_c5, best_de_z
        c5 = objective(x, m)
        if c5 < best_de_c5:
            best_de_c5 = c5
            best_de_z = x.copy()
        
    try:
        result = differential_evolution(
            objective,
            bounds,
            args=(m,),
            maxiter=150,
            popsize=30,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            callback=de_callback,
            disp=False
        )
        if result.success and result.fun < best_de_c5:
            best_de_c5 = result.fun
            best_de_z = result.x.copy()
    except:
        pass
    
    # Use best found solution from DE or initial set
    final_z = best_de_z if best_de_z is not None else best_z
    if final_z is None:
        # use fallback if not found
        final_z = np.random.rand(2*m)
    
    # Refinement phase
    ref_budget = 0.15 * budget_s
    ref_start = time.time()
    s, e = decode_params(final_z, m)
    
    # Local refinement
    s_refined, e_refined, refined_c5 = local_refinement(s, e, max_iter=100)
    
    # Final evaluation with exact computation
    h_final = build_grid_from_intervals(s_refined, e_refined, n_points)
    final_c5 = compute_c5_fft(h_final)
    
    # Fallback if no improvement
    if final_c5 >= best_c5:
        # Random sampling fallback
        best_random_c5 = float('inf')
        best_random_h = None
        for _ in range(1000):
            rand_z = np.random.rand(2*m)
            s_rand, e_rand = decode_params(rand_z, m)
            h_rand = build_grid_from_intervals(s_rand, e_rand, n_points)
            c5_rand = compute_c5_fft(h_rand)
            if c5_rand < best_random_c5:
                best_random_c5 = c5_rand
                best_random_h = h_rand.copy()
        if best_random_h is not None:
            final_c5 = best_random_c5
            h_final = best_random_h
    
    # Final verification with exact correlation
    corr = correlate(h_final, 1-h_final, mode='full')
    final_c5_exact = dx * np.max(corr)
    
    return (h_final.tolist(), float(final_c5_exact), n_points)
