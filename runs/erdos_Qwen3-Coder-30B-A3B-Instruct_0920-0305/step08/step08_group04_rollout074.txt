import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import time

# Global variables
TARGET_SUM = None
n_points = 300
dx = 2.0 / n_points
n_coarse = 150

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling projection  
    # Sort values
    x_sorted = np.sort(x)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(x_sorted)
    # Find the cut point 
    idx = np.argmax(x_sorted - (target_sum - cumsum + x_sorted) / np.arange(1, len(x_sorted) + 1) <= 0) 
    mu = (target_sum - cumsum[idx]) / (len(x) - idx)
    result = np.clip(x - mu, 0, 1)
    return result

def rasterise_and_project(theta):
    """Map 16-dimensional vector to n-point wavelet-like h"""
    K = 8  # 8 blocks
    widths = np.exp(theta[:K])
    widths = 2 * widths / widths.sum()  # Normalized widths
    heights = 1 / (1 + np.exp(-theta[K:]))
    
    # Create the piecewise constant function
    h = np.zeros(n_points)
    cum_width = 0
    for i in range(K):
        w = widths[i]
        h_start = int(cum_width / dx)
        h_end = int((cum_width + w) / dx)
        h[h_start:h_end] = heights[i]
        cum_width += w
    
    # Project to sum & box
    h_projected = project_to_sum_box(h, n_points / 2)
    return h_projected

def obj(θ):
    h = rasterise_and_project(θ)
    c5 = evaluate_c5(h)
    return c5

            if time.time() - start_time > budget_s - 20:
