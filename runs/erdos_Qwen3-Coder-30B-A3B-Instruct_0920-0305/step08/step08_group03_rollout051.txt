import numpy as np

import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    # That is shifts s from -j to (n-1-j)

        # In code the total is over shifts where i+s valid (and i-s valid for reverse),
        # Let's just compute on individual positions and updates
    
    # Detailed induced changes
    # shift from -i to n-1-i, add delta * b[i+s] to c[s + (n-1)]
    # shift from i-(n-1) to i, subtract delta * a[i-s] to c[s + (n-1)]
    
    # Handle i contribution (start from -i to n-1-i):
    for s in range(-i, n-1-i+1):
        c_shift_idx = s + n - 1
        if 0 <= c_shift_idx < 2*n - 1:
            if 0 <= i + s < n:  # b[i+s] valid
                c[c_shift_idx] += delta * b[i+s]
    
    # Handle i contribution from a (subtract delta * a[i-s]):
    for s in range(i-(n-1), i+1):
        c_shift_idx = s + n - 1
        if 0 <= c_shift_idx < 2*n - 1:
            if 0 <= i-s < n:  # a[i-s] valid
                c[c_shift_idx] -= delta * h[i-s]
    
    # Handle j contribution (reverse logic, but contrib to full range):
    # shift from -j to n-1-j, add (-delta) * b[j+s] to c[s + (n-1)]
    for s in range(-j, n-1-j+1):
        c_shift_idx = s + n - 1
        if 0 <= c_shift_idx < 2*n - 1:
            if 0 <= j + s < n:  # b[j+s] valid
                c[c_shift_idx] += (-delta) * b[j+s]
    
    # Handle j contribution from a (subtract (-delta) * a[j-s] to c[s + (n-1)]):
    # actual delta already dealt with for now, just with opposite sign
    for s in range(j-(n-1), j+1):
        c_shift_idx = s + n - 1
        if 0 <= c_shift_idx < 2*n - 1:
            if 0 <= j-s < n:  # a[j-s] valid
                c[c_shift_idx] -= (-delta) * h[j-s]
                
    # Add cross terms from change of both: delta * delta = delta^2
    # For i to j, delta terms are (i to j = delta_i = -delta, delta_j = +delta)
    # ij contributes when final shift = j - i
    c_ij = j - i + n - 1  # shift s = j - i corresponds to index s + n - 1
    if 0 <= c_ij < 2*n - 1:
        c[c_ij] += delta * delta  # The delta * -delta = -delta^2 is canceling, but since 
        # delta_i = -delta, delta_j = +delta, therefore delta_i * delta_j = -delta^2.
        # The second change in i makes this h[i] change from -delta to 0, contributing nothing.
        # But there is other term: both contributions negated. Let's simplify:
        
    # Correct approach rather than generic:
    # Note that diff from a_i changes by -delta, and b_i changes by +delta
    # In full convolution, a_i * b_j yields contribution to some correlation: say shift s = j - i.
    # When changing a_i by -delta, b_j by +delta ("delta + delta" for a_i*b_j and vice versa)
    # So contrib delta * (b[j] - a[i]) where i was changed to i-s, j = i + s.
    # But in our case we only move delta between two indices.
    
    # Keep as safe assumption - the right value for full change
    # We handled cross changes as part of larger computation above.
    
    return c
