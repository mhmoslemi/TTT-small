import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import GLPK, CBC, SCIP

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def compute_milp_solution(n_points, solver):
    """Solve the exact MILP for optimal h"""
    # Variables
    h = cp.Variable(n_points, boolean=True)
    C = cp.Variable()  # upper bound on max correlation

    # Constraints:
    # 1. Sum constraint
    constraints = [cp.sum(h) == n_points / 2]
    # 2. C is an upper bound on all correlations
    # For each shift k = -(n-1) to n-1
    # We compute the correlation sum via auxiliary variables
    for k in range(-n_points + 1, n_points):
        # This is tricky due to shifting
        # Let's compute it using enumeration / auxiliary constraints
        corr_sum = 0
        # When k >= 0, indices go from 0 to n-k-1 for h[i], and k to n-1 for h[i+k]
        # When k < 0, indices go from -k to n-1 for h[i], and 0 to n+k-1 for h[i+k]
        if k >= 0:
            # sum h[i] (1 - h[i+k]) for i = 0..n-k-1
            # Iterator
            offset = 0
        else:  # k < 0
            # sum h[i] (1 - h[i+k]) for i = -k..n-1
            offset = -k

        # For all indices where both h[i] and h[i+k] exist:
        for i in range(n_points):
            i_plus_k = i + k
            if 0 <= i_plus_k < n_points:
                # Use auxiliary variable y[i] = h[i] * h[i+k]
                y = cp.Variable(boolean=True)
                constraints.extend([
                    y <= h[i],
                    y <= h[i_plus_k],
                    y >= h[i] + h[i_plus_k] - 1
                ])
                corr_sum += h[i] - y
            else:
                corr_sum += h[i] * (1 - 0)  # assumes h[i] is not affected by out of range h[i+k]

        # Optional: better formulation of correlation
        # This is more precise, but tricky in CVXPy
        # Instead, use different logic for clarity:
        
        # Below is a direct approximated version:
        # Actually, let's compute in a loop;
        # However, let's simplify via explicit split:
        corr_sum = 0
        if k >= 0:
            indices = range(n_points - k)
            for i in indices:
                # Compute h[i] * (1 - h[i + k])
                corr_sum += h[i] * (1 - h[i + k])
        else:
            indices = range(-k, n_points)
            for i in indices:
                corr_sum += h[i] * (1 - h[i + k])

        # Add bound constraint
        constraints.append(corr_sum <= C)

    # Objective
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=solver, verbose=False, max_iters=5000)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            return h.value
    except Exception:
        pass
    return None

def round_and_refine(h_val, n_points):
    """Use rounding and local search on the relaxed solution."""
    best_h = h_val.copy()
    
    # Rounding - Keep top n/2 as 1
    indices_sorted = np.argsort(h_val)[::-1]
    h_rounded = np.zeros(n_points)
    for i in range(n_points // 2):
        h_rounded[indices_sorted[i]] = 1
    # Reassign to maintain sum
    h_rounded = np.clip(h_rounded, 0, 1)
    
    # Local hill climbing
    dx = 2.0 / n_points
    best_c5 = evaluate_c5(h_rounded)
    iters = 0
    while iters < 100:
        improved = False
        for i in range(n_points):
            for j in range(i+1, n_points):
                if h_rounded[i] == h_rounded[j]:
                    continue
                h_trial = h_rounded.copy()
                h_trial[i], h_trial[j] = h_trial[j], h_trial[i]
                c5_trial = evaluate_c5(h_trial)
                if c5_trial < best_c5:
                    best_h = h_trial
                    best_c5 = c5_trial
                    improved = True
        if not improved:
            break
        iters += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Use n=120
    n_points = 120
    dx = 2.0 / n_points
    
    # Try exact MILP first
    solver = CBC
    h_sol = compute_milp_solution(n_points, solver)
    best_h = None
    best_c5 = float('inf')
    
    if h_sol is not None:
        h_sol = np.array(h_sol)
        best_c5 = evaluate_c5(h_sol)
        best_h = h_sol.tolist()
    else:
        # Fallback: initialize a clean binary pattern
        h_init = np.ones(n_points)
        h_init[n_points // 2:] = 0
        h_init = np.clip(h_init + np.random.normal(0, 0.01, n_points), 0, 1)
        # Round to binary
        indices_sorted = np.argsort(h_init)[::-1]
        h_init_rounded = np.zeros(n_points)
        for i in range(n_points // 2):
            h_init_rounded[indices_sorted[i]] = 1
        best_h = h_init_rounded.tolist()
        best_c5 = evaluate_c5(best_h)
    
    # Local refinement for better solution
    local_end = start_time + 0.8 * budget_s
    while time.time() < local_end:
        h_trial = list(best_h)
        # pick two random positions
        i, j = np.random.randint(0, n_points, 2)
        h_trial[i], h_trial[j] = h_trial[j], h_trial[i]
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < best_c5:
            best_h = h_trial
            best_c5 = c5_trial
    
    return (best_h, best_c5, n_points)
