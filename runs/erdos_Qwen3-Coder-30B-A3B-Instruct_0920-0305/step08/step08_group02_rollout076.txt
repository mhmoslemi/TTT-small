```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from scipy.fft import fft, ifft

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using waterfilling"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    # Binary search in [-1,1] for the correct lambda
    a, b = -1.0, 1.0
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def greedy_binary_initialization():
    """Create a greedy binary initialization with high minimum autocorrelation"""
    # Place ones with some spacing to maximize minimum autocorrelation
    y = np.zeros(n_points, dtype=int)
    M = n_points // 2  # target sum
    
    # Start placing 1s, attempt to space them well 
    # Initial idea: place ones every ~2 positions while keeping a good distribution
    for i in range(M):
        # A simple spacing heuristic
        pos = (i * 2) % n_points
        y[pos] = 1
        
    # Adjust to get exactly M ones if needed
    current_sum = np.sum(y)
    if current_sum < M:
        # Fill in zeros with ones
        zeros = np.where(y == 0)[0]
        fill_indices = zeros[:M - current_sum]
        y[fill_indices] = 1
    elif current_sum > M:
        # Remove ones from the end to get to M
        ones = np.where(y == 1)[0]
        remove_indices = ones[current_sum - M:]
        y[remove_indices] = 0
    
    return y.astype(float)

def solve_binary_milp(strategy='MILP'):
    """Solve the binary MILP for the Erdos overlap problem"""
    if strategy == 'MILP':
        # Setup for CVXPY + CBC
        try:
            # Create variables
            y = cp.Variable(n_points, boolean=True)
            t = cp.Variable()
            
            # Constraints
            constraints = [
                cp.sum(y) == n_points // 2,  # sum constraint
                t >= 0, 
                t <= n_points // 2  # upper bound on t
            ]
            
            # Add constraints for all shifts k=1 to N-1
            for k in range(1, n_points):
                # Add auxiliary variable w_k,i for y_i * y_{i+k} 
                w = cp.Variable(n_points - k, nonneg=True)  # for i=0 to N-k-1
                for i in range(n_points - k):
                    # Linearization constraints for w_k,i = y_i * y_{i+k}
                    constraints.append(w[i] <= y[i])
                    constraints.append(w[i] <= y[i+k])
                    constraints.append(w[i] >= y[i] + y[i+k] - 1)
                    constraints.append(w[i] >= 0)
                # Add constraint sum(w_k,i) >= t for this k
                constraints.append(cp.sum(w) >= t)
            
            # Objective
            objective = cp.Maximize(t)
            prob = cp.Problem(objective, constraints)
            
            # Try CBC solver first
            try:
                prob.solve(solver=cp.CBC, max_seconds=0.4*600)
            except:
                try:
                    # Try GLPK_MI if CBC fails
                    prob.solve(solver=cp.GLPK_MI, max_seconds=0.4*600)
                except:
                    # Fallback to ECOS_BB
                    try:
                        prob.solve(solver=cp.ECOS_BB, max_seconds=0.4*600)
                    except:
                        # If no solver available, return none
                        return None, None
                        
            return y.value, prob.value
        except:
            return None, None
    else:
        return None, None

def compute_subgradient(h, k_star):
    """Compute subgradient of the worst shift k_star"""
    n = len(h)
    g = np.zeros(n)
    for i in range(n):
        g[i] = 1 - h[(i+k_star) % n] - h[(i-k_star) % n]
    return g

def subgradient_descent(h0, max_time=0.35 * 600):
    """Perform subgradient descent on the continuous problem"""
    start_time = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Initial step size
    alpha_0 = 0.5 * dx
    t = 1  # iteration counter
    
    while time.time() - start_time < max_time:
        # Compute all correlations
        corr = compute_corr_fft(h)
        k_star = np.argmax(corr)
        
        # Compute subgradient
        g = compute_subgradient(h, k_star)
        
        # Take step
        alpha_t = alpha_0 / np.sqrt(t)
        h_new = h - alpha_t * g
        
        # Project
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        c5_new = evaluate_c5(h_new)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Update for next iteration
        h = h_new
        t += 1
        
    return best_h, best_c5

def two_index_swap_move(h, delta):
    """Perform two-index swap move"""
    indices = np.random.choice(n_points, size=2, replace=False)
    i, j = indices[0], indices[1]
    
    # Adjust values
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    
    # Clip
    h_new = np.clip(h_new, 0, 1)
    
    # Project to sum
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    
    return h_new

def stochastic_refinement(h0, max_time=0.25 * 600):
    """Apply stochastic 2-index swap annealing"""
    start_time = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    T_start = 5 * dx
    t0 = time.time()
    
    while time.time() - start_time < max_time:
        T = T_start * (1 - (time.time() - t0) / max_time)
        
        # Make 2-index move
        delta = np.random.uniform(-0.02, 0.02)
        h_new = two_index_swap_move(h, delta)
        
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            best_h = h_new.copy()
        elif T > 0:
            p_accept = np.exp(-(c5_new - best_c5) / T)
            if np.random.random() < p_accept:
                h = h_new.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Phase 1: Binary optimization
    binary_solution, objective_val = solve_binary_milp('MILP')
    
    # If solver fails, use greedy binary initialization
    if binary_solution is None:
        h0 = greedy_binary_initialization()
    else:
        h0 = binary_solution
        
    # Ensure sum constraint exactly
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    # Phase 2: Continuous subgradient descent  
    h1, c5_1 = subgradient_descent(h0, max_time=0.35*budget_s)
    
    # Phase 3: Stochastic refinement
    h_final, c5_final = stochastic_refinement(h1, max_time=0.25*budget_s)
    
    # Return best solution found
    best_h = h_final.copy()
    best_c5 = c5_final
    
    return (best_h.tolist(), float(best_c5), n_points)
```
