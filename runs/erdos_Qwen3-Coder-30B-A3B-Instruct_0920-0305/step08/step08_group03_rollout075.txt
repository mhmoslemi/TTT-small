def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def compute_J(k, L, A):
    """Analytic computation of the overlap integral for shift k"""
    m = len(L)
    # Compute cumulative interval endpoints
    S = [0.0]
    for j in range(m):
        S.append(S[-1] + L[j])
    # Linear part: integral of h(x) from 0 to 2-k
    linear = 0.0
    for j in range(m):
        s1 = S[j]
        s2 = S[j+1]
        if s1 >= 2 - k:
            break
        if s2 <= 2 - k:
            linear += A[j] * (s2 - s1)
        else:
            linear += A[j] * ((2 - k) - s1)
            break
    # Quadratic part: integral of h(x) * h(x+k)
    product = 0.0
    # When intervals I_p and I_q overlap
    for p in range(m):
        for q in range(m):
            s1_p = S[p]
            s2_p = S[p+1]
            s1_q = S[q]
            s2_q = S[q+1]
            # Overlap in shifted coordinate system
            # Left boundary of shifted I_q is at s1_q + k.
            # Right boundary of shifted I_q is s2_q + k.
            left = max(s1_p, s1_q + k)
            right = min(s2_p, s2_q + k)
            if left < right:
                w = right - left
                product += A[p] * A[q] * w
    return linear - product

def build_critical_shifts(L):
    """Generate all critical shift values based on interval endpoints"""
    m = len(L)
    S = [0.0]
    for j in range(m):
        S.append(S[-1] + L[j])
    # All possible shift values where a boundary of one interval
    # aligns with a boundary of another
    shifts = set()
    for p in range(m):
        for q in range(m):
            # |s_p - s_q|
            shifts.add(abs(S[p] - S[q]))
            shifts.add(abs((2 - S[p]) - (2 - S[q])))
            # Check if s_p and s_q are relevant boundary positions
            shifts.add(S[p])
            shifts.add(2 - S[q])
    # Remove values outside [0,2]
    shifts = [s for s in shifts if 0 <= s <= 2]
    return sorted(list(set(shifts)))

def compute_C(L, A):
    """Compute the maximum overlap integral for given parameters"""
    shifts = build_critical_shifts(L)
    max_J = -float('inf')
    for k in shifts:
        J = compute_J(k, L, A)
        if J > max_J:
            max_J = J
    return max_J

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result
