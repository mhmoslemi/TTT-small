import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def random_binary_solution():
    """Create a random binary solution with target sum ones"""
    h = np.zeros(n_points)
    indices = np.random.choice(n_points, size=int(TARGET_SUM), replace=False)
    h[indices] = 1.0
    return h

def simulated_annealing_binary():
    """Run simulated annealing on binary patterns"""
    best_h = random_binary_solution()
    best_c5 = evaluate_c5(best_h)
    
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Annealing parameters
    initial_temp = 0.015
    final_temp = 0.0001
    max_iter = 200000
    start_time = time.time()
    budget_s = 580  # Leave 20s for final processing
    
    temp = initial_temp
    temp_decay = (final_temp / initial_temp) ** (1.0 / max_iter)
    
    # Track progress
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Sample two indices: one from 1s, one from 0s
        ones_indices = np.where(current_h == 1)[0]
        zeros_indices = np.where(current_h == 0)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            # Resample if no 1s or 0s
            current_h = random_binary_solution()
            current_c5 = evaluate_c5(current_h)
            continue
            
        i_idx = np.random.choice(ones_indices)
        j_idx = np.random.choice(zeros_indices)
        
        # Swap
        new_h = current_h.copy()
        new_h[i_idx] = 0.0
        new_h[j_idx] = 1.0
        
        # Evaluate
        new_c5 = evaluate_c5(new_h)
        
        # Accept or reject
        delta = current_c5 - new_c5
        
        if delta > 0 or np.random.random() < np.exp(delta / temp):
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                last_improvement = i
                
        # Update temperature
        temp *= temp_decay
        
        # Occasionally restart if no improvement for a while
        if i - last_improvement > 10000 and i > 10000:
            current_h = random_binary_solution()
            current_c5 = evaluate_c5(current_h)
            last_improvement = i
            temp = initial_temp  # Reset temperature
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h, best_c5 = simulated_annealing_binary()
    
    # Final evaluation to make sure the result is accurate
    final_c5 = evaluate_c5(best_h)
    
    # If we found a better solution, update
    if final_c5 < best_c5:
        best_c5 = final_c5
	
    return (best_h.tolist(), float(best_c5), n_points)
