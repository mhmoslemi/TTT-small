import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_sum_box(v, target_sum):
    """Project vector v onto sum = target_sum with box constraints [0,1]"""
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-10:
        return v.copy()
    
    # Use a bisection approach to find the correct Lagrange multiplier
    def objective(lam):
        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = objective(m)
        if abs(fm) < 1e-10:
            break
        if fm * objective(a) < 0:
            b = m
        else:
            a = m
    lam = (a + b) / 2
    result = np.clip(v + lam, 0, 1)
    return result

def solve_milp_binary(n, m):
    """Solve integer programming for binary sequence that maximizes minimum correlation"""
    # Variables
    y = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(y) == m]
    
    # Product variables and constraints for each shift k
    shift_vars = []
    for k in range(1, n):
        indices = list(range(n - k))
        w_k = cp.Variable(len(indices), nonneg=True)
        shift_vars.append((k, w_k))
        # Link y[i] and y[i+k] with w_k[i]
        for i in range(len(indices)):
            constraints.extend([
                w_k[i] <= y[i],
                w_k[i] <= y[i + k],
                w_k[i] >= y[i] + y[i + k] - 1,
            ])
        # Constraint: sum of products >= t
        constraints.append(cp.sum(w_k) >= t)
        
    # Add t constraint
    constraints.append(t >= 0)
    constraints.append(t <= m)
    
    # Objective: maximize t
    prob = cp.Problem(cp.Maximize(t), constraints)
    
    # Try different solvers
    solver = None
    solutions = [cp.CBC, cp.GLPK_MI, cp.ECOS_BB]
    for s in solutions:
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                solver = s
                break
        except:
            continue
    
    if solver is None or prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None
        
    y_sol = np.round(y.value).astype(int)
    return y_sol

        # Find worst shift k
        corr = correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr) - (n_points - 1)
        
        # Compute gradient (subgradient) for current worst shift
        g = np.ones(n_points)
        if k_max >= 0:
            g[:n_points - k_max] -= h[k_max:]
        else:
            k = -k_max
            g[k:] -= h[:n_points - k]
        
        # Add symmetric term
        if k_max > 0:
            g[k_max:] -= h[:n_points - k_max]
        elif k_max < 0:
            k = -k_max
            g[:n_points - k] -= h[k:]
        
        # Step size
        alpha = 0.01 / np.sqrt(t)
        
        # Gradient step
        h_new = h - alpha * g
        
        # Project back to constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

        h = h_new.copy()
        
    return best_h, best_c5

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)
