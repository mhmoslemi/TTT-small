import numpy as np

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_qp_with_constraints(h0):
    """Solve quadratic programming problem with box constraints"""
    from cvxpy import Variable, Minimize, Problem, sum_squares, Constraint
    
    h = Variable(n_points)
    objective = Minimize(sum_squares(h - h0))
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == TARGET_SUM
    ]
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver='ECOS', verbose=False)
        return np.array(h.value).flatten()
    except:
        return h0.copy()

def insert_peaks(h):
    """Add two sharp peaks to make better overlap suppression"""
    # Locate areas with low values and add two strong spikes
    thresholds = [0.2, 0.5, 0.8]
    for thres in thresholds:
        mask = (h < thres)
        if np.sum(mask) > 10:
            # Set some low values to 1
            indices = np.where(mask)[0]
            for _ in range(min(2, len(indices)//3)):
                idx = np.random.choice(indices)
                h[idx] = 1.0
                # Smooth out nearby values by reducing them slightly
                for offset in [-1, 0, 1]:
                    if 0 <= idx + offset < len(h):
                        h[idx + offset] = max(0, h[idx + offset] - 0.2)
            break
    
    return project_to_sum_box(h, TARGET_SUM)

def local_search_improve(h0, max_iter=300):
    """Improve solution with local search"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Generate neighbors by making small changes
        h_new = h.copy()
        # Pick random indices
        idxs = np.random.choice(n_points, size=3, replace=False)
        # Make small adjustments
        for idx in idxs:
            noise = np.random.uniform(-0.02, 0.02)
            h_new[idx] += noise
        # Project back to constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            
        # Occasionally, try injecting peaks
        if i % 50 == 0 and np.random.random() < 0.1:
            h = insert_peaks(h.copy())
            best_c5 = evaluate_c5(h)
    
    return h

        c5_qp = evaluate_c5(h_qp)
        if c5_qp < best_c5:
            best_h = h_qp.copy()
            best_c5 = c5_qp
            
    return (best_h.tolist(), float(best_c5), n_points)
