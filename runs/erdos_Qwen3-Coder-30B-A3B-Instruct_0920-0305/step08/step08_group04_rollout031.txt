def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def piecewise_const_from_params(params, n_points):
    """Convert parameters to piecewise constant function h"""
    # params: [log_widths[0] to log_widths[K-1], heights_raw[0] to heights_raw[K-1]]
    K = 8

    heights_raw = params[K-1:]
    
    # Process widths
    widths = np.exp(log_widths)
    total_width = np.sum(widths)
    widths = 2 * widths / total_width
    
    # Pad width to have K elements
    widths = np.append(widths, 2 - np.sum(widths))
    
    # Process heights (logistic transform to [0,1])
    heights = 1 / (1 + np.exp(-heights_raw))
    heights = np.clip(heights, 0, 1)
    
    # Create piecewise constant function
    h = np.zeros(n_points)
    cum_width = 0
    for i in range(K):
        start_idx = int(cum_width / 2 * n_points)
        end_idx = int((cum_width + widths[i]) / 2 * n_points)
        h[start_idx:end_idx] = heights[i]
        cum_width += widths[i]
    
    # Handle any remaining points
    h[end_idx:] = heights[-1]
    
    return h
