import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def neighbor_sum(h, lag):
    """Compute neighbor sum vector for given lag"""
    lag_abs = abs(lag)
    if lag_abs == 0:
        return 2 * h
    else:
        w = np.zeros_like(h)
        if lag > 0:
            w[:n_points-lag_abs] += h[lag_abs:]
            w[lag_abs:] += h[:n_points-lag_abs]
        else:
            w[lag_abs:] += h[:-lag_abs]
            w[:-lag_abs] += h[lag_abs:]
        return w

def mass_transfer_step(h, step_size, eps=1e-6):
    """Perform one step of mass transfer to reduce maximum overlap"""
    corr = np.correlate(h, 1-h, mode='full') * dx
    idx_max = np.argmax(corr)
    lag_max = idx_max - (n_points - 1)
    
    # Compute neighbor sum w for the current worst lag
    w = neighbor_sum(h, lag_max)
    
    # Find indices with low and high w values
    low_indices = np.where(h > eps)[0]
    high_indices = np.where(h < 1 - eps)[0]
    
    if len(low_indices) == 0 or len(high_indices) == 0:
        return h, False
    
    # Find low w index with highest h value
    low_idx = low_indices[np.argmin(w[low_indices])]
    
    # Find high w index with lowest h value
    high_idx = high_indices[np.argmax(w[high_indices])]
    
    # Compute transfer amount
    delta = min(step_size, h[low_idx], 1 - h[high_idx])
    
    if delta < 1e-8:
        return h, False
    
    # Perform transfer
    h_new = h.copy()
    h_new[low_idx] -= delta
    h_new[high_idx] += delta
    
    return h_new, True
