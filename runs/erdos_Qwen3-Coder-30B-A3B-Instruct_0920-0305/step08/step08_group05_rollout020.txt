def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def neighbor_sum(h, lag):
    """Compute neighbor sum for given lag"""
    if lag == 0:
        return 2 * h
    else:
        lag_abs = abs(lag)
        w = np.zeros_like(h)
        if lag > 0:
            w[:n_points-lag_abs] += h[lag_abs:]
            w[lag_abs:] += h[:n_points-lag_abs]
        else:
            w[lag_abs:] += h[:n_points-lag_abs]
            w[:n_points-lag_abs] += h[lag_abs:]
        return w

    L = 2 * len(h) - 1

    # Find the worst lag

    lag_max = idx_max - (n_points - 1)
    
    # Compute neighbor sum for this lag
    w = neighbor_sum(h, lag_max)
    
    # Find low and high points for transfer
    low_idx = -1
    high_idx = -1
    
    # Find low w point with h > eps
    for i in range(n_points):
        if h[i] > eps and (low_idx == -1 or w[i] < w[low_idx]):
            low_idx = i
    
    # Find high w point with h < 1-eps
    for i in range(n_points):
        if h[i] < 1 - eps and (high_idx == -1 or w[i] > w[high_idx]):
            high_idx = i
    
    if low_idx == -1 or high_idx == -1:
        return h, False
    
    # Compute transfer amount
    delta = min(step_size, h[low_idx], 1 - h[high_idx])
    if delta < 1e-12:
        return h, False
    
    # Update h
    h_new = h.copy()
    h_new[low_idx] -= delta
    h_new[high_idx] += delta
    
    # Project back to sum constraint
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    
    return h_new, True
