import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import differential_evolution
import math

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def create_initial_solution():
    """Create initial solution with pattern that suppresses overlaps well"""
    # Start with a piecewise constant function that has values near 0 and 1
    # This creates good separation in the correlation
    h = np.zeros(n_points)
    
    # Place two spikes of high value
    spike_width = n_points // 5
    start1 = n_points // 4
    start2 = 3 * n_points // 4
    
    h[start1:start1 + spike_width] = 1.0
    h[start2:start2 + spike_width] = 1.0
    
    # Adjust to meet sum constraint
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def perturb_solution(h, strength=0.05):
    """Create a neighbor by small random changes"""
    h_new = h.copy()
    
    # Randomly modify some elements
    indices = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
    
    for idx in indices:
        # Add small random perturbation
        delta = np.random.uniform(-strength, strength)
        h_new[idx] += delta
        
        # Project onto feasible region
        if h_new[idx] < 0:
            h_new[idx] = 0
        elif h_new[idx] > 1:
            h_new[idx] = 1
    
    # Ensure sum constraint is met
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    return h_new

def evolutionary_optimization():
    """Use differential evolution with constraint handling"""
    def objective(h):
        return evaluate_c5(h)
    
    # Define bounds: [0, 1] for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Start with a good initial solution
    h0 = create_initial_solution()
    
    # Use differential evolution with our custom bounds
    result = differential_evolution(
        objective,
        bounds,
        maxiter=1000,
        popsize=15,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=42,
        callback=lambda x, convergence: None
    )
    
    return project_to_sum_box(result.x, TARGET_SUM)

def simulated_annealing_optimization():
    """Use simulated annealing approach"""
    np.random.seed(42)
    
    # Initial solution
    current_h = create_initial_solution()
    current_c5 = evaluate_c5(current_h)
    
    # Parameters
    T = 0.1
    T_min = 1e-6
    alpha = 0.99
    max_iter = 10000
    
    best_h = current_h.copy()
    best_c5 = current_c5
    
    for iteration in range(max_iter):
        # Perturb solution
        new_h = perturb_solution(current_h, strength=T)
        new_c5 = evaluate_c5(new_h)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Cool down
        T *= alpha
        if T < T_min:
            T = T_min
            
        # Every few iterations, add a random perturbation to escape local optima
        if iteration % 500 == 0 and iteration > 0:
            current_h = create_initial_solution()
            current_c5 = evaluate_c5(current_h)
    
    return best_h

def multi_start_method():
    """Combine multiple starting points and methods"""
    best_h = None
    best_c5 = np.inf
    
    # Try different initial structures
    initial_configs = [
        lambda: create_initial_solution(),
        lambda: np.random.uniform(0, 1, n_points),
        lambda: np.clip(np.random.normal(0.5, 0.2, n_points), 0, 1),
        lambda: np.random.choice([0, 1], size=n_points, p=[0.4, 0.6]),
    ]
    
    for i, initial_func in enumerate(initial_configs):
        # Run evolutionary optimization from this starting point
        try:
            h = initial_func()
            # Limit the optimization to avoid spending too much time on one
            h_opt = evolutionary_optimization()
            c5_opt = evaluate_c5(h_opt)
            
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
        except:
            continue
            
    return best_h if best_h is not None else create_initial_solution()

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    
    # Budget management: spend time on different phases
    remaining_time = budget_s
    
    # Phase 1: Multi-start optimization
    start_phase1 = time.time()
    h_opt = multi_start_method()
    phase1_time = time.time() - start_phase1
    remaining_time -= phase1_time
    
    # Phase 2: Fine-tune with simulated annealing
    start_phase2 = time.time()
    if remaining_time > 10:
        h_opt = simulated_annealing_optimization()
    phase2_time = time.time() - start_phase2
    remaining_time -= phase2_time
    
    # Final evaluation
    final_c5 = evaluate_c5(h_opt)
    
    return (h_opt.tolist(), float(final_c5), n_points)
