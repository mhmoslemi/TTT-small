import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def create_binary_construction():
    """Create a binary-like construction that's more structured"""
    h = np.zeros(n_points)
    
    # Create a structure with carefully placed 0s and 1s
    # The goal is to be very sparse in areas that would create overlap
    # and dense in areas that are well-separated
    
    # Start with a basic pattern: 1s at positions that are far apart
    # and 0s where overlap might be problematic
    
    # Place a few large blocks of 1s
    block_start = int(n_points * 0.2)
    block_width = int(n_points * 0.3)
    h[block_start:block_start+block_width] = 1.0
    
    # Add another block
    block_start2 = int(n_points * 0.6)
    h[block_start2:block_start2+block_width] = 1.0
    
    # Make sure it's properly normalized
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def create_smooth_piecewise():
    """Create a piecewise smooth function with better control"""
    h = np.zeros(n_points)
    
    # Create a smooth step-like shape
    # This method creates a combination of ramps and steps
    
    # Smooth transition from 0 to 1 over the first portion
    transition_length = n_points // 4
    h[:transition_length] = np.linspace(0, 1, transition_length)
    
    # Flat region at 1
    flat_start = transition_length
    flat_end = flat_start + n_points // 4
    h[flat_start:flat_end] = 1.0
    
    # Smooth decrease
    transition_length = n_points // 6
    h[flat_end:flat_end+transition_length] = np.linspace(1, 0, transition_length)
    
    # Remaining values from 0 to 1, less steep
    remaining_start = flat_end + transition_length
    h[remaining_start:remaining_start+n_points//6] = np.linspace(0, 1, n_points//6)
    
    # Normalize
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def create_sine_modulated():
    """Create a sine-wave modulated construction"""
    x = np.linspace(0, 2, n_points)
    # Add a component that will push values up and down to avoid overlap
    h = 0.5 + 0.3 * np.sin(3 * np.pi * x) + 0.2 * np.cos(5 * np.pi * x)
    h = np.clip(h, 0, 1)
    
    # Adjust so that sum matches
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def smart_local_search(h0, max_iter=2000):
    """Improved local search with adaptive step sizes and multiple strategies"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for iteration in range(max_iter):
        # Strategy 1: Try single moves
        if iteration % 5 == 0:
            idx = np.random.randint(0, n_points)
            step_size = 0.02 * (1 + np.random.random() * 2)
            delta = step_size * (np.random.random() - 0.5)
            h_new = h.copy()
            h_new[idx] += delta
            
            if h_new[idx] < 0:
                h_new[idx] = 0
            elif h_new[idx] > 1:
                h_new[idx] = 1
                
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            new_c5 = evaluate_c5(h_new)
            
            if new_c5 < best_c5:
                h = h_new.copy()
                best_c5 = new_c5
        
        # Strategy 2: Try nearby group moves
        if iteration % 10 == 0:
            start_idx = np.random.randint(0, n_points - 5)
            end_idx = min(start_idx + 5, n_points)
            step_size = 0.02 * (1 + np.random.random())
            deltas = np.full(end_idx - start_idx, step_size * (np.random.random() - 0.5))
            h_new = h.copy()
            h_new[start_idx:end_idx] += deltas
            
            # Clip
            h_new = np.clip(h_new, 0, 1)
            
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            new_c5 = evaluate_c5(h_new)
            
            if new_c5 < best_c5:
                h = h_new.copy()
                best_c5 = new_c5
        
        # Strategy 3: Random population for diversity
        if iteration % 50 == 0:
            new_h = create_binary_construction()
            new_h = project_to_sum_box(new_h, TARGET_SUM)
            new_c5 = evaluate_c5(new_h)
            if new_c5 < best_c5:
                h = new_h.copy()
                best_c5 = new_c5
                
        # Occasionally reset with a better construction if it's not improving
        if iteration % 100 == 0 and best_c5 > 0.38:
            temp_h = create_binary_construction()
            temp_h = project_to_sum_box(temp_h, TARGET_SUM)
            temp_c5 = evaluate_c5(temp_h)
            if temp_c5 < best_c5:
                h = temp_h.copy()
                best_c5 = temp_c5
                
    return h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Generate multiple different starting constructions and optimize each
    construction_funcs = [
        create_binary_construction,
        create_smooth_piecewise,
        create_sine_modulated,
        lambda: np.random.uniform(0, 1, n_points),
        lambda: np.random.choice([0, 1], size=n_points, p=[0.4, 0.6]),
    ]
    
    # Run for as much of budget as possible
    while time.time() - start_time < budget_s - 1:
        # Choose random construction method
        func_index = random.randint(0, len(construction_funcs) - 1)
        h0 = construction_funcs[func_index]()
        
        # Make sure it's properly normalized
        h0 = project_to_sum_box(h0, TARGET_SUM)
        
        # Run local search
        h_final, c5_final = smart_local_search(h0)
        
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
            
    return (best_h.tolist(), float(best_c5), n_points)
