import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    
    # Use differential evolution for global optimization
    def objective(h):
        return evaluate_c5(h)
    
    # Define bounds: [0, 1] for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use differential evolution to get a good starting point
    de_result = differential_evolution(
        objective,
        bounds,
        maxiter=500,
        popsize=15,
        tol=1e-6,

        recombination=0.7,
        seed=seed,
        callback=lambda x, convergence: None,
        disp=False
    )

    best_c5 = evaluate_c5(best_h)
    
    # Refine with local optimization
    from scipy.optimize import minimize
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Try local optimization from several starting points
    for attempt in range(5):

            )
            if res.success:
                h_candidate = project_to_sum_box(res.x, TARGET_SUM)
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
