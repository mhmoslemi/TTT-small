from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def evaluate_c5_correlate(h):
    """Evaluation using numpy correlate as used in final evaluation"""
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def neighbor_sum(h, lag):
    """Compute the neighbor sum w_i = h[i+lag] + h[i-lag] for valid indices"""
    lag_abs = abs(lag)
    if lag_abs >= len(h):
        return np.zeros_like(h)
    
    w = np.zeros_like(h)
    if lag >= 0:
        w[:len(h)-lag_abs] += h[lag_abs:]
        w[lag_abs:] += h[:len(h)-lag_abs]
    else:
        w[lag_abs:] += h[:-lag_abs]
        w[:len(h)-lag_abs] += h[lag_abs:]
    return w

def mass_transfer_step(h, step_size, eps=1e-6):
    """Perform one mass transfer step to improve the solution"""
    # Compute full correlation to find worst shift
    corr = np.correlate(h, 1-h, mode="full")
    idx_max = np.argmax(corr)
    lag_max = idx_max - (len(h) - 1)
    
    # Compute neighbor sum for the worst shift
    w = neighbor_sum(h, lag_max)
    
    # Find indices with low and high w values that satisfy constraints
    low_indices = np.where(h > eps)[0]
    high_indices = np.where(h < 1 - eps)[0]
    
    if len(low_indices) == 0 or len(high_indices) == 0:
        return h, False
    
    # For each candidate low/high pair, try to transfer
    improved = False
    for i in range(len(low_indices)):
        low_idx = low_indices[i]
        if w[low_idx] < w[low_indices[0]] + 1e-10:  # Near minimum
            for j in range(len(high_indices)):
                high_idx = high_indices[j]
                if w[high_idx] > w[high_indices[-1]] - 1e-10:  # Near maximum
                    delta = min(step_size, h[low_idx], 1 - h[high_idx])
                    if delta > 1e-12:
                        h_new = h.copy()
                        h_new[low_idx] -= delta
                        h_new[high_idx] += delta
                        improved = True
                        return h_new, True
    return h, improved
