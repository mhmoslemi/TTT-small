import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import time

# Global constants
n_points = 300
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water-filling"""
    # Sort the vector in descending order
    sorted_indices = np.argsort(x)[::-1]
    sorted_x = x[sorted_indices]
    
    # Find the optimal lambda
    cumsum = np.cumsum(sorted_x)
    n = len(sorted_x)
    lambdas = (cumsum - target_sum) / np.arange(1, n+1)
    
    # Find the largest lambda that makes all elements non-negative
    if sorted_x[-1] >= 0:
        # Find the largest lambda such that projected values are non-negative
        # Let k be the number of elements that become zero or negative when we add lambda
        # We want to find the smallest k s.t. (sorted_x[k-1] + lambda) <= 0 or (sorted_x[k-1] + lambda) >= 1 
        # But since sorted_x is in descending order, the smallest lambda value that makes some element=0
        # or the largest lambda that makes some element 1 is what matters
        # iterative handle this correctly
        
        # Water filling approach
        # Sort values and compute cumulative sums
        # Find critical lambda that makes the projection correct
        lambda_value = 0.0
        remaining_sum = target_sum
        remaining_entries = n
        for i in range(n):
            if sorted_x[i] > 1:
                # An entry that will be clipped to 1, so we don't use it to adjust lambda
                pass
            elif sorted_x[i] < 0:
                # This shouldn't happen, just to be safe
                raise ValueError("In project_to_sum_box, unexpected negative value")
            else:
                # Compute threshold lambda for this set of elements
                threshold = (remaining_sum - 1.0*(n-i-1)) / (n-i)
                if sorted_x[i] >= threshold:
                    lambda_value = threshold
                else:
                    lambda_value = sorted_x[i]
                break
        else:
            # If every element is 0, then we need to determine lambda based on sum
            lambda_value = target_sum / n
            
        # Adjust all entries
        if lambda_value < 0:
            lambda_value = 0.0
            
        # Compute projection
        temp = x + lambda_value
        result = np.clip(temp, 0, 1)
        actual_sum = np.sum(result)
        if abs(actual_sum - target_sum) > 1e-10:
            # Fine adjustment to make sum exact
            # Need to change some values from 1 to 0 or vice versa
            # Simple linear search approach is slow but safe
            if actual_sum > target_sum:
                # Need to decrease some values
                order_dec = np.argsort(result)
                for idx in reversed(order_dec):
                    if result[idx] > 0:
                        delta = min(result[idx], actual_sum - target_sum)
                        result[idx] -= delta
                        actual_sum -= delta
                        if abs(actual_sum - target_sum) < 1e-10:
                            break
            else:  
                # Need to increase some values
                order_inc = np.argsort(result)
                for idx in order_inc:
                    if result[idx] < 1:
                        delta = min(1 - result[idx], target_sum - actual_sum)
                        result[idx] += delta
                        actual_sum += delta
                        if abs(actual_sum - target_sum) < 1e-10:
                            break
    else:
        # Need to adjust each element using cumulative sum
        # Continue the water filing approach correctly
        if n == 0:
            return x
        # This should not generally happen but safety
        # Sort in descending order to apply the algorithm correctly
        zipped = [(x[i], i) for i in range(n)]
        zipped.sort(key=lambda a: a[0], reverse=True)
        sorted_x_val = [a[0] for a in zipped]
        sorted_indices = [a[1] for a in zipped]
        # Compute cumulative sums of sorted values
        cumsum = np.cumsum(sorted_x_val)
        # Compute lambda_arr for each potential cut
        lambda_arr = (cumsum - target_sum) / np.arange(1, n+1)
        # Lambda needed to make the first i+1 elements equal to 1.0
        lambda_1 = (cumsum[-1] + 1 - cumsum) / np.arange(1, n+1)
        lambda_1 = np.array([max(0.0, np.clip(a, 0, 1)) for a in 1 - sorted_x_val])
        # We need to find lambda such that result would match target_sum
        # We first save the sorted val and then sort back
        # Using a corrected approach
        sorted_x = np.sort(x)[::-1]
        cumsum = np.cumsum(sorted_x)
        max_lambda = 1.0
        min_lambda = 0.0
        lambda_found = 0.0
        
        def compute_sum_for_lambda(lam):
            result = np.clip(sorted_x + lam, 0, 1.)
            return np.sum(result)
            
        tolerance = 1e-12
        for _ in range(50):
            test_lambda = (min_lambda + max_lambda) / 2
            test_sum = compute_sum_for_lambda(test_lambda)
            if abs(test_sum - target_sum) < tolerance:
                lambda_found = test_lambda
                break
            elif test_sum > target_sum:
                min_lambda = test_lambda
            else:
                max_lambda = test_lambda
                
        # Final adjustment
        lambda_found = (max_lambda + min_lambda) / 2
        result = np.clip(x + lambda_found, 0, 1)
        
    if np.isclose(np.sum(result), target_sum, atol=1e-10):
        pass
    else:
        # Last resort method to correct
        # Find indices to decrease and increase
        diff = np.sum(result) - target_sum
        if diff > 0:
            # Need to decrease
            order = np.argsort(result)[::-1]
            for i, idx in enumerate(order):
                if diff <= 0:
                    break
                if result[idx] > 0:
                    delta = min(diff, result[idx])
                    result[idx] -= delta
                    diff -= delta
        else:
            # Need to increase
            order = np.argsort(result)
            for i, idx in enumerate(order):
                if diff >= 0:
                    break
                if result[idx] < 1:
                    delta = min(-diff, 1 - result[idx])
                    result[idx] += delta
                    diff += delta
    
    return result

def rasterise_and_project(θ):
    """Transforms the parameter vector into an h vector and applies projection"""
    K = 8
    # Recover widths
    widths_raw = np.exp(np.clip(θ[:K], -15, 15))
    widths = 2 * widths_raw / np.sum(widths_raw)
    
    # Recover heights
    heights = 1 / (1 + np.exp(-θ[K:]))
    
    # Allocate h vector
    h = np.zeros(n_points)
    
    # Set values
    cumsum_widths = np.cumsum(widths)
    x_positions = np.linspace(0, 2, n_points + 1)
    x_positions = (x_positions[1:] + x_positions[:-1]) / 2
    
    block_indices = np.searchsorted(cumsum_widths, x_positions, side='right')
    # Adjust indices to be 0-based
    block_indices = np.clip(block_indices, 0, K - 1)
    
    # Assign corresponding heights
    for i in range(K):
        mask = block_indices == i
        h[mask] = heights[i]
    
    # Project to desired sum
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def objective_de(θ):
    h = rasterise_and_project(θ)
    return evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Step 1: Coarse DE on n=150
    n_coarse = 150
    dx_coarse = 2.0 / n_coarse
    target_sum_coarse = n_coarse / 2
    
    def coarse_project_to_sum_box(x, target_sum):
        sorted_x = np.sort(x)[::-1]
        cumsum = np.cumsum(sorted_x)
        n = len(sorted_x)
        lambda_val = 0.0
        remaining_sum = target_sum
        remaining_entries = n
        
        # Binary search on lambda
        low, high = -10.0, 10.0
        for _ in range(50):
            test_lambda = (low + high) / 2
            projected = np.clip(x + test_lambda, 0, 1)
            actual_sum = np.sum(projected)
            if abs(actual_sum - target_sum) < 1e-12:
                lambda_val = test_lambda
                break
            elif actual_sum > target_sum:
                high = test_lambda
            else:
                low = test_lambda
        else:
            lambda_val = (low + high)/2
        return np.clip(x + lambda_val, 0, 1)
    
    def coarse_rasterise_and_project(θ):
        K = 8
        widths_raw = np.exp(np.clip(θ[:K], -15, 15))
        widths = 2 * widths_raw / np.sum(widths_raw)
        heights = 1 / (1 + np.exp(-θ[K:]))
        
        h = np.zeros(n_coarse)
        cumsum_widths = np.cumsum(widths)
        x_positions = np.linspace(0, 2, n_coarse + 1)
        x_positions = (x_positions[1:] + x_positions[:-1]) / 2
        
        block_indices = np.searchsorted(cumsum_widths, x_positions, side='right')
        block_indices = np.clip(block_indices, 0, K - 1)
        
        for i in range(K):
            mask = block_indices == i
            h[mask] = heights[i]
        
        h = coarse_project_to_sum_box(h, target_sum_coarse)
        return h
    
    def coarse_obj(θ):
        h = coarse_rasterise_and_project(θ)
        # For simplicity, just compute the full correlation using a detailed search path
        # Evaluate the correlation correctly
        h_rev = h[::-1]
        b = 1 - h
        L = 2 * len(h) - 1
        corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
        return dx_coarse * np.max(corr)
    
    # Define bounds for DE
    bounds = [(-5, 5) for _ in range(16)]
    
    coarse_de_res = differential_evolution(
        coarse_obj, bounds, 
        popsize=12, maxiter=120, 
        seed=seed+1, 
        disp=False, 
        atol=1e-8, 
        rtol=1e-8
    )
    
    h_coarse = coarse_rasterise_and_project(coarse_de_res.x)
    
    # Step 2: Local refinement on coarse grid
    def coarse_evaluate_c5(h):
        h_rev = h[::-1]
        b = 1 - h
        L = 2 * len(h) - 1
        corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
        return dx_coarse * np.max(corr)
    
    def pgd_refinement_coarse(h0, max_iter=500):
        h = h0.copy()
        best_h = h.copy()
        best_c5 = coarse_evaluate_c5(h)
        
        step_size = 0.5
        eta_max = 2.0
        
        for i in range(max_iter):
            # Compute correlation and find active lags
            h_rev = h[::-1]
            b = 1 - h
            L = 2 * len(h) - 1
            corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
            max_val = corr.max()
            active = np.where(corr >= max_val - 1e-12)[0] 
            shifts = active - (len(h) - 1)  # convert to signed shift
            
            # Assemble subgradient
            g = np.zeros(len(h))
            for s in shifts:
                if s >= 0:
                    g[:len(h)-s] += -dx_coarse * h[s:]
                    g[s:] += -dx_coarse * h[:len(h)-s]
                else:
                    t = -s
                    g[t:] += -dx_coarse * h[:len(h)-t]
                    g[:len(h)-t] += -dx_coarse * h[t:]
            g /= len(shifts)
            
            # Perform descent step
            d = -g  # descent direction
            h_new = h + step_size * d
            
            # Project
            h_new = coarse_project_to_sum_box(h_new, target_sum_coarse)
            
            # Evaluate
            c5_new = coarse_evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                step_size = min(eta_max, step_size * 1.05)  # Increase
            else:
                step_size = max(1e-4, step_size * 0.7)  # Decrease
            h = h_new
            
        return best_h, best_c5
    
    h_coarse_ref, c5_coarse = pgd_refinement_coarse(h_coarse, max_iter=500)
    
    # Step 3: Upsample to fine resolution
    h_fine_start = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), h_coarse_ref)
    h_fine_start = project_to_sum_box(h_fine_start, TARGET_SUM)
    
    # Step 4: Fine grid PSGD refinement
    def fine_evaluate_c5(h):
        h_rev = h[::-1]
        b = 1 - h
        L = 2 * len(h) - 1
        corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
        return dx * np.max(corr)
    
    def pgd_refinement_fine(h0, max_iter=5000):
        h = h0.copy()
        best_h = h.copy()
        best_c5 = fine_evaluate_c5(h)
        
        step_size = 0.5
        eta_max = 2.0
        no_imp_count = 0
        swap_period = 30
        max_no_imp = 200
        delta = 1e-4
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute correlation and find active lags
            h_rev = h[::-1]
            b = 1 - h
            L = 2 * len(h) - 1
            corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
            max_val = corr.max()
            active = np.where(corr >= max_val - 1e-12)[0]
            shifts = active - (len(h) - 1)  # convert to signed shift
            
            # Assemble subgradient
            g = np.zeros(len(h))
            for s in shifts:
                if s >= 0:
                    g[:len(h)-s] += -dx * h[s:]
                    g[s:] += -dx * h[:len(h)-s]
                else:
                    t = -s
                    g[t:] += -dx * h[:len(h)-t]
                    g[:len(h)-t] += -dx * h[t:]
            g /= len(shifts)
            
            # Perform descent step
            d = -g  # descent direction
            h_new = h + step_size * d
            
            # Project
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            # Evaluate
            c5_new = fine_evaluate_c5(h_new)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                step_size = min(eta_max, step_size * 1.05)  # Increase
                no_imp_count = 0
            else:
                step_size = max(1e-4, step_size * 0.7)  # Decrease
                no_imp_count += 1
            
            h = h_new
            
            # Mass transfer moves
            if i % swap_period == 0 or no_imp_count >= max_no_imp:
                # Pick two points
                valid_indices = np.where((h > 0) & (h < 1))[0]
                if len(valid_indices) >= 2:
                    idx1, idx2 = np.random.choice(valid_indices, 2, replace=False)
                    # Transfer small amount
                    transfer_amount = min(0.02, h[idx2])
                    if transfer_amount > 0:
                        amount = np.random.random() * transfer_amount
                        h[idx1] += amount
                        h[idx2] -= amount
                        h = project_to_sum_box(h, TARGET_SUM)
                        c5_test = fine_evaluate_c5(h)
                        if c5_test <= best_c5 + delta:
                            if c5_test < best_c5:
                                best_c5 = c5_test
                                best_h = h.copy()
                            no_imp_count = 0
                        else:
                            # Maybe revert?
                            pass
            
            if no_imp_count >= max_no_imp:
                no_imp_count = 0
                # Try a stronger move
                valid_indices = np.where((h > 0) & (h < 1))[0]
                if len(valid_indices) >= 2:
                    idx1, idx2 = np.random.choice(valid_indices, 2, replace=False)
                    transfer_amount = min(0.05, h[idx2])
                    if transfer_amount > 0:
                        amount = np.random.random() * transfer_amount
                        h[idx1] += amount
                        h[idx2] -= amount
                        h = project_to_sum_box(h, TARGET_SUM)
                        c5_test = fine_evaluate_c5(h)
                        if c5_test <= best_c5 + delta:
                            if c5_test < best_c5:
                                best_c5 = c5_test
                                best_h = h.copy()
                            no_imp_count = 0
                        else:
                            pass
    
        return best_h, best_c5
    
    final_h, final_c5 = pgd_refinement_fine(h_fine_start, max_iter=5000)
    
    return (final_h.tolist(), float(final_c5), n_points)
