# Global variables
n_points = 256
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def compute_overlap_integral(k, L, A):
    """Compute J(k) = int h(x) (1 - h(x+k)) dx for given interval lengths and heights"""
    # Total interval endpoints
    s = np.concatenate([[0], np.cumsum(L)])
    m = len(L)  # number of intervals
    
    # Linear term: int h(x) dx over x \in [0, 2-k]
    linear_term = 0
    x_end = 2 - k
    x = 0
    for j in range(m):
        if x >= x_end:
            break
        x_next = s[j+1]
        if x_next > x_end:
            # Partial interval
            linear_term += A[j] * (x_end - x)
            break
        else:
            # Full interval
            linear_term += A[j] * (x_next - x)
            x = x_next
            continue
    
    # Product term: sum over pairs of intervals
    product_term = 0
    for p in range(m):
        for q in range(m):
            # Overlap of interval p with shifted interval q by k
            s_p, s_p1 = s[p], s[p+1]
            s_q, s_q1 = s[q], s[q+1]
            # Overlap of [s_p, s_p1] with [s_q + k, s_q1 + k]
            left = max(s_p, s_q + k)
            right = min(s_p1, s_q1 + k)
            if right > left:
                overlap_length = right - left
                product_term += A[p] * A[q] * overlap_length
    
    return linear_term - product_term

def compute_critical_shifts(L):
    """Compute all possible critical shifts k where overlap changes"""
    s = np.concatenate([[0], np.cumsum(L)])
    shifts = set()
    for i in range(len(s)):
        for j in range(len(s)):
            shifts.add(abs(s[i] - s[j]))
    shifts = sorted(list(shifts))
    # Remove very small numerical noise
    shifts = [s for s in shifts if s >= 1e-10 and s <= 2 + 1e-10]
    return list(shifts)

def c5_from_params(L, A):
    """Compute C5 for given parameters"""
    # Ensure last a_m computed from constraint
    a_m = (1 - np.sum(A[:-1] * L[:-1])) / L[-1] if L[-1] > 1e-10 else 0
    A_full = np.append(A[:-1], a_m)
    if a_m < 0:
        return float('inf')
    # Compute critical shifts
    K_crit = compute_critical_shifts(L)
    # Evaluate J for each k in critical shifts
    J_values = [compute_overlap_integral(k, L, A) for k in K_crit]
    return max(J_values)
