def generate_critical_shifts(intervals):
    """
    Generate the finite set of critical shift values for the given intervals.
    These are the points where the overlap changes due to boundary alignment.
    """
    s = [0] + list(np.cumsum(intervals))
    critical_k = set()
    
    for i in range(len(s) - 1):
        for j in range(len(s) - 1):
            # Interval boundaries align
            critical_k.add(abs(s[i] - s[j]))
            critical_k.add(2 - abs(s[i] - s[j]))
            # Interval boundaries are at endpoints
            if s[i] >= 0 and s[i] <= 2:
                critical_k.add(s[i])
            if s[j] >= 0 and s[j] <= 2:
                critical_k.add(s[j])
    return sorted(list(critical_k))

def compute_J(k, lengths, heights):
    """
    Compute the overlap integral J(k) for a given shift k.
    """
    # Interval boundaries
    s = [0] + list(np.cumsum(lengths))
    total_mass = 0.0
    product_term = 0.0
    
    # Compute linear term: ∫ h(x) dx over [0, 2-k]
    for i in range(len(lengths)):
        a = heights[i]
        start = s[i]
        end = s[i+1]
        # Clip with [0, 2-k]
        clip_start = max(0, start)
        clip_end = min(2-k, end)
        if clip_start < clip_end:
            total_mass += a * (clip_end - clip_start)
    
    # Compute quadratic term: ∫ h(x) h(x+k) dx
    for i in range(len(lengths)):
        a = heights[i]
        start1 = s[i]
        end1 = s[i+1]
        for j in range(len(lengths)):
            b = heights[j]
            start2 = s[j]
            end2 = s[j+1]
            # Overlap region
            overlap_start = max(start1, k + start2)
            overlap_end = min(end1, k + end2)
            if overlap_start < overlap_end:
                product_term += a * b * (overlap_end - overlap_start)
    
    return total_mass - product_term

def compute_C(L, A, critical_shifts):
    """
    Compute C(L, A) = max_{k in critical_shifts} J(k; L, A)
    """
    if len(L) != len(A) - 1:
        raise ValueError("Length mismatch in L and A")
    
    # Compute last height from mass constraint: sum a_j * l_j = 1
    # We have sum_{j=1}^{m-1} a_j * l_j + a_m * l_m = 1
    # So: a_m * l_m = 1 - sum_{j=1}^{m-1} a_j * l_j
    # But we need a_m = (1 - sum_{j=1}^{m-1} a_j * l_j) / l_m, l_m = 2 - sum_{j=1}^{m-1} l_j
    
    sum_prod = sum(A[i] * L[i] for i in range(len(L)))
    last_length = 2 - sum(L)
    if last_length <= 0:
        return float('inf')
    last_height = (1.0 - sum_prod) / last_length
    if last_height < 0:
        return float('inf')
    if last_height > 1:
        return float('inf')
    heights_full = list(A) + [last_height]
    
    values = []
    for k in critical_shifts:
        try:
            val = compute_J(k, L, heights_full)
            values.append(val)
        except:
            # Handle numerical precision issues
            values.append(float('inf'))
    return max(values) if values else float('inf')
