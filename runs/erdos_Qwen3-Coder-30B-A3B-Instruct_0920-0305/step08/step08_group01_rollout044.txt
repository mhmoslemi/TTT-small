import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import SCS

def create_shift_matrices(n):
    """Create shift matrices R_s for s = 1..floor(n/2)"""
    R_list = []
    for s in range(1, (n // 2) + 1):
        R = np.zeros((n, n))
        for i in range(n - s):
            R[i, i + s] = 1
        R_list.append(R)
    return R_list

def solve_sdp(n, max_time_s=30):
    """Solve the SDP relaxation"""
    # Create variables
    X = cp.Variable((n, n), symmetric=True)
    mu = cp.Variable(n)
    lambda_var = cp.Variable()

    # Build shift matrices
    R_list = create_shift_matrices(n)
    
    # Constraints
    constraints = [
        X >> 0,  # PSD
        cp.diag(X) == mu,  # Diagonal constraints
        0 <= mu,
        mu <= 1,
        cp.sum(mu) == n / 2,
        mu[0] == 1,  # Symmetry breaking
    ]

    # Add overlap constraints
    for i, R in enumerate(R_list):
        constraints.append(cp.trace(R @ X) >= lambda_var)

    # Objective
    objective = cp.Maximize(lambda_var)

    # Problem
    prob = cp.Problem(objective, constraints)

    # Solve
    try:
        prob.solve(solver=SCS, time_limit=max_time_s, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return X.value, mu.value
    except:
        pass
    return None, None

def hyperplane_round(X_val, n, n_rounds=200):
    """Randomized hyperplane rounding to get binary solution"""
    try:
        # Eigenvalue decomposition
        V, _ = np.linalg.eigh(X_val)
        V = V[:, -min(V.shape[1], 64):]  # Keep only relevant components
        
        best_x = None
        best_c5 = np.inf
        
        for _ in range(n_rounds):
            g = np.random.randn(V.shape[1])
            scores = V @ g
            
            # Sort scores to find threshold
            perm = np.argsort(-scores)
            indices = perm[:n//2]
            x = np.zeros(n)
            x[indices] = 1
            
            # Evaluate using FFT
            corr = correlate(x, 1 - x, mode='full')
            c5 = np.max(corr)
            
            if c5 < best_c5:
                best_c5 = c5
                best_x = x
                
        return best_x, best_c5
    except:
        return None, np.inf

def local_refinement(x, n_points):
    """Targeted local refinement based on worst shift"""
    # Compute full correlation
    corr_full = correlate(x, 1 - x, mode='full')
    max_idx = np.argmax(corr_full)
    # The shift that causes max overlap is max_idx - (n-1)
    k = max_idx - (n_points - 1)
    if k < 0:
        k += n_points 
    # Find positions that are 1 but would help reduce overlap k
    # and will be 0 and would benefit from being 1
    Bad = np.where((x == 1) & (np.roll(x, k) == 0))[0]
    Good = np.where((x == 0) & (np.roll(x, k) == 1))[0]
    
    x_best = x.copy()
    c5_best = np.max(corr_full)
    
    for _ in range(2000):
        if len(Bad) == 0 or len(Good) == 0:
            break
            
        # Sample one from each
        i = Bad[np.random.randint(len(Bad))]
        j = Good[np.random.randint(len(Good))]
        
        # Make swap
        x_new = x.copy()
        x_new[i], x_new[j] = x_new[j], x_new[i]
        
        # Recompute max correlation (efficiently just update)
        corr_new = correlate(x_new, 1 - x_new, mode='full')
        c5_new = np.max(corr_new)
        
        if c5_new <= c5_best:
            x_best = x_new.copy()
            c5_best = c5_new
            # Update Bad and Good
            Bad = np.where((x_new == 1) & (np.roll(x_new, k) == 0))[0]
            Good = np.where((x_new == 0) & (np.roll(x_new, k) == 1))[0]
    
    return x_best, c5_best

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Fix n to 120
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    best_h = None
    best_c5 = np.inf
    
    # SDP solve
    while time.time() < start_time + 0.5 * budget_s:
        try:
            X_val, mu_val = solve_sdp(n_points, 25)
            if X_val is not None and mu_val is not None:
                # Try several random rounds
                for _ in range(30):
                    x, c5 = hyperplane_round(X_val, n_points, 30)
                    if x is not None and c5 < best_c5:
                        best_c5 = c5
                        best_h = x
                        
        except Exception as e:
            pass
            
        if time.time() > start_time + 0.4 * budget_s:
            break
    
    if best_h is None:
        # Fallback to a simple construction
        best_h = np.ones(n_points) / 2
        best_h[0] = 1
        best_h[-1] = 0
        best_h = np.clip(best_h, 0, 1)
        # Project to sum
        sum_h = np.sum(best_h)
        ratio = TARGET_SUM / sum_h
        best_h = np.clip(best_h * ratio, 0, 1)
        best_c5 = evaluate_c5(best_h)
    
    # Refinement
    while time.time() < start_time + 0.8 * budget_s:
        try:
            refined_h, refined_c5 = local_refinement(best_h, n_points)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h.copy()
                
            if np.abs(refined_c5 - best_c5) < 1e-8:
                break
        except Exception as e:
            break
    
    # Final verification
    final_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), final_c5 * dx, n_points)
