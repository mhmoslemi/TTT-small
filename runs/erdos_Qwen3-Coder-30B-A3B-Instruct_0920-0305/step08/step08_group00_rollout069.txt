import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Minimize, Problem, Variable, sum as cp_sum, maximum as cp_max

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points // 2  # exactly half ones
    
    # We will use a hybrid approach: exact MILP if small enough,
    # otherwise LP relaxation + rounding with polishing
    
    # Step 1: Attempt formulation as MILP (smaller n)
    # We'll go with n=120 which should be manageable for CBC or SCIP
    # Variables h_{i} in {0,1}, sum = n/2
    
    # To use CVXPY with MILP:
    # minimize C
    # subject to:
    #   m_k <= C for all k
    #   m_k = sum_i h_i * (1 - h_{i+k}) = sum_i h_i - sum_i h_i * h_{i+k}
    # We introduce auxiliary binary variables y_{i,k} = h_i * h_{i+k} for all valid i,k
    # Then:
    #   m_k = sum_i h_i - sum_i y_{i,k}
    #   y_{i,k} <= h_i; y_{i,k} <= h_{i+k}; y_{i,k} >= h_i + h_{i+k} - 1; y_{i,k} in {0,1}
    
    # Liberate `target_sum` into global scope for clarity
    global TARGET_SUM
    TARGET_SUM = target_sum
    
    if n_points <= 150:  # Safety: limit size to avoid long timeouts
        print("Using MILP approach for n =", n_points)
        
        # === MILP formulation ===
        h = Variable(n_points, boolean=True)
        C = Variable()
        
        # Constraints
        constraints = [cp_sum(h) == target_sum, C >= 0]
        
        # Compute all shifts k from -n+1 to n-1, compute mismatches
        shifted_products = {}
        m = []
        
        for k in range(-(n_points-1), n_points):
            if k >= 0:
                # i from 0 to n-1-k
                indices = list(range(n_points - k))
                h_i = h[indices]
                h_i_k = h[indices + k]
            else:
                # i from -k to n-1
                indices = list(range(-k, n_points))
                h_i = h[indices]
                h_i_k = h[indices - k]
            y_i_k = Variable(len(indices), boolean=True)
            shifted_products[(k, indices)] = y_i_k
            
            # Constraints for y_i_k:
            # y_i_k <= h_i
            # y_i_k <= h_i_k
            # y_i_k >= h_i + h_i_k - 1
            for i in range(len(indices)):
                constraints.append(y_i_k[i] <= h_i[i])
                constraints.append(y_i_k[i] <= h_i_k[i])
                constraints.append(y_i_k[i] >= h_i[i] + h_i_k[i] - 1)
            
            # Sum of mismatch for this shift k
            m_k = cp_sum(h_i) - cp_sum(y_i_k)
            m.append(m_k)
            constraints.append(m_k <= C)
            
        # Objective
        objective = Minimize(C)
        
        # Solve
        prob = Problem(objective, constraints)
        
        # Try CBC first (free solver)
        try:
            prob.solve(solver='CBC')
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_opt = h.value
                c5_bound = prob.value * dx
                print(f"MILP result: C5 = {c5_bound}")
            else:
                print(f"MILP failed with status: {prob.status}")
                h_opt = None
        except Exception as e:
            print(f"MILP solver failed: {e}")
            h_opt = None
    else:
        h_opt = None
        
    # If MILP failed or was not attempted, fallback to LP relaxation + rounding + polishing
    if h_opt is None:
        print("Falling back to LP relaxation + rounding")
        h_relax = Variable(n_points, continuous=True)
        C = Variable()
        
        # LP relaxation
        constraints = [cp_sum(h_relax) == target_sum, 0 <= h_relax, h_relax <= 1, C >= 0]
        
        # For now, simplified restrictions (some manual pre-processing can help)
        # We'll unfortunately have to compute the mismatch terms manually with index tricks
        # to avoid iterating over all shift values, as CVXPY doesn't allow us to cleanly write full matrix logic inline.
        # Instead, we will precompute alignment matrices, but that's not easy here.
        
        # However, we can give a more conservative approach:
        # Let's not use CVXPY's high-level indexing yet; instead, we'll use a hybrid.
        # We'll use an LP relaxation but use a custom generation algorithm to get good initial point.
        # More problematic than that — re-writing this is best achieved with a BREAKTHROUGH,
        # which is to formulate the total constraints as an LP in the form:
        # minimize max_k (sum h_i (1 - h_{i+k}))
        # We can model this as a standard LP after introducing auxiliary variables.
        
        # This is difficult to model directly in cvxpy with expensive indexing; risk going into large RAM usage.
        
        # So we'll skip to a simpler path with fixed construction and resort to refinements.
        
        h_opt = None
        h0 = np.zeros(n_points)
        h0[0 : n_points // 2] = 1
        np.random.shuffle(h0)
        h0 = np.clip(h0, 0, 1)
        h0 = np.array(h0)
        # Ensure sum = n_points / 2
        if np.abs(np.sum(h0) - target_sum) > 0.1:
            h0 = np.clip(np.random.rand(n_points), 0, 1)
            h0 = project_to_sum_box(h0, target_sum)
        
        # Steal the best smart start from `initial_h_values` if available
        from initial_h_values import initial_h_values
        try:
            h_init = np.array(initial_h_values)
            if len(h_init) == n_points:
                h0 = h_init.copy()
                print("Using initial_h_values as starting point")
        except:
            pass
        
        c5_min = float('inf')
        best_h = h0.copy()
        
        # Iteratively refine using simple swaps and hill-climbing
        for n_iter in range(200):
            if time.time() > start_time + 0.6 * budget_s:
                break
            # Swap a 1 and a 0 (locally)
            indices_1 = np.where(best_h == 1)[0]
            indices_0 = np.where(best_h == 0)[0]
            if len(indices_1) < 1 or len(indices_0) < 1:
                continue
            i1 = np.random.choice(indices_1)
            i0 = np.random.choice(indices_0)
            h_new = best_h.copy()
            h_new[i1] = 0
            h_new[i0] = 1
            c5_new = evaluate_c5(h_new)
            if c5_new < c5_min:
                best_h = h_new.copy()
                c5_min = c5_new
                
        h_opt = best_h
        c5_bound = c5_min
    
    # Final refinement layer (precise) using FFT + lazy local search  
    h_final = h_opt.copy()
    last_c5 = evaluate_c5(h_final)
    
    # Iterate for further improvement
    max_refinement_iter = 100
    for i in range(max_refinement_iter):
        if time.time() > start_time + budget_s:
            break
        indices_1 = np.where(h_final == 1)[0]
        indices_0 = np.where(h_final == 0)[0]
        if len(indices_1) < 1 or len(indices_0) < 1:
            break
        i1 = np.random.choice(indices_1)
        i0 = np.random.choice(indices_0)
        h_test = h_final.copy()
        h_test[i1] = 0
        h_test[i0] = 1
        c5_test = evaluate_c5(h_test)
        if c5_test < last_c5:
            h_final = h_test
            last_c5 = c5_test
    
    c5_final = last_c5 * dx  # Rescale to match original dx
    
    return (h_final.tolist(), float(c5_final), n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Clip x in-place
    x = np.clip(x, 0, 1)
    
    # Use a simple projection iteration: adjust by small amounts
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result
