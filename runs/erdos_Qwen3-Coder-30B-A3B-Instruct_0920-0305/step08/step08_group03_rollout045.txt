import numpy as np

n_points = 120

def generate_critical_shifts(intervals):
    """Generate all critical shifts from interval boundaries"""
    s = [0]
    for l in intervals:
        s.append(s[-1] + l)
    
    shifts = set()
    for i in range(len(s)):
        for j in range(len(s)):
            # |s_i - s_j| for all pairs
            shifts.add(abs(s[i] - s[j]))
            
            # ensure that the shift is in [0, 2]
            if s[i] < 2 and s[j] < 2:
                shifts.add(abs((2 - s[i]) - (2 - s[j])))
            
            # include endpoints if they are in range
            if i < len(s) - 1:
                shifts.add(s[i])
                shifts.add(2 - s[i])
            if j < len(s) - 1:
                shifts.add(s[j])
                shifts.add(2 - s[j])

    return sorted(shifts)

def evaluate_h_on_intervals(h_vals, interval_lengths, interval_heights, k):
    """Fast evaluation of the integral for a given k"""

    h_func = lambda x: np.interp(x, np.concatenate([[0], np.cumsum(interval_lengths)]),

    # Accelerated computation of piecewise overlaps for given k
    total_mass = 0
    total_product = 0
    
    # Evaluate h in [0, 2-k]
    # In practice this equates to integrating h(x) over [0, 2-k]
    for i, (length, height) in enumerate(zip(interval_lengths, interval_heights)):
        s_start = np.sum(interval_lengths[:i])
        s_end = s_start + length
        # Consider overlap [max(0, s_start), min(2-k, s_end)]
        overlap_start = max(0, s_start)
        overlap_end = min(2 - k, s_end)
        if overlap_end > overlap_start:
            total_mass += height * (overlap_end - overlap_start)

    # Evaluate h(x) * (1 - h(x+k)) 
    # This is over [0, 2-k] but we also need to compute h(x+k) part
    # For a fixed k, we need total integral over [0, 2-k]
    # Over each interval i_interval, we compute where the shifted version h(x+k)
    # overlaps with h(x). That's where x in [0, 2-k] and (x+k) in [s_j, s_{j+1})
    # which means x in [0, 2-k] AND x in [s_j - k, s_{j+1} - k], so 
    # x in [max(0, s_j - k), min(2-k, s_{j+1} - k)]

    # Key optimization: only one pass over all intervals
    for i, (length, height) in enumerate(zip(interval_lengths, interval_heights)):
        s_start = np.sum(interval_lengths[:i])
        s_end = s_start + length
        
        for j, (length2, height2) in enumerate(zip(interval_lengths, interval_heights)):
            s_start2 = np.sum(interval_lengths[:j]) 
            s_end2 = s_start2 + length2
            
            # Shifting interval j by k
            shift_start = s_start2 + k
            shift_end = s_end2 + k
            
            # Overlap of [s_start, s_end] with [shift_start, shift_end] -> [0, 2-k]
            overlap_start = max(s_start, shift_start, 0)
            overlap_end = min(s_end, shift_end, 2-k)
            if overlap_end > overlap_start:
                total_product += height * height2 * (overlap_end - overlap_start)
                
    return total_mass - total_product

def evaluate_c5_general(h_vals, shifts):
    """Evaluate C5 by computing max over critical points"""
    # Step function is defined over n_points, but processed in intervals
    # If we're in 'piecewise' mode, h_vals represent heights and intervals 
    # For direct integration query, convert step function to piecewise
    
    # extract number of possible intervals
    m = len(h_vals) // 2
    interval_heights = h_vals[:m]
    interval_lengths = h_vals[m:]
    
    # Ensure last length is determined correctly
    total_len = sum(interval_lengths)
    if abs(total_len - 2.0) > 1e-10:
        # re-normalize so that last interval's length makes sum = 2
        scale = 2.0 / total_len
        interval_lengths = [l * scale for l in interval_lengths]
    
    # Final list of critical shifts
    critical_shifts = generate_critical_shifts(interval_lengths)
    
    # Evaluate J(k) only over critical shifts
    Js = []
    for k in critical_shifts:
        js = evaluate_h_on_intervals(None, interval_lengths, interval_heights, k)
        Js.append(js)
    
    return max(Js)
