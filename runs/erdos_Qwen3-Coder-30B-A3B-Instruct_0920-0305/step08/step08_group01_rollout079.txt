import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_initial_solution():
    """Generate a better initial solution using a more systematic approach"""
    # Start with a periodic pattern
    h = np.zeros(n_points)
    
    # Create a pattern that resembles a sawtooth wave
    # with sharp transitions to help lower the maximum overlap
    
    # We start by placing a few steep drops to create a pattern
    # that reduces overlap significantly
    step_size = n_points // 6
    for i in range(0, n_points, step_size):
        if i + 1 < n_points:
            # Create a sharp transition
            h[i] = 1.0
            if i + 1 < n_points:
                h[i+1] = 0.0
        else:
            h[i] = 1.0
    
    # Randomize the remaining points
    for i in range(n_points):
        if h[i] == 0.0:
            h[i] = random.uniform(0, 0.5)
        elif h[i] == 1.0:
            h[i] = random.uniform(0.5, 1.0)
    
    # Project to correct sum
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def evolutionary_search():
    """Use evolutionary algorithm to search for better solutions"""
    population_size = 30
    generations = 50
    mutate_prob = 0.1
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h = generate_initial_solution()
        population.append(h)
    
    best_solution = population[0].copy()
    best_c5 = evaluate_c5(best_solution)
    
    for gen in range(generations):
        # Evaluate fitness (lower C5 is better)
        fitness = []
        for h in population:
            c5 = evaluate_c5(h)
            fitness.append(c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        
        # Keep best solutions
        new_population = [population[i] for i in sorted_indices[:5]]
        
        # Create new children
        for _ in range(population_size - 5):
            # Tournament selection
            parent1 = population[random.choice(sorted_indices[:10])]
            parent2 = population[random.choice(sorted_indices[:10])]
            
            # Crossover
            crossover_point = random.randint(1, n_points-1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            if random.random() < mutate_prob:
                mutation_point = random.randint(0, n_points-1)
                child[mutation_point] = random.uniform(0, 1)
            
            # Project to sum constraint
            child = project_to_sum_box(child, TARGET_SUM)
            
            new_population.append(child)
        
        population = new_population
        
        # Update best
        current_best = population[0]
        current_c5 = evaluate_c5(current_best)
        if current_c5 < best_c5:
            best_solution = current_best.copy()
            best_c5 = current_c5
            
    return best_solution, best_c5

def local_improvement(h, max_iter=300):
    """Local search improvement"""
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    for i in range(max_iter):
        # Try small modifications to several points
        h_current = best_h.copy()
        indices = random.sample(range(n_points), min(15, n_points//4))
        
        for idx in indices:
            old_val = h_current[idx]
            # Try several small steps
            steps = [-0.03, -0.01, 0.01, 0.03]
            for step in steps:
                new_val = old_val + step
                if 0 <= new_val <= 1:
                    h_current[idx] = new_val
                    h_current = project_to_sum_box(h_current, TARGET_SUM)
                    new_c5 = evaluate_c5(h_current)
                    if new_c5 < best_c5:
                        best_h = h_current.copy()
                        best_c5 = new_c5
                        break
            else:
                # No improvement found, revert
                h_current[idx] = old_val
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points / 2
    
    # Use evolutionary search
    best_h, best_c5 = evolutionary_search()
    
    # Local improvement
    final_h, final_c5 = local_improvement(best_h)
    
    # Try some additional local improvements
    while time.time() - start_time < 0.8 * budget_s:
        # Quick refinement
        perturbed = final_h + np.random.normal(0, 0.005, n_points)
        perturbed = project_to_sum_box(perturbed, TARGET_SUM)
        refined_h, refined_c5 = local_improvement(perturbed)
        
        if refined_c5 < final_c5:
            final_h = refined_h
            final_c5 = refined_c5
    
    return (final_h.tolist(), float(final_c5), n_points)
