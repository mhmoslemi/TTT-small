def compute_overlap_for_shift(a, lengths, k):
    """Compute the overlap integral for a specific shift k"""
    total_length = 2.0
    if k >= total_length:
        return 0.0
    
    # Compute the part of h(x) * (1 - h(x + k)) for given k
    # Each interval contributes based on its overlap with shifted interval
    intervals = []
    s_prev = 0.0
    for i in range(len(a)):
        s_curr = s_prev + lengths[i]
        intervals.append((s_prev, s_curr, a[i]))
        s_prev = s_curr

    # Overlap area for h(x) * (1 - h(x + k)) over domain [0, 2-k]
    integral = 0.0
    for s_l, s_r, h_val in intervals:
        if s_r <= k or s_l >= total_length - k:
            continue
        # Find intersection of [s_l, s_r] with [k, 2-k] = [k, 2-k]
        t_start = max(s_l, k)
        t_end = min(s_r, total_length - k)
        if t_start < t_end:
            # Overlap length with shifted interval
            overlap_length = t_end - t_start
            integral += h_val * overlap_length

    # Now compute the product h(x) * (1 - h(x + k))
    integral2 = 0.0
    # For each pair of intervals (p,q), we compute how much they intersect
    for i, (s_l1, s_r1, h1_val) in enumerate(intervals):
        s1_start = max(s_l1, k)
        s1_end = min(s_r1, total_length - k)
        if s1_start >= s1_end:
            continue
        for j, (s_l2, s_r2, h2_val) in enumerate(intervals):
            s2_start = max(s_l2, 0)
            s2_end = min(s_r2, total_length - k + k)
            if s2_start >= s2_end:
                continue
            # We compute integral h1(x) * h2(x + k)
            overlap_len = max(0, min(s1_end, s2_end) - max(s1_start, s2_start))
            if overlap_len > 0:
                integral2 += h1_val * h2_val * overlap_len

    return integral - integral2

def evaluate_all_critical_shifts(a, lengths):
    """Evaluate the maximum overlap over critical shifts"""
    critical_shifts = []
    # Critical shifts: 0, and all endpoints of intervals + k
    # s_i values:
    s = [0.0]
    for l in lengths:
        s.append(s[-1] + l)
    
    # There are at most 2*len(intervals) + 1 critical shifts
    # But let's enumerate over all zero crossings, considering 'k' as congruent to 0
    # interval boundaries and zero-interval borders should be present
    for i, s1 in enumerate(s):
        if s1 < 2:
            critical_shifts.append(s1)
            if i > 0:
                critical_shifts.append(s1 - s[i-1])  # shifts between intervals
            if i + 1 < len(s):
                critical_shifts.append(s1 + (s[i+1] - s1) / 2)

    # Also check critical values
    critical_shifts.extend([0.0, 1.0, 2.0])  # essential fractions

    # Evaluate all overlaps for each critical shift
    max_overlap = 0.0
    for k in critical_shifts:
        if k < 0 or k > 2:
            continue
        val = compute_overlap_for_shift(a, lengths, k)
        if val > max_overlap:
            max_overlap = val
    return max_overlap

def parametric_objective(vars):
    """Evaluate the objective function for a parametrized step function"""
    n = n_intervals
    # vars[0..n-1] are intervals lengths
    # vars[n..2*n-1] are heights
    lengths = vars[0:n]
    a = vars[n:2*n]
    
    total_len = sum(lengths)
    if abs(total_len - 2.0) > 1e-6:
        # force a flat penalty
        return 1e12
    
    # Normalize heights to satisfy mass constraint
    norm_factor = 2.0 / sum(a[j] * lengths[j] for j in range(n))
    a_norm = [a[j] * norm_factor for j in range(n)]
    
    # Evaluate all critical overlaps
    max_overlap = evaluate_all_critical_shifts(a_norm, lengths)
    return max_overlap

def project_to_constraints(vars):
    """Project vector onto feasible domain"""
    n = n_intervals
    lengths = np.array(vars[0:n])
    a = np.array(vars[n:2*n])
    # Project lengths to [0, 2] and sum to 2.0
    lengths = np.clip(lengths, 0.0, 2.0)
    lengths *= (2.0 / np.sum(lengths))
    # Project a to [0, 1]
    a = np.clip(a, 0.0, 1.0)
    return np.concatenate([lengths, a])
