import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from scipy.optimize import differential_evolution

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    h0 = adaptive_sampling_initialization()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use differential evolution for global optimization
    def objective(h):
        return evaluate_c5(h)
    
    # Constrained optimization using differential evolution
    de_result = differential_evolution(
        objective,
        bounds,
        maxiter=500,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        tol=1e-6,
        callback=lambda x, convergence: None
    )
    
    if de_result.success:
        h_de = project_to_sum_box(de_result.x, TARGET_SUM)
        c5_de = evaluate_c5(h_de)
        if c5_de < best_c5:
            best_h = h_de.copy()
            best_c5 = c5_de
    
    # Fine-tune using local optimization
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Run several local optimizations from different starting points
    for i in range(5):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Random perturbation of the best solution
        h_perturbed = best_h + np.random.normal(0, 0.02, n_points)
        h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
        
        try:
            res = minimize(
                objective, 
                h_perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8}
            )
            if res.success:
                h_local = project_to_sum_box(res.x, TARGET_SUM)
                c5_local = evaluate_c5(h_local)
                if c5_local < best_c5:
                    best_h = h_local.copy()

    for _ in range(100):

        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
    
    return (best_h.tolist(), float(best_c5), n_points)
