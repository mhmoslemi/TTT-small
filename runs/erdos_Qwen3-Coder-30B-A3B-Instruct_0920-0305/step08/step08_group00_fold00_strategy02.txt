```
def run(seed=42, budget_s=600, **kwargs):

    np.random.seed(seed)
    start = time.time()
    n = 200
    dx = 2.0 / n
    # Helper functions:
    def max_mismatches(h):
        # Use FFT correlation for speed
        h = h.astype(np.float64)
        b = 1 - h
        rev = h[::-1]
        L = 2*n - 1
        corr = np.fft.ifft(np.fft.fft(rev, L) * np.fft.fft(b, L)).real
        # rounding
        corr_int = np.rint(corr).astype(int)
        return int(corr_int.max())
    # candidate best
    best_h = None
    best_max = n  # high
    # allocate time for SA
    time_for_sa = budget_s * 0.85  # remainder for final refinement
    # SA parameters
    # We want to run many restarts
    while time.time() - start < time_for_sa:
        # random initial
        h = np.zeros(n, dtype=int)
        h[:n//2] = 1
        np.random.shuffle(h)
        cur_h = h.copy()
        cur_max = max_mismatches(cur_h)
        # track best globally
        if cur_max < best_max:
            best_max = cur_max
            best_h = cur_h.copy()
        # temperature schedule
        T0 = 5.0
        T = T0
        cooling = 0.9998
        # number of steps per SA session depending on remaining time
        steps = 0
        while time.time() - start < time_for_sa and steps < 20000:
            # random swap
            ones_idx = np.where(cur_h == 1)[0]
            zeros_idx = np.where(cur_h == 0)[0]
            i_one = np.random.choice(ones_idx)
            i_zero = np.random.choice(zeros_idx)
            # perform swap
            cur_h[i_one] = 0
            cur_h[i_zero] = 1
            new_max = max_mismatches(cur_h)
            delta = new_max - cur_max
            accept = False
            if delta <= 0:
                accept = True
            else:
                # Metropolis acceptance using temperature in mismatches units
                prob = math.exp(-delta / max(T, 1e-12))
                if np.random.rand() < prob:
                    accept = True
            if accept:
                cur_max = new_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = cur_h.copy()
                # keep swapped arrangement
            else:
                # revert swap
                cur_h[i_one] = 1
                cur_h[i_zero] = 0
            # cooling
            T *= cooling
            steps += 1
```

The problem is a *min‑max* optimisation:

\[
C_5=\max_k\int_0^2 h(x)\bigl(1-h(x+k)\bigr)\,dx ,
\qquad
h:[0,2]\to[0,1],\;
\int_0^2h=1 .
\]

With a uniform discretisation (`n` points, `dx=2/n`) the integral becomes  

\[
C_5 =dx\;\max_{k}\; \sum_{i=0}^{n-1} h_i\bigl(1-h_{i+k}\bigr) .
\]

If we restrict ourselves to *step* functions with values **0** or **1** (the
optimal solution is known to be attained at an extreme point of the feasible
polytope) the inner sum simply counts the *mismatches* for a given shift

\[
m_k = \#\{\,i\;|\;h_i=1,\;h_{i+k}=0\}.
\]

Hence  

\[
C_5 =dx\; \max_k m_k .
\]

For a fixed number of points `n` the target bound
`C₅ ≤ 0.3807` is equivalent to a bound on the integer mismatches

\[
m_{\max}\le M_{\text{target}}=
\Bigl\lfloor\frac{0.3807\,n}{2}\Bigr\rfloor .
\]

For `n=200` we have `dx=0.01` and `M_target = 38`.  
The task is therefore **find a binary vector of length 200 with exactly 100
ones such that the maximal mismatch over all shifts does not exceed 38**.

The search is performed in three stages:

1. **Fast evaluation** – mismatches are obtained with a single FFT
   (`h_rev` * (1‑h)) and rounding, i.e.  
   `max_mismatch(h) = max( round( ifft( fft(h_rev)*fft(1‑h) ) ) )`.

2. **Simulated annealing** – many random *swap* moves (exchange a `1` and a
   `0`) are tried.  The Metropolis rule accepts any move that lowers the
   current maximum mismatch, otherwise it accepts with probability
   `exp(-Δ/T)`.  Temperature decays geometrically.

3. **Greedy hill‑climbing** – after the annealing phase we repeatedly try
   random swaps and keep any that strictly improve the maximum mismatch.
   This deterministic polishing phase finishes the optimisation.

All phases are limited by the global time budget (`budget_s`).  Multiple
restarts are automatically performed while time permits, and the best
solution seen throughout the run is returned.

The implementation uses only NumPy, `time` and `math`, respects the
required function signature, and works entirely in memory.

```python
import numpy as np
import time
import math

# ----------------------------------------------------------------------
#  Helper functions – all are top‑level (no closures, no lambdas)
# ----------------------------------------------------------------------


def max_mismatch(h: np.ndarray) -> int:
    """Return max_k  Σ_i h_i (1‑h_{i+k})  for a binary vector h."""
    # h is 0/1 integer array
    n = h.shape[0]
    # reverse h and compute convolution with (1‑h) via FFT
    h_rev = h[::-1].astype(np.float64)
    b = (1 - h).astype(np.float64)
    L = 2 * n - 1
    # FFT‑based convolution (real → real)
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    # rounding eliminates tiny floating errors
    max_val = int(np.rint(corr).max())
    return max_val


def c5_from_h(h: np.ndarray) -> float:
    """Compute the C5 value for the binary step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    return dx * max_mismatch(h)


def random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random binary vector with exactly n//2 ones."""
    h = np.zeros(n, dtype=int)
    h[: n // 2] = 1
    rng.shuffle(h)
    return h


def greedy_improve(
    h: np.ndarray,
    cur_max: int,
    rng: np.random.Generator,
    max_iters: int = 500,
) -> tuple[np.ndarray, int]:
    """
    Simple hill‑climbing: try random swaps, keep any that lower the current max.
    """
    n = h.shape[0]
    for _ in range(max_iters):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i_one = rng.choice(ones)
        i_zero = rng.choice(zeros)

        # perform swap
        h[i_one] = 0
        h[i_zero] = 1

        new_max = max_mismatch(h)
        if new_max < cur_max:
            cur_max = new_max
            # keep the improvement
        else:
            # revert
            h[i_one] = 1
            h[i_zero] = 0
    return h, cur_max


def anneal(
    n: int,
    rng: np.random.Generator,
    time_limit: float,
    target_max: int,
) -> tuple[np.ndarray, int]:
    """
    Simulated‑annealing search for a binary vector with a small maximal mismatch.
    Returns the best vector found and its max_mismatch value.
    """
    start = time.time()
    best_h = None
    best_max = n  # worst possible

    # ----------- main loop – several independent restarts ------------
    while time.time() - start < time_limit:
        # initialise a random feasible vector
        cur_h = random_binary(n, rng)
        cur_max = max_mismatch(cur_h)

        # update global best
        if cur_max < best_max:
            best_h, best_max = cur_h.copy(), cur_max
            if best_max <= target_max:
                return best_h, best_max

        # annealing parameters
        T = 5.0                     # initial temperature (in mismatch units)
        cooling = 0.9995            # geometric cooling factor
        # inner SA loop – stop either when time runs out or after a fixed #steps
        steps = 0
        max_steps = 25000
        while time.time() - start < time_limit and steps < max_steps:
            # pick a random 1↔0 swap
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            i_one = rng.choice(ones)
            i_zero = rng.choice(zeros)

            # apply swap
            cur_h[i_one] = 0
            cur_h[i_zero] = 1
            new_max = max_mismatch(cur_h)
            delta = new_max - cur_max

            accept = False
            if delta <= 0:
                accept = True
            else:
                prob = math.exp(-delta / max(T, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_max = new_max
                if cur_max < best_max:
                    best_h, best_max = cur_h.copy(), cur_max
                    if best_max <= target_max:
                        return best_h, best_max
            else:
                # revert swap
                cur_h[i_one] = 1
                cur_h[i_zero] = 0

            # temperature schedule
            T *= cooling
            steps += 1

        # a short greedy polish for the current restart
        cur_h, cur_max = greedy_improve(cur_h, cur_max, rng, max_iters=300)
        if cur_max < best_max:
            best_h, best_max = cur_h.copy(), cur_max
            if best_max <= target_max:
                return best_h, best_max

    return best_h, best_max


# ----------------------------------------------------------------------
#  Main entry point required by the competition framework
# ----------------------------------------------------------------------


def run(seed: int = 42, budget_s: float = 600, **kwargs):
    """
    Return a step‑function (as a list of floats), its C5 value and the number of
    discretisation points.
    """
    rng = np.random.default_rng(seed)

    # Choose a modest size – 200 points give dx = 0.01 and a target of 38 mismatches.
    n = 200
    dx = 2.0 / n
    target_max = int(math.floor(0.3807 * n / 2))   # 38 for n=200

    # Allocate 85 % of the budget to the annealing phase, the rest to final polishing
    t0 = time.time()
    anneal_time = budget_s * 0.85
    best_h, best_max = anneal(n, rng, anneal_time, target_max)

    # Final deterministic improvement (still respecting the overall budget)
    remaining = budget_s - (time.time() - t0)
    if remaining > 0.0:
        best_h, best_max = greedy_improve(best_h, best_max, rng, max_iters=2000)

    # Compute final C5 value
    final_c5 = dx * best_max

    # Convert to the required output format (list of floats in [0,1])
    h_values = best_h.astype(float).tolist()
    return (h_values, float(final_c5), n)
```
