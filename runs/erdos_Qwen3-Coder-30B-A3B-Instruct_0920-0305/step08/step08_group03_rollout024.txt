import numpy as np
from scipy.optimize import minimize
from itertools import product
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
m = 5  # number of intervals

def compute_J(k, L, A):
    """Compute J(k; L, A) = int_0^{2-k} h(x)(1-h(x+k)) dx"""
    # Build intervals
    s = [0] + list(np.cumsum(L))
    # Compute linear term: int h(x) dx over [0, 2-k]
    linear = 0.0
    k_end = 2 - k
    for i in range(len(L)):
        if s[i] >= k_end:
            break
        a = A[i]
        left = max(s[i], 0)
        right = min(s[i+1], k_end)
        if right > left:
            linear += a * (right - left)
    
    # Compute quadratic term: sum a_p * a_q * omega_{pq}(k)
    quadratic = 0.0
    for p in range(len(L)):
        for q in range(len(L)):
            a_p = A[p]
            a_q = A[q]
            left_p = s[p]
            right_p = s[p+1]
            left_q = s[q] + k
            right_q = s[q+1] + k
            overlap_left = max(left_p, left_q)
            overlap_right = min(right_p, right_q)
            if overlap_right > overlap_left:
                omega = overlap_right - overlap_left
                quadratic += a_p * a_q * omega
                
    return linear - quadratic

def compute_critical_shifts(s):
    """Given interval endpoints s, return critical shift values"""
    K = set()
    for i, j in product(range(len(s)), repeat=2):
        K.add(abs(s[i] - s[j]))
        K.add(abs(2 - s[i] - (2 - s[j])))
    K.add(0.0)
    K.add(2.0)
    # Filter to [0,2]
    K = [k for k in K if 0 <= k <= 2]
    return sorted(list(set(K)))

def get_constraints(L, A):
    """Return list of constraints on L and A"""
    constraints = []
    # Lengths are positive
    for l in L:
        constraints.append({'type': 'ineq', 'fun': lambda x: x})
    # A_j are between 0 and 1
    for a in A:
        constraints.append({'type': 'ineq', 'fun': lambda x: x})
        constraints.append({'type': 'ineq', 'fun': lambda x: 1 - x})
    # Sum of a_j * l_j = 1
    def mass_constraint(x):
        a = x[m-1:m*2-1]  # Get the first m-1 heights from x
        l = x[0:m-1]      # Get first m-1 lengths from x
        last_a = (1 - sum(a[i]*l[i] for i in range(len(l)))) / (2 - sum(l)) if sum(l) < 2 else (1 - sum(a[i]*l[i] for i in range(len(l))))
        # Clamp last_a to [0, 1] to avoid numerical issues
        last_a = max(0, min(1, last_a))
        return 1 - sum(a[i]*l[i] for i in range(len(l))) - last_a*(2 - sum(l))
    constraints.append({'type': 'eq', 'fun': mass_constraint})
    return constraints

def evaluate_c5_from_params(L, A):
    """Evaluate the full C5 from piecewise constant parameters"""
    # Compute s = cumulative sums (interval endpoints)
    s = [0] + list(np.cumsum(L))
    # Compute critical shifts
    K = compute_critical_shifts(s)
    # Evaluate J(k) for each k
    J_vals = [compute_J(k, L, A) for k in K]
    return max(J_vals)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_L_a = None
    best_C5 = float('inf')
    
    # Precompute critical shifts for m=5 (smaller m, faster)
    # This is safe to do once since number of intervals is fixed
    s = [0, 1, 1, 1, 1, 2]  # default start with an efficient layout
    K = compute_critical_shifts(s)
    
    # Reuse a single constraint manager and minimize function setup overhead
    
    def objective(x):
        # x = [L1, L2, ..., L_{m-1}, A1, A2, ..., A_{m-1}]
        # Last elements are derived from constraints
        L = x[:m-1]
        A = x[m-1:2*m-2]
        
        # Derive last length and height based on constraints
        sum_L = sum(L)
        if sum_L >= 2:  # Don't let it be >=2, even if unintentional
            L = [max(1e-6, l*2/sum_L) for l in L]
            sum_L = 2.0
            
        last_L = 2.0 - sum_L
        if last_L <= 0:
            last_L = 1e-6
            
        # Derive last A from mass constraint
        sum_weights = sum(A[i] * L[i] for i in range(len(L)))
        if sum_weights >= 1:
            A = [max(0, min(1, a*1/sum_weights)) for a in A]
            sum_weights = 1.0
        last_A = (1.0 - sum_weights) / last_L if last_L > 0 else 0.0
        last_A = max(0, min(1, last_A))
    
        A_full = A + [last_A]
        L_full = L + [last_L]
        
        # Circuitously compute J(k) for all critical shifts
        J_vals = [compute_J(k, L_full, A_full) for k in K]
        return max(J_vals)

    def constraint_eq(x):
        L = x[:m-1]
        A = x[m-1:2*m-2]
        # Derive last elements
        sum_L = sum(L)
        if sum_L >= 2:  # Prevent edge case
            sum_L = 2.0 - 1e-9
        last_L = 2.0 - sum_L
        if last_L <= 0:
            last_L = 1e-6
        sum_weights = sum(A[i]*L[i] for i in range(len(L)))
        last_A = (1.0 - sum_weights) / last_L if last_L > 0 else 0.0
        last_A = max(0, min(1, last_A))
        # Constraint on masses
        return 1.0 - (sum_weights + last_A * last_L)  # Should be zero

    # Optimizer settings
    constraints = [{'type': 'eq', 'fun': constraint_eq}]
    
    bounds = []
    # Bounds for lengths [0,2] (prevent bad values)
    for _ in range(m-1):
        bounds.append((1e-9, 2.0 - 1e-9))
    # Bounds for heights [0,1]
    for _ in range(m-1):
        bounds.append((1e-9, 1.0 - 1e-9))
        
    # Use multiple restarts
    num_restarts = 50
    best_sol = None
    best_val = float('inf')
    
    for i in range(num_restarts):
        # Random initial guess
        L0 = np.random.uniform(0.2, 0.6, m-1)
        A0 = np.random.uniform(0.2, 0.8, m-1)
        x0 = np.concatenate([L0, A0])
        try:
            res = minimize(
                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                val = res.fun
                if val < best_val:
                    best_val = val
                    best_sol = res.x
        except:
            continue

        # Meditate on time remaining
        if time.time() - start_time > budget_s * 0.9:
            break

    # Build the final solution (now using best_sol)
    best_L = best_sol[:m-1]
    best_A = best_sol[m-1:2*m-2]
    
    # Derive last elements
    sum_L = sum(best_L)
    if sum_L >= 2:
        sum_L = 2.0 - 1e-9
    last_L = 2.0 - sum_L
    if last_L <= 0:
        last_L = 1e-6
    sum_weights = sum(best_A[i]*best_L[i] for i in range(len(best_L)))
    last_A = (1.0 - sum_weights) / last_L if last_L > 0 else 0.0
    last_A = max(0, min(1, last_A))
    
    final_L = best_L + [last_L]
    final_A = best_A + [last_A]

    # Re-evaluate the system with proper handling of uncertainties
    c5_final = evaluate_c5_from_params(final_L, final_A)
    
    # Generate step function from the final solution
    step_h = []
    s = [0] + list(np.cumsum(final_L))
    for i in range(len(final_L)):
        h_val = final_A[i]
        num_points = int(round(final_L[i]/2.0 * n_points))
        step_h.extend([h_val]*num_points)
    
    # Normalize
    total_mass = sum(step_h) * dx
    step_h = [h * (1.0 / total_mass) for h in step_h]
    
    # Iterate to further optimize if time remains
    time_remaining = budget_s - (time.time() - start_time)
    if time_remaining > 10 and c5_final > 0.379:
        # Use to interpolate and fine tune 120-point series
        # Create fine grid to be compatible with the evaluator
        x_fine = np.linspace(0, 2, n_points)
        h_fine = np.array(step_h)
        # Small incremental optimization for final polish
        from scipy.optimize import minimize as minimize_scipy
        
        def corr_obj(h):
            # see how close it is to not violating the constraint
            corr = np.correlate(h, 1-h, mode="full") * dx
            return np.max(corr)
        
        # Reset as vector, apply parametric interpolater
        h_step = np.array(step_h)
        bounds = [(0, 1) for _ in range(n_points)]
        
        def obj_fn(h):
            return corr_obj(h)
            
        try:
            res = minimize(
                obj_fn,

                options={'maxiter': 100}
            )
            if res.success:
                h_final = res.x
                c5_final = corr_obj(h_final)
        except:
            pass

    # Return h, c5, and n_points
    return (h_final.tolist(), float(c5_final), n_points)
