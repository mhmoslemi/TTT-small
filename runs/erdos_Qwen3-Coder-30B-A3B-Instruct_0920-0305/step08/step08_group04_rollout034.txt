import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import time

# Global variables
n_points = 300
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling projection
    sorted_x = np.sort(x)[::-1]
    cumsum = np.cumsum(sorted_x)
    n = len(x)
    # Find the threshold
    threshold_idx = -1
    for i in range(n):
        if cumsum[i] <= target_sum:
            threshold_idx = i
        else:
            break
    if threshold_idx == -1:
        # All elements are above threshold
        threshold = sorted_x[0] + (target_sum - cumsum[0]) / n
    else:
        if threshold_idx == n - 1:
            threshold = (target_sum - cumsum[threshold_idx]) / (n - threshold_idx - 1)
        else:
            threshold = (target_sum - cumsum[threshold_idx]) / (n - threshold_idx - 1)
    
    # Apply threshold
    result = np.clip(x - threshold, 0, 1)
    # Adjust to enforce exact sum
    current_sum = np.sum(result)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Adjust by distributing the difference
        indices = np.argsort(result)
        for i in range(int(abs(diff) / 1e-10)):
            idx = indices[i % len(indices)]
            if diff > 0:
                if result[idx] < 1:
                    result[idx] = min(result[idx] + 1e-10, 1)
            else:
                if result[idx] > 0:
                    result[idx] = max(result[idx] - 1e-10, 0)
    
    return result

def rasterise_and_project(θ):
    """Map parametric θ to a discrete h with sum constraint"""
    K = 8  # number of blocks
    widths = np.exp(θ[:K])
    widths = 2 * widths / np.sum(widths)
    
    heights = 1.0 / (1.0 + np.exp(-θ[K:]))
    
    # Create h
    h = np.zeros(n_points)
    
    start = 0
    for i in range(K):
        end = int(start + round(widths[i] / dx))
        if end > n_points:
            end = n_points
        h[start:end] = heights[i]
        start = end
        if start >= n_points:
            break
    
    # Project onto sum-box
    h = project_to_sum_box(h, TARGET_SUM)
    
    return h

def obj(θ):
    """Objective function for DE"""
    h = rasterise_and_project(θ)
    c5 = evaluate_c5(h)
    return c5

def compute_gradient(h, corr, active_shifts):
    """Compute sub-gradient for current h and active shifts"""
    g = np.zeros(len(h))
    n = len(h)
    
    # Subtract one over all elements
    g += -dx
    
    for shift in active_shifts:
        if shift >= 0:
            g[:n-shift] += -dx * h[shift:]
            g[shift:] += -dx * h[:n-shift]
        else:
            t = -shift
            g[t:] += -dx * h[:n-t]
            g[:n-t] += -dx * h[t:]
    
    # Average over active shifts
    g /= len(active_shifts)
    
    return -g

def psgd_optimize(h_initial, t0, budget_s):
    """Perform projected sub-gradient descent"""
    h = h_initial.copy()
    c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    eta = 0.5
    eta_max = 2.0
    no_imp = 0
    max_no_imp = 200
    swap_interval = 30
    
    corr = np.real(np.fft.ifft(np.fft.fft(h[::-1]) * np.fft.fft(1-h))) * dx
    max_val = corr.max()
    active = np.where(corr >= max_val - 1e-12)[0]
    shifts = active - (n_points - 1)
    
    # Convert to signed shifts
    for i in range(10000):
        # Compute updates
        if time.time() - t0 > budget_s - 10:
            break
            
        g = compute_gradient(h, corr, shifts)
        h_new = h - eta * g
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_imp = 0
            # Increase step size
            eta = min(eta * 1.05, eta_max)
        else:
            no_imp += 1
            # Decrease step size
            eta = max(eta * 0.7, 1e-4)
            
        h = h_new
        c5 = c5_new
        
        # Check for mass transfer swaps
        if i % swap_interval == 0 or no_imp > max_no_imp:
            # Perform mass transfer
            # Find valid indices
            valid_pos = np.where(h < 1)[0]
            valid_neg = np.where(h > 0)[0]
            
            if len(valid_pos) > 0 and len(valid_neg) > 0:
                i_pos = np.random.choice(valid_pos)
                i_neg = np.random.choice(valid_neg)
                # Transfer small amount
                amount = min(0.02, h[i_neg]) * np.random.random()
                h[i_neg] -= amount
                h[i_pos] += amount
                h = project_to_sum_box(h, TARGET_SUM)
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
                    
            no_imp = 0
            
        # Recompute correlation and active shifts
        corr = np.real(np.fft.ifft(np.fft.fft(h[::-1]) * np.fft.fft(1-h))) * dx
        max_val = corr.max()
        active = np.where(corr >= max_val - 1e-12)[0]
        shifts = active - (n_points - 1)
        
        if no_imp > max_no_imp:
            break
            
    return best_h, best_c5

def coarse_de_optimize(h_initial, t0, budget_s):
    """Run differential evolution on coarse grid"""
    # Coarse grid parameters
    n_coarse = 150
    dx_coarse = 2.0 / n_coarse
    target_sum_coarse = n_coarse / 2
    
    # Extract coarse version of initial guess
    h_coarse = np.interp(np.linspace(0, 2, n_coarse), np.linspace(0, 2, len(h_initial)), h_initial)
    h_coarse = project_to_sum_box(h_coarse, target_sum_coarse)
    
    # Run DE on coarse grid with early termination
    def obj_coarse(θ):
        K = 8
        widths = np.exp(θ[:K])
        widths = 2 * widths / np.sum(widths)
        heights = 1.0 / (1.0 + np.exp(-θ[K:]))
        
        # Create h
        h = np.zeros(n_coarse)
        start = 0
        for i in range(K):
            end = int(start + round(widths[i] / dx_coarse))
            if end > n_coarse:
                end = n_coarse
            h[start:end] = heights[i]
            start = end
            if start >= n_coarse:
                break
        
        # Project
        h = project_to_sum_box(h, target_sum_coarse)
        corr = np.real(np.fft.ifft(np.fft.fft(h[::-1]) * np.fft.fft(1-h))) * dx_coarse
        return np.max(corr)
    
    bounds = [(-5, 5)] * 16  # 8 widths + 8 heights
    
    def callback(θ, **kwargs):
        if time.time() - t0 > budget_s - 100:
            return True
        return False
            
    # Run DE with fewer iterations
    res = differential_evolution(
        obj_coarse, 
        bounds, 
        maxiter=120, 
        popsize=12, 
        disp=False,
        callback=callback
    )
    
    if not res.success:
        return h_initial
    
    # Reconstruct coarse solution
    h_coarse = rasterise_and_project(res.x)
    
    return h_coarse
