import numpy as np

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def update_correlation_delta(c, a, b, i, j, delta_i, delta_j):
    """Update correlation array after moving mass from i to j"""
    n = len(a)
    delta_c = np.zeros(2 * n - 1)
    
    # Handle delta_i from index i
    if delta_i != 0:
        # Update contribution from b[i+s] term 
        s_start_b = max(-i, -(n-1))
        s_end_b = min(n-1-i, n-1)
        for s in range(s_start_b, s_end_b + 1):
            if 0 <= i + s <= n - 1:
                delta_c[s + n - 1] += delta_i * b[i + s]
        
        # Update contribution from a[i-s] term
        s_start_a = max(i - (n-1), 0)
        s_end_a = i
        for s in range(s_start_a, s_end_a + 1):
            if 0 <= i - s <= n - 1:
                delta_c[s + n - 1] -= delta_i * a[i - s]

    # Handle delta_j from index j
    if delta_j != 0:
        # Update contribution from b[j+s] term
        s_start_b = max(-j, -(n-1))
        s_end_b = min(n-1-j, n-1)
        for s in range(s_start_b, s_end_b + 1):
            if 0 <= j + s <= n - 1:
                delta_c[s + n - 1] += delta_j * b[j + s]
        
        # Update contribution from a[j-s] term
        s_start_a = max(j - (n-1), 0)
        s_end_a = j
        for s in range(s_start_a, s_end_a + 1):
            if 0 <= j - s <= n - 1:
                delta_c[s + n - 1] -= delta_j * a[j - s]
    
    # Add cross product terms: delta_i * delta_j
    if delta_i != 0 and delta_j != 0:
        delta_ij = delta_i * delta_j  # This should be -delta_i^2
        s1 = j - i + n - 1  # shift where contribution comes
        s2 = i - j + n - 1
        if 0 <= s1 < 2 * n - 1:
            delta_c[s1] += delta_ij
        if 0 <= s2 < 2 * n - 1:
            delta_c[s2] += delta_ij

    return c + delta_c
