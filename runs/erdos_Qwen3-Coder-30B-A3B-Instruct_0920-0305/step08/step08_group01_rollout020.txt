import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def random_binary_solution():
    """Generate random binary solution with target sum ones"""
    h = np.zeros(n_points)
    ones_indices = np.random.choice(n_points, size=int(TARGET_SUM), replace=False)
    h[ones_indices] = 1.0
    return h

def simulated_annealing_binary(max_iter=20000):
    """Simulated annealing optimization for binary patterns"""
    # Initialize
    h = random_binary_solution()
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Temperature schedule
    initial_temp = 0.015
    final_temp = 0.0001
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > 550:  # Leave buffer for cleanup
            break
            
        # Temperature cooling schedule
        elapsed = time.time() - start_time
        temperature = initial_temp * (final_temp / initial_temp) ** (elapsed / 550)
        
        # Sample indices for swap
        ones_indices = np.where(h == 1)[0]
        zeros_indices = np.where(h == 0)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            continue
            
        i = np.random.choice(ones_indices)
        j = np.random.choice(zeros_indices)
        
        # Perform swap
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        
        # Compute new C5
        new_c5 = evaluate_c5(h_new)
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
        else:
            # Accept with probability
            if np.random.random() < np.exp((current_c5 - new_c5) / temperature):
                h = h_new
                current_c5 = new_c5
    
    return best_h, best_c5

def improve_solution_with_gradients(h0, max_iter=1000):
    """Use local search with gradient methods to improve solution"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Run multiple restarts of simulated annealing
    restart_count = 0
    max_restarts = 20  # Limit restarts to avoid too much time commitment
    
    while restart_count < max_restarts and (time.time() - start_time) < (budget_s * 0.95):
        h, c5 = simulated_annealing_binary(max_iter=10000)  # 10k iterations per restart
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        restart_count += 1
    
    # Final refinement if we have a good solution
    if best_h is not None and (time.time() - start_time) < (budget_s * 0.98):
        refined_h = improve_solution_with_gradients(best_h, max_iter=500)
        refined_c5 = evaluate_c5(refined_h)
        
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
