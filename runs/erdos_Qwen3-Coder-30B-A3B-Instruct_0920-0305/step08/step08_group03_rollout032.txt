def project_to_box_sum(h, target_sum):
    # Project onto box [0,1] with sum = target_sum using binary search on lambda
    h = np.asarray(h)
    lb, ub = -1.0, 1.0
    while True:
        mid = (lb + ub) / 2
        ph = np.clip(h + mid, 0, 1)
        s = np.sum(ph)
        if abs(s - target_sum) < 1e-12:
            return ph
        if s > target_sum:
            ub = mid
        else:
            lb = mid

    # Using full mode to match evaluator
    c = correlate(h, b, mode='full')
    return c

    # foreach indices l, s with corr = sum h[l]b[l-s]  
    # h[l] contributes to c[s]. For l = i, this is h[i] * b[i-s] 
    # If h[i] is changed by -delta  
    # Then corr update is: c[s] -= delta * b[i - s]  
    # For s with i - s in [0,n)

    # => index = i - 1 + (n-1) = i + n - 2

    # This is super dangerous to trust with e.g. results of invalid indexing  
    # Guard correct lower bound, higher bound based on confirmed slice  no ensure valid access

    # Slice i-n to i-1, (n elements)

    # [ i-1 to i+n-2 ]
