import numpy as np
from scipy.signal import correlate
import time

# Global constants
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Clip to [0,1]
    x_clipped = np.clip(x, 0, 1)
    
    # If already satisfied, return
    actual_sum = np.sum(x_clipped)
    if abs(actual_sum - target_sum) < 1e-12:
        return x_clipped
    
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlation_update(c, h, b, i, j, delta):
    """Efficiently compute correlation change from a mass transfer"""
    n = len(h)
    h_rev = h[::-1]
    
    # Compute the new correlation vector in-place
    c_new = c.copy()
    
    # Part 1: contribution from h[i] -> b
    if 0 <= i + (n-1) < 2*n-1:
        start = i
        if start >= 0 and i + n <= n:
            c_new[start:start+n] += delta * b[:n-start]
        else:
            # Handle boundary cases
            for s in range(n):
                if 0 <= i + s < n:
                    idx = i + s + (n-1) - (n-1)
                    c_new[idx] += delta * b[i+s]
    
    # Part 2: contribution from b[j] -> h
    if 0 <= j + (n-1) < 2*n-1:
        start = j  
        if start >= 0 and j + n <= n:
            c_new[start:start+n] -= delta * h_rev[:n-start]
        else:
            # Handle boundary cases
            for s in range(n):
                if 0 <= j - s < n:
                    idx = j - s + (n-1)
                    c_new[idx] -= delta * h_rev[j-s]
    
    return c_new

def eval_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def estimate_c5_from_correlation(corr):
    """Get C5 estimate from precomputed correlation"""
    return dx * np.max(corr)

    temp_interval = 2500  # Steps to decrease temp

        old_h_j = h[j]

        h[j] += delta

        b[i] = 1 - h[i]
        b[j] = 1 - h[j]

            no_improvement_count = 0

            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            # Revert changes
            h[i] = old_h_i
            h[j] = old_h_j
            b[i] = 1-h[i]  
            b[j] = 1-h[j]
            no_improvement_count += 1
        
        # Update acceptance rate every 1000 steps
        acceptance_total += 1
        if acceptance_total % 1000 == 0 and acceptance_total > 0:
            acc_rate = acceptance_count / acceptance_total
            if acc_rate < 0.05:
                delta *= 0.9
            elif acc_rate > 0.3:
                delta *= 1.05

        # Update parameters periodically  
        if step - last_delta_update > decay_interval:
            delta *= 0.9
            if delta < 1e-4:
                delta = 1e-4
            last_delta_update = step
                
        if step - last_temp_update > temp_interval:
            T *= alpha_T
            if T < 1e-6:
                T = 1e-6
            last_temp_update = step
        
        # Reset temperature if no improvement  
        if no_improvement_count > no_improvement_patience:
            T = max(T * 2, 0.01)  # Double T up to 0.01
            no_improvement_count = 0

    # Final evaluation of best solution
    final_c5 = eval_c5(best_h)
    return (best_h.tolist(), float(final_c5), n_points)
