import numpy as np
from scipy.signal import correlate

import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def update_correlation(c, h, b, i, delta):
    """Update correlation efficiently for a single change at position i"""
    n = len(h)
    # Update c with changes from index i
    # For b contribution: delta * b[i+s] where i+s is valid (0 <= i+s <= n-1)
    # This affects c from s = -i to s = n-1-i
    s_start = max(-i, -(n-1))
    s_end = min(n-1-i, n-1)
    
    # Since indices are from 0 to n-1, for shift s (from -i to n-1-i)
    # b[i+s] is valid when 0 <= i+s <= n-1, which means -i <= s <= n-1-i
    if s_start <= s_end:
        # Map s to c indices
        c_start = s_start + n - 1
        c_end = s_end + n - 1
        
        # Vectorized update
        b_slice = b[i+s_start:i+s_end+1]  # b[i+s] for valid s
        c[c_start:c_end+1] += delta * b_slice
    
    # For a contribution: -delta * a[i-s] where i-s is valid (0 <= i-s <= n-1)  
    # This affects c from s = i-(n-1) to s = i
    # Shift s ranges from i-(n-1) to i
    s_start2 = i-(n-1)
    s_end2 = i
    
    if s_start2 <= s_end2:
        # Map s to c indices
        c_start2 = s_start2 + n - 1
        c_end2 = s_end2 + n - 1
        
        # We want to extract a[i-s] for s from i-(n-1) to i
        # i-s ranges from (i-(n-1)) to i, so we reverse indices a[n-1] to a[0]
        # which gives a[i-s] for s from i-(n-1) to i
        # This corresponds to a[n-1] to a[n-1-(i-(n-1))] = a[n-1-i+(n-1)] = a[2n-2-i] down to a[0]
        # But since we work with Python slicing: from index i-(n-1) to i
        # We need a[s] with s from i-(n-1) to i, so a indices are from i-(n-1) to i
        # Also a[0] to a[n-1]. For s from i-(n-1) to i:
        # Lower index = max(0, i-(n-1)) = i-(n-1) if i >= n-1
        # Upper index0 = min(n-1, i) so slices from n-1 to n-1 
        # Actually it's more direct: 
        # We're adding Delta * a[i-s] for shift s from i-(n-1) to i
        # i-s goes from (i-(n-1)) to i 
        # These are a[0] to a[n-1], so use reverse array for proper shifting
        # The correct slice is a from index (i-(n-1)) to index (i+1) (exclusive) counting backwards after reversing
        # Simpler yet: reverse the view
        s_indices = np.arange(s_start2, s_end2+1)
        a_indices = i - s_indices # negative to positive
		
        # Now handle bounds still
        valid_mask = (a_indices >= 0) & (a_indices < n)
		
        # Actual update
        if np.any(valid_mask):
            c_slice_indices = s_indices + n - 1
            # Since we're going to subtract negative delta, we add
            c[c_slice_indices[valid_mask]] -= delta * h[a_indices[valid_mask]]

    return c

    TARGET_SUM = n_points / 2

    # Gradient based optimization along with simulated annealing
    # We'll define a more effective transformed optimization that uses:
    # 1. Multiple restarts with tabu search as in earlier work
    # 2. Global optimization to escape local minima
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    # Optimizer with diverse search strategies
    improvement_attempts = 0
    max_improvement = 20000  # We can run longer with good heuristics
    
    # Maintained temperature for simulated annealing
    temp_start = 0.01
    temp_decay = 0.9995

    # Try population based global optimization
    try:
        from scipy.optimize import differential_evolution
        
        res_de = differential_evolution(
            objective,
            bounds,
            constraints={'type': 'eq', 'fun': constraint_eq},
            maxiter=500,
            popsize=20,
            mutation=(0.5, 1),
            recombination=0.7,
            atol=1e-12,
            rtol=1e-12,
            seed=seed,
            disp=False
        )
        if res_de.success:
            h_candidate = project_to_sum_box(res_de.x, TARGET_SUM)
            c5_candidate = evaluate_c5(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
    except Exception as e:
        pass

    # Add local search refinement
    h_current = best_h.copy()
    c5_current = best_c5
    temp = temp_start

    # Main loop with simulated annealing
    while time.time() - start_time < budget_s and improvement_attempts < max_improvement:
        improvement_attempts += 1
        
        # Current state
        h_current = best_h.copy()
        c5_current = best_c5
        
        # Simulated Annealing step
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-0.01, 0.01)
        h_trial = h_current.copy()
        h_trial[i] += delta
        h_trial[j] -= delta
        
        # Handle boundary constraints
        h_trial = project_to_sum_box(h_trial, TARGET_SUM)
        
        # Evaluate new solution
        c5_trial = evaluate_c5(h_trial)
        
        # Accept or reject
        if c5_trial < c5_current:
            h_current = h_trial
            c5_current = c5_trial
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()
        else:
            # Accept worse solution with probability
            if np.random.uniform() < np.exp(-(c5_trial - c5_current) / temp):
                h_current = h_trial
                c5_current = c5_trial
        
        # Increase iteration and decrease temperature
        temp *= temp_decay

        # Occasionally enhance by local search
        if improvement_attempts % 500 == 0:
            # Take more structured moves for local exploration
            h_local = best_h.copy()
            indices = np.random.choice(n_points, size=n_points//8, replace=False)
            for idx in indices:
                delta = np.random.uniform(-0.005, 0.005)
                h_local[idx] += delta
            h_local = project_to_sum_box(h_local, TARGET_SUM)
            c5_local = evaluate_c5(h_local)
            if c5_local < best_c5:
                best_c5 = c5_local
                best_h = h_local.copy()

        # Occasionally reset temperature
        if improvement_attempts % 1000 == 0:
            temp = temp_start

    # Final check
    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h.tolist(), float(best_c5), n_points)
