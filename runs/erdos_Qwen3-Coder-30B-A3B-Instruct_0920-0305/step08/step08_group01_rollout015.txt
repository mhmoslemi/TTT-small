import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time
import cvxpy as cp
from cvxpy import SCS, MOSEK, ECOS

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def build_shift_matrices(n):
    """Build shift matrices for all valid shifts s in 1..floor(n/2)"""
    R = []
    for s in range(1, n // 2 + 1):
        # Construct shift matrix R_s where only (i, i+s) entries are 1
        R_s = np.zeros((n, n))
        for i in range(n - s):
            R_s[i, i + s] = 1
        R.append(R_s)
    return R

def solve_sdp(n, shift_matrices, seed=42, solver='SCS'):
    """Solve the SDP relaxation for the optimal overlap"""
    np.random.seed(seed)
    # Variables
    X = Semidef(n, name='X')
    mu = Variable(n, name='mu')
    lam = Variable(name='lam')

    # Constraints involving mass and marginal
    constraints = [
        mu >= 0,
        mu <= 1,
        sum(mu) == n / 2,
        X << 1,  # bounded from above
    ] + [X[i, i] == mu[i] for i in range(n)]

    # Add shift constraints
    for s, R_s in enumerate(shift_matrices):
        constraints.append(trace(R_s @ X) >= lam)

    # Objective
    objective = Maximize(lam)

    # Solve
    prob = Problem(objective, constraints)
    try:
        if solver == 'MOSEK':
            prob.solve(solver=MOSEK, verbose=False, time_limit=150)
        elif solver == 'ECOS':
            prob.solve(solver=ECOS, verbose=False, max_iters=5000)
        else:
            prob.solve(solver=SCS, verbose=False, max_iters=5000, eps=1e-4)
    except Exception as e:
        print(f"SDP solver failed: {e}")
        return None, None

    if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
        return X.value, mu.value
    else:
        print(f"SDP solver status: {prob.status}")
        return None, None

def round_binary(X_solution, mu_solution, n):
    """Round the SDP solution to binary using hyperplane rounding"""
    # Compute the Cholesky decomposition or SVD
    try:
        try:
            V = np.linalg.cholesky(X_solution)
        except np.linalg.LinAlgError:
            U, _, _ = np.linalg.svd(X_solution)
            V = U @ np.diag(np.sqrt(np.linalg.eig(X_solution)[0]))
    except:
        # Fallback: compute principal components
        V = np.random.rand(n, n)
        # This is a simplified method, in practice use stable SVD
        V = np.eye(n)
    # Random Gaussian vector
    g = np.random.randn(V.shape[1])
    # Scores for each row (will be used to set binary positions)
    scores = V @ g
    # Sort indices by score to get top half
    indices_sorted = np.argsort(scores)[::-1]
    # Take exactly n/2 elements
    chosen_indices = set(indices_sorted[:n // 2])
    x = np.array([1 if i in chosen_indices else 0 for i in range(n)])
    return x

def compute_shift_contributions(x, s, n):
    """Compute contribution to overlap for a particular shift s"""
    count = 0
    for i in range(n - s):
        count += x[i] * x[i + s]
    return count

def local_improve(h, n):
    """Local refinement using targeted swaps based on worst shift"""
    # Compute correlations once
    corr = np.correlate(h, 1 - h, mode='full')
    best_shift = np.argmax(corr)
    best_corr = corr[best_shift]
    
    max_iter = 10000  # Should be within budget
    iter_count = 0
    
    while iter_count < max_iter:
        s = best_shift
        # Identify indices where a 1 contributes to this shift, and 0 does not
        bad_indices = []  # locations x_i = 1 AND x_{i+s} = 0
        good_indices = []  # locations x_j = 0 AND x_{j+s} = 1
        
        for i in range(n - s):
            if h[i] == 1 and h[i + s] == 0:
                bad_indices.append(i)
            if h[i] == 0 and h[i + s] == 1:
                good_indices.append(i)
        
        # If not enough to swap, skip
        if not bad_indices or not good_indices:
            break
        
        # Perform a swap
        i = np.random.choice(bad_indices)
        j = np.random.choice(good_indices)
        
        # Clone and try the flip
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # Compute new correlations
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_max = np.max(corr_new)
        
        if new_max <= best_corr:
            # Accept the swap
            h = h_new
            best_corr = new_max
            # Update shift if needed
            best_shift = np.argmax(corr_new)
            if best_shift != s:
                corr = corr_new
        else:
            # Reject (could record as stall)
            pass
            
        iter_count += 1
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 120
    dx = 2.0 / n
    shift_matrices = build_shift_matrices(n)
    
    def solve_and_round():
        X_sol, mu_sol = solve_sdp(n, shift_matrices, seed=seed, solver='SCS')
        if X_sol is None:
            return None
        best_x = round_binary(X_sol, mu_sol, n)
        return best_x

    best_c5 = float('inf')
    best_h = None
    backups = []

    # Phase 1: SDP + Rounding
    phase1_end = start_time + 200
    num_rounding = 500
    for i in range(num_rounding):
        if time.time() > phase1_end:
            break
        try:
            h_try = solve_and_round()
            if h_try is not None:
                c5_val = evaluate_c5(h_try)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_try.copy()
                backups.append((h_try.copy(), c5_val))
        except:
            pass

    # Phase 2: Local refinement
    phase2_end = start_time + 320
    if best_h is not None:
        refined_h = local_improve(best_h.copy(), n)
        refined_c5 = evaluate_c5(refined_h)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h.copy()

    # Phase 3: Use backups if no solution found
    if best_h is None:
        for h_backup, c5_backup in backups:
            if c5_backup < best_c5:
                best_c5 = c5_backup
                best_h = h_backup.copy()

    if best_h is None:
        # Fallback
        best_h = np.zeros(n)
        best_h[:n//2] = 1
        best_c5 = evaluate_c5(best_h)

    return (best_h.tolist(), float(best_c5 * dx), n)
