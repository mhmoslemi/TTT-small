def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def evaluate_c5_correlate(h):
    """Alternative evaluation using numpy correlate for accuracy"""
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def neighbor_sum(h, lag):
    """Compute neighbor sum for given lag efficiently"""
    lag_abs = abs(lag)
    if lag_abs >= len(h):
        return np.zeros_like(h)
    
    w = np.zeros_like(h)
    if lag_abs == 0:
        w = 2 * h
    else:
        if lag > 0:
            w[:len(h)-lag_abs] += h[lag_abs:]
            w[lag_abs:] += h[:len(h)-lag_abs]
        else:
            w[lag_abs:] += h[:-lag_abs]
            w[:len(h)-lag_abs] += h[lag_abs:]
    return w

def mass_transfer_step(h, step_size, eps=1e-6):
    """Perform one step of mass transfer to improve h"""
    # Compute correlation to find worst lag
    corr = np.correlate(h, 1-h, mode="full")

    lag_max = idx_max - (len(h) - 1)
    
    # Compute neighbor sum for the worst lag
    w = neighbor_sum(h, lag_max)
    
    # Find indices where we can move mass
    low_candidates = np.where(h > eps)[0]
    high_candidates = np.where(h < 1 - eps)[0]
    
    if len(low_candidates) == 0 or len(high_candidates) == 0:
        return h, False
    
    # Find the index with minimum w among low candidates
    min_w_idx = low_candidates[np.argmin(w[low_candidates])]
    # Find the index with maximum w among high candidates
    max_w_idx = high_candidates[np.argmax(w[high_candidates])]
    
    # If they're the same index, find different ones
    if min_w_idx == max_w_idx:
        # Try to find better candidates
        if len(low_candidates) > 1 and len(high_candidates) > 1:
            # Try to find candidates with different w values
            sorted_low = low_candidates[np.argsort(w[low_candidates])]
            sorted_high = high_candidates[np.argsort(-w[high_candidates])]
            
            # Try a few candidates
            for low_idx in sorted_low:
                for high_idx in sorted_high:
                    if low_idx != high_idx and h[low_idx] > eps and h[high_idx] < 1 - eps:
                        min_w_idx = low_idx
                        max_w_idx = high_idx
                        break
                else:
                    continue
                break
    
    # Compute the amount to transfer
    if min_w_idx == max_w_idx:
        return h, False
        
    delta = min(step_size, h[min_w_idx], 1 - h[max_w_idx])
    
    if delta < 1e-10:
        return h, False
    
    # Apply the transfer
    h_new = h.copy()
    h_new[min_w_idx] -= delta
    h_new[max_w_idx] += delta
    
    return h_new, True
