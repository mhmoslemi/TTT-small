import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

# Global parameters
N = 120
dx = 2.0 / N
M = N // 2  # Target sum

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    # Return max correlation value scaled by dx
    return dx * np.max(corr)

def project_to_sum_box(h, target_sum):
    """Projection onto box constraints [0,1] with sum = target_sum using water filling"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    def sum_with_lambda(lam):
        # Compute sum of clamp(h + lam, 0, 1)
        proj = np.clip(h + lam, 0, 1)
        return np.sum(proj)
    
    # Binary search for lambda
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = sum_with_lambda(m) - target_sum
        if abs(fm) < 1e-10:
            break
        if fm * (sum_with_lambda(a) - target_sum) < 0:
            b = m
        else:
            a = m
    
    return np.clip(h + (a + b)/2, 0, 1)

def solve_binary_milp():
    """Solve binary MILP for maximum minimum autocorrelation"""
    # Try to solve using CBC
    try:
        y = cp.Variable(N, boolean=True)
        w = cp.Variable((N-1, N), nonneg=True)  # w[k,i] = y[i]*y[i+k]
        t = cp.Variable()
        
        constraints = [
            cp.sum(y) == M,
            t >= 0,
            t <= M
        ]
        
        # Add product linearization constraints
        for k in range(1, N):
            for i in range(N - k):
                # w[k,i] <= y[i]
                constraints.append(w[k-1, i] <= y[i])
                # w[k,i] <= y[i+k]  
                constraints.append(w[k-1, i] <= y[i+k])
                # w[k,i] >= y[i] + y[i+k] - 1
                constraints.append(w[k-1, i] >= y[i] + y[i+k] - 1)
                # w[k,i] >= 0
                constraints.append(w[k-1, i] >= 0)
        
        # Add minimum autocorrelation constraints
        for k in range(1, N):
            constraints.append(cp.sum(w[k-1, :N-k]) >= t)
            
        # Objective
        objective = cp.Maximize(t)
        
        prob = cp.Problem(objective, constraints)
        
        # Solve with time limit
        prob.solve(solver=cp.CBC, max_seconds=240)
        
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            return y.value.astype(int)
            
    except Exception:
        pass
    
    return None

def greedy_binary_initialization():
    """Construct a feasible binary solution with good autocorrelation"""
    y = np.zeros(N, dtype=int)
    # Place 1s with spacing to avoid autocorrelation issues
    spacing = N // M  # Target about 2 positions between 1s
    # Fill with 1s, skipping some to create gaps for better correlation
    count = 0
    for i in range(N):
        if count < M and (i % spacing == 0 or (i+1) % spacing == 0):
            y[i] = 1
            count += 1
        if count >= M:
            break
    # Ensure we have exactly M ones in the right places 
    ones_positions = np.where(y == 1)[0]
    if len(ones_positions) > M:
        y[ones_positions[M:]] = 0
    elif len(ones_positions) < M:
        # Add more ones not in current positions
        unused_positions = np.where(y == 0)[0]
        to_add = M - len(ones_positions)
        for pos in unused_positions[:to_add]:
            y[pos] = 1
    
    return y

def compute_subgradient(h, k_star):
    """Compute subgradient of overlap at shift k_star"""
    # h_{j+k_star} - with circular wrap-around for simplicity in loop
    g = np.ones_like(h)
    if k_star > 0:
        g[:-k_star] -= h[k_star:]
        g[-k_star:] -= h[:k_star]
    else:
        g -= h
    return g

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    
    # Phase 1: Binary MILP or greedy initialization (40% of time)
    phase1_end = start_time + 0.4 * budget_s
    
    # Try to solve binary MILP
    binary_solution = solve_binary_milp()
    if binary_solution is not None:
        h0 = binary_solution.astype(float)
    else:
        # Fallback to greedy initialization
        h0 = greedy_binary_initialization().astype(float)
    
    # Ensure sum is correct
    h0 = project_to_sum_box(h0, M)
    current_c5 = evaluate_c5(h0)
    best_h = h0.copy()
    best_c5 = current_c5
    
    # Phase 2: Sub-gradient descent (35% of time)
    phase2_end = start_time + 0.75 * budget_s
    alpha_0 = 0.5 * dx  # Initial step size
    
    iter_count = 0
    while time.time() < phase2_end:
        iter_count += 1
        
        # Evaluate correlations to find worst shift
        corr = correlate(1 - best_h, best_h, mode='full')
        k_star = np.argmax(corr)
        if k_star > N - 1:
            k_star = k_star - N  # Adjust for FFT offset
        
        # Compute subgradient
        g = compute_subgradient(best_h, k_star)
        
        # Compute step size
        alpha = alpha_0 / np.sqrt(iter_count)
        
        # Update with step
        h_new = best_h - alpha * g
        
        # Project back to feasible set
        h_new = project_to_sum_box(h_new, M)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new)
        
        # Update if better
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
    
    # Phase 3: Simulated annealing refinement (25% of time)
    phase3_end = start_time + budget_s
    temp_start = 5 * dx
    
    while time.time() < phase3_end:
        # Two-index swap
        i, j = np.random.choice(N, size=2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Clip and re-project
        h_new = project_to_sum_box(h_new, M)
        
        new_c5 = evaluate_c5(h_new)
        
        # Accept with Metropolis probability
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        else:
            # Accept with probability
            delta_c5 = new_c5 - best_c5
            temp = temp_start * (1 - (time.time() - start_time) / budget_s)
            if temp > 0 and np.random.rand() < np.exp(-delta_c5 / temp):
                best_h = h_new.copy()
                best_c5 = new_c5
    
    return (best_h.tolist(), float(best_c5), N)
