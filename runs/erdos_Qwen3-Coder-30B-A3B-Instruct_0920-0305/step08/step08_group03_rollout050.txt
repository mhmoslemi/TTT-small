import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def update_correlation(c, h, b, i, delta, j, delta2):
    """Update correlation array with change at indices i,j"""
    # For each shift s, compute changes to correlation
    # We compute contribution from both changes
    
    # Change on i: increment by delta
    # Change on j: increment by delta2 = -delta (mass conservation)
    
    # For index i, we look at correlations with all valid shifts where i participates
    # Correlation at shift s includes term h[i] * b[i+s]
    # For shift s, this term is in c[s + n - 1]
    
    # From delta * b[i+s] for valid indices
    s_start = max(0, -i)
    s_end = min(n_points - 1 - i, n_points - 1)
    
    c_start = s_start + n_points - 1  
    c_end = s_end + n_points - 1 + 1  # exclusive upper bound
    
    if c_start < len(c) and c_end > 0:
        c[c_start:c_end] += delta * b[i:i + (c_end - c_start)]
    
    # Subtract h[i-s] * delta for valid i-s indices (reverse from b term)
    # This happens when shift s is such that i - s is valid
    s_start_j = max(0, i - (n_points - 1))
    s_end_j = i
    
    c_start_j = s_start_j + n_points - 1
    c_end_j = s_end_j + n_points - 1 + 1
    
    if c_start_j < len(c) and c_end_j > 0:
        # Reverse of a vector for indexing  
        a_rev = h[::-1]
        # only use elements where valid
        valid_len = min(c_end_j - c_start_j, n_points)
        c_slice = c[c_start_j:c_end_j]
        c_slice -= delta * a_rev[i:(i+valid_len)] if valid_len > 0 else 0
    
    # same for j 
    # For index j, we look at correlations with all valid shifts where j participates
    # From delta2 * b[j+s] for valid indices
    s_start2 = max(0, -j)
    s_end2 = min(n_points - 1 - j, n_points - 1)
    
    c_start2 = s_start2 + n_points - 1  
    c_end2 = s_end2 + n_points - 1 + 1
    
    if c_start2 < len(c) and c_end2 > 0:
        c[c_start2:c_end2] += delta2 * b[j:j + (c_end2 - c_start2)]
        
    # Subtract h[j-s] * delta2 for valid j-s indices  
    s_start_j2 = max(0, j - (n_points - 1))
    s_end_j2 = j
    
    c_start_j2 = s_start_j2 + n_points - 1
    c_end_j2 = s_end_j2 + n_points - 1 + 1
    
    if c_start_j2 < len(c) and c_end_j2 > 0:
        a_rev2 = h[::-1]
        valid_len = min(c_end_j2 - c_start_j2, n_points)
        c[c_start_j2:c_end_j2] -= delta2 * a_rev2[j:(j+valid_len)] if valid_len > 0 else 0

    # Add delta^2 term (isotropic interaction)
    if i != j:
        delta_sq = delta * delta
        # For i -> j: shift = j - i
        shift_index = j - i + n_points - 1
        if 0 <= shift_index < len(c):
            c[shift_index] += delta_sq
        # For j -> i: shift = i - j
        shift_index2 = i - j + n_points - 1
        if 0 <= shift_index2 < len(c):
            c[shift_index2] += delta_sq

    return c

    b = 1 - h
    c = correlate(h, b, mode="full") * dx
    
    # Simple simulated annealing with incremental updates
    iteration = 0
    max_iterations = 500000
    
    # Temperature schedule
    T = 0.01
    cooling_rate = 0.99995
    
    while iteration < max_iterations and time.time() - start_time < budget_s:
        iteration += 1
        
        # Randomly select two points to exchange mass
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Ensure i and j are different
        if i == j:
            continue
            
        # Proposed change: move small delta amount
        delta = np.random.uniform(0.0001, 0.005)
        delta_i = -delta
        delta_j = delta
        
        # Store values
        h_old_i = h[i]
        h_old_j = h[j]
        
        # Update h
        h[i] += delta_i
        h[j] += delta_j
        
        # Project to constraints
        h = project_to_sum_box(h, TARGET_SUM)
        
        # If projection changed the values, recompute
        if abs(h[i] - h_old_i) > 1e-10 or abs(h[j] - h_old_j) > 1e-10:
            # Recompute correlation from scratch
            b = 1 - h
            c = correlate(h, b, mode="full") * dx
            current_c5 = np.max(c)
        else:
            # Update correlation incrementally
            c = update_correlation(c, h, b, i, delta_i, j, delta_j)
            current_c5 = np.max(c)
        
        # Accept or reject the change
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        elif np.random.random() < np.exp(-(current_c5 - best_c5) / T):
            # Accept worse solution with some probability
            pass
        else:
            # Reject and revert
            h[i] = h_old_i
            h[j] = h_old_j
        
        # Adjust temperature
        T *= cooling_rate
    
    return (best_h.tolist(), float(best_c5), n_points)
