import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import time

# Global variables
n_points = 300
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def rasterise_and_project(theta, n=n_points):
    """Convert piecewise constant block parameters to h"""
    K = 8  # Number of blocks
    # Split theta into widths and heights
    widths = np.exp(theta[:K])
    widths = 2 * widths / np.sum(widths)
    heights = 1 / (1 + np.exp(-theta[K:]))
    
    # Compute cumulative widths for searchsorted
    cum_widths = np.concatenate([[0], np.cumsum(widths)])
    
    # Grid points
    x_grid = (np.arange(n) + 0.5) * dx
    
    # Find block index for each grid point
    block_indices = np.searchsorted(cum_widths, x_grid, side="right") - 1
    h = heights[block_indices]
    
    # Project to sum constraint
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def obj(θ):
    h = rasterise_and_project(θ)
    c5 = evaluate_c5(h)
    return c5

def compute_active_shifts(corr, tol=1e-12):
    """Return indices of active shifts"""
    max_val = corr.max()
    active = np.where(corr >= max_val - tol)[0]
    return active - (n_points - 1)  # Convert to signed shifts

def compute_subgradient(h, shifts, dx=dx):
    """Compute average subgradient from active shifts"""
    g = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            g[:n_points-s] += -dx * h[s:]
            g[s:] += -dx * h[:n_points-s]
        else:
            t = -s
            g[t:] += -dx * h[:n_points-t]
            g[:n_points-t] += -dx * h[t:]
    return -g / len(shifts)  # descent direction

def psgd_refinement(h0, max_iter=5000, time_limit=460):
    """Projected sub-gradient descent with mass transfer moves"""
    h = h0.copy()
    c5_best = evaluate_c5(h)
    h_best = h.copy()
    
    eta = 0.5
    eta_min, eta_max = 1e-4, 2.0
    iteration = 0
    no_improvement_count = 0
    t0 = time.time()

    while iteration < max_iter and time.time() - t0 < time_limit:
        # Compute correlation and active shifts
        corr = correlate(h[::-1], 1-h, mode='full') * dx
        active_shifts = compute_active_shifts(corr)
        g = compute_subgradient(h, active_shifts)
        
        # Descent step
        h_trial = h - eta * g
        h_trial = project_to_sum_box(h_trial, TARGET_SUM)
        
        c5_trial = evaluate_c5(h_trial)
        
        if c5_trial < c5_best:
            h = h_trial
            c5_best = c5_trial
            h_best = h.copy()
            no_improvement_count = 0
            eta = min(eta * 1.05, eta_max)  # Increase eta on success
        else:
            no_improvement_count += 1
            eta = max(eta * 0.7, eta_min)  # Decrease eta on failure
        
        # Random mass transfer every N steps or when stuck
        if iteration % 30 == 0 or no_improvement_count > 200:
            i_plus = np.random.choice(n_points)
            i_minus = np.random.choice(n_points)
            if h[i_plus] < 1 and h[i_minus] > 0:
                transfer_amount = min(0.02, h[i_minus]) * np.random.random()
                h[i_plus] += transfer_amount
                h[i_minus] -= transfer_amount
                h = project_to_sum_box(h, TARGET_SUM)
        
        iteration += 1
    
    return h_best, c5_best
