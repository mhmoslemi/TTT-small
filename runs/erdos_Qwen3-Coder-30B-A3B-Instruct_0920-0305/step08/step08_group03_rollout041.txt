def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    s_start = max(-i, -(n-1))

    if s_start <= s_end:
        c_idx = s_start + n - 1
        c[c_idx:c_idx + (s_end - s_start + 1)] += delta * b[i+s_start:i+s_end+1]
    
    # Handle contribution from a change (h change)
    # For shift s where i-s is valid (0 <= i-s <= n-1), contribution to c[s+n-1] is -delta * a[i-s]
    # This means s ranges from i-(n-1) to i
    s_start_a = max(i-(n-1), 0)  # s should be >= i-(n-1)
    s_end_a = min(i, n-1)        # s should be <= i

        a_idx = i - np.arange(s_start_a, s_end_a + 1)  # a[i-s] = a[a_idx]
        c_idx = s_start_a + n - 1
        c[c_idx:c_idx + (s_end_a - s_start_a + 1)] -= delta * h[a_idx[::-1]]
    
    return c
