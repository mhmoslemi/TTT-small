```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import CBC, GLPK_MI, ECOS_BB

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def clip_and_project(h, target_sum):
    """Clip to [0,1] and project onto sum constraint"""
    h = np.clip(h, 0, 1)
    
    # Water-filling projection
    def proj_sum(h, target_sum):
        a, b = -1.0, 1.0
        def f(lam):
            projected = np.clip(h + lam, 0, 1)
            return np.sum(projected) - target_sum
        
        for _ in range(50):
            m = (a + b) / 2
            fm = f(m)
            if abs(fm) < 1e-10:
                break
            if fm * f(a) < 0:
                b = m
            else:
                a = m
        return np.clip(h + (a + b)/2, 0, 1)
    
    h = proj_sum(h, target_sum)
    return h

def greedy_binary_initialization():
    """Generate a good binary start using greedy spacing"""
    y = np.zeros(n_points, dtype=int)
    # Place 1s with roughly equal spacing
    spacing = n_points // TARGET_SUM
    for i in range(int(TARGET_SUM)):
        idx = i * spacing
        if idx < n_points:
            y[idx] = 1
    
    # Ensure exactly TARGET_SUM ones
    diff = int(TARGET_SUM - np.sum(y))
    if diff > 0:
        # Add ones to reduce autocorrelation
        for i in range(n_points):
            if y[i] == 0 and diff > 0:
                y[i] = 1
                diff -= 1
            if diff == 0:
                break
    elif diff < 0:
        # Remove some ones
        for i in range(n_points-1, -1, -1):
            if y[i] == 1 and diff < 0:
                y[i] = 0
                diff += 1
            if diff == 0:
                break
                
    return y.astype(float)

def solve_binary_milp():
    """Solve the binary MILP for the best 0/1 pattern"""
    M = int(TARGET_SUM)
    
    # Create variables
    y = cp.Variable(n_points, boolean=True)
    w = cp.Variable((n_points, n_points), nonneg=True)
    t = cp.Variable()
    
    # Constraints for linearization of y_i * y_{i+k}
    constraints = []
    
    # Cardinality constraint
    constraints.append(cp.sum(y) == M)
    
    # Linearization constraints for each k, i
    for k in range(1, n_points):
        for i in range(n_points - k):
            constraints.append(w[i, i+k] <= y[i])
            constraints.append(w[i, i+k] <= y[i+k])
            constraints.append(w[i, i+k] >= y[i] + y[i+k] - 1)
            constraints.append(w[i, i+k] >= 0)
    
    # min autocorrelation constraint
    for k in range(1, n_points):
        constraints.append(cp.sum(w[:, k]) >= t)
    
    # Bounds on t
    constraints.append(t >= 0)
    constraints.append(t <= M)
    
    # Add symmetry breaking
    constraints.append(y[0] == 1)
    
    # Objective
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers with time limits
    solvers = [CBC, GLPK_MI, ECOS_BB]
    for solver in solvers:
        try:
            prob.solve(solver=solver, max_seconds=0.4 * 600)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                y_sol = y.value
                return y_sol
        except:
            continue
    
    # If MILP fails, fall back to greedy initialization
    return greedy_binary_initialization()

def subgradient_descent(h_start, max_seconds):
    """Perform deterministic subgradient descent"""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    t = 1
    alpha_0 = 0.5 * dx
    
    while time.time() - start_time < max_seconds:
        # Compute correlations
        corr = compute_corr_fft(h)
        k_star = np.argmax(corr)
        
        # Gradient computation (subgradient of worst shift)
        g = np.zeros(n_points)
        for i in range(n_points):
            if i + k_star < n_points:
                g[i] += -h[i + k_star]
            if i - k_star >= 0:
                g[i] += -h[i - k_star]
            g[i] += 1
        
        # Take step
        h_new = h - (alpha_0 / np.sqrt(t)) * g
        h_new = clip_and_project(h_new, TARGET_SUM)
        
        # Test improvement
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        t += 1
        
        # Check for early termination on tiny steps
        if t > 100000:
            break
            
    return best_h, best_c5

def swap_move_annealing(h_start, max_seconds):
    """Perform simulated annealing with two-index swaps"""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    T_start = 5 * dx
    current_time = start_time
    
    T = T_start
    step = 0
    while current_time - start_time < max_seconds:
        # Choose 2 indices
        i, j = np.random.choice(n_points, size=2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        
        # Update
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        h_new = np.clip(h_new, 0, 1)
        h_new = clip_and_project(h_new, TARGET_SUM)
        
        c5_new = evaluate_c5(h_new)
        delta_c5 = c5_new - best_c5
        
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        step += 1
        current_time = time.time()
        T = max(0, T_start * (1 - (current_time - start_time) / max_seconds))
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    
    # Phase 1: Binary optimization via MILP or greedy
    h_current = solve_binary_milp()
    c5_current = evaluate_c5(h_current)
    
    # Phase 2: Subgradient descent
    h_optimized, c5_optimized = subgradient_descent(h_current, 0.35 * budget_s)
    if c5_optimized < c5_current:
        h_current = h_optimized
        c5_current = c5_optimized
    
    # Phase 3: Simulated annealing
    h_final, c5_final = swap_move_annealing(h_current, 0.25 * budget_s)
    if c5_final < c5_current:
        h_current = h_final
        c5_current = c5_final
        
    return (h_current.tolist(), float(c5_current), n_points)
```
