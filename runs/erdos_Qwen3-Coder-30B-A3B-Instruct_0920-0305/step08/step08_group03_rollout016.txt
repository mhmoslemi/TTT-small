import numpy as np
from scipy.signal import correlate
import time
from itertools import product
import math

def evaluate_overlap_integrals(h_values, n_points):
    """Compute C5 using discrete correlation; fast version."""
    dx = 2.0 / n_points
    h = np.array(h_values)
    b = 1 - h
    # compute full convolution, but we only need max over shifts 0..2
    corr = correlate(h, b, mode="full")
    # Correlation corresponds to shift k from 0 to 2
    # The full correlation is indexed from -len+1 to len-1
    # We need only the positive shifts in [0, 2]
    # So take the relevant part of the correlation array
    len_h = len(h)
    start_idx = len_h - 1      # k = 0 is the middle of full corr
    corr_full = corr[start_idx:]  # k from 0 to 2
    max_overlap = np.max(corr_full)
    return dx * max_overlap

def generate_piecewise_step_function(lengths, heights, n_points):
    """
    Generate a step function from lengths and heights.
    lengths: list of interval lengths summing to 2
    heights: list of interval heights in [0,1]  
    """
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    pos = 0
    for i, (length, height) in enumerate(zip(lengths, heights)):
        start_idx = int(pos / dx)
        end_idx = int((pos + length) / dx)
        if end_idx >= n_points:
            end_idx = n_points
        h[start_idx:end_idx] = height
        pos += length
    return h

def find_optimal_piecewise_function(num_intervals, n_points, budget_s):
    """
    Find optimal piecewise constant h using deterministic search:
    - Represent h as intervals with variable lengths and heights. 
    - Model the overlap integral exactly.
    - Use a global optimization or grid-based strategy.
    """
    def objective(params):
        # s[0], s[1], ..., s[m-2] = lengths
        # h[0], h[1], ..., h[m-2] = heights
        # m = num_intervals
        lengths = list(params[:num_intervals-1])
        total_length = sum(lengths)
        if total_length >= 2:
            return float('inf')
        lengths.append(2 - total_length)
        
        heights = list(params[num_intervals-1:])
        # enforce height constraints
        total_mass = sum(h * l for h, l in zip(heights, lengths))
        if abs(total_mass - 1.0) > 1e-10:
            # Slightly infeasible
            return float('inf')
        
        # Compute the corresponding h function
        h_func = generate_piecewise_step_function(lengths, heights, n_points)
        return evaluate_overlap_integrals(h_func.tolist(), n_points)
        
    def grid_search_with_local_refinement():
        best_c5 = float('inf')
        best_params = None
        grid_size = 5  # per dimension, larger is slower but finer
        
        # Intervals: 2 * (m-1) = 8 variables.
        # We discretize each length to be in range [0.05, 1.95] 
        # and each height to [0.05, 0.95].
        # Use a restricted grid for trial points
        search_space = []
        for _ in range(300):
            # Randomly sampled constraints
            lengths = np.random.rand(num_intervals-1) * 1.9 + 0.05
            total = sum(lengths)
            lengths = [l * 2 / total for l in lengths]
            
            heights = np.random.rand(num_intervals-1) * 0.9 + 0.05
            # Keep height of last interval via mass constraint
            h_last = (1 - sum(h * l for h, l in zip(heights, lengths))) / lengths[-1]
            if not (0 <= h_last <= 1):
                continue
            heights = list(heights) + [h_last]
            
            params = lengths + heights
            cost = objective(params)
            if cost < best_c5:
                best_c5 = cost
                best_params = params.copy()
        
        # Local polishing
        if best_params:
            # Use a very simple local random walk in low dimensions
            for _ in range(500):
                new_params = [p + np.random.normal(0, 0.01) for p in best_params]
                # Project to feasible
                new_lengths = new_params[:num_intervals-1]
                new_heights = new_params[num_intervals-1:]
                if sum(new_lengths) >= 2:
                    continue
                    
                # Spread the lengths
                new_lengths.append(2 - sum(new_lengths))
                total_mass = sum(h * l for h, l in zip(new_heights, new_lengths))
                mass_ratio = 1.0 / total_mass if total_mass > 0 else 1
                new_heights = [h * mass_ratio for h in new_heights]
                
                # Bound to [0,1]
                new_heights = [max(0, min(1, h)) for h in new_heights]
                new_params = new_lengths + new_heights
                
                cost = objective(new_params)
                if cost < best_c5:
                    best_c5 = cost
                    best_params = new_params.copy()

        return best_params, best_c5

    def adaptive_search_with_early_stop():
        best_c5 = float('inf')
        best_params = None        
        start_time = time.time()
        
        for _ in range(10000):
            if time.time() - start_time > budget_s * 0.9: # leave some margin for final polishing
                break
                
            # Sample candidate configuration
            # heights = random sample limited to [0.1, 0.9] to get non-trivial functions
            lengths = []
            heights = []
            for _ in range(num_intervals - 1):
                lengths.append(np.random.uniform(0.05, 1.9)) 
                heights.append(np.random.uniform(0.1, 0.9))
            
            # adjust so last item fits
            total_len = sum(lengths)
            lengths.append(2 - total_len)
            
            # compute corresponding height based on mass constraint
            total_mass = sum(h * l for h, l in zip(heights, lengths[:-1]))
            h_last = (1.0 - total_mass) / lengths[-1]
            if not (0 <= h_last <= 1):
                continue
            heights.append(h_last)
                
            params = lengths + heights
            
            cost = objective(params)
            if cost < best_c5:
                best_c5 = cost
                best_params = params.copy()
        
        return best_params, best_c5

    # initial parameter values – mostly in center of [0,2] and [0,1]
    search_result = adaptive_search_with_early_stop()
    best_params, best_value = search_result
    if best_params is None:
        return None, float('inf')
    
    # polish the solution
    for _ in range(1000):
        # make slight changes with probability 0.5
        if np.random.rand() < 0.5:
            new_params = best_params.copy()
            # pick one random index to modify
            idx = np.random.randint(len(best_params))
            new_params[idx] += np.random.normal(0, 0.01)
            
            # handle bounds for non-negative lengths and 0-1 height
            if idx < num_intervals - 1:
                new_params[idx] = max(0.01, new_params[idx])
            else: # height
                new_params[idx] = max(0.0, min(1.0, new_params[idx]))
            
            # Ensure mass constraint is preserved
            lengths = list(new_params[:num_intervals-1]) + [2 - sum(new_params[:num_intervals-1])]
            heights = list(new_params[num_intervals-1:])
            if abs(sum(h * l for h, l in zip(heights, lengths)) - 1.0) > 1e-8:
                # revert
                continue
            
            new_cost = objective(new_params)
            if new_cost < best_value:
                best_value = new_cost
                best_params = new_params.copy()

    return best_params, best_value

def run(seed=42, budget_s=600, **kwargs):
    """
    Main solver function. 
    """
    np.random.seed(seed)
    start_time = time.time()

    # Use fewer intervals for faster optimization
    num_intervals = 5
    n_points = 240  # consistent with provided solutions
    
    # Optimise via parametric piecewise
    best_params, best_c5 = find_optimal_piecewise_function(num_intervals, n_points, budget_s)
    
    if best_params is None:
        # fallback: use a simple construction
        h = np.array([0.0]*n_points)
        # Try piecewise constant with 100 steps
        h[:n_points//2] = 1.0
        best_h = h.tolist()
        best_c5 = evaluate_overlap_integrals(best_h, n_points)
        return best_h, best_c5, n_points

    # Reconstruct h from solved parameters
    lengths = best_params[:num_intervals-1]
    total_len = sum(lengths)
    lengths = lengths + [2 - total_len]
    
    heights = best_params[num_intervals-1:]
    
    h = generate_piecewise_step_function(lengths, heights, n_points)
    
    final_c5 = evaluate_overlap_integrals(h.tolist(), n_points)
    
    return h.tolist(), float(final_c5), n_points
