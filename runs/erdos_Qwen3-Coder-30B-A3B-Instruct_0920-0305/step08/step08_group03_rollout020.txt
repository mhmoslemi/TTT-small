def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def update_correlation(c, a, b, i, delta):
    """Incrementally update correlation when changing a[i]"""
    n = len(a)
    # For indices i, we compute contributions to all correlation values
    # Contribution from b: for s where i+s in bounds [0,n-1] 
    # (i.e. s in [-i, n-1-i])
    s_start = max(-i, -(n-1))
    s_end = min(n-1-i, n-1)
    for s in range(s_start, s_end+1):
        if 0 <= i+s < n:
            c[s + n - 1] += delta * b[i+s]

    # (i.e. s in [i-(n-1), i])
    s_start = max(i-(n-1), 0)
    s_end = min(i, n-1)
    for s in range(s_start, s_end+1):
        if 0 <= i-s < n:
            c[s + n - 1] -= delta * a[i-s]
    
    return c
