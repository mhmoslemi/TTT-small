import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def update_correlation_incremental(c, h, b, i, j, delta):
    """Incrementally update correlation when moving mass between indices i and j"""
    # Original arrays
    a = h.copy()
    b = 1 - h
    
    delta_i = -delta
    delta_j = +delta
    
    # Process index i
    if i != j:
        # b contribution: b[i+s] * delta_i for s in range(-i, n-1-i)
        s_start = max(-i, -(n_points - 1))
        s_end = min(n_points - 1 - i, n_points - 1)
        s_range = np.arange(s_start, s_end + 1)
        c_idx = (s_range + n_points - 1).astype(int)
        if len(s_range) > 0:
            c[c_idx] += delta_i * b[i + s_range]
        
        # a contribution: -a[i-s] * delta_i for s in range(i-(n_points-1), i)
        s_start = max(i - (n_points - 1), 0)
        s_end = i
        s_range = np.arange(s_start, s_end + 1)
        c_idx = (s_range + n_points - 1).astype(int)
        a_indices = i - s_range
        if len(s_range) > 0:
            c[c_idx] -= delta_i * a[a_indices]
        
        # Add the cross term contribution from delta_i * delta_j
        cross_shift1 = j - i + n_points - 1
        cross_shift2 = i - j + n_points - 1
        if 0 <= cross_shift1 < len(c):
            c[cross_shift1] += delta * delta
        if 0 <= cross_shift2 < len(c):
            c[cross_shift2] += delta * delta
    
    # Process index j
    if i != j:
        # b contribution: b[j+s] * delta_j for s in range(-j, n_points-1-j)
        s_start = max(-j, -(n_points - 1))
        s_end = min(n_points - 1 - j, n_points - 1)
        s_range = np.arange(s_start, s_end + 1)
        c_idx = (s_range + n_points - 1).astype(int)
        if len(s_range) > 0:
            c[c_idx] += delta_j * b[j + s_range]
        
        # a contribution: -a[j-s] * delta_j for s in range(j-(n_points-1), j)
        s_start = max(j - (n_points - 1), 0)
        s_end = j
        s_range = np.arange(s_start, s_end + 1)
        c_idx = (s_range + n_points - 1).astype(int)
        a_indices = j - s_range
        if len(s_range) > 0:
            c[c_idx] -= delta_j * a[a_indices]
        
        # Add the cross term contribution from delta_i * delta_j
        cross_shift1 = j - i + n_points - 1
        cross_shift2 = i - j + n_points - 1
        if 0 <= cross_shift1 < len(c):
            c[cross_shift1] += delta * delta
        if 0 <= cross_shift2 < len(c):
            c[cross_shift2] += delta * delta

    return c

    best_h = h_initial.copy()

    TARGET_SUM = n_points / 2
    
    # Initialize correlation matrix
    c_full = np.correlate(best_h, 1 - best_h, mode="full") * dx
    
    # Main optimization loop
    max_iter = 200000
    iter_count = 0
    
    # Temperature schedule for simulated annealing
    temp_schedule = np.logspace(0, -3, max_iter)
    
    while iter_count < max_iter and time.time() - start_time < budget_s:
        iter_count += 1
        
        # Anneal temperature
        current_temp = temp_schedule[min(iter_count, len(temp_schedule) - 1)]
        
        # Randomly choose two indices to modify
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        if i == j:
            continue
            
        # Choose delta amount with adaptive schedule
        delta = np.random.uniform(0.001, 0.02)
        
        # Make a copy of current solution for rollback
        h_new = best_h.copy()
        c_new = c_full.copy()
        
        # Apply change to h
        h_new[i] = h_new[i] - delta
        h_new[j] = h_new[j] + delta
        
        # Re-project to constraint
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Update correlation incrementally 
        c_new = update_correlation_incremental(c_new, h_new, 1 - h_new, i, j, delta)
        
        # Compute new C5
        new_c5 = np.max(c_new)
        
        # Accept or reject based on simulated annealing
        if new_c5 < best_c5 or random.random() < np.exp(-(new_c5 - best_c5) / current_temp):
            best_h = h_new.copy()
            c_full = c_new.copy()
            best_c5 = new_c5
            
    return (best_h.tolist(), float(best_c5), n_points)
