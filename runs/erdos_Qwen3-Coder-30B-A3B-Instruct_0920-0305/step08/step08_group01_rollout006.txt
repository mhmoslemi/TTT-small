import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

# Global constants
n_points = 180  # Grid resolution for binary function
dx = 2.0 / n_points
TARGET_SUM = n_points / 2  # Required integral constraint
m_intervals = 5  # Number of intervals

def decode_intervals(z):
    """Convert a parameter vector z to interval endpoints s and e"""
    # z is [l1,...,lm, g1,...,gm], 2m elements
    l = np.array(z[:m_intervals])
    g = np.array(z[m_intervals:])
    
    # Ensure lengths positive and renormalize
    l = np.clip(l, 1e-6, 1.0)
    l = l / np.sum(l)  # total mass = 1
    
    # Compute endpoints
    s = np.zeros(m_intervals)
    e = np.zeros(m_intervals)
    s[0] = 0
    e[0] = l[0]
    for i in range(1, m_intervals):
        s[i] = e[i-1] + g[i-1]
        e[i] = s[i] + l[i]
        
    # If end exceeds 2, compress the gaps
    total_length = e[-1]
    if total_length > 2:
        scale = 2.0 / total_length
        l *= scale
        e = s + l
    
    return s, e

def build_binary_from_intervals(s, e, n=n_points):
    """Build binary vector h from interval endpoints"""
    h = np.zeros(n)
    grid = np.linspace(0, 2, n, endpoint=False) + dx/2  # grid cell centers
    
    for i in range(len(s)):
        mask = (grid >= s[i]) & (grid < e[i])
        h[mask] = 1.0
        
    return h

def evaluate_c5_fft(h):
    """Fast evaluation of C5 using FFT"""
    # h is binary, compute correlation
    h_rev = h[::-1]
    b = 1 - h
    corr = correlate(h_rev, b, mode='full')
    return np.max(corr) * dx

def objective(z):
    """Objective function - minimize max correlation"""
    s, e = decode_intervals(z)
    h = build_binary_from_intervals(s, e)
    return evaluate_c5_fft(h)

def local_refinement(theta_DE):
    """Refine the interval positions with deterministic edge moves"""
    s, e = decode_intervals(theta_DE)
    h = build_binary_from_intervals(s, e)
    
    # Quick check
    best_c5 = evaluate_c5_fft(h)
    best_s, best_e = s.copy(), e.copy()
    
    max_iter = 1000  # Max number of local moves
    
    for _ in range(max_iter):
        improved = False
        for i in range(len(s)):
            # Try moving left edge
            delta_s = dx / 2
            new_s = s[i] - delta_s
            if i == 0:
                new_s = max(0, new_s)
            else:
                new_s = max(s[i-1] + 0.01*dx, new_s)
            
            # Try moving right edge
            delta_e = dx / 2
            new_e = e[i] + delta_e
            if i == len(s) - 1:
                new_e = min(2, new_e)
            else:
                new_e = min(e[i+1] - 0.01*dx, new_e)
            
            # If our moves are valid test them
            if new_s < new_e:
                if i == 0:
                    new_s_vals = [new_s]
                else:
                    new_s_vals = [new_s, s[i]]
                    
                if i == len(s) - 1:
                    new_e_vals = [new_e]
                else:
                    new_e_vals = [new_e, e[i]]
                
                # Update interval and recomputes
                s[i] = new_s
                e[i] = new_e
                h = build_binary_from_intervals(s, e)
                c5 = evaluate_c5_fft(h)
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_s = s.copy()
                    best_e = e.copy()
                    improved = True
                    break

        if not improved:
            break
            
    return best_s, best_e

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: Collect initial population
    initial_pop = []
    best_c5 = float('inf')
    best_params = None
    
    # Generate some random interval configurations to seed DE
    for _ in range(20):
        z = np.random.rand(2 * m_intervals)
        # Make sure last l + g is not zero
        z[-1] = np.clip(z[-1], 1e-4, 1.0)
        
        try:
            s, e = decode_intervals(z)
            if e[-1] <= 2.0:
                c5 = objective(z)
                if c5 < best_c5:
                    best_c5 = c5
                    best_params = z.copy()
                initial_pop.append(z)
        except Exception:
            continue
    
    # Phase 2: Run differential evolution
    bounds = [(0, 1) for _ in range(2 * m_intervals)]
    de_options = {
        'strategy': 'best1bin',
        'maxiter': 100,
        'popsize': 15,
        'tol': 1e-6,
        'mutation': (0.5, 1.0),
        'recombination': 0.9,
        'atol': 1e-6,
        'updating': 'immediate'
    }
    
    def callback_de(xk, convergence):
        try:
            s, e = decode_intervals(xk)
            h = build_binary_from_intervals(s, e)
            c5 = evaluate_c5_fft(h)
            nonlocal best_c5, best_params
            if c5 < best_c5:
                best_c5 = c5
                best_params = xk.copy()
        except Exception:
            pass
        return False  # Don't stop early in DE
    
    try:
        result = differential_evolution(
            objective,
            bounds,
            seed=seed,
            callback=callback_de,
            maxiter=100,
            popsize=15,
            strategy='best1bin',
            polish=False
        )
        
        if result.success:
            final_params = result.x
        else:
            final_params = best_params
    except:
        final_params = best_params
    
    # Phase 3: Local refinement (neighborhood search)
    if final_params is not None:
        s, e = decode_intervals(final_params)
        refined_s, refined_e = local_refinement(final_params)
        
        h = build_binary_from_intervals(refined_s, refined_e)
        c5_value = evaluate_c5_fft(h)
        
        if c5_value < best_c5:
            best_c5 = c5_value
            s_final, e_final = refined_s, refined_e
        else:
            s_final, e_final = s, e
    else:
        # Fallback if something goes wrong
        s_final = np.zeros(m_intervals)
        e_final = np.linspace(0, 1, m_intervals + 1)[1:]
        
    # Final evaluation with standard correlate (to make sure)
    h_final = build_binary_from_intervals(s_final, e_final)
    c5_final = evaluate_c5_fft(h_final)
    
    # Re-evaluate with the exact numpy correlate method
    # To match the test suite exactly
    h_np = h_final.copy()
    corr = correlate(h_np, 1-h_np, mode='full')
    c5_exact = dx * np.max(corr)
    
    # Build final binary B
    B = h_final.tolist()
    
    return B, float(c5_exact), n_points
