import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

# Global variables for the problem
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_overlap(h):
    """Compute the maximum overlap using optimized FFT approach"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def evaluate_overlap(h):
    """Evaluate overlap using convolution with full mode"""
    corr = correlate(h, 1-h, mode="full")
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Projection onto [0,1]^n with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlation_fast(h):
    """Compute correlations using forward pass"""
    corr = np.correlate(h, 1-h, mode="full")
    # Convert from index-based to shift-based numbering
    n = len(h)
    shifts = np.arange(-(n-1), n)
    return shifts, corr

def solve_milp_erdos(n, time_limit_s=60):
    """Solve the MILP formulation for Erdős constant"""
    
    # Create binary variables for h_i
    x = cp.Variable(n, boolean=True)
    
    # Epigraph variable
    t = cp.Variable()
    
    # Constraints for mass
    mass_constraint = cp.sum(x) == TARGET_SUM
    
    # Set of shifts to consider (start with a small subset)
    shifts = list(range(1, 6))  # Only small shifts to start
    
    # Initialize auxiliary variables and constraints
    y_vars = {}
    aux_constraints = []
    
    # Build initial constraints
    for s in shifts:
        for i in range(n):
            if i + s < n:
                y_vars[(i, s)] = cp.Variable(boolean=True)
                aux_constraints.extend([
                    y_vars[(i, s)] <= x[i],
                    y_vars[(i, s)] <= x[i + s],
                    y_vars[(i, s)] >= x[i] + x[i + s] - 1
                ])
    
    # Epigraph constraints
    epigraph_constraints = []
    for s in shifts:
        expr = sum(y_vars[(i, s)] for i in range(n) if i + s < n)
        epigraph_constraints.append(expr >= TARGET_SUM - t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Build the problem
    prob = cp.Problem(objective, 
                      [mass_constraint] + aux_constraints + epigraph_constraints + [t >= 0])
    
    # Solve iteratively
    max_iter = 500
    iter_count = 0
    
    # Track best solution
    best_solution = None
    best_value = float('inf')
                    
    while iter_count < max_iter:
        try:
            prob.solve(solver=CPLEX, time_limit=time_limit_s, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                solution = x.value.round().astype(int).tolist()
                solution_float = [float(val) for val in solution]
                
                if not best_solution or evaluate_overlap(solution_float) < best_value:
                    best_solution = solution_float
                    best_value = evaluate_overlap(solution_float)
                
                # Evaluate exact worst correlation
                shifts_idx, corr = compute_correlation_fast(solution_float)
                max_corr_idx = np.argmax(corr)
                max_corr_val = corr[max_corr_idx]
                
                if prob.value - max_corr_val < 1e-8:
                    break  # Meets optimality condition
                    
                # Add the most violating shift
                shift = shifts_idx[max_corr_idx]
                if shift not in shifts:
                    shifts.append(shift)
                    
                    # Add new auxiliary variables and constraints for this shift
                    for i in range(n):
                        if i + shift < n:
                            y_vars[(i, shift)] = cp.Variable(boolean=True)
                            aux_constraints.extend([
                                y_vars[(i, shift)] <= x[i],
                                y_vars[(i, shift)] <= x[i + shift],
                                y_vars[(i, shift)] >= x[i] + x[i + shift] - 1
                            ])
                    
                    # Add epigraph constraint for new shift
                    expr_new = sum(y_vars[(i, shift)] for i in range(n) if i + shift < n)
                    epigraph_constraints.append(expr_new >= TARGET_SUM - t)
                    
            else:
                break
                
        except Exception as e:
            break
            
        iter_count += 1
    
    return best_solution, best_value

def smooth_optimize_constrained(h_start, t_init):
    """Smooth optimization to refine binary solution"""
    # Softmax-based smooth upper bound
    gamma = 100.0
    
    def smooth_objective(h, t):
        # Compute correlations using FFT
        shifts, corr = compute_correlation_fast(h)
        max_corr = np.max(corr)
        # Soft approximation of max
        total_exp = np.sum(np.exp(gamma * (corr - t)))
        return t + (1/gamma) * np.log(total_exp)
    
    def constraint(h):
        return np.sum(h) - TARGET_SUM
    
    # Wrapped objective for scipy
    def obj(h):
        return smooth_objective(h, t_init)
    
    # Constraints
    cons = {'type': 'eq', 'fun': constraint}
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimizer
    res = minimize(obj, h_start, method='SLSQP', bounds=bounds, 
                   constraints=cons, options={'maxiter': 100, 'ftol': 1e-12})
    
    if res.success:
        return project_to_sum_box(res.x, TARGET_SUM)
    else:
        return h_start

def upscale_initial_binary(h_coarse):
    """Create a better initial solution on a refined grid"""
    n_fine = 200
    dx_fine = 2.0 / n_fine
    h_fine = np.zeros(n_fine)
    
    # Fill with coarse solution values
    for j in range(n_fine):
        idx_coarse = int((j * n_points) // n_fine)
        h_fine[j] = h_coarse[idx_coarse]
    
    # Project to correct sum
    h_fine = project_to_sum_box(h_fine, n_fine / 2.0)
    return h_fine

    np.random.seed(seed)
    start_time = time.time()

    h_best = None

    time_used = 0
    
    for attempt in range(3):

            break
            
        remaining_time = budget_s - time_used
        time_limit = min(100, remaining_time // 2)
        
        h_milp, c5_milp = solve_milp_erdos(n_points, time_limit)
        
        if h_milp is not None and c5_milp < c5_best:
            h_best = h_milp
            c5_best = c5_milp
            
        time_used = time.time() - start_time
    
    # If coarse solution was found, try upsampling to finer grid
    if h_best is not None and time_used < 0.8 * budget_s:
        try:
            h_fine = upscale_initial_binary(h_best)
            c5_fine = compute_overlap(h_fine)
            if c5_fine < c5_best:
                h_best = h_fine
                c5_best = c5_fine
        except:
            pass
    
    # Phase 2: Continuous polishing
    if h_best is not None and time_used < 0.95 * budget_s:
        # Compute epigraph value directly from FFT
        shifts, corr = compute_correlation_fast(h_best)
        t_estimate = np.max(corr)
        
        h_polished = smooth_optimize_constrained(h_best, t_estimate)
        c5_polished = compute_overlap(h_polished)
        
        if c5_polished < c5_best:
            h_best = h_polished
            c5_best = c5_polished
    
    # Final validation with FFT and return
    if h_best is not None:
