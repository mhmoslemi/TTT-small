import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def project_sum_box(v, M):
    """Project vector v onto the box [0,1]^n and sum constraint sum = M"""
    # Clip to box
    v = np.clip(v, 0, 1)
    # Adjust to sum constraint
    diff = M - np.sum(v)
    if abs(diff) < 1e-12:
        return v
    # Use bisection to find lambda for waterfilling
    a, b = -1.0, 1.0
    tol = 1e-10
    
    def f(lam):
        v_test = np.clip(v + lam, 0, 1)
        return np.sum(v_test) - M
    
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    lam = (a + b) / 2
    return np.clip(v + lam, 0, 1)

def solve_milp_binary(N, M, max_time_s):
    """
    Solve the binary MILP for maximum minimum overlap.
    Returns the binary vector y (0/1) if successful.
    """
    try:
        # Variables
        y = cp.Variable(N, boolean=True)
        t = cp.Variable()
        
        # Constraints
        constraints = [cp.sum(y) == M]
        
        # w variables for each shift k
        w_vars = {}
        shift_sums = []
        
        for k in range(1, N):
            # non-zero indices for the shift k
            indices = range(N - k)
            if len(indices) <= 0:
                continue
            w_k = cp.Variable(len(indices))  # Continuous
            w_vars[k] = w_k
            
            # Linearization constraints
            constraints += [w_k <= y[indices]]
            constraints += [w_k <= y[k:]]
            constraints += [w_k >= y[indices] + y[k:] - 1]
            constraints += [w_k >= 0]
            
            shift_sum = cp.sum(w_k)
            shift_sums.append(shift_sum)
            constraints += [shift_sum >= t]
        
        # Bound t
        constraints += [t >= 0, t <= M]
        
        # Objective
        prob = cp.Problem(cp.Maximize(t), constraints)
        
        # Solve
        prob.solve(solver=cp.CBC, max_seconds=max_time_s, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            y_opt = y.value.astype(int)
            return y_opt
        
    except Exception as e:
        return None
    return None

def evaluate_c5(h):
    """Fast evaluation using FFT."""
    corr = correlate(h, 1-h, mode="full")
    return np.max(corr) * (2.0 / len(h))

def subgradient_descent(h0, M, max_iter=10000):
    """Run deterministic subgradient descent on h0."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    alpha0 = 0.01
    
    for t in range(1, max_iter + 1):
        # Compute correlation and find max shift
        corr = correlate(h, 1-h, mode="full")
        k_max = np.argmax(corr) - (len(h) - 1)
        
        # Compute subgradient
        g = np.ones_like(h)
        if k_max >= 0:
            g[:len(h)-k_max] -= h[k_max:]
        else:
            k = -k_max
            g[k:] -= h[:len(h)-k]
            
        if k_max > 0:
            g[k_max:] -= h[:len(h)-k_max]
        elif k_max < 0:
            k = -k_max
            g[:len(h)-k] -= h[k:]
        
        # Gradient step
        alpha = alpha0 / np.sqrt(t)
        h_new = h - alpha * g
        
        # Project back
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, M)
        
        # Evaluate and maybe accept
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new.copy()
    
    return best_h, best_c5

def stochastic_refinement(h0, M, max_iter=5000):
    """Simple local stochastic refinement."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    temp_schedule = lambda t: 1.0 / (1 + t / 1000)
    
    for t in range(max_iter):
        # Random swap
        i, j = np.random.choice(len(h), size=2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum_box(h_new, M)
        
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        elif np.random.rand() < np.exp(-(c5_new - evaluate_c5(h)) / temp_schedule(t)):
            h = h_new.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    
    # Parameters
    N = 120  # Same as existing setup
    M = N // 2
    dx = 2.0 / N
    
    # Time slices
    tau1 = int(0.4 * budget_s)
    tau2 = int(0.35 * budget_s)
    tau3 = budget_s - tau1 - tau2
    
    # Phase 1: Solve MILP
    y_opt = None
    h0 = None
    c5_opt = float('inf')
    try:
        y_opt = solve_milp_binary(N, M, tau1)
        if y_opt is not None:
            h0 = y_opt.astype(float)
            c5_opt = evaluate_c5(h0)
    except:
        y_opt = None
        
    if y_opt is None:
        # Fallback: random initial vector
        h0 = np.random.uniform(0, 1, N)
        h0 = project_sum_box(h0, M)
        c5_opt = evaluate_c5(h0)
    
    # Phase 2: Subgradient descent
    if time.time() - start < tau1 + tau2:
        h_descent, c5_descent = subgradient_descent(h0, M, max_iter=tau2*50)
        if c5_descent < c5_opt:
            h0 = h_descent
            c5_opt = c5_descent
    
    # Phase 3: Stochastic refinement
    if time.time() - start < budget_s:
        h_refine, c5_refine = stochastic_refinement(h0, M, max_iter=tau3*10)
        if c5_refine < c5_opt:
            h0 = h_refine
            c5_opt = c5_refine
    
    return (h0.tolist(), float(c5_opt), N)
