import numpy as np
from scipy.signal import correlate
import time

def project_to_box_sum(h, target_sum):
    """Project h onto box [0,1] with sum = target_sum using bisection."""
    h = np.array(h, dtype=np.float64)
    lb, ub = 0.0, 1.0
    if np.sum(h) == target_sum:
        return h
    
    def f(lam):
        h_proj = np.clip(h + lam, lb, ub)
        return np.sum(h_proj) - target_sum
    
    # Find initial bracketing values
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Bisection search
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    return np.clip(h + (a + b)/2, lb, ub)

def update_correlation_single_idx(c, h, b, i, delta):
    """Update correlation c for a single index change at i."""
    n = len(h)
    h_rev = h[::-1]
    
    # Update for the contribution from b[i+s] part (s from index -n+1 to n-1)
    s_offset = n-1
    start_b = s_offset - i
    if start_b < 0:
        start_b = 0
    end_b = start_b + n - i
    if end_b > len(c):
        end_b = len(c)
    if (n-1-i) < 0:
        # Some elements from b need to be skipped
        c[start_b:end_b] += delta * b[0:i+1]
    else:
        c[start_b:end_b] += delta * b[i:]
    
    # Update for the contribution from h[i-s] part
    start_h = i
    end_h = i + n
    if end_h > len(c):
        end_h = len(c)
    if start_h < 0:
        c[0:end_h] -= delta * h_rev[-i:]
    else:
        c[start_h:end_h] -= delta * h_rev[:n]

def update_correlation_double_idx(c, h, b, i, j, delta):
    """Update correlation c for moving delta mass from i to j."""
    # Handle first index (h[i] -= delta)
    update_correlation_single_idx(c, h, b, i, -delta)
    # Handle second index (h[j] += delta)  
    update_correlation_single_idx(c, h, b, j, delta)
    # Add correction due to quadratic terms (approx 0 for small delta)
    shift1 = j - i
    shift2 = i - j
    idx1 = shift1 + (len(c) // 2)
    idx2 = shift2 + (len(c) // 2)
    if 0 <= idx1 < len(c):
        c[idx1] += delta * delta
    if 0 <= idx2 < len(c):
        c[idx2] += delta * delta

def compute_correlation(h):
    """Compute full correlation (used for sanity check)."""
    b = 1 - h
    c = correlate(h, b, mode='full')
    return c
