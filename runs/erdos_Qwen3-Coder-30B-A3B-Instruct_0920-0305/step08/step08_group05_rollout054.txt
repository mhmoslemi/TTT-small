import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_initial_h():
    """Generate a better initial construction based on known patterns"""
    h = np.zeros(n_points)
    
    # Try a more sophisticated pattern inspired by optimal constructions
    # Start with a piecewise constant function with a bump in the middle
    # We'll use 4 segments: low, high, low, high
    seg_size = n_points // 4
    h[:seg_size] = 0.2
    h[seg_size:2*seg_size] = 0.8
    h[2*seg_size:3*seg_size] = 0.2
    h[3*seg_size:] = 0.8
    
    # Normalize to the correct sum
    current_sum = np.sum(h)
    h *= TARGET_SUM / current_sum
    
    # Apply box constraints
    h = np.clip(h, 0, 1)
    
    # Project to sum constraint
    h = project_to_sum_box(h, TARGET_SUM)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = generate_initial_h()
    best_c5 = evaluate_c5(best_h)
    
    # Try various optimization approaches
    strategies = []
    
    # Strategy 1: High-quality initial construction
    strategies.append(('initial', best_h.copy()))
    
    # Strategy 2: Random perturbations with guided constraints  
    for _ in range(3):
        h_start = np.random.uniform(0, 1, n_points)
        h_start = project_to_sum_box(h_start, TARGET_SUM)
        strategies.append(('random', h_start))
    
    # Strategy 3: Piecewise constant with specific pattern
    h_piece = np.zeros(n_points)
    segments = [0.15, 0.7, 0.15, 0.7]
    seg_size = n_points // 4
    for i in range(4):
        start = i * seg_size
        end = (i+1) * seg_size if i < 3 else n_points
        h_piece[start:end] = segments[i]
    h_piece = project_to_sum_box(h_piece, TARGET_SUM)
    strategies.append(('piecewise', h_piece))
    
    # Strategy 4: Gaussian-like shape
    x = np.linspace(0, 2, n_points)
    center = 1.0
    sigma = 0.3
    h_gauss = np.exp(-0.5 * ((x - center) / sigma)**2)
    h_gauss = project_to_sum_box(h_gauss, TARGET_SUM)
    strategies.append(('gauss', h_gauss))
    
    # Strategy 5: Binary pattern with more smoothness
    h_binary = np.zeros(n_points)
    for i in range(n_points):
        if (i // (n_points // 8)) % 2 == 0:
            h_binary[i] = 0.3
        else:
            h_binary[i] = 0.8
    h_binary = project_to_sum_box(h_binary, TARGET_SUM)
    strategies.append(('binary', h_binary))
    
    # Optimize each starting point
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Try optimization from each starting point
    for strategy_name, h_start in strategies:
        if time.time() - start_time > budget_s:
            break
            
        h_in = h_start.copy()
        
        try:
            res = minimize(
                objective, 
                h_in, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 3000, 'ftol': 1e-13, 'gtol': 1e-13}
            )
            if res.success:
                h_candidate = project_to_sum_box(res.x, TARGET_SUM)
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
        except:
            continue

    # Perform some additional local Newton-like optimizations
    refine_iter = 0
    max_refine = 500
    while time.time() - start_time < budget_s and refine_iter < max_refine:
        refine_iter += 1
        
        # Make a more aggressive perturbation
        h_changed = best_h + np.random.normal(0, 0.01, n_points)  
        h_changed = project_to_sum_box(h_changed, TARGET_SUM)
        
        try:
            res = minimize(
                objective, 
                h_changed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-13, 'gtol': 1e-13}
            )
            if res.success:
                h_candidate = project_to_sum_box(res.x, TARGET_SUM)
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
        except:
            continue
            

    return (best_h.tolist(), float(best_c5), n_points)
