import time

def run(seed=42, budget_s=600, **kwargs):

    # Setup
    n = 120
    dx = 2.0 / n
    TARGET_SUM = n // 2
    
    def compute_corr_fft(h):
        """Compute correlation using FFT"""
        h_rev = h[::-1]
        b = 1 - h
        L = 2 * len(h) - 1
        corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
        return corr

    def c5_from_corr(corr, n):
        """Convert correlation result to C5 value"""
        return dx * np.max(corr)

    def evaluate_c5(h):
        """Fast evaluation of C5 using FFT"""
        corr = compute_corr_fft(h)
        return c5_from_corr(corr, n)

    # Generate shift matrices for SDP
    def get_shift_matrices(n):
        R_list = []
        for s in range(1, n//2 + 1):
            R = np.zeros((n, n))
            for i in range(n - s):
                R[i, i + s] = 1
            R_list.append(R)
        return R_list

    # SDP solution
    def solve_sdp(n, max_time=150):
        R_list = get_shift_matrices(n)
        
        # Variables
        X = cp.Variable((n, n), symmetric=True)
        mu = cp.Variable(n)
        lam = cp.Variable()
        
        # Constraints
        constraints = [
            X >> 0,  # PSD
            cp.diag(X) == mu,
            0 <= mu,
            mu <= 1,
            cp.sum(mu) == TARGET_SUM,
            lam <= 0.5  # Initial guess
        ]
        
        # Add overlapping constraints
        for R in R_list:
            constraints.append(cp.trace(R @ X) >= lam)
        
        # Objective
        objective = cp.Maximize(lam)
        prob = cp.Problem(objective, constraints)
        
        # Solve with SCS solver
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=3000, time_limit_secs=max_time)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                X_opt = X.value
                lam_opt = lam.value
                mu_opt = mu.value
                return X_opt, mu_opt, lam_opt
        except:
            pass
        return None, None, None

                    if x[i] == 0 and x[i + shift] == 1:

                        Good.append(i)

                    corr = update_corr(corr, h, i, j)

            if c5 < best_c5:
