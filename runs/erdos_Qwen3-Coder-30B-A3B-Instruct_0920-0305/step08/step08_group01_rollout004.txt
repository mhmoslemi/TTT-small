import numpy as np

import cvxpy as cp
import time

def compute_shift_matrices(n):
    """Return list of shift matrices R_s for s = 1..floor(n/2)"""
    Rs = []
    for s in range(1, (n//2) + 1):
        R = np.zeros((n, n))
        for i in range(n - s):
            R[i, i + s] = 1
        Rs.append(R)
    return Rs

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = correlate(h, 1-h, mode="full")
    return np.max(corr)

def solve_sdp(n, Rs):
    """Solve the SDP relaxation"""
    # Variables
    X = cp.Variable((n, n), symmetric=True)
    mu = cp.Variable(n)
    lam = cp.Variable()
    
    # Constraints
    constraints = [X >> 0]  # PSD
    for i in range(n):
        constraints.append(X[i,i] == mu[i])
        constraints.append(0 <= mu[i])
        constraints.append(mu[i] <= 1)
    
    constraints.append(cp.sum(mu) == n/2)
    
    # Add shift constraints
    for R in Rs:
        constraints.append(cp.trace(R @ X) >= lam)
    
    # Objective
    objective = cp.Maximize(lam)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INFEASIBLE]:
            return X.value, mu.value
        return None, None
    except:
        return None, None

                good_positions.append(i)
