import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import Minimize, Variable, sum as cp_sum, Problem

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
C5_BOUND = float('inf')

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Exact projection onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def waterfilling_projection(h, target_sum):
    """Waterfilling projection with O(N) complexity for sum constraint + box"""
    h = np.clip(h, 0, 1)
    
    def project_sum(x):
        """Projection onto sum constraint only"""
        def f(lam):
            projected = np.clip(x + lam, 0, 1)
            return np.sum(projected) - target_sum
            
        a, b = -1.0, 1.0
        while f(a) * f(b) > 0:
            a -= 0.1
            b += 0.1

        tol = 1e-10
        for _ in range(50):
            m = (a + b) / 2
            fm = f(m)
            if abs(fm) < tol:
                break
            if fm * f(a) < 0:
                b = m
            else:
                a = m

        return np.clip(x + (a + b)/2, 0, 1)
    
    return project_sum(h)

def greedy_binary_construction():
    """Greedy spacing heuristic for binary construction"""
    h = np.zeros(n_points)
    ones_positions = set()
    slots = list(range(n_points))
    
    # Simple spacing heuristic: try to place 1s with minimum overlap
    for _ in range(n_points // 2):  # Need to place exactly n_points / 2 ones
        best_pos = -1
        best_score = -float('inf')
        
        for pos in slots:
            new_h = h.copy()
            new_h[pos] = 1.0
            
            if len(ones_positions) > 0:  # Check overlaps
                # Binary case, easy to compute overlaps
                # Score is inverse of total overlap (helps remove overlap)
                score = 0
                for one_pos in ones_positions:
                    k = abs(one_pos - pos)
                    if k < n_points:
                        # This would be contribution to overlap from current placement
                        score += 1
            else:
                score = 0
                
            if score > best_score:
                best_score = score
                best_pos = pos
                
        if best_pos >= 0:
            h[best_pos] = 1.0
            ones_positions.add(best_pos)
            slots.remove(best_pos)
    
    # Ensure exact sum
    current_sum = np.sum(h)
    diff = current_sum - TARGET_SUM
    if abs(diff) > 1e-8:
        # Adjust by flipping elements
        for i in range(n_points):
            if h[i] == 1.0:
                h[i] = 0.0
                diff -= 1
                if abs(diff) < 1e-8:
                    break
            if diff <= 0:
                break
        for i in range(n_points):
            if diff > 0 and h[i] == 0.0:
                h[i] = 1.0
                diff -= 1
                if abs(diff) < 1e-8:
                    break
    
    return project_to_sum_box(h, TARGET_SUM)

def binary_milp():
    """Mixed Integer Linear Program (MILP) formulation for highest overlap symmetric"""
    try:
        # Variables
        y = Variable(n_points, boolean=True)
        w = Variable((n_points - 1, n_points))
        t = Variable()

        # Constraints
        constraints = [
            cp_sum(y) == TARGET_SUM,  # Sum constraint
            t >= 0,
            t <= TARGET_SUM
        ]

        # Add binary constraints
        for k in range(1, n_points):
            for i in range(n_points - k):
                constraints += [
                    w[k-1, i] <= y[i],
                    w[k-1, i] <= y[i + k],
                    w[k-1, i] >= y[i] + y[i + k] - 1,
                    w[k-1, i] >= 0,
                ]
                # Ensure w equals y[i] * y[i+k]
                constraints += [
                    cp_sum(w[k-1, :]) >= t
                ]

        # Objective
        objective = Minimize(-t)

        # Create problem
        prob = Problem(objective, constraints)

        # Solve
        prob.solve(solver=cp.CBC, max_seconds=0.4 * 600, verbose=False)
        
        if prob.status in ["optimal", "optimal_inaccurate"]:
            y_values = y.value
            h = y_values.astype(float)
            c5_val = dx * (TARGET_SUM - (-prob.value))
            return h, c5_val
        
        return None, None
        
    except Exception as e:
        return None, None

def sub_gradient_descent(h0, max_time_s):
    """Deterministic sub-gradient descent to refine continuous solution"""
    start_time = time.time()
    h = h0.copy()

    t_iter = 1
    alpha0 = 0.5 * dx

    while time.time() < start_time + max_time_s:
        # Compute all correlations using FFT
        corr = compute_corr_fft(h)
        k_star_idx = np.argmax(corr)
        k_star = k_star_idx if k_star_idx < n_points else k_star_idx - n_points
        
        # Compute sub-gradient g
        g = np.zeros_like(h)
        if k_star < n_points - 1:
            for i in range(n_points):
                # Avoid negative index
                idx1 = (i + k_star) % n_points
                idx2 = (i - k_star) % n_points
                g[i] = 1.0 - h[idx1] - h[idx2]
        else:
            # For k = 0 to n-1
            for i in range(n_points):
                idx1 = (i + k_star - n_points + 1) % n_points
                idx2 = (i - k_star + n_points - 1) % n_points
                g[i] = 1.0 - h[idx1] - h[idx2]
        
        # Step size
        alpha = alpha0 / np.sqrt(t_iter)
        
        h_new = h - alpha * g
        h_new = waterfilling_projection(h_new, TARGET_SUM)
        
        h = h_new
        t_iter += 1

    return h

def simulated_annealing(h0, max_time_s):
    """Simulated Annealing after sub-gradient"""
    start_time = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    T_start = 5 * dx
    T_min = 1e-8
    T = T_start
    
    while time.time() < start_time + max_time_s:
        # 2-index swap
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
		
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Re-project to sum = M
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        delta_c5 = new_c5 - best_c5
        
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new.copy()
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                
        # Cool down
        T = max(T_min, T * 0.999)

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx, C5_BOUND
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2

    # Phase 1: Binary MILP or greedy start
    h0 = None
    phase1_end = start_time + 0.4 * budget_s
    try:
        h0, c5_0 = binary_milp()
        if h0 is None or c5_0 is None:
            raise RuntimeError("MILP failed or timed out")
        else:
            C5_BOUND = min(C5_BOUND, c5_0)
    except:
        h0 = greedy_binary_construction()
        c5_0 = evaluate_c5(h0)
        C5_BOUND = min(C5_BOUND, c5_0)
    
    # Phase 2: Continuous sub-gradient descent
    phase2_end = start_time + 0.75 * budget_s
    h = sub_gradient_descent(h0, max_time_s=phase2_end - start_time)
    c5_cand = evaluate_c5(h)
    C5_BOUND = min(C5_BOUND, c5_cand)
    
    # Usually the sub-gradient step achieves the best value,
    # but also test if the stochastic stage gives an improvement
    phase3_end = start_time + budget_s
    h_final, c5_final = simulated_annealing(h, max_time_s=phase3_end - phase2_end)
    C5_BOUND = min(C5_BOUND, c5_final)
    
    return (h_final.tolist(), C5_BOUND, n_points)
