def project_to_box_sum(h, target_sum):
    """Project h onto {x : 0 ≤ xᵢ ≤ 1, sum(x) = target_sum}"""
    # Binary search for lambda
    h_clipped = np.clip(h, 0, 1)
    sum_clipped = np.sum(h_clipped)
    
    if abs(sum_clipped - target_sum) < 1e-12:
        return h_clipped
    
    # find lambda such that sum(clamp(h + lambda, 0, 1)) = target_sum
    a, b = -1.0, 1.0
    def func(lam):
        temp = np.clip(h + lam, 0, 1)
        return np.sum(temp) - target_sum
    
    # Ensure a and b bracket the root
    while func(a) * func(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Bisection search
    for _ in range(50):
        m = (a + b) / 2
        fm = func(m)
        if abs(fm) < 1e-12:
            break
        if fm * func(a) < 0:
            b = m
        else:
            a = m
    
    return np.clip(h + (a + b) / 2, 0, 1)

def compute_c_full(h):
    # Re-compute full size correlation
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return corr
