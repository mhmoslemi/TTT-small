import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h, w_norm):
    """Compute gradient using convolution approach"""
    n = len(h)
    L = 2 * n - 1
    offset = n - 1
    
    # Using convolution for both terms
    conv_wh = np.convolve(w_norm, h, mode='full')
    conv_rev = np.convolve(w_norm[::-1], h, mode='full')
    
    # Extract contributions for valid range
    grad = 1.0 - (conv_rev[offset:offset+n] + conv_wh[offset:offset+n])
    
    return grad

    """Initialize random h vector with correct sum"""
    h = np.random.random(n)
    h = project_to_sum_box(h, target_sum)
    return h
