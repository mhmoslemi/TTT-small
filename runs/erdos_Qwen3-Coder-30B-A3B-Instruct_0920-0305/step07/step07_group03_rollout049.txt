import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def simulated_annealing(h0, max_iter=5000, initial_temp=0.1, cooling_rate=0.9995):
    """Simulated annealing optimization"""
    h = h0.copy()
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    temp = initial_temp
    
    for i in range(max_iter):
        # Create neighbor solution
        h_new = h.copy()
        
        # Perturb a few random points
        num_changes = max(1, min(10, n_points // 20))
        indices = random.sample(range(n_points), num_changes)
        for idx in indices:
            delta = random.uniform(-0.05, 0.05)
            h_new[idx] += delta
            
        # Project back to feasible region
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        
        # Accept or reject the move
        if new_c5 < current_c5 or random.random() < np.exp((current_c5 - new_c5) / temp):
            h = h_new
            current_c5 = new_c5
            
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Cool down temperature
        temp *= cooling_rate
        
        if temp < 1e-8:
            break
            
    return best_h, best_c5

def genetic_algorithm(max_generations=100, population_size=50):
    """Genetic algorithm optimization"""
    def create_individual():
        h = np.random.uniform(0, 1, n_points)
        return project_to_sum_box(h, TARGET_SUM)
    
    def fitness(h):
        return -evaluate_c5(h)  # Negative because we minimize
    
    def crossover(parent1, parent2):
        # Single-point crossover
        crossover_point = random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        return project_to_sum_box(child, TARGET_SUM)
    
    def mutate(individual, mutation_rate=0.1):
        mutated = individual.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                mutated[i] += random.uniform(-0.05, 0.05)
        return project_to_sum_box(mutated, TARGET_SUM)
    
    # Initialize population
    population = [create_individual() for _ in range(population_size)]
    
    for generation in range(max_generations):
        # Evaluate fitness
        fitness_scores = [fitness(ind) for ind in population]
        
        # Keep best individuals
        sorted_indices = np.argsort(fitness_scores)[::-1]
        elite_size = population_size // 4
        new_population = [population[i] for i in sorted_indices[:elite_size]]
        
        # Create offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitness_scores, 3)
            parent2 = tournament_selection(population, fitness_scores, 3)
            
            child = crossover(parent1, parent2)
            child = mutate(child)
            
            new_population.append(child)
            
        population = new_population
    
    # Return best individual
    fitness_scores = [fitness(ind) for ind in population]
    best_idx = np.argmax(fitness_scores)
    return population[best_idx], -fitness_scores[best_idx]

def tournament_selection(population, fitness_scores, k):
    """Select individual using tournament selection"""
    tournament_indices = random.sample(range(len(population)), k)
    tournament_fitness = [fitness_scores[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmax(tournament_fitness)]
    return population[winner_idx]
