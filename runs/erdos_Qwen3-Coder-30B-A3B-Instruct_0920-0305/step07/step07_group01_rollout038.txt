def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr)

def fourier_basis(x, m):
    """Generate cosine basis function"""
    return np.cos(m * np.pi * x)

def construct_h_from_coeffs(coeffs, n_points=200):
    """Construct h from Fourier coefficients"""
    x = np.linspace(0, 2, n_points)
    # Use constant term and cos terms
    h = coeffs[0] * np.ones_like(x)  # constant term
    for m in range(1, len(coeffs)):
        h += coeffs[m] * fourier_basis(x, m)
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    return h
