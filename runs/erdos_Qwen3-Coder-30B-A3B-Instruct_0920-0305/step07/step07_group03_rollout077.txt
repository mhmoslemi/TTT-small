def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        prev_h = h.copy()
        prev_c5 = best_c5
        
        for idx in indices:
            old_val = h[idx]

                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    h = prev_h.copy()  # Revert
                else:
                    h[idx] = old_val  # Revert if out of bounds
        
        if best_c5 < prev_c5:
            h = best_h.copy()
        else:
            # Occasionally try random small perturbations
            if np.random.random() < 0.1:
                idx = np.random.randint(0, n_points)
                h[idx] += np.random.uniform(-0.01, 0.01)
                h = project_to_sum_box(h, TARGET_SUM)
                new_c5 = evaluate_c5(h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
    
    return best_h

def improve_peak_distribution(h):
    """Better algorithm for peak construction and refinement"""
    h = h.copy()
    
    # Add periodic-like peaks but with optimized placement
    # Look for low-value regions and inject strong values
    threshold = 0.3
    low_indices = np.where(h < threshold)[0]

            # Remove from potential low indices to avoid re-use
            low_indices = low_indices[low_indices != idx]
