import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_c5(h):
    """Fast evaluation of C5 using FFT"""
    n = len(h)
    # compute linear convolution via FFT
    nfft = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    B = np.fft.rfft(1 - h, n=nfft)
    conv = np.fft.irfft(H * B, n=nfft)[:2*n-1]
    return np.max(conv) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def heuristic_initialization():
    """Create a heuristic initial solution"""
    # Start with a pattern based on initial_h_values if available
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        x_base = np.linspace(0, 2, len(h_base))
        x_new = np.linspace(0, 2, n_points)
        h_interp = np.interp(x_new, x_base, h_base)
    else:
        # Create simple pattern: alternating intervals
        h_interp = np.zeros(n_points)
        interval_length = 2.0 / n_points
        for i in range(n_points):
            if (i // (n_points // 8)) % 2 == 0:
                h_interp[i] = 1.0
    # Ensure correct sum
    h_interp = project_to_sum_box(h_interp, TARGET_SUM)
    return h_interp

def binary_from_continuous(h_cont):
    """Convert continuous solution to binary while maintaining sum"""
    # Simple rounding approach: use threshold 0.5
    h_bin = (h_cont > 0.5).astype(float)
    # Ensure correct sum
    h_bin = project_to_sum_box(h_bin, TARGET_SUM)
    return h_bin

def greedy_improvement(h):
    """Apply greedy improvement targeting maximum overlap shift"""
    n = len(h)
    max_iter = 10000  # Limit iterations
    iter_count = 0
    
    while iter_count < max_iter:
        # Compute correlation
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n - 1)
        
        # Determine overlapping region indices for worst shift
        if max_shift >= 0:
            i_range = np.arange(0, n - max_shift)
            j_range = i_range + max_shift
        else:
            i_range = np.arange(-max_shift, n)
            j_range = i_range + max_shift  # shift negative pushes left
            
        # Find mismatched pairs
        # (1,0) mismatches: h[i] == 1 and h[j] == 0
        im1 = i_range[(h[i_range] == 1) & (h[j_range] == 0)]
        # (0,1) mismatches: h[i] == 0 and h[j] == 1
        im2 = i_range[(h[i_range] == 0) & (h[j_range] == 1)]
        
        if im1.size == 0 or im2.size == 0:
            break
            
        # pick random i, j and swap
        i = np.random.choice(im1)
        j = np.random.choice(im2)
        
        # swap
        h[i] = 0
        h[j] = 1
        
        iter_count += 1
    
    return h

def local_gradient_refinement(h0, max_iter=1000):
    """Refine with gradient-based optimization on small subset"""
    h = h0.copy()
    
    # Focus on best regions found
    corr_full = correlate(h, 1-h, mode="full")
    max_idx = np.argmax(corr_full)
    best_region_start = max_idx - 30 if max_idx >= 30 else 0
    best_region_end = min(max_idx + 30, n_points)
    
    # Optimize these regions with reference to the full function
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_local):
        return compute_c5(h_local)
    
    def constraint_eq(h_local):
        return np.sum(h_local) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use fewer iterations and local optimizer
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
    except:
        pass
    
    # Project to maintain constraints
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    best_h = heuristic_initialization()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Local improvements with greedy algorithms
    time_allocated = 0.8 * budget_s
    phase1_end = start_time + time_allocated
    
    runs = 0
    while time.time() < phase1_end:
        runs += 1
        # Random restart
        h_rnd = np.random.random(n_points)
        h_rnd = project_to_sum_box(h_rnd, TARGET_SUM)
        h_rnd = binary_from_continuous(h_rnd)
        h_improved = greedy_improvement(h_rnd)
        c5_improved = compute_c5(h_improved)
        
        if c5_improved < best_c5:
            best_h = h_improved.copy()
            best_c5 = c5_improved
            
        # Also try random binary mutation
        h_mut = best_h.copy()
        num_swaps = max(1, n_points // 100)
        for _ in range(num_swaps):
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            h_mut[i], h_mut[j] = h_mut[j], h_mut[i]
        h_mut = project_to_sum_box(h_mut, TARGET_SUM)
        c5_mut = compute_c5(h_mut)
        if c5_mut < best_c5:
            best_h = h_mut.copy()
            best_c5 = c5_mut

    # Phase 2: Final local refinement
    phase2_end = start_time + budget_s
    if time.time() < phase2_end:
        # Convert to continuous (weighted for gradient refined) and apply local refinement
        h_cont = best_h.copy()
        h_cont = local_gradient_refinement(h_cont, max_iter=200)

        final_c5 = compute_c5(h_cont)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_cont.copy()

    return (best_h.tolist(), float(best_c5), n_points)
