from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_with_gradients(h0, max_iter=1000):
    """Optimize using gradient-based method"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def local_improve(h0, max_iter=200):
    """Local search improving solution"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Make perturbations to a few elements
        idxs = np.random.choice(n_points, size=min(5, n_points//6), replace=False)
        improved = False
        
        for idx in idxs:
            old_val = h[idx]
            best_delta = 0
            best_new_c5 = best_c5
            
            # Try several small changes
            for delta in [-0.02, -0.01, 0.01, 0.02]:
                if 0 <= old_val + delta <= 1:
                    h_new = h.copy()
                    h_new[idx] = old_val + delta
                    
                    # Ensure sum constraint is satisfied
                    h_new = project_to_sum_box(h_new, TARGET_SUM)
                    new_c5 = evaluate_c5(h_new)
                    
                    if new_c5 < best_new_c5:
                        best_new_c5 = new_c5
                        best_delta = delta
            
            if best_delta != 0:
                h[idx] += best_delta
                h = project_to_sum_box(h, TARGET_SUM)
                best_c5 = best_new_c5
                improved = True
                
        # If no improvement, break early
        if not improved:
            break
            
    return h

def custom_adjustment(h0, k=2):
    """Apply k-length pattern adjustments to refine solution"""
    h = h0.copy()
    
    # Try to identify and enhance structure
    # Find positions of high and low values
    indices_high = np.where(h > 0.7)[0]
    indices_low = np.where(h < 0.3)[0]
    
    # If there are enough high areas, push some to 1.0

                if 0 <= i + offset < len(h):
                    h[i + offset] = max(0, h[i + offset] - 0.2)

        h_candidate = optimize_with_gradients(best_h, max_iter=200)
        c5_candidate = evaluate_c5(h_candidate)

            best_h = h_candidate
            best_c5 = c5_candidate

            h_optimized = optimize_with_gradients(h_perturbed, max_iter=100)
            c5_optimized = evaluate_c5(h_optimized)
