    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_overlap(h):
        # Use FFT for fast convolution
        h_rev = h[::-1]
        b = 1 - h
        L = 2 * len(h) - 1
        corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
        # Extract valid correlations (full mode) for k=0 to 2*n_points-2
        return dx * np.max(corr)

    def project_to_box_and_mass(x, target_sum):
        """Project x onto [0,1]^n and sum= target_sum"""
        lb = 0.0
        ub = 1.0
        
        def f(lam):
            # Projected x
            x_proj = np.clip(x + lam, lb, ub)
            return np.sum(x_proj) - target_sum
            
        # Binary search for lambda to fulfill sum constraint
        a, b = -1.0, 1.0
        for _ in range(50):
            m = (a + b) / 2
            fm = f(m)
            if abs(fm) < 1e-10:
                break
            if fm * f(a) < 0:
                b = m
            else:
                a = m
        
        x_proj = np.clip(x + (a + b) / 2, lb, ub)
        return x_proj

    def create_mcormick_lp(h_init, T):
        """Create LP for McCormick relaxation with fixed T."""
        # Convex relaxation of the bilinear terms (h[i] * h[j])
        h = cp.Variable(n_points, name='h')
        p = cp.Variable((n_points, n_points), name='p')  # Normalized product matrices 
        # Set initial values to match input h
        h.value = h_init
        # Add McCormick constraints for all pairs (i,j) where i <= j
        constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
        for i in range(n_points):
            for j in range(i, n_points):
                # P[i, j] >= 0
                constraints.append(p[i, j] >= 0)
                # P[i, j] >= h[i] + h[j] - 1
                constraints.append(p[i, j] >= h[i] + h[j] - 1)
                # P[i, j] <= h[i]
                constraints.append(p[i, j] <= h[i])
                # P[i, j] <= h[j]
                constraints.append(p[i, j] <= h[j])
                
        # Define overlap terms
        # For a shift k, overlap is sum_{i} h[i] * h[i+k] (where 0 <= i < n-k)
        # So O_k = sum_i p[i][i+k] for valid indices i
        # R_k = sum h - O_k = target_sum - O_k
        # Add the constraint that R_k <= T for all k -> O_k >= target_sum - T
        
        # Consider only shifts k from -(n-1) to n-1
        for k in range(-(n_points-1), n_points):
            start_idx = max(0, -k)
            end_idx = min(n_points, n_points - k)
            if start_idx == 0 or end_idx > 0:
                overlap_sum = 0
                for i in range(start_idx, end_idx):
                    overlap_sum += p[i, i+k]
                # O_k >= target_sum - T  ->  overlap_sum >= target_sum - T
                constraints.append(overlap_sum >= target_sum - T)
                
        # Minimize T (not actually optimizing but checking feasibility)
        # Return the problem
        return cp.Problem(cp.Minimize(0), constraints)

    def solve_relaxation(h0, T):
        """Solve the convex relaxation to check if T is feasible."""
        prob = create_mcormick_lp(h0, T)
        try:
            prob.solve(solver=cp.ECOS, max_iters=3000, verbose=False, feastol=1e-7, abstol=1e-7)
            return prob.status == 'optimal' or prob.status == 'optimal_inaccurate'
        except:
            return False

    def evaluate_and_verify(h_asolution, T):

    x2 = np.linspace(0, 2, n_points)

    h_init = project_to_box_and_mass(h_init, target_sum)

    # Binary search on C5 bound
    T_low = 0.0
    T_high = target_sum  # Upper bound on max overlap: sum h = n/2, each product is at most 1
    tolerance = 1e-4 * dx
    best_h = h_init
    best_T = T_high

    while T_high - T_low > tolerance and time.time() - start_time < budget_s * 0.3:
        T_mid = (T_low + T_high) / 2.0
        feasible = solve_relaxation(h_init, T_mid)
        if feasible:
            # Now verify with the true overlap
            # Let's compute using a warmed start
            try:
                prob = create_mcormick_lp(h_init, T_mid)
                prob.solve(solver=cp.ECOS, max_iters=5000, verbose=False)
                if prob.status == 'optimal' or prob.status == 'optimal_inaccurate':
                    h_solution = h_init.copy()
                    h_solution = np.clip(prob.variables()[0].value, 0, 1)
                    h_solution = project_to_box_and_mass(h_solution, target_sum)
                    # Evaluate the true C5
                    c5_true = compute_overlap(h_solution)
                    if c5_true <= T_mid * dx:
                        best_h = h_solution.copy()
                        best_T = T_mid
                        T_high = T_mid
                    else:
                        T_low = T_mid
                else:
                    T_low = T_mid
            except:
                T_low = T_mid
        else:

    # Use best_h as starting point for subgradient optimization
    def compute_subgradient(h, k_star):
        """Compute subgradient of C5 at shift k_star."""
        g = np.zeros_like(h)
        # From definition of h[i] * (1 - h[i+k])
        # d/dh_i = (1 - h[i+k]) if k_star >= 0, otherwise -h[i+|k_star|]
        if k_star >= 0:
            # g[i] = (1 - h[i+k_star]) - (1 - h[i-k_star]) if k_star is valid
            for i in range(len(h)):
                if i + k_star < len(h):
                    # Forward term
                    g[i] += (1 - h[i + k_star])
                if i - k_star >= 0:
                    # Backward term
                    g[i] -= h[i - k_star]
        else:
            # Reverse direction
            k_abs = -k_star
            for i in range(len(h)):
                if i + k_abs < len(h):
                    g[i] += (1 - h[i + k_abs])
                if i - k_abs >= 0:
                    g[i] -= h[i - k_abs]
        return g

    def compute_active_shift(h):
        """Find shift k that maximizes overlap sum."""
        corr_list = []
        for k in range(-(n_points-1), n_points):
            start_idx = max(0, -k)
            end_idx = min(n_points, n_points - k)
            if start_idx < end_idx:
                s = sum(h[i] * h[i + k] for i in range(start_idx, end_idx))
                corr_list.append((k, s))
        return max(corr_list, key=lambda x: x[1])[0]

    # Subgradient polishing
    h_polished = best_h.copy()
    best_c5 = compute_overlap(h_polished)
    max_iter_subgrad = 2000
    alpha0 = 0.1
    for t in range(max_iter_subgrad):
        if time.time() - start_time > budget_s * 0.7:
            break
        k_best = compute_active_shift(h_polished)
        grad = compute_subgradient(h_polished, k_best)
        alpha = alpha0 / math.sqrt(t + 1)
        h_new = h_polished - alpha * grad
        h_new = project_to_box_and_mass(h_new, target_sum)
        c5_new = compute_overlap(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            h_polished = h_new.copy()
            
    # Exchange refinement (limited sweep)
    if time.time() - start_time < budget_s * 0.9:
        h_refined = h_polished.copy()
        n_iter_exchange = 5000  # About 400K pair operations at 5% batch
        for i in range(n_iter_exchange):
            if time.time() - start_time > budget_s:
                break
            i1 = np.random.randint(0, n_points)
            i2 = np.random.randint(0, n_points)
            if i1 == i2:
                continue
            # Safe transfer
            h1 = h_refined[i1]
            h2 = h_refined[i2]
            delta_max = min(h2, 1.0 - h1)
            if delta_max <= 0:
                continue
            delta = np.random.uniform(0.1, 0.9) * delta_max
            if delta <= 0:
                continue
            # Try making the change
            h_refined[i1] += delta
            h_refined[i2] -= delta
            # Project
            h_refined = project_to_box_and_mass(h_refined, target_sum)
            new_c5 = compute_overlap(h_refined)
            if new_c5 < best_c5:
                best_c5 = new_c5
            else:
                # Revert
                h_refined[i1] -= delta
                h_refined[i2] += delta
                h_refined = project_to_box_and_mass(h_refined, target_sum)

    c5_final = compute_overlap(h_refined)
    return (h_refined.tolist(), float(c5_final), n_points)
