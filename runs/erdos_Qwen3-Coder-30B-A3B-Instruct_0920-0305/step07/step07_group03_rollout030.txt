from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_of_corr_at_lag(h, lag):
    """Compute gradient of correlation w.r.t h for given lag"""
    n = len(h)
    g = np.ones(n)
    
    if lag >= 0:
        # i from 0 to n-lag-1
        if lag < n:
            g[:n-lag] = 1 - h[lag:]
            g[lag:] -= h[:n-lag]
    else:
        # lag negative, let d = -lag
        d = -lag
        if d < n:
            g[d:] = 1 - h[:n-d]
            g[:n-d] -= h[d:]
            
    return g

def subgradient_descent(h0, max_iter=1000):
    """Refine solution using subgradient descent on maximum correlation"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        corr = compute_corr(h)
        max_val = np.max(corr)
        max_indices = np.where(np.abs(corr - max_val) < 1e-10)[0]
        max_lag = max_indices[0] - (n_points - 1)
        
        # Compute gradient for the max lag
        grad = gradient_of_corr_at_lag(h, max_lag)
        
        # Adaptive step size with diminishing step
        step = 0.1 / np.sqrt(i + 1)
        h_new = h - step * grad
        
        # Project onto feasible set
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            
    return h

def bit_swap(h, i, j):
    """Swap bits at positions i and j"""
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new

    h = np.zeros(n_points)
    ones_indices = np.random.choice(n_points, size=n_points//2, replace=False)
    h[ones_indices] = 1.0
    return h

    # Add small noise and project to satisfy sum constraint
    h_perturbed = h_interp + np.random.normal(0, 0.01, n_points)
    h_perturbed = np.clip(h_perturbed, 0, 1)
    h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()

    TARGET_SUM = n_points / 2
    best_c5 = float('inf')

    # Phase 1: Multiple restarts with simulated annealing on binary configurations

    for restart in range(10):

            break

        if restart < 5:
            # Use random binary

        else:

        local_c5 = evaluate_c5(h)

        refined_c5 = evaluate_c5(refined_h)
        if refined_c5 < best_c5:
            best_h = refined_h.copy()
            best_c5 = refined_c5
