import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    h_perturbed = np.clip(h_perturbed, 0, 1)

def pattern_based_search(h0, max_iter=1000):
    """Improves using targeted pattern-based neighborhood search"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    corr_full = compute_corr_fft(h)
    
    for i in range(max_iter):
        if i % 50 == 0:
            corr_full = compute_corr_fft(h)
        
        # Find the index of maximum correlation
        max_corr_idx = np.argmax(corr_full)
        # Modify interval around the maximum correlation
        start = max(0, max_corr_idx - 20)
        end = min(n_points, max_corr_idx + 20)
        
        # Try several modifications to the interval
        for _ in range(5):
            h_new = h.copy()
            
            # Try increasing/decreasing in small steps
            delta = np.random.normal(0, 0.01)
            if h_new[start:end].sum() + delta > 0 and h_new[start:end].sum() + delta < len(h_new[start:end]):
                # Select a random subset of indices to modify
                indices = np.random.choice(range(start, end), size=max(1, int(0.2 * (end-start))), replace=False)
                for idx in indices:
                    h_new[idx] += delta * np.random.choice([-1, 1])
            
            # Ensure bounds
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5:
                h = h_new.copy()
                best_c5 = new_c5
                corr_full = compute_corr_fft(h)
                    
    return h

def local_gradient_refinement(h0, max_iter=1000):
    """Refine with trust-region optimization on a subset of variables"""
    h = h0.copy()
    
    # Focus on a random subset of variables that seem to have to high impact
    indices_to_modify = np.random.choice(n_points, size=max(20, n_points//6), replace=False)
    
    bounds = [(0, 1) for _ in range(len(indices_to_modify))]
    
    def objective(h_subset):
        h_full = h.copy()
        for i, idx in enumerate(indices_to_modify):
            h_full[idx] = h_subset[i]
        return evaluate_c5(h_full)
    
    def constraint_eq(h_subset):
        h_full = h.copy()
        for i, idx in enumerate(indices_to_modify):
            h_full[idx] = h_subset[i]
        return np.sum(h_full) - TARGET_SUM
    
    initial_guess = h[indices_to_modify]
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            initial_guess, 
            method='trust-constr', 
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'gtol': 1e-10, 'ftol': 1e-10}
        )
        if res.success:
            h_full = h.copy()
            for i, idx in enumerate(indices_to_modify):
                h_full[idx] = res.x[i]
            h = h_full.copy()
    except:
        pass
    
    # Project to maintain constraints
    h = project_to_sum_box(h, TARGET_SUM)
    return h

    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = heuristic_initialization()
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: Pattern-based search
    phase1_end = start_time + 0.3 * budget_s
    for run_id in range(15):

        h_refined = pattern_based_search(best_h.copy(), max_iter=300)
        c5_refined = evaluate_c5(h_refined)
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    # Phase 2: Hybrid refinement with local optimization
    phase2_end = start_time + 0.6 * budget_s
    for run_id in range(15):
        if time.time() > phase2_end:
            break
            
        h_mixed = pattern_based_search(best_h.copy(), max_iter=200)
        h_mixed = local_gradient_refinement(h_mixed.copy(), max_iter=200)
        c5_mixed = evaluate_c5(h_mixed)
        
        if c5_mixed < best_c5:
            best_h = h_mixed.copy()
            best_c5 = c5_mixed
    
    # Phase 3: Final refinement with systematic exploration
    phase3_end = start_time + budget_s
    while time.time() < phase3_end:
        # Systematically try small adjustments to high-contribution regions
        h_noise = best_h + np.random.normal(0, 0.01, n_points)
        h_noise = project_to_sum_box(h_noise, TARGET_SUM)
        
        h_final = local_gradient_refinement(h_noise, max_iter=200)
        c5_final = evaluate_c5(h_final)
        
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final
    
    return (best_h.tolist(), float(best_c5), n_points)
