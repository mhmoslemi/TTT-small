import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

# Global variables
TARGET_SUM = None
n_points = 150
dx = 2.0 / n_points
MAX_ITER_SUBGRAD = 2000
MAX_ITER_CMAES = 100

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_lp_relaxation(cutoff_time_s=150):
    """Solve the LP relaxation with McCormick envelopes"""
    start_t = time.time()
    # Variables: h[0], ..., h[n-1] in [0,1]
    h = cp.Variable(n_points, name="h")
    # Variables: p_ij for i <= j representing h_i * h_j
    p = {}
    print(f"Setting up LP for n={n_points}, up to {n_points * (n_points + 1) // 2} variables")
    
    # Constraints for McCormick
    constraints = [
        cp.sum(h) == TARGET_SUM,
        h >= 0,
        h <= 1
    ]
    
    # Add McCormick constraints
    for i in range(n_points):
        for j in range(i, n_points):
            p_ij = cp.Variable(name=f"p_{i}_{j}")
            p[i, j] = p_ij
            constraints += [
                p_ij >= 0,
                p_ij >= h[i] + h[j] - 1,
                p_ij <= h[i],
                p_ij <= h[j]
            ]
    
    # Objective: minimize t such that for all k, sum over valid products <= n/2 - t
    # Let's define t as a scalar variable representing the upper bound
    t = cp.Variable(name="t", nonneg=True)
    
    # Shift k from 0 to n-1 (symmetric cases can be derived from these)
    # For shift k: valid indices are i = 0..(n-1-k)
    # O_k = sum_{i=0}^{n-1-k} h[i] * h[i+k]
    # We want: int h(x) * (1-h(x+k)) dx = sum h_i * (1 - h_{i+k}) = (n/2) - O_k
    # So: O_k = n/2 - t => O_k + t = n/2
    # Or: t = (n/2) - O_k
    # Minimise t => Maximize O_k
    # But since O_k = sum_{i} p_{i, i+k}
    for k in range(0, n_points):
        if k > 0:
            # Sum over i = 0..n-1-k
            valid_indices = [(i, i + k) for i in range(n_points - k)]
            if len(valid_indices) > 0:
                shift_sum = cp.sum([p[i, j] for (i, j) in valid_indices])
                constraints.append(shift_sum >= TARGET_SUM / 2 - t)

    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)

    print(f"Starting solve of LP with {len(constraints)} constraints and {len(h) + len(p)} variables")
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000, verbose=False, time_limit=cutoff_time_s)
        print("LP solution status:", prob.status)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h.value
            return h_sol
        else:
            print("Failed to solve LP to optimality")
            return None
    except Exception as e:
        print(f"Error solving LP: {e}")
        return None

def subgradient_optimize(h0, max_iter=MAX_ITER_SUBGRAD):
    """Project subgradient descent for minimizing C5"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    lr = 0.5 * dx
    print(f"Starting subgradient optimization with {max_iter} iterations")

    for it in range(max_iter):
        # Compute correlation
        corr = compute_corr_fft(h)
        k_star = np.argmax(corr)
        # For simplification, thrust k_star in range 0 to n-1 even though includes -n+1 to n-1
        # correct way: compute backwards for negative lags and forward for positive
        # But since |k| is mostly small in general, correct booltract info with both directions
        # Simplified
        shift_ind = k_star if k_star < n_points else k_star - 2*n_points 
        
        # Subgradient for the active k_star term
        # For shift k_star, we compute:
        # g_i = (1-h_{i+k_star}) - h_{i-k_star} , 
        # where out-of-range indices = 0
        # This derivative of integral h_i * (1 - h_{i+k_star}) w.r.t. h_i
        # So how do we ensure valid indices?
        # If shift is k*, then f(h) = sum h_i * (1 - h_{i+k*})
        # gradient w.r.t h_j is (1 - h_{j+k*}) if j+k* is valid,
        # and we also get -h_{j-k*} if j-k* >= 0
        # So g_j = (1 - h_{j+k*}) - h_{j-k*} (where out-of-range are 0)

        g = np.zeros_like(h)
        if k_star < n_points:
            k_star = k_star
            # Deprecate lines below which actually does not work for nonlinear optimization...
            # Instead we directly compute gradient
            for i in range(n_points):
                hi = h[i]
                h_ipk = h[(i + k_star) % n_points] if (i + k_star) < n_points else 0
                h_imk = h[i - k_star] if i >= k_star else 0
                g[i] = (1 - h_ipk) - h_imk 

        # Apply subgradient update
        h_new = h - lr * g
        # Project onto [0,1] and preserve sum constraint
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        # Evaluate new C5
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

        h = h_new
        if it % 500 == 0:
            print(f"Subgrad iteration {it}: C5 = {c5_new:.6f}, Best = {best_c5:.6f}")
        if time.time() > start_t + 0.8 * budget_s:
            break

    return best_h, best_c5

def cma_es_optimize(h0, budget_s=100):
    """Basic CMA-ES local search"""
    print("Starting CMA-ES; not implemented in this version")
    # No-op for now, but could implement later if needed
    return h0, evaluate_c5(h0)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx, start_t
    np.random.seed(seed)
    start_t = time.time()

    TARGET_SUM = n_points / 2
    print(f"Starting optimization with n={n_points}")

    # Phase 1: Solve the LP relaxation
    h_lp = solve_lp_relaxation(cutoff_time_s=150)
    if h_lp is None:
        print("LP relaxation failed; using a robust initial solution")
        h_lp = np.random.uniform(0, 1, n_points)
        h_lp = project_to_sum_box(h_lp, TARGET_SUM)
    else:
        print("LP relaxation successful")

    # Evaluate the current LP-bound h and refine via deterministic subgradient
    c5_bound = evaluate_c5(h_lp)
    print(f"Initial C5 from LP relaxation: {c5_bound}")

    # Refine with project subgradient descent (local), but only if not good enough yet
    refined_h, refined_c5 = h_lp, c5_bound
    if c5_bound > 0.379:
        print("LP result not good enough, refining with projected subgradient iteration...")
        refined_h, refined_c5 = subgradient_optimize(h_lp, MAX_ITER_SUBGRAD)

    # Try optional CMA-ES if still below threshold
    if refined_c5 > 0.379:
        try:
            refined_h, refined_c5 = cma_es_optimize(refined_h, budget_s=50)
        except Exception:
            print("CMA-ES failed, skipping")

    return (refined_h.tolist(), float(refined_c5), n_points)
