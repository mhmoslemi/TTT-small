import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def create_thue_morse(n):
    """Generate Thue-Morse sequence up to n bits"""
    if n <= 0:
        return []
    seq = [0]
    for i in range(n):
        seq.extend([1-x for x in seq])
    return seq[:n]

def create_rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence up to n terms"""
    if n <= 0:
        return []
    # Use the recursive definition
    seq = [0]
    for i in range(1, n):
        # Rudin-Shapiro is defined recursively
        # For our purposes, we use the first 100 values of known sequence
        # For n=256, just some samples from calculated values
        # Placeholder implementation; we'll take a simple version
        seq.append(seq[i-1] ^ 1) if i % 2 == 0 else seq.append(seq[i-1])
    return seq[:n]

def greedy_worst_shift_improvement(h):
    """Improve h using greedy swaps for worst shift"""
    h = h.copy()
    corr_full = compute_corr_fft(h)
    max_shift_idx = np.argmax(corr_full)
    
    # Try swapping elements to reduce overlap for the identified max shift
    max_shift = max_shift_idx - (n_points - 1)
    
    # Identify all shifts where h[i] = 1, h[i+k] = 0; and h[j] = 0, h[j+k] = 1
    # These are the pairs that can be swapped to improve overlap
    mask1 = h[:-abs(max_shift)] if max_shift < 0 else h[abs(max_shift):]
    mask2 = h[abs(max_shift):] if max_shift < 0 else h[:-abs(max_shift)]
    
    want1 = np.where(mask1 == 1)[0]
    want2 = np.where(mask2 == 0)[0]
    
    # Get matching indices in original array
    if max_shift < 0:
        i_indices = want1
        j_indices = want2 + abs(max_shift)
    else:
        i_indices = want1 + abs(max_shift)
        j_indices = want2
    
    # Actually we just check whether there are such pairs that can be swapped
    # to decrease the overlap for this shift
    try:
        # Choose random such swap if exists
        if len(i_indices) > 0 and len(j_indices) > 0:
            i = i_indices[random.randint(0, len(i_indices)-1)]
            j = j_indices[random.randint(0, len(j_indices)-1)]
            
            if max_shift < 0:
                # Shift is negative, rearrange indices
                i, j = i, j - abs(max_shift)
            else:
                # Shift is positive, rearrange indices
                i, j = i - abs(max_shift), j
            
            # Make the swap if it makes sense, update
            if h[i] == 1 and h[j] == 0:
                h[i], h[j] = 0, 1
                
    except IndexError:
        pass
        
    return h

def local_search_with_greedy_restarts(budget_s):
    """Perform random restarts with greedy improvement"""
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    n_restarts = 300  # Will likely be fewer
    for i in range(n_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Generate binary initial solution
        h_candidate = np.random.randint(0, 2, n_points).astype(float)
        h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
        
        for _ in range(100): # Max iterations per restart
            if time.time() - start_time > budget_s * 0.95:
                break
            h_candidate = greedy_worst_shift_improvement(h_candidate)
        
        c5 = evaluate_c5(h_candidate)
        if c5 < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5
    
    return best_h, best_c5

def simulate_annealing(h0, T_init, T_final, alpha, budget_s):
    """Simulated Annealing with geometric decay"""
    h = h0.copy()
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    start_time = time.time()
    
    T = T_init
    
    while T > T_final and time.time() - start_time < budget_s * 0.2:
        # Try one move
        # We make a local modification: swap two random positions
        i, j = random.randint(0, n_points-1), random.randint(0, n_points-1)
        
        if i != j:
            h[i], h[j] = h[j], h[i]  # Swap
            # Maybe do boundary check: project h into [0,1]?
            h = project_to_sum_box(h, TARGET_SUM)
            new_c5 = evaluate_c5(h)
            
            delta = new_c5 - current_c5
            
            if delta < 0 or np.exp(-delta / T) > random.random():
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = new_c5
            else:
                h[i], h[j] = h[j], h[i]  # Revert swap
        else:
            pass  # No swap, do nothing, just continue
        
        T *= alpha
    
    return best_h, best_c5

def refine_solution_continuous(h0, max_iter=1000):
    """Refine using gradient descent"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_local):
        return evaluate_c5(h_local)
    
    def constraint_eq(h_local):
        return np.sum(h_local) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
    except:
        pass
    
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = None
    best_c5 = float('inf')
    
    # Phase 1: Local search with greedy restarts
    h_candidate, c5_candidate = local_search_with_greedy_restarts(budget_s)
    if c5_candidate < best_c5:
        best_h = h_candidate.copy()
        best_c5 = c5_candidate
        
    # Phase 2: Simulated Annealing refinement
    if best_h is not None:
        annealed_h, annealed_c5 = simulate_annealing(
            best_h.copy(), T_init=1.0, T_final=1e-5, alpha=0.95, budget_s=budget_s
        )
        if annealed_c5 < best_c5:
            best_h = annealed_h.copy()
            best_c5 = annealed_c5
    
    # Phase 3: Continuous refinement
    if best_h is not None:
        final_h = refine_solution_continuous(best_h.copy())
        final_c5 = evaluate_c5(final_h)
        if final_c5 < best_c5:
            best_h = final_h.copy()
            best_c5 = final_c5
    
    return (best_h.tolist(), float(best_c5), n_points)
