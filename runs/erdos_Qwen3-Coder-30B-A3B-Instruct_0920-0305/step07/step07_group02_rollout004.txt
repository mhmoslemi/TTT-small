# Global variables
TARGET_SUM = None

def c5_fft(h):
    """Compute C5 using FFT-based linear correlation"""
    n = len(h)
    nfft = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, nfft)
    B = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * B, nfft)[:2*n-1]
    return np.max(conv) * dx

def project_to_sum(h, target_sum):
    """Project vector onto sum constraint using binary search on Lagrange multiplier"""
    h = np.array(h)
    def f(lam):
        proj = np.clip(h + lam, 0, 1)
        return np.sum(proj) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def thue_morse(n):
    """Generate Thue-Morse sequence of length n"""
    tm = [0]
    for i in range(n):
        tm.extend([1-x for x in tm])
    return np.array(tm[:n])

def rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence of length n"""
    rs = np.zeros(n, dtype=int)
    for i in range(1, n):
        rs[i] = (rs[i-1] + (1 if (bin(i-1).count('1') & 1) == 0 else 0)) % 2
    return rs

def greedy_descent(h_start):

    h = project_to_sum(h, target_sum)
    
    max_iter = n
    for _ in range(max_iter):
        corr = correlate(h, 1-h, mode="full")
        # Find the index of maximum (the worst shift)
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)
        
        # Determine mismatched pairs
        if shift >= 0:
            idx1 = np.where((h[:-shift] == 1) & (h[shift:] == 0))[0]
            idx2 = np.where((h[:-shift] == 0) & (h[shift:] == 1))[0]
        else:
            idx1 = np.where((h[-shift:] == 1) & (h[:-shift] == 0))[0]
            idx2 = np.where((h[-shift:] == 0) & (h[:-shift] == 1))[0]
        
        if len(idx1) == 0 or len(idx2) == 0:
            break
            
        # Select random pair
        i = random.choice(idx1)
        j = random.choice(idx2)
        
        # Swap the bits
        if shift >= 0:
            h[i], h[j + shift] = h[j + shift], h[i]
        else:
            h[i - shift], h[j] = h[j], h[i - shift]
        
        # Project back to sum constraint
        h = project_to_sum(h, target_sum)
    
    return h, c5_fft(h)
