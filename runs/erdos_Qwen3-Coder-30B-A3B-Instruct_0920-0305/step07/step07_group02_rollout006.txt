import time

def compute_c5_fft(h):
    """Compute C5 using FFT-based correlation"""
    dx = 2.0 / len(h)
    corr_full = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr_full)

def greedy_descent(h_init):
    """Improve solution by greedy swaps on worst shift"""
    h = h_init.copy()
    dx = 2.0 / len(h)
    
    while True:
        # Compute all correlations
        corr_full = correlate(h, h, mode='full')
        # Worst shift corresponds to minimum correlation
        worst_shift = np.argmin(corr_full)
        corr_worst = corr_full[worst_shift]
        
        # Convert to overlap value
        overlap_worst = 1 - corr_worst * dx
        
        # Find mismatched pairs
        # This is the set where h[i]=1, h[i+shift]=0 (contributes to C5)
        # And where h[i]=0, h[i+shift]=1
        shift_idx = worst_shift - (len(h) - 1)
        A = []
        B = []
        
        for i in range(len(h)):
            j = i + shift_idx
            if 0 <= j < len(h):
                if h[i] == 1 and h[j] == 0:
                    A.append(i)
                elif h[i] == 0 and h[j] == 1:
                    B.append(i)
        
        if not A or not B:
            break
            
        # Pick random elements and swap
        i = np.random.choice(A)
        j = np.random.choice(B)
        
        # Perform swap and recompute
        h[i], h[j] = h[j], h[i]
    
    return h, compute_c5_fft(h)

def anneal_binary(h_init, remaining_time):

    Tmax = int(10000)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = current_c5
    
    start_time = time.time()
    
    for t in range(Tmax):
        if time.time() - start_time > remaining_time - 1:
            break
            
        # Temperature
        T = T0 * (1 - t / Tmax)
        if T <= 0:
            break
            
        # Pick two random indices
        i = np.random.randint(len(h))
        j = np.random.randint(len(h))
        if i == j:
            continue
            
        # Swap and compute new overlap
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5_fft(h)
        
        # Accept or reject
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # Revert swap
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_c5
