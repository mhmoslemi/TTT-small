import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def create_balanced_initial():
    """Create a more balanced initial solution"""
    h = np.zeros(n_points)
    
    # Place two symmetric peaks
    peak1_x = n_points // 4
    peak2_x = 3 * n_points // 4
    
    # Create Gaussian peaks
    sigma = n_points // 12
    for i in range(n_points):
        h[i] += 0.8 * np.exp(-0.5 * ((i - peak1_x) / sigma)**2)
        h[i] += 0.8 * np.exp(-0.5 * ((i - peak2_x) / sigma)**2)
    
    # Normalize to sum = TARGET_SUM
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def create_sparse_initial():
    """Create a sparse initial solution with concentrated mass"""
    h = np.zeros(n_points)
    
    # Distribute mass in two locations
    left_pos = n_points // 3
    right_pos = 2 * n_points // 3
    
    # Place points with high values
    h[left_pos] = 1.0
    h[right_pos] = 1.0
    
    # Adjust to get correct sum
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def local_search(h0, max_iter=500):

    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Try to improve by small changes
        idx = np.random.randint(0, n_points)
        old_val = h[idx]
        
        # Try different step sizes
        step_sizes = [0.01, 0.02, 0.05]
        np.random.shuffle(step_sizes)
        
        for step in step_sizes:
            # Try both directions
            for delta in [-step, step]:
                h_new = h.copy()
                h_new[idx] += delta
                
                if h_new[idx] < 0 or h_new[idx] > 1:
                    continue
                    
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                new_c5 = evaluate_c5(h_new)
                
                if new_c5 < best_c5:
                    h = h_new.copy()
                    best_c5 = new_c5
                    break
            else:
                continue
            break
        else:
            # Perturb randomly if no improvement found
            if np.random.random() < 0.3:
                idx = np.random.randint(0, n_points)
                delta = np.random.uniform(-0.02, 0.02)
                h[idx] += delta
                h = project_to_sum_box(h, TARGET_SUM)
                best_c5 = evaluate_c5(h)
    
    return h

def optimize_with_constraints(h0):
    """Optimize with scipy SLSQP with constraints"""
    h = h0.copy()
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
    except:
        pass
    
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def adaptive_resize(h0):
    """Adaptively resize solution based on current behavior"""
    # Try to place mass more efficiently by_iteratively adjusting 
    h = h0.copy()
    
    # Find areas with high and low values to rebalance
    low_indices = np.where(h < 0.3)[0]
    high_indices = np.where(h > 0.7)[0]
    
    # Move mass from high to low areas
    for _ in range(10):
        if len(low_indices) > 0 and len(high_indices) > 0:
            from_idx = np.random.choice(high_indices)
            to_idx = np.random.choice(low_indices)
            
            # Move a small amount of mass
            move_amount = min(0.1, h[from_idx])
            if move_amount > 0:
                h[from_idx] -= move_amount
                h[to_idx] += move_amount
    
    h = project_to_sum_box(h, TARGET_SUM)
    return h
