import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Use Newton-Raphson method for better convergence
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Find initial bracket
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    # Newton-Raphson iteration
    for _ in range(50):
        fa = f(a)
        if abs(fa) < 1e-12:
            break
        # Numerical derivative
        eps = 1e-8
        fp = (f(a + eps) - fa) / eps
        if abs(fp) < 1e-12:
            break
        a_new = a - fa / fp
        if abs(a_new - a) < 1e-12:
            break
        a = a_new
        
    result = np.clip(x + a, 0, 1)
    return result

def optimize_with_gradients(h0, max_iter=1000):
    """Optimize using gradient-based method with projection"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def genetic_algorithm_search():
    """Use a simple genetic algorithm for global search"""
    pop_size = 30
    num_generations = 50
    mutation_rate = 0.05
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = np.random.uniform(0, 1, n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    
    for generation in range(num_generations):
        # Evaluate fitness
        fitness = [evaluate_c5(h) for h in population]
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        
        # Keep top 50%
        survivors = population[:pop_size//2]
        
        # Create new population through crossover and mutation
        new_population = survivors.copy()
        while len(new_population) < pop_size:
            # Tournament selection
            idx1, idx2 = np.random.choice(len(survivors), 2, replace=False)
            parent1, parent2 = survivors[idx1], survivors[idx2]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points-1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] = np.clip(child1[i] + np.random.normal(0, 0.02), 0, 1)
                if np.random.rand() < mutation_rate:
                    child2[i] = np.clip(child2[i] + np.random.normal(0, 0.02), 0, 1)
            
            child1 = project_to_sum_box(child1, TARGET_SUM)
            child2 = project_to_sum_box(child2, TARGET_SUM)
            
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
    
    # Return best individual
    fitness = [evaluate_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    return population[best_idx]
