import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
from cvxpy import Minimize, Problem
from scipy.optimize import minimize

# Global constants
n_points = 150
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_relaxation_lp():
    """Solve the linear program relaxation with McCormick envelopes"""
    # Variables
    h = cp.Variable(n_points)
    p = cp.Variable((n_points, n_points), symmetric=True)  # product variables
    
    # Constraints
    constraints = []
    
    # Box constraints
    constraints += [h[i] >= 0 for i in range(n_points)]
    constraints += [h[i] <= 1 for i in range(n_points)]
    
    # McCormick constraints for all pairs i,j
    for i in range(n_points):
        for j in range(i, n_points):
            constraints += [p[i, j] >= 0]
            constraints += [p[i, j] >= h[i] + h[j] - 1]
            constraints += [p[i, j] <= h[i]]
            constraints += [p[i, j] <= h[j]]
            
    # Mass constraint
    constraints += [cp.sum(h) == TARGET_SUM]
    
    # Shift variables - add dummy variable to bound max
    t = cp.Variable()
    constraints += [t >= 0]
    
    # Optimization: minimize max correlation through t
    # For each shift k compute overlap
    for k in range(-n_points+1, n_points):
        if k >= 0:
            start_idx = 0
            end_idx = n_points - k
        else:
            start_idx = -k
            end_idx = n_points
        
        if end_idx > start_idx:
            O_k = cp.sum([p[i, i+k] for i in range(start_idx, end_idx)])
            constraints += [O_k >= TARGET_SUM - t]
    
    # Objective
    objective = Minimize(t)
    prob = Problem(objective, constraints)
    
    # Solve with timeout
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000, abstol=1e-6, reltol=1e-6)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return h.value
    except:
        pass
    return None


def compute_subgradient(h, k_star):
    """Compute subgradient for the shift k_star"""
    g = np.zeros_like(h)
    for i in range(len(h)):
        idx1 = i + k_star
        idx2 = i - k_star
        if 0 <= idx1 < len(h):
            g[i] -= h[idx1]
        if 0 <= idx2 < len(h):
            g[i] -= h[idx2]
    return g

def projected_gradient_descent(h_start, max_iter=2000):
    """Perform projected gradient descent for local refinement"""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    step_size = 0.5 * dx
    decay_rate = 0.995
    
    for i in range(max_iter):
        # Find current best shift
        corr = np.correlate(h, 1-h, mode="full")
        k_star = np.argmax(corr) - (len(h) - 1)
        
        # Compute subgradient
        g = compute_subgradient(h, k_star)
        
        # Update
        h_new = h - step_size * g
        
        # Project to box constraints [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project to sum constraint
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate objective
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new.copy()
        step_size *= decay_rate
        
        # Early stopping if improvement is small
        if i > 100 and abs(c5_new - best_c5) < 1e-8:
            break
            
    return best_h, best_c5
