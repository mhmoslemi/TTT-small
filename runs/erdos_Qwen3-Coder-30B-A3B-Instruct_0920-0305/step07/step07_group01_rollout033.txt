import time

# Global variables
TARGET_SUM = None
dx = None

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_gradient(h, corr, beta):
    """Compute smooth max gradient"""
    # Subtract max for numerical stability
    corr_centered = corr - np.max(corr)
    exp_corr = np.exp(beta * corr_centered)
    weights = exp_corr / np.sum(exp_corr)
    
    # Create gradient from weighted sum of all lags
    grad = np.zeros_like(h)
    n = len(h)
    
    # For each lag, compute contribution to gradient
    for lag in range(-n+1, n):
        if lag < 0:
            # lag is negative, so h[i] contributes to h[i+lag] term
            lag_term = np.zeros_like(h)
            lag_term[-lag:] = h[:-lag]
            grad += weights[lag + n - 1] * (1 - lag_term)
        elif lag == 0:
            # lag is zero, so h[i] contributes to itself term
            grad += weights[n-1] * (1 - h)
        else:
            # lag is positive, so h[i] contributes to h[i-lag] term
            lag_term = np.zeros_like(h)
            lag_term[:-lag] = h[lag:]
            grad += weights[lag + n - 1] * (1 - lag_term)
    
    return grad * dx

def max_lag_gradient(h, corr):
    """Compute exact sub-gradient of max-over-lags"""
    max_idx = np.argmax(corr)
    n = len(h)
    
    # Convert index from full correlation to actual lag
    lag = max_idx - (n - 1)
    
    grad = np.zeros_like(h)
    if lag < 0:
        # h[i] contributes to h[i+lag] probs, i.e., i+lag must be valid
        for i in range(n):
            if i + lag >= 0:
                grad[i] += (1 - h[i+lag]) * dx
    elif lag > 0:
        # h[i] contributes to h[i-lag] probs, i.e., i-lag must be valid
        for i in range(n):
            if i - lag < n:
                grad[i] -= h[i-lag] * dx
    else:
        # lag = 0
        for i in range(n):
            grad[i] = (1 - h[i]) * dx
    
    return grad

def mass_transfer(h, grad, epsilon):
    """Perform greedy mass transfer move"""
    d = -grad  # Point in direction of steepest ascent
    
    sources = np.where(d < 0)[0]
    sinks = np.where(d > 0)[0]
    
    # Sort by decreasing effect (absolute values for sources, increasing for sinks)
    sources = sources[np.argsort(-d[sources])]
    sinks = sinks[np.argsort(d[sinks])]
    
    h_new = h.copy()
    remaining_mass = epsilon * TARGET_SUM
    
    i, j = 0, 0
    while i < len(sources) and j < len(sinks) and remaining_mass > 0:
        s = sources[i]
        t = sinks[j]
        
        # Transfer amount
        transfer = min(h_new[s], 1 - h_new[t], remaining_mass)
        if transfer <= 0:
            if h_new[s] <= 0:
                i += 1
            if 1 - h_new[t] <= 0:
                j += 1
            continue
            
        h_new[s] -= transfer
        h_new[t] += transfer
        remaining_mass -= transfer

    # Re-project onto feasible set
    h_new = project_sum_box(h_new, TARGET_SUM)
    
    return h_new

def line_search_step(h, grad, corr, beta, α0=0.2, α_min=5e-4):
    """Back-tracking line search"""
    α = α0
    cmax = evaluate_c5(h)
    
    while α >= α_min:
        h_new = h - α * grad
        h_new = project_sum_box(h_new, TARGET_SUM)
        cmax_new = evaluate_c5(h_new)
        
        # Accept if improvement
        if cmax_new < cmax:
            return h_new, cmax_new, α
        
        α /= 2
    
    # If we didn't improve, return original h
    return h, cmax, α
