import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def stochastic_local_search(h0, max_iter=2000):
    """Stochastic local search with adaptive step size"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    step_size = 0.02
    
    for i in range(max_iter):
        # Randomly select indices to modify
        num_changes = max(1, min(5, n_points // 20))
        idxs = np.random.choice(n_points, size=num_changes, replace=False)
        
        # Generate trial changes
        h_trial = h.copy()
        for idx in idxs:
            # Adaptive step size based on current value
            adapt_step = step_size * (1.0 - h[idx]) if h[idx] < 0.5 else step_size * h[idx]
            delta = np.random.uniform(-adapt_step, adapt_step)
            h_trial[idx] += delta
            
        # Project to feasible region
        h_trial = project_to_sum_box(h_trial, TARGET_SUM)
        new_c5 = evaluate_c5(h_trial)
        
        # Accept with probability based on improvement
        if new_c5 < best_c5:
            h = h_trial.copy()
            best_c5 = new_c5
            # Increase step size if we've made progress
            step_size = min(0.05, step_size * 1.05)
        else:
            # Occasionally accept worse solutions to escape local optima
            if np.random.random() < 0.01:
                h = h_trial.copy()
                best_c5 = new_c5
                step_size = max(0.005, step_size * 0.95)
    
    return h

    peak_width = n_points // 10
    
    # Create first peak
    for i in range(max(0, peak1_pos - peak_width), min(n_points, peak1_pos + peak_width)):

    h += np.random.normal(0, 0.02, n_points)
    
    # Ensure feasibility and sum constraint
    h = np.clip(h, 0, 1)
    h = project_to_sum_box(h, TARGET_SUM)
