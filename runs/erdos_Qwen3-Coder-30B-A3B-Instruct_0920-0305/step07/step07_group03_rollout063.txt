import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_search(h0, iterations=500):
    """Enhanced local search with adaptive step sizes"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(iterations):
        # Adaptive perturbation based on current state
        if i < iterations // 3:
            step_size = 0.02
        elif i < 2 * iterations // 3:
            step_size = 0.01
        else:
            step_size = 0.005
        
        # Try several random perturbations
        improved = False
        for _ in range(10):
            h_new = h.copy()
            
            # Select a few points to modify
            indices = np.random.choice(n_points, size=min(5, n_points//10), replace=False)
            
            # Apply small changes
            for idx in indices:
                delta = np.random.uniform(-step_size, step_size)
                h_new[idx] += delta
                
                # Ensure bounds
                h_new[idx] = np.clip(h_new[idx], 0, 1)
            
            # Project to correct sum
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            # Evaluate
            c5_new = evaluate_c5(h_new)
            
            if c5_new < best_c5:
                h = h_new.copy()
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                break
        
        # If no improvement, reduce step size and try again
        if not improved:
            step_size *= 0.8
            
    return best_h, best_c5

def optimize_with_cobyla(h0, max_iter=100):
    """Use COBYLA optimization with multiple restarts"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    def constraint_bound(h, idx):
        return h[idx] - 1  # h[idx] <= 1
    def constraint_bound_low(h, idx):
        return -h[idx]  # h[idx] >= 0
    
    # Define constraints
    cons = [{'type': 'eq', 'fun': constraint_eq}]
    for i in range(n_points):
        cons.append({'type': 'ineq', 'fun': lambda h, i=i: 1 - h[i]})
        cons.append({'type': 'ineq', 'fun': lambda h, i=i: h[i]})
    
    for _ in range(5):
        try:
            res = minimize(
                objective, 
                h, 
                method='COBYLA', 
                constraints=cons,
                options={'maxiter': max_iter, 'rhobeg': 0.05, 'tol': 1e-6}
            )
            if res.success:
                h_candidate = res.x
                h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
            
        # Perturb initial point for next restart
        h = h + np.random.normal(0, 0.01, n_points)
        h = project_to_sum_box(h, TARGET_SUM)
    
    return best_h, best_c5
