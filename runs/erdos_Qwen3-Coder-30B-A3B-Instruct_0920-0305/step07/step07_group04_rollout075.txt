def project_to_sum_box(x, target_sum, dx):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return 2.0 / n * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def subgradient_refinement(h, max_iter=2000):
    """Improves upon h with projected subgradient descent"""
    h = np.array(h)
    t0 = time.time()
    for i in range(max_iter):
        if time.time() - t0 > 200:  # 200s remaining for this step
            break
            
        # Find active shift k*
        corr = compute_corr_fft(h)
        max_k = np.argmax(corr)
        
        # Compute gradient df/dh_i for the active shift
        k_star = max_k - len(h) + 1  # Convert to actual shift
        g = np.zeros_like(h)
        for i in range(len(h)):
            left_idx = i - k_star
            right_idx = i + k_star
            
            if 0 <= left_idx < len(h):
                g[i] -= h[left_idx]
            if 0 <= right_idx < len(h):
                g[i] -= h[right_idx]
            g[i] += 1
            
        # Gradient descent step
        h_new = h - 0.02 * g  # Step size
        
        # Project and enforce sum constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, len(h) / 2, 2.0 / len(h))
        
        # Update h with accepted move at least 
        h = h_new
    
    return h
