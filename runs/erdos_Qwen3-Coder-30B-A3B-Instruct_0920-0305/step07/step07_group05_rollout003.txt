import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global variables
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmin_weights(s, alpha):
    """Compute soft-min weights for the given s and alpha"""
    s_max = np.max(s)
    exp_s = np.exp(-alpha * (s - s_max))
    return exp_s / np.sum(exp_s)

def weighted_gradient(h, alpha):
    """Compute gradient of softmin over correlations"""
    s = correlate(h, h, mode='full')
    w = softmin_weights(s, alpha)
    grad = np.zeros_like(h)
    n = len(h)
    
    # For each lag, add contribution from shifted h
    for i in range(len(w)):
        lag = i - (n - 1)
        if lag >= 0:
            # h shifted by lag
            grad[lag:] += w[i] * h[:n-lag]
        else:
            # h shifted by -lag
            grad[:n+lag] += w[i] * h[-lag:]
    
    return grad

def pairwise_move(h, grad, delta, rng):
    """Move mass from smallest gradient to largest gradient"""
    i_src = np.argmin(grad)
    i_tgt = np.argmax(grad)
    
    # Ensure source is positive and target is less than 1
    if h[i_src] <= 1e-10 or h[i_tgt] >= 1 - 1e-10:
        return False, h
    
    # Determine transfer amount
    transfer = min(delta, h[i_src], 1 - h[i_tgt])
    if transfer <= 1e-10:
        return False, h
    
    # Apply move
    h[i_src] -= transfer
    h[i_tgt] += transfer
    
    # Maintain constraints
    h[i_src] = max(0, h[i_src])
    h[i_tgt] = min(1, h[i_tgt])
    
    return True, h

def annealed_gradient_ascent(h0, time_budget, alpha0, delta0, temp0, rng):
    """Annealed gradient ascent with pairwise moves"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    alpha = alpha0
    delta = delta0
    temperature = temp0
    
    start_time = time.time()
    iter_count = 0
    
    while time.time() - start_time < time_budget:
        iter_count += 1
        if iter_count % 200 == 0:
            alpha = min(alpha * 1.2, 200)
            delta *= 0.9995
            temperature *= 0.9995
            
        grad = weighted_gradient(h, alpha)
        old_c5 = evaluate_c5(h)
        
        # Perform move
        success, h_new = pairwise_move(h, grad, delta, rng)
        if not success:
            continue
            
        new_c5 = evaluate_c5(h_new)
        
        # Accept with Metropolis rule if worse
        if new_c5 > old_c5:
            if np.random.random() < np.exp(-(new_c5 - old_c5) / temperature):
                h = h_new
            else:
                # Revert the move
                h = h0
        else:
            h = h_new
            
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
        # Periodically reproject to maintain feasibility
        if iter_count % 1000 == 0:
            h = project_to_sum_box(h, TARGET_SUM)
    
    return best_h, best_c5

def random_binary_start(n, rng):
    """Create a binary start vector"""
    h = np.zeros(n)
    ones_indices = rng.sample(range(n), k=n//2)
    h[ones_indices] = 1
    return h

def local_optimize_restart(start_h, remaining_time, rng):
    """Run a short gradient ascent from a start vector"""
    h = start_h.copy()
    h = project_to_sum_box(h, TARGET_SUM)
    
    alpha0 = 10.0
    delta0 = 0.5
    temp0 = 5e-3
    
    best_h, best_c5 = annealed_gradient_ascent(
        h, remaining_time * 0.8, alpha0, delta0, temp0, rng
    )
    
    return best_h, best_c5
