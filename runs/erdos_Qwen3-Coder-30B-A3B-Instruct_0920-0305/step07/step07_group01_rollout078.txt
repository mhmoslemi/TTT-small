TARGET_SUM = 1.0

    """Fast evaluation of C5 using correlation"""

def fourier_basis(x, m):
    """Return m-th cosine basis function evaluated at points x"""
    return np.cos(m * np.pi * x)

def construct_h_from_coeffs(coeffs, n_points=N_POINTS):
    """Construct h(x) from Fourier coefficients"""
    x = np.linspace(0, 2, n_points)
    # We'll use a simple cosine basis with constant term
    h_vals = 0.5 * np.ones_like(x)  # baseline
    for i, coeff in enumerate(coeffs):
        h_vals += coeff * fourier_basis(x, i + 1)
    # Clip to [0,1] to respect constraints
    h_vals = np.clip(h_vals, 0, 1)
    return h_vals

def objective_with_penalty(coeffs, n_points=N_POINTS, penalty_weight=1e6):
    """Objective function with penalty for constraint violations"""
    h_vals = construct_h_from_coeffs(coeffs, n_points)
    # Check sum constraint - add penalty if violated
    sum_violation = abs(np.sum(h_vals) * DX - TARGET_SUM)
    if sum_violation > 1e-4:
        penalty = penalty_weight * sum_violation
    else:
        penalty = 0
    
    c5 = evaluate_c5(h_vals)
    return c5 + penalty
