import numpy as np

import time

# Global variables
TARGET_SUM = None
n_points = 120

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def initialize_h_random(n, target_sum):

    h = np.random.uniform(0, 1, n)
    return project_to_sum_box(h, target_sum)

def compute_gradient(h, w_norm):
    """Compute gradient using convolution method"""
    n = len(h)
    offset = n - 1
    
    # Compute convolutions efficiently
    conv_wh = np.convolve(w_norm, h, mode='full')
    conv_rev = np.convolve(w_norm[::-1], h, mode='full')
    
    # Extract relevant portions
    grad = 1.0 - (conv_rev[offset:offset+n] + conv_wh[offset:offset+n])
    return grad

def gradient_descent(h_start, n_iters, lr, eps, target_sum):
    """Perform gradient descent with smoothing"""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for iter in range(n_iters):
        # Compute correlation
        corr = np.correlate(h, 1-h, mode='full')
        max_corr = np.max(corr)
        
        # Compute softmax weights
        w_raw = np.exp((corr - max_corr) / eps)
        w = w_raw / w_raw.sum()
        
        # Compute gradient
        grad = compute_gradient(h, w)
        
        # Update step
        h_new = h - lr * grad
        
        # Projection onto constraints
        h_new = project_to_sum_box(h_new, target_sum)
        h = h_new
        
        # Evaluate
        if iter % 10 == 0:
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5
