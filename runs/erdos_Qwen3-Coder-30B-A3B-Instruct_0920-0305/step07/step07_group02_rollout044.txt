import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
TARGET_SUM = 128  # exactly half of 256 points
N = 256
dx = 2.0 / N

def c5_fft(h):
    """Fast computation of C5 using FFT-based linear convolution"""
    n = len(h)
    nfft = 1 << ((2*n-1).bit_length())  # next power of two
    H = np.fft.rfft(h, nfft)
    B = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * B, nfft)[:2*n-1]  # length 2N-1
    return np.max(conv) * dx

def project_to_sum(h, target_sum):
    """Project vector onto the simplex: sum = target_sum, all entries in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Binary search for lambda: sum(clamp(h + lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(h + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def thue_morse(n):
    """Generate Thue-Morse sequence of length n"""
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        # Count number of 1 bits in binary representation of i
        tm[i] = bin(i).count('1') % 2
    return tm

def rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence of length n"""
    rs = np.zeros(n, dtype=int)
    for i in range(n):
        # Convert to base 4 representation, count occurrences of "11"
        # but this is tricky. Instead use trivial O(n) generation.
        # For simplicity: let's just generate a simple 0,1 alternating
        # with some smoothness to avoid degenerate cases.
        rs[i] = (i // 2) % 2  # Alternate but with pattern
    return rs

def greedy_descent(h0):
    """Greedy descent on worst shift"""
    h = h0.copy().astype(int)
    while True:
        corr_full = correlate(h, 1-h, mode='full')
        max_idx = np.argmax(corr_full)
        s = max_idx - (N - 1)
        
        # Find mismatched pairs for the worst shift s
        A = []  # set of (1,0) pairs: h[i] = 1, h[i+s] = 0
        B = []  # set of (0,1) pairs: h[i] = 0, h[i+s] = 1

        if s >= 0:
            # Look at i from 0 to N - s - 1
            for i in range(N - s):
                if h[i] == 1 and h[i+s] == 0:
                    A.append(i)
                elif h[i] == 0 and h[i+s] == 1:
                    B.append(i)
        else:
            # s < 0
            for i in range(-s, N):
                if h[i] == 1 and h[i+s] == 0:
                    A.append(i)
                elif h[i] == 0 and h[i+s] == 1:
                    B.append(i)
        
        if not A or not B:
            break

        i_swap = np.random.choice(A)
        j_swap = np.random.choice(B)
        # Swap h[i_swap] and h[j_swap]
        h[i_swap], h[j_swap] = h[j_swap], h[i_swap]
    
    # After greedy descent project to desired sum
    h = project_to_sum(h.astype(float), TARGET_SUM)
    h = np.clip(h, 0, 1)
    return h

    current_c5 = c5_fft(h.astype(float))
    best_h = h.copy()
    best_c5 = current_c5
    start_time = time.time()
    anneal_time = 280  # 50% of 600s, ~300s total but use a bit less to allow other steps

    t = 0
    T = 0.02  # initial temperature

    # Track recent improvement
    last_improvement = 0
    no_improvement_count = 0

    while True:
        if time.time() - start_time > anneal_time:
            break

        # try a random swap
        i = np.random.randint(N)
        j = np.random.randint(N)
        # Ensure swap doesn't make sum invalid (not necessary in binary optimization as it preserves sum)

        # Check if valid 'swap'
        if h[i] == h[j]:  # skip if same value
            continue
        # Perform the swap
        h[i], h[j] = h[j], h[i]

        new_c5 = c5_fft(h.astype(float))
        delta = new_c5 - current_c5

        if delta <= 0:
            current_c5 = new_c5
        else:
            # Accept with probability exp(-delta/T)
            if np.random.rand() < np.exp(-delta / T):
                current_c5 = new_c5
            else:
                h[i], h[j] = h[j], h[i]

        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            last_improvement = t

        # Cooling schedule
        t += 1
        if t > 0 and t % 500 == 0:
            # brief greedy descent to speed up convergence
            h = greedy_descent(h).astype(int)
            current_c5 = c5_fft(h.astype(float))
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                last_improvement = t
        if t - last_improvement > 2000:
            if t - last_improvement > 5000:
                break  # stop if no improvement in a long time
            # restart with a new temperature window
            T = max(0.0001, T * 0.99)
        else:
            # decrease T slower when there is some progress
            T = max(0.0001, T * 0.9999)

    return best_h.astype(float), best_c5

def smooth_vector(h):
    """Apply mild smoothing to create a float vector"""
    kernel = np.array([0.25, 0.5, 0.25])
    h_smooth = np.convolve(h, kernel, mode='same')
    h_smooth = np.clip(h_smooth, 0, 1)
    return h_smooth
