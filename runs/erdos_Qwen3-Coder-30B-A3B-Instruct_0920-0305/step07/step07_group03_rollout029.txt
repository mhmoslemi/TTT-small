import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def evaluate_with_gradients(h):
    """Evaluate C5 and return gradient information"""
    # Simple finite difference gradient approximation
    eps = 1e-6
    base_c5 = evaluate_c5(h)
    grad = np.zeros_like(h)
    
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_sum_box(h_plus, TARGET_SUM)
        c5_plus = evaluate_c5(h_plus)
        grad[i] = (c5_plus - base_c5) / eps
    
    return base_c5, grad

def local_descent(h0, max_iter=1000):
    """Local descent optimization using gradient information"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        c5, grad = evaluate_with_gradients(h)
        
        # Adaptive step size
        step_size = 0.05 / (1.0 + i / 100.0)
        h_new = h - step_size * grad
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < c5:
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # If no improvement, reduce step size and try again
            new_c5 = evaluate_c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    
    return best_h, best_c5

def improve_solution_with_mutation(h0, max_iter=1000):
    """Mutation-based improvement"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    for i in range(max_iter):
        # Large mutation
        if np.random.random() < 0.3:
            idxs = np.random.choice(n_points, size=min(20, n_points//4), replace=False)
            for idx in idxs:
                h[idx] = np.random.random()
        else:
            # Small mutation
            idxs = np.random.choice(n_points, size=min(5, n_points//8), replace=False)
            for idx in idxs:
                delta = np.random.normal(0, 0.02)
                h[idx] += delta
                h[idx] = np.clip(h[idx], 0, 1)
        
        h = project_to_sum_box(h, TARGET_SUM)
        new_c5 = evaluate_c5(h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    return best_h, best_c5
