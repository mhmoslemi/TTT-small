import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time
from scipy.optimize import minimize

# Global variables
TARGET_SUM = None
n_points = 150
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    # Use cvxpy to set up and solve the LP

    # Try an initial guess that's eccentric, appropriate for refining
    initial_guess = np.zeros(n_points)

    center = n_points // 2
    left_start = n_points // 4
    right_start = 3 * n_points // 4

def subgradient_descent(h0, max_iter=2000, learning_rate=0.5):
    """Subgradient descent for refining the solution"""
    h = h0.copy()
    
    def compute_subgradient(k_star):
        # Compute subgradient with respect to shift k_star
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_star
            if 0 <= j < n_points:
                grad[i] += 1 - h[j]
            else:
                pass  # out of bounds
            j = i - k_star
            if 0 <= j < n_points:
                grad[i] -= h[j]
            else:
                pass  # out of bounds
        return grad

    for it in range(max_iter):
        # Compute current max correlation
        corr = compute_corr_fft(h)
        k_star = np.argmax(corr[:len(corr)//2 + 1]) - (len(corr)//2)
        if k_star == 0:
            k_star = 1
        # Compute subgradient
        g = compute_subgradient(k_star)
        # Step
        h_new = h - learning_rate * g
        # Project
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        # Accept
        if evaluate_c5(h_new) < evaluate_c5(h):
            h = h_new.copy()
            learning_rate *= 0.999
        else:
            learning_rate *= 0.999
        if it % 100 == 0:
            # Periodically check for early stopping
            if learning_rate < 1e-6:
                break
            
    return h


def run(seed=42, budget_s=600, **kwargs):

    h0 = lp_initialization()

    best_c5 = evaluate_c5(h0)
    
    # Phase 2: Refine with subgradient descent
    phase2_end = start_time + 0.8 * budget_s
    
    while time.time() < phase2_end:
        # Perform sub-gradient refinement
        h_refined = subgradient_descent(best_h.copy(), max_iter=300)
        c5_refined = evaluate_c5(h_refined)
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()

            h_noisy = best_h + noise

            h_noisy = subgradient_descent(h_noisy, max_iter=200)
            c5_noisy = evaluate_c5(h_noisy)
            if c5_noisy < best_c5:
                best_h = h_noisy.copy()
                best_c5 = c5_noisy
    
    # Phase 3: Maybe run a quick local refinement if time permits
    if time.time() < start_time + budget_s:
        h_final = subgradient_descent(best_h.copy(), max_iter=200)
        c5_final = evaluate_c5(h_final)
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final
    
    return (best_h.tolist(), float(best_c5), n_points)
