```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5_fft(h):
    """Fast evaluation of C5 using FFT"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def greedy_descent(h0, max_iter=1000):
    """Improve solution with deterministic worst shift descent"""
    h = h0.copy()
    best_c5 = compute_c5_fft(h)
    
    for _ in range(max_iter):
        corr_full = np.correlate(h, h, mode="full")
        max_corr_idx = np.argmax(corr_full)
        worst_shift = max_corr_idx - (len(h) - 1)
        
        # Find mismatched positions
        A = []
        B = []
        for i in range(len(h)):
            j = i + worst_shift
            if 0 <= j < len(h):
                if h[i] == 1 and h[j] == 0:
                    A.append(i)
                elif h[i] == 0 and h[j] == 1:
                    B.append(i)
        
        if not A or not B:
            break
            
        # Pick random elements for swap
        i = np.random.choice(A)
        j = np.random.choice(B)
        
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5_fft(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            # Restore
            h[i], h[j] = h[j], h[i]
    return h, best_c5

def anneal_binary(h0, time_budget):
    """Lightweight simulated annealing"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5_fft(h)
    
    start = time.time()
    T0 = 0.02
    Tmax = 10000
    
    for t in range(Tmax):
        if time.time() - start > time_budget - 1:
            break
            
        # Pick two random positions
        i = np.random.randint(len(h))
        j = np.random.randint(len(h))
        
        if i == j:
            continue
            
        # Try swap
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5_fft(h)
        
        # Accept or reject
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / (T0 * (1 - t/Tmax))):
            best_c5 = new_c5
            best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
            
        # Occasionally apply greedy step to accelerate
        if t % 100 == 0:
            h, _ = greedy_descent(h, 50)
            new_c5 = compute_c5_fft(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    N = 120
    dx = 2.0 / N
    target_sum = N // 2
    
    # Attempt MILP solution first
    h_best = None
    c5_best = float('inf')
    
    try:
        # Build the MILP model
        h = cp.Variable(N, boolean=True)
        lam = cp.Variable(integer=True)
        
        # Constraints
        constraints = [cp.sum(h) == target_sum]
        constraints.append(h[0] == 1)  # symmetry breaking
        
        # Index k, from -(N-1) to N-1, but only valid range from 0 to N-1
        w_vars = []
        
        for k in range(-(N-1), N):
            # compute valid i indices
            i_valid = np.arange(max(0, -k), min(N, N-k))
            if len(i_valid) == 0:
                continue
                
            # w[i,k] = h[i] * h[i+k]
            w = cp.Variable(len(i_valid), boolean=True)
            constraints.append(w <= h[i_valid])  # w <= h[i]
            constraints.append(w <= h[i_valid + k])  # w <= h[i+k]
            constraints.append(w >= h[i_valid] + h[i_valid + k] - 1)  # w >= h[i] + h[i+k] -1
            w_vars.append((w, i_valid, k))
            
        # Add constraints that correlation is at least lambda for all shifts
        for w, i_valid, k in w_vars:
            shift_sum = cp.sum(w)
            constraints.append(shift_sum >= lam)  # min lambda constraint
        
        # Objective
        objective = cp.Maximize(lam)
        
        # Solve
        prob = cp.Problem(objective, constraints)
        solver_options = {"max_seconds": max(10, 0.55 * budget_s)}
        prob.solve(solver='CBC', verbose=False, **solver_options)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE]:
            h_sol = np.round(h.value).astype(int)
            # Project to exactly correct sum
            current_sum = np.sum(h_sol)
            diff = int(current_sum - target_sum)
            if diff != 0:
                # Flip diff bits from 1 to 0
                indices_1 = np.where(h_sol == 1)[0]
                if len(indices_1) >= abs(diff):
                    flip_indices = indices_1[:abs(diff)]
                    h_sol[flip_indices] = 0
                    
            c5 = 1.0 - (2.0 * lam.value) / N
            h_best = h_sol
            c5_best = c5
        else:
            h_best = np.random.randint(0, 2, N)
            h_best = project_to_sum_box(h_best.astype(float), target_sum).astype(int)
            c5_best = compute_c5_fft(h_best)
    except Exception as e:
        # Fallback to heuristic
        h_best = np.random.randint(0, 2, N)
        h_best = project_to_sum_box(h_best.astype(float), target_sum).astype(int)
        c5_best = compute_c5_fft(h_best)
    
    # Greedy descent polishing
    elapsed = time.time() - start_time
    if elapsed < 0.75 * budget_s:
        h_best, c5_best = greedy_descent(h_best)
    
    # Simulated annealing polish if time permits
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        h_best, c5_best = anneal_binary(h_best, remaining)
    
    return (h_best.tolist(), float(c5_best), N)
```
