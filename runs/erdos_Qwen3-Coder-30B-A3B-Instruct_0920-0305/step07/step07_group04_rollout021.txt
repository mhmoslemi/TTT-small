    n_points = 120

        for j in range(n_points):
            constraints.append(P[i,j] >= 0)
            constraints.append(P[i,j] <= h[i])
            constraints.append(P[i,j] <= h[j])
            constraints.append(P[i,j] >= h[i] + h[j] - 1)
    
    # Force symmetry for products (due to h_i * h_j = h_j * h_i)
    for i in range(n_points):
        for j in range(i+1, n_points):
            constraints.append(P[i,j] == P[j,i])
    
    # Add constraints for diagonal terms
    for i in range(n_points):
        constraints.append(P[i,i] >= 0)
        constraints.append(P[i,i] <= h[i])
        constraints.append(P[i,i] >= 2*h[i] - 1)

    # We'll solve for time efficiency using relatively few shifts
    max_shift = min(n_points, 50)  # limit shifts for efficiency

    # Build correlation constraints
    correlations = []

        if not (k >= 0):  # shifting left 
            # correlation for shift k (where k < 0 means shifting left)
            # The term h[i]*(1-h[i-k])

            corr_k = 0

                # At index n-1+k (where k is lag), the correlation corresponds to 
                # sum_{i=0}^{n-1-abs(k)} h[i]*1_{i+k} for terms where both i and i+k in [0,n-1]
                # = sum_{i=0}^{n-1-abs(k)} h[i]*h[i+k] when valid
                
                # for k > 0: sum_{i=0}^{n-1-k} h[i]*(1-h[i+k]) = sum h[i] - sum h[i]*h[i+k]
                # = sum h[i] - sum_{i=0}^{n-1-k} P[i,i+k] = S - sum_{i=0}^{n-1-k} P[i,i+k] 
                # For negative k, consider how correlation works with convolution
                # Global correlation by full: index corresponds to shift k
                # index i corresponds to shift from -n+1 to n-1 inclusive  
                # Let's compute manually conversion from numpy pattern:
                
                # Actually let's go directly by associating product variables

                if k >= 0:
                    corr_k = cp.sum(h[:n_points-k]) - cp.sum(P[np.arange(n_points-k), np.arange(k, n_points)])
                else:
                    # k < 0: shift left, need compute correlations indices such that
                    #  i-k < n means i < n + abs(k), i >= 0. So valid integral: range(0, n+k) 
                    # But from correlation definition and numpy: 
                    # mode='full', so shift k from -n+1 to n-1, corresponding to valid index 
                    # for correlation value = sum_{i} h[i] * b[i+k]  where b is the vector  
                    # (1-h) of same length but zero padded, so b is itself equal to h simply

                    # So actually b is NOT systematic from shifting h values by 1-h. Let's avoid it.

                    # So i from [0, n-1], l from [-n+1, n-1]

                    # so correlation = sum_{i=max(0,-k)}^{min(n-1,n-1-k)} h[i] * (1 - h[i+k])
                    # = sum_{i=max(0,-k)}^{min(n-1,n-1-k)} h[i] - sum_{i=max(0,-k)}^{min(n-1,n-1-k)} h[i] * h[i+k]
                    # = sum_{i=max(0,-k)}^{min(n-1,n-1-k)} h[i] 
                    #                     - sum_{i=max(0,-k)}^{min(n-1,n-1-k)} P[i,i+k]
                    
                    corr_k = cp.sum(h[max(0,-k):min(n_points,n_points-k)]) \

                correlations.append(corr_k)

    max_corr = cp.Variable(name='max_corr')

    # Objective is to minimize max_corr, i.e., minimal correlation, reverse of maximizing overlap.
    objective = cp.Minimize(max_corr)
    
    # In practice, it might be helpful to derive the final constraint matrix from the full computation
    # of correlation for each shift. Let's directly solve optimization problem with constraints
    # on correlation values rather than large matrix manipulations.
    
    # Create full symmetry checker
    # Optimize for overlap expressions directly:
    # The main goal is to minimize maximum of all correlation terms.
    # z = sum h[i] * (1 - h[i+k]) for valid i sum if is indeed i+k in [0,n-1].
    # For shift k, the constraint defines valid range:
    # i ranges from max(0, -k) to min(n-1, n-1-k)  
    # i from max(0, -k) to min(n-1, n-1-k)  
    # = i from max(0, -k) to min(n-1, n-1-k)
    
    # Define intermediate helpful calculations:
    def compute_corr(h_var, k_shift):
        start = max(0, -k_shift)
        end = min(n_points, n_points - k_shift)
        if k_shift >= 0:
            valid_indices = slice(start, end)
            corr_value = cp.sum(h_var[valid_indices]) - cp.sum(P[valid_indices, valid_indices+k_shift])
        else:
            valid_indices = slice(start, end)
            corr_value = cp.sum(h_var[valid_indices]) - cp.sum(P[valid_indices, valid_indices+k_shift])
        return corr_value
    
    # Manually build list of shift values to consider: reasonable subset
