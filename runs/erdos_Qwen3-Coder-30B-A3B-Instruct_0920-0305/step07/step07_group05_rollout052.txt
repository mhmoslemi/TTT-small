import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global variables
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmin_weights(s, alpha):
    """Compute normalized weights for softmin"""
    s_shifted = s - np.max(s)  # For numerical stability
    exps = np.exp(-alpha * s_shifted)
    return exps / np.sum(exps)

def weighted_gradient(h, alpha):
    """Compute gradient of softmin with respect to h"""
    # Compute autocorrelation
    s = correlate(h, h, mode='full')
    # Compute weights
    w = softmin_weights(s, alpha)
    # Compute gradient
    grad = np.zeros_like(h)
    n = len(h)
    for i in range(len(s)):
        lag = i - (n - 1)
        if lag >= 0:
            # h[i] * h[i+lag] contribution
            if i + lag < n:
                grad[i] += w[i] * h[i + lag]
            if i - lag >= 0:
                grad[i] += w[i] * h[i - lag]
        else:
            # h[i] * h[i-lag] contribution
            if i - lag < n:
                grad[i] += w[i] * h[i - lag]
            if i + lag >= 0:
                grad[i] += w[i] * h[i + lag]
    return grad

def pairwise_move(h, grad, delta, rng):
    """Attempt to transfer mass between indices"""
    i_src = np.argmin(grad)
    i_tgt = np.argmax(grad)
    
    # Ensure indices are valid
    if h[i_src] <= 0 or h[i_tgt] >= 1:
        return False, h.copy(), 0
    
    # Compute transfer amount
    transfer = min(delta, h[i_src], 1 - h[i_tgt])
    if transfer <= 1e-10:
        return False, h.copy(), 0
    
    # Perform transfer
    h[i_src] -= transfer
    h[i_tgt] += transfer
    h = project_to_sum_box(h, TARGET_SUM)
    return True, h, transfer

def annealed_gradient_ascent(h0, time_budget, alpha0, delta0, T0, rng):
    """Main optimization loop using annealed gradient ascent"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    alpha = alpha0
    delta = delta0
    T = T0
    iter_count = 0
    
    start_time = time.time()
    while time.time() - start_time < time_budget:
        # Compute weighted gradient (this is the key difference from previous algo)
        grad = weighted_gradient(h, alpha)
        
        # Attempt a move
        accepted, h_new, _ = pairwise_move(h, grad, delta, rng)
        if accepted:
            h = h_new
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Update parameters
        iter_count += 1
        if iter_count % 500 == 0:
            # Gradually increase alpha (sharpness)
            alpha = min(alpha * 1.1, 200.0)
            # Gradually decrease delta
            delta *= 0.9995
            # Gradually decrease temperature
            T *= 0.9995
            # Occasionally reset delta to avoid getting stuck
            if iter_count % 5000 == 0:
                delta = min(delta * 1.2, 0.5)
    
    return best_h, best_c5

def random_binary_start(n, rng):
    """Create a feasible binary start"""
    h = np.zeros(n)
    ones_indices = rng.sample(range(n), k=n//2)
    h[ones_indices] = 1
    return project_to_sum_box(h, TARGET_SUM)

def local_optimize_restart(start_h, remaining_time, rng):
    """Run a local optimization on a start point"""
    # Run for a fraction of remaining time
    time_budget = remaining_time * 0.8
    
    # Run main ascent
    h_final, c5_final = annealed_gradient_ascent(
        start_h, time_budget, alpha0=10.0, delta0=0.4, T0=5e-3, rng=rng
    )
    
    return h_final, c5_final
