import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Constants
N = 256
dx = 2.0 / N
TARGET_SUM = N // 2

def c5_fft(h):
    """Fast C5 evaluation using FFT."""
    n = len(h)
    nfft = 1 << ((2*n-1).bit_length())  # next power of two
    H = np.fft.rfft(h, nfft)
    B = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * B, nfft)[:2*n-1]
    return np.max(conv) * dx

def project_to_sum(h, target_sum):
    """Project vector h onto the simplex with sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Use water-filling approach
    sorted_h = np.sort(h)[::-1]
    cumsum = np.cumsum(sorted_h)
    # Find where we can set values to 1
    n_above = 0
    while n_above < len(sorted_h) and sorted_h[n_above] > 1:
        n_above += 1
    if n_above > 0:
        # This case can be handled directly
        h = np.clip(h, 0, 1)
    else:
        # Use water-filling
        while n_above < len(sorted_h) and (sorted_h[n_above] * (n_above + 1) < target_sum):
            n_above += 1
        # Set everything above the threshold to 1 or max 1
        threshold = target_sum / n_above if n_above > 0 else 0.0
        h = np.clip(h, 0, 1)
        h = np.where(h > threshold, 1.0, h)
        h = np.where(h > 0, h, 0.0)
        # Normalize to target_sum
        current = np.sum(h)
        if current > target_sum:
            diff = current - target_sum
            # Reduce from the highest ones
            indices = np.argsort(h)[::-1]
            for i in indices:
                if diff <= 0: break
                remain = min(diff, h[i])
                h[i] -= remain
                diff -= remain
        elif current < target_sum:
            diff = target_sum - current
            # Increase from the lowest zeros
            indices = np.argsort(h)
            for i in indices:
                if diff <= 0: break
                remain = min(diff, 1 - h[i])
                h[i] += remain
                diff -= remain
    return h

def thue_morse(n):
    """Generate Thue-Morse sequence with exactly n/2 ones and zeros."""
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        count = bin(i).count('1') % 2
        tm[i] = count
    # Balance: if more 1s than needed, flip some 1s to 0s
    ones = np.sum(tm)
    excess = ones - n//2
    if excess > 0:
        flip_indices = np.where(tm == 1)[0][:excess]
        tm[flip_indices] = 0
    elif excess < 0:
        flip_indices = np.where(tm == 0)[0][:abs(excess)]
        tm[flip_indices] = 1
    return tm.astype(float)

def rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence with exactly n/2 ones and zeros."""
    # Rudin-Shapiro just zaps between more 1s and 0s
    # Simple approach: just components
    rs = np.zeros(n, dtype=int)
    for i in range(n):
        # Count number of non-overlapping pairs of consecutive bits
        count = 0
        j = 0
        while j < i and j + 1 < i:
            if (i & (1 << j)) and (i & (1 << (j + 1))):
                count += 1
            j += 2
        # Mod 2 of count
        rs[i] = count % 2
    # Count 1s
    ones = np.sum(rs)
    excess = ones - n//2
    if excess > 0:
        flip_indices = np.where(rs == 1)[0][:excess]
        rs[flip_indices] = 0
    elif excess < 0:
        flip_indices = np.where(rs == 0)[0][:abs(excess)]
        rs[flip_indices] = 1
    return rs.astype(float)

def greedy_descent(h_init):
    """Deterministic greedy descent on binary vector."""
    h = h_init.copy().astype(float)
    num_ones = int(np.sum(h))
    
    # Ensure the correct sum exactly
    if num_ones > TARGET_SUM:
        # Remove excess ones (from the beginning or randomly)
        ones_indices = np.where(h == 1)[0]
        to_zero = ones_indices[:num_ones - TARGET_SUM]
        h[to_zero] = 0
    elif num_ones < TARGET_SUM:
        zeros_indices = np.where(h == 0)[0]
        to_one = zeros_indices[:TARGET_SUM - num_ones]
        h[to_one] = 1
    
    max_iter = N * 2  # Safety
    iter = 0

    while iter < max_iter:
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (N - 1)
        corr_max = corr[max_idx]
        
        if shift >= 0:
            a = list(range(N - shift))
            b = list(range(shift, N))
        else:
            a = list(range(-shift, N))
            b = list(range(N + shift))

        # Find indices of mismatched pairs
        A = []
        B = []
        
        for i in range(len(a)):
            i_a = a[i]
            if h[i_a] == 1 and h[b[i]] == 0:
                A.append(i_a)
            if h[i_a] == 0 and h[b[i]] == 1:
                B.append(i_a)
        
        if len(A) == 0 or len(B) == 0:
            break

        # Choose a random pair to swap
        swap_i = np.random.choice(A)
        swap_j = np.random.choice(B)
        
        # Ensure we swap and keep sum correct
        h[swap_i] = 0
        h[swap_j] = 1
        iter += 1
    
    return h, c5_fft(h)

def simulated_annealing(h0, T0=0.02, cooling=0.99998, max_steps=200000):
    """Simulated Annealing for binary vector."""
    h_current = h0.copy()
    c5_current = c5_fft(h_current)
    h_best = h_current.copy()
    c5_best = c5_current
    
    T = T0
    steps = 0
    last_improved = 0

    while steps < max_steps and T > 1e-6:
        # Propose a random swap of two elements
        idx1 = np.random.randint(0, N)
        idx2 = np.random.randint(0, N)
        
        # Make sure they’re different and values are different
        if h_current[idx1] == h_current[idx2]:
            steps += 1
            continue
            
        # Swap
        h_new = h_current.copy()
        h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
        
        # Keep the correct sum
        current_sum = np.sum(h_new)
        if current_sum > TARGET_SUM:
            # Find 1 to flip to 0
            ones = np.where(h_new == 1)[0]
            if len(ones) > TARGET_SUM:
                to_flip = ones[:len(ones) - TARGET_SUM]
                for i in to_flip:
                    h_new[i] = 0
        elif current_sum < TARGET_SUM:
            zeros = np.where(h_new == 0)[0]
            if len(zeros) > N - TARGET_SUM:
                to_flip = zeros[:TARGET_SUM - len(ones)]
                for i in to_flip:
                    h_new[i] = 1
        
        c5_new = c5_fft(h_new)
        
        if c5_new <= c5_current or np.random.rand() < np.exp(-(c5_new - c5_current) / T):
            h_current = h_new
            c5_current = c5_new
            
            if c5_current < c5_best:
                h_best = h_new.copy()
                c5_best = c5_current
                last_improved = steps
        
        steps += 1
        # Cooling
        T *= cooling
        # Optional mini-greedy descent every few steps
        if steps % 500 == 0 and c5_best < 0.4:
            _, c5_greedy = greedy_descent(h_best.copy())
            if c5_greedy < c5_best:
                h_best = h_best.copy()
                c5_best = c5_greedy
        
        # Terminate early if no improvement for a while
        if steps > last_improved + 50000:
            break
    
    return h_best, c5_best

def continuous_refinement(h_binary, max_iter=500):
    """Continuously refine the binary solution."""
    h = h_binary.copy()
    h_smooth = np.convolve(h, np.array([0.25, 0.5, 0.25]), mode='same')
    h = np.clip(h_smooth, 0, 1)
    h = project_to_sum(h, TARGET_SUM)
    
    def objective(h_in):
        return c5_fft(h_in)
    
    def constraint_eq(h_in):
        return np.sum(h_in) - TARGET_SUM
    
    bounds = [(0, 1) for _ in range(N)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective,
            h,
            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'gtol': 1e-8, 'ftol': 1e-12}
        )
        if res.success:
            h_final = res.x
        else:
            h_final = h
    except:
        h_final = h
    
    # Final projection to ensure constraint
    h_final = project_to_sum(h_final, TARGET_SUM)
    return h_final, c5_fft(h_final)
