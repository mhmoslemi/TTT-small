import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
from scipy.optimize import minimize

# Global constants
n_points = 150
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Step 1: Set up the LP variables and constraints
    h = cp.Variable(n_points)
    p = cp.Variable((n_points, n_points), symmetric=True)
    
    # Objective: minimize t (upper bound on the maximum correlation)
    t = cp.Variable()
    
    # Constraints
    mass_constraint = cp.sum(h) == TARGET_SUM
    box_constraints = [0 <= h[i] <= 1 for i in range(n_points)]
    
    # McCormick envelopes
    mccormick_constraints = []
    for i in range(n_points):
        for j in range(i, n_points):
            mccormick_constraints += [
                p[i, j] >= 0,
                p[i, j] >= h[i] + h[j] - 1,
                p[i, j] <= h[i],
                p[i, j] <= h[j]
            ]
    
    # Shift constraints: compute the overlap term O_k = sum p_ij over valid pairs
    all_shifts = []
    for k in range(-n_points+1, n_points):
        # Indices to avoid boundary effects where k extends beyond n_points
        if k >= 0:
            idx1 = slice(0, n_points-k)
            idx2 = slice(k, n_points)
        else:
            idx1 = slice(-k, n_points)
            idx2 = slice(0, n_points+k)
        if len(idx1) > 0 and len(idx2) > 0:
            # Sum over product variables
            overlap = cp.sum([p[i, i+k] for i in range(len(idx1))])
            all_shifts.append(overlap)
    
    # Each overlap is bounded by t
    regression_constraints = [overlap <= TARGET_SUM - t for overlap in all_shifts]
    
    # Composite problem
    constraints = [mass_constraint] + box_constraints + mccormick_constraints + regression_constraints + [t >= 0]
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve the LP
    solver_timeout = 150
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=5000, abstol=1e-8, reltol=1e-8)
        if prob.status == cp.OPTIMAL:
            h_val = h.value
        else:

            h_val = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), h_val)
    except:
        h_val = np.array(initial_h_values)
        h_val = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), h_val)

    # Normalize and clip to ensure feasibility
    h_val = np.clip(h_val, 0, 1)
    h_val = h_val * (TARGET_SUM / np.sum(h_val))

    # Step 2: Refine with projected sub-gradient
    def evaluate_overlap(h_vec):
        # Compute all correlations using convolution of h and (1-h)
        hrev = np.flip(h_vec)
        b = 1 - h_vec
        corr = correlate(hrev, b, mode="full")
        corr = corr[len(corr)//2:] * dx
        return np.max(corr)

    def project_to_sum(x, target_sum):
        # Find lambda such that sum(clamp(x + lambda, 0, 1)) = target_sum
        def f(lam):
            proj = np.clip(x + lam, 0, 1)
            return np.sum(proj) - target_sum
        
        a, b = -1.0, 1.0
        for _ in range(100):
            if f(a) * f(b) > 0:
                a -= 0.1
                b += 0.1
            else:
                break
        for _ in range(50):
            m = (a + b) / 2
            fm = f(m)
            if abs(fm) < 1e-12:
                break
            if fm * f(a) < 0:
                b = m
            else:
                a = m
        return np.clip(x + (a+b)/2, 0, 1)
    
    def calculate_subgradient(h_vec, k_active):
        # k_active: shift index that maximizes overlap
        if k_active >= 0:
            # h_vec[0:n-k] * (1 - h_vec[k:n])
            g = np.zeros_like(h_vec)
            g[:-k_active] = (1 - h_vec[k_active:])  # contribution from 1 - h_{i+k}
            g[k_active:] += -h_vec[:-k_active]       # contribution from h_{i-k}
        else:
            # h_vec[0:k] * (1 - h_vec[n+k:n])
            g = np.zeros_like(h_vec)
            # h_{i+k} when k < 0: h_{i-k}, so k_active=0 means no shift
            g[:k_active] = (1 - h_vec[-k_active:])  # from 1-h+i+k
            g[-k_active:] += -h_vec[:k_active]      # from h+i-k (shifted)
        return g
    
    h_current = h_val.copy()
    current_c5 = evaluate_overlap(h_current)
    
    iter_count = 0
    max_iter = 5000
    step_scale = 0.5 * dx
    patience = 50

    while time.time() - start_time < 0.7 * budget_s and iter_count < max_iter:
        # Identify dominant shift
        hrev = np.flip(h_current)
        b = 1 - h_current
        corr = correlate(hrev, b, mode="full")
        corr = corr[len(corr)//2:] * dx
        k_best = np.argmax(corr)
        if k_best > n_points - 1:
            k_best -= n_points
        
        # Compute tentative gradient (subgradient)
        g = calculate_subgradient(h_current, k_best)
        
        # Step along gradient
        h_trial = h_current - step_scale * g
        
        # Apply projection
        h_projected = project_to_sum(h_trial, TARGET_SUM)
        
        # Evaluate new objective
        new_c5 = evaluate_overlap(h_projected)
        
        if new_c5 < current_c5:
            h_current = h_projected
            current_c5 = new_c5
            step_scale = min(step_scale * 1.05, 0.5)
            patience = 50
        else:
            patience -= 1
            if patience <= 0:
                step_scale *= 0.99
                patience = 50
                
        iter_count += 1
        
    return (h_current.tolist(), float(current_c5), n_points)
