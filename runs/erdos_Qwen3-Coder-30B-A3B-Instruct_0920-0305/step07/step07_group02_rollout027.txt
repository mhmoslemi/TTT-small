import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

# Global variables for the problem
N = 120
dx = 2.0 / N
TARGET_SUM = N // 2

def compute_c5_fft(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def greedy_descent(h0):
    """Iteratively improve by fixing the worst shift"""
    h = h0.copy()
    c5 = compute_c5_fft(h)
    
    for iteration in range(1000):
        # Compute all correlations
        corr_full = correlate(h, h, mode='full')
        max_idx = np.argmax(corr_full)
        k = max_idx - (len(h) - 1)  # actual shift parameter
        
        # Positions that contribute to the maximum correlation
        contrib_indices = []
        for i in range(len(h)):
            j = i + k
            if 0 <= j < len(h):
                contrib_indices.append((i, j))
                
        # Identify mismatched pairs
        A = [] # positions where h[i]=1 and h[i+k]=0
        B = [] # positions where h[i]=0 and h[i+k]=1
        
        for i, j in contrib_indices:
            if h[i] == 1 and h[j] == 0:
                A.append(i)
            elif h[i] == 0 and h[j] == 1:
                B.append(i)
                
        if not A or not B:
            break
            
        # Pick a random pair to swap
        i_swap = np.random.choice(A)
        j_swap = np.random.choice(B)
        
        # Perform swap
        h[i_swap], h[j_swap] = h[j_swap], h[i_swap]
        
        new_c5 = compute_c5_fft(h)
        if new_c5 >= c5:
            # Undo if not helpful (but likely rare)
            h[i_swap], h[j_swap] = h[j_swap], h[i_swap]
            break
        else:
            c5 = new_c5
    
    return h, c5

def anneal_binary(h0, max_time_s):
    """Lightweight simulated annealing on binary vectors"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5_fft(h)
    
    # Cooling schedule
    T0 = 0.02
    Tmax = 10000
    start_time = time.time()
    
    for t in range(Tmax):
        if time.time() - start_time > max_time_s - 1:
            break
            
        T = T0 * (1 - t / Tmax)
        if T <= 0:
            break
            
        # Generate move: pick two random indices
        i = np.random.randint(0, len(h))
        j = np.random.randint(0, len(h))
        
        # Ensure they are different
        if i == j:
            continue
            
        # Try to swap a 1 and a 0
        if h[i] == 0:
            continue
        if h[j] == 1:
            continue
            
        # Save current state and simulate swap
        old_c5 = compute_c5_fft(h)
        
        # Swap bits
        h[i], h[j] = 0, 1
        
        new_c5 = compute_c5_fft(h)
        
        # Accept or reject
        delta = new_c5 - old_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        else:
            # Revert swap
            h[i], h[j] = 1, 0
            
        # Occasionally perform greedy descent
        if t % 100 == 0:
            h, c5 = greedy_descent(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    
    return best_h, best_c5

def initial_binary_h():
    """Construct initial binary vector via interpolation from initial_h_values"""
    init_vals = np.array(initial_h_values)
    x_base = np.linspace(0, 2, len(init_vals))
    x_new = np.linspace(0, 2, N)
    h_interp = np.interp(x_new, x_base, init_vals)
    h = np.round(h_interp).astype(int)
    
    # Ensure correct sum
    diff = TARGET_SUM - np.sum(h)
    if diff > 0:
        # Add ones
        idxs = np.argsort(h)
        for i in range(diff):
            h[idxs[i]] = 1
    elif diff < 0:
        # Subtract ones
        idxs = np.argsort(h)[::-1]
        for i in range(-diff):
            h[idxs[i]] = 0
            
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Timing budget allocation
    milp_time = 0.55 * budget_s
    remaining_time = budget_s - (time.time() - start_time)
    
    # Solve MILP (if feasible)
    h_best = None
    c5_best = float('inf')
    try:
        # Variables
        h = cp.Variable(N, boolean=True)
        lam = cp.Variable(integer=True)
        
        # Constraints
        constraints = [cp.sum(h) == TARGET_SUM, lam >= 0, lam <= TARGET_SUM]
        
        # Precompute all valid indices for each shift
        w_vars = {}
        for k in range(-(N-1), N):
            indices_i = np.arange(max(0,-k), min(N, N-k))
            if len(indices_i) > 0:
                idxs_j = indices_i + k
                # Create auxiliary variables
                w = cp.Variable(len(indices_i), boolean=True)
                w_vars[(k,indices_i[0])] = (w, indices_i, idxs_j)
                
                # McCormick constraints for each w[i,k] = h[i]*h[i+k]
                for idx_i, idx_j in zip(indices_i, idxs_j):
                    constraints.extend([
                        w[idx_i - indices_i[0]] <= h[idx_i],
                        w[idx_i - indices_i[0]] <= h[idx_j],
                        w[idx_i - indices_i[0]] >= h[idx_i] + h[idx_j] - 1
                    ])
                
                # Constraint for correlation sum
                constraints.append(cp.sum(w) >= lam)
        
        # Break symmetry
        constraints.append(h[0] == 1)
        
        # Objective
        problem = cp.Problem(cp.Maximize(lam), constraints)
        
        # Solve
        problem.solve(solver=cp.CBC, max_seconds=milp_time, verbose=False)
        
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE]:
            h_val = np.round(h.value).astype(int)
            lambda_val = int(lam.value)
            c5_milp = 1 - 2 * lambda_val / N
            h_best = h_val
            c5_best = c5_milp
        else:
            # Fallback to binary initialization
            h_best = initial_binary_h()
            c5_best = compute_c5_fft(h_best)
            
    except Exception as e:
        # Reset to fallback solution
        h_best = initial_binary_h()
        c5_best = compute_c5_fft(h_best)

    # Greedy refinement
    elapsed = time.time() - start_time
    if elapsed < 0.75 * budget_s:
        h_best, c5_best = greedy_descent(h_best)

    # Annealing polishing
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 30:
        h_best, c5_best = anneal_binary(h_best, remaining)

    return (h_best.tolist(), float(c5_best), N)
