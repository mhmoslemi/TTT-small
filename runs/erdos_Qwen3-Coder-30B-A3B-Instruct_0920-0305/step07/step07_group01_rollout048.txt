```python
import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = 1.0
n_points = 200
dx = 2.0 / n_points
MAX_FREQS = 8  # Number of cosine modes to use

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def fourier_basis(x, coeffs, max_freq):
    """Build h(x) from cosine basis with given coefficients"""
    h = np.full_like(x, coeffs[0])  # baseline
    for i in range(1, min(len(coeffs), max_freq + 1)):
        h += coeffs[i] * np.cos(i * np.pi * x)
    return np.clip(h, 0, 1)

def sample_constraints(coeffs, max_freq, n_samples=1000):
    """Return constraints that h(x) stays in [0,1]"""
    x = np.linspace(0, 2, n_samples)
    h_vals = fourier_basis(x, coeffs, max_freq)
    return np.min(h_vals), np.max(h_vals)

def objective_with_constraints(coeffs, max_freq):
    """Objective function with penalty for constraint violations"""
    # Ensure the integral equals 1 (automatic with cos basis)
    # Sample constraints
    h_low, h_high = sample_constraints(coeffs, max_freq)
    penalty = 0
    if h_low < 0:
        penalty += 1e6 * (0 - h_low)**2
    if h_high > 1:
        penalty += 1e6 * (h_high - 1)**2
    
    # Compute actual C5
    x = np.linspace(0, 2, n_points)
    h = fourier_basis(x, coeffs, max_freq)
    c5 = evaluate_c5(h)
    return c5 + penalty

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize coefficients (small number as we're optimizing over fewer variables)
    initial_coeffs = np.zeros(MAX_FREQS + 1)
    initial_coeffs[0] = 0.5  # baseline to make integral equal 1
    
    # Set bounds for each coefficient
    bounds = [(-2, 2)] * (MAX_FREQS + 1)
    
    # Run differential evolution with budget control
    result = differential_evolution(
        objective_with_constraints,
        bounds,
        args=(MAX_FREQS,),
        maxiter=max(100, int(2000 * budget_s / 600)),
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        disp=False,
        polish=True
    )
    
    # Refine final result with local optimization
    final_coeffs = result.x
    final_x = np.linspace(0, 2, n_points)
    final_h = fourier_basis(final_x, final_coeffs, MAX_FREQS)
    final_c5 = evaluate_c5(final_h)
    
    # Return the best solution found within budget
    return (final_h.tolist(), float(final_c5), n_points)
```
