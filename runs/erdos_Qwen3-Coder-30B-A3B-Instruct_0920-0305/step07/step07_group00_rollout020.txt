def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr_and_weights(h, eps):
    """Compute correlation and softmax weights"""
    # Compute correlation
    corr = correlate(h, 1 - h, mode="full")
    # Compute softmax weights
    max_corr = np.max(corr)
    w_raw = np.exp((corr - max_corr) / eps)
    w_norm = w_raw / np.sum(w_raw)
    return corr, w_norm

def compute_gradient(h, w_norm):
    """Compute gradient using convolution method"""
    # Compute using direct fft approach
    n = len(h)
    
    # Invert weights for convolution
    w_rev = w_norm[::-1]
    
    # Compute convolutions
    L = 2 * n - 1
    
    # For h_conv = sum_s w_s * h[i - s]
    conv_wh = np.convolve(w_norm, h, mode='full')
    # For h_rev_conv = sum_s w_s * h[i + s] 
    conv_rev_wh = np.convolve(w_rev, h, mode='full')
    
    # Extract middle portion
    offset = n - 1
    # We want len n values
    grad = 1.0 - (conv_rev_wh[offset:offset+n] + conv_wh[offset:offset+n])
    return grad

def gradient_descent(h_start, eps_schedule, iters_per_eps, lr_schedule, target_sum):
    """Run gradient descent with given schedule"""
    h = h_start.copy()
    best_h = h.copy()

    iter_count = 0
    total_iters = sum(iters_per_eps)
    
    for eps_idx, (eps, num_iters) in enumerate(zip(eps_schedule, iters_per_eps)):
        lr = lr_schedule[eps_idx] if eps_idx < len(lr_schedule) else 0.01
        
        for _ in range(num_iters):
            iter_count += 1
            corr, w_norm = compute_corr_and_weights(h, eps)
            gradient = compute_gradient(h, w_norm)
            
            h_new = h - lr * gradient
            h_new = project_to_sum_box(h_new, target_sum)
            h = h_new
            
            if iter_count % 50 == 0:
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    
    return best_h, best_c5

    h = np.random.uniform(0, 1, n)
