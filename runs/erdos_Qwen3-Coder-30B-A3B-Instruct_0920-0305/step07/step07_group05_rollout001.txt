import numpy as np
import time

# Global variables
n_points = None
dx = None
TARGET_SUM = None

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    global dx
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    x = np.asarray(x, dtype=np.float64)
    # simple binary search for lambda such that sum = target_sum.
    lo, hi = -1.0, 1.0
    # expand until sign change
    def f(lam):
        return np.sum(np.clip(x + lam, 0.0, 1.0)) - target_sum
    while f(lo) > 0:
        lo -= 1.0
    while f(hi) < 0:
        hi += 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        val = f(mid)
        if val > 0:
            hi = mid
        else:
            lo = mid
    lam = (lo + hi) / 2.0
    return np.clip(x + lam, 0.0, 1.0)

def compute_gradient(h, lag):
    """Compute gradient of correlation for a specific lag"""
    n = len(h)
    grad = np.ones(n)
    if lag >= 0:
        lag = int(lag)
        if lag < n:
            grad[:n-lag] -= h[lag:]
            grad[lag:] -= h[:n-lag]
    else:
        lag = -int(lag)
        if lag < n:
            grad[lag:] -= h[:n-lag]
            grad[:lag] -= h[lag:]
    return grad

def random_binary(n):
    """Create random binary vector with exactly n/2 ones"""
    ones = np.zeros(n)
    idx = np.random.choice(n, n//2, replace=False)
    ones[idx] = 1.0
    return ones

        if src_candidates.size > 0 and tgt_candidates.size > 0:
            src = src_candidates[np.argmax(grad[src_candidates])]
            tgt = tgt_candidates[np.argmin(grad[tgt_candidates])]
            delta_allowed = min(h[src], 1.0 - h[tgt])
            if delta_allowed > 1e-12:
                delta = max_delta * np.random.rand()
                delta = min(delta, delta_allowed)
                # apply move
                h[src] -= delta
                h[tgt] += delta
                c5_new = evaluate_c5(h)
                if c5_new < c5:
                    c5 = c5_new
                    # recompute correlation and lag_max
                    corr = np.correlate(h, 1 - h, mode='full')
                    lag_max = int(np.argmax(corr) - (n - 1))
                else:
                    # revert
                    h[src] += delta
                    h[tgt] -= delta
        else:
            # no greedy move possible; do random move
            i, j = np.random.choice(n, 2, replace=False)
            if h[i] > 1e-12 and h[j] < 1 - 1e-12:
                delta_allowed = min(h[i], 1.0 - h[j])
                delta = max_delta * np.random.rand()
                delta = min(delta, delta_allowed)
                h[i] -= delta
                h[j] += delta
                c5_new = evaluate_c5(h)
                if c5_new < c5:
                    c5 = c5_new
                    corr = np.correlate(h, 1 - h, mode='full')
                    lag_max = int(np.argmax(corr) - (n - 1))
                else:
                    # revert
                    h[i] += delta
                    h[j] -= delta
        # occasionally do a random perturbation to escape local minima
        if iteration % 500 == 0:
            # Random perturb of few entries
            for _ in range(3):
                i, j = np.random.choice(n, 2, replace=False)

                    # transfer a small fraction
                    delta = max_delta * 0.1 * np.random.rand()
                    delta = min(delta, h[i], 1.0 - h[j])

                    h[j] += delta
            c5_new = evaluate_c5(h)
            if c5_new < c5:
                c5 = c5_new
                corr = np.correlate(h, 1 - h, mode='full')

            c5 = evaluate_c5(h)
            corr = np.correlate(h, 1 - h, mode='full')
            lag_max = int(np.argmax(corr) - (n - 1))
    return h, c5
