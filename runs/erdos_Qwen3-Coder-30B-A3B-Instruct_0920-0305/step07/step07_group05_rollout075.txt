import numpy as np
import cvxpy as cp
from scipy.linalg import eigh
from scipy.signal import correlate
import time

def build_lag_matrices(n):
    """Build sparse matrices A_ℓ for all lags ℓ ∈ {-(n-1), ..., n-1}"""
    A_list = []
    for l in range(-(n-1), n):
        # Create indices for the lag l
        row = np.arange(n)
        col = row + l
        # Only keep valid indices
        valid = (col >= 0) & (col < n)
        if np.any(valid):
            A = np.zeros((n, n))
            A[row[valid], col[valid]] = 1
            A_list.append((l, A))
    return A_list

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def round_and_evaluate(X_opt, target_sum, n_points):
    """Round the relaxed solution to a feasible vector"""
    # Eigendecomposition
    _, V = eigh(X_opt)
    # Take the first (largest) eigenvector
    v = V[:, -1]
    # Scale to approximate target sum
    scale_factor = target_sum / np.sum(np.abs(v))
    h = v * scale_factor    
    # Clip and reproject
    h = np.clip(h, 0, 1)
    h = project_to_sum_box(h, target_sum)
    return h
