import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def adaptive_pattern_search(h0, max_iter=1000):
    """Improves using adaptive pattern search that focuses on problem areas"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        corr_full = compute_corr_fft(h)
        max_corr_idx = np.argmax(corr_full)
        
        # Sample around the maximum correlation point to better understand the landscape
        start_idx = max(0, max_corr_idx - 20)
        end_idx = min(len(h), max_corr_idx + 20)
        
        # Modify the region around worst correlation with adaptive step sizes
        step_size = 0.01 * (0.9 ** (i // 100))
        
        # Randomly select a few points to modify
        indices = np.random.choice(range(start_idx, end_idx), size=min(8, end_idx - start_idx), replace=False)
        
        for idx in indices:
            # Try a small positive or negative step
            direction = np.random.choice([-1, 1])
            h_new = h.copy()
            h_new[idx] += step_size * direction
            
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5:

def direct_optimization_refinement(h0, max_iter=1000):

        # Build full h vector from subset

        pass
    
    return h
