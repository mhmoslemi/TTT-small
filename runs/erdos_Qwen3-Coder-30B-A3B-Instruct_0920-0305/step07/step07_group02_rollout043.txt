import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global constants
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_binary_initial():
    """Create a binary initial solution using fast construction"""
    h = np.zeros(n_points)
    
    # Use Thue-Morse sequence or similar for good spread
    # Simple approach: fill with 1's then randomize
    # Start with alternating 0,1 pattern for symmetry
    for i in range(n_points):
        h[i] = i % 2
    
    # Randomize a bit while maintaining sum
    count_ones = int(n_points / 2)
    indices = np.random.choice(n_points, count_ones, replace=False)
    h.fill(0)
    h[indices] = 1
    
    return h

def greedy_worst_shift_improve(h_start):
    """Improve solution by greedy swap on worst shift"""
    h = h_start.copy()
    old_c5 = evaluate_c5(h)
    
    iter_count = 0
    while iter_count < 10000:  # Safety limit
        iter_count += 1
        corr = compute_corr_fft(h)
        max_shift_idx = np.argmax(corr)
        shift = max_shift_idx - (n_points - 1)
        
        if shift >= 0:
            left = h[:-shift]
            right = h[shift:]
        else:
            left = h[-shift:]
            right = h[:shift]
        
        # Identify mismatched pairs for current shift
        # (1,0) and (0,1) in [(h_i, h_{i+shift})] are problematic
        ones_first = np.where((left == 1) & (right == 0))[0]
        zeros_second = np.where((left == 0) & (right == 1))[0]
        
        if len(ones_first) == 0 or len(zeros_second) == 0:
            break
            
        # Pick one such pair and attempt swap
        i = ones_first[random.randint(0, len(ones_first)-1)]
        j = zeros_second[random.randint(0, len(zeros_second)-1)]
        
        # Swap elements
        h[i], h[j] = h[j], h[i]
        
        # Re-evaluate
        new_c5 = evaluate_c5(h)
        
        if new_c5 >= old_c5:
            # Revert the change
            h[i], h[j] = h[j], h[i]
            break
        else:
            old_c5 = new_c5
            
    return h

def simulated_annealing(h0, T_init=1e-2, T_final=1e-6, steps=1000):
    """Simulated Annealing with temperature decay"""
    h_current = h0.copy()
    c5_current = evaluate_c5(h_current)
    
    T = T_init
    decay = (T_final / T_init) ** (1.0 / steps)
    
    for _ in range(steps):
        # Propose a small random change (swap two points)
        i = random.randint(0, n_points - 1)
        j = random.randint(0, n_points - 1)
        if i == j:
            continue
        
        # Swap
        h_current[i], h_current[j] = h_current[j], h_current[i]
        
        c5_new = evaluate_c5(h_current)
        
        if c5_new < c5_current:
            # Accept improvement
            c5_current = c5_new
        else:
            # Accept with probability
            prob = np.exp(-(c5_new - c5_current) / T)
            if random.random() > prob:
                # Revert swap
                h_current[i], h_current[j] = h_current[j], h_current[i]
        
        T *= decay
    
    return h_current

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    best_c5 = float('inf')
    best_h = None
    
    # Keep running random restarts until timeout
    restarts = 0
    while time.time() < start_time + budget_s:
        restarts += 1
        
        h = generate_binary_initial()
        
        # Apply local greedy improvement
        h_improved = greedy_worst_shift_improve(h)
        
        # Simulated annealing exploration
        h_annealed = simulated_annealing(h_improved, steps=500, T_init=0.02)
        
        # Continuous refinement using SLSQP
        def objective(h_local):
            return evaluate_c5(h_local)
        
        def constraint_eq(h_local):
            return np.sum(h_local) - TARGET_SUM
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                h_annealed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_final = res.x
            else:
                h_final = h_annealed
        except:
            h_final = h_annealed
            
        h_final = project_to_sum_box(h_final, TARGET_SUM)
        c5_final = evaluate_c5(h_final)
        
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final
            
    assert best_h is not None
    
    return (best_h.tolist(), float(best_c5), n_points)
