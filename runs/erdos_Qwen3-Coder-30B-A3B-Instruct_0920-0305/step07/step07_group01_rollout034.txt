import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def perturb_solution(h, strength=0.01):
    """Create a perturbed version of the solution"""
    perturbed = h + np.random.normal(0, strength, n_points)
    return project_to_sum_box(perturbed, TARGET_SUM)

def adaptive_gradient_guided_search(h0, max_iter=1000, patience=50):
    """Gradient-guided local search with adaptive step sizes"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Track improvement to adapt step size
    last_improvement = 0
    step_size = 0.01
    
    for i in range(max_iter):
        if i - last_improvement > patience:
            step_size *= 0.9  # Reduce step size if no improvement
            
        # Compute gradient approximation by finite differences
        grad = np.zeros(n_points)
        for k in range(n_points):
            # Estimate gradient by making small perturbations
            test_h = h.copy()
            delta = np.random.normal(0, step_size)
            test_h[k] = np.clip(test_h[k] + delta, 0, 1)
            new_c5 = evaluate_c5(test_h)
            grad[k] = (best_c5 - new_c5) / (delta + 1e-12)
        
        # Move in gradient direction
        if np.any(np.abs(grad) > 1e-10):
            # Find the best direction to improve
            idx_min = np.argmin(grad)
            idx_max = np.argmax(grad)
            
            if idx_min != idx_max:
                delta = min(h[idx_min], 1 - h[idx_max], step_size)
                if delta > 1e-6:
                    h[idx_min] -= delta
                    h[idx_max] += delta
                    h = project_to_sum_box(h, TARGET_SUM)
                    c5 = evaluate_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                        last_improvement = i
                    
    return best_h, best_c5

def optimized_particle_swarm_search(h0, max_iter=2000):
    """Enhanced particle swarm optimization with better update rules"""
    # Initialize particles
    num_particles = 20
    particles = [h0.copy() for _ in range(num_particles)]
    velocities = [np.random.normal(0, 0.01, n_points) for _ in range(num_particles)]
    particle_best = [p.copy() for p in particles]
    particle_best_scores = [evaluate_c5(p) for p in particles]
    
    # Global best
    global_best_idx = np.argmin(particle_best_scores)
    global_best = particle_best[global_best_idx].copy()
    global_best_score = particle_best_scores[global_best_idx]
    
    # PSO parameters
    w = 0.7  # Inertia weight
    c1 = 1.5  # Cognitive coefficient
    c2 = 1.5  # Social coefficient
    
    for iter_num in range(max_iter):
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (particle_best[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            
            # Update position
            new_pos = particles[i] + velocities[i]
            particles[i] = project_to_sum_box(new_pos, TARGET_SUM)
            
            # Evaluate new position
            new_score = evaluate_c5(particles[i])
            
            # Update personal best
            if new_score < particle_best_scores[i]:
                particle_best_scores[i] = new_score
                particle_best[i] = particles[i].copy()
                
                # Update global best
                if new_score < global_best_score:
                    global_best_score = new_score
                    global_best = particles[i].copy()
    
    return global_best, global_best_score
