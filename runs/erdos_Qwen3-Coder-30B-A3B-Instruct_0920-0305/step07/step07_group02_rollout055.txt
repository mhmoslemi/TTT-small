import numpy as np
from scipy.signal import correlate

def c5_fft(h):
    n = len(h)
    nfft = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, nfft)
    B = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * B, nfft)[:2*n-1]
    return np.max(conv) * (2.0 / n)

def project_to_sum(h, target_sum):
    """Project h onto simplex with sum = target_sum using binary search"""
    h = np.array(h)
    def f(lam):
        proj = np.clip(h + lam, 0, 1)
        return np.sum(proj) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def thue_morse(n):
    """Generate Thue-Morse sequence of length n"""
    if n == 0: return []
    seq = [0]
    while len(seq) < n:
        seq += [1 - x for x in seq]
    return seq[:n]

def rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence of length n"""
    if n == 0: return []
    seq = [0]
    for i in range(1, n):
        # Rudin-Shapiro does not directly produce 0,1 uniformly,
        # so we convert accordingly
        count = bin(i).count('1') % 2
        seq.append(count)
    return seq[:n]

def greedy_descent(h_start):
    """Greedy descent focusing on worst shift"""
    h = np.array(h_start).copy()
    n = len(h)
    target_sum = n // 2

    while True:
        corr = correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n - 1)
        
        # Find mismatched pairs
        A = []  # (1,0) pairs
        B = []  # (0,1) pairs
        
        if max_shift >= 0:
            for i in range(n - max_shift):
                if h[i] == 1 and h[i + max_shift] == 0:
                    A.append(i)
                elif h[i] == 0 and h[i + max_shift] == 1:
                    B.append(i)
        else:
            shift = -max_shift
            for i in range(n - shift):
                if h[i] == 1 and h[i + shift] == 0:
                    A.append(i)
                elif h[i] == 0 and h[i + shift] == 1:
                    B.append(i)
        
        if not A or not B:
            break
            
        i = np.random.choice(A)
        j = np.random.choice(B)
        
        # Swap bits at positions i and j
        h[i], h[j] = h[j], h[i]
        
    return h, c5_fft(h)
