import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_gradient_refinement(h0, max_iter=200):
    """Refine with gradient-based optimization"""
    h = h0.copy()
    
    # Focus on best regions found
    corr_full = compute_corr_fft(h)
    max_idx = np.argmax(corr_full)
    
    # Optimize with bounds and constraint using SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_local):
        return evaluate_c5(h_local)
    
    def constraint_eq(h_local):
        return np.sum(h_local) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
    except:
        pass
    
    # Project to maintain constraints
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def binary_greedy_improvement(h):
    """Improve binary solution by focusing on worst shift"""
    n = len(h)
    max_iter = 1000
    
    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)
        
        # Find mismatched pairs (1,0) and (0,1) for this shift
        if shift >= 0:
            i_range = np.arange(0, n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
            
        h_i = h[i_range]
        h_j = h[j_range]
        
        # (1,0) mismatches: h_i == 1 and h_j == 0
        im1 = i_range[(h_i == 1) & (h_j == 0)]
        # (0,1) mismatches: h_i == 0 and h_j == 1
        im2 = i_range[(h_i == 0) & (h_j == 1)]
        
        if im1.size == 0 or im2.size == 0:
            break
            
        # Choose random i, j from mismatched pairs
        i = np.random.choice(im1)
        j = np.random.choice(im2)
        
        # Perform swap to reduce overlap for this shift
        h[i] = 0
        h[j] = 1
        
    return h

        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h
    
    return (best_h.tolist(), float(best_c5), n_points)
