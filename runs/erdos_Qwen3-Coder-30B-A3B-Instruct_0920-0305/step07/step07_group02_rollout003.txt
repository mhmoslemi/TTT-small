def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_thue_morse(n):
    """Generate Thue-Morse sequence of length n"""
    tm = [0]
    for i in range(n):
        if i >= len(tm):
            tm += [1-j for j in tm]
    return np.array(tm[:n], dtype=float)

def generate_rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence of length n"""
    rs = [0] * n
    for i in range(1, n):
        rs[i] = (rs[i-1] + (1 if (i & (i-1)) == 0 else 0)) % 2
    return np.array(rs, dtype=float)

def greedy_swap_improvement(h0, max_iter=10000):
    """Improve given h by swapping (1,0) and (0,1) pairs for maximum shift"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for iteration in range(max_iter):
        # Recompute correlation
        corr = compute_corr_fft(h)
        max_shift_idx = np.argmax(corr)
        max_shift = max_shift_idx - (n_points - 1)
        
        # Find mismatched pairs for this shift
        if max_shift >= 0:
            # h[i] and h[i+shift] should both be 1  
            # h[i] and h[i+shift] are (1,0) or (0,1) for overlap
            h1 = h[:-max_shift]
            h2 = h[max_shift:]
            mismatched = np.where((h1 == 1) & (h2 == 0) | (h1 == 0) & (h2 == 1))[0]
        else:
            # h[i] and h[i+shift] 
            h1 = h[-max_shift:]
            h2 = h[:-max_shift]
            mismatched = np.where((h1 == 1) & (h2 == 0) | (h1 == 0) & (h2 == 1))[0]
        
        if len(mismatched) == 0:
            break
        
        # Pick a random mismatch and attempt a swap
        i = random.choice(mismatched)

            j = i + max_shift
        else:
            j = i - max_shift
        
        # Try to swap h[i] and h[j]
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # Project back to constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)

def simulated_annealing(h0, temp_schedule='exp', max_iter=5000):
    """Simulated Annealing with targeted swaps"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    temp_start = 0.1
    temp_end = 0.0001
    decay_rate = 0.999
    
    temperature = temp_start
    
    for iteration in range(max_iter):
        if temp_schedule == 'linear':
            temperature = temp_end + (temp_start - temp_end) * (1 - iteration/max_iter)
        elif temp_schedule == 'exp':
            temperature *= decay_rate
        
        # Choose two random indices
        i = random.randint(0, n_points-1)
        j = random.randint(0, n_points-1)
        if i == j:
            continue
            
        # Perform swap
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # Project to satisfy constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        delta = new_c5 - best_c5
        
        if delta < 0 or random.random() < np.exp(-delta / temperature):
            h = h_new.copy()
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
    
    return best_h

def continuous_refinement(h0, max_iter=500):
    """Refine with gradient-based optimization"""
    h = h0.copy()
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_local):
        return evaluate_c5(h_local)
    
    def constraint_eq(h_local):
        return np.sum(h_local) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
    except:
        pass
    
    # Project to maintain constraints
    h = project_to_sum_box(h, TARGET_SUM)
    
    return h
