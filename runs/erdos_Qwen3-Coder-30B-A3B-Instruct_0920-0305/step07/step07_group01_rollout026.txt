import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def perturb_solution(h, strength=0.01):
    """Create a perturbed version of the solution"""
    perturbed = h + np.random.normal(0, strength, n_points)
    return project_to_sum_box(perturbed, TARGET_SUM)

def construct_initial_solution():
    """Construct a good initial solution"""
    # Start with a piecewise constant construction to avoid small overlaps
    h = np.zeros(n_points)
    
    # Place a peak in the middle, then use slopes for gentle transition
    center = n_points // 2
    width = n_points // 6
    
    # Create a shape with a central peak and gradual slopes
    for i in range(n_points):
        dist = abs(i - center)
        if dist < width:
            h[i] = 1.0 - dist / width
        else:
            h[i] = 0.0
            
    # Apply some smoothing
    h = np.convolve(h, np.ones(5)/5, mode='same')
    
    # Project to correct sum
    return project_to_sum_box(h, TARGET_SUM)

def adaptive_gradient_guided_search(h0, max_iter=1000, patience=50):
    """Gradient-guided local search with adaptive step sizes"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Track improvement to adapt step size
    last_improvement = 0
    step_size = 0.01
    
    for i in range(max_iter):
        if i - last_improvement > patience:
            step_size *= 0.9  # Reduce step size if no improvement
            
        # Compute gradient approximation using finite differences
        grad = np.zeros(n_points)
        for k in range(n_points):
            # Small perturbations to estimate gradient
            delta = 0.001
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[k] = np.clip(h_plus[k] + delta, 0, 1)
            h_minus[k] = np.clip(h_minus[k] - delta, 0, 1)
            
            c5_plus = evaluate_c5(h_plus)
            c5_minus = evaluate_c5(h_minus)
            grad[k] = (c5_minus - c5_plus) / (2 * delta)
        
        # Move in direction of steepest descent
        idx_min = np.argmin(grad)
        idx_max = np.argmax(grad)
        
        if idx_min != idx_max:
            delta = min(h[idx_min], 1 - h[idx_max], step_size)
            if delta > 1e-6:
                h[idx_min] -= delta
                h[idx_max] += delta
                h = project_to_sum_box(h, TARGET_SUM)
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    last_improvement = i
                    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    
    # Initial construction with better starting point
    best_h = construct_initial_solution()
    best_c5 = evaluate_c5(best_h)
    
    # Multiple runs with different random starting points
    phase1_end = start_time + 0.4 * budget_s
    for run_idx in range(10):
        if time.time() >= phase1_end:
            break
            
        # Vary starting point significantly
        h_start = perturb_solution(best_h, 0.05)
        h_refined, c5_refined = adaptive_gradient_guided_search(h_start, 500, 50)
        
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    
    # Use more sophisticated optimization
    phase2_end = start_time + 0.7 * budget_s
    while time.time() < phase2_end:
        # Random perturbation around current best
        h_perturbed = perturb_solution(best_h, 0.01)
        
        # Try different optimization approaches
        bounds = [(0, 1) for _ in range(n_points)]
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
            
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            # Try multiple strategies
            strategies = [
                ('SLSQP', {'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10}),
                ('L-BFGS-B', {'maxiter': 500})
            ]
            
            for method, options in strategies:
                if time.time() >= phase2_end:
                    break
                    
                try:
                    res = minimize(
                        objective, 
                        h_perturbed, 
                        method=method, 
                        bounds=bounds, 
                        constraints=cons,
                        options=options
                    )
                    if res.success:
                        h_opt = res.x
                        h_opt = project_to_sum_box(h_opt, TARGET_SUM)
                        c5_opt = evaluate_c5(h_opt)
                        if c5_opt < best_c5:
                            best_c5 = c5_opt
                            best_h = h_opt.copy()
                except:
                    continue
        except:
            pass
    
    # Final fine-tuning
    phase3_end = start_time + budget_s
    while time.time() < phase3_end:
        # Use a hybrid approach with both gradient and heuristic search
        h_fine = perturb_solution(best_h, 0.002)
        h_final, c5_final = adaptive_gradient_guided_search(h_fine, 150, 30)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
