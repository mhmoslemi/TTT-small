import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_lp_relaxation(h0, n_points, dx):
    """Solve the LP relaxation for the max overlap"""
    # Variables
    h = cp.Variable(n_points, bounds=(0, 1))
    p = cp.Variable((n_points, n_points), entrywise=True)
    T = cp.Variable(1, nonneg=True)

    # Constraints
    constraints = [
        cp.sum(h) == n_points / 2,
        T >= 0
    ]

    # McCormick envelopes
    for i in range(n_points):
        for j in range(i, n_points):
            constraints += [
                p[i, j] >= 0,
                p[i, j] >= h[i] + h[j] - 1,
                p[i, j] <= h[i],
                p[i, j] <= h[j]
            ]

    # Overlap constraints for all shifts
    max_shift = n_points - 1
    for k in range(-max_shift, max_shift+1):
        if k >= 0:
            # Forward shift
            O_k = cp.sum([p[i, i+k] for i in range(n_points - k)])
        else:
            # Backward shift
            O_k = cp.sum([p[i-k, i] for i in range(-k, n_points)])
        constraints += [O_k >= n_points/2 - T]

    # Objective
    objective = cp.Minimize(T)
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.GLPK, time_limit=180)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value
    except:
        pass

    return h0  # Return the original if optimization fails

def binary_search_corrected_c5(h0, n_points, dx, target_bound):
    """Binary search approach to get a correct value"""
    low = 0.0
    high = n_points / 2
    tolerance = 1e-4 * dx

    best_h = h0.copy()
    best_C5 = evaluate_c5(h0)

    while high - low > tolerance:
        mid = (low + high) / 2
        h_solution = solve_lp_relaxation(h0, n_points, dx)
        h_solution = project_to_sum_box(h_solution, n_points / 2)

        actual_c5 = evaluate_c5(h_solution)
        if actual_c5 <= mid * dx:
            # Valid, rejection test fails
            best_h = h_solution.copy()
            best_C5 = actual_c5
            high = mid
        else:
            # This indicates an error, but use mid anyway
            low = mid

    return best_h, best_C5

def subgradient_polish(h0, max_iter=2000):
    """Polish the solution using subgradients"""
    h = h0.copy()
    t = 0
    alpha0 = 1e-3

    for _ in range(max_iter):
        dc = compute_corr_fft(h)
        k_star = np.argmax(dc)
        g = np.zeros_like(h)

        # Compute subgradient
        for i in range(len(h)):
            if i + k_star < len(h):
                g[i] += (1 - h[i + k_star])
            if i - k_star >= 0:
                g[i] -= h[i - k_star]

        # Subgradient step
        alpha = alpha0 / np.sqrt(t + 1)
        h = h - alpha * g

        # Project onto feasible set
        h = project_to_sum_box(h, len(h) / 2)
        t += 1

    return h

def local_exchange_refinement(h, max_iter=100):
    """Perform local exchange of mass"""
    h = h.copy()
    n_points = len(h)

    for _ in range(max_iter):
        # Randomly sample pairs
        i, j = np.random.choice(n_points, 2, replace=False)
        delta_max = min(h[j], 1 - h[i])
        if delta_max <= 0:
            continue

        step = 0.1 * delta_max
        # Apply exchange
        h[i] += step
        h[j] -= step

        # Project
        h = project_to_sum_box(h, n_points / 2)

    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Parameters
    n_points = 150
    dx = 2.0 / n_points

    # Initialize with provided values if available
    if 'initial_h_values' in globals():
        # Interpolate to the current n_points
        h_ref = np.array(initial_h_values)
        x_ref = np.linspace(0, 2, len(h_ref))
        x_new = np.linspace(0, 2, n_points)
        h_interp = np.interp(x_new, x_ref, h_ref)
        h0 = h_interp
    else:
        h0 = np.ones(n_points) * 0.5

    h0 = project_to_sum_box(h0, n_points / 2)

    # Phase 1: Binary search with LP relaxation
    h_lp = solve_lp_relaxation(h0, n_points, dx)
    h_lp = project_to_sum_box(h_lp, n_points / 2)

    h_final, C5 = binary_search_corrected_c5(h_lp, n_points, dx, 1)
    best_C5 = C5

    # Phase 2: Sub-gradient polishing
    h_p = subgradient_polish(h_final, max_iter=2000)
    C5_p = evaluate_c5(h_p)
    if C5_p < best_C5:
        best_C5 = C5_p
        h_final = h_p

    # Phase 3: Local exchange refinement
    h_e = local_exchange_refinement(h_final, max_iter=100)
    C5_e = evaluate_c5(h_e)
    if C5_e < best_C5:
        best_C5 = C5_e
        h_final = h_e

    return (h_final.tolist(), float(best_C5), n_points)
