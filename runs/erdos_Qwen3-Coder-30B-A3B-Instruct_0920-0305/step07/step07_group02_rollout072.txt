import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random
from math import log

# Global constants
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points
MAX_RESTARTS = 300
MAX_GREEDY_ITER = 2000
MAX_ANNEALING_ITER = 10000
TEMP_START = 1.0
TEMP_DECAY = 0.9995
NOISE_MAG = 0.01

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_thue_morse(n):
    """Generate Thue-Morse sequence of length n"""
    tm = [0]
    for i in range(int(log(n, 2)) + 1):
        tm = tm + [1 - x for x in tm]
    return np.array(tm[:n])

def generate_rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence of length n, normalized to 0/1"""
    rs = np.zeros(n)
    rs[0] = 1
    for i in range(1, n):
        if i % 2 == 0:
            rs[i] = rs[i//2]
        else:
            # Rudin-Shapiro has some complexity, simplified here
            rs[i] = 1 - rs[(i-1)//2]
    return rs

def generate_binary_initial(n):
    """Generate binary initial vector with exactly n/2 ones"""
    h = np.zeros(n)
    ones_indices = np.random.choice(n, n//2, replace=False)
    h[ones_indices] = 1
    return h

def perform_swap(h, i, j, k):
    """Perform a swap of positions i and j for shift k"""
    h_new = h.copy()
    # Check if we can perform a beneficial swap
    if h_new[i] != h_new[j]:
        h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new

def find_worst_shift_indices(h):
    """Find indices in worst shift for changeable flip pairs"""
    corr = compute_corr_fft(h)
    max_idx = np.argmax(corr)
    max_shift = max_idx - (n_points - 1)
    return max_shift

def greedy_improve(h, max_iter=1000):
    """Improved greedy algorithm for better overlap reduction"""
    h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for iteration in range(max_iter):
        corr = compute_corr_fft(h)
        max_idx = np.argmax(corr)
        shift = max_idx - (n_points - 1)
        
        if shift >= 0:
            # Left shift
            left = h[:n_points-shift]
            right = h[shift:]
        else:
            # Right shift
            left = h[-shift:]
            right = h[:n_points+shift]
        
        mismatches = np.where((left == 1) & (right == 0))[0]
        mismatches2 = np.where((left == 0) & (right == 1))[0]
        
        # Try to make beneficial swaps
        if len(mismatches) > 0 and len(mismatches2) > 0:
            i = random.choice(mismatches)
            j = random.choice(mismatches2)
            
            # Try to swap when possible
            if shift >= 0 and i + shift < n_points:
                h_new = h.copy()
                h_new[i], h_new[i+shift] = h_new[i+shift], h_new[i]
                new_c5 = evaluate_c5(h_new)
                if new_c5 < best_c5:
                    h = h_new
                    best_c5 = new_c5
        else:
            # No improvement possible; proceed or break
            break
    
    return h

def simulated_annealing(h, max_iter, temp_start, temp_decay):
    """Perform simulated annealing from current state"""
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    temp = temp_start
    for i in range(max_iter):
        if temp < 1e-10:
            break
            
        # Make a random swap
        i1 = np.random.randint(n_points)
        i2 = np.random.randint(n_points)
        if i1 != i2:
            nh = current_h.copy()
            nh[i1], nh[i2] = nh[i2], nh[i1]
            nh = project_to_sum_box(nh, TARGET_SUM)  # Ensure constraints
            
            new_c5 = evaluate_c5(nh)
            delta = new_c5 - current_c5
            
            # Accept or reject based on Metropolis criterion
            if delta < 0 or random.random() < np.exp(-delta / temp):
                current_h = nh
                current_c5 = new_c5
                
                if new_c5 < best_c5:
                    best_h = nh.copy()
                    best_c5 = new_c5
                    
        temp *= temp_decay
        
    return best_h, best_c5

def continuous_refinement(h):
    """Use SLSQP for continuous refinement"""
    bounds = [(0, 1) for _ in range(n_points)]
    h_float = h.astype(float)
    
    def objective(h_local):
        return evaluate_c5(h_local)
    
    def constraint_eq(h_local):
        return np.sum(h_local) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h_float, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_float = res.x
    except:
        pass
    
    return project_to_sum_box(h_float, TARGET_SUM)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    
    TARGET_SUM = n_points / 2
    
    # Generate multiple starting points from different sequences
    initial_conditions = []
    
    # Thue-Morse sequence
    tm = generate_thue_morse(n_points)
    initial_conditions.append(tm / np.sum(tm) * TARGET_SUM)
    
    # Rudin-Shapiro sequence
    rs = generate_rudin_shapiro(n_points)
    initial_conditions.append(rs / np.sum(rs) * TARGET_SUM)
    
    # Random binary initialization
    for _ in range(10):
        rand_bin = generate_binary_initial(n_points)
        initial_conditions.append(rand_bin / np.sum(rand_bin) * TARGET_SUM)
    
    # Additional random initialization
    for _ in range(20):
        h_rand = np.random.random(n_points)
        initial_conditions.append(h_rand / np.sum(h_rand) * TARGET_SUM)
    
    # Main optimization loop
    remaining_time = budget_s
    for run_id in range(MAX_RESTARTS):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Try several different initial conditions
        h_init = random.choice(initial_conditions)
        
        # Greedy improvement
        h_greedy = greedy_improve(h_init, MAX_GREEDY_ITER)
        
        # Simulated Annealing
        h_sa, c5_sa = simulated_annealing(h_greedy, MAX_ANNEALING_ITER, TEMP_START, TEMP_DECAY)
        
        # Continuous refinement with SLSQP
        h_final = continuous_refinement(h_sa)
        c5_final = evaluate_c5(h_final)
        
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final
    
    # Ensure final output is within bounds
    best_h = np.clip(best_h, 0, 1)
    best_h = project_to_sum_box(best_h, TARGET_SUM)
    best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
