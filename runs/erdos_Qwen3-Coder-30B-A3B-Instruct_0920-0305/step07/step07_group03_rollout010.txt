import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient_of_corr_at_lag(h, lag):
    """Compute gradient of correlation w.r.t. h at specific lag"""
    n = len(h)
    g = np.ones(n)
    
    # handle negative lag by symmetry
    if lag < 0:
        lag_abs = -lag
        if lag_abs < n:
            g[lag_abs:] = 1 - h[:n-lag_abs]
            g[:n-lag_abs] -= h[lag_abs:]
    else:
        if lag < n:
            g[:n-lag] = 1 - h[lag:]
            g[lag:] -= h[:n-lag]
            
    return g

def compute_subgradient(h, corr):
    """Compute subgradient for the max correlation"""
    max_val = np.max(corr)
    indices = np.where(np.abs(corr - max_val) < 1e-10)[0]
    
    grad_sum = np.zeros(len(h))
    for idx in indices:
        # need to map correlation index back to lag
        lag = idx - (len(h) - 1)
        grad_sum += compute_gradient_of_corr_at_lag(h, lag)
    
    if len(indices) > 0:
        return grad_sum / len(indices)
    else:
        return np.zeros(len(h))

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    TARGET_SUM = n_points / 2
    best_h = None
    best_c5 = float('inf')
    
    # Randomized binary search with simulated annealing on binary solutions
    n_restart = 10
    time_per_restart = budget_s * 0.6 / n_restart
    
    for restart in range(n_restart):
        if time.time() > start_time + budget_s - 10:
            break
            
        # Generate random binary initial solution
        h = np.zeros(n_points)
        ones_indices = np.random.choice(n_points, int(TARGET_SUM), replace=False)
        h[ones_indices] = 1.0
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Simulated annealing on binary pattern
        current_c5 = evaluate_c5(h)
        temp = 0.5
        temp_min = 1e-6
        cooling_rate = 0.999
        max_iter = int(time_per_restart * 1000)
        iter_count = 0
        
        while temp > temp_min and iter_count < max_iter and time.time() < start_time + budget_s - 10:
            # Try a bit flip
            old_h = h.copy()
            idx_1 = np.random.choice(np.where(h > 0.5)[0]) if len(np.where(h > 0.5)[0]) > 0 else 0
            idx_0 = np.random.choice(np.where(h < 0.5)[0]) if len(np.where(h < 0.5)[0]) > 0 else 0
            
            if idx_1 != idx_0:
                h[idx_1] = 0
                h[idx_0] = 1
                h = project_to_sum_box(h, TARGET_SUM)
                
                new_c5 = evaluate_c5(h)
                delta = new_c5 - current_c5
                
                if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                    current_c5 = new_c5
                else:
                    h = old_h
                
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                    
                temp *= cooling_rate
                iter_count += 1
    
    # Refine with continuous optimization
    if best_h is None:
        # fallback to random initialization with projection
        h_init = np.random.rand(n_points)
        h_init = project_to_sum_box(h_init, TARGET_SUM)
        best_h = h_init.copy()
        best_c5 = evaluate_c5(best_h)
    
    # Gradient descent refinement with subgradient method
    current_h = best_h.copy()
    current_c5 = evaluate_c5(current_h)
    max_iter = 10000
    step = 0.05
    decay_factor = 0.999
    
    for i in range(max_iter):
        if time.time() > start_time + budget_s - 10:
            break
            
        corr = compute_corr(current_h)
        subgrad = compute_subgradient(current_h, corr)
        
        # Update with gradient descent step
        new_h = current_h - step * subgrad
        new_h = project_to_sum_box(new_h, TARGET_SUM)
        
        new_c5 = evaluate_c5(new_h)
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            step = min(0.1, step * 1.01)  # increase step size if improving
        else:
            step *= decay_factor  # decrease step size if not improving
            if step < 1e-6:
                break
                
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    # Final evaluation to ensure correctness
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
