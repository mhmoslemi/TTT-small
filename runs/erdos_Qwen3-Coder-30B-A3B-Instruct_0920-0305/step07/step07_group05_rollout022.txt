import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global constants
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmin_weights(s, alpha):
    """Return normalized soft-min weights for given s and alpha"""
    exp_s = np.exp(-alpha * s)
    exp_s = exp_s / np.sum(exp_s)
    return exp_s

def weighted_gradient(h, alpha):
    """Compute gradient of softmin with respect to h"""
    s = correlate(h, h, mode='full')
    w = softmin_weights(s, alpha)
    
    grad = np.zeros_like(h)
    n = len(h)
    
    # For each lag, add weighted contribution from shifted h
    for i in range(len(s)):
        lag = i - (n - 1)
        if lag >= 0:
            # h shifted by +lag
            if i + lag < len(s):
                grad[:n-lag] += w[i] * h[lag:]
            # h shifted by -lag (equivalent to +(-lag))
            if i >= lag:
                grad[lag:] += w[i] * h[:n-lag]
        else:
            lag_abs = -lag
            # h shifted by -lag
            if i >= lag_abs:
                grad[lag_abs:] += w[i] * h[:n-lag_abs]
            # h shifted by +(-lag) = -lag
            if i + lag_abs < len(s):
                grad[:lag_abs] += w[i] * h[lag_abs:]
    
    return grad

def pairwise_move(h, grad, delta, rng):
    """Perform pairwise mass transfer move based on gradient"""
    # Find indices with most extreme gradient values
    neg_grad_idx = np.argmin(grad)
    pos_grad_idx = np.argmax(grad)
    
    # Ensure indices are valid and have room to move
    if h[neg_grad_idx] <= 0 or h[pos_grad_idx] >= 1:
        # fallback to random move
        i = rng.randint(0, len(h) - 1)
        j = rng.randint(0, len(h) - 1)
        if i != j and h[i] > 0 and h[j] < 1:
            delta = min(delta, h[i], 1 - h[j])
            h[i] -= delta
            h[j] += delta
            return True
        else:
            return False
    
    # Perform the transfer
    delta = min(delta, h[neg_grad_idx], 1 - h[pos_grad_idx])
    if delta <= 0:
        return False
    
    h[neg_grad_idx] -= delta
    h[pos_grad_idx] += delta
    
    # Clip to maintain constraint
    h[neg_grad_idx] = max(0, h[neg_grad_idx])
    h[pos_grad_idx] = min(1, h[pos_grad_idx])
    
    return True

def annealed_gradient_ascent(h0, time_budget, alpha0=10.0, delta0=0.2, T0=5e-3, rng=None):
    """Perform annealed gradient ascent with pairwise moves"""
    if rng is None:
        rng = random.Random(42)
    
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Annealing schedules
    alpha = alpha0
    delta = delta0
    T = T0
    alpha_factor = 1.05
    delta_decay = 0.9995
    T_decay = 0.9995
    last_improvement = 0
    iter_count = 0
    
    start_time = time.time()
    
    while time.time() - start_time < time_budget:
        iter_count += 1
        
        # Compute gradient of softmin
        grad = weighted_gradient(h, alpha)
        
        # Perform pairwise move
        accepted = pairwise_move(h, grad, delta, rng)
        
        # Evaluate new c5
        new_c5 = evaluate_c5(h)
        
        # Accept with Metropolis rule if not accepted
        if not accepted and new_c5 > best_c5 and T > 0:
            if rng.random() < np.exp(-(new_c5 - best_c5) / T):
                accepted = True
                # Keep the change
                pass # already updated in move
        
        if accepted:
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                last_improvement = 0
            else:
                last_improvement += 1
        
        # Anneal hyperparameters
        if iter_count % 200 == 0:
            alpha = min(alpha * alpha_factor, 200.0)
            
        if iter_count % 50 == 0:
            delta *= delta_decay
            if iter_count % 1000 == 0:
                T *= T_decay
                
        # Early termination if no improvement for a long time
        if last_improvement > 5000:
            break
    
    return best_h, best_c5

def random_binary_start(n, rng):
    """Create a random binary start vector with sum = n/2"""
    h = np.zeros(n)
    ones = rng.sample(range(n), n // 2)
    h[ones] = 1.0
    return h

def local_optimize_restart(start_h, remaining_time, rng):
    """Run a short optimization with a given start vector"""
    return annealed_gradient_ascent(start_h, remaining_time, rng=rng)
