import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * 2.0 / len(h)

def project_sum_box(h, target_sum):
    """Project vector h onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(h+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(h + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search with 50 iterations
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def smooth_gradient(h, corr, beta):
    """Compute gradient of smooth maximum surrogate"""
    corr_shifted = corr.copy()
    max_corr = np.max(corr_shifted)
    corr_shifted -= max_corr  # Prevent overflow
    exp_corr = np.exp(beta * corr_shifted)
    weights = exp_corr / np.sum(exp_corr)
    grad = np.zeros_like(h)
    n = len(h)
    for i in range(n):
        for j in range(n):
            lag = j - i
            if -n+1 <= lag <= n-1:
                weights_idx = lag + n - 1
                if 0 <= weights_idx < len(weights):
                    grad[i] += weights[weights_idx] * (1 - h[j] if j < n else 0)
    
    return -grad * 2.0 / n

def max_lag_gradient(h, corr):
    """Compute exact sub-gradient of max over shifts, only for the worst lag"""
    max_idx = np.argmax(corr)
    max_lag = max_idx - len(h) + 1
    grad = np.zeros_like(h)
    n = len(h)
    if max_lag >= 0:
        for i in range(n - max_lag):
            grad[i] += (1 - h[i + max_lag]) * 2.0 / n
        for i in range(max_lag, n):
            grad[i] -= h[i - max_lag] * 2.0 / n
    else:
        for i in range(-max_lag, n):
            grad[i] += (1 - h[i + max_lag]) * 2.0 / n
        for i in range(n + max_lag):
            grad[i] -= h[i - max_lag] * 2.0 / n
    return grad

def mass_transfer(h, grad, epsilon):
    """Perform greedy mass transfer moves"""
    d = -grad
    sources = np.where(d < 0)[0]
    sinks = np.where(d > 0)[0]
    
    if len(sources) == 0 or len(sinks) == 0:
        return h
    
    # Sort by descending importance
    sources = sources[np.argsort(-d[sources])]
    sinks = sinks[np.argsort(-d[sinks])]
    
    h_new = h.copy()
    for s in sources:
        if np.sum(h_new) < epsilon:
            break
        remaining = h_new[s]
        for t in sinks:
            if remaining <= epsilon or h_new[t] >= 1 - epsilon:
                continue
            transfer = min(remaining, 1.0 - h_new[t], epsilon)
            if transfer > epsilon:
                h_new[s] -= transfer
                h_new[t] += transfer
                remaining -= transfer
    
    return project_sum_box(h_new, len(h) / 2)

def backtracking_line_search(h, grad, c5_val, alpha0=0.2, alpha_min=5e-4):
    """Perform backtracking line search for gradient descent"""
    alpha = alpha0
    while alpha >= alpha_min:
        h_new = h - alpha * grad
        h_new = project_sum_box(h_new, len(h) / 2)
        c5_new = evaluate_c5(h_new)
        if c5_new < c5_val - 1e-12:
            return h_new, c5_new, alpha
        alpha /= 2
    return h, c5_val, alpha
