import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def adaptive_pattern_search(h0, max_iter=500):
    """Adaptive pattern-based search focusing on worst regions"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    # Find current worst overlaps in different parts of the domain
    corr_full = compute_corr_fft(h)
    # Analyze worst indices and focus search there
    worst_indices = np.argsort(-corr_full)[:20]  # Top 20 worst overlaps
    
    for i in range(max_iter):
        if i % 50 == 0:
            corr_full = compute_corr_fft(h)
        
        # Try to modify with focus on worst overlap regions
        for _ in range(5):
            # Choose a random worst overlap index
            idx = np.random.choice(worst_indices)
            # Create a window around that index
            start = max(0, idx - 15)
            end = min(n_points, idx + 15)
            
            # Make a targeted modification
            direction = np.random.choice([-1, 1])
            delta = 0.01 * direction
            
            h_new = h.copy()
            h_new[start:end] += delta
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5:
                h = h_new.copy()
                best_c5 = new_c5
                corr_full = compute_corr_fft(h)
    
    return h

def genetic_local_search(h0, population_size=10, generations=50):
    """Simple genetic algorithm approach with local refinement"""
    population = []
    fitness = []
    
    # Create initial population
    for _ in range(population_size):
        h_perturb = h0 + np.random.normal(0, 0.01, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = project_to_sum_box(h_perturb, TARGET_SUM)
        population.append(h_perturb)
        fitness.append(evaluate_c5(h_perturb))
    
    best_h = population[np.argmin(fitness)]
    best_c5 = min(fitness)
    
    for gen in range(generations):
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        top_individuals = [population[i] for i in sorted_indices[:population_size//2]]
        top_fitness = [fitness[i] for i in sorted_indices[:population_size//2]]
        
        # Create new population
        new_population = top_individuals.copy()
        new_fitness = top_fitness.copy()
        
        # Crossover and mutation
        for _ in range(population_size - population_size//2):
            parent1, parent2 = np.random.choice(top_individuals, 2, replace=False)
            
            # Simple crossover
            crossover_point = np.random.randint(0, n_points)
            h_child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            mutate_indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
            for idx in mutate_indices:
                h_child[idx] += np.random.normal(0, 0.01)
            
            h_child = np.clip(h_child, 0, 1)
            h_child = project_to_sum_box(h_child, TARGET_SUM)
            
            new_population.append(h_child)
            new_fitness.append(evaluate_c5(h_child))
        
        population = new_population
        fitness = new_fitness
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
    
    return best_h

    # Phase 1: Refinement with pattern search
    phase1_end = start_time + 0.2 * budget_s

        c5_refined = evaluate_c5(h_refined)
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    # Phase 2: Genetic search hybrid
    phase2_end = start_time + 0.4 * budget_s
    for run_id in range(10):
        if time.time() > phase2_end:
            break
            
        h_genetic = genetic_local_search(best_h.copy(), population_size=8, generations=40)
        c5_genetic = evaluate_c5(h_genetic)
        
        if c5_genetic < best_c5:
            best_h = h_genetic.copy()
            best_c5 = c5_genetic
    
    # Phase 3: Final hybrid refinement
    phase3_end = start_time + budget_s
    while time.time() < phase3_end:
        # Create several candidates with different strategies
        candidates = []
        
        # Strategy 1: Slight random perturbation
        h_noise = best_h + np.random.normal(0, 0.005, n_points)
        h_noise = project_to_sum_box(h_noise, TARGET_SUM)
        candidates.append(h_noise)
        
        # Strategy 2: Pattern search
        h_pattern = adaptive_pattern_search(best_h.copy(), max_iter=500)
        candidates.append(h_pattern)
        
        # Evaluate all candidates
        for h_candidate in candidates:
            c5_candidate = evaluate_c5(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    
    return (best_h.tolist(), float(best_c5), n_points)
