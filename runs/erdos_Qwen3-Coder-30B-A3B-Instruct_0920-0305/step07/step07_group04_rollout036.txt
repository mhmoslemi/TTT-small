import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def direct_optimization_approach(h0):
    """A more robust direct optimization approach"""
    h = h0.copy()
    
    # First, try a coarse optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        # Try multiple optimization algorithms
        methods_to_try = ['SLSQP', 'L-BFGS-B', 'TNC']
        for method in methods_to_try:
            try:
                res = minimize(
                    objective, 
                    h, 
                    method=method, 
                    bounds=bounds, 
                    constraints=cons,
                    options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
                )
                if res.success:
                    h = res.x
                    break
            except:
                continue
        
        # Apply final projection to ensure sum constraint
        h = project_to_sum_box(h, TARGET_SUM)
        
    except Exception as e:
        pass
        
    return h

def improved_local_search(h0, max_iter=1000):
    """Enhanced local search with multiple perturbation strategies"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Iteratively improve using different local search strategies
    for i in range(max_iter):
        # Strategy 1: Random small changes
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.03, 0.03)
        h_new[idx] += delta
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            h = h_new.copy()
        
        # Strategy 2: Block-wise changes
        if np.random.rand() < 0.3:
            block_size = max(1, n_points // 20)

            h_new = h.copy()
            h_new[start_idx:end_idx] += delta
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                h = h_new.copy()
    
    return best_h
