import numpy as np
from scipy.signal import correlate
import time
import math

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_gradient(h, corr, beta):
    """Compute the gradient of the smooth max surrogate"""
    # Compute softmax weights: exp(beta*corr) / sum(exp(beta*corr))
    max_corr = np.max(corr)
    exp_corr = np.exp(beta * (corr - max_corr))
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute the gradient: dx * (1 - sum of weighted shifts)
    grad = np.zeros_like(h)
    n = len(h)
    
    # sum over all lags
    for lag in range(-n+1, n):
        if lag >= 0:
            # h[i] * (1 - h[i+lag]) contributes to grad[i]
            for i in range(n - lag):
                grad[i] += weights[lag + n - 1] * (-h[i + lag])
        else:
            # h[i] * (1 - h[i+lag]) contributes to grad[i]
            for i in range(-lag, n):
                grad[i] += weights[lag + n - 1] * (-h[i + lag])
                
    # Add the 1-dx term (weighted sum over identity)
    grad += dx * np.sum(weights)
    return grad

def max_lag_gradient(h, corr):
    """Compute the exact sub-gradient for the true max-over-shifts"""
    # Find the worst lag
    k_best = np.argmax(corr)
    grad = np.zeros_like(h)
    
    # Apply gradient based on the worst lag
    if k_best >= 0:
        # grad[i] += (1 - h[i + k_best]) * dx if i + k_best < n
        # grad[i] -= h[i - k_best] * dx if i >= k_best
        for i in range(len(h)):
            if i + k_best < len(h):
                grad[i] += (1 - h[i + k_best]) * dx
            if i >= k_best:
                grad[i] -= h[i - k_best] * dx
    else:
        k_best = -k_best
        # grad[i] += (1 - h[i + k_best]) * dx if i >= k_best
        # grad[i] -= h[i - k_best] * dx if i + k_best < n
        for i in range(len(h)):
            if i >= k_best:
                grad[i] += (1 - h[i + k_best]) * dx
            if i + k_best < len(h):
                grad[i] -= h[i - k_best] * dx
    return grad

def mass_transfer(h, grad, epsilon=0.01):
    """Perform greedy mass transfer based on gradient sign"""
    d = -grad  # direction we want to move in
    
    # Find sources and sinks
    sources = []
    sinks = []
    for i in range(len(h)):
        if d[i] < 0:  # decreasing h[i] is beneficial
            sources.append((i, -d[i], h[i]))
        elif d[i] > 0:  # increasing h[i] is beneficial
            sinks.append((i, d[i], 1 - h[i]))
            
    # Sort by decreasing benefit
    sources.sort(key=lambda x: x[1], reverse=True)
    sinks.sort(key=lambda x: x[1], reverse=True)
    
    # Perform transfers
    for s_idx, _, rem in sources:
        for t_idx, _, add in sinks:
            if rem <= 1e-12 or add <= 1e-12:
                continue
            transfer = min(rem, add, epsilon)
            if transfer > 1e-12:
                h[s_idx] -= transfer
                h[t_idx] += transfer
                rem -= transfer
                add -= transfer
                if rem <= 1e-12:
                    break
    
    # Re-project to ensure feasibility
    return project_sum_box(h, TARGET_SUM)

def back_tracking_line_search(h, grad, corr_old, beta, alpha0=0.2, alpha_min=5e-4):
    """Backtracking line search for sufficient descent"""
    alpha = alpha0
    h_proj = project_sum_box(h - alpha * grad, TARGET_SUM)
    corr_new = correlate(h_proj, 1 - h_proj, mode='full') * dx
    cmax_new = np.max(corr_new)
    
    while cmax_new >= corr_old and alpha >= alpha_min:
        alpha /= 2
        h_proj = project_sum_box(h - alpha * grad, TARGET_SUM)
        corr_new = correlate(h_proj, 1 - h_proj, mode='full') * dx
        cmax_new = np.max(corr_new)
        
    return h_proj, cmax_new, alpha
