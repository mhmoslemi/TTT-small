import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
from scipy.optimize import minimize

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    n = len(h)
    dx = 2.0 / n
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlation(h):
    n = len(h)
    dx = 2.0 / n
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def solve_relaxed_lp(h0, target_sum, n, T_guess=None):
    """Solve the relaxed LP for the current guess"""
    dx = 2.0 / n
    # Variables
    h = cp.Variable(n)
    p = cp.Variable((n, n), symmetric=True)
    T = cp.Variable(1)
    
    # Constraints
    constraints = []
    constraints.append(cp.sum(h) == target_sum)
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # McCormick envelopes
    for i in range(n):
        for j in range(i, n):
            constraints.append(p[i, j] >= 0)
            constraints.append(p[i, j] >= h[i] + h[j] - 1)
            constraints.append(p[i, j] <= h[i])
            constraints.append(p[i, j] <= h[j])
    
    # Overlap constraints
    if T_guess is not None:
        T.value = T_guess
    else:
        T.value = 0.5
    
    # Define the shifts
    for shift in range(-(n-1), n):
        if shift >= 0:
            # Overlap: sum p[i, i+shift] for i = 0 to n-1-shift
            overlap = cp.sum([p[i, i+shift] for i in range(n-shift)])
        else:
            # Overlap: sum p[i-shift, i] for i = -shift to n-1
            overlap = cp.sum([p[i-shift, i] for i in range(-shift, n)])
        # O_k >= (n/2) - T
        constraints.append(overlap >= target_sum - T)
    
    # Minimize T
    if T_guess is None:
        objective = cp.Minimize(T)
        prob = cp.Problem(objective, constraints)
    else:
        prob = cp.Problem(cp.Minimize(T), constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000, feastol=1e-8, abstol=1e-8)
        return prob.status == cp.OPTIMAL, h.value
    except:
        return False, None

def binary_search_lp(h0, target_sum, n, t_low=0.0, t_high=None):
    """Binary search for the best T using relaxed LP"""
    if t_high is None:
        t_high = target_sum
    
    dx = 2.0 / n
    tol = 1e-4 * dx
    while t_high - t_low > tol:
        t_mid = (t_low + t_high) / 2.0
        feasible, h_val = solve_relaxed_lp(h0, target_sum, n, T_guess=t_mid)
        
        if feasible:
            # Evaluate the true C5
            c5 = evaluate_c5(h_val)
            if c5 <= t_mid * dx:
                # This is good, accept it
                t_high = t_mid
                best_h = h_val
            else:
                # False positive, relax further
                t_low = t_mid
        else:
            # Infeasible, tighten the bound
            t_low = t_mid
    
    return t_high * dx, best_h

def compute_subgradient(h, active_shift):
    """Compute the subgradient for the true overlap with no relaxation"""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        # Gradient from h_i(1 - h_{i+active_shift})
        if i + active_shift >= 0 and i + active_shift < n:
            grad[i] += (1 - h[i + active_shift])
        # Gradient from -h_{i-active_shift}h_i if appropriate
        if i - active_shift >= 0 and i - active_shift < n:
            grad[i] -= h[i - active_shift]
    return grad

def polish_with_subgradients(h0, target_sum, n, max_iter=2000):
    """Improve solution with deterministic subgradient method"""
    h = h0.copy()
    dx = 2.0 / n
    
    for t in range(max_iter):
        # Compute correlation
        corr = compute_correlation(h)
        active_shift = np.argmax(corr)
        
        # Compute gradient
        grad = compute_subgradient(h, active_shift)
        
        # Step size decay
        step_size = 1.0 / (1.0 + t**0.5) # adaptive
        h -= step_size * grad
        
        # Project
        h = project_to_sum_box(h, target_sum)
    
    return h

    num_exchanges = int(exchange_ratio * n)
    
    for _ in range(num_passes):
        # Shuffle exchanges
        pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        np.random.shuffle(pairs)
        
        for i, j in pairs[:num_exchanges]:
            # Compute max safe transfer
            delta_max = min(h[j], 1 - h[i])
            if delta_max == 0:
                continue
            
            # Try a few different transfers
            for eta in [0.1, 0.5, 0.9]:
                delta = eta * delta_max
                h_new = h.copy()
                h_new[i] += delta
                h_new[j] -= delta
                
                # Ensure bounds
                if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
                    continue
                
                h_new = project_to_sum_box(h_new, target_sum)
                c5_new = evaluate_c5(h_new)
                
                if c5_new < evaluate_c5(h):
                    h = h_new
                    break
    
    return h
