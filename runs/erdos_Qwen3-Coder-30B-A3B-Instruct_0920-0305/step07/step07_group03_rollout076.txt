import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_with_gradients(h0, max_iter=1000):
    """Optimize using gradient-based method with projection"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def local_improvement(h, max_iter=500):
    """Perform local search to improve solution"""
    h_current = h.copy()
    best_c5 = evaluate_c5(h_current)
    
    for i in range(max_iter):
        # Pick random indices to modify
        indices = np.random.choice(n_points, size=min(5, n_points//10), replace=False)
        best_improvement = 0
        best_h = h_current.copy()

            old_val = h_current[idx]

                    best_h = h_new.copy()
                    best_c5 = new_c5
                    best_improvement = 1
                    break

def find_peaks_and_valleys(h):
    """Identify peaks and valleys to optimize their placement"""
    smooth_h = np.convolve(h, np.ones(5)/5, mode='same')
    peaks = np.where(np.diff(np.diff(smooth_h)) < -0.01)[0]
    valleys = np.where(np.diff(np.diff(smooth_h)) > 0.01)[0]
    return peaks, valleys

def improve_structure(h):
    """Try to improve solution structure by placing sharper transitions"""
    h_new = h.copy()
    
    # Find effective peak regions 
    peaks, valleys = find_peaks_and_valleys(h_new)
    
    # Make some valleys shallower, some peaks higher
    if len(peaks) > 0:
        # Increase the height of some peaks
        for i in range(min(3, len(peaks))):
            idx = np.random.choice(peaks)
            h_new[idx] = min(1.0, h_new[idx] + 0.1)
    
    if len(valleys) > 0:
        # Flatten some valleys
        for i in range(min(3, len(valleys))):
            idx = np.random.choice(valleys)
            h_new[idx] = max(0.0, h_new[idx] - 0.1)
    
    # Normalize to correct sum
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    return h_new

        initial_solutions.append(h_init)
    
    # Evaluate all initial solutions
    best_c5 = float('inf')
    best_h = None
    
    for h in initial_solutions:
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 2: Simulated Annealing inspired local search
    current_h = best_h.copy()
    current_c5 = best_c5
    
    temp = 1.0
    min_temp = 1e-6
    cooling_rate = 0.9995
    
    remaining_time = budget_s * 0.7
    
    while temp > min_temp and time.time() - start_time < 0.7 * budget_s:
        # Try a random modification
        h_new = current_h.copy()
        idx = np.random.randint(0, n_points)
        h_new[idx] = np.clip(h_new[idx] + np.random.normal(0, 0.03), 0, 1)
        
        # Project to correct sum
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        
        # Accept or reject the change
        if new_c5 < current_c5 or np.random.random() < np.exp(-(new_c5 - current_c5) / temp):
            current_h = h_new
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        
        # Cool the system
        temp *= cooling_rate
        
        if time.time() - start_time > remaining_time + 0.2 * budget_s:
            break

    # Phase 3: Refine with gradient and local search
    refined_h = optimize_with_gradients(best_h.copy(), max_iter=500)
    refined_c5 = evaluate_c5(refined_h)
    
    if refined_c5 < best_c5:
        best_h = refined_h
        best_c5 = refined_c5

    # Final local improvement with structure-aware updates
    final_h = local_improvement(best_h.copy(), max_iter=300)
    final_c5 = evaluate_c5(final_h)
    
    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5

    return (best_h.tolist(), float(best_c5), n_points)
