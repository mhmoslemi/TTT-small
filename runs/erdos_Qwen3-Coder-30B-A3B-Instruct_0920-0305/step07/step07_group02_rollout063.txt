import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_thue_morse(n):
    """Generate Thue-Morse sequence of length n"""
    if n <= 0:
        return np.array([])
    seq = [0]
    for i in range(int(np.ceil(np.log2(n)))):
        seq = seq + [1 - x for x in seq]
    return np.array(seq[:n])

def generate_rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence of length n"""
    if n <= 0:
        return np.array([])
    # Rudin-Shapiro sequence directly
    seq = [0]
    for i in range(int(np.ceil(np.log2(n)))):
        new_seq = []
        for x in seq:
            new_seq.append(x)
            new_seq.append(1-x)
        seq = new_seq
    return np.array(seq[:n])

def compute_shift_mismatch_indices(h, shift):
    """Find indices where h[i] = 1 and h[i+shift] = 0 (for positive shift)"""
    if shift >= 0:
        # Compare h[:n-shift] with h[shift:]
        h1 = h[:n_points-shift]
        h2 = h[shift:]
        mismatches = np.where((h1 == 1) & (h2 == 0))[0]
    else:
        # Compare h[-shift:] with h[:-shift]
        h1 = h[-shift:]
        h2 = h[:-shift]
        mismatches = np.where((h1 == 1) & (h2 == 0))[0]
    return mismatches

def greedy_worst_shift_improvement(h, max_iter=2000):
    """Improves solution by greedily swapping mismatches for the current worst shift"""
    h = h.copy()
    
    for iteration in range(max_iter):
        # Compute correlation
        corr = compute_corr_fft(h)
        max_shift_idx = np.argmax(corr)
        best_shift = max_shift_idx - (n_points - 1)
        
        # If no improvement is possible, stop
        if corr[max_shift_idx] < 1e-10:
            break
            
        # Find mismatched pairs for this shift
        mismatches1 = compute_shift_mismatch_indices(h, best_shift)
        mismatches2 = compute_shift_mismatch_indices(1 - h, best_shift)
        
        # Create a list of potential swaps
        potential_swaps = []
        for i in mismatches1:
            j = i + best_shift
            # Make sure it's valid 
            if j >= 0 and j < n_points:
                potential_swaps.append((i, j))
        
        # Try to find beneficial swaps
        if potential_swaps:
            # Pick a random beneficial swap if exists; if none, still continue
            best_swap = None
            best_reduction = 0
            
            for i, j in potential_swaps:
                # Swapping h[i] with h[j] reduces overlap for shift k
                # Since h[i]=1 and h[i+k]=0, and h[j]=0, h[j+k]=1, swapping changes
                # to h[i]=0, h[i+k]=1, h[j]=1, h[j+k]=0, reducing overlap by 1 for this shift
                # Check how much this impacts overall C5 - we'll be pessimistic
                temp_h = h.copy()
                temp_h[i], temp_h[j] = temp_h[j], temp_h[i]  # Swap
                new_corr = compute_corr_fft(temp_h)
                new_c5 = c5_from_corr(new_corr, n_points)
                old_c5 = c5_from_corr(corr, n_points)
                
                # We want to minimize C5, so if it goes down we take it
                if new_c5 < old_c5 - 1e-12:
                    reduction = old_c5 - new_c5
                    if reduction > best_reduction:
                        best_reduction = reduction
                        best_swap = (i, j)
            
            if best_swap:
                i, j = best_swap
                h[i], h[j] = h[j], h[i]
            else:
                break  # No beneficial swaps found
        else:
            break  # No mismatches found
    
    return h

def simulated_annealing(h0, initial_temp=0.01, cooling_rate=0.98, max_iter=500):
    """Run Simulated Annealing from a starting solution"""
    h_current = h0.copy()
    current_c5 = evaluate_c5(h_current)
    
    temp = initial_temp
    for i in range(max_iter):
        # Make a small perturbation
        idx1 = random.randint(0, n_points - 1)
        idx2 = random.randint(0, n_points - 1)
        
        h_new = h_current.copy()
        h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
        h_new = project_to_sum_box(h_new, TARGET_SUM)  # Ensure sum constraint maintained
        
        new_c5 = evaluate_c5(h_new)
        
        # Accept based on Metropolis criterion
        if new_c5 < current_c5 or random.random() < np.exp((current_c5 - new_c5) / temp):
            h_current = h_new
            current_c5 = new_c5
        
        temp *= cooling_rate  # Cool down
    
    return h_current

def continuous_refinement(h0, max_iter=300):
    """Refine continuous solution using SLSQP with bounds and sum constraint"""
    def objective(h_local):
        return evaluate_c5(h_local)
    
    def constraint_eq(h_local):
        return np.sum(h_local) - TARGET_SUM
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            return res.x
    except:
        pass
    
    return h0

def heuristic_initialization():
    """Create a heuristic initial solution"""
    # Generate several candidate sequences
    candidates = []
    
    # 1. Thue-Morse sequence
    tm = generate_thue_morse(n_points)
    candidates.append(project_to_sum_box(tm.astype(float), TARGET_SUM))
    
    # 2. Rudin-Shapiro
    rs = generate_rudin_shapiro(n_points)
    candidates.append(project_to_sum_box(rs.astype(float), TARGET_SUM))
    
    # 3. Random binary with correct sum
    rand = np.random.random(n_points)
    rand = project_to_sum_box(rand, TARGET_SUM)
    candidates.append(rand)
    
    # 4. Uniform + small random perturbations
    uniform = np.ones(n_points) * 0.5
    uniform = project_to_sum_box(uniform, TARGET_SUM)
    candidates.append(uniform)
    
    # Evaluate all and return the best
    best_h = None
    best_c5 = float('inf')
    
    for h_candidate in candidates:
        c5_candidate = evaluate_c5(h_candidate)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate
    
    return best_h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    best_h = heuristic_initialization()
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: Multiple random restarts with greedy improvement
    restarts = 300  # Large enough to reasonably explore climate space
    restart_interval = budget_s / restarts
    
    for i in range(restarts):
        if time.time() - start_time > budget_s:
            break
            
        # Start with an initial sequence again, not just random
        h0 = heuristic_initialization()
        
        # Apply greedy worst-shift improvement
        improved_h = greedy_worst_shift_improvement(h0)
        c5_improved = evaluate_c5(improved_h)
        
        if c5_improved < best_c5:
            best_h = improved_h
            best_c5 = c5_improved
    
    # Phase 2: Fine tuning with simulated annealing (if time permits)
    if time.time() - start_time < budget_s * 0.9:
        sa_h = simulated_annealing(best_h.copy(), initial_temp=0.01, cooling_rate=0.99, max_iter=200)
        sa_c5 = evaluate_c5(sa_h)
        if sa_c5 < best_c5:
            best_h = sa_h
            best_c5 = sa_c5
    
    # Phase 3: Output
    return (best_h.tolist(), float(best_c5), n_points)
