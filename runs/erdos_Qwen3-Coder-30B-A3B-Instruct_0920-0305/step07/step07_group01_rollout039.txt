import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr)

def fourier_basis(x, M):
    """Generate Fourier basis functions: 1, cos(πx), cos(2πx), ..., cos(Mπx)"""
    basis = np.ones((len(x), M+1))
    for m in range(1, M+1):
        basis[:, m] = np.cos(m * np.pi * x)
    return basis

def construct_h_from_coeffs(coeffs, x, M):
    """Construct h(x) from Fourier coefficients"""
    basis = fourier_basis(x, M)
    h = np.dot(basis, coeffs)
    return np.clip(h, 0, 1)

    """Objective function with penalty for violating box constraints"""
    h = construct_h_from_coeffs(coeffs, x, M)

    # Add penalty for constraint violations
    h_min = np.min(h)
    h_max = np.max(h)

        penalty += penalty_weight * (0 - h_min)**2
    if h_max > 1:
        penalty += penalty_weight * (h_max - 1)**2
    
    return c5 + penalty
