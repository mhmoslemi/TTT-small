import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global constants
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmin_weights(s, alpha):
    """Compute normalized soft-min weights"""
    # Subtract max for numerical stability
    s_shifted = s - np.max(s)
    weights = np.exp(-alpha * s_shifted)
    return weights / np.sum(weights)

def weighted_gradient(h, alpha):
    """Compute gradient of softmin over autocorrelations"""
    s = correlate(h, h, mode='full')
    w = softmin_weights(s, alpha)
    grad = np.zeros_like(h)
    
    # Use full correlation length
    n = len(h)
    for i in range(2 * n - 1):
        lag = i - (n - 1)
        # Only consider weights where w[i] > 0
        if abs(w[i]) > 1e-15:
            if lag >= 0:
                # h[i] * h[i+lag] contributes to gradient
                # Gradient w.r.t. h[i] is h[i+lag]
                grad[:n-lag] += w[i] * h[lag:]
                # Gradient w.r.t. h[i+lag] is h[i]
                grad[lag:] += w[i] * h[:n-lag]
            else:
                # lag < 0
                # h[i] * h[i+lag] contributes to gradient
                # Gradient w.r.t. h[i] is h[i+lag]
                grad[-lag:] += w[i] * h[:n+lag]
                # Gradient w.r.t. h[i+lag] is h[i]
                grad[:n+lag] += w[i] * h[-lag:]
    
    return grad

def pairwise_move(h, grad, delta_max, rng):
    """Transfer mass from min grad to max grad indices"""
    # Find source (min gradient) and target (max gradient)
    src_idx = np.argmin(grad)
    tgt_idx = np.argmax(grad)
    
    h_src = h[src_idx]
    h_tgt = h[tgt_idx]
    
    # If indices have same sign, try to disentangle
    if (grad[src_idx] >= 0) == (grad[tgt_idx] >= 0):
        # Find indices with opposite signs to resolve
        if grad[src_idx] >= 0:
            # Find a negative gradient index to swap
            neg_indices = np.where(grad < 0)[0]
            if len(neg_indices) > 0:
                alt_src = neg_indices[np.argmax(grad[neg_indices])]
                if h[alt_src] > 0:
                    src_idx = alt_src
    
    # Calculate transfer amount
    delta = min(delta_max, h[src_idx], 1 - h[tgt_idx])
    
    if delta <= 1e-10:
        return False, h, 0
    
    # Transfer
    h[src_idx] -= delta
    h[tgt_idx] += delta
    
    # Project onto feasible region (reduces any numerical errors)
    h = project_to_sum_box(h, TARGET_SUM)
    
    return True, h, delta

def annealed_gradient_ascent(h0, time_budget, alpha_0, delta_0, T0, rng):
    """Annealed projected gradient ascent with pairwise moves"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Parameters that decay
    alpha = alpha_0
    delta = delta_0
    T = T0
    last_improvement = 0
    
    time_start = time.time()
    
    # Annealing schedule parameters
    alpha_factor = 1.2
    delta_factor = 0.9995
    T_factor = 0.9995
    max_iter_no_improvement = 20000
    
    iter_num = 0
    while time.time() - time_start < time_budget:
        iter_num += 1
        
        # Periodic updates
        if iter_num % 1000 == 0:
            if alpha < 200.0:
                alpha *= alpha_factor
            delta *= delta_factor
            T *= T_factor
        
        # Compute the gradient
        grad = weighted_gradient(h, alpha)
        
        # Perform a pairwise move
        accepted, h_new, _ = pairwise_move(h.copy(), grad, delta, rng)
        
        if accepted:
            # Compute new C5
            c5_new = evaluate_c5(h_new)
            
            # Accept or not
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                last_improvement = iter_num
            elif np.random.random() < np.exp(-(c5_new - best_c5) / T):
                h = h_new.copy()
                
        else:
            # Try small random move for exploration
            i = rng.randint(0, n_points - 1)
            j = rng.randint(0, n_points - 1)
            if i != j:
                delta_move = min(h[i], 1 - h[j], 0.001)
                if delta_move > 1e-6:
                    h[i] -= delta_move
                    h[j] += delta_move
                    h = project_to_sum_box(h, TARGET_SUM)
        
        # Check for early exit if no improvement for many iterations
        if iter_num - last_improvement > max_iter_no_improvement:
            break
            
    return best_h, best_c5

def random_binary_start(n, rng):
    """Generate a random binary start vector"""
    h = np.zeros(n)
    ones_indices = rng.sample(range(n), n // 2)
    h[ones_indices] = 1.0
    return h

def local_optimize_restart(start_h, remaining_time, rng):

    return annealed_gradient_ascent(

        min(remaining_time, 20.0),

        delta_0=0.5, 
        T0=5e-3, 
        rng=rng
    )
