import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize

# Global parameters
n_points = 150
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_lp_relaxation(h_init, T_guess):
    """Solve the LP relaxation of the Erdos problem with fixed T"""
    n = n_points
    # Variables
    h = cp.Variable(n, bounds=(0, 1))
    p = cp.Variable((n, n), symmetric=True, bounds=(0, 1))
    
    # Mass constraint
    mass_con = cp.sum(h) == TARGET_SUM
    
    # McCormick envelopes
    McCormick_con = []
    for i in range(n):
        for j in range(i, n):
            # p_ij >= h_i + h_j - 1
            McCormick_con.append(p[i, j] >= h[i] + h[j] - 1)
            # p_ij <= h_i
            McCormick_con.append(p[i, j] <= h[i])
            # p_ij <= h_j
            McCormick_con.append(p[i, j] <= h[j])
    
    # Shift constraints
    shift_con = []
    for k in range(-n+1, n):
        if k >= 0:
            O_k = cp.sum([p[i, i+k] for i in range(n-k)])
        else:  # k < 0
            O_k = cp.sum([p[i-k, i] for i in range(-k, n)])
        # O_k >= n/2 - T
        shift_con.append(O_k >= TARGET_SUM - T_guess)
    
    # Formulate and solve the LP
    objective = cp.Minimize(0)  # dummy objective
    prob = cp.Problem(objective, [mass_con] + McCormick_con + shift_con)
    
    try:
        prob.solve(solver=cp.GLPK, verbose=False, max_iters=1000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_OUT]:
            return h.value
        else:
            return None
    except:
        return None

def binary_search_and_verify():
    """Binary search on the objective using LP relaxation"""
    low, high = 0.0, TARGET_SUM
    tolerance = 1e-4 * dx
    best_h = None
    best_c5 = float('inf')

    while high - low > tolerance:
        mid = (low + high) / 2
        h_sol = solve_lp_relaxation(None, mid)
        if h_sol is not None:
            h_sol = project_to_sum_box(h_sol, TARGET_SUM)
            actual_c5 = evaluate_c5(h_sol)
            if actual_c5 <= mid * dx:
                best_h = h_sol
                best_c5 = actual_c5
                high = mid
            else:
                low = mid
        else:
            low = mid

    return best_h, best_c5

def subgradient_polish(h0, max_iter=2000):
    """Deterministic polishing via subgradient descent on true objective"""
    h = h0.copy()
    
    for t in range(max_iter):
        # Compute correlation
        corr = compute_corr_fft(h)
        k_active = int(np.argmax(corr))
        
        # Compute subgradient
        g = np.zeros_like(h)
        for i in range(len(h)):
            # Term: h[i] * (1 - h[i + k_active])
            if i + k_active < len(h):
                g[i] += (1 - h[i + k_active])
            # Term: h[i - k_active] * (1 - h[i])
            if i - k_active >= 0:
                g[i] -= h[i - k_active]
        
        # Step size
        alpha = 1e-4 / np.sqrt(t + 1)
        
        # Update
        h = h - alpha * g
        
        # Project
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Early stopping
        if t > 50 and t % 50 == 0:
            if evaluate_c5(h) > 1.01 * np.min([evaluate_c5(h0), best_c5]):
                break
    
    return h

def exchange_refine(h0, max_iterations=1000):
    """Exchange refinement to improve the solution"""
    h = h0.copy()
    n = len(h)
    
    for _ in range(max_iterations):
        # Take random shuffle of indices
        indices = np.random.permutation(n)
        improved = False
        
        for i in indices:
            # Try to move some mass from a neighbor
            if i >= 1:
                j = i - 1
                delta_max = min(h[j], 1 - h[i])
                if delta_max > 0:
                    try_deltas = [0.1, 0.5, 0.9]
                    for delta in try_deltas:
                        delta_move = delta * delta_max
                        if delta_move < 1e-8:
                            continue
                        h_new = h.copy()
                        h_new[i] += delta_move
                        h_new[j] -= delta_move
                        h_new = project_to_sum_box(h_new, TARGET_SUM)
                        if evaluate_c5(h_new) < evaluate_c5(h):
                            h = h_new
                            improved = True
                            break
            if improved:
                break
    
    return h
