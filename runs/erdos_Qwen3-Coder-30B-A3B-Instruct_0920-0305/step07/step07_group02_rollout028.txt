import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evolutionary_search(h0, max_iter=1000):
    """Evolutionary approach with mutation and selection"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    population = [h]
    
    for i in range(max_iter):
        if i % 50 == 0 and len(population) < 20:
            # Add some diversity
            for _ in range(3):
                h_new = h + np.random.normal(0, 0.01, n_points)
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                population.append(h_new.copy())
        
        # Select parent
        parent_idx = np.random.randint(len(population))
        parent = population[parent_idx]
        
        # Create offspring through mutation
        offspring = parent.copy()
        # Mutate randomly
        for j in range(n_points // 10):
            idx = np.random.randint(n_points)
            offspring[idx] += np.random.normal(0, 0.01)
        
        offspring = np.clip(offspring, 0, 1)
        offspring = project_to_sum_box(offspring, TARGET_SUM)
        
        # Evaluate offspring
        c5_offspring = evaluate_c5(offspring)
        
        if c5_offspring < best_c5:
            h = offspring.copy()
            best_c5 = c5_offspring
            # Add to population
            if len(population) < 20:
                population.append(offspring.copy())
            else:
                # Replace worst performing individual
                worst_idx = np.argmax([evaluate_c5(p) for p in population])
                population[worst_idx] = offspring.copy()
    
    return h, best_c5

def multi_start_optimization(h0, max_iter=2000):
    """Run multiple optimizations from different starting points"""
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Try several random starting points with varied pattern
    for i in range(10):
        h_start = h0.copy()
        
        if i > 0:
            # Vary the initial values slightly
            h_start *= (0.9 + 0.2 * np.random.rand())
            h_start = np.clip(h_start, 0, 1)
            h_start = project_to_sum_box(h_start, TARGET_SUM)
        
        # Try evolutionary search
        h_final, c5_final = evolutionary_search(h_start, max_iter // 10)
        
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final
    
    return best_h, best_c5

        def objective(h_local):
            return evaluate_c5(h_local)
        
        def constraint_eq(h_local):
            return np.sum(h_local) - TARGET_SUM
        
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective,

            pass

        c5_local = evaluate_c5(h_local)
        
        if c5_local < best_c5:
            best_h = h_local.copy()
            best_c5 = c5_local
    
    return (best_h.tolist(), float(best_c5), n_points)
