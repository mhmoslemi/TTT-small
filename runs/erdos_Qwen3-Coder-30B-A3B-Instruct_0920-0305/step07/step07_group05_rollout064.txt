import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import SCS, ECOS
from scipy.linalg import eigh

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def build_correlation_matrices(n):
    """Build the sparse matrices A_ell for each lag"""
    A_list = []
    for ell in range(-n+1, n):
        # index arrays for a sparse matrix
        rows = []
        cols = []
        for i in range(n):
            j = i + ell
            if 0 <= j < n:
                rows.append(i)
                cols.append(j)
        if len(rows) > 0:
            A = np.zeros((n, n))
            A[rows, cols] = 1.0
            A_list.append((ell, A))
    return A_list

def solve_sdp(n_points, dx, A_list):
    """Use CVXPY to solve the semidefinite program relaxation"""
    # Create variables
    X = cp.Variable((n_points, n_points), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        X >> 0,  # Positive semidefinite
        cp.diag(X) >= 0,
        cp.diag(X) <= 1,
        cp.sum(cp.diag(X)) == n_points / 2
    ]

    # Add constraint: trace(A_ell * X) >= n/2 - t/dx for all ell
    for ell, A_ell in A_list:
        constraints.append(cp.trace(A_ell @ X) >= n_points / 2 - t / dx)

    # Objective
    objective = cp.Minimize(t)

    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=SCS, verbose=False, max_iters=1000)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return X.value, t.value
    except Exception:
        pass
    return None, None

def round_sdp_solution(X_opt, n_points, dx):
    """Round the SDP solution to get a feasible h vector"""
    # Use principal eigenvector rounding
    try:
        eigenvals, eigenvecs = eigh(X_opt)
        v = eigenvecs[:, -1]  # largest eigenvector
        h_raw = v.copy()
        h_clip = np.clip(h_raw, 0, 1)
        h_projected = project_to_sum_box(h_clip, n_points / 2.0)
        return h_projected
    except Exception:
        return None

def local_refinement(h_initial, max_iter=300):
    """Perform local refinement around the current solution"""
    h = h_initial.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for iter_num in range(max_iter):
        # heuristic: move mass from max to min gradient points
        corr = correlate(h, 1 - h, mode='full')
        max_lag = np.argmax(corr)
        grad = np.zeros_like(h)
        eps = 1e-6
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_sum_box(h_plus, len(h)/2)
            c5_plus = evaluate_c5(h_plus)
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_sum_box(h_minus, len(h)/2)
            c5_minus = evaluate_c5(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Move from highest negative gradient to lowest positive:
        idx_min = np.argmin(grad)
        idx_max = np.argmax(grad)
        if grad[idx_min] < 0 and grad[idx_max] > 0:
            delta = min(h[idx_min], 1 - h[idx_max], 0.01)
            if delta > 1e-6:
                h[idx_min] -= delta
                h[idx_max] += delta
                h = project_to_sum_box(h, len(h)/2)
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    n_points = 140
    dx = 2.0 / n_points

    # Build all correlation matrices
    A_list = build_correlation_matrices(n_points)
    
    # Solve SDP (using a reasonable portion of time)
    sdp_end_time = start_time + 0.3 * budget_s
    X_opt, t_opt = None, None
    while time.time() < sdp_end_time and X_opt is None:
        X_opt, t_opt = solve_sdp(n_points, dx, A_list)
    
    if X_opt is None:
        # fallback to a simple random construction
        h = np.random.rand(n_points)
        h = project_to_sum_box(h, n_points / 2)
        c5 = evaluate_c5(h)
        return (h.tolist(), float(c5), n_points)
    
    # Rounding and local refinement
    h_rounded = round_sdp_solution(X_opt, n_points, dx)
    if h_rounded is None:
        # fallback
        h = np.random.rand(n_points)
        h = project_to_sum_box(h, n_points / 2)
        c5 = evaluate_c5(h)
        return (h.tolist(), float(c5), n_points)

    h_refined, c5_refined = local_refinement(h_rounded, 400)
    final_h = h_refined.copy()
    final_c5 = c5_refined

    # Further refine with a smart distance-perturbed search
    refine_end_time = start_time + 0.9 * budget_s
    while time.time() < refine_end_time:
        # small random perturbation
        h_perturbed = final_h + np.random.normal(0, 0.005, n_points)
        h_perturbed = project_to_sum_box(h_perturbed, n_points / 2)
        h_local, c5_local = local_refinement(h_perturbed, 200)
        if c5_local < final_c5:
            final_c5 = c5_local
            final_h = h_local.copy()

    return (final_h.tolist(), float(final_c5), n_points)
