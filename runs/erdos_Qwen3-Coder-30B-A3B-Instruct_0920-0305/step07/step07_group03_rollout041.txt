import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_of_corr_at_lag(h, lag):
    """Compute gradient of correlation at given lag"""
    n = len(h)
    g = np.ones(n)
    
    if lag >= 0:
        # i from 0 to n-lag-1.
        if lag < n:
            g[:n-lag] = 1 - h[lag:]  # for positions i where i+lag in-range
            g[lag:] -= h[:n-lag]     # subtract h_{i-ℓ}
    else:  # lag < 0, let d = -lag
        d = -lag
        # i from d to n-1 have i+lag = i-d in range.
        if d < n:
            g[d:] = 1 - h[:n-d]
            g[:n-d] -= h[d:]
    return g

def subgradient_descent(h0, max_iter=5000, step_size=0.1):
    """Perform subgradient descent on the maximum correlation"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    for i in range(max_iter):
        corr = compute_corr(h)
        max_val = np.max(corr)
        
        # Find indices where max is achieved
        max_indices = np.where(np.abs(corr - max_val) < 1e-10)[0]
        
        # Compute average gradient for all max indices
        grad_sum = np.zeros(len(h))
        for idx in max_indices:
            # Convert index to actual lag
            lag = idx - (len(h) - 1)
            grad_sum += gradient_of_corr_at_lag(h, lag)
        
        gradient = grad_sum / len(max_indices)
        
        # Update step size with diminishing returns
        current_step = step_size / np.sqrt(i + 1)
        
        # Move in negative gradient direction
        h_new = h - current_step * gradient
        
        # Project to constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new
    
    return best_h, best_c5

def simulated_annealing_binary(h0, max_time=200):
    """Perform simulated annealing on binary sequences"""
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    current_h = h0.copy()
    current_c5 = best_c5
    
    # Initialize temperature
    temp = 1.0
    start_time = time.time()
    
    while time.time() - start_time < max_time:
        # Binary swap: choose two indices to swap (one 1, one 0)
        ones = np.where(current_h > 0.5)[0]
        zeros = np.where(current_h < 0.5)[0]
        
        if len(ones) > 0 and len(zeros) > 0:
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Create new candidate by swapping values
            new_h = current_h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]
            
            # Project to maintain sum constraint
            new_h = project_to_sum_box(new_h, TARGET_SUM)
            
            new_c5 = evaluate_c5(new_h)
            
            # Accept with probability based on temperature
            if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / temp):
                current_h = new_h
                current_c5 = new_c5
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
            
            # Cool down slowly
            temp = max(1e-6, temp * 0.9999)
    
    return best_h, best_c5

def initial_binary_guess():
    """Create a reasonable binary initialization"""
    h = np.zeros(n_points)
    # Place ones in a pattern that tries to avoid concentrating them
    ones_positions = np.random.choice(n_points, size=n_points//2, replace=False)
    h[ones_positions] = 1.0
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Phase 1: Multiple random restarts with simulated annealing
    best_h = None
    best_c5 = float('inf')
    
    for restart in range(15):
        if time.time() - start_time > 0.6 * budget_s:
            break
            
        h_initial = initial_binary_guess()
        h_sa, c5_sa = simulated_annealing_binary(h_initial, max_time=40)
        
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
    
    # Phase 2: Refine with gradient descent
    if best_h is not None and time.time() - start_time < 0.9 * budget_s:
        best_h, best_c5 = subgradient_descent(best_h, max_iter=2000)
    
    # Phase 3: Final local refinement
    if time.time() - start_time < budget_s:
        # Small improvement
        h_final = best_h.copy()
        current_c5 = best_c5
        
        for _ in range(1000):
            if time.time() - start_time > budget_s:
                break
                
            # Small perturbation
            perturb_idx = np.random.randint(0, n_points)
            old_val = h_final[perturb_idx]
            if old_val > 0.5:
                new_val = max(0, old_val - 0.02)
            else:
                new_val = min(1, old_val + 0.02)
            
            h_final[perturb_idx] = new_val
            
            # Project to sum
            h_final = project_to_sum_box(h_final, TARGET_SUM)
            
            new_c5 = evaluate_c5(h_final)
            if new_c5 < current_c5:
                current_c5 = new_c5
            else:
                h_final[perturb_idx] = old_val  # revert
                
        best_h = h_final
        best_c5 = current_c5
    
    return (best_h.tolist(), float(best_c5), n_points)
