import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 150
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    # Compute correlation using correlation (not FFT to keep exact fits)
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return dx * max_corr

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def initialize_h_random(n, target_sum):
    """Initialize with random values that sum to target_sum"""
    h = np.random.uniform(0, 1, n)
    h = project_to_sum_box(h, target_sum)
    return h

def gradient_descent(h_start, n_iters, lr, eps, target_sum, verbose=False):
    """Run gradient descent with smoothing of the maximum"""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    if verbose:
        print(f"Starting gradient descent with eps={eps}, lr={lr}")
    
    for iter in range(n_iters):
        # Compute correlation: full convolution
        corr = correlate(h, 1 - h, mode="full")
        
        # Compute smoothed version of the maximum
        # Use softmax approximation with eps smoothing
        m = np.max(corr)
        # Avoid numerical issues by subtracting max
        w_raw = np.exp((corr - m) / eps)
        w = w_raw / w_raw.sum()
        
        # Both convolutions for gradient computation
        # Gradient of sum_s w_s * corr_s = sum_s w_s * gradient of corr_s
        # Where corr_s = sum_i h_i * (1 - h_{i+s})
        # Derivative d corr_s / d h_l = (1 - h_{l+s}) - h_{l-s}
        # So gradient[l] = sum_s w_s * [(1 - h_{l+s}) - h_{l-s}]
        
        # Compute via loops for make sure it's correct
        gradient = np.zeros_like(h)
        L = len(corr)
        offset = len(h) - 1  # corr indices shifted by len(h)-1
        
        for s in range(-(len(h)-1), len(h)):  # valid shift indices
            if s >= 0:
                ws = w[s + offset]
                if abs(ws) < 1e-15:
                    continue
                gradient[:len(h)-s] += ws * (1 - h[s:])
                gradient[s:] -= ws * h[:len(h)-s]
            else:
                t = -s
                ws = w[s + offset]
                if abs(ws) < 1e-15:
                    continue
                gradient[t:] += ws * (1 - h[:len(h)-t])
                gradient[:len(h)-t] -= ws * h[t:]
        
        # Update step
        h_new = h - lr * gradient
        
        # Project onto constraints
        h_new = project_to_sum_box(h_new, target_sum)
        h = h_new
        
        # Check if improvement
        if iter % 10 == 0:
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5

    seg_size = n_points // segments
    for i in range(segments):

        h_piecewise[start:end] = np.random.uniform(0.2, 0.8)

    initial_guesses.append(h_piecewise)
    
    # 4. Tent style shape

    x_points = np.linspace(0, 2, n_points)
    center = 1.0
    width = 0.4
    for i in range(n_points):
        x = x_points[i]
        h_tent[i] = max(0, 1 - abs(x - center) / width)
    h_tent = project_to_sum_box(h_tent, TARGET_SUM)

    for i, h_start in enumerate(initial_guesses):
        if time.time() - start_time > budget_s - 10:
            break

        c5_current = evaluate_c5(h_current)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()

        for eps, num_iter in zip(eps_schedule, iter_schedule):

                    options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10}
                )

                    c5_refined = evaluate_c5(h_refined)
                    if c5_refined < best_c5:
                        best_c5 = c5_refined
                        best_h = h_refined.copy()

                pass

    # Final refinement using same approach as initial code
    if time.time() - start_time < budget_s - 10:
        bounds = [(0, 1) for _ in range(n_points)]
        def objective(h):
            return evaluate_c5(h)
            
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        cons = {'type': 'eq', 'fun': constraint_eq}
        try:
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_final = project_to_sum_box(res.x, TARGET_SUM)
                c5_final = evaluate_c5(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final.copy()
        except:
            pass
                
    return (best_h.tolist(), float(best_c5), n_points)
