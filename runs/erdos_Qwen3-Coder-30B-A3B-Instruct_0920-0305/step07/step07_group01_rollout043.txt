import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * 2.0 / len(h)

def project_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_gradient(corr, beta):
    """Compute gradient of smooth max surrogate"""
    # Subtract max for numerical stability
    corr_shifted = corr - np.max(corr)
    exp_corr = np.exp(beta * corr_shifted)
    weights = exp_corr / np.sum(exp_corr)
    
    grad = np.zeros_like(corr)
    
    # Compute gradient using full correlation result
    full_corr = correlate(corr, 1 - corr, mode='full')
    # Correct the indexing
    n = len(corr)
    # For each i compute weighted average of all shifts
    for i in range(n):
        for k in range(-n+1, n):
            if 0 <= i+k < n:
                grad[i] += weights[k + n - 1] * (1 - corr[i+k])
    
    return 2.0/len(corr) * (1 - grad)

def max_lag_gradient(corr, max_lag_idx):
    """Compute exact sub-gradient of maximum"""
    n = len(corr)
    grad = np.zeros(n)
    # Add gradient for the worst lag
    # Interpret max_lag_idx as the shift in the full correlation
    shift = max_lag_idx - (n - 1)  # convert to shift index
    # Add gradient contribution from the auto-correlation term
    if shift >= 0:
        # Gradient from the (1 - h[i+shift])
        for i in range(n - shift):
            grad[i] += (1 - corr[i + shift]) * 2.0 / n
        # Gradient from the h[i-shift]
        for i in range(shift, n):
            grad[i] -= corr[i - shift] * 2.0 / n
    else:
        # Shift < 0 
        for i in range(-shift, n):
            grad[i] += (1 - corr[i + shift]) * 2.0 / n
        for i in range(n + shift):
            grad[i] -= corr[i - shift] * 2.0 / n

    return grad

def mass_transfer(h, grad, epsilon):
    """Perform a greedy mass transfer move"""
    d = -grad.copy()
    # Identify sources (negative gradient, so decreasing h improves)
    sources = [(i, d[i]) for i in range(len(d)) if d[i] < -1e-10]
    # Identify sinks (positive gradient, so increasing h improves)
    sinks = [(i, d[i]) for i in range(len(d)) if d[i] > 1e-10]
    
    sources.sort(key=lambda x: x[1])  # Sort by decreasing negative values 
    sinks.sort(key=lambda x: x[1], reverse=True)  # Sort by decreasing positive values
    
    h_new = h.copy()
    
    for i, _ in sources:
        if epsilon <= 0:
            break
        
        max_removable = h_new[i]
        transfer_amount = min(max_removable, epsilon)
        if transfer_amount <= 0:
            continue

        # Find the best sink
        j = -1
        for idx, _ in sinks:
            if h_new[idx] < 1:
                j = idx
                break
        if j == -1:
            continue
            
        max_addable = 1 - h_new[j]
        transfer = min(transfer_amount, max_addable)
        if transfer > 0:
            h_new[i] -= transfer
            h_new[j] += transfer
            epsilon -= transfer
    
    return h_new

def backtracking_descent(h, corr, grad, alpha=0.2, alpha_min=1e-4):
    """Backtracking line search for descent step"""
    c0 = evaluate_c5(h)
    while alpha >= alpha_min:
        h_new = h - alpha * grad
        h_new = project_sum_box(h_new, len(h) * 0.5)
        c_new = evaluate_c5(h_new)
        if c_new < c0:
            return h_new, c_new
        alpha *= 0.5
    return h, c0  # fallback to no step
