```python
import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use n=120 as default since it's manageable and was used in previous work
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    # Create CVXPY variables
    h = cp.Variable(n_points, name="h")
    P = cp.Variable((n_points, n_points), name="P")  # Product variables
    
    # Define constraints
    constraints = [
        cp.sum(h) == TARGET_SUM,
        h >= 0,
        h <= 1
    ]
    
    # Add McCormick constraints for all pairs
    for i in range(n_points):
        for j in range(n_points):
            # McCormick envelopes for h[i] * h[j]
            constraints.append(P[i, j] >= 0)
            constraints.append(P[i, j] >= h[i] + h[j] - 1)
            constraints.append(P[i, j] <= h[i])
            constraints.append(P[i, j] <= h[j])
    
    # Variables for max correlation terms
    correlations = []
    t = cp.Variable(name="t")
    
    # Compute correlation for each shift
    for k in range(n_points):
        # Overlap region for shift k: indices where both i and i+k exist
        overlap_indices = list(range(n_points - k))
        if overlap_indices:  # Only add if there's overlap
            # correlation_k = sum_i h_i - sum_{i in overlap} h_i * h_{i+k}
            corr_k = cp.sum(h[overlap_indices]) - cp.sum(P[overlap_indices, np.array(overlap_indices) + k])
            correlations.append(corr_k)
            # t >= correlation_k
            constraints.append(t >= corr_k)
    
    # Objective: minimize maximum correlation
    objective = cp.Minimize(t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # We solve with a timeout to respect budget, but we still want best result
    try:
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=budget_s - 5)
        if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
            h_opt = h.value
            # Reconstruct C5 using actual computations
            correlations_actual = []
            for k in range(n_points):
                overlap_indices = list(range(n_points - k))
                if overlap_indices:
                    corr_k = np.sum(h_opt[overlap_indices]) - np.sum(h_opt[overlap_indices] * h_opt[np.array(overlap_indices) + k])
                    correlations_actual.append(corr_k)
            c5_bound = max(correlations_actual) * dx
        else:
            # Fall back to heuristic approach if LP fails
            # Generate a well-behaved step function
            h_opt = np.zeros(n_points)
            mid = n_points // 2
            # Use an asymmetric pattern: more ones in the center, tapering
            # Generate more ones in middle, reduce at edges
            for i in range(n_points):
                # Create a smooth tapering pattern with high concentration in middle
                dist_from_center = abs(i - mid)
                if dist_from_center < 30:
                    h_opt[i] = 1.0 - 0.1 * (dist_from_center / 30)
                elif dist_from_center < 40:
                    h_opt[i] = 0.9 - 0.8 * ((dist_from_center - 30) / 10)
                else:
                    h_opt[i] = 0.1
            
            # Project to correct sum
            actual_sum = np.sum(h_opt)
            scale_factor = TARGET_SUM / actual_sum
            h_opt = h_opt * scale_factor
            
            # Evaluate actual C5
            correlations_actual = []
            for k in range(n_points):
                overlap_indices = list(range(n_points - k))
                if overlap_indices:
                    corr_k = np.sum(h_opt[overlap_indices]) - np.sum(h_opt[overlap_indices] * h_opt[np.array(overlap_indices) + k])
                    correlations_actual.append(corr_k)
            c5_bound = max(correlations_actual) * dx
    
    except Exception as e:
        # If LP fails, fall back to a good heuristic
        h_opt = np.zeros(n_points)
        mid = n_points // 2
        # Conservative step pattern: more ones concentrated in center
        for i in range(n_points):
            dist_from_center = abs(i - mid)
            if dist_from_center < 40:
                h_opt[i] = 1.0
            elif dist_from_center < 50:
                h_opt[i] = 0.5
            else:
                h_opt[i] = 0.0
        
        # Project to correct sum
        actual_sum = np.sum(h_opt)
        scale_factor = TARGET_SUM / actual_sum
        h_opt = h_opt * scale_factor
        
        # Evaluate actual C5
        correlations_actual = []
        for k in range(n_points):
            overlap_indices = list(range(n_points - k))
            if overlap_indices:
                corr_k = np.sum(h_opt[overlap_indices]) - np.sum(h_opt[overlap_indices] * h_opt[np.array(overlap_indices) + k])
                correlations_actual.append(corr_k)
        c5_bound = max(correlations_actual) * dx
    
    return (h_opt.tolist(), float(c5_bound), n_points)
```
