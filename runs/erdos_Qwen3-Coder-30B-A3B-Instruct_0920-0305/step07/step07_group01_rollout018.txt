import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr)

def fourier_basis(x, coeffs, n_modes=8):
    """Evaluate Fourier series basis: h(x) = sum_{m=0}^{n_modes} coeffs[m] * phi_m(x)"""
    # phi_0 = 1 (constant), phi_m = cos(m * pi * x) for m = 1, ..., n_modes
    h = coeffs[0] * np.ones_like(x)  # Start with constant term
    for m in range(1, n_modes + 1):
        h += coeffs[m] * np.cos(m * np.pi * x)
    return h

def project_to_bounds(h_vals, x_vals, bounds):
    """Project h(x) to stay in [0,1] at sample points"""
    h_clipped = np.clip(h_vals, 0, 1)
    return h_clipped

def objective_function(coeffs, n_points=200, n_modes=8):
    """Objective to minimize: C5"""
    # Define the domain [0,2] with n_points
    x = np.linspace(0, 2, n_points)
    
    # Evaluate h(x) using Fourier basis
    h_val = fourier_basis(x, coeffs, n_modes)
    
    # Enforce bounds by penalty if violated
    h_clipped = project_to_bounds(h_val, x, [0, 1])
    
    # Calculate sum constraint (should be 1) and apply penalty if violated
    integral = np.trapz(h_clipped, x)
    sum_penalty = 10000 * abs(integral - 1.0)
    
    # Evaluate overlap integral C5
    c5 = evaluate_c5(h_clipped)
    
    # Return combined objective with penalties
    return c5 + sum_penalty

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Number of Fourier modes to use
    n_modes = 8
    n_points = 200  # fine resolution for evaluation
    
    # Define bounds for each coefficient (tighter initial bounds)
    bounds = [(-1.0, 1.0) for _ in range(n_modes + 1)]
    
    # Initial guess approaches
    initial_guess = [0.5] + [0.0] * n_modes  # Start with constant baseline
    
    best_c5 = float('inf')
    best_coeffs = None
    
    # Run differential evolution for the entire budget
    result = differential_evolution(
        objective_function,
        bounds,
        args=(n_points, n_modes),
        seed=seed,
        maxiter=max(100, int(budget_s / 10)),
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.7,
        atol=1e-6,
        rtol=1e-6,
        workers=1,
        disp=0
    )
    
    # Refine with local optimization if converged
    refined_result = result
    if result.success:
        refined_result = differential_evolution(
            objective_function,
            bounds,
            args=(n_points, n_modes),
            seed=seed,
            maxiter=200,
            popsize=10,
            mutation=(0.8, 1.0),
            recombination=0.9,
            atol=1e-8,
            rtol=1e-8,
            workers=1,
            disp=0,
            x0=result.x
        )
    
    best_coeffs = refined_result.x
    
    # Reconstruct h at high resolution and evaluate if smaller
    x_fine = np.linspace(0, 2, n_points)
    h_opt = fourier_basis(x_fine, best_coeffs, n_modes)
    h_opt = project_to_bounds(h_opt, x_fine, [0, 1])
    c5_final = evaluate_c5(h_opt)
    
    # Return vector form of h
    return (h_opt.tolist(), float(c5_final), len(h_opt))
