import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import math

# Global variables
TARGET_SUM = None
N = 256
dx = 2.0 / N

def c5_fft(h):
    """Fast evaluation using FFT for linear cross-correlation"""
    n = len(h)
    nfft = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, nfft)
    B = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * B, nfft)[:2*n-1]
    return np.max(conv) * dx

def project_to_sum(h, target_sum):
    """Project h onto the affine subspace with sum = target_sum"""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Binary search for lambda
    def func(lam):
        proj = np.clip(h + lam, 0, 1)
        return np.sum(proj) - target_sum
    
    a, b = -1.0, 1.0
    while func(a) * func(b) > 0:
        a -= 0.1
        b += 0.1
    
    for _ in range(50):
        m = (a + b) / 2
        fm = func(m)
        if abs(fm) < 1e-10:
            break
        if fm * func(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def thue_morse(n):
    """Generate Thue-Morse sequence of length n"""
    # Generate the sequence up to 2n terms then truncate to n
    tm = [0]
    for i in range(int(math.log2(n)) + 2):
        tm = tm + [1 - x for x in tm]
    tm = tm[:n]
    
    # Ensure count of 1s equals target_sum
    current_ones = sum(tm)
    if current_ones > TARGET_SUM:
        # Flip some 1s to 0s
        indices_ones = [i for i in range(n) if tm[i] == 1]
        to_flip = current_ones - TARGET_SUM
        for i in range(to_flip):
            tm[indices_ones[i]] = 0
    elif current_ones < TARGET_SUM:
        # Flip some 0s to 1s
        indices_zeros = [i for i in range(n) if tm[i] == 0]
        to_flip = TARGET_SUM - current_ones
        for i in range(to_flip):
            tm[indices_zeros[i]] = 1
    return np.array(tm)

def rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence of length n"""
    # Generate Rudin-Shapiro up to 2n then truncate
    rs = [0]
    for i in range(int(math.log2(n)) + 2):
        new_rs = []
        for x in rs:
            new_rs.append(x)
            new_rs.append(x + 1)
        rs = new_rs
    rs = rs[:n]
    
    # Convert to binary 0,1
    binary = [int(x % 2 == 0) for x in rs]
    
    # Adjust to have exactly target_sum 1s
    current_ones = sum(binary)
    if current_ones > TARGET_SUM:
        # Flip some 1s to 0s
        indices_ones = [i for i in range(n) if binary[i] == 1]
        to_flip = current_ones - TARGET_SUM
        for i in range(to_flip):
            binary[indices_ones[i]] = 0
    elif current_ones < TARGET_SUM:
        # Flip some 0s to 1s
        indices_zeros = [i for i in range(n) if binary[i] == 0]
        to_flip = TARGET_SUM - current_ones
        for i in range(to_flip):
            binary[indices_zeros[i]] = 1
    return np.array(binary)

def greedy_descent(h):
    """Perform greedy descent on worst shift"""
    h = h.copy()
    target_sum = TARGET_SUM
    ones_count = int(np.sum(h))
    
    # Ensure sum constraint
    if ones_count != target_sum:
        indices_0 = np.where(h == 0)[0]
        indices_1 = np.where(h == 1)[0]
        if ones_count > target_sum:
            for i in range(ones_count - target_sum):
                if len(indices_1) > 0:
                    h[indices_1[i]] = 0
        else:
            for i in range(target_sum - ones_count):
                if len(indices_0) > 0:
                    h[indices_0[i]] = 1
    
    # Iteratively reduce overlap at worst shift
    while True:
        corr = correlate(h, 1 - h, mode="full")
        worst_idx = np.argmax(corr)
        shift = worst_idx - (N - 1)
        corr_val = corr[worst_idx]
        
        # Identify matching (1,0) and (0,1) pairs
        A = []
        B = []
        if shift >= 0:
            for i in range(N - shift):
                if h[i] == 1 and h[i + shift] == 0:
                    A.append(i)
                if h[i] == 0 and h[i + shift] == 1:
                    B.append(i)
        else:
            shift = -shift
            for i in range(N - shift):
                if h[i + shift] == 1 and h[i] == 0:
                    A.append(i + shift)
                if h[i + shift] == 0 and h[i] == 1:
                    B.append(i + shift)
        
        if len(A) == 0 or len(B) == 0:
            break
        
        # Perform a swap
        i = np.random.choice(A)
        j = np.random.choice(B)
        
        h[i], h[j] = h[j], h[i]
    
    return h

        pass
    
    for sidx in range(min(restarts, len(seeds))):
        h0 = seeds[sidx].copy()
        h_greedy = greedy_descent(h0)
        c5_greedy = c5_fft(h_greedy)
        if c5_greedy < best_c5:
            best_c5 = c5_greedy
            best_h = h_greedy.copy()
    
    # Phase 2: Simulated annealing
    if best_h is None:
        best_h = np.random.rand(N)
        best_h = project_to_sum(best_h, TARGET_SUM)
    h_anneal = best_h.copy()
    c5_anneal = best_c5
    
    T0 = 0.02
    max_anneal_time = 0.5 * budget_s
    anneal_end = start_time + max_anneal_time
    
    T = T0
    best_energy = c5_anneal
    
    iter_count = 0
    last_improvement = 0
    no_improvement_counter = 0
    
    # Run SA loop
    while time.time() < anneal_end and T > 1e-5:
        if iter_count % 500 == 0 and iter_count > 0:
            h_anneal = greedy_descent(h_anneal)
            c5_anneal = c5_fft(h_anneal)
            if c5_anneal < best_energy:
                best_energy = c5_anneal
                last_improvement = iter_count
            no_improvement_counter = 0
        
        # Try a swap
        a = np.random.randint(0, N)
        b = np.random.randint(0, N)
        while h_anneal[a] == h_anneal[b]:
            b = np.random.randint(0, N)
        
        # Evaluate what happens with this swap
        old_energy = c5_anneal
        h_temp = h_anneal.copy()
        h_temp[a], h_temp[b] = h_temp[b], h_temp[a]
        new_energy = c5_fft(h_temp)
        
        if new_energy <= old_energy:
            h_anneal = h_temp
            c5_anneal = new_energy
            if new_energy < best_energy:
                best_energy = new_energy
                last_improvement = iter_count
        else:
            prob = math.exp(-(new_energy - old_energy) / T)
            if np.random.rand() < prob:
                h_anneal = h_temp
                c5_anneal = new_energy
        
        iter_count += 1
        no_improvement_counter += 1
        if no_improvement_counter > 10000:
            T *= 0.98
            no_improvement_counter = 0
    
    best_c5 = best_energy
    best_h = h_anneal.copy()
    
    # Phase 3: Continuous refinement with trust-constr
    cont_end = start_time + budget_s
    if time.time() < cont_end - 10:
        h_init = best_h.copy()
        # Create initial guess with smoothing
        kernel = np.array([0.25, 0.5, 0.25])
        h_smooth = np.convolve(h_init, kernel, mode='same')
        
        h_smooth = np.clip(h_smooth, 0, 1)
        h_refined = project_to_sum(h_smooth, TARGET_SUM)
        
        # Create objective function using FFT
        def obj(h):
            # Avoid singularities near boundary by adjusting
            return c5_fft(h)
        
        try:
            # Try optimization with trust-constr
            res = minimize(
                obj,
                h_refined,
                method='trust-constr',
                bounds=[(0, 1) for _ in range(N)],
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM},

                h_final = project_to_sum(np.clip(res.x, 0, 1), TARGET_SUM)

                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_final
        except:
            pass
    
    # Final validation
    best_h = project_to_sum(np.clip(best_h, 0, 1), TARGET_SUM)
    final_c5 = c5_fft(best_h)
    
    return (best_h.tolist(), float(final_c5), N)
