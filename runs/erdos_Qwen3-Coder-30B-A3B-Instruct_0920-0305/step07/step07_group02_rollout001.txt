```python
import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def random_binary_initial(n):
    """Create a random binary initialization with exactly n/2 ones"""
    h = np.zeros(n)
    ones_indices = np.random.choice(n, n//2, replace=False)
    h[ones_indices] = 1.0
    return h

def greedy_improvement(h):
    """Improve by targeting the worst shift"""
    n = len(h)
    max_iter = 1000  # prevent infinite loops
    iter_count = 0
    
    while iter_count < max_iter:
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n - 1)
        
        # Determine overlapping region indices
        if max_shift >= 0:
            i_range = np.arange(0, n - max_shift)
            j_range = i_range + max_shift
        else:
            i_range = np.arange(-max_shift, n)
            j_range = i_range + max_shift
        
        # Find mismatched pairs
        h_i = h[i_range]
        h_j = h[j_range]
        
        # (1,0) mismatches: h_i == 1 and h_j == 0
        im1 = i_range[(h_i == 1) & (h_j == 0)]
        # (0,1) mismatches: h_i == 0 and h_j == 1
        im2 = i_range[(h_i == 0) & (h_j == 1)]
        
        if im1.size == 0 or im2.size == 0:
            break
            
        # Pick random i, j
        i = np.random.choice(im1)
        j = np.random.choice(im2)
        
        # Swap
        h[i] = 0
        h[j] = 1
        
        iter_count += 1
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    
    # Allocate time for main search (80% of budget)
    search_time = 0.8 * budget_s
    
    # Run multiple random starts
    max_restarts = 50
    restart_count = 0
    
    while restart_count < max_restarts and (time.time() - start_time) < search_time:
        # Initialize binary solution
        h = random_binary_initial(n_points)
        
        # Apply greedy improvement
        h_improved = greedy_improvement(h.copy())
        
        # Evaluate
        c5 = evaluate_c5(h_improved)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_improved.copy()
        
        restart_count += 1
        
        # Early stopping if we found good enough solution
        if best_c5 <= 0.379:
            break
    
    # If no good solution found, start with known values
    if best_h is None:
        best_h = random_binary_initial(n_points)
        best_c5 = evaluate_c5(best_h)
    
    # Final refinement using local search
    final_h = best_h.copy()
    final_c5 = best_c5
    num_refinements = 0
    
    # Quick refinement using swaps around current best
    while (time.time() - start_time) < budget_s - 2.0 and num_refinements < 1000:
        # Try a few random swaps
        for _ in range(50):
            if (time.time() - start_time) >= budget_s - 2.0:
                break
            
            # Pick two random indices to swap
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            
            if i != j and final_h[i] != final_h[j]:
                # Perform swap
                final_h[i] = 1 - final_h[i]
                final_h[j] = 1 - final_h[j]
                
                new_c5 = evaluate_c5(final_h)
                
                if new_c5 < final_c5:
                    final_c5 = new_c5
                else:
                    # Revert
                    final_h[i] = 1 - final_h[i]
                    final_h[j] = 1 - final_h[j]
        
        num_refinements += 1
    
    return (final_h.tolist(), float(final_c5), n_points)
```
