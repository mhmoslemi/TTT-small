```python
import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
from cvxpy import SCS
from scipy.linalg import eigh

# Global constants for the problem
n_points = 140
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def create_lag_matrices(n):
    """Create sparse matrices A_ell for each lag ℓ"""
    # We'll need matrices for lags from -(n-1) to (n-1)
    # But we only need the non-zero entries
    A_list = []
    for ell in range(-(n-1), n):
        A = np.zeros((n, n))
        for i in range(n):
            j = i + ell
            if 0 <= j < n:
                A[i, j] = 1
                A[j, i] = 1  # Symmetric
        A_list.append(A)
    return A_list

def evaluate_c5(h):
    """Fast evaluation using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto [0,1] box with sum = target_sum"""
    def f(lam):
        proj = np.clip(x + lam, 0, 1)
        return np.sum(proj) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_sdp_and_round(A_list):
    """Solve SDP relaxation then round solution to get a feasible vector""" 
    n = n_points
    X = cp.Variable((n, n), PSD=True)
    t = cp.Variable()
    
    # Build constraints
    constraints = [
        cp.diag(X) >= 0,
        cp.diag(X) <= 1,
        cp.sum(cp.diag(X)) == TARGET_SUM
    ]
    # Add all autocorrelation constraints
    for A_ell in A_list:
        constraints.append(cp.trace(A_ell @ X) >= TARGET_SUM - t / dx)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve with SCS solver
    prob.solve(solver=SCS, verbose=False, max_iters=5000, eps=1e-5)
    
    # Get solution matrix
    X_opt = X.value
    if X_opt is None:
        return None, float('inf')
    
    # Rounding strategy: use principal eigenvector
    lamb, V = eigh(X_opt)
    v = V[:, -1]  # Largest eigenvalue's eigenvector
    h_unscaled = np.abs(v)  # Take absolute value
    
    # Scale to the target sum and clip
    h_raw = project_to_sum_box(h_unscaled, TARGET_SUM)
    
    return h_raw, evaluate_c5(h_raw)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Generate the Lag matrices
    A_list = create_lag_matrices(n_points)
    
    best_h = np.ones(n_points) * 0.5
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: Try SDP approach first
    time_left = budget_s - (time.time() - start_time) - 1.0
    if time_left > 1.0:
        h_sdp, c5_sdp = solve_sdp_and_round(A_list)
        if h_sdp is not None and c5_sdp < best_c5:
            best_h = h_sdp
            best_c5 = c5_sdp
    
    # Phase 2: Local refinement from SDP solution if available
    if time.time() < start_time + budget_s - 10.0:
        h_refined = best_h.copy()
        # compute gradient with finite differences
        for _ in range(1000):
            if time.time() > start_time + budget_s - 1.0:
                break
            grad = np.zeros(n_points)
            eps = 1e-5
            h_base = h_refined.copy()
            for i in range(n_points):
                h_plus = h_base.copy()
                h_plus[i] += eps
                h_plus = project_to_sum_box(h_plus, TARGET_SUM)
                c5_plus = evaluate_c5(h_plus)
                
                h_minus = h_base.copy()
                h_minus[i] -= eps
                h_minus = project_to_sum_box(h_minus, TARGET_SUM)
                c5_minus = evaluate_c5(h_minus)
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Drop small gradients
            idx_min = np.argmin(grad)
            idx_max = np.argmax(grad)
            if grad[idx_min] < 0.5 * grad[idx_max] and grad[idx_min] < -1e-6:
                delta = min(h_refined[idx_min], 1 - h_refined[idx_max], 0.01)
                if delta > 1e-6:
                    h_refined[idx_min] -= delta
                    h_refined[idx_max] += delta
                    h_refined = project_to_sum_box(h_refined, TARGET_SUM)
                    nc5 = evaluate_c5(h_refined)
                    if nc5 < best_c5:
                        best_c5 = nc5
                        best_h = h_refined.copy()
    
    # Final polishing with random generation if needed
    time_left = budget_s - (time.time() - start_time) - 10.0
    if time_left > 1.0:
        for _ in range(100):
            if time.time() > start_time + budget_s - 1.0:
                break
            # Perturb current best
            h_pert = best_h + np.random.normal(0, 0.01, n_points)
            h_pert = project_to_sum_box(h_pert, TARGET_SUM)
            c5_pert = evaluate_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
        
        # One more gradient-based refinement
        for _ in range(200):
            if time.time() > start_time + budget_s - 1.0:
                break
            h_base = best_h.copy()
            grad = np.zeros(n_points)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h_base.copy()
                h_plus[i] += eps
                h_plus = project_to_sum_box(h_plus, TARGET_SUM)
                c5_plus = evaluate_c5(h_plus)
                
                h_minus = h_base.copy()
                h_minus[i] -= eps
                h_minus = project_to_sum_box(h_minus, TARGET_SUM)
                c5_minus = evaluate_c5(h_minus)
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            idx_min = np.argmin(grad)
            idx_max = np.argmax(grad)
            if grad[idx_min] < 0.5 * grad[idx_max] and grad[idx_min] < -1e-6:
                delta = min(best_h[idx_min], 1 - best_h[idx_max], 0.005)
                if delta > 1e-6:
                    best_h[idx_min] -= delta
                    best_h[idx_max] += delta
                    best_h = project_to_sum_box(best_h, TARGET_SUM)
                    nc5 = evaluate_c5(best_h)
                    if nc5 < best_c5:
                        best_c5 = nc5

    return (best_h.tolist(), float(best_c5), n_points)
```
