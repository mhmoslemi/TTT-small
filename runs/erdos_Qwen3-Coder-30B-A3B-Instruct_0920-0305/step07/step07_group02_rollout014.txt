import numpy as np
from scipy.signal import correlate

import time

# Global constants
N = 256
dx = 2.0 / N
TARGET_SUM = N // 2

def c5_fft(h):
    """Fast evaluation using FFT"""
    n = len(h)
    nfft = 1 << ((2*n-1).bit_length())          # next power of two
    H = np.fft.rfft(h, nfft)
    B = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * B, nfft)[:2*n-1]   # length 2N‑1
    return np.max(conv) * dx

def project_to_sum(h, target_sum):
    """Project h onto the simplex with sum = target_sum using bisection"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    def f(lam):
        proj = np.clip(h + lam, 0, 1)
        return np.sum(proj) - target_sum
    
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    return np.clip(h + (a + b)/2, 0, 1)

def thue_morse(n):
    """Generate Thue-Morse sequence of length n"""
    t = [0]
    while len(t) < n:
        t.extend(1 - np.array(t))
    return np.array(t[:n])

def rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence of length n"""
    # simplistic generator
    s = [0] * n
    for i in range(1, n):
        s[i] = (s[i-1] + (i & (i-1)) == 0) & 1
    return np.array(s)

def greedy_descent(h):
    """Improve h by swapping mismatched pairs for worst shift"""
    h = h.copy()
    h = np.clip(h, 0, 1)
    max_iter = 50000
    for it in range(max_iter):
        corr = np.correlate(h, 1-h, mode="full")
        max_shift_idx = np.argmax(corr)
        shift = max_shift_idx - (N-1)
        
        # get pairs
        if shift >= 0:
            A = np.where((h[:-shift] == 1) & (h[shift:] == 0))[0]
            B = np.where((h[:-shift] == 0) & (h[shift:] == 1))[0]
        else:
            A = np.where((h[-shift:] == 1) & (h[:-shift] == 0))[0]
            B = np.where((h[-shift:] == 0) & (h[:-shift] == 1))[0]
        
        if len(A) == 0 or len(B) == 0:
            break
        
        i = np.random.choice(A)
        j = np.random.choice(B)
        
        if shift >= 0:
            h[i], h[j] = h[j], h[i]
        else:
            h[j], h[i] = h[i], h[j]
    
    return h

def simulate_annealing(seed_h, max_time):
    """Perform simulated annealing on binary vectors"""
    start = time.time()
    h = seed_h.copy()
    current_c5 = c5_fft(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    T0 = 0.02
    t = 0
    
    while t < max_time:
        if time.time() - start > max_time - 0.1:
            break
        t = time.time() - start
        
        # linear cooling
        T = T0 * (1 - t / max_time)
        if T < 1e-5:
            break
            
        # propose swap
        one_indices = np.where(h == 1)[0]
        zero_indices = np.where(h == 0)[0]
        if len(one_indices) == 0 or len(zero_indices) == 0:
            continue
            
        i = np.random.choice(one_indices)
        j = np.random.choice(zero_indices) 
            
        # make new state
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i] 
        new_c5 = c5_fft(h_new)
        
        # accept or reject
        if new_c5 <= current_c5:
            h = h_new
            current_c5 = new_c5
        else:
            delta = new_c5 - current_c5
            if np.random.rand() < np.exp(-delta / T):
                h = h_new
                current_c5 = new_c5
                
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # mini-greedy improvement every k steps
        if it % 500 == 0 and it > 0:
            h = greedy_descent(h)
            current_c5 = c5_fft(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    return best_h, best_c5
