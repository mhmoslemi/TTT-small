import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from scipy.fft import fft, ifft

# Global constants
n_points = 256
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(ifft(fft(h_rev, L) * fft(b, L)))
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_thue_morse(n):
    """Generate Thue-Morse sequence of length n"""
    tm = [0]
    while len(tm) < n:
        tm.extend(1 - np.array(tm))
    return np.array(tm[:n])

def generate_rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence of length n"""
    rs = np.zeros(n, dtype=int)
    rs[0] = 1
    for i in range(1, n):
        rs[i] = rs[i-1] ^ (i & (i-1) == 0)
    return rs

def binary_to_float(binary_h):
    """Map binary to continuous values"""
    return binary_h.astype(float)

def float_to_binary(float_h, threshold=0.5):
    """Convert float to binary"""
    return (float_h > threshold).astype(int)

def find_worst_shift_indices(h, k):
    """Find indices where (h_i, h_{i+k}) = (1,0) and (h_j, h_{j+k}) = (0,1)"""
    N = len(h)
    if k >= 0:
        h1 = h[:N-k]
        h2 = h[k:]
    else:
        h1 = h[-k:]
        h2 = h[:N+k]
    
    # Find index pairs where (1,0) in first and (0,1) in second
    idx10 = np.where((h1 == 1) & (h2 == 0))[0]
    idx01 = np.where((h1 == 0) & (h2 == 1))[0]
    return idx10, idx01

def greedy_swap_step(h, max_shift_idx, current_c5):
    """Perform one greedy swap step on worst shift"""
    # Find shift that maximizes overlap
    N = len(h)
    
    # Forward calculation
    corr = compute_corr_fft(h)
    max_shift_idx = np.argmax(corr)
    
    # Find pairs for this shift
    k = max_shift_idx - N + 1 # shift amount = max_shift_idx - (N - 1)
    idx10, idx01 = find_worst_shift_indices(h, k)
    
    # If no beneficial swaps exist, skip
    if len(idx10) == 0 or len(idx01) == 0:
        return h, current_c5, False
    
    # Swap first pair that reduces overlap
    try:
        i = idx10[0]
        j = idx01[0]
        if k >= 0:
            h[i], h[i+k] = h[i+k], h[i]
        else:
            h[i], h[j] = h[j], h[i]
        return h, evaluate_c5(h), True
    except:
        return h, current_c5, False

def greedy_improvement(h0):
    """Improve with local greedy swap"""
    h = h0.copy()
    current_c5 = evaluate_c5(h)
    
    while True:
        corr = compute_corr_fft(h)
        max_shift_idx = np.argmax(corr)
        old_c5 = current_c5
        h, current_c5, improved = greedy_swap_step(h, max_shift_idx, current_c5)
        if not improved:
            break
    
    return h

def random_binary_restart():
    """Generate a random binary initial solution"""
    h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
    h = project_to_sum_box(h.astype(float), TARGET_SUM)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    iterations = 0
    
    # We'll use 1000 random restarts with time control
    max_restarts = 1000
    restart_count = 0
    
    # Initialize with known sequences
    sequences = [
        generate_thue_morse(n_points),
        generate_rudin_shapiro(n_points),
    ]
    for i, seq in enumerate(sequences):
        h = project_to_sum_box(seq.astype(float), TARGET_SUM)
        h = greedy_improvement(h)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Random restarts
    while restart_count < max_restarts and (time.time() - start_time) < budget_s:
        h = random_binary_restart()
        h = greedy_improvement(h)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        restart_count += 1
        
        # Periodic status check to prevent exhausting budget
        if restart_count % 50 == 0 and (time.time() - start_time) > 0.9 * budget_s:
            break
    
    # Continuous refinement
    if best_h is not None:
        # Refine with smooth optimization
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_eq(h_local):
            return np.sum(h_local) - TARGET_SUM
            
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        # Use SLSQP with sum constraint
        try:
            res = minimize(
                evaluate_c5, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-14}
            )
            if res.success:
                best_h = res.x
        except:
            pass
            
        best_c5 = evaluate_c5(best_h)
        
        # Final project to make sure sums and bounds respected
        best_h = project_to_sum_box(best_h, TARGET_SUM)
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
