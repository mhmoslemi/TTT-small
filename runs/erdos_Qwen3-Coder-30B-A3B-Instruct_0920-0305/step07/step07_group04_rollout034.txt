import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
from cvxpy import Minimize, Problem
from scipy.optimize import minimize

# Global variables
TARGET_SUM = None
n_points = 150
dx = 2.0 / n_points
TIME_LIMIT = 600  # seconds

def evaluate_c5(h):
    """Evaluate the overlap integral using FFT."""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_relaxed_lp():
    """Solve the linear relaxation using McCormick envelopes."""
    h = cp.Variable(n_points, name="h")
    p = cp.Variable((n_points, n_points), name="p")
    
    # Mass constraint
    mass_constraint = cp.sum(h) == TARGET_SUM
    
    # McCormick inequalities
    mccormick_constraints = []
    for i in range(n_points):
        for j in range(i, n_points):
            mccormick_constraints.append(p[i, j] >= 0)
            mccormick_constraints.append(p[i, j] >= h[i] + h[j] - 1)
            mccormick_constraints.append(p[i, j] <= h[i])
            mccormick_constraints.append(p[i, j] <= h[j])
    
    # Objective
    t = cp.Variable(name="t")
    shift_constraints = []
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            O_k = cp.sum([p[i, i + k] for i in range(n_points - k)])
        else:
            O_k = cp.sum([p[i - k, i] for i in range(-k, n_points)])
        shift_constraints.append(O_k >= TARGET_SUM / 2 - t)
    
    # Final constraints
    constraints = [h >= 0, h <= 1] + mccormick_constraints + [mass_constraint] + shift_constraints
    
    # Problem formulation
    prob = Problem(Minimize(t), constraints)
    prob.solve(solver=cp.ECOS, max_iters=3000, abstol=1e-6, reltol=1e-6)
    
    if prob.status in ["optimal", "optimal_inaccurate"]:
        return h.value, prob.value
    else:
        raise RuntimeError("LP solver failed, status = " + prob.status)

def compute_exact_c5(h):
    """Compute exact C5 with positive evaluation."""
    corr = correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def main_local_refinement(h_init):
    """Perform projected sub-gradient descent on the exact objective."""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_exact_c5(h)
    
    # For each iteration:
    for iter in range(5000):
        if time.time() > start_time + TIME_LIMIT - 20:
            break
        
        # Find the shift k* that maximizes the overlap
        corr = correlate(h, 1 - h, mode="full")
        k_star = np.argmax(corr)
        shift_idx = k_star - (n_points - 1)
        
        # Compute the gradient
        grad = np.zeros_like(h)
        if shift_idx >= 0:
            # grad[i] = (1 - h[i + shift_idx]) - h[i - shift_idx] when valid
            for i in range(n_points - shift_idx):
                grad[i] += (1 - h[i + shift_idx])
            for i in range(shift_idx, n_points):
                grad[i] -= h[i - shift_idx]
        else:
            # shift_idx < 0: i - shift_idx > i, so shift_idx is negative
            for i in range(-shift_idx, n_points):
                grad[i] += (1 - h[i + shift_idx])
            for i in range(n_points + shift_idx):
                grad[i] -= h[i - shift_idx]
        
        # Gradient step with decreasing step size
        alpha = 0.5 * dx * (0.99 ** (iter // 100))
        h_new = h - alpha * grad
        
        # Project to unit cube [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project to sum constraint
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate and update
        c5_new = compute_exact_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new.copy()
        else:
            # Accept with probability to escape plateaus
            if np.random.rand() < 0.01:  # 1% chance to accept bad move
                h = h_new.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx, start_time
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up
    TARGET_SUM = n_points / 2
    
    try:
        # Step 1: Solve LP relaxation (150s max)
        h_relaxed, _ = solve_relaxed_lp()
        
        # Step 2: Compute exact C5 for the relaxed solution
        h_feasible = h_relaxed.copy()
        h_feasible = project_to_sum_box(h_feasible, TARGET_SUM)
        c5_feasible = compute_exact_c5(h_feasible)
        
        # Step 3: Refine with gradient descent
        h_final, c5_best = main_local_refinement(h_feasible)
        
        # Maybe use CMA-ES for further refining if needed
        if c5_best > 0.3795:
            # Simple randomization + local refinement
            for _ in range(5):
                h_perturbed = h_final + np.random.normal(0, 0.005, n_points)
                h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
                h_refined, c5_refined = main_local_refinement(h_perturbed)
                if c5_refined < c5_best:
                    h_final, c5_best = h_refined, c5_refined
        
    except Exception as e:
        # Fallback to less structured approach
        h_final = np.ones(n_points) / 2
        h_final = project_to_sum_box(h_final, TARGET_SUM)
        c5_best = compute_exact_c5(h_final)
    
    return (h_final.tolist(), float(c5_best), n_points)
