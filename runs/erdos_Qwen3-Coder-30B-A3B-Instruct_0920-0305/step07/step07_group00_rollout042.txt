import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Generate initial solution using a modified approach
    # Start with piecewise constant with more structured variation
    h0 = np.zeros(n_points)
    
    # Create a more sophisticated piecewise constant pattern
    segments = 6
    seg_size = n_points // segments
    
    # Alternate between higher and lower values in segments
    for i in range(segments):
        start = i * seg_size
        end = (i+1) * seg_size if i < segments-1 else n_points
        if i % 2 == 0:
            h0[start:end] = 0.6
        else:
            h0[start:end] = 0.2
    
    # Ensure sum matches target
    h0 = project_to_sum_box(h0, n_points / 2)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    TARGET_SUM = n_points / 2

    # Try different optimization strategies
    # Try multiple restarts with different perturbation patterns
    for restart in range(20):
        if time.time() - start_time > budget_s:
            break
        
        # Create new initial guess with more diverse perturbations
        h_start = h0.copy()
        
        # Different types of perturbations
        if restart < 5:
            # Random slight perturbations
            h_start += np.random.normal(0, 0.01, n_points)
        elif restart < 10:
            # Add some sine wave modulation to break symmetry
            x_vals = np.linspace(0, 4*np.pi, n_points)
            h_start += 0.03 * np.sin(x_vals)
        else:
            # Stronger random perturbation
            h_start += np.random.normal(0, 0.05, n_points)
        
        h_start = project_to_sum_box(h_start, TARGET_SUM)
        
        # Optimization with L-BFGS-B for better convergence
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                h_start, 
                method='L-BFGS-B', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500}
            )
            if res.success:
                h_candidate = project_to_sum_box(res.x, TARGET_SUM)
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
        except:
            continue

    # Final local refinement
    refinement_attempts = 0
    while time.time() - start_time < budget_s and refinement_attempts < 500:
        refinement_attempts += 1
        
        # Try small random changes
        h_test = best_h + np.random.normal(0, 0.005, n_points)
        h_test = project_to_sum_box(h_test, TARGET_SUM)
        
        # Try to improve with local minimization
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                h_test, 
                method='L-BFGS-B', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 100}
            )
            if res.success:
                h_candidate = project_to_sum_box(res.x, TARGET_SUM)
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
        except:
            continue
            
    return (best_h.tolist(), float(best_c5), n_points)
