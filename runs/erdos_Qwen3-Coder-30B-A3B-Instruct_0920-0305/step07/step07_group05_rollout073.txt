TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmin_weights(s, alpha):
    """Compute the normalized soft-min weights"""
    s_shifted = s - np.max(s)  # Stability trick
    exp_s = np.exp(-alpha * s_shifted)
    return exp_s / np.sum(exp_s)

def weighted_gradient(h, alpha):

    n = len(h)
    
    for i in range(len(s)):
        lag = i - (n - 1)
        if lag >= 0:

            # gradient d/dh_i s_{-lag} = h_{i-lag} (including i-lag >=0)

def pairwise_move(h, grad, delta_max, rng):
    """Transfer mass according to gradient."""
    # Find indices where gradient suggests movement
    src_idx = np.argmin(grad)
    tgt_idx = np.argmax(grad)
    
    if grad[src_idx] >= 0 or grad[tgt_idx] <= 0:
        # If both gradients are not far from zero, 
        # fallback to random valid pair
        src_idx = np.random.randint(0, len(h))
        tgt_idx = np.random.randint(0, len(h))
        while src_idx == tgt_idx or h[src_idx] <= 0 or h[tgt_idx] >= 1:
            src_idx = np.random.randint(0, len(h))
            tgt_idx = np.random.randint(0, len(h))
    
    # Ensure valid moves and small amount to reduce potential large impact
    delta = min(delta_max, h[src_idx], 1 - h[tgt_idx])
    if delta <= 1e-12:
        return False, h
    
    h[src_idx] -= delta
    h[tgt_idx] += delta
    
    return True, h

def annealed_gradient_ascent(h0, time_budget, alpha0, delta0, T0, rng):
    """Run the continuous annealed gradient ascent process."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    n = len(h)
    
    alpha = alpha0
    delta = delta0
    T = T0
    
    # Updated schedule parameters:
    alpha_inc = 1.005
    delta_decay = 0.9995
    T_decay = 0.9998
    
    start_time = time.time()
    iter_count = 0
    
    while time.time() - start_time < time_budget:
        iter_count += 1
        alpha = max(alpha * alpha_inc, 10.0)
        delta = max(delta * delta_decay, 0.001)
        T = T * T_decay
        
        if iter_count % 300 == 0:
            delta = min(delta * 1.02, 0.4)  # Occasionally increase
    
        grad = weighted_gradient(h, alpha)
        accepted, h = pairwise_move(h, grad, delta, rng)
        
        if accepted:
            c5_val = evaluate_c5(h)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
                
            # Acceptance probability for worsening moves (Metropolis rule)
            # We should derive this more carefully:
            # delta_c5 = c5_val - best_c5 (we want minimizing)
            # if c5_val >= best_c5, accepting with p = exp(-delta_c5 / T)
            # But since delta_c5 is positive, always accept if better; 
            # else accept else with prob = exp(-delta_c5 / T)  
            # Actually delta_c5 = new - old => if new < old -> positive log(-log) =positive
            # But we want to use exp(-(delta_c5) / T), only when new > old
    
            if c5_val > best_c5:
                # if random number bigger than exp(-delta_c5/T) we reject it
                delta_c5 = c5_val - best_c5
                accept_prob = np.exp(-delta_c5 / T)
                if rng.random() > accept_prob:
                    # reject: undo the move
                    h = best_h.copy()
                    # No update to best_h or best_c5, but this is fine.
            
        # Periodic projection to enforce bounds
        if iter_count % 1000 == 0:
            h = np.clip(h, 0, 1)
            h = project_to_sum_box(h, TARGET_SUM)
            if evaluate_c5(h) < best_c5:
                best_c5 = evaluate_c5(h)
                best_h = h.copy()
    
    return best_h, best_c5

def random_binary_start(n, rng):
    """Create random binary start with exactly n/2 ones""" 
    h = np.zeros(n)
    indices = rng.sample(range(n), n//2)
    h[indices] = 1.0
    return h

def local_optimize_restart(start_h, remaining_time, rng):
    """Execute short restart"""
    if remaining_time <= 1.0:
        return start_h, evaluate_c5(start_h)
        
    h_star, c5 = annealed_gradient_ascent(
        start_h, remaining_time*0.6, 10.0, 0.3, 5e-3, rng
    )
    return h_star, c5
