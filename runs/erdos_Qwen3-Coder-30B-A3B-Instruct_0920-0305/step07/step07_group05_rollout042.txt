import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global variables
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmin_weights(s, alpha):
    """Compute normalized soft-min weights from correlation values"""
    s_max = np.max(s)
    exp_s = np.exp(-alpha * (s - s_max))
    return exp_s / np.sum(exp_s)

def weighted_gradient(h, alpha):
    """Compute gradient of softmin of correlations"""
    # Compute full autocorrelation of h
    s = correlate(h, h, mode='full')
    w = softmin_weights(s, alpha)
    
    # Initialize gradient
    grad = np.zeros_like(h)
    
    # Add contributions from each lag
    n = len(h)
    for i in range(len(s)):
        lag = i - (n - 1)  # Convert from full index to actual lag
        if lag >= 0:
            # h[i] contributes to s[i+lag] term: h[i+lag]
            if i + lag < n:
                grad[i] += w[i] * h[i + lag]
            # h[i+lag] contributes to s[i] term: h[i]
            if i - lag >= 0:
                grad[i] += w[i] * h[i - lag]
        else:
            # lag < 0
            # h[i] contributes to s[i+lag] term: h[i+lag] 
            if i - lag < n:
                grad[i] += w[i] * h[i - lag]
            # h[i-lag] contributes to s[i] term: h[i]
            if i + lag >= 0:
                grad[i] += w[i] * h[i + lag]
                
    return grad

def pairwise_move(h, grad, delta, rng):
    """Move mass from the most negative gradient index to the most positive"""
    # Find indices for source and target
    src_idx = np.argmin(grad)
    tgt_idx = np.argmax(grad)
    
    # Ensure we can move mass (not at boundary)
    if h[src_idx] <= 1e-10 or h[tgt_idx] >= 1 - 1e-10:
        return False, h, 0
    
    # Calculate transfer amount
    max_transfer = min(delta, h[src_idx], 1 - h[tgt_idx])
    if max_transfer <= 1e-10:
        return False, h, 0
    
    # Move mass
    h[src_idx] -= max_transfer
    h[tgt_idx] += max_transfer
    
    return True, h, max_transfer

def random_binary_start(n, rng):
    """Create a feasible random binary vector"""
    h = np.zeros(n)
    ones_indices = rng.sample(range(n), k=n//2)
    h[ones_indices] = 1.0
    return h

def annealed_gradient_ascent(h0, time_budget, alpha0, delta0, T0, rng):
    """Core optimisation loop using annealed gradient ascent"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    alpha = alpha0
    delta = delta0
    T = T0
    
    start_time = time.time()
    step = 0
    accepted_moves = 0
    
    while time.time() - start_time < time_budget:
        # Compute gradient
        grad = weighted_gradient(h, alpha)
        
        # Perform a move
        accepted, h_new, delta_transfer = pairwise_move(h, grad, delta, rng)
        
        if accepted:
            h = h_new.copy()
            c5_new = evaluate_c5(h)
            
            # Accept if better
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                
            # Accept with Metropolis probability if worse
            elif rng.random() < np.exp(-(c5_new - best_c5) / T):
                best_c5 = c5_new
                best_h = h.copy()
            
            accepted_moves += 1
            
            # Occasionally re-project to maintain feasibility
            if accepted_moves % 1000 == 0:
                h = project_to_sum_box(h, TARGET_SUM)
                
        # Adaptive parameters
        if step % 50 == 0:
            delta *= 0.9995
            T *= 0.9995
            if alpha < 200 and step % 500 == 0:
                alpha *= 1.1
                
        step += 1
        
    return best_h, best_c5

def local_optimize_restart(start_h, remaining_time, rng):
    """Optimize a given start vector for a time limit"""
    h0 = start_h.copy()
    # Start with moderate alpha and Δ
    h_opt, c5_opt = annealed_gradient_ascent(
        h0, remaining_time, 
        alpha0=10, delta0=0.3, T0=1e-2, rng=rng
    )
    return h_opt, c5_opt
