import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def construct_initial_solution():
    """Construct an initial solution based on the known better construction"""
    # The solution should have a simple structure that's easy to optimize
    # Let's try a piece-wise constant construction that's symmetric
    
    # Strategy: Create a simple structure with large values in the middle
    h = np.zeros(n_points)
    
    # Put some mass in the middle regions
    mid_start = n_points // 4
    mid_end = 3 * n_points // 4
    h[mid_start:mid_end] = 1.0
    
    # Normalize to sum = n_points/2
    current_sum = np.sum(h)
    h = h * (TARGET_SUM / current_sum)
    
    return h

def modified_gradient_descent(h0, max_iter=3000):
    """Improved gradient descent approach"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # More sophisticated initialization
    h = project_to_sum_box(h, TARGET_SUM)
    
    for iter_num in range(max_iter):
        # Take a step in direction of gradient
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_sum_box(h_plus, TARGET_SUM)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_sum_box(h_minus, TARGET_SUM)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Apply gradient step with adaptive learning rate
        learning_rate = 0.01 / (1.0 + iter_num / 1000)
        h_new = h - learning_rate * grad
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Store current solution and perturb if no improvement
            h = h_new
        
        if best_c5 < 0.375:  # Early stopping if we're close to goal
            break
        
        if iter_num % 500 == 0:
            # Occasionally perform a refinement search
            h_refined, c5_refined = local_refinement(h, 100)
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined.copy()
                h = h_refined

    return best_h, best_c5

def local_refinement(h, num_steps):
    """Perform local refinement around current solution"""
    h_current = h.copy()
    best_c5 = evaluate_c5(h_current)
    
    for _ in range(num_steps):
        # Make small random perturbations
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        
        h_new = h_current.copy()
        h_new[idx1] = np.clip(h_new[idx1] + delta, 0, 1)
        h_new[idx2] = np.clip(h_new[idx2] - delta, 0, 1)
        
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            h_current = h_new.copy()
    
    return h_current, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Phase 1: Generate lower bound from known construction
    h_base = construct_initial_solution()
    h_base = project_to_sum_box(h_base, TARGET_SUM)
    
    best_h = h_base.copy()
    best_c5 = evaluate_c5(h_base)
    
    # Phase 2: Optimizer with adaptive gradient descent
    phase_end = start_time + 0.8 * budget_s
    while time.time() < phase_end:
        # Multiple restarts with different initializations
        for restart in range(15):
            # Generate a new initial point
            h_start = h_base + np.random.normal(0, 0.02, n_points)
            h_start = project_to_sum_box(h_start, TARGET_SUM)
            
            # Optimize from this point
            h_opt, c5_opt = modified_gradient_descent(h_start, 500)
            
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        
        # Cool-down period
        time.sleep(0.01)
    
    # Phase 3: Final refinement
    phase_end = start_time + budget_s - 1.0
    while time.time() < phase_end:
        h_fine, c5_fine = local_refinement(best_h, 200)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
