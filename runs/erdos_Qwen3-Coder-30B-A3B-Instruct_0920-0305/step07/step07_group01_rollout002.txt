import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
MAX_N = 360  # Maximum resolution to use if time permits

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_gradient(h, corr, beta):
    """Compute gradient of smooth max surrogate"""
    corr_shifted = corr - np.max(corr)  # Avoid overflow
    exp_corr = np.exp(beta * corr_shifted)
    weights = exp_corr / np.sum(exp_corr)
    
    # Start with all ones and subtract weighted influences from both directions
    grad = np.ones_like(h)
    for i in range(len(h)):
        for l in range(len(corr)):
            if l == len(corr) // 2:
                continue
            lag = l - len(corr) // 2
            # When indexing to the left (pos lag): h[i+lag] exists if i+lag < N
            # When indexing to the right (negative lag): h[i+lag] exists if i+lag >= 0
            if lag > 0:
                j = i + lag
                if j < len(h):
                    grad[i] -= weights[l] * h[j]
            else:
                j = i + lag
                if j >= 0:
                    grad[i] -= weights[l] * h[j]
    return dx * grad

def max_lag_gradient(h, corr, idx_best):
    """Compute exact sub-gradient of the max function over lags"""
    k_star = idx_best - len(corr) // 2

    # Construct gradient vector for lag k_star
    grad = np.zeros_like(h)
    # Add contribution from front part (i <= N-1-k_star)
    for i in range(len(h)):
        if i <= len(h)-1-k_star:
            grad[i] += (1-h[i+k_star]) * dx
    # Add contribution from back part (i >= k_star)
    for i in range(len(h)):
        if i >= k_star:
            grad[i] -= h[i-k_star] * dx

    return grad

def mass_transfer(h, grad, epsilon=0.01):
    """Perform linearized mass transfer move to make step more efficient"""
    d = -grad
    sources = []
    sinks = []
    
    for i in range(len(h)):
        if d[i] < 0:
            sources.append((i, -d[i], h[i]))
        elif d[i] > 0:
            sinks.append((i, d[i], 1-h[i]))
    
    # Sort by descending d-values
    sources.sort(key=lambda x: x[1], reverse=True)
    sinks.sort(key=lambda x: x[1], reverse=True)
    
    # Greedily transfer mass
    for s_idx, _, rem in sources:
        if rem <= 1e-10: continue
        for t_idx, _, add in sinks:
            if add <= 1e-10: continue
            transfer = min(rem, add, epsilon)
            if transfer <= 1e-10: continue
            h[s_idx] -= transfer
            h[t_idx] += transfer
            rem -= transfer
            add -= transfer
            if rem <= 1e-10: break
    
    # Re-project to satisfy sum and box constraints
    return project_sum_box(h, TARGET_SUM)

def backtracking_line_search(h, grad, beta, corr, idx_best, alpha_init=0.2, alpha_min=5e-4):
    """Backtracking line search with projected step"""
    alpha = alpha_init
    h_old = h.copy()
    corr_old = corr.copy()
    c_old = evaluate_c5(h_old)
    
    # This approximation works mostly in early stages
    if beta < 20:
        grad_func = lambda x: smooth_gradient(x, corr_old, beta)
    else:
        grad_func = lambda x: max_lag_gradient(x, corr_old, idx_best)
    
    while alpha >= alpha_min:
        h_new = h_old - alpha * grad
        h_new = project_sum_box(h_new, TARGET_SUM)
        try:
            c_new = evaluate_c5(h_new)
            if c_new < c_old:
                return h_new, c_new, alpha
        except:
            pass
        alpha /= 2
    return h_old, c_old, 0.0
