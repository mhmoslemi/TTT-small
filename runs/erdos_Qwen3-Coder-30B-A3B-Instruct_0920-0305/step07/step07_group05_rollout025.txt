import numpy as np

# Fixed parameters
n_points = 140
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def build_lag_matrices(n):
    """Build sparse A_ℓ matrices for all lags."""
    A_list = []
    for ell in range(-(n-1), n):
        # Construct sparse matrix for lag ell
        rows = []
        cols = []
        data = []
        for i in range(n):
            j = i + ell
            if 0 <= j < n:
                rows.append(i)
                cols.append(j)
                data.append(1)
        A = np.array(data)
        rows = np.array(rows)
        cols = np.array(cols)
        A_list.append((ell, rows, cols, A))
    return A_list

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def solve_sdp(A_list, n):
    """Solve the SDP relaxation to get a lower bound."""
    # Create variables
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        X >> 0,  # PSD
        cp.diag(X) >= 0,  # Non-negativity
        cp.diag(X) <= 1,  # Upper bound
        cp.sum(cp.diag(X)) == TARGET_SUM,  # Sum constraint
    ]

    # Add the autocorrelation constraints
    for ell, rows, cols, data in A_list:
        # Compute trace(A_ell * X)
        trace_con = cp.sum(cp.multiply(data, X[rows, cols]))
        constraints.append(trace_con >= TARGET_SUM - t)

    # Objective
    objective = cp.Minimize(t)

    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False, eps=1e-5, max_iters=1000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return X.value, t.value
    except:
        pass
    return None, None

def round_sdp_solution(X_opt, n):
    """Round SDP solution to feasible vector h."""
    # Perform eigen-decomposition
    _, V = eigh(X_opt)
    v = V[:, -1]  # largest eigenvector
    v = v / np.linalg.norm(v)  # Normalize

    h_raw = v.copy()
    h_clip = np.clip(h_raw, 0, 1)
    h_proj = project_to_sum_box(h_clip, TARGET_SUM)
    return h_proj

def gradient_guided_search(h0, max_iter):
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    for iter_num in range(max_iter):
        # Simple gradient estimation
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_sum_box(h_plus, TARGET_SUM)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_sum_box(h_minus, TARGET_SUM)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Move toward negative gradient
        idx_min = np.argmin(grad)
        idx_max = np.argmax(grad)
        
        if grad[idx_min] < grad[idx_max]:
            delta = min(h[idx_min], 1 - h[idx_max], 0.005)
            if delta > 1e-6:
                h[idx_min] -= delta
                h[idx_max] += delta
                h = project_to_sum_box(h, TARGET_SUM)
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        else:
            # Try small random move
            i = np.random.choice(n_points)
            j = np.random.choice(n_points)
            if i != j:
                delta = min(h[i], 1 - h[j], 0.005)
                if delta > 1e-6:
                    h[i] -= delta
                    h[j] += delta
                    h = project_to_sum_box(h, TARGET_SUM)
                    c5 = evaluate_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
    
    return best_h, best_c5
