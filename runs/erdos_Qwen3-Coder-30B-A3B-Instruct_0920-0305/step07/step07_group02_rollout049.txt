import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def improved_pattern_search(h0, max_iter=1000):

    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Every 50 iterations, recompute correlation for efficiency
        if i % 50 == 0:
            corr_full = compute_corr_fft(h)
        
        # Sample multiple best locations if they exist
        # Smooth out the correlation to find more diverse promising locations
        corr_smoothed = np.convolve(corr_full, np.ones(5)/5, mode='same')
        max_idx = np.argmax(corr_smoothed)
        
        # Focus on area around maximum for exploration
        start = max(0, max_idx - 20)
        end = min(len(h), max_idx + 20)

            interval_end = np.random.randint(interval_start + 3, min(interval_start + 15, end))
        else:
            interval_start = np.random.randint(0, len(h) - 10)
            interval_end = min(interval_start + 10, len(h))

def hybrid_optimization(h0, max_iter=1000):
    """Hybrid approach mixing pattern search and gradient-based refinement"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    # Phase 1: Pattern search to find promising regions
    for i in range(max_iter//2):
        h = improved_pattern_search(h, max_iter=10)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            # Small random perturb if we haven't improved for a while
            if i % 20 == 0:
                h = h + np.random.normal(0, 0.005, len(h))
                h = project_to_sum_box(h, TARGET_SUM)
    
    # Phase 2: Local gradient refinement
    idxs = np.where(h > 0.01)[0]
    if len(idxs) < 5:
        return h
    
    # Take a subset of indices to optimize for less computational cost
    selected_idxs = np.random.choice(idxs, min(30, len(idxs)), replace=False)
    def objective(h_subset):
        h_full = h.copy()
        h_full[selected_idxs] = h_subset
        return evaluate_c5(h_full)
    
    initial_h_subset = h[selected_idxs]
    bounds = [(0, 1) for _ in range(len(selected_idxs))]
    
    # Optimize the selected indices
    try:
        res = minimize(objective, initial_h_subset, method='L-BFGS-B', bounds=bounds, options={'maxiter': 200})
        if res.success:
            h[selected_idxs] = res.x
            h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    
    return h
