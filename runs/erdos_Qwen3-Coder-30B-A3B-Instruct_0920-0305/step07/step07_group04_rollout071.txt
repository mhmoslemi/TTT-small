import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem setup
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    # Variables
    h = cp.Variable(n_points, name="h")
    
    # Constraints
    constraints = [h >= 0, h <= 1, cp.sum(h) == TARGET_SUM]
    
    # McCormick relaxation for bilinear terms
    # We'll compute correlations for all shifts, but only consider the meaningful ones
    # For shift k, we compute correlation = sum_i h_i * (1 - h_{i+k}) = sum_i h_i - sum_i h_i * h_{i+k}
    
    # Define product variables for all pairs
    P = cp.Variable((n_points, n_points), name="P")
    
    # Add McCormick relaxation constraints
    for i in range(n_points):
        for j in range(n_points):
            constraints.extend([
                P[i, j] >= h[i] + h[j] - 1,
                P[i, j] <= h[i],
                P[i, j] <= h[j],
                P[i, j] >= 0
            ])
    
    # Compute maximum correlation over all shifts
    max_corr = cp.Variable(1, name="max_corr")
    
    # For each shift, compute correlation term
    # We'll compute full correlation for all possible shifts (indices -n+1 to n-1) 
    # and take max
    
    # Simplification: We can compute for k from 0 to n-1 since correlation for negative k 
    # should be the same due to symmetry in this case
    
    # Each correlation corresponds to a sum over overlapping pairs
    correlations = []
    
    for k in range(n_points):
        # Overlap length is n - k (when k > 0)
        # For shift k, we consider terms h_i * (1 - h_{i+k}) for i where both are valid
        # This equals sum_i h_i - sum_{i : i+k < n} h_i * h_{i+k} 
        # = sum_i h_i - sum_{i : i+k < n} P[i, i+k]
        if k == 0:
            corr = cp.sum(h) - 0  # No overlap products for zero shift
        else:
            # Consider only pairs where j = i + k and both indices are valid
            corr = cp.sum(h) - cp.sum([P[i, i+k] for i in range(n_points-k)])
        correlations.append(corr)
    
    # Take maximum over all correlations
    for i, corr in enumerate(correlations):
        constraints.append(max_corr >= corr)
    
    # Objective: minimize the maximum correlation
    objective = cp.Minimize(max_corr)
    
    problem = cp.Problem(objective, constraints)
    
    # Solve using CBC solver (suitable for this problem size)
    try:
        warm_start = True
