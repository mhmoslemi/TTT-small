import numpy as np

# Global variables
TARGET_SUM = None

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr_and_weights(h, eps):
    """Compute correlation and softmax weights for gradient calculation"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    # Stabilize exponentiation
    w_raw = np.exp((corr - max_corr) / eps)
    w = w_raw / np.sum(w_raw)
    return corr, w

    for s in range(-(n-1), n):
        ws = w[s + (n-1)]  # index in w array
        if abs(ws) < 1e-15:
            continue
        if s >= 0:
            # Gradient contribution from positive shifts
            # For i = 0 to n-s-1: h[i+s] exists
            # Add ws to grad[0:n-s] for (1-h[i+s]) part
            if s < n:  # to ensure indexing doesn't go too far
                grad[:n-s] += ws * (1 - h[s:])
            # Subtract ws from grad[s:] for h[i-s] part  
            if s > 0:
                grad[s:] -= ws * h[:n-s]
        else:
            # s < 0; we need to be careful about i starting at min(0, -s) etc
            t = -s  # t > 0
            # Contribution for indexes i where i+s >=0
            # i >= -s => range from max(0, t) to n-1; i from t to n-1 (since t>0)
            if t < n:
                grad[t:] += ws * (1 - h[:n-t])
            # Contribution for h[i-s] = h[i+t], only if i-t >= 0 => i >= t
            # i from 0 to n-1-t (since would be <= n-1 for h[i+t])
            if t < n:
                grad[:n-t] -= ws * h[t:]
                
    return grad

def optimize_h(start_h, n_iters, lr, eps_list, target_sum, best_c5_so_far):
    """Optimize h with gradient descent using a schedule"""
    h = start_h.copy()
    best_h = h.copy()
    c5 = evaluate_c5(h)
    if c5 < best_c5_so_far:
        best_c5_so_far = c5
        best_h = h.copy()
    
    for i, eps in enumerate(eps_list):
        iter_per_eps = max(500, 1000 - i * 200)  # Decreasing iterations
        if iter_per_eps < 100:
            iter_per_eps = 100
        for step in range(iter_per_eps):
            # Compute correlation and weights
            corr, w = compute_corr_and_weights(h, eps)
            # Compute gradient
            grad = compute_gradient(h, w)
            # Update step
            h_new = h - lr * grad
            # Project onto constraints
            h_new = project_to_sum_box(h_new, target_sum)
            h = h_new
            # Evaluate

                if c5 < best_c5_so_far:
                    best_c5_so_far = c5
                    best_h = h.copy()
    return best_h, best_c5_so_far
