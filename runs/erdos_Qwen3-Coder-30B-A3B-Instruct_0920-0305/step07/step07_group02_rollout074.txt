def c5_fft(h):
    n = len(h)
    nfft = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, nfft)
    B = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * B, nfft)[:2*n-1]
    return np.max(conv) * (2.0 / n)

def project_to_sum(h, target_sum):
    """Project h onto the simplex with sum = target_sum"""
    h = np.array(h, dtype=float)
    # Binary search on lambda
    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        proj = np.clip(h + mid, 0, 1)
        s = np.sum(proj)
        if s < target_sum:
            low = mid
        else:
            high = mid
    h_proj = np.clip(h + (low + high) / 2, 0, 1)
    # Renormalize to exact sum
    h_proj *= target_sum / np.sum(h_proj)
    return h_proj

def thue_morse(n):
    """Generate Thue-Morse sequence of length n"""
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    return tm

def rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence of length n"""
    rs = np.zeros(n, dtype=int)
    for i in range(n):
        count = bin(i).count('1')  # count of 1s in binary representation
        # Rudin-Shapiro sequence based on the number of adjacent 1s in binary
        k = 0
        while count > 0:
            if count & 1:
                k += 1
            count >>= 1
        rs[i] = k % 2
    return rs

def greedy_descent(h_start, target_sum):
    h = h_start.copy()
    while True:
        corr = correlate(h, 1 - h, mode='full')
        # Find the shift that maximizes overlap
        max_shift_idx = np.argmax(corr)
        shift = max_shift_idx - (len(h) - 1)
        # Determine mismatched pairs
        A = set()
        B = set()
        if shift >= 0:
            for i in range(len(h) - shift):
                if h[i] == 1 and h[i + shift] == 0:
                    A.add(i)
                elif h[i] == 0 and h[i + shift] == 1:
                    B.add(i)
        else:
            for i in range(-shift, len(h)):
                if h[i] == 1 and h[i + shift] == 0:
                    A.add(i)
                elif h[i] == 0 and h[i + shift] == 1:
                    B.add(i)
        
        if not A or not B:
            break
            
        # Pick a random pair to swap
        i = np.random.choice(list(A))
        j = np.random.choice(list(B))
        h[i], h[j] = 0, 1
        
    return h
