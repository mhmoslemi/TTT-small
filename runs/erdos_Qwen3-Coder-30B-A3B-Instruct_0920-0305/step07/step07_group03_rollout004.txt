import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def insert_peaks(h):
    """Add two sharp peaks to make better overlap suppression"""
    # Locate areas with low values and add two strong spikes
    thresholds = [0.2, 0.5, 0.8]
    for thres in thresholds:
        mask = (h < thres)
        if np.sum(mask) > 10:
            # Set some low values to 1
            indices = np.where(mask)[0]
            for _ in range(min(2, len(indices)//3)):
                idx = np.random.choice(indices)
                h[idx] = 1.0
                # Smooth out nearby values by reducing them slightly
                for offset in [-1, 0, 1]:
                    if 0 <= idx + offset < len(h):
                        h[idx + offset] = max(0, h[idx + offset] - 0.2)
            break
    
    return project_to_sum_box(h, TARGET_SUM)

def local_improvement(h, max_iter=500):
    """Local improvement using gradient-free method"""
    h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Use differential evolution for local search
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(x):
        x_proj = project_to_sum_box(x, TARGET_SUM)
        return evaluate_c5(x_proj)
    
    try:
        res = differential_evolution(
            objective, 
            bounds,
            maxiter=300,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=42
        )
        if res.success:
            h = project_to_sum_box(res.x, TARGET_SUM)
            new_c5 = evaluate_c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
    except:
        pass
    
    # Fine tune with small local changes
    for _ in range(max_iter):
        idxs = np.random.choice(n_points, size=min(5, n_points//8), replace=False)
        improved = False
        for idx in idxs:
            old_val = h[idx]

            h[idx] += delta
            h = project_to_sum_box(h, TARGET_SUM)
            best_c5 = evaluate_c5(h)
    
    return h, best_c5
