import time

# Global variables
TARGET_SUM = None
n_points = 120

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        )

            h = res.x

def gradient_descent_with_momentum(h0, learning_rate=0.01, momentum=0.9, max_iter=1000):
    """Gradient descent with momentum for faster convergence"""
    h = h0.copy()
    velocity = np.zeros_like(h)
    
    for i in range(max_iter):
        # Evaluate gradient using finite differences
        epsilon = 1e-5
        grad = np.zeros_like(h)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += epsilon
            h_plus = project_to_sum_box(h_plus, TARGET_SUM)
            f_plus = evaluate_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[j] -= epsilon
            h_minus = project_to_sum_box(h_minus, TARGET_SUM)
            f_minus = evaluate_c5(h_minus)
            
            grad[j] = (f_plus - f_minus) / (2 * epsilon)
        
        # Update with momentum
        velocity = momentum * velocity - learning_rate * grad
        h = h + velocity
        
        # Project back to feasible region
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Early stopping if improvement is minimal
        if i > 10 and np.allclose(grad, 0, atol=1e-8):
            break
    
    return h

def hybrid_search(h0, max_iter=2000):
    """Combine global and local search strategies"""
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    c5_1 = evaluate_c5(h1)
    if c5_1 < best_c5:
        best_h = h1.copy()
        best_c5 = c5_1
    
    # Phase 2: fine gradient descent
    h2 = gradient_descent_with_momentum(best_h, learning_rate=0.01, momentum=0.9, max_iter=500)
    c5_2 = evaluate_c5(h2)
    if c5_2 < best_c5:
        best_h = h2.copy()
        best_c5 = c5_2

    for _ in range(20):
