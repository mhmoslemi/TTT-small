import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    TARGET_SUM = n_points / 2

    # 1. Original interpolated solution
    initial_guesses.append(h_interpolated.copy())

    h_sine = 0.5 + 0.3 * np.sin(np.linspace(0, 4*np.pi, n_points))
    h_sine = project_to_sum_box(h_sine, TARGET_SUM)
    initial_guesses.append(h_sine)
    
    # 3. Piecewise constant with random heights
    h_piecewise = np.zeros(n_points)
    segments = 6
    seg_size = n_points // segments
    for i in range(segments):

        end = (i+1) * seg_size if i < segments-1 else n_points
        h_piecewise[start:end] = np.random.uniform(0.2, 0.8)
    h_piecewise = project_to_sum_box(h_piecewise, TARGET_SUM)
    initial_guesses.append(h_piecewise)

    x_vals = np.linspace(0, 2, n_points)
    h_bell = np.exp(-10 * (x_vals - 1)**2)
    h_bell = project_to_sum_box(h_bell, TARGET_SUM)
    initial_guesses.append(h_bell)

    for _ in range(3):
        h_pert = h_interpolated + np.random.normal(0, 0.03, n_points)

        initial_guesses.append(h_pert)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Try optimization from each initial point
    for i, h_start in enumerate(initial_guesses):
        if time.time() - start_time > budget_s:

        try:

                objective,

        neighbor_h = current_h + np.random.normal(0, T, n_points)

            stagnation_count = 0
        else:
            # Accept with probability
            delta = neighbor_c5 - current_c5
            if np.random.rand() < np.exp(-delta / T):
                current_h = neighbor_h
                current_c5 = neighbor_c5
                stagnation_count = 0
            else:
                stagnation_count += 1
        
        # Cool down
        T *= alpha
        
        # Early stopping if no improvement for a while
        if stagnation_count > stagnation_limit:
            break
    
    return (best_h.tolist(), float(best_c5), n_points)
