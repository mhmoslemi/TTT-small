import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def heuristic_initialization():
    """Create a heuristic initial solution based on known good patterns"""
    # Create a piecewise linear function that's a good starting point
    h = np.zeros(n_points)
    
    # Simple 3-segment construction
    segments = [0, 0.3, 0.7, 1.0]
    values = [0, 0.8, 0.8, 0]
    
    x = np.linspace(0, 2, n_points)
    for i in range(len(segments)-1):
        start_idx = int(segments[i] * n_points / 2)
        end_idx = int(segments[i+1] * n_points / 2)
        h[start_idx:end_idx] = values[i]
    
    h = np.clip(h, 0, 1)
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def genetic_algorithm_search():
    """Use genetic algorithm with small population"""
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h = heuristic_initialization()
        h = np.clip(h + np.random.normal(0, 0.02, n_points), 0, 1)
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_c5(h) for h in population]
    
    for generation in range(generations):
        # Select top 50%
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:population_size // 2]
        survivors = [population[i] for i in top_indices]
        fitness_values = [fitness[i] for i in top_indices]
        
        # Create new population via crossover and mutation
        new_population = survivors[:]
        
        # Add offspring
        while len(new_population) < population_size:
            # Tournament selection
            a_idx = np.random.choice(len(survivors))
            b_idx = np.random.choice(len(survivors))
            
            parent_a = survivors[a_idx]
            parent_b = survivors[b_idx]
            
            # Crossover (uniform)
            mask = np.random.rand(n_points) > 0.5
            child = np.where(mask, parent_a, parent_b)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation_idx = np.random.randint(0, n_points)
                child[mutation_idx] = np.random.rand()
                
            child = np.clip(child, 0, 1)
            child = project_to_sum_box(child, TARGET_SUM)
            new_population.append(child)
            
        population = new_population[:population_size]
        fitness = [evaluate_c5(h) for h in population]
        
    # Return best individual
    best_idx = np.argmin(fitness)
    return population[best_idx]

def local_refinement(h0, max_iter=500):
    """Refine using gradient-based approach on a small region"""
    h = h0.copy()
    corr_full = compute_corr_fft(h)
    max_idx = np.argmax(corr_full)
    
    # Focus on a localized region
    start = max(0, max_idx - 15)
    end = min(len(h), max_idx + 15)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_local):
        return evaluate_c5(h_local)
    
    def constraint_eq(h_local):
        return np.sum(h_local) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use fewer iterations and local optimizer
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
    except:
        pass
    
    # Project to maintain constraints
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = genetic_algorithm_search()
    best_c5 = evaluate_c5(best_h)
    
    # Local refinement 
    best_h = local_refinement(best_h, max_iter=300)
    best_c5 = evaluate_c5(best_h)
    
    # Final exploration phase 
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        iter_count = 0
        while time.time() - start_time < budget_s - 5:
            # Perturb and refine
            h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
            h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
            h_refined = local_refinement(h_perturbed, max_iter=200)
            c5_refined = evaluate_c5(h_refined)
            
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
                
            iter_count += 1
    
    return (best_h.tolist(), float(best_c5), n_points)
