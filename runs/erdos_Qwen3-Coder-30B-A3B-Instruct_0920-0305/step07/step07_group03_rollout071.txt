import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_of_corr_at_lag(h, lag):
    """Compute gradient of correlation at specific lag"""
    n = len(h)
    g = np.ones(n)
    
    if lag >= 0:
        if lag < n:
            g[:n-lag] = 1 - h[lag:]  
            g[lag:] -= h[:n-lag]
    else:  # lag < 0
        d = -lag
        if d < n:
            g[d:] = 1 - h[:n-d]
            g[:n-d] -= h[d:]
    
    return g

def gradient_descent_refinement(h0, max_iter=1000, step_size=0.1):
    """Refine solution using gradient descent"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        corr = compute_corr(h)
        max_idx = np.argmax(corr)
        max_lag = max_idx - (n_points - 1)
        
        g = gradient_of_corr_at_lag(h, max_lag)
        
        h_new = h - step_size * g
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            step_size = min(0.5, step_size * 1.1)  # Increase step slightly
        else:
            step_size *= 0.9  # Decrease step size
            
        if step_size < 1e-6:
            break
    
    return h, best_c5

def simulated_annealing_binary(max_time, initial_h=None):
    """Simulated Annealing on binary patterns"""
    if initial_h is None:
        h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
        # Fix to have exactly target sum
        ones_indices = np.where(h == 1)[0]
        zeros_indices = np.where(h == 0)[0]
        num_ones_to_fix = TARGET_SUM - np.sum(h)
        
        if num_ones_to_fix > 0:
            # Need to convert some zeros to ones
            for _ in range(int(num_ones_to_fix)):
                if len(zeros_indices) > 0:
                    idx = np.random.choice(zeros_indices)
                    h[idx] = 1
                    zeros_indices = np.delete(zeros_indices, np.where(zeros_indices == idx))
        elif num_ones_to_fix < 0:
            # Need to convert some ones to zeros
            for _ in range(int(-num_ones_to_fix)):
                if len(ones_indices) > 0:
                    idx = np.random.choice(ones_indices)
                    h[idx] = 0
                    ones_indices = np.delete(ones_indices, np.where(ones_indices == idx))
    else:
        h = initial_h.copy()
        h = project_to_sum_box(h, TARGET_SUM)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Initialize temperature
    start_time = time.time()
    elapsed = 0
    
    while elapsed < max_time:
        # Perturb solution by swapping two bits
        idx1 = np.random.choice(np.where(h == 1)[0])
        idx2 = np.random.choice(np.where(h == 0)[0])
        
        h_test = h.copy()
        h_test[idx1] = 0
        h_test[idx2] = 1
        
        c5_test = evaluate_c5(h_test)
        
        if c5_test < best_c5:
            h = h_test.copy()
            best_h = h.copy()
            best_c5 = c5_test
        else:
            # Accept with probability
            delta = c5_test - best_c5
            if delta > 0:
                acceptance_prob = np.exp(-delta / (max_time * 1e-4 + 1e-5))
                if np.random.random() < acceptance_prob:
                    h = h_test.copy()
        
        elapsed = time.time() - start_time
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Phase 1: Simulated Annealing on binary patterns
    best_binary_h = None
    best_binary_c5 = float('inf')
    
    for run_id in range(10):
        if time.time() - start_time > 0.5 * budget_s:
            break
            
        try:
            h, c5 = simulated_annealing_binary(0.05 * budget_s)
            if c5 < best_binary_c5:
                best_binary_h = h.copy()
                best_binary_c5 = c5
        except:
            continue
    
    # Phase 2: Refinement with gradient descent
    if best_binary_h is not None:
        try:
            refined_h, refined_c5 = gradient_descent_refinement(best_binary_h, max_iter=2000)
            if refined_c5 < best_binary_c5:
                best_h = refined_h
                best_c5 = refined_c5
            else:
                best_h = best_binary_h
                best_c5 = best_binary_c5
        except:
            best_h = best_binary_h
            best_c5 = best_binary_c5
    else:
        # Fallback: random solution
        h = np.random.random(n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        best_h = h
        best_c5 = evaluate_c5(h)
    
    # Phase 3: Final refinement with local search
    final_time = time.time()
    while time.time() - final_time < 0.3 * budget_s:
        # Small perturbations
        h_test = best_h + np.random.normal(0, 0.005, n_points)
        h_test = project_to_sum_box(h_test, TARGET_SUM)
        c5_test = evaluate_c5(h_test)
        
        if c5_test < best_c5:
            best_h = h_test
            best_c5 = c5_test
    
    return (best_h.tolist(), float(best_c5), n_points)
