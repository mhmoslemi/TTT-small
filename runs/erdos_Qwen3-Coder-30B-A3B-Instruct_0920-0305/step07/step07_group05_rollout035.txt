import numpy as np
import cvxpy as cp
from scipy.signal import correlate
from scipy.linalg import eigh
import time

def build_autocorrelation_matrices(n):
    """Build sparse matrices A_ell for each lag."""
    A_list = []
    for ell in range(-(n-1), n):
        rows, cols = [], []
        for i in range(n):
            j = i + ell
            if 0 <= j < n:
                rows.append(i)
                cols.append(j)
        if rows:
            A = np.zeros((n, n))
            A[rows, cols] = 1
            A_list.append((ell, A))
    return A_list

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr)
