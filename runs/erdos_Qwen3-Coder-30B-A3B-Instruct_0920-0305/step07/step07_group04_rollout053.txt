import numpy as np

import time

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT correlation"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_search(h0, max_iter=2000):
    """Local search with random perturbations"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Make small perturbations
        h_new = h.copy()
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        delta = np.random.normal(0, 0.02, 2)
        h_new[idx1] += delta[0]
        h_new[idx2] += delta[1]
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            h = h_new.copy()
            best_h = h.copy()
            best_c5 = new_c5
            
    return best_h, best_c5

            fitness.append(c5)

        # Tournament selection
        selected = []
        for _ in range(pop_size):
            idx1, idx2 = np.random.choice(pop_size, 2, replace=False)
            if fitness[idx1] < fitness[idx2]:
                selected.append(population[idx1].copy())
            else:
                selected.append(population[idx2].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i+1) % pop_size]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points-1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutation
            for child in [child1, child2]:

                    child = np.clip(child, 0, 1)
                child = project_to_sum_box(child, TARGET_SUM)
                new_population.append(child)
        
        population = new_population
    
    return best_h, best_c5
