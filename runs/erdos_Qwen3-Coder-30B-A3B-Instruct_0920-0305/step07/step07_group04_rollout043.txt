import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_descent_with_projection(h0, max_iter=2000):
    """Optimize using gradient descent with projection"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Learning rate schedule
    lr = 0.01
    for i in range(max_iter):
        # Simple gradient approximation via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_to_sum_box(h_plus, TARGET_SUM)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = project_to_sum_box(h_minus, TARGET_SUM)
            c5_minus = evaluate_c5(h_minus)
            
            grad[j] = (c5_plus - c5_minus) / (2 * eps)
        
        # Gradient step
        h_new = h - lr * grad
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Update parameters
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new.copy()
        
        # Reduce learning rate if stuck
        if i % 100 == 0 and i > 0:
            lr *= 0.95
    
    return best_h, best_c5

def generate_orthogonal_initial():
    """Create a more structured initial solution that can potentially improve convergence"""
    # Start with piecewise constant function with some variation
    h = np.zeros(n_points)
    
    # Create a few segments with different heights
    segments = 4
    segment_width = n_points // segments
    
    for i in range(segments):
        start = i * segment_width
        end = (i + 1) * segment_width if i < segments - 1 else n_points
        if i % 2 == 0:
            h[start:end] = 0.7
        else:
            h[start:end] = 0.3
    
    # Rebalance to meet sum constraint
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def local_search_improvement(h0, max_iter=800):
    """Refine solution using local neighborhood search"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for _ in range(max_iter):
        # Try small modifications
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        # Perturb one element
        delta = np.random.uniform(-0.1, 0.1)
        h_new[idx] += delta
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            h = h_new.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    best_h = generate_orthogonal_initial()
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: Local improvement with structured initialization
    phase1_end = start_time + 0.3 * budget_s
    while time.time() < phase1_end:
        h_improved, c5_improved = local_search_improvement(best_h, max_iter=200)
        if c5_improved < best_c5:
            best_h = h_improved
            best_c5 = c5_improved
    
    # Phase 2: Gradient descent with projection  
    phase2_end = start_time + 0.6 * budget_s
    while time.time() < phase2_end:
        h_refined, c5_refined = gradient_descent_with_projection(best_h, max_iter=500)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    
    # Phase 3: Final local search with gradient start
    phase3_end = start_time + budget_s
    while time.time() < phase3_end:
        h_final, c5_final = local_search_improvement(best_h, max_iter=300)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    
    return (best_h.tolist(), float(best_c5), n_points)
