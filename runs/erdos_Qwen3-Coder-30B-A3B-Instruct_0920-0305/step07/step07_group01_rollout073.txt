from scipy.signal import correlate
import time

# Global parameters
TARGET_SUM = None
n_points = 120

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_gradient(h, corr, beta):
    """Compute gradient of smooth max approximation"""
    # Compute weights for soft-max
    max_corr = np.max(corr)
    corr_shifted = corr - max_corr
    exp_corr = np.exp(beta * corr_shifted)
    weights = exp_corr / np.sum(exp_corr)
    
    # Convolution with weights
    grad = np.zeros_like(h)
    n = len(h)
    
    # For each position, we compute the weighted sum of all shifted values
    for i in range(n):
        for k in range(-n+1, n):
            j = i + k
            if 0 <= j < n:
                grad[i] += weights[k + n - 1] * (1 - h[j])
                
    return dx * (np.ones_like(h) - grad)

def max_lag_gradient(h, corr, worst_lag):
    """Compute exact sub-gradient at worst lag"""
    n = len(h)
    grad = np.zeros_like(h)
    
    # Lag indices offset correctly (full correlation)
    lag_offset = worst_lag - (n - 1)
    
    if lag_offset >= 0:
        # Positive shift
        for i in range(n - lag_offset):
            grad[i] += (1 - h[i + lag_offset]) * dx
        for i in range(lag_offset, n):
            grad[i] -= h[i - lag_offset] * dx
    else:
        # Negative shift
        for i in range(-lag_offset, n):
            grad[i] += (1 - h[i + lag_offset]) * dx
        for i in range(n + lag_offset):
            grad[i] -= h[i - lag_offset] * dx
            
    return grad

def mass_transfer(h, grad, epsilon):
    """Perform greedy mass transfer to improve solution"""
    d = -grad  # gradient direction for descent
    n = len(h)
    
    # Sort sources by decreasing negative d and sinks by increasing d
    sources = [(i, d[i]) for i in range(n) if d[i] < -1e-10]
    sinks = [(i, d[i]) for i in range(n) if d[i] > 1e-10]
    
    sources.sort(key=lambda x: x[1])  # Increasing order of d (most negative first)
    sinks.sort(key=lambda x: -x[1])   # Decreasing order of d (most positive first)
    
    h_new = h.copy()
    
    for i, _ in sources:
        if epsilon <= 0:
            break
        for j, _ in sinks:
            if epsilon <= 0:
                break
            transfer = min(h_new[i], 1 - h_new[j], epsilon)
            if transfer > 0:
                h_new[i] -= transfer
                h_new[j] += transfer
                epsilon -= transfer
                
    # Re-project
    h_new = project_sum_box(h_new, TARGET_SUM)
    return h_new

def backtracking_line_search(h, grad, current_val, alpha0=0.2, alpha_min=5e-4, max_iter=10):
    """Backtracking line search for descent"""
    alpha = alpha0
    n = len(h)
    
    for _ in range(max_iter):
        h_new = h - alpha * grad
        h_new = project_sum_box(h_new, TARGET_SUM)
        new_val = evaluate_c5(h_new)
        
        if new_val < current_val:
            return h_new, new_val, alpha
        alpha /= 2
        if alpha < alpha_min:
            break
            
    return h, current_val, alpha
