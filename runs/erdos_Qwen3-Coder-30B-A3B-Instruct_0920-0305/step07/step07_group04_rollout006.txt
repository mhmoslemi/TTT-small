import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    h0 = adaptive_sampling_initialization()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Phase 1: Multiple gradient optimizations with different starting points
    phase1_end = start_time + 0.4 * budget_s
    for run_id in range(20):
        if time.time() > phase1_end:
            break
            
        # Try different initialization strategies
        initial_modes = [
            adaptive_sampling_initialization,
            lambda: np.random.uniform(0, 1, n_points),
            lambda: np.random.choice([0, 1], size=n_points, p=[0.5, 0.5]),
            lambda: np.ones(n_points) * 0.5
        ]
        h_start = initial_modes[run_id % len(initial_modes)]()
        h_start = project_to_sum_box(h_start, TARGET_SUM)
        
        # Simple but effective optimization approach
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
            
            res = minimize(
                evaluate_c5, 
                h_start, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_refined = res.x
                h_refined = project_to_sum_box(h_refined, TARGET_SUM)
                c5_refined = evaluate_c5(h_refined)
                
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
        except Exception as e:
            pass

    # Phase 2: Local search with different neighborhood types
    phase2_end = start_time + 0.7 * budget_s
    while time.time() < phase2_end:
        # Try perturbation using different strategies
        h_candidate = best_h.copy()
        
        # Random perturbations of different magnitudes
        mag = 0.05
        idx = np.random.randint(0, n_points)
        h_candidate[idx] += np.random.uniform(-mag, mag)
        
        # Apply multiple random steps
        for _ in range(5):
            idx = np.random.randint(0, n_points)
            h_candidate[idx] += np.random.uniform(-mag, mag)
            
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
        
        c5_candidate = evaluate_c5(h_candidate)
        
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            
    # Phase 3: Global search with stochastic sampling
    phase3_end = start_time + budget_s
    while time.time() < phase3_end:
        # Random stochastic sampling around current best
        h_new = best_h + np.random.normal(0, 0.02, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
    
    return (best_h.tolist(), float(best_c5), n_points)
