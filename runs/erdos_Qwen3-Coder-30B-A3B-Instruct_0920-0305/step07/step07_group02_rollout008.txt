import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def heuristic_initialization():
    """Create a heuristic initial solution based on known good patterns"""
    # Start with known good values from earlier work
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h_base = np.array(initial_h_values)
        x_base = np.linspace(0, 2, len(h_base))
        x_new = np.linspace(0, 2, n_points)
        h_interp = np.interp(x_new, x_base, h_base)
    else:
        # Simple heuristic if no initial values available
        h_interp = np.ones(n_points) * 0.5
    
    # Add small perturbations
    h_perturbed = h_interp + np.random.normal(0, 0.02, n_points)
    h_perturbed = np.clip(h_perturbed, 0, 1)
    h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
    return h_perturbed

def pattern_based_search(h0, max_iter=200):
    """Improves using pattern-based neighborhood search with some randomness"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Try to modify several segments around the worst overlap
        corr_full = compute_corr_fft(h)
        max_corr_idx = np.argmax(corr_full)
        
        # Modify a broader range around the worst point
        start = max(0, max_corr_idx - 20)
        end = min(len(h), max_corr_idx + 20)
        
        # Randomly select a small region and perturb it
        r_start = np.random.randint(max(0, end - 10), end)
        r_end = min(len(h), r_start + 5)
        
        # Try several modifications
        for _ in range(5):
            sign = np.random.choice([-1, 1])
            delta = 0.01 * sign
            
            h_new = h.copy()
            h_new[r_start:r_end] += delta
            # Ensure we stay in bounds
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5:
                h = h_new.copy()
                best_c5 = new_c5
                corr_full = compute_corr_fft(h)
    
    return h

def adaptive_local_search(h0, max_iter=100):
    """Adaptive local search with varying step sizes and depths"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    step_sizes = [0.01, 0.005, 0.002, 0.001]
    
    for i in range(max_iter):
        # Randomly select a step size
        step_size = np.random.choice(step_sizes)
        
        # Perturb a small random region
        start = np.random.randint(0, n_points - 10)
        end = min(start + 10, n_points)
        
        # Try a random sign
        sign = np.random.choice([-1, 1])
        delta = step_size * sign
        
        h_new = h.copy()
        h_new[start:end] += delta
        # Ensure we stay in bounds
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = heuristic_initialization()
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: Pattern-based search
    phase1_end = start_time + 0.4 * budget_s
    for run_id in range(20):
        if time.time() > phase1_end:
            break
            
        h_refined = pattern_based_search(best_h.copy(), max_iter=200)
        c5_refined = evaluate_c5(h_refined)
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    # Phase 2: Adaptive local search
    phase2_end = start_time + 0.7 * budget_s
    for run_id in range(30):
        if time.time() > phase2_end:
            break
            
        h_adapt = adaptive_local_search(best_h.copy(), max_iter=150)
        c5_adapt = evaluate_c5(h_adapt)
        
        if c5_adapt < best_c5:
            best_h = h_adapt.copy()
            best_c5 = c5_adapt
    
    # Phase 3: Final diversity attempts
    phase3_end = start_time + budget_s
    attempts = 0
    while time.time() < phase3_end and attempts < 20:
        attempts += 1
        # Try a different random initialization with better base
        h_new = heuristic_initialization()
        # Do some pattern refinement around worst correlation
        h_refined = pattern_based_search(h_new, max_iter=100)
        c5_refined = evaluate_c5(h_refined)
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    return (best_h.tolist(), float(best_c5), n_points)
