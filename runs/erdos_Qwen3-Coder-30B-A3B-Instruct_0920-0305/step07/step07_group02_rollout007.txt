```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

# Global variables
N = 120
dx = 2.0 / N
TARGET_SUM = N // 2

def compute_c5_fft(h):
    """Fast evaluation of C5 using FFT"""
    corr_full = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr_full)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def greedy_descent(h0):
    """Greedy descent on the worst shift"""
    h = h0.copy()
    best_c5 = compute_c5_fft(h)
    
    for _ in range(1000):
        # Compute correlations
        corr_full = np.correlate(h, h, mode='full')
        # Find worst shift
        max_idx = np.argmax(corr_full)
        worst_shift = max_idx - (len(h) - 1)
        # Compute positions that contribute to overlap
        A = np.where((h == 1) & (np.roll(h, -worst_shift) == 0))[0]
        B = np.where((h == 0) & (np.roll(h, -worst_shift) == 1))[0]
        if len(A) == 0 or len(B) == 0:
            break
        # Pick a random swap
        i = np.random.choice(A)
        j = np.random.choice(B)
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5_fft(h)
        if new_c5 >= best_c5:
            # Revert if worse
            h[i], h[j] = h[j], h[i]
        else:
            best_c5 = new_c5
    
    return h, best_c5

def anneal_binary(h0, time_budget_s):
    """Simple simulated annealing on binary vectors"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5_fft(h)
    
    start_time = time.time()
    T0 = 0.02
    Tmax = 5000
    t = 0
    
    while time.time() - start_time < time_budget_s:
        # Temperature
        T = T0 * (1 - t / Tmax) if t < Tmax else 0
        
        # Make a move (swap a 1 and a 0)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            continue
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Store old state
        old_h = h.copy()
        old_c5 = compute_c5_fft(h)
        
        # Make the move
        h[i], h[j] = h[j], h[i]
        
        # Compute change
        new_c5 = compute_c5_fft(h)
        delta = new_c5 - old_c5
        
        # Accept or reject
        if delta < 0 or (T > 0 and np.random.rand() < np.exp(-delta / T)):
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        else:
            h = old_h.copy()
            
        t += 1
        
        # Occasionally do a greedy sweep
        if t % 200 == 0:
            h, c5 = greedy_descent(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                
    return best_h, best_c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Solve the MILP
    try:
        # Create CVXPY variables
        h = cp.Variable(N, boolean=True)
        lam = cp.Variable(integer=True)
        
        # Constraints
        constraints = [cp.sum(h) == TARGET_SUM]
        
        # Create w variables for each valid shift
        w_vars = {}  # w[i, k]
        
        for k in range(-(N - 1), N):
            # Valid indices for shift k
            valid_i = np.arange(max(0, -k), min(N, N - k))
            if len(valid_i) == 0:
                continue
            # Create auxiliary variables
            w = cp.Variable(len(valid_i), boolean=True)
            # Linearization constraints
            i_vals = valid_i
            j_vals = i_vals + k
            constraints.extend([
                w <= h[i_vals],
                w <= h[j_vals],
                w >= h[i_vals] + h[j_vals] - 1
            ])
            # Constraint: sum of w >= lambda
            constraints.append(cp.sum(w) >= lam)
            # Store for later use
            w_vars[k] = (w, i_vals, j_vals)
        
        # Add symmetry breaking
        constraints.append(h[0] == 1)
        
        # Objective
        objective = cp.Maximize(lam)
        
        # Solve the problem
        problem = cp.Problem(objective, constraints)
        
        # Solve with time limit
        remaining_time = budget_s * 0.55
        problem.solve(solver=cp.CBC, max_seconds=remaining_time, verbose=False)
        
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_val = np.round(h.value).astype(int)
            # Repair the sum if needed
            current_sum = np.sum(h_val)
            if current_sum != TARGET_SUM:
                # Find some 0s and 1s to swap
                zeros = np.where(h_val == 0)[0]
                ones = np.where(h_val == 1)[0]
                n_swap = abs(current_sum - TARGET_SUM)
                if len(zeros) >= n_swap and len(ones) >= n_swap:
                    for i in range(n_swap):
                        h_val[zeros[i]] = 1
                        h_val[ones[i]] = 0
            # Get λ value
            lambda_val = int(lam.value)
            c5_bound = 1 - (2 * lambda_val) / N
        else:
            # Fallback to heuristic
            h_val = np.random.randint(0, 2, N)
            # Project to correct sum
            h_val = project_to_sum_box(h_val, TARGET_SUM)
            c5_bound = compute_c5_fft(h_val)
            
    except Exception as e:
        # If everything fails, use heuristic approach
        h_val = np.random.randint(0, 2, N)
        h_val = project_to_sum_box(h_val, TARGET_SUM)
        c5_bound = compute_c5_fft(h_val)
    
    # Greedy refinement
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.75 * budget_s:
        h_val, c5_bound = greedy_descent(h_val)
    
    # Optional simulated annealing
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 30:
        h_val, c5_bound = anneal_binary(h_val, remaining_time)
    
    return (h_val.tolist(), float(c5_bound), N)
```
