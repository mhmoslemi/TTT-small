import numpy as np
from scipy.linalg import eigh
from cvxpy import *
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_A_matrices(n):
    """Generate all sparse A_ell matrices for lags -n+1 to n-1"""
    A_list = []
    for ell in range(-(n-1), n):
        A = np.zeros((n, n))
        if ell >= 0:
            A[np.arange(n-ell), np.arange(ell, n)] = 1
        else:
            A[np.arange(-ell, n), np.arange(n+ell)] = 1
        A_list.append(A)
    return A_list

def solve_sdp(n_points, A_list, dx, time_limit_s):
    """Solve the SDP relaxation to get a lower bound"""
    start_time = time.time()
    # Variables
    X = Semidef(n_points, name='X')
    t = Variable(name='t')
    
    # Constraints
    constraints = [diag(X) >= 0, diag(X) <= 1, sum(diag(X)) == n_points/2]
    
    # Add constraints for all lags
    for A in A_list:
        constraints.append(trace(A @ X) >= n_points/2 - t/dx)
    
    # Objective
    objective = Minimize(t)
    
    # Problem
    prob = Problem(objective, constraints)
    
    # Solve with SCS
    prob.solve(solver=SCS, timeout=time_limit_s, verbose=False)
    
    end_time = time.time()
    if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
        return X.value, t.value, end_time - start_time
    else:
        return None, None, end_time - start_time

def round_and_improve(X_opt, n_points, dx, n_samples=2000):
    """Sample from the SDP solution and refine"""
    # Extract eigenvalues and eigenvectors
    eigvals, eigvecs = eigh(X_opt)
    v = eigvecs[:, -1]  # Largest eigenvector
    
    best_h = None
    best_c5 = np.inf
    
    # Try multiple random samples
    for _ in range(n_samples):
        # Sample y from N(0, X_opt) using eigendecomposition
        y = eigvecs @ (np.sqrt(np.abs(eigvals)) * np.random.randn(n_points))
        h = np.clip(y, 0, 1)
        h = project_to_sum_box(h, n_points/2)
        
        # Compute C5
        corr = np.correlate(h, 1-h, mode='full')
        c5 = dx * np.max(corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def refine_solution(h, n_points, dx, budget_s):
    """Run a short gradient-guided local refinement"""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx
    
    # Simple local move: move mass between indices with high/low gradients
    while time.time() - start_time < budget_s:
        # Compute local gradients
        grad = np.zeros(n_points)
        eps = 1e-3
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_sum_box(h_plus, n_points/2)
            c5_plus = np.max(np.correlate(h_plus, 1-h_plus, mode='full')) * dx
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_sum_box(h_minus, n_points/2)
            c5_minus = np.max(np.correlate(h_minus, 1-h_minus, mode='full')) * dx
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Find index with max gradient (most increasing)
        idx_max = np.argmax(grad)
        # Find index with min gradient (least increasing)
        idx_min = np.argmin(grad)
        
        if grad[idx_min] < grad[idx_max]:
            delta = min(h[idx_min], 1 - h[idx_max], 0.01)
            if delta > 1e-6:
                h[idx_min] -= delta
                h[idx_max] += delta
                h = project_to_sum_box(h, n_points/2)
                c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        else:
            i = np.random.choice(n_points)
            j = np.random.choice(n_points)
            if i != j:
                delta = min(h[i], 1 - h[j], 0.005)
                if delta > 1e-6:
                    h[i] -= delta
                    h[j] += delta
                    h = project_to_sum_box(h, n_points/2)
                    c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
    
    return best_h, best_c5
