import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def thue_morse_sequence(length):
    """Generate Thue-Morse sequence up to given length"""
    if length <= 0:
        return []
    seq = [0]
    while len(seq) < length:
        seq.extend([1 - x for x in seq])
    return seq[:length]

def rudin_shapiro_sequence(length):
    """Generate Rudin-Shapiro sequence up to given length"""
    if length <= 0:
        return []
    seq = [1]
    for i in range(1, length):
        if i % 2 == 0:
            seq.append(seq[i//2])
        else:
            seq.append(seq[i//2] * (-1)**((i-1)//2))
    return [1 if x > 0 else 0 for x in seq[:length]]

def binary_to_fractional(h_binary):
    """Convert binary to fractional version"""
    return h_binary.astype(float)

def constraint_sum(h):
    return np.sum(h) - TARGET_SUM

def local_search_greedy(h0, max_iter=5000):
    """Apply greedy local search focusing on worst shift"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    corr_full = compute_corr_fft(h)
    max_shift = np.argmax(corr_full)
    
    for i in range(max_iter):
        if i % 500 == 0:
            corr_full = compute_corr_fft(h)
            max_shift = np.argmax(corr_full)
            
        # Find indices for swap: (1,0) and (0,1) for max_shift
        h_shifted = h[max_shift:]  # h_k
        h_cut = h[:-max_shift]    # h_0
        
        # Find when (h_i, h_{i+k}) = (1, 0)
        ones_zeroes = np.where((h_cut == 1) & (h_shifted == 0))[0]
        # Find when (h_j, h_{j+k}) = (0, 1)
        zeroes_ones = np.where((h_cut == 0) & (h_shifted == 1))[0]
        
        if len(ones_zeroes) == 0 or len(zeroes_ones) == 0:
            # No good swaps available, do different shift
            other_shifts = np.argsort(corr_full)[-10:]  # Top 10 correlations
            found = False
            for new_shift in other_shifts:
                if new_shift == max_shift:
                    continue
                h_shifted_new = h[new_shift:]
                h_cut_new = h[:-new_shift]
                ones_zeroes_new = np.where((h_cut_new == 1) & (h_shifted_new == 0))[0]
                zeroes_ones_new = np.where((h_cut_new == 0) & (h_shifted_new == 1))[0]
                if len(ones_zeroes_new) > 0 and len(zeroes_ones_new) > 0:
                    max_shift = new_shift
                    ones_zeroes = ones_zeroes_new
                    zeroes_ones = zeroes_ones_new
                    found = True
                    break
            if not found:
                break
            
        # Choose a random swap pair
        idx1 = random.choice(ones_zeroes)
        idx2 = random.choice(zeroes_ones)
        
        # Perform swap
        if idx1 != idx2:
            h[idx1], h[idx2] = h[idx2], h[idx1]
            new_c5 = evaluate_c5(h)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
            else:
                # Restore if no improvement
                h[idx1], h[idx2] = h[idx2], h[idx1]
    
    return h

def anneal_with_temperature_schedule(h0, max_iter=5000, temp_start=1e-2, temp_end=1e-5):
    """Perform simulated annealing"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    current_c5 = best_c5
    
    def acceptance_probability(old_c5, new_c5, temp):
        if new_c5 < old_c5:
            return 1.0
        return np.exp((old_c5 - new_c5) / temp) if temp > 0 else 0.0
    
    for i in range(max_iter):
        # Cool down
        temp = temp_start * (temp_end / temp_start) ** (i / max_iter)
        if temp <= temp_end:
            break
        
        # Propose a swap in valid range
        idx1 = random.randint(0, n_points - 1)
        idx2 = random.randint(0, n_points - 1)
        if idx1 == idx2:
            continue
            
        # Swap and evaluate
        h[idx1], h[idx2] = h[idx2], h[idx1]
        new_c5 = evaluate_c5(h)
        
        if new_c5 < current_c5:
            # Accept
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        else:
            # Accept with probability
            if random.random() > acceptance_probability(current_c5, new_c5, temp):
                # Reject
                h[idx1], h[idx2] = h[idx2], h[idx1]
            else:
                current_c5 = new_c5
    
    return best_h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Prepare candidates from known sequences and configurations
    initial_vectors = []
    
    # Binary from Thue-Morse
    tm_seq = thue_morse_sequence(n_points)
    h_tm = np.array(tm_seq, dtype=float)
    h_tm = project_to_sum_box(h_tm, TARGET_SUM)
    initial_vectors.append(h_tm)
    
    # Binary from Rudin-Shapiro
    rs_seq = rudin_shapiro_sequence(n_points)
    h_rs = np.array(rs_seq, dtype=float)
    h_rs = project_to_sum_box(h_rs, TARGET_SUM)
    initial_vectors.append(h_rs)
    
    # Random binary start
    h_rand = np.random.randint(0, 2, n_points).astype(float)
    h_rand = project_to_sum_box(h_rand, TARGET_SUM)
    initial_vectors.append(h_rand)
    
    # Use initial_h_values if available
    try:
        initial_h_values_np = np.array(initial_h_values, dtype=float)
        # Interpolate to current resolution
        x_old = np.linspace(0, 2, len(initial_h_values))
        x_new = np.linspace(0, 2, n_points)
        h_interpolated = np.interp(x_new, x_old, initial_h_values_np)
        h_interpolated = project_to_sum_box(h_interpolated, TARGET_SUM)
        initial_vectors.append(h_interpolated)
    except:
        pass
    
    # Optimized restart loop
    best_h = None
    best_c5 = float('inf')
    
    restart_count = 0
    total_restarts = 300  # Allowing 300 restarts
    
    while time.time() - start_time < 0.8 * budget_s and restart_count < total_restarts:
        # Select candidate
        if len(initial_vectors) > 0:
            h_init = random.choice(initial_vectors)
        else:
            h_init = np.random.rand(n_points)
            h_init = project_to_sum_box(h_init, TARGET_SUM)
        
        # Phase 1: Greedy local search
        h_greedy = local_search_greedy(h_init, max_iter=1000)
        c5_greedy = evaluate_c5(h_greedy)
        
        # Phase 2: Annealing phase
        h_anneal = anneal_with_temperature_schedule(h_greedy, max_iter=2000)
        c5_anneal = evaluate_c5(h_anneal)
        
        # Phase 3: Final deterministic refinement
        def objective(h_local):
            return evaluate_c5(h_local)
        
        def constraint_eq(h_local):
            return np.sum(h_local) - TARGET_SUM
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        try:
            res = minimize(
                objective, 
                h_anneal, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_final = res.x
                h_final = project_to_sum_box(h_final, TARGET_SUM)
                c5_final = evaluate_c5(h_final)
                if c5_final < c5_anneal:
                    c5_anneal = c5_final
                    h_anneal = h_final
        except:
            pass
        
        if c5_anneal < best_c5:
            best_c5 = c5_anneal
            best_h = h_anneal.copy()
            
        restart_count += 1
    
    # Final refinement using SLSQP directly if time permits
    if time.time() - start_time < 0.95 * budget_s:
        try:
            c5_final_ref = evaluate_c5(best_h)
            bounds_final = [(0, 1) for _ in range(n_points)]
            cons_final = {'type': 'eq', 'fun': constraint_eq}
            res_final = minimize(
                lambda x: evaluate_c5(x),
                best_h,
                method='SLSQP', 
                bounds=bounds_final, 
                constraints=cons_final,
                options={'maxiter': 1000, 'ftol': 1e-15, 'gtol': 1e-15}
            )
            if res_final.success:
                h_refined = res_final.x
                h_refined = project_to_sum_box(h_refined, TARGET_SUM)
                c5_refined = evaluate_c5(h_refined)
                if c5_refined < c5_final_ref:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
