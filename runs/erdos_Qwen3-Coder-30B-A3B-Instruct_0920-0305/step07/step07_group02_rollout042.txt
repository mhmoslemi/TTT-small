import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_thue_morse(n):
    """Generate Thue-Morse sequence approximately"""
    # We build a small sequence and repeat to get length n
    tm = [0, 1]
    while len(tm) < n:
        next_block = [1 - x for x in tm]
        tm.extend(next_block)
    return np.array(tm[:n], dtype=float)

def generate_rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence approximately"""
    rs = [1]
    for i in range(1, n):
        # Rudin-Shapiro is hard to compute in closed form, 
        # so let's just generate a simple version that's good
        # For now, use a switch pattern that might be good
        rs.append(1 if (i % 4) < 2 else 0)
    return np.array(rs[:n], dtype=float)

def find_worst_shift(h):
    """Find the shift that creates maximum overlap"""
    corr = compute_corr_fft(h)
    return np.argmax(corr)

def initialize_binary():
    """Initialize a binary pattern (0/1) with exactly TARGET_SUM ones"""
    # Start with Thue-Morse
    h0 = generate_thue_morse(n_points)
    # Ensure exactly half ones
    current_sum = np.sum(h0)
    needed = TARGET_SUM - current_sum
    
    if needed > 0:
        # Flip some zeros to ones
        zeros = np.where(h0 == 0)[0]
        flip_indices = np.random.choice(zeros, size=int(needed), replace=False)
        h0[flip_indices] = 1
    elif needed < 0:
        # Flip some ones to zeros
        ones = np.where(h0 == 1)[0]
        flip_indices = np.random.choice(ones, size=int(-needed), replace=False)
        h0[flip_indices] = 0
    
    return h0

def greedy_worst_shift_swap(h):
    """Perform greedy swaps to reduce worst shift overlap"""
    original_h = h.copy()
    h = h.copy()
    max_shift = find_worst_shift(h)
    iterations = 0
    max_iterations = 1000
    
    while iterations < max_iterations:
        # Check if we need to recomputed
        if iterations % 50 == 0:
            max_shift = find_worst_shift(h)
        
        # Look at the current worst shift for improvement
        shift = max_shift
        # Find indices where pattern (1,0) exists
        if shift < 0:
            left = h[:shift]  # First part
            right = h[-shift:]  # Shifted part
            mismatch1 = np.where((left == 1) & (right == 0))[0]
            mismatch2 = np.where((left == 0) & (right == 1))[0]
        else:
            left = h[:-shift] # First part
            right = h[shift:]  # Shifted part
            mismatch1 = np.where((left == 1) & (right == 0))[0]
            mismatch2 = np.where((left == 0) & (right == 1))[0]
            
        if len(mismatch1) == 0 or len(mismatch2) == 0:
            break
            
        # Pick a random pair to swap
        i = np.random.choice(mismatch1)
        j = np.random.choice(mismatch2)
        
        # Perform swap
        h[i], h[j] = h[j], h[i]
        iterations += 1
    
    return h

def simulated_annealing(h0, temp_schedule):
    h = h0.copy()
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Convert to binary (bit flip for SA) - should be quick
    for _ in range(1000):  # limited iterations
        swap_i = np.random.randint(0, n_points)
        swap_j = np.random.randint(0, n_points)
        
        # Make a move
        h_new = h.copy()
        h_new[swap_i], h_new[swap_j] = h_new[swap_j], h_new[swap_i]
        
        # Project to maintain sum constraint
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        # Accept with probability based on temperature
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5)/temp_schedule()):
            h = h_new.copy()
            current_c5 = new_c5
    
    return best_h, best_c5

def continuous_refinement(h0):
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_local):
        return evaluate_c5(h_local)
    
    def constraint_eq(h_local):
        return np.sum(h_local) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
    except:
        pass

    h = project_to_sum_box(h, TARGET_SUM)
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Times assigned to:
    # Greedy search (0.7)  
    # Simulated Annealing (0.2) 
    # Continuous refinement (0.1)
    
    # Phase 1: Multiple Random Restarts of Greedy
    phase1_end = start_time + 0.7 * budget_s
    num_restarts = 300
    for i in range(num_restarts):
        if time.time() > phase1_end:
            break
        
        # Try initialization from different sources
        if i % 3 == 0:
            h = generate_thue_morse(n_points)
        elif i % 3 == 1:
            h = generate_rudin_shapiro(n_points)
        else:
            # Random binary pattern 
            h = np.random.randint(0, 2, n_points).astype(float)
            
        # Ensure proper sum
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Improve by greedy move
        h_refined = greedy_worst_shift_swap(h)
        c5_refined = evaluate_c5(h_refined)
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    # Phase 2: Simulated Annealing
    phase2_end = start_time + 0.9 * budget_s
    temp_schedule = lambda: max(0.01, 0.05 * (1 - (time.time() - (start_time + 0.7 * budget_s)) / (0.2 * budget_s)))
    for _ in range(10):
        if time.time() > phase2_end:
            break
        h_temp, c5_temp = simulated_annealing(best_h, temp_schedule)
        if c5_temp < best_c5:
            best_h = h_temp.copy()
            best_c5 = c5_temp
    
    # Phase 3: Final continuous refinement
    if best_h is not None:
        final_h = continuous_refinement(best_h)
        final_c5 = evaluate_c5(final_h)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    return (best_h.tolist(), float(best_c5), n_points)
