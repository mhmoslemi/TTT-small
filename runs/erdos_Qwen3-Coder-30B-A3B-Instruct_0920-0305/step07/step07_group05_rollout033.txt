import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global constants
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmin_weights(s, alpha):
    """Compute soft-min weights for given correlation values and alpha"""
    s_shifted = s - np.max(s)  # For numerical stability
    exps = np.exp(-alpha * s_shifted)
    return exps / np.sum(exps)

def weighted_gradient(h, alpha):
    """Compute gradient of softmin s with respect to h"""
    s = correlate(h, h, mode='full')
    w = softmin_weights(s, alpha)
    
    grad = np.zeros_like(h)
    n = len(h)
    
    for i in range(2 * n - 1):
        if w[i] > 0:
            lag = i - (n - 1)
            if lag >= 0:
                # Add contribution from h[i] * h[i+lag] 
                # which comes from s[i] term
                if i + lag < n:
                    grad[i] += w[i] * h[i + lag]
                if i - lag >= 0:
                    grad[i] += w[i] * h[i - lag]
            else:
                # lag < 0: contribution from h[i] * h[i+lag] where lag < 0
                if i - lag < n:
                    grad[i] += w[i] * h[i - lag]
                if i + lag >= 0:
                    grad[i] += w[i] * h[i + lag]
    
    return grad

def pairwise_move(h, grad, delta, rng):
    """Apply a pairwise mass transfer move"""
    # Find indices with extreme gradient values
    idx_src = np.argmin(grad)
    idx_tgt = np.argmax(grad)
    
    # Check if swap is valid
    if h[idx_src] <= 1e-10 or h[idx_tgt] >= 1 - 1e-10:
        return False, h, 0
    
    # Compute transfer amount
    transfer = min(delta, h[idx_src], 1 - h[idx_tgt])
    
    # Apply transfer
    h[idx_src] -= transfer
    h[idx_tgt] += transfer
    
    return True, h, transfer

def annealed_gradient_ascent(h0, time_budget, alpha0, delta0, temp0, rng):
    """Annealed gradient ascent with pairwise moves"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    alpha = alpha0
    delta = delta0
    temp = temp0
    
    start_time = time.time()
    iterations = 0
    
    while time.time() - start_time < time_budget:
        # Step size schedule
        if iterations % 100 == 0:
            delta *= 0.9995
            temp *= 0.9995
            if alpha < 200:
                alpha *= 1.05
        
        # Compute weighted gradient
        grad = weighted_gradient(h, alpha)
        
        # Perform move
        accepted, h_new, moved = pairwise_move(h, grad, delta, rng)
        
        if accepted:
            # Evaluate new solution
            c5_new = evaluate_c5(h_new)
            
            # Accept or reject based on temperature
            if c5_new < best_c5 or rng.random() < np.exp(-(c5_new - best_c5) / temp):
                h = h_new.copy()
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                # Revert the move if rejected
                pass
        
        iterations += 1
    
    return best_h, best_c5

def random_binary_start(n, rng):
    """Generate a binary start vector with exactly n/2 ones"""
    h = np.zeros(n)
    indices = rng.sample(range(n), n // 2)
    h[indices] = 1
    return h

def local_optimize_restart(start_h, remaining_time, rng):
    """Perform local optimization on a start vector"""
    # Parameters for the annealing schedule
    alpha0 = 10.0
    delta0 = 0.4
    temp0 = 5e-3
    
    h, c5 = annealed_gradient_ascent(start_h, remaining_time, alpha0, delta0, temp0, rng)
    return h, c5
