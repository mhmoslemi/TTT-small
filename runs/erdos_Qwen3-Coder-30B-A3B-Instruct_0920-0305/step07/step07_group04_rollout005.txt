```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import Minimize, Variable, sum as cp_sum, norm, Problem

# Global variables
TARGET_SUM = None
n_points = 150
dx = 2.0 / n_points

def evaluate_c5(h):
    """Compute C5 value using FFT correlation"""
    corr = np.correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_lp_relaxation():
    """Solve the linear programming relaxation using McCormick envelopes"""
    n = n_points
    h = Variable(n,.bounds=(0,1))
    
    # Build the McCormick constraints for all pairs
    # Only define p_ij for i <= j to save space
    p_vars = {}
    p_list = []
    
    for i in range(n):
        for j in range(i, n):
            p_vars[i,j] = Variable()
            p_list.append(p_vars[i,j])
    
    # McCormick constraints
    constraints = []
    for i in range(n):
        for j in range(i, n):
            p = p_vars[i,j]
            constraints.extend([
                p >= 0,
                p >= h[i] + h[j] - 1,
                p <= h[i],
                p <= h[j]
            ])
    
    # Sum constraint
    constraints.append(cp_sum(h) == TARGET_SUM)
    
    # Overlaps for all valid shifts
    t = Variable()  # upper bound on max (n/2 - overlap_k)
    
    # This is minimum over objective function
    # We maximize t s.t. sum(h_i) - overlap_k <= t
    # or equivalently overlap_k >= (n/2 - t)
    for k in range(1 - n, n):
        if k >= 0:
            # For k >= 0, overlap_k = sum_{i=0}^{n-1-k} h_i * h_{i+k}
            overlap_k = cp_sum([p_vars[i,i+k] for i in range(n-k)])
        else:
            # For k < 0, overlap_k = sum_{i=0}^{n-1+k} h_i * h_{i+k}
            overlap_k = cp_sum([p_vars[i+k,i] for i in range(-k, n)])
            
        constraints.append(overlap_k >= TARGET_SUM - t)
    
    objective = Minimize(t)
    prob = Problem(objective, constraints)
    
    # We want to allow at most 150 seconds to solve the LP before falling back
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000, abstol=1e-6, reltol=1e-6, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_opt = h.value
            return h_opt
    except Exception as e:
        pass

    # fallback to uniformly distributed
    return np.ones(n) * 0.5

def projected_subgradient_descent(h0, max_iter=2000, time_budget_s=200):
    """Improve LP solution with projected subgradient descent"""
    h = h0.copy()
    start_time = time.time()
    
    alpha = 0.5 * dx
    decay = 0.995
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        if time.time() - start_time > time_budget_s:
            break
            
        # compute gradient of max correlation term
        corr = np.correlate(h, 1 - h, mode="full")[n_points-1:]
        k_star = np.argmax(corr)
        
        # compute gradient vector g
        g = np.zeros_like(h)
        for i in range(len(h)):
            if i + k_star < len(h):
                g[i] += -(1 - h[i + k_star])
            if i - k_star >= 0:
                g[i] += -h[i - k_star]
        
        # update h
        h_new = h - alpha * g
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # evaluate new h
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        
        h = h_new.copy()
        alpha *= decay
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    
    # Phase 1: Solve LP relaxation
    h0 = solve_lp_relaxation()
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    # Evaluate C5 for initial solution
    c5_initial = evaluate_c5(h0)
    best_h = h0.copy()
    best_c5 = c5_initial
    
    # Phase 2: Refine using projected subgradient descent
    refine_time = min(200, budget_s * 0.3)
    h_refined, c5_refined = projected_subgradient_descent(h0, 2000, refine_time)
    
    if c5_refined < best_c5:
        best_h = h_refined.copy()
        best_c5 = c5_refined
    
    # Phase 3: Optional randomized refinement with CMA-ES style step
    if time.time() - start_time < budget_s * 0.9:
        h_cma = best_h + np.random.normal(0, 0.01, n_points)
        h_cma = project_to_sum_box(h_cma, TARGET_SUM)
        h_final, c5_final = projected_subgradient_descent(h_cma, max_iter=500, time_budget_s=budget_s * 0.1)
        
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final
    
    # Return final result
    return (best_h.tolist(), float(best_c5), n_points)
```
