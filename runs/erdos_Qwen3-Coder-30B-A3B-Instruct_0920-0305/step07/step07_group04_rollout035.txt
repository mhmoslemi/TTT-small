import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 150
dx = 2.0 / n_points
TIME_BUDGET = 600

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_lp_relaxation():
    """Solve the linear programming relaxation to get a good starting point"""
    start_time = time.time()
    
    # Decision variables
    h = cp.Variable(n_points, lower_bound=0, upper_bound=1)
    
    # Product variables p[i,j] for i <= j
    p = {}
    for i in range(n_points):
        for j in range(i, n_points):
            p[(i,j)] = cp.Variable(lower_bound=0, upper_bound=1)
    
    # Constraints
    constrs = []
    
    # Mass constraint
    constrs.append(cp.sum(h) == TARGET_SUM)
    
    # McCormick constraints for products
    for i in range(n_points):
        for j in range(i, n_points):
            # p_ij >= 0
            constrs.append(p[(i,j)] >= 0)
            # p_ij >= h_i + h_j - 1
            constrs.append(p[(i,j)] >= h[i] + h[j] - 1)
            # p_ij <= h_i
            constrs.append(p[(i,j)] <= h[i])
            # p_ij <= h_j
            constrs.append(p[(i,j)] <= h[j])
    
    # Objective variable
    t = cp.Variable(pos=True)
    
    # For each shift k, compute overlap and add constraints
    # Only focus on non-negative shifts for symmetry - we compute all over shifts later
    for k in range(n_points - 1):
        # Compute sum of products for this shift: sum_{i=0}^{n-1-k} p[i][i+k]
        overlap = cp.sum([p[(i, i+k)] for i in range(n_points - k)])
        # R_k = (n/2) - overlap
        # So: R_k <= t => (n/2) - overlap <= t => overlap >= (n/2) - t
        constrs.append(overlap >= TARGET_SUM - t)

    # Objective: minimize t
    prob = cp.Problem(cp.Minimize(t), constrs)
    
    # Solve with time limit
    try:
        prob.solve(solver=cp.ECOS, max_iters=10000, abstol=1e-6, reltol=1e-6, verbose=False)
    except:
        return None, 1.0

    if prob.status == cp.OPTIMAL:
        h_solution = h.value
        if h_solution is not None:
            return np.array(h_solution), t.value
    return None, 1.0

def projected_gradient_descent(h0, max_iter=2000):
    """Refine solution using projected subgradient descent"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    alpha0 = 0.5 * dx
    for i in range(max_iter):
        # Find the active shift
        corr = compute_corr_fft(h)
        k_star = np.argmax(corr)
        
        # Compute subgradient (gradient of active term)
        g = np.zeros(n_points)
        for j in range(n_points):
            # h_{j+k*}
            if j+k_star < n_points:
                g[j] = (1 - h[j+k_star])
            else:
                g[j] = 0
            
            # h_{j-k*}
            if j-k_star >= 0:
                g[j] -= h[j-k_star]
            else:
                g[j] -= 0
        
        # Gradient step
        h_new = h - alpha0 * g
        
        # Project to box [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project to sum constraint
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        h = h_new.copy()
        
        # Decrease step-size
        alpha0 *= 0.999
        
        # Break early if no improvement
        if i > 100 and i % 50 == 0:
            if best_c5 < 0.379:
                return best_h, best_c5
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialization
    TARGET_SUM = n_points / 2
    
    # Step 1: LP relaxation to get a good starting point
    print("Solving LP relaxation...")
    h_lp, t_lp = solve_lp_relaxation()
    if h_lp is None:
        # Fallback to random initialization
        h0 = np.random.uniform(0, 1, n_points)
        h0 = project_to_sum_box(h0, TARGET_SUM)
    else:
        h0 = h_lp.copy()
        h0 = project_to_sum_box(h0, TARGET_SUM)
    
    # Step 2: Evaluate the candidate
    best_c5 = evaluate_c5(h0)
    best_h = h0.copy()
    print(f"LP yielded C5 = {best_c5:.6f}")
    
    # Step 3: Local refinement with subgradient method
    print("Refining solution with projected subgradient descent...")
    h_refined, c5_refined = projected_gradient_descent(best_h)
    
    if c5_refined < best_c5:
        best_h = h_refined
        best_c5 = c5_refined
        print(f"Subgradient improved C5 to {best_c5:.6f}")
    
    # Step 4: Quick fallback method with CMA-ES if still too high
    if best_c5 > 0.38:
        print("Performing fallback optimization with Random Search + Local Refinement")
        
        # Start from better of the two previous solutions
        for _ in range(10):
            # Small random noise to vary the initial point a bit.
            h_noise = best_h + np.random.normal(0, 0.005, n_points)
            h_noise = project_to_sum_box(h_noise, TARGET_SUM)
            
            # Refine a bit
            h_local, c5_local = projected_gradient_descent(h_noise, max_iter=500)
            if c5_local < best_c5:
                best_h = h_local
                best_c5 = c5_local
                print(f"Fallback local search yielded C5 = {best_c5:.6f}")
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
