import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def perturb_solution(h, strength=0.01):
    """Create a perturbed version of the solution"""
    perturbed = h + np.random.normal(0, strength, n_points)
    return project_to_sum_box(perturbed, TARGET_SUM)

    last_improvement = 0
    step_size = 0.02

        if i - last_improvement > patience:
            step_size *= 0.95  # Reduce step size if no improvement

        grad = np.zeros(n_points)

            h_plus = h.copy()
            h_plus[k] = np.clip(h_plus[k] + step_size, 0, 1)
            h_minus = h.copy()
            h_minus[k] = np.clip(h_minus[k] - step_size, 0, 1)
            
            c5_plus = evaluate_c5(h_plus)
            c5_minus = evaluate_c5(h_minus)
            grad[k] = (c5_minus - c5_plus) / (2 * step_size)

        update = step_size * grad

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            last_improvement = i
            h = h_new.copy()
        else:
            h = h_new.copy()
            
    return best_h, best_c5
