```python
import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = 1.0
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def fourier_basis(x, M):
    """Generate Fourier basis functions up to order M"""
    basis = np.ones((len(x), M + 1))
    for m in range(1, M + 1):
        basis[:, m] = np.cos(m * np.pi * x)
    return basis

def construct_h_from_coeffs(coeffs, M):
    """Construct h from Fourier coefficients"""
    # Set up x grid
    x = np.linspace(0, 2, n_points)
    
    # Generate basis matrix
    basis = fourier_basis(x, M)
    
    # Compute h
    h = np.dot(basis, coeffs)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    return h

def objective_fourier(coeffs, M):
    """Objective function for Fourier coefficients"""
    h = construct_h_from_coeffs(coeffs, M)
    c5 = evaluate_c5(h)
    return c5

def constraint_sum(coeffs, M):
    """Constraint that h integrates to 1"""
    x = np.linspace(0, 2, n_points)
    basis = fourier_basis(x, M)
    h = np.dot(basis, coeffs)
    integral = np.sum(h) * dx
    return integral - TARGET_SUM

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different numbers of basis functions
    best_c5 = float('inf')
    best_coeffs = None
    best_M = None
    
    # Test different numbers of Fourier modes
    for M in [5, 7, 9, 11]:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Set up bounds for coefficients
        bounds = [(-1.0, 1.0) for _ in range(M + 1)]
        
        # Set up the optimization
        def obj(coeffs):
            return objective_fourier(coeffs, M)
        
        # Differential evolution
        try:
            result = differential_evolution(
                obj,
                bounds,
                maxiter=500,
                popsize=15,
                mutation=(0.5, 1.0),
                recombination=0.7,
                seed=seed,
                disp=False,
                atol=1e-8,
                ftol=1e-8
            )
            
            if result.success:
                coeffs = result.x
                h = construct_h_from_coeffs(coeffs, M)
                c5 = evaluate_c5(h)
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_coeffs = coeffs
                    best_M = M
                    
        except Exception as e:
            continue
    
    # Local refinement with L-BFGS
    if best_coeffs is not None and time.time() - start_time < budget_s * 0.98:
        from scipy.optimize import minimize
        
        def obj_limited(coeffs):
            return objective_fourier(coeffs, best_M)
            
        try:
            result = minimize(
                obj_limited,
                best_coeffs,
                method='L-BFGS-B',
                bounds=[(-1.0, 1.0) for _ in range(best_M + 1)],
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            
            if result.success:
                refined_coeffs = result.x
                h_refined = construct_h_from_coeffs(refined_coeffs, best_M)
                c5_refined = evaluate_c5(h_refined)
                
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_coeffs = refined_coeffs
                    
        except Exception as e:
            pass
    
    # Final solution
    h_final = construct_h_from_coeffs(best_coeffs, best_M)
    
    return (h_final.tolist(), float(best_c5), n_points)
```
