import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    # Create CVXPY variables
    h = cp.Variable(n_points, name="h")
    # Constraint: h in [0,1]
    constraints = [h >= 0, h <= 1]
    
    # Constraint: sum(h) = TARGET_SUM
    constraints.append(cp.sum(h) == TARGET_SUM)
    
    # Create McCormick relaxation for all pairwise products
    # We'll create a matrix P where P[i,j] approximates h[i] * h[j]
    P = cp.Variable((n_points, n_points), name="P")
    
    # Add McCormick constraints for all pairs
    for i in range(n_points):
        for j in range(n_points):
            constraints.append(P[i, j] >= 0)
            constraints.append(P[i, j] <= h[i])
            constraints.append(P[i, j] <= h[j])
            constraints.append(P[i, j] >= h[i] + h[j] - 1)
    
    # Compute correlations for all shifts (including negative shifts for full correlation)
    # For positive shifts k=0..n-1, the overlap is h_i * (1 - h_{i+k}) over i: 0 to n-1-k
    # For negative shifts k=-n+1..-1, overlap is over i: 0 to n-1+k 
    correlations = []
    
    # For verification against simple implementation, we'll compute all correlations manually
    # and use the best (but still valid) result
    
    # Use convex relaxation to minimize maximum correlation
    # Objective: minimize max_k correlation_k
    # correlation_k = sum_i h_i - sum_{i where i+k < n} h_i * h_{i+k}
    # This uses what's actually in the overlap. So for each k:
    #   correlation_k = sum(h_i) - sum_{i for which both i, i+k valid} P[i, i+k]
    # We'll define t as upper bound on all correlations
    
    # Define t as scalar variable for maximum correlation bound
    t = cp.Variable(name="t")
    # Add constraint that t is greater than all correlation values
    for k in range(-n_points + 1, n_points):  # Include all valid shifts
        # Calculate valid indices for shift k
        if k >= 0:
            valid_indices = list(range(n_points - k))
            # Sum of h * (1 - h.shifted) corresponds to h - sum P[i, i+k]
            corr = cp.sum(h[valid_indices]) - cp.sum(P[valid_indices, valid_indices + k])
        else:
            # For negative k: valid indices are those where i + k >= 0
            valid_indices = list(range(-k, n_points))
            corr = cp.sum(h[valid_indices]) - cp.sum(P[valid_indices, valid_indices + k])
        
        constraints.append(corr <= t)
    
    # Objective is to minimize t (maximum correlation)
    objective = cp.Minimize(t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Use time budget to solve within 90% of time limit
    timeout_sec = 0.9 * budget_s
    
    try:
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=timeout_sec)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            # Extract solution
            h_solution = h.value
            # Compute actual C5 using numpy correlation
            corr = np.correlate(h_solution, 1 - h_solution, mode="full")
            c5_value = np.max(corr) * dx
            return (h_solution.tolist(), float(c5_value), n_points)
        else:
            # Fall back to a simple construction
            h_simple = np.ones(n_points)
            h_simple[:n_points//2] = 0
            h_simple = np.clip(h_simple, 0, 1)
            h_simple = h_simple * (TARGET_SUM / np.sum(h_simple))
            corr = np.correlate(h_simple, 1 - h_simple, mode="full")
            c5_value = np.max(corr) * dx
            return (h_simple.tolist(), float(c5_value), n_points)
    except:
        # Fall back to a simple construction
        h_simple = np.ones(n_points)
        h_simple[:n_points//2] = 0

        h_simple = h_simple * (TARGET_SUM / np.sum(h_simple))
        corr = np.correlate(h_simple, 1 - h_simple, mode="full")
        c5_value = np.max(corr) * dx
        return (h_simple.tolist(), float(c5_value), n_points)
