import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    """Evaluate the true C5 bound using FFT correlation"""
    n_points = len(h)
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def binary_search_lp_relaxation(n_points, max_iter=50, tol=1e-4):
    """Solve the LP relaxation with binary search to get certified C5 upper bound"""
    dx = 2.0 / n_points
    midpoints = []
    
    # Create the convex relaxation
    h = cp.Variable(n_points, bounds=(0, 1))
    p = cp.Variable((n_points, n_points), symmetric=True, bounds=(0, 1))
    T = cp.Variable()
    
    # Mass constraint
    constraints = [cp.sum(h) == n_points / 2]
    
    # McCormick envelopes for p[i,j] = h[i] * h[j]
    for i in range(n_points):
        for j in range(i, n_points):
            constraints += [
                p[i, j] >= 0,
                p[i, j] >= h[i] + h[j] - 1,
                p[i, j] <= h[i],
                p[i, j] <= h[j]
            ]
    
    # Shift correlation constraints
    # Using a simplified approach: for each valid shift k, approximate as
    # O_k ≈ sum_{i=0}^{n-1-|k|} p[i, i+|k|]
    for k in range(-n_points+1, n_points):
        # Determine valid indices for this shift
        if k > 0:
            # positive shift: i ranges over [0, n-1-k]
            start = 0
            end = n_points - k
        else:
            # negative shift: i ranges over [0, n-1+|k|] 
            start = 0
            end = n_points + k
        if end > start:
            # Compute overlap for this shift
            O_k = cp.sum([p[i, i+abs(k)] for i in range(start, end)])
            # Add constraint O_k >= n/2 - T
            constraints += [O_k >= n_points/2 - T]
    
    # Solving with binary search
    low, high = 0.0, n_points / 2
    best_h = None
    best_T = high
    for it in range(max_iter):
        if high - low < tol * dx:
            break
        T_val = (low + high) / 2
        constraints.append(T == T_val)  # Fix T for this iteration
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False, time_limit=180)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                # Now evaluate the true C5
                h_sol = h.value
                if h_sol is not None:
                    c5_true = evaluate_erdos_solution(h_sol)
                    if c5_true <= T_val * dx:
                        high = T_val
                        best_h = h_sol.copy()
                        best_T = T_val
                    else:
                        low = T_val
                else:
                    low = T_val
            else:
                low = T_val
        except Exception:
            low = T_val
        finally:
            # Remove the T constraint for next iteration
            constraints.pop()

    if best_h is not None:
        # Make sure final mass is enforced
        h_sol = best_h.copy()
        current_sum = np.sum(h_sol)
        delta = (n_points / 2 - current_sum) / n_points
        h_sol += delta
        h_sol = np.clip(h_sol, 0, 1)
        # Enforce the exact fix via projection
        def project_sum(h_test, target):
            # Binary search for lambda
            a, b = -1.0, 1.0
            def f(lam):
                h_proj = np.clip(h_test + lam, 0, 1)
                return np.sum(h_proj) - target
            for _ in range(50):
                m = (a + b) / 2
                fm = f(m)
                if abs(fm) < 1e-12:
                    break
                if fm * f(a) < 0:
                    b = m
                else:
                    a = m
            return np.clip(h_test + (a + b) / 2, 0, 1)
        h_sol = project_sum(h_sol, n_points / 2)
        return h_sol, best_T * dx
    return None, None

def project_to_sum(h, target_sum):
    """Project h to sum to target_sum with box constraints [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    delta = (target_sum - current_sum) / len(h)
    h += delta
    h = np.clip(h, 0, 1)
    
    def f(lam):
        h_proj = np.clip(h + lam, 0, 1)
        return np.sum(h_proj) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    return np.clip(h + (a + b)/2, 0, 1)

def subgradient_optimize(h0, max_iter=2000):
    """Perform deterministic sub-gradient descent to polish the solution"""
    h = h0.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    h = project_to_sum(h, n_points / 2)
    
    # Parameters
    alpha0 = 0.01
    for t in range(max_iter):
        # Evaluate original function and identify active term
        corr = correlate(h, 1 - h, mode="full")
        active_k = np.argmax(corr)
        
        # Compute sub-gradient for active shift k
        grad = np.zeros_like(h)
        if active_k < n_points:
            # Positive shift
            for i in range(n_points - active_k):
                grad[i] += (1 - h[i + active_k])
            for i in range(active_k, n_points):
                grad[i] -= h[i - active_k]
        else:
            # Negative shift
            k = active_k - n_points + 1
            for i in range(n_points - k):
                grad[i] -= h[i + k]
            for i in range(k, n_points):
                grad[i] += (1 - h[i - k])
        
        alpha_t = alpha0 / np.sqrt(t + 1)
        h -= alpha_t * grad
        h = project_to_sum(h, n_points / 2)
        
    return h

def local_exchange_refinement(h0, max_iters=1000):
    """Apply local mass exchange refinement"""
    h = h0.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    
    def calculate_c5(h):
        corr = correlate(h, 1 - h, mode="full")
        return dx * np.max(corr)
    
    c5 = calculate_c5(h)
    for i in range(max_iters):
        # sample pairs
        i1, i2 = np.random.choice(n_points, 2, replace=False)
        if i1 == i2:
            continue
        max_transfer = min(h[i2], 1 - h[i1])
        if max_transfer < 1e-10:
            continue
        delta = np.random.random() * 0.9 * max_transfer
        h_new = h.copy()
        h_new[i1] += delta
        h_new[i2] -= delta
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum(h_new, n_points / 2)
        c5_new = calculate_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 150
    dx = 2.0 / n_points
    
    h_values, c5_bound = binary_search_lp_relaxation(n_points, max_iter=30)
    
    if h_values is None or c5_bound is None:
        return [], 1.0, n_points
    
    # polish with sub-gradient
    h_values = subgradient_optimize(h_values, max_iter=1500)
    c5_bound = evaluate_erdos_solution(h_values)
    
    # local exchange refinement
    h_values = local_exchange_refinement(h_values, max_iters=500)
    c5_bound = evaluate_erdos_solution(h_values)
    
    return (h_values.tolist(), float(c5_bound), n_points)
