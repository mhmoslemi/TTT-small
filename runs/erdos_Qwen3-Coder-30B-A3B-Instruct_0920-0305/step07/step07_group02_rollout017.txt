```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

# Global variables for consistent usage
N = 120
dx = 2.0 / N
TARGET_SUM = N // 2

def compute_c5_fft(h):
    """Fast evaluation of C5 using FFT"""
    corr_full = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr_full)

def round_and_repair_h(h, target_sum):
    """Round h to binary values and repair sum constraint"""
    h_rounded = np.round(h).astype(int)
    diff = target_sum - np.sum(h_rounded)
    if diff > 0:
        # Need to flip some 0s to 1s
        zero_indices = np.where(h_rounded == 0)[0]
        for i in range(diff):
            if i < len(zero_indices):
                h_rounded[zero_indices[i]] = 1
    elif diff < 0:
        # Need to flip some 1s to 0s
        one_indices = np.where(h_rounded == 1)[0]
        for i in range(-diff):
            if i < len(one_indices):
                h_rounded[one_indices[i]] = 0
    return h_rounded

def greedy_descent(h):
    """Greedy local descent to improve worse shift"""
    h = h.copy()
    c5 = compute_c5_fft(h)
    max_iter = 200
    
    for _ in range(max_iter):
        corr_full = np.correlate(h, h, mode='full')
        # Find shift with highest overlap (corresponds to min overlap)
        worst_shift = np.argmax(corr_full)
        # Get right half of correlation array (shifts from 0 to N-1)
        shift_values = corr_full[N-1:]
        min_shift_idx = np.argmin(shift_values)
        worst_shift = min_shift_idx
        
        # Identify mismatched pairs for this shift
        A = []
        B = []
        for i in range(N):
            j = i + worst_shift
            if j < N:
                if h[i] == 1 and h[j] == 0:
                    A.append(i)
                elif h[i] == 0 and h[j] == 1:
                    B.append(i)
        
        if not A or not B:
            break
            
        # Pick a random pair and swap
        i = np.random.choice(A)
        j = np.random.choice(B)
        h[i], h[j] = h[j], h[i]
        
        new_c5 = compute_c5_fft(h)
        if new_c5 >= c5:
            # Revert if it didn't improve
            h[i], h[j] = h[j], h[i]
            break
        else:
            c5 = new_c5
            
    return h, c5

def anneal_binary(h, time_limit):
    """Lightweight simulated annealing"""
    h = h.copy()
    c5_base = compute_c5_fft(h)
    start_time = time.time()
    
    T0 = 0.02
    Tmax = 10000
    
    for t in range(Tmax):
        if time.time() - start_time > time_limit - 1:
            break
            
        # Choose two random positions
        i = np.random.randint(0, N)
        j = np.random.randint(0, N)
        if i == j:
            continue
            
        # Swap them
        h[i], h[j] = h[j], h[i]
        
        new_c5 = compute_c5_fft(h)
        delta = new_c5 - c5_base
        
        if delta < 0:
            # Always accept if better
            c5_base = new_c5
        else:
            # Accept with probability
            T = T0 * (1.0 - t / Tmax)
            if T > 1e-10:
                prob = np.exp(-delta / T)
                if np.random.rand() < prob:
                    c5_base = new_c5
                else:
                    # Revert swap
                    h[i], h[j] = h[j], h[i]
        
        # Occasionally do a greedy sweep
        if t % 500 == 0:
            h, c5_base = greedy_descent(h)
            
    return h, c5_base

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        # Build MILP model
        h_vars = cp.Variable(N, boolean=True)
        lambda_var = cp.Variable(integer=True)
        
        # Constraints for mass
        constraints = [cp.sum(h_vars) == TARGET_SUM]
        
        # Linearize product terms
        w_vars = {}
        shift_constraints = []
        
        for k in range(-(N-1), N):
            # valid indices
            i_indices = np.arange(max(0, -k), min(N, N - k))
            if len(i_indices) == 0:
                continue
            # Create auxiliary variables for this shift
            w_list = cp.Variable(len(i_indices), boolean=True)
            w_vars[k] = w_list
            
            # Linearization constraints
            i_list = i_indices
            j_list = i_list + k
            h_i = h_vars[i_list]
            h_j = h_vars[j_list]
            constraints.append(w_list <= h_i)
            constraints.append(w_list <= h_j)
            constraints.append(w_list >= h_i + h_j - 1)
            
            shift_constraints.append(cp.sum(w_list) >= lambda_var)
        
        # Add constraints
        constraints.extend(shift_constraints)
        
        # Add symmetry breaking
        constraints.append(h_vars[0] == 1)
        
        # Objective: maximize lambda
        objective = cp.Maximize(lambda_var)
        prob = cp.Problem(objective, constraints)
        
        # Solve with solver time limit
        remaining_time = budget_s * 0.55
        prob.solve(solver=cp.CBC, max_seconds=remaining_time, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE]:
            h_sol = h_vars.value
            h_sol = np.round(h_sol).astype(int)
            # Repair sum if needed
            h_sol = round_and_repair_h(h_sol, TARGET_SUM)
            c5_sol = compute_c5_fft(h_sol)
        else:
            # Fallback
            h_sol = np.random.choice([0, 1], size=N, p=[0.5, 0.5])
            h_sol = round_and_repair_h(h_sol, TARGET_SUM)
            c5_sol = compute_c5_fft(h_sol)
            
    except Exception as e:
        # Ensure fallback
        h_sol = np.random.choice([0, 1], size=N, p=[0.5, 0.5])
        h_sol = round_and_repair_h(h_sol, TARGET_SUM)
        c5_sol = compute_c5_fft(h_sol)
    
    best_h = h_sol
    best_c5 = c5_sol
    
    # Greedy improvement
    elapsed = time.time() - start_time
    if elapsed < 0.75 * budget_s:
        best_h, best_c5 = greedy_descent(best_h)
        
    # Simulated annealing if time permits
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 30:
        best_h, best_c5 = anneal_binary(best_h, remaining_time)
    
    return (best_h.tolist(), float(best_c5), N)
```
