import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global variables
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmin_weights(s, alpha):
    """Compute soft-min weights for given s and alpha"""
    s_centered = s - np.max(s)  # For numerical stability
    exp_s = np.exp(-alpha * s_centered)
    return exp_s / np.sum(exp_s)

def weighted_gradient(h, alpha):
    """Compute weighted gradient for softmin objective"""
    s = correlate(h, h, mode='full')
    w = softmin_weights(s, alpha)
    
    grad = np.zeros_like(h)
    n = len(h)
    
    for lag in range(-n+1, n):
        if lag >= 0:
            # h[i] * h[i+lag] contribution
            if lag < n:
                grad[lag:] += w[lag + n - 1] * h[:n-lag]
            if lag > 0:
                grad[:n-lag] += w[lag + n - 1] * h[lag:]
        else:
            # h[i] * h[i+lag] contribution (neg lag)
            lag_abs = -lag
            if lag_abs < n:
                grad[lag_abs:] += w[lag + n - 1] * h[:n-lag_abs]
            if lag_abs > 0:
                grad[:lag_abs] += w[lag + n - 1] * h[lag_abs:]
    
    # Compute negative gradient for descent (since we want to minimize -F)
    return -grad

def pairwise_move(h, grad, delta, rng):
    """Perform a pairwise mass transfer move"""
    src = np.argmin(grad)
    tgt = np.argmax(grad)
    
    # Ensure source and target are valid
    if grad[src] >= 0 or grad[tgt] <= 0:
        # Fall back to random
        src = rng.randint(0, len(h)-1)
        tgt = rng.randint(0, len(h)-1)
    
    # Clamp to valid range
    if h[src] <= 0 or h[tgt] >= 1:
        return h, False
    
    # Transfer mass
    delta = min(delta, h[src], 1 - h[tgt])
    if delta <= 1e-8:
        return h, False
    
    h[src] -= delta
    h[tgt] += delta
    
    # Clip to bounds
    h[src] = max(0, min(1, h[src]))
    h[tgt] = max(0, min(1, h[tgt]))
    
    return h, True

def annealed_gradient_ascent(h0, time_budget, alpha0, delta0, T0, rng):
    """Main annealing loop with adaptive parameters"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    tot_time = time.time()
    alpha = alpha0
    delta = delta0
    T = T0
    
    iter_count = 0
    accepted_count = 0
    no_improvement_count = 0
    
    max_no_improvement = 20000
    start_time = time.time()
    
    while time.time() - start_time < time_budget:
        iter_count += 1
        
        # Compute gradient with current alpha
        grad = weighted_gradient(h, alpha)
        
        # Choose source and target based on gradient
        src = np.argmin(grad)
        tgt = np.argmax(grad)
        
        if grad[src] >= 0 or grad[tgt] <= 0:
            # Fall back to random
            src = rng.randint(0, len(h)-1)
            tgt = rng.randint(0, len(h)-1)
        
        if h[src] <= 0 or h[tgt] >= 1:
            # Random move
            src = rng.randint(0, len(h)-1)
            tgt = rng.randint(0, len(h)-1)
            
        # Transfer mass
        delta_transfer = min(delta, h[src], 1 - h[tgt])
        if delta_transfer <= 1e-8:
            # Skip if tiny transfer
            continue
        
        h_old = h.copy()
        h[src] -= delta_transfer
        h[tgt] += delta_transfer
        
        # Clip to bounds
        h[src] = max(0, min(1, h[src]))
        h[tgt] = max(0, min(1, h[tgt]))
        
        # Evaluate
        c5_new = evaluate_c5(h)
        c5_old = evaluate_c5(h_old)
        
        # Accept or reject based on Metropolis criterion
        if c5_new < c5_old:
            # Improvement - always accept
            accepted_count += 1
            best_c5 = min(best_c5, c5_new)
            if c5_new < best_c5:
                best_h[:] = h[:]
            no_improvement_count = 0
        else:
            # Possibly reject
            if rng.random() < np.exp(-(c5_new - c5_old) / T):
                accepted_count += 1
                best_c5 = min(best_c5, c5_new)
                if c5_new < best_c5:
                    best_h[:] = h[:]
                no_improvement_count = 0
            else:
                # Revert the move
                h[:] = h_old[:]
                no_improvement_count += 1
        
        # Parameter updates
        if iter_count % 100 == 0:
            # Update alpha
            if alpha < 200:
                alpha *= 1.02
            
            # Update delta
            delta *= 0.999
            
            # Update temperature
            T *= 0.9995
        
        # Prevent stagnation
        if no_improvement_count > max_no_improvement:
            # Reset some parameters
            delta = min(delta * 1.2, 0.5)
            T = min(T * 1.5, 1e-2)
            no_improvement_count = 0
        
        # Fairly balanced time reporting
        if iter_count % 10000 == 0:
            elapsed = time.time() - start_time
            if elapsed > time_budget * 0.7:
                break
    
    return best_h, best_c5

def random_binary_start(n, rng):
    """Create a binary starting vector with correct sum"""
    h = np.zeros(n)
    indices = rng.sample(range(n), n // 2)
    h[indices] = 1.0
    return h

    alpha0 = 10.0
    delta0 = 0.2
    T0 = 5e-3
    
    return annealed_gradient_ascent(start_h, remaining_time, alpha0, delta0, T0, rng)
