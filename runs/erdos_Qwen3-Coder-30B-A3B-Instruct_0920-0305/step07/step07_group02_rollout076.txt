import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5_fft(h):
    """Fast evaluation of C5 using FFT"""
    dx = 2.0 / len(h)
    corr_full = np.correlate(h, 1 - h, mode="full")
    return dx * np.max(corr_full)

def greedy_descent(h_original):
    """Greedy descent on worst shift"""
    h = h_original.copy()
    dx = 2.0 / len(h)
    while True:
        corr = np.correlate(h, h, mode='full')
        worst_shift_idx = np.argmax(corr)
        # Find positions for worst shift
        if worst_shift_idx < len(h):
            s = worst_shift_idx
        else:
            s = worst_shift_idx - len(h) + 1
        
        # Find mismatched pairs
        positions_h1 = np.where(h == 1)[0]
        # Set of i such that h[i]=1 and h[i+s]=0
        A = positions_h1[np.where(h[np.clip(positions_h1 + s, 0, len(h)-1)] == 0)[0]]
        # Set of i such that h[i]=0 and h[i+s]=1
        B = np.where(h == 0)[0]
        B = B[np.where(h[np.clip(B + s, 0, len(h)-1)] == 1)[0]]

        if len(A) == 0 or len(B) == 0:
            break  # No more improvements possible for this shift
        i = np.random.choice(A)
        j = np.random.choice(B)
        
        # Swap bits
        h[i] = 0
        h[j] = 1
    return h, compute_c5_fft(h)

def anneal_binary(h_original, time_limit):
    """Lightweight simulated annealing"""

    Tmin = 1e-6
    Tmax = int(10000 * time_limit / 30)  # Scale with time
    
    for t in range(Tmax):
        if time.time() - start_time > time_limit - 1:
            break
            
        T = T0 * (1 - t / Tmax)
        if T < Tmin:
            break
            
        i = np.random.randint(0, len(h))
        # Find a 0 position to swap with
        j_candidates = np.where(h == 0)[0]
        if len(j_candidates) == 0:
            continue

        # Swap bits
        h[i], h[j] = h[j], h[i]
        new_C5 = compute_c5_fft(h)
        
        delta = new_C5 - C5
        
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            C5 = new_C5
        else:
            # Revert swap
            h[i], h[j] = h[j], h[i]
        
        # Every 100 moves, do a quick greedy fix
        if t % 100 == 0:
            h, C5 = greedy_descent(h)
            
    return h, C5
