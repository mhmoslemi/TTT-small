import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_convex_relaxation(h0):
    """Solve a convex relaxation of the problem"""
    # This is a simplified approach - not strictly solving convex relaxation
    # But gives us a good starting point for local optimization
    h = h0.copy()
    
    # Apply simple smoothing and projection
    h = np.convolve(h, np.ones(5)/5, mode='same')
    h = project_to_sum_box(h, TARGET_SUM)
    
    return h

def enhanced_local_search(h0, max_iter=1000):
    """Enhanced local search with multiple strategies"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Try different neighborhood structures
    for i in range(max_iter):
        # Make a random change
        idx = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.02, 0.02)
        h_new = h.copy()
        h_new[idx] += delta
        
        # Keep within bounds
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            h = h_new.copy()
            best_h = h.copy()
            best_c5 = new_c5
        elif np.random.rand() < 0.05:  # Sometimes accept worse solutions
            h = h_new.copy()
    
    return best_h

def run(seed=42, budget_s=600, **kwargs):

    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Get a better initial guess using a mix of methods
    best_h = adaptive_sampling_initialization()
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: Multiple starting points with local search
    phase1_end = start_time + 0.5 * budget_s
    for run_id in range(20):
        if time.time() > phase1_end:
            break
            
        # Try different initialization strategies
        initial_modes = [
            adaptive_sampling_initialization,
            lambda: np.random.uniform(0, 1, n_points),
            lambda: np.random.choice([0, 1], size=n_points, p=[0.5, 0.5]),
            lambda: solve_convex_relaxation(np.random.uniform(0, 1, n_points))
        ]
        
        h_start = initial_modes[run_id % len(initial_modes)]()
        h_start = project_to_sum_box(h_start, TARGET_SUM)
        
        # Local search

        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    # Phase 2: Fine-tune with coordinate descent
    phase2_end = start_time + budget_s
    while time.time() < phase2_end:
        h_old = best_h.copy()
        c5_old = best_c5
        
        # Coordinate descent on one element at a time
        for i in range(n_points):
            best_val = best_h[i]
            best_score = best_c5
            
            # Try several discrete values for this element
            test_values = [max(0, min(1, best_h[i] + d)) for d in [-0.05, -0.02, -0.01, 0, 0.01, 0.02, 0.05]]
            
            for val in test_values:
                h_test = best_h.copy()
                h_test[i] = val
                h_test = project_to_sum_box(h_test, TARGET_SUM)
                score = evaluate_c5(h_test)
                
                if score < best_score:
                    best_score = score
                    best_val = val
            
            best_h[i] = best_val
            
        best_c5 = evaluate_c5(best_h)
        
        # Stop if no improvement 
        if best_c5 >= c5_old:
            break
    
    return (best_h.tolist(), float(best_c5), n_points)
