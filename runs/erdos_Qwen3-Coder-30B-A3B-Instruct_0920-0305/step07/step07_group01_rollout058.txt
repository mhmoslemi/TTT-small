import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

# Global variables
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def fourier_basis(x, M):
    """Generate Fourier basis functions up to order M"""
    basis = np.ones((len(x), M + 1))
    for m in range(1, M + 1):
        basis[:, m] = np.cos(m * np.pi * x)
    return basis

def construct_h_from_coeffs(coeffs, M):
    """Construct h(x) from Fourier coefficients"""
    x = np.linspace(0, 2, n_points)
    basis = fourier_basis(x, M)
    
    # Add constant term to ensure integral = 1
    # h(x) = coeffs[0] + sum_{m=1}^M coeffs[m] * cos(m * pi * x)
    # To make integral = 1, we need sum(coeffs[1:]) = 0 since integral of cos(m*pi*x) = 0
    h = coeffs[0] + np.dot(basis[:, 1:], coeffs[1:])
    
    # Ensure h is in [0,1], this is a key constraint
    h = np.clip(h, 0, 1)
    return h

def objective_with_penalty(coeffs, M, penalty_weight=1e6):
    """Objective function with penalty for violating bounds"""
    h = construct_h_from_coeffs(coeffs, M)
    c5 = evaluate_c5(h)
    
    # Add penalty for values out of [0,1] 
    violations = np.sum(np.maximum(0, h - 1) + np.maximum(0, -h))
    if violations > 0:
        c5 += penalty_weight * violations
    
    return c5
