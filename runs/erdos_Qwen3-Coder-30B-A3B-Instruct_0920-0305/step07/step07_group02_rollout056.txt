```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5_fft(h):
    """Fast evaluation of C5 using FFT"""
    dx = 2.0 / len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def greedy_descent(h0, max_iter=1000):
    """Greedy local improvement for worst shift"""
    h = h0.copy()
    for _ in range(max_iter):
        corr = correlate(h, h, mode='full')
        # Convert to overlap (1 - correlation)
        overlap = len(h) / 2.0 - corr
        worst_shift_idx = np.argmax(overlap)
        worst_shift = worst_shift_idx - (len(h) - 1)
        
        # Find mismatched pairs
        A = np.where((h == 1) & (np.roll(h, -worst_shift) == 0))[0]  # 1 in h, 0 in shifted
        B = np.where((h == 0) & (np.roll(h, -worst_shift) == 1))[0]  # 0 in h, 1 in shifted
        
        if len(A) == 0 or len(B) == 0:
            break  # No useful swap left

        # Pick random pair
        i = np.random.choice(A)
        j = np.random.choice(B)
        h[i], h[j] = h[j], h[i]  # swap bits
    return h, compute_c5_fft(h)

def anneal_binary(h0, time_limit_s):
    """Light simulated annealing with binary swap moves"""
    h = h0.copy()
    c5_cur = compute_c5_fft(h)
    start = time.time()
    T0 = 0.02
    Tmax = 2000
    t = 0

    while time.time() - start < time_limit_s:
        if t % 100 == 0 and t > 0:
            # Short greedy descent
            h, c5_cur = greedy_descent(h, max_iter=100)

        # Pick random 1 and 0 positions
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Proposed swap
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        c5_new = compute_c5_fft(h_new)
        
        # Accept or reject
        delta = c5_new - c5_cur
        T = T0 * (1 - t / Tmax)
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new.copy()
            c5_cur = c5_new
            
        t += 1
    return h, c5_cur

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    N = 120
    dx = 2.0 / N
    target_sum = N // 2
    
    # ---- MILP ----
    h_vars = cp.Variable(N, boolean=True)
    lam_var = cp.Variable(integer=True)
    constraints = []
    
    # Mass constraint
    constraints.append(cp.sum(h_vars) == target_sum)
    
    # Symmetry breaking
    constraints.append(h_vars[0] == 1)
    
    # Declare W variables and constraints for each shift
    W = {}  # dict of auxiliary binary variables w[i,k]
    for k in range(-(N-1), N):
        # valid indices for shift k
        # valid i range: so 0 <= i < N and 0 <= i+k < N
        i_valid = np.arange(max(0, -k), min(N, N - k))
        if len(i_valid) > 0:
            # Create binary auxiliary variable for each valid i,k pair
            w = cp.Variable(len(i_valid), boolean=True)
            W[(k, tuple(i_valid))] = w
            
            # Linearization: w <= h[i], w <= h[i+k], w >= h[i]+h[i+k]-1
            # So for each valid i with i+k in range:
            for idx, i in enumerate(i_valid):
                j = i + k
                constraints.append(w[idx] <= h_vars[i])
                constraints.append(w[idx] <= h_vars[j])
                constraints.append(w[idx] >= h_vars[i] + h_vars[j] - 1)
                
                # Also enforce that w[i,k] is in the sum for k shift
                # i.e., the constraint: sum{w[i,k]} >= lam
                # This is already handled in the lambda constraint below
            # Add constraint for this shift k
            constraints.append(cp.sum(w) >= lam_var)

    # Objective
    prob = cp.Problem(cp.Maximize(lam_var), constraints)
    
    # Use seconnd half of budget for solving
    remaining_time = budget_s * 0.55
    prob.solve(solver=cp.CBC, max_seconds=remaining_time, verbose=False)
    
    c5_best = 1.0 
    best_h = np.random.binomial(1, 0.5, N)
    best_h = np.clip(best_h, 0, 1)
    
    # Extract solution if feasible
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_sol = np.round(h_vars.value).astype(int)
        # repair to exact sum
        current_sum = np.sum(h_sol)
        if current_sum != target_sum:
            # Swap a few bits to meet the constraint exactly
            ones = np.where(h_sol == 1)[0]
            zeros = np.where(h_sol == 0)[0]
            swaps_needed = abs(target_sum - current_sum)
            for _ in range(min(swaps_needed, len(ones), len(zeros))):
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                h_sol[i] = 0
                h_sol[j] = 1
                ones = np.where(h_sol == 1)[0]
                zeros = np.where(h_sol == 0)[0]
        lam_val = int(lam_var.value)
        c5_best = 1.0 - (2.0 * lam_val) / N
        best_h = h_sol.copy()
    else:
        # Fallback to a random valid binary solution
        best_h = np.random.binomial(1, 0.5, N)
        best_h = np.clip(best_h, 0, 1)
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        diff = len(ones) - target_sum
        for _ in range(abs(diff)):
            if diff > 0:
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                best_h[i] = 0
                best_h[j] = 1
                ones = np.where(best_h == 1)[0]
                zeros = np.where(best_h == 0)[0]
        c5_best = compute_c5_fft(best_h)
    
    # Greedy descent
    elapsed = time.time() - start_time
    if elapsed < 0.75 * budget_s:
        best_h, c5_best = greedy_descent(best_h)
    
    # Simulated annealing
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        best_h, c5_best = anneal_binary(best_h, remaining)
        
    return (best_h.tolist(), float(c5_best), N)
```
