import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from cvxpy import *

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_convex_relaxation(h0, n_points):
    """Solve the convex relaxation of the problem."""
    dx = 2.0 / n_points
    n = n_points
    h = cp.Variable(n, name="h")
    p = cp.Variable((n, n), name="p")  # p[i,j] = h[i]*h[j] for i<=j
    T = cp.Variable(name="T")  # Overlap bound

    constraints = []
    
    # Mass constraint
    constraints.append(cp.sum(h) == n/2)
    
    # McCormick envelopes
    for i in range(n):
        for j in range(i, n):
            constraints.append(p[i, j] >= 0)
            constraints.append(p[i, j] >= h[i] + h[j] - 1)
            constraints.append(p[i, j] <= h[i])
            constraints.append(p[i, j] <= h[j])

    # Overlap constraints
    # compute the overlaps by convolution (since the domain is [0,2], shifts are (-n+1, ..., n-1))
    # But to be simple, loop over known shift indices k = 0,1,..., n-1
    # Then O_k = sum_{i=0}^{n-1-|k|} p[i, i+k] = sum_{i=0}^{n-1-|k|} p[i, i+k] 
    # We'll use defaulting if i+k < 0 or i+k >= n. For index shifts, let's be more careful:
    # Pairs such that the shift makes the window valid.
    for k in range(1, n):
        max_i = n - k
        contrib = 0
        for i in range(max_i):
            contrib += p[i, i+k]
            
        # O_k = contrib
        # Rewrite to: O_k >= n/2 - T
        constraints.append(contrib >= n/2 - T)

    # Objective
    objective = cp.Minimize(T)
    prob = cp.Problem(objective, constraints)
    
    try:
        # Use a reliable solver like ECOS for speed
        prob.solve(solver=cp.ECOS, max_iters=1000, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            sol_h = h.value
            return project_to_sum_box(sol_h, n/2)
        else:
            return h0
    except Exception:
        return h0

def subgradient_policing(h0, max_iter=2000):
    """Refine h0 via sub-gradient descent on the true objective"""
    h = h0.copy()
    dx = 2.0 / len(h0)
    for t in range(max_iter):
        # Compute correlation
        corr = compute_corr_fft(h)
        
        # Find the maximizing shift k*
        k_star = np.argmax(corr)
        # Compute sub-gradient for that term
        g = np.zeros_like(h)
        if k_star < len(h):
            for i in range(len(h)):
                if i + k_star < len(h):
                    g[i] += 1 - h[i + k_star]
                if i >= k_star:
                    g[i] -= h[i - k_star]

        # Projective step
        alpha = 0.01 / np.sqrt(t + 1)
        h_new = h - alpha * g

        # Projection
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, len(h_new)/2)
        h = h_new
        
    return h

def local_exchange_refinement(h0, max_iter=1000):
    """Refine with local mass transfers"""
    h = h0.copy()
    n = len(h0)
    for _ in range(max_iter):
        # Create random index pairs
        i, j = np.random.randint(0, n, 2)
        if i > j:
            i, j = j, i
        if i == j:
            continue
            
        # Safe transfer from j to i
        delta_max = min(h[j], 1.0 - h[i])
        if delta_max <= 0:
            continue
            
        eta = np.random.choice([0.1, 0.5, 0.9])
        delta = eta * delta_max
        
        h_test = h.copy()
        h_test[i] += delta
        h_test[j] -= delta
        
        # Ensure we stay in box
        if np.any(h_test < 0) or np.any(h_test > 1):
            continue
            
        h_test = project_to_sum_box(h_test, n/2)
        
        # Check improvement
        if evaluate_c5(h_test) < evaluate_c5(h):
            h = h_test

    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Small problem size for speed and control
    n_points = 150
    dx = 2.0 / n_points

    # Initialization using the provided values
    h_ref = np.array(initial_h_values)
    x_ref = np.linspace(0, 2, len(h_ref))
    x_new = np.linspace(0, 2, n_points)
    h0 = np.interp(x_new, x_ref, h_ref)
    h0 = project_to_sum_box(h0, n_points/2)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Phase 1: Convex relaxation to get a lower bound
    if time.time() - start_time < 0.4 * budget_s:
        relaxed_h = solve_convex_relaxation(h0, n_points)
        c5_relaxed = evaluate_c5(relaxed_h)
        if c5_relaxed < best_c5:
            best_h = relaxed_h.copy()
            best_c5 = c5_relaxed
    
    # Phase 2: Sub-gradient polishing
    if time.time() - start_time < 0.6 * budget_s:
        polished_h = subgradient_policing(best_h, max_iter=2000)
        c5_polished = evaluate_c5(polished_h)
        if c5_polished < best_c5:
            best_h = polished_h.copy()
            best_c5 = c5_polished
    
    # Phase 3: Local exchange refinement
    if time.time() - start_time < budget_s:
        refined_h = local_exchange_refinement(best_h, max_iter=1000)
        c5_refined = evaluate_c5(refined_h)
        if c5_refined < best_c5:
            best_h = refined_h.copy()
            best_c5 = c5_refined
    
    return (best_h.tolist(), float(best_c5), n_points)
