import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def initialize_h_random(n, target_sum):
    """Initialize a random step function"""
    h = np.random.uniform(0, 1, n)
    return project_to_sum_box(h, target_sum)

def gradient_descent_with_smoothing(h_start, eps_schedule, n_iters_per_eps, step_size, target_sum):
    """Gradient descent with smoothed max objective"""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for eps_idx, eps in enumerate(eps_schedule):
        n_iters = n_iters_per_eps[eps_idx]
        for i in range(n_iters):
            # Compute correlation via FFT
            h_rev = h[::-1]
            b = 1 - h
            L = 2 * len(h) - 1
            corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
            
            # Compute weights (softmax) with numerical stability
            corr_max = np.max(corr)
            corr_shifted = corr - corr_max
            w_raw = np.exp(corr_shifted / eps)
            w = w_raw / np.sum(w_raw)
            
            # Compute gradient using convolution (more efficient than loops)
            # gradient = sum_s w_s * ( (1 - h[l+s]) - h[l-s] )
            # This corresponds to:
            # conv_rev = sum_s w_s * h[l+s] and conv = sum_s w_s * h[l-s]
            # gradient = 1 - conv_rev - conv
            
            conv_wh = np.convolve(w, h, mode='full')
            conv_rev = np.convolve(w[::-1], h, mode='full')
            offset = len(h) - 1
            grad = 1.0 - (conv_rev[offset:offset + len(h)] + conv_wh[offset:offset + len(h)])
            
            # Update step (using learning rate schedule)
            lr = step_size * eps
            h_new = h - lr * grad
            
            # Project onto constraints
            h_new = project_to_sum_box(h_new, target_sum)
            h = h_new
            
            # Evaluate and track best
            if i % 10 == 0:
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    
    return best_h, best_c5

            best_c5 = c5_result

    if time.time() - start_time < budget_s - 2:
        bounds = [(0, 1) for _ in range(n_points)]

        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM

                c5_candidate = evaluate_c5(h_candidate)

                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
        except:
            pass

    return (best_h.tolist(), float(best_c5), n_points)
