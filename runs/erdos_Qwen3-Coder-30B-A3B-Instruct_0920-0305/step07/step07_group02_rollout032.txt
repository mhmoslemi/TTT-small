import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_thue_morse(n):
    """Generate Thue-Morse sequence of length n"""
    # Create Thue-Morse sequence using bitwise XOR
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    return tm.astype(float)

def generate_rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence of length n"""
    # Rudin-Shapiro sequence: a(n) = number of pairs (0,0) in binary expansion
    rs = np.zeros(n, dtype=int)
    for i in range(n):
        # Count number of bit pairs (0,0) in binary representation
        binary = format(i, 'b').zfill(8)  # 8-bit representation
        count = 0
        for j in range(len(binary)-1):
            if binary[j:j+2] == '00':
                count += 1
        rs[i] = count % 2
    return rs.astype(float)

def greedy_worst_shift_improve(h):
    """Perform greedy improvement based on worst shift"""
    h = h.copy()
    while True:
        corr = compute_corr_fft(h)
        max_shift_idx = np.argmax(corr)
        max_shift = max_shift_idx - (n_points - 1) // 2  # Map index to actual shift
        
        # Find positions where h[i]=1 and h[i+shift]=0, and where h[j]=0 and h[j+shift]=1
        if max_shift >= 0:
            # h[:n_points-max_shift] and h[max_shift:]
            h1 = h[:n_points-max_shift]
            h2 = h[max_shift:]
            mismatches1 = np.where((h1 == 1) & (h2 == 0))[0]
            mismatches2 = np.where((h1 == 0) & (h2 == 1))[0]
        else:
            # h[-max_shift:] and h[:n_points+max_shift]
            h1 = h[-max_shift:]
            h2 = h[:n_points+max_shift]
            mismatches1 = np.where((h1 == 1) & (h2 == 0))[0]
            mismatches2 = np.where((h1 == 0) & (h2 == 1))[0]
        
        if len(mismatches1) == 0 or len(mismatches2) == 0:
            break
            
        # Swap first mismatched positions
        i = mismatches1[0]
        j = mismatches2[0]
        
        if max_shift >= 0:
            i, j = i, j + max_shift  # Adjust indices for current shift
        else:
            i, j = i - max_shift, j  # Adjust indices for current shift

        # Perform swap
        h[i], h[j] = h[j], h[i]
        
        # Early termination if no more swaps improve
        corr_new = compute_corr_fft(h)
        if corr_new[max_shift_idx] >= corr[max_shift_idx]:
            break
    
    return h

def local_gradient_refinement(h0, max_iter=200):
    """Refine with gradient-based optimization on small subset"""
    h = h0.copy()
    
    # Focus on best regions found
    corr_full = compute_corr_fft(h)
    max_idx = np.argmax(corr_full)
    best_region_start = max_idx - 30 if max_idx >= 30 else 0
    best_region_end = min(max_idx + 30, n_points)
    
    # Optimize these regions with reference to the full function
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_local):
        return evaluate_c5(h_local)
    
    def constraint_eq(h_local):
        return np.sum(h_local) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
    except:
        pass
    
    # Project to maintain constraints
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    best_c5 = float('inf')
    best_h = None
    
    # Number of random restarts within the budget
    total_restarts = 300
    
    for restart in range(total_restarts):
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Try different initializations
        choices = [
            np.random.rand(n_points),
            generate_thue_morse(n_points),
            generate_rudin_shapiro(n_points),
            np.ones(n_points) * 0.5,
        ]
        
        h_initial = choices[restart % len(choices)]
        h_initial = project_to_sum_box(h_initial, TARGET_SUM)
        
        # First, improve with greedy algorithm
        h_current = greedy_worst_shift_improve(h_initial)
        
        # Then refine locally
        h_converged = local_gradient_refinement(h_current, max_iter=150)
        c5_result = evaluate_c5(h_converged)
        
        if c5_result < best_c5:
            best_c5 = c5_result
            best_h = h_converged.copy()
            
    # Do a final refinement on the best found solution
    if best_h is not None:
        final_h = local_gradient_refinement(best_h, max_iter=100)
        final_c5 = evaluate_c5(final_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
