import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import differential_evolution
import math

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with a custom function that matches the known good values
    # Use a piecewise linear construction inspired by the solution structure
    h_initial = np.zeros(n_points)
    
    # Create a pattern that is reminiscent of optimal solutions
    # This is inspired by heuristics that work well for similar problems
    mid = n_points // 2
    width = n_points // 3
    x_vals = np.arange(n_points)
    
    # Create an asymmetric but balanced pattern
    for i in range(n_points):
        if i < mid - width//2:
            h_initial[i] = 0.0
        elif i < mid + width//2:
            h_initial[i] = 1.0
        else:
            h_initial[i] = 0.0
    
    # Add some gradient in the middle
    for i in range(mid - width//2, mid + width//2):
        weight = (i - (mid - width//2)) / width
        h_initial[i] = 0.3 + 0.7 * weight
    
    # Slightly adjust to respect sum constraint
    h_initial = project_to_sum_box(h_initial, n_points / 2)
    
    best_h = h_initial.copy()
    best_c5 = evaluate_c5(h_initial)
    
    # Multi-start optimization with simulated annealing approach
    n_starts = 20
    for iteration in range(n_starts):
        if time.time() - start_time > budget_s:
            break
            
        # Random perturbations with decreasing amplitude
        amplitude = max(0.02, 0.1 * (1 - iteration / n_starts))
        
        h_perturbed = h_initial + np.random.normal(0, amplitude, n_points)
        h_perturbed = project_to_sum_box(h_perturbed, n_points / 2)
        
        try:
            # Use differential evolution as global optimizer
            bounds = [(0, 1) for _ in range(n_points)]
            res = differential_evolution(
                lambda x: evaluate_c5(x),
                bounds,
                maxiter=50,
                popsize=15,
                mutation=(0.5, 1),
                recombination=0.7,
                tol=1e-10,
                seed=seed + iteration
            )
            if res.success:
                h_optimized = project_to_sum_box(res.x, n_points / 2)
                c5_optimized = evaluate_c5(h_optimized)
                if c5_optimized < best_c5:
                    best_c5 = c5_optimized
                    best_h = h_optimized.copy()
        except:
            continue
        
        # Occasionally try more drastic changes
        if iteration % 5 == 0:
            h_alt = best_h.copy()

            positions = np.random.choice(n_points, size=min(10, n_points//10), replace=False)
            for pos in positions:
                h_alt[pos] = np.random.uniform(0, 1)
            h_alt = project_to_sum_box(h_alt, n_points / 2)
            c5_alt = evaluate_c5(h_alt)
            if c5_alt < best_c5:
                best_c5 = c5_alt
                best_h = h_alt.copy()

    # Final local refinement with gradient descent approach
    for _ in range(100):
        if time.time() - start_time > budget_s:

        # Small random changes with adaptive step size
        h_candidate = best_h + np.random.normal(0, 0.001, n_points)

        c5_candidate = evaluate_c5(h_candidate)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
