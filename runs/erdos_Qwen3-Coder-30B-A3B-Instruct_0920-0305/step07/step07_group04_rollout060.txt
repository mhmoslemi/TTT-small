import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use n=120 as default (larger n might improve bound but slower)
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    # Create variables
    h = cp.Variable(n_points, name='h')
    
    # Constraints: h in [0,1] and sum = TARGET_SUM
    constraints = [h >= 0, h <= 1, cp.sum(h) == TARGET_SUM]
    
    # Define all pairwise products using McCormick envelopes
    # For each pair (i,j), create product variable p_ij that approximates h_i * h_j
    # We'll create P matrix to store these products
    
    # For efficiency, just generate all pairs
    all_pairs = [(i,j) for i in range(n_points) for j in range(i, n_points)]
    
    # Create full matrix of products
    P = cp.Variable((n_points, n_points), name='P')
    
    # Add McCormick constraints for all pairs
    for i in range(n_points):
        for j in range(n_points):
            # Add McCormick relaxation constraints
            constraints.append(P[i,j] >= h[i] + h[j] - 1)
            constraints.append(P[i,j] <= h[i])
            constraints.append(P[i,j] <= h[j])
    
    # For diagonal elements, P[i,i] represents h[i] * h[i] = h[i]^2
    # The McCormick bounds are tight for diagonal since h[i] >= h[i] + h[i] - 1 = 2*h[i] - 1
    # So P[i,i] >= 2*h[i] - 1 and P[i,i] <= h[i] and P[i,i] >= 0
    # When h[i] >= 0.5, P[i,i] >= 2*h[i] - 1 >= 0, and when h[i] < 0.5, h[i] >= 0
    # But since we want P[i,i] = h[i]^2, tight constraint for diagonal is P[i,i] = h[i]^2
    # But McCormick gives convex upper bound and lower bound including P[i,i] = h[i]^2.
    # Since we solve LP maximizing min overlap in each shift, we'll just apply the Hessian bounds.
    
    # For each shift k in [-n+1, n-1], compute overlap and apply constraints
    # But we'll simplify and consider only k in [0, n-1] due to symmetry
    # The correlation for shift k using zero-padding mode = sum_{i=0}^{n-1-k} h_i * (1 - h_{i+k})
    # Which simplifies to sum_i h_i - sum_{i=0}^{n-1-k} P[i,i+k] (this uses McCormick relaxation)
    
    # To minimize max correlation across all shifts, we introduce t and add constraints
    # that for each shift k we have correlation_k <= t
    t = cp.Variable(1, name='t')
    max_shift = n_points - 1
    
    # For each shift, compute the corresponding corr_k
    # For shift k, overlaps are at indices i and i+k where both are valid
    corr_constraints = []
    
    for k in range(max_shift + 1):
        # Compute correlation using McCormick relaxation
        # For shift k, we sum over i from 0 to n-1-k
        # correlation_k = sum_{i=0}^{n-1-k} h_i - sum_{i=0}^{n-1-k} P[i,i+k]
        if k == 0:
            # Special case: no shift, so h_i * (1 - h_i) = h_i - h_i^2
            # These are just sums over all valid pairs
            corr = cp.sum(h) - cp.sum(P[0:n_points-k, 0:n_points-k])
        else:
            # For shift k, overlapping indices go from 0 to n-1-k
            corr = cp.sum(h[0:n_points-k]) - cp.sum(P[0:n_points-k, k:n_points])
        
        # Add constraint that corr <= t (we minimize t)
        corr_constraints.append(corr <= t)
    
    # Add all constraints
    constraints.extend(corr_constraints)
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    
    # Create optimization problem
    prob = cp.Problem(objective, constraints)
    
    # Solve using CBC (fast, open-source)
    solver_options = {'max_seconds': budget_s - 0.5}
    try:
        prob.solve(solver=solver, verbose=False, **solver_options)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_EOF]:
            h_solution = h.value
            final_t = t.value
            # Compute actual C5 value for this solution
            # Use true computation using correlation function
            # Not using linear approximation for accurate result
            actual_c5 = compute_actual_c5(h_solution)
            return (h_solution.tolist(), float(actual_c5), n_points)
        else:
            # Fallback to simpler solution if optimization failed
            pass
    except:
        # Fallback if LP fails
        pass
    
    # Simple fallback solution: piecewise constant with zeros at edges
    h_fallback = np.zeros(n_points)
    center_len = int(n_points * 0.5)
    start_idx = (n_points - center_len) // 2
    h_fallback[start_idx:start_idx+center_len] = 1.0
    # Scale to match sum
    h_fallback = h_fallback * TARGET_SUM / np.sum(h_fallback)
    actual_c5 = compute_actual_c5(h_fallback)
    
    return (h_fallback.tolist(), float(actual_c5), n_points)

def compute_actual_c5(h_values):
    # This evaluates the actual function without linear approximations
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Full correlation using numpy 
    corr_full = np.correlate(h, 1 - h, mode='full')
    # Take maximum of full correlation times dx
    max_corr = np.max(corr_full) * dx
    return max_corr
