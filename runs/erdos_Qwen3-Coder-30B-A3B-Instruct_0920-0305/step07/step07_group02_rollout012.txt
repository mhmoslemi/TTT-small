import numpy as np

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_thue_morse(n):
    """Generate Thue-Morse sequence with 0s and 1s"""
    tm = [0]
    for i in range(n):
        # Extend to length 2*i
        tm.extend([1-x for x in tm])
        if len(tm) >= n:
            break
    return np.array(tm[:n], dtype=float)

def generate_rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence (0s and 1s)"""
    rs = [0] * n
    for i in range(1, n):
        # Rudin-Shapiro can be computed via bit counting
        rs[i] = (bin(i).count('1') % 2)
    return np.array(rs, dtype=float)

def find_indices_at_shift(h1, h2, shift):
    """Find indices where (h1[i], h2[i+shift]) = (1, 0) or (0, 1)"""
    indices = []
    if shift >= 0:
        left = h1[:-shift] if shift > 0 else h1
        right = h2[shift:] if shift > 0 else h2
        mismatches = np.where((left == 1) & (right == 0))[0]
    else:
        left = h1[-shift:] if shift < 0 else h1
        right = h2[:shift] if shift < 0 else h2
        mismatches = np.where((left == 1) & (right == 0))[0]
    return mismatches

def greedy_improvement(h0):

def change_sum(h, target_sum):
    """Change sum to target_sum exactly (fixed sum issue)"""
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    indices = np.argsort(h)  # index from smallest to largest
    if diff > 0:
        # Need to turn some 0s to 1s
        for i in range(int(diff + 0.5)):
            if h[indices[i]] < 1:
                h[indices[i]] = 1
    else:
        # Need to turn some 1s to 0s
        for i in range(int(-diff + 0.5)):
            if h[indices[-i-1]] > 0:
                h[indices[-i-1]] = 0
                
    return h

def simulated_annealing(h0, T_start=10.0, T_end=0.001, decay=0.999):
    """Simulated Annealing for global optimization"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    current_c5 = best_c5
    T = T_start
    
    while T > T_end:
        # Try large swaps in 1D
        i = random.randint(0, n_points - 1)
        j = random.randint(0, n_points - 1)
        
        # Swap two values
        h_prev = h[i]
        h[i] = h[j]
        h[j] = h_prev
        
        # Ensure sum constraint still holds approximately
        new_c5 = evaluate_c5(h)
        
        # Accept according to Metropolis criterion
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_c5 = new_c5
        else:
            # Revert swap
            h_prev = h[i]
            h[i] = h[j]
            h[j] = h_prev
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        T *= decay
    
    return best_h
