import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_initial_h():
    """Create a smoothed initial solution"""
    # Start with a basic piecewise constant solution
    h = np.ones(n_points) * 0.5
    
    # Create a more interesting pattern with varying heights
    # Create 4 segments with different values
    segments = 4
    seg_size = n_points // segments
    for i in range(segments):
        start_idx = i * seg_size
        end_idx = (i + 1) * seg_size if i < segments - 1 else n_points
        if i % 2 == 0:
            h[start_idx:end_idx] = 0.7
        else:
            h[start_idx:end_idx] = 0.3
    
    # Adjust sum to match target
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def simulated_annealing(h0, max_iter=2000):
    """Improvement using simulated annealing approach"""
    h = h0.copy()
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Annealing parameters
    temp_start = 0.01
    temp_end = 0.0001
    cooling_rate = 0.999
    
    for i in range(max_iter):
        # Cool temperature
        temp = temp_start * (cooling_rate ** i)
        if temp < temp_end:
            temp = temp_end

        # Make a small random change
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        change = np.random.normal(0, 0.02)
        h_new[idx] += change
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        
        # Accept or reject based on Metropolis criterion
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h = h_new
            current_c5 = new_c5
            
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    return best_h, best_c5

def optimize_with_pattern_smoothing(h0, max_iter=1000):
    """Optimizes using local smoothing around maximum overlap regions"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Find the current worst overlap position
        corr = compute_corr_fft(h)
        max_corr_idx = np.argmax(corr)
        
        # Create a focal region around the maximum correlation
        focal_start = max(0, max_corr_idx - 20) 
        focal_end = min(len(h), max_corr_idx + 20)
        
        # Modify the focal region
        h_new = h.copy()
        # Smooth the values within the focal region
        for j in range(focal_start, focal_end):
            h_new[j] = np.clip(h_new[j] + np.random.normal(0, 0.005), 0, 1)
        
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Use a better initial construction based on known patterns
    best_h = generate_initial_h()   
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: Simulated Annealing
    phase1_end = start_time + 0.4 * budget_s
    iter_count = 0
    while time.time() < phase1_end and iter_count < 1000:
        iter_count += 1
        h_sa, c5_sa = simulated_annealing(best_h.copy(), max_iter=10)
        if c5_sa < best_c5:
            best_h = h_sa.copy()
            best_c5 = c5_sa
    
    # Phase 2: Pattern-based smoothing
    phase2_end = start_time + 0.7 * budget_s
    iter_count = 0
    while time.time() < phase2_end and iter_count < 1000:
        iter_count += 1
        h_smooth = optimize_with_pattern_smoothing(best_h.copy(), max_iter=20)
        c5_smooth = evaluate_c5(h_smooth)
        if c5_smooth < best_c5:
            best_h = h_smooth.copy()
            best_c5 = c5_smooth
    
    # Phase 3: Local refinement
    phase3_end = start_time + budget_s
    while time.time() < phase3_end:
        # Random small perturbations to escape local minima
        h_noise = best_h + np.random.normal(0, 0.01, n_points) * 0.3
        h_noise = np.clip(h_noise, 0, 1)
        h_noise = project_to_sum_box(h_noise, TARGET_SUM)
        
        # Try local optimization
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h_local):
            return evaluate_c5(h_local)
        
        def constraint_eq(h_local):
            return np.sum(h_local) - TARGET_SUM
        
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                h_noise, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 50, 'ftol': 1e-10}
            )
            if res.success:
                h_refined = project_to_sum_box(res.x, TARGET_SUM)
                c5_refined = evaluate_c5(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
