import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h, w_norm):
    """Compute gradient of smoothed max correlation"""
    n = len(h)
    # Compute convolution contributions
    # w_norm has length 2*n-1 corresponding to shifts -n+1 to n-1
    conv_wh = np.convolve(w_norm, h, mode='full')
    conv_rev = np.convolve(w_norm[::-1], h, mode='full')
    offset = n - 1
    # Extract the relevant portion
    grad = 1.0 - (conv_rev[offset:offset+n] + conv_wh[offset:offset+n])
    return grad

def initialize_h_random(n, target_sum):
    """Generate random initial function values"""
    h = np.random.random(n)
    h = project_to_sum_box(h, target_sum)
    return h

def optimize_h(start_h, eps_schedule, iter_schedule, lr, target_sum):
    """Optimize h using gradient descent with smoothing"""
    h = start_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for eps_idx, (eps, max_iters) in enumerate(zip(eps_schedule, iter_schedule)):
        for i in range(max_iters):
            # Compute correlation
            corr = np.correlate(h, 1 - h, mode='full')
            
            # Compute softmax weights
            max_corr = np.max(corr)
            w_raw = np.exp((corr - max_corr) / eps)
            w_norm = w_raw / w_raw.sum()
            
            # Compute gradient
            gradient = compute_gradient(h, w_norm)
            
            # Update h
            h_new = h - lr * gradient
            h_new = project_to_sum_box(h_new, target_sum)
            h = h_new
            
            # Evaluate and update best
            if i % 10 == 0:
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    
    return best_h, best_c5

    iter_schedule = [2000, 2000, 1500, 1000, 800, 600, 500]

    TARGET_SUM = n_points / 2
    best_h = h_interpolated.copy()
    best_c5 = evaluate_c5(h_interpolated)
    
    # Try multiple different starting points
    initial_guesses = []
    
    # Perturbation 1: slightly adjust initial values
    h1 = h_interpolated.copy()

    h1 = project_to_sum_box(h1, TARGET_SUM)
    initial_guesses.append(h1)
    
    # Perturbation 2: symmetric bell-shaped
    h2 = np.zeros(n_points)
    center = n_points // 2
    std = n_points // 6
    for i in range(n_points):
        x = (i - center) / std
        h2[i] = np.exp(-0.5 * x**2)
    h2 = project_to_sum_box(h2, TARGET_SUM)
    initial_guesses.append(h2)
    
    # Perturbation 3: random initialization
    h3 = initialize_h_random(n_points, TARGET_SUM)
    initial_guesses.append(h3)
    
    # Perturbation 4: piecewise constant with random levels
    h4 = np.zeros(n_points)
    segments = 5
    seg_size = n_points // segments
    for i in range(segments):
        start = i * seg_size
        end = (i+1) * seg_size if i < segments-1 else n_points
        h4[start:end] = np.random.uniform(0, 1)
    h4 = project_to_sum_box(h4, TARGET_SUM)
    initial_guesses.append(h4)
    
    # Try optimization from multiple starting points
    for i, h_start in enumerate(initial_guesses):
        if time.time() - start_time > budget_s:
            break
            
        try:
            h_opt, c5_opt = optimize_h(h_start, eps_schedule, iter_schedule, lr, TARGET_SUM)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception as e:
            continue
    
    # Refinement using scipy minimize for final improvement
    if time.time() - start_time < budget_s - 10:
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            def objective(h):
                return evaluate_c5(h)
            
            def constraint_eq(h):
                return np.sum(h) - TARGET_SUM
            
            cons = {'type': 'eq', 'fun': constraint_eq}
            
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_candidate = project_to_sum_box(res.x, TARGET_SUM)
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
        except:
            pass

    return (best_h.tolist(), float(best_c5), n_points)
