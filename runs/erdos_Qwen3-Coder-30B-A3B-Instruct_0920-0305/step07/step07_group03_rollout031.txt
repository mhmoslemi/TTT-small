import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr(h):
    """Compute correlation using FFT for speed"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_of_corr_at_lag(h, lag):
    """Compute gradient of correlation w.r.t h for specific lag"""
    n = len(h)
    g = np.ones(n)
    
    if lag >= 0:
        # i from 0 to n-lag-1
        if lag < n:
            g[:n-lag] = 1 - h[lag:]  # for positions where i+lag in-range
            g[lag:] -= h[:n-lag]     # subtract h_{i-lag} (if i-lag >= 0)
    else:  # lag < 0, let d = -lag
        d = -lag
        # i from d to n-1 
        if d < n:
            g[d:] = 1 - h[:n-d]
            g[:n-d] -= h[d:]
    
    return g

def subgradient_descent(h0, max_iter=10000, step_size=0.2):
    """Perform subgradient descent with diminishing step sizes""" 
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for i in range(max_iter):
        corr = compute_corr(h)
        max_val = np.max(corr)
        max_indices = np.where(np.abs(corr - max_val) < 1e-10)[0]
        
        grad_sum = np.zeros(len(h))
        for idx in max_indices:
            lag = idx - (n_points - 1)
            grad_sum += gradient_of_corr_at_lag(h, lag)
        
        if len(max_indices) > 0:
            g = grad_sum / len(max_indices)
        else:
            g = np.zeros(len(h))
            
        step = step_size / np.sqrt(i + 1)
        h_new = h - step * g
        
        # Project back to constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new
    
    return best_h, best_c5

def local_search_binary(h0, num_iterations=5000):
    """Perform local search with swaps on binary vectors"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for i in range(num_iterations):
        # Try a swap
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        if len(ones) > 0 and len(zeros) > 0:
            i1 = np.random.choice(ones)
            i0 = np.random.choice(zeros)
            
            h_new = h.copy()
            h_new[i1] = 0
            h_new[i0] = 1
            
            # Ensure sum is still correct
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            c5_new = compute_c5(h_new)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                
            h = h_new
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: Global optimization with random restarts
    best_h = None
    best_c5 = float('inf')
    
    # Try multiple random restarts
    for restart in range(10):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Initialize binary random solution
        h = np.zeros(n_points)
        ones_indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h[ones_indices] = 1
        
        # Local search on binary solution
        h_binary, c5_binary = local_search_binary(h, num_iterations=1000)
        
        # Upgrade to continuous solution
        h_continuous, c5_continuous = subgradient_descent(h_binary, max_iter=2000)
        
        if c5_continuous < best_c5:
            best_h = h_continuous.copy()
            best_c5 = c5_continuous
            
    # Phase 2: Final refinement 
    if best_h is not None:
        final_h, final_c5 = subgradient_descent(best_h, max_iter=3000)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    return (best_h.tolist(), float(best_c5), n_points)
