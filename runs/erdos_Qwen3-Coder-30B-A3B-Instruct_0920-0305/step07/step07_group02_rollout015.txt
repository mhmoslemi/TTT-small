def c5_fft(h):
    """FFT-based computation of C5"""
    n = len(h)
    nfft = 1 << ((2*n-1).bit_length())  # Next power of two
    H = np.fft.rfft(h, nfft)
    B = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * B, nfft)[:2*n-1]
    return np.max(conv) * dx

def project_to_sum(h, target_sum):
    """Project h onto the simplex: sum(h) = target_sum, h in [0,1]"""
    h = np.array(h)
    # Binary search on lambda
    def sum_lam(lam):
        ph = np.clip(h + lam, 0, 1)
        return np.sum(ph) - target_sum

    a, b = -1.0, 1.0
    for _ in range(100):
        m = (a + b) / 2
        fm = sum_lam(m)
        if abs(fm) < 1e-12:
            break
        if fm * sum_lam(a) < 0:
            b = m
        else:
            a = m
    h_proj = np.clip(h + (a + b)/2, 0, 1)
    return h_proj

def thue_morse(length):
    """Generate Thue-Morse sequence"""
    tm = np.zeros(length, dtype=int)
    for i in range(length):
        tm[i] = bin(i).count('1') % 2
    return tm

def rudin_shapiro(length):
    """Generate Rudin-Shapiro sequence"""
    rs = np.zeros(length, dtype=int)
    for i in range(length):
        ones = bin(i).count('1')
        ones_mod_2 = ones % 2
        neg = bin(i & (i >> 1)).count('1') % 2
        rs[i] = (ones_mod_2 + neg) % 2
    return rs

def greedy_descent(h0):
    """Perform greedy descent on the worst shift"""
    h = h0.copy()
    while True:
        corr = correlate(h, 1-h, mode='full')
        s = np.argmax(corr) - (N-1)
        if s >= 0:
            pos = np.where((h[:N-s] == 1) & (h[s:] == 0))[0]
            neg = np.where((h[:N-s] == 0) & (h[s:] == 1))[0]
        else:
            pos = np.where((h[-s:] == 1) & (h[:-s] == 0))[0]
            neg = np.where((h[-s:] == 0) & (h[:-s] == 1))[0]
        
        if len(pos) == 0 or len(neg) == 0:
            break
        
        i = np.random.choice(pos)
        j = np.random.choice(neg)
        if s >= 0:
            h[i] = 0
            h[j] = 1
        else:
            h[i] = 0
            h[j] = 1 
    return h
