```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def random_binary_initial(n):
    """Create random binary initial solution with exactly n/2 ones"""
    h = np.zeros(n)
    ones_idx = np.random.choice(n, n//2, replace=False)
    h[ones_idx] = 1.0
    return h

def greedy_improvement(h):
    """Improve by targeting the worst correlation shift"""
    n = len(h)
    max_iter = 10000
    iter_count = 0
    while iter_count < max_iter:
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n - 1)
        
        # Determine overlapping region indices:
        if max_shift >= 0:
            i_range = np.arange(0, n - max_shift)
            j_range = i_range + max_shift
        else:
            i_range = np.arange(-max_shift, n)
            j_range = i_range + max_shift  # shift negative pushes left
            
        # Among these positions, find mismatched pairs
        # i_range corresponds to indices of first vector
        h_i = h[i_range]
        h_j = h[j_range]
        
        # (1,0) mismatches: h_i == 1 and h_j == 0
        im1 = i_range[(h_i == 1) & (h_j == 0)]
        # (0,1) mismatches: h_i == 0 and h_j == 1
        im2 = i_range[(h_i == 0) & (h_j == 1)]
        
        # If no mismatched pairs, can't improve this shift
        if im1.size == 0 or im2.size == 0:
            break
            
        # pick random i, j
        i = np.random.choice(im1)
        j = np.random.choice(im2)
        
        # swap
        h[i] = 0
        h[j] = 1
        
        iter_count += 1
        
    return h

def local_gradient_refinement(h0, max_iter=200):
    """Refine with gradient-based optimization"""
    h = h0.copy()
    
    # Focus on best regions found
    corr_full = compute_corr_fft(h)
    max_idx = np.argmax(corr_full)
    
    # Optimize with reference to the full function
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_local):
        return compute_c5(h_local)
    
    def constraint_eq(h_local):
        return np.sum(h_local) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use fewer iterations and local optimizer
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
    except:
        pass
    
    # Project to maintain constraints
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Phase 1: Multiple random restarts with greedy improvement
    restarts = 10
    phase1_end = start_time + 0.8 * budget_s
    
    for restart in range(restarts):
        if time.time() > phase1_end:
            break
            
        # Random binary initialization
        h = random_binary_initial(n_points)
        h = greedy_improvement(h)
        c5 = compute_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 2: Fine-tune with gradient refinement
    phase2_end = start_time + 0.95 * budget_s
    
    if best_h is not None:
        h_refined = local_gradient_refinement(best_h, max_iter=500)
        c5_refined = compute_c5(h_refined)
        
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    
    # Phase 3: Final refinement using more gradient steps
    if time.time() < phase2_end and best_h is not None:
        h_final = local_gradient_refinement(best_h, max_iter=1000)
        c5_final = compute_c5(h_final)
        
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    # Ensure we have a solution
    if best_h is None:
        best_h = random_binary_initial(n_points)
        best_c5 = compute_c5(best_h)
        
    return (best_h.tolist(), float(best_c5), n_points)
```
