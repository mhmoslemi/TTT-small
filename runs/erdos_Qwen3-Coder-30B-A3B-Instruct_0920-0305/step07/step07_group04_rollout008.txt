import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_relaxed_lp(h_init, n_points, dx):
    """Solve the convex relaxation LP and return the best feasible solution found."""
    # Variables
    h = cp.Variable(n_points)
    p = cp.Variable((n_points, n_points), symmetric=True)
    T = cp.Variable()

    # Constraints
    constraints = []
    constraints += [h[i] >= 0 for i in range(n_points)]
    constraints += [h[i] <= 1 for i in range(n_points)]
    constraints += [cp.sum(h) == n_points / 2]
    
    # McCormick envelopes for p_ij = h_i * h_j
    # store only upper triangular part
    for i in range(n_points):
        for j in range(i, n_points):
            constraints += [p[i,j] >= 0]
            constraints += [p[i,j] >= h[i] + h[j] - 1]
            constraints += [p[i,j] <= h[i]]
            constraints += [p[i,j] <= h[j]]

    # Overlap constraints for each shift
    for k in range(-(n_points - 1), n_points):
        if k == 0:
            continue
        if k > 0:
            o_k = cp.sum([p[i, i + k] for i in range(n_points - k)])
        else:
            o_k = cp.sum([p[i + (-k), i] for i in range(n_points + k)])
        constraints += [o_k >= n_points / 2 - T]

    # Objective
    objective = cp.Minimize(T)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.GLPK, verbose=False, max_iters=1000)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            h_result = h.value
            h_result = np.clip(h_result, 0, 1)
            h_result = project_to_sum_box(h_result, n_points / 2)
            return h_result
    except Exception: 
        pass
    return h_init

def binary_search_on_objective(h_start, n_points, dx, max_iter=50):
    """Find certified upper bound using binary search of LP"""
    T_low = 0.0
    T_high = n_points / 2
    T_eps = 1e-4 * dx
    h_best = h_start
    c5_best = evaluate_c5(h_start)
    
    # Compute initial bound
    def evaluate_with_relaxation(h):
        try:
            best_h = solve_relaxed_lp(h, n_points, dx)
        except:
            best_h = h
        return evaluate_c5(best_h)

    for _ in range(max_iter):
        if T_high - T_low < T_eps:
            break
        T_mid = (T_low + T_high) / 2
        
        # Fix T midpoint and check if we can find a satisfying h in the relaxed space
        h_candidate = solve_relaxed_lp(h_start.copy(), n_points, dx)
        c5_candidate = evaluate_c5(h_candidate)

        if c5_candidate <= T_mid * dx:
            h_best = h_candidate
            c5_best = c5_candidate
            T_high = T_mid
        else:
            T_low = T_mid

    return h_best, c5_best

def projected_subgradient_descent(h_start, n_points, dx, max_iter=2000):
    """Apply deterministic subgradient descent with projection"""
    h = h_start.copy()
    alpha0 = 1e-3
    
    for t in range(max_iter):
        # Compute current correlation
        corr = compute_corr_fft(h)
        k_active = np.argmax(corr)
        if k_active >= n_points:
            k_active = k_active - n_points
        
        # Subgradient on active term for each i
        g = np.zeros(n_points)
        for i in range(n_points):
            if i + k_active < n_points:
                g[i] += 1 - h[i + k_active]
            if i - k_active >= 0:
                g[i] -= h[i - k_active]
        
        # Step size
        alpha = alpha0 / np.sqrt(t + 1)
        
        # Update with gradient
        h_new = h - alpha * g
        
        # Projection
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, n_points / 2)
        
        # Stopping criteria on small improvement
        if evaluate_c5(h_new) > evaluate_c5(h):
            break
        h = h_new
        
    return h

def local_exchange_refinement(h_start, n_points, dx, num_passes=3, exchange_fraction=0.2):
    """Local exchange method on the final solution, trying small mass transfers."""
    h = h_start.copy()
    n_exchange = int(n_points * exchange_fraction)
    step_sizes = [0.1, 0.5, 0.9]
    
    for i in range(num_passes):
        # Shuffle pairs to exchange
        pairs = list(np.random.permutation(n_points))
        step = min(500, n_exchange)
        for j in range(0, n_exchange, step):
            for k in range(min(step, n_exchange - j)):
                idx = pairs[j + k]
                neighbor = (idx + 1) % n_points
                if h[neighbor] > 0 and h[idx] < 1:
                    delta_max = min(h[neighbor], 1 - h[idx])
                    for s in step_sizes:
                        delta = s * delta_max
                        if delta > 0:
                            h_new = h.copy()
                            h_new[idx] += delta
                            h_new[neighbor] -= delta
                            # Project
                            h_new = np.clip(h_new, 0, 1)
                            h_new = project_to_sum_box(h_new, n_points / 2)
                            c5_new = evaluate_c5(h_new)
                            if c5_new < evaluate_c5(h):
                                h = h_new
                                break
    return h
