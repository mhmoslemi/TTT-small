import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * 2.0 / len(h)

def project_sum_box(h, target_sum):
    """Project vector h onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(h + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def smooth_gradient(h, corr, beta):
    """Compute gradient of smooth max surrogate"""
    # Center the correlation for numerical stability
    corr_centered = corr - np.max(corr)
    exp_corr = np.exp(beta * corr_centered)
    w = exp_corr / np.sum(exp_corr)
    
    # Gradient of smooth max = dx * (1 - sum of weighted shifts)
    grad = np.zeros_like(h)
    n = len(h)
    
    # Manually construct the gradient based on full correlation
    for i in range(n):
        for j in range(n):
            k = j - i
            if -n+1 <= k <= n-1:
                idx = k + n - 1  # Index in correlation array
                if 0 <= idx < 2*n - 1:
                    grad[i] += w[idx] * (1 - h[j])
        grad[i] -= w[n-1] * h[i]  # Weight for zero lag
    
    return 2.0 / n * grad

def max_lag_gradient(h, corr, idx_best):
    """Compute exact sub-gradient for the max lag"""
    grad = np.zeros_like(h)
    n = len(h)
    k_star = idx_best - (n - 1)  # Convert correlation index to actual lag
    
    # Split into separate cases for k_star > 0 and k_star < 0
    if k_star > 0:
        # contribution from (1 - h[i + k_star])
        for i in range(n - k_star):
            grad[i] += (1 - h[i + k_star]) * 2.0 / n
        # contribution from (-h[i - k_star])
        for i in range(k_star, n):
            grad[i] -= h[i - k_star] * 2.0 / n
    elif k_star < 0:
        # contribution from (1 - h[i + k_star])
        k_abs = -k_star
        for i in range(n - k_abs):
            grad[i] += (1 - h[i + k_abs]) * 2.0 / n
        # contribution from (-h[i - k_abs])
        for i in range(k_abs, n):
            grad[i] -= h[i - k_abs] * 2.0 / n
    else:
        # k_star == 0
        for i in range(n):
            grad[i] -= h[i] * 2.0 / n
    
    return grad

def mass_transfer(h, grad, epsilon):
    """Move mass greedily based on gradient directions"""
    d = -grad
    n = len(h)
    sources = []
    sinks = []
    
    # Identify sources and sinks
    for i in range(n):
        if d[i] < 0:
            sources.append((i, -d[i], h[i]))
        elif d[i] > 0:
            sinks.append((i, d[i], 1 - h[i]))
    
    # Sort by decreasing d values
    sources.sort(key=lambda x: x[1], reverse=True)
    sinks.sort(key=lambda x: x[1], reverse=True)
    
    h_copy = h.copy()
    # Transfer mass
    for s_idx, _, s_max in sources:
        s_amt = s_max
        for t_idx, _, t_max in sinks:
            if s_amt <= 0:
                break
            transfer = min(s_amt, t_max, epsilon)
            if transfer <= 0:
                continue
            h_copy[s_idx] -= transfer
            h_copy[t_idx] += transfer
            s_amt -= transfer
            # Update sink's remaining amount
            for i in range(len(sinks)):
                if sinks[i][0] == t_idx:
                    sinks[i] = (t_idx, sinks[i][1], t_max - transfer)
                    break

    # Re-project onto box and sum
    h_copy = project_sum_box(h_copy, len(h) / 2)
    return h_copy

def backtracking_line_search(h, grad, corr, cmax, beta, max_iter=20):
    alpha = 0.2
    min_alpha = 5e-4
    alpha_decay = 0.5
    
    for _ in range(max_iter):
        h_new = h - alpha * grad
        h_new = project_sum_box(h_new, len(h) / 2)
        cmax_new = evaluate_c5(h_new)
        
        if cmax_new < cmax:
            return h_new, cmax_new, alpha
        else:
            alpha *= alpha_decay
            if alpha < min_alpha:
                break
    
    return h, cmax, 0.0
