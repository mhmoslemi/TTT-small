import numpy as np
import time

# Global variables
n_points = None
dx = None
TARGET_SUM = None

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h, lag):
    """Compute gradient of correlation at given lag"""
    n = len(h)
    grad = np.ones(n)
    if lag >= 0:
        if lag < n:
            grad[:n-lag] -= h[lag:]
            grad[lag:] -= h[:n-lag]
    else:
        lag = -lag
        if lag < n:
            grad[lag:] -= h[:n-lag]
            grad[:lag] -= h[lag:]
    return grad

        src_candidates = np.where((grad > 1e-8) & (h > 1e-12))[0]

        if src_candidates.size > 0 and tgt_candidates.size > 0:
            # Pick source with max gradient and target with min gradient
            src = src_candidates[np.argmax(grad[src_candidates])]
            tgt = tgt_candidates[np.argmin(grad[tgt_candidates])]
            
            # Compute available delta
            delta_allowed = min(h[src], 1.0 - h[tgt])

                h[src] -= delta
                h[tgt] += delta
                
                # Evaluate new C5
                c5_new = evaluate_c5(h)
                if c5_new < c5:
                    c5 = c5_new
                    # Recompute correlation and lag
                    corr = np.correlate(h, 1 - h, mode='full')
                    lag_max = int(np.argmax(corr) - (n - 1))
                else:
                    # Revert move
                    h[src] += delta
                    h[tgt] -= delta
        else:
            # Fallback: random move
            i, j = np.random.choice(n, 2, replace=False)
            if h[i] > 1e-12 and h[j] < 1 - 1e-12:
                delta_allowed = min(h[i], 1.0 - h[j])

                delta = min(delta, delta_allowed)
                
                h[i] -= delta

                c5_new = evaluate_c5(h)
                if c5_new < c5:

                    corr = np.correlate(h, 1 - h, mode='full')

                else:
                    # Revert
                    h[i] += delta
                    h[j] -= delta

        # Occasionally do random perturbation

            for _ in range(2):

                corr = np.correlate(h, 1 - h, mode='full')

            c5 = evaluate_c5(h)
            corr = np.correlate(h, 1 - h, mode='full')
            lag_max = int(np.argmax(corr) - (n - 1))
    
    return h, c5

def random_binary(n):
    """Create random binary vector with exact sum constraint"""
    ones = np.zeros(n)
    idx = np.random.choice(n, n//2, replace=False)
    ones[idx] = 1.0
    return ones
