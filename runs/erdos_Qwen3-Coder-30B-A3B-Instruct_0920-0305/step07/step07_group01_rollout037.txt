import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def perturb_solution(h, strength=0.01):
    """Create a perturbed version of the solution"""
    perturbed = h + np.random.normal(0, strength, n_points)
    return project_to_sum_box(perturbed, TARGET_SUM)

def generate_initial_solution():
    """Generate a good initial solution by combining analytical and heuristic approaches"""
    # Start with a simple construction: a step function with a peak at center and tails
    h = np.zeros(n_points)
    
    # Center peak
    peak_width = n_points // 4
    peak_start = n_points // 2 - peak_width // 2
    peak_end = peak_start + peak_width
    h[peak_start:peak_end] = 1.0
    
    # Add small tails to better satisfy the constraint
    tail_width = n_points // 8
    h[:tail_width] = 0.2
    h[-tail_width:] = 0.2
    
    # Normalize to target sum
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def gradient_based_local_search(h0, max_iter=500):

    best_h = h.copy()

    last_improvement = 0
    step_size = 0.005
    
    for i in range(max_iter):
        if i - last_improvement > 30:
            step_size *= 0.95  # Reduce step size if no improvement
            
        # Get a better gradient approximation by finite differences
        grad = np.zeros(n_points)
        
        # Evaluate on a few directions
        for k in range(n_points):
            # Forward difference
            h_test = h.copy()
            h_test[k] = min(1.0, h[k] + step_size)
            c5_plus = evaluate_c5(h_test)
            
            h_test = h.copy()
            h_test[k] = max(0.0, h[k] - step_size)
            c5_minus = evaluate_c5(h_test)
            
            grad[k] = (c5_plus - c5_minus) / (2 * step_size + 1e-12)
        
        # Try to improve by moving along negative gradient
        # But be more conservative and adaptive
        improvement_found = False
        for _ in range(10):
            # Find the point that it's most beneficial to decrease
            idx_decrease = np.argmin(grad)
            # And the point that it's most beneficial to increase
            idx_increase = np.argmax(grad)
            
            if grad[idx_decrease] >= 0 or grad[idx_increase] <= 0:
                break  # No improvement possible
                
            # Determine the amount to move
            move_amount = min(step_size, h[idx_decrease], 1 - h[idx_increase])
            if move_amount < 1e-6:
                break
                
            # Apply the change
            new_h = h.copy()
            new_h[idx_decrease] -= move_amount
            new_h[idx_increase] += move_amount
            
            # Project to valid region
            new_h = project_to_sum_box(new_h, TARGET_SUM)
            
            c5_new = evaluate_c5(new_h)
            if c5_new < best_c5:
                h = new_h
                best_c5 = c5_new
                best_h = h.copy()
                improvement_found = True
                last_improvement = i
                break
        
        if not improvement_found:
            # Simple random moves if no gradient improvement 
            if np.random.random() < 0.1:
                k1 = np.random.randint(0, n_points)
                k2 = np.random.randint(0, n_points)
                move = min(step_size, h[k1], 1 - h[k2])
                if move > 1e-6:
                    h[k1] -= move
                    h[k2] += move
                    h = project_to_sum_box(h, TARGET_SUM)
                    c5_new = evaluate_c5(h)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                        last_improvement = i
    
    return best_h, best_c5
