import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 180
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_grad_max_shift(h, shift, dx):
    """Compute gradient of C5 for the maximum shift"""
    grad = np.zeros_like(h)
    n = len(h)
    
    if shift >= 0:
        # Positive shift: C5(k) = sum_{i=0}^{n-1-k} h[i] * (1 - h[i+k])
        # Derivative w.r.t h[i]: (1 - h[i+k]) if i+k < n, -h[i-k] if i-k >= 0
        if shift < n:
            grad[0:n-shift] += (1 - h[shift:]) * dx
            grad[shift:] += -h[:n-shift] * dx
    else:
        # Negative shift: C5(k) = sum_{i=|k|}^{n-1} h[i] * (1 - h[i+k])
        # Derivative: (1 - h[i+k]) if i+k >= 0, -h[i-k] if i-k < n
        abs_shift = -shift
        if abs_shift < n:
            grad[abs_shift:] += (1 - h[:n-abs_shift]) * dx
            grad[:n-abs_shift] += -h[abs_shift:] * dx
    
    return grad

    h = h_interpolated.copy()
    
    # Main optimization loop with gradient descent
    total_iter = 20000

    for iter in range(total_iter):

        c5_vals = corr * dx
        k_idx = np.argmax(c5_vals) - (n_points - 1)

        grad = compute_grad_max_shift(h, k_idx, dx)
        
        # Perform line search for appropriate step size
        current_c5 = evaluate_c5(h)

        while step > min_step:

            h_new = project_to_sum_box(h_new, TARGET_SUM)
            new_c5 = evaluate_c5(h_new)
