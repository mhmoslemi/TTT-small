import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

# Global variables
n_points = 150
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5_exact(h):
    """Compute the exact C5 value using FFT correlation"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(h, target_sum):
    """Project vector to box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(h + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def build_mccormick_constraints(p, h, i, j):
    """Build McCormick constraints for product h[i]*h[j]"""
    constraints = []
    constraints.append(p[i,j] >= 0)
    constraints.append(p[i,j] >= h[i] + h[j] - 1)
    constraints.append(p[i,j] <= h[i])
    constraints.append(p[i,j] <= h[j])
    return constraints

def solve_lp_relaxation(h_init=None):
    """Solve the LP relaxation of the problem"""
    h = cp.Variable(n_points, bounds=(0, 1))
    p = cp.Variable((n_points, n_points), symmetric=True, bounds=(0, 1))
    T = cp.Variable(1, nonneg=True)
    
    # Constraints for McCormick envelopes
    constraints = []
    
    # McCormick envelopes for all pairs
    for i in range(n_points):
        for j in range(i, n_points):
            constraints += build_mccormick_constraints(p, h, i, j)
    
    # Mass constraint
    constraints.append(cp.sum(h) == TARGET_SUM)
    
    # Shift constraints
    # For shifts k from -(n-1) to n-1
    k_shifts = range(-(n_points - 1), n_points)
    
    # From each shift k compute the overlap term O_k
    # But for simplicity, we just use the CC
    # Let's compute simpler: O_k = sum_{i} h[i]*h[i+k]  (both within range)
    # We'll create a separate variable for the shift
    # This is more direct
    
    # But simpler method: for each shift k, build the overlap term
    
    # We use an explicit loop on shifts
    # But for efficiency in the LP, we correspond each pair of indices to 
    # their shift. Let L = number of unique shifts. In our case they interpolate  
    # back to just about 2*n-1 shifts. This can be done fully on a dense graph.
    
    # Equivalent to compute maximum of all O_k = sum_{i: valid} h[i]*h[i+k]
    # So we build variables O_k and link them to sum of h[i]*h[i+k]
    # Which we express via p_ij
    
    # Less direct: Relax all the overlaps
    # Each O_k = sum_{i: valid} h[i]*h[i+k] 
    # So that O_k = sum_{i=0}^{n-1-k} p[i, i+k]
    # Now write these sums in LP
    
    # Precompute shifts to matrix entries for efficiency
    # But we'll just do a direct calculation using helper indicator arrays 
    
    # More direct approach:
    # Build the sums directly.
    # Since the operation is: max_{k} sum_{i} h[i]*h[i+k] = max_{k} (n/2 - sum of overlaps = sum_{i} h[i]*h[i+k])
    # So objective becomes minimize T 
    # subject to max_k (n/2 - sum_{i=0}^{n-1-k} h[i] h[i+k] ) <= T
    # i.e. sum_{i=0}^{n-1-k} h[i]*h[i+k] >= n/2 - T
    # Use slack - replace above constraint with (sum_{i=0}^{n-1-k} p[i, i+k]) >= n/2 - T 
    # therefore for full range of k : sum_{i=0}^{n-1-k} p[i, i+k] >= n/2 - T
    # Quick matrices: if shifted position [i+k] is valid, i+k < n, so i < n-k
    # So avoid creating this for really big negative k.
    # Let's just compute overlaps and expressions explicitly.
    
    # It's simpler just do the matrix logic:
    
    for k in k_shifts:
        # Ensure we use only valid elements
        # Sum over i where (i+k) is valid
        L = min(n_points, n_points - abs(k)) if k >= 0 else min(n_points - k, n_points)
        if L == 0:
            continue # skip
        # Construct sum of p[i, i+k] for i in [0, L-1]
        # if k<0 we have i+k < i so i+k in [0,n-1] means i in [k,n-1+k]  
        # if k>=0 we have i in [0,n-k-1] 
        # if k<0: i+k < n-1 => i<n-1-k, so i in [0,n-1-k]
        sum_p_k_eq = 0
        for i in range(L):
            if i + k >= 0 and i + k < n_points:
                sum_p_k_eq += p[i, i+k]
            else:
                # skip
                pass
        if k >= 0:
            # range [0, n-1-k]
            expected_len = n_points - k
            actual_len = L
            if expected_len == actual_len:
                constraints.append(sum_p_k_eq >= n/2 - T)
        else: 
            # range [0, n-1+k]
            expected_len = n_points + k
            actual_len = L
            if expected_len == actual_len:
                constraints.append(sum_p_k_eq >= n/2 - T)

    # Objective
    objective = cp.Minimize(T)
    prob = cp.Problem(objective, constraints)
    
    # Solve with small solver budget
    prob.solve(solver=cp.GLPK, max_iters=10000, time_limit=180)

    if prob.status == cp.OPTIMAL:
        h_sol = h.value
        return h_sol
    else:
        # Return initial guess when infeasible
        return h_init if h_init is not None else np.ones(n_points) * 0.5

def exchange_refinement(h0, n_attempts=1000):
    """Local exchange refinement"""
    h = h0.copy()
    h = np.clip(h, 0, 1)
    h = project_to_sum_box(h, TARGET_SUM)
    
    for _ in range(n_attempts):
        i = np.random.randint(0, len(h))
        j = np.random.randint(0, len(h))
        if i == j:
            continue
            
        # Compute max transfer
        delta_max = min(h[j], 1 - h[i])
        if delta_max <= 0:
            continue
            
        # Try small transfer
        eta = 0.5
        delta = eta * delta_max
        h[i] += delta
        h[j] -= delta
        h = np.clip(h, 0, 1)
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Check if improvement
        new_c5 = evaluate_c5_exact(h)
        if new_c5 >= evaluate_c5_exact(h0):
            # Revert
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0, 1)
            h = project_to_sum_box(h, TARGET_SUM)
    
    return h

def intp_new_grid(old_h, n_new):
    """Interpolate solution to a finer grid"""
    x_old = np.linspace(0, 2, len(old_h))
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, old_h)
    return h_new

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Ensure we can work with the scipy fashion

        h_base = np.array(initial_h_values)
        x_base = np.linspace(0, 2, len(h_base))

        init_h = np.interp(x_new, x_base, h_base)
    else:
        init_h = np.ones(n_points) * 0.5
    
    init_h = project_to_sum_box(init_h, TARGET_SUM)
    
    # Phase 1: Solve LP relaxation
    if start_time + (0.3 * budget_s) > time.time(): 
        h_lp = solve_lp_relaxation(init_h)
        if h_lp is not None:
            h_sol = h_lp.copy()
        else:
            h_sol = init_h.copy()
    else:
        h_sol = init_h.copy()
    
    # Phase 2: Binary search for certified bound
    if start_time + (0.7 * budget_s) > time.time():
        h_final, c5_bound = binary_search_on_bound(h_sol, 0.0, 1.0, 1e-6)
    else:
        h_final = h_sol
        c5_bound = evaluate_c5_exact(h_final)
    
    # Phase 3: Subgradient polishing
    if start_time + budget_s > time.time():
        h_final = subsgradient_descent(h_final, max_iter=2000, alpha0=1e-3)
    
    # Compute final evaluation
    c5_final = evaluate_c5_exact(h_final)
    
    # Phase 4: Exchange refinement
    if start_time + budget_s > time.time():
        h_final = exchange_refinement(h_final, n_attempts=1000)
        c5_final = evaluate_c5_exact(h_final)
    
    # Try once more intergrid finer refinement if we have time  
    if start_time + budget_s > time.time():
        try:
            n_new = n_points + 30
            dx_new = 2.0 / n_new
            h_fine = intp_new_grid(h_final, n_new)
            h_fine = project_to_sum_box(h_fine, n_new/2)
            # Try quick polished approach
            h_fine = subsgradient_descent(h_fine, max_iter=1000, alpha0=1e-3)
            c5_fine = evaluate_c5_exact(h_fine)
            if c5_fine < c5_final:
                h_final = h_fine
                c5_final = c5_fine
        except:
            pass
    
    return (h_final.tolist(), float(c5_final), n_points)
