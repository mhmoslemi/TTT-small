import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 150
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmax_weights(corr, eps):
    """Compute softmax weights for smoothing"""
    m = np.max(corr)
    # Stable computation: subtract max to avoid overflow
    w_raw = np.exp((corr - m) / eps)
    return w_raw / w_raw.sum()

def compute_gradient(h, w_norm):
    """Compute gradient using direct sum (O(n^2)) for better control"""
    n = len(h)
    offset = n - 1
    gradient = np.zeros(n)
    
    # For each shift s
    for s in range(-n+1, n):
        ws = w_norm[s + offset]
        if ws == 0:
            continue
            
        if s >= 0:
            # Gradient contribution from shift s
            # sum_i w_s * (1 - h[i+s]) for i in 0..n-s-1
            # sum_i w_s * h[i-s] for i in s..n-1
            if s < n:  # Avoid out-of-bounds
                gradient[:n-s] += ws * (1 - h[s:])
                gradient[s:] -= ws * h[:n-s]
        else:
            # s < 0
            t = -s  # t > 0
            # Gradient contribution from shift s
            # sum_i w_s * (1 - h[i+s]) for i in t..n-1
            # sum_i w_s * h[i-s] for i in 0..n-t-1
            if t < n:  # Avoid out-of-bounds
                gradient[t:] += ws * (1 - h[:n-t])
                gradient[:n-t] -= ws * h[t:]
    
    return gradient

def gradient_descent(h0, eps_schedule, iterations_per_eps, lr_schedule, target_sum):
    """Perform gradient descent with smoothing"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for eps_idx, (eps, max_iter, lr) in enumerate(zip(eps_schedule, iterations_per_eps, lr_schedule)):
        # Optimization loop for current epsilon
        for iter in range(max_iter):
            # Compute correlation
            corr = np.correlate(h, 1 - h, mode="full")
            
            # Compute weights
            w_norm = softmax_weights(corr, eps)
            
            # Compute gradient
            grad = compute_gradient(h, w_norm)
            
            # Gradient step
            h_new = h - lr * grad
            
            # Project to feasible region
            h_new = project_to_sum_box(h_new, target_sum)
            
            h = h_new
            
            # Check for improvement
            if iter % 5 == 0:
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    
    return best_h, best_c5

def initialize_h_random(n, target_sum):
    """Initialize a random step function"""
    h = np.random.uniform(0, 1, n)
    h = project_to_sum_box(h, target_sum)
    return h
