import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    # Create CVXPY variables
    h = cp.Variable(n_points, name="h")
    # Use matrix to store all pairwise products
    P = cp.Variable((n_points, n_points), name="P")
    
    # Constraints for h variables
    constraints = [
        cp.sum(h) == TARGET_SUM,
        h >= 0,
        h <= 1
    ]
    
    # Add McCormick constraints for all pairs
    for i in range(n_points):
        for j in range(n_points):
            constraints.extend([
                P[i, j] >= h[i] + h[j] - 1,
                P[i, j] <= h[i],
                P[i, j] <= h[j],
                P[i, j] >= 0
            ])
    
    # For each shift k, compute the correlation with (1-h) 
    # The maximum correlation over all shifts is what we want to minimize
    max_corr = cp.Variable(1, name="max_corr")
    
    # We consider each shift k from 0 to n_points-1
    for k in range(n_points):
        # Compute correlation for shift k, but only if indices are valid
        corr_k = 0
        for i in range(n_points - k):
            corr_k += h[i] - P[i, i + k]

        # Since we're computing over all shifts, we want to find the maximum
        # So we need to ensure that max_corr is greater than or equal to
        # all computed correlations (n_points in this case). 
        # Now define variable t to represent the correlation that we are trying to minimize
        # We'll keep track through a constraint that max_corr >= corr_k for all k
        constraints.append(max_corr >= corr_k)
        
    # Objective: minimize the maximum correlation
    objective = cp.Minimize(max_corr)
    
    # Create problem and solve
    prob = cp.Problem(objective, constraints)
