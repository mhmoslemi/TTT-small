def compute_gradient(h, lag):
    n = len(h)
    grad = np.ones(n)
    if lag >= 0:
        lag = int(lag)
        if lag < n:
            grad[:n-lag] -= h[lag:]
            grad[lag:] -= h[:n-lag]
    else:
        lag = -lag
        if lag < n:
            grad[lag:] -= h[:n-lag]
            grad[:lag] -= h[lag:]
    return grad

def project_to_sum_box(x, target_sum):
    """Project vector onto box [0,1] and sum constraint by shifting and clipping."""
    x = np.asarray(x, dtype=np.float64)
    # simple binary search for lambda such that sum = target_sum.
    lo, hi = -1.0, 1.0
    # expand until sign change
    def f(lam):
        return np.sum(np.clip(x + lam, 0.0, 1.0)) - target_sum
    while f(lo) > 0:
        lo -= 1.0
    while f(hi) < 0:
        hi += 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        val = f(mid)
        if val > 0:
            hi = mid
        else:
            lo = mid
    lam = (lo + hi) / 2.0
    return np.clip(x + lam, 0.0, 1.0)

def compute_total_gradient(h, max_lags):
    # max_lags list of integer lags
    n = len(h)
    grad = np.zeros(n)
    for lag in max_lags:
        if lag >= 0:
            grad[:n-lag] -= h[lag:]
            grad[lag:] -= h[:n-lag]
        else:
            lag_abs = -lag
            grad[lag_abs:] -= h[:n-lag_abs]
            grad[:lag_abs] -= h[lag_abs:]
    grad += len(max_lags)  # since each lag has +1 from sum term
