import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global variables
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmin_weights(s, alpha):
    """Compute soft-min weights for given s vector and alpha"""
    # Subtract max for numerical stability
    s_shifted = s - np.max(s)
    exp_s = np.exp(-alpha * s_shifted)
    return exp_s / np.sum(exp_s)

def weighted_gradient(h, alpha):
    """Compute gradient of softmin with respect to h"""
    n = len(h)
    s = correlate(h, h, mode='full')  # Full correlation
    w = softmin_weights(s, alpha)
    
    # Build gradient vector
    grad = np.zeros(n)
    for lag in range(-n+1, n):
        if lag >= 0:
            # h[i+lag] contribution
            if lag < n:
                grad[lag:] += w[lag + (n-1)] * h[:-lag]
            # h[i-lag] contribution (only if lag != 0)
            if lag > 0:
                grad[:-lag] += w[lag + (n-1)] * h[lag:]
        else:
            # h[i+lag] contribution (lag negative)
            if -lag < n:
                grad[:n+lag] += w[lag + (n-1)] * h[-lag:]
            # h[i-lag] contribution (lag negative)
            if lag < 0:
                grad[-lag:] += w[lag + (n-1)] * h[:n+lag]
    
    return grad

def pairwise_move(h, grad, delta_max, rng):
    """Perform a mass transfer move based on gradient signs"""
    n = len(h)
    # Find indices with most negative and positive gradient
    src_idx = np.argmin(grad)
    tgt_idx = np.argmax(grad)
    
    if src_idx == tgt_idx:
        # Try to find a sufficiently different pair
        grad_sorted = np.argsort(grad)
        src_idx = grad_sorted[0]
        tgt_idx = grad_sorted[-1]
        
        # If still same, pick random
        if src_idx == tgt_idx:
            src_idx = np.random.randint(0, n)
            tgt_idx = np.random.randint(0, n)
            while tgt_idx == src_idx:
                tgt_idx = np.random.randint(0, n)
    
    # Check feasibility
    if h[src_idx] <= 1e-10 or h[tgt_idx] >= 1 - 1e-10:
        return h, False
    
    # Calculate maximum transfer amount
    delta = min(delta_max, h[src_idx], 1 - h[tgt_idx])
    
    if delta <= 1e-10:
        return h, False
    
    # Apply move
    h[src_idx] -= delta
    h[tgt_idx] += delta
    
    # Ensure sum constraint
    total = np.sum(h)
    if abs(total - TARGET_SUM) > 1e-10:
        # Adjust to meet constraint exactly
        diff = total - TARGET_SUM
        h[0] -= diff
        
        # Keep in bounds
        h = np.clip(h, 0.0, 1.0)
        
        # Project to sum
        h = project_to_sum_box(h, TARGET_SUM)
        
    return h, True

def annealed_gradient_ascent(h0, time_budget, alpha0, delta0, T0, rng):
    """Core gradient ascent with annealing"""
    h = h0.copy()

    best_c5 = evaluate_c5(h)
    
    alpha = alpha0
    delta = delta0
    T = T0
    
    start_time = time.time()
    iterations = 0
    temperature_factor = 0.9995
    alpha_factor = 1.2
    alpha_increase_interval = 2000
    
    while time.time() - start_time < time_budget:
        iterations += 1
        
        # Compute gradient
        grad = weighted_gradient(h, alpha)
        
        # Choose source and target based on gradient
        src_idx = np.argmin(grad)
        tgt_idx = np.argmax(grad)
        
        # Perturb a bit to avoid degeneracy
        if src_idx == tgt_idx or (h[src_idx] <= 1e-8 or h[tgt_idx] >= 1 - 1e-8):
            if len(np.where(h > 1e-8)[0]) > 1:
                src_idx = np.random.choice(np.where(h > 1e-8)[0])
            else:
                src_idx = np.random.randint(0, len(h))
                
            if len(np.where(h < 1 - 1e-8)[0]) > 1:
                tgt_idx = np.random.choice(np.where(h < 1 - 1e-8)[0])
            else:
                tgt_idx = np.random.randint(0, len(h))
                
            while tgt_idx == src_idx:
                tgt_idx = np.random.randint(0, len(h))
                
        # Determine transfer amount
        delta_transfer = min(delta, h[src_idx], 1 - h[tgt_idx])
        
        if delta_transfer <= 1e-10:
            continue
            
        # Make the move
        h[src_idx] -= delta_transfer
        h[tgt_idx] += delta_transfer
        
        # Project to sum constraint
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Evaluate the new C5
        c5_new = evaluate_c5(h)
        
        # Accept or reject the move (Metropolis criterion)
        if c5_new <= best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        else:
            # Accept with probability based on temperature
            diff = c5_new - best_c5
            if rng.random() < np.exp(-diff / T):
                best_c5 = c5_new
                best_h = h.copy()
            else:
                # Revert the move
                h[src_idx] += delta_transfer
                h[tgt_idx] -= delta_transfer
                h = project_to_sum_box(h, TARGET_SUM)
        
        # Update parameters periodically
        if iterations % alpha_increase_interval == 0 and alpha < 200:
            alpha *= alpha_factor
            if alpha > 200:
                alpha = 200
            
        if iterations % 100 == 0:
            delta *= 0.9998
            T *= temperature_factor
            
        # Early stopping if improvement becomes very slow
        if iterations > 10000 and iterations % 1000 == 0 and abs(delta) < 1e-6:
            break
            
    return best_h, best_c5

def random_binary_start(n, rng):
    """Create a random binary start vector"""
    h = np.zeros(n)
    indices = rng.sample(range(n), n//2)
    h[indices] = 1.0
    return h

def local_optimize_restart(start_h, remaining_time, rng):
    """Run a short optimization with a given start"""
    # Parameters for this run
    initial_alpha = 10.0
    initial_delta = 0.4
    initial_T = 5e-3
    
    local_time = min(20.0, remaining_time / 2.0)
    try:
        h_opt, c5_opt = annealed_gradient_ascent(
            start_h, local_time, initial_alpha, initial_delta, initial_T, rng
        )
        return h_opt, c5_opt
    except:
        return start_h.copy(), evaluate_c5(start_h)
