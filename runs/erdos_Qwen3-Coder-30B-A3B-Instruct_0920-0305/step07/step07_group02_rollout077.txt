```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5_fft(h):
    """Fast evaluation of C5 using FFT"""
    dx = 2.0 / len(h)
    corr_full = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr_full)

def greedy_descent(h0, max_iter=1000):
    """Greedy descent on worst shift"""
    h = h0.copy()
    for _ in range(max_iter):
        corr = np.correlate(h, h, mode='full')
        # Correlation of h with (1-h) is (sum(h) - corr) = 1 - corr
        # So C5 = dx * max(1 - corr) = dx * max(1 - corr)
        worst_shift_idx = np.argmax(1 - corr)
        # positions where h[i] =1 and h[i+shift] =0
        A = np.where((h == 1) & (np.roll(h, -worst_shift_idx) == 0))[0]
        # positions where h[i] =0 and h[i+shift] =1
        B = np.where((h == 0) & (np.roll(h, -worst_shift_idx) == 1))[0]
        if len(A) == 0 or len(B) == 0:
            break
        # swap one element from each
        i = np.random.choice(A)
        j = np.random.choice(B)
        h[i], h[j] = h[j], h[i]
    return h, compute_c5_fft(h)

def anneal_binary(h0, time_limit_s):
    """Lightweight simulated annealing on binary vector"""
    np.random.seed(int(time.time() * 1000) % 2**32)
    h = h0.copy()
    dx = 2.0 / len(h)
    c5 = compute_c5_fft(h)
    start = time.time()
    T0 = 0.02
    Tmax = 10000
    t = 0
    while time.time() - start < time_limit_s:
        if t % 1000 == 0:
            T = T0 * (1 - t / Tmax) if t < Tmax else 0.001
        else:
            T = T0 * (1 - t / Tmax) if t < Tmax else 0.001

        # propose a swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # perform swap temporarily
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5_fft(h)
        delta = c5_new - c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            c5 = c5_new
        else:
            # undo swap
            h[i], h[j] = h[j], h[i]
        t += 1
        # occasional greedy refinement every 500 steps
        if t % 500 == 0:
            h, c5 = greedy_descent(h, 50)
    return h, c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use N = 120 as it balances tractability with resolution
    N = 120
    dx = 2.0 / N
    target_sum = N // 2

    # Variable declarations
    h = cp.Variable(N, boolean=True)
    lam = cp.Variable(integer=True)
    w_list = []
    constraints = [cp.sum(h) == target_sum, h[0] == 1]  # symmetry breaking

    # Linearization for all possible shifts and valid indices
    for k in range(-(N - 1), N):
        idxs_i = np.arange(max(0, -k), min(N, N - k))
        if len(idxs_i) > 0:
            i = idxs_i
            j = i + k
            w = cp.Variable(len(i), boolean=True)
            constraints += [
                w <= h[i],
                w <= h[j],
                w >= h[i] + h[j] - 1
            ]
            constraints += [cp.sum(w) >= lam]
            w_list.append(w)
    
    # Set up and solve the problem
    objective = cp.Maximize(lam)
    prob = cp.Problem(objective, constraints)
    
    elapsed = time.time() - start_time
    solver_time = min(0.55 * budget_s, budget_s - elapsed - 5)
    
    try:
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=solver_time)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_val = np.round(h.value).astype(int)
            lam_val = int(lam.value)
            c5 = 1 - (2 * lam_val) / N
        else:
            # fallback to random binary
            h_val = np.random.choice([0, 1], size=N, p=[0.5, 0.5])
            h_val = np.clip(h_val, 0, 1)
            # Project onto sum constraint
            ones_indices = np.where(h_val == 1)[0]
            if len(ones_indices) > target_sum:
                # Remove some ones
                remove_indices = np.random.choice(ones_indices, 
                                                  len(ones_indices) - target_sum, 
                                                  replace=False)
                h_val[remove_indices] = 0
            elif len(ones_indices) < target_sum:
                # Add some zeros
                zero_indices = np.where(h_val == 0)[0]
                add_indices = np.random.choice(zero_indices, 
                                               target_sum - len(ones_indices), 
                                               replace=False)
                h_val[add_indices] = 1
            c5 = compute_c5_fft(h_val)
    except Exception:
        # Fallback on exception
        h_val = np.random.choice([0, 1], size=N, p=[0.5, 0.5])
        h_val = np.clip(h_val, 0, 1)
        # project onto sum constraint
        ones_indices = np.where(h_val == 1)[0]
        if len(ones_indices) > target_sum:
            remove_indices = np.random.choice(ones_indices, 
                                              len(ones_indices) - target_sum, 
                                              replace=False)
            h_val[remove_indices] = 0
        elif len(ones_indices) < target_sum:
            zero_indices = np.where(h_val == 0)[0]
            add_indices = np.random.choice(zero_indices, 
                                           target_sum - len(ones_indices), 
                                           replace=False)
            h_val[add_indices] = 1
        c5 = compute_c5_fft(h_val)

    best_h = h_val.copy()
    best_c5 = c5
    
    # Greedy local improvement
    elapsed = time.time() - start_time
    if elapsed < 0.75 * budget_s:
        best_h, best_c5 = greedy_descent(best_h)

    # Annealing if remaining time
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 30:
        best_h, best_c5 = anneal_binary(best_h, remaining)
    
    return (best_h.tolist(), float(best_c5), N)
```
