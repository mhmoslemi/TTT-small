import numpy as np

def create_A_matrices(n):
    """Create sparse A_matrices for all lags."""
    A_list = []
    for lag in range(-(n-1), n):
        # This creates a sparse matrix A_lag with 1s at positions (i, i+lag)
        rows, cols = [], []
        for i in range(n):
            if 0 <= i + lag < n:
                rows.append(i)
                cols.append(i + lag)
        A = np.zeros((n, n))
        A[rows, cols] = 1
        A_list.append(A)
    return A_list

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def round_and_evaluate(X_opt, n, dx, target_sum, num_samples=2000):
    """Round SDP solution X_opt to feasible h vector"""
    # Compute dominant eigenvector of X_opt
    _, V = eigh(X_opt)
    v = V[:, -1]
    h_raw = v.copy()
    h_clip = np.clip(h_raw, 0, 1)
    h_projected = project_to_sum_box(h_clip, target_sum)
    best_c5 = evaluate_c5(h_projected)
    best_h = h_projected.copy()

    # Sample multiple random vectors from the distribution defined by X_opt
    for sample in range(num_samples):
        # Sample from N(0, X_opt) by eigendecomposition
        L = np.linalg.cholesky(X_opt)
        z = np.random.randn(n)
        y = L @ z  # y ~ N(0, X_opt)
        h = np.clip(y, 0, 1)
        h = project_to_sum_box(h, target_sum)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5
