```python
import numpy as np
from scipy.signal import correlate
import time
try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False

# Global variables for the problem
N = 120
dx = 2.0 / N
TARGET_SUM = N // 2

def compute_c5_fft(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Check if cvxpy available
    if not HAS_CVXPY:
        # Fallback to a heuristic approach
        h = np.random.rand(N)
        h = project_to_sum_box(h, TARGET_SUM)
        c5 = compute_c5_fft(h)
        return (h.tolist(), float(c5), N)

    # 1. Solve MILP
    try:
        # Variables
        h = cp.Variable(N, boolean=True)
        lam = cp.Variable(integer=True)
        
        # Constraints
        constraints = [cp.sum(h) == TARGET_SUM, h[0] == 1]  # symmetry breaking
        
        # Auxiliary variables and constraints for all shifts
        w_vars = {}
        
        for k in range(-(N-1), N):
            idxs_i = np.arange(max(0,-k), min(N, N-k))
            if len(idxs_i) == 0:
                continue
            
            # Create auxiliary variables for this shift
            w = cp.Variable(len(idxs_i), boolean=True)
            w_vars[(k, idxs_i[0])] = w
            
            # McCormick constraints
            i_indices = idxs_i
            j_indices = i_indices + k
            constraints += [
                w <= h[i_indices],
                w <= h[j_indices],
                w >= h[i_indices] + h[j_indices] - 1
            ]
            
            # Constraint for minimum correlation
            constraints += [cp.sum(w) >= lam]

        # Objective: maximize lambda
        prob = cp.Problem(cp.Maximize(lam), constraints)
        
        # Solve with time limit
        remaining_budget = budget_s * 0.55
        prob.solve(solver=cp.CBC, max_seconds=remaining_budget, verbose=False)
        
        if prob.status in ['optimal', 'optimal_inaccurate']:
            h_opt = np.round(h.value).astype(int)
            lambda_opt = int(lam.value)
            c5_opt = 1 - (2 * lambda_opt) / N
        else:
            # Fallback with random binary
            h_opt = np.random.randint(0, 2, N)
            # Ensure sum constraint
            current_sum = np.sum(h_opt)
            if current_sum > TARGET_SUM:
                # Need to flip some 1s to 0s
                ones = np.where(h_opt == 1)[0]
                flips = current_sum - TARGET_SUM
                for i in range(flips):
                    h_opt[ones[i]] = 0
            elif current_sum < TARGET_SUM:
                # Need to flip some 0s to 1s
                zeros = np.where(h_opt == 0)[0]
                flips = TARGET_SUM - current_sum
                for i in range(flips):
                    h_opt[zeros[i]] = 1
            
            c5_opt = compute_c5_fft(h_opt)
    except Exception as e:
        # Fallback
        h_opt = np.random.randint(0, 2, N)
        # Ensure sum constraint
        current_sum = np.sum(h_opt)
        if current_sum > TARGET_SUM:
            ones = np.where(h_opt == 1)[0]
            flips = current_sum - TARGET_SUM
            for i in range(flips):
                h_opt[ones[i]] = 0
        elif current_sum < TARGET_SUM:
            zeros = np.where(h_opt == 0)[0]
            flips = TARGET_SUM - current_sum
            for i in range(flips):
                h_opt[zeros[i]] = 1
        
        c5_opt = compute_c5_fft(h_opt)
    
    # 2. Greedy polishing
    elapsed = time.time() - start_time
    if elapsed < 0.75 * budget_s:
        h_purified = h_opt.copy()
        corr_full = correlate(h_purified, h_purified, mode="full")
        for i in range(100):  # max iterations for polishing
            max_corr_idx = np.argmax(corr_full)
            # Monte Carlo sampling for better coverage
            if max_corr_idx < N:
                max_shift = max_corr_idx - (N-1)
            else:
                max_shift = max_corr_idx - (N-1)
            
            # Identify mismatches in the worst shift
            mismatches_i_1 = np.where((h_purified == 1) & (np.roll(h_purified, max_shift) == 0))[0]
            mismatches_j_0 = np.where((h_purified == 0) & (np.roll(h_purified, max_shift) == 1))[0]
            
            if len(mismatches_i_1) > 0 and len(mismatches_j_0) > 0:
                swap_idx_i = np.random.choice(mismatches_i_1)
                swap_idx_j = np.random.choice(mismatches_j_0)
                h_purified[swap_idx_i], h_purified[swap_idx_j] = h_purified[swap_idx_j], h_purified[swap_idx_i]
                corr_full = correlate(h_purified, h_purified, mode="full")
            else:
                break  # No more beneficial swaps
    
        c5_new = compute_c5_fft(h_purified)
        if c5_new < c5_opt:
            h_opt = h_purified.copy()
            c5_opt = c5_new
    
    # 3. Simulated Annealing (if time remains)
    elapsed = time.time() - start_time
    if elapsed < budget_s - 10:
        h_sa = h_opt.copy()
        current_c5 = c5_opt
        T = 0.02
        max_iter = 5000  # about 5 seconds total at 1000 iterations/sec
        iter_count = 0
        while iter_count < max_iter:
            # Make a move
            pos1 = np.random.choice(np.where(h_sa == 1)[0])
            pos2 = np.random.choice(np.where(h_sa == 0)[0])
            h_new = h_sa.copy()
            h_new[pos1] = 0
            h_new[pos2] = 1
            
            new_c5 = compute_c5_fft(h_new)
            if new_c5 < current_c5:
                h_sa = h_new
                current_c5 = new_c5
            else:
                # Accept with probability
                delta = new_c5 - current_c5
                prob = np.exp(-delta / T)
                if np.random.rand() < prob:
                    h_sa = h_new
                    current_c5 = new_c5
            iter_count += 1
            # Cool down
            T *= 0.99995
            elapsed = time.time() - start_time
            if elapsed > budget_s - 10:
                break
        
        if current_c5 < c5_opt:
            h_opt = h_sa
            c5_opt = current_c5
    
    return (h_opt.tolist(), float(c5_opt), N)
```
