import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    n = len(h)
    dx = 2.0 / n
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_lp_relaxation_with_t(h0, T_val, n_points):
    """Solve relaxed LP with fixed T"""
    # Initialize
    dx = 2.0 / n_points
    h = cp.Variable(n_points, name="h")
    p = cp.Variable(int(n_points * (n_points + 1) / 2), name="p")
    
    # 0 <= h <= 1
    constraints = [h >= 0, h <= 1]
    
    # Mass constraint
    constraints.append(cp.sum(h) == n_points / 2)
    
    # McCormick envelopes for bilinear terms
    # We'll create pairs in full storage format
    pairs = []
    for i in range(n_points):
        for j in range(i, n_points):
            pairs.append((i, j))
            
    p_idx = 0
    for i, j in pairs:
        p_ij = p[p_idx]
        constraints.append(p_ij >= 0)
        constraints.append(p_ij >= h[i] + h[j] - 1)
        constraints.append(p_ij <= h[i])
        constraints.append(p_ij <= h[j])
        p_idx += 1
        
    # Overlap constraints
    # For each shift k = -(n-1) to n-1
    # O_k = sum_{i = 0}^{n-1-|k|} p_{i, i+|k|}
    # Then C_k = n/2 - O_k <= T_val
    # So we require O_k >= n/2 - T_val
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            # For k>=0, overlap = sum_{i=0 to n-k-1} p_{i, i+k}
            indices = []
            for i in range(n_points - k):
                # p_{i, i+k}
                # Upper triangular storage: row `i`, col `i+k`
                ni = i
                nj = i + k
                if ni <= nj:
                    idx = ni * n_points + nj - ni * (ni + 1) // 2
                else:
                    idx = nj * n_points + ni - nj * (nj + 1) // 2
                indices.append(idx)
        else:
            # For k<0, overlap = sum_{i=0 to n+|k|-1} p_{i+|k|, i}
            indices = []
            abs_k = -k
            for i in range(n_points - abs_k):
                ni = i + abs_k
                nj = i
                if ni <= nj:
                    idx = ni * n_points + nj - ni * (ni + 1) // 2
                else:
                    idx = nj * n_points + ni - nj * (nj + 1) // 2
                indices.append(idx)
        # For p_{i, i+k}, we have pointers to p_ij (symmetric)
        # Sum these values over all valid i
        o_k = cp.sum([p[i] for i in indices])
        constraints.append(o_k >= n_points / 2 - T_val)
        
    # Objective: minimize T (not used in feasibility phase)
    # No need to set an objective, just add constraints
    prob = cp.Problem(cp.Minimize(0), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
        return prob.status == cp.OPTIMAL, h.value, p.value
    except:
        return False, None, None

def binary_search_for_t(h0, n_points, target_eps=1e-6):
    """Binary search for minimal T that allows a feasible solution"""
    dx = 2.0 / n_points
    T_low = 0.0
    T_high = n_points / 2.0
    h_best = h0
    
    while T_high - T_low > target_eps:
        T_mid = (T_low + T_high) / 2
        feasible, h_val, _ = solve_lp_relaxation_with_t(h0, T_mid, n_points)
        if feasible:
            h_test = project_to_sum_box(h_val, n_points / 2)
            true_c5 = evaluate_c5(h_test)
            if true_c5 <= T_mid * dx:
                h_best = h_test.copy()
                T_high = T_mid
            else:
                T_low = T_mid
        else:
            T_low = T_mid
            
    return h_best, T_high * dx

def sub_gradient_polish(h0, n_points, num_iters=2000):
    """Sub-gradient descent"""
    h = h0.copy()
    dx = 2.0 / n_points
    for t in range(num_iters):
        # Identify active shift
        h_rev = h[::-1]
        b = 1 - h
        L = 2 * n_points - 1
        corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
        active_k = np.argmax(corr)
        
        # Generate subgradient for the active term
        g = np.zeros(n_points)
        if active_k == 0:
            # Most common case, the case of lag 0, we use standard formula
            # Actually, the objective is max_k \int h(x)(1-h(x+k)) dx
            # We differentiate with respect to h_i and get the gradient of each term
            # For correlation term R_i(k) = \int h_i h_{i+k} dx, the influence:
            # dR_i(k)/dh_i = h_{i+k}, dR_i(k)/dh_{i+k} = h_i 
            # and for the total objective, we need the derivative of 1 - R_k
            # which is -dR_k/dh_i for all terms in correlation k
            
            g = np.zeros(n_points)
            for i in range(n_points):
                i_shifted = (i + active_k) % n_points
                if i_shifted < n_points:  # boundary checks would be complicated due to circular nature but for simplicity and correctness usually can be ignored
                    g[i] += -(1 - h[i_shifted])
                elif i_shifted >= n_points:
                    i_shifted_adj = i_shifted - n_points
                    if i_shifted_adj < n_points:
                        g[i] += -(1 - h[i_shifted_adj])
                else:
                    for i_start in range(n_points):
                        g[i_start] = 0
            g = -g  # Because optimization target is to minimize, and we're looking at the derivative of 1 - overlap
        elif active_k > 0:
            # k = 1..n-1, go forward shift
            for i in range(n_points - active_k):
                g[i] -= (1 - h[i + active_k])
            for i in range(active_k, n_points):
                g[i] -= h[i - active_k]
        else:
            # k < 0, backward shift
            active_k = abs(active_k)
            for i in range(n_points - active_k):
                g[i] -= h[i + active_k]
            for i in range(active_k, n_points):
                g[i] -= (1 - h[i - active_k])
        
        # Update step
        alpha = 1.0 / np.sqrt(t + 1)
        h = h - alpha * g
        h = project_to_sum_box(h, n_points / 2)
        
    return h

def exchange_refine(h0, n_points, num_cycles=500):
    """Local exchange sweep to refine solution"""
    h = h0.copy()
    
    # Shuffle all pairs
    pair_list = []
    for i in range(n_points):
        for j in range(i + 1, n_points):
            pair_list.append((i, j))
            
    np.random.shuffle(pair_list)
    
    for i_pair, (i, j) in enumerate(pair_list):
        if i_pair > num_cycles:  # Limit iterations
            break
        delta_max = min(h[j], 1 - h[i])
        if delta_max <= 0:
            continue
        # Try a small fraction
        for eta in [0.1, 0.5, 0.9]:
            delta = eta * delta_max
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
            
            # Projection
            h_new = project_to_sum_box(h_new, n_points / 2)
            if evaluate_c5(h_new) < evaluate_c5(h):
                h = h_new
                break  # Accept first improvement
            
    return h

def initial_interpolation(h_ref, n_out):
    """Interpolate to higher resolution"""
    h_ref = np.array(h_ref)
    x_ref = np.linspace(0, 2, len(h_ref))
    x_new = np.linspace(0, 2, n_out)
    h_interp = np.interp(x_new, x_ref, h_ref)
    return h_interp
