import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        return evaluate_c5(h)

        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def local_descent(h0, max_iter=3000):
    """Gradient-based local descent with smarter step selection"""
    h = h0.copy()
    
    for i in range(max_iter):
        # Compute gradient approximation
        grad = np.zeros_like(h)
        eps = 1e-6
        
        # Finite difference gradient estimation
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_to_sum_box(h_plus, TARGET_SUM)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = project_to_sum_box(h_minus, TARGET_SUM)
            
            grad[j] = (evaluate_c5(h_plus) - evaluate_c5(h_minus)) / (2 * eps)
            
        # Gradient step
        step_size = 0.01  # adaptive if needed
        h_new = h - step_size * grad
        
        # Project to constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Accept if improvement
        if evaluate_c5(h_new) < evaluate_c5(h):
            h = h_new.copy()
        else:
            # Try with smaller step
            step_size *= 0.5
            h_new = h - step_size * grad
            h_new = project_to_sum_box(h_new, TARGET_SUM)

        h_new = h.copy()
        
        if strategy == 'gaussian':
            # Add small Gaussian noise
            noise = np.random.normal(0, 0.01, n_points)
            h_new += noise
            h_new = project_to_sum_box(h_new, TARGET_SUM)
