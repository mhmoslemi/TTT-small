import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 150
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_lp_relaxation():
    """Solve the convex relaxation LP to get a strong initial solution"""
    h = cp.Variable(n_points, name="h")
    p = cp.Variable((n_points, n_points), name="p")  # Product variables
    
    # Define the constraints
    constraints = []
    
    # Box constraints on h
    constraints += [0 <= h, h <= 1]
    
    # Mass constraint
    constraints += [cp.sum(h) == TARGET_SUM]
    
    # McCormick constraints for p[i,j] = h[i] * h[j]
    for i in range(n_points):
        for j in range(i, n_points):
            constraints += [
                p[i, j] >= 0,
                p[i, j] >= h[i] + h[j] - 1,
                p[i, j] <= h[i],
                p[i, j] <= h[j]
            ]
    
    # Objective: minimize the maximum overlap
    t = cp.Variable(name="t")
    # For each shift k, compute the overlap at that shift
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            # Correlation overlap calculation
            overlap_k = cp.sum([p[i, i + k] for i in range(n_points - k)])
        else:
            overlap_k = cp.sum([p[i - k, i] for i in range(-k, n_points)])
        
        # Add constraint that overlap_k <= (n/2) - t => t <= (n/2) - overlap_k
        constraints += [overlap_k >= TARGET_SUM - t]
    
    # The objective is to minimize t
    objective = cp.Minimize(t)
    
    prob = cp.Problem(objective, constraints)
    
    # Solve the problem
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=3000, feastol=1e-8)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            h_val = h.value
            return h_val
    except:
        pass
    
    # If solver fails or returns infeasible, use a simple uniform distribution
    return np.ones(n_points) * 0.5

def projected_subgradient_descent(h0, max_iter=2000):
    """Refine using projected sub-gradient descent"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    alpha = 0.5 * dx  # Initial step size
    decay_rate = 0.99
    
    for iter in range(max_iter):
        # Find the active shift
        corr = compute_corr_fft(h)
        k_star = np.argmax(corr)
        
        # Compute subgradient
        g = np.zeros_like(h)
        # Handle the case k_star >= 0 (positive shift)
        if k_star >= 0:
            for i in range(len(h) - k_star):
                g[i] += 1 - h[i + k_star]
                if i - k_star >= 0:
                    g[i] -= h[i - k_star]
        else:
            # Handle negative shift
            for i in range(-k_star, len(h)):
                g[i] += 1 - h[i + k_star]
                if i - k_star >= 0:
                    g[i] -= h[i - k_star]
        
        # Gradient step
        h_new = h - alpha * g
        
        # Project onto box [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project onto sum constraint
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Update for next iteration
        h = h_new.copy()
        alpha *= decay_rate
        
        # Early stopping based on improvement rate
        if iter > 100 and best_c5 < 0.379:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points / 2
    
    # Solve LP relaxation to get a good starting point
    h0 = solve_lp_relaxation()
    
    # If the relaxation fails, fall back to a uniform distribution
    if h0 is None:
        h0 = np.ones(n_points) * 0.5
    
    # Ensure the sum is correct
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    best_c5 = evaluate_c5(h0)
    best_h = h0.copy()

    # Local refinement with projected subgradient descent
    refine_start = time.time()
    refined_h, refined_c5 = projected_subgradient_descent(h0, 1000)
    
    if refined_c5 < best_c5:
        best_h = refined_h.copy()
        best_c5 = refined_c5
    
    # Additional optimizations
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5:
        # Run CMA-ES for a short time
        try:
            from scipy.optimize import differential_evolution
            def objective(h):
                return evaluate_c5(h)
            
            # Define bounds for each variable
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Ensure sum constraint
            def constraint_sum(x):
                return np.sum(x) - TARGET_SUM
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            
            # Run differential evolution with limited time
            res = differential_evolution(objective, bounds, constraints=cons,
                                         seed=seed, maxiter=100, popsize=10)
            if res.success:
                h_test = res.x
                c5_test = evaluate_c5(h_test)
                if c5_test < best_c5:
                    best_h = h_test.copy()
                    best_c5 = c5_test
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
