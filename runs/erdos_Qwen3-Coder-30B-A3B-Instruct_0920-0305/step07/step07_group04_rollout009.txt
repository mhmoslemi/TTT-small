import numpy as np
from scipy.signal import correlate

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    n = len(h)
    dx = 2.0 / n
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_relaxed_lp(h0, n_points, dx, T_val):
    """Solve the relaxed LP for a fixed T value"""
    h = cp.Variable(n_points, name='h')
    p = cp.Variable((n_points, n_points), name='p')
    
    # Constraints for the McCormick envelope
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2  # mass constraint
    ]
    
    # McCormick envelopes
    for i in range(n_points):
        for j in range(i, n_points):
            constraints.extend([
                p[i, j] >= 0,
                p[i, j] >= h[i] + h[j] - 1,
                p[i, j] <= h[i],
                p[i, j] <= h[j]
            ])
    
    # Overlap constraints
    for k in range(-n_points+1, n_points):
        o = 0
        if k >= 0:
            for i in range(n_points - k):
                o += p[i, i + k]
        else:
            for i in range(n_points + k):
                o += p[i - k, i]
        constraints.append(o >= n_points / 2 - T_val)

    # Minimize T
    prob = cp.Problem(cp.Minimize(0), constraints)
    try:
        prob.solve(solver=cp.GLPK, verbose=False, max_iters=5000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURATE]:
            return h.value
    except:
        pass
    return None

def binary_search_then_verify(h0, n_points, dx, target_bound):
    """Binary search over T values to find the minimum feasible bound"""
    T_low, T_high = 0.0, n_points / 2
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    for _ in range(30):
        if T_high - T_low < 1e-4 * dx:
            break
        T_mid = (T_low + T_high) / 2
        h_sol = solve_relaxed_lp(h0, n_points, dx, T_mid)
        if h_sol is not None:
            # Evaluate exact C5
            c5_exact = evaluate_c5(h_sol)
            if c5_exact <= T_mid * dx:
                best_h = h_sol.copy()
                best_c5 = c5_exact
                T_high = T_mid
            else:
                T_low = T_mid
        else:
            T_low = T_mid
    
    return best_h, best_c5

def subgradient_polish(h0, max_iter=2000):
    """Apply deterministic subgradient method to improve the solution"""
    h = h0.copy()
    dx = 2.0 / len(h)
    learning_rate = 0.01
    
    for t in range(max_iter):
        if t % 100 == 0:
            corr = np.correlate(h, 1 - h, mode='full')
            k_star = np.argmax(corr)
            if k_star >= len(h):
                k_star = k_star - len(h) + 1
        else:
            k_star = 0

        g = np.zeros_like(h)
        if k_star > 0:
            g[k_star:] = 1 - h[:-k_star]
            g[:k_star] = 0
        else:
            g = 1 - h
            
        # Adjust for backward shift
        if len(h) + k_star > 0:
            offset = max(0, k_star) 
            if offset < len(h):
                g[:len(h)-offset] -= h[offset:]
        else:
            g[:] = 1 - h[:]
            
        h -= learning_rate * g
        h = project_to_sum_box(h, len(h) / 2)

    return h

    np.random.seed(seed)

    # Use a slightly larger grid size for better resolution

    h0 = project_to_sum_box(h_interp, n_points / 2)

    best_c5 = evaluate_c5(h0)
    
    # Phase 1: Binary search approach
    h_searched, c5_searched = binary_search_then_verify(h0, n_points, dx, best_c5)
    if c5_searched < best_c5:
        best_h = h_searched.copy()
        best_c5 = c5_searched
    
    # Phase 2: Subgradient polishing
    h_polished = subgradient_polish(best_h.copy(), 2000)

    if c5_polished < best_c5:
        best_h = h_polished.copy()
        best_c5 = c5_polished
    
    # Final accuracy check and potentially warm-start a better refinement
    final_h = project_to_sum_box(best_h, n_points / 2)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h.tolist(), float(final_c5), n_points)
