```python
import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_of_corr_at_lag(h, lag):
    """Compute gradient of correlation at specific lag"""
    n = len(h)
    g = np.ones(n)
    
    if lag >= 0:
        if lag < n:
            g[:n-lag] = 1 - h[lag:]
            g[lag:] -= h[:n-lag]
    else:
        d = -lag
        if d < n:
            g[d:] = 1 - h[:n-d]
            g[:n-d] -= h[d:]
            
    return g

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    def compute_c5(h):
        corr = compute_corr(h)
        return dx * np.max(corr)
    
    def compute_gradient(h, lag):
        return gradient_of_corr_at_lag(h, lag)
    
    best_h = None
    best_c5 = float('inf')
    
    # Phase 1: Simulated Annealing on binary patterns
    sa_time_limit = 0.5 * budget_s
    sa_end_time = start_time + sa_time_limit
    
    for restart in range(20):
        if time.time() > sa_end_time:
            break
            
        # Generate random binary pattern
        h = np.zeros(n_points)
        ones_indices = np.random.choice(n_points, size=int(TARGET_SUM), replace=False)
        h[ones_indices] = 1.0
        
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Simulated Annealing
        T = 0.1
        cooling_rate = 0.9995
        min_T = 1e-8
        iter_count = 0
        
        while time.time() < sa_end_time and T > min_T:
            # Propose neighbor move: swap 0 and 1
            ones_positions = np.where(h == 1)[0]
            zeros_positions = np.where(h == 0)[0]
            
            if len(ones_positions) > 0 and len(zeros_positions) > 0:
                i = np.random.choice(ones_positions)
                j = np.random.choice(zeros_positions)
                
                # Swap
                h_new = h.copy()
                h_new[i] = 0
                h_new[j] = 1
                
                # Project to maintain sum
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                
                new_c5 = compute_c5(h_new)
                
                # Metropolis acceptance
                delta = new_c5 - current_c5
                if delta < 0 or np.random.random() < np.exp(-delta / T):
                    h = h_new
                    current_c5 = new_c5
                    
                    if current_c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = current_c5
                        
            T *= cooling_rate
            iter_count += 1
            
    # Phase 2: Continuous refinement with gradient descent
    if best_h is not None:
        h_refined = best_h.copy()
        current_c5 = best_c5
        
        # Convert to continuous and refine
        iter_count = 0
        max_iter = 1000
        step_size = 0.05
        
        while time.time() < start_time + budget_s and iter_count < max_iter:
            corr = compute_corr(h_refined)
            max_index = np.argmax(corr)
            max_lag = max_index - (n_points - 1)
            
            # Compute gradient for this lag
            grad = compute_gradient(h_refined, max_lag)
            
            # Gradient descent step
            h_new = h_refined - step_size * grad
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            new_c5 = compute_c5(h_new)
            
            if new_c5 < current_c5:
                h_refined = h_new
                current_c5 = new_c5
                step_size = min(0.1, step_size * 1.1)
            else:
                step_size = max(1e-6, step_size * 0.7)  
            
            iter_count += 1
            
        # Keep the best solution found
        if current_c5 < best_c5:
            best_h = h_refined
            best_c5 = current_c5
        
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
```
