def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * 2.0 / len(h)

def project_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_gradient(h, corr, beta):
    """Compute the smooth max gradient"""
    # Subtract max for numerical stability
    max_corr = np.max(corr)
    exp_corr = np.exp(beta * (corr - max_corr))
    w = exp_corr / np.sum(exp_corr)
    
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        for l in range(-n+1, n):
            j = i + l
            if 0 <= j < n:
                grad[i] += w[l + n - 1] * (h[j] + h[j])
    
    return 2.0 * grad - np.ones(n)

def max_lag_gradient(h, corr):
    """Compute the exact sub-gradient of maximum correlation"""
    n = len(h)
    max_idx = np.argmax(corr)
    grad = np.zeros(n)
    
    # Convert flat index to lag parameter
    lag = max_idx - (n - 1)
    
    if lag > 0:
        # Positive lag: h[i] contributes to correlation only when i+lag < n
        for i in range(n - lag):
            grad[i] += (1 - h[i + lag])
        # h[i+lag] contributes to correlation when i >= 0
        for i in range(lag, n):
            grad[i - lag] -= h[i]
    elif lag < 0:
        # Negative lag: h[i] contributes to correlation when i-lag >= 0 
        for i in range(-lag, n):
            grad[i] += (1 - h[i + lag])
        # h[i-lag] contributes to correlation when i-lag < n
        for i in range(0, n + lag):
            grad[i - lag] -= h[i]
    else:
        # lag == 0: different term
        for i in range(n):
            grad[i] += 1 - 2 * h[i]
    return grad

def mass_transfer_move(h, grad, epsilon):
    """Perform greedy mass transfer in descending gradient direction"""
    n = len(h)
    d = -grad
    sources = []
    sinks = []
    # Identify sources (where gradient is negative) and sinks (where positive)
    for i in range(n):
        if d[i] < 0:
            sources.append((i, -d[i], h[i]))
        elif d[i] > 0:
            sinks.append((i, d[i], 1 - h[i]))
    
    # Sort by decreasing d value
    sources.sort(key=lambda x: -x[1])
    sinks.sort(key=lambda x: -x[1])
    
    # Perform greedy transfers
    total_transferred = 0
    for i, (s, _, max_rem) in enumerate(sources):
        if total_transferred >= epsilon:
            break
        for j, (_, _, max_add) in enumerate(sinks):
            if total_transferred >= epsilon:
                break
            transfer = min(max_rem, max_add, epsilon - total_transferred)
            if transfer > 0:
                h[s] -= transfer
                h[j] += transfer
                total_transferred += transfer
                max_rem -= transfer
                max_add -= transfer
                if max_rem <= 0 or max_add <= 0:
                    break
    return project_sum_box(h, len(h) / 2)

def backtracking_line_search(h, grad, corr, beta, max_iter=20):
    """Backtracking line search with deterministic descent"""
    alpha = 0.2
    alpha_min = 5e-4
    step_size = 0.5  # For exponential decrease
    n = len(h)
    c_orig = evaluate_c5(h)
    
    for _ in range(max_iter):
        h_new = h - alpha * grad
        h_new = project_sum_box(h_new, n / 2)
        
        c_new = evaluate_c5(h_new)
        if c_new < c_orig:
            return h_new, c_new, alpha
        alpha *= step_size
    
    return h, c_orig, 0.0

def stochastic_perturbation(h, T):
    """Stochastic escape using temperature"""
    n = len(h)
    i, j = np.random.randint(0, n, 2)
    tau = T * 0.1
    delta = np.random.uniform(0, min(tau, h[i], 1 - h[j]))
    
    h[i] -= delta
    h[j] += delta
    h = project_sum_box(h, n / 2)
    
    return h
