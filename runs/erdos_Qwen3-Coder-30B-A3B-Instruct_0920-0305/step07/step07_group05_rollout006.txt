from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_guided_search(h0, max_iter=1000):
    """Refine solution with gradient-guided search"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for iter_num in range(max_iter):
        # Compute gradient using finite differences of the objective
        eps = 1e-5
        grad = np.zeros(n_points)

            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_sum_box(h_plus, TARGET_SUM)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_sum_box(h_minus, TARGET_SUM)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Apply update
        idx_min = np.argmin(grad)
        idx_max = np.argmax(grad)
        
        # Move mass from idx_min to idx_max
        delta = min(h[idx_min], 1 - h[idx_max], 0.01)
        if delta > 1e-6:
            h[idx_min] -= delta
            h[idx_max] += delta
            h = project_to_sum_box(h, TARGET_SUM)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Occasionally make random small moves
        if iter_num % 100 == 0 and best_c5 > 0.375:
            i = np.random.choice(n_points)
            j = np.random.choice(n_points)
            if i != j:
                delta = min(h[i], 1 - h[j], 0.005)
                if delta > 1e-6:
                    h[i] -= delta
                    h[j] += delta
                    h = project_to_sum_box(h, TARGET_SUM)
                    c5 = evaluate_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
    
    return best_h, best_c5
