import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 180
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_grad_max_shift(h, shift, dx):
    """Compute gradient of C5 for the shift that maximizes it"""
    n = len(h)
    grad = np.zeros(n)
    
    if shift >= 0:
        # Shift is positive
        # For i in [0, n-shift-1], add (1 - h[i+shift]) * dx
        grad[0:n-shift] += (1 - h[shift:]) * dx
        # For i in [shift, n-1], add -h[i-shift] * dx
        grad[shift:] += -h[:n-shift] * dx
    else:
        # Shift is negative (i.e., shift = -k)
        k = -shift
        # For i in [k, n-1], add (1 - h[i-k]) * dx
        grad[k:] += (1 - h[:n-k]) * dx
        # For i in [0, n-k-1], add -h[i+k] * dx
        grad[:n-k] += -h[k:] * dx
    
    return grad
