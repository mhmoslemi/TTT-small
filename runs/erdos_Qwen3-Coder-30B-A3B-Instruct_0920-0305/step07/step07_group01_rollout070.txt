import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 180
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_grad_max_shift(h, shift_k, dx):
    """Compute gradient for the maximum shift"""
    n = len(h)
    grad = np.zeros(n)
    
    if shift_k >= 0:
        # Forward shift: h[i] * (1 - h[i+shift_k])
        # Derivative w.r.t h[i]: (1 - h[i+shift_k]) - h[i-shift_k] (if i-shift_k >= 0)
        if shift_k > 0:
            grad[0:n-shift_k] += (1 - h[shift_k:]) * dx
            grad[shift_k:] += -h[:-shift_k] * dx
        else:  # shift_k == 0
            grad += (1 - 2 * h) * dx
    else:
        # Negative shift: h[i] * (1 - h[i-shift_k]) where shift_k < 0
        # i - shift_k >= 0 => i >= abs(shift_k)
        shift_neg = -shift_k
        if shift_neg > 0:
            grad[shift_neg:] += (1 - h[:-shift_neg]) * dx
            grad[:-shift_neg] += -h[shift_neg:] * dx

    return grad

    best_h = h_interpolated.copy()
    best_c5 = evaluate_c5(h_interpolated)
    TARGET_SUM = n_points / 2
    
    # Gradient descent with smart step size and line search
    max_iter = 50000
    step_size = 0.2
    min_step = 1e-6
    iter = 0
    
    while iter < max_iter and (time.time() - start_time) < budget_s:
        # Compute correlation values
        corr = correlate(best_h, 1 - best_h, mode='full')
        c5_vals = corr * dx
        cur_c5 = np.max(c5_vals)
        
        # Find shift index of maximum
        max_idx = np.argmax(c5_vals)
        actual_shift = max_idx - (n_points - 1)
        
        # Compute gradient for this shift
        grad = compute_grad_max_shift(best_h, actual_shift, dx)
        
        # Line search for step size
        h_candidate = best_h - step_size * grad
        h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
        candidate_c5 = evaluate_c5(h_candidate)
        
        # Accept if improvement, or backtracking line search
        if candidate_c5 < cur_c5:
            h_new = h_candidate
            c5_new = candidate_c5
        else:
            # Backtracking line search
            current_step = step_size
            while current_step > min_step:
                current_step *= 0.5
                h_candidate = best_h - current_step * grad
                h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
                candidate_c5 = evaluate_c5(h_candidate)
                
                if candidate_c5 < cur_c5:
                    h_new = h_candidate
                    c5_new = candidate_c5
                    break
            else:
                # If we couldn't find an improving step, just proceed with current
                h_new = best_h
                c5_new = cur_c5
        
        # Update best if we have improvement
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        iter += 1
        # Decay step size gradually
        step_size *= 0.9995
        
        # Occasionally restart with small random noise
        if iter % 2000 == 0:

            noisy_h = best_h + noise

                best_h = noisy_h
                best_c5 = noisy_c5
