import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_with_gradients(h0, max_iter=2000):
    """Optimize using gradient-based method with projection"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def improve_solution_local(h0, max_iter=2000):
    """Iteratively improve solution using local search with policy changes"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    # Different update strategies for diversity
    strategies = [
        lambda h, idx: h[idx] + np.random.choice([-0.02, -0.01, 0.01, 0.02]),
        lambda h, idx: np.clip(h[idx] + np.random.normal(0, 0.01), 0, 1),
        lambda h, idx: h[idx] * np.random.uniform(0.95, 1.05)
    ]
    
    for i in range(max_iter):
        idx = np.random.randint(0, n_points)
        strategy = np.random.choice(strategies)
        
        h_new = h.copy()
        new_val = strategy(h, idx)
        h_new[idx] = new_val
        
        if h_new[idx] < 0 or h_new[idx] > 1:
            continue
            
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
    
    return h

    """Create better initial shape by using more structured approach"""
    # Create a piecewise constant function and adjust it to meet the sum constraint
    h = np.zeros(n_points)
    
    # Start with a few intervals that sum to 1
    intervals = 3
    segments = n_points // intervals
    for i in range(intervals):
        start = i * segments

        h[start:end] = 1.0 / intervals
        
    # Adjust so that sum equals TARGET_SUM

    diff = TARGET_SUM - current_sum
    if abs(diff) > 1e-6:
        # Distribute the difference evenly
        h += diff / n_points
        
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    h = project_to_sum_box(h, TARGET_SUM)
    
    return h

        # Local search optimization
        h_refined = improve_solution_local(h_start, max_iter=200)

        h_local = improve_solution_local(h_noise, max_iter=100)
        c5_local = evaluate_c5(h_local)
