import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT for speed"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient_of_corr_at_lag(h, lag):
    """Compute gradient of correlation for given lag"""
    n = len(h)
    g = np.ones(n)
    
    if lag >= 0:
        if lag < n:
            # Gradient for terms where i+lag is within bounds
            g[:n-lag] = 1 - h[lag:]
            # Subtract contribution from h_{i-lag}
            g[lag:] -= h[:n-lag]
    else:
        # For negative lag, shift the indices appropriately
        d = -lag
        if d < n:
            # Gradient for terms where i-d is within bounds
            g[d:] = 1 - h[:n-d]
            # Subtract contribution from h_{i+d}
            g[:n-d] -= h[d:]
            
    return g

def gradient_descent_refinement(h0, max_iter=2000):
    """Refine solution with gradient-based method"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    step_size = 0.1
    for i in range(max_iter):
        if i % 100 == 0:
            corr = compute_corr(h)
            max_index = np.argmax(corr)
            max_lag = max_index - (len(h) - 1)
            
            # Compute gradient for the max lag
            g = compute_gradient_of_corr_at_lag(h, max_lag)
            
            # Take a step in negative gradient direction
            h_new = h - step_size * g
            
            # Project to [0,1] and maintain sum constraint
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            new_c5 = compute_c5(h_new)
            
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                step_size = 0.1  # Reset step size on improvement
            else:
                step_size *= 0.95  # Reduce step size
                if step_size < 1e-6:
                    break
                
            h = h_new.copy()
            
    return best_h, best_c5

def simulated_annealing_binary(h0, temp_start, temp_end, max_iter=5000):
    """Simulated Annealing optimization for binary sequences"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    temp = temp_start
    temp_decay = (temp_end / temp_start) ** (1.0 / max_iter)
    
    for i in range(max_iter):
        # Randomly select two positions to swap
        indices_ones = np.where(h > 0.5)[0]
        indices_zeros = np.where(h < 0.5)[0]
        
        if len(indices_ones) > 0 and len(indices_zeros) > 0:
            i1 = np.random.choice(indices_ones)
            i2 = np.random.choice(indices_zeros)
            
            # Perform swap
            h_new = h.copy()
            h_new[i1] = 0
            h_new[i2] = 1
            
            # Project to maintain sum constraint
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            new_c5 = compute_c5(h_new)
            current_c5 = compute_c5(h)
            
            delta = new_c5 - current_c5
            
            # Accept with Metropolis probability
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                h = h_new.copy()
                if new_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = new_c5
                    
        # Cool down temperature
        temp *= temp_decay
        
    return best_h, best_c5

        if c5_sa < best_c5:
            best_h = h_sa.copy()

    if best_h is not None:
        # Convert to continuous values in [0,1] and refine
        continuous_h = best_h.copy()
        # Use SLSQP to refine with continuous variables
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return compute_c5(h)
            
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
            
        cons = {'type': 'eq', 'fun': constraint_eq}

            )

            # Project to maintain sum and bounds
