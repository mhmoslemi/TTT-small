import numpy as np
from scipy.signal import correlate
import time

# Global parameters
TARGET_SUM = None
MAX_N = 360
n_points = 120
dx = 2.0 / n_points
start_time = None

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_gradient(h, corr, beta):
    """Compute gradient of smooth max surrogate"""
    max_corr = np.max(corr)
    # Subtract max to prevent overflow
    exp_corr = np.exp(beta * (corr - max_corr))
    w = exp_corr / np.sum(exp_corr)
    
    # Compute gradient
    grad = np.zeros_like(h)
    for ell in range(len(corr)):
        if ell < len(h):
            grad[ell] += w[ell] * (1 - h[ell])
        if ell >= len(h):
            i = ell - len(h) + 1
            if i < len(h):
                grad[i] += w[ell] * (1 - h[i])
        if ell >= len(h):
            i = ell - len(h) + 1
            if i < len(h):
                grad[i] -= w[ell] * h[i]
        else:
            grad[ell] -= w[ell] * h[ell] 
    return -dx * grad

def max_lag_gradient(h, idx_best):
    """Compute exact subgradient for the max of correlations"""
    grad = np.zeros_like(h)
    # worst lag
    k = idx_best - len(h) + 1
    # Handle negative lag
    if k > 0:  # k = idx_best - (N-1)
        # Elements 0 to N-k-1 
        # grad[i] += (1 - h[i+k]) * dx  if i < N-k
        # grad[i] -= h[i-k] * dx if i >= k
        grad[:len(h)-k] += (1 - h[k:]) * dx
        grad[k:] -= h[:-k] * dx
    else:
        # k = idx_best - (N-1) < 0
        # Elements 0 to N+k-1 
        # grad[i] += (1 - h[i-k]) * dx   0 <= i < N-k 
        # grad[i] -= h[i+k] * dx   k <= i < N
        grad[:len(h)+k] += (1 - h[-k:]) * dx
        grad[-k:] -= h[:k] * dx
    return -grad

def mass_transfer(h, grad, epsilon):
    """Perform greedy mass transfer move"""
    d = -grad
    sources = np.where(d < -1e-10)[0]
    sinks = np.where(d > 1e-10)[0]
    
    if len(sources) == 0 or len(sinks) == 0:
        return h
    
    # Sort sources by decreasing -d and sinks by decreasing d
    sources = sources[np.argsort(-d[sources])]
    sinks = sinks[np.argsort(-d[sinks])]
    
    h_copy = h.copy()
    # Set total amount to move, limit to epsilon fraction of total mass
    total_move = epsilon * np.sum(h_copy)
    moved = 0
    s_idx = 0
    t_idx = 0
    
    while moved < total_move and s_idx < len(sources) and t_idx < len(sinks):
        s = sources[s_idx]
        t = sinks[t_idx]
        max_rem = h_copy[s]
        max_add = 1.0 - h_copy[t]
        transfer = min(max_rem, max_add, total_move - moved)
        if transfer < 1e-12:
            if d[s] < -1e-10:
                s_idx += 1  # advance to next source
            if d[t] > 1e-10:
                t_idx += 1  # advance to next sink
            continue
        else:
            h_copy[s] -= transfer
            h_copy[t] += transfer
            moved += transfer
    return project_sum_box(h_copy, TARGET_SUM)

def backtracking_line_search(h, grad, corr, idx_best, beta, alpha0=0.2, alpha_min=1e-4):
    """Backtracking line search"""
    alpha = alpha0
    cmax = np.max(corr)
    max_iter = 100
    for _ in range(max_iter):
        h_new = h - alpha * grad
        h_new = project_sum_box(h_new, TARGET_SUM)
        cmax_new = evaluate_c5(h_new)
        if cmax_new < cmax:
            return h_new, cmax_new
        alpha *= 0.5
        if alpha < alpha_min:
            break
    return h, cmax

    return (best_h.tolist(), float(best_c5), n_points)
