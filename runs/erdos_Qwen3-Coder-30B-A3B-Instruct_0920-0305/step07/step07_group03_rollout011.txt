import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_of_corr_at_lag(h, lag):
    """Compute the gradient of correlation at a specific lag"""
    n = len(h)
    g = np.ones(n)
    
    if lag >= 0:
        if lag < n:
            # For i from 0 to n-lag-1, we have h[i] * (1 - h[i+lag])
            g[:n-lag] = 1 - h[lag:]  # derivative of (1 - h[i+lag]) w.r.t h[i]
            g[lag:] -= h[:n-lag]     # derivative of -h[i] * h[i+lag] w.r.t h[i]
    else:
        # lag < 0: equivalent to positive lag but shifted indices
        d = -lag
        if d < n:
            # i from d to n-1: derivative of (1 - h[i+lag]) = 1 - h[i-d] 
            g[d:] = 1 - h[:n-d]      # derivative of (1 - h[i+lag])
            # derivative of -h[i] * h[i+lag] w.r.t. h[i] = -h[i-d] (for i-d in range)
            g[:n-d] -= h[d:]         # subtract h[i-d]
    
    return g

def subgradient_descent_step(h, step_size):
    """Perform one step of subgradient descent"""
    corr = compute_corr(h) 
    max_val = np.max(corr)
    max_indices = np.where(np.abs(corr - max_val) < 1e-10)[0]
    
    grad_sum = np.zeros_like(h)
    for idx in max_indices:
        lag = idx - (len(h) - 1)
        grad_sum += gradient_of_corr_at_lag(h, lag)
    
    if len(max_indices) > 0:
        grad = grad_sum / len(max_indices)
    else:
        grad = np.zeros_like(h)
    
    h_new = h - step_size * grad
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Phase 1: Simulated Annealing on binary patterns
    max_sa_time = 0.7 * budget_s
    sa_end_time = start_time + max_sa_time
    
    # Try multiple restarts with different strategies
    for restart in range(10):
        if time.time() > sa_end_time:
            break
            
        # Strategy 1: Random binary
        h = np.zeros(n_points)
        ones = np.random.choice(n_points, size=n_points//2, replace=False)
        h[ones] = 1
        
        # Strategy 2: Block patterns
        if restart % 3 == 0:
            h = np.zeros(n_points)
            # Create two blocks of ones
            block1_start = np.random.randint(0, n_points//2)
            block2_start = np.random.randint(n_points//2, n_points-10)
            block_length = 10 + np.random.randint(0, 20)
            h[block1_start:block1_start+block_length] = 1
            h[block2_start:block2_start+block_length] = 1
            
        # Strategy 3: Simple pattern 
        elif restart % 3 == 1:
            h = np.zeros(n_points)
            # Place ones at regular intervals
            interval = n_points // 60 + np.random.randint(-2, 3)
            positions = np.arange(0, n_points, interval)
            positions = positions[:n_points//2]
            h[positions] = 1
            
        current_h = h.copy()
        current_c5 = evaluate_c5(current_h)
        
        # Simulated Annealing
        temp = current_c5 * 0.5 if current_c5 > 0 else 0.1
        cooling_rate = 0.9995
        min_temp = 1e-6
        iterations = 0
        
        while time.time() < sa_end_time and temp > min_temp:
            # Make a move
            h_new = current_h.copy()
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            if h_new[i] == h_new[j]:
                # Try to make a swap
                i = np.random.choice(np.where(h_new > 0.5)[0]) if len(np.where(h_new > 0.5)[0]) > 0 else i
                j = np.random.choice(np.where(h_new < 0.5)[0]) if len(np.where(h_new < 0.5)[0]) > 0 else j
                h_new[i], h_new[j] = h_new[j], h_new[i]
            else:
                h_new[i] = 1 - h_new[i]
            
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            new_c5 = evaluate_c5(h_new)
            
            # Accept or reject
            delta = new_c5 - current_c5
            if delta < 0 or np.random.random() < np.exp(-delta / temp):
                current_h = h_new
                current_c5 = new_c5
                
            temp *= cooling_rate
            iterations += 1
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
    
    # Phase 2: Gradient-based refinement using custom algorithm
    max_refinement_time = budget_s - (time.time() - start_time)
    refinement_end_time = start_time + max_refinement_time
    
    if best_h is not None:
        current_h = best_h.copy()
        current_c5 = best_c5
        
        # Subgradient descent
        step_size = 0.02
        iteration = 0
        
        while time.time() < refinement_end_time:
            iteration += 1
            
            # Decrease step size over iterations
            if iteration % 100 == 0:
                step_size *= 0.95
                if step_size < 1e-5:
                    break
                    
            h_new = subgradient_descent_step(current_h, step_size)
            new_c5 = evaluate_c5(h_new)
            
            if new_c5 < current_c5:
                current_h = h_new
                current_c5 = new_c5
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
            else:
                # Occasionally try another strategy
                if np.random.random() < 0.1:
                    noise_h = current_h + np.random.normal(0, 0.01, n_points)
                    noise_h = project_to_sum_box(noise_h, TARGET_SUM)
                    noise_c5 = evaluate_c5(noise_h)
                    if noise_c5 < current_c5:
                        current_h = noise_h
                        current_c5 = noise_c5
    
    # Final check and optimization
    if best_h is None:
        # Fallback to random construction  
        best_h = np.random.uniform(0, 1, n_points)
        best_h = project_to_sum_box(best_h, TARGET_SUM)
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
