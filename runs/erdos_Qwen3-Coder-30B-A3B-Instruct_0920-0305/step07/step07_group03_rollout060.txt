import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_c5(h):
    """Compute C5 value"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_of_corr_at_lag(h, lag):
    """Compute gradient of correlation at specific lag"""
    n = len(h)
    g = np.ones(n)
    
    if lag >= 0:
        # i from 0 to n-lag-1
        if lag < n:
            g[:n-lag] = 1 - h[lag:]
            g[lag:] -= h[:n-lag]
    else:
        # lag < 0, let d = -lag
        d = -lag
        if d < n:
            g[d:] = 1 - h[:n-d]
            g[:n-d] -= h[d:]
    
    return g

def subgradient_descent(h0, max_iter=5000, step_size=0.1):
    """Subgradient descent for minimizing max correlation"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for i in range(max_iter):
        corr = compute_corr(h)
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Compute subgradient for the maximum correlation
        lag = max_idx - (len(h) - 1)
        g = gradient_of_corr_at_lag(h, lag)
        
        # Take step
        step = step_size / np.sqrt(i + 1)
        h_new = h - step * g
        
        # Project
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new
        
    return best_h, best_c5

def simulated_annealing_binary(h0, max_iter=10000, temp_start=0.1):
    """Binary simulated annealing with swaps"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    temp = temp_start
    temp_decay = 0.9995
    
    for i in range(max_iter):
        # Pick two positions to swap: one with 1 and one with 0
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        if len(ones) > 0 and len(zeros) > 0:
            idx1 = np.random.choice(ones)
            idx2 = np.random.choice(zeros)
            
            # Swap
            h[idx1], h[idx2] = h[idx2], h[idx1]
            
            c5_new = compute_c5(h)
            
            # Accept or reject
            delta = c5_new - best_c5
            
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
            else:
                # Revert
                h[idx1], h[idx2] = h[idx2], h[idx1]
        
        temp *= temp_decay
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Use multiple random restarts
    num_restarts = 10
    for restart in range(num_restarts):
        if time.time() - start_time > 0.6 * budget_s:
            break
            
        # Create binary initial solution
        h_binary = np.zeros(n_points)
        ones_indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h_binary[ones_indices] = 1.0
        
        # Simulated annealing on binary
        h_sa, c5_sa = simulated_annealing_binary(h_binary, max_iter=5000)
        
        # Refine with continuous optimization
        h_refined, c5_refined = subgradient_descent(h_sa, max_iter=2000)
        
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
            
        # Simple local search for further improvement
        if time.time() - start_time > 0.7 * budget_s:
            break
    
    # Final refinement with gradient descent
    if best_h is not None and time.time() - start_time < budget_s - 10:
        h_final, c5_final = subgradient_descent(best_h, max_iter=2000)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
            
    return (best_h.tolist(), float(best_c5), n_points)
