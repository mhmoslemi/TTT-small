```python
import numpy as np
from scipy.linalg import eigh
from cvxpy import *
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 140
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2  # This must be the exact sum for the integral constraint
    
    # Build the A_lag matrices using sparse format for efficiency
    def build_A_matrices(n):
        A_list = []
        for l in range(-(n-1), n):
            rows = []
            cols = []
            data = []
            for i in range(n):
                if 0 <= i + l < n:
                    rows.append(i)
                    cols.append(i + l)
                    data.append(1)
            # Use scipy sparse matrix structure
            A_list.append((l, rows, cols, data))
        return A_list
    
    A_matrices = build_A_matrices(n_points)
    
    # Construct CVXPY problem 
    X = Semidef(n_points, name="X")
    t = Variable(name="t")  # The objective variable
    
    # Constraints for X
    constraints = [
        0 <= X.diag(),  # All diagonal entries are >= 0
        X.diag() <= 1,  # All diagonal entries are <= 1
        sum(X.diag()) == TARGET_SUM,
    ]
    
    # Add constraints for lagged terms
    for l, rows, cols, data in A_matrices:
        # Use cvxpy's trace and matrix multiplication
        A_l = np.zeros((n_points, n_points))
        A_l[rows, cols] = data
        constraints.append(trace(A_l @ X) >= TARGET_SUM - t / dx)
    
    # Objective: minimize t
    objective = Minimize(t)
    prob = Problem(objective, constraints)

    # Solve SDP
    try:
        # Use SCS for faster solution, reasonably accurate for our scale
        prob.solve(solver=SCS, verbose=False, max_iters=10000, eps=1e-5)
        print(f"SDP solved. Status: {prob.status}")
    except Exception as e:
        print(f"Error solving SDP: {e}")
        # Fallback: return a simple construction
        h_simple = np.ones(n_points) * 0.5
        h_simple = h_simple / np.sum(h_simple) * TARGET_SUM
        h_simple = np.clip(h_simple, 0, 1)
        return (h_simple.tolist(), 1.0, n_points)

    # Extract solution
    X_opt = X.value
    t_opt = t.value
    
    best_h = None
    best_c5 = float('inf')
    
    # Rounding with median eigenvalue and satisfying clips and sum
    def project_to_sum_box(x, target_sum):
        # Binary search for lambda
        def eval_shift(lam):
            projected = np.clip(x + lam, 0, 1)
            return np.sum(projected) - target_sum
        
        a, b = -1, 1
        while eval_shift(a) * eval_shift(b) > 0:
            a -= 0.1
            b += 0.1
        tol = 1e-10
        for _ in range(50):
            m = (a + b) / 2
            fm = eval_shift(m)
            if abs(fm) < tol:
                break
            if fm * eval_shift(a) < 0:
                b = m
            else:
                a = m
        result = np.clip(x + (a + b)/2, 0, 1)
        return result

    # Sample from the eigen-decomposition of X_opt
    # Sample 500 candidates via random rounding
    samples = 500
    for _ in range(samples):
        try:
            # Eigenvalue decomposition 
            # This is an approximation because cvxpy give us a semidefinite matrix
            w, v = eigh(X_opt)
            
            # We will sample from the improved approximation
            # Use a normalized random vector to simulate the sampling
            v_prime = v[:, -1]  # Take the eigenvector corresponding to largest eigenvalue
            # Project it to get a feasible vector
            if np.any(np.isnan(v_prime)) or np.any(np.isinf(v_prime)):
                continue
                
            h_candidate = v_prime.copy()  # Could be negative or unnormalized
            h_candidate = np.abs(h_candidate)  # Make sure non-negative (handle all negative case)
            
            h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)

            # Compute C5 value carefully using correlation
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_value = dx * np.max(corr)
            
            if c5_value < best_c5:
                best_c5 = c5_value
                best_h = h_candidate.copy()
        except:
            continue

    # If we failed to find a good solution through sampling, make simple search
    if best_h is None:
        h_simple = np.ones(n_points) * 0.5
        h_simple = h_simple / np.sum(h_simple) * TARGET_SUM
        h_simple = np.clip(h_simple, 0, 1)
        best_h = h_simple.copy()
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = dx * np.max(corr)
    
    # Final local refinement (any improvement?)
    def evaluate_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return dx * np.max(corr)

    local_search_end = start_time + 0.2 * budget_s
    for _ in range(1000):
        if time.time() >= local_search_end:
            break
        # Perform small random modifications
        h_new = best_h.copy()
        i = np.random.randint(n_points)
        j = np.random.randint(n_points)
        delta = np.random.uniform(0.001, 0.02)
        h_new[i] = np.clip(h_new[i] - delta, 0, 1)
        h_new[j] = np.clip(h_new[j] + delta, 0, 1)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    return (best_h.tolist(), float(best_c5), n_points)
```
