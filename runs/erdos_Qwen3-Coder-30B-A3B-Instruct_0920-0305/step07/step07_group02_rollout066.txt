import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_c5_fft(h):
    """Fast evaluation of C5 using FFT"""
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using convolution"""
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def greedy_descent(h0):
    """Determistic descent on worst shift"""
    h = h0.copy()
    c5 = evaluate_c5(h)
    max_iter = 10000
    for _ in range(max_iter):
        corr_full = np.correlate(h, h, mode='full')
        # Worst shift: where h[i] * h[i+k] is minimized in overlap
        # We compute 1-h[i] * h[i+k] to find maximizing overlap
        corr_neg = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr_neg)
        worst_shift = max_corr_idx - len(h) + 1
        
        # Find the positions that contribute to the worst case
        # h[i] = 1 and h[i + k] = 0 give +dx to overlap
        # h[i] = 0 and h[i + k] = 1 give +dx to overlap
        # Our goal is to minimize total overlap
        # For shift k, we look at positions where h[i] = 1, h[i+k] = 0
        # This contributes to overlap
        # The goal is to reduce such mismatch

        # This corresponds to positions where h[i] = 1 and h[i+k] = 0
        # But better approach: look at worst shift criterion more directly
        # Using actual convolution of h and (1-h)
        corr = np.correlate(h, 1 - h, mode='full')
        worst_i = np.argmax(corr)
        shift = worst_i - len(h) + 1
        # Compute shift index properly
        # Position i is shifted by shift to get i + shift, this should be looked up
        
        # Faulty way -- rewrite this better
        # Find change that would reduce the worst overlap
        shift_index = worst_i
        k = shift_index - (len(h) - 1)  # i + k = shift_index
        
        # Positions i where h[i] * h[i+k] contributes to the worst case
        # Actually, h[i] * h[i+k] contributes only to the correlation the sum counts
        # So h[i] * (1 - h[i+k]) contributes to overlap
        # Thus we want to reduce sum h[i] * (1 - h[i+k]) = sum h[i] - sum h[i] * h[i+k]
        # So we decrease sum h[i] * h[i+k] (minimize this for the worst shift)
        # We have h[i] * h[i+k]
        # So for a given shift k, we want to change h so reduce sum of h[i]*h[i+k]
        # Which means changing to 0 for some i where h[i] = 1 and h[i+k] = 1 or increase where h[i] = 0 and h[i+k] = 1
        
        # Find mismatch in this shift that contributes most to overlap
        # Positions intersecting with shift k:
        # for i in range(len(h)):
        #     if i + k >= 0 and i + k < len(h):
        #         j = i + k
        #         if h[i] == 1 and h[j] == 0:  # contributes to C5
        #             good 1-0 pairs but not helping reduce

        # Actually, better idea:
        # Let the 'one' appears in i, zero appears in i+k 
        # Now sum h[i] * (1 - h[i+k]) should reduce
        # So change 1 to 0 in i and 0 to 1 in j = i+k,
        # To reduce h[i] * (1 - h[i+k]) = 1 * (1 - 0) = 1

        # So we want to look for:
        # i where h[i] == 1
        # j where h[j] == 0 AND j == i + k
        # If both exist, we can swap them
        a_positions = []
        b_positions = []
        for i in range(len(h)):
            j = i + k
            if 0 <= j < len(h):
                if h[i] == 1 and h[j] == 0:
                    a_positions.append(i)  # 1 at i, 0 at j
                elif h[i] == 0 and h[j] == 1:
                    b_positions.append(i)  # 0 at i, 1 at j
        
        if a_positions and b_positions:
            # Pick a random one from each
            i = np.random.choice(a_positions)
            j = np.random.choice(b_positions)
            # Apply the swap
            h[i], h[j] = h[j], h[i]
            new_c5 = evaluate_c5(h)
            if new_c5 < c5:
                c5 = new_c5
            else:
                # Undo the swap if not good
                h[i], h[j] = h[j], h[i]
        else:
            break  # No beneficial moves left
    return h, c5

def anneal_binary(h0, remaining_time):
    """Simulated annealing on binary vectors"""
    h = h0.copy()
    c5 = evaluate_c5(h)
    T0 = 0.02
    T_min = 1e-5
    max_steps = 50000
    step = 0
    
    start_time = time.time()
    
    while step < max_steps and (time.time() - start_time) < remaining_time - 1:
        # Random bit swap
        i = np.random.randint(len(h))
        j = np.random.randint(len(h))
        # Swap
        h[i], h[j] = h[j], h[i]
        new_c5 = evaluate_c5(h)
        
        if new_c5 < c5:
            c5 = new_c5
        else:
            # Accept with probability
            delta = new_c5 - c5
            T = T0 * (1 - step / max_steps)  # Cooling schedule
            if T < T_min:
                T = T_min
            if np.random.rand() > np.exp(-delta / T):
                # Revert swap
                h[i], h[j] = h[j], h[i]
        
        step += 1
        
        # Periodically do a greedy sweep
        if step % 1000 == 0:
            h, c5 = greedy_descent(h)
    
    return h, c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    global TARGET_SUM, n_points, dx
    start_time = time.time()
    
    # Model and solve with MILP
    N = 120
    dx = 2.0 / N
    target_sum = N // 2

    # Build MILP model
    
    h = cp.Variable(N, boolean=True)
    lam = cp.Variable(integer=True)
    
    # We'll build constraints for each shift
    # w[i,k] indicates h[i] * h[i+k]
    # Construct w as a variable for each (i,k) pair
    # First collect all (i, k) pairs with valid indices
    constraints = [cp.sum(h) == target_sum]
    
    # Build w variables and constraints
    w_list = []
    w_map = {}
    for k in range(-(N - 1), N):
        idxs_i = np.arange(max(0, -k), min(N, N - k))
        i = idxs_i
        j = i + k
        
        if len(i) > 0:
            # Create w as a list of auxiliary binary variables
            w = cp.Variable(len(i), boolean=True)
            # Store indices in map for constraints
            w_map[(k,)] = (i, j, w)
            constraints += [w <= h[i], w <= h[j], w >= h[i] + h[j] - 1]
            w_list.append(w)
            
    # Also add constraint that correlation >= lambda for each shift
    for k in range(-(N - 1), N):
        idxs_i = np.arange(max(0, -k), min(N, N - k))
        if len(idxs_i) > 0:
            i = idxs_i
            j = i + k
            w = w_map[(k,)][2]
            constraints += [cp.sum(w) >= lam]
    
    # Define objective
    objective = cp.Maximize(lam)
    prob = cp.Problem(objective, constraints)
    
    # Solve with a time limit (MILP solver config)
    remaining_time = budget_s * 0.55
    try:
        prob.solve(solver=cp.CBC, max_seconds=remaining_time, verbose=False)
        # Check if we got a solution
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_best = np.round(h.value).astype(int)
            if np.sum(h_best) != target_sum:
                # Repair the solution
                # Find positions where we are off by 1
                diff = int(target_sum - np.sum(h_best))
                if diff > 0:
                    # Need to convert some 0s to 1s
                    zero_indices = np.where(h_best == 0)[0]
                    if len(zero_indices) >= diff:
                        # Flip randomly
                        indices = np.random.choice(zero_indices, diff, replace=False)
                        for idx in indices:
                            h_best[idx] = 1
                else:
                    # Need to convert some 1s to 0s
                    one_indices = np.where(h_best == 1)[0]
                    if len(one_indices) >= -diff:
                        indices = np.random.choice(one_indices, -diff, replace=False)
                        for idx in indices:
                            h_best[idx] = 0
            # Compute C5 directly from the result
            c5_best = 1 - (2 * lam.value) / N
        else:
            h_best = np.random.randint(0, 2, N)
            h_best = np.round(h_best).astype(int)
            h_best = h_best / np.sum(h_best) * target_sum
            h_best = np.round(h_best).astype(int)
            # Fix the sum if needed
            diff = target_sum - np.sum(h_best)
            if diff > 0:
                # Convert some 0s to 1s
                zero_indices = np.where(h_best == 0)[0]
                if len(zero_indices) >= diff:
                    for i in range(diff):
                        h_best[zero_indices[i]] = 1
            elif diff < 0:
                # Convert some 1s to 0s
                one_indices = np.where(h_best == 1)[0]
                if len(one_indices) >= -diff:
                    for i in range(-diff):
                        h_best[one_indices[i]] = 0
            
            c5_best = compute_c5_fft(h_best)
    except:
        h_best = np.random.randint(0, 2, N)
        h_best = np.round(h_best).astype(int)
        h_best = h_best / np.sum(h_best) * target_sum
        h_best = np.round(h_best).astype(int)
        # Fix
        diff = target_sum - np.sum(h_best)
        if diff > 0:
            zero_indices = np.where(h_best == 0)[0]
            if len(zero_indices) >= diff:
                for i in range(diff):
                    h_best[zero_indices[i]] = 1
        elif diff < 0:
            one_indices = np.where(h_best == 1)[0]
            if len(one_indices) >= -diff:
                for i in range(-diff):
                    h_best[one_indices[i]] = 0
        c5_best = compute_c5_fft(h_best)

    # Greedy improvement 
    elapsed_time = time.time() - start_time
    if elapsed_time < 0.75 * budget_s:
        try:
            h_best, c5_best = greedy_descent(h_best)
        except:
            pass

    # Simulated annealing
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 30:
        try:
            h_best, c5_best = anneal_binary(h_best, remaining_time)
        except:
            pass
    
    return (h_best.tolist(), float(c5_best), N)
