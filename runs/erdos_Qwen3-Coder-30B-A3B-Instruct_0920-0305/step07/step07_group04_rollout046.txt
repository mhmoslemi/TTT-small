import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_optimization(h0, max_iter=2000):
    """Optimize using gradient-based method with projection"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def simulated_annealing(h0, max_iter=2000):
    """Simulated annealing approach to escape local minima"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Temperature schedule
    T = 0.1
    T_min = 1e-6
    alpha = 0.95
    
    for i in range(max_iter):
        if T < T_min:
            break
            
        # Make a small random change
        idx = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.05, 0.05)
        h_new = h.copy()
        h_new[idx] += delta
        
        # Keep within bounds
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        delta_c5 = new_c5 - best_c5
        
        # Accept or reject based on simulated annealing
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new.copy()
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                
        T *= alpha
    
    return best_h

def evolutionary_search(h0, max_iter=1000):
    """Evolutionary algorithm inspired approach"""
    population_size = 20
    population = [h0.copy()]
    
    # Generate initial population
    for _ in range(population_size - 1):
        h_perturbed = h0 + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
        population.append(h_perturbed)
    
    for generation in range(max_iter):
        # Evaluate fitness
        fitness = [evaluate_c5(h) for h in population]
        min_fitness_idx = np.argmin(fitness)
        
        if generation % 100 == 0:
            # Keep the best individual
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
        
        # Selection: keep top 50%
        sorted_indices = np.argsort(fitness)[:population_size//2]
        selected = [population[i] for i in sorted_indices]
        
        # Create new population via crossover and mutation
        new_population = selected.copy()
        while len(new_population) < population_size:
            # Crossover
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            crossover_point = np.random.randint(1, n_points-1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            mutation_strength = 0.03 * (1 - generation/max_iter)  # Decrease over time
            for i in range(n_points):
                if np.random.rand() < 0.1:  # 10% chance to mutate
                    child[i] += np.random.normal(0, mutation_strength)
                    
            # Project to valid range and constraints
            child = np.clip(child, 0, 1)
            child = project_to_sum_box(child, TARGET_SUM)
            new_population.append(child)
        
        population = new_population
    
    # Return best from final population
    fitness = [evaluate_c5(h) for h in population]
    min_fitness_idx = np.argmin(fitness)
    return population[min_fitness_idx]
