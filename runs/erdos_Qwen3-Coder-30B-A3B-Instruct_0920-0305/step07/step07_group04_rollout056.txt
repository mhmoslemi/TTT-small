from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_with_gradients(h0, max_iter=1000):
    """Optimize using gradient-based method with projection"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def adaptive_local_search(h0, max_iter=1000):
    """Improved local search combining multiple strategies"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Try different perturbation strategies
    for i in range(max_iter):
        # Adaptively choose perturbation strategy
        strategy = i % 4
        h_new = h.copy()
        
        if strategy == 0:  # Random perturbations with different magnitudes
            idx = np.random.randint(0, n_points)
            delta = np.random.uniform(-0.08, 0.08)
            h_new[idx] += delta
        elif strategy == 1:  # Clustered perturbations
            idx = np.random.randint(0, n_points)
            # Apply same delta to nearby points
            for j in range(max(0, idx-2), min(n_points, idx+3)):
                h_new[j] += np.random.uniform(-0.05, 0.05)
        elif strategy == 2:  # Global scaling
            scaling = np.random.uniform(0.9, 1.1)
            h_new = h_new * scaling
            # Add small noise to avoid getting stuck
            h_new += np.random.normal(0, 0.005, n_points)
        else:  # Gradient-based adaptation with small random steps
            # Get gradient estimates through finite differences
            step = 1e-4
            grad = np.zeros(n_points)
            base_c5 = evaluate_c5(h)
            for j in range(n_points):

                h_test[j] += step

            h_new = h - 0.01 * grad + np.random.normal(0, 0.01, n_points)
