import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

# Global variables
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def fourier_basis(x, m):
    """Cosine basis function"""
    return np.cos(m * np.pi * x)

def reconstruct_h(a, x):
    """Reconstruct h from Fourier coefficients"""
    h = np.zeros_like(x)
    for m in range(len(a)):
        h += a[m] * fourier_basis(x, m+1)
    h += 0.5  # Constant term to ensure integral = 1
    return np.clip(h, 0, 1)

def objective(a):
    """Objective function to minimize C5"""
    # Reconstruct h for the full domain
    x = np.linspace(0, 2, n_points)
    h = reconstruct_h(a, x)
    
    # Evaluate C5 using correlation
    c5 = evaluate_c5(h)
    
    return c5

    """Constraint function to ensure h(x) in [0,1] at sampling points"""
    x = np.linspace(0, 2, n_points)
    h = reconstruct_h(a, x)
    # Penalty for values outside [0,1]
    penalty = np.sum(np.maximum(0, h - 1)) + np.sum(np.maximum(0, -h))
    return -penalty  # Return negative because we want to MAXIMIZE constraint
