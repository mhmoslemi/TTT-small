import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 150
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmax_weights(corr, eps):
    """Compute softmax weights from correlation values"""
    m = np.max(corr)
    # Stable computation to avoid overflow
    w_raw = np.exp((corr - m) / eps)
    return w_raw / w_raw.sum()

def compute_gradient(h, w_norm):
    """Compute gradient using convolution approach for better efficiency"""
    n = len(h)
    L = 2 * n - 1
    
    # Convolution approach
    conv_wh = np.convolve(w_norm, h, mode='full')
    conv_rev = np.convolve(w_norm[::-1], h, mode='full')
    
    offset = n - 1
    grad = 1.0 - (conv_rev[offset:offset+n] + conv_wh[offset:offset+n])
    return grad

def optimize_h(start_h, eps_schedule, iterations_per_eps, lr, target_sum):
    """Optimize h using gradient descent with smoothing"""
    h = start_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for eps_idx, (eps, max_iter) in enumerate(zip(eps_schedule, iterations_per_eps)):
        for iter in range(max_iter):
            # Compute correlation
            corr = np.correlate(h, 1 - h, mode='full')
            
            # Compute weights
            w_norm = softmax_weights(corr, eps)
            
            # Compute gradient
            grad = compute_gradient(h, w_norm)
            
            # Update
            h_new = h - lr * grad
            
            # Project onto constraints
            h_new = project_to_sum_box(h_new, target_sum)
            
            h = h_new
            
            # Occasionally evaluate and track best
            if iter % 20 == 0 or iter == max_iter - 1:
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        
        # Decrease learning rate a bit for next epsilon
        lr *= 0.95
    
    return best_h, best_c5
