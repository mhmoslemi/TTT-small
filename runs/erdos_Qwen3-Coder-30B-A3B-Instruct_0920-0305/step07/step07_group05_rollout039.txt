import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def simulated_annealing(h0, max_iter=5000, temp_start=1e-2, cool_rate=0.9995):
    """Simulated annealing optimization"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    temp = temp_start
    for i in range(max_iter):
        # Create candidate solution
        h_candidate = h.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.01)
        h_candidate[idx] += delta
        h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
        
        # Evaluate
        c5_candidate = evaluate_c5(h_candidate)
        
        # Accept or reject
        if c5_candidate < best_c5 or np.random.rand() < np.exp(-(c5_candidate - best_c5) / temp):
            h = h_candidate.copy()
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h.copy()
        
        # Cool down
        temp *= cool_rate
        
        # Early stopping if close to target
        if best_c5 < 0.3795:
            break
            
    return best_h, best_c5

def improved_gradient_guided_search(h0, max_iter=1000):
    """Improved gradient-guided search with direction-aware updates"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for iter_num in range(max_iter):
        # Compute approximate gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        
        for i in range(n_points):
            # Forward difference
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_sum_box(h_plus, TARGET_SUM)
            c5_plus = evaluate_c5(h_plus)
            
            # Backward difference
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_sum_box(h_minus, TARGET_SUM)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Find indices to update based on gradient direction
        sorted_indices = np.argsort(grad)
        idx_min = sorted_indices[0]  # Most negative gradient
        idx_max = sorted_indices[-1]  # Most positive gradient
        
        # Apply updates intelligently
        if grad[idx_min] < -abs(grad[idx_max]) * 0.1:  # Strong enough negative gradient
            # Move mass from idx_min to idx_max if possible
            delta = min(h[idx_min], 1 - h[idx_max], 0.01)
            if delta > 1e-6:
                h[idx_min] -= delta
                h[idx_max] += delta
                h = project_to_sum_box(h, TARGET_SUM)
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        else:
            # Small random perturbations
            i, j = np.random.choice(n_points, 2, replace=False)
            delta = min(h[i], 1 - h[j], 0.002)
            if delta > 1e-6:
                h[i] -= delta
                h[j] += delta
                h = project_to_sum_box(h, TARGET_SUM)
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        
        # Periodic improvement check
        if iter_num % 50 == 0 and best_c5 < 0.38:
            h = best_h.copy()  # Gradually stick to good solutions
    
    return best_h, best_c5
