import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
from scipy.linalg import eigh

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr)

def build_lag_matrices(n):
    """Construct sparse matrix representations of A_ell for all lags."""
    A_list = []
    for lag in range(-(n-1), n):
        rows, cols = [], []
        # For each valid pair (i, i+lag), add indices
        for i in range(n):
            j = i + lag
            if 0 <= j < n:
                rows.append(i)
                cols.append(j)
        A_list.append((lag, rows, cols))
    return A_list

def solve_sdp_lagrangian(n, dx, A_list, max_time=45):
    """Solve the SDP relaxation and return solution X and bound t."""
    start_time = time.time()

    # Decision variables
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    constraints = [
        cp.diag(X) >= 0,
        cp.diag(X) <= 1,
        cp.sum(cp.diag(X)) == n / 2,
        t >= 0
    ]

    for lag, rows, cols in A_list:
        if len(rows) == 0:
            continue
        A = np.zeros((n, n))
        A[rows, cols] = 1.0
        # Add symmetry: each (i,j) and (j,i) gets 1
        # Trace(A X) >= n/2 - t/dx
        constraints.append(cp.trace(A @ X) >= n/2 - t/dx)

    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=10000)
        # print(f"SDP solve time: {time.time() - start_time:.2f}s")
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return X.value, t.value, True
        else:
            print(f"SDP did not converge. Status: {prob.status}")
            return None, None, False
    except Exception as e:
        print(f"SDP solver exception: {e}")
        return None, None, False

def round_sdp_solution(X_opt, n, target_sum, num_samples=2000):
    """Round the SDP solution using principal eigenvector and random samples."""
    # Eigendecomposition of X_opt
    lamb, V = eigh(X_opt)
    v = V[:, -1]  # largest eigenvector
    
    # Principal eigenvector rounding
    h_raw = v.copy()
    h_clip = np.clip(h_raw, 0, 1)
    h_projected = project_to_sum_box(h_clip, target_sum)
    
    best_h = h_projected.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Random rounding: sample from N(0, X_opt)
    # Use eigendecomposition: X = VΛV^T, so sample Y = V * Z where Z~N(0,I)
    for _ in range(num_samples):
        if time.time() > start_time + 550:
            break
        # Generate sample from distribution with covariance X_opt
        Z = np.random.randn(n)
        y = V @ (np.sqrt(np.abs(lamb)) * Z)
        
        h = np.clip(y, 0, 1)
        h = project_to_sum_box(h, target_sum)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
