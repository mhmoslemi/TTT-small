import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    x = np.asarray(x)
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_thue_morse(n):
    """Generate Thue-Morse sequence of length n"""
    if n == 0: return []
    seq = [0]
    while len(seq) < n:
        # Append complement of current sequence
        seq.extend(1 - x for x in seq)
    return np.array(seq[:n], dtype=float)

def generate_rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence of length n"""
    # Rudin-Shapiro sequence: a(n) = number of pairs (i,j) with i+j = n and i&j = 0
    # But for simplicity, we'll generate a ballanced one
    # Let's use a specific construction that's known to give good result
    # This is a placeholder - a more complex construction needed
    # We'll make a round-robin pick - alternating between 0, 1, 0, 1 pattern modified 
    # to achieve exactly n/2 ones
    arr = np.zeros(n)
    indices = np.random.choice(n, n//2, replace=False)
    arr[indices] = 1
    return arr

def find_worst_shift_indices(h):
    """Find the indices for the worst shift that has max overlap"""
    corr = compute_corr_fft(h)
    max_shift_idx = np.argmax(corr)
    return max_shift_idx

def find_mismatched_indices(h, shift):
    """Find indices where (h_i, h_{i+shift}) = (1,0) and (h_j, h_{j+shift}) = (0,1)"""
    N = len(h)
    if shift >= 0:
        h1 = h[:N-shift]
        h2 = h[shift:]
    else:
        h1 = h[-shift:]
        h2 = h[:-shift]
    
    # Find mismatches: (1,0) and (0,1)
    both_ones = (h1 == 1) & (h2 == 1)
    both_zeros = (h1 == 0) & (h2 == 0)
    mismatches = (h1 == 1) & (h2 == 0)  # Should swap with (0,1) mismatches
    correct_pairs = (h1 == 0) & (h2 == 1)
    
    masked = np.where(mismatches)[0]
    correct = np.where(correct_pairs)[0]
    return masked, correct

def swap_indices(h, i, j, shift):
    """Swap h[i] and h[j] (but only valid for the given shift)"""
    h = h.copy()
    if i < len(h) and j < len(h):
        h[i], h[j] = h[j], h[i]
    return h

def greedy_improve(h, max_iter=1000):
    """Improve by swapping in the worst-shift region"""
    h = h.copy()
    for _ in range(max_iter):
        max_shift_idx = find_worst_shift_indices(h)
        shift = max_shift_idx - (len(h) - 1)  # convert full corr index back to actual shift
        masked, correct = find_mismatched_indices(h, shift)
        if len(masked) == 0 or len(correct) == 0:
            break
        
        # find one random swap
        idx_i = np.random.choice(masked)
        idx_j = np.random.choice(correct)
        if idx_i < len(h) and idx_j < len(h):
            h[idx_i], h[idx_j] = h[idx_j], h[idx_i]
    return h

def simulated_annealing(h0, total_iter=5000, max_temp=1.0, min_temp=1e-6):
    """Run simulated annealing with temperature schedule"""
    h = h0.copy()
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # T(t) = T0 * (min_temp/T0)^(t/N)
    temp_schedule = [max_temp * (min_temp/max_temp)**(t/total_iter) for t in range(total_iter)]
    
    for t in range(total_iter):
        temp = temp_schedule[t]
        if temp < min_temp: break
        
        # propose a random swap in the array  
        i, j = np.random.randint(len(h)), np.random.randint(len(h))
        if i == j: continue
        
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # enforce sum constraint via projection if necessary
        # We'll insert a simple hack instead of full projection
        # instead of full projection, just make sure we didn't violate sum constraint
        if abs(np.sum(h_new) - TARGET_SUM) > 0.1:
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
        new_c5 = evaluate_c5(h_new)
        delta = new_c5 - current_c5
        
        if delta < 0 or np.exp(-delta / temp) > np.random.rand():
            h = h_new
            current_c5 = new_c5
            
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = None
    best_c5 = float('inf')
    
    # Generate several starting points
    start_configs = []
    
    # Thue-Morse sequence
    start_configs.append(('tm', generate_thue_morse(n_points)))
    
    # Rudin-Shapiro (lot of variation required)
    start_configs.append(('rs', generate_rudin_shapiro(n_points)))
    
    # Random binary with correct sum
    start_configs.append(('rand', np.random.choice([0.0, 1.0], n_points, p=[0.5, 0.5])))
    
    # 0,1, 0,1 ... pattern
    start_configs.append(('half', np.array([i % 2 for i in range(n_points)])))
    
    # Random start
    start_configs.append(('random', np.random.rand(n_points)))
    
    # Initial sweep over all start configurations
    for name, init_h in start_configs:
        if time.time() - start_time > budget_s - 5: break
        # Project to correct sum
        init_h = project_to_sum_box(init_h, TARGET_SUM)
        
        # Run greedy improve
        improved_h = greedy_improve(init_h, max_iter=500)
        
        # Check the C5 value now
        improved_c5 = evaluate_c5(improved_h)
        
        if improved_c5 < best_c5:
            best_h = improved_h
            best_c5 = improved_c5
    
    # Now apply simulated annealing on the best config found so far
    if best_h is not None and time.time() - start_time < budget_s - 2:
        anneal_iter = min(2000, int((budget_s - (time.time() - start_time)) * 5))
        if anneal_iter > 50:
            better_h, better_c5 = simulated_annealing(best_h, total_iter=anneal_iter)
            if better_c5 < best_c5:
                best_h = better_h.copy()
                best_c5 = better_c5
    
    # Final numerical refinement
    if best_h is not None and (time.time() - start_time) < budget_s - 1:
        # Convert to float, keep represented as [0,1] values
        best_h = project_to_sum_box(best_h, TARGET_SUM)
        
        # Constraint optimization: use trust-constr with bounds policy
        bounds = [(0, 1) for _ in range(n_points)]
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        def objective(h):
            return evaluate_c5(h)
        
        try:
            res = minimize(
                objective, 
                best_h, 
                method='trust-constr', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 200, 'gtol': 1e-12}
            )
            if res.success:
                final_h = res.x
                final_c5 = evaluate_c5(final_h)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except Exception:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
