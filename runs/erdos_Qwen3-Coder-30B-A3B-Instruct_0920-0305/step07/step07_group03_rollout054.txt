# Global variables
TARGET_SUM = None

dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def stochastic_local_search(h0, max_iter=1500):
    """Stochastic local search with adaptive step sizes and better diversification"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    # Define parameters for adaptive search
    step_sizes = [0.01, 0.005, 0.001, 0.0005]
    step_idx = 0
    
    for i in range(max_iter):
        # Use larger steps at beginning, smaller ones later
        current_step = step_sizes[min(step_idx, len(step_sizes)-1)]
        
        # Select random indices to modify
        indices_to_modify = np.random.choice(n_points, size=max(1, n_points//20), replace=False)
        
        improved = False
        for idx in indices_to_modify:
            # Try multiple small changes
            delta = np.random.choice([-current_step, -current_step/2, current_step/2, current_step])
            h_new = h.copy()
            h_new[idx] += delta
            
            if h_new[idx] < 0 or h_new[idx] > 1:
                continue
                
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            new_c5 = evaluate_c5(h_new)
            
            if new_c5 < best_c5:
                h = h_new.copy()
                best_c5 = new_c5
                best_h = h.copy()
                improved = True
            else:
                # Accept with low probability to escape local optima
                if np.random.random() < 0.01:
                    h = h_new.copy()
                    best_c5 = new_c5
                    improved = True
                    
        if not improved:
            # Occasionally restart from a better solution or use small global changes
            if np.random.random() < 0.02:
                h = project_to_sum_box(h + np.random.normal(0, 0.005, n_points), TARGET_SUM)
        else:
            step_idx = max(0, step_idx - 1)  # Decrease step size if we improved
            
        if i % 100 == 0:
            step_idx = min(step_idx + 1, len(step_sizes) - 1)  # Increase step size gradually
            
    return best_h, best_c5
