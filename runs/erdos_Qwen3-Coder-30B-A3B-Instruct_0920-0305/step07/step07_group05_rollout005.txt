import numpy as np

from scipy.signal import correlate
from cvxpy import Variable, Minimize, Problem, diag, sum as cp_sum, trace, PSD, norm

# Static constants
n_points = 140
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def create_A_matrices():
    """Create all sparse A_ell matrices for lags in -n+1 to n-1"""
    A_list = []
    for l in range(-(n_points - 1), n_points):
        # Create sparse matrix A_l where A_{i,i+l}=1 for valid indices
        rows, cols = [], []
        for i in range(n_points):
            j = i + l
            if 0 <= j < n_points:
                rows.append(i)
                cols.append(j)
        A_list.append((l, rows, cols))
    return A_list

def solve_sdp(A_list):
    """Solve SDP relaxation to get a lower bound"""
    # Create variable
    X = Variable((n_points, n_points), PSD=True)
    t = Variable()
    
    # Build constraints
    constraints = [0 <= diag(X), diag(X) <= 1, cp_sum(diag(X)) == TARGET_SUM]
    
    # Add the trace constraints
    for l, rows, cols in A_list:
        # Build sparse matrix A_l
        A_l = np.zeros((n_points, n_points))
        A_l[rows, cols] = 1.0
        if l != 0:
            A_l[cols, rows] = 1.0
        constraints.append(trace(A_l @ X) >= TARGET_SUM - t/dx)
    
    # Objective
    objective = Minimize(t)
    
    # Form and solve the problem
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000, eps=1e-5)
        if prob.status == 'optimal':
            return X.value, t.value
        else:
            print(f"SDP solver status: {prob.status}")
            return None, None
    except Exception as e:
        print(f"SDP solver error: {e}")
        return None, None

def round_and_refine(X_opt, num_samples=2000):
    """Round SDP solution to feasible vector via sampling"""
    best_h = None
    best_c5 = np.inf
    
    # Sample from the distribution defined by X_opt
    # For improvement: get eigenvalue decomposition of X_opt
    try:
        # Decompose X_opt and sample from its eigenvectors
        eigvals, eigvecs = eigh(X_opt)
        # Sort in descending order
        idx = np.argsort(eigvals)[::-1]
        eigvals = eigvals[idx]
        eigvecs = eigvecs[:, idx]
        
        # Keep only meaningful eigenvalues
        # Consider only significant eigenvalues for sampling
        eps = 1e-8
        valid_eig = eigvals > eps
        if np.sum(valid_eig) == 0:
            # If no valid eigenvalues, try using the largest one
            valid_eig = [True] * len(eigvals)
        
        # If rank is 1, just take the first eigenvector
        if np.sum(valid_eig) == 1:
            v = eigvecs[:, 0]
            h = v / np.sqrt(np.sum(v**2))  # normalize
            h = project_to_sum_box(h, TARGET_SUM)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # Sample M vectors y ~ N(0, X_opt) and then smooth the result
            for _ in range(num_samples):
                # Generate a sample using eigendecomposition
                L = eigvecs @ np.diag(np.sqrt(np.maximum(eigvals, 0)))
                y = np.random.randn(n_points)
                sample = L @ y
                
                # Create a candidate vector
                h = sample.copy()
                h = project_to_sum_box(h, TARGET_SUM)
                if not np.allclose(h, 0):
                    c5 = evaluate_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
    except Exception as e:
        print(f"Error during rounding and refinement: {e}")
        # Fallback to principal eigenvector if sampling fails
        try:
            w, v = eigh(X_opt)
            v = v[:, -1]  # take the largest eigenvalue
            h = project_to_sum_box(v, TARGET_SUM)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        except:
            pass
    
    return best_h, best_c5

def local_refinement(h0, max_iter=150):

    for iter_num in range(max_iter):
        # Compute approximate gradient with finite diff
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_sum_box(h_plus, TARGET_SUM)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_sum_box(h_minus, TARGET_SUM)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Move mass from low gradient to high gradient
        idx_min = np.argmin(grad)
        idx_max = np.argmax(grad)
        
        if grad[idx_min] < grad[idx_max]:
            delta = min(h[idx_min], 1 - h[idx_max], 0.01)
            if delta > 1e-6:
                h[idx_min] -= delta
                h[idx_max] += delta
                h = project_to_sum_box(h, TARGET_SUM)
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        else:
            # Random small move
            i = np.random.choice(n_points)
            j = np.random.choice(n_points)
            if i != j:
                delta = min(h[i], 1 - h[j], 0.005)
                if delta > 1e-6:
                    h[i] -= delta
                    h[j] += delta
                    h = project_to_sum_box(h, TARGET_SUM)
                    c5 = evaluate_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
    
    return best_h, best_c5
