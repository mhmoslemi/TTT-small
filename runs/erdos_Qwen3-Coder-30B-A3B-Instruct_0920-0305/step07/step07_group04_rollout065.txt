import numpy as np

import time

import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 150
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT for performance"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_relaxation_lp(n):
    """Solve the LP relaxation using cvxpy"""
    # Create variables
    h = cp.Variable(n)
    p = cp.Variable((n,n), symmetric=True)
    
    # Constraints
    constraints = [
        0 <= h,
        h <= 1,
        cp.sum(h) == n / 2,
    ]

    # McCormick constraints for p_ij
    for i in range(n):
        for j in range(i, n):
            constraints.extend([
                p[i, j] >= 0,
                p[i, j] >= h[i] + h[j] - 1,
                p[i, j] <= h[i],
                p[i, j] <= h[j],
            ])

    # Shift constraints
    t = cp.Variable()
    for k in range(1, n):  # k ranges from 1 to n-1
        # Compute overlap on shift k
        o = 0
        for i in range(n - k):
            o += p[i, i+k]
        constraints.append(o >= n/2 - t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    
    # Try multiple solvers
    solvers = ['ECOS', 'CBC', 'GLPK_MI']
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False, max_iters=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_ERR]:
                break
        except:
            continue

    if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_ERR]:
        return h.value
    else:
        # Fall back to numerical solution
        return None

def refine_with_subgradient(h0, max_iter=2000):
    """Refine h using projected sub-gradient descent"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    alpha = 0.5 * dx
    decay_rate = 0.995
    
    for i in range(max_iter):
        # Find active shift
        corr = compute_corr_fft(h)
        k_star = np.argmax(corr)
        
        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j = (i + k_star) % len(h)
            grad[i] = 1 - h[j]
            if i >= k_star:
                grad[i] -= h[i - k_star]
        
        # Update
        h_new = h - alpha * grad
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        
        # Project
        h = project_to_sum_box(h_new, TARGET_SUM)
        
        # Annealing-like step decay
        alpha *= decay_rate
    
    return best_h
