# Global variables
TARGET_SUM = None
n_points = 150
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def steepest_descent_refinement(h0, max_iter=2000):
    """Refine h0 using projected sub-gradient descent with a schedule"""
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    t_start = time.time()
    for i in range(max_iter):
        if time.time() - t_start > 200:
            break
            
        # Find shift k* that maximizes overlap
        h_rev = h[::-1]
        b = 1 - h
        L = 2 * len(h) - 1
        corr_raw = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
        k_star = np.argmax(corr_raw[:L-1]) - len(h) + 1  # adjust index properly

        # Compute sub-gradient for the active shift k_star
        g = np.zeros_like(h)
        for i in range(len(h)):
            if i + k_star >= 0 and i + k_star < len(h):
                g[i] = (1 - h[i + k_star])
            else:
                g[i] = 0
            if i - k_star >= 0 and i - k_star < len(h):
                g[i] -= h[i - k_star]
            else:
                g[i] -= 0
                
        # Take step
        step_size = 0.5 * dx * np.exp(-0.001 * i)
        h -= step_size * g
        
        # Project back to [0,1]
        h = np.clip(h, 0, 1)
        
        # Project into correct mass
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Evaluate and update
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
    return best_h, best_c5
