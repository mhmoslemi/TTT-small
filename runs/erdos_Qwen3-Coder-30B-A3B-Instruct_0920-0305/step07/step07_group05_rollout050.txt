import numpy as np
import time

# Global variables
n_points = None
dx = None
TARGET_SUM = None

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    global dx
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    x = np.asarray(x, dtype=np.float64)
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0.0, 1.0)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(60):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0.0, 1.0)
    return result

def compute_gradient(h, lag):
    """Compute gradient of correlation at lag for current h"""
    n = len(h)
    grad = np.ones(n)
    if lag >= 0:
        # lag = int(lag)  # already integer
        if lag < n:
            grad[:n-lag] -= h[lag:]
            grad[lag:] -= h[:n-lag]
    else:
        lag = -lag
        if lag < n:
            grad[lag:] -= h[:n-lag]
            grad[:lag] -= h[lag:]
    return grad

    c5 = evaluate_c5(h)

    corr = np.correlate(h, 1 - h, mode='full')

    # parameters
    max_delta = 0.5  # Start with larger delta
    min_delta = 1e-5
    increment_factor = 0.9995
    
    start = time.time()
    iteration = 0
    
    while True:
        now = time.time()
        if now - start > time_limit:
            break
        iteration += 1
        
        # shrink delta
        max_delta = max(min_delta, max_delta * increment_factor)
        
        # compute gradient for current max lag
        grad = compute_gradient(h, lag_max)
        
        # candidate sources and targets
        src_candidates = np.where((grad > 1e-8) & (h > 1e-12))[0]
        tgt_candidates = np.where((grad < -1e-8) & (h < 1 - 1e-12))[0]
        
        if src_candidates.size > 0 and tgt_candidates.size > 0:
            # Find source and target candidates
            src = src_candidates[np.argmax(grad[src_candidates])]
            tgt = tgt_candidates[np.argmin(grad[tgt_candidates])]
            
            delta_allowed = min(h[src], 1.0 - h[tgt])
            if delta_allowed > 1e-12:
                delta = max_delta * np.random.rand()
                delta = min(delta, delta_allowed)
                
                # apply move
                h[src] -= delta
                h[tgt] += delta
                
                c5_new = evaluate_c5(h)
                if c5_new < c5:
                    c5 = c5_new
                    # recompute correlation and lag_max
                    corr = np.correlate(h, 1 - h, mode='full')
                    lag_max = int(np.argmax(corr) - (n - 1))
                else:
                    # revert
                    h[src] += delta
                    h[tgt] -= delta
        else:
            # no greedy move possible; do random move
            i, j = np.random.choice(n, 2, replace=False)
            if h[i] > 1e-12 and h[j] < 1 - 1e-12:
                delta_allowed = min(h[i], 1.0 - h[j])
                delta = max_delta * np.random.rand()
                delta = min(delta, delta_allowed)
                
                h[i] -= delta
                h[j] += delta
                
                c5_new = evaluate_c5(h)
                if c5_new < c5:
                    c5 = c5_new
                    corr = np.correlate(h, 1 - h, mode='full')
                    lag_max = int(np.argmax(corr) - (n - 1))
                else:
                    # revert
                    h[i] += delta
                    h[j] -= delta
        
        # occasionally do a random perturbation to escape local minima
        if iteration % 500 == 0:
            # Random perturb of few entries
            for _ in range(3):
                i, j = np.random.choice(n, 2, replace=False)
                if h[i] > 1e-12 and h[j] < 1 - 1e-12:
                    # transfer a small fraction
                    delta = max_delta * 0.1 * np.random.rand()
                    delta = min(delta, h[i], 1.0 - h[j])
                    h[i] -= delta
                    h[j] += delta
            c5_new = evaluate_c5(h)
            if c5_new < c5:
                c5 = c5_new
                corr = np.correlate(h, 1 - h, mode='full')
                lag_max = int(np.argmax(corr) - (n - 1))
        
        # maintain sum sum via small corrections
        if abs(np.sum(h) - TARGET_SUM) > 1e-8:
            h = project_to_sum_box(h, TARGET_SUM)
            c5 = evaluate_c5(h)
            corr = np.correlate(h, 1 - h, mode='full')
            lag_max = int(np.argmax(corr) - (n - 1))
    
    return h, c5

def random_binary(n):
    """Generate random binary vector with exactly TARGET_SUM ones"""
    ones = np.zeros(n)
    idx = np.random.choice(n, int(TARGET_SUM), replace=False)
    ones[idx] = 1.0
    return ones
