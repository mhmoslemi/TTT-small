import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def adaptive_pattern_search(h0, max_iter=1000):
    """Improved pattern search with adaptive steps"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    params = {"window": 10, "step_size": 0.01, "n_trials": 10}
    
    for i in range(max_iter):
        if i % 100 == 0:
            corr_full = compute_corr_fft(h)
            max_corr_idx = np.argmax(corr_full)
        
        # Try to target the highest correlation region with increased intensity
        start = max(0, max_corr_idx - params["window"])
        end = min(n_points, max_corr_idx + params["window"])
        
        # Try various modifications to this region
        for _ in range(params["n_trials"]):
            # Select a random sub-region
            sub_start = np.random.randint(start, min(end, n_points - 5))
            sub_end = min(sub_start + 5, n_points)

            mod_strength = np.random.choice([0.005, 0.01, 0.02])

def local_gradient_refinement(h0, max_iter=300):
    """Refine with gradient-based optimization using L-BFGS-B"""
    h = h0.copy()
    
    def objective(h_local):
        return evaluate_c5(h_local)
    
    def constraint_eq(h_local):
        return np.sum(h_local) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use bounds for variables
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(
            objective, 
            h, 
            method='L-BFGS-B', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
    except:
        pass
    
    # Project to maintain constraints
    h = project_to_sum_box(h, TARGET_SUM)
    return h
