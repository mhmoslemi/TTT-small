import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

# Global variables
n_points = 150
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Evaluate C5 using FFT."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_convex_relaxation(h0):
    """Solve convex relaxation via LP with McCormick envelopes."""
    h = cp.Variable(n_points, name="h")
    p = cp.Variable((n_points, n_points), name="p")
    T = cp.Variable(nonneg=True, name="T")
    
    # Constraints on h
    constraints = [0 <= h, h <= 1, cp.sum(h) == TARGET_SUM]
    
    # McCormick envelopes
    for i in range(n_points):
        for j in range(i, n_points):
            constraints += [
                p[i, j] >= 0,
                p[i, j] >= h[i] + h[j] - 1,
                p[i, j] <= h[i],
                p[i, j] <= h[j],
            ]
    
    # Shift constraints
    # Compute all valid overlaps based on shift k
    for shift in range(-(n_points - 1), n_points):
        if shift >= 0:
            # Overlap term O_k = sum_{i=0}^{n-1-shift} p[i, i+shift]
            if shift == 0:
                O = cp.sum(h)  # all h_i * 1
            else:
                # Convex relaxation of sum_{i=0}^{n-1-shift} p[i, i+shift]
                O = cp.sum([p[i, i + shift] for i in range(n_points - shift)])
            constraints.append(O >= TARGET_SUM/2 - T)
    
    # Objective: minimize T
    problem = cp.Problem(cp.Minimize(T), constraints)
    problem.solve(solver=cp.GLPK, timelimit=180)
    
    if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return h.value
    else:
        return h0

def binary_search_lp(h_start, max_iter=50):
    """Binary search on C5 bound using LP relaxation and exact evaluation."""
    h_best = h_start.copy()
    best_c5 = evaluate_c5(h_best)

    # Bound interval
    T_low = 0.0
    T_high = TARGET_SUM
    
    tol = 1e-4 * dx
    for _ in range(max_iter):
        if T_high - T_low < tol:
            break
        T = (T_low + T_high) / 2
        h = solve_convex_relaxation(h_start)
        # Again solve with the specific T value (both at same time cannot be done)
        # This is a feasibility test. Be compliant or just solve with T_fixed.
        # We'll do a multistart optimization to treat it as a residual
        # Simpler approach: just evaluate and see.
        c5_eval = evaluate_c5(h)
        if c5_eval <= T * dx:
            best_c5 = min(best_c5, c5_eval)
            h_best = h.copy()
            T_high = T
        else:
            T_low = T
            
    return h_best, best_c5

def subgradient_polish(h0, max_iter=2000):
    """Deterministic subgradient-based polishing."""
    h = h0.copy()
    lr = 0.01
    decay = 0.995
    
    for t in range(max_iter):
        # Compute correlation to find active shift
        corr = correlate(h, 1 - h, mode="full")
        k_star = np.argmax(corr)
        # Adjust k to be within [-n+1, n-1]
        if k_star > n_points - 1:
            k_star = k_star - (2 * n_points - 2)
        else:
            k_star = k_star - (n_points - 1)

        # Compute subgradient
        g = np.zeros_like(h)
        for i in range(n_points):
            if i + k_star >= 0 and i + k_star < n_points:
                g[i] += (1 - h[i + k_star])
            if i - k_star >= 0 and i - k_star < n_points:
                g[i] -= h[i - k_star]

        # Step
        lr *= decay
        h = h - lr * g

        h = project_to_sum_box(h, TARGET_SUM)
        
        # Early break if improvement stops or stalls
        if t % 100 == 0:
            if t > 0:
                c5 = evaluate_c5(h)
                if c5 > 0.99 * best_c5:
                    break

    return h

def local_exchange_refinement(h0, max_iter=5000):
    """Local pairwise mass transfers."""
    h = h0.copy()
    c5 = evaluate_c5(h)
    
    for _ in range(max_iter):
        i, j = np.random.choice(n_points, 2, replace=False)
        if i > j:
            i, j = j, i
        
        delta_max = min(h[j], 1 - h[i])
        if delta_max == 0:
            continue
            
        delta = np.random.choice([0.1, 0.5, 0.9]) * delta_max
        if delta == 0:
            continue
            
        h_prev = h.copy()
        h[i] += delta
        h[j] -= delta
        h = project_to_sum_box(h, TARGET_SUM)
        c5_new = evaluate_c5(h)
        if c5_new < c5:
            c5 = c5_new
        else:
            h = h_prev
    return h
