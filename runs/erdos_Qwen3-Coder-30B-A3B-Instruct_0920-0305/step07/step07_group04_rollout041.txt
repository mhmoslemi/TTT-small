```python
import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    # Variables
    h = cp.Variable(n_points, name="h")
    P = cp.Variable((n_points, n_points), name="P")
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == TARGET_SUM
    ]
    
    # McCormick envelopes for all pairs
    for i in range(n_points):
        for j in range(n_points):
            constraints += [
                P[i, j] >= 0,
                P[i, j] <= h[i],
                P[i, j] <= h[j],
                P[i, j] >= h[i] + h[j] - 1
            ]
    
    # Compute correlations for all shifts
    max_corr = cp.Variable(name="max_corr")
    
    # For each shift k from -n+1 to n-1
    # But we only consider shifts where overlap exists
    correlations = []
    for k in range(-n_points + 1, n_points):
        # Region where both indices exist: [0, n-1] intersected with [k, n-1+k] 
        # This is [max(0,k), min(n-1, n-1+k)] = [max(0,k), min(n-1, n-1+k)]
        # But also for k<0, need min(n-1, n-1+k) = n-1 (assuming k >= -n+1)
        # For k>0, need max(0,k) = k
        # So overlap region for shift k:
        if k >= 0:
            start = k
            end = n_points
        else:
            start = 0
            end = n_points + k
        
        if start >= end:  # No overlap case
            corr = 0
        else:
            corr = cp.sum(h[start:end]) - cp.sum(P[range(start, end), range(start+k, end+k)])
        
        correlations.append(corr)
    
    # For better numerical performance, tie the first correlations to the maximum
    for i, corr in enumerate(correlations):
        constraints.append(max_corr >= corr)
        
    # Objective: minimize maximum correlation
    objective = cp.Minimize(max_corr)
    
    # Formulate and solve the problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CBC, max_seconds=budget_s-10)
    except:
        # If CBC fails, fall back to default solver or early return
        pass
    
    # Extract solution
    h_solution = h.value
    if h_solution is None:
        # Fallback to previous approach if optimization fails
        h_solution = np.zeros(n_points)
        h_solution[:n_points//2] = 1
        h_solution = h_solution / np.sum(h_solution) * TARGET_SUM
    
    # Evaluate actual C5
    max_corr_val = 0
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            start = k
            end = n_points
        else:
            start = 0
            end = n_points + k
            
        if start >= end:  # No overlap
            corr = 0
        else:
            corr = np.sum(h_solution[start:end]) - np.sum(h_solution[start:end] * h_solution[start+k:end+k])
            
        max_corr_val = max(max_corr_val, corr)
    
    C5_bound = max_corr_val * dx
    
    # Return best solution
    return (h_solution.tolist(), float(C5_bound), n_points)
```
