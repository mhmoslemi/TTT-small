import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    N = 120
    dx = 2.0 / N
    target_sum = N // 2
    
    # Initialize the best solution
    best_h = None
    best_c5 = float('inf')
    
    # Solve MILP if possible
    try:
        # Create decision variables
        h = cp.Variable(N, boolean=True)
        lam = cp.Variable(integer=True)
        
        # Constraints: sum of h equals target_sum
        constraints = [cp.sum(h) == target_sum]
        
        # Declare auxiliary variables w[i,k]
        # For each shift k, we define valid indices i such that 0 <= i < N and 0 <= i+k < N
        w_vars = {}
        
        # Linear constraints for all possible shifts
        for k in range(-(N-1), N):
            # Valid indices for this shift
            start_i = max(0, -k)
            end_i = min(N, N - k)
            if end_i > start_i:
                i_vals = np.arange(start_i, end_i)
                j_vals = i_vals + k
                w_key = f"k_{k}"
                w = cp.Variable(len(i_vals), boolean=True)
                w_vars[w_key] = (w, i_vals, j_vals)
                
                # McCormick linearization constraints
                constraints += [w <= h[i_vals]]
                constraints += [w <= h[j_vals]]
                constraints += [w >= h[i_vals] + h[j_vals] - 1]
                
                # Constraint: sum of w elements >= lambda
                constraints += [cp.sum(w) >= lam]
        
        # Fix one variable to break symmetry
        constraints.append(h[0] == 1)
        
        # Objective: maximize lambda
        objective = cp.Maximize(lam)
        
        # Create problem
        prob = cp.Problem(objective, constraints)
        
        # Solve with CBC
        remaining = budget_s * 0.55
        if remaining > 0:
            try:
                prob.solve(solver=CP.SCS, max_iters=10000, verbose=False)
                h_sol = np.round(h.value).astype(int)
                lam_val = lam.value
                
                if h_sol is not None and lam_val is not None:
                    # Repair any deviation from target sum
                    current_sum = np.sum(h_sol)
                    if current_sum != target_sum:
                        # Fix by swapping bits
                        ones_idx = np.where(h_sol == 1)[0]
                        zeros_idx = np.where(h_sol == 0)[0]
                        if len(ones_idx) > target_sum:
                            # Remove excess 1s
                            excess = len(ones_idx) - target_sum
                            for i in range(excess):
                                if len(zeros_idx) > 0:
                                    h_sol[ones_idx[i]] = 0
                                    h_sol[zeros_idx[0]] = 1
                        elif len(ones_idx) < target_sum:
                            # Add missing 1s
                            missing = target_sum - len(ones_idx)
                            for i in range(missing):
                                if len(zeros_idx) > 0:
                                    h_sol[zeros_idx[i]] = 1
                                    h_sol[ones_idx[i]] = 0

                    # Compute C5
                    c5_val = 1 - (2 * lam_val) / N
                    if c5_val < best_c5:
                        best_h = h_sol
                        best_c5 = c5_val
            except:
                pass
    except:
        pass

    # Fallback to heuristic if MILP fails
    if best_h is None:
        # Use existing heuristic (or random) initialization
        initial = np.array(initial_h_values)
        best_h = np.interp(np.linspace(0, 2, N), np.linspace(0, 2, len(initial)), initial)
        best_h = np.clip(best_h + np.random.normal(0, 0.01, N), 0, 1)
        # Project to correct sum
        best_h = best_h * target_sum / np.sum(best_h)
        if np.sum(best_h) > target_sum:
            # Scale down a bit
            best_h *= target_sum / np.sum(best_h)
        else:
            # Scale up a bit
            best_h *= target_sum / np.sum(best_h)
        best_h = project_to_sum_box(best_h, target_sum)
        best_c5 = evaluate_c5_fast(best_h)

    # Post-process: greedy descent
    if time.time() - start_time < budget_s * 0.75:
        best_h, best_c5 = greedy_descent(best_h, budget_s - (time.time() - start_time))

    # Post-process: simulated annealing
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 30:
        best_h, best_c5 = anneal_binary(best_h, remaining_time)

    return (best_h.tolist(), float(best_c5), N)


# Helper functions
def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    scale_factor = target_sum / current_sum
    x = x * scale_factor
    # Ensure exact sum by small adjustments
    diff = target_sum - np.sum(x)
    idx = np.argsort(x)
    if diff > 0:
        for i in range(int(np.ceil(diff))):
            if i < len(idx) and x[idx[i]] < 1:
                x[idx[i]] = min(1, x[idx[i]] + diff/len(idx))
    elif diff < 0:
        for i in range(int(np.ceil(-diff))):
            if i < len(idx) and x[idx[i]] > 0:
                x[idx[i]] = max(0, x[idx[i]] + diff/len(idx))
    return x


def compute_corr_fft(h):
    """Compute full correlation via FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr


def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / len(h))


def greedy_descent(h0, remaining_time=300):
    """Greedy descent on the worst shift"""
    h = h0.copy()
    start_t = time.time()
    
    while time.time() - start_t < remaining_time:
        corr = compute_corr_fft(h)
        k_worst = np.argmax(corr)
        # This gives the match counts for various shifts
        # For the current shift k_worst where we compute corr[k_worst],
        # correlate h[0:119] with 1-h[k_worst:119+k_worst] (if k_worst >= 0)
        # or diffuse it accordingly

        # Let's compute overlap in a more explicit way
        # correlation corresponds to h[i]*h[i+k] but we want h[i](1-h[i+k])
        
        # The overlap C5 = max_k integral h*(1-h[k]) = max_k (sum of h[i] - sum of h[i]*h[i+k])
        # So, we're trying to minimize max_k (sum of h[i] - Σ h[i]*h[i+k])
        # and we know sum(h) is fixed
        
        # Since we know a worst shift k_worst:
        # let's scan the possible swaps on h[j]*h[j-k_worst] where
        # one is 1 and one is 0
        # Think of all i such that h[i] = 1, h[i+k] = 0 for the worst k
        # and all i such that h[i] = 0, h[i+k] = 1 for the worst k
        
        # However for small N, we could just recompute the max correlation
        ilist = np.arange(len(h))
        shift = k_worst - len(h) + 1  # corresponds to the right offset in indexing for reverse correlation
        
        # Compute the two mismatched sets for the worst shift
        A = np.where((h != 0) & (np.roll(h, shift) == 0))[0]
        B = np.where((h == 0) & (np.roll(h, shift) != 0))[0]
        if len(A) > 0 and len(B) > 0:
            # Swap a random element of A with a random element of B
            a = np.random.choice(A)
            b = np.random.choice(B)
            
            h[a], h[b] = h[b], h[a]
        else:
            break
    return h, evaluate_c5_fast(h)


def anneal_binary(h0, remaining_time=300):
    """Fast simulated annealing on binary vector"""
    h = h0.copy()
    T0 = 0.02
    T = T0
    T_min = 1e-6
    alpha = 0.995
    
    best_h = h.copy()
    best_c5 = evaluate_c5_fast(h)
    
    start_t = time.time()
    
    while time.time() - start_t < remaining_time and T > T_min:
        # Pick random i, j with different values
        i = np.random.randint(0, len(h))
        j = np.random.randint(0, len(h))
        
        if i == j:
            continue
            
        # Try to swap h[i] and h[j]
        old_h = h.copy()
        old_c5 = evaluate_c5_fast(h)
        
        if h[i] == h[j]:
            continue  # entropy not affected
            
        h[i], h[j] = h[j], h[i]
        
        new_c5 = evaluate_c5_fast(h)
        
        delta = new_c5 - old_c5
        
        # Accept or reject
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            # Accept
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Reject
            h[:] = old_h[:]
            
        # Cool down
        T *= alpha
    
    return best_h, best_c5
