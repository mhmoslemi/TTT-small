import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_with_gradients(h0, max_iter=1000):
    """Optimize using gradient-based method with projection"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def local_search_improve(h0, max_iter=500):
    """Apply local search to improve solution"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Try to improve by making small changes to a subset of points
        idxs = np.random.choice(n_points, size=min(30, n_points//3), replace=False)
        improved = False
        
        for idx in idxs:
            old_val = h[idx]
            # Try different small changes
            deltas = [-0.03, -0.02, -0.01, 0.01, 0.02, 0.03]
            for delta in deltas:
                h_new = h.copy()
                h_new[idx] += delta
                
                if h_new[idx] < 0 or h_new[idx] > 1:
                    continue
                    
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                new_c5 = evaluate_c5(h_new)
                
                if new_c5 < best_c5:
                    h = h_new.copy()
                    best_c5 = new_c5
                    improved = True
                    break
            if improved:
                break
        
        # If no improvement, make random perturbations
        if not improved:
            if np.random.random() < 0.3:

                h[idx] += delta
                h = project_to_sum_box(h, TARGET_SUM)
                best_c5 = evaluate_c5(h)
    
    return h

def add_peaks(h, peak_positions=None, peak_heights=None):
    """Add peaks to enhance overlap suppression"""
    h_new = h.copy()
    
    if peak_positions is None:
        # Select random positions for peaks
        peak_positions = np.random.choice(n_points, size=3, replace=False)
    
    if peak_heights is None:
        peak_heights = [0.8, 0.7, 0.9]
    
    for pos, height in zip(peak_positions, peak_heights):
        h_new[pos] = max(h_new[pos], height)
        # Apply smoothing effect
        for offset in [-2, -1, 0, 1, 2]:
            if 0 <= pos + offset < n_points:
                h_new[pos + offset] = max(0, h_new[pos + offset] - 0.1)
    
    return project_to_sum_box(h_new, TARGET_SUM)

    phase2_end = start_time + 0.7 * budget_s

            best_h = h_optimized.copy()
            best_c5 = c5_optimized
