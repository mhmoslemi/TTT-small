```python
import numpy as np
from scipy.signal import correlate
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, Maximum, sum_entries, norm
import cvxpy as cp
import time
from scipy.optimize import minimize

n_points = 150
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def solve_lp_relaxation():
    """Solve the LP relaxation via McCormick envelopes."""
    h = Variable(n_points, bounds=(0, 1))
    p = Variable((n_points, n_points), symmetric=True)
    t = Variable()

    # Sum constraint
    constraints = [cp_sum(h) == TARGET_SUM]

    # McCormick envelopes
    for i in range(n_points):
        for j in range(i, n_points):
            constraints += [
                p[i, j] >= 0,
                p[i, j] >= h[i] + h[j] - 1,
                p[i, j] <= h[i],
                p[i, j] <= h[j]
            ]

    # Overlap constraints
    # For all shifts k from -n+1 to n-1, the overlap for shift k is sum over valid i of p[i, i+k]
    shifts = range(-n_points+1, n_points)
    for k in shifts:
        if k >= 0:
            overlap = cp_sum([p[i, i+k] for i in range(n_points - k)])
        else:
            overlap = cp_sum([p[i-k, i] for i in range(-k, n_points)])
        # R_k = (n/2) - O_k; max R_k <= t means O_k >= (n/2) - t
        constraints += [overlap >= TARGET_SUM - t]

    objective = Minimize(t)
    prob = Problem(objective, constraints)

    # Try fastest solvers
    solver = 'ECOS'
    try:
        prob.solve(solver=solver, verbose=False, max_iters=10000, feastol=1e-8, abstol=1e-8)
    except:
        solver = 'CBC'
        try:
            prob.solve(solver=solver, verbose=False, timelimit=100)
        except:
            solver = 'GLPK'
            try:
                prob.solve(solver=solver, verbose=False, tmlim=100)
            except:
                return None  # Fallback if LP fails completely

    if prob.status in ['optimal', 'optimal_inaccurate']:
        return h.value
    else:
        return None

def compute_c5(h):
    """Return C5 = max_k ∫ h(x) (1 - h(x+k)) dx"""
    corr = correlate(h, 1-h, mode="full")
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto sum = target_sum and box [0,1]"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def projected_subgrad_descent(h0, max_iter=2000):
    """Loosely based on projected subgradient method for C5 minimization"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for i in range(max_iter):
        # Find active shift for current h
        corr = correlate(h, 1-h, mode="full")
        k_active = np.argmax(corr)
        # The 1D shift index of the max
        shift = k_active - (len(h) - 1)

        # Compute gradient for this shift
        g = np.zeros_like(h)
        for i in range(len(h)):
            j_i = i + shift
            if 0 <= j_i < len(h):
                g[i] += (1 - h[j_i])
            j_i = i - shift
            if 0 <= j_i < len(h):
                g[i] -= h[j_i]

        alpha = 0.5 * dx
        h_new = h - alpha * g
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

        h = h_new

    return best_h, best_c5

def evaluate_c5_fast(h):
    """Compute C5 using fast FFT correlation."""
    corr = correlate(h, 1-h, mode="full")
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()

    # Initialize
    np.random.seed(seed)

    # Step 1: Solve the LP relaxation to get a strong initial vector
    h_initial = solve_lp_relaxation()
    if h_initial is None:
        h_initial = np.ones(n_points) * 0.5
        h_initial = project_to_sum_box(h_initial, TARGET_SUM)
    
    h_final = h_initial.copy()
    c5_best = evaluate_c5_fast(h_final)

    # Step 2: Local refinement with projected subgradient descent
    if time.time() - start_time < budget_s - 10:
        h_refined, c5_refined = projected_subgrad_descent(h_initial, max_iter=1500)
        if c5_refined < c5_best:
            h_final = h_refined
            c5_best = c5_refined

    # Final validation
    c5_final = evaluate_c5_fast(h_final)

    return (h_final.tolist(), float(c5_final), n_points)
```
