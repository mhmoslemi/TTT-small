def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * 2.0 / len(h)

def project_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_gradient(h, corr, beta):
    """Compute gradient of smooth max surrogate"""
    # Subtract max for numerical stability
    corr_stable = corr - np.max(corr)
    exp_corr = np.exp(beta * corr_stable)
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient of smooth max
    grad = np.zeros_like(h)
    n = len(h)
    
    for ell in range(-n+1, n):
        if ell >= 0:
            # Forward shift
            shifted = np.concatenate([h[ell:], np.zeros(ell)])
        else:
            # Backward shift
            shifted = np.concatenate([np.zeros(-ell), h[:ell]])
        
        # Avoid computing for indices that are out of bounds
        offset = max(0, ell)
        length = len(h) - abs(ell)
        if ell >= 0:
            grad[:length] += weights[ell + n - 1] * (1 - shifted[:length])
        else:
            grad[-length:] += weights[ell + n - 1] * (1 - shifted[-length:])
    
    return 2.0 / n * (1 - np.sum(h) / (n/2) - grad)

def max_lag_gradient(h, corr):
    """Compute exact sub-gradient of max-over-shifts using the worst lag"""
    max_lag_idx = np.argmax(corr)
    n = len(h)
    grad = np.zeros_like(h)
    
    # Add contribution from h[i + max_lag_idx]
    if max_lag_idx >= 0:
        # Forward shift case
        for i in range(n - max_lag_idx):
            grad[i] += (1 - h[i + max_lag_idx]) * 2.0 / n
    else:
        # Backward shift case
        for i in range(-max_lag_idx, n):
            grad[i] -= h[i + max_lag_idx] * 2.0 / n
    
    return grad

def mass_transfer(h, grad, epsilon=0.01):
    """Perform mass transfer from decreasing entries to increasing entries"""
    d = -grad
    sources = np.where(d < -1e-10)[0]
    sinks = np.where(d > 1e-10)[0]
    
    # Sort by decreasing importance
    sources = sources[np.argsort(-d[sources])]
    sinks = sinks[np.argsort(d[sinks])]
    
    h_new = h.copy()
    total_transfer = 0
    
    i, j = 0, 0
    while i < len(sources) and j < len(sinks):
        source = sources[i]
        sink = sinks[j]
        
        max_remove = h_new[source]
        max_add = 1 - h_new[sink]
        transfer = min(max_remove, max_add, epsilon)
        
        if transfer > 1e-10:
            h_new[source] -= transfer
            h_new[sink] += transfer
            total_transfer += transfer
            
            if h_new[source] < 1e-10:
                i += 1
            if h_new[sink] > 1 - 1e-10:
                j += 1
        else:
            i += 1
            j += 1
            
    return project_sum_box(h_new, len(h) / 2)

def backtracking_line_search(h, grad, corr, beta):
    """Perform backtracking line search for sufficient descent"""
    alpha = 0.2
    alpha_min = 5e-4
    cmax = np.max(corr)
    
    while alpha >= alpha_min:
        h_trial = h - alpha * grad
        h_trial = project_sum_box(h_trial, len(h) / 2)
        corr_trial = correlate(h_trial, 1 - h_trial, mode='full')
        cmax_trial = np.max(corr_trial) * 2.0 / len(h)
        
        if cmax_trial < cmax:
            return h_trial, corr_trial, cmax_trial
        
        alpha /= 2
    
    return h, corr, cmax
