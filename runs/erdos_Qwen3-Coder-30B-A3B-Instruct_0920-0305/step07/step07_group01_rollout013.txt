import numpy as np
from scipy.signal import correlate
import time

# Global variables for the optimization
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
start_time = None

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_gradient(h, corr, beta):
    """Compute gradient of smooth max surrogate"""
    # Add small value to avoid numerical overflow
    corr_shifted = corr - np.max(corr)
    exp_corr = np.exp(beta * corr_shifted)
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute the gradient of the smooth max with respect to h
    grad = np.zeros_like(h)
    # Full correlation result has length 2*N-1, we need to handle indices properly
    for i in range(len(h)):
        for lag in range(-len(h) + 1, len(h)):
            if 0 <= i + lag < len(h):
                # For lag i, if h[i] appears in the shift, we're considering it as an element
                # of the correlation interface including element at i+lag
                # When computing full correlation with h and (1-h), we consider:
                # correlation at lag l = sum h[i] * (1 - h[i+l])
                # So for element at position i, we contribute to correlation:
                # - when it has index k and l = k - i; then h[k] contributes to correlation with coefficient (1 - h[i+l])
                # But we want to compute gradient with respect to h[i], so ...
                # For any lag l, we're contributing to the term:
                # h[i] * (1 - h[i+l]) if i+l exists in h and is valid
                # So we contribute to term (1-h[i+l]) for position i-l
                # For a fixed i, let's compute contribution to d/dh[i] :
                # gradient = sum_{lag} weight[lag] * (d/dh[i]) [h[i] * (1 - h[i+lag])]
                #          = [(1 - h[i+lag]) * w[lag] for lag s.t. i+lag is valid]
                #          = sum_{lag} w[lag] * (1 - h[i+lag]) if 0 <= i+lag < N else 0
                
                # We compute correlation using scipy, so we work with full correlation vector.
                # In the full correlation of h and (1-h), the element corr[lag + len(h)-1] corresponds
                # to the correlation with lag = lag. 
                # Let's compute the contribution properly:
                try:
                    actual_lag = lag
                    corr_ind = actual_lag + len(h) - 1  # Index in full correlation vector
                    if 0 <= corr_ind < len(corr):
                        # If this is an outbound index, it contributes
                        # gradient contribution to h[i] = sum of: 
                        # if i+lag >= 0, contribution += w[lag] * (1 - h[i + lag]) 
                        # when 0 <= i + lag < N, and we're in the lag window we're looking at
                        
                        # However, need to be more careful.
                        #цающимся коэффициентом в сумме для h[i]: 
                        if i + lag >= 0 and i + lag < len(h):
                            # the contribution in term delta_j - delta_j+lag is the gradient per delta_j 
                            # where in corr = h * (1-h), the contribution to 
                            # the gradient w.r.t. h[i] from term h[i] * (1 - h[i + lag]) is
                            # just (1 - h[i + lag]) * w[lag]
                            # But the in our expression, w[lag] is with respect to lag index l,
                            # here l = actual_lag, so we should take exp(beta * corr[lag +len(h)-1]) 
                            # i.e. corr_index + len(h) - 1 is the lag index
                            # But we are interested in correlation about an arbitrary lag.
                            # This is tricky because with "full", we get a full correlation of length 2N - 1
                            # Index 0 is lag = -(N-1), index (N-1) is lag = 0, index (2N-2) is lag = N-1
                            # If we loop over i and look at contribution in lag l = actual_lag, 
                            # it's the coefficient corresponding to w[actual_lag + N-1]
                            # But as done in typical computation of derivative of smooth maximum:
                            # gradient wrt h[i] = sum_{l in -N+1 to N-1} w[l] * (1 - h[i+l])
                            # which is a cross-term of h[i] * (1 - h[i+l])
                            # But instead we compute gradients of one term as a total derivative of sum_i h[i] * (1 - h[i+l]), 
                            # which is only one term term per derivative. 
                            # So:
                            grad[i] += weights[actual_lag + len(h) - 1] * (1 - h[i + lag])
                except IndexError:
                    # Should never happen
                    continue
    
    # Reverse direction for descent
    grad = -grad
    return grad

def max_lag_gradient(h, corr, idx_best):
    """Compute exact sub-gradient of the max (this computes gradient only where the max is attained)"""
    grad = np.zeros_like(h)
    best_lag = idx_best - len(h) + 1  # Convert index in full correlation to actual lag
    
    # The gradient is a modified form of correlation in h* (1-h)
    # specifically for the lag where max is attained. 
    # This is the gradient of just that correlation coeff.
    # High level idea: derivative of h_i * (1 - h_{i+lag}) w.r.t. h_i is (1 - h_{i+lag})
    # So for fixed lag, gradient consists of terms:
    # grad[i] = (1 - h[i+lag]) if i+lag is valid; and -h[i-lag] if i-lag is valid
    # but this interpretation is incomplete. 
    # We actually compute derivative of h * 1-h over all overlapping shifts to get actual gradient
    # of the objective.
    
    # Understanding from correlation:
    # Objective is max_lag sum_i h_i (1 - h_{i+lag})  
    # so for this fixed lag, we consider the index i in [0, N-1], and the index i+lag 
    # for this part to be valid it must be in [0, N-1], i.e. i >= 0 and i+lag <= N - 1, hence i <= N-1-lag
    # and i + lag >= 0, i.e. i >= -lag, hence if lag >= 0 then i >= lag else i >= 0
    
    # Correct path:
    # For the fixed lag with max at lag 'l':  
    # Function is sum_i h_i (1 - h_{i+l})  
    # For index i, we are computing (d/d h_i) sum_j h_j(1-h_{j+l}) = (1 - h_{i+l}) if l is valid  
    #                  (d/d h_i) sum_j h_j(1-h_{j+l}) = - h_{i-l} if l is invalid  
    # Actually it's easier, derivative = (1 - h_{i+l} ) + (-h_{i-l}) part if it exists  
    # But since we are considering full-correlation (with all comparisons), i need to be careful.
    
    # Concise derivation:
    # Let R = correlation, where R[tau + N-1] corresponds to lag tau = tau.
    # Objective = max_tau R[tau].
    # Say the maximizing tau is tau_idx. 
    # Then the subgradient at h is: specific added terms scaled by the weights (which, for infinite β, simplifies to fixed treatment).
    # The actual gradient w.r.t. h_i of  sum_j h_j (1 - h_{j-\tau_idx}) for tau_idx (lag)  
    # = 1 - h_{i+\tau_idx} if i+\tau_idx < N AND 0 <= i  
    # = - h_{i-\tau_idx} if i-\tau_idx >= 0 AND  i < N
    
    # cruder code, right hand side:
    # F(i + l) = 
    # so for i in 0 to N-1, let's compute how this contributes to gradient for h[i]
    
    # For one gradient direction (1-h[i+l]) 
    # when l < 0, it's h[i-l]  
    # when l > 0, it's h[i+l]
    # if h[i+l] valid, then + (1 - h[i+l]), otherwise gradient is 0

    # Computation of exact sub-gradient for the maximizing lag:
    # Let's re-index the full index properly.
    # For i-th element, and for the maximizing lag k = idx_best - (N-1),
    # the gradient w.r.t. h_i is:
    # if (i + k >= 0 and i + k < len(h)): 
    #    Gradient contribution = -1 * (1 - h[i + k]) // (1 - h[i+k]);
    # So update h[i] contributes to derivative via h[i] term, vs h[i+k] via (1-h[i+k])  
    #    That is为啥 we create the form...)
    # However looking at the function being minimized:
    # correlations max_{l} sum_i h_i (1 - h_{i+l})  
    # thus, for h_i, derivative is sum_l (active 1_h relevant to l @ i) = 
    # For l = idx_best:   - (1-h[i + idx_best-N+1]) if i+idx_best-N+1 >= 0  
    #  i.e., i <= idx_best-N+1 and 
    # For l = idx_best:  0, if i < 0, 
    #                   1 if i - idx_best+N-1 >= 0 and i + idx_best-N+1 < N  
    #                   But this is looking at terms in code variable
    # 
    #  That means relative system:
    #    cor[in] = sum_{j=0}^{N-1} h[j] (1 - h[j-in+N-1]) 
    #    For each i = 0 to 2*N-2  
    #    Gradients wrt h_k =   \sum_{i=0}^{2N-2} w[i] (d/dh_k) cor[i] = w[i] (d/dh_k) sum_j h_j * (1-h_{j+(in-N+1)})  
    #    We THEN take in index = idx-best, corresponds to some l = idx_best - (N-1) 
    #    THEN the grad wrt h_k =  - w[idx_best]* (1 - h[k + l]) if 0 <= k+l < N
    #                           - w[idx_best]* h[k-l] if 0 <= k-l < N
    # 
    # part 1 unit (vanishing when not in range):
    # heat term from (1-h_{k+l})  
    # Let _idcor = idx_best
    #    k in range(0, N): 
    #    corresponding to idcor = (k + l + N-1) in full corr vector  
    #    contribution to d/d h_k = - w[_idx_cor] * (1-h_{k_labeled}) when k_labeled validity in [0,N-1]
    corr_id = idx_best
    lag = corr_id - (len(h)-1)
    
    # Compute sub-gradient of the correlation for specific lag:
    # It's the gradient of sum_{i=0}^{N-1} h_i * (1 - h_{i+lag})
    for i in range(len(h)):
        # gradient of h_i * (1 - h_{i+lag}) w.r.t. h_i is (1 - h_{i+lag})
        # Gradient of h_i * (1 - h_{i+lag}) w.r.t. h_{i+lag} is  -h_i 
        # but  for fixed h, we compute gradient per h_{i}.
        # In our objective, it makes a contribution to max_{l} sum_i h_i (1 - h_{i+l})  
        
        # There is an expanded formulation.
        # for fixed l, the gradient shows:
        # objective = sum_i h_i * (1 - h_{i+l})  
        # d/dh_i obj = (1 - h_{i+l}) 
        # (but only if i+l is valid)
        # Also d/dh_{i+l} obj = -h_i
        if i + lag >= 0 and i + lag < len(h):  # i+l in valid range
            grad[i] += -(1 - h[i + lag]) # Got compensation
        if i - lag >= 0 and i - lag < len(h): # i-lag in valid range
            grad[i] += h[i - lag]
    
    return -grad  # For descending gradient

def mass_transfer(h, grad, epsilon):
    """Perform mass transfer for constraint respecting heuristic to improve solution quickly"""
    d = -grad
    # Identify sources (where decreasing h improves)
    sources = np.where(d < 0)[0]
    # Identify sinks (where increasing h improves)
    sinks = np.where(d > 0)[0]
    
    # Sort by marginal improvement
    sources_sorted = sources[np.argsort(-d[sources])]
    sinks_sorted = sinks[np.argsort(d[sinks])]
    
    # Create copies to not modify during iteration
    h_new = h.copy()
    for i in sources_sorted:
        if epsilon <= 0:
            break  # all done
        for j in sinks_sorted:
            if epsilon <= 0:
                break
            # Primitive amount
            transfer = min(h_new[i], 1 - h_new[j], epsilon)
            if transfer > 0:
                h_new[i] -= transfer
                h_new[j] += transfer
                epsilon -= transfer
    
    return project_sum_box(h_new, TARGET_SUM)

def line_search(h, grad, c0, beta):
    """Backtracking line search with alpha reduction for optimal descent"""
    alpha_max = 0.2
    alpha_min = 5e-4
    step_size = alpha_max
    h_new = h - step_size * grad
    
    # Project h_new first (recompute lambda for that specific alpha)
    h_new_projected = project_sum_box(h_new, TARGET_SUM)
    c_new = evaluate_c5(h_new_projected)
    
    # Accept if improvement, else decrease alpha until improvement or threshold
    while c_new >= c0 and step_size > alpha_min:
        step_size /= 2.0
        h_new = h - step_size * grad
        h_new_projected = project_sum_box(h_new, TARGET_SUM)
        c_new = evaluate_c5(h_new_projected)
    
    return h_new_projected if step_size > alpha_min else h, c_new
