# Global variables
TARGET_SUM = None

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def pattern_based_search(h0, max_iter=1000):
    """A new heuristic that focuses on patterned moves"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Try different patterns for moving mass
    patterns = [
        lambda x: x,  # Identity
        lambda x: np.sin(x * np.pi / len(x)),  # Sine pattern
        lambda x: np.exp(-x * x / (2 * (len(x)/4)**2)),  # Gaussian
    ]
    
    for iter_num in range(max_iter):
        # Try a pattern-based move
        pattern_idx = iter_num % len(patterns)
        pattern = patterns[pattern_idx]
        pattern_values = pattern(np.arange(n_points))
        
        # Create a move pattern
        move_direction = pattern_values - np.mean(pattern_values)
        move_direction = move_direction / np.linalg.norm(move_direction)
        
        # Try moving mass along the pattern direction
        step_size = 0.01
        h_candidate = h + step_size * move_direction
        h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
        
        c5_candidate = evaluate_c5(h_candidate)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
            h = h_candidate.copy()
        else:
            # Reduce step size if no improvement
            step_size *= 0.95
            
        # Occasionally do random local moves
        if iter_num % 50 == 0:
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            delta = min(h[i], 1 - h[j], 0.01)
            if delta > 1e-6:
                h[i] -= delta
                h[j] += delta
                h = project_to_sum_box(h, TARGET_SUM)
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        
        if iter_num % 100 == 0 and best_c5 < 0.38:
            break
        
    return best_h, best_c5
