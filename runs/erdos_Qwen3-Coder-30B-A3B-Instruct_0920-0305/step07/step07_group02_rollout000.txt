import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_initialization():
    """Initialize binary pattern with exactly half ones"""
    h = np.zeros(n_points)
    ones_indices = np.random.choice(n_points, n_points // 2, replace=False)
    h[ones_indices] = 1.0
    return h

def greedy_improvement(h):
    """Improve h by targeting the worst overlap shift"""
    n = len(h)
    max_iter = 10000
    for _ in range(max_iter):
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n - 1)
        
        # Find regions affected by this shift
        if max_shift >= 0:
            indices_i = np.arange(0, n - max_shift)
            indices_j = indices_i + max_shift
        else:
            indices_i = np.arange(-max_shift, n)
            indices_j = indices_i + max_shift
            
        # Find mismatches: (1,0) pairs and (0,1) pairs
        h_i = h[indices_i]
        h_j = h[indices_j]
        
        mismatches_10 = indices_i[(h_i == 1) & (h_j == 0)]
        mismatches_01 = indices_i[(h_i == 0) & (h_j == 1)]
        
        if len(mismatches_10) == 0 or len(mismatches_01) == 0:
            break
            
        # Choose one pair to swap
        i = np.random.choice(mismatches_10)
        j = np.random.choice(mismatches_01)
            
        # Perform the swap
        h[i], h[j] = h[j], h[i]
        
        # Prevent infinite loops
        if _ > max_iter - 10:
            break
            
    return h

def simulated_annealing_binary(h0, time_limit, seed):
    """Run simulated annealing with binary flips"""
    np.random.seed(seed)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    T0 = 0.05  # Initial temperature
    
    ones_indices = np.where(h == 1)[0]
    zeros_indices = np.where(h == 0)[0]
    
    # Precompute correlation for initial state
    current_c5 = evaluate_c5(h)
    
    iteration = 0
    while time.time() - start_time < time_limit:
        iteration += 1
        # Adaptive temperature schedule
        elapsed = time.time() - start_time
        T = T0 * (1 - elapsed / time_limit)
        if T <= 0:
            break
            
        # Choose two indices to swap (one with 1, one with 0)
        i = np.random.choice(ones_indices)
        j = np.random.choice(zeros_indices)
        
        # Swap values
        h[i] = 0
        h[j] = 1
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h)
        delta_c5 = new_c5 - current_c5
        
        # Accept with probability based on temperature
        if delta_c5 < 0 or np.exp(-delta_c5 / T) > np.random.rand():
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                
            # Update index lists
            ones_indices = np.where(h == 1)[0]
            zeros_indices = np.where(h == 0)[0]
        else:
            # Revert swap
            h[i] = 1
            h[j] = 0
            
    return best_h, best_c5

def local_gradient_refinement(h0, max_iter=1000):
    """Refine with gradient-based optimization on continuous variables"""
    h = h0.copy()
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_local):
        return evaluate_c5(h_local)
    
    def constraint_eq(h_local):
        return np.sum(h_local) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
    except:
        pass
    
    # Project to maintain constraints
    h = project_to_sum_box(h, TARGET_SUM)
    return h

    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    TARGET_SUM = n_points // 2
    best_h = None
    best_c5 = float('inf')
    time_left = budget_s * 0.9  # Reserve 10% for final refinement
    
    # Multiple restarts to avoid local optima
    num_restarts = 10
    restarts_per_run = 10
    
    for run_id in range(num_restarts):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Initial binary pattern
        h_initial = binary_initialization()
        
        # Improve with greedy algorithm first
        h_greedy = greedy_improvement(h_initial.copy())
        c5_greedy = evaluate_c5(h_greedy)
        
        # After greedy, run local search with simulated annealing
        h_final, c5_final = simulated_annealing_binary(
            h_greedy, 
            time_left / (num_restarts * restarts_per_run), 
            seed + run_id
        )
        
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    # Final refinement
    if best_h is not None and time.time() - start_time < budget_s * 0.95:
        final_h = local_gradient_refinement(best_h.copy(), max_iter=500)
        final_c5 = evaluate_c5(final_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
