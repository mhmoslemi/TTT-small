import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_of_corr_at_lag(h, lag):
    """Compute gradient of correlation at a specific lag"""
    n = len(h)
    g = np.ones(n)
    
    # Handle positive lag
    if lag >= 0:
        if lag < n:
            g[:n-lag] = 1 - h[lag:]
            g[lag:] -= h[:n-lag]
        else:
            g[:] = 0  # No overlap
    else:  # Handle negative lag
        d = -lag
        if d < n:
            g[d:] = 1 - h[:n-d]
            g[:n-d] -= h[d:]
        else:
            g[:] = 0  # No overlap
    
    return g

def gradient_descent_refinement(h0, max_iter=1000, step_size=0.05):
    """Refine solution using gradient descent on max correlation"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Compute correlation and find max
        corr = compute_corr(h)
        max_idx = np.argmax(corr)
        max_lag = max_idx - (n_points - 1)
        
        # Compute gradient for max lag
        grad = gradient_of_corr_at_lag(h, max_lag)
        
        # Step update
        h_new = h - step_size * grad
        
        # Project to feasible set
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        
        h = h_new
    
    return best_h, best_c5

def simulated_annealing_binary(n_iter=5000, temp_start=0.1, temp_end=1e-5):
    """Simulated annealing on binary patterns"""
    best_h = None
    best_c5 = float('inf')
    
    # Initialize with random binary pattern
    h = np.zeros(n_points)
    ones_indices = np.random.choice(n_points, size=n_points//2, replace=False)
    h[ones_indices] = 1
    
    current_c5 = evaluate_c5(h)
    current_temp = temp_start
    
    for i in range(n_iter):
        # Annealing schedule
        if i > 0:
            current_temp = temp_start * (temp_end/temp_start)**(i/n_iter)
        
        # Generate neighbor: swap one 1 and one 0
        ones_locs = np.where(h == 1)[0]
        zeros_locs = np.where(h == 0)[0]
        
        if len(ones_locs) > 0 and len(zeros_locs) > 0:
            idx_1 = np.random.choice(ones_locs)
            idx_0 = np.random.choice(zeros_locs)
            
            h_new = h.copy()
            h_new[idx_1] = 0
            h_new[idx_0] = 1
            
            new_c5 = evaluate_c5(h_new)
            delta = new_c5 - current_c5
            
            if delta < 0 or np.random.random() < np.exp(-delta / current_temp):
                h = h_new
                current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Phase 1: Simulated Annealing for binary patterns
    phase1_end = start_time + 0.4 * budget_s
    for run_id in range(10):
        if time.time() > phase1_end:
            break
            
        h, c5 = simulated_annealing_binary(n_iter=5000)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Phase 2: Gradient descent refinement
    phase2_end = start_time + 0.8 * budget_s
    if best_h is not None:
        h_refined, c5_refined = gradient_descent_refinement(best_h, max_iter=1000, step_size=0.05)
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    # Phase 3: Final stochastic refinement
    phase3_end = start_time + budget_s
    if best_h is not None:
        h_final = best_h.copy()
        current_c5 = best_c5
        
        while time.time() < phase3_end:
            # Small random perturbation
            h_perturbed = h_final + np.random.normal(0, 0.005, n_points)
            h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
            
            # Local search using gradient descent
            h_local, c5_local = gradient_descent_refinement(h_perturbed, max_iter=500, step_size=0.02)
            
            if c5_local < current_c5:
                h_final = h_local.copy()
                current_c5 = c5_local
        
        best_h = h_final.copy()
        best_c5 = current_c5
    
    return (best_h.tolist(), float(best_c5), n_points)
