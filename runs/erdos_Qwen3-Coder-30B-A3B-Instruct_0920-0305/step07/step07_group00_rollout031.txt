# Global variables
TARGET_SUM = None
n_points = 120

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def initialize_h_random(n, target_sum):
    """Generate random h vector with given sum"""
    h = np.random.uniform(0, 1, n)
    h = project_to_sum_box(h, target_sum)
    return h

def gradient_descent_with_smoothing(h0, eps_schedule, iters_per_eps, lr_schedule, target_sum):
    """Perform gradient descent with smoothed max objective"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for eps_idx, (eps, max_iter, lr) in enumerate(zip(eps_schedule, iters_per_eps, lr_schedule)):
        if max_iter <= 0:
            continue
        for iter in range(max_iter):
            # Compute correlation
            corr = np.correlate(h, 1 - h, mode="full")
            
            # Compute softmax weights (stable computation)
            max_corr = np.max(corr)
            w_raw = np.exp((corr - max_corr) / eps)
            w_norm = w_raw / np.sum(w_raw)
            
            # Compute gradient via convolution approach
            # Gradient = sum_s w_s * (1 - h_{i+s}) - h_{i-s}
            # Use FFT for convolution
            L = len(corr)
            offset = len(h) - 1
            
            conv_wh = np.convolve(w_norm, h, mode='full')
            conv_rev = np.convolve(w_norm[::-1], h, mode='full')
            
            grad = np.ones_like(h) - (conv_rev[offset:offset + len(h)] + conv_wh[offset:offset + len(h)])
            
            # Update
            h_new = h - lr * grad
            
            # Project onto constraints
            h_new = project_to_sum_box(h_new, target_sum)
            h = h_new
            
            # Evaluate performance occasionally
            if iter % 10 == 0:
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    
    return best_h, best_c5
