# Global variables
TARGET_SUM = None

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

        fm = f(m)
        if abs(fm) < tol:
            break

            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_of_corr_at_lag(h, lag):
    """Compute gradient of correlation at lag with respect to h"""

    else:
        d = -lag

            g[d:] = 1 - h[:n-d]
            g[:n-d] -= h[d:]
    
    return g
