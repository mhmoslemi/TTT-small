import numpy as np
from scipy.signal import correlate
import time
try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False

def compute_c5_fft(h):
    """Fast evaluation of C5 using FFT"""
    dx = 2.0 / len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def greedy_descent(h0):
    h = h0.copy()
    prev_c5 = compute_c5_fft(h)
    for _ in range(1000):
        corr = correlate(h, h, mode='full')
        # To get the shifted overlap: we want to minimize int h(x)(1-h(x+k)) = \sum h[i] (1 - h[i+k]) = \sum h[i] - \sum h[i]h[i+k]
        # = 1 - \sum h[i] h[i+k]. Maximize \sum h[i] h[i+k] becomes minimize 1 - \sum h[i]h[i+k]
        # The index of maximum correlation
        max_corr_idx = np.argmax(corr)
        corr_k = max_corr_idx - (len(h) - 1)
        # find mismatched positions
        A = []
        B = []
        for i in range(len(h)):
            j = i + corr_k
            if 0 <= j < len(h):
                if h[i] == 1 and h[j] == 0:
                    A.append(i)
                elif h[i] == 0 and h[j] == 1:
                    B.append(i)
        if len(A) == 0 or len(B) == 0:
            break
        # swap one element from A with one from B
        i = np.random.choice(A)
        j = np.random.choice(B)
        # make the swap in the original array
        h[i], h[j] = h[j], h[i]
        c5 = compute_c5_fft(h)
        if c5 >= prev_c5:
            h[i], h[j] = h[j], h[i]  # roll back
            break
        else:
            prev_c5 = c5
    return h, compute_c5_fft(h)

def anneal_binary(h0, timeout_s):
    start_time = time.time()
    h = h0.copy()
    current_c5 = compute_c5_fft(h)
    T0 = 0.02
    max_iter = 10000
    for i in range(max_iter):
        if time.time() - start_time > timeout_s:
            break
        # select two random positions
        pos1 = np.random.randint(0, len(h))
        pos2 = np.random.randint(0, len(h))
        if pos1 == pos2:
            continue
        if h[pos1] == h[pos2]:
            continue
        # perform the swap
        h[pos1], h[pos2] = h[pos2], h[pos1]
        new_c5 = compute_c5_fft(h)
        delta = new_c5 - current_c5
        if delta < 0:
            # accept always
            current_c5 = new_c5
        else:
            # accept with probability based on temperature
            T = T0 * (1.0 - i / max_iter)
            if T <= 0 or np.random.rand() > np.exp(-delta / T):
                # reject
                h[pos1], h[pos2] = h[pos2], h[pos1]
        # do local improvement every 100 steps
        if i % 100 == 0:
            h, current_c5 = greedy_descent(h)
    return h, current_c5

                end = min(N, N-k)
                if start >= end:

                i_range = list(range(start, end))

                    constraints.append(w <= h_vars[j])
                    constraints.append(w >= h_vars[i] + h_vars[j] - 1)
                    w_list.append(w)

                constraints.append(cp.sum(w_list) >= lam)
            
            # symmetry break
            constraints.append(h_vars[0] == 1)
