import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global constants
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmin_weights(s, alpha):
    """Compute normalized weights for softmin"""
    s_shifted = s - np.max(s)  # Numerical stability
    exp_s = np.exp(-alpha * s_shifted)
    return exp_s / np.sum(exp_s)

def weighted_gradient(h, alpha):
    """Compute gradient of softmin with respect to h"""
    # Compute correlation s = h * h (full correlation)
    s = correlate(h, h, mode='full')
    
    # Compute weights
    w = softmin_weights(s, alpha)
    
    # Compute weighted gradient
    grad = np.zeros_like(h)
    n = len(h)
    
    for i in range(len(s)):
        lag = i - (n - 1)
        if lag >= 0:
            # Add h[lag:] to grad[:n-lag]
            if lag < n:
                grad[:n-lag] += w[i] * h[lag:]
        else:
            # Add h[:n+lag] to grad[-lag:]
            if -lag < n:
                grad[-lag:] += w[i] * h[:n+lag]
    
    return grad

def pairwise_move(h, grad, delta, rng):
    """Perform a pairwise mass transfer move"""
    # Find source (min gradient) and target (max gradient)
    src_idx = np.argmin(grad)
    tgt_idx = np.argmax(grad)
    
    # Ensure indices are valid
    if h[src_idx] <= 1e-10 or h[tgt_idx] >= 1 - 1e-10:
        return False, h
    
    # Determine transfer amount
    transfer = min(delta, h[src_idx], 1 - h[tgt_idx])
    
    # Apply the transfer
    if transfer > 1e-10:
        h[src_idx] -= transfer
        h[tgt_idx] += transfer
        return True, h
    else:
        return False, h

def annealed_gradient_ascent(h0, time_budget, alpha0, delta0, temp0, rng):
    """Main optimisation loop"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    t0 = time.time()
    alpha = alpha0
    delta = delta0
    temp = temp0
    
    iter_count = 0
    last_improvement = 0
    
    while time.time() - t0 < time_budget:
        # Compute gradient
        grad = weighted_gradient(h, alpha)
        
        # Perform a move
        accepted, h_new = pairwise_move(h, grad, delta, rng)
        if accepted:
            h = h_new.copy()
            c5_new = evaluate_c5(h)
            
            # Metropolis acceptance criterion
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                last_improvement = iter_count
            elif rng.random() < np.exp(-(c5_new - best_c5) / temp):
                best_c5 = c5_new
                best_h = h.copy()
                last_improvement = iter_count
            else:
                # Revert if not accepted
                h = h_new.copy()
        
        # Update parameters
        if iter_count % 100 == 0:
            alpha *= 1.05
            alpha = min(alpha, 200)
            temp *= 0.999
            delta *= 0.9995
            if delta < 1e-6:
                delta = 1e-6
                
        iter_count += 1
        
        # Early stopping if no improvement in a while
        if iter_count - last_improvement > 10000:
            break
            
        # Occasionally re-project to handle round-off
        if iter_count % 1000 == 0:
            h = project_to_sum_box(h, TARGET_SUM)
    
    return best_h, best_c5

def random_binary_start(n, rng):
    """Create a random binary start vector"""
    h = np.zeros(n)
    ones_indices = rng.sample(range(n), n // 2)
    h[ones_indices] = 1.0
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def local_optimize_restart(start_h, remaining_time, rng):
    """Run a local optimization restart"""
    if remaining_time < 5:
        return start_h, evaluate_c5(start_h)
    
    h = start_h.copy()
    h = project_to_sum_box(h, TARGET_SUM)
    
    # Run gradient ascent with shorter time budget
    time_budget = min(remaining_time - 5, 20.0)
    best_h, best_c5 = annealed_gradient_ascent(h, time_budget, 10.0, 0.4, 5e-3, rng)
    
    return best_h, best_c5
