```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_to_float(h_binary):
    """Convert binary representation to float vector"""
    return h_binary.astype(float)

def float_to_binary(h_float):
    """Convert float vector to binary representation"""
    return (h_float > 0.5).astype(int)

def compute_shift_mismatches(h, shift):
    """Find mismatched positions for a given shift"""
    if shift >= 0:
        matched = h[:-shift] * (1 - h[shift:])
    else:
        matched = h[-shift:] * (1 - h[:shift])
    return matched

def get_worst_shift_and_mismatches(h):
    """Finds the shift with maximum overlap and mismatched positions"""
    corr = compute_corr_fft(h)
    max_shift = np.argmax(corr)
    max_val = corr[max_shift]
    
    # Convert shift to actual shift (positive for right shift)
    shift = max_shift - (len(h) - 1)
    
    return shift, max_val

def greedy_swap_improvement(h):
    """Improve the solution through local swaps"""
    h = h.copy()
    prev_max_corr = evaluate_c5(h)
    
    for _ in range(10000):  # Limit iterations
        shift, max_corr = get_worst_shift_and_mismatches(h)
        if max_corr < prev_max_corr - 1e-10:
            break
            
        # Find all mismatching pairs for current shift
        if shift >= 0:
            h_left = h[:-shift]
            h_right = h[shift:]
        else:
            h_left = h[-shift:]
            h_right = h[:shift]
            
        mismatches = np.where((h_left == 1) & (h_right == 0))[0]
        matched = np.where((h_left == 0) & (h_right == 1))[0]
        
        if len(mismatches) == 0 or len(matched) == 0:
            continue
            
        # Pick a random mismatch and match to swap
        i = np.random.choice(mismatches)
        j = np.random.choice(matched)
        
        # Apply swap
        if shift >= 0:
            h[i], h[i + shift] = h[i + shift], h[i]
        else:
            h[j], h[j - shift] = h[-shift + j], h[j]
            
        # Ensure sum constraint
        h = project_to_sum_box(h, TARGET_SUM)
        
        new_c5 = evaluate_c5(h)
        if new_c5 >= prev_max_corr - 1e-10:
            h = h.copy()
            break
        else:
            prev_max_corr = new_c5
            
    return h

def simulated_annealing(h_start, iterations=2000):
    """Improve with simulated annealing"""
    h_current = h_start.copy()
    current_c5 = evaluate_c5(h_current)
    
    temp = 0.1
    cool_rate = 0.999
    min_temp = 1e-5
    
    for _ in range(iterations):
        if temp < min_temp:
            break
            
        # Propose a move: randomly swap two indices
        i, j = np.random.randint(0, n_points, 2)
        h_new = h_current.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # Ensure sum constraint maintained
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        delta_c5 = new_c5 - current_c5
        
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temp):
            h_current = h_new
            current_c5 = new_c5
            
        temp *= cool_rate
            
    return h_current

def generate_thue_morse_sequence(n):
    """Generate Thue-Morse sequence of length n"""
    sequence = [0]
    for i in range(int(np.log2(n))):
        sequence = sequence + [1 - x for x in sequence]
    return np.array(sequence[:n])

def generate_rudin_shapiro_sequence(n):
    """Generate Rudin-Shapiro sequence of length n"""
    seq = np.zeros(n, dtype=int)
    seq[0] = 0
    for i in range(1, n):
        # Rudin-Shapiro definition  
        # This is a rough implementation - for precise construction
        # the real Rudin-Shapiro is more complex 
        seq[i] = (i & (i >> 1)) & 1
    return seq

def local_gradient_refinement(h0, max_iter=500):
    """Refine with gradient-based optimization"""
    h = h0.copy()
    
    # Focus on best regions found
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_local):
        return evaluate_c5(h_local)
    
    def constraint_eq(h_local):
        return np.sum(h_local) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
    except:
        pass
    
    # Project to maintain constraints
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # 150 restarts
    for restart in range(150):
        if time.time() - start_time > budget_s:
            break
        
        # Generate random binary vector
        h_init = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
        # Ensure sum is correct
        current_sum = np.sum(h_init)
        if current_sum > TARGET_SUM:
            ones_to_flips = int(current_sum - TARGET_SUM)
            indices = np.where(h_init == 1)[0]
            flip_indices = np.random.choice(indices, size=ones_to_flips, replace=False)
            h_init[flip_indices] = 0
        elif current_sum < TARGET_SUM:
            zeros_to_flips = int(TARGET_SUM - current_sum)
            indices = np.where(h_init == 0)[0]
            flip_indices = np.random.choice(indices, size=zeros_to_flips, replace=False)
            h_init[flip_indices] = 1
        
        # Ensure integer constraint and avoid double-counting
        h_init = project_to_sum_box(h_init.astype(float), TARGET_SUM)
        
        # Apply greedy improvement
        h_greedy = greedy_swap_improvement(h_init)
        
        # Apply simulated annealing improvement
        h_anneal = simulated_annealing(h_greedy)
        
        # Apply gradient refinement
        h_final = local_gradient_refinement(h_anneal, max_iter=300)
        
        c5_final = evaluate_c5(h_final)
        
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final
    
    return (best_h.tolist(), float(best_c5), n_points)
```
