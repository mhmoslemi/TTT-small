TARGET_SUM = None

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def create_thue_morse(n):
    """Creates Thue-Morse sequence of length n"""
    s = [0]
    while len(s) < n:
        s += [1 - x for x in s]
    return np.array(s[:n], dtype=float)

def create_rudin_shapiro(n):
    """Creates Rudin-Shapiro sequence of length n (simplified version)"""
    h = np.zeros(n)
    # Start with 0s and 1s, but use pattern that leads to good overlap properties
    for i in range(n):
        h[i] = (i & (i >> 1)) == 0  # Simple approximation
    return h.astype(float)

    return indices1, indices2, max_shift

def swap_indices(indices1, indices2):
    """Swap pairs from indices1 and indices2"""
    # Choose one of the first few mismatched pairs
    if len(indices1) == 0 or len(indices2) == 0:
        return False
    
    # Select a random pair
    i1 = random.choice(indices1)
    i2 = random.choice(indices2)
    if i1 == i2:
        return False
        
    return (i1, i2)
