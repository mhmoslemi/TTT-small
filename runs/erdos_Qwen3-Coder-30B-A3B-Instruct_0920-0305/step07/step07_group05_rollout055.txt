import numpy as np
import cvxpy as cp
from scipy.linalg import eigh
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 140
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def build_lag_matrices(n):
    """Build sparse matrices A_ell for each lag"""
    A_list = []
    for ell in range(-(n-1), n):
        rows = []
        cols = []
        for i in range(n):
            j = i + ell
            if 0 <= j < n:
                rows.append(i)
                cols.append(j)
        if len(rows) > 0:
            A_list.append((ell, np.array(rows), np.array(cols)))
    return A_list

def solve_sdp(n, A_list, max_time=30):
    """Solve SDP relaxation to get a lower bound"""
    start_time = time.time()

    # Variables
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        X >> 0,                                    # PSD
        cp.diag(X) >= 0,                           # Lower bound
        cp.diag(X) <= 1,                           # Upper bound
        cp.sum(cp.diag(X)) == n/2                   # Sum constraint
    ]

    # Lag constraints
    for ell, rows, cols in A_list:
        if len(rows) > 0:
            # Create sparse matrix for A_ell
            A_ell = np.zeros((n, n))
            A_ell[rows, cols] = 1
            A_ell[cols, rows] = 1  # symmetrize
            constraints.append(cp.trace(A_ell @ X) >= n/2 - t/dx)
    
    # Objective
    objective = cp.Minimize(t)

    # Problem
    prob = cp.Problem(objective, constraints)

    # Solver
    try:
        prob.solve(solver=cp.SCS, verbose=False, time_limit=max_time)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return X.value, t.value
        else:
            return None, None
    except:
        return None, None

def round_sdp_solution(X_sol, n_samples=2000):

    best_h = None
    best_c5 = np.inf
    
    # Compute leading eigenvector
    try:
        eigvals, eigvecs = eigh(X_sol)
        v = eigvecs[:, -1]  # largest eigenvector
        v = v / np.linalg.norm(v)  # normalize
        h_initial = v.copy()
        
        # Randomized rounding: sample from N(0, X_sol)
        # Cholesky decomposition version for numerical stability
        try:
            L = np.linalg.cholesky(X_sol)
            for i in range(n_samples):
                z = np.random.randn(n)
                y = L @ z 
                h = np.clip(y, 0, 1)
                h = project_to_sum_box(h, n_points / 2)
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        except:
            pass
        
        # Also try direct rounding from principal eigenvector
        h_direct = np.clip(v, 0, 1)
        h_direct = project_to_sum_box(h_direct, n_points / 2)
        c5_direct = evaluate_c5(h_direct)
        if c5_direct < best_c5:
            best_c5 = c5_direct
            best_h = h_direct.copy()
            
        return best_h, best_c5
    except:
        return None, np.inf

def local_improve(h, max_iter=1000):
    """Simple local improvement using gradient descent."""
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for _ in range(max_iter):
        # Simple finite difference approximation of gradient
        eps = 1e-4
        grad = np.zeros(n_points)
        
        h_orig = best_h.copy()
        for i in range(n_points):
            h_plus = h_orig.copy()
            h_plus[i] += eps
            h_plus = project_to_sum_box(h_plus, TARGET_SUM)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = h_orig.copy()
            h_minus[i] -= eps
            h_minus = project_to_sum_box(h_minus, TARGET_SUM)
            c5_minus = evaluate_c5(h_minus) 
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update position with gradient descent: move from low gradient to high
        idx_min = np.argmin(grad)
        idx_max = np.argmax(grad)
        
        delta = min(h_orig[idx_min], 1 - h_orig[idx_max], 0.01)
        if delta > 1e-6:
            h_orig[idx_min] -= delta
            h_orig[idx_max] += delta
            h_orig = project_to_sum_box(h_orig, TARGET_SUM)
            c5_new = evaluate_c5(h_orig)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_orig.copy()
        else:
            break  # Converged

    return best_h, best_c5
