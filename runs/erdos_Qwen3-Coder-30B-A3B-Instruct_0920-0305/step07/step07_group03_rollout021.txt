import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

# Global variables for the optimization
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr(h):
    """Compute correlation using FFT for efficiency"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Use binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_of_corr_at_lag(h, lag):
    """Compute gradient of correlation at a specific lag"""
    n = len(h)
    g = np.ones(n)
    
    if lag >= 0:
        # i from 0 to n-lag-1.
        if lag < n:
            g[:n-lag] = 1 - h[lag:]  # derivative w.r.t. h_i where i+lag in range
            g[lag:] -= h[:n-lag]     # subtract derivative of product term
    else:  # lag < 0, let d = -lag
        d = -lag
        # i from d to n-1 have i+lag = i-d in range.
        if d < n:
            g[d:] = 1 - h[:n-d]
            g[:n-d] -= h[d:]
    
    return g

def compute_subgradient(h, corr):
    """Compute subgradient based on the maximizing lag"""
    max_val = np.max(corr)
    indices = np.where(np.abs(corr - max_val) < 1e-10)[0]
    
    # Return average gradient over all maximizing lags
    g_sum = np.zeros(len(h))
    for idx in indices:
        # Convert index back to lag (shift)
        lag = idx - (n_points - 1)
        g_lag = gradient_of_corr_at_lag(h, lag)
        g_sum += g_lag
    
    if len(indices) > 0:
        return g_sum / len(indices)
    else:
        return np.zeros(len(h))

def subgradient_descent(h0, max_iter=2000, tolerance=1e-8):
    """Refine a solution using subgradient descent with diminishing step size"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for iteration in range(max_iter):
        corr = compute_corr(h)
        g = compute_subgradient(h, corr)
        
        # Adaptive step size
        step_size = 0.05 / np.sqrt(iteration + 1)
        
        # Update h
        h_new = h - step_size * g
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Early stopping
        if abs(best_c5 - compute_c5(h)) < tolerance:
            break
            
        h = h_new
        
    return best_h, best_c5

def binary_search_restart():
    """Run Simulated Annealing on binary patterns to get a good starting point"""
    best_c5 = float('inf')
    best_h = None
    
    # Try several random binary restarts
    for seed in range(5):
        np.random.seed(seed)
        
        # Create balanced binary pattern
        h = np.zeros(n_points)
        ones_indices = np.random.choice(n_points, size=int(TARGET_SUM), replace=False)
        h[ones_indices] = 1.0
        
        # Simple local improvement
        for _ in range(100):
            # Try a few swaps
            i = np.random.choice(np.where(h > 0.5)[0])  # position of one
            j = np.random.choice(np.where(h < 0.5)[0])  # position of zero
            
            if i != j:
                h_copy = h.copy()
                h_copy[i] = 0
                h_copy[j] = 1
                
                c5_new = compute_c5(h_copy)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_copy.copy()
                    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Phase 1: Binary Search + Refinement
    # Run binary search restarts for 50% of budget
    phase1_end = start_time + 0.5 * budget_s
    best_h = None
    best_c5 = float('inf')
    
    while time.time() < phase1_end:
        h_restart, c5_restart = binary_search_restart()
        if c5_restart < best_c5:
            best_c5 = c5_restart
            best_h = h_restart.copy()

    # Phase 2: Continuous refinement using subgradient descent
    # Use remaining budget for refinement
    remaining_budget = budget_s - (time.time() - start_time)
    
    if best_h is not None and remaining_budget > 10:
        # Refine the best binary solution with continuous optimization
        refined_h, refined_c5 = subgradient_descent(best_h, max_iter=1000)
        
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h.copy()

    # Final evaluation and return
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
