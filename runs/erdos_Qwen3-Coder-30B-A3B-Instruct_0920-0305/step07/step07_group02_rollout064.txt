import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
N = 256
dx = 2.0 / N
TARGET_SUM = N // 2

def c5_fft(h):
    """Compute C5 using FFT-based linear correlation"""
    n = len(h)
    nfft = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, nfft)
    B = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * B, nfft)[:2*n-1]
    return np.max(conv) * dx

def project_to_sum(h, target_sum):
    """Project h onto sum constraint using water-filling"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Binary search for lambda
    def f(lam):
        proj = np.clip(h + lam, 0, 1)
        return np.sum(proj) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b)/2, 0, 1)
    # Ensure exact sum due to floating point precision
    scale = target_sum / np.sum(result)
    result = result * scale
    return result

def thue_morse(n):
    """Generate Thue-Morse sequence of length n"""
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    return tm

def rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence of length n"""
    rs = np.zeros(n, dtype=int)
    for i in range(n):
        ones = bin(i).count('1') 
        rs[i] = (ones % 2)
    return rs

def greedy_descent(h0):
    """Greedy optimization focused on the worst shift"""
    h = h0.copy()
    while True:
        corr = correlate(h, 1-h, mode='full')
        worst_shift_idx = np.argmax(corr)
        shift = worst_shift_idx - (N - 1)
        
        if shift >= 0:
            A = np.where((h[:-shift] == 1) & (h[shift:] == 0))[0]
            B = np.where((h[:-shift] == 0) & (h[shift:] == 1))[0]
        else:
            A = np.where((h[-shift:] == 1) & (h[:-shift] == 0))[0]
            B = np.where((h[-shift:] == 0) & (h[:-shift] == 1))[0]
        
        if len(A) == 0 or len(B) == 0:
            break
        
        # Select random matching pair (a,b) and swap
        a = np.random.choice(A)
        b = np.random.choice(B)
        
        if shift >= 0:
            h[a], h[b] = 0, 1
        else:
            h[a], h[b] = 0, 1
    return h, c5_fft(h)

def simulated_annealing(h0, max_time):
    """Simulated annealing with cooling schedule and mini-greedy steps"""
    current_h = h0.copy()
    current_c5 = c5_fft(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    start_time = time.time()
    
    T0 = 0.02
    decay_factor = 0.999
    T = T0
    swaps = 0
    
    # Track recent acceptance rate for dynamic adjustment
    recent_accepted = 0
    recent_proposals = 0
    
    while time.time() - start_time < max_time and T > 1e-5:
        # Proposal: swap 2 random bits
        i = np.random.randint(0, N)
        j = np.random.randint(0, N)
        
        # Avoid swapping same element
        if i == j:
            continue
            
        # Save original values
        orig_i, orig_j = current_h[i], current_h[j]
        
        # Swap
        current_h[i], current_h[j] = orig_j, orig_i
        new_c5 = c5_fft(current_h)
        
        # Accept or reject
        delta = new_c5 - current_c5
        if delta <= 0 or np.random.random() < np.exp(-delta / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = new_c5
            recent_accepted += 1
        else:
            # Revert
            current_h[i], current_h[j] = orig_i, orig_j
        
        # Every 500 swaps, do a mini-greedy descent to help convergence
        swaps += 1
        if swaps % 500 == 0:
            local_h, local_c5 = greedy_descent(best_h)
            if local_c5 < best_c5:
                best_h = local_h.copy()
                best_c5 = local_c5
                
        # Adjust temperature
        recent_proposals += 1
        if recent_proposals >= 1000:
            acceptance_rate = recent_accepted / recent_proposals
            if acceptance_rate < 0.05:
                T = min(T * 1.1, T0)  # Heat up if too cold
            else:
                T = T * decay_factor  # Cool down
            recent_accepted = 0
            recent_proposals = 0
        
        if best_c5 < 0.379:
            break
            
    return best_h, best_c5

def continuous_refinement(h0):
    """Refine resulting binary solution with continuous optimization"""
    h = h0.copy()
    
    # Smooth with a triangular kernel to create a better start
    kernel = np.array([0.25, 0.5, 0.25])
    h_smooth = np.convolve(h, kernel, mode='same')
    h_smooth = np.clip(h_smooth, 0, 1)
    h_smooth = project_to_sum(h_smooth, TARGET_SUM)
    
    def objective(h_in):
        return c5_fft(h_in)
        
    def constraint_eq(h_in):
        return np.sum(h_in) - TARGET_SUM

    bounds = [(0, 1) for _ in range(N)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(objective, h_smooth, method='trust-constr', bounds=bounds, constraints=cons,
                       options={'maxiter': 500, 'gtol': 1e-8, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            return project_to_sum(h_final, TARGET_SUM)
    except:
        pass
        
    return h_smooth

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: Multiple greedy starts
    seeds = create_seeds()
    best_h = None
    best_c5 = float('inf')
    
    for name, h_seed in seeds:
        _, c5 = greedy_descent(h_seed)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_seed.copy()
    
    # Phase 2: Simulated Annealing with damping
    anneal_time = 0.5 * budget_s
    best_h_anneal, best_c5_anneal = simulated_annealing(best_h, anneal_time)
    if best_c5_anneal < best_c5:
        best_h = best_h_anneal
        best_c5 = best_c5_anneal

    # Phase 3: Continuous refinement
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        refined_h = continuous_refinement(best_h)
        refined_c5 = c5_fft(refined_h)
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
            
    # Final check
    best_h = project_to_sum(best_h, TARGET_SUM)
    final_c5 = c5_fft(best_h)
    
    return (best_h.tolist(), float(final_c5), N)
