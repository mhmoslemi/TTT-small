import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import differential_evolution

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    # Add some random perturbation
    h_perturbed = h_interp + np.random.normal(0, 0.01, n_points)
    h_perturbed = np.clip(h_perturbed, 0, 1)
    h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
    return h_perturbed

def improve_with_differential_evolution(h0, max_iter=100):
    """Use differential evolution for global optimization"""
    def objective(h):
        return evaluate_c5(h)
    
    # Define bounds (0 to 1 for each variable)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Add constraint that sum must be TARGET_SUM
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Run differential evolution
    res = differential_evolution(
        objective, 
        bounds, 
        constraints=cons,
        maxiter=max_iter,
        popsize=15,
        tol=1e-6,
        seed=42,
        recombination=0.7,
        mutation=(0.5, 1.0)
    )
    
    if res.success:
        h = res.x
    else:
        h = h0.copy()
    
    # Project to satisfy sum constraint
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def smooth_solution(h):
    """Apply Gaussian smoothing to make the solution more regular"""
    # Create a Gaussian kernel
    kernel_size = min(21, n_points // 4)
    if kernel_size % 2 == 0:
        kernel_size += 1
    sigma = kernel_size / 6.0
    x = np.arange(kernel_size) - kernel_size // 2
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel = kernel / np.sum(kernel)
    
    # Apply convolution (wrap around)
    h_smoothed = np.zeros_like(h)
    for i in range(n_points):
        total = 0.0
        for j in range(kernel_size):
            idx = (i + j - kernel_size // 2) % n_points
            total += h[idx] * kernel[j]
        h_smoothed[i] = total
    
    # Project to satisfy sum constraint
    h_smoothed = project_to_sum_box(h_smoothed, TARGET_SUM)
    return h_smoothed

def run(seed=42, budget_s=600, **kwargs):

    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = adaptive_sampling_initialization()
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: Differential evolution starting from different initial conditions
    phase1_end = start_time + 0.5 * budget_s
    initial_strategies = [
        adaptive_sampling_initialization,

        lambda: np.random.choice([0, 1], size=n_points, p=[0.3, 0.7]),

    ]
    
    for i in range(10):
        if time.time() > phase1_end:
            break
        h_start = initial_strategies[i % len(initial_strategies)]()
        h_start = project_to_sum_box(h_start, TARGET_SUM)
        h_candidate = improve_with_differential_evolution(h_start, max_iter=50)
        c5_candidate = evaluate_c5(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
    
    # Phase 2: Apply smoothing to the best solution and evaluate
    phase2_end = start_time + 0.8 * budget_s
    while time.time() < phase2_end:
        h_smoothed = smooth_solution(best_h)
        c5_smoothed = evaluate_c5(h_smoothed)
        if c5_smoothed < best_c5:
            best_h = h_smoothed.copy()
            best_c5 = c5_smoothed
    
    # Phase 3: Local refinement using small perturbations
    phase3_end = start_time + budget_s
    while time.time() < phase3_end:
        h_vary = best_h + np.random.normal(0, 0.005, n_points)
        h_vary = project_to_sum_box(h_vary, TARGET_SUM)
        c5_vary = evaluate_c5(h_vary)
        if c5_vary < best_c5:
            best_h = h_vary.copy()
            best_c5 = c5_vary
    
    return (best_h.tolist(), float(best_c5), n_points)
