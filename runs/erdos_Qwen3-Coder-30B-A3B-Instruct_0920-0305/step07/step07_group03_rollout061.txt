import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_c5(h):
    """Compute C5 value"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_of_corr_at_lag(h, lag):
    """Computes gradient of correlation w.r.t. h for a specific lag"""
    n = len(h)
    g = np.ones(n)
    
    if lag >= 0:
        if lag < n:
            # i from 0 to n-lag-1
            g[:n-lag] = 1 - h[lag:]
            # subtract term for i-lag where i-lag >= 0
            g[lag:] -= h[:n-lag]
    else:
        # lag negative
        d = -lag
        if d < n:
            # i from d to n-1, i+lag = i-d in range
            g[d:] = 1 - h[:n-d]
            # subtract term for i+d where i+d < n
            g[:n-d] -= h[d:]
    
    return g

def subgradient_descent(h0, max_iter=2000, step_size=0.1):
    """Subgradient descent to minimize max correlation"""
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for i in range(max_iter):
        corr = compute_corr(h)
        max_idx = np.argmax(corr)
        max_lag = max_idx - (n_points - 1)
        
        g = gradient_of_corr_at_lag(h, max_lag)
        h_new = h - step_size * g
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            step_size = min(0.1, step_size * 1.1)
        else:
            step_size *= 0.99
        
        h = h_new.copy()
    
    return best_h, best_c5

def simulated_annealing_binary(h0, T_start=0.1, T_end=1e-5, max_iter=10000):
    """Binary simulated annealing using swap moves"""
    h = h0.copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    T = T_start
    cooling_rate = (T_end / T_start) ** (1.0 / max_iter)
    
    for i in range(max_iter):
        # Generate neighbor: swap one 1 with one 0
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        if len(ones) > 0 and len(zeros) > 0:
            i_swap = np.random.choice(ones)
            j_swap = np.random.choice(zeros)
            
            h_new = h.copy()
            h_new[i_swap] = 0
            h_new[j_swap] = 1
            
            new_c5 = compute_c5(h_new)
            delta = new_c5 - current_c5
            
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = h_new.copy()
                current_c5 = new_c5
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        
        T *= cooling_rate
        if T < T_end:
            T = T_end
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = None
    best_c5 = float('inf')
    
    # Multi-start with binary SA
    num_restarts = 10
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s * 0.7:
            break
            
        # Random binary initial solution
        h_binary = np.zeros(n_points)
        ones_indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h_binary[ones_indices] = 1
        
        h_bin, c5_bin = simulated_annealing_binary(h_binary, max_iter=1000)
        
        # Convert to float and refine with gradient descent
        h_cont = h_bin.astype(float)
        h_refined, c5_refined = subgradient_descent(h_cont, max_iter=2000)
        
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    
    # Final refinement with gradient descent if needed
    if time.time() - start_time < budget_s * 0.95:
        h_final, c5_final = subgradient_descent(best_h, max_iter=1000)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    return (best_h.tolist(), float(best_c5), n_points)
