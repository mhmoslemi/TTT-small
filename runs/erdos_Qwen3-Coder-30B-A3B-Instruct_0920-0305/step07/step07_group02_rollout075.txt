import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
N = 256
dx = 2.0 / N
TARGET_SUM = N // 2
MAX_ITER_GREEDY = N  # upper bound on swaps needed
MAX_ITER_ANNEAL = 200000
T0 = 0.02
COOLING_RATE = 0.9999
MIN_TEMP = 1e-5

def c5_fft(h):
    """Fast evaluation of C5 using FFT"""
    n = len(h)
    nfft = 1 << ((2*n-1).bit_length())          # next power of two
    H = np.fft.rfft(h, nfft)
    B = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * B, nfft)[:2*n-1]   # length 2N‑1
    return np.max(conv) * dx

def project_to_sum(h, target_sum):
    """Project vector onto sum constraint [0,1] with sum = target_sum"""
    # Water filling algorithm using binary search
    def f(lam):
        ph = np.clip(h + lam, 0, 1)
        return np.sum(ph) - target_sum

    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def thue_morse(n):
    """Generate Thue-Morse sequence up to n bits"""
    tm = [0]
    for _ in range(n.bit_length()):
        tm = tm + [1 - x for x in tm]
    return np.array(tm[:n])

def rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence up to n bits"""
    rs = [0]
    for i in range(1, n):
        # Use recurrence or lookup table for speed
        # For small n, we can hardcode or generate efficiently
        count = bin(i).count('1')  # number of 1s in binary representation
        rs.append((count % 2) ^ (bin(i).count('11') % 2))
        # Default fallback for simplicity: fill evenly
    rs = rs + [0] * (n - len(rs))
    return np.array(rs[:n])

def greedy_descent(h_in):
    """Local descent improving the worst overlap shift"""
    h = h_in.copy()
    # Avoid overflow in counting
    max_iter_greedy = min(MAX_ITER_GREEDY, len(h))
    count = 0
    while count < max_iter_greedy:
        corr = correlate(h, 1-h, mode="full")
        best_shift_idx = np.argmax(corr)
        worst_shift = best_shift_idx - (len(h) - 1)
        corr_max = corr[best_shift_idx]

        # Compute all mismatched pairs
        A = []  # (1,0) pairs
        B = []  # (0,1) pairs
        
        if worst_shift >= 0:
            for i in range(len(h) - worst_shift):
                if h[i] == 1 and h[i + worst_shift] == 0:
                    A.append(i)
                if h[i] == 0 and h[i + worst_shift] == 1:
                    B.append(i)
        else:
            shift_abs = -worst_shift
            for i in range(len(h) - shift_abs):
                if h[i + shift_abs] == 1 and h[i] == 0:
                    A.append(i + shift_abs)
                if h[i + shift_abs] == 0 and h[i] == 1:
                    B.append(i + shift_abs)
        
        if not A or not B:
            break
            
        # Swap a (1,0) pair with a (0,1) pair
        i = np.random.choice(A)
        j = np.random.choice(B)
        if worst_shift >= 0:
            h[i], h[j] = 0, 1
        else:
            h[i], h[j] = 0, 1  # same effect (i < j both are swapped)
        count += 1

    return h, c5_fft(h)

def simulated_annealing(h_start, time_end):
    """Simulated Annealing on binary vectors"""
    h_current = h_start.copy()
    c5_current = c5_fft(h_current)
    h_best = h_current.copy()
    c5_best = c5_current
    T = T0
    
    step = 0
    while time.time() < time_end:
        if T < MIN_TEMP and step > 1000:
            break
        step += 1

        # Propose a move: swap 1 with 0 randomly
        ones = np.where(h_current == 1)[0]
        zeros = np.where(h_current == 0)[0]
        if len(ones) > 0 and len(zeros) > 0:
            idx1 = np.random.choice(ones)
            idx2 = np.random.choice(zeros)
            h_new = h_current.copy()
            h_new[idx1] = 0
            h_new[idx2] = 1
            
            c5_new = c5_fft(h_new)
            
            if c5_new <= c5_current:
                h_current = h_new
                c5_current = c5_new
                if c5_current < c5_best:
                    h_best = h_new.copy()
                    c5_best = c5_current
            else:
                p_accept = np.exp(-(c5_new - c5_current) / T)
                if np.random.random() < p_accept:
                    h_current = h_new
                    c5_current = c5_new
        # Cool down
        T *= COOLING_RATE
        if step % 500 == 0:
            # periodic boost: greedy descent
            h_current, c5_current = greedy_descent(h_current)
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
    return h_best, c5_best

def continuous_refinement(h_binary, time_end):
    """Final continuous refinement using constraint optimizer"""
    # Convert to float and smooth with low-pass
    h_smooth = np.convolve(h_binary, [0.25, 0.5, 0.25], mode='same')
    h0 = np.clip(h_smooth, 0, 1)
    h0 = project_to_sum(h0, TARGET_SUM)

    def objective(h):
        return c5_fft(h)

    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM

    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(N)]

    try:
        # If we don't have much time, skip
        if time.time() + 1.0 > time_end:
            return h_binary.copy(), c5_fft(h_binary)
        
        res = minimize(
            objective,
            h0,
            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'gtol': 1e-8, 'ftol': 1e-10}
        )
        if res.success:
            h_final = res.x.copy()
            # Ensure feasibility
            h_final = np.clip(h_final, 0, 1)
            h_final = project_to_sum(h_final, TARGET_SUM)
            c5_final = c5_fft(h_final)
            return h_final, c5_final
    except Exception:
        pass
    return h_binary.copy(), c5_fft(h_binary)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_end = start_time + budget_s
    
    # Phase 1: Multiple greedy descent runs
    h_seeds = initialize_h_samples()
    best_h = None
    best_c5 = float('inf')
    for h_seed in h_seeds:
        h_greedy, c5_greedy = greedy_descent(h_seed)
        if c5_greedy < best_c5:
            best_h = h_greedy
            best_c5 = c5_greedy
    
    # Phase 2: Simulated Annealing
    if time.time() + 1.0 < time_end:
        h_anneal, c5_anneal = simulated_annealing(best_h, time_end)
        if c5_anneal < best_c5:
            best_h = h_anneal
            best_c5 = c5_anneal
    
    # Phase 3: Continuous refinement
    if time.time() + 1.0 < time_end:
        h_final, c5_final = continuous_refinement(best_h, time_end)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    
    # Final sanity check
    best_h = np.clip(best_h, 0, 1)
    best_h = project_to_sum(best_h, TARGET_SUM)
    final_c5 = c5_fft(best_h)
    
    return (best_h.tolist(), float(final_c5), N)
