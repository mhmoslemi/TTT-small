import numpy as np
from scipy.signal import correlate
import time
import math

# Global constants
TARGET_SUM = None
MAX_N = 360
INITIAL_N = 120
DX = 2.0 / INITIAL_N

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * DX

def project_sum_box(h, target_sum):
    """Project vector h onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(h + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def smooth_gradient(h, corr, beta):
    """Compute smooth gradient using soft max of all correlations"""
    corr_shifted = corr - np.max(corr)
    exp_corr = np.exp(beta * corr_shifted)
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute full correlation vector with shifts
    N = len(h)
    grad = np.zeros(N)
    for i in range(N):
        for k in range(N):
            if i + k < N:
                grad[i] += weights[k] * (1 - h[i + k])
            if i - k >= 0:
                grad[i] -= weights[k] * h[i - k]
    return DX * grad

def max_lag_gradient(h, corr):
    """Compute exact sub-gradient for the max-over-shifts"""
    N = len(h)
    idx_best = np.argmax(corr)
    grad = np.zeros(N)
    
    # If idx_best is positive, then we shift h forward
    if idx_best < N:
        # We shift the current array left by idx_best positions
        for i in range(N - idx_best):
            grad[i] += (1 - h[i + idx_best]) * DX
        for i in range(idx_best, N):
            grad[i] -= h[i - idx_best] * DX
    else:
        # If idx_best >= N, we shift the current array backward
        shift = idx_best - N + 1
        for i in range(shift, N):
            grad[i] += (1 - h[i - shift]) * DX
        for i in range(shift):
            grad[i] -= h[N - shift + i] * DX
    
    return grad

def mass_transfer(h, grad, epsilon=0.01):
    """Perform greedy mass transfer to improve solution"""
    N = len(h)
    d = -grad
    sources = [(i, -d[i]) for i in range(N) if d[i] < -1e-10]
    sinks = [(i, d[i]) for i in range(N) if d[i] > 1e-10]
    sources.sort(key=lambda x: x[1])
    sinks.sort(key=lambda x: x[1], reverse=True)
    
    # Create copies for transfer
    hcopy = h.copy()
    sources_copy = sources.copy()
    sinks_copy = sinks.copy()
    
    # Apply transfers
    for i, _ in sources_copy:
        for j, _ in sinks_copy:
            if i == j:
                continue
            delta = min(hcopy[i], 1 - hcopy[j], epsilon)
            if delta > 1e-10:
                hcopy[i] -= delta
                hcopy[j] += delta
                epsilon -= delta
                if epsilon <= 1e-10:
                    break
            if epsilon <= 1e-10:
                break
        if epsilon <= 1e-10:
            break
    
    return hcopy

def backtracking_line_search(h, grad, corr, beta, func_available=True, max_attempts=50):
    """Back-tracking line search for gradient descent"""
    alpha = 0.2
    alpha_min = 1e-4
    cmax = corr.max()
    
    while alpha >= alpha_min:
        h_new = h - alpha * grad
        h_new = project_sum_box(h_new, TARGET_SUM)
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        cmax_new = corr_new.max()
        
        if cmax_new < cmax:
            return h_new, corr_new, True
        alpha *= 0.5
    
    # If we couldn't make progress, do a mass transfer step
    h_new = mass_transfer(h, grad, epsilon=0.005)
    h_new = project_sum_box(h_new, TARGET_SUM)
    corr_new = correlate(h_new, 1 - h_new, mode='full')
    cmax_new = corr_new.max()
    
    return h_new, corr_new, False

def stochastic_perturbation(h, T):
    """Perform stochastic perturbation for escape from local minima"""
    N = len(h)
    i = np.random.randint(0, N)
    j = np.random.randint(0, N)
    delta = np.random.uniform(0, min(h[i], 1 - h[j], T * 0.1))
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    h_new = project_sum_box(h_new, TARGET_SUM)
    return h_new
