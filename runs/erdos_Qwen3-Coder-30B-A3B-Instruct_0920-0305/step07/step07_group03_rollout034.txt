import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from scipy.optimize import differential_evolution

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_initial_population(n_pop, n_dim):
    """Generate diverse initial population"""
    pop = []
    for _ in range(n_pop):
        # Mix of initialization strategies
        mode = np.random.choice([
            lambda: adaptive_sampling_initialization(),
            lambda: np.random.uniform(0, 1, n_dim),
            lambda: np.random.choice([0, 1], size=n_dim, p=[0.4, 0.6]),
            lambda: np.clip(np.random.normal(0.5, 0.2, n_dim), 0, 1),
        ])
        ind = mode()
        pop.append(ind)
    return pop

def objective_with_regularization(h):
    """Objective with regularization to avoid trivial solutions"""
    c5 = evaluate_c5(h)
    # Add small penalty for too uniform solutions
    var = np.var(h)
    regularization = 0.01 * np.exp(-var * 10)
    return c5 + regularization

def optimize_with_de(h0, max_iter=100):
    """Use differential evolution for global optimization"""
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return objective_with_regularization(h)
    
    try:
        res = differential_evolution(
            objective, 
            bounds, 
            maxiter=max_iter, 
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=42
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
        return h
    except:
        return h0

def iterate_improvement(h0, max_iter=1000):
    """Iteratively improve the solution using local search and function evaluations"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Try small changes to elements
        idxs = np.random.choice(n_points, size=min(8, n_points//6), replace=False)
        improved = False
        for idx in idxs:
            old_val = h[idx]
            # Try a few small changes
            for delta in [-0.02, -0.01, 0.01, 0.02]:
                h_new = h.copy()
                h_new[idx] += delta
                
                if h_new[idx] < 0 or h_new[idx] > 1:
                    continue
                    
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                new_c5 = evaluate_c5(h_new)
                
                if new_c5 < best_c5:
                    h = h_new.copy()
                    best_c5 = new_c5
                    improved = True
                    break
            if improved:
                break
        if not improved:
            # Add noise to escape local minimum

            best_c5 = evaluate_c5(h)

        if i % 100 == 0 and i > 0:
            h = optimize_with_de(h, max_iter=50)
            best_c5 = evaluate_c5(h)
    
    return h

            h_improved = iterate_improvement(best_h.copy(), max_iter=max_local_iter)

                best_h = h_improved.copy()
