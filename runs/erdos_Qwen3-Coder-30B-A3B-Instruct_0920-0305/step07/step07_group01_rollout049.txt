```python
import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

# Global variables
n_points = 200
dx = 2.0 / n_points
target_sum = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def fourier_basis(x, M):
    """Generate Fourier basis functions for x in [0,2]"""
    basis = np.ones((len(x), M+1))
    for m in range(1, M+1):
        basis[:, m] = np.cos(m * np.pi * x)
    return basis

def reconstruct_h_from_coeffs(coeffs, M=5):
    """Reconstruct h from Fourier coefficients"""
    x = np.linspace(0, 2, n_points)
    basis = fourier_basis(x, M)
    
    # Total function
    h = basis @ coeffs
    
    # Project to [0,1] constraint
    h = np.clip(h, 0, 1)
    
    # Ensure integral = 1
    integral = np.sum(h) * dx
    h = h * (target_sum / integral)
    
    return h

def objective_with_penalty(coeffs, M=5, penalty_weight=1e6):
    """Objective function with penalty for constraint violations"""
    x = np.linspace(0, 2, n_points)
    basis = fourier_basis(x, M)
    h = basis @ coeffs
    
    # Apply bounds
    h = np.clip(h, 0, 1)
    
    # Integral constraint
    integral = np.sum(h) * dx
    integral_penalty = 0 if abs(integral - 1.0) < 1e-8 else penalty_weight * (abs(integral - 1.0))
    
    # Evaluate c5
    c5 = evaluate_c5(h)
    
    return c5 + integral_penalty

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess with small number of components
    M = 6
    # Initial coefficients: [constant, cos(πx), cos(2πx), ...]
    initial_coeffs = [0.5] + [0.0] * M  # First element is intercept, enforce integral=1
    
    # Parameter bounds for differential evolution
    bounds = [(0.0, 1.0)] + [(-1.0, 1.0)] * M  # Bounds: 0,1 for constant, -1,1 for others
    
    best_c5 = float('inf')
    best_coeffs = None
    
    # Run differential evolution for a while
    remaining_time = budget_s
    
    while remaining_time > 10 and best_c5 > 0.378:
        # Limit the number of function evaluations based on time remaining
        max_evals = min(10000, int(remaining_time * 200))
        
        result = differential_evolution(
            lambda x: objective_with_penalty(x, M),
            bounds,
            maxiter=max_evals,
            popsize=10,
            strategy='best1bin',
            seed=seed,
            disp=False,
            atol=1e-6,
            ftol=1e-6
        )
        
        if result.success:
            candidate_coeffs = result.x
            candidate_h = reconstruct_h_from_coeffs(candidate_coeffs, M)
            candidate_c5 = evaluate_c5(candidate_h)
            
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_coeffs = candidate_coeffs.copy()
                
        # Break if time is up
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 30:
            break
    
    # If better bounds can be found, try increasing dimension
    if best_c5 > 0.377 and remaining_time > 60:
        M2 = 8
        bounds2 = [(0.0, 1.0)] + [(-1.0, 1.0)] * M2
        result2 = differential_evolution(
            lambda x: objective_with_penalty(x, M2),
            bounds2,
            maxiter=min(15000, int(remaining_time * 150)),
            popsize=15,
            strategy='best1bin',
            seed=seed,
            disp=False,
            atol=1e-6,
            ftol=1e-6
        )
        
        if result2.success:
            candidate_coeffs = result2.x
            candidate_h = reconstruct_h_from_coeffs(candidate_coeffs, M2)
            candidate_c5 = evaluate_c5(candidate_h)
            
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_coeffs = candidate_coeffs.copy()
                M = M2

    # Final refinement step using gradient-based method
    if best_coeffs is not None:
        final_h = reconstruct_h_from_coeffs(best_coeffs, M)
        final_c5 = evaluate_c5(final_h)
        
        # Local refinement with bounds
        def local_objective(coeffs):
            return evaluate_c5(reconstruct_h_from_coeffs(coeffs, M))
        
        try:
            bounds_local = [(-1.0, 1.0)] * len(best_coeffs)
            result_ref = differential_evolution(
                local_objective,
                bounds_local,
                maxiter=500,
                popsize=10,
                strategy='best1bin',
                seed=seed,
                disp=False
            )
            if result_ref.success:
                final_h_ref = reconstruct_h_from_coeffs(result_ref.x, M)
                final_c5_ref = evaluate_c5(final_h_ref)
                if final_c5_ref < final_c5:
                    final_c5 = final_c5_ref
                    final_h = final_h_ref
        except:
            pass

    # Return as needed by the problem interface
    return (final_h.tolist(), float(final_c5), n_points)
```
