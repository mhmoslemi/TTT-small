import numpy as np
import cvxpy as cp
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 140
dx = 2.0 / n_points

def build_A_matrices(n):
    """Build sparse matrices A_ell for all lags ell = -(n-1) to n-1"""
    A_list = []
    for lag in range(-(n-1), n):
        # Create sparse matrix for lag
        rows = []
        cols = []
        data = []
        for i in range(n):
            j = i + lag
            if 0 <= j < n:
                rows.append(i)
                cols.append(j)
                data.append(1)
        A = np.zeros((n, n))
        A[rows, cols] = data
        A_list.append(A)
    return A_list

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_sdp(A_matrices):
    """Solve the SDP relaxation. Returns (X_opt, t_opt)"""
    n = A_matrices[0].shape[0]
    X = cp.Variable((n, n), PSD=True)
    t = cp.Variable()
    
    constraints = [
        cp.diag(X) >= 0,
        cp.diag(X) <= 1,
        cp.sum(cp.diag(X)) == n/2,
    ]
    
    # Add constraints for all A_ell matrices
    for ell, A in enumerate(A_matrices):
        # trace(A_ell * X) >= n/2 - t/dx
        # This is equivalent to trace(A_ell * X) + t/dx >= n/2
        constraints.append(cp.trace(A @ X) >= n/2 - t/dx)
        
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return X.value, t.value
    except:
        return None, None
    return None, None

def random_round_sdp(X_opt, n, num_samples=2000):
    """Sample feasible vectors from PSD matrix for random rounding"""
    # Compute eigendecomposition: X ≈ v v^T for some vector v (assuming rank 1 approximation)  
    evals, evecs = np.linalg.eigh(X_opt)
    # Take the principal (largest) eigenvector
    v = evecs[:, -1]
    
    # Generate multiple samples via Gaussian sampling with covariance X_opt
    best_h = None
    best_c5 = float('inf')
    
    for _ in range(num_samples):
        # Sample z ~ N(0, I)
        z = np.random.randn(n)
        # y = L z where X = L L^T (Cholesky)
        try:
            L = np.linalg.cholesky(X_opt)
            y = L @ z
        except:
            # Fallback using eigendecomposition: X = V D V^T, y = V sqrt(D) z
            y = evecs @ np.sqrt(np.maximum(evals, 0)) * z
        
        # Clip and project
        h_raw = y
        h_clip = np.clip(h_raw, 0, 1)
        h = project_to_sum_box(h_clip, n/2)
        
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

            h_minus[i] -= eps

                    best_h[j] += delta
                    best_h = project_to_sum_box(best_h, TARGET_SUM)
