import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 180
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_grad_max_shift(h, shift, dx):
    """Compute gradient of C5 for the specific shift that achieves max."""
    n = len(h)
    g = np.zeros(n)
    
    if shift >= 0:
        # Positive shift case: h[i] * (1 - h[i+shift])
        # Derivative w.r.t. h[i]: (1 - h[i+shift]) if i+shift < n, else 0
        # Derivative w.r.t. h[i+shift]: -h[i] if i < n-shift, else 0
        if shift < n:
            g[0:n-shift] += (1 - h[shift:]) * dx
            g[shift:] += -h[:n-shift] * dx
    else:
        # Negative shift case: h[i] * (1 - h[i+shift]) 
        # Derivative w.r.t. h[i]: (1 - h[i+shift]) if i+shift >= 0, else 0
        # Derivative w.r.t. h[i-shift]: -h[i] if i-shift < n, else 0
        shift_abs = -shift
        if shift_abs < n:
            g[shift_abs:] += (1 - h[:n-shift_abs]) * dx
            g[:n-shift_abs] += -h[shift_abs:] * dx
    
    # Return gradient direction to move h to decrease max C5
    return -g
