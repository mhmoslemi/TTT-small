def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr)

def fourier_basis(x, coeffs, n_modes=10):
    """Evaluate h(x) using cosine basis functions"""
    h = 0.5 + np.sum(coeffs[i] * np.cos((i+1) * np.pi * x) for i in range(n_modes))
    return np.clip(h, 0, 1)
