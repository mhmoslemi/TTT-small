import numpy as np
import cvxpy as cp
import time

    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    # Variables
    h = cp.Variable(n_points, name="h")
    P = cp.Variable((n_points, n_points), name="P")
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == TARGET_SUM
    ]
    
    # McCormick envelope constraints for all pairs
    for i in range(n_points):
        for j in range(n_points):
            constraints.extend([
                P[i, j] >= 0,
                P[i, j] >= h[i] + h[j] - 1,
                P[i, j] <= h[i],
                P[i, j] <= h[j]
            ])
    
    # Objective: minimize maximum correlation
    # Compute all correlation terms for different shifts
    # We want to maximize min overlap, or minimize max correlation
    # correlation_k = sum_i h_i - sum_{i, i+k} P[i, i+k] 
    # max_k correlation_k 
    # we use a variable t such that t >= correlation_k for all k
    
    # For consistency with the earlier implementation, compute using
    # full convolution and then take maximum
    
    # Create a single slack variable to bound max correlation, 
    # but for direct LP solve, we'll mimic the correlation structure:
    # Actually let's directly optimize max_k (S - overlap_k) where overlap_k = sum of P[i, i+k]
    
    # The final solution will be a continuous h vector that minimizes max_k correlation_k
    # We will evaluate actual C5 with numpy for reporting
    
    # Since our combination is LP directly solvable:
    # minimize max_k (S - sum_i P[i, i+k]) for all k
    # => minimize max_k (S - sum_{i | i+k < n} P[i, i+k]) for all k
    # Which is equivalent to maximize min_k sum_{i | i+k < n} P[i, i+k]
    
    # Instead let's penalize the max correlation directly:
    # introduce t for max correlation, constraints t >= (S - sum_{i+k < n} P[i, i+k) for all k.
    
    t = cp.Variable()

    shift_indices = range(n_points)

        overlap_indices = list(range(n_points - k))
        # Compute correlation with this shift
        corr_sum = cp.sum([P[i, i + k] for i in overlap_indices])
        corr = TARGET_SUM - corr_sum
        constraints.append(t >= corr)
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)

    remaining_time = budget_s - solver_time

            prob.solve(solver=cp.GLPK, verbose=False, max_seconds=remaining_time)
        except:
