import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    # Create CVXPY variables
    h = cp.Variable(n_points, name="h")
    P = cp.Variable((n_points, n_points), name="P")
    
    # Constraints
    constraints = [
        cp.sum(h) == TARGET_SUM,
        h >= 0,
        h <= 1
    ]
    
    # Add McCormick constraints for all pairs
    for i in range(n_points):
        for j in range(n_points):
            constraints.extend([
                P[i, j] >= h[i] + h[j] - 1,
                P[i, j] <= h[i],
                P[i, j] <= h[j],
                P[i, j] >= 0
            ])
    
    # Objective: minimize maximum correlation
    # correlation_k = sum_i h_i - sum_{i: i+k < n} h_i h_{i+k}
    # We'll compute all correlations and minimize the maximum
    max_corr = cp.Variable()

    # For each shift k, compute correlation (linearized)
    correlations = []
    for k in range(n_points):
        # For shift k, overlapping indices are from 0 to n-k-1
        corr_expr = 0
        for i in range(n_points - k):
            corr_expr += h[i] - P[i, i + k]
        correlations.append(corr_expr)
    
    # Add constraint that max_corr is greater than or equal to all correlations
    for corr in correlations:
        constraints.append(max_corr >= corr)
    
    # Objective: minimize max correlation
    objective = cp.Minimize(max_corr)
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    solver_options = {"max_seconds": budget_s * 0.95}
    try:
        prob.solve(solver=cp.GLPK, verbose=False, **solver_options)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            actual_c5 = max_correlations(h_opt, n_points, dx)
            return (h_opt.tolist(), float(actual_c5), n_points)
        else:
            # Fallback to previous approach if optimization fails
            pass
    except Exception:
        pass

    # Fallback to heuristic method if LP fails
    return fallback_heuristic(seed, budget_s)

    # Convert to numpy and ensure proper size

    # Compute correlation of h with (1-h) using full convolution
    corr = np.correlate(h, 1 - h, mode="full")
    
    # The maximum correlation occurs at the central point in the correlation array
    # due to zero padding
    max_corr_idx = len(corr) // 2
    max_corr_value = corr[max_corr_idx]

    # Adjust index to get proper maximum - we compute all correlations
    # and find the maximum
    max_corr = np.max(corr) * dx  # Scale by dx
    
    return max_corr
