# Problem setup
n_points = 140
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def build_autocorrelation_matrices(n):
    """Build sparse matrices A_ell for all lags"""
    A_list = []
    for ell in range(-(n-1), n):
        # Create matrix A_ell that picks elements h_i * h_{i+ell}
        rows = []
        cols = []
        data = []
        for i in range(n):
            j = i + ell
            if 0 <= j < n:
                rows.append(i)
                cols.append(j)
                data.append(1.0)
        
        # Construct sparse matrix
        A_ell = np.zeros((n,n))
        A_ell[np.array(rows), np.array(cols)] = data 
        A_list.append(A_ell)
        
    return A_list

def sdpa_rounding(X, n, target_sum):
    """Rounding step using principal eigenvector"""
    evals, evecs = eigh(X)
    # Use the largest eigenvector, which is most likely to be the closest to rank-1
    v = evecs[:, -1]
    h_raw = np.abs(v)
    h_clip = np.clip(h_raw, 0, 1)
    
    # Project to sum constraint
    def project_to_sum(h, target_sum):
        def f(lam):
            projected = np.clip(h + lam, 0, 1)
            return np.sum(projected) - target_sum
        
        a, b = -1.0, 1.0
        for _ in range(50):
            m = (a + b) / 2
            fm = f(m)
            if abs(fm) < 1e-12:
                break
            if fm * f(a) < 0:
                b = m
            else:
                a = m
            
        lam = (a + b) / 2
        return np.clip(h + lam, 0, 1)
    
    h = project_to_sum(h_clip, target_sum)
    return h

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)
