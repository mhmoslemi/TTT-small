import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_c5(h):
    """Compute C5 using FFT for speed"""
    n = len(h)
    # Use real FFT for efficiency
    nfft = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    B = np.fft.rfft(1 - h, n=nfft)
    conv = np.fft.irfft(H * B, n=nfft)[:2*n-1]
    return np.max(conv) * dx

def random_binary_initial(n):
    """Create random binary initial solution"""
    h = np.zeros(n)
    ones_idx = np.random.choice(n, n//2, replace=False)
    h[ones_idx] = 1.0
    return h

def greedy_improvement(h):

        corr = np.correlate(h, 1 - h, mode='full')

        # Determine which indices are involved in the worst case
        if max_shift >= 0:
            i_range = np.arange(0, n - max_shift)
            j_range = i_range + max_shift
        else:
            i_range = np.arange(-max_shift, n)
            j_range = i_range + max_shift
            
        # Find mismatched pairs
        h_i = h[i_range]
        h_j = h[j_range]
        
        # (1,0) mismatches: h_i == 1 and h_j == 0  
        im1 = i_range[(h_i == 1) & (h_j == 0)]
        # (0,1) mismatches: h_i == 0 and h_j == 1
        im2 = i_range[(h_i == 0) & (h_j == 1)]
        
        if im1.size == 0 or im2.size == 0:
            break
            
        # Pick random pair and swap
        i = np.random.choice(im1)
        j = np.random.choice(im2)
        
        h[i] = 0
        h[j] = 1
        
        iter_count += 1
        
    return h

            return np.sum(h_local) - TARGET_SUM
            
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
    except:
        pass

    h = np.clip(h, 0, 1)

        else:
            for i in range(int(-diff)):
                if idx[-(1+i)] > 0 and h[idx[-(1+i)]] > 0:
                    h[idx[-(1+i)]] -= 1
    
    return h
