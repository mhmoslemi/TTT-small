TARGET_SUM = None

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smart_perturb_solution(h, strength=0.01):
    """Create a more intelligent perturbed version of the solution"""
    # Create perturbation with structured variation
    perturbed = h + np.random.normal(0, strength, n_points)
    
    # Apply a small amount of projection to avoid too extreme changes
    perturbed = h + 0.3 * (perturbed - h)
    
    return project_to_sum_box(perturbed, TARGET_SUM)

            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = i
                
    return best_h, best_c5
