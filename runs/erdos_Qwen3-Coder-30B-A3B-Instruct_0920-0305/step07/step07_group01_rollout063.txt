import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * 2.0 / len(h)

def project_sum_box(h, target_sum):
    """Project vector h onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(h + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def smooth_gradient(h, corr, beta):
    """Compute gradient of smooth max approximation"""
    # Numerical stability: subtract max
    corr_centered = corr - np.max(corr)
    exp_corr = np.exp(beta * corr_centered)
    weights = exp_corr / np.sum(exp_corr)
    # The gradient is dx * [1 - sum_{l} w_l * (h_{i+l} + h_{i-l})]
    # Since we're using full correlation, shift the weights appropriately
    grad = np.zeros_like(h)
    n = len(h)
    
    # Compute weights contribution, sum over all lags
    # This is equivalent to dx * [1 - sum_{l} w_l * (h_{i+l} + h_{i-l})] for valid indices
    for l in range(-n+1, n):
        if l < 0:
            # h_{i+l} where i+l >= 0 and h_{i-l} where i-l < n
            for i in range(-l, n):
                if i-l < n:
                    grad[i] -= weights[l+n-1] * h[i+l]
                if i-l >= 0:
                    grad[i] -= weights[l+n-1] * h[i-l]
        elif l > 0:
            for i in range(0, n-l):
                if i+l < n:
                    grad[i] -= weights[l+n-1] * h[i+l]
                if i-l >= 0:
                    grad[i] -= weights[l+n-1] * h[i-l]
        else:
            # l == 0 case
            for i in range(0, n):
                grad[i] -= weights[l+n-1] * h[i+l]  # h[i+0] = h[i] only
                grad[i] -= weights[l+n-1] * h[i-l]  # h[i-0] = h[i] only

    return 2.0/len(h) * (np.ones_like(h) - grad)

def max_lag_gradient(h, corr):
    """Compute exact sub-gradient of maximum correlation. """
    max_lag_index = np.argmax(corr)
    # Convert from full correlation indexing to standard index
    n = len(h)
    shift = max_lag_index - (n-1)
   
    grad = np.zeros_like(h)
    # Only the non-zero terms
    if shift >= 0:
        # This means the lag is positive
        # For i >= shift, the term (1-h_{i-shift}) contributes to the gradient
        # This is handled by the shift direction
        for i in range(shift, len(h)):
            grad[i] += 2.0/n * (-h[i-shift])  # The 2.0/n factor
    else:
        # shift < 0, lag is negative, both parts
        # For i in range(0, len(h) + shift), h_{i-shift} contributes to the gradient
        for i in range(0, len(h) + shift):
            grad[i] += 2.0/n * (-h[i-shift])
        # For i in range(-shift, len(h)), (1-h_{i+shift}) contributes to the gradient
        # which affects the term for index i -> (1-h_{i+shift})
        # But here's the key insight: we're computing the derivative of max_lag_i \int h(x) (1 - h(x+k)) dx
        # for that particular k = shift. Gradient w.r.t. h_i is (1-h_{i+shift}) if i < len(h)-shift
        # and -h_{i-shift} (i >= shift) - (1-h_{i-shift}) is computed in the other case
        for i in range(0, len(h)):
            if i - shift < len(h):
                # gradient of term h_i * (1 - h_{i+shift}) wrt h_i
                # is (1 - h_{i+shift})
                # We apply to grad_i the correction
                grad[i] += 2.0/n * (1 - h[i-shift])
    
    # Slight correction: to match the original convention:
    # Let's be more careful:
    # For above we have full correlation = correlation(h, 1-h, mode='full')
    # It has len 2*n - 1
    # index `k = i - n + 1` in 'self' corresponds to shift = k
    # So shift = argmax(corr) - (n-1)
    # and then it matters for which i the term is non-zero
    # Correlation lags go from -n+1 to n-1
    
    # Let's correct more carefully:
    shift_ = np.argmax(corr) - (n-1)
    # shift_ may be positive or negative or 0, indicating shift of h_i to h_{i+shift_}
    # gradient of sum h_i (1 - h_{i+shift_}) w.r.t h_i is (1 - h_{i+shift_}), i < len(h)-shift_
    # gradient of sum h_i (1 - h_{i+shift_}) w.r.t h_{i-shift_} is -h_{i-shift_}, i >= shift_
    # This is for w.r.t h_i = 1 - h_{i+shift_}  (gradient of term + d_i) and -h_{i-shift_} (d_i)
    grad = np.zeros_like(h)
   
    # Now compute a cleaner version:
    # This is the dp/dh_i derivative for shifted overlap
    if shift_ >= 0:
        # Positive shift. So we need to count terms h[i] * (1 - h[i-shift_])  
        # gradient w.r.t. h[i] is (1 - h[i-shift_]) if i+shift_ < n = i < n-shift_
        for i in range(n-shift_):
            grad[i] += 2.0/n * (1 - h[i+shift_])
        # Gradient w.r.t h[i+shift_] should be -h[i]
        # But we're assuming upper bound for i = i-shift_, the range of i is i衩 = 0, ... shift_-1
        #  
        # No. let me write the correct expressions:
        # You sum over i: h_i * (1-h_{i+k})
        # Then d/dh_i = (1 - h_{i+k})
        # The correlation sum over shift from -n+1 to n-1
        # which is h_0 (1 - h_{k}) + h_1 (1 - h_{1+k}) + ..., but now k=shift_
        # The nieve version was wrong - what we're differentiating wrt:
        # g = sum_i h[i] * (1 - h[i+shift_]) for shift_ = shift
        # that's automatically 0 if there's any out-of-bounds 
        # All i such that i + shift_ < n  contribute the term (1 - h[i+shift_])
        # All i such that i-shift_ >= 0 contribute - h[i-shift_]
        for i in range(n-shift_):
            grad[i] += 2.0/n * (1 - h[i+shift_])
        for i in range(shift_):
            grad[i-shift_] -= 2.0/n * h[i]
    else:
        # Negative shift k = shift_ < 0, so Lag is negative, need to reverse
        shift_neg = -shift_
        for i in range(shift_neg):
            grad[i] += 2.0/n * (1 - h[i-shift_neg])
        for i in range(shift_neg, n):
            grad[i] -= 2.0/n * h[i-shift_neg]
        
    return grad

def mass_transfer(h, grad, eps=0.01):
    """Perform greedy mass transfer moves."""
    # We want to move mass: take negative grad components (decrease h_i) to positive grad components (increase h_i)
    # max rem = h_i = 1 - h_i to add
    d = -grad
    sources = np.where(d < 0)[0]
    sinks = np.where(d > 0)[0]
    
    # Sort for greedy selection
    sort_sources = np.argsort(-d[sources])
    sort_sinks = np.argsort(-d[sinks])

    sources = sources[sort_sources]
    sinks = sinks[sort_sinks]
    
    # Calculate transfer amounts
    h_rem = h.copy()
    h_add = 1.0 - h.copy()
    h_new = h.copy()
    
    # Total amount to transfer
    total = eps * np.sum(h)
    
    total_transferred = 0
    for s in sources:
        if total_transferred >= total:
            break
        for t in sinks:
            if total_transferred >= total:
                break
            # Min of what we have, what we want to add, and eps limit
            transfer = min(h_rem[s], h_add[t], total - total_transferred)
            if transfer <= 0:
                continue
            
            h_new[s] -= transfer
            h_new[t] += transfer
            h_rem[s] -= transfer
            h_add[t] -= transfer
            total_transferred += transfer
    
    # Apply the projection step again
    h_new = project_sum_box(h_new, np.sum(h) + 0.0)
    return h_new

def line_search(h, grad, beta, corr, alpha0=0.2, alpha_min=5e-4):
    """Backtracking line search"""
    dx = 2.0 / len(h)
    c0 = evaluate_c5(h)
    # Compute the direction
    alpha = alpha0
    while alpha >= alpha_min:
        h_new = h - alpha * grad
        h_new = project_sum_box(h_new, np.sum(h))
        c_new = evaluate_c5(h_new)
        if c_new < c0:
            return h_new, c_new, alpha
        alpha /= 2
    return h, c0, 0.0
