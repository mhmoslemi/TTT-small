import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import math

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def c5_fft(h):
    # linear (non-circular) convolution of h with (1-h)
    n = len(h)
    nfft = 1 << ((2*n-1).bit_length())          # next power of two
    H = np.fft.rfft(h, nfft)
    B = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * B, nfft)[:2*n-1]   # length 2N-1
    return np.max(conv) * dx

def project_to_sum(h, target_sum):
    """Project h onto the constraint sum(h) = target_sum"""
    # Water-filling approach
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if abs(current_sum - target_sum) < 1e-10:
        return h_clipped
    
    # Binary search for lambda
    a, b = -1.0, 1.0
    tol = 1e-12
    for _ in range(100):
        lam = (a + b) / 2
        h_test = np.clip(h_clipped + lam, 0, 1)
        test_sum = np.sum(h_test)
        if abs(test_sum - target_sum) < tol:
            return h_test
        if test_sum < target_sum:
            a = lam
        else:
            b = lam

    # Fallback
    h_final = np.clip(h_clipped + (a + b) / 2, 0, 1)
    return h_final

def thue_morse(n):
    """Generate Thue-Morse sequence of length n"""
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    return tm

def rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence of length n"""
    rs = np.zeros(n, dtype=int)
    for i in range(n):
        # Count number of overlapping 11 patterns in binary of i
        count = 0
        j = i
        while j > 0:
            if (j & 3) == 3:
                count += 1
            j >>= 1
        rs[i] = count % 2
    return rs

def initialize_h(seed):
    """Generate an initial binary vector of exactly TARGET_SUM ones"""
    np.random.seed(seed)
    # Random binary start
    h0 = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
    current_sum = np.sum(h0)
    
    # Adjust to exactly TARGET_SUM
    if current_sum > TARGET_SUM:
        # Need to flip some 1's to 0's
        ones = np.where(h0 == 1)[0]
        flip_indices = np.random.choice(ones, int(current_sum - TARGET_SUM), replace=False)
        h0[flip_indices] = 0
    elif current_sum < TARGET_SUM:
        # Need to flip some 0's to 1's
        zeros = np.where(h0 == 0)[0]
        flip_indices = np.random.choice(zeros, int(TARGET_SUM - current_sum), replace=False)
        h0[flip_indices] = 1

    return h0

def greedy_descent(h0, max_iter=None):
    """Perform deterministic greedy descent"""
    h = h0.copy()
    c5 = c5_fft(h)
    
    if max_iter is None:
        max_iter = 2 * n_points
    
    for _ in range(max_iter):
        # Compute correlation function
        corr = correlate(h, 1 - h, mode='full') * dx
        max_idx = np.argmax(corr)
        max_shift = max_idx - (len(h) - 1)
        max_val = corr[max_idx]

        if max_val <= c5:
            break  # No improvement

        # Find mismatched pairs for the worst shift
        if max_shift >= 0:
            # h[i] and h[i + max_shift] should be (1,0) for maximum shift
            # Find indices where (1,0) and (0,1) patterns occur
            pairs_10 = np.where((h[:-max_shift] == 1) & (h[max_shift:] == 0))[0]
            pairs_01 = np.where((h[:-max_shift] == 0) & (h[max_shift:] == 1))[0]
        else:
            # Case for negative shift
            max_shift = -max_shift
            pairs_10 = np.where((h[max_shift:] == 1) & (h[:-max_shift] == 0))[0]
            pairs_01 = np.where((h[max_shift:] == 0) & (h[:-max_shift] == 1))[0]

        if len(pairs_10) == 0 or len(pairs_01) == 0:
            break  # No improvement possible with single swaps

        # Pick random pair to swap
        i = np.random.choice(pairs_10)
        j = i + max_shift if max_shift > 0 else i - max_shift
        if len(pairs_01) > 0:
            j = np.random.choice(pairs_01)
            j = j if max_shift > 0 else j + max_shift

        # Perform swap to fix mismatch
        h[i], h[j] = h[j], h[i]
        c5 = c5_fft(h)
        
    return h, c5

def simulated_annealing(h_init, max_time, verbose=False):
    """Run simulated annealing on binary vectors."""
    T0 = 0.02
    start_time = time.time()
    current_h = h_init.copy()
    current_c5 = c5_fft(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5

    # Reset after 5000 attempts without improvement
    stable_count = 0
    last_improvement = time.time()

    # Temperature schedule over time
    while time.time() - start_time < max_time:
        # Compute current temperature
        elapsed = time.time() - start_time
        if elapsed < 0.01:
            continue
            
        T = T0 * (1.0 - elapsed / max_time)  # Linear cooling
        if T < 1e-5:
            break

        # Pick two random indices
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)

        if current_h[i] == current_h[j]:
            continue  # No useful swap

        # Temporarily swap
        current_h[i], current_h[j] = current_h[j], current_h[i]
        new_c5 = c5_fft(current_h)

        # Accept or reject the swap
        delta_E = new_c5 - current_c5
        if delta_E <= 0 or np.random.rand() < np.exp(-delta_E / T):
            # Accept the move
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
                stable_count = 0
                last_improvement = time.time()
        else:
            # Reject the move
            current_h[i], current_h[j] = current_h[j], current_h[i]
            stable_count += 1

        # Sometimes trigger local refinement
        if stable_count % 500 == 0 and time.time() - start_time < max_time:
            h_local, c5_local = greedy_descent(current_h, 10)  # Small local descent
            if c5_local < current_c5:
                current_h = h_local.copy()
                current_c5 = c5_local
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
                stable_count = 0
                last_improvement = time.time()

        # Ensure monotonicity
        if time.time() - last_improvement > 20:
            # Reset the temperature
            stable_count = 0
        
        if time.time() - start_time > max_time:
            break

    return best_h, best_c5

def fine_tune_continuous(h_binary, maxtime):
    """Use continuous optimization to refine fractional values"""
    if maxtime < 5:
        return h_binary, c5_fft(h_binary)

    # Flatten to float
    h0 = h_binary.astype(float)
    
    # Apply smoothing
    kernel = np.array([0.25, 0.5, 0.25])
    h_smooth = np.convolve(h0, kernel, mode='same')
    h0 = np.clip(h_smooth, 0, 1)
    
    # Project exactly to required sum
    h0 = project_to_sum(h0, TARGET_SUM)

    def obj_func(h):
        return c5_fft(h)

    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM

    bounds = [(0, 1) for _ in range(len(h0))]
    cons = {'type': 'eq', 'fun': constraint_eq}

    try:
        res = minimize(
            obj_func,
            h0,
            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'gtol': 1e-8, 'ftol': 1e-10}
        )
        if res.success:
            h_final = project_to_sum(res.x, TARGET_SUM)
            c5_final = c5_fft(h_final)
            if c5_final < c5_fft(h_binary):
                return h_final, c5_final
    except:
        pass
    
    return h_binary, c5_fft(h_binary)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    TARGET_SUM = n_points / 2

    # Initial generation from seeds
    # Generate multiple candidates from different seeds
    best_c5 = float('inf')
    best_h = None

    # Use different seeds for initialization
    seeds = [0, 1, 2, 3]
    restarts = len(seeds) * 2  # Because we'll do both greedy and SA

    for i, s in enumerate(seeds):
        h0 = initialize_h(s)
        # Try both regular and Thue-Morse
        if i % 2 == 0:
            h0 = h0
        else:
            # Try Thue-Morse vector
            tm = thue_morse(n_points)
            tm_sum = np.sum(tm)
            # adjust if not correct sum
            h0 = tm.copy()
            if tm_sum > TARGET_SUM:
                zeros = np.where(h0 == 0)[0]
                flip_indices = np.random.choice(zeros, int(tm_sum - TARGET_SUM), replace=False)
                h0[flip_indices] = 1
            elif tm_sum < TARGET_SUM:
                ones = np.where(h0 == 1)[0]
                flip_indices = np.random.choice(ones, int(TARGET_SUM - tm_sum), replace=False)
                h0[flip_indices] = 0
        
        # Greedy descent to improve the candidate
        h_greedy, c5_greedy = greedy_descent(h0, max_iter=100)
        if c5_greedy < best_c5:
            best_c5 = c5_greedy
            best_h = h_greedy.copy()
        
        # Simulated annealing from this point
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time > 5 and i < len(seeds) - 1:
            # Do simulated annealing or skip if time is low
            h_sa, c5_sa = simulated_annealing(h_greedy, min(remaining_time * 0.5, 300))
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()

        # Early termination check
        if time.time() - start_time > 0.9 * budget_s:
            break

    # Refine using continuous method if time permits
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        continuous_h, c5_continuous = fine_tune_continuous(best_h, remaining_time)
        if c5_continuous < best_c5:
            best_h = continuous_h
            best_c5 = c5_continuous

    # Final correction to ensure constraints
    final_h = project_to_sum(best_h, TARGET_SUM)
    final_c5 = c5_fft(final_h)
    
    return (final_h.tolist(), float(final_c5), n_points)
