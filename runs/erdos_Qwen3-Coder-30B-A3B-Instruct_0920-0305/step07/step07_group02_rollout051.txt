```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_c5(h):
    """Fast evaluation of C5 using FFT"""
    n = len(h)
    # compute linear convolution via FFT
    nfft = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    B = np.fft.rfft(1 - h, n=nfft)
    conv = np.fft.irfft(H * B, n=nfft)[:2*n-1]
    return np.max(conv) * dx

def greedy_improvement(h):
    """Apply greedy improvement focusing on the worst shift"""
    n = len(h)
    max_iter = 10000
    iter_count = 0
    while iter_count < max_iter:
        # compute correlation and find maximum shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n - 1)
        
        # Determine indices that contribute to overlap at max shift
        if max_shift >= 0:
            i_range = np.arange(0, n - max_shift)
            j_range = i_range + max_shift
        else:
            i_range = np.arange(-max_shift, n)
            j_range = i_range + max_shift
            
        # Find mismatched pairs: h[i] == 1 and h[j] == 0
        h_i = h[i_range]
        h_j = h[j_range]
        im1 = i_range[(h_i == 1) & (h_j == 0)]
        # Find mismatched pairs: h[i] == 0 and h[j] == 1
        im2 = i_range[(h_i == 0) & (h_j == 1)]
        
        if im1.size == 0 or im2.size == 0:
            break
            
        # Pick random mismatched pairs and swap
        i = np.random.choice(im1)
        j = np.random.choice(im2)
        h[i] = 0
        h[j] = 1
        iter_count += 1
    return h

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_gradient_refinement(h0, max_iter=1000):
    """Refine with gradient-based optimization on small subset"""
    h = h0.copy()
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_local):
        return compute_c5(h_local)
    
    def constraint_eq(h_local):
        return np.sum(h_local) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use fewer iterations and local optimizer
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
    except:
        pass
    
    # Project to maintain constraints
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Run multiple annealing runs with different seeds
    num_runs = 10
    for run_id in range(num_runs):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Initialize binary vector with half ones and half zeros
        h = np.zeros(n_points)
        ones_indices = np.random.choice(n_points, int(TARGET_SUM), replace=False)
        h[ones_indices] = 1.0
        
        # Apply greedy improvement
        h = greedy_improvement(h)
        
        # Compute C5
        c5 = compute_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Run final refinement with SLSQP if time allows
    if time.time() - start_time < budget_s * 0.95:
        try:
            # Refine using SLSQP
            refined_h = local_gradient_refinement(best_h.copy(), max_iter=200)
            refined_c5 = compute_c5(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h.copy()
                best_c5 = refined_c5
        except:
            # If SLSQP fails, keep best binary
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
```
