# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_thue_morse(n):
    """Generate Thue-Morse sequence of length n"""
    if n <= 1:
        return np.array([0])
    
    # Start with [0]
    seq = [0]
    while len(seq) < n:
        seq += [1 - x for x in seq]
    
    return np.array(seq[:n], dtype=float)

def generate_rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence of length n"""
    if n <= 1:
        return np.array([0])
    
    seq = [0]
    for i in range(1, n):
        if i % 2 == 0:
            seq.append(seq[i//2])
        else:
            seq.append(1 - seq[i//2])
    
    return np.array(seq[:n], dtype=float)

def greedy_worst_shift_improvement(h0):
    """Improve binary solution using greedy worst-shift swaps"""
    h = h0.copy()
    
    for _ in range(10000):  # Limit iterations
        corr = compute_corr_fft(h)
        max_shift_idx = np.argmax(corr)
        
        # Find the actual shift amount
        max_shift = max_shift_idx - (n_points - 1)
        
        # For a given shift, find mismatched pairs
        # We are looking for positions (i, i+shift) where h[i]=1 and h[i+shift]=0,
        # and (j, j+shift) where h[j]=0 and h[j+shift]=1.
        # A swap at those positions will reduce the overlap.
        
        # Get the slices
        if max_shift >= 0:
            left_part = h[:n_points - max_shift]
            right_part = h[max_shift:]
        else:
            left_part = h[-max_shift:]
            right_part = h[:n_points + max_shift]
        
        # Indices for 1,0 mismatches in the shift
        indices_10 = np.where((left_part == 1) & (right_part == 0))[0]
        indices_01 = np.where((left_part == 0) & (right_part == 1))[0]
        
        if len(indices_10) == 0 or len(indices_01) == 0:
            break  # No more beneficial swaps
        
        # Pick random one from each
        i10 = indices_10[np.random.randint(0, len(indices_10))]
        i01 = indices_01[np.random.randint(0, len(indices_01))]
        
        # Make pair swap
        if max_shift >= 0:
            h[i10], h[i01 + max_shift] = h[i01 + max_shift], h[i10]
        else:
            h[i10 - max_shift], h[i01] = h[i01], h[i10 - max_shift]
        
    return h

def simulated_annealing(h_start):
    """Run simulated annealing on refined binary solution"""
    h_current = h_start.copy()
    c5_current = evaluate_c5(h_current)
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # Temperature schedule: from 0.1 to 0.001 over 10000 steps
    T0 = 0.1
    T_final = 0.001
    steps = 10000
    
    for i in range(steps):
        temp = T0 * (T_final / T0) ** (i / steps)
        
        # Make proposal
        h_proposed = h_current.copy()
        # Select two random indices to swap
        idx1 = np.random.randint(0, n_points)
        idx2 = np.random.randint(0, n_points)
        h_proposed[idx1], h_proposed[idx2] = h_proposed[idx2], h_proposed[idx1]
        # Must project back to valid state
        h_proposed = project_to_sum_box(h_proposed, TARGET_SUM)
        
        c5_proposed = evaluate_c5(h_proposed)
        
        # Metropolis acceptance
        if c5_proposed < c5_current or np.exp(-(c5_proposed - c5_current) / temp) > np.random.rand():
            h_current = h_proposed
            c5_current = c5_proposed
            
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                
        # Early exit if improvement is stationary
        if i > 2000 and abs(c5_current - best_c5) < 1e-8:
            break
    
    return best_h, best_c5

def finalize_solution(h0):
    """Refine solution with constrained optimization"""
    h = h0.copy()
    
    # Convert to continuous with bounds [0,1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_local):
        return evaluate_c5(h_local)
    
    def constraint_eq(h_local):
        return np.sum(h_local) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Optimize with bounds and constraint
    res = minimize(
        objective, 
        h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
    )
    
    if res.success:
        h_final = res.x
    else:
        h_final = h
    
    # Project to sum constraint
    h_final = project_to_sum_box(h_final, TARGET_SUM)
    return h_final

    start_time = time.time()

        # Final refinement
        h_final = finalize_solution(h_anneal)

        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    # Final check on best solution
    assert best_h is not None, "No solution found"
    
    return (best_h.tolist(), float(best_c5), n_points)
