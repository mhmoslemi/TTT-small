import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 180
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_grad_max_shift(h, shift_k, dx):
    """Compute gradient of C5 for specific shift k"""
    n = len(h)
    g = np.zeros(n)
    
    if shift_k >= 0:
        # For positive shift k
        # derivative of h_i * (1 - h_{i+k}) w.r.t h_i is (1 - h_{i+k})
        # derivative of h_{i-k} * (1 - h_i) w.r.t h_i is -h_{i-k}
        if shift_k < n:
            g[0:n-shift_k] += (1 - h[shift_k:]) * dx
            g[shift_k:n] += -h[:n-shift_k] * dx
    else:
        # For negative shift k, treat as positive shift -k
        shift_k_neg = -shift_k
        if shift_k_neg < n:
            g[shift_k_neg:n] += (1 - h[:n-shift_k_neg]) * dx
            g[:n-shift_k_neg] += -h[shift_k_neg:] * dx
    
    return g

    h_interpolated = project_to_sum_box(h_interpolated, n_points/2)
    
    best_h = h_interpolated.copy()
    best_c5 = evaluate_c5(h_interpolated)
    TARGET_SUM = n_points / 2
    
    # Parameters
    max_iter = 30000
    step_size = 0.1
    min_step = 1e-6
    step_decay = 0.999
    tolerance = 1e-6
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation and find worst shift
        corr = correlate(best_h, 1 - best_h, mode='full')
        c5_vals = corr * dx
        k_idx = np.argmax(c5_vals) - (n_points - 1)  # actual shift value
        
        # Compute gradient for the worst shift
        grad = compute_grad_max_shift(best_h, k_idx, dx)
        
        # Line search for appropriate step size
        current_c5 = np.max(c5_vals)
        step = step_size
        
        while step > min_step:
            # Compute candidate solution
            h_candidate = best_h - step * grad
            h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
            
            # Evaluate new C5
            new_c5 = evaluate_c5(h_candidate)
            
            if new_c5 < current_c5 - tolerance:
                break
            step *= step_decay
        
        # Accept the move if improvement or keep if no improvement
        if step > min_step:
            best_h = h_candidate
            best_c5 = new_c5
        
        # Occasionally add some randomness to escape local minima
        if iter % 200 == 0 and iter > 0:
            noise = np.random.normal(0, 0.001, n_points)
            best_h = project_to_sum_box(best_h + noise, TARGET_SUM)
            best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
