import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    step_h = np.zeros(n_points)

    step_h[n_points//3:2*n_points//3] = 0.5
    step_h[2*n_points//3:] = 0.0
    initial_configs.append(step_h)
    
    # 4. Annealed random + projection
    for _ in range(5):
        rand_h = np.random.rand(n_points) * 0.5
        rand_h = project_to_sum_box(rand_h, TARGET_SUM)
        initial_configs.append(rand_h)
    
    # Optimization loop
    for run_id, h0 in enumerate(initial_configs):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        h0 = project_to_sum_box(h0, TARGET_SUM)
        current_h = h0.copy()
        current_c5 = evaluate_c5(h0)
        
        # Smooth optimization approach using simulated annealing
        T = 0.1
        decay = 0.999
        min_T = 1e-5
        for i in range(2000):
            if time.time() - start_time > budget_s:
                break
                
            # Generate neighbor solution
            h_new = current_h.copy()
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.02)
            h_new[idx] += delta
            
            # Project to valid space
            h_new = project_to_sum_box(h_new, TARGET_SUM)

            # Accept or reject
            if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):

                current_c5 = new_c5
                
            T *= decay
            if T < min_T:
                T = min_T
                
            if current_c5 < best_c5:
                best_c5 = current_c5
