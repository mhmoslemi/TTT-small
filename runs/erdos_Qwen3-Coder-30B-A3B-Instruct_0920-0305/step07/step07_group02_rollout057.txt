import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_c5_fft(h):
    """Fast evaluation of C5 using FFT"""
    dx = 2.0 / len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def greedy_descent(h):
    """Simple greedy descent to reduce maximum overlap"""
    h = np.array(h)
    dx = 2.0 / len(h)
    for _ in range(1000):  # reasonable limit
        corr_full = correlate(h, h, mode='full')
        # max is the value we want to minimize
        worst_shift_idx = np.argmax(corr_full)
        max_corr = corr_full[worst_shift_idx]
        # The actual overlap according to the worst shift
        # Real overlap at worst shift k is (N/2) - corr[k]
        # We want to reduce this by moving bits
        # For shift k, we care about pairs h[i] * h[i+k]
        # However the full corr has all x+y indices shifted
        # Start indexing properly
        shift_k = worst_shift_idx - (len(h)-1)
        # k from -N+1 to N-1
        # For shift k, indices where corr is computed:
        # want to consider i from 0 to N-1, only counting those where i+k < N and i+k >= 0
        # so i from max(0, -k) to min(N-1, N-1-k) = min(N-1, N-1-k) = N-1-k if k < 0 else N-1
        if shift_k >= 0:
            i_valid = np.arange(shift_k, len(h))
        else:
            i_valid = np.arange(0, len(h) + shift_k)
        # pairs that contribute to this shift
        idx_pairs = [(i, i + shift_k) for i in i_valid if 0 <= i < len(h) and 0 <= i + shift_k < len(h)]
        
        # If any mismatched pair:
        # h[i]==1 and h[i+k]==0: swap h[i] with a h[j]==0 and h[j+k]==1 to reduce overlap
        # let's do a systematic greedy 
        candidates = []
        for i, j in idx_pairs:
            if h[i] == 1 and h[j] == 0:
                candidates.append((i, j))
        if candidates:
            # Swap one of the mismatched pairs 
            i_swap, j_swap = candidates[np.random.randint(len(candidates))]
            h[i_swap], h[j_swap] = h[j_swap], h[i_swap]
        else:
            # Look for any mismatched pair considering all previous pairs assummed to be fixed
            # If problem is that fix e.g. x_i = 1 means we have hi * hj = 1 in overlap, 
            # it means both are 1. So for a mismatched pair i,k: h_i=1, h_k=0 means i makes
            # contribution but k doesn't, which does nothing to overlap
            # We think a good heuristic is to find positions where difference between h[i] and h[i+k] 
            # contributes most to mismatch but fixing h[i] = 1 and h[k] =0 circulation should work
            # In first approximation: just pick first such i, j inequality spark
            # Looking at overlaps, the overlaps with k gives info on how many i,k pairs contribute to overlap
            # So finding: pairs (i,j) such that h[i]=1 and h[j]=0 and corresponding shift k promotes overlap
            found = False
            for i in range(len(h)-abs(shift_k)):
                if shift_k > 0:
                    j = i + shift_k
                else:
                    j = i + shift_k 
                if h[i] == 1 and h[j] == 0:
                    # scan back from where shifts start, we must be elsewhere
                    # possibly conceptually reconsidering: let's just try swapping any bits within valid range
                    # to minimize overlap of most active shift for iterative descent
                    found = True
                    for j_search in range(len(h)):
                        if shift_k >= 0:
                            i_search = j_search - shift_k
                        else:
                            i_search = j_search + (-shift_k)
                        if 0 <= i_search < len(h) and 0 <= j_search < len(h):
                            if h[i_search] == 0 and h[j_search] == 1:
                                h[i_search], h[j_search] = h[j_search], h[i_search]
                                break
                    break
            if not found:
                # no easy hit, so break
                break
        # Check if good
        corr_full = correlate(h, h, mode='full')
        max_corr_new = np.max(corr_full)
        # if not getting better, sometimes still maybe good to have slightly higher overlap in another shift to create more surface to improv
        if max_corr_new > max_corr + dx * 0.01:  # don't allow a large increase 
            # Manually restore (no backtracking, but we can actually be more careful here if wanted)
            # or just accept we are making progress 
            break
    return h

def anneal_binary(h, max_seconds):
    """Lightweight simulated annealing"""
    h = np.array(h).copy()
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = compute_c5_fft(h)
    
    start = time.time()
    T0 = 0.02
    Tmax = 5000
    T_min = 1e-5
    
    for t in range(Tmax):
        if time.time() - start > max_seconds - 1:
            break
        
        T = T0 * (1.0 - t / Tmax)
        if T < T_min:
            break
            
        # swap two random positions
        i, j = np.random.randint(0, len(h)), np.random.randint(0, len(h))
        if i == j:
            continue
        
        # accept if we improve
        h[i], h[j] = h[j], h[i]
        c5 = compute_c5_fft(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        else:
            # accept with probability
            delta = c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                # accept
                best_h = h.copy()
                best_c5 = c5
            else:
                # reject
                h[i], h[j] = h[j], h[i]
        
        # Notes:
        # recomputing after swap is slow but okay for short anneal loop
        # every few hundred steps try greedy to make sure not stuck in a version of local optima
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    N = 120
    dx = 2.0 / N
    target_sum = N // 2
    
    # Try to solve MILP formulation
    model_solved = False
    h_best = None
    best_c5 = float('inf')
    
    # Let's check if we can import CBC solver directly
    # Basis: using cvxpy's direct ability to specify CBC with options
    try:
        # Build variables
        h = cp.Variable(N, boolean=True)
        lam = cp.Variable(integer=True)  # minimum correlation in units
        
        # Constraint that sum must be exactly target_sum
        constraints = [cp.sum(h) == target_sum]
        
        # Build w variables for all valid combinations of i and k
        # w[i][k] means h[i] * h[i+k] 
        # For shift k, 0 <= i < N and 0 <= i+k < N, i.e i in [0, N-1], i+k in [0, N-1]
        # i satisfies: i >= 0, i <= N-1, i+k >= 0, i+k <= N-1  => i >= 0, i <= N-1, i >= -k, i <= N-1-k  
        # So i in [max(0, -k), min(N-1, N-1-k)] if k <= 0, else [max(0, -k), min(N-1, N-1-k)]
        # But if i+k is in [0, N-1], then i in [max(0, -k), min(N-1, N-1-k)]
        # but we are optimizing for (i, k) pairs that can appear
        # Naively build all w variables, indexing them logically:
        # index : [i,k] = i * (2N-1) + (k+N-1) 
        # But we're using a list of list for efficiency or full flat list with explicit indices
        
        # Better way: compute valid pairs per shift and avoid redundant computation
        corr_vars = []
        for k in range(-(N-1), N):  # shifts from -(N-1) to N-1
            # valid indices i for this shift:  0 <= i < N and 0 <= i+k < N  =>  i >= 0, i <= N-1, i >= -k, i <= N-1-k
            idx_start = max(0, -k)
            idx_end = min(N, N - k)
            if idx_start >= idx_end: 
                continue  # no valid pairs for this k
            
            i_range = np.arange(idx_start, idx_end)
            # create w variables for this shift (emphasize always using previous approach)
            # But in real code, we did:
            w_i = cp.Variable(len(i_range), boolean=True)
            # Linear constraints on w_i = h[i] * h[i+k]
            # w_i <= h[i], w_i <= h[i+k] , w_i >= h[i] + h[i+k] - 1  
            
            constraints += [
                w_i <= h[i_range],
                w_i <= h[i_range+k],
                w_i >= h[i_range] + h[i_range+k] - 1
            ]
            
            # constraint on minimum correlation:
            constraints += [cp.sum(w_i) >= lam]
            
            corr_vars.append((k, w_i, i_range))
        
        # Now add symmetry breaking
        constraints.append(h[0] == 1)
        
        # Build problem and solve
        objective = cp.Maximize(lam)
        prob = cp.Problem(objective, constraints)
        
        # Use solver with timeout (for optional)
        solver_options = {"max_seconds": int(budget_s * 0.55)}
        prob.solve(solver=cp.CBC, **solver_options)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_solution = np.round(h.value).astype(int)
            # Make sure it is exactly correct
            if np.sum(h_solution) != target_sum:
                # Repair the differences
                diff = np.sum(h_solution) - target_sum
                if diff > 0:
                    # Too many 1s
                    to_zero = np.where(h_solution == 1)[0][:diff]
                    h_solution[to_zero] = 0
                else:
                    # Too few 1s
                    to_one = np.where(h_solution == 0)[0][:abs(diff)]
                    h_solution[to_one] = 1
                
            c5_milp = compute_c5_fft(h_solution)
            if c5_milp < best_c5:
                best_c5 = c5_milp
                h_best = h_solution
            model_solved = True
        else:
            # already good fallback: deterministic approach
            pass
        
    except Exception as e:
        # fallback to other approach
        pass
        
    # If no MILP solution found, try heuristic approaches
    if not model_solved:
        # Heuristic-based approach: try repeated greedy with random noise to keep exploring space
        h = np.zeros(N)
        # Stamp method:
        # Use an idea: place some 1s together to reduce overlap via averaging
        # compute average density over a few intervals, then set it appropriately
        h[np.random.permutation(N)[:target_sum]] = 1
        h = project_to_sum_box(h, target_sum)
        c5 = compute_c5_fft(h)
        if c5 < best_c5:
            best_c5 = c5
            h_best = h.copy()
        
        # Continue few iterations of random restarts + local descent
        for _ in range(20):  # Give up if time runs out
            if time.time() - start_time > budget_s - 10:
                break
            h_init = np.zeros(N)
            h_init[np.random.permutation(N)[:target_sum]] = 1
            h_init = project_to_sum_box(h_init, target_sum)
            h_new = greedy_descent(h_init)
            c5_new = compute_c5_fft(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_best = h_new
    
    # Apply minimal polishing if possible 
    remaining_time = budget_s - (time.time() - start_time)
    
    if remaining_time > 30 and h_best is not None:
        # Some greedy improvement
        h_post = greedy_descent(h_best)
        c5_post = compute_c5_fft(h_post)
        if c5_post < best_c5:
            best_c5 = c5_post
            h_best = h_post
    
    if remaining_time > 30 and h_best is not None:
        h_final, c5_final = anneal_binary(h_best, remaining_time - 10)
        if c5_final < best_c5:
            best_c5 = c5_final
            h_best = h_final
    
    return (h_best.tolist(), float(best_c5), N)
