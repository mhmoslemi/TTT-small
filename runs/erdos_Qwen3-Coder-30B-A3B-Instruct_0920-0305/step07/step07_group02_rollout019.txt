import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def neighborhood_search(h0, max_iter=1000):
    """Enhanced neighborhood search with focus on problem structure"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        if i % 100 == 0:
            corr_full = compute_corr_fft(h)
        
        # Focus on the largest correlation
        max_corr_idx = np.argmax(corr_full)
        # Target a local region of the correlation
        start_region = max(0, max_corr_idx - 20)
        end_region = min(len(corr_full), max_corr_idx + 20)
        
        # Try local modifications
        for _ in range(5):
            # Select a random local region to modify
            region_start = np.random.randint(0, n_points - 5)
            region_end = min(region_start + 10, n_points)
            
            # Try to reduce values in the region
            h_new = h.copy()
            
            # Apply a simple gradient-ascent-like move
            amplitude = np.random.uniform(0.005, 0.02)

def simulated_annealing(h0, T0=0.05, alpha=0.95, max_iter=1000):
    """Simulated annealing with hard constraints"""
    h = h0.copy()
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    T = T0
    
    for i in range(max_iter):
        # Generate neighbor by small random changes
        h_new = h.copy()
        
        # Modify several small regions
        for _ in range(3):
            start = np.random.randint(0, n_points - 5)
            end = min(start + 5, n_points)
            
            # Small random change
            change = np.random.normal(0, 0.01) * T
            h_new[start:end] += change
            
        # Apply constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        
        # Accept or reject
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new.copy()
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        
        # Cool down
        T = max(1e-6, T * alpha)
        
    return best_h, best_c5

            best_c5 = c5_refined
            
        # Phase 2: Simulated annealing for global search
        if time.time() - start_time < 0.8 * budget_s:
            h_sa, c5_sa = simulated_annealing(h_refined, max_iter=500)
            
            if c5_sa < best_c5:
                best_h = h_sa.copy()
                best_c5 = c5_sa

    # Final refinement with local optimization
    if best_h is not None and time.time() - start_time < budget_s:
        def objective(h_local):
            return evaluate_c5(h_local)
        
        def constraint_eq(h_local):
            return np.sum(h_local) - TARGET_SUM
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                best_h = res.x
        except:
            pass
            
        best_h = project_to_sum_box(best_h, TARGET_SUM)
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
