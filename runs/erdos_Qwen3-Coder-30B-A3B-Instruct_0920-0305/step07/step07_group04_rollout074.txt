import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp
from scipy.optimize import minimize

# Global parameters
n_points = 150
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_lp_relaxation():
    """Solve the LP relaxation to get a warm start"""
    # Define variables
    h = Variable(n_points)
    p = Variable((n_points, n_points), symmetric=True)
    
    # Define constraints
    constraints = []
    # Mass constraint
    constraints.append(sum(h) == TARGET_SUM)
    
    # Box constraints
    constraints.extend([0 <= h, h <= 1])
    
    # McCormick envelope constraints for products
    for i in range(n_points):
        for j in range(i, n_points):
            constraints.append(p[i, j] >= 0)
            constraints.append(p[i, j] >= h[i] + h[j] - 1)
            constraints.append(p[i, j] <= h[i])
            constraints.append(p[i, j] <= h[j])
    
    # Create new variables for overlaps
    t = Variable()
    shifts = list(range(1 - n_points, n_points))  # All valid shifts
    
    # Overlaps
    for k in shifts:
        if k >= 0:
            start = 0
            end = n_points - k
        else:
            start = -k
            end = n_points
            
        if start < end:
            overlap = sum(p[i, i + k] for i in range(start, end))
            # Adjust for t: R_k = n/2 - overlap, so we want: R_k <= t, i.e., overlap >= n/2 - t
            constraints.append(overlap >= TARGET_SUM - t)
    
    # Objective
    objective = Minimize(t)
    
    # Create and solve the problem
    prob = Problem(objective, constraints)
    
    # Set solver options
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
        if prob.status == 'optimal':
            return h.value
    except:
        pass
    return None

def subgradient_descent(h0, max_iter=2000, step_size=0.5):
    """Use projected subgradient descent to refine the solution"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    step = step_size * dx
    
    for i in range(max_iter):
        # Find k that achieves maximum
        corr = compute_corr_fft(h)
        k_star = np.argmax(corr)
        
        # Compute subgradient
        g = np.zeros(n_points)
        if k_star < n_points:
            # Positive shift
            for i in range(n_points - k_star):
                g[i] += 1 - h[i + k_star]
            for i in range(k_star, n_points):
                g[i] -= h[i - k_star]
        else:
            # Negative shift
            k_neg = k_star - n_points
            for i in range(n_points + k_neg):
                g[i] += 1 - h[i - k_neg]
            for i in range(-k_neg, n_points):
                g[i] -= h[i + k_neg]
        
        # Update with negative gradient
        h_new = h - step * g
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            step = min(1.0, step * 1.1)  # Adaptive step size
        else:
            h = h_new.copy()
            if i % 100 == 0:
                step *= 0.95  # Reduce step size occasionally
    
    return best_h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Solve LP relaxation
    h_lp = solve_lp_relaxation()
    
    if h_lp is None or np.any(np.isnan(h_lp)):
        # Fallback: use initial construction
        h_init = np.array(initial_h_values)
        x_ref = np.linspace(0, 2, len(h_init))
        x_new = np.linspace(0, 2, n_points)
        h_interp = np.interp(x_new, x_ref, h_init)
        h_opt = project_to_sum_box(h_interp, TARGET_SUM)
    else:
        h_opt = project_to_sum_box(h_lp, TARGET_SUM)
    
    c5_opt = evaluate_c5(h_opt)
    
    # Local refinement with projected subgradients
    h_refined = subgradient_descent(h_opt, max_iter=2000, step_size=0.5)
    c5_refined = evaluate_c5(h_refined)
    
    # If we improved, use the refined solution
    if c5_refined < c5_opt:
        h_opt = h_refined
        c5_opt = c5_refined
    
    # Potentially re-optimize with local methods
    try:
        # For fine-tuning, reinitialize with current best
        def obj(h):
            return -evaluate_c5(h)  # Negative because scipy minimizes
        
        res = minimize(obj, h_opt, method='L-BFGS-B', bounds=[(0,1) for _ in range(n_points)],
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM},
                       options={'maxiter': 500, 'ftol': 1e-9})
        if res.success:
            h_final = res.x
            h_final = project_to_sum_box(h_final, TARGET_SUM)
            c5_final = evaluate_c5(h_final)
            if c5_final < c5_opt:
                h_opt = h_final
                c5_opt = c5_final
    except:
        pass
    
    return (h_opt.tolist(), float(c5_opt), n_points)
