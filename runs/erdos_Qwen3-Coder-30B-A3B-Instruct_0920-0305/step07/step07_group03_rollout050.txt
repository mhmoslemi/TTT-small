import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_of_corr_at_lag(h, lag):
    """Compute gradient of correlation at given lag"""
    n = len(h)
    g = np.ones(n)
    
    if lag >= 0:
        # For lag >= 0, h[i] contributes to correlation with h[i+lag]
        # If i+lag < n, then derivative = 1 - h[i+lag]
        if lag < n:
            g[:n-lag] = 1 - h[lag:]
            g[lag:] -= h[:n-lag]
    else:
        # For lag < 0, we have h[i] with h[i+lag], where i+lag = i-abs(lag) < i
        # Derivative from h[i+lag] contribution: -h[i+lag] = -h[i-abs(lag)]
        d = -lag
        if d < n:
            g[d:] = 1 - h[:n-d]
            g[:n-d] -= h[d:]
    
    return g

def subgradient_descent_on_max_corr(h0, max_iter=2000):
    """Subgradient descent to minimize the maximum correlation"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Compute correlation array
        corr = compute_corr(h)
        max_val = np.max(corr)
        
        # Find all lags that achieve maximum value
        max_lags = np.where(np.abs(corr - max_val) < 1e-10)[0]
        
        # Compute average gradient for those lags
        grad_sum = np.zeros_like(h)
        for lag_idx in max_lags:
            # Convert from array index to actual lag
            actual_lag = lag_idx - (n_points - 1)
            grad = gradient_of_corr_at_lag(h, actual_lag)
            grad_sum += grad
        
        # Normalize gradient
        if len(max_lags) > 0:
            grad_sum /= len(max_lags)
        
        # Apply gradient descent step
        step_size = 0.05 / (1 + i**0.5)
        h_new = h - step_size * grad_sum
        
        # Project to [0,1] and sum constraint
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate and accept improvement
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        
        h = h_new
    
    return best_h, best_c5

def simulated_annealing_binary(h0, max_iter=10000):
    """Simulated annealing on binary patterns"""
    h = h0.copy()
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Cooling schedule
    T_start = current_c5
    T_end = 1e-6
    alpha = (T_end / T_start) ** (1.0 / (max_iter * 0.5))
    
    T = T_start
    for i in range(max_iter):
        # Fast cooling schedule
        if i > 0 and i % 1000 == 0:
            T *= alpha
            
        # Propose move: swap one 0 with one 1
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        if len(ones) > 0 and len(zeros) > 0:
            i_swap = np.random.choice(ones)
            j_swap = np.random.choice(zeros)
            
            # Create new solution by swapping
            h_new = h.copy()
            h_new[i_swap] = 0.0
            h_new[j_swap] = 1.0
            
            # Evaluate new solution
            new_c5 = evaluate_c5(h_new)
            
            # Accept with simulated annealing probability
            delta = new_c5 - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = h_new
                current_c5 = new_c5
                
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Phase 1: Multiple random restarts with simulated annealing
    phase1_end = start_time + 0.5 * budget_s
    for restart in range(5):
        if time.time() > phase1_end:
            break
            
        # Generate random binary solution
        h = np.zeros(n_points)
        ones_indices = np.random.choice(n_points, size=int(n_points/2), replace=False)
        h[ones_indices] = 1.0
        
        # Run simulated annealing for this restart
        h_sa, c5_sa = simulated_annealing_binary(h, max_iter=2000)
        
        if c5_sa < best_c5:
            best_h = h_sa.copy()
            best_c5 = c5_sa
    
    # Phase 2: Subgradient descent refinement
    phase2_end = start_time + 0.8 * budget_s
    if best_h is not None:
        h_refined, c5_refined = subgradient_descent_on_max_corr(best_h, max_iter=2000)
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    # Phase 3: Final local search if time permits
    if time.time() < phase2_end:
        # Apply gradient descent from final binary solution
        h_final, c5_final = subgradient_descent_on_max_corr(best_h, max_iter=1000)
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final
    
    return (best_h.tolist(), float(best_c5), n_points)
