import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def thue_morse_sequence(n):
    """Generate Thue-Morse sequence of length n"""
    seq = [0]
    while len(seq) < n:
        seq.extend(1 - x for x in seq)
    return np.array(seq[:n], dtype=float)

def rudin_shapiro_sequence(n):
    """Generate Rudin-Shapiro sequence of length n"""
    # Rudin-Shapiro sequence: start with 1, then for each step, 
    # append mirrored complement of current sequence  
    # For a better approximation, we compute first few terms directly
    s = np.array([1], dtype=float)
    while len(s) < n:
        # Reverse complement
        complement = 1 - s[::-1]
        s = np.concatenate([s, complement])
    return s[:n]

def swap_indices(h, i, j):
    """Swap elements at indices i and j"""
    h = h.copy()
    h[i], h[j] = h[j], h[i]
    return h

def greedy_worst_shift_improvement(h0):
    """Improve using greedy swaps focused on worst shifts"""
    h = h0.copy()
    
    max_iter = 10000
    for _ in range(max_iter):
        corr = compute_corr_fft(h)
        max_shift_idx = np.argmax(corr)
        if corr[max_shift_idx] <= 0.0001:
            break
            
        # Find the shift we're looking at
        shift = max_shift_idx - (n_points - 1)
        
        # Identify mismatched pairs for that shift 
        # where (h[i], h[i+shift]) transition from (1,0) to (0,1)
        left = h[:-abs(shift)] if shift < 0 else h[:len(h)+shift]
        right = h[abs(shift):] if shift < 0 else h[:len(h)-shift]
        
        # We want positions where (1,0) becomes (0,1)
        # i.e. h[i] = 1, h[i+shift] = 0; and h[j] = 0, h[j+shift] = 1
        pos10 = np.where((left == 1) & (right == 0))[0]
        pos01 = np.where((left == 0) & (right == 1))[0]
        
        if len(pos10) == 0 or len(pos01) == 0:
            continue
            
        # Pick random pair for swap
        i = np.random.choice(pos10)
        j = np.random.choice(pos01)
        if shift < 0:
            i, j = j, i - shift
        else:
            j = j + shift
            
        # Perform the swap and validate it lowers energy
        h_new = swap_indices(h, i, j)
        new_c5 = evaluate_c5(h_new)
        old_c5 = evaluate_c5(h)
        
        if new_c5 < old_c5:
            h = h_new
        else:
            # Slight randomness to escape local optima
            if np.random.rand() < 0.01:
                h = h_new
    return h

def local_gradient_refinement(h0, max_iter=500):
    """Refine with gradient-based optimization"""
    h = h0.copy()
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_local):
        return evaluate_c5(h_local)
    
    def constraint_eq(h_local):
        return np.sum(h_local) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
    except:
        pass
    
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None
    
    num_restarts = 300  # 5000 in total with buffer
    for run_id in range(num_restarts):
        if time.time() > start_time + 0.8 * budget_s:
            break
            
        # For each restart, try several strategies
        strategies = [
            # Random binary start
            np.random.choice([0, 1], size=n_points),
            # Thue-Morse
            thue_morse_sequence(n_points),
            # Rudin-Shapiro
            rudin_shapiro_sequence(n_points),
            # Some heuristic overlaps that are known bad and others good
            np.random.rand(n_points) > 0.5
        ]
        
        h_start = strategies[run_id % len(strategies)]
        h_start = project_to_sum_box(h_start, TARGET_SUM)
        
        # Greedy local search
        h_improved = greedy_worst_shift_improvement(h_start)
        c5_current = evaluate_c5(h_improved)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_improved.copy()
            
        # Small refinement around best
        if run_id % 10 == 0:
            try:
                h_final = local_gradient_refinement(best_h.copy(), max_iter=200)
                c5_final = evaluate_c5(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final.copy()
            except:
                pass
    
    # Final SLSQP refinement of the best solution found
    if best_h is not None:
        best_h = local_gradient_refinement(best_h, max_iter=500)
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
