import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import Problem, Minimize, sum as cp_sum, Variable, Matrix, quad_form, norm
import math

# For linear programming, define helper to create blocks
def build_mcormick_constraints(h, p):
    """Generate McCormick constraints for variables h and product p."""
    constraints = []
    n = len(h)
    for i in range(n):
        for j in range(i, n):
            constraints += [
                p[i, j] >= 0,
                p[i, j] >= h[i] + h[j] - 1,
                p[i, j] <= h[i],
                p[i, j] <= h[j]
            ]
    return constraints

def build_shift_constraints(h, p, t):
    """Generate shift constraints: O_k >= n/2 - t for all k."""
    n = len(h)
    constraints = []
    for k in range(-n+1, n):
        if k >= 0:
            o = cp_sum([p[i, i+k] for i in range(n-k)])
        else:
            o = cp_sum([p[i-k, i] for i in range(-k, n)])
        constraints += [o <= (n/2) - t]
    return constraints

def c5_fft(h):
    """Evaluate C5 using FFT fast correlation."""
    h = np.array(h)
    b = 1 - h
    corr_full = correlate(h, b, mode="full")
    return np.max(corr_full) * (2 / len(h))

def compute_c5_exact(h):
    """Exact evaluation using loops."""
    h = np.array(h)
    n = len(h)
    max_corr = 0.0
    for k in range(-n+1, n):
        corr = 0.0
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                corr += h[i] * (1 - h[j])
        if corr > max_corr:
            max_corr = corr
    return max_corr

def project_to_box_and_sum(h, target_sum):
    """Project h onto [0,1]^n and ensure sum is target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    delta = target_sum - current_sum
    if abs(delta) < 1e-12:
        return h
    
    # Use bisection to find lambda such that sum(clip(h + lambda)) = target_sum
    a, b = -1.0, 1.0
    for _ in range(50):
        mid = (a + b) / 2
        test = np.clip(h + mid, 0, 1)
        if np.sum(test) < target_sum:
            a = mid
        else:
            b = mid
    lambda_opt = (a + b) / 2
    h = np.clip(h + lambda_opt, 0, 1)
    return h

def solve_lp(n, dx):
    """Solve a linear programming relaxation of C5."""
    h = Variable(n)
    # Products矩阵
    p = {}
    for i in range(n):
        for j in range(i, n):
            p[(i, j)] = Variable()

    # Constraints
    constraints = []
    # Mass constraint
    constraints.append(cp_sum(h) == n / 2)

    # McCormick constraints
    for i in range(n):
        for j in range(i, n):
            constraints.append(p[(i, j)] >= 0)
            constraints.append(p[(i, j)] >= h[i] + h[j] - 1)
            constraints.append(p[(i, j)] <= h[i])
            constraints.append(p[(i, j)] <= h[j])

    # Max overlap shift constraints
    t = Variable(1)
    for k in range(-n+1, n):
        if k >= 0:
            total = cp_sum([p[i, i + k] for i in range(n - k)])
        else:
            total = cp_sum([p[i - k, i] for i in range(-k, n)])
        constraints.append(total >= (n / 2) - t)

    # Objective
    objective = Minimize(t)

    # Create and solve the problem
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000, abstol=1e-6, reltol=1e-6)
    except:
        return None, None

    # Extract h and ensure mass
    h_val = np.array(h.value).flatten()
    if h_val is None:
        return None, None
    h_val = project_to_box_and_sum(h_val, n / 2)
    return h_val, prob.value

def sub_gradient_refinement(h0, max_iter=2000):
    """Refine h using projected sub-gradient descent."""
    h = h0.copy()
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = c5_fft(h)
    
    # Gradient descent with exponential decay step size
    alpha = 0.5 * dx
    decay_factor = 0.995
    
    for i in range(max_iter):
        # Compute index k* of max overlap
        imax = -1
        max_corr = -1
        for k in range(-len(h)+1, len(h)):
            corr = 0.0
            for i in range(len(h)):
                j = i + k
                if 0 <= j < len(h):
                    corr += h[i] * (1 - h[j])
            if corr > max_corr:
                max_corr = corr
                imax = k

        if imax == -1:
            break
            
        # Compute subgradient g
        g = np.zeros_like(h)
        for i in range(len(h)):
            j = i + imax
            if 0 <= j < len(h):
                g[i] -= (1 - h[j])
            j = i - imax
            if 0 <= j < len(h):
                g[i] -= h[j]

        # Gradient step
        h = h - alpha * g
        
        # Project
        h = project_to_box_and_sum(h, len(h) / 2)
        
        c5 = c5_fft(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        alpha *= decay_factor
        if alpha < 1e-10:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 150
    dx = 2.0 / n_points
    
    # Solve linear programming relaxation
    h0, lp_value = solve_lp(n_points, dx)
    if h0 is None:
        # Fallback to some constructed h
        h0 = np.ones(n_points) * 0.5
        
    # Evaluate this LP solution
    c5_initial = c5_fft(h0)
    
    # Refine using projected gradient descent
    h_refined, c5_opt = sub_gradient_refinement(h0, max_iter=2000)
    
    # Optionally do one more polishing with CMA-ES small population,
    # but instead just sample a few random perturbations and keep the best
    
    final_c5 = c5_opt
    best_h = h_refined
    
    # Time budget enforcement
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    
    if remaining > 5.0:
        # Try up to 3 random restarts to improve
        for _ in range(3):
            h_test = h_refined + np.random.normal(0, 0.005, n_points)
            h_test = project_to_box_and_sum(h_test, n_points / 2)
            c5_test = c5_fft(h_test)
            if c5_test < final_c5:
                final_c5 = c5_test
                best_h = h_test.copy()
    
    return (best_h.tolist(), float(final_c5), n_points)
