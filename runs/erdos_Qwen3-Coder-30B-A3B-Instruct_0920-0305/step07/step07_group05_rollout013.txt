import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global variables
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmin_weights(s, alpha):
    """Compute soft-min weights for a vector s with smoothing parameter alpha"""
    s_shifted = s - np.max(s)  # Numerical stability
    exp_s = np.exp(-alpha * s_shifted)
    return exp_s / np.sum(exp_s)

def weighted_gradient(h, alpha):
    """Compute weighted gradient of softmin with respect to h"""
    s = correlate(h, h, mode='full')
    w = softmin_weights(s, alpha)
    
    grad = np.zeros_like(h)
    n = len(h)
    
    # Loop over all lags
    for i in range(len(w)):
        lag = i - (n - 1)
        if lag >= 0:
            # h[i] contributes to s[i+lag] and we get h[i+lag]
            if i + lag < n:
                grad[i] += w[i] * h[i + lag]
            # h[i+lag] contributes to s[i] and we get h[i]
            if i - lag >= 0:
                grad[i] += w[i] * h[i - lag]
        else:
            # lag < 0, so same logic with reversed contributions
            lag_abs = -lag
            if i - lag_abs >= 0:
                grad[i] += w[i] * h[i - lag_abs]
            if i + lag_abs < n:
                grad[i] += w[i] * h[i + lag_abs]

    return grad

def pairwise_move(h, grad, delta, rng):
    """Move mass between indices based on gradient"""
    # Find indices with most positive and negative gradients
    idx_src = np.argmin(grad)
    idx_tgt = np.argmax(grad)
    
    if idx_src >= len(h) or idx_tgt >= len(h):
        return False, h.copy(), False
    
    # Make the move
    move_amount = min(delta, h[idx_src], 1.0 - h[idx_tgt])
    if move_amount < 1e-10:
        return False, h.copy(), False
    
    h_new = h.copy()
    h_new[idx_src] -= move_amount
    h_new[idx_tgt] += move_amount
    
    # Apply bounds
    h_new = np.clip(h_new, 0, 1)
    
    return True, h_new, True

def annealed_gradient_ascent(h0, time_budget, alpha0=10, delta0=0.5, temp0=5e-3, rng=None):
    """Annealed gradient ascent with pairwise moves"""
    if rng is None:
        rng = random.Random(42)
    
    # Initialize
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    alpha = alpha0
    delta = delta0
    temp = temp0
    
    # Track iterations
    iteration = 0
    start_time = time.time()
    
    # We reuse an expensive computation in a way that should be effective
    while True:
        # Check remaining time
        elapsed = time.time() - start_time
        if elapsed >= time_budget:
            break
            
        # Compute weighted gradient and softmin
        grad = weighted_gradient(h, alpha)
        
        # Compute current C5 for comparison
        current_c5 = evaluate_c5(h)
        
        # Perform move
        accepted, h_new, move_made = pairwise_move(h, grad, delta, rng)
        
        if accepted:
            # Evaluate new C5
            new_c5 = evaluate_c5(h_new)
            
            # Accept if better
            if new_c5 < current_c5:
                h = h_new
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
            else:
                # Possibly accept with probability
                if temp > 0:
                    prob = np.exp(-(new_c5 - current_c5) / temp)
                    if rng.random() < prob:
                        h = h_new
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h_new.copy()
        
        # Update annealing parameters every 100 iterations
        if iteration % 100 == 0:
            if iteration > 0:
                alpha *= 1.05
                if alpha > 200:
                    alpha = 200
                delta *= 0.9995
                temp *= 0.9995
                
        iteration += 1
    
    return best_h, best_c5

def random_binary_start(n, rng):
    """Create a random binary start vector with exactly n/2 ones"""
    h = np.zeros(n)
    indices = rng.sample(range(n), n//2)
    h[indices] = 1.0
    return h

def local_optimize_restart(start_h, remaining_time, rng):
    """Run a local optimization on start_h for a limited time"""
    start_time = time.time()
    best_h, best_c5 = start_h.copy(), evaluate_c5(start_h)
    
    # Try to run some iterations of gradient ascent
    time_left = remaining_time
    if time_left > 1.0:
        h_opt, c5_opt = annealed_gradient_ascent(
            start_h, 
            time_left * 0.8, 
            alpha0=10,
            delta0=0.4,
            temp0=5e-3,
            rng=rng
        )
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
    
    return best_h, best_c5
