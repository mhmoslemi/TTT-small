import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    TARGET_SUM = n_points / 2

        # Try different initialization strategies
        initial_modes = [
            adaptive_sampling_initialization,

        ]
        
        # Randomly select initialization
        h_start = initial_modes[np.random.randint(0, len(initial_modes))]()
        h_start = project_to_sum_box(h_start, TARGET_SUM)
        
        # Update the solution if the restart improves it
        c5_start = evaluate_c5(h_start)
        if c5_start < best_c5:

        h_current = h_start.copy()

            grad = np.zeros(n_points)

                h_plus = h_current.copy()

            best_h = h_perturbed.copy()

    return (best_h.tolist(), float(best_c5), n_points)
