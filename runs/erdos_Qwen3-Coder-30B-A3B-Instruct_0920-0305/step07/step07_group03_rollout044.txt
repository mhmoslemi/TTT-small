import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_with_gradients(h0, max_iter=1000):
    """Optimize using gradient-based method with projection"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def local_search_improvement(h0, max_iter=1000):
    """Local search improving the solution"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Small tweaks to few elements
        idxs = np.random.choice(n_points, size=min(10, n_points//4), replace=False)
        improved = False
        
        for idx in idxs:
            old_val = h[idx]
            # Try multiple small changes
            for delta in [-0.02, -0.01, 0.01, 0.02]:
                h_new = h.copy()
                h_new[idx] += delta
                
                if h_new[idx] < 0 or h_new[idx] > 1:
                    continue
                    
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                new_c5 = evaluate_c5(h_new)
                
                if new_c5 < best_c5:
                    h = h_new.copy()
                    best_c5 = new_c5
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Perturb slightly
            idx = np.random.randint(0, n_points)
            delta = np.random.uniform(-0.01, 0.01)
            h[idx] += delta
            h = project_to_sum_box(h, TARGET_SUM)
            best_c5 = evaluate_c5(h)
    
    return h

def genetic_algorithm_search():
    """Use a simple genetic algorithm approach"""
    population_size = 20
    generations = 100
    elite_size = 5
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h_init = np.random.uniform(0, 1, n_points)
        h_init = project_to_sum_box(h_init, TARGET_SUM)
        population.append(h_init)
    
    for gen in range(generations):
        # Evaluate fitness
        fitness = [evaluate_c5(h) for h in population]
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        
        # Keep elite
        new_population = population[:elite_size]
        
        # Generate offspring
        for _ in range(population_size - elite_size):
            parent1 = population[np.random.randint(0, elite_size)]
            parent2 = population[np.random.randint(0, elite_size)]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points-1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            mutation_rate = 0.1
            for i in range(n_points):
                if np.random.random() < mutation_rate:
                    child[i] = np.clip(child[i] + np.random.normal(0, 0.05), 0, 1)
            
            child = project_to_sum_box(child, TARGET_SUM)
            new_population.append(child)
        
        population = new_population
    
    # Return best individual
    final_fitness = [evaluate_c5(h) for h in population]
    best_idx = np.argmin(final_fitness)
    return population[best_idx]

def create_adaptive_initialization():

    # Start from a good baseline
    h = adaptive_sampling_initialization()
    
    # Try different improvement strategies
    candidates = []
    
    # Strategy 1: Local search
    h1 = local_search_improvement(h, max_iter=200)
    candidates.append(h1)
    
    # Strategy 2: Adaptive initialization with different noise levels
    for noise_level in [0.005, 0.01, 0.02]:
        h_noise = h + np.random.normal(0, noise_level, n_points)
        h_noise = project_to_sum_box(h_noise, TARGET_SUM)
        h2 = local_search_improvement(h_noise, max_iter=150)
        candidates.append(h2)

        h3 = genetic_algorithm_search()
        candidates.append(h3)
    except:
        pass
    
    # Evaluate all and choose best
    fitness = [evaluate_c5(h) for h in candidates]
    best_idx = np.argmin(fitness)
