import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import SDP, Variable, Minimize, Problem
from scipy.linalg import eigh
import random

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def build_A_matrices(n):
    """Construct sparse A matrices for each lag"""
    A_list = []
    for l in range(-n+1, n):
        A = np.zeros((n, n))
        for i in range(n):
            j = i + l
            if 0 <= j < n:
                A[i, j] = 1
        A_list.append(A)
    return A_list

def solve_sdp(n_points, A_matrices, dx):
    """Solve the SDP relaxation"""
    # Variables
    X = Variable((n_points, n_points), PSD=True)
    t = Variable()
    
    # Constraints
    constraints = [X[i, i] >= 0 for i in range(n_points)]
    constraints += [X[i, i] <= 1 for i in range(n_points)]
    constraints += [cp.sum(X[i, i]) == n_points / 2]
    
    # Add constraints for all lags
    for i, A in enumerate(A_matrices):
        lag = i - (n_points - 1)  # lag from -n+1 to n-1
        # trace(A * X) >= n/2 - t/dx
        constraints += [cp.trace(A @ X) >= n_points / 2 - t / dx]

    # Objective
    objective = Minimize(t)
    
    # Problem
    problem = Problem(objective, constraints)
    
    # Solve (with timeout)
    max_time = 40  # seconds
    problem.solve(solver=cp.SCS, verbose=False, time_limit=max_time)
    
    return X.value, t.value

def round_sdp_solution(X_opt, n_points, dx, num_samples=2000):
    """Round SDP solution to get a feasible vector h"""
    # Principal eigenvector rounding
    eigenvals, eigenvecs = eigh(X_opt)
    v = eigenvecs[:, -1]
    h_raw = v.copy()
    h_clip = np.clip(h_raw, 0, 1)
    h_projected = project_to_sum_box(h_clip, n_points / 2)
    
    best_h = h_projected.copy()
    best_c5 = evaluate_c5(h_projected)
    
    # Random rounding
    for _ in range(num_samples):
        # Sample from N(0, X_opt)
        L = np.linalg.cholesky(X_opt)
        z = np.random.normal(0, 1, n_points)
        y = L @ z
        h_sample = np.clip(y, 0, 1)
        h_sample = project_to_sum_box(h_sample, n_points / 2)
        c5_sample = evaluate_c5(h_sample)
        if c5_sample < best_c5:
            best_c5 = c5_sample
            best_h = h_sample.copy()
    
    return best_h, best_c5

def local_improvement(h, n_points, dx, budget_s_remaining):
    """Simple local refinement"""
    max_iter = int(budget_s_remaining * 50)  # Adjust for remaining time
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Create a small perturbation
        h_pert = h + np.random.normal(0, 0.001, n_points)
        h_pert = project_to_sum_box(h_pert, n_points / 2)
        
        c5_pert = evaluate_c5(h_pert)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
        
        if best_c5 < 0.380:
            break  # Early stop if we're getting close
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 140
    dx = 2.0 / n_points
    
    # Build A matrices
    A_matrices = build_A_matrices(n_points)
    
    # Solve SDP relaxation
    remaining_time = budget_s - 10
    X_opt, t_opt = solve_sdp(n_points, A_matrices, dx)
    
    # Rounding
    best_h, best_c5 = round_sdp_solution(X_opt, n_points, dx)
    
    # Local improvement
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1.0:
        best_h, best_c5 = local_improvement(best_h, n_points, dx, remaining_time)
    
    return (best_h.tolist(), float(best_c5), n_points)
