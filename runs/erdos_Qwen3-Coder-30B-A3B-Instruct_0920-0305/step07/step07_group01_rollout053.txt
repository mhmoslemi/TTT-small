import numpy as np
from scipy.signal import correlate
import time

# Global variables for the optimization
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
beta_schedule = None
alpha_min = 1e-4
epsilon = 0.01
T0 = 0.05
gamma = 5
max_iter_per_stage = 5000
K = 400  # iterations between stochastic perturbations
P = 500  # iterations without improvement before perturbation

def evaluate_c5(h):
    """Evaluate C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_sum_box(x, target_sum):
    """Project x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_gradient(h, corr, beta):
    """Compute the smooth max gradient"""
    max_corr = np.max(corr)
    exp_corr = np.exp(beta * (corr - max_corr))
    weights = exp_corr / np.sum(exp_corr)
    grad = np.zeros_like(h)
    
    # Compute gradient direction
    for i in range(len(h)):
        for l in range(len(corr)):
            if l < len(h) and i + l - (len(h) - 1) >= 0:
                grad[i] += weights[l] * (1 - h[i + l - (len(h) - 1)]) * dx
            if l < len(h) and i - l >= 0:
                grad[i] -= weights[l] * h[i - l] * dx
    
    return grad

def max_lag_gradient(h, corr):
    """Compute exact sub-gradient for the maximum over shifts"""
    max_idx = np.argmax(corr)
    # Compute the gradient in one direction only
    grad = np.zeros_like(h)
    lag = max_idx - (len(h) - 1)
    # i + lag = j (or i = j - lag) means we're in the valid range
    # when i >= 0 and i + lag < len(h), we have a valid term
    
    # If lag < 0, we consider i + lag >= 0 and i < len(h)
    # If lag > 0, we consider i >= lag and i < len(h)
    for i in range(len(h)):
        if lag < 0 and i >= 0 and i + lag < len(h):
            grad[i] -= h[i + lag] * dx
        elif lag > 0 and i >= lag:
            grad[i] += (1 - h[i - lag]) * dx

    
    return grad

def mass_transfer_move(h, grad):
    """Perform mass transfer between bad and good locations"""
    d = -grad
    sources = np.where(d < 0)[0]
    sinks = np.where(d > 0)[0]
    
    # Sort based on priority
    sources = sources[np.argsort(-d[sources])]
    sinks = sinks[np.argsort(-d[sinks])]
    
    h_copy = h.copy()
    for s_idx in sources:
        amount_left = h_copy[s_idx]
        for t_idx in sinks:
            if amount_left <= 0:
                break
            transfer = min(amount_left, 1 - h_copy[t_idx], epsilon)
            if transfer > 0:
                h_copy[s_idx] -= transfer
                h_copy[t_idx] += transfer
                amount_left -= transfer
    
    # Re-project after mass transfer
    h_copy = project_sum_box(h_copy, TARGET_SUM)
    return h_copy

def backtracking_line_search(h, grad, corr, beta, max_lag_idx):
    """Perform backtracking line search"""
    h_new = h.copy()
    alpha = 0.2
    
    while alpha > alpha_min:
        h_candidate = h_new - alpha * grad
        h_candidate = project_sum_box(h_candidate, TARGET_SUM)
        c5_candidate = evaluate_c5(h_candidate)
        if c5_candidate < corr.max() - 1e-8:
            return h_candidate, c5_candidate, True
        alpha /= 2
    
    # Failed to find suitable step size, fall back to mass transfer
    return mass_transfer_move(h, grad), evaluate_c5(mass_transfer_move(h, grad)), False

def stochastic_perturbation(h, T):
    """Perform stochastic perturbation to escape local minima"""
    i = np.random.randint(0, len(h))
    j = np.random.randint(0, len(h))
    
    delta = np.random.uniform(0, min(h[i], 1 - h[j], T * 0.1))
    
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    h_new = project_sum_box(h_new, TARGET_SUM)
    
    return h_new
