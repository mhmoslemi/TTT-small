import numpy as np

import time

# Global constants
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmin_weights(s, alpha):
    """Compute normalized soft-min weights"""
    s_max = np.max(s)
    exp_s = np.exp(-alpha * (s - s_max))
    return exp_s / np.sum(exp_s)

def weighted_gradient(h, alpha):
    """Compute gradient of softmin with respect to h"""
    s = correlate(h, h, mode='full')
    w = softmin_weights(s, alpha)
    
    grad = np.zeros_like(h)
    n = len(h)
    
    # Loop over all lags
    for i in range(len(w)):
        lag = i - (n - 1)
        if lag >= 0:
            # h[i] contributes to s[i+lag] and later positions
            if i + lag < n:
                grad[:n-lag] += w[i] * h[lag:]
            if i - lag >= 0:
                grad[lag:] += w[i] * h[:n-lag]
        else:
            # lag < 0
            lag_abs = -lag
            if i + lag_abs < n:
                grad[lag_abs:] += w[i] * h[:n-lag_abs]
            if i - lag_abs >= 0:
                grad[:lag_abs] += w[i] * h[lag_abs:]
    
    return grad

def pairwise_move(h, grad, delta, rng):
    """Move mass from indices with most negative gradient to most positive"""
    n = len(h)
    
    # Find source and target indices
    idx_src = np.argmin(grad)
    idx_tgt = np.argmax(grad)
    
    if idx_src == idx_tgt:
        # If they're the same, pick another source
        idx_src = np.argsort(grad)[0]  # smallest gradient
        idx_tgt = np.argsort(-grad)[0]  # largest gradient
    
    # Check constraints
    if h[idx_src] <= 1e-10 or h[idx_tgt] >= 1.0 - 1e-10:
        return False, h
    
    # Transfer amount
    transfer = min(delta, h[idx_src], 1.0 - h[idx_tgt])
    if transfer < 1e-12:
        return False, h
    
    # Apply move
    h[idx_src] -= transfer
    h[idx_tgt] += transfer
    
    # Ensure feasibility due to floating point errors
    h[idx_src] = max(0.0, h[idx_src])
    h[idx_tgt] = min(1.0, h[idx_tgt])
    
    return True, h

def annealed_gradient_ascent(h0, time_budget, alpha0, delta0, temp0, rng):

    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Parameters
    alpha = alpha0
    delta = delta0
    temp = temp0
    alpha_factor = 1.02
    delta_factor = 0.9995
    temp_factor = 0.9995
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < time_budget:
        # Update parameters every few iterations
        if iteration % 500 == 0 and iteration > 0:
            alpha *= alpha_factor
            delta *= delta_factor
            temp *= temp_factor
            if alpha > 200:
                alpha = 200

        # Compute weighted gradient
        grad = weighted_gradient(h, alpha)
        
        # Perform move
        accepted, h_new = pairwise_move(h, grad, delta, rng)
        if accepted:
            h = h_new
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Accept with probability based on temperature if not better
            if not accepted and temp > 1e-6 and rng.random() < np.exp(-(c5 - best_c5) / temp):
                # Revert the move
                pass
        else:
            # Small random move to avoid getting stuck
            i = rng.randint(0, n_points - 1)
            j = rng.randint(0, n_points - 1)
            if i != j:
                delta_move = min(h[i], 1 - h[j], 0.005)
                if delta_move > 1e-6:
                    h[i] -= delta_move
                    h[j] += delta_move
                    h = project_to_sum_box(h, TARGET_SUM)
                    c5 = evaluate_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
        
        iteration += 1
    
    return best_h, best_c5

def random_binary_start(n, rng):
    """Create a binary starting vector with exactly n/2 ones"""
    h = np.zeros(n)
    ones_indices = rng.sample(range(n), n // 2)
    h[ones_indices] = 1.0
    return h
