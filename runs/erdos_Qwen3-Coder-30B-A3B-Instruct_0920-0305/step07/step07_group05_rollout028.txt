from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

                h[i] += np.random.normal(0, 0.002)
            h = project_to_sum_box(h, TARGET_SUM)
            
        # Early stopping if improvement is small
        if iter_num > 200 and abs(best_c5 - evaluate_c5(h)) < 1e-8:
            break
    
    return best_h, best_c5

def multistart_optimization(seed, initial_h):

    global TARGET_SUM
    TARGET_SUM = n_points / 2
    
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(initial_h)
    
    # Random perturbations with different intensities
    for i in range(20):
        np.random.seed(seed + i)
        
        # Try different perturbation approaches
        if i < 5:
            # Large perturbations
            h_pert = initial_h + np.random.normal(0, 0.05, n_points) * (1 + 0.2 * np.random.random(n_points))
        elif i < 10:
            # Medium perturbations
            h_pert = initial_h + np.random.normal(0, 0.02, n_points) * (1 + 0.1 * np.random.random(n_points))
        else:
            # Small perturbations
            h_pert = initial_h + np.random.normal(0, 0.005, n_points) * (1 + 0.05 * np.random.random(n_points))
