import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def heuristic_initialization():
    """Create a heuristic initial solution based on known good patterns"""
    # Start with known good values from earlier work or use binary pattern
    h = np.zeros(n_points)
    # Fill half positions randomly
    ones_indices = np.random.choice(n_points, n_points // 2, replace=False)
    h[ones_indices] = 1.0
    
    return h

def greedy_improvement(h):
    """Apply greedy local improvement to reduce worst-case overlap"""
    n = len(h)
    max_iter = 10000
    iter_count = 0
    
    while iter_count < max_iter:
        # Compute correlation
        corr = compute_corr_fft(h)
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n - 1)
        
        # Determine overlapping region indices
        if max_shift >= 0:
            i_range = np.arange(0, n - max_shift)
            j_range = i_range + max_shift
        else:
            i_range = np.arange(-max_shift, n)
            j_range = i_range + max_shift
            
        # Find mismatched pairs
        h_i = h[i_range]
        h_j = h[j_range]
        
        # (1,0) mismatches: h_i == 1 and h_j == 0
        im1 = i_range[(h_i == 1) & (h_j == 0)]
        # (0,1) mismatches: h_i == 0 and h_j == 1
        im2 = i_range[(h_i == 0) & (h_j == 1)]
        
        if im1.size == 0 or im2.size == 0:
            break
            
        # Perform swap
        i = np.random.choice(im1)
        j = np.random.choice(im2)
        
        h[i] = 0
        h[j] = 1
        
        iter_count += 1
        
    return h

def local_gradient_refinement(h0, max_iter=2000):
    """Refine with gradient-based optimization on small subset"""
    h = h0.copy()
    
    # Focus on best regions found
    corr_full = compute_corr_fft(h)
    max_idx = np.argmax(corr_full)
    best_region_start = max_idx - 30 if max_idx >= 30 else 0
    best_region_end = min(max_idx + 30, n_points)
    
    # Optimize these regions with reference to the full function
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_local):
        return evaluate_c5(h_local)
    
    def constraint_eq(h_local):
        return np.sum(h_local) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use fewer iterations and local optimizer
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
    except:
        pass
    
    # Project to maintain constraints
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = None
    best_c5 = float('inf')
    
    # Run multiple restarts
    num_restarts = 20
    restart_budget = budget_s * 0.8 / num_restarts
    
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Generate initial binary solution
        h_initial = heuristic_initialization()
        
        # Apply greedy improvement
        h_reduced = greedy_improvement(h_initial)
        
        # Evaluate C5
        c5_val = evaluate_c5(h_reduced)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_reduced.copy()
    
    # Final refinement with continuous optimization
    if best_h is not None:
        final_h = local_gradient_refinement(best_h.copy(), max_iter=500)
        final_c5 = evaluate_c5(final_h)
        
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    return (best_h.tolist(), float(best_c5), n_points)
