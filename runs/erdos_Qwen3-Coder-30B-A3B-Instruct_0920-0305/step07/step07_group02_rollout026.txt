import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5_fft(h):
    """Fast evaluation of C5 using FFT"""
    dx = 2.0 / len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def greedy_descent(h0):
    """Deterministic local improvement on worst shift"""
    h = h0.copy()
    dx = 2.0 / len(h)
    
    while True:
        corr_full = correlate(h, h, mode='full')
        # Find the index for worst shift (smallest correlation)
        worst_shift_idx = np.argmax(corr_full)
        # The actual shift k corresponding to this index
        k = worst_shift_idx - (len(h) - 1)
        
        # Positions where h[i]=1 and h[i+k]=0 (these contribute to C5)
        A = np.where((h == 1) & (np.roll(h, -k) == 0))[0]
        # Positions where h[i]=0 and h[i+k]=1
        B = np.where((h == 0) & (np.roll(h, -k) == 1))[0]
        
        if len(A) == 0 or len(B) == 0:
            break
            
        # Pick a random pair to swap
        i = np.random.choice(A)
        j = np.random.choice(B)
        
        # Perform the swap
        h[i], h[j] = h[j], h[i]
    
    return h, compute_c5_fft(h)

def anneal_binary(h0, max_time):
    """Lightweight simulated annealing"""
    h = h0.copy()
    dx = 2.0 / len(h)
    current_c5 = compute_c5_fft(h)
    
    T0 = 0.02
    Tmax = 5000
    start_time = time.time()
    
    for t in range(Tmax):
        if time.time() - start_time > max_time:
            break
        
        # Temperature schedule
        T = T0 * (1 - t / Tmax)
        if T <= 0:
            break
            
        # Pick two random positions
        i = np.random.randint(len(h))
        j = np.random.randint(len(h))
        if i == j:
            continue
            
        # Compute change in C5
        old_h = h.copy()
        h[i], h[j] = h[j], h[i]
        
        new_c5 = compute_c5_fft(h)
        delta = new_c5 - current_c5
        
        # Accept or reject
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_c5 = new_c5
        else:
            h = old_h
    
    return h, current_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    N = 120
    dx = 2.0 / N
    target_sum = N // 2  # N must be even
    
    best_h = None
    best_c5 = float('inf')
    
    # Step 1: Solve MILP with CBC
    try:
        # Define variables
        h = cp.Variable(N, boolean=True)
        lam = cp.Variable(integer=True)
        
        # Constraints for correlation
        constraints = [cp.sum(h) == target_sum]
        w_vars = {}
        
        # For each possible shift
        for k in range(-N+1, N):
            # valid indices for shift k
            i_vals = np.arange(max(0, -k), min(N, N-k))
            if len(i_vals) == 0:
                continue
                
            # build w[i,k] variables
            w = cp.Variable(len(i_vals), boolean=True)
            w_vars[(k, i_vals[0])] = w
            
            # Add linearization constraints
            constraints += [w <= h[i_vals], w <= h[i_vals+k], w >= h[i_vals] + h[i_vals+k] - 1]
            
            # Add correlation constraint
            constraints += [cp.sum(w) >= lam]
        
        # Symmetry breaking: fix h[0] = 1
        constraints += [h[0] == 1]
        
        # Objective
        prob = cp.Problem(cp.Maximize(lam), constraints)
        
        # Solve
        remaining_time = budget_s * 0.55
        prob.solve(solver=cp.CBC, max_seconds=remaining_time, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = np.round(h.value).astype(int)
            if abs(np.sum(h_sol) - target_sum) > 1e-6:
                # repair discrete constraint
                ones = np.where(h_sol == 1)[0]
                zeros = np.where(h_sol == 0)[0]
                if len(ones) > target_sum:
                    # remove extra 1s
                    to_remove = len(ones) - target_sum
                    for _ in range(to_remove):
                        i = np.random.choice(ones)
                        h_sol[i] = 0
                        ones = np.where(h_sol == 1)[0]
                else:
                    # add missing 1s
                    to_add = target_sum - len(ones)
                    for _ in range(to_add):
                        i = np.random.choice(zeros)
                        h_sol[i] = 1
                        zeros = np.where(h_sol == 0)[0]
            
            # Compute C5
            C5 = 1 - 2*lam.value/N
            if C5 < best_c5:
                best_h = h_sol
                best_c5 = C5
                
        else:
            raise Exception("Failed to solve MILP")
    except Exception as e:
        # fallback to heuristics
        h_random = np.random.rand(N) < 0.5
        tot = np.sum(h_random)
        if tot > target_sum:
            # set some 1s to 0
            ones = np.where(h_random == 1)[0]
            for i in ones[tot-target_sum:]:
                h_random[i] = 0
        elif tot < target_sum:
            # set some 0s to 1
            zeros = np.where(h_random == 0)[0]
            for i in zeros[:target_sum-tot]:
                h_random[i] = 1
                
        best_h = h_random.astype(int)
        best_c5 = compute_c5_fft(best_h)
    
    # Step 2: Greedy local improvement
    elapsed = time.time() - start_time
    if elapsed < 0.9 * budget_s:
        try:
            improved_h, improved_c5 = greedy_descent(best_h)
            if improved_c5 < best_c5:
                best_h = improved_h
                best_c5 = improved_c5
        except:
            pass
    
    # Step 3: Simulated annealing
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 30:
        try:
            final_h, final_c5 = anneal_binary(best_h, remaining)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), N)
