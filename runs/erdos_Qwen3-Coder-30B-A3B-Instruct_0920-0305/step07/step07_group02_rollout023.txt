import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from itertools import combinations

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def init_from_binary_pattern(n, pattern_type="random"):
    """Initialize from a binary pattern"""
    if pattern_type == "random":
        h = np.random.randint(0, 2, n)
        # Ensure correct sum
        current_sum = np.sum(h)
        if current_sum > n // 2:
            # Turn off some 1s
            ones_to_flip = current_sum - n // 2
            flip_indices = np.random.choice(np.where(h == 1)[0], size=ones_to_flip, replace=False)
            h[flip_indices] = 0
        elif current_sum < n // 2:
            # Turn on some 0s
            zeros_to_flip = n // 2 - current_sum
            flip_indices = np.random.choice(np.where(h == 0)[0], size=zeros_to_flip, replace=False)
            h[flip_indices] = 1
    elif pattern_type == "thue_morse":
        h = np.array([int(x) for x in format(i, 'b').replace('0','') for i in range(n)])
        # Truncate to length n if needed
        if len(h) > n:
            h = h[:n]
        elif len(h) < n:
            h = np.pad(h, (0, n-len(h)), 'wrap')
        # Fix length
        h = h[:n]
        # Adjust if necessary
        current_sum = np.sum(h)
        if current_sum != n // 2:
            # Adjust by flipping appropriate bits
            if current_sum > n // 2:
                excess = current_sum - n // 2
                ones_indices = np.where(h == 1)[0]
                flip = np.random.choice(ones_indices, size=excess, replace=False)
                h[flip] = 0
            else:
                deficit = n // 2 - current_sum
                zeros_indices = np.where(h == 0)[0]
                flip = np.random.choice(zeros_indices, size=deficit, replace=False)
                h[flip] = 1
    elif pattern_type == "rudin_shapiro":
        # Rudin-Shapiro sequence based on number of adjacent 11s
        h = np.zeros(n, dtype=int)
        for i in range(n):
            count = bin(i).count('11') & 1  # number of adjacent 11s mod 2
            h[i] = count
    else:
        h = np.random.randint(0, 2, n)
        # Ensure correct sum
        current_sum = np.sum(h)
        if current_sum > n // 2:
            # Turn off some 1s
            ones_to_flip = current_sum - n // 2
            flip_indices = np.random.choice(np.where(h == 1)[0], size=ones_to_flip, replace=False)
            h[flip_indices] = 0
        elif current_sum < n // 2:
            # Turn on some 0s
            zeros_to_flip = n // 2 - current_sum
            flip_indices = np.random.choice(np.where(h == 0)[0], size=zeros_to_flip, replace=False)
            h[flip_indices] = 1

    return h.astype(float)

def greedy_swap_improvement(h):
    """Improve h by swaps that remove worst overlaps"""
    # Repeat until no improvement possible
    old_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = old_c5
    
    while True:
        corr = compute_corr_fft(best_h)
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n_points - 1) // 2
        
        # Find pairs for potential swap
        L = n_points
        if max_shift >= 0:
            h1 = best_h[:L - max_shift]
            h2 = best_h[max_shift:]
        else:
            h1 = best_h[-max_shift:]
            h2 = best_h[:L + max_shift]
        
        # Find wrong pairs
        mismatch_id1 = np.where((h1 == 1) & (h2 == 0))[0]
        mismatch_id2 = np.where((h1 == 0) & (h2 == 1))[0]
        
        # Try swapping incompatible pairs
        improvement_made = False
        nm = min(len(mismatch_id1), len(mismatch_id2))
        if nm > 0:
            indices1 = np.random.choice(mismatch_id1, size=nm, replace=False)
            indices2 = np.random.choice(mismatch_id2, size=nm, replace=False)
            for i1, i2 in zip(indices1, indices2):
                if max_shift >= 0:
                    if i1 + max_shift < len(best_h) and i2 + max_shift < len(best_h):
                        h = best_h.copy()
                        h[i1], h[i2] = h[i2], h[i1]
                        h[i1 + max_shift], h[i2 + max_shift] = h[i2 + max_shift], h[i1 + max_shift]
                        h = project_to_sum_box(h, TARGET_SUM)
                        new_c5 = evaluate_c5(h)
                        if new_c5 < best_c5:
                            best_h = h.copy()
                            best_c5 = new_c5
                            improvement_made = True
        if not improvement_made:
            break
    
    return best_h, best_c5

def simulated_annealing(h, initial_temperature=0.1, cooling_rate=0.95, max_iters=1000):
    """Perform simulated annealing"""
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    temp = initial_temperature
    iter_count = 0

    while iter_count < max_iters and temp > 1e-8:
        # Randomly perturb
        h_new = current_h.copy()
        idx1 = np.random.randint(0, n_points)
        idx2 = np.random.randint(0, n_points)
        h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        new_c5 = evaluate_c5(h_new)
        
        # Accept or reject
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            current_h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
        
        temp *= cooling_rate
        iter_count += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Number of restarts based on time limit
    num_restarts = max(100, int(budget_s / 3))  # Allow 3 seconds per restart
    restart_budget = budget_s * 0.7
    restart_interval = restart_budget / num_restarts
    restarts_made = 0
    
    # List of initialization strategies
    init_types = ["random", "thue_morse", "rudin_shapiro"]
    init_weights = [0.6, 0.2, 0.2]
    
    for restart_id in range(num_restarts):
        if time.time() - start_time > restart_budget:
            break
        
        # Choose initialization randomly
        init_type = np.random.choice(init_types, p=init_weights)
        h0 = init_from_binary_pattern(n_points, init_type)
        
        # Greedy improvement
        h_greedy, c5_greedy = greedy_swap_improvement(h0)
        
        # Simulated annealing
        h_sa, c5_sa = simulated_annealing(h_greedy)
        
        if c5_sa < best_c5:
            best_h = h_sa.copy()
            best_c5 = c5_sa
        
        restarts_made += 1
    
    # Final continuous refinement with local optimizer
    if best_h is not None:
        def objective(x):
            return evaluate_c5(x)
        
        def constraint_eq(x):
            return np.sum(x) - TARGET_SUM
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                best_h = res.x
            else:
                best_h = project_to_sum_box(best_h, TARGET_SUM)
        except:
            best_h = project_to_sum_box(best_h, TARGET_SUM)
        
        best_c5 = evaluate_c5(best_h)
    
    # Final projection to ensure bounds
    best_h = np.clip(best_h, 0, 1)
    
    return (best_h.tolist(), float(best_c5), n_points)
