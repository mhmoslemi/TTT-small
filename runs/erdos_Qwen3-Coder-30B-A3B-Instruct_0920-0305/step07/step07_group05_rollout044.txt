def build_autocorrelation_matrices(n):
    """Build sparse A_ell matrices for each lag."""
    A_list = []
    for ell in range(-(n-1), n):
        rows = []
        cols = []
        data = []
        for i in range(n):
            j = i + ell
            if 0 <= j < n:
                rows.append(i)
                cols.append(j)
                data.append(1)
        A = np.zeros((n, n))
        A[rows, cols] = data
        A_list.append(A)
    return A_list

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation."""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def round_sdp_solution(X_opt, n):
    """Extract h from PSD solution X_opt using principal eigenvector rounding."""
    _, V = eigh(X_opt)
    # Use the leading left eigenvector
    v = V[:, -1]
    h_raw = np.abs(v)
    h_clip = np.clip(h_raw, 0, 1)
    h_projected = project_to_sum_box(h_clip, n / 2)
    return h_projected
