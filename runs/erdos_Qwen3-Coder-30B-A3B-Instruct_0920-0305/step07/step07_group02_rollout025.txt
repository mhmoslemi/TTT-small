import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def c5_fft(h):
    # linear (non-circular) convolution of h with (1-h)
    n = len(h)
    nfft = 1 << ((2*n-1).bit_length())  # next power of two
    H = np.fft.rfft(h, nfft)
    B = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * B, nfft)[:2*n-1]   # length 2N-1
    dx = 2.0 / n
    return np.max(conv) * dx

def project_to_sum(h, target_sum):
    """Project h to have sum = target_sum, maintaining bounds [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Binary search for the Lagrange multiplier
    def f(lam):
        h_trial = np.clip(h + lam, 0, 1)
        return np.sum(h_trial) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    h_proj = np.clip(h + (a + b)/2, 0, 1)
    return h_proj

def thue_morse(n):
    """Generate Thue-Morse sequence of length n"""
    tm = [0]
    for _ in range(int(np.log2(n)) + 2):
        tm = tm + [1 - x for x in tm]
    return np.array(tm[:n]) % 2

def rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence of length n"""
    rs = np.zeros(n, dtype=int)
    for i in range(n):
        # Compute the number of runs of 1s at even positions in the binary
        # representation of i, modulo 2
        # This is a simplified version for the first few entries
        # The real Rudin-Shapiro sequence is more complex, but we just want
        # a balanced sequence
        count = 0
        j = i
        while j:
            if j & 1:
                count += 1
            j >>= 1
        rs[i] = count & 1
    return rs

def greedy_descent(h):
    """Perform greedy descent on the worst shift until no improvement possible"""
    h = h.copy()
    target_sum = len(h) / 2
    n = len(h)
    max_iter = int(1.5 * n)  # a safe upper bound
    
    for _ in range(max_iter):
        corr = correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        s = max_idx - (n - 1)
        
        # Find mismatched indices
        A = []  # indices where h[i]=1 and h[i+s]=0
        B = []  # indices where h[i]=0 and h[i+s]=1
        
        if s >= 0:
            for i in range(n - s):
                if h[i] == 1 and h[i+s] == 0:
                    A.append(i)
                elif h[i] == 0 and h[i+s] == 1:
                    B.append(i)
        else:
            s = -s
            for i in range(n - s):
                if h[i+s] == 1 and h[i] == 0:
                    A.append(i+s)
                elif h[i+s] == 0 and h[i] == 1:
                    B.append(i+s)
        
        if not A or not B:  # No improving swaps available
            break

        # Pick a random improving swap
        i = np.random.choice(A)
        j = np.random.choice(B)
        
        h[i], h[j] = h[j], h[i]
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    n = 256
    dx = 2.0 / n
    target_sum = n / 2
    
    # Phase I: Multiple greedy descent from various seeds
    seeds = get_initial_seeds(n, target_sum)
    best_c5 = float('inf')
    best_h = None
    
    phase1_time = 0.2 * budget_s
    for h_seed in seeds:
        if time.time() - start_time > phase1_time:
            break
        h_greedy = greedy_descent(h_seed)
        c5 = c5_fft(h_greedy)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_greedy.copy()
    
    # Phase II: Simulated Annealing
    phase2_time = 0.5 * budget_s
    if time.time() - start_time > phase2_time:
        return (best_h.tolist(), float(best_c5), n)
    
    # Initialize with best solution from greedy phase
    h_current = best_h.copy()
    c5_current = c5_fft(h_current)
    
    # Set up temperature and counters
    T0 = 0.02
    T = T0
    max_iter = 200000
    accepted = 0
    last_improvement = 0
    last_c5 = c5_current
    iter_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < phase2_time:
        # Try a random swap of two entries
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        
        if i == j or h_current[i] == h_current[j]:
            iter_count += 1
            continue
            
        # Temporarily update the two entries
        h_new = h_current.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        
        c5_new = c5_fft(h_new)
        
        # Accept or reject
        delta = c5_new - c5_current
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            h_current = h_new
            c5_current = c5_new
            accepted += 1
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()
            last_improvement = accepted
            
        iter_count += 1
        
        # Cooling schedule
        t = (time.time() - start_time) / phase2_time
        T = T0 * (1 - t)  # Linear cooling to 0
        
        # Occasionally perform greedy descent on the worst shift
        if iter_count % 500 == 0:
            h_current = greedy_descent(h_current)
            c5_current = c5_fft(h_current)
    
    # Phase III: Continuous refinement
    phase3_time = min(0.3 * budget_s, budget_s - (time.time() - start_time))
    
    if phase3_time > 5:
        # Smooth the binary solution
        kernel = np.array([0.25, 0.5, 0.25])
        h0 = np.convolve(best_h, kernel, mode='same')
        h0 = project_to_sum(h0, target_sum)
        h0 = np.clip(h0, 0, 1)
        
        def objective(h):
            return c5_fft(h)
        
        bounds = [(0, 1)] * n
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        
        try:
            res = minimize(
                objective,
                h0,
                method='trust-constr',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500, 'gtol': 1e-8, 'ftol': 1e-12}
            )
            if res.success:
                h_final = res.x
                c5_final = c5_fft(h_final)
                h_final = project_to_sum(h_final, target_sum)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final
        except:
            # Fall back to binary if optimization fails
            pass

    # Final sanity check
    best_h = project_to_sum(best_h, target_sum)
    final_c5 = c5_fft(best_h)
    
    return (best_h.tolist(), float(final_c5), n)
