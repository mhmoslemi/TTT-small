import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def c5_fft(h):
    """Fast C5 computation using FFT-based linear convolution"""
    n = len(h)
    nfft = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, nfft)
    B = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * B, nfft)[:2*n-1]
    return np.max(conv) * dx

def project_to_sum(h, target_sum):
    """Project vector h onto sum = target_sum with clipping to [0,1]"""
    def f(lam):
        temp = np.clip(h + lam, 0, 1)
        return np.sum(temp) - target_sum
    
    # Binary search for lambda
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b) / 2, 0, 1)
    return result

def thue_morse(n):
    """Generate Thue-Morse sequence up to n bits"""
    if n == 0:
        return np.array([])
    t = [0]
    while len(t) < n:
        t.extend(1 - np.array(t))
    return np.array(t[:n])

def rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence up to n bits"""
    if n == 0:
        return np.array([])
    # Compute Rudin-Shapiro using recurrence
    rs = np.zeros(n, dtype=int)
    rs[0] = 1
    for i in range(1, n):
        # Count number of 1s in binary representation of i where consecutive 1s don't overlap
        # For simplicity, we generate up to n via a simpler method
        rs[i] = (-1)**sum(1 for j in range(i.bit_length()) if (i>>j)&1 and (i>>(j+1))&1)
    # We first generate simpler version: 0,0,1,0,1,1,0,0...
    # Actually, easy implementation:
    rs = np.zeros(n, dtype=int)
    rs[0] = 1
    for i in range(1, n):
        k = bin(i)[2:]  # binary of i without '0b'
        ones = k.count('1')
        if ones % 2 == 0:
            rs[i] = 1
        else:
            rs[i] = -1
    rs = (rs + 1) // 2  # Convert to 0,1
    return rs

def greedy_descent(h0):
    """Greedy descent from a binary starting vector to improve the worst overlap"""
    h = h0.copy().astype(float)
    target_sum = TARGET_SUM
    # Ensure the sum is exactly right
    h = project_to_sum(h, target_sum)
    
    # Determine all indices matching the constraints
    while True:
        corr_full = correlate(h, 1-h, mode="full")
        max_idx = np.argmax(corr_full)
        # The resulting correlation index starts from N-1
        shift = max_idx - (n_points - 1)
        
        # Get mismatched pairs (1,0) and (0,1)
        A = []
        B = []
        if shift >= 0:
            for i in range(n_points - shift):
                if h[i] == 1 and h[i + shift] == 0:
                    A.append(i)
                elif h[i] == 0 and h[i + shift] == 1:
                    B.append(i)
        else:
            for i in range(-shift, n_points):
                if h[i] == 1 and h[i + shift] == 0:
                    A.append(i)
                elif h[i] == 0 and h[i + shift] == 1:
                    B.append(i)
        
        if len(A) == 0 or len(B) == 0:
            break
    
        # Pick random pair and make swap
        i = np.random.choice(A)
        j = np.random.choice(B)
        
        # Swap bit values at indices i and j
        h[i], h[j] = h[j], h[i]
        
        # Re-project to ensure sum constraint maintained
        h = project_to_sum(h, target_sum)
    
    return h, c5_fft(h)

def simulated_annealing(h_start, max_time):
    """Run simulated annealing on binary vectors"""
    h_current = h_start.copy()
    current_c5 = c5_fft(h_current)
    best_h = h_current.copy()
    best_c5 = current_c5
    start_time = time.time()
    
    # Parameters
    T0 = 0.02
    decay = 0.99999  # use a slow decay
    T = T0
    accept_count = 0
    total_count = 0
    
    while time.time() - start_time < max_time:
        total_count += 1
        # Make a swap
        idx1 = np.random.randint(0, n_points)
        idx2 = np.random.randint(0, n_points)
        
        # Ensure they are different indices and have different values
        if h_current[idx1] == h_current[idx2]:
            continue
            
        # Save original state
        old_value1 = h_current[idx1]
        old_value2 = h_current[idx2]
        
        # Perform swap
        h_current[idx1] = old_value2
        h_current[idx2] = old_value1
        
        # Evaluate new C5
        new_c5 = c5_fft(h_current)
        
        # Accept or reject
        if new_c5 <= current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = new_c5
            accept_count += 1
        else:
            # Metropolis criterion
            if np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                current_c5 = new_c5
                accept_count += 1
            else:
                # Revert
                h_current[idx1] = old_value1
                h_current[idx2] = old_value2
        
        T *= decay
        
        # Perform mini-greedy descent occasionally
        if total_count % 500 == 0:
            h_greedy, c5_greedy = greedy_descent(h_current)
            if c5_greedy < current_c5:
                h_current = h_greedy.copy()
                current_c5 = c5_greedy
            if c5_greedy < best_c5:
                best_h = h_greedy.copy()
                best_c5 = c5_greedy
    
    return best_h, best_c5

def continuous_refinement(h_initial):
    """Refine with a continuous optimizer"""
    try:
        # Convert to float and smooth slightly
        h0 = h_initial.astype(float)
        kernel = np.array([0.25, 0.5, 0.25])
        h0 = np.convolve(h0, kernel, mode='same')
        h0 = np.clip(h0, 0, 1)
        h0 = project_to_sum(h0, TARGET_SUM)
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return c5_fft(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
            
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        # Use trust-constr optimizer
        res = minimize(
            objective,
            h0,
            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'gtol': 1e-8}
        )
        
        if res.success:
            h_refined = res.x
        else:
            h_refined = h0
        
        h_refined = project_to_sum(h_refined, TARGET_SUM)
        h_refined = np.clip(h_refined, 0, 1)
        
        return h_refined, c5_fft(h_refined)
    except:
        return h_initial, c5_fft(h_initial)
