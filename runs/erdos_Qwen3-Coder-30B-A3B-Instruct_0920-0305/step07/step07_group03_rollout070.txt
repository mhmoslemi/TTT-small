import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_of_corr_at_lag(h, lag):
    """Compute gradient of correlation at lag with respect to h"""
    n = len(h)
    g = np.ones(n)
    
    if lag >= 0:
        if lag < n:
            g[:n-lag] = 1 - h[lag:]
            g[lag:] -= h[:n-lag]
    else:
        d = -lag
        if d < n:
            g[d:] = 1 - h[:n-d]
            g[:n-d] -= h[d:]
    
    return g

def subgradient_descent_step(h, step_size):
    """Perform one step of subgradient descent"""
    corr = compute_corr(h)
    max_val = np.max(corr)
    
    # Find indices where maximum occurs
    indices = np.where(np.abs(corr - max_val) < 1e-10)[0]
    
    # Compute average gradient from max indices
    grad_sum = np.zeros(len(h))
    for idx in indices:
        # Convert index back to lag
        lag = idx - (len(h) - 1)
        grad_sum += gradient_of_corr_at_lag(h, lag)
    
    avg_grad = grad_sum / len(indices)
    
    # Update h
    h_new = h - step_size * avg_grad
    
    # Project to sum and bounds
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    
    return h_new

def simulated_annealing_binary(n_iter=5000, temp_start=0.1, temp_end=1e-5):
    """Simulated annealing for binary sequences"""
    h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
    # Ensure sum is exactly TARGET_SUM
    ones_indices = np.where(h == 1)[0]
    if len(ones_indices) != TARGET_SUM:
        # Adjust to have exactly TARGET_SUM ones
        if len(ones_indices) > TARGET_SUM:
            # Remove some 1s
            to_remove = len(ones_indices) - TARGET_SUM
            remove_indices = np.random.choice(ones_indices, to_remove, replace=False)
            h[remove_indices] = 0
        else:
            # Add some 1s
            zero_indices = np.where(h == 0)[0]
            to_add = TARGET_SUM - len(ones_indices)
            add_indices = np.random.choice(zero_indices, to_add, replace=False)
            h[add_indices] = 1
    
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    temp = temp_start
    alpha = (temp_end / temp_start) ** (1.0 / n_iter)
    
    for i in range(n_iter):
        temp *= alpha
        
        # Propose neighbor: swap 0 and 1 at random positions
        ones_positions = np.where(h == 1)[0]
        zeros_positions = np.where(h == 0)[0]
        
        if len(ones_positions) > 0 and len(zeros_positions) > 0:
            idx1 = np.random.choice(ones_positions)
            idx2 = np.random.choice(zeros_positions)
            
            h_new = h.copy()
            h_new[idx1] = 0
            h_new[idx2] = 1
            
            new_c5 = evaluate_c5(h_new)
            
            delta = new_c5 - current_c5
            
            if delta < 0 or np.random.random() < np.exp(-delta / temp):
                h = h_new
                current_c5 = new_c5
                
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Phase 1: Simulated Annealing with multiple restarts
    phase1_end = start_time + 0.5 * budget_s
    for restart in range(10):
        if time.time() > phase1_end:
            break
            
        h_binary, c5_binary = simulated_annealing_binary(n_iter=1000)
        
        if c5_binary < best_c5:
            best_h = h_binary.copy()
            best_c5 = c5_binary
    
    # Phase 2: Local search refinement with continuous optimization
    phase2_end = start_time + 0.8 * budget_s
    
    # Convert to float with proper initialization
    h_continuous = best_h.astype(float)
    current_c5 = evaluate_c5(h_continuous)
    
    # Use subgradient descent to improve continuously
    step_size = 0.02
    max_iter = 5000
    
    for i in range(max_iter):
        if time.time() > phase2_end:
            break
            
        h_new = subgradient_descent_step(h_continuous, step_size)
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < current_c5:
            h_continuous = h_new
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = h_continuous.copy()
                best_c5 = current_c5
                
            # Adaptive step size
            step_size = min(0.1, step_size * 1.05)
        else:
            step_size *= 0.95
            
        if step_size < 1e-6:
            break
    
    # Phase 3: Final optimization on best result    
    phase3_end = start_time + budget_s
    
    # Perturb slightly and optimize
    h_optimized = best_h.copy()
    
    # Try several small random adjustments
    for i in range(1000):
        if time.time() > phase3_end:
            break
            
        # Pick two different positions
        idx1 = np.random.randint(0, n_points)
        idx2 = np.random.randint(0, n_points)
        
        if idx1 != idx2:
            # Store original values
            original1 = h_optimized[idx1]
            original2 = h_optimized[idx2]
            
            # Try small change to both
            h_optimized[idx1] += np.random.uniform(-0.05, 0.05)
            h_optimized[idx2] -= np.random.uniform(-0.05, 0.05)
            
            # Project back to [0, 1] and maintain sum
            h_optimized = project_to_sum_box(h_optimized, TARGET_SUM)
            
            new_c5 = evaluate_c5(h_optimized)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_optimized.copy()
            else:
                # Revert if no improvement
                h_optimized[idx1] = original1
                h_optimized[idx2] = original2
    
    return (best_h.tolist(), float(best_c5), n_points)
