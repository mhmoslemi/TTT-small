import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def enhance_solution(h):
    """Refine solution with gradient descent and random perturbations"""
    # Try to improve using scipy optimizer with different methods
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Try several approaches
    methods = ['SLSQP', 'L-BFGS-B']
    
    for method in methods:
        try:
            # Start with slight random noise
            h_start = h + np.random.normal(0, 0.001, n_points)
            h_start = project_to_sum_box(h_start, TARGET_SUM)
            
            res = minimize(
                objective, 
                h_start, 
                method=method, 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            
            if res.success:
                h_candidate = res.x
                h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
                candidate_c5 = evaluate_c5(h_candidate)
                
                if candidate_c5 < evaluate_c5(h):
                    h = h_candidate
            
        except Exception:
            continue
            
    return h

def local_search_step(h, attempts=20):
    """Perform hill-climbing local search"""
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    for _ in range(attempts):
        h_trial = h.copy()
        # Modify a few elements
        indices = np.random.choice(n_points, size=min(5, n_points//10), replace=False)
        for i in indices:
            # Small perturbation
            delta = np.random.uniform(-0.02, 0.02)
            h_trial[i] += delta
            # Ensure it stays in bounds
            h_trial[i] = np.clip(h_trial[i], 0, 1)
        
        # Ensure sum constraint
        h_trial = project_to_sum_box(h_trial, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_trial)
        if new_c5 < best_c5:
            best_h = h_trial
            best_c5 = new_c5
    
    return best_h, best_c5

    if np.sum(low_mask) > 0:

        # Insert two sharp peaks

                        else:
                            h[idx] = max(0, h[idx] - 0.3)
