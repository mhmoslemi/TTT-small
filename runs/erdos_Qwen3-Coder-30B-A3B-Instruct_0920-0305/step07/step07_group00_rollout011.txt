from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h, weights):
    """Compute gradient using convolution method to avoid explicit loops"""
    n = len(h)
    offset = n - 1
    
    # Compute the weighted sum of shifted vectors
    # This computes sum_s w_s * (1 - h[i+s]) and sum_s w_s * h[i-s]
    conv_wh = np.convolve(weights, h, mode='full')
    conv_rev = np.convolve(weights[::-1], h, mode='full')
    
    # Extract relevant parts corresponding to total_weight (sum of weights = 1)  
    # and the two components: sum_s w_s * h[i+s] and sum_s w_s * h[i-s]
    gradient = 1.0 - (conv_rev[offset:offset+n] + conv_wh[offset:offset+n])
    return gradient

def main_gradient_descent(h_start, epsilon_schedule, max_iters_per_eps, lr_factor, target_sum):
    """Main gradient descent optimization with scheduled epsilon smoothing"""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for eps_idx, eps in enumerate(epsilon_schedule):
        iter_count = max_iters_per_eps[eps_idx]
        lr = lr_factor * eps  # Adaptive learning rate
        
        for iter in range(iter_count):
            # Compute correlation
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            
            # Compute exponential weights (with numerical stability)
            w_raw = np.exp((corr - max_corr) / eps)
            w_norm = w_raw / w_raw.sum()
            
            # Compute gradient
            gradient = compute_gradient(h, w_norm)
            
            # Update step
            h_new = h - lr * gradient
            h_new = project_to_sum_box(h_new, target_sum)
            h = h_new
            
            # Periodic evaluation
            if iter % 20 == 0:
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    
    return best_h, best_c5
