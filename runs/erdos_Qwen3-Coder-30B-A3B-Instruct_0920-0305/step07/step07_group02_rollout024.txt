import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def c5_fft(h):
    # linear (non-circular) convolution of h with (1-h)
    n = len(h)
    nfft = 1 << ((2*n-1).bit_length())          # next power of two
    H = np.fft.rfft(h, nfft)
    B = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * B, nfft)[:2*n-1]   # length 2N-1
    dx = 2.0 / n
    return np.max(conv) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def thue_morse(n):
    """Generate Thue-Morse sequence up to n bits"""
    tm = [0]
    for i in range(n):
        tm.extend(1 - tm[i])
        if len(tm) >= n:
            break
    return np.array(tm[:n])

def rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence up to n bits"""
    rs = np.zeros(n, dtype=int)
    for i in range(n):
        # Count number of carries in binary addition of i and bit-reversed i
        count = 0
        j = i
        k = 0
        while j > 0:
            if (j & 1) and (k & 1):
                count += 1
            j >>= 1
            k >>= 1
        if (count & 1) == 0:
            rs[i] = 1
    return rs

def random_binary(n, target_sum):
    """Generate random binary vector with target sum"""
    h = np.zeros(n)
    ones = np.random.choice(n, target_sum, replace=False)
    h[ones] = 1
    return h

def greedy_descent(h, target_sum):
    """Greedy descend to minimize C5"""
    h = h.copy()
    old_c5 = c5_fft(h)
    max_iter = 2 * target_sum  # Safety limit
    for i in range(max_iter):
        # Compute overlap for all shifts
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_shift_idx = np.argmax(corr)
        shift = max_shift_idx - (len(h) - 1)
        
        # Determine pairs for the current shift
        if shift >= 0:
            pairs_10 = np.where((h[:-shift] == 1) & (h[shift:] == 0))[0]
            pairs_01 = np.where((h[:-shift] == 0) & (h[shift:] == 1))[0]
        else:
            # shift < 0
            pairs_10 = np.where((h[-shift:] == 1) & (h[:-shift] == 0))[0]
            pairs_01 = np.where((h[-shift:] == 0) & (h[:-shift] == 1))[0]
            
        if len(pairs_10) == 0 or len(pairs_01) == 0:
            break
        
        # Pick a random pair to swap
        i0 = np.random.choice(pairs_10)
        j0 = np.random.choice(pairs_01)
        
        # swap h[i0] and h[j0]
        h[i0], h[j0] = h[j0], h[i0]
    
    return h, c5_fft(h)
