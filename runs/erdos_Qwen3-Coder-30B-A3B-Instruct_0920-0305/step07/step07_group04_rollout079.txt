import numpy as np
import cvxpy as cp
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables for discretization
n_points = 150
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_relaxed_lp(h_initial):
    h_vars = cp.Variable(n_points, name='h')
    p_vars = cp.Variable((n_points, n_points), name='p', symmetric=True)
    
    # Mass constraint
    mass_constraint = cp.sum(h_vars) == TARGET_SUM
    
    # McCormick envelopes
    mc_constraints = []
    for i in range(n_points):
        for j in range(i, n_points):
            # p_ij >= h_i + h_j - 1
            mc_constraints.append(p_vars[i, j] >= h_vars[i] + h_vars[j] - 1)
            # p_ij >= 0
            mc_constraints.append(p_vars[i, j] >= 0)
            # p_ij <= h_i
            mc_constraints.append(p_vars[i, j] <= h_vars[i])
            # p_ij <= h_j
            mc_constraints.append(p_vars[i, j] <= h_vars[j])
    
    # Correlation constraints
    T = cp.Variable(name='T', nonneg=True)
    overlap_vars = {}
    for k in range(-(n_points - 1), n_points):
        if k >= 0:
            # sum p_{i, i+k} for i = 0 to n_points-1-k
            overlap = cp.sum([p_vars[i, i + k] for i in range(n_points - k)])
        else:
            # sum p_{i+k, i} for i = 0 to n_points+ k -1 (with proper limits)
            overlap = cp.sum([p_vars[i + (-k), i] for i in range(n_points + k)])
        overlap_vars[k] = overlap
        # O_k >= (n/2 - T)
        mc_constraints.append(overlap >= TARGET_SUM - T)
    
    objective = cp.Minimize(T)
    constraints = [mass_constraint] + mc_constraints
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.GLOP, verbose=False, time_limit=180)
    
    if prob.status != cp.OPTIMAL:
        return h_initial.copy()
    
    h_final = h_vars.value
    return h_final

def binary_search_lp_bound(h_start):
    h_best = h_start.copy()
    t_low = 0.0
    t_high = TARGET_SUM
    tol = 1e-4 * dx
    
    while t_high - t_low > tol:
        t_mid = (t_low + t_high) / 2
        # Solve LP feasibility test
        h_test = solve_relaxed_lp(h_best)
        # Evaluate true C5
        c5_test = evaluate_c5(h_test)
        if c5_test <= t_mid * dx and t_mid > 0:
            t_high = t_mid
            h_best = h_test.copy()
        elif t_mid < TARGET_SUM:
            t_low = t_mid
        else:
            break
    return h_best, t_high * dx

def sub_gradient_polish(h0, max_iter=2000):
    h = h0.copy()
    step_size = 0.01
    
    for t in range(max_iter):
        # Compute correlation
        corr = compute_corr_fft(h)
        # Find the index of the maximum overlap
        k_star = np.argmax(corr)
        # Compute subgradient for the active correlation
        g = np.zeros(n_points)
        # For k_star, the gradient of h_i * (1 - h[i+k_star]) is (1 - h[i+k_star]) - h[i-k_star]
        # We construct the gradient part by part:
        # Contribution from h_i (1-h_{i+k_star}):
        for i in range(n_points):
            if i + k_star < n_points:
                g[i] += 1 - h[i + k_star]
        # Contribution from h_{i-k_star} (when applicable)
        if k_star <= n_points - 1:
            for i in range(k_star, n_points):
                g[i] -= h[i - k_star]
        
        # Update h
        h = h - step_size * g
        # Project onto [0, 1] and adjust mass
        h = project_to_sum_box(h, TARGET_SUM)
        # Decrease step size
        step_size *= 0.999
        
    return h

def local_exchange_refinement(h0, max_iter=1000):
    h = h0.copy()
    for it in range(max_iter):
        i, j = np.random.choice(n_points, 2, replace=False)
        if i > j:
            i, j = j, i
        delta_max = min(h[j], 1 - h[i])
        if delta_max <= 0:
            continue
        delta = np.random.choice([0.1, 0.5, 0.9]) * delta_max
        # Save old values
        old_c5 = evaluate_c5(h)
        h[i] += delta
        h[j] -= delta
        # Clip to [0,1]
        h[i] = np.clip(h[i], 0, 1)
        h[j] = np.clip(h[j], 0, 1)
        # Project to correct mass
        h = project_to_sum_box(h, TARGET_SUM)
        # Evaluate new C5
        new_c5 = evaluate_c5(h)
        # Accept only if improvement
        if new_c5 >= old_c5:
            # Revert
            h[i] -= delta
            h[j] += delta
            h[i] = np.clip(h[i], 0, 1)
            h[j] = np.clip(h[j], 0, 1)
            h = project_to_sum_box(h, TARGET_SUM)
    return h
