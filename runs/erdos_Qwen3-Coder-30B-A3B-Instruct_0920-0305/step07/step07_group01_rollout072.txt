def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * 2.0 / len(h)

def project_sum_box(h, target_sum):
    """Project vector h onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(h + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def smooth_gradient(h, corr, beta):
    """Compute gradient of smooth max approximation"""
    # Compute weights
    max_corr = np.max(corr)
    shifted_corr = corr - max_corr
    exp_corr = np.exp(beta * shifted_corr)
    weights = exp_corr / np.sum(exp_corr)
    # Compute gradient: dx * [1 - sum(w_l * (h_{i+l} + h_{i-l}) for all l)]
    N = len(h)
    grad = np.ones(N) * 2.0 / N
    for l in range(-N+1, N):
        if l >= 0:
            # l >= 0, so h_{i+l} contributes
            shifted_h = np.roll(h, -l)
            grad -= weights[l+N-1] * shifted_h
        else:
            # l < 0, so h_{i+l} = h_{i-abs(l)} contributes
            shifted_h = np.roll(h, -l)
            grad -= weights[l+N-1] * shifted_h
    return grad

def max_lag_gradient(h, corr):
    """Compute exact sub-gradient for the true maximum"""
    N = len(h)
    idx_best = np.argmax(corr)
    grad = np.zeros(N)
    # For the worst lag, we know that exactly one term contributes
    # The contribution of shift k to corr[i] is h[i] * (1-h[i+k]) 
    # -> exact gradient at i:
    #     if i+k < N: grad[i] += (1-h[i+k]) * 2/N  
    #     if i-k >= 0: grad[i] -= h[i-k] * 2/N
    if idx_best >= 0:
        # We shift by idx_best to the right
        for i in range(N-idx_best):
            grad[i] += 2.0/N * (1-h[i+idx_best])
        for i in range(idx_best, N):
            grad[i] -= 2.0/N * h[i-idx_best]
    else:
        # We shift by -idx_best to the left
        for i in range(-idx_best, N):
            grad[i] += 2.0/N * (1-h[i+idx_best])
        for i in range(N+idx_best):
            grad[i] -= 2.0/N * h[i-idx_best]
    return grad

def mass_transfer(h, grad, epsilon_percent=0.01):
    """Perform greedy mass transfer moves"""
    N = len(h)
    epsilon = epsilon_percent * N / 2
    d = -grad
    sources = [(i, d[i]) for i in range(N) if d[i] < -1e-10]  
    sinks = [(i, d[i]) for i in range(N) if d[i] > 1e-10]  
    sources.sort(key=lambda x: x[1], reverse=True)
    sinks.sort(key=lambda x: x[1], reverse=True)
    h_copy = h.copy()
    for s, _ in sources:
        if epsilon <= 0: break
        for t, _ in sinks:
            if epsilon <= 0: break
            delta = min(h_copy[s], 1 - h_copy[t], epsilon)
            if delta > 1e-12:
                h_copy[s] -= delta
                h_copy[t] += delta
                epsilon -= delta
    return project_sum_box(h_copy, N/2)
