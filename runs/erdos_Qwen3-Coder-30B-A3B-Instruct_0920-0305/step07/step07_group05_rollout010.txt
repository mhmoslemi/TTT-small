import numpy as np
import time

# Global variables
n_points = None
dx = None
TARGET_SUM = None

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    global dx
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    x = np.asarray(x, dtype=np.float64)
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0.0, 1.0)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(60):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0.0, 1.0)
    return result

def compute_gradient(h, lag):
    """Compute gradient for a given lag"""
    global n_points
    n = len(h)
    grad = np.ones(n)
    if lag >= 0:
        if lag < n:
            grad[:n-lag] -= h[lag:]
            grad[lag:] -= h[:n-lag]
    else:
        lag_abs = -lag
        if lag_abs < n:
            grad[lag_abs:] -= h[:n-lag_abs]
            grad[:lag_abs] -= h[lag_abs:]
    return grad

                h[src] -= delta
                h[tgt] += delta
                c5_new = evaluate_c5(h)
                if c5_new < c5:
                    c5 = c5_new
                    corr = np.correlate(h, 1 - h, mode='full')
                    lag_max = int(np.argmax(corr) - (n - 1))
                else:
                    h[src] += delta
                    h[tgt] -= delta
        else:

            i, j = np.random.choice(n, 2, replace=False)
            if h[i] > 1e-12 and h[j] < 1 - 1e-12:
                delta_allowed = min(h[i], 1.0 - h[j])

                delta = min(delta, delta_allowed)
                h[i] -= delta
                h[j] += delta
                c5_new = evaluate_c5(h)
                if c5_new < c5:
                    c5 = c5_new
                    corr = np.correlate(h, 1 - h, mode='full')
                    lag_max = int(np.argmax(corr) - (n - 1))
                else:
                    h[i] += delta
                    h[j] -= delta
        
        # Occasionally make a perturbation to escape local minima
        if iteration % 500 == 0:
            for _ in range(3):
                i, j = np.random.choice(n, 2, replace=False)
                if h[i] > 1e-12 and h[j] < 1 - 1e-12:

                    delta = min(delta, h[i], 1.0 - h[j])

            if c5_new < c5:
                c5 = c5_new
                corr = np.correlate(h, 1 - h, mode='full')

            c5 = evaluate_c5(h)
            corr = np.correlate(h, 1 - h, mode='full')
            lag_max = int(np.argmax(corr) - (n - 1))
    
    return h, c5

def random_binary(n):
    """Generate a random binary vector with exactly TARGET_SUM ones"""
    ones = np.zeros(n)
    idx = np.random.choice(n, int(TARGET_SUM), replace=False)
    ones[idx] = 1.0
    return ones
