import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    n = len(h)
    nfft = 1 << ((2*n-1).bit_length())  # Next power of 2
    H = np.fft.rfft(h, n=nfft)
    B = np.fft.rfft(1 - h, n=nfft)
    conv = np.fft.irfft(H * B, n=nfft)[:2*n-1]
    return np.max(conv) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def random_binary_initial(n):
    """Create a random binary initial solution"""
    indices = np.random.choice(n, n//2, replace=False)
    h = np.zeros(n)
    h[indices] = 1.0
    return h

def greedy_improvement(h):
    """Improve using deterministic swaps targeting worst shift"""
    n = len(h)
    max_iter = 10000  # Prevent infinite loops
    iter_count = 0
    while iter_count < max_iter:
        iter_count += 1
        corr = correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n - 1)
        
        if max_shift >= 0:
            i_range = np.arange(0, n - max_shift)
            j_range = i_range + max_shift
        else:
            i_range = np.arange(-max_shift, n)
            j_range = i_range + max_shift
            
        h_i = h[i_range]
        h_j = h[j_range]
        
        im1 = i_range[(h_i == 1) & (h_j == 0)]
        im2 = i_range[(h_i == 0) & (h_j == 1)]
        
        if im1.size == 0 or im2.size == 0:
            break
            
        i = np.random.choice(im1)
        j = np.random.choice(im2)
        
        h[i] = 0
        h[j] = 1
        
    return h

def local_gradient_refinement(h0, max_iter=300):
    """Refine with gradient-based optimization"""
    h = h0.copy()

    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_local):
        return compute_c5(h_local)
    
    def constraint_eq(h_local):
        return np.sum(h_local) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
    except:
        pass
    
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    best_h = None
    best_c5 = float('inf')
    
    # Multiple random restarts
    num_restarts = 20
    iter_per_restart = 5000
    
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Random binary initialization
        h = random_binary_initial(n_points)
        
        # Apply greedy improvement
        h_improved = greedy_improvement(h)
        
        # Evaluate
        c5 = compute_c5(h_improved)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_improved.copy()
    
    # Final refinement with continuous optimization
    if best_h is not None:
        best_h_refined = local_gradient_refinement(best_h, max_iter=500)
        final_c5 = compute_c5(best_h_refined)
        
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = best_h_refined.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
