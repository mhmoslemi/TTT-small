import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmin_weights(s, alpha):
    """Compute normalized soft-min weights for the given alpha"""
    s_shifted = s - np.max(s)
    exp_vals = np.exp(-alpha * s_shifted)
    return exp_vals / np.sum(exp_vals)

def weighted_gradient(h, alpha):
    """Compute ∇_h softmin_α(s(h)) efficiently using loops"""
    s = correlate(h, h, mode='full')
    w = softmin_weights(s, alpha)
    
    grad = np.zeros_like(h)
    
    # Process each lag ℓ
    for ell in range(len(s)):
        w_ell = w[ell]
        if w_ell <= 1e-15:  # Skip negligible weights
            continue
            
        # Convert full-length index back to relative lag
        lag = ell - (len(s) - 1) // 2
        
        if lag >= 0:
            # Add contribution from h[i] * h[i+l] for all valid i
            # This is: h shifted by +lag (i.e., h[i+lag])
            if lag < len(h):
                grad[lag:] += w_ell * h[:-lag]  # h[:len(h)-lag] * h[lag:]
        else:
            # lag < 0: Add contribution from h[i] * h[i+l] for all valid i
            # This is: h[i-lag] term where i-lag >= 0, i.e., h shifted by -lag
            lag_abs = -lag
            if lag_abs < len(h):
                grad[:len(h)-lag_abs] += w_ell * h[lag_abs:]  # h[lag_abs:] * h[:len(h)-lag_abs]

    return grad

def pairwise_move(h, grad, delta_max, rng):
    """Perform a single pairwise mass transfer move on h based on gradient"""
    # Find source (min gradient) and target (max gradient)
    src = np.argmin(grad)
    tgt = np.argmax(grad)
    
    # Ensure src and tgt are different, should happen typically
    if src == tgt:
        # Find alternative source and target
        sorted_grad_idx = np.argsort(grad)
        src = sorted_grad_idx[0]
        tgt = sorted_grad_idx[-1]
        if src == tgt:
            # Fallback to random
            src = rng.choice(len(h))
            tgt = rng.choice(len(h))
    
    # Compute allowable transfer amount
    delta = min(delta_max, h[src], 1 - h[tgt])
    if delta < 1e-10:
        return False  # No significant transfer

    # Apply transfer
    h[src] -= delta
    h[tgt] += delta
    return True

def annealed_gradient_ascent(h0, time_budget, alpha_start, delta_start, temp_start, rng):
    """Main optimization loop with temperature annealing"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    alpha = alpha_start
    delta = delta_start
    temp = temp_start
    
    start_time = time.time()
    iter_count = 0
    accept_count = 0
    
    while time.time() - start_time < time_budget:
        iter_count += 1
        
        # Compute gradient using soft minimum
        grad = weighted_gradient(h, alpha)
        
        # Attempt a move
        moved = pairwise_move(h, grad, delta, rng)
        if moved:
            c5_new = evaluate_c5(h)
            # Accept based on Metropolis rule
            if c5_new < best_c5:
                # Always accept improvements
                best_c5 = c5_new
                best_h = h.copy()
                accept_count += 1
            else:
                # Accept with probability e^(-ΔC/T)
                delta_c5 = c5_new - best_c5
                if temp > 1e-12:
                    p_accept = np.exp(-delta_c5 / temp)
                    if rng.random() < p_accept:
                        accept_count += 1
                else:
                    # Revert if no acceptance
                    h = best_h.copy()
        
        # Schedule updates
        if iter_count % 10 == 0:
            # Gradual cooling and smoothing
            alpha *= 1.01
            if alpha > 200:
                alpha = 200
            delta *= 0.9995
            temp *= 0.9995
            
        if iter_count % 500 == 0:
            # Reset stepsize occasionally to escape local minima
            delta = 0.2
            
    return best_h, best_c5

def random_binary_start(n, rng):
    """Generate a randomized binary starting point"""
    h = np.zeros(n)
    ones_indices = rng.choice(n, size=n//2, replace=False)
    h[ones_indices] = 1.0
    return h

def local_optimize_restart(start_h, remaining_time, rng):
    """Optimize a given starting point with optimal temperature schedule"""
    # Short run for exploratory restarts
    restart_budget = min(20, remaining_time * 0.8)
    h_opt, c5_opt = annealed_gradient_ascent(
        start_h, restart_budget, 10.0, 0.4, 5e-3, rng
    )
    return h_opt, c5_opt
