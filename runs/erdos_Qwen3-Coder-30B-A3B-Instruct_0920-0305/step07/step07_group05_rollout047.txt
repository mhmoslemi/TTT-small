import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def annealed_local_search(h0, max_iter=5000):
    """Refine solution with simulated annealing style local search"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Initialize temperature
    temp = 0.1
    decay_rate = 0.999
    
    for iter_num in range(max_iter):
        # Create neighbor solution
        h_new = h.copy()
        
        # Perturb a few positions
        n_perturb = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        for i in indices:
            delta = np.random.normal(0, 0.02)
            h_new[i] += delta
        
        # Project back to constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject based on temperature
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / temp):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        # Cool down
        temp *= decay_rate
        
        # Early stopping if good enough
        if best_c5 < 0.3795:
            break
    
    return best_h, best_c5

def gradient_guided_search_v2(h0, max_iter=2000):
    """Enhanced gradient-guided search"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for iter_num in range(max_iter):
        # Compute gradient using finite differences on a subset
        grad = np.zeros(n_points)
        eps = 1e-4
        
        # Sample indices for gradient computation
        sample_indices = np.random.choice(n_points, min(30, n_points//2), replace=False)
        
        for i in sample_indices:
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_sum_box(h_plus, TARGET_SUM)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_sum_box(h_minus, TARGET_SUM)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Find best improvement direction
        improvement_direction = np.argsort(grad)[:max(1, n_points//20)]
        
        # Try moves in improvement directions
        for i in improvement_direction:
            if np.random.rand() < 0.3:  # Stochastic selection
                delta = min(h[i], 1 - h[i], 0.01)
                if delta > 1e-6:
                    h[i] -= delta
                    h = project_to_sum_box(h, TARGET_SUM)
                    c5 = evaluate_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
        
        # Occasionally do random moves
        if np.random.rand() < 0.1:
            i = np.random.choice(n_points)
            j = np.random.choice(n_points)
            if i != j:
                delta = min(h[i], 1 - h[j], 0.005)
                if delta > 1e-6:
                    h[i] -= delta
                    h[j] += delta
                    h = project_to_sum_box(h, TARGET_SUM)
                    c5 = evaluate_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
    
    return best_h, best_c5
