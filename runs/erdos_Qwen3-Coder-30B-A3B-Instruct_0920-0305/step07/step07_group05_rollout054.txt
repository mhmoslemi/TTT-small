def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_A_matrices(n):
    """Construct sparse A_lag matrices"""
    A_list = []
    for lag in range(-(n-1), n):
        rows = []
        cols = []
        data = []
        for i in range(n):
            j = i + lag
            if 0 <= j < n:
                rows.append(i)
                cols.append(j)
                data.append(1.0)
        A = csc_matrix((data, (rows, cols)), shape=(n, n))
        A_list.append(A)
    return A_list

def solve_sdp(n, A_list, dx, max_time=30):
    """Solve SDP relaxation using cvxpy"""
    start_time = time.time()
    
    # Variables
    X = cp.Variable((n, n), PSD=True)
    t = cp.Variable()

    # Constraints
    constraints = []
    
    # Diagonal bounds
    for i in range(n):
        constraints.append(X[i, i] >= 0)
        constraints.append(X[i, i] <= 1)
    
    # Sum constraint
    constraints.append(cp.trace(X) == n/2)
    
    # Autocorrelation constraints
    for A in A_list:
        constraints.append(cp.trace(A @ X) >= n/2 - t/dx)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000, eps=1e-5)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            X_opt = X.value
            t_opt = t.value
            return X_opt, t_opt, True
    except:
        pass
    
    return None, None, False

def rounding_and_sampling(X_opt, n, dx, num_samples=2000):
    """Round SDP solution using eigenvector and Gaussian sampling"""
    # Principal eigenvector rounding
    eigenvals, eigenvecs = eigh(X_opt)
    v = eigenvecs[:, -1]  # largest eigenvalue
    h_eig = np.clip(v, 0, 1)
    h_eig = project_to_sum_box(h_eig, n/2)
    c5_eig = evaluate_c5(h_eig)
    
    best_h = h_eig.copy()
    best_c5 = c5_eig
    
    # Gaussian sampling
    # Cholesky decomposition to sample from N(0, X_opt)
    try:
        L = np.linalg.cholesky(X_opt) if np.allclose(X_opt, X_opt.T) else np.linalg.svd(X_opt)[0] @ np.diag(np.sqrt(np.linalg.svd(X_opt)[1]))
    except:
        L = np.linalg.svd(X_opt)[0] @ np.diag(np.sqrt(np.linalg.svd(X_opt)[1]))
    
    for _ in range(num_samples):
        # Sample from N(0, X_opt)  
        z = np.random.randn(n)
        y = L @ z
        h_sample = np.clip(y, 0, 1)
        h_sample = project_to_sum_box(h_sample, n/2)
        c5_sample = evaluate_c5(h_sample)
        if c5_sample < best_c5:
            best_c5 = c5_sample
            best_h = h_sample.copy()
    
    return best_h, best_c5

    # Compute initial correlation

        eps = 1e-5
        for i in range(n_points):

            c5_plus = evaluate_c5(h_plus)

            h_minus = project_to_sum_box(h_minus, n_points / 2)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Find best indices to modify
