import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def adaptive_pattern_search(h0, max_iter=1000):

    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    # Track recent improvements for adaptive step sizing
    recent_improvements = []
    
    for i in range(max_iter):
        if i % 100 == 0:
            corr_full = compute_corr_fft(h)
        
        # Get current maximum correlation
        max_corr_idx = np.argmax(corr_full)
        
        # Adaptive modification: 
        # - If we recently improved, try larger changes
        # - If we're stuck, try smaller, more focused changes
        
        # Try several modifications to the interval around the worst overlap
        step_size = 0.01 if len(recent_improvements) < 5 else 0.02
        modification_count = 5 if len(recent_improvements) < 5 else 10
        
        for _ in range(modification_count):
            # Try modifying a random interval around the worst overlap
            start = max(0, max_corr_idx - 20 + np.random.randint(-10, 10))
            end = min(n_points, start + 10 + np.random.randint(0, 15))
            
            # Try several modifications to the interval
            sign = np.random.choice([-1, 1])

            h_new = h.copy()
