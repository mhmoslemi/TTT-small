import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global constants
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmin_weights(s, alpha):
    """Compute soft-min weights for the given alpha"""
    s_shifted = s - np.max(s)  # numerical stability
    exps = np.exp(-alpha * s_shifted)
    return exps / np.sum(exps)

def weighted_gradient(h, alpha):
    """Compute the gradient of the softmin with respect to h"""
    # Compute the autocorrelation s = h * h
    s = correlate(h, h, mode='full')
    # Get weights
    w = softmin_weights(s, alpha)
    # Compute gradient
    grad = np.zeros_like(h)
    n = len(h)
    
    # Compute weighted sum of shifted vectors
    for i in range(len(w)):
        lag = i - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[:n-lag] += w[i] * h[lag:]
        else:
            lag_abs = -lag
            if lag_abs < n:
                grad[lag_abs:] += w[i] * h[:n-lag_abs]
    return grad

def pairwise_move(h, grad, delta, rng):
    """Perform a pairwise mass transfer move based on gradient"""
    # Find source and target indices
    neg_indices = np.where(grad < 0)[0]
    pos_indices = np.where(grad > 0)[0]
    
    if len(neg_indices) == 0 or len(pos_indices) == 0:
        # fall back to random move
        i_src = rng.choice(np.arange(len(h)))
        i_tgt = rng.choice(np.arange(len(h)))
        while i_src == i_tgt:
            i_tgt = rng.choice(np.arange(len(h)))
    else:
        i_src = neg_indices[np.argmin(grad[neg_indices])]
        i_tgt = pos_indices[np.argmax(grad[pos_indices])]
    
    # Compute transfer amount
    transfer_amount = min(delta, h[i_src], 1 - h[i_tgt])
    if transfer_amount < 1e-10:
        return False, h.copy(), False
    
    # Perform the transfer
    h[i_src] -= transfer_amount
    h[i_tgt] += transfer_amount
    
    # Clip to bounds
    h = np.clip(h, 0, 1)
    
    return True, h.copy(), True

def annealed_gradient_ascent(h0, time_budget, alpha0, delta0, T0, rng):
    """Run the annealed gradient ascent algorithm"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Schedule parameters
    alpha = alpha0
    delta = delta0
    T = T0
    
    start_time = time.time()
    iteration = 0
    accepted_moves = 0
    
    while time.time() - start_time < time_budget:
        # Compute weighted gradient
        grad = weighted_gradient(h, alpha)
        
        # Perform a move
        accepted, h_new, moved = pairwise_move(h, grad, delta, rng)
        
        if moved:
            c5_new = evaluate_c5(h_new)
            
            # Accept or reject based on Metropolis criterion if not improving
            if c5_new < best_c5:
                h = h_new.copy()
                best_c5 = c5_new
                best_h = h.copy()
                accepted_moves += 1
            else:
                # Accept with probability
                if accepted and T > 0:
                    if rng.random() < np.exp(-(c5_new - best_c5) / T):
                        h = h_new.copy()
                        accepted_moves += 1
                else:
                    # If not accepted, revert
                    pass
        
        # Update schedules
        if iteration % 20 == 0:
            if alpha < 200:
                alpha *= 1.1
            if delta > 1e-4:
                delta *= 0.9995
            T *= 0.9995
        
        iteration += 1
        
        # Occasionally re-project to maintain feasibility
        if iteration % 500 == 0:
            h = project_to_sum_box(h, TARGET_SUM)
    
    return best_h, best_c5

def random_binary_start(n, rng):
    """Create a random binary start vector"""
    h = np.zeros(n)
    ones_indices = rng.sample(range(n), k=n//2)
    h[ones_indices] = 1.0
    return h

def local_optimize_restart(start_h, remaining_time, rng):
    """Run local optimization from a start vector for a fixed time"""
    if remaining_time < 5:
        return start_h, evaluate_c5(start_h)
    
    # Run with aggressive annealing in short time
    h, c5 = annealed_gradient_ascent(
        start_h, 
        remaining_time * 0.8, 
        alpha0=10, 
        delta0=0.4, 
        T0=5e-3, 
        rng=rng
    )
    return h, c5
