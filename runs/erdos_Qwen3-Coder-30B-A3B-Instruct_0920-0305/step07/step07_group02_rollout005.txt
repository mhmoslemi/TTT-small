# Global constants
TARGET_SUM = 128  # N/2 for N=256

dx = 2.0 / N

def c5_fft(h):
    """Compute C5 using fast FFT-based convolution."""
    n = len(h)
    nfft = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, nfft)
    B = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * B, nfft)[:2*n-1]
    return np.max(conv) * dx

def project_to_sum(h, target_sum):
    """Project h onto the simplex with sum = target_sum."""
    h = np.asarray(h)
    h = np.clip(h, 0, 1)

    # Use water-filling to enforce sum == target_sum
    h_sorted = np.sort(h)[::-1]
    cumulative = np.cumsum(h_sorted)
    n = len(h_sorted)
    k = np.argmax(cumulative > target_sum)
    if k == 0 and cumulative[0] <= target_sum:
        k = n
    theta = (cumulative[k-1] - target_sum) / k if k > 0 else 0
    h_final = np.maximum(h - theta, 0)
    h_final = np.clip(h_final, 0, 1)
    return h_final

def thue_morse(n):
    """Generate Thue-Morse sequence with exactly n/2 ones."""
    tm = [0]
    for i in range(n // 2):
        tm.extend(1 - x for x in tm)
    tm = tm[:n]
    if sum(tm) > TARGET_SUM:
        # Need to flip some 1s to 0s
        ones = np.where(np.array(tm) == 1)[0]
        to_flip = len(ones) - TARGET_SUM
        for i in range(to_flip):
            tm[ones[i]] = 0
    elif sum(tm) < TARGET_SUM:
        zeros = np.where(np.array(tm) == 0)[0]
        to_flip = TARGET_SUM - sum(tm)
        for i in range(to_flip):
            tm[zeros[i]] = 1
    return np.array(tm)

def rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence."""
    rs = [0] * n
    b = 0
    for i in range(n):
        if i == 0:
            rs[i] = 0
        else:
            b = (b + (i & (~i))) & i  # Gosper's hack to count trailing zeros
            rs[i] = rs[i-1] + (1 if b == 0 else -1)
    rs = np.array([1 if x >= 0 else 0 for x in rs])
    # Make sum = 128
    ones = np.where(rs == 1)[0]
    if len(ones) > TARGET_SUM:
        to_flip = len(ones) - TARGET_SUM
        for i in range(to_flip):
            rs[ones[i]] = 0
    elif len(ones) < TARGET_SUM:
        zeros = np.where(rs == 0)[0]
        to_flip = TARGET_SUM - len(ones)
        for i in range(to_flip):
            rs[zeros[i]] = 1
    return rs

def greedy_descent(h0):
    """Improve h0 using greedy descent on the worst shift."""
    h = h0.copy()
    while True:
        corr = correlate(h, 1-h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (len(h) - 1)
        
        if shift >= 0:
            matching = np.where((h[:-shift] == 1) & (h[shift:] == 0))[0]
            mismatching = np.where((h[:-shift] == 0) & (h[shift:] == 1))[0]
        else:
            matching = np.where((h[-shift:] == 1) & (h[:-shift] == 0))[0]
            mismatching = np.where((h[-shift:] == 0) & (h[:-shift] == 1))[0]
        
        if len(matching) == 0 or len(mismatching) == 0:
            break
        
        # Swap one (1,0) with one (0,1)
        i = np.random.choice(matching)
        j = np.random.choice(mismatching)
        index = abs(j - i) if shift >= 0 else abs(i + shift - j)
        if shift >= 0:
            h[i], h[j+shift] = h[j+shift], h[i]
        else:
            h[i-shift], h[j] = h[j], h[i-shift]
    return h
