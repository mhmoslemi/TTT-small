import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global constants
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmin_weights(s, alpha):
    """Computes normalized soft-min weights"""
    s_max = np.max(s)
    exp_s = np.exp(-alpha * (s - s_max))
    return exp_s / np.sum(exp_s)

def weighted_gradient(h, alpha):
    """Computes gradient of softmin with respect to h"""
    s = correlate(h, h, mode='full')
    w = softmin_weights(s, alpha)
    
    grad = np.zeros_like(h)
    n = len(h)
    
    for lag in range(-n+1, n):
        if w[lag + n - 1] > 0:  # Only compute if weight is non-zero
            if lag >= 0:
                grad[:n-lag] += w[lag + n - 1] * h[lag:]
                grad[lag:] += w[lag + n - 1] * h[:n-lag]
            else:
                lag_abs = -lag
                grad[lag_abs:] += w[lag + n - 1] * h[:n-lag_abs]
                grad[:lag_abs] += w[lag + n - 1] * h[lag_abs:]
    
    return grad

def pairwise_move(h, grad, delta, rng):
    """Perform mass transfer move and return (moved, accepted)"""
    # Find indices with extreme gradient values
    src_idx = np.argmin(grad)
    tgt_idx = np.argmax(grad)
    
    # Ensure valid indices
    if h[src_idx] <= 0.001 or h[tgt_idx] >= 0.999:
        return False, False
    
    # Compute move amount
    move_amount = min(delta, h[src_idx], 1 - h[tgt_idx])
    if move_amount < 1e-6:
        return False, False
    
    # Apply move
    h[src_idx] -= move_amount
    h[tgt_idx] += move_amount
    
    return True, True

def annealed_gradient_ascent(h0, time_budget, alpha0, delta0, temp0, rng):
    """Runs the annealed gradient ascent loop"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Annealing variables
    alpha = alpha0
    delta = delta0
    temp = temp0
    
    start_time = time.time()
    
    # Schedule parameters
    alpha_factor = 1.05
    delta_factor = 0.9995
    temp_factor = 0.9995
    update_interval = 200
    
    iteration = 0
    
    while time.time() - start_time < time_budget:
        # Sample from the gradient
        grad = weighted_gradient(h, alpha)
        
        # Perform move
        moved, accepted = pairwise_move(h, grad, delta, rng)
        
        if moved and accepted:
            # Evaluate new C5
            c5_new = evaluate_c5(h)
            
            # Accept update if better
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            else:
                # occasionally accept worsening moves
                delta_c = c5_new - best_c5
                if delta_c > 0 and np.random.random() < np.exp(-delta_c / temp):
                    # accept anyway but revert after the step
                    pass  # not exactly reverting but kind of okay as per objective of this algorithm
                
        # Update annealing parameters
        iteration += 1
        if iteration % update_interval == 0:
            alpha = min(alpha * alpha_factor, 200.0)
            delta *= delta_factor
            temp *= temp_factor
            
    return best_h, best_c5

def random_binary_start(n, rng):
    """Create a random binary vector with exactly n/2 ones"""
    start = np.zeros(n)
    indices = rng.sample(range(n), n//2)
    start[indices] = 1.0
    return start

def local_optimize_restart(start_h, remaining_time, rng):
    """Local optimization from a given start"""
    return annealed_gradient_ascent(
        start_h, 
        remaining_time, 
        alpha0=10.0, 
        delta0=0.3, 
        temp0=5e-3, 
        rng=rng
    )
