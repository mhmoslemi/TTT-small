import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 180
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_grad_max_shift(h, shift, dx):
    """Compute gradient of C5 for the specific shift with maximum correlation"""
    n = len(h)
    grad = np.zeros(n)
    
    if shift >= 0:
        # For shift >= 0: h[i] contributes to derivative in terms of h[i+shift] and h[i-shift]
        if shift > 0:
            # derivative w.r.t h[i] = (1 - h[i+shift]) * dx + (-h[i-shift]) * dx  
            # for i+shift < n: part1 = (1 - h[i+shift]) * dx
            # for i-shift >= 0: part2 = -h[i-shift] * dx
            grad[0:n-shift] += (1 - h[shift:]) * dx
            grad[shift:n] += -h[:n-shift] * dx
        else:
            # shift == 0: derivative is dx * (1 - 2 * h[i])
            grad += -2 * dx * h
    else:
        # For shift < 0 (negative), we still treat as absolute shift
        # shift = -t, so h[i] contributes to derivative in terms of h[i+t] and h[i-t]
        abs_shift = -shift
        grad[abs_shift:n] += (1 - h[:n-abs_shift]) * dx
        grad[:n-abs_shift] += -h[abs_shift:] * dx
        
    return grad

def line_search_step(h, grad, c5_current, max_step=0.1, min_step=1e-6):
    """Perform line search to find appropriate step size"""
    step = max_step
    h_new = project_to_sum_box(h - step * grad, TARGET_SUM)
    c5_new = evaluate_c5(h_new)
    
    # If improvement, return
    if c5_new < c5_current:
        return step, h_new
    
    # Otherwise try smaller steps
    while step > min_step and c5_new >= c5_current:
        step *= 0.5
        h_new = project_to_sum_box(h - step * grad, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
    
    return step, h_new
