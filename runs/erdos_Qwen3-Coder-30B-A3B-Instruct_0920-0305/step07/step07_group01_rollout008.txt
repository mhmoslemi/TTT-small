def fourier_basis(x, m):
    """Cosine basis function"""
    return np.cos(m * np.pi * x)

def fourier_h(x, coeffs):
    """Reconstruct h from Fourier coefficients"""
    # coeffs[0] is for constant term, coeffs[1] for cos(πx), etc.
    h = coeffs[0] * np.ones_like(x)
    for i in range(1, len(coeffs)):
        h += coeffs[i] * fourier_basis(x, i)
    return h
