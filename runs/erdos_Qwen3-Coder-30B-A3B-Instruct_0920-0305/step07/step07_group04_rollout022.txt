import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_h(h):
    """Apply a smoothing operation to make solution more regular"""
    # Simple moving average filter
    kernel = np.ones(5) / 5
    h_smooth = np.convolve(h, kernel, mode='same')
    h_smooth = np.clip(h_smooth, 0, 1)
    return project_to_sum_box(h_smooth, TARGET_SUM)

def optimize_with_constraints(h0, max_iter=500):
    """Optimize using gradient-based method"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Multiple starts with different strategies
    best_h = None
    best_c5 = float('inf')
    
    # Strategy 1: Use reference initialization
    h0 = initialize_from_ref()
    h0 = project_to_sum_box(h0, TARGET_SUM)
    h_refined = optimize_with_constraints(h0, max_iter=500)
    c5_refined = evaluate_c5(h_refined)
    if c5_refined < best_c5:
        best_h = h_refined.copy()
        best_c5 = c5_refined
    
    # Strategy 2: Random initialization + optimization
    for _ in range(5):
        if time.time() - start_time > 0.7 * budget_s:
            break
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = project_to_sum_box(h_rand, TARGET_SUM)
        h_rand = optimize_with_constraints(h_rand, max_iter=300)
        c5_rand = evaluate_c5(h_rand)
        if c5_rand < best_c5:
            best_h = h_rand.copy()
            best_c5 = c5_rand
    
    # Strategy 3: Sparse + smooth optimization
    for _ in range(3):

        h_sparse = np.zeros(n_points)

        c5_sparse = evaluate_c5(h_sparse)
        if c5_sparse < best_c5:
            best_h = h_sparse.copy()
            best_c5 = c5_sparse
    
    # Strategy 4: Local refinement with smoothing
    h_current = best_h.copy()

        if time.time() - start_time > budget_s - 5:
            break
        # Apply smoothing
        h_smooth = smooth_h(h_current)
        # Refine with optimization
        h_refined = optimize_with_constraints(h_smooth, max_iter=100)
        c5_refined = evaluate_c5(h_refined)
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
        h_current = h_refined
    
    return (best_h.tolist(), float(best_c5), n_points)
