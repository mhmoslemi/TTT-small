import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_c5_exact(h, dx):
    """Evaluate the actual C5 value using FFT-based correlation."""
    # Compute h(x) * (1 - h(x + k)) for all shifts k
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return dx * max_corr

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_lp_relaxation(h0, n_points, dx):
    """Solve LP relaxation to get lower bound on C5."""
    # Use the old implementation for LP only
    # Just do the linear relaxation with convex hull via McCormick
    
    # Variables
    h = cp.Variable(n_points, name="h")
    p = cp.Variable((n_points, n_points), name="p")
    T = cp.Variable(name="T")  # Upper bound on max overlap
    
    # Constraints for McCormick envelopes for all pairs (i,j)
    constraints = []
    for i in range(n_points):
        for j in range(i, n_points):
            # McCormick envelopes for h_i * h_j
            constraints.append(p[i, j] >= 0)
            constraints.append(p[i, j] >= h[i] + h[j] - 1)
            constraints.append(p[i, j] <= h[i])
            constraints.append(p[i, j] <= h[j])
    
    # Mass constraint
    constraints.append(cp.sum(h) == n_points / 2)
    
    # Correlation pairs
    # For each shift k, define overlap O_k
    # O_k = sum_{i=0}^{length-k} h[i]*h[i+k]
    # We do it in the fully expanded form 
    
    # Instead of complete overlap for each k, we'll compute a convex relaxation of the overlap:
    # Although we can compute realistic overlaps, this is much more complex, so let's be efficient
    
    # For an initial try, just validating with small grids
    # Perform a coarse binomial relaxation approach
    # Here, we focus directly on binary search via convex relaxations through solver
    
    # Not fully modeling the full constraint; see more folk approach
    # We hardcode basic weakest or most conservative approach to test efficiency:
    # bound as largest affine term, just given total mass:
    # O_k >= (n/2 - T) = sum_{i} h_i * h_{i+k}
    # Thus p_{ij} >= h_i * h_j
    
    # Refine approach
    constraints.append(T >= 0)
    
    # Get any feasible solution back for a quickly checked value
    # As there are so many variables, it's better to just cut down proportionally
    
    # Convexify on full problem
    # This core idea is:
    # We constraint: for each i,j:  p_{ij} >= h_i * h_j (McCormick)
    # Also we bound the overlap such that for each shift:
    # sum_{i} h_i*(1 - h_i+k) >= sum_{i} h_i h_{i+k}  == O_k
    # So  O_k = sum_{i,k} h_i h_{i+k} = sum_{i,k} p_{i,i+k} 
    
    # We implement it as much as practical with sparse matrix approach and convergence
    
    # Construction:
    # Force relaxation that is fast to verify with solver:
    #       Now implement one-step checking with CVXPY and a careful granularity
    
    # Instead of solving the actual full LP for huge matrix because of memory
    # Just implement what matters for small memory and reevaluate it
    
    # Submit static constraints with this LP to a simple solver
    problem = cp.Problem(cp.Minimize(T), constraints)
    
    try:
        # Solve using solver (may have limitations in parallel or quad precision)
        problem.solve(solver=cp.GLPK, verbose=False, time_limit=120)
        if problem.status in [cp.OPTIMAL, cp.OPTIMIZER_ERR]:
            return h.value, T.value
    except Exception as e:
        pass
    return h0, np.inf

def compute_overlap_vector(h):
    """Fast Computation of the correlation vector using FFT."""
    return correlate(h, 1 - h, mode='full')

def compute_subgradient(h, k_star, dx):
    """Compute subgradient of max overlap term at shift k_star."""
    n = len(h)
    g = np.zeros(n)
    
    # For each i, we compute d/dh_i of h_i * (1 - h_{i+k_star})
    for i in range(n):
        # term h_i * (1 - h_{i+k_star})
        # gradient w.r.t. h_i is 1 - h_{i+k_star} (assuming k_star is positive)
        if i + k_star < n:
            g[i] = 1 - h[i + k_star]
        else:
            g[i] = 0  # h_{i+k_star} is out of bounds
        
        # Consider also the negative term
        # recall 1-h_i where h is the original
        # but we one-hot per position aggregation across correlations
        # not more extensive, just focus on main error in direction
        # Actually, it reuses classic as sub-gradient in computation over scalar
    
    # Badly apply the energy component of (1-h_i) components for all:
    # The proper one is to repeated. THANKS. 
    for i in range(n):
        if i - k_star >= 0:
            g[i] -= h[i - k_star]
    
    return g

def subgradient_descent(h0, dx, max_iter=2000, alpha0=0.01):
    """Perform deterministic subgradient descent of the max overlap."""
    h = h0.copy()
    # Project to target sum even before descent
    h = project_to_sum_box(h, len(h) / 2)
    
    for t in range(max_iter):
        # Find current active shift k*
        corr = compute_overlap_vector(h)
        k_star = int(np.argmax(corr) - len(corr) / 2 + 0.5)
        if abs(k_star) < 0.1 * len(h):
            # Bound or roughness, no valid signal
            break
        # If it points to a very edge or trunk, better path to fallback etc.
        
        g = compute_subgradient(h, k_star, dx)
        alpha = alpha0 / np.sqrt(t + 1)
        h = h - alpha * g

        # Project back to CVX set
        h = project_to_sum_box(h, len(h) / 2)
        
        # Tolerance for early stop for stability
        if t > 100 and t % 50 == 0:
            old_c5 = evaluate_c5_exact(h, dx) 
            if old_c5 > 0.381:
                pass  # we stay in case wants to fly
    return h

def local_exchange_refinement(h_start, dx, max_iter=1000):
    """Perform exchange of mass between pairs to reduce overlap."""
    h = h_start.copy()
    n = len(h)
    best_h = h.copy()
    best_value = evaluate_c5_exact(h, dx)
    
    n_pairs = n * (n - 1) // 2
    max_pairs = min(1000, int(n_pairs * 0.05))  # limit to min(a,b) to save time
    
    for i in range(max_iter):
        if i % 100 == 0:
            current = evaluate_c5_exact(h, dx)
            if current < best_value:
                best_h = h.copy()
                best_value = current
        # Sample randomly from pair indices
        try:
            indices = np.random.choice(n, size=(2, max_pairs), replace=True)
            indices = indices.T
        except:
            break
            
        for i1, i2 in indices:
            if i1 == i2:
                continue
                
            capacity = min(h[i2], 1 - h[i1])
            if capacity <= 0:
                continue
                
            delta = capacity * np.random.uniform(0.01, 0.1)
            
            # Try movement
            h[i1] += delta
            h[i2] -= delta
            h = np.clip(h, 0, 1)
            h = project_to_sum_box(h, n / 2)
            
            new_c5 = evaluate_c5_exact(h, dx)
            if new_c5 < best_value:
                best_value = new_c5
                best_h = h.copy()
            else:
                # revert
                h[i1] -= delta
                h[i2] += delta
                h = np.clip(h, 0, 1)
                h = project_to_sum_box(h, n / 2)
    
    return best_h
