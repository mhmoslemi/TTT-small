def eval_c5(h_vals, dx):
    """Likely same approximation code used in main version"""
    h = np.array(h_vals)
    # Compute convolution using FFT directly
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_sum(arr, target_sum):
    """Project vector onto sum = target_sum"""
    current_sum = np.sum(arr)
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return arr
    
    # Simple projection via mean adjustment
    mean_adjust = diff / len(arr)
    new_arr = arr + mean_adjust
    
    # Clip to stay within [0,1]
    return np.clip(new_arr, 0, 1)
