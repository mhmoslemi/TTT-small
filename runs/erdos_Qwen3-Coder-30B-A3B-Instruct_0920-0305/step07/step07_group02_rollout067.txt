import numpy as np

import time

    import cvxpy as cp

except ImportError:
    HAS_CVXPY = False

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def greedy_descent(h0):
    """Greedy descent on the worst shift"""
    h = h0.copy()
    original_c5 = evaluate_c5(h)
    converged = False
    while not converged:
        corr = compute_corr_fft(h)
        # Compute max overlap
        max_k = np.argmax(corr)
        max_overlap = corr[max_k]
        if max_overlap <= 0 or max_overlap == original_c5:
            break
        # Identify mismatched pairs for the worst shift
        # Get the shift value
        shift = max_k - (len(h) - 1)  # Convert index to actual shift value
        # For shift k, corresponding h[i] * h[i+k] that contributes to the overlap
        # The overlap is computing int h(x) * (1-h(x+k)) dx = sum h[i] * (1-h[i+k])
        # Which can be rewritten as sum h[i] - sum h[i] * h[i+k]

        # We look for pairs that will reduce the overlap
        # Find indices i where h[i] = 1 and h[i+k] = 0 (contributes to overlap)
        # Find indices j where h[j] = 0 and h[j+k] = 1 (contributes to overlap)
        # For better overlap, swap one from both groups
        a_indices = []
        b_indices = []
        for i in range(len(h)):
            j = i + shift
            if 0 <= j < len(h):
                if h[i] == 1 and h[j] == 0:
                    a_indices.append(i)
                elif h[i] == 0 and h[j] == 1:
                    b_indices.append(i)
                    
        if len(a_indices) == 0 or len(b_indices) == 0:
            break
        
        # Perform a swap
        i_a = np.random.choice(a_indices)
        i_b = np.random.choice(b_indices)
        h[i_a], h[i_b] = h[i_b], h[i_a]
        converged = False  # Try again
    return h, evaluate_c5(h)

def anneal_binary(h0, max_time_s):
    """Lightweight simulated annealing"""
    h = h0.copy()
    current_c5 = evaluate_c5(h)
    start_time = time.time()
    # Initial temperature
    temp = 0.02
    tmax = 10000
    t = 0
    # To make scanning more efficient without recalculating entirely
    corr = compute_corr_fft(h)
    while time.time() - start_time < max_time_s and t < tmax:
        t += 1
        temp = max(1e-6, 0.02 * (1 - t / tmax))
        
        # Pick two random positions to swap: one 1, one 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)
        
        # Try the swap
        h[i1], h[i0] = h[i0], h[i1]
        new_c5 = evaluate_c5(h)
        delta = new_c5 - current_c5
        
        if delta <= 0 or np.random.rand() < np.exp(-delta / temp):
            current_c5 = new_c5
        else:
            # Revert swap
            h[i1], h[i0] = h[i0], h[i1]
            
        # Periodically do a greedy sweep
        if t % 100 == 0:
            h, current_c5 = greedy_descent(h)
            
    return h, current_c5

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    """Improves using pattern-based neighborhood search"""

    for i in range(max_iter):
        if i % 100 == 0:
            corr_full = compute_corr_fft(h)

            end = min(start + 10, n_points)
            sign = np.random.choice([-1, 1])
            delta = 0.01 * sign
            h_new = h.copy()
            h_new[start:end] += delta
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5:
                h = h_new.copy()
