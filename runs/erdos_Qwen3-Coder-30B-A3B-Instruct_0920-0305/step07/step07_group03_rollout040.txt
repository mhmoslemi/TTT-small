import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_of_corr_at_lag(h, lag):
    """Compute gradient of correlation at specific lag with respect to h"""
    n = len(h)
    g = np.ones(n)
    
    if lag >= 0:
        # i from 0 to n-lag-1 where i+lag < n
        if lag < n:
            g[:n-lag] = 1 - h[lag:]
            g[lag:] -= h[:n-lag]
    else:
        # lag negative: shift to left
        d = -lag
        if d < n:
            g[d:] = 1 - h[:n-d]
            g[:n-d] -= h[d:]
    return g

def minimize_max_correlation_subgradient(h0, max_iter=2000, step_size=0.05):
    """Use subgradient method to minimize max correlation"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # compute correlations for all lags
        corr = compute_corr(h)
        max_val = np.max(corr)
        
        # find indices where maximum is achieved
        indices = np.where(np.abs(corr - max_val) < 1e-10)[0]
        
        # compute average gradient of these lags
        grad_sum = np.zeros_like(h)
        for idx in indices:
            # convert index back to lag (corr has indices from -n+1 to n-1)
            lag = idx - (n_points - 1)
            grad_sum += gradient_of_corr_at_lag(h, lag)
        
        # normalize gradient
        if len(indices) > 0:
            grad = grad_sum / len(indices)
        else:
            grad = np.zeros_like(h)
        
        # step size schedule
        step = step_size / np.sqrt(i + 1)
        
        # update h
        h_new = h - step * grad
        
        # project to bounds and sum constraint
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # evaluate
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new
        
    return best_h, best_c5

def random_binary_initialization():
    """Create random binary initial solution"""
    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    h = np.zeros(n_points)
    h[indices] = 1.0
    return h

def simulated_annealing_binary(h0, temp_start=1.0, temp_end=1e-5, max_iter=10000):
    """Simulated annealing for binary patterns"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    temp = temp_start
    cooling_rate = (temp_end / temp_start) ** (1.0 / max_iter)
    
    iter_count = 0
    while iter_count < max_iter:
        # propose neighbor: swap two random elements (0 and 1)
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Ensure i != j  
        while j == i:
            j = np.random.randint(0, n_points)
            
        # Current and proposed states
        h_old = h.copy()
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # Evaluate new state
        c5_old = evaluate_c5(h_old)
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject
        delta = c5_new - c5_old
        if delta < 0 or np.random.random() < np.exp(-delta / temp):
            h = h_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        
        # Cool temperature
        temp *= cooling_rate
        iter_count += 1
        
    return best_h, best_c5

def insert_peaks(h):
    """Add two sharp peaks to make better overlap suppression"""
    # Locate areas with low values and add two strong spikes
    thresholds = [0.2, 0.5, 0.8]
    for thres in thresholds:
        mask = (h < thres)
        if np.sum(mask) > 10:
            # Set some low values to 1
            indices = np.where(mask)[0]
            for _ in range(min(2, len(indices)//3)):
                idx = np.random.choice(indices)
                h[idx] = 1.0
                # Smooth out nearby values by reducing them slightly
                for offset in [-1, 0, 1]:
                    if 0 <= idx + offset < len(h):
                        h[idx + offset] = max(0, h[idx + offset] - 0.2)
            break
    
    return project_to_sum_box(h, TARGET_SUM)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = None
    best_c5 = float('inf')
    
    # Phase 1: Multiple random restarts with simulated annealing
    phase1_end = start_time + 0.6 * budget_s
    for run_id in range(15):
        if time.time() > phase1_end:
            break
            
        # Initialize with random binary pattern
        h0 = random_binary_initialization()
        
        # Run simulated annealing
        h_sa, c5_sa = simulated_annealing_binary(h0, max_iter=2000)
        
        if c5_sa < best_c5:
            best_h = h_sa.copy()
            best_c5 = c5_sa
    
    # Phase 2: Refine best solution using gradient descent
    phase2_end = start_time + 0.9 * budget_s
    while time.time() < phase2_end:
        h_refined, c5_refined = minimize_max_correlation_subgradient(
            best_h, max_iter=200, step_size=0.05
        )
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
        else:
            # Fine-tune the best solution in case gradient descent doesn't improve
            h_local, c5_local = minimize_max_correlation_subgradient(
                best_h, max_iter=100, step_size=0.02
            )
            if c5_local < best_c5:
                best_h = h_local.copy()
                best_c5 = c5_local
            break
    
    # Phase 3: Final improvement with local refinement
    if best_h is not None:
        for i in range(5):
            if time.time() > start_time + budget_s:
                break
            # Add some small random adjustments
            h_perturbed = best_h + np.random.normal(0, 0.005, n_points)
            h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
            
            # Refine with gradient descent
            h_refined, c5_refined = minimize_max_correlation_subgradient(
                h_perturbed, max_iter=150, step_size=0.03
            )
            
            if c5_refined < best_c5:
                best_h = h_refined.copy()
                best_c5 = c5_refined
    
    # Ensure bounds are respected even after refining
    if best_h is not None:
        best_h = project_to_sum_box(best_h, TARGET_SUM)
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
