def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def fourier_basis(x, m):
    """Return m-th cosine basis function"""
    return np.cos(m * np.pi * x)

def h_from_coeffs(coeffs, x_vals):
    """Reconstruct h from Fourier coefficients"""
    h = np.full_like(x_vals, coeffs[0])
    for m in range(1, len(coeffs)):
        h += coeffs[m] * fourier_basis(x_vals, m)
    return np.clip(h, 0, 1)
