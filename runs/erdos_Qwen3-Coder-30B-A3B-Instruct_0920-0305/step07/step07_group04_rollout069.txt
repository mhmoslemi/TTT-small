import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5_exact(h):
    """Evaluate the true C5 using full correlation"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(h, target_sum):
    """Project onto box [0, 1] with sum = target_sum using binary search"""
    def f(lam):
        proj = np.clip(h + lam, 0, 1)
        return np.sum(proj) - target_sum
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    return np.clip(h + (a + b)/2, 0, 1)

def solve_lp_relaxation(n_points, target_sum, max_time=180):
    """Solve the convex relaxation LP with McCormick envelopes"""
    dx = 2.0 / n_points
    h = cp.Variable(n_points, name='h')
    p = cp.Variable((n_points, n_points), name='p')
    T = cp.Variable(1, name='T')

    # Mass constraint
    constraints = [cp.sum(h) == target_sum, h >= 0, h <= 1]

    # McCormick envelopes
    for i in range(n_points):
        for j in range(i, n_points):
            constraints.extend([
                p[i, j] >= 0,
                p[i, j] >= h[i] + h[j] - 1,
                p[i, j] <= h[i],
                p[i, j] <= h[j]
            ])

    # Overlap constraints
    for k in range(-n_points+1, n_points):
        if k >= 0:
            O_k = cp.sum([p[i, i+k] for i in range(n_points-k)])
        else:
            O_k = cp.sum([p[i-k, i] for i in range(-k, n_points)])
        constraints.append(O_k >= target_sum - T)

    objective = cp.Minimize(T)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.GLPK, time_limit=max_time)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURATE]:
            return h.value, T.value
    except:
        pass
    return None, None

def binary_search_certified_bound(h_initial, n_points, target_sum, dx, budget_s):
    """Binary search for the smallest feasible T with verified bound"""
    start_time = time.time()
    T_low, T_high = 0.0, target_sum
    best_h = h_initial.copy()
    best_c5 = evaluate_c5_exact(h_initial)
    
    while time.time() - start_time < 0.8 * budget_s and T_high - T_low > 1e-6 * dx:
        T_mid = (T_low + T_high) / 2
        def check_feasible(T_val):
            h = cp.Variable(n_points)
            p = cp.Variable((n_points, n_points))
            
            constraints = [cp.sum(h) == target_sum, h >= 0, h <= 1]
            for i in range(n_points):
                for j in range(i, n_points):
                    constraints.extend([
                        p[i, j] >= 0,
                        p[i, j] >= h[i] + h[j] - 1,
                        p[i, j] <= h[i],
                        p[i, j] <= h[j]
                    ])
            for k in range(-n_points+1, n_points):
                if k >= 0:
                    O_k = cp.sum([p[i, i+k] for i in range(n_points-k)])
                else:
                    O_k = cp.sum([p[i-k, i] for i in range(-k, n_points)])
                constraints.append(O_k >= target_sum - T_val)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.GLPK, time_limit=10)
                return prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURATE]
            except:
                return False
        
        if check_feasible(T_mid):
            h_sol, _ = solve_lp_relaxation(n_points, target_sum, 10)
            if h_sol is not None:
                h_sol = project_to_sum_box(h_sol, target_sum)
                c5_sol = evaluate_c5_exact(h_sol)
                if c5_sol <= T_mid * dx:
                    best_h = h_sol
                    best_c5 = c5_sol
                    T_high = T_mid
                else:
                    T_low = T_mid
            else:
                T_low = T_mid
        else:
            T_low = T_mid
    
    return best_h, best_c5

def subgradient_refinement(h_initial, n_points, target_sum, dx, budget_s):
    """Deterministic subgradient descent for polishing"""
    start_time = time.time()
    h = h_initial.copy()
    step_size = 0.01
    
    num_iterations = int(2500 * 0.4)  # For 40% of the remaining budget
    for i in range(num_iterations):
        if time.time() - start_time > 0.4 * budget_s:
            break
            
        corr = np.correlate(h, 1 - h, mode='full')
        k_active = np.argmax(corr)
        
        # Compute subgradient
        g = np.zeros(n_points)
        if k_active < n_points:
            # Forward gradient for shift k_active
            for i in range(n_points):
                g[i] += (1 - h[(i + k_active) % n_points]) if (i + k_active) < n_points else 0
        else:
            # Backward gradient for negative shift
            k_active = n_points - k_active - 1
            for i in range(n_points):
                g[i] += -h[(i - k_active) % n_points] if (i - k_active) >= 0 else 0
        
        # Update with subgradient step
        h = h - step_size * g
        h = project_to_sum_box(h, target_sum)
        
    return h

def exchange_refinement(h_initial, n_points, target_sum, dx, budget_s):
    """Local exchange refinement for fine-tuning"""
    start_time = time.time()
    h = h_initial.copy()
    num_exchanges = 20000
    
    def safe_transfer(h, i, j, delta):
        """Check and perform a transfer"""
        if h[i] >= delta and h[j] <= 1 - delta:
            h[i] -= delta
            h[j] += delta
            return True
        return False
    
    for _ in range(num_exchanges):
        if time.time() - start_time > 0.2 * budget_s:
            break
        i, j = np.random.randint(0, n_points, 2)
        if i != j:
            max_transfer = min(h[i], 1 - h[j])
            if max_transfer > 0:
                delta = 0.1 * max_transfer
                old_c5 = evaluate_c5_exact(h)
                if safe_transfer(h, i, j, delta):
                    new_c5 = evaluate_c5_exact(h)
                    if new_c5 >= old_c5:
                        # Revert if not improving
                        h[i] += delta
                        h[j] -= delta
    
    return h
