from scipy.signal import correlate
import time

# Global variables
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def fourier_basis(x, m):
    """Generate cosine basis function"""
    return np.cos(m * np.pi * x)

def reconstruct_h(a, x):
    """Reconstruct h from Fourier coefficients"""
    h = a[0] * np.ones_like(x)  # Constant term
    for m in range(1, len(a)):
        h += a[m] * fourier_basis(x, m)
    return h
