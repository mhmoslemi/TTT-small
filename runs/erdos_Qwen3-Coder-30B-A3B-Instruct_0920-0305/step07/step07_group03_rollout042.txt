import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_with_gradients(h0, max_iter=1000):
    """Optimize using gradient-based method with projection"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def improve_solution_iterative(h0, max_iter=1000):

    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Small tweaks to few elements
        idxs = np.random.choice(n_points, size=min(10, n_points//4), replace=False)
        for idx in idxs:
            old_val = h[idx]
            # Try multiple small changes
            for delta in [-0.03, -0.015, 0.015, 0.03]:
                h_new = h.copy()
                h_new[idx] += delta
                
                if h_new[idx] < 0 or h_new[idx] > 1:
                    continue
                    
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                new_c5 = evaluate_c5(h_new)
                
                if new_c5 < best_c5:
                    h = h_new.copy()
                    best_c5 = new_c5
                    break
            else:
                continue
            break
        else:
            # If no improvement, perturb slightly
            if np.random.random() < 0.3:
                idx = np.random.randint(0, n_points)
                delta = np.random.uniform(-0.015, 0.015)
                h[idx] += delta
                h = project_to_sum_box(h, TARGET_SUM)
                best_c5 = evaluate_c5(h)
    
    return h

def insert_peaks(h):
    """Add two sharp peaks to the solution"""
    # First, make the values more spread out
    h_copy = h.copy()
    
    # Determine the time points with lower values
    # Prefer placing spikes where they can suppress maximum overlap
    values_sorted = np.sort(h_copy)
    first_quartile = values_sorted[int(0.25*len(values_sorted))]
    low_values = h_copy < first_quartile
    
    # Try to add peaks at low-value locations
    low_indices = np.where(low_values)[0]
    if len(low_indices) > 10:
        # Choose 2 of the lowest indices for peaks
        chosen_indices = np.random.choice(low_indices, size=min(2, len(low_indices)), replace=False)

            h_copy[idx] = 1.0
            # Reduce nearby values slightly
            for offset in [-2, -1, 0, 1, 2]:
                if 0 <= idx + offset < len(h_copy):
                    h_copy[idx + offset] = max(0, h_copy[idx + offset] - 0.1)

        # Fallback: just add some peaks randomly

            lambda x: optimize_with_gradients(x, max_iter=300),

        c5_refined = evaluate_c5(h_refined)
        
        if c5_refined < best_c5:

        c5_final = evaluate_c5(h_final)

            best_h = h_final.copy()
            best_c5 = c5_final

    phase4_end = start_time + budget_s
    while time.time() < phase4_end:
        # Add noise and fine tune
        h_noise = best_h + np.random.normal(0, 0.003, n_points)
        h_noise = project_to_sum_box(h_noise, TARGET_SUM)
        
        h_optimized = optimize_with_gradients(h_noise, max_iter=200)
        c5_optimized = evaluate_c5(h_optimized)
        
        if c5_optimized < best_c5:
            best_h = h_optimized.copy()
            best_c5 = c5_optimized
