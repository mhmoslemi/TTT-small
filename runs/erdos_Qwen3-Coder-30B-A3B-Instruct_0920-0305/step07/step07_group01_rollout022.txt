import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def project_sum_box(h, target_sum):
    """Project vector h onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(h + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def smooth_gradient(h, corr, beta):
    """Smooth gradient using soft-max over all lags"""
    # Center the correlation values for numerical stability
    corr_centered = corr - np.max(corr)
    exp_corr = np.exp(beta * corr_centered)
    weights = exp_corr / np.sum(exp_corr)
    # Compute gradient of the smooth max
    grad = np.zeros_like(h)
    N = len(h)
    for k in range(-N+1, N):
        if k >= 0:
            # Forward shift
            shifted = np.concatenate([h[k:], np.zeros(k)])
        else:
            # Backward shift
            shifted = np.concatenate([np.zeros(-k), h[:k]])
        grad += weights[k+N-1] * (1 - shifted)
        # Subtract the conjugate term (accounting for h(i+k) when computing h(i) contribution)
        if k >= 0:
            grad += weights[k+N-1] * (-shifted)
        else:
            # Backward shift contribution
            grad -= weights[-k+N-1] * h
    
    return (2.0 / N) * grad

def max_lag_gradient(h, corr):
    """Gradient of exact max over all lags (sub-gradient)"""
    idx_worst = np.argmax(corr)
    N = len(h)
    grad = np.zeros_like(h)
    if idx_worst < N:  # Forward shift
        k = idx_worst
        grad[:N-k] += (1 - h[k:]) * (2.0 / N)
        grad[k:] -= h[:-k] * (2.0 / N)
    else:  # Backward shift
        k = idx_worst - N + 1
        grad[k:] += (1 - h[:-k]) * (2.0 / N)
        grad[:N-k] -= h[k:] * (2.0 / N)
    return grad

def mass_transfer_move(h, grad, epsilon=0.01):
    """Perform greedy mass transfer to speed up convergence."""
    d = -grad
    N = len(h)

    # Identify sources (indices where to take mass) and sinks
    sources = np.where(d < 0)[0]
    sinks = np.where(d > 0)[0]
    
    if len(sources) == 0 or len(sinks) == 0:
        return h

    # Sort to maximize effectiveness
    sources = sources[np.argsort(-d[sources])]
    sinks = sinks[np.argsort(d[sinks])]

    # Create a working copy to avoid side effects
    h_new = h.copy()
    
    # Perform greedy transfers
    for s in sources:
        if h_new[s] <= 0:
            continue
        for t in sinks:
            if h_new[t] >= 1:
                continue
            transfer = min(h_new[s], 1 - h_new[t], epsilon)
            if transfer == 0:
                continue
            h_new[s] -= transfer
            h_new[t] += transfer
            if h_new[s] <= 0:
                break
            
    # Project back to ensure feasibility
    h_new = project_sum_box(h_new, len(h_new) / 2)
    return h_new

def backtracking_line_search(h, grad, corr, beta, alpha0=0.2, alpha_min=5e-4):
    """Backtracking line search for descent with decrease in C5"""
    alpha = alpha0
    c5_prev = evaluate_c5(h)
    while alpha >= alpha_min:
        h_trial = h - alpha * grad
        h_trial = project_sum_box(h_trial, len(h) / 2)
        c5_new = evaluate_c5(h_trial)
        if c5_new < c5_prev:
            return h_trial, c5_new, alpha
        alpha /= 2
    return h, c5_prev, 0.0
