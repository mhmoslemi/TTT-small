import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_search(h0, max_iter=500):
    """Local search with coordinate descent"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Try small adjustments to each coordinate
    for i in range(max_iter):
        improved = False
        for j in range(n_points):
            # Try positive and negative small steps
            for step in [0.01, -0.01]:
                h_new = h.copy()
                h_new[j] += step
                h_new = np.clip(h_new, 0, 1)
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5:
                    h = h_new.copy()
                    best_h = h.copy()
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    
    return best_h

def genetic_algorithm(max_generations=50, population_size=30):
    """Genetic algorithm for global search"""
    # Initialize population
    population = []
    for _ in range(population_size):
        h = adaptive_sampling_initialization()
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    for generation in range(max_generations):
        # Evaluate fitness (lower C5 is better)
        fitness = []
        for h in population:
            c5 = evaluate_c5(h)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Selection (tournament selection)
        selected = []
        for _ in range(population_size):
            idx1, idx2 = np.random.randint(0, population_size, 2)
            if fitness[idx1] < fitness[idx2]:
                selected.append(population[idx1].copy())
            else:
                selected.append(population[idx2].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < population_size else selected[0]
            
            # Uniform crossover
            mask = np.random.random(n_points) < 0.5
            child1 = np.where(mask, parent1, parent2)
            child2 = np.where(mask, parent2, parent1)
            
            # Mutation
            mutation_rate = 0.1 / (generation + 1)
            mutation_mask = np.random.random(n_points) < mutation_rate
            child1[mutation_mask] = np.random.uniform(0, 1, np.sum(mutation_mask))
            child2[mutation_mask] = np.random.uniform(0, 1, np.sum(mutation_mask))
            
            # Project to constraints
            child1 = project_to_sum_box(child1, TARGET_SUM)
            child2 = project_to_sum_box(child2, TARGET_SUM)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
    
    return best_h, best_c5

            ga_h, ga_c5 = genetic_algorithm(max_generations=10, population_size=20)
            if ga_c5 < best_c5:

        ls_c5 = evaluate_c5(ls_h)
        if ls_c5 < best_c5:
            best_h = ls_h.copy()
            best_c5 = ls_c5
    
    # Phase 3: Fine-tuning with gradient-free optimization
    final_end = start_time + budget_s
    while time.time() < final_end:
        # Add small random perturbations
        h_perturbed = best_h + np.random.normal(0, 0.002, n_points)

        h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
        
        # Local search on perturbed solution
        try:
            refined_h = local_search(h_perturbed, max_iter=100)
            refined_c5 = evaluate_c5(refined_h)

                best_h = refined_h.copy()
                best_c5 = refined_c5
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
