import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global variables
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmin_weights(s, alpha):
    """Compute normalized soft-min weights"""
    # Subtract max for numerical stability
    s_shifted = s - np.max(s)
    exp_s = np.exp(-alpha * s_shifted)
    return exp_s / np.sum(exp_s)

def weighted_gradient(h, alpha):
    """Compute gradient of softmin with respect to h"""
    # Compute autocorrelation
    s = correlate(h, h, mode='full')
    
    # Compute weights
    w = softmin_weights(s, alpha)
    
    # Compute weighted gradient
    grad = np.zeros_like(h)
    n = len(h)
    
    # Loop over all lags
    for lag in range(-(n-1), n):
        if lag >= 0:
            # h[i] * h[i+lag] term
            if lag < n:
                grad[lag:] += w[lag + (n-1)] * h[:n-lag]
            # h[i] * h[i-lag] term  
            if lag > 0:
                grad[:n-lag] += w[lag + (n-1)] * h[lag:]
        else:
            # h[i] * h[i+lag] term (lag is negative) 
            lag_abs = -lag
            if lag_abs < n:
                grad[:n-lag_abs] += w[lag + (n-1)] * h[lag_abs:]
            # h[i] * h[i-lag] term (lag is negative, so -lag is positive)
            if lag < 0:
                grad[lag_abs:] += w[lag + (n-1)] * h[:n-lag_abs]
    
    return grad

def pairwise_move(h, grad, delta, rng):
    """Perform a pairwise mass transfer move"""
    # Find source (most negative gradient) and target (most positive gradient)
    neg_indices = np.where(grad < 0)[0]
    pos_indices = np.where(grad > 0)[0]
    
    if len(neg_indices) == 0 or len(pos_indices) == 0:
        # Fallback to random indices
        i_src = rng.randint(0, len(h)-1)
        i_tgt = rng.randint(0, len(h)-1)
    else:
        i_src = neg_indices[np.argmin(grad[neg_indices])]
        i_tgt = pos_indices[np.argmax(grad[pos_indices])]
        
    # Ensure indices are valid
    if h[i_src] <= 0 or h[i_tgt] >= 1:
        return False, h.copy(), False
    
    # Compute transfer amount
    delta = min(delta, h[i_src], 1 - h[i_tgt])
    if delta <= 1e-10:
        return False, h.copy(), False
    
    # Perform transfer
    h[i_src] -= delta
    h[i_tgt] += delta
    
    # Ensure bounds
    h[i_src] = max(0, h[i_src])
    h[i_tgt] = min(1, h[i_tgt])
    
    return True, h.copy(), True

def annealed_gradient_ascent(h0, time_budget, alpha0, delta0, T0, rng):
    """Annealed gradient ascent with simulated annealing"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    alpha = alpha0
    delta = delta0
    T = T0
    
    start_time = time.time()
    iterations = 0
    accepted_moves = 0
    
    while time.time() - start_time < time_budget:
        iterations += 1
        
        # Compute gradient
        grad = weighted_gradient(h, alpha)
        
        # Try a move
        moved, h_new, accepted = pairwise_move(h, grad, delta, rng)
        
        if moved:
            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
            
            # Accept with Metropolis rule (if not better)
            if not accepted:
                if new_c5 <= best_c5 or np.random.rand() < np.exp(-(new_c5 - best_c5) / T):
                    h = h_new.copy()
                    accepted_moves += 1
                else:
                    # Revert move
                    pass
            else:
                h = h_new.copy()
                accepted_moves += 1
            
            # Update parameters with schedule
            if iterations % 500 == 0:
                alpha *= 1.2
                alpha = min(alpha, 200)
                delta *= 0.9995
                T *= 0.9995
                
                # Reset step size occasionally
                if iterations % 5000 == 0:
                    delta = 0.4  # Reset to larger value
                
                # Occasionally reproject to maintain feasibility
                if iterations % 1000 == 0:
                    h = project_to_sum_box(h, TARGET_SUM)
    
    return best_h, best_c5

def random_binary_start(n, rng):
    """Create a random binary start vector"""
    h = np.zeros(n)
    ones_indices = rng.sample(range(n), n//2)
    h[ones_indices] = 1.0
    return h

def local_optimize_restart(start_h, remaining_time, rng):
    """Run a local optimization with time limit"""
    # Use a low alpha for discovery, higher for refinement
    alpha0 = 10.0
    delta0 = 0.4
    T0 = 5e-3
    
    # Run for a fraction of remaining time
    run_time = min(remaining_time, 20.0)
    return annealed_gradient_ascent(start_h, run_time, alpha0, delta0, T0, rng)
