import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
N = 256
dx = 2.0 / N
TARGET_SUM = N // 2

def c5_fft(h):
    """Compute C5 using FFT-based correlation"""
    h = np.asarray(h)
    n = len(h)
    nfft = 1 << ((2*n-1).bit_length())  # next power of two
    H = np.fft.rfft(h, nfft)
    B = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * B, nfft)[:2*n-1]
    return np.max(conv) * dx

def project_to_sum(h, target_sum):
    """Project h onto the simplex with sum = target_sum"""
    h = np.asarray(h)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Water filling approach
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Find where to cut
    i = 0
    while i < len(h_sorted) and cumsum[i] > (i + 1) * target_sum / (i + 1):
        i += 1
    
    # Compute threshold
    if i == 0:
        threshold = target_sum / n
        h_proj = np.clip(h, 0, threshold)
    else:
        threshold = (target_sum - cumsum[i-1]) / (n - i)
        h_proj = np.clip(h, 0, threshold)
    
    # Normalize and ensure exact sum
    h_proj = h_proj * target_sum / np.sum(h_proj)
    return np.clip(h_proj, 0, 1)

def thue_morse(n):
    """Generate Thue-Morse sequence of length n"""
    if n <= 0:
        return np.array([])
    seq = [0]
    for i in range(n):
        seq += [1 - x for x in seq]
    return np.array(seq[:n]) % 2

def rudin_shapiro(n):
    """Generate Rudin-Shapiro sequence of length n"""
    if n <= 0:
        return np.array([])
    # Start with [0] and iteratively build
    s = [0]
    for k in range(n.bit_length()):
        s_next = []
        for x in s:
            s_next.append(x)
            s_next.append(1 - x)
        s = s_next
    return np.array(s[:n])

def greedy_descent(h_start):
    """Greedy descent on the worst shift"""
    h = np.array(h_start)
    h = project_to_sum(h, TARGET_SUM)
    
    max_iter = 10000
    for _ in range(max_iter):
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (N - 1)
        
        # Get mismatched pairs
        if shift >= 0:
            ones_i = np.where((h[:N-shift] == 1) & (h[shift:] == 0))[0]
            zeros_i = np.where((h[:N-shift] == 0) & (h[shift:] == 1))[0]
        else:
            ones_i = np.where((h[-shift:] == 1) & (h[:-shift] == 0))[0]
            zeros_i = np.where((h[-shift:] == 0) & (h[:-shift] == 1))[0]
        
        if len(ones_i) == 0 or len(zeros_i) == 0:
            break
        
        i = np.random.choice(ones_i)
        j = np.random.choice(zeros_i)
        if shift >= 0:
            j += shift
        else:
            i -= shift
        
        # Perform swap
        h[i], h[j] = h[j], h[i]
    
    return h, c5_fft(h)

def simulated_annealing(start_h, timeout):
    """Simulated Annealing on binary vectors"""
    np.random.seed(int(time.time() * 1000) % (2**32 - 1))
    current_h, current_c5 = greedy_descent(start_h)
    best_h, best_c5 = current_h.copy(), current_c5
    t0 = 0.02
    last_improvement = 0
    iteration = 0
    
    start_time = time.time()
    
    while time.time() - start_time < timeout:
        # Temperature schedule
        elapsed = time.time() - start_time
        t = t0 * (1.0 - elapsed / timeout)
        if t <= 1e-8:
            break
            
        # Generate move
        ones = np.where(current_h == 1)[0]
        zeros = np.where(current_h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Try the swap
        h_new = current_h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_c5 = c5_fft(h_new)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5)/t):
            current_h = h_new.copy()
            current_c5 = new_c5
            
            if new_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = new_c5
                last_improvement = iteration
                
        iteration += 1
        
        if iteration % 500 == 0:
            # Mini-greedy step
            h_greedy, c5_greedy = greedy_descent(best_h)
            if c5_greedy < best_c5:
                best_h = h_greedy.copy()
                best_c5 = c5_greedy
                
    return best_h, best_c5
