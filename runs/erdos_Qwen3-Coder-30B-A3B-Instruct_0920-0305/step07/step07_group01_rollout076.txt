import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def perturb_solution(h, strength=0.01):
    """Create a perturbed version of the solution"""
    perturbed = h + np.random.normal(0, strength, n_points)
    return project_to_sum_box(perturbed, TARGET_SUM)

def local_search_step(h, step_size=0.01, max_iter=50):
    """Simple local search step"""
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for _ in range(max_iter):
        # Create neighbor by slightly modifying two points
        i1, i2 = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-step_size, step_size)
        
        # Try to adjust both points
        h_new = h.copy()
        h_new[i1] = np.clip(h[i1] + delta, 0, 1)
        h_new[i2] = np.clip(h[i2] - delta, 0, 1)
        
        # Project to correct sum
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
    
    return best_h, best_c5

    return project_to_sum_box(perturbed, TARGET_SUM)
