import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5_fft(h):
    """Fast evaluation of C5 using FFT"""
    dx = 2.0 / len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def greedy_descent(h):
    """Greedy descent on the worst shift"""
    h = h.copy()
    dx = 2.0 / len(h)
    max_iter = 1000
    for _ in range(max_iter):
        corr = correlate(h, h, mode="full")
        # The overlap integral for shift k is dx*(len(h) - corr[k + len(h) - 1])
        # So overlap is maximized when (len(h) - corr) is minimized
        # i.e. when corr is maximized
        worst_shift_idx = np.argmax(corr)
        overlap_at_worst = corr[worst_shift_idx]
        worst_shift = worst_shift_idx - (len(h) - 1)
        
        # identify mismatched pairs for the worst shift
        A = []
        B = []
        k = worst_shift
        for i in range(len(h)):
            if k >= 0:
                j = i + k
                if j < len(h):
                    if h[i] == 1 and h[j] == 0:
                        A.append(i)
                    elif h[i] == 0 and h[j] == 1:
                        B.append(i)
            else:
                j = i + k
                if 0 <= j < len(h):
                    if h[i] == 1 and h[j] == 0:
                        A.append(i)
                    elif h[i] == 0 and h[j] == 1:
                        B.append(i)
        
        if len(A) == 0 or len(B) == 0:
            break
            
        i = np.random.choice(A)
        j = np.random.choice(B)
        
        # Perform the swap
        h[i], h[j] = h[j], h[i]
        
    return h, compute_c5_fft(h)

def anneal_binary(h, max_time):
    """Simple simulated annealing on binary vectors"""
    dx = 2.0 / len(h)
    current_c5 = compute_c5_fft(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    start_time = time.time()
    T0 = 0.02
    Tmax = 10000
    t = 0
    accepted = 0
    while time.time() - start_time < max_time:
        t += 1
        if t > Tmax:
            break
            
        # temperature schedule
        T = T0 * (1 - t / Tmax)
        if T <= 0:
            break
            
        # propose a move: swap two bits
        i = np.random.randint(len(h))
        j = np.random.randint(len(h))
        if i == j:
            continue
            
        # save old
        old_h = h.copy()
        old_c5 = current_c5
        
        # perform swap
        h[i], h[j] = h[j], h[i]
        
        # compute new C5
        new_c5 = compute_c5_fft(h)
        
        # decide whether to accept
        delta = new_c5 - old_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_c5 = new_c5
            accepted += 1
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # revert swap
            h = old_h
            current_c5 = old_c5
            
        if accepted % 100 == 0:
            # every 100 accepted moves, do a greedy sweep
            h, c5 = greedy_descent(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    N = 120
    dx = 2.0 / N
    target_sum = N // 2
    
    # ---- 1. MILP ----
    h = cp.Variable(N, boolean=True)
    lam = cp.Variable(integer=True)
    
    # auxiliary binary variables
    w_vars = {}
    constraints = [cp.sum(h) == target_sum]
    
    # Define domain shifts
    for k in range(-(N-1), N):
        # valid indices for this shift k:
        start_idx = max(0, -k)
        end_idx = min(N, N-k)
        idxs = np.arange(start_idx, end_idx)
        if len(idxs) > 0:
            # We want to define w[i,k] for i in idxs, j = i + k:
            w = cp.Variable(len(idxs), boolean=True)
            i = idxs
            j = i + k
            # Add linearization constraints:
            constraints += [
                w <= h[i],
                w <= h[j],
                w >= h[i] + h[j] - 1
            ]
            # Define correlation constraint for this shift:
            constraints += [cp.sum(w) >= lam]
            w_vars[k] = w
    
    # Objective: maximize λ
    prob = cp.Problem(cp.Maximize(lam), constraints)
    
    # Add symmetry breaking constraint:
    constraints.append(h[0] == 1)
    prob = cp.Problem(cp.Maximize(lam), constraints)
    
    # Solve with time limit
    remaining_budget = 0.55 * budget_s
    try:
        prob.solve(solver=cp.CBC, max_seconds=remaining_budget, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_vals = np.round(h.value).astype(int)
            h_vals = np.clip(h_vals, 0, 1)
            # correct sum
            actual_sum = np.sum(h_vals)
            if actual_sum != target_sum:
                # need to fix the sum via swapping
                ones = np.where(h_vals == 1)[0]
                zeros = np.where(h_vals == 0)[0]
                # convert excess 1's to 0's
                while actual_sum > target_sum:
                    idx = np.random.choice(ones)
                    h_vals[idx] = 0
                    actual_sum -= 1
                # convert excess 0's to 1's
                while actual_sum < target_sum:
                    idx = np.random.choice(zeros)
                    h_vals[idx] = 1
                    actual_sum += 1
            
            # Extract lambda from optimization
            lambda_val = int(round(lam.value))
            c5 = 1 - (2 * lambda_val) / N
            h_best = h_vals
            
        else:
            # fallback: random binary
            h_vals = np.random.choice([0, 1], size=N, p=[0.5, 0.5])
            while np.sum(h_vals) != target_sum:
                if np.sum(h_vals) > target_sum:
                    idx = np.random.choice(np.where(h_vals == 1)[0])
                    h_vals[idx] = 0
                else:
                    idx = np.random.choice(np.where(h_vals == 0)[0])
                    h_vals[idx] = 1
            h_best = h_vals
            c5 = compute_c5_fft(h_best)
    except Exception:
        # Fallback to heuristic
        h_vals = np.random.choice([0, 1], size=N, p=[0.5, 0.5])
        while np.sum(h_vals) != target_sum:
            if np.sum(h_vals) > target_sum:
                idx = np.random.choice(np.where(h_vals == 1)[0])
                h_vals[idx] = 0
            else:
                idx = np.random.choice(np.where(h_vals == 0)[0])
                h_vals[idx] = 1
        h_best = h_vals
        c5 = compute_c5_fft(h_best)
    
    best_h = h_best
    best_c5 = c5
    
    # ---- 2. Greedy local improvement ----
    elapsed = time.time() - start_time
    if elapsed < 0.75 * budget_s:
        best_h, best_c5 = greedy_descent(best_h)
    
    # ---- 3. Simulated annealing (optional) ----
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        best_h, best_c5 = anneal_binary(best_h, remaining)
    
    return (best_h.tolist(), float(best_c5), N)
