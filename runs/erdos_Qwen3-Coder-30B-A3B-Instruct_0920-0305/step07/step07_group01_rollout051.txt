import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 180
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def evaluate_c5_all(h):
    """Evaluate C5 for all shifts"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * corr

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_grad_max_shift(h, shift, dx):
    """Compute gradient of C5 for the specific shift that maximizes C5"""
    n = len(h)
    grad = np.zeros(n)
    
    # Convert shift to a lag index in correlation array
    lag = shift + (n - 1)
    
    if shift >= 0:
        # Handle positive shift case
        # g[i] += (1 - h[i+shift]) * dx for i in valid range
        # g[i] += -h[i-shift] * dx for i in valid range
        if shift < n:
            grad[0:n-shift] += (1 - h[shift:]) * dx
            grad[shift:] += -h[:n-shift] * dx
    else:
        # Handle negative shift case
        shift_abs = -shift
        # g[i] += (1 - h[i-shift_abs]) * dx for i >= shift_abs
        # g[i] += -h[i+shift_abs] * dx for i < n-shift_abs
        if shift_abs < n:
            grad[shift_abs:] += (1 - h[:n-shift_abs]) * dx
            grad[:n-shift_abs] += -h[shift_abs:] * dx
    
    return grad
