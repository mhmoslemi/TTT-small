import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr_and_weights(h, eps):
    """Compute correlation and softmax weights"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    w_raw = np.exp((corr - max_corr) / eps)
    w_norm = w_raw / np.sum(w_raw)
    return corr, w_norm

def compute_gradient(h, w_norm):
    """Compute gradient using convolution"""
    # The gradient is sum_s w_s * ( (1 - h_{i+s}) - h_{i-s} )
    # This can be computed as:
    # Gradient = 1 - (conv_rev + conv_wh)
    # where conv_rev is convolution of w_rev with h, 
    # and conv_wh is convolution of w with h
    
    conv_wh = np.convolve(w_norm, h, mode='full')
    conv_rev = np.convolve(w_norm[::-1], h, mode='full')
    
    offset = len(h) - 1
    gradient = 1.0 - (conv_rev[offset:offset+len(h)] + conv_wh[offset:offset+len(h)])
    return gradient

def gradient_descent_optimization(h_start, eps_schedule, iterations_per_eps, target_sum):
    """Perform gradient descent optimization"""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for eps_idx, (eps, max_iter) in enumerate(zip(eps_schedule, iterations_per_eps)):
        for iteration in range(max_iter):
            # Compute correlation and weights
            corr, w_norm = compute_corr_and_weights(h, eps)
            
            # Compute gradient
            grad = compute_gradient(h, w_norm)
            
            # Gradient step
            step_size = 0.1 * eps  # Adjustable step size
            h_new = h - step_size * grad
            
            # Project to feasible set
            h_new = project_to_sum_box(h_new, target_sum)
            h = h_new
            
            # Evaluate and update best
            if iteration % 20 == 0:
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    
    return best_h, best_c5

def initialize_h_random(n, target_sum):
    """Initialize random h with proper sum"""
    h = np.random.uniform(0, 1, n)
    h = project_to_sum_box(h, target_sum)
    return h
