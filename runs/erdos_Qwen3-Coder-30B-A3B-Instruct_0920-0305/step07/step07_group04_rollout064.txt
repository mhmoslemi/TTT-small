def compute_corr_fft(h):
    """Fast evaluation of correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto [0,1] box constraints with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def subsgrad_refine_h(h0, max_iter=2000):
    """Refine h with projected sub Gradients"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        corr = compute_corr_fft(h)
        k_star = np.argmax(corr)
        
        # Compute subgradient for active shift
        g = np.zeros(n_points)
        for i in range(n_points):
            next_idx = i + k_star
            prev_idx = i - k_star
            g[i] = (1 - h[next_idx % n_points]) - (h[prev_idx] if prev_idx >= 0 else 0)
        
        # Simple gradient step
        alpha = 0.5 * dx / (i + 1) ** 0.5
        h_new = h - alpha * g
        
        # Clip and project as needed
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
    return best_h, best_c5
