import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def project_to_sum(h, target_sum):
    """Project h onto sum = target_sum"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum == target_sum:
        return h
    diff = target_sum - h_sum
    if diff > 0:
        # Need to increase some values
        # We increase the values that were originally at 0 (in order to keep bounds)
        indices = np.argsort(h)
        j = 0
        while diff > 1e-12 and j < len(h):
            if h[indices[j]] < 1:
                inc = min(1 - h[indices[j]], diff)
                h[indices[j]] += inc
                diff -= inc
            j += 1
    else:
        # Need to decrease some values
        indices = np.argsort(h)[::-1]
        j = 0
        while diff < -1e-12 and j < len(h):
            if h[indices[j]] > 0:
                dec = min(h[indices[j]], -diff)
                h[indices[j]] -= dec
                diff += dec
            j += 1
    return h

def evaluate_c5(h):
    """Evaluate the C5 value for a given h"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def solve_convex_relaxation(h0, n_points, dx, T, solver='ECOS'):
    """Solve the convex relaxation of the problem with fixed T"""
    # Use cvxpy
    h = cp.Variable(n_points)
    # Variables for the product h_i * (1 - h_j) for all ordered pairs (i,j)
    p = cp.Variable((n_points, n_points), PSD=False)  # could be sparse

    # Add equality constraint for mass
    mass = cp.sum(h) == n_points / 2.
    constraints = [0 <= h, h <= 1, mass]
    
    # Add McCormick envelopes
    for i in range(n_points):
        for j in range(i, n_points):
            # McCormick envelopes:
            # p[i, j] >= h[i] + h[j] - 1
            # p[i, j] >= 0
            # p[i, j] <= h[i] 
            # p[i, j] <= h[j]
            constraints.append(p[i, j] >= h[i] + h[j] - 1)
            constraints.append(p[i, j] >= 0)
            constraints.append(p[i, j] <= h[i])
            constraints.append(p[i, j] <= h[j])
            # Symmetry
            if i != j:
                constraints.append(p[j, i] == p[i, j])
    
    # Add overlap bounds (we want all overlaps to be <= T * dx)
    # The overlap for shift k is sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k])
    # We define the contribution via p[i, i+k]
    for k in range(-n_points+1, n_points):  # iterate over all shifts
        shift_sum = 0
        num = 0
        for i in range(n_points):
            if i + k >= 0 and i + k < n_points:
                shift_sum += p[i, i+k]
                num += 1
        if num > 0:  # avoid redundant constraints
            # The overlap is: sum h[i] * (1 - h[i+k])
            # = sum h[i] - sum h[i] * h[i+k]
            # = n/2 - sum h[i]*h[i+k]
            # = n/2 - shift_sum because h[i+k] is paired with h[i]
            # So the shift overlap is n/2 - shift_sum
            # We want overlap <= T * dx 
            # That is: n/2 - shift_sum <= T * dx
            # So shift_sum >= n/2 - T * dx
            # But we want the correlation to be ≤ T*dx  
            # Here, we compute: R_k = (n/2) - O_k, O_k = sum h[i]*h[i+k]  (there's another way to define!)
            # Let's fix the indexing via correlation definition
            # We want h[i] * (1 - h[i+k]) becomes expand: h[i] - h[i]*h[i+k]
            # Correlation R_k = sum h[i] - sum h[i] * h[i+k] 
            # To make the solution consistent, the interpretation should be:
            # We say the overlap O_k is the sum over i of h[i] * h[i+k]  
            # Then R_k = (sum h[i]) - O_k = n/2 - O_k 
            # If we want R_k <= T*dx --> O_k >= n/2 - T*dx
            # But this seems inconsistent with the result of using the actual overlap integral via correlate
            # Let's write down the full correlation:
            # h(x) * (1 - h(x+shift)) = h(x) - h(x)*h(x+shift)  
            # The integral is ∫ h(x) dx - ∫ h(x)*h(x+shift) dx = (n/2) - ∫ h(x)*h(x+shift) dx
            # Say Correlation(k) = ∫ h(x) * (1 - h(x+k)) dx = (n/2) - ∫ h(x)*h(x+k) dx 
            # We want ∫ h(x) * (1 - h(x+k)) dx <= T*dx, so 
            # (n/2) - ∫ h(x)*h(x+k) dx <= T*dx
            # ∫ h(x)*h(x+k) dx >= (n/2) - T*dx
            # Let p[i,j] = h[i] * h[j]
            # The overlap contribution for shift k is sum_{i=0}^{n-k-1} p[i, i+k]  
            # OR do: sum_{i=0}^{n-k-1} h[i] * h[i+k]  = shift_sum
            # We have this constraint:
            # shift_sum >= (n/2) - T*dx
            # But we can also write: shift_sum = sum h[i] * h[i+k]  
            # Then: Sum_{k=-n+1}^{n-1} (n/2 - shift_sum_k) <= T * dx * (2*n-1) 
            # But not that one.
            # Maybe language is for the overlap, which is 
            # ∫ h(x) h(x+k) dx, and correlation of the 1-h is what we minimize:
            # R_k = ∫ h(x) * (1-h(x+k)) dx = ∫ h(x) dx - ∫ h(x)*h(x+k) dx = (n/2) - O_k
            # We want max_k R_k <= T*dx, over all possible shifts. Therefore the corresponding
            # worst-case constraint is: max_k (n/2 - O_k) <= T*dx.  
            # So min_k O_k >= n/2 - T*dx
            # Let's put them into a semidefinite form for (some)
            # Let's attempt the dual constraint:
            # For any shift k, shift_sum_k >= (n/2) - T*dx.            
            # This is a constraint. But we also want to minimize this.
            constraints.append(shift_sum >= (n_points / 2.) - T * dx)
    
    # Now the constraint to involve p[i,j] explicitly for overlaps:
    # The expression for O_k = sum_{i=0}^{n-k-1} h[i] * h[i+k], 
    # can write as sum_{i=0}^{n-k-1} p[i, i+k].  This should be put into constraint.
    # num is n-k for shift k = [n-k, n-k+1, ... n-1] such that j = i+k gives valid i, j indices.
    # But then the sum loop is for 
    # for k in range(1-n, n):  
    #     shift_sum = 0
    #     for i = 0 to n-k-1:   # i in [0: n-k-1]  
    #         shift_sum += p[i, i+k]
    #     constraints.append(shift_sum >= (n/2) - T*dx)
    
    # Correct way is:
    for k in range(1-n_points, n_points):  # x-shift in [-n+1, n-1]
        # this is shift k
        # for i in [0, ... max(0, n-k-1)] i.e. the domain over which we form the sum h[i] * h[i+k]
        shift_sum = 0
        num = 0
        # case k >= 0, indices i, i+k where 0 <= i <= n-k-1, so i+k <= n-1
        # case k < 0, i in [0], ..., i <= n-1+k, so it's i in [0, n+k-1]  
        if k >= 0:
            # i from 0 to n-k-1, j = i+k  
            for i in range(0, n_points-k):
                shift_sum += p[i, i+k]
                num += 1
        else:
            # i from -k to n-1, let i+k >= 0, so i >= -k, i+k <= n-1, so i <= n-1-k, so i in [-k, n-k-1]
            for i in range(-k, n_points+k):
                shift_sum += p[i, i+k]
                num += 1
        if num > 0:
            constraints.append(shift_sum >= (n_points / 2.) - T * dx)
    
    # We don't need to minimize anything, just check feasibility
    # The problem is: minimize T subject to feasibility of all these constraints
    # But it's a feasibility check with fixed T.
    prob = cp.Problem(cp.Minimize(0), constraints)
    try:
        prob.solve(solver=solver, verbose=False, max_iters=3000, abstol=1e-9, reltol=1e-9)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value
        else:
            return None
    except:
        return None

def check_and_evaluate(h, T_min, T_max, dx):
    """Evaluates if a solution H corresponding to fixed T is valid"""
    h_eval = np.array(h)
    corr = compute_corr_fft(h_eval)
    max_correlation = c5_from_corr(corr, len(h_eval)) * dx
    
    return max_correlation <= T_max * dx

def refine_solution_subgrad(h, max_iter=2000):
    """Improve the solution using sub-gradient descent""" 
    h = np.array(h).copy()
    h_sum = np.sum(h)
    target_sum = len(h) / 2
    if h_sum != target_sum:
        diff = target_sum - h_sum
        h = h + diff/len(h)  # Try direct additive
        h = np.clip(h, 0, 1)
        h = project_to_sum(h, target_sum)
    
    alpha0 = 0.1  # Initial learning rate
    for t in range(max_iter):
        # Compute the correlation
        corr = compute_corr_fft(h)
        # Find the index where maximum occurs
        max_k = np.argmax(corr)
        # Compute the subgradient
        sub_grad = np.zeros_like(h)
        # case: h_i * (1 - h_{i+k}) contributes to correlation 
        # sub_grad_i = (1 - h_{i+k}) - h_{i-k} (where i-k exists)
        for i in range(len(h)):
            # Partial derivative d/dh_i of the term h_i * (1 - h_{i+k}) 
            # This long one vanishes if there's no k-shift
            # contribution from region i (i+max_k corresponds to 
            # but testing again on actual script...
            if i + max_k < len(h):
                sub_grad[i] += (1 - h[i + max_k])  # from left  
            if i - max_k >= 0:
                sub_grad[i] -= h[i - max_k]  # from right
        # check if the result is all zeros
        if np.linalg.norm(sub_grad) < 1e-15:
            break
        
        # Take the step with adaptive learning rate
        alpha = alpha0 / np.sqrt(t + 1)
        h_new = h - alpha * sub_grad
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum(h_new, target_sum)
        
        # Stop or evaluate
        h = h_new
    return h.tolist() 

def exchange_refinement(h, num_passes=3):
    """Refine using local exchange moves"""
    h = np.array(h)
    n = len(h)
    target_sum = n / 2
    total_rounds = 0
    for _ in range(num_passes):
        # Shuffle the pairs
        pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        np.random.shuffle(pairs) 
        for i, j in pairs:
            # Compute delta_max
            delta_max = min(h[j], 1 - h[i])
            if delta_max <= 1e-8:
                continue
            # Try small increments
            for eta in [0.1, 0.5, 0.9]:
                delta = eta * delta_max
                # Compute new h vector
                h_new = h.copy()
                h_new[i] += delta
                h_new[j] -= delta
                if np.all(h_new >= 0) and np.all(h_new <= 1):
                    h_new = project_to_sum(h_new, target_sum)
                    if np.abs(np.sum(h_new) - target_sum) > 1e-8:
                        continue
                    # Approximate C5 using FFT
                    corr_new = compute_corr_fft(h_new)
                    new_c5 = c5_from_corr(corr_new, n)
                    corr_old = compute_corr_fft(h)
                    old_c5 = c5_from_corr(corr_old, n)
                    if new_c5 < old_c5: 
                        h = h_new.copy()
                        break
    return h.tolist()
