import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def improve_with_neighbor_search(h0, max_iter=2000):
    """Improve solution using local neighbor search"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Try changing a few random indices
        idxs = np.random.choice(n_points, size=max(1, n_points // 50), replace=False)
        improved = False
        
        for idx in idxs:
            old_val = h[idx]
            # Try a few small changes
            for delta in [-0.03, -0.02, -0.01, 0.01, 0.02, 0.03]:
                h_new = h.copy()
                h_new[idx] += delta
                
                # Ensure it stays in [0,1]
                if h_new[idx] < 0 or h_new[idx] > 1:
                    continue
                    
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                new_c5 = evaluate_c5(h_new)
                
                if new_c5 < best_c5:
                    h = h_new.copy()
                    best_c5 = new_c5
                    improved = True
                    break
            if improved:
                break
        
        # Occasionally make a larger random perturbation
        if np.random.random() < 0.02:
            perturb_idx = np.random.choice(n_points)
            h[perturb_idx] = np.random.uniform(0, 1)
            h = project_to_sum_box(h, TARGET_SUM)
            best_c5 = evaluate_c5(h)
    
    return h

def dynamic_sampling_initialization():
    """Create an initial solution with dynamic sampling approach"""
    # Start with a smooth template that's easy to optimize
    x = np.linspace(0, 2, n_points)
    # Create a simple smooth function that approximates a good solution
    h_base = 0.5 * (1 + np.sin(np.pi * x - np.pi/2))  # Sine-like shape
    
    # Add noise to break symmetry
    noise = np.random.normal(0, 0.05, n_points)
    h_noisy = h_base + noise
    
    # Clip and normalize
    h_final = np.clip(h_noisy, 0, 1)
    h_final = project_to_sum_box(h_final, TARGET_SUM)
    
    return h_final

def refine_peaks(h):
    """Refine any potential peaks in the function"""
    # Find local minima which might be good places for peaks
    h_smooth = np.convolve(h, np.ones(5)/5, mode='same')  # Smooth a bit
    peaks = []
    
    for i in range(1, len(h_smooth)-1):
        if h_smooth[i] < h_smooth[i-1] and h_smooth[i] < h_smooth[i+1]:
            peaks.append(i)
    
    # If we found peaks, try to enhance them
    if len(peaks) > 0:
        num_peaks = min(3, len(peaks))
        selected = np.random.choice(peaks, size=num_peaks, replace=False)
        
        for peak_idx in selected:
            h[peak_idx] = 1.0
            # Reduce nearby values to make it a clear spike
            for offset in range(-2, 3):
                idx = peak_idx + offset
                if 0 <= idx < n_points and idx != peak_idx:
                    h[idx] = max(0, h[idx] - 0.2)
    
    # Ensure constraints still hold
    return project_to_sum_box(h, TARGET_SUM)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = dynamic_sampling_initialization()
    best_c5 = evaluate_c5(best_h)
    
    # Iteratively improve using different techniques
    remaining_time = budget_s
    iteration_count = 0
    
    while remaining_time > 10:
        iteration_count += 1
        time_left = budget_s - (time.time() - start_time)
        
        # Small optimizations each step 
        h_candidate = improve_with_neighbor_search(best_h.copy(), max_iter=50)
        c5_candidate = evaluate_c5(h_candidate)
        
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
        
        # Occasionally add sharp peaks
        if iteration_count % 3 == 0:
            h_peaks = refine_peaks(best_h.copy())
            c5_peaks = evaluate_c5(h_peaks)
            if c5_peaks < best_c5:
                best_h = h_peaks.copy()
                best_c5 = c5_peaks
        
        # Occasionally try a fresh start to escape local minima
        if iteration_count % 10 == 0:
            h_fresh = dynamic_sampling_initialization()
            h_fresh = improve_with_neighbor_search(h_fresh, max_iter=100)
            c5_fresh = evaluate_c5(h_fresh)
            if c5_fresh < best_c5:
                best_h = h_fresh.copy()
                best_c5 = c5_fresh
                
        remaining_time = budget_s - (time.time() - start_time)
    
    return (best_h.tolist(), float(best_c5), n_points)
