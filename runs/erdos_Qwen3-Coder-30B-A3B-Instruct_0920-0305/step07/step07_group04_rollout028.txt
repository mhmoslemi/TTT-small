def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr) * (2.0 / len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_relaxed_lp(h0, n_points, T_guess):
    """Solve the convex relaxation LP with fixed T"""
    dx = 2.0 / n_points
    h = cp.Variable(n_points, name="h")
    p = cp.Variable((n_points, n_points), name="p")  # upper triangular part
    T = cp.Parameter(name="T")
    T.value = T_guess
    
    # mass constraint
    constraints = [cp.sum(h) == n_points / 2]
    
    # McCormick envelopes
    for i in range(n_points):
        for j in range(i, n_points):  # only upper triangular part
            constraints += [
                p[i, j] >= 0,
                p[i, j] >= h[i] + h[j] - 1,
                p[i, j] <= h[i],
                p[i, j] <= h[j]
            ]
    
    # Shift constraints
    overlaps = []
    for k in range(-n_points + 1, n_points):
        start_idx = max(0, k)
        end_idx = min(n_points, n_points + k)
        if k >= 0:
            o = cp.sum([p[i, i+k] for i in range(start_idx, end_idx)])
        else:
            o = cp.sum([p[i-k, i] for i in range(start_idx, end_idx)])
        overlaps.append(o)
    
    # relaxation of the hard constraint
    for k in range(len(overlaps)):
        constraints.append(n_points / 2 - overlaps[k] >= T)
    
    # objective is to minimize T (which is fixed)
    prob = cp.Problem(cp.Minimize(T), constraints)
    
    try:
        prob.solve(solver=cp.GLPK, verbose=False, eps=1e-8)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value
        else:
            return None
    except:
        return None

def binary_search_on_c5(h0, n_points, max_iter=200):
    """Binary search for the smallest valid upper bound"""
    T_low, T_high = 0.0, n_points / 2
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    tol = 1e-4 * (2.0 / n_points)
    
    for _ in range(max_iter):
        if T_high - T_low < tol:
            break
        T_mid = (T_low + T_high) / 2
        h_sol = solve_relaxed_lp(h0, n_points, T_mid)
        if h_sol is None:
            T_low = T_mid
            continue
            
        # Verifying if this feasible LP solution is acceptable
        c5_eval = evaluate_c5(h_sol)
        if c5_eval <= T_mid * (2.0 / n_points):
            best_h = h_sol
            best_c5 = c5_eval
            T_high = T_mid
        else:
            T_low = T_mid
            
    return best_h, T_high * (2.0 / n_points)

def subgradient_polish(h0, max_iter=2000, alpha0=0.01):
    """Deterministic subgradient polishing"""
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    
    for t in range(max_iter):
        # Compute the full correlation (FFT-based)
        h_rev = h[::-1]
        b = 1 - h
        L = 2 * n - 1
        corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
        k_star = np.argmax(corr)
        
        # compute subgradient
        g = np.zeros(n)
        if k_star <= n-1:
            for i in range(n):
                g[i] = (1 - h[i + k_star]) if i + k_star < n else 0
                if i - k_star >= 0:
                    g[i] -= h[i - k_star]
        
        # Update
        alpha_t = alpha0 / np.sqrt(t + 1)
        h = h - alpha_t * g
        h = project_to_sum_box(h, n / 2)
        
        # Early stopping if no improvement over recent iterations
        if t > 100 and (t % 100 == 0):
            if evaluate_c5(h) > evaluate_c5(h0):
                break
                
    return h

def exchange_refinement(h0, max_iter=5000):

    h = h0.copy()

        indices = list(range(n))

                break

            delta_max = min(h[b], 1 - h[a])

                h[a] += delta
                h[b] -= delta
                h = project_to_sum_box(h, n / 2)
    
    return h
