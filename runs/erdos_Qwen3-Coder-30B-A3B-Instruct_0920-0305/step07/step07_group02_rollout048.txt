import time

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def perturb_with_local_search(h0, num_attempts=20):
    """Improve solution by perturbing and local optimization"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for _ in range(num_attempts):
        # Add a small random perturbation
        h_perturbed = h + np.random.normal(0, 0.01, n_points) 
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
        
        # Local optimization around this point
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            def objective(h_local):
                return evaluate_c5(h_local)
            
            def constraint_eq(h_local):
                return np.sum(h_local) - TARGET_SUM
            
            cons = {'type': 'eq', 'fun': constraint_eq}
            
            res = minimize(
                objective, 
                h_perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-10}
            )
