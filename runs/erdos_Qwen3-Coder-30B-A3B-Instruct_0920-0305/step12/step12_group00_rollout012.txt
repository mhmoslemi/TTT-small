import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
import math

def balanced_sturmian(n):
    """Generate a balanced Sturmian binary sequence."""
    phi = (math.sqrt(5) - 1) / 2  # golden ratio conjugate
    x = np.array([int(((i * phi) % 1) < 0.5) for i in range(n)])
    # Ensure exactly n//2 ones
    ones_count = np.sum(x)
    if ones_count > n // 2:
        # Flip some 1s to 0s
        indices = np.where(x == 1)[0]
        flip_count = ones_count - n // 2
        flip_indices = np.random.choice(indices, flip_count, replace=False)
        x[flip_indices] = 0
    elif ones_count < n // 2:
        # Flip some 0s to 1s
        indices = np.where(x == 0)[0]
        flip_count = n // 2 - ones_count
        flip_indices = np.random.choice(indices, flip_count, replace=False)
        x[flip_indices] = 1
    return x

def compute_autocorr_fft(h):
    """Compute autocorrelations efficiently using FFT."""
    h_rev = h[::-1]
    b = h  # h * h is the autocorrelation we're computing for like terms, 
    # but we actually compute s_l = sum_i h_i * h_{i+l}
    # we compute h * h (convolution with reversed h) and take all terms 
    # with lags from -n+1 to n-1
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr[len(h)-1:]  # take the middle part with lags 0,1,...,n-1

def get_best_swap(x, S, ones, zeros):
    """Find the best swap to improve the minimum autocorrelation."""
    n = len(x)
    best_delta = 0
    best_i, best_j = -1, -1
    best_S_prime = None
    min_S = np.min(S[1:])  # exclude lag 0
    
    # Try several random swaps
    for _ in range(100):
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Compute changes in autocorrelations for the swap
        delta_S = np.zeros(n)
        for l in range(1, n):
            if i + l < n and j + l < n:
                delta_S[l] += x[i] - x[j]
            if i - l >= 0 and j - l >= 0:
                delta_S[l] += x[i] - x[j]
            if i + l < n:
                delta_S[l] += x[j] - x[i]
            if i - l >= 0:
                delta_S[l] += x[j] - x[i]
        
        S_prime = S + delta_S
        min_S_prime = np.min(S_prime[1:])
        if min_S_prime > min_S + best_delta:
            best_delta = min_S_prime - min_S
            best_i, best_j = i, j
            best_S_prime = S_prime
    
    return best_i, best_j, best_S_prime

def refine_solution(x, n_points, dx, target_sum):
    """Refine binary solution with local swaps."""
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    S = compute_autocorr_fft(x)
    min_S = np.min(S[1:])
    c5 = dx * (target_sum - min_S)
    best_c5 = c5
    best_x = x.copy()
    
    # Hill climbing
    total_swaps = 0
    max_swaps = 5 * n_points
    start_time = time.time()
    
    while total_swaps < max_swaps and time.time() - start_time < 150:
        i, j, S_new = get_best_swap(x, S, ones, zeros)
        if i != -1:
            # Perform swap
            x[i], x[j] = x[j], x[i]
            ones = np.where(x == 1)[0]
            zeros = np.where(x == 0)[0]
            S = S_new
            min_S = np.min(S[1:])
            c5 = dx * (target_sum - min_S)
            if c5 < best_c5:
                best_c5 = c5
                best_x = x.copy()
            total_swaps += 1
        else:
            # No improvement found, try more swaps
            total_swaps += 1
    
    return best_x, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 128  # Primary discretization
    dx = 2.0 / n_points
    target_sum = n_points // 2
    
    best_h = None
    best_c5 = float('inf')
    
    # -----------------------------------------
    # Stage 1: Exact Integer Linear Programming
    # -----------------------------------------
    try:
        # Set up the ILP model
        x = cp.Variable(n_points, boolean=True)
        y = {}
        S = {}
        
        # Create auxiliary variables and constraints for all lags
        for l in range(1, n_points):
            for i in range(n_points - l):
                y[(i, l)] = cp.Variable(boolean=True)
                # McCormick constraints
                cp.add_constraint(y[(i, l)] <= x[i])
                cp.add_constraint(y[(i, l)] <= x[i + l])
                cp.add_constraint(y[(i, l)] >= x[i] + x[i + l] - 1)
        
        # Define the autocorrelations
        for l in range(1, n_points):
            # Compute sum of y[i,l] for i from 0 to n-l-1
            S[l] = cp.sum([y[(i, l)] for i in range(n_points - l)])
        
        # Epigraph variable for minimum autocorrelation
        m = cp.Variable()
        
        # Add the mass constraint
        constraints = [cp.sum(x) == target_sum]
        
        # Add the epigraph constraints
        for l in range(1, n_points):
            constraints.append(S[l] >= m)
        
        # Add symmetry-breaking constraints
        constraints.append(x[0] == 1)
        for i in range(1, n_points // 2):
            constraints.append(x[i] >= x[i + 1])  # decreasing
        
        # Objective: maximize m
        objective = cp.Maximize(m)
        prob = cp.Problem(objective, constraints)
        
        # Solve with timeout
        prob.solve(solver=cp.CBC, max_seconds=0.45 * budget_s)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_EOF]:
            # Extract the solution
            x_val = np.round(x.value).astype(int)
            # Ensure correct sum
            current_sum = np.sum(x_val)
            if current_sum > target_sum:
                # Flip some 1s to 0s
                one_indices = np.where(x_val == 1)[0]
                flip_indices = np.random.choice(one_indices, current_sum - target_sum, replace=False)
                x_val[flip_indices] = 0
            elif current_sum < target_sum:
                # Flip some 0s to 1s
                zero_indices = np.where(x_val == 0)[0]
                flip_indices = np.random.choice(zero_indices, target_sum - current_sum, replace=False)
                x_val[flip_indices] = 1
            
            # Compute c5 from the solution
            corr = np.correlate(x_val, x_val, mode='full')
            c5 = dx * np.max(corr)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = x_val
            
    except Exception as e:
        pass  # Fall back to heuristic
    
    # -----------------------------------------
    # Stage 2: Refinement using local search
    # -----------------------------------------
    if best_h is None:
        # Fallback to Sturmian solution
        best_h = balanced_sturmian(n_points)
    
    # Refine with local swaps
    refined_h, refined_c5 = refine_solution(best_h, n_points, dx, target_sum)
    
    if refined_c5 < best_c5:
        best_c5 = refined_c5
        best_h = refined_h
    
    # -----------------------------------------
    # Stage 3: Potential improvement via random restarts
    # -----------------------------------------
    for i in range(5):
        if time.time() - start_time > 0.95 * budget_s:
            break
        # Random restart
        h = np.random.rand(n_points)
        h = np.round(h).astype(int)
        # Ensure correct sum
        current_sum = np.sum(h)
        if current_sum > target_sum:
            indices = np.where(h == 1)[0]
            flip_indices = np.random.choice(indices, current_sum - target_sum, replace=False)
            h[flip_indices] = 0
        elif current_sum < target_sum:
            indices = np.where(h == 0)[0]
            flip_indices = np.random.choice(indices, target_sum - current_sum, replace=False)
            h[flip_indices] = 1
        
        refined_h, refined_c5 = refine_solution(h, n_points, dx, target_sum)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h
    
    if best_h is None:
        # Fallback solution
        best_h = balanced_sturmian(n_points)
        corr = np.correlate(best_h, best_h, mode='full')
        best_c5 = dx * np.max(corr)
    
    return (best_h.tolist(), float(best_c5), n_points)
