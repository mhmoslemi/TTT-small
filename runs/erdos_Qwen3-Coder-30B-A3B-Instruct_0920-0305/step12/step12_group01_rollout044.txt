def evaluate_c5(h):
    """Fast evaluation of C5 using FFT correlation"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr) * (2.0 / len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_binary_pattern(n):
    """Create a binary pattern that's likely to be good"""
    # Try making a pattern with alternating high/low regions
    pattern = np.zeros(n)
    # Place high values in alternating regions
    for i in range(0, n, 4):
        if i < n:
            pattern[i] = 1.0
        if i + 2 < n:
            pattern[i + 2] = 1.0
    # Normalize to sum to n/2
    pattern = pattern * (n/2) / np.sum(pattern)
    return pattern
