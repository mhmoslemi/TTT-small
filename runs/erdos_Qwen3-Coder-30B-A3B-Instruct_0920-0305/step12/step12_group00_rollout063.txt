import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
import math

def balanced_sturmian(n):
    """Generate a balanced Sturmian word as initial solution"""
    phi = (math.sqrt(5) - 1) / 2
    x = np.zeros(n, dtype=int)
    for i in range(n):
        if (i * phi) % 1 < 0.5:
            x[i] = 1
    # Adjust to make sum exactly n//2
    current_sum = np.sum(x)
    target_sum = n // 2
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        ones_indices = np.where(x == 1)[0]
        flip_count = current_sum - target_sum
        flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
        x[flip_indices] = 0
    elif current_sum < target_sum:
        # Need to flip some 0s to 1s
        zeros_indices = np.where(x == 0)[0]
        flip_count = target_sum - current_sum
        flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
        x[flip_indices] = 1
    return x

def compute_autocorr_fft(x):
    """Compute autocorrelations via FFT for all lags"""
    n = len(x)
    x_rev = x[::-1]
    # Cross-correlation of x with itself
    corr = correlate(x, x, mode='full')
    # Take only non-negative lags (1 to n-1)
    autocorr = corr[n-1:]
    return autocorr

def swap_delta(autocorr, i, j, ones, zeros, n):
    """Compute the change in autocorrelation when swapping bits at positions i and j"""
    # Create a copy of autocorrelations to compute the delta
    delta = np.zeros_like(autocorr)
    for l in range(1, n):  # lags from 1 to n-1
        # Add contribution of the swap
        if i + l < n:
            delta[l-1] += (x[j+l] - x[i+l])  # x[i] becomes 0, x[j] becomes 1
        if i - l >= 0:
            delta[l-1] += (x[j-l] - x[i-l])  # x[i] becomes 0, x[j] becomes 1
        if j + l < n:
            delta[l-1] += (x[i+l] - x[j+l])  # x[j] becomes 0, x[i] becomes 1
        if j - l >= 0:
            delta[l-1] += (x[i-l] - x[j-l])  # x[j] becomes 0, x[i] becomes 1
    return delta

def hill_climb_swap(x, max_time):
    """Perform hill climbing by swapping pairs of bits"""
    start_time = time.time()
    n = len(x)
    zeros = set(np.where(x == 0)[0])
    ones = set(np.where(x == 1)[0])
    
    # Precompute autocorrelations
    autocorr = compute_autocorr_fft(x)
    min_autocorr = np.min(autocorr[1:])  # Exclude lag 0
    
    iterations = 0
    max_iter = 1000000  # hard limit
    patience = 0
    patience_limit = 5 * n
    
    # For simulated annealing
    temp = 0.01
    
    while time.time() - start_time < max_time and iterations < max_iter:
        iterations += 1
        
        # Try a random swap
        if ones and zeros:
            i = np.random.choice(list(ones))
            j = np.random.choice(list(zeros))
            
            # Compute the delta in autocorrelations
            delta = swap_delta(autocorr, i, j, ones, zeros, n)
            new_autocorr = autocorr + delta
            new_min_autocorr = np.min(new_autocorr[1:])
            
            # Accept the swap with probability based on improvement
            if new_min_autocorr > min_autocorr:
                # Accept the swap
                x[i], x[j] = x[j], x[i]
                zeros.remove(i)
                ones.remove(j)
                ones.add(i)
                zeros.add(j)
                autocorr += delta
                min_autocorr = new_min_autocorr
                patience = 0
            elif np.random.rand() < math.exp((new_min_autocorr - min_autocorr) / temp):
                # Accept with low probability (simulated annealing)
                x[i], x[j] = x[j], x[i]
                zeros.remove(i)
                ones.remove(j)
                ones.add(i)
                zeros.add(j)
                autocorr += delta
                min_autocorr = new_min_autocorr
                patience = 0
            else:
                patience += 1
                if patience > patience_limit:
                    # Reset temperature and patience
                    patience = 0
                    temp *= 0.99
            
        if patience > patience_limit:
            # Reset if no progress
            patience = 0
            temp = max(temp * 0.99, 0.0001)
    
    return x, min_autocorr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Parameters
    n = 128
    dx = 2.0 / n
    N = n // 2
    best_c5 = float('inf')
    best_h = None
    best_m = 0
    
    # -------------------------------------------------
    # Stage 1 – exact ILP (max-min autocorrelation)
    # -------------------------------------------------
    print("Solving ILP...")
    # Create variables
    x = cp.Variable(n, integer=True)
    y = {}
    constraints = []
    
    # Constraints for each lag
    for l in range(1, n):
        for i in range(n - l):
            y[i, l] = cp.Variable(boolean=True)
            
    # Add McCormick constraints
    for l in range(1, n):
        for i in range(n - l):
            # y[i,l] <= x[i]
            constraints.append(y[i, l] <= x[i])
            # y[i,l] <= x[i+l]
            constraints.append(y[i, l] <= x[i + l])
            # y[i,l] >= x[i] + x[i+l] - 1
            constraints.append(y[i, l] >= x[i] + x[i + l] - 1)
    
    # Mass constraint
    constraints.append(cp.sum(x) == N)
    
    # Symmetry breaking constraints
    # Fix first element to 1
    constraints.append(x[0] == 1)
    # Lexicographic ordering: ensure first N elements are non-increasing
    for i in range(N - 1):
        constraints.append(x[i] >= x[i + 1])
        
    # Compute autocorrelation terms
    S = []
    
    # Use the constraint that S_ell >= m instead of actually computing S
    # We'll set a variable m and maximize it subject to constraints
    
    m = cp.Variable()
    
    # Each S_ell >= m (for ell from 1 to n-1)
    for l in range(1, n):
        sum_term = 0
        for i in range(n - l):
            sum_term += y[i, l]
        constraints.append(sum_term >= m)
    
    # Objective
    objective = cp.Maximize(m)
    
    # Solve the ILP
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CBC, max_seconds=0.45 * budget_s)
    
    # Check if a solution was found
    if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_USING_CUTS]:
        x_sol = np.round(x.value).astype(int)
        print(f"ILP solution found. Status: {prob.status}")
        # Compute the actual minimum autocorrelation
        autocorr_vals = compute_autocorr_fft(x_sol)
        min_autocorr = np.min(autocorr_vals[1:])
        actual_c5 = dx * (N - min_autocorr)
        best_c5 = actual_c5
        best_h = x_sol
        best_m = min_autocorr
        print(f"ILP solution achieved C5 = {actual_c5}")
    else:
        print(f"ILP did not converge within time limit. Status: {prob.status}")
        # Fall back to heuristic search on Sturmian word
        x_sol = balanced_sturmian(n)
        best_h = x_sol.copy()
    
    # -------------------------------------------------
    # Stage 2 – greedy swap hill-climb (binary refinement)
    # -------------------------------------------------
    print("Refining solution with hill-climbing...")
    remaining_time = 0.35 * budget_s
    refined_h, refined_m = hill_climb_swap(best_h.copy(), remaining_time)
    
    # Compute the final C5
    final_autocorr = compute_autocorr_fft(refined_h)
    final_min_autocorr = np.min(final_autocorr[1:])
    final_c5 = dx * (N - final_min_autocorr)
    
    print(f"After hill-climb: C5 = {final_c5}")
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = refined_h.copy()
        best_m = final_min_autocorr

    # -------------------------------------------------
    # Stage 3 – Random restarts
    # -------------------------------------------------
    print("Performing random restart search...")
    max_restarts = 10
    restart_count = 0
    restart_time_limit = 0.1 * budget_s
    
    while restart_count < max_restarts and time.time() - t0 < budget_s * 0.95:
        restart_count += 1
        x_restart = np.random.randint(0, 2, n)
        # Ensure sum is correct
        current_sum = np.sum(x_restart)
        if current_sum > N:
            ones_indices = np.where(x_restart == 1)[0]
            flip_count = current_sum - N
            flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
            x_restart[flip_indices] = 0
        elif current_sum < N:
            zeros_indices = np.where(x_restart == 0)[0]
            flip_count = N - current_sum
            flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
            x_restart[flip_indices] = 1
            
        x_refined, _ = hill_climb_swap(x_restart, restart_time_limit / max_restarts)
        
        # Compute the C5
        autocorr_vals = compute_autocorr_fft(x_refined)
        min_autocorr = np.min(autocorr_vals[1:])
        c5 = dx * (N - min_autocorr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = x_refined.copy()
            best_m = min_autocorr
        if time.time() - t0 > budget_s * 0.95:
            break

    # Convert to list for return
    return (best_h.tolist(), float(best_c5), n)
