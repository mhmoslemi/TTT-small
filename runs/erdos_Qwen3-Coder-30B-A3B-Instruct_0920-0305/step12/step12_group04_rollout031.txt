# Global variables
TARGET_SUM = None
n_points = 200

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = correlate(h, 1 - h, mode="full")
    O = corr * dx
    return np.max(O)

def compute_gradient_for_lag(h, lag):
    """Compute gradient of O(k) for a specific lag k"""
    N = len(h)
    forward = np.zeros(N)
    backward = np.zeros(N)
    if lag >= 0:
        forward[:N - lag] = h[lag:]
        backward[lag:] = h[:N - lag]
    else:
        k = -lag
        forward[k:] = h[:N - k]
        backward[:N - k] = h[k:]
    grad = 1.0 - forward - backward
    return grad

        current_val = evaluate_c5(h)

            h = h_new
        else:
            # Accept with small probability to escape local minima
            if np.random.random() < 0.001:
                h = h_new
                
        iteration += 1
    
    return best_h, best_val
