import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort the vector
    sorted_x = np.sort(x)[::-1]
    
    # Find lambda using binary search
    a, b = -1.0, 1.0
    while True:
        m = (a + b) / 2
        projected = np.clip(sorted_x + m, 0, 1)
        current_sum = np.sum(projected)
        if abs(current_sum - target_sum) < 1e-10:
            break
        elif current_sum > target_sum:
            a = m
        else:
            b = m
    
    result = np.clip(x + m, 0, 1)
    return result

def softmax_weights(corr, alpha):
    """Compute softmax weights"""
    exp_corr = np.exp(alpha * corr)
    return exp_corr / np.sum(exp_corr)

def grad_softmax(corr, alpha, h):
    """Compute gradient of softmax with respect to h"""
    weights = softmax_weights(corr, alpha)
    
    # Convolution with h
    L = len(h) * 2 - 1
    L_fft = 1 << (L - 1).bit_length()
    
    W = np.fft.fft(weights, L_fft)
    H = np.fft.fft(h, L_fft)
    conv1 = np.fft.ifft(W * H).real[:len(h)]
    
    W_rev = np.fft.fft(weights[::-1], L_fft)
    conv2 = np.fft.ifft(W_rev * H).real[:len(h)]
    
    return 1.0 - conv1 - conv2

def adam_step(h, m, v, grad, t, alpha, beta1=0.9, beta2=0.999, epsilon=1e-8):
    """Perform Adam step with projection"""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    
    h_new = h - alpha * m_hat / (np.sqrt(v_hat) + epsilon)
    
    return h_new, m, v

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points // 2
    best_c5 = float('inf')
    best_h = None
    
    # Sturmian word construction
    phi = (np.sqrt(5) - 1) / 2
    sturmian_h = np.array([1 if (i * phi) % 1 < 0.5 else 0 for i in range(n_points)])
    sturmian_h = project_to_sum_box(sturmian_h.astype(float), TARGET_SUM)
    
    # Random initialization
    random_h = np.random.rand(n_points)
    random_h = project_to_sum_box(random_h, TARGET_SUM)
    
    # Perturbed Sturmian
    perturbed_h = sturmian_h + np.random.normal(0, 0.05, n_points)
    perturbed_h = project_to_sum_box(perturbed_h, TARGET_SUM)
    
    # Run multiple restarts
    initial_points = [sturmian_h, random_h, perturbed_h]
    
    # Adam optimizer parameters
    alpha_init = 4.0
    alpha_growth = 1.5
    max_alpha = 200.0
    learning_rate = 0.12
    max_iters = 8000
    patience = 1000
    
    for i, h0 in enumerate(initial_points):
        if time.time() - start_time > 0.45 * budget_s:
            break
            
        h = h0.copy()
        m = np.zeros(n_points)
        v = np.zeros(n_points)
        alpha = alpha_init
        t = 1
        
        prev_c5 = float('inf')
        patience_count = 0
        
        while t < max_iters and time.time() - start_time < 0.45 * budget_s:
            # Compute correlation
            corr = compute_corr_fft(h)
            current_c5 = c5_from_corr(corr, n_points)
            
            # Compute gradient
            grad = grad_softmax(corr, alpha, h)
            
            # Adam step
            h, m, v = adam_step(h, m, v, grad, t, learning_rate)
            
            # Project to feasible set
            h = project_to_sum_box(h, TARGET_SUM)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            # Update alpha (schedule)
            if t > 0 and t % 200 == 0:
                alpha = min(alpha * alpha_growth, max_alpha)
                
            # Check for convergence
            if abs(current_c5 - prev_c5) < 1e-8:
                patience_count += 1
            else:
                patience_count = 0
            prev_c5 = current_c5
            
            if patience_count > patience:
                break
                
            t += 1
    
    # Binary rounding and hill-climb
    if best_h is not None:
        # Round to binary
        binary_h = (best_h >= 0.5).astype(float)
        
        # Adjust sum
        ones_count = np.sum(binary_h)
        if ones_count > TARGET_SUM:
            ones_indices = np.where(binary_h == 1)[0]
            flip_count = int(ones_count - TARGET_SUM)
            flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
            binary_h[flip_indices] = 0
        elif ones_count < TARGET_SUM:
            zeros_indices = np.where(binary_h == 0)[0]
            flip_count = int(TARGET_SUM - ones_count)
            flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
            binary_h[flip_indices] = 1
        
        binary_h = project_to_sum_box(binary_h, TARGET_SUM)
        best_binary_c5 = c5_from_corr(compute_corr_fft(binary_h), n_points)
        
        # Save the binary solution
        if best_binary_c5 < best_c5:
            best_c5 = best_binary_c5
            best_h = binary_h.copy()
            
        # Swap hill-climb
        current_binary_h = binary_h.copy()
        current_c5 = best_binary_c5
        max_swaps = 5 * n_points
        swap_count = 0
        t_end = start_time + 0.35 * budget_s
        t_end_swap = time.time() + (0.35 * budget_s / 3)  # Inside time limit
        
        while time.time() < t_end_swap and swap_count < max_swaps:
            i = np.random.randint(n_points)
            j = np.random.randint(n_points)
            
            if current_binary_h[i] == current_binary_h[j]:
                continue
                
            # Store current lags
            current_corr = compute_corr_fft(current_binary_h)
            current_min = np.min(current_corr)
            
            # Simulate the swap
            temp_h = current_binary_h.copy()
            temp_h[i], temp_h[j] = temp_h[j], temp_h[i]
            
            new_corr = compute_corr_fft(temp_h)
            new_min = np.min(new_corr)
            
            # Accept if it improves
            if new_min > current_min:
                current_binary_h = temp_h.copy()
                current_c5 = c5_from_corr(new_corr, n_points)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_binary_h.copy()
            
            swap_count += 1
        
        # Additional random binary restarts
        t_restart = time.time() + (0.35 * budget_s / 3)
        for _ in range(10):
            if time.time() > t_restart:
                break
            
            random_bin = np.random.rand(n_points)
            random_bin = project_to_sum_box(random_bin, TARGET_SUM)
            binary_h = (random_bin >= 0.5).astype(float)
            
            # Adjust sum
            ones_count = np.sum(binary_h)
            if ones_count > TARGET_SUM:
                ones_indices = np.where(binary_h == 1)[0]
                flip_count = int(ones_count - TARGET_SUM)
                flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
                binary_h[flip_indices] = 0
            elif ones_count < TARGET_SUM:
                zeros_indices = np.where(binary_h == 0)[0]
                flip_count = int(TARGET_SUM - ones_count)
                flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
                binary_h[flip_indices] = 1
            
            binary_h = project_to_sum_box(binary_h, TARGET_SUM)
            trial_c5 = c5_from_corr(compute_corr_fft(binary_h), n_points)
            if trial_c5 < best_c5:
                best_c5 = trial_c5
                best_h = binary_h.copy()
    
    # If no solution found, return the initial one
    if best_h is None:
        best_h = sturmian_h.copy()
        best_c5 = c5_from_corr(compute_corr_fft(best_h), n_points)
        
    return (best_h.tolist(), float(best_c5), n_points)
