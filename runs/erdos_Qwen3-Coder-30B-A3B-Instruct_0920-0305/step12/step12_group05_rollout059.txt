from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_search(h, current_c5, start_time, budget_s):
    """Local search with gradient-based refinement"""
    if time.time() - start_time > budget_s - 10:
        return h, current_c5
    
    # Simple gradient-based local search
    for _ in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
        
        # Estimate gradient by finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_sum_box(h_plus, TARGET_SUM)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_sum_box(h_minus, TARGET_SUM)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update with gradient descent
        step_size = 0.01
        h_new = h - step_size * grad
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        c5_new = evaluate_c5(h_new)
        
        if c5_new < current_c5:
            h = h_new
            current_c5 = c5_new
            
    return h, current_c5

def simulated_annealing(h, current_c5, start_time, budget_s):
    """Simulated Annealing approach"""
    if time.time() - start_time > budget_s - 10:
        return h, current_c5
    
    temp = 1.0
    min_temp = 1e-8
    cooling_rate = 0.999
    max_iter = 5000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Create neighbor solution
        h_new = h.copy()
        idx = np.random.randint(len(h))
        delta = np.random.normal(0, 0.05)
        h_new[idx] += delta
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject the new solution
        if c5_new < current_c5 or np.random.random() < np.exp(-(c5_new - current_c5) / temp):
            h = h_new
            current_c5 = c5_new
            
        # Cool down
        temp *= cooling_rate
        if temp < min_temp:
            temp = min_temp
            
    return h, current_c5

            idx = np.random.randint(len(h))

    return h, current_c5
