import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_softmax(h, beta):
    """Compute gradient of smoothed max correlation"""
    c = compute_corr_fft(h)
    c_max = np.max(c)
    # Numerically stable softmax
    w_raw = np.exp(beta * (c - c_max))
    p = w_raw / np.sum(w_raw)
    
    # reversed weights
    w_rev = p[::-1]
    
    # Use np.convolve for efficient computation of gradient terms
    conv2 = np.convolve(w_rev, 1 - h, mode='full')
    conv3 = np.convolve(w_rev, h, mode='full')
    
    # Slice to get the correct indices
    start_idx = len(h) - 1
    end_idx = 2 * len(h) - 1
    g = dx * (conv2[start_idx:end_idx] - conv3[start_idx:end_idx])
    
    return g, c

    best_c5 = evaluate_c5(h0)
    
    # Multiple restarts with different starting points
    num_restarts = 5
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Random initialization with some structure
        if restart == 0:
            h = h0.copy()
        else:
            # Start with a slightly perturbed version of the best found so far
            h = project_to_sum_box(best_h + 0.1 * np.random.randn(n_points), TARGET_SUM)
        
        # Gradient descent with momentum
        beta = 1.0
        eta = 0.1
        momentum = 0.9
        v = np.zeros(n_points)
        
        # Store best for this restart
        restart_best_h = h.copy()
        restart_best_c5 = evaluate_c5(h)
        
        iterations = 0
        max_iterations = 5000
        
        while iterations < max_iterations and time.time() - start_time < budget_s - 10:
            # Compute gradient
            g, c = gradient_softmax(h, beta)
            
            # Update with momentum
            v = momentum * v - eta * g
            h_new = h + v
            
            # Project to feasible region
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            # Line search
            c5_new = evaluate_c5(h_new)
            if c5_new < restart_best_c5:
                h = h_new
                restart_best_c5 = c5_new
                restart_best_h = h.copy()
                
                # Update global best
                if restart_best_c5 < best_c5:
                    best_c5 = restart_best_c5
                    best_h = restart_best_h.copy()
            
            # Update parameters
            iterations += 1
            beta = min(beta * 1.01, 100)

            # Occasionally random perturbation
            if iterations % 500 == 0 and time.time() - start_time < budget_s - 30:
                h = project_to_sum_box(h + 0.02 * np.random.randn(n_points), TARGET_SUM)
        
        # Final refinement with local optimization
        if time.time() - start_time < budget_s - 30:
            try:
                def objective(h_vec):
                    return evaluate_c5(h_vec)
                
                def constraint_eq(h_vec):
                    return np.sum(h_vec) - TARGET_SUM
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons = {'type': 'eq', 'fun': constraint_eq}
                
                res = minimize(

                    best_h = h_final.copy()
            except:
                pass
    
    return (best_h.tolist(), float(best_c5), n_points)
