import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

# Global constants
n_points = 200
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def build_h_from_params(params, num_segments):
    """Build a step function from parametrization"""
    # First num_segments are raw lengths (unconstrained)
    raw_lengths = params[:num_segments]
    # Next num_segments are raw heights (unconstrained)
    raw_heights = params[num_segments:2*num_segments]
    
    # Compute lengths (positive)
    lengths = np.exp(raw_lengths)
    lengths = lengths / lengths.sum() * 2.0
    
    # Compute heights (in [0,1])
    heights = 1.0 / (1.0 + np.exp(-raw_heights))
    
    # Create h array
    x = (np.arange(n_points) + 0.5) * dx
    cum_lengths = np.cumsum(lengths)
    seg_indices = np.searchsorted(cum_lengths, x, side='right')
    h = heights[seg_indices]
    
    return h
