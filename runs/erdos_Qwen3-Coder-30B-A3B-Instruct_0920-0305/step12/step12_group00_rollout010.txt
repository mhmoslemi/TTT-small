import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort the vector to get the water-filling projection
    sorted_x = np.sort(x)[::-1]
    
    # Find lambda using bisection
    def f(lam):
        projected = np.clip(sorted_x - lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    lambda_val = (a + b) / 2
    result = np.clip(x - lambda_val, 0, 1)
    return result

def softmax_weights(corr, alpha):
    """Compute softmax weights for given correlation and alpha"""
    exp_corr = np.exp(alpha * corr)
    return exp_corr / np.sum(exp_corr)

def grad_softmax(corr, alpha, h):
    """Compute gradient of softmax with respect to h"""
    w = softmax_weights(corr, alpha)
    # Compute forward convolution
    conv1 = np.convolve(w, h, mode='full')[:len(h)]
    # Compute backward convolution  
    conv2 = np.convolve(w[::-1], h, mode='full')[:len(h)]
    # Ensure same length
    grad = 1.0 - conv1 - conv2
    return grad

def adam_step(h, grad, m, v, t, lr=0.12, beta1=0.9, beta2=0.999, epsilon=1e-8):
    """Perform Adam update step"""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
    return h_new, m, v

def compute_c5_exact(h):
    """Compute exact C5 using full correlation"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def create_balanced_sturmian():
    """Create balanced Sturmian word"""
    phi = (np.sqrt(5) - 1) / 2
    h = np.array([int(((i * phi) % 1) < 0.5) for i in range(n_points)])
    # Adjust to ensure sum = TARGET_SUM
    current_sum = np.sum(h)
    if current_sum > TARGET_SUM:
        ones_indices = np.where(h == 1)[0]
        flip_count = int(current_sum - TARGET_SUM)
        flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
        h[flip_indices] = 0
    elif current_sum < TARGET_SUM:
        zeros_indices = np.where(h == 0)[0]
        flip_count = int(TARGET_SUM - current_sum)
        flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
        h[flip_indices] = 1
    
    h = project_to_sum_box(h.astype(float), TARGET_SUM)
    return h

def create_random_vector():
    """Create random vector and project to feasible set"""
    h = np.random.rand(n_points)
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def create_perturbed_sturmian():
    """Create perturbed sturmian word"""
    h = create_balanced_sturmian()
    noise = np.random.normal(0, 0.05, n_points)
    h = h + noise
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def binary_refinement(h_fractional):
    """Convert fractional solution to binary and refine with swaps"""
    # Round to binary
    h_binary = (h_fractional >= 0.5).astype(float)
    
    # Adjust sum to be exactly TARGET_SUM
    current_sum = np.sum(h_binary)
    if current_sum > TARGET_SUM:
        ones_indices = np.where(h_binary == 1)[0]
        flip_count = int(current_sum - TARGET_SUM)
        flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
        h_binary[flip_indices] = 0
    elif current_sum < TARGET_SUM:
        zeros_indices = np.where(h_binary == 0)[0]
        flip_count = int(TARGET_SUM - current_sum)
        flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
        h_binary[flip_indices] = 1
        
    h_binary = project_to_sum_box(h_binary, TARGET_SUM)
    
    # Compute current overlaps
    corr = compute_corr_fft(h_binary)
    current_c5 = c5_from_corr(corr, n_points)
    
    # Keep track of best solution
    best_h = h_binary.copy()
    best_c5 = current_c5
    
    # Fast computation of overlaps for swap updates
    ones_indices = np.where(h_binary == 1)[0]
    zeros_indices = np.where(h_binary == 0)[0]
    
    # Precompute all correlations
    corr_all = np.zeros(2 * n_points - 1)
    for i in range(n_points):
        corr_all[i] += h_binary[i] * (1 - h_binary[0])
        corr_all[n_points - 1 + i] += h_binary[i] * (1 - h_binary[i])
    
    # Swap hill-climb
    max_swaps = 5 * n_points
    swap_count = 0
    misfits = []
    t1 = time.time() + 150  # 150 seconds for swaps
    
    while swap_count < max_swaps and time.time() < t1:
        # Select pair (i in ones, j in zeros)
        i = np.random.choice(ones_indices)
        j = np.random.choice(zeros_indices)
        
        # Compute change in overlaps
        delta_corr = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            # Indices where we might add points
            l = k - (n_points - 1)
            # Check i (now 0): contribution to corr[k] 
            if (i + l >= 0) and (i + l < n_points):
                delta_corr[k] += h_binary[i+l] - (1-h_binary[i+l])
            # Check j (now 1): contribution to corr[k] 
            if (j + l >= 0) and (j + l < n_points):
                delta_corr[k] += (1-h_binary[j+l]) - h_binary[j+l]
        
        # Compute new maximum correlation
        new_corr = corr + delta_corr
        new_c5 = dx * np.max(new_corr)
        
        if new_c5 < current_c5:
            h_binary[i], h_binary[j] = h_binary[j], h_binary[i]
            corr = new_corr
            current_c5 = new_c5
            
            # Update indices
            ones_indices = np.where(h_binary == 1)[0]
            zeros_indices = np.where(h_binary == 0)[0]
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_binary.copy()
        
        swap_count += 1
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Keep track of timing
    t_total = start_time + budget_s
    
    # Try several initializations
    initializations = [
        create_balanced_sturmian(),
        create_random_vector(),
        create_perturbed_sturmian()
    ]
    
    # Adam optimization phase
    max_iters = 8000
    alpha = 4.0
    alpha_growth = 1.5
    max_alpha = 200.0
    
    # Initialize Adam moments
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    # Run from each initialization
    for init_idx, h0 in enumerate(initializations):
        if time.time() > t_total:
            break
            
        h = h0.copy()
        current_best_c5 = compute_c5_exact(h)
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = h.copy()
            
        # Run Adam optimization
        iter_count = 0
        t1 = start_time + 0.45 * budget_s  # 45% for optimization
        
        while iter_count < max_iters and time.time() < t1:
            iter_count += 1
            
            # Compute correlation and gradient
            corr = compute_corr_fft(h)
            grad = grad_softmax(corr, alpha, h)
            
            # Take Adam step
            h_new, m, v = adam_step(h, grad, m, v, iter_count)
            
            # Project
            h = project_to_sum_box(h_new, TARGET_SUM)
            
            # Update alpha
            if iter_count % 200 == 0:
                alpha = min(alpha * alpha_growth, max_alpha)
            
            # Check for improved solution
            current_c5 = compute_c5_exact(h)
            if current_c5 < current_best_c5:
                current_best_c5 = current_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        
        # Early termination if budget is nearly done
        if time.time() > t_total - 20:
            break
    
    # Binary refinement
    if best_h is not None and time.time() < t_total - 30:
        binary_h, binary_c5 = binary_refinement(best_h)
        if binary_c5 < best_c5:
            best_c5 = binary_c5
            best_h = binary_h
    
    # Additional restart experiments
    if time.time() < t_total - 30:
        t2 = start_time + 0.85 * budget_s
        for _ in range(5):
            if time.time() > t2:
                break
            h_random = create_random_vector()
            binary_h, binary_c5 = binary_refinement(h_random)
            if binary_c5 < best_c5:
                best_c5 = binary_c5
                best_h = binary_h
    
    # Return best solution found
    if best_h is None:
        best_h = create_balanced_sturmian()
        best_c5 = compute_c5_exact(best_h)
        
    return (best_h.tolist(), float(best_c5), n_points)
