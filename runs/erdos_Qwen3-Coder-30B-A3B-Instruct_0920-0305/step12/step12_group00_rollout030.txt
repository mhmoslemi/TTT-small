import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort the vector
    sorted_x = np.sort(x)[::-1]
    
    # Find lambda using bisection
    def f(lam):
        projected = np.clip(sorted_x - lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x - (a + b)/2, 0, 1)
    return result

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def compute_softmax_weights(corr, alpha):
    """Compute softmax weights for smooth approximation"""
    exp_corr = np.exp(alpha * corr)
    return exp_corr / np.sum(exp_corr)

def grad_softmax(corr, alpha, h):
    """Compute gradient of smooth max with respect to h"""
    weights = compute_softmax_weights(corr, alpha)
    # Forward convolution: w * h
    forward_conv = np.convolve(weights, h, mode='full')[:len(h)]
    # Backward convolution: w_rev * h 
    backward_conv = np.convolve(weights[::-1], h, mode='full')[:len(h)]
    grad = 1.0 - forward_conv - backward_conv
    return grad

def adam_step(h, grad, m, v, t, lr=0.12, beta1=0.9, beta2=0.999, epsilon=1e-8):
    """Adam optimization step with projection"""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
    return h_new, m, v

def balanced_sturmian(n, phi=(np.sqrt(5)-1)/2):
    """Generate balanced Sturmian word"""
    h = np.array([(i * phi) % 1 < 0.5 for i in range(n)], dtype=float)
    return h

    # 2. Random uniform with projection

    h_random = project_to_sum_box(h_random, TARGET_SUM)
    initial_candidates.append(("random", h_random))
    
    # 3. Perturbed balanced word
    h_perturbed = h_sturm + np.random.normal(0, 0.05, n_points)
    h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
    initial_candidates.append(("perturbed", h_perturbed))
