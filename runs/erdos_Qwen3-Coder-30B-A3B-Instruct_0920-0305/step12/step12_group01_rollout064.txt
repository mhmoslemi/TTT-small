import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr) * (2.0 / len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_candidate_solution():
    """Generate a candidate solution based on a combinatorial construction"""
    # Start with a simple piecewise constant construction
    n_points = 120
    h = np.zeros(n_points)
    
    # Use a construction that tries to distribute mass to minimize overlap
    # This is inspired by difference sets and low autocorrelation sequences
    segments = 4
    segment_size = n_points // segments
    
    # Alternate between high and low values in segments
    for i in range(segments):
        start_idx = i * segment_size
        end_idx = start_idx + segment_size
        if i % 2 == 0:
            h[start_idx:end_idx] = 0.8
        else:
            h[start_idx:end_idx] = 0.2
    
    # Ensure the sum constraint is met
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = project_to_sum_box(h, target_sum)
    
    return h

def optimize_with_local_search(h, budget_s):
    """Perform local search with simulated annealing approach"""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Annealing parameters
    temperature = 1.0
    cooling_rate = 0.999
    min_temperature = 1e-6
    iterations_per_temp = 200
    
    current_h = h.copy()
    current_c5 = evaluate_c5(h)
    
    while temperature > min_temperature and time.time() - start_time < budget_s - 5:
        for _ in range(iterations_per_temp):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Make a small random perturbation
            h_new = current_h.copy()
            idx = np.random.randint(0, len(h_new))
            delta = np.random.normal(0, 0.05)
            h_new[idx] += delta
            
            # Project back to feasible region
            h_new = project_to_sum_box(h_new, len(h_new) / 2)
            
            c5_new = evaluate_c5(h_new)
            
            # Accept or reject based on simulated annealing
            if c5_new < current_c5 or np.random.random() < np.exp((current_c5 - c5_new) / temperature):
                current_h = h_new
                current_c5 = c5_new
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
        
        temperature *= cooling_rate
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Generate initial candidate solution
    h = generate_candidate_solution()
    c5 = evaluate_c5(h)
    
    # Local optimization using simulated annealing
    best_h, best_c5 = optimize_with_local_search(h, budget_s)
    
    # Final refinement with gradient-free optimization
    try:
        from scipy.optimize import differential_evolution
        def objective(h):
            return evaluate_c5(h)
        
        bounds = [(0, 1) for _ in range(len(best_h))]
        
        res = differential_evolution(
            objective, 
            bounds, 
            maxiter=100,
            popsize=15,
            seed=seed,
            tol=1e-10
        )
        if res.success:
            final_h = res.x
        else:
            final_h = best_h
            
        final_h = project_to_sum_box(final_h, len(final_h) / 2)
        final_c5 = evaluate_c5(final_h)
        
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), len(best_h))
