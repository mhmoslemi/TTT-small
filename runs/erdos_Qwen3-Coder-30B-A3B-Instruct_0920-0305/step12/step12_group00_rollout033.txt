import numpy as np

def balanced_sturmian(n):
    """Generate a balanced Sturmian sequence with n/2 ones."""
    phi = (np.sqrt(5) - 1) / 2
    x = np.array([int((i * phi) % 1 < 0.5) for i in range(n)])
    # Adjust to have exactly n//2 ones
    ones_count = np.sum(x)
    if ones_count > n // 2:
        zeros_indices = np.where(x == 0)[0]
        flip_count = ones_count - n // 2
        flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
        x[flip_indices] = 1
    elif ones_count < n // 2:
        ones_indices = np.where(x == 1)[0]
        flip_count = n // 2 - ones_count
        flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
        x[flip_indices] = 0
    return x

def compute_autocorr_fft(h):
    """Compute autocorrelations using FFT."""
    h_rev = h[::-1]
    b = h  # h(x) * h(x+k) 
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def swap_delta(S, i, j, n):
    """Compute ΔS_ℓ for swapping positions i and j."""
    delta_S = np.zeros(n-1)
    for l in range(1, n):
        # Only consider valid indices
        if i-l >= 0 and i+l < n:
            delta_S[l-1] += h[i-l] - h[i+l]
        else:
            if i-l >= 0:
                delta_S[l-1] += h[i-l]
            if i+l < n:
                delta_S[l-1] -= h[i+l]
        if j-l >= 0 and j+l < n:
            delta_S[l-1] -= h[j-l] - h[j+l]
        else:
            if j-l >= 0:
                delta_S[l-1] -= h[j-l]
            if j+l < n:
                delta_S[l-1] += h[j+l]
    return delta_S

def hill_climb_swap(x, S, n, max_iter=5000, max_time=0.35):
    """Greedy swap hill-climb with simulated annealing fallback."""
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    current_m = np.min(S[1:])  # exclude lag 0
    best_x = x.copy()
    best_m = current_m
    start_time = time.time()
    
    temp = 0.01 * (1.0 - current_m / (n/2))
    iteration = 0
    no_improve_count = 0
    
    while time.time() - start_time < max_time and iteration < max_iter:
        iteration += 1
        # Pick random pair (one from ones, one from zeros)
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Compute delta
        delta_S = swap_delta(S, i, j, n)
        new_S = S + delta_S
        new_m = np.min(new_S[1:])
        
        # Accept or reject
        if new_m > current_m:
            # Accept
            x[i], x[j] = x[j], x[i]
            S = new_S
            current_m = new_m
            ones = np.where(x == 1)[0]
            zeros = np.where(x == 0)[0]
            if new_m > best_m:
                best_m = new_m
                best_x = x.copy()
            no_improve_count = 0
        else:
            # Simulated annealing acceptance
            if temp > 0 and np.random.rand() < np.exp((new_m - current_m) / temp):
                x[i], x[j] = x[j], x[i]
                S = new_S
                current_m = new_m
                ones = np.where(x == 1)[0]
                zeros = np.where(x == 0)[0]
                if new_m > best_m:
                    best_m = new_m
                    best_x = x.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
                
        # Annealing decay
        temp *= 0.99
        if no_improve_count > 5 * n:
            # Reset temperature to prevent stalling
            temp = 0.01 * (1.0 - current_m / (n/2))
            no_improve_count = 0
    
    return best_x, best_m
