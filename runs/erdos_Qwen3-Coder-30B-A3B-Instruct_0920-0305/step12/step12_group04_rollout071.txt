def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):

    corr = compute_corr_fft(h)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5_and_corr(h):
    """Evaluate C5 and return correlation array"""
    corr = np.correlate(h, 1 - h, mode="full")
    O = corr * dx
    max_O = np.max(O)
    return max_O, O

def compute_gradient_for_lag(h, lag):
    """Compute gradient of O(k) for given lag k"""
    N = len(h)
    forward = np.zeros(N)
    backward = np.zeros(N)
    
    if lag >= 0:
        forward[:N - lag] = h[lag:]
        backward[lag:] = h[:N - lag]
    else:
        k = -lag
        forward[k:] = h[:N - k]
        backward[:N - k] = h[k:]
    
    grad = 1.0 - forward - backward
    return grad

def subgradient_descent(h_init, target_sum, max_iter, step0, time_limit):
    """Run subgradient descent optimization"""
    h = h_init.copy()
    best_h = h.copy()
    best_val, _ = evaluate_c5_and_corr(h)
    
    t_start = time.time()
    iteration = 0
    
    while time.time() - t_start < time_limit and iteration < max_iter:
        max_O, O_arr = evaluate_c5_and_corr(h)
        
        # Find the lag that gives maximum O
        lag = np.argmax(O_arr) - (n_points - 1)
        
        # Compute gradient for this lag
        grad = compute_gradient_for_lag(h, lag)
        
        # Adaptive step size
        step = step0 / np.sqrt(iteration + 1)
        
        # Update
        h_new = h - step * grad
        h_new = project_to_sum_box(h_new, target_sum)
        
        new_val, _ = evaluate_c5_and_corr(h_new)
        
        # Accept the new solution
        h = h_new
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()
        
        iteration += 1
    
    return best_h, best_val
