def toeplitz_A(n, lag):
    """Construct Toeplitz matrix A_ℓ for lag ℓ."""
    # Create a sparse representation of A_ℓ
    i_vals = np.arange(n)
    j_vals = np.arange(n)
    i_vals, j_vals = np.meshgrid(i_vals, j_vals, indexing='ij')
    mask = np.abs(i_vals - j_vals) == lag
    # Return coo format for efficient handling
    rows, cols = np.where(mask)
    data = np.ones(len(rows))
    from scipy.sparse import coo_matrix
    return coo_matrix((data, (rows, cols)), shape=(n, n)).tocsr()

def compute_corr_fft(h):
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def eval_autocorr(h, A_list):
    """Result is a list of S_ℓ values."""
    return [h @ A @ h for A in A_list]
