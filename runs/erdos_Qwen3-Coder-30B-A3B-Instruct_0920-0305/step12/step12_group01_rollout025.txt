import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode='full')
    # The maximum correlation corresponds to the maximum overlap
    return np.max(corr) * (2.0 / len(h))

def project_to_sum_box(h, target_sum):
    """Project h onto sum constraint with box constraints [0,1]"""
    h = np.array(h)
    # Simple projection: adjust values to meet constraint
    current_sum = np.sum(h)
    if current_sum == target_sum:
        return h
    
    # Use clipping and rescaling approach 
    if current_sum > target_sum:
        # Reduce values to meet sum target
        sorted_h = np.sort(h)[::-1]
        cumulative = np.cumsum(sorted_h)
        # Find the largest index where sum is <= target_sum
        idx = np.argmax(cumulative >= target_sum) 
        if cumulative[idx] > target_sum:
            idx -= 1
        # Must reduce values at indices > idx
        h = np.array(h)
        for i in range(idx + 1, len(h)):
            h[i] = 0
        # Rescale remaining values to meet sum exactly
        remaining_sum = np.sum(h[:idx+1])
        remaining = target_sum - remaining_sum
        if remaining_sum > 0:
            scale = remaining / remaining_sum
            h[:idx+1] *= scale
    else:
        # Increase values to meet sum target
        sorted_h = np.sort(h)
        cumulative = np.cumsum(sorted_h)
        # Find the smallest index where sum is >= target_sum
        idx = np.argmax(cumulative >= target_sum) 
        if idx == 0 and cumulative[0] < target_sum:
            # Not enough to meet target, just set all to 1 except last
            h = np.ones_like(h)
            h[-1] = (target_sum - np.sum(h[:-1])) / (1.0 - h[-1]) if h[-1] != 1 else 0
        else:
            # Increase values to meet sum exactly
            # Set values at indices > idx to 1
            h = np.array(h)
            for i in range(idx+1, len(h)):
                h[i] = 1
            # Rescale first idx+1 values to meet target exactly
            remaining_sum = np.sum(h[:idx+1])
            remaining = target_sum - remaining_sum
            if remaining_sum > 0:
                scale = remaining / remaining_sum
                h[:idx+1] *= scale
    # Ensure all values are in [0,1]
    h = np.clip(h, 0, 1)
    return h

def generate_good_initial():
    """Generate initial construction based on mathematical insight"""
    n = 120
    dx = 2.0 / n
    # Start with an alternating pattern with a few key features
    h = np.zeros(n)
    
    # Create a pattern that's 1 in some regions, 0 in others
    # We want to smooth this to get a better starting point
    # Place the high values in regions where we can best minimize overlap
    
    # Use a construction inspired by difference sets and graph theory
    # Place 1s and 0s in a pattern that tries to spread out high values
    period = int(n / 6)  # maybe 20 points
    for i in range(n):
        if i % period in [0, 1, 2, period//2, period//2+1, period//2+2]:
            h[i] = 1.0
        else:
            h[i] = 0.0
    
    # Scale to meet sum constraint
    target_sum = n / 2
    h = project_to_sum_box(h, target_sum)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 120
    dx = 2.0 / n_points
    
    # Start with a good construction
    h_current = generate_good_initial()
    c5_current = evaluate_c5(h_current)
    
    best_h = h_current.copy()
    best_c5 = c5_current
    
    print(f"Initial C5: {best_c5}")
    
    # Use differential evolution for global optimization
    # This is a stochastic global search method that may escape local minima
    bounds = [(0, 1) for _ in range(n_points)]
    target_sum = n_points / 2
    
    def objective(h):
        # Project to unit box and sum constraint
        h_projected = project_to_sum_box(h, target_sum)
        return evaluate_c5(h_projected)
    
    # Try different parameterizations
    maxiter = 500
    popsize = 15
    
    try:
        # Run differential evolution with multiple restarts
        for restart in range(10):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Random restart
            np.random.seed(int(time.time() * 1000) % 2**32)
            h0 = np.random.rand(n_points)
            h0 = project_to_sum_box(h0, target_sum)
            
            # Apply differential evolution
            res = differential_evolution(
                objective,
                bounds,
                maxiter=maxiter,
                popsize=popsize,
                seed=seed + restart,
                disp=False,
                atol=1e-10,
                tol=1e-10
            )
            
            if res.success:
                h_new = res.x
                h_new = project_to_sum_box(h_new, target_sum)
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    print(f"Improved C5 to: {best_c5}")
                    if best_c5 < 0.3795:
                        break  # Early stopping
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
