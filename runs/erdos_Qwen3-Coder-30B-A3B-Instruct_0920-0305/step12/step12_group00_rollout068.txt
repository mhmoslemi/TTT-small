def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""

        return np.sum(projected) - target_sum

    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10

        fm = f(m)
        if abs(fm) < tol:
            break

            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def convolve_fft(a, b):
    """Perform convolution using FFT, returns same length as b"""
    L = len(a) + len(b) - 1

    A = np.fft.fft(a, L_fft)
    B = np.fft.fft(b, L_fft)
    result = np.fft.ifft(A * B).real[:len(b)]
    return result

    grad = 1.0 - conv1 - conv2
