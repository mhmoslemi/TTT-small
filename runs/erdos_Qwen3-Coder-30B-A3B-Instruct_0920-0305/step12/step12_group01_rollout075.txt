import numpy as np
from scipy.signal import correlate
import time
import math

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr) * (2.0 / len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_combinatorial_sequence(n):
    """Generate a combinatorial sequence with good properties"""
    # Try to construct a sequence with low auto-correlations
    # Using a construction based on difference sets or related combinatorial designs
    
    # For small n, we can try a simple pattern
    if n <= 20:
        # Simple binary pattern
        h = np.zeros(n)
        # Place 1s in well-separated positions
        for i in range(n):
            if i % 3 == 0:  # Every third position
                h[i] = 1
        return h
    
    # For larger n, use a more sophisticated approach
    h = np.zeros(n)
    
    # Start with a uniform distribution, then tweak
    for i in range(n):
        h[i] = 0.5 + 0.3 * math.sin(2 * math.pi * i / n)
    
    # Normalize to desired sum
    current_sum = np.sum(h)
    h = h * (n / 2) / current_sum
    
    # Project onto constraints
    h = np.clip(h, 0, 1)
    h = project_to_sum_box(h, n / 2)
    
    return h

def perturb_solution(h, noise_level=0.02):
    """Perturb solution with noise"""
    noise = np.random.normal(0, noise_level, len(h))
    h_perturbed = h + noise
    return project_to_sum_box(h_perturbed, len(h) / 2)

def optimize_with_guided_search(h0, budget_s, start_time):
    """Use a guided local search approach"""
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Parameters
    max_iterations = 10000
    perturbation_magnitude = 0.05
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate candidate by perturbing current best
        candidate_h = perturb_solution(best_h, perturbation_magnitude)
        
        # Evaluate candidate
        candidate_c5 = evaluate_c5(candidate_h)
        
        # Accept if better
        if candidate_c5 < best_c5:
            best_h = candidate_h.copy()
            best_c5 = candidate_c5
            
            # Reduce perturbation magnitude as we get closer to optimum
            perturbation_magnitude = max(0.001, perturbation_magnitude * 0.999)
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use a different approach: generate based on known good constructions
    n_points = 120
    dx = 2.0 / n_points
    
    # Start with a combinatorial construction
    h_initial = generate_combinatorial_sequence(n_points)
    
    # Refine with guided local search
    final_h, final_c5 = optimize_with_guided_search(h_initial, budget_s, start_time)
    
    # Try a few more specialized constructions if time allows
    for i in range(5):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Alternative: try a symmetric construction
        h_alt = np.zeros(n_points)
        mid = n_points // 2
        
        # Create a symmetric pattern with a central peak
        for j in range(n_points):
            dist = abs(j - mid)
            # Gaussian-like shape
            h_alt[j] = max(0, 1.0 - dist * 0.01)
            
        # Normalize
        h_alt = h_alt * (n_points / 2) / np.sum(h_alt)
        h_alt = project_to_sum_box(h_alt, n_points / 2)
        c5_alt = evaluate_c5(h_alt)
        
        if c5_alt < final_c5:
            final_h = h_alt
            final_c5 = c5_alt
    
    return (final_h.tolist(), float(final_c5), n_points)
