TARGET_SUM = None

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort the vector 
    sorted_x = np.sort(x)
    # Binary search for lambda
    def f(lam):
        projected = np.clip(sorted_x - lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    lambda_val = (a + b) / 2
    result = np.clip(x - lambda_val, 0, 1)
    return result

def softmax_weights(corr, alpha):
    """Compute softmax weights for smooth max"""
    exp_corr = np.exp(alpha * corr)
    return exp_corr / np.sum(exp_corr)

def grad_softmax(corr, alpha, h):
    """Compute gradient of soft max with respect to h"""
    weights = softmax_weights(corr, alpha)
    
    # Compute forward and backward convolution
    h_rev = h[::-1]
    weights_rev = weights[::-1]
    
    # Zero pad to avoid circular convolution
    L = 2 * len(h) - 1
    L_fft = 1 << (L - 1).bit_length()
    
    W = np.fft.fft(weights, L_fft)
    H = np.fft.fft(h, L_fft)
    W_rev = np.fft.fft(weights_rev, L_fft)
    
    conv1 = np.fft.ifft(W * H).real[:len(h)]
    conv2 = np.fft.ifft(W_rev * H).real[:len(h)]
    
    grad = np.ones_like(h) - conv1 - conv2
    return grad

def adam_step(h, grad, m, v, t, lr=0.12, beta1=0.9, beta2=0.999, epsilon=1e-8):
    """Adam optimizer step"""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
    return h_new, m, v
