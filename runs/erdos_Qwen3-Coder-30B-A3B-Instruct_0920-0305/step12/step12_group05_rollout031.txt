import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_softmax(h, beta):
    """Compute gradient of smoothed max correlation with respect to h"""
    c = compute_corr_fft(h)
    c_max = np.max(c)
    # Numerical stability: subtract max before exponentiation
    w_raw = np.exp(beta * (c - c_max))
    p = w_raw / np.sum(w_raw)
    
    # reversed weights for convolution
    w_rev = p[::-1]
    
    # Compute convolutions
    conv2 = np.convolve(w_rev, 1 - h, mode='full')
    conv3 = np.convolve(w_rev, h, mode='full')
    
    # Extract relevant indices
    start_idx = n_points - 1
    end_idx = 2 * n_points - 1
    
    g = dx * (conv2[start_idx:end_idx] - conv3[start_idx:end_idx])
    return g, c

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    best_h = None
    best_c5 = np.inf
    
    # Multiple restarts with different random seeds
    num_restarts = 5
    for restart in range(num_restarts):
        # Random initialization
        h = np.random.rand(n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Try initial solution if available
        if restart == 0 and 'initial_h_values' in globals():
            try:
                h0 = np.array(initial_h_values)
                h0 = project_to_sum_box(h0, TARGET_SUM)
                c5_0 = evaluate_c5(h0)
                if c5_0 < evaluate_c5(h):
                    h = h0.copy()
            except:
                pass
        
        # Gradient descent optimization
        beta = 1.0
        eta = 0.1
        iter_count = 0
        max_iter = 5000
        
        while iter_count < max_iter and time.time() - start_time < budget_s - 10:
            # Compute gradient
            g, corr = gradient_softmax(h, beta)
            
            # Line search
            step_size = eta
            improved = False
            
            for _ in range(5):  # At most 5 line search steps
                h_new = project_to_sum_box(h - step_size * g, TARGET_SUM)
                c5_new = evaluate_c5(h_new)
                if c5_new < evaluate_c5(h):
                    h = h_new
                    improved = True
                    break
                step_size *= 0.5
            
            if not improved:
                # Random perturbation if no improvement
                perturbation = 0.01 * np.random.randn(n_points)
                h = project_to_sum_box(h + perturbation, TARGET_SUM)
            
            # Update best solution
            c5_current = evaluate_c5(h)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h.copy()
            
            # Adapt parameters
            iter_count += 1
            if iter_count % 100 == 0:
                beta = min(beta * 1.2, 100)
                eta = max(eta * 0.95, 1e-6)
            
            # Occasionally add random perturbation to escape local optima
            if np.random.rand() < 0.05:
                perturbation = 0.02 * np.random.randn(n_points)
                h = project_to_sum_box(h + perturbation, TARGET_SUM)
        
        # Early stopping if we found good enough solution
        if best_c5 <= 0.3807:
            break
    
    if best_h is None:
        # Fallback to initial solution
        h0 = np.array(initial_h_values)
        h0 = project_to_sum_box(h0, TARGET_SUM)
        best_h = h0.copy()
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
