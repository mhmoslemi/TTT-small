import numpy as np
import time

def compute_autocorr(h):
    """Compute full autocorrelation of h with 1-h using numpy"""
    return np.correlate(h, 1-h, mode="full")

def evaluate_solution(h):
    """Evaluate the C5 value of a solution"""
    corr = compute_autocorr(h)
    return np.max(corr) * (2.0 / len(h))

def generate_beatty_pattern(n):
    """Generate a Beatty sequence pattern as initial guess"""
    phi = (1 + np.sqrt(5)) / 2
    positions = np.floor(np.arange(n) * phi) % n
    positions = np.sort(positions.astype(int))
    
    h = np.zeros(n)
    h[positions[:n//2]] = 1
    return h

def generate_random_pattern(n):
    """Generate a random binary pattern with exactly n/2 ones"""
    h = np.zeros(n)
    ones_indices = np.random.permutation(n)[:n//2]
    h[ones_indices] = 1
    return h

    """Update correlation vector C_raw when swapping h[i] and h[j]"""
    # i changes from 0 to 1
    # j changes from 1 to 0
    
    n = len(h)
    delta_C = np.zeros(2*n - 1)
    
    # When h[i] goes from 0 to 1, it contributes to correlations:
    # For each k where i+k is in valid range [0, n-1]
    for k in range(n):
        if i + k < n:
            delta_C[k] -= 1 - h[i + k]  # h[i+k] was 0 earlier, now 1-0=1, so added -1
    
    # When h[i] goes from 0 to 1, it contributes to correlations with negative shifts:
    # For each k where i-k is in valid range [0, n-1]  
    for k in range(1, n):
        if i - k >= 0:
            delta_C[-k] -= h[i - k]  # contribution to reverse shift
    
    # When h[j] goes from 1 to 0:
    # For each k where j+k is in valid range [0, n-1]
    for k in range(n):
        if j + k < n:
            delta_C[k] += 1 - h[j + k]  # h[j+k] was 1 earlier, now 1-1=0, so added +1
    
    # When h[j] goes from 1 to 0:
    # For each k where j-k is in valid range [0, n-1]
    for k in range(1, n):
        if j - k >= 0:
            delta_C[-k] += h[j - k]  # contribution to reverse shift
    
    return delta_C
