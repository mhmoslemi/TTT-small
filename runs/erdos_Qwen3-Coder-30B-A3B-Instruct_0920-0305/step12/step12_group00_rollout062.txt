import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def balanced_sturmian(n):
    """Generate a balanced Sturmian word as starting point"""
    phi = (np.sqrt(5) - 1) / 2
    x = np.zeros(n, dtype=int)
    for i in range(n):
        if (i * phi) % 1 < 0.5:
            x[i] = 1
    # Ensure sum is exactly n//2
    current_sum = np.sum(x)
    target_sum = n // 2
    if current_sum > target_sum:
        # Flip some 1s to 0s
        ones = np.where(x == 1)[0]
        to_flip = int(current_sum - target_sum)
        for i in range(to_flip):
            x[ones[i]] = 0
    elif current_sum < target_sum:
        # Flip some 0s to 1s
        zeros = np.where(x == 0)[0]
        to_flip = int(target_sum - current_sum)
        for i in range(to_flip):
            x[zeros[i]] = 1
    return x

def compute_autocorr_fft(x):
    """Compute autocorrelations using FFT"""
    x_rev = x[::-1]
    b = x
    L = 2 * len(x) - 1
    corr = np.fft.ifft(np.fft.fft(x_rev, L) * np.fft.fft(b, L)).real
    return corr

def swap_delta(S, x, i, j, n):
    """Compute change in autocorrelations from swapping x[i] and x[j]"""
    delta_S = np.zeros(n-1)
    for l in range(1, n):
        if i-l >= 0:
            delta_S[l-1] -= x[i-l]
        if i+l < n:
            delta_S[l-1] -= x[i+l]
        if j-l >= 0:
            delta_S[l-1] += x[j-l]
        if j+l < n:
            delta_S[l-1] += x[j+l]
        # Also consider the effect on the lags from the actual swap
        delta_S[l-1] += 2*(x[i] - x[j])
    return delta_S

def hill_climb_swap(x, S, n, max_time):
    """Hill-climb with simulated annealing fallback"""
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    current_m = np.min(S[1:])  # excluding lag 0
    current_c5 = 0.379  # dummy upper bound for max_time check
    
    # For simulated annealing
    T = 0.001 * current_c5
    cooling_rate = 0.99
    
    start_time = time.time()
    
    attempt_count = 0
    while time.time() - start_time < max_time:
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        # Pick a random swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        delta_S = swap_delta(S, x, i, j, n)
        new_S = S + delta_S
        new_m = np.min(new_S[1:])
        
        if new_m > current_m:
            x[i], x[j] = x[j], x[i]
            S = new_S
            current_m = new_m
            # Update C5 using exact computation
            corr = compute_autocorr_fft(x)
            current_c5 = np.max(corr) * (2.0 / n)
            attempt_count = 0  # reset counter
        else:
            # Simulated annealing
            if np.random.rand() < np.exp((new_m - current_m) / T):
                x[i], x[j] = x[j], x[i]
                S = new_S
                current_m = new_m
                # Update C5 using exact computation
                corr = compute_autocorr_fft(x)
                current_c5 = np.max(corr) * (2.0 / n)
                attempt_count = 0  # reset counter
            else:
                attempt_count += 1
        
        # Switch to simulated annealing if stuck
        if attempt_count > 5 * n:
            T = max(T * cooling_rate, 1e-8)
            attempt_count = 0

    return x, current_m, current_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_values = [128, 144]  # Try different sizes
    best_c5 = float('inf')
    best_h = None
    best_m = -1
    
    for n in n_values:
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        dx = 2.0 / n
        N = n // 2  # Target sum
        
        # Time allocation: 45% for main ILP
        ilp_time = 0.45 * budget_s
        try:
            # Build the model
            x = cp.Variable(n, boolean=True)
            y = cp.Variable((n, n-1), boolean=True)
            m = cp.Variable()
            
            # Constraints: sum constraint
            constraints = [cp.sum(x) == N]
            
            # McCormick linearization for y[i,l] = x[i] * x[i+l]
            for l in range(1, n):
                for i in range(n - l):
                    constraints.append(y[i, l-1] <= x[i])
                    constraints.append(y[i, l-1] <= x[i+l])
                    constraints.append(y[i, l-1] >= x[i] + x[i+l] - 1)
            
            # Define S_l as sum of y[i,l]
            S = cp.Variable(n-1)
            for l in range(1, n):
                i_start = 0
                i_end = n - l
                if i_end > i_start:
                    constraints.append(S[l-1] == cp.sum(y[i_start:i_end, l-1]))
            
            # Ensure m is lower bound on all S_l
            constraints.extend([S[l-1] >= m for l in range(1, n)])
            
            # Objective: maximize m
            objective = cp.Maximize(m)
            prob = cp.Problem(objective, constraints)
            
            # Solve with time limit
            prob.solve(solver=cp.CBC, max_seconds=ilp_time)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                x_sol = np.round(x.value).astype(int)
                if x_sol is not None:
                    # Verify it's valid and compute exact C5
                    if np.sum(x_sol) == N:
                        corr = compute_autocorr_fft(x_sol)
                        c5 = np.max(corr) * dx
                        m_opt = N - np.min(corr[1:])  # since C5 = dx*(N-m)
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = x_sol
                            best_m = m_opt
        except Exception:
            pass
        
        # Break if we already have a solution with good bound
        if best_c5 < 0.380:
            break
    
    # If no ILP succeeded, use stochastic hill-climbing
    if best_h is None:
        # Start with a Sturmian sequence if no ILP solution found
        h0 = balanced_sturmian(n)
        corr = compute_autocorr_fft(h0)
        current_c5 = np.max(corr) * (2.0 / n)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h0.copy()
            best_m = N - np.min(corr[1:])
    
    # Local search refinement
    max_refine_time = 0.35 * budget_s
    if best_h is not None:
        # Compute exact autocorrelations
        S = compute_autocorr_fft(best_h)
        best_h, best_m, final_c5 = hill_climb_swap(best_h, S, n, max_refine_time)
        if final_c5 < best_c5:
            best_c5 = final_c5
        # Compute final correct c5
        corr = compute_autocorr_fft(best_h)
        best_c5 = np.max(corr) * (2.0 / n)  # final correction
    
    # Final fallback: restarts with random balanced binary vectors
    final_h = best_h.copy()
    final_c5 = best_c5
    final_m = best_m
    
    max_fallback_time = 0.05 * budget_s
    start_fallback = time.time()
    while time.time() - start_fallback < max_fallback_time:
        # Generate a random balanced binary vector
        x = np.random.rand(n)
        x = (x > 0.5).astype(int)
        
        # Adjust sum to exactly N
        current_sum = np.sum(x)
        if current_sum > N:
            ones = np.where(x == 1)[0]
            to_flip = int(current_sum - N)
            for i in range(to_flip):
                x[ones[i]] = 0
        elif current_sum < N:
            zeros = np.where(x == 0)[0]
            to_flip = int(N - current_sum)
            for i in range(to_flip):
                x[zeros[i]] = 1
                
        if np.sum(x) != N:
            continue
            
        # Refine with hill-climb
        S = compute_autocorr_fft(x)
        refined_x, m_new, c5_new = hill_climb_swap(x, S, n, 0.02)
        if c5_new < final_c5:
            final_c5 = c5_new
            final_h = refined_x.copy()
            final_m = m_new
    
    return (final_h.tolist(), float(final_c5), n)
