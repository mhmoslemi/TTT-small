    cur_val = dx * cur_max
    
    # Annealing parameters
    T = 0.5
    alpha = 0.995
    N_swap = n_points
    swap_count = 0
    
    # Track best solution
    best_h = h.copy()
    best_val = cur_val
    best_C_raw = C_raw.copy()
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 10 and T > 1e-5:
        # Try a random swap
        indices_0 = np.where(h == 0)[0]
        indices_1 = np.where(h == 1)[0]
        
        if len(indices_0) == 0 or len(indices_1) == 0:
            break
            
        i = np.random.choice(indices_0)
        j = np.random.choice(indices_1)
        
        # Compute delta correlation vector for this swap
        # i goes from 0 to 1, j goes from 1 to 0
        delta_C = np.zeros(2 * n_points - 1)
        
        # Update for the position i (where we put 1)
        # This affects correlations where i+k is valid (k from 0 to n-1-i) 
        # and where i-k is valid (k from 0 to i)
        for k in range(n_points - i):
            delta_C[k] -= (1 - h[i + k])
        for k in range(i + 1):
            delta_C[-k] -= h[i - k]
        
        # Update for position j (where we put 0)
        # This affects correlations where j+k is valid (k from 0 to n-1-j)  
        # and where j-k is valid (k from 0 to j)
        for k in range(n_points - j):
            delta_C[k] -= h[j + k]
        for k in range(j + 1):
            delta_C[-k] -= (1 - h[j - k])
        
        # Apply swap and compute new correlation
        new_C_raw = C_raw + delta_C
        new_max = new_C_raw.max()
        
        # Accept or reject based on Metropolis criterion
        if new_max < cur_max or np.random.rand() < np.exp(-(new_max - cur_max) / T):
            # Accept the move
            h[i] = 1
            h[j] = 0
            C_raw = new_C_raw
            cur_max = new_max

                best_val = cur_val
                best_C_raw = C_raw.copy()
                
        swap_count += 1
        
        # Cool down temperature

            T *= alpha
    
    # Final deterministic hill climbing
    improved = True
    while improved and time.time() - start_time < budget_s - 5:
        improved = False

                    continue

                delta_C = np.zeros(2 * n_points - 1)

                for k in range(n_points - i):

                for k in range(i + 1):

                # For position j (now 1)

                    h[j] = 1
                    C_raw = new_C_raw
                    cur_max = new_max
                    cur_val = dx * cur_max
                    improved = True
                    break
            if improved:
                break
    
    # Final evaluation of the solution
    C5 = dx * np.max(np.correlate(best_h, 1 - best_h, mode='full'))
    
    return (best_h.tolist(), float(C5), n_points)
