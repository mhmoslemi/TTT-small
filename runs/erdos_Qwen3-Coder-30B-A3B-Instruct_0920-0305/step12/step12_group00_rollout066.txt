import numpy as np
from scipy.signal import correlate
from scipy.sparse import diags
import cvxpy as cp
import time
import math

# Global constants
n_points = 128
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def build_toeplitz_matrices(n):
    """Precompute all Toeplitz matrices A_ℓ for lags ℓ = 1,...,n-1"""
    A_list = []
    for lag in range(1, n):
        # Create the Toeplitz matrix with 1s at distance lag
        data = [1] * (n - lag)
        offsets = [lag]
        A = diags(data, offsets, shape=(n, n), format='csr')
        A = A + A.T  # Make symmetric
        A_list.append(A)
    return A_list

def sdp_rounding(X, n, target_sum):
    """Randomized hyperplane rounding with top-N projection"""
    try:
        # Eigen decomposition
        eigenvals, eigenvecs = np.linalg.eigh(X)
        # Keep only positive eigenvalues
        mask = eigenvals > 1e-8
        eigenvals = eigenvals[mask]
        eigenvecs = eigenvecs[:, mask]
        
        # Form the factor V * sqrt(λ)
        V = eigenvecs @ np.diag(np.sqrt(eigenvals))
        
        best_h = None
        best_c5 = float('inf')
        
        # Try multiple random vectors
        for _ in range(5):
            g = np.random.randn(V.shape[1])
            scores = V @ g
            
            # Top-N threshold
            indices = np.argsort(-scores)
            h = np.zeros(n)
            h[indices[:target_sum]] = 1
            
            # Ensure exact sum
            current_sum = int(np.sum(h))
            if current_sum > target_sum:
                # Remove extra 1s
                to_zero = indices[target_sum:current_sum]
                h[to_zero] = 0
            elif current_sum < target_sum:
                # Add missing 1s
                to_one = indices[current_sum:target_sum]
                h[to_one] = 1
            
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        return best_h, best_c5
    except Exception:
        # Fallback to deterministic rounding
        eigenvals, eigenvecs = np.linalg.eigh(X)
        # Just use the first principal component
        scores = eigenvecs[:, -1]
        indices = np.argsort(-scores)
        h = np.zeros(n)
        h[indices[:target_sum]] = 1
        # Ensure exact sum
        current_sum = int(np.sum(h))
        if current_sum > target_sum:
            to_zero = indices[target_sum:current_sum]
            h[to_zero] = 0
        elif current_sum < target_sum:
            to_one = indices[current_sum:target_sum]
            h[to_one] = 1
        return h, evaluate_c5(h)

def hill_climb_swap(h, target_sum, max_iters=5000, time_limit=None):
    """Hill-climb with swapping 0s and 1s"""
    start_time = time.time()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    # Precompute correlations
    corr = compute_corr_fft(h)
    max_corr = np.max(corr)
    c5 = dx * max_corr
    
    iters = 0
    while iters < max_iters:
        if time_limit and (time.time() - start_time) > time_limit:
            break
            
        # Pick random indices
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Fast update of correlations
        delta_corr = np.zeros(n_points)
        for k in range(n_points):
            if (i+k) % n_points < n_points and (i-k) % n_points < n_points:
                delta_corr[k] += h[(i+k)%n_points] - h[(i-k)%n_points]
            if (j+k) % n_points < n_points and (j-k) % n_points < n_points:
                delta_corr[k] += h[(j-k)%n_points] - h[(j+k)%n_points]
                
        new_corr = corr + delta_corr
        new_max_corr = np.max(new_corr)
        new_c5 = dx * new_max_corr
        
        if new_c5 < c5:
            # Accept the swap
            h[i] = 0
            h[j] = 1
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            corr = new_corr
            c5 = new_c5
            
        iters += 1
        
    return h, c5
