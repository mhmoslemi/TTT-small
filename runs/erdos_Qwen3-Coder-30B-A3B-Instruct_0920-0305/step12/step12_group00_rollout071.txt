import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import math

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort the vector
    sorted_x = np.sort(x)[::-1]
    
    # Find lambda using binary search
    a, b = -1.0, 1.0
    tol = 1e-10
    
    def f(lam):
        projected = np.clip(sorted_x - lam, 0, 1)
        return np.sum(projected) - target_sum
    
    # Refine bounds
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    lambda_val = (a + b) / 2
    result = np.clip(x - lambda_val, 0, 1)
    return result

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def compute_corr_exact(h):
    """Compute exact correlation for binary vectors"""
    corr = np.correlate(h, 1 - h, mode='full')
    return corr

def softmax_weights(corr, alpha):
    """Compute softmax weights"""
    exp_corr = np.exp(alpha * corr)
    return exp_corr / np.sum(exp_corr)

def grad_softmax(corr, alpha, h):
    """Compute gradient of softmax"""
    weights = softmax_weights(corr, alpha)
    # Forward convolution: w * h
    L = len(h) + len(weights) - 1
    L_fft = 1 << (L - 1).bit_length()
    w_fft = np.fft.fft(weights, L_fft)
    h_fft = np.fft.fft(h, L_fft)
    conv1 = np.fft.ifft(w_fft * h_fft).real[:len(weights)]
    
    # Backward convolution: w_rev * h
    w_rev = weights[::-1]
    w_rev_fft = np.fft.fft(w_rev, L_fft)
    conv2 = np.fft.ifft(w_rev_fft * h_fft).real[:len(weights)]
    
    # Combine gradient
    grad = 1.0 - conv1 - conv2
    return grad

def adam_step(h, m, v, grad, t, alpha, beta1=0.9, beta2=0.999, eps=1e-8, lr=0.12):
    """Perform Adam step"""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
    return h_new, m, v

def sturmian_sequence(n, phi=(math.sqrt(5) - 1) / 2):
    """Generate balanced Sturmian sequence"""
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1 if (i * phi) % 1 < 0.5 else 0
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points // 2
    best_c5 = float('inf')
    best_h = None
    
    # Store best solution so far
    def update_best(h, c5):
        nonlocal best_c5, best_h
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Run Adam optimization with different initializations
    initializations = []
    
    # 1. Sturmian sequence
    h_sturm = sturmian_sequence(n_points)
    initializations.append(("sturmian", h_sturm))
    
    # 2. Random vectors
    for i in range(3):
        h_random = np.random.rand(n_points)
        initializations.append((f"random_{i}", h_random))
    
    # 3. Perturbed Sturmian
    h_perturbed = h_sturm + np.random.normal(0, 0.05, n_points)
    initializations.append(("perturbed", h_perturbed))
    
    # Adam optimization with warm start
    max_iter = 8000
    alpha = 4.0
    alpha_growth = 1.5
    alpha_max = 200.0
    patience = 200
    
    for init_name, init_h in initializations:
        if time.time() - start_time > 0.45 * budget_s:
            break
            
        h = project_to_sum_box(init_h, TARGET_SUM)
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        t = 1
        
        prev_c5 = float('inf')
        patience_count = 0
        
        for i in range(max_iter):
            if time.time() - start_time > 0.45 * budget_s:
                break
            
            corr = compute_corr_fft(h)
            c5 = dx * np.max(corr)
            
            # Early stopping if no improvement
            if abs(c5 - prev_c5) < 1e-8:
                patience_count += 1
            else:
                patience_count = 0
                
            if patience_count > patience:
                break
                
            prev_c5 = c5
            
            # Gradient computation
            grad = grad_softmax(corr, alpha, h)
            
            # Adam step
            h, m, v = adam_step(h, m, v, grad, t, alpha)
            h = project_to_sum_box(h, TARGET_SUM)
            
            update_best(h, c5)
            
            t += 1
            
            # Increase smoothness parameter
            if i % 200 == 0:
                alpha = min(alpha * alpha_growth, alpha_max)
    
    # Convert best fractional solution to binary
    if best_h is None:
        # Fall back to initial sturmian if no better solution found
        best_h = sturmian_sequence(n_points)
    
    # Round to binary with correct sum
    h_binary = (best_h >= 0.5).astype(float)
    
    # Adjust sum to be exactly TARGET_SUM
    current_sum = np.sum(h_binary)
    if current_sum > TARGET_SUM:
        # Flip some 1's to 0's
        ones_indices = np.where(h_binary == 1)[0]
        to_flip = int(current_sum - TARGET_SUM)
        flip_indices = np.random.choice(ones_indices, to_flip, replace=False)
        h_binary[flip_indices] = 0
    elif current_sum < TARGET_SUM:
        # Flip some 0's to 1's
        zeros_indices = np.where(h_binary == 0)[0]
        to_flip = int(TARGET_SUM - current_sum)
        flip_indices = np.random.choice(zeros_indices, to_flip, replace=False)
        h_binary[flip_indices] = 1
        
    h_binary = project_to_sum_box(h_binary, TARGET_SUM)
    
    # Evaluate exact C5 for the binary solution
    exact_c5 = evaluate_c5(h_binary)
    update_best(h_binary, exact_c5)
    
    # Binary hill-climbing (swap moves)
    if best_h is not None:
        # Create efficient data structures for swaps
        current_h = best_h.copy()
        current_c5 = exact_c5
        
        # Calculate current overlaps
        current_corr = compute_corr_fft(current_h)
        current_max_overlap = dx * np.max(current_corr)
        
        # Store indices
        ones = np.where(current_h == 1)[0]
        zeros = np.where(current_h == 0)[0]
        
        n_swaps = 0
        max_swaps = 5 * n_points
        while time.time() - start_time < 0.8 * budget_s and n_swaps < max_swaps:
            # Pick two random indices
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Compute the effect of swapping
            old_corr = compute_corr_fft(current_h)
            
            # Get the effect of swapping h[i] and h[j]
            new_h = current_h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]
            
            new_corr = compute_corr_fft(new_h)
            new_c5 = dx * np.max(new_corr)
            
            if new_c5 < current_c5:
                current_h = new_h.copy()
                current_c5 = new_c5
                update_best(current_h, current_c5)
                
                # Update indices
                ones = np.where(current_h == 1)[0]
                zeros = np.where(current_h == 0)[0]
                
            n_swaps += 1
    
    # Additional binary restarts
    if time.time() - start_time < 0.95 * budget_s:
        for _ in range(5):
            # Generate random binary with correct sum
            h_rand = np.random.rand(n_points)
            h_rand = project_to_sum_box(h_rand, TARGET_SUM)
            
            h_rand_binary = (h_rand >= 0.5).astype(float)
            
            # Adjust sum
            current_sum = np.sum(h_rand_binary)
            if current_sum > TARGET_SUM:
                ones_indices = np.where(h_rand_binary == 1)[0]
                to_flip = int(current_sum - TARGET_SUM)
                flip_indices = np.random.choice(ones_indices, to_flip, replace=False)
                h_rand_binary[flip_indices] = 0
            elif current_sum < TARGET_SUM:
                zeros_indices = np.where(h_rand_binary == 0)[0]
                to_flip = int(TARGET_SUM - current_sum)
                flip_indices = np.random.choice(zeros_indices, to_flip, replace=False)
                h_rand_binary[flip_indices] = 1
                
            h_rand_binary = project_to_sum_box(h_rand_binary, TARGET_SUM)
            
            # Evaluate
            rand_c5 = evaluate_c5(h_rand_binary)
            update_best(h_rand_binary, rand_c5)
    
    # Final evaluation of best solution
    if best_h is None:
        # Fallback to initial sturmian
        best_h = sturmian_sequence(n_points)
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
