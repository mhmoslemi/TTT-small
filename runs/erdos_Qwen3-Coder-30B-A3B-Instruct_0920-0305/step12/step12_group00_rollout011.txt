import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort the vector
    sorted_indices = np.argsort(x)[::-1]
    sorted_x = x[sorted_indices]
    
    # Find lambda using binary search
    def f(lam):
        projected = np.clip(sorted_x - lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = sorted_x[0] - 1.0, sorted_x[-1] + 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    lambda_val = (a + b) / 2
    result = np.clip(x - lambda_val, 0, 1)
    return result

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def softmax_weights(corr, alpha):
    """Compute softmax weights for the correlation values"""
    exp_corr = np.exp(alpha * corr)
    return exp_corr / np.sum(exp_corr)

def grad_softmax(corr, alpha, h):
    """Compute gradient of softmax with respect to h"""
    weights = softmax_weights(corr, alpha)
    # Compute (w * h) and (w_rev * h)
    conv1 = np.convolve(weights, h, mode='full')[:len(h)]
    conv2 = np.convolve(weights[::-1], h, mode='full')[:len(h)]
    grad = 1.0 - conv1 - conv2
    return grad

def adam_step(h, m, v, grad, t, alpha, beta1=0.9, beta2=0.999, epsilon=1e-8):
    """Perform Adam step with projection"""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    h_new = h - alpha * m_hat / (np.sqrt(v_hat) + epsilon)
    return project_to_sum_box(h_new, TARGET_SUM), m, v

def sturmian_sequence(n, phi=(np.sqrt(5)-1)/2):
    """Generate balanced Sturmian word sequence"""
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1 if ((i * phi) % 1) < 0.5 else 0
    return h

def init_h_sturmian():
    """Initialize with Sturmian sequence"""
    h = sturmian_sequence(n_points)
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def init_h_random():
    """Initialize with random uniform distribution"""
    h = np.random.rand(n_points)
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def init_h_perturbed_sturmian():
    """Initialize with perturbed Sturmian sequence"""
    h = sturmian_sequence(n_points) + np.random.normal(0, 0.05, n_points)
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points // 2
    best_c5 = float('inf')
    best_h = None
    
    # Initial solutions
    init_solutions = [
        init_h_sturmian(),
        init_h_random(),
        init_h_perturbed_sturmian()
    ]
    
    alpha = 4.0
    max_iter = 8000
    beta1, beta2, epsilon = 0.9, 0.999, 1e-8
    learning_rate = 0.12
    
    # Run Adam optimization for different initializations
    # Allocate 45% of time to this stage
    t1 = start_time + 0.45 * budget_s
    
    for init_h in init_solutions:
        if time.time() >= t1:
            break
        h = init_h.copy()
        m = np.zeros(n_points)
        v = np.zeros(n_points)
        t = 1
        
        for iter_num in range(max_iter):
            if time.time() >= t1:
                break
                
            corr = compute_corr_fft(h)
            grad = grad_softmax(corr, alpha, h)
            
            h, m, v = adam_step(h, m, v, grad, t, learning_rate, beta1, beta2, epsilon)
            
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            t += 1
            
            # Increase alpha to focus on worst lag
            if t % 200 == 0:
                alpha = min(alpha * 1.5, 200.0)
    
    # Binary rounding and swap hill-climb
    # Allocate 35% of time to this stage
    if best_h is not None:
        t2 = start_time + 0.8 * budget_s
        
        # Round to binary
        median_val = np.median(best_h)
        binary_h = (best_h >= median_val).astype(float)
        
        # Adjust to correct sum
        ones_count = np.sum(binary_h)
        if ones_count > TARGET_SUM:
            ones_indices = np.where(binary_h == 1)[0]
            flip_count = int(ones_count - TARGET_SUM)
            flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
            binary_h[flip_indices] = 0
        elif ones_count < TARGET_SUM:
            zeros_indices = np.where(binary_h == 0)[0]
            flip_count = int(TARGET_SUM - ones_count)
            flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
            binary_h[flip_indices] = 1
        
        binary_h = project_to_sum_box(binary_h, TARGET_SUM)
        best_binary_c5 = evaluate_c5(binary_h)
        
        # Store the current overlap counts
        corr_full = compute_corr_fft(binary_h)
        current_overlaps = corr_full[:n_points]  # Only considering non-negative lags
        
        # Swap hill-climb 
        current_binary_h = binary_h.copy()
        current_c5 = best_binary_c5
        max_swaps = 5 * n_points
        swap_count = 0
        
        while time.time() < t2 and swap_count < max_swaps:
            # Pick two random indices
            i = np.random.randint(n_points)
            j = np.random.randint(n_points)
            
            if current_binary_h[i] == current_binary_h[j]:
                continue
            
            # Simulate the swap
            # Update overlaps efficiently by calculating the change for affected lags
            old_overlaps = current_overlaps.copy()
            
            # Calculate change in overlaps
            current_overlaps = np.zeros(2 * n_points - 1)
            for ell in range(-n_points + 1, n_points):
                if ell >= 0:
                    current_overlaps[ell + n_points - 1] = np.sum(
                        current_binary_h[:n_points-ell] * current_binary_h[ell:]
                    )
                else:
                    current_overlaps[ell + n_points - 1] = np.sum(
                        current_binary_h[-ell:] * current_binary_h[:n_points+ell]
                    )
                    
            # Actually implement the swap
            temp_h = current_binary_h.copy()
            temp_h[i], temp_h[j] = temp_h[j], temp_h[i]
            
            # Recalculate full correlation without FFT for small binary case
            new_corr = compute_corr_fft(temp_h)
            new_c5 = c5_from_corr(new_corr, n_points)
            
            if new_c5 < current_c5:
                current_binary_h = temp_h.copy()
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_binary_h.copy()
            
            swap_count += 1
            
        # Additional random binary restarts
        # Allocate 15% of time to this stage
        t3 = start_time + 0.95 * budget_s
        for _ in range(50):
            if time.time() >= t3:
                break
            
            # Create a new random binary sequence with same sum
            rand_h = np.random.rand(n_points)
            rand_h = project_to_sum_box(rand_h, TARGET_SUM)
            # Round to binary
            median_val = np.median(rand_h)
            rand_binary = (rand_h >= median_val).astype(float)
            # Fix sum
            ones_count = np.sum(rand_binary)
            if ones_count > TARGET_SUM:
                ones_indices = np.where(rand_binary == 1)[0]
                flip_count = int(ones_count - TARGET_SUM)
                flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
                rand_binary[flip_indices] = 0
            elif ones_count < TARGET_SUM:
                zeros_indices = np.where(rand_binary == 0)[0]
                flip_count = int(TARGET_SUM - ones_count)
                flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
                rand_binary[flip_indices] = 1
            rand_binary = project_to_sum_box(rand_binary, TARGET_SUM)
            trial_c5 = evaluate_c5(rand_binary)
            if trial_c5 < best_c5:
                best_c5 = trial_c5
                best_h = rand_binary.copy()
    
    if best_h is None:
        best_h = init_h_sturmian()
        
    return (best_h.tolist(), float(best_c5), n_points)
