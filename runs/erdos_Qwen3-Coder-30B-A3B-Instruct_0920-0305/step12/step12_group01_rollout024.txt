def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr) * (2.0 / len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_with_local_search(h0, target_sum, max_iter=1000):
    """Use a local search approach with adaptive steps"""
    h = h0.copy()
    current_c5 = evaluate_c5(h)
    
    # Gradient-free optimization with adaptive local search
    for iteration in range(max_iter):
        # Try small random perturbations
        perturbation_magnitude = 0.01 / (1 + iteration / 100)
        
        # Create multiple candidates
        candidates = []
        for _ in range(10):
            h_trial = h + np.random.normal(0, perturbation_magnitude, len(h))
            h_trial = project_to_sum_box(h_trial, target_sum)
