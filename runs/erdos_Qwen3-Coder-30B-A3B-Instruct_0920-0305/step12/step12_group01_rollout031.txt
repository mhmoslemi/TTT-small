import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import time
import random

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Use a hybrid approach with multiple strategies
    best_h = None
    best_c5 = float('inf')
    
    # Strategy 1: Start with initial construction and refine
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_to_sum_box(h0, TARGET_SUM)
        current_c5 = evaluate_c5(h0)
        if current_c5 < best_c5:
            best_h = h0.copy()
            best_c5 = current_c5
    
    # Strategy 2: Simulated Annealing with periodic cooling
    def simulated_annealing():
        nonlocal best_h, best_c5
        h_current = project_to_sum_box(np.random.rand(n_points), TARGET_SUM)
        c5_current = evaluate_c5(h_current)
        
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            
        # Cooling schedule
        temperature = 0.5
        min_temp = 1e-6
        alpha = 0.999
        max_iter = 100000
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Generate neighbor solution
            h_new = h_current.copy()
            idx = np.random.randint(0, n_points)
            h_new[idx] = np.random.uniform(0, 1)
            
            # Project to constraints
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            c5_new = evaluate_c5(h_new)
            
            # Accept or reject
            if c5_new < c5_current or np.random.random() < np.exp(-(c5_new - c5_current) / temperature):
                h_current = h_new
                c5_current = c5_new
                
                if c5_current < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_current
            
            # Cool down
            temperature *= alpha
            if temperature < min_temp:
                temperature = min_temp
        
        return best_h, best_c5
    
    # Strategy 3: Pattern-based optimization with fewer parameters
    def pattern_optimization():
        nonlocal best_h, best_c5
        # Try different block configurations
        for _ in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Create a 5-block pattern: [0, a], [a, b], [b, c], [c, d], [d, 2]
            # Assign random heights to each block
            heights = np.random.rand(5)
            lengths = np.random.rand(5)
            lengths = lengths / np.sum(lengths) * 2
            
            # Ensure lengths sum to 2
            h = np.zeros(n_points)
            cum_length = 0
            for i in range(5):
                start_idx = int(cum_length / 2 * n_points)
                end_idx = int((cum_length + lengths[i]) / 2 * n_points)
                if end_idx > n_points:
                    end_idx = n_points
                if start_idx >= n_points:
                    break
                h[start_idx:end_idx] = heights[i]
                cum_length += lengths[i]
            
            # Normalize to have correct integral
            integral = np.sum(h) * dx
            if integral > 0:
                h = h * (TARGET_SUM / integral)
            
            # Ensure constraints
            h = np.clip(h, 0, 1)
            h = project_to_sum_box(h, TARGET_SUM)
            
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    
    # Strategy 4: Differential Evolution with bounds
    def differential_evolution_opt():
        nonlocal best_h, best_c5
        
        # Create bounds for each variable
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Objective function for differential evolution
        def objective(x):
            # Project to constraints
            x_projected = project_to_sum_box(x, TARGET_SUM)
            return evaluate_c5(x_projected)
        
        try:
            result = differential_evolution(
                objective, 
                bounds, 
                maxiter=1000, 
                popsize=15,
                seed=seed,
                strategy='best1bin'
            )
            if result.success:
                h_final = project_to_sum_box(result.x, TARGET_SUM)
                c5_final = evaluate_c5(h_final)
                if c5_final < best_c5:
                    best_h = h_final.copy()
                    best_c5 = c5_final
        except:
            pass

    # Run strategies
    try:
        # Run simulated annealing
        sa_h, sa_c5 = simulated_annealing()
        if sa_c5 < best_c5:
            best_h = sa_h
            best_c5 = sa_c5
    except:
        pass
    
    try:
        # Run pattern optimization
        pattern_optimization()
    except:
        pass
        
    try:
        # Run differential evolution
        differential_evolution_opt()
    except:
        pass
    
    # Final refinement with local search
    if best_h is not None:
        try:
            # Use LBFGSB for final refinement
            from scipy.optimize import minimize
            
            def objective(h):
                return evaluate_c5(h)
            
            def constraint_eq(h):
                return np.sum(h) - TARGET_SUM
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': constraint_eq}
            
            res = minimize(
                objective, 
                best_h, 
                method='L-BFGS-B', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                final_h = project_to_sum_box(res.x, TARGET_SUM)
                final_c5 = evaluate_c5(final_h)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    if best_h is None:
        # Fallback to initial construction
        best_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.ones(n_points) * 0.5
        best_h = project_to_sum_box(best_h, TARGET_SUM)
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
