TARGET_SUM = None

phi = (math.sqrt(5) - 1) / 2.0  # Golden ratio

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort the vector
    sorted_x = np.sort(x)[::-1]
    # Find lambda using binary search
    a, b = -1.0, 1.0
    while True:
        m = (a + b) / 2
        projected = np.clip(sorted_x - m, 0, 1)
        if np.sum(projected) >= target_sum:
            a = m
        else:
            b = m
        if abs(a - b) < 1e-12:
            break
    lambda_val = (a + b) / 2
    result = np.clip(x - lambda_val, 0, 1)
    return result

def softmax_weights(corr, alpha):
    """Compute softmax weights"""
    exp_corr = np.exp(alpha * corr)
    return exp_corr / np.sum(exp_corr)

def grad_softmax(corr, alpha, h):
    """Compute gradient of softmax with respect to h"""
    weights = softmax_weights(corr, alpha)
    # Convolution with h (forward and backward)
    L = len(h) * 2 - 1
    L_fft = 1 << (L - 1).bit_length()
    W = np.fft.fft(weights, L_fft)
    H = np.fft.fft(h, L_fft)
    conv_forward = np.fft.ifft(W * H).real[:len(h)]
    conv_backward = np.fft.ifft(W[::-1] * H).real[:len(h)]
    grad = 1.0 - conv_forward - conv_backward
    return grad

def adam_step(h, m, v, grad, t, alpha, beta1, beta2, epsilon):
    """Adam optimizer step"""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    h_new = h - alpha * m_hat / (np.sqrt(v_hat) + epsilon)
    return h_new, m, v
