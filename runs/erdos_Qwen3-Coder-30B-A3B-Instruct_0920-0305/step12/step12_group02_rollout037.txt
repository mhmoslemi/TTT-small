import numpy as np
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Precompute distances from centre
    x = np.arange(n_points) * dx
    d = np.abs(x - 1.0)  # distance from centre
    
    best_c5 = np.inf
    best_h = None
    
    # Grid search over p values
    p_values = np.linspace(0.3, 5.0, 200)
    
    for p in p_values:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute alpha
        alpha_raw = (p + 1) / (2 * p)
        alpha = min(alpha_raw, 1.0)
        
        # Compute raw shape
        ratio = d / alpha
        f = np.where(ratio <= 1, 1 - np.power(ratio, p), 0.0)
        
        sum_f = f.sum()
        
        if sum_f == 0:
            continue
            
        # Compute feasible range for c
        c_max1 = target_sum / sum_f
        
        # Constraint from b + c <= 1 where b = (target_sum - c * sum_f) / n_points
        # (target_sum - c * sum_f) / n_points + c <= 1
        # target_sum/n_points + c*(1 - sum_f/n_points) <= 1
        # c*(1 - sum_f/n_points) <= 1 - target_sum/n_points = 0.5
        denom = 1 - sum_f / n_points
        if denom > 0:
            c_max2 = 0.5 / denom
        else:
            c_max2 = np.inf
            
        c_max = min(c_max1, c_max2)
        if c_max <= 0:
            continue
            
        # Golden section search on c to minimize C5
        def obj(c):
            if c < 0 or c > c_max:
                return np.inf
            b = (target_sum - c * sum_f) / n_points
            h = b + c * f
            h = np.clip(h, 0, 1)
            return compute_c5(h)
        
        # Simple ternary search for optimization
        lo = 0.0
        hi = c_max
        for _ in range(30):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = obj(m1)
            f2 = obj(m2)
            if f1 < f2:
                hi = m2
            else:
                lo = m1
                
        c_opt = (lo + hi) / 2.0
        try:
            c5_val = obj(c_opt)
            if c5_val < best_c5:
                best_c5 = c5_val
                b = (target_sum - c_opt * sum_f) / n_points
                best_h = b + c_opt * f
                best_h = np.clip(best_h, 0, 1)
                # Apply projection to ensure exact sum constraint
                best_h = project_to_sum_box(best_h, target_sum)
                best_c5 = compute_c5(best_h)
        except:
            continue
    
    # Final local refinement with L-BFGS-B
    if best_h is not None and time.time() - start_time < budget_s - 5:
        def obj_proj(x):
            h = project_to_sum_box(x, TARGET_SUM)
            return compute_c5(h)
            
        try:
            x0 = best_h.copy()
            res = minimize(
                obj_proj, 
                x0, 
                method='L-BFGS-B', 
                bounds=[(0,1)]*n_points,
                options={'maxiter': 300, 'ftol': 1e-12}
            )
            if res.success:
                h_opt = project_to_sum_box(res.x, TARGET_SUM)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
