import numpy as np
import time

def compute_correlation(h):
    """Compute full correlation of h with (1-h)"""
    return np.correlate(h, 1 - h, mode='full')

def evaluate_c5(h):
    """Evaluate C5 from the correlation"""
    corr = compute_correlation(h)
    return np.max(corr) * (2.0 / len(h))

def get_beatty_pattern(n):
    """Generate Beatty sequence pattern"""
    phi = (1 + np.sqrt(5)) / 2
    indices = np.floor(np.arange(n) * phi) % n
    indices = np.sort(indices.astype(int))
    h = np.zeros(n)
    h[indices[:n//2]] = 1
    return h

def update_correlation_delta(C_raw, h, i, j):
    """Update correlation delta for swapping h[i] and h[j]"""
    delta = np.zeros_like(C_raw)
    
    # h[i] changes from 0 to 1, h[j] changes from 1 to 0
    # For all k such that i+k is valid and j-k is valid
    for k in range(-i, min(n - i, n)):
        # When i appears as first argument in h[i]*h[i+k] 
        delta[k + n - 1] -= (1 - h[i + k])
    for k in range(max(0, -j), n - j):
        # When j appears as second argument in h[j-k]*h[j] 
        delta[-k + n - 1] -= h[j - k]
    
    return delta

def run(seed=42, budget_s=600, **kwargs):
    global n
    n = 120
    dx = 2.0 / n
    
    np.random.seed(seed)
    start_time = time.time()
    
    # Generate initial patterns
    initial_patterns = get_initial_patterns(n)
    
    # Find the best initial pattern
    best_h = None
    best_corr = None
    best_val = float('inf')
    
    for h in initial_patterns:
        corr = compute_correlation(h)
        max_val = np.max(corr) * dx
        if max_val < best_val:
            best_val = max_val
            best_h = h.copy()
            best_corr = corr.copy()
    
    # Simulated annealing
    T = 0.5
    alpha = 0.995
    N_swap = n
    swap_count = 0
    
    current_h = best_h.copy()
    current_corr = best_corr.copy()
    current_max = np.max(current_corr)
    
    best_h = current_h.copy()
    best_corr = current_corr.copy()
    best_max = current_max
    
    while time.time() - start_time < budget_s - 10:
        # Pick two indices: one 0 and one 1
        zeros = np.where(current_h == 0)[0]
        ones = np.where(current_h == 1)[0]
        
        if len(zeros) == 0 or len(ones) == 0:
            break
            
        i = np.random.choice(zeros)
        j = np.random.choice(ones)
        
        # Compute the change in correlation
        delta = update_correlation_delta(current_corr, current_h, i, j)
        new_corr = current_corr + delta
        new_max = np.max(new_corr)
        
        # Accept or reject the move
        if new_max < current_max:
            # Always accept improvements
            current_h[i] = 1
            current_h[j] = 0
            current_corr = new_corr
            current_max = new_max
            if new_max < best_max:
                best_h = current_h.copy()
                best_corr = current_corr.copy()
                best_max = new_max
        else:
            # Accept with probability
            prob = np.exp(-(new_max - current_max) / T)
            if np.random.rand() < prob:
                current_h[i] = 1
                current_h[j] = 0
                current_corr = new_corr
                current_max = new_max
        
        swap_count += 1
        if swap_count % N_swap == 0:
            T *= alpha
            if T < 1e-5:
                break
    
    # Hill climbing to refine the solution
    improved = True
    while improved and time.time() - start_time < budget_s - 5:
        improved = False
        for i in np.where(best_h == 0)[0]:
            for j in np.where(best_h == 1)[0]:
                delta = update_correlation_delta(best_corr, best_h, i, j)
                new_corr = best_corr + delta
                new_max = np.max(new_corr)
                if new_max < best_max:
                    best_h[i] = 1
                    best_h[j] = 0
                    best_corr = new_corr
                    best_max = new_max
                    improved = True
                    break
            if improved:
                break
    
    # Final evaluation
    final_corr = compute_correlation(best_h)
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), float(final_c5), n)
