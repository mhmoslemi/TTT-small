import time

# Global configuration
n_points = 128
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def generate_toeplitz_matrices(n):
    """Generate Toeplitz autocorrelation matrices A_ell for all lags ell = 1..n-1"""
    A_list = [None] * (n - 1) # store them as sparse matrices
    for ell in range(1, n):
        # Create a sparse diagonal matrix with 1 at position (i, i+ell)
        # and (i+ell, i) for i=0..n-ell-1, with proper indexing
        row = np.arange(n - ell)
        col = np.arange(ell, n)
        data = np.ones(n - ell)
        A = csr_matrix((data, (row, col)), shape=(n, n))
        A += A.T  # make symmetric
        A_list[ell-1] = A
    return A_list

def sdp_solve_with_rounding(A_list, n, N, t_sdp=240):
    """Solve the SDP relaxation and return best rounded solution"""
    # Create variable
    X = cp.Variable((n, n), symmetric=True)
    m = cp.Variable()
    
    # Constraints
    constraints = [
        cp.trace(X) == N,
        X >> 0,  # PSD
        X >= 0,
        X <= 1
    ]
    
    # Add lagwise constraints
    for A in A_list:
        constraints.append(cp.trace(A @ X) >= m)
    
    # Objective
    objective = cp.Maximize(m)
    
    # Create problem
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-5, verbose=False,
                   time_limit=t_sdp)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            X_val = X.value
        else:
            # If not optimal, return best primal iterate
            X_val = prob.solver_stats.extra_stats['x']
    except Exception:
        # Fallback to a coarse solution if full solve fails
        return None, None

    # Randomized rounding
    best_h = None
    best_c5 = float('inf')
    tolerance = 1e-8
    
    # Perform 5 rounds of rounding
    for _ in range(5):
        # Eigen-decomposition
        try:
            evals, evecs = eigh(X_val)
            # Keep only positive eigenvalues
            pos_evals = evals[evals > tolerance]
            pos_evecs = evecs[:, evals > tolerance]
            
            if len(pos_evals) == 0:
                continue
                
            # Form V
            V = pos_evecs @ np.diag(np.sqrt(pos_evals))
            r = V.shape[1]
            
            # Sample random Gaussian vector
            g = np.random.randn(r)
            scores = V @ g
            
            # Threshold to get binary vector
            indices = np.argsort(-scores)  # descending
            h = np.zeros(n)
            h[indices[:N]] = 1.0
            
            # Evaluate C5
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        except Exception:
            continue
    
    return best_h, best_c5

def compute_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    # Take max of all lags 
    max_overlap = np.max(corr)
    return dx * max_overlap

def local_hill_climb(h, A_list, max_iter=10000):
    """Perform local swaps to improve the binary vector"""
    n = len(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    # Compute initial S_ell values quickly via FFT
    h_fft = np.fft.fft(h)
    ones_fft = np.fft.fft(1 - h)
    corr = np.fft.ifft(h_fft * ones_fft.conj()).real
    S = corr[:n]  # First n values (circular correlation)
    min_S = np.min(S[1:]) if len(S) > 1 else np.inf
    
    current_c5 = dx * (n/2 - min_S)
    best_h = h.copy()
    best_c5 = current_c5
    
    iter_count = 0
    for _ in range(max_iter):
        if iter_count >= max_iter:
            break
            
        # Select a random 1 and 0 to swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Update S_ell
        # For lag k where we are computing S_k = sum_i h_i h_{i+k}, 
        # we flip h_i and h_j
        delta_S = np.zeros(n)
        for k in range(1, n):
            # If h_i and h_j both have a neighbor at k, delta is 0
            # If h_i only has one, we remove it (delta_h_i = -1)
            # If h_j only has one, we add it (delta_h_j = +1)
            # We do this carefully for fast permutation
            contrib = 0
            
            # Check neighbors around i:
            i_pos = i
            j_pos = j
            # Consider direct neighbors in both directions but use
            # real part of circulant convolution which is efficient
            if (i_pos - k) >= 0:
                contrib -= h[(i_pos - k) % n]  # remove i's neighbor
            if (i_pos + k) < n:
                contrib -= h[(i_pos + k) % n]  # remove i's neighbor
            if (j_pos - k) >= 0:
                contrib += h[(j_pos - k) % n]  # add j's neighbor
            if (j_pos + k) < n:
                contrib += h[(j_pos + k) % n]  # add j's neighbor
                
            delta_S[k] = contrib
            
        new_S = S + delta_S
        new_min_S = np.min(new_S[1:])
        new_c5 = dx * (n/2 - new_min_S)
        
        if new_min_S > min_S:
            # Accept swap
            h[i], h[j] = h[j], h[i]
            S = new_S
            min_S = new_min_S
            current_c5 = new_c5
            # Update index lists
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                
        iter_count += 1
        
    return best_h, best_c5
