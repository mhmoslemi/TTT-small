def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT for efficiency
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    """Create a binary pattern that minimizes overlap"""
    # Try to create a pattern with low autocorrelation

    for i in range(n_points):
        # Create a binary pattern with regions of 0 and 1
        # The periodic nature helps reduce overlap
        if (i // 15) % 2 == 0:
            h[i] = 1.0
        else:
            h[i] = 0.0
            
    # Apply smoothing to reduce sharp transitions while maintaining sum
    h = project_to_sum_box(h, n_points / 2)
    return h

    h = project_to_sum_box(h, n_points / 2)
