import numpy as np
import time

def compute_correlation(h):
    """Compute full correlation of h with (1-h) using np.correlate"""
    return np.correlate(h, 1 - h, mode='full')

def evaluate_c5(h, dx=2.0/120):
    """Compute C5 from h values"""
    corr = compute_correlation(h)
    return dx * np.max(corr)

def generate_beatty_pattern(n):
    """Generate binary pattern using Beatty sequence with golden ratio"""
    phi = (1 + np.sqrt(5)) / 2
    indices = np.floor(np.arange(n) * phi) % n
    indices = np.sort(indices.astype(int))
    h = np.zeros(n)
    h[indices[:n//2]] = 1
    return h

def generate_random_pattern(n):
    """Generate random binary pattern with exactly n/2 ones"""
    h = np.zeros(n)
    ones_indices = np.random.choice(n, n//2, replace=False)
    h[ones_indices] = 1
    return h

def update_correlation_incremental(C_raw, h, i, j):
    """Update correlation vector C_raw by swapping h[i] and h[j]"""
    delta = np.zeros_like(C_raw)
    
    # For each shift k where i+k is valid
    for k in range(1, len(h)):
        if i + k < len(h):
            delta[k] -= (1 - h[i + k])
        if i - k >= 0:
            delta[-k] -= h[i - k]
    
    # For j
    for k in range(1, len(h)):
        if j + k < len(h):
            delta[k] += (1 - h[j + k])
        if j - k >= 0:
            delta[-k] += h[j - k]
    
    return C_raw + delta
