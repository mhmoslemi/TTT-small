import numpy as np
from scipy.signal import correlate
import cvxpy as cp

def balanced_sturmian(n):
    """Generate a balanced Sturmian word with n bits and exactly n//2 ones."""
    phi = (np.sqrt(5) - 1) / 2
    x = np.array([int(((i * phi) % 1) < 0.5) for i in range(n)])
    # Adjust to have exactly n//2 ones
    ones_count = np.sum(x)
    if ones_count > n // 2:
        # Need to flip some 1s to 0s
        ones_indices = np.where(x == 1)[0]
        flip_count = ones_count - n // 2
        flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
        x[flip_indices] = 0
    elif ones_count < n // 2:
        # Need to flip some 0s to 1s
        zeros_indices = np.where(x == 0)[0]
        flip_count = n // 2 - ones_count
        flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
        x[flip_indices] = 1
    return x

def compute_autocorr_fft(h):
    """Compute autocorrelations using FFT."""
    n = len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def swap_update(S, i, j, n):
    """Compute delta in autocorrelations when swapping bits at positions i and j."""
    # S[i] = sum h_k * h_{k+i} over valid k
    delta = np.zeros(n)
    # For each lag ℓ, compute change in S_ℓ
    for l in range(1, n):
        # When swapping h[i] and h[j], impact is:
        # ΔS_ℓ depends on how h[i], h[i+ℓ] and h[j], h[j+ℓ] change
        # Contribute if the conditions (i or i+ℓ) in ones or (j or j+ℓ) in ones
        # Simplified update:
        contrib_i = 0
        contrib_j = 0
        # If both are involved in lag l
        if i != l and j != l:  # i and j not directly contributing
            if i + l < n:
                contrib_i += h[i] - h[j]  # contribution to S_l
                contrib_j += h[j] - h[i]
            if j + l < n:
                contrib_i -= h[i] - h[j]
                contrib_j -= h[j] - h[i]
        else:
            contrib_i = 0
            contrib_j = 0
        # This needs to account for the full logic properly
        # But for simplicity in this optimized version:
        # Update for each lag l:
        for l in range(1, n):
            if i + l < n or i - l >= 0 or j + l < n or j - l >= 0:
                delta[l] += (h[i] - h[j]) * (h[i + l] - h[j + l] if i + l < n and j + l < n else 0)
                delta[l] += (h[j] - h[i]) * (h[j + l] - h[i + l] if i + l < n and j + l < n else 0)
    return delta

def swap_update_simple(S, i, j, h):
    """Efficiently compute change in autocorrelations when swapping h[i] with h[j]."""
    n = len(h)
    delta = np.zeros(n)
    # Compute impact of swapping h[i] with h[j]
    # For all lags ℓ from 1 to n-1:
    for l in range(1, n):
        if i+l < n and j+l < n:
            delta[l] = (h[i] - h[j]) * (h[i+l] - h[j+l]) + (h[j] - h[i]) * (h[j+l] - h[i+l])
        elif i+l < n and j+l >= n:
            delta[l] = (h[i] - h[j]) * (h[i+l] - 0) + (h[j] - h[i]) * (0 - h[i+l])
        elif i+l >= n and j+l < n:
            delta[l] = (h[i] - h[j]) * (0 - h[j+l]) + (h[j] - h[i]) * (h[j+l] - 0)
        else:  # both overflow
            delta[l] = 0

        # Add row-specific updates
        # Just direct handling for loops is safer:
        # For each lag ℓ:
        # S_ℓ = Σ_k x_k * x_{k+ℓ}
        # When swapping h[i], h[j]:
        # ∂S_ℓ / ∂x_i = x_{i+ℓ}
        # ∂S_ℓ / ∂x_j = x_{j+ℓ}
        # In a swap with old values x[i], x[j] we get new values 1-x[i], 1-x[j]
        # ΔS_ℓ = x_{i+ℓ} * (1-x[j]) + x_j * (1-x[i]) - (x_{i+ℓ} * x_j + x_j * x_i) = x_{i+ℓ}*(1-x[j] - x_j) + x_j*(1 - x[i])

    # More correctly capture the change
    # Original: S_ℓ = Σ_k x_k x_{k+ℓ}
    # When swapping x_i with x_j, contributions to difference:
    delta = np.zeros(n)
    for l in range(1, n):
        contrib_i = 0
        contrib_j = 0
        # Contribution from x_i to lag l (i.e. x_i * x_{i+l})
        if i+l < n:
            contrib_i += h[i]*h[i+l]
            contrib_j += h[j]*h[i+l]  # Different from original due to trade
        # Contribution from x_j to lag l (i.e. x_j * x_{j+l})
        if j+l < n:
            contrib_i += h[i]*h[j+l]
            contrib_j += h[j]*h[j+l]
        delta[l] = contrib_i - contrib_j  # Note the logic of the swap much more carefully

    return delta

def swap_delta(S, x, i, j):
    """Swap bits at positions i and j, compute the change in autocorrelation scores."""
    # Current values
    ni = x[i]
    nj = x[j]
    # Split handling by lags for two indices
    delta = np.zeros(len(S))
    n = len(x)
    # For each lag l from 1 to n-1
    for l in range(1, n):
        # Index a = i and index b = j
        # Change in S_l when swapping x_i and x_j:
        # contribution to S_l from pair (x_i, x_{i+l})
        # contribution to S_l from pair (x_j, x_{j+l})
        # When swapping, instead have (x_i, x_{j+l}) and (x_j, x_{i+l})
        a = i
        b = j
        # delta will include contributions where this affects pair
        if i+l < n and j+l < n:
            # original contribution to S_l from indices a, a+l and b, b+l
            old_a = x[a] * x[a+l]
            old_b = x[b] * x[b+l]
            # new contribution to S_l from indices a, b+l and b, a+l
            new_a = x[a] * x[b+l]
            new_b = x[b] * x[a+l]
            delta[l] = (new_a + new_b) - (old_a + old_b)
        elif i+l < n and j+l >= n:
            old_a = x[a] * x[a+l]
            new_a = x[a] * 0  # since b+l >= n, x[b+l] = 0 (beyond bounds)
            # Original not only has a sigma but also original b*b+l we ignore if b-l is invalid
            delta[l] = new_a - old_a
        elif i+l >= n and j+l < n:
            # Same logic
            old_b = x[b] * x[b+l]
            new_b = 0 * x[a+l]
            delta[l] = new_b - old_b
        else:
            # Both are out of bound
            delta[l] = 0
    return delta

def greedy_hill_climb(x, S, n_points, dx, budget_s, t0):
    """Run a greedy hill climb to improve the solution."""
    # Split x into ones and zeros
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return x, max(S) * dx
    
    m = min(S[1:])
    best_c5 = m * dx
    
    total_swaps = 0
    patience = 0
    max_patience = 5 * n_points
    prev_m = m
    
    while time.time() < t0 + budget_s and patience < max_patience:
        # Pick a random pair (i in ones, j in zeros)
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # Compute delta in S
        delta = swap_delta(S, x, i, j)
        new_S = S + delta
        new_m = min(new_S[1:])  # min of all lags except l=0
        
        if new_m > m:
            # Accept the swap
            x[i], x[j] = x[j], x[i]
            S = new_S
            m = new_m
            best_c5 = m * dx
            ones = np.where(x == 1)[0]
            zeros = np.where(x == 0)[0]
            patience = 0
        else:
            patience += 1
            
        total_swaps += 1
        
    return x, best_c5
