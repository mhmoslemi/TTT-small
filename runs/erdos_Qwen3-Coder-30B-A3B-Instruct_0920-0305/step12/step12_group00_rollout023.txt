```python
import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
from math import sqrt

# Global constants
n_points = 128
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def balanced_sturmian(n):
    """Generate a balanced Sturmian sequence which typically has good autocorrelation properties."""
    phi = (sqrt(5) - 1) / 2
    x = np.array([int(((i * phi) % 1) < 0.5) for i in range(n)])
    
    # Ensure correct sum
    ones_count = np.sum(x)
    if ones_count > TARGET_SUM:
        ones_indices = np.where(x == 1)[0]
        flip_count = int(ones_count - TARGET_SUM)
        flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
        x[flip_indices] = 0
    elif ones_count < TARGET_SUM:
        zeros_indices = np.where(x == 0)[0]
        flip_count = int(TARGET_SUM - ones_count)
        flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
        x[flip_indices] = 1
    
    return x

def compute_autocorr_fft(h):
    """Compute autocorrelations using FFT."""
    n = len(h)
    auto_corr = correlate(h, h, mode="full")
    # Take only the relevant part (excluding the zero-lag correlation)
    return auto_corr[n-1:]

def swap_delta(S, i, j, n):
    """Compute change in autocorrelation if we swap bits at positions i and j."""
    delta = np.zeros(2 * n - 1)
    for l in range(1, n):
        if 0 <= i - l < n and 0 <= j - l < n:
            delta[l] += (h[i] - h[j]) * (h[i-l] - h[j-l]) if h[i-l] != h[j-l] else 0
        if 0 <= i + l < n and 0 <= j + l < n:
            delta[l] += (h[i] - h[j]) * (h[i+l] - h[j+l]) if h[i+l] != h[j+l] else 0
        if l < n and i != j:
            delta[-l] = delta[l]
    return delta

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage 1: Exact ILP approach
    best_c5 = float('inf')
    best_h = None
    best_m = -1
    
    # Build the model
    x = cp.Variable(n_points, boolean=True)
    m = cp.Variable()
    
    # Constraints for sum
    constraints = [cp.sum(x) == TARGET_SUM]
    
    # Symmetry breaking
    constraints.append(x[0] == 1)
    for i in range(TARGET_SUM):
        constraints.append(x[i] >= x[i+1])
    
    # Create y variables and constraints for products
    y = {}
    S = []
    for l in range(1, n_points):
        y[l] = cp.Variable((n_points - l), boolean=True)
        constraints.append(y[l] <= x[:n_points - l])
        constraints.append(y[l] <= x[l:])
        constraints.append(y[l] >= x[:n_points - l] + x[l:] - 1)
        
        s = cp.sum(y[l])
        constraints.append(s >= m)
        S.append(s)
    
    # Objective
    objective = cp.Maximize(m)
    problem = cp.Problem(objective, constraints)
    
    # Solve
    time_left = budget_s * 0.45
    try:
        problem.solve(solver=cp.CBC, max_seconds=time_left)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_val = np.round(x.value).astype(int)
            if np.sum(x_val) == TARGET_SUM:
                # Compute the actual C5 bound
                auto_corr = compute_autocorr_fft(x_val)
                m_val = np.min(auto_corr[:n_points-1])
                c5_val = dx * (TARGET_SUM - m_val)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = x_val
                    best_m = m_val
    except:
        pass
    
    # Stage 2: Local search refinement
    if best_h is None:
        # Fallback: start with Sturmian word
        current_h = balanced_sturmian(n_points)
    else:
        current_h = best_h.copy()
    
    # Compute initial autocorrelations
    auto_corr = compute_autocorr_fft(current_h)
    current_m = np.min(auto_corr[:n_points-1])
    
    # Refresh optimal value
    current_c5 = dx * (TARGET_SUM - current_m)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
        best_m = current_m

    # Optimization via greedy swap hill-climbing
    ones = np.where(current_h == 1)[0]
    zeros = np.where(current_h == 0)[0]
    
    # Calculate minimum k such that the correlation at lags 1,...,k is symmetric
    time_limit = start_time + budget_s * 0.35
    
    patience = 0
    T = 0.01 * current_c5
    swap_count = 0
    max_swaps = 5 * n_points

    while time.time() < time_limit and swap_count < max_swaps:
        # Pick random pair (i in ones, j in zeros)
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Compute delta in correlations
        delta = swap_delta(auto_corr, i, j, n_points)
        new_auto_corr = auto_corr + delta
        potential_m = np.min(new_auto_corr[:n_points-1])
        
        if potential_m > current_m:
            # Accept the improvement
            current_h[i] = 0
            current_h[j] = 1
            auto_corr = new_auto_corr
            current_m = potential_m
            ones = np.where(current_h == 1)[0]
            zeros = np.where(current_h == 0)[0]
            current_c5 = dx * (TARGET_SUM - current_m)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                best_m = current_m
            patience = 0
        else:
            # Possibly accept with simulated annealing probability
            if np.random.rand() < np.exp((potential_m - current_m) / T) and T > 1e-8:
                current_h[i] = 0
                current_h[j] = 1
                auto_corr = new_auto_corr
                current_m = potential_m
                ones = np.where(current_h == 1)[0]
                zeros = np.where(current_h == 0)[0]
                current_c5 = dx * (TARGET_SUM - current_m)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
                    best_m = current_m
                patience = 0
            else:
                patience += 1
                if patience > 5 * n_points:
                    # Reset temperature
                    T *= 0.99
                    patience = 0
        
        swap_count += 1
    
    # Stage 3: Fallback to random restarts if constraint is violated or we are still suboptimal
    if best_h is None or best_c5 > 0.38:
        for _ in range(5):
            try:
                # Random binary sequence
                h = np.zeros(n_points)
                indices = np.random.choice(n_points, TARGET_SUM, replace=False)
                h[indices] = 1
                
                # Get final reformulation
                auto_corr = compute_autocorr_fft(h)
                new_m = np.min(auto_corr[:n_points-1])
                new_c5 = dx * (TARGET_SUM - new_m)
                
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h
                    best_m = new_m
            except Exception:
                continue

    if best_h is None:
        # Return the best fallback solution
        best_h = balanced_sturmian(n_points)
        auto_corr = compute_autocorr_fft(best_h)
        best_m = np.min(auto_corr[:n_points-1])
        best_c5 = dx * (TARGET_SUM - best_m)
    
    return (best_h.tolist(), float(best_c5), n_points)
```
