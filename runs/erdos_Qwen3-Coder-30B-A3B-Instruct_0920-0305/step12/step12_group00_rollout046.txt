# Global constants
n_points = 128
dx = 2.0 / n_points
N = n_points // 2
J = np.ones((n_points, n_points))

def build_toeplitz_matrices(n):
    """Generate Toeplitz matrices A_ℓ for each lag ℓ = 1..n-1"""
    A = []
    for l in range(1, n):
        A_l = np.zeros((n, n))
        # Fill diagonals
        np.fill_diagonal(A_l[l:], 1)
        np.fill_diagonal(A_l[:, l:], 1)
        A.append(A_l)
    return A

def compute_autocorrelations(h, A_matrices):
    """Fast computation of all autocorrelations S_ℓ using FFT for each A_ℓ"""
    S = []
    for A in A_matrices:
        # For Toeplitz matrix A_ℓ, we compute h^T A_ℓ h
        # Since A_ℓ has ones on a specific diagonal only, we can do this efficiently.
        # This is equivalent to a specific convolution: h_i * (1 - h_{i+k})
        # So we use the standard way: compute A_ℓ @ h, then h^T @ (A_ℓ @ h)
        # which is simple matrix vector product: trace(A_ℓ @ x) = x^T A_ℓ x
        s = np.dot(h, A @ h)
        S.append(s)
    return np.array(S)

def sdpsolve(n, A_matrices, t_sdp):
    """Solve the SDP relaxation: max m s.t. trace(A_ℓ X) ≥ m, trace(JX)=N, X ≥ 0, 0 ≤ X ≤ 1"""
    X = cp.Variable((n,n), PSD=True)
    m = cp.Variable()
    
    # Constraints
    constraints = [
        cp.trace(J @ X) == N,
        X >= 0,
        X <= 1
    ]
    for A in A_matrices:
        constraints.append(cp.trace(A @ X) >= m)
        
    # Objective
    objective = cp.Maximize(m)
    
    # Solve with SCS
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-5, time_limit=t_sdp, verbose=False)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            return X.value, m.value
    except Exception:
        pass
    # Fallback to largest feasible iterate
    return None, None

def round_to_binary(X, num_rounds=5):
    """Random hyperplane rounding from the SDP solution"""
    # Eigen-decompose X
    try:
        evals, evecs = eigh(X)
        evals = evals[evals > 1e-8]
        evecs = evecs[:, evals > 1e-8]
        V = evecs @ np.diag(np.sqrt(evals))
    except Exception:
        # Fallback: use diagonal
        V = np.diag(np.sqrt(np.diag(X)))
        if V.ndim == 1:
            V = V.reshape(-1, 1)
    best_h = None
    best_c5 = np.inf
    
    for _ in range(num_rounds):
        g = np.random.normal(size=V.shape[1])
        s = V @ g
        h = (s >= np.partition(s, -N)[-N]).astype(int)
        if np.sum(h) != N:
            # Adjust to reach exactly N ones
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) > N:
                flip_indices = np.random.choice(ones, len(ones) - N, replace=False)
                h[flip_indices] = 0
            elif len(zeros) > N - len(ones):
                flip_indices = np.random.choice(zeros, N - len(ones), replace=False)
                h[flip_indices] = 1
        # Compute C5 with FFT
        h = np.array(h)
        corr = correlate(h, 1-h, mode='full')[n_points-1:]
        c5 = dx * np.max(corr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    return best_h, best_c5

def swap_hill_climb(h, A_matrices, max_iter=5000, t_lim=None):
    """Perform swap-based hill climb on binary vector to minimize C5"""
    start = time.time()
    h = h.copy()
    S = compute_autocorrelations(h, A_matrices)
    m = np.minimum.reduce(S[1:])  # min S_ℓ for ℓ >= 1

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    iter_count = 0
    while iter_count < max_iter and (t_lim is None or time.time() < t_lim):
        # Pick random indices
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Compute changes in S_ℓ for this swap
        delta = np.zeros(len(A_matrices))
        for l_idx, A in enumerate(A_matrices):
            # This is efficient for Toeplitz structure
            # Directly compute the dot products
            # For A_ℓ: update S_ℓ by changing h_i and h_j
            # Only relevant terms are at indices (i±l) and (j±l) if they exist
            k = l_idx + 1
            delta[l_idx] = - (h[i+k] + h[i-k]) + (h[j+k] + h[j-k])
        
        S_new = S + delta
        m_new = np.minimum.reduce(S_new[1:])
        
        if m_new > m:
            h[i] = 0
            h[j] = 1
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            S = S_new
            m = m_new
            iter_count = 0 # Reset counter after improvement
            
        iter_count += 1

    corr = correlate(h, 1-h, mode='full')[n_points-1:]
    c5 = dx * np.max(corr)
    return h, c5
