```python
import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
from math import floor

def balanced_sturmian(n):
    """Generate a balanced Sturmian word for given n"""
    phi = (np.sqrt(5) - 1) / 2
    x = np.array([int(((i * phi) % 1) < 0.5) for i in range(n)])
    # Adjust to have exactly n/2 ones
    ones_count = np.sum(x)
    if ones_count > n // 2:
        # Turn some 1s to 0s
        ones_indices = np.where(x == 1)[0]
        flip_indices = np.random.choice(ones_indices, int(ones_count - n // 2), replace=False)
        x[flip_indices] = 0
    elif ones_count < n // 2:
        # Turn some 0s to 1s
        zeros_indices = np.where(x == 0)[0]
        flip_indices = np.random.choice(zeros_indices, int(n // 2 - ones_count), replace=False)
        x[flip_indices] = 1
    return x

def compute_autocorr_fft(x):
    """Compute full autocorrelation using FFT"""
    x_rev = x[::-1]
    L = 2 * len(x) - 1
    corr = np.fft.ifft(np.fft.fft(x_rev, L) * np.fft.fft(x, L)).real
    return corr

def update_autocorr_delta(S, i, j, x, n):
    """Update autocorrelation values with a swap between positions i and j"""
    # Calculate change in autocorrelation for all lags
    delta_S = np.zeros(n)
    for l in range(1, n):
        if i - l >= 0:
            delta_S[l] += x[i-l] - x[j-l]
        if i + l < n:
            delta_S[l] += x[i+l] - x[j+l]
        if j - l >= 0:
            delta_S[l] -= x[i-l] - x[j-l]
        if j + l < n:
            delta_S[l] -= x[i+l] - x[j+l]
    return delta_S

def hill_climb_binary(x, n, N, dx):
    """Perform greedy hill-climb refinement on binary vector"""
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    
    # Compute initial autocorrelations
    S = np.real(compute_autocorr_fft(x))[:n]
    m = np.min(S[1:])  # minimum autocorrelation (excluding lag 0)
    best_m = m
    best_x = x.copy()
    best_c5 = dx * (N - m)
    
    iter_count = 0
    max_iter = 50000
    accept_count = 0
    T = 0.01 * best_c5  # Initial temperature
    
    while iter_count < max_iter:
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Compute change in autocorrelation when swapping x[i] and x[j]
        delta_S = update_autocorr_delta(S, i, j, x, n)
        new_S = S + delta_S
        new_m = np.min(new_S[1:])
        
        if new_m > m:
            # Accept improvement
            x[i], x[j] = x[j], x[i]
            S = new_S
            m = new_m
            accept_count += 1
            if m > best_m:
                best_m = m
                best_x = x.copy()
                best_c5 = dx * (N - m)
        else:
            # Accept worse solution with probability
            if T > 0 and np.random.rand() < np.exp((new_m - m) / T):
                x[i], x[j] = x[j], x[i]
                S = new_S
                m = new_m
                accept_count += 1
                
        iter_count += 1
        if accept_count >= 1000:
            T *= 0.99
            accept_count = 0
            
        # Occasionally check for improved solution
        if iter_count % 1000 == 0 and m > best_m:
            best_m = m
            best_x = x.copy()
            best_c5 = dx * (N - m)
            
        if iter_count % 10000 == 0:
            T = max(1e-6, T * 0.95)
            
    return best_x, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n = 128
    dx = 2.0 / n
    N = n // 2
    
    best_c5 = float('inf')
    best_h = None
    best_m = 0
    
    # Stage 1: Try exact ILP
    try:
        # Build variable
        x = cp.Variable(n, boolean=True)
        
        # Add mass constraint
        mass_constraint = cp.sum(x) == N
        
        # Add symmetry-breaking constraints 
        symmetry_constraints = [
            x[0] == 1,  # Fix first value to 1
        ]
        
        # Add lexicographic ordering constraint on first N elements
        if N > 1:
            for i in range(N - 1):
                symmetry_constraints.append(x[i] >= x[i+1])
        
        # Create auxiliary variables for products
        y = {}
        S = {}
        
        # Add McCormick constraints and compute sums
        constraints = [mass_constraint] + symmetry_constraints
        
        for l in range(1, n):
            for i in range(n - l):
                y[i, l] = cp.Variable(boolean=True)
                # McCormick constraints
                constraints.append(y[i, l] <= x[i])
                constraints.append(y[i, l] <= x[i + l])
                constraints.append(y[i, l] >= x[i] + x[i + l] - 1)
                
                if l not in S:
                    S[l] = 0
                S[l] += y[i, l]
        
        # Add epigraph constraint
        m = cp.Variable()
        for l in range(1, n):
            constraints.append(S[l] >= m)
            
        # Create the problem
        objective = cp.Maximize(m)
        prob = cp.Problem(objective, constraints)
        
        # Solve with time limit
        time_remaining = budget_s * 0.45
        prob.solve(solver=cp.CBC, max_seconds=time_remaining)
        
        # Check if we have a solution
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_sol = np.round(x.value).astype(int)
            # Re-evaluate using exact correlation
            full_corr = correlate(x_sol, x_sol, mode='full')[:2*n-1]
            # Remove lag 0 and compute C5
            m_val = np.min(full_corr[n:])
            c5_val = dx * (N - m_val)
            
            # If we get a better solution, save it
            if c5_val < best_c5:
                best_h = x_sol
                best_c5 = c5_val
                best_m = m_val
                
    except:
        pass
        
    # If ILP failed or did not finish, use fallback approach
    if best_h is None or best_c5 >= 0.4:
        # Try sturmian and refine
        sturmian_x = balanced_sturmian(n)
        refined_x, refined_c5 = hill_climb_binary(sturmian_x, n, N, dx)
        
        if refined_c5 < best_c5:
            best_h = refined_x
            best_c5 = refined_c5
            # Re-compute m properly
            full_corr = correlate(refined_x, refined_x, mode='full')[:2*n-1]
            best_m = np.min(full_corr[n:]) # Proper minimum from full correlation
            
    # Stage 2: Try larger discretization if time allows
    if best_h is not None and time.time() - start_time < budget_s * 0.6:
        try:
            n_large = 144
            dx_large = 2.0 / n_large
            N_large = n_large // 2
            
            x_large = cp.Variable(n_large, boolean=True)
            mass_constraint = cp.sum(x_large) == N_large
            
            symmetry_constraints = [x_large[0] == 1]
            for i in range(N_large - 1):
                symmetry_constraints.append(x_large[i] >= x_large[i+1])
                
            y_large = {}
            S_large = {}
            
            constraints = [mass_constraint] + symmetry_constraints
            
            for l in range(1, n_large):
                for i in range(n_large - l):
                    y_large[i, l] = cp.Variable(boolean=True)
                    constraints.append(y_large[i, l] <= x_large[i])
                    constraints.append(y_large[i, l] <= x_large[i + l])
                    constraints.append(y_large[i, l] >= x_large[i] + x_large[i + l] - 1)
                    
                    if l not in S_large:
                        S_large[l] = 0
                    S_large[l] += y_large[i, l]
            
            m_large = cp.Variable()
            for l in range(1, n_large):
                constraints.append(S_large[l] >= m_large)
                
            objective = cp.Maximize(m_large)
            prob = cp.Problem(objective, constraints)
            
            time_remaining = budget_s * 0.15
            prob.solve(solver=cp.CBC, max_seconds=time_remaining)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                x_sol = np.round(x_large.value).astype(int)
                full_corr = correlate(x_sol, x_sol, mode='full')[:2*n_large-1]
                m_val = np.min(full_corr[n_large:])
                c5_val = dx_large * (N_large - m_val)
                
                if c5_val < best_c5:
                    best_h = x_sol
                    best_c5 = c5_val
                    best_m = m_val
                    
        except:
            pass
    
    # Stage 3: Final refinement with local search
    if best_h is not None:
        refined_x, refined_c5 = hill_climb_binary(best_h, n, N, dx)
        if refined_c5 < best_c5:
            best_h = refined_x
            best_c5 = refined_c5
    
    # Stage 4: Random restarts for safety
    if best_h is None:
        best_c5 = float('inf')
        
    if best_h is not None:
        # Try several random restarts with hill climbing
        for _ in range(5):
            if time.time() - start_time > budget_s * 0.95:
                break
            rand_x = np.random.rand(n)
            # Project to sum constraint
            rand_x = np.clip(rand_x + np.mean((N - np.sum(rand_x)) / n), 0, 1)
            # Round and adjust
            rand_x = (rand_x > 0.5).astype(int)
            ones_count = np.sum(rand_x)
            if ones_count > N:
                ones_indices = np.where(rand_x == 1)[0]
                flip_count = int(ones_count - N)
                flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
                rand_x[flip_indices] = 0
            elif ones_count < N:
                zeros_indices = np.where(rand_x == 0)[0]
                flip_count = int(N - ones_count)
                flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
                rand_x[flip_indices] = 1
                
            refined_x, refined_c5 = hill_climb_binary(rand_x, n, N, dx)
            if refined_c5 < best_c5:
                best_h = refined_x
                best_c5 = refined_c5
                
    return (best_h.tolist(), float(best_c5), n)
```
