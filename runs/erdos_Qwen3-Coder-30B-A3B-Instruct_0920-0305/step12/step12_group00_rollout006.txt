from scipy.signal import correlate
from scipy.sparse import diags, csr_matrix
import cvxpy as cp
from scipy.linalg import eigh
import time

# Global constants
n = 128
dx = 2.0 / n
N = n // 2
J = np.ones((n, n))

# Precompute Toeplitz matrices
def build_toeplitz_matrices(n):
    A = []
    for k in range(1, n):
        # Create Toeplitz matrix for lag k
        # Diagonal at offset k
        data = [1] * (n - k)
        offsets = [k]
        diags_ = diags(data, offsets, shape=(n, n), format='csr')
        A.append((diags_ + diags_.T).tocsr())
    return A

A_matrices = build_toeplitz_matrices(n)
