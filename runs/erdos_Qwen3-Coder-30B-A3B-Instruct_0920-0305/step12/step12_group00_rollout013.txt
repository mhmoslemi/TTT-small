import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
import math

def balanced_sturmian(n):
    """Generate a balanced Sturmian word for given length n"""
    phi = (math.sqrt(5) - 1) / 2
    x = np.array([int((i * phi) % 1 < 0.5) for i in range(n)])
    # Ensure exactly n/2 ones
    ones_count = np.sum(x)
    if ones_count > n // 2:
        zeros_indices = np.where(x == 0)[0]
        flip_count = int(ones_count - n // 2)
        flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
        x[flip_indices] = 1
    elif ones_count < n // 2:
        ones_indices = np.where(x == 1)[0]
        flip_count = int(n // 2 - ones_count)
        flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
        x[flip_indices] = 0
    return x

def compute_autocorr_fft(x):
    """Compute autocorrelations using FFT for efficiency"""
    n = len(x)
    x_rev = x[::-1]
    b = x
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(x_rev, L) * np.fft.fft(b, L)).real
    return corr[:n]

def swap_delta(S, i, j, ones, zeros, n):
    """Efficiently compute the change in autocorrelations due to swapping x[i] and x[j]"""
    delta_S = np.zeros(n)
    # For each lag ℓ, compute how the swap affects S_ℓ
    for l in range(1, n):
        # Before swap: x[i]·x[i+l] + x[j]·x[j+l]
        # After swap: x[i]·x[i+l] + x[j]·x[j+l] + (x[i] - x[j])·(x[i+l] - x[j+l])
        if i + l < n:
            delta_S[l] += (x[i] - x[j]) * (x[i+l] - x[j+l])
        if j + l < n:
            delta_S[l] += (x[j] - x[i]) * (x[j+l] - x[i+l])
        # Effect due to index i and j in x[i+l] and x[j+l] respectively
        # Check if x[i] changes or x[j] changes
        # Bounds checks to avoid index errors
        if i >= l and i + l < n:
            delta_S[l] -= (x[i] - x[j]) * (x[i-l] - x[j-l]) if i >= l and i-l >= 0 else 0
        if j >= l and j + l < n:
            delta_S[l] -= (x[j] - x[i]) * (x[j-l] - x[i-l]) if j >= l and j-l >= 0 else 0
            
    return delta_S

def local_search_improve(h, target_sum, dx, max_time):
    """Improve a solution using greedy swap hill-climb with simulated annealing fallback"""
    # Convert to int array if not already
    x = np.array(h, dtype=int)
    n = len(x)
    ones = set(np.where(x == 1)[0])
    zeros = set(np.where(x == 0)[0])
    
    # Compute autocorrelations
    corr = compute_autocorr_fft(x)
    S = corr[:n-1]  # S[l-1] = S_l
    m = np.min(S[1:])  # exclude lag 0
    
    start_time = time.time()
    best_m = m
    best_x = x.copy()
    
    # Greedy hill-climb
    iterations = 0
    no_improve_count = 0
    T = 0.01 * (1 - dx * (n//2 - m))  # Initial temperature
    
    while time.time() - start_time < max_time:
        # Try random swap
        i = np.random.choice(list(ones))
        j = np.random.choice(list(zeros))
        
        # Compute effect on autocorrelations
        delta_S = np.zeros(n-1)
        for l in range(1, n-1):
            if i+l < n:
                delta_S[l] += -x[i]*x[i+l] + x[j]*x[j+l]
            if j+l < n:
                delta_S[l] += -x[j]*x[j+l] + x[i]*x[i+l]
            if l <= i:
                delta_S[l] += -x[i-l]*x[i] + x[i-l]*x[j]
            if l <= j:
                delta_S[l] += -x[j-l]*x[j] + x[j-l]*x[i]
            
        S_new = S + delta_S
        m_new = np.min(S_new[1:])  # exclude lag 0
        
        # Accept or reject
        if m_new > m or (np.random.rand() < np.exp((m_new - m) / T)):
            # Perform swap
            x[i], x[j] = x[j], x[i]
            ones.remove(i)
            ones.add(j)
            zeros.remove(j)
            zeros.add(i)
            S = S_new
            m = m_new
            
            if m > best_m:
                best_m = m
                best_x = x.copy()
                no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Simulated annealing cooling schedule
        if no_improve_count > 5 * n:
            T *= 0.99
        
        # Reset no-improve counter every 1000 iterations
        if iterations % 1000 == 0:
            no_improve_count = 0
            
        iterations += 1
    
    # Final evaluation
    final_corr = compute_autocorr_fft(best_x)
    final_m = np.min(final_corr[1:])
    final_c5 = dx * (n//2 - final_m)
    return best_x.tolist(), float(final_c5), int(final_m)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_values = [128, 144, 160]
    dx = 2.0 / 128  # default
    N = 128 // 2  # mass constraint
    
    # Time allocation
    ilp_time = 0.45 * budget_s
    search_time = 0.35 * budget_s
    fallback_time = 0.15 * budget_s
    
    best_c5 = float('inf')
    best_h = None
    best_n = None
    best_m = None
    
    # Try different discretizations
    for n in n_values:
        if time.time() - start_time > budget_s - 2:
            break
            
        dx = 2.0 / n
        N = n // 2
        
        # Create the ILP model
        try:
            # Binary variables
            x = cp.Variable(n, boolean=True)
            # Auxiliary variables for products x[i]*x[i+ℓ]
            y = {}
            S = {}
            m = cp.Variable()
            
            # Constraints for McCormick envelopes 
            constraints = []
            
            # Set sum constraint
            constraints.append(cp.sum(x) == N)
            
            # Fix first element to break symmetry, and ensure first half is descending
            constraints.append(x[0] == 1)
            for i in range(1, N):
                constraints.append(x[i-1] >= x[i])
            
            # Build the autocorrelations and constraints
            for l in range(1, n):
                # Create auxiliary variables for y[i,l] = x[i]*x[i+l]
                for i in range(n - l):
                    key = (i, l)
                    y[key] = cp.Variable(boolean=True) 
                    constraints.append(y[key] <= x[i])
                    constraints.append(y[key] <= x[i+l])
                    constraints.append(y[key] >= x[i] + x[i+l] - 1)
                
                # Compute S_l
                S[l] = cp.sum([y[(i,l)] for i in range(n-l)])
                constraints.append(S[l] >= m)
            
            # Objective: maximize m
            objective = cp.Maximize(m)
            problem = cp.Problem(objective, constraints)
            
            # Solve with timeout
            problem.solve(solver=cp.CBC, verbose=False, max_seconds=ilp_time)
            
            # Check solution
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                x_val = np.array([int(np.round(val.value)) for val in x])
                # Ensure the sum is correct
                actual_sum = np.sum(x_val)
                if abs(actual_sum - N) > 1e-5:
                    # Repair if necessary
                    if actual_sum < N:
                        zero_indices = np.where(x_val == 0)[0]
                        to_flip = N - actual_sum
                        if len(zero_indices) >= to_flip:
                            flip_indices = np.random.choice(zero_indices, to_flip, replace=False)
                            x_val[flip_indices] = 1
                    elif actual_sum > N:
                        ones_indices = np.where(x_val == 1)[0]
                        to_flip = actual_sum - N
                        if len(ones_indices) >= to_flip:
                            flip_indices = np.random.choice(ones_indices, to_flip, replace=False)
                            x_val[flip_indices] = 0
                
                # Check sum again
                if abs(np.sum(x_val) - N) < 1e-5:
                    # Compute actual C5
                    corr = compute_autocorr_fft(x_val)
                    m_val = np.min(corr[1:])  # compute minimum correlation (excluding lag 0)
                    c5_val = dx * (N - m_val)
                    
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = x_val.tolist()
                        best_n = n
                        best_m = m_val
                        
        except Exception:
            continue
    
    # If no ILP solution found, use fallback: local search on Sturmian sequence
    if best_h is None:
        # Initialize with Sturmian word
        h_initial = balanced_sturmian(128)
        h_initial = np.array(h_initial)
        corr = compute_autocorr_fft(h_initial)
        m_initial = np.min(corr[1:])
        c5_initial = dx * (128//2 - m_initial)
        best_h = h_initial.tolist()
        best_c5 = c5_initial
        best_n = 128
        best_m = m_initial
    
    # Refinement with local search
    if time.time() - start_time < budget_s - 10:
        final_h, final_c5, final_m = local_search_improve(best_h, 128//2, dx, search_time)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
            best_m = final_m
    
    return (best_h, float(best_c5), best_n)
