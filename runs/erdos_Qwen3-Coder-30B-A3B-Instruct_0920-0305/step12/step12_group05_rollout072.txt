def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def simulated_annealing(h_start, current_c5, start_time, budget_s):
    """Implement simulated annealing for global optimization"""
    h_current = h_start.copy()
    c5_current = current_c5
    best_h = h_current.copy()
    best_c5 = c5_current
    
    T = 1.0  # Initial temperature
    T_min = 1e-8  # Minimum temperature
    alpha = 0.95  # Cooling rate
    max_iter = 1000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Generate neighbor solution
        h_neighbor = h_current.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.05)
        h_neighbor[idx] += delta
        
        # Project to valid region
        h_neighbor = project_to_sum_box(h_neighbor, TARGET_SUM)
        c5_neighbor = evaluate_c5(h_neighbor)
        
        # Accept or reject the new solution
        if c5_neighbor < c5_current:
            h_current = h_neighbor
            c5_current = c5_neighbor
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_neighbor.copy()
        else:
            # Accept with probability based on temperature
            if np.random.random() < np.exp(-(c5_neighbor - c5_current) / T):
                h_current = h_neighbor
                c5_current = c5_neighbor
                
        # Cool down
        T = max(T_min, T * alpha)
    
    return best_h, best_c5
