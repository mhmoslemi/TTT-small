def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max(corr, alpha):
    """Compute smooth approximation to max with parameter alpha"""
    exp_corr = np.exp(alpha * corr)
    return np.log(np.sum(exp_corr)) / alpha

def smooth_grad(corr, alpha, h):
    """Compute gradient of smooth max with respect to h"""
    exp_corr = np.exp(alpha * corr)
    weights = exp_corr / np.sum(exp_corr)
    # Compute convolution with weights
    L = len(h) + len(weights) - 1
    L_fft = 1 << (L - 1).bit_length()
    A = np.fft.fft(weights, L_fft)
    B = np.fft.fft(h, L_fft)
    conv_result = np.fft.ifft(A * B).real[:len(weights)]
    return 1.0 - conv_result
