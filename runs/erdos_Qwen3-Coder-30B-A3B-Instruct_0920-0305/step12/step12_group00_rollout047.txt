from scipy.linalg import eigh
from scipy.signal import correlate
from scipy.optimize import minimize
import time

from cvxpy import Variable, trace, sum, Minimize, Problem, SCS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 128
    dx = 2.0 / n
    N = n // 2
    
    # Precompute Toeplitz matrices for all lags
    def create_toeplitz_matrix(lag, n):
        # Create Toeplitz matrix A_lag where (A_lag)_{ij} = 1 if |i-j| = lag
        A = np.zeros((n, n))
        for i in range(n):
            j = (i + lag) % n
            A[i, j] = 1
        return A
    
    # Precompute all matrices
    A_matrices = {}
    for lag in range(1, n):
        A_matrices[lag] = create_toeplitz_matrix(lag, n)
    
    # Matrix for mass constraint
    J = np.ones((n, n))
    
    # SDP solver parameters
    t_sdp = 0.4 * budget_s
    t_hc = 0.3 * budget_s
    
    best_c5 = float('inf')
    best_h = None
    
    # Function to solve SDP
    def solve_sdp(A_matrices, n, N, t_sdp, J):
        # Variables
        X = Variable((n, n), PSD=True)
        m = Variable()
        
        # Constraints
        constraints = [
            trace(J @ X) == N,
            X >= 0,
            X <= 1
        ]
        
        # Add autocorrelation constraints
        for lag in range(1, n):
            constraints.append(trace(A_matrices[lag] @ X) >= m)
        
        # Objective
        objective = Minimize(-m)  # We maximize m, so minimize -m
        
        # Problem
        prob = Problem(objective, constraints)
        
        # Solve
        try:
            prob.solve(solver=SCS, max_iters=5000, eps=1e-5, time_limit=t_sdp, verbose=False)
            return X.value, m.value
        except:
            # Fallback: return an infeasible point if solver fails
            return None, None
    
    # Rounding function (hyperplane rounding)
    def round_solution(X_val, n, N):
        num_rounds = 5
        best_h = None
        best_c5 = float('inf')
        
        for _ in range(num_rounds):
            # Eigen-decomposition
            try:
                eigenvals, eigenvecs = eigh(X_val)
                # Keep positive eigenvalues
                positive_indices = eigenvals > 1e-8
                if np.sum(positive_indices) == 0:
                    # Fallback to simpler approach
                    scores = np.diag(X_val)
                else:
                    V = eigenvecs[:, positive_indices] * np.sqrt(eigenvals[positive_indices])
                    # Sample random vector
                    g = np.random.randn(V.shape[1])
                    scores = V @ g
                # Threshold to get binary vector
                sorted_indices = np.argsort(-scores)
                h = np.zeros(n)
                h[sorted_indices[:N]] = 1
                # Ensure exact mass
                current_sum = np.sum(h)
                if current_sum > N:
                    # Need to zero out some 1s
                    ones_indices = np.where(h == 1)[0]
                    indices_to_zero = ones_indices[:int(current_sum - N)]
                    h[indices_to_zero] = 0
                elif current_sum < N:
                    # Need to set some 0s to 1s
                    zeros_indices = np.where(h == 0)[0]
                    indices_to_one = zeros_indices[:int(N - current_sum)]
                    h[indices_to_one] = 1
                
                # Check if new best
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            except:
                # If eigendecomposition fails, try another approach
                continue
                
        return best_h, best_c5
    
    # Function to evaluate C5 (exact value via correlation)
    def evaluate_c5(h):
        # Compute correlation using FFT
        h_rev = h[::-1]
        b = 1 - h
        corr = np.fft.ifft(np.fft.fft(h_rev, 2*n-1) * np.fft.fft(b, 2*n-1)).real
        corr = corr[:n]  # only keep the first n correlation values
        max_corr = np.max(corr)
        return dx * max_corr
    
    # Swap-based hill climb
    def hill_climb(h, t_hc):
        all_lags = list(range(1, n))
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        
        # Compute initial correlation
        corr = np.zeros(n)
        for lag in all_lags:
            cross_corr = correlate(h, np.roll(h, lag), mode='valid')
            corr[lag] = cross_corr[0]
        
        current_c5 = dx * np.max(np.sum(h) - corr[1:])  # since corr[0] is when lag=0

        max_swaps = 100000
        improvement_count = 0
        early_stop_count = 0
        
        while swap_attempts < max_swaps and time.time() - start_time < t_hc:
            if early_stop_count > 5000:
                break
            # Pick random indices to swap
            i = np.random.choice(ones_idx)
            j = np.random.choice(zeros_idx)
            
            # Simulate the swap
            new_h = h.copy()
            new_h[i] = 0
            new_h[j] = 1
            
            # Compute new correlations for affected lags (signed, so scan nearby lags)
            lag_shifts = []
            for lag in all_lags:
                corr_i = h[i] * h[(i+lag)%n] + h[i] * h[(i-lag)%n]
                corr_j = h[j] * h[(j+lag)%n] + h[j] * h[(j-lag)%n]
                corr_new_i = new_h[i] * new_h[(i+lag)%n] + new_h[i] * new_h[(i-lag)%n]
                corr_new_j = new_h[j] * new_h[(j+lag)%n] + new_h[j] * new_h[(j-lag)%n]
                # Update magnitude change
                diff = (corr_new_i + corr_new_j) - (corr_i + corr_j)
                lag_shifts.append((lag, diff))
            
            # Find updated overlaps
            # Use FFT to get new full correlation
            corr_new = np.zeros(n)
            for lag in all_lags:
                cross_corr_new = correlate(new_h, np.roll(new_h, lag), mode='valid')
                corr_new[lag] = cross_corr_new[0]
            
            new_min_corr = np.min(np.sum(h) - corr_new[1:])
            
            if new_min_corr > current_min_corr:
                # Accept the change
                h = new_h
                current_min_corr = new_min_corr
                current_c5 = dx * (N - current_min_corr)
                improvement_count += 1
                early_stop_count = 0
            else:
                early_stop_count += 1
                if improvement_count > 0:
                    improvement_count -= 1
            
            swap_attempts += 1
            
        return h, current_c5
    
    # Main SDP loop
    best_sdp_h = None
    best_sdp_c5 = float('inf')

    X_val, m_opt = solve_sdp(A_matrices, n, N, t_sdp, J)
    
    if X_val is not None and m_opt is not None:
        # Round the solution
        rounded_h, rounded_c5 = round_solution(X_val, n, N)
        if rounded_c5 < best_sdp_c5:
            best_sdp_c5 = rounded_c5
            best_sdp_h = rounded_h.copy()
    
    # If SDP didn't work, try solving a smaller SDP and blow up
    if best_sdp_h is None:
        n_coarse = 64
        dx_coarse = 2.0 / n_coarse
        N_coarse = n_coarse // 2
        
        # Create Toeplitz matrices for coarse grid
        A_matrices_coarse = {}
        for lag in range(1, n_coarse):
            A_matrices_coarse[lag] = create_toeplitz_matrix(lag, n_coarse)
        
        J_coarse = np.ones((n_coarse, n_coarse))
        
        # Solve SDP on small grid
        X_coarse, m_coarse = solve_sdp(A_matrices_coarse, n_coarse, N_coarse, t_sdp/2, J_coarse)
        if X_coarse is not None and m_coarse is not None:
            # Blow up to full size
            expanded_h = np.zeros(n)
            b = n // n_coarse
            for i in range(n_coarse):
                expanded_h[i*b:(i+1)*b] = X_coarse[i, i] if X_coarse[i, i] > 0.5 else 0
            # Adjust mass
            current_sum = np.sum(expanded_h)
            if current_sum > N:
                ones_indices = np.where(expanded_h == 1)[0]
                flip_indices = np.random.choice(ones_indices, int(current_sum - N), replace=False)
                expanded_h[flip_indices] = 0
            elif current_sum < N:
                zeros_indices = np.where(expanded_h == 0)[0]
                flip_indices = np.random.choice(zeros_indices, int(N - current_sum), replace=False)
                expanded_h[flip_indices] = 1
            
            # Hill climb
            final_h, final_c5 = hill_climb(expanded_h, t_hc)
            if final_c5 < best_sdp_c5:
                best_sdp_c5 = final_c5
                best_sdp_h = final_h.copy()
    
    # At least try random restarts
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        
        # Create random binary vector
        h_random = np.zeros(n)
        indices_one = np.random.choice(n, N, replace=False)
        h_random[indices_one] = 1
        
        # Find the best from this random start in a short run
        h_local, c5_local = hill_climb(h_random, t_hc/10)
        if c5_local < best_sdp_c5:
            best_sdp_c5 = c5_local
            best_sdp_h = h_local.copy()
