def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5_and_corr(h):
    """Evaluate C5 and return correlation array"""
    corr = np.correlate(h, 1 - h, mode="full")
    O = corr * dx
    max_O = np.max(O)
    return max_O, O

def compute_gradient_for_lag(h, lag):
    """Compute gradient of O(k) for given lag"""
    N = len(h)
    forward = np.zeros(N)
    backward = np.zeros(N)
    
    if lag >= 0:
        forward[:N - lag] = h[lag:]
        backward[lag:] = h[:N - lag]
    else:
        k = -lag
        forward[k:] = h[:N - k]
        backward[:N - k] = h[k:]
    
    grad = np.ones(N) - forward - backward
    return grad

        lag = np.argmax(O_arr) - (len(h) - 1)
        
        # Compute gradient for this lag
        grad = compute_gradient_for_lag(h, lag)
        
        # Update step size with diminishing schedule
        step = step_size / np.sqrt(iteration + 1)
        
        # Update h
        h_new = h - step * grad
        h_new = project_to_sum_box(h_new, target_sum)

        new_val, _ = evaluate_c5_and_corr(h_new)

            h += np.random.normal(0, 0.01, len(h))
            h = project_to_sum_box(h, target_sum)
    
    return best_h, best_val
