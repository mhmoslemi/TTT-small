import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort the vector
    sorted_x = np.sort(x)[::-1]
    
    # Binary search for lambda
    def f(lam):
        projected = np.clip(sorted_x - lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    lambda_val = (a + b) / 2
    result = np.clip(x - lambda_val, 0, 1)
    return result

def softmax_weights(corr, alpha):
    """Compute softmax weights"""
    exp_corr = np.exp(alpha * corr)
    return exp_corr / np.sum(exp_corr)

def grad_softmax(corr, alpha, h):
    """Compute gradient of softmax with respect to h"""
    weights = softmax_weights(corr, alpha)
    # Compute forward and backward convolutions
    conv1 = np.convolve(weights, h, mode='valid')
    conv2 = np.convolve(weights[::-1], h, mode='valid')
    # Adjust indices
    grad = 1.0 - conv1 - conv2
    return grad

def adam_step(h, m, v, grad, t, alpha, beta1=0.9, beta2=0.999, epsilon=1e-8):
    """Adam optimization step with projection"""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    
    h_new = h - alpha * m_hat / (np.sqrt(v_hat) + epsilon)
    
    return h_new, m, v

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points // 2
    best_c5 = float('inf')
    best_h = None
    
    # Sturmian word initialization
    phi = (np.sqrt(5) - 1) / 2
    sturmian = np.array([(i * phi) % 1 < 0.5 for i in range(n_points)], dtype=float)
    h0 = project_to_sum_box(sturmian, TARGET_SUM)
    
    # Random initialization
    h1 = project_to_sum_box(np.random.rand(n_points), TARGET_SUM)
    
    # Perturbed Sturmian
    h2 = project_to_sum_box(sturmian + np.random.normal(0, 0.05, n_points), TARGET_SUM)
    
    initial_solutions = [h0, h1, h2]
    
    # Adam optimization
    alpha0 = 4.0
    alpha_growth = 1.5
    alpha_max = 200.0
    max_iters = 8000
    learning_rate = 0.12
    
    best_solutions = []
    
    for i, h_start in enumerate(initial_solutions):
        if time.time() - start_time > 0.45 * budget_s:
            break
            
        h = h_start.copy()
        m = np.zeros(n_points)
        v = np.zeros(n_points)
        
        current_alpha = alpha0
        prev_c5 = float('inf')
        
        for t in range(1, max_iters + 1):
            if time.time() - start_time > 0.45 * budget_s:
                break
                
            corr = compute_corr_fft(h)
            current_c5 = c5_from_corr(corr, n_points)
            
            # Update best solution
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            # Grad of softmax approximation
            grad = grad_softmax(corr, current_alpha, h)
            
            # Adam step with projection
            h_new, m, v = adam_step(h, m, v, grad, t, learning_rate)
            h = project_to_sum_box(h_new, TARGET_SUM)
            
            # Increase alpha for sharper max approximation
            if t % 200 == 0:
                current_alpha = min(current_alpha * alpha_growth, alpha_max)
    
    # Convert to binary and optimize further
    if best_h is not None:
        # Rounding
        binary_h = (best_h >= 0.5).astype(float)
        
        # Adjust sum
        ones_count = np.sum(binary_h)
        if ones_count > TARGET_SUM:
            flip_count = int(ones_count - TARGET_SUM)
            ones_indices = np.where(binary_h == 1)[0]
            flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
            binary_h[flip_indices] = 0
        elif ones_count < TARGET_SUM:
            flip_count = int(TARGET_SUM - ones_count)
            zeros_indices = np.where(binary_h == 0)[0]
            flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
            binary_h[flip_indices] = 1
        
        binary_h = project_to_sum_box(binary_h, TARGET_SUM)
        
        # Update binary solution
        current_c5 = c5_from_corr(compute_corr_fft(binary_h), n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = binary_h.copy()
        
        # Swap hill-climb
        ones = np.where(binary_h == 1)[0]
        zeros = np.where(binary_h == 0)[0]
        current_binary = binary_h.copy()
        current_c5 = c5_from_corr(compute_corr_fft(current_binary), n_points)
        
        # Precompute correlations
        corr = compute_corr_fft(current_binary)
        S = corr
        
        # Swap optimization
        unsuccessful = 0
        max_unsuccessful = 5 * n_points
        t_end = start_time + 0.8 * budget_s
        while time.time() < t_end and unsuccessful < max_unsuccessful:
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Compute change in correlations
            delta_S = np.zeros(2 * n_points - 1)
            for k in range(-n_points + 1, n_points):
                if i + k >= 0 and i + k < n_points:
                    delta_S[k + n_points - 1] -= current_binary[i + k]
                if j + k >= 0 and j + k < n_points:
                    delta_S[k + n_points - 1] += current_binary[j + k]
                        
                if i - k >= 0 and i - k < n_points:
                    delta_S[k + n_points - 1] -= current_binary[i - k]
                if j - k >= 0 and j - k < n_points:
                    delta_S[k + n_points - 1] += current_binary[j - k]
            
            new_S = S + delta_S
            new_c5 = dx * np.max(new_S)
            
            if new_c5 < current_c5:
                # Accept swap
                current_binary[i] = 0
                current_binary[j] = 1
                S = new_S
                current_c5 = new_c5
                
                # Update indices
                ones = np.where(current_binary == 1)[0]
                zeros = np.where(current_binary == 0)[0]
                
                # Update best
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_binary.copy()
                
                unsuccessful = 0
            else:
                unsuccessful += 1
                
            # Occasionally random restart to escape local minimum
            if np.random.rand() < 0.01 and unsuccessful > 100:
                # Restart from best solution
                current_binary = best_h.copy()
                S = compute_corr_fft(current_binary)
                current_c5 = dx * np.max(S)
                unsuccessful = 0
        
        # Additional restarts
        t_end = start_time + budget_s
        for _ in range(5):
            if time.time() >= t_end:
                break
            # Random binary initialization
            rand_h = np.random.rand(n_points)
            rand_h = project_to_sum_box(rand_h, TARGET_SUM)
            rand_binary = (rand_h >= 0.5).astype(float)
            
            # Adjust sum
            ones_count = np.sum(rand_binary)
            if ones_count > TARGET_SUM:
                flip_count = int(ones_count - TARGET_SUM)
                ones_indices = np.where(rand_binary == 1)[0]
                flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
                rand_binary[flip_indices] = 0
            elif ones_count < TARGET_SUM:
                flip_count = int(TARGET_SUM - ones_count)
                zeros_indices = np.where(rand_binary == 0)[0]
                flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
                rand_binary[flip_indices] = 1
            
            rand_binary = project_to_sum_box(rand_binary, TARGET_SUM)
            
            # Hill-climb
            ones = np.where(rand_binary == 1)[0]
            zeros = np.where(rand_binary == 0)[0]
            current_h = rand_binary.copy()
            corr = compute_corr_fft(current_h)
            current_c5 = dx * np.max(corr)
            
            unsuccessful = 0
            max_unsuccessful = 2 * n_points
            while time.time() < t_end and unsuccessful < max_unsuccessful:
                if len(ones) == 0 or len(zeros) == 0:
                    break
                    
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                
                # Compute change in correlation
                delta_S = np.zeros(2 * n_points - 1)
                for k in range(-n_points + 1, n_points):
                    if i + k >= 0 and i + k < n_points:
                        delta_S[k + n_points - 1] -= current_h[i + k]
                    if j + k >= 0 and j + k < n_points:
                        delta_S[k + n_points - 1] += current_h[j + k]
                            
                    if i - k >= 0 and i - k < n_points:
                        delta_S[k + n_points - 1] -= current_h[i - k]
                    if j - k >= 0 and j - k < n_points:
                        delta_S[k + n_points - 1] += current_h[j - k]
                
                new_S = corr + delta_S
                new_c5 = dx * np.max(new_S)
                
                if new_c5 < current_c5:
                    current_h[i] = 0
                    current_h[j] = 1
                    corr = new_S
                    current_c5 = new_c5
                    
                    # Update indices
                    ones = np.where(current_h == 1)[0]
                    zeros = np.where(current_h == 0)[0]
                    
                    # Update best
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = current_h.copy()
                    
                    unsuccessful = 0
                else:
                    unsuccessful += 1
    
    return (best_h.tolist(), float(best_c5), n_points)
