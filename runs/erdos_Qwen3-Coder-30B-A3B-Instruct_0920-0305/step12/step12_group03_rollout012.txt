import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Parameters
    n_points = 120

    target_sum = n_points / 2
    max_iter = 500000  # For time management

    phi = (1 + np.sqrt(5)) / 2
    beatty_indices = np.floor(np.arange(n_points) * phi) % n_points
    beatty_indices = np.sort(beatty_indices.astype(int))

    h[beatty_indices[:int(target_sum)]] = 1
    
    # Generate other initial patterns for seeding
    patterns = [h.copy()]
    
    # Random seed
    random_indices = np.random.permutation(n_points)[:int(target_sum)]
    h_random = np.zeros(n_points, dtype=int)
    h_random[random_indices] = 1
    patterns.append(h_random)

    for i, h_seed in enumerate(patterns):
        # Compute initial correlation
        h = h_seed.copy().astype(float)
        C_raw = np.correlate(h, 1 - h, mode='full')
        cur_max = np.max(C_raw)
        current_val = dx * cur_max
        
        if current_val < best_val:
            best_val = current_val
            best_h = h.copy()
    
    h = best_h.copy()

    # Update correlation efficiently with O(n) operations
    def update_correlation(C_raw, h, i, j):
        # i: 0, j: 1
        delta_C = np.zeros_like(C_raw)
        
        # For each shift k where i+k is valid
        # C_raw[k + n - 1] corresponds to shift k (since full correlation indexed from -n+1 to n-1)
        for k in range(-n_points + 1, n_points):
            if 0 <= i + k < n_points:
                delta_C[k + n_points - 1] -= 1 - h[i + k]
            if 0 <= j - k < n_points:
                delta_C[k + n_points - 1] -= h[j - k]
        return delta_C

    # Simulated Annealing
    T = 0.5
    alpha = 0.995
    N_swap = n_points
    swap_count = 0
    N_accept = 0
    
    cur_max = np.max(np.correlate(h, 1 - h, mode='full'))
    current_val = dx * cur_max
    
    while time.time() - start_time < budget_s - 10 and swap_count < max_iter:
        # Pick random i where h[i] = 0 and j where h[j] = 1
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        
        if len(zeros) == 0 or len(ones) == 0:
            break
            
        i = np.random.choice(zeros)
        j = np.random.choice(ones)
        
        # Compute change in correlation
        delta_C = update_correlation(C_raw, h, i, j)
        new_C_raw = C_raw + delta_C
        new_max = np.max(new_C_raw)
        new_val = dx * new_max
        
        # Accept or reject the move
        if new_val < current_val:
            # Always accept improvement
            h[i] = 1
            h[j] = 0
            C_raw = new_C_raw
            current_val = new_val
            N_accept += 1
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(new_val - current_val) / T):
                h[i] = 1
                h[j] = 0
                C_raw = new_C_raw
                current_val = new_val
                N_accept += 1
        
        swap_count += 1
        
        # Temperature cooling
        if swap_count % N_swap == 0:
            T *= alpha
            if T < 1e-5:
                break
    
    # Final deterministic hill climb

    while improved and time.time() - start_time < budget_s - 5:
        improved = False
        cur_max = np.max(C_raw)
        for i in range(n_points):

                continue

                    continue
                # Try the swap
                delta_C = update_correlation(C_raw, h, j, i)
                new_C_raw = C_raw + delta_C
                new_max = np.max(new_C_raw)
                new_val = dx * new_max
                
                if new_val < current_val:
                    h[i] = 0
                    h[j] = 1
                    C_raw = new_C_raw
                    current_val = new_val
                    improved = True
                    break
            if improved:
                break
                
    # Final check
    final_corr = np.correlate(h, 1 - h, mode='full')
    final_max = np.max(final_corr)
    final_C5 = dx * final_max
    
    return (h.tolist(), float(final_C5), n_points)
