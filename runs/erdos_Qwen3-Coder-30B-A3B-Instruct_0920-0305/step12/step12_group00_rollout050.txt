import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort the vector
    sorted_x = np.sort(x)[::-1]
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x - lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x - (a + b)/2, 0, 1)
    return result

def compute_softmax_weights(corr, alpha):
    """Compute softmax weights"""
    exp_corr = np.exp(alpha * corr)
    return exp_corr / np.sum(exp_corr)

def grad_softmax(corr, alpha, h):
    """Compute gradient of softmax with respect to h"""
    weights = compute_softmax_weights(corr, alpha)
    # Compute forward convolution: w * h
    L = 2 * len(h) - 1
    L_fft = 1 << (L - 1).bit_length()
    w_fft = np.fft.fft(weights, L_fft)
    h_fft = np.fft.fft(h, L_fft)
    conv_forward = np.fft.ifft(w_fft * h_fft).real[:len(h)]
    
    # Compute backward convolution: w_rev * h
    w_rev_fft = np.fft.fft(weights[::-1], L_fft)
    conv_backward = np.fft.ifft(w_rev_fft * h_fft).real[:len(h)]
    
    grad = 1.0 - conv_forward - conv_backward
    return grad

def adam_step(h, m, v, grad, t, alpha, beta1, beta2, epsilon, lr):
    """Adam optimization step with projection"""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
    h_proj = project_to_sum_box(h_new, TARGET_SUM)
    
    return h_proj, m, v

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points // 2
    best_c5 = float('inf')
    best_h = None
    
    # Helper functions for Sturmian word
    def sturmian_word(n, phi=(np.sqrt(5)-1)/2):
        h = np.zeros(n)
        for i in range(n):
            if (i * phi) % 1 < 0.5:
                h[i] = 1.0
        return h
    
    # Generate initial solutions
    initial_solutions = []
    
    # 1. Balanced Sturmian word
    h_sturm = sturmian_word(n_points)
    h_sturm = project_to_sum_box(h_sturm, TARGET_SUM)
    initial_solutions.append(("sturmian", h_sturm))
    
    # 2. Random initialization with projection
    h_rand = np.random.rand(n_points)
    h_rand = project_to_sum_box(h_rand, TARGET_SUM)
    initial_solutions.append(("random", h_rand))
    
    # 3. Perturbed Sturmian word
    h_perturbed = h_sturm + np.random.normal(0, 0.05, n_points)
    h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
    initial_solutions.append(("perturbed", h_perturbed))
    
    # Adam optimization
    beta1, beta2, epsilon = 0.9, 0.999, 1e-8
    lr = 0.12
    
    # Process each initial solution
    for name, h0 in initial_solutions:
        if time.time() - start_time > 0.4 * budget_s:
            break
            
        h = h0.copy()
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        alpha = 4.0
        t = 1
        
        max_iter = 8000
        t_max = start_time + 0.4 * budget_s
        last_improvement = 0
        patience = 100
        
        while time.time() < t_max and t < max_iter:
            corr = compute_corr_fft(h)
            c5_val = c5_from_corr(corr, n_points)
            
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
                last_improvement = t
            
            grad = grad_softmax(corr, alpha, h)
            h, m, v = adam_step(h, m, v, grad, t, alpha, beta1, beta2, epsilon, lr)
            
            # Update alpha
            if t % 200 == 0:
                alpha = min(alpha * 1.5, 200.0)
            
            t += 1
            
            if t - last_improvement > patience:
                break
    
    # Binary rounding and improvement
    if best_h is not None:
        # Round to binary
        binary_h = (best_h >= 0.5).astype(float)
        
        # Adjust to correct sum
        ones_count = np.sum(binary_h)
        if ones_count > TARGET_SUM:
            ones_indices = np.where(binary_h == 1)[0]
            flip_count = int(ones_count - TARGET_SUM)
            flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
            binary_h[flip_indices] = 0
        elif ones_count < TARGET_SUM:
            zeros_indices = np.where(binary_h == 0)[0]
            flip_count = int(TARGET_SUM - ones_count)
            flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
            binary_h[flip_indices] = 1
        
        binary_h = project_to_sum_box(binary_h, TARGET_SUM)
        current_c5 = c5_from_corr(compute_corr_fft(binary_h), n_points)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = binary_h.copy()
        
        # Swap hill-climb
        t_max = start_time + 0.75 * budget_s
        if time.time() < t_max:
            # Convert to binary with exact sum
            binary_h = (best_h >= 0.5).astype(float)
            ones_count = np.sum(binary_h)
            if ones_count > TARGET_SUM:
                ones_indices = np.where(binary_h == 1)[0]
                flip_count = int(ones_count - TARGET_SUM)
                flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
                binary_h[flip_indices] = 0
            elif ones_count < TARGET_SUM:
                zeros_indices = np.where(binary_h == 0)[0]
                flip_count = int(TARGET_SUM - ones_count)
                flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
                binary_h[flip_indices] = 1
            
            binary_h = project_to_sum_box(binary_h, TARGET_SUM)
            current_c5 = c5_from_corr(compute_corr_fft(binary_h), n_points)
            
            # Store current overlap metrics
            current_corr = compute_corr_fft(binary_h)
            current_min = np.min(current_corr)  # min overlap is negative (so we minimize it)
            # Actually, we compute min corresponding to max C5 (which means we are looking for
            # the most negative value in correlation)
            overlapping_indices = list(range(-n_points+1, n_points))
            current_min_overlap = np.min(current_corr[overlapping_indices])
            
            # Better way: find the smallest positive value
            # Actually, it's sum of h * (1-h shifted) = sum of h - sum of h * h shifted
            # Maximum of that is the smallest min, so we want max correlation
            current_c5_exact = -current_min_overlap # As the correl contains overlaps, the max is the one we want to avoid
            
            if current_c5_exact < best_c5:
                best_c5 = current_c5_exact
                best_h = binary_h.copy()
            
            # Optimized swap procedure
            ones = np.where(binary_h == 1)[0]
            zeros = np.where(binary_h == 0)[0]
            
            # Re-evaluate using exact maximum
            current_c5_exact = c5_from_corr(compute_corr_fft(binary_h), n_points)
            
            max_swaps = 5 * n_points
            swap_count = 0
            
            while time.time() < t_max and swap_count < max_swaps:
                if len(ones) == 0 or len(zeros) == 0:
                    break
                    
                # Pick two random indices
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                
                # Make a temporary copy
                temp_h = binary_h.copy()
                temp_h[i], temp_h[j] = temp_h[j], temp_h[i]
                
                # Evaluate the effect of this swap
                new_corr = compute_corr_fft(temp_h)
                new_c5_exact = c5_from_corr(new_corr, n_points)
                
                # Accept if better
                if new_c5_exact < current_c5_exact:
                    binary_h = temp_h.copy()
                    ones = np.where(binary_h == 1)[0]
                    zeros = np.where(binary_h == 0)[0]
                    current_c5_exact = new_c5_exact
                    if current_c5_exact < best_c5:
                        best_c5 = current_c5_exact
                        best_h = binary_h.copy()
                
                swap_count += 1
            
            # Final evaluation of best solution
            final_corr = compute_corr_fft(best_h)
            final_c5 = c5_from_corr(final_corr, n_points)
            if final_c5 < best_c5:
                best_c5 = final_c5
        
        # Additional random restarts
        t_max = start_time + budget_s
        for _ in range(50):
            if time.time() >= t_max:
                break
            # Generate random binary with correct sum
            rand_h = np.random.rand(n_points)
            rand_h = project_to_sum_box(rand_h, TARGET_SUM)
            # Round to binary
            rand_h = (rand_h >= 0.5).astype(float)
            # Ensure correct sum
            ones_count = np.sum(rand_h)
            if ones_count > TARGET_SUM:
                ones_indices = np.where(rand_h == 1)[0]
                flip_count = int(ones_count - TARGET_SUM)
                flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
                rand_h[flip_indices] = 0
            elif ones_count < TARGET_SUM:
                zeros_indices = np.where(rand_h == 0)[0]
                flip_count = int(TARGET_SUM - ones_count)
                flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
                rand_h[flip_indices] = 1
            rand_h = project_to_sum_box(rand_h, TARGET_SUM)
            rand_c5 = c5_from_corr(compute_corr_fft(rand_h), n_points)
            if rand_c5 < best_c5:
                best_c5 = rand_c5
                best_h = rand_h.copy()
    
    # Fall back to initial solution if nothing found
    if best_h is None:
        # Use the sturmian solution as fallback
        best_h = sturmian_word(n_points)
        best_h = project_to_sum_box(best_h, TARGET_SUM)
        best_c5 = c5_from_corr(compute_corr_fft(best_h), n_points)
    
    return (best_h.tolist(), float(best_c5), n_points)
