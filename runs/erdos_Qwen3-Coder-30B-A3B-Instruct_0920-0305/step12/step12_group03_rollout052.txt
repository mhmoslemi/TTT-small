def compute_correlation(h):
    """Compute full correlation using numpy.correlate"""
    return np.correlate(h, 1-h, mode='full')

def evaluate_c5(h):

def update_correlation_diff(C_raw, h, i, j):
    """Update correlation when swapping h[i] and h[j]"""
    # Create difference vector
    delta = np.zeros_like(C_raw)
    n = len(h)
    
    # For shift k where i+k is valid, subtract contribution from h[i] 
    # (i+k is the first index in h_i * h_{i+k})
    for k in range(n):
        if 0 <= i+k < n:
            delta[k] -= (1 - h[i+k])
    
    # For shift k where i-k is valid, subtract contribution from h[i] 
    # (i-k is the second index in h_{i-k} * h_i)
    for k in range(n):
        if 0 <= i-k < n:
            delta[2*n-2-k] -= h[i-k]
    
    # Add the opposite contributions for j
    for k in range(n):
        if 0 <= j+k < n:
            delta[k] += (1 - h[j+k])
    
    for k in range(n):
        if 0 <= j-k < n:
            delta[2*n-2-k] += h[j-k]
    
    return delta

    start_time = time.time()

        # Try a swap: find a 0 and 1 to exchange

        delta_C = update_correlation_diff(C_raw, h, i, j)
        new_C_raw = C_raw + delta_C
