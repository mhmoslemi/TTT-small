import numpy as np
from scipy.linalg import eigh
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp
import time
import random

def toeplitz_A(lag, n):
    """Generate a sparse Toeplitz matrix A_lag for autocorrelation."""
    data = np.ones(2 * (n - lag))
    i = np.arange(n)
    j = np.arange(lag, n)
    rows = np.concatenate([i, i])
    cols = np.concatenate([j, i])
    return (data, (rows, cols)), n

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n = 128
    dx = 2.0 / n
    N = n // 2
    
    # Precompute all Toeplitz matrices
    A_list = []
    for l in range(1, n):
        data, shape = toeplitz_A(l, n)
        A_list.append(data)
    
    # Multi-stage approach: first solve SDP with n=64, then use for seeding
    n_coarse = 64
    dx_coarse = 2.0 / n_coarse
    N_coarse = n_coarse // 2
    
    # For coarsening, we replicate each coarse point into b = n//n_coarse consecutive fine points
    b = n // n_coarse
    
    # Helper function to compute exact C5 from FFT
    def compute_c5_exact(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return dx * max_corr
    
    # SDP solver with max_iterations limited
    def solve_sdp(A_list, n, N, t_sdp):
        X = cp.Variable((n, n), PSD=True)
        m = cp.Variable()
        
        # Constraints
        constraints = [
            cp.trace(cp.sum([A[0] @ X for A in A_list])) >= m * len(A_list),  # Simplified version
            cp.trace(cp.ones((n, n)) @ X) == N,
            X >= 0,
            X <= 1
        ]
        
        # Objective
        objective = cp.Maximize(m)
        
        # Solve
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-5, verbose=False,
                       time_limit=t_sdp)
            return X.value, m.value
        except:
            return None, None

    # Rounding via hyperplane method
    def round_sdp(X, n, N, seed=None):
        np.random.seed(seed)
        if X is None:
            return None
            
        try:
            # SVD-based rounding
            try:
                # Get eigenvectors/vectors in descending order
                eigenvals, eigenvecs = eigh(X)
                # Keep the first few that have significant eigenvalue
                idx = np.argsort(eigenvals)[::-1]
                eigenvals = eigenvals[idx]
                eigenvecs = eigenvecs[:, idx]
                
                k = min(20, len(eigenvals))
                # Form the low-rank approximation
                V = eigenvecs[:, :k] @ np.diag(np.sqrt(eigenvals[:k]))
            except:
                # Fall back to diagonal
                V = np.diag(np.sqrt(np.diag(X)))
                
            # Random hyperplane rounding
            g = np.random.randn(V.shape[1])
            scores = V @ g
            sorted_indices = np.argsort(-scores)
            h = np.zeros(n)
            h[sorted_indices[:N]] = 1
            
            return h
        except:
            return None

    # Cowling test to get early results with a simpler strategy
    def get_simple_initial():
        h = np.zeros(n)
        idxs = np.random.choice(n, N, replace=False)
        h[idxs] = 1
        return h

    # Stage 1: Try coarse grid SDP for early seeding (if time allows)
    if time.time() - start_time < 0.8 * budget_s:
        # Coarse grid SDP
        coarse_X, coarse_m = solve_sdp(A_list[::n//n_coarse], n_coarse, N_coarse, 0.3 * budget_s)
        if coarse_X is not None and coarse_m is not None:
            try:
                coarse_h = round_sdp(coarse_X, n_coarse, N_coarse, seed=seed)
                if coarse_h is not None:
                    # Expand to fine resolution
                    expanded_h = np.repeat(coarse_h, b)
                    if len(expanded_h) > n:
                        expanded_h = expanded_h[:n]
                    elif len(expanded_h) < n:
                        expanded_h = np.pad(expanded_h, (0, n - len(expanded_h)), mode='constant', constant_values=0)
                    # Adjust sum to target
                    current_sum = np.sum(expanded_h)
                    if current_sum > N:
                        ones_idx = np.where(expanded_h == 1)[0]
                        flip_count = int(current_sum - N)
                        flip_indices = np.random.choice(ones_idx, flip_count, replace=False)
                        expanded_h[flip_indices] = 0
                    elif current_sum < N:
                        zeros_idx = np.where(expanded_h == 0)[0]
                        flip_count = int(N - current_sum)
                        flip_indices = np.random.choice(zeros_idx, flip_count, replace=False)
                        expanded_h[flip_indices] = 1
                    best_h = expanded_h
                    best_c5 = compute_c5_exact(expanded_h)
            except:
                pass
        else:
            # Fallback to simpler solution
            best_h = get_simple_initial()
            best_c5 = compute_c5_exact(best_h)

    # Fallback case: simple run with a random solution
    if best_h is None:
        best_h = get_simple_initial()
        best_c5 = compute_c5_exact(best_h)

    # Hillclimb near the best solution
    def hill_climb_h(s, n, N, max_iters=5000):
        current_h = s.copy()
        current_c5 = compute_c5_exact(current_h)
        best_h = current_h.copy()
        best_c5 = current_c5
        changes = 0
        for i in range(max_iters):
            if time.time() - start_time > budget_s * 0.9:
                break
            ones = np.where(current_h == 1)[0]
            zeros = np.where(current_h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i_swap = np.random.choice(ones)
            j_swap = np.random.choice(zeros)
            new_h = current_h.copy()
            new_h[i_swap] = 0
            new_h[j_swap] = 1
            new_c5 = compute_c5_exact(new_h)
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = new_c5
                changes += 1
            if changes == 1000:
                # Escalate conversion with phase change like pseudo-temperature.
                changes = 0
                if np.random.rand() < 0.1:
                    # Small chance to make even worse move
                    pass
                else:
                    pass
        return best_h, best_c5

    # Apply hill-climb to the seed
    best_h, best_c5 = hill_climb_h(best_h, n, N, max_iters=5000)

    # Final refinement by trying multiple random seeds
    for i in range(5):
        if time.time() - start_time > 0.9 * budget_s:
            break
        h = get_simple_initial()
        h, c5 = hill_climb_h(h, n, N, max_iters=500)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    return (best_h.tolist(), float(best_c5), n)
