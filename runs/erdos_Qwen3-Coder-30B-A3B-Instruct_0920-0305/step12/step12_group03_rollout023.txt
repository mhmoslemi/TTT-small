import numpy as np
import time

def compute_correlation(h):
    """Compute full correlation using numpy correlate"""
    return np.correlate(h, 1 - h, mode='full')

def evaluate_c5(h):
    """Evaluate C5 using full correlation"""
    corr = compute_correlation(h)
    return np.max(corr) * (2.0 / len(h))

def generate_beatty_pattern(n):
    """Generate a Beatty sequence pattern based on golden ratio"""
    phi = (1 + np.sqrt(5)) / 2
    indices = np.floor(np.arange(n) * phi) % n
    indices = np.sort(indices.astype(int))
    h = np.zeros(n)
    h[indices[:n//2]] = 1
    return h

def generate_random_pattern(n):
    """Generate a random binary pattern with exactly n/2 ones"""
    h = np.zeros(n)
    ones_indices = np.random.choice(n, n//2, replace=False)
    h[ones_indices] = 1
    return h

def update_correlation_incremental(h, C_raw, i, j):
    """Update correlation vector incrementally when swapping h[i] and h[j]"""
    # Create delta vector
    delta_C = np.zeros_like(C_raw)
    
    # For shift k where i+k is valid
    for k in range(0, len(h)):
        if i + k < len(h):
            delta_C[k] -= (1 - h[i + k]) * (h[j] - h[i])
        if i - k >= 0:
            delta_C[len(h) - 1 - k] -= h[i - k] * (h[j] - h[i])
            
    return delta_C
