def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def simulated_annealing(h_start, c5_start, start_time, budget_s):
    """Simulated Annealing optimization"""
    h = h_start.copy()
    c5 = c5_start
    temperature = 0.1
    min_temp = 1e-8
    alpha = 0.95
    max_iter = 1000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Create neighbor by perturbing some elements
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, temperature)
        h_new[idx] += delta
        
        # Project to valid constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject based on Metropolis criterion
        if c5_new < c5 or np.random.random() < np.exp(-(c5_new - c5) / temperature):
            h = h_new
            c5 = c5_new
            
        temperature *= alpha
        
        if temperature < min_temp:
            temperature = min_temp
            
    return h, c5

def adaptive_gradient_descent(h_start, c5_start, start_time, budget_s):

    c5 = c5_start
    
    # Simple gradient descent with adaptive step size
    max_iter = 1000
    step_size = 0.01

            break
            
        # Estimate gradient using finite differences
        grad = np.zeros(n_points)
        epsilon = 1e-6
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += epsilon
            h_plus = project_to_sum_box(h_plus, TARGET_SUM)
            
            c5_plus = evaluate_c5(h_plus)
            grad[j] = (c5_plus - c5) / epsilon
            
        # Update with gradient descent
        h_new = h - step_size * grad
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        c5_new = evaluate_c5(h_new)
        
        # Adaptive step size adjustment
        if c5_new < c5:
            step_size *= 1.1  # Increase step size if improvement
            h = h_new
            c5 = c5_new
        else:
            step_size *= 0.5  # Decrease step size
            
        # Prevent step size from becoming too small
        step_size = max(step_size, 1e-8)
        
    return h, c5

def pattern_search(h_start, c5_start, start_time, budget_s):

    h = h_start.copy()
    c5 = c5_start
    max_iter = 2000
    
    # Try systematic perturbations
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Try different patterns of perturbation
        h_new = h.copy()
        
        # Add random perturbations in batches
        batch_size = max(1, n_points // 20)
        indices = np.random.choice(n_points, batch_size, replace=False)
        
        for idx in indices:
            # Try to improve by adjusting the value
            direction = np.random.choice([-1, 1])
            delta = np.random.uniform(0.001, 0.05)
            h_new[idx] += direction * delta
            
        # Project to valid region
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        c5_new = evaluate_c5(h_new)
        
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            
    return h, c5

def direct_optimization(h_start, start_time, budget_s):
    """Direct optimization using scipy minimize"""
    h = h_start.copy()
    
    def objective(h_vec):
        return evaluate_c5(h_vec)
    
    def constraint_eq(h_vec):
        return np.sum(h_vec) - TARGET_SUM
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Try different optimizers
    optimizers = ['SLSQP', 'L-BFGS-B', 'TNC']
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for method in optimizers:
        if time.time() - start_time > budget_s - 5:
            break
            
        try:
            res = minimize(
                objective, 
                h, 
                method=method, 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            
            if res.success:
                h_test = project_to_sum_box(res.x, TARGET_SUM)
                c5_test = evaluate_c5(h_test)
                if c5_test < best_c5:
                    best_h = h_test
                    best_c5 = c5_test
                    
        except:
            continue
            
    return best_h, best_c5
