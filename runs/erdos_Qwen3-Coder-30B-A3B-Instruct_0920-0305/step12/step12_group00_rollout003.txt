import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def balanced_sturmian(n):
    """Generate a balanced Sturmian word as starting point"""
    phi = (np.sqrt(5) - 1) / 2
    x = np.zeros(n, dtype=int)
    for i in range(n):
        if (i * phi) % 1 < 0.5:
            x[i] = 1
    # Ensure exactly n//2 ones
    ones_count = np.sum(x)
    if ones_count > n//2:
        # Find indices with 1s and flip some to 0
        ones_indices = np.where(x == 1)[0]
        flip_count = ones_count - n//2
        flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
        x[flip_indices] = 0
    elif ones_count < n//2:
        # Find indices with 0s and flip some to 1
        zeros_indices = np.where(x == 0)[0]
        flip_count = n//2 - ones_count
        flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
        x[flip_indices] = 1
    return x

def compute_autocorr_fft(x):
    """Compute autocorrelations efficiently using FFT"""
    # Compute correlation using FFT
    x_rev = x[::-1]
    b = x
    L = 2 * len(x) - 1
    corr = np.fft.ifft(np.fft.fft(x_rev, L) * np.fft.fft(b, L)).real
    return corr

def swap_delta(S, i, j, n):
    """Compute change in autocorrelations when swapping positions i and j"""
    delta = np.zeros(n - 1)
    # For each lag ℓ, compute the change in S_ℓ
    for l in range(1, n):
        # Check if indices are valid
        if i - l >= 0:
            delta[l-1] -= x[i-l] * x[j]  # Going from i to j
        if i + l < n:
            delta[l-1] -= x[i] * x[i+l]  # For i+l
        if j - l >= 0:
            delta[l-1] += x[j-l] * x[i]  # From j to i
        if j + l < n:
            delta[l-1] += x[j] * x[j+l]  # For j+l
    return delta

def compute_c5_manual(h):
    """Compute C5 value manually using full convolution"""
    n = len(h)
    dx = 2.0 / n
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    corr = corr[:n-1]  # Only the valid shifts (not full correlation)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    ones_count = np.sum(x)
    if ones_count == target_sum:
        return x
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 128
    dx = 2.0 / n
    N = n // 2
    solver_time_limit = 0.45 * budget_s
    
    # Stage 1: Exact ILP with cvxpy
    best_c5 = float('inf')
    best_h = None
    best_m = None
    
    # Create variables
    x = cp.Variable(n, boolean=True)
    m = cp.Variable(1)
    S = []
    
    # Build constraints
    constraints = []
    
    # Mass constraint
    constraints.append(cp.sum(x) == N)
    
    # Add symmetry-breaking constraints
    constraints.append(x[0] == 1)  # Set first element to 1
    
    # Lexicographic ordering of first half (make first N elements non-increasing)
    for i in range(N - 1):
        constraints.append(x[i] >= x[i+1])
    
    # Auxiliary variables and constraints for autocorrelations
    aux_vars = {}
    for l in range(1, n):
        # We have (n-l) valid i's for this lag
        for i in range(n - l):
            var_name = f'y_{i}_{l}'
            y = cp.Variable(boolean=True)
            aux_vars[(i, l)] = y
            # McCormick constraints for y = x[i] * x[i+l]
            constraints.append(y <= x[i])
            constraints.append(y <= x[i+l])
            constraints.append(y >= x[i] + x[i+l] - 1)
    
    # Set up autocorrelations
    for l in range(1, n):
        S_l = cp.sum([aux_vars[(i, l)] for i in range(n - l)])
        S.append(S_l)
        # Add constraint that S_l >= m
        constraints.append(S_l >= m)
    
    # Objective
    objective = cp.Maximize(m)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CBC, max_seconds=solver_time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_sol = np.round(x.value).astype(int)
            sol_m = m.value
            # Compute actual C5 value
            sol_c5 = compute_c5_manual(x_sol)
            if sol_c5 < best_c5:
                best_c5 = sol_c5
                best_h = x_sol
                best_m = sol_m
    except Exception:
        pass
    
    # Stage 2: Fallback local search with hill climb
    # If no ILP solution found, or we want to refined it
    if best_h is None:
        # Use Sturmian word as fallback
        x = balanced_sturmian(n)
        x = project_to_sum_box(x, N)
        best_h = x
        best_c5 = compute_c5_manual(x)
    
    # Hill climb with greedy swaps
    current_h = best_h.copy()
    current_c5 = best_c5
    ones = set(np.where(current_h == 1)[0])
    zeros = set(np.where(current_h == 0)[0])  
    # Compute initial autocorrelations
    S_vals = compute_autocorr_fft(current_h)
    min_autocorr = min(S_vals[1:]) # Exclude lag 0
    
    # Stages 3 and 4: Hill climb
    max_swaps = 5000
    swap_count = 0
    
    t_end = start_time + 0.9 * budget_s
    while time.time() < t_end and swap_count < max_swaps:
        # Pick a random one and zero for swap
        i = np.random.choice(list(ones))
        j = np.random.choice(list(zeros))
        
        # Compute delta
        delta = swap_delta(S_vals, i, j, n)
        new_S_vals = S_vals + delta
        new_min_autocorr = min(new_S_vals[1:])
        
        if new_min_autocorr > min_autocorr:
            # Accept swap
            current_h[i] = 0
            current_h[j] = 1
            ones.remove(i)
            zeros.remove(j)
            ones.add(j)
            zeros.add(i)
            S_vals = new_S_vals
            min_autocorr = new_min_autocorr
            new_c5 = dx * (N - min_autocorr)
            if new_c5 < current_c5:
                current_c5 = new_c5
                best_h = current_h.copy()
                best_c5 = new_c5
        else:
            # Possibly accept with low probability (SA)
            pass
        swap_count += 1
    
    # Final check with FFT
    if best_h is not None:
        final_c5 = compute_c5_manual(best_h)
    else:
        # Fallback to Sturmian
        final_h = balanced_sturmian(n)
        final_h = project_to_sum_box(final_h, N)
        best_h = final_h
        final_c5 = compute_c5_manual(final_h)
    
    return (best_h.tolist(), float(final_c5), n)
