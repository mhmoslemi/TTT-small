def balanced_sturmian(n):
    """Generate a balanced Sturmian word with n bits and n//2 ones."""
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    x = np.array([int((i * phi) % 1 < 0.5) for i in range(n)])
    # Adjust to ensure exactly n//2 ones
    ones_count = np.sum(x)
    if ones_count > n // 2:
        # Flip some 1s to 0s
        ones_indices = np.where(x == 1)[0]
        flip_indices = np.random.choice(ones_indices, ones_count - n // 2, replace=False)
        x[flip_indices] = 0
    elif ones_count < n // 2:
        # Flip some 0s to 1s
        zeros_indices = np.where(x == 0)[0]
        flip_indices = np.random.choice(zeros_indices, n // 2 - ones_count, replace=False)
        x[flip_indices] = 1
    return x

def compute_autocorr_fft(x):
    """Compute autocorrelations efficiently using FFT."""
    x_rev = x[::-1]
    # Full correlation (includes negative lags)
    corr = correlate(x_rev, x, mode='full')
    # Extract positive lags only, excluding lag 0
    n = len(x)
    # Correlation indices: -n+1 to n-1
    lag_indices = np.arange(-n+1, n)
    # Get lags 1 to n-1
    valid_indices = np.where((lag_indices > 0) & (lag_indices < n))[0]
    return corr[valid_indices]

def swap_delta(S, i, j, n):
    """Compute change in autocorrelations if we swap bits at positions i,j."""
    # Create the effects on autocorrelations
    new_S = S.copy()
    # For lag l, the change is:
    # - x[i] * x[i+l] + x[i-l] * x[i]  →  x[j] * x[j+l] + x[j-l] * x[j]
    # That is: (i+l is affected when x[i] changes)
    # Effect on S_l = - (x[i+l] + x[i-l]) + (x[j+l] + x[j-l])
    for l in range(1, n):
        if i + l < n and i - l >= 0:
            new_S[l-1] += (x[j+l] - x[i+l]) + (x[j-l] - x[i-l])  # lag l is at index l-1
        elif i + l < n:
            new_S[l-1] += (x[j+l] - x[i+l]) - x[i-l]  # lag l is at index l-1
        elif i - l >= 0:
            new_S[l-1] += (x[j-l] - x[i-l]) - x[i+l]  # lag l is at index l-1
        else:  # both out-of-bounds (shouldn't happen in valid indices)
            pass
    return new_S
