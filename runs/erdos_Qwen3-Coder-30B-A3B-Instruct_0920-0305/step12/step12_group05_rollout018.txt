# Global variables
TARGET_SUM = None

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def adaptive_evolutionary_strategy(start_time, budget_s, h_start, best_c5):
    # Use different population sizes and mutation rates
    population_size = 20
    generations = 150
    
    # Adaptive parameters
    mutation_rates = [0.08, 0.06, 0.04, 0.02]
    elite_count = 4
    
    # Generate initial population with diverse perturbations
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        # Mix different types of perturbations
        h = h_start.copy()
        # Add some structured changes
        for i in range(0, n_points, 10):
            if np.random.random() < 0.3:
                h[i] += np.random.normal(0, 0.05)
        # Add Gaussian noise
        noise = np.random.normal(0, 0.03, n_points)
        h += noise
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    
    # Evolutionary optimization
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate fitness
        fitness = [evaluate_c5(h) for h in population]
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        current_best_fitness = fitness[sorted_indices[0]]
        
        if current_best_fitness < best_c5:
            best_c5 = current_best_fitness
            
        # Create new population
        new_population = []
        
        # Keep elite
        for i in range(elite_count):
            new_population.append(population[sorted_indices[i]].copy())
        
        # Generate offspring through mutation with adaptive rate
        current_mutation_rate = mutation_rates[min(gen // 20, len(mutation_rates) - 1)]
        while len(new_population) < population_size:
            # Tournament selection
            parent_idx = sorted_indices[np.random.randint(0, elite_count)]
            parent = population[parent_idx].copy()
            
            # Mutation with adaptive rate
            for i in range(n_points):
                if np.random.random() < current_mutation_rate:
                    parent[i] += np.random.normal(0, 0.05)
            
            # Project to valid region
            parent = project_to_sum_box(parent, TARGET_SUM)
            new_population.append(parent)
        
        population = new_population
    
    return population[sorted_indices[0]], best_c5

def simulated_annealing(start_time, budget_s, h_start, best_c5):
    # Simulated annealing with adaptive temperature
    current_h = h_start.copy()
    current_c5 = evaluate_c5(current_h)
    
    # Initial temperature
    temperature = 0.1
    cooling_rate = 0.95
    min_temperature = 1e-6
    
    # Cooling schedule
    steps = 0
    max_steps = 5000
    
    while temperature > min_temperature and steps < max_steps:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate neighbor solution
        neighbor_h = current_h.copy()
        # Randomly modify some entries
        for i in range(n_points):
            if np.random.random() < 0.1:
                neighbor_h[i] += np.random.normal(0, 0.02)
        
        # Project to valid region
        neighbor_h = project_to_sum_box(neighbor_h, TARGET_SUM)
        
        # Evaluate neighbor
        neighbor_c5 = evaluate_c5(neighbor_h)
        
        # Accept or reject based on Metropolis criterion
        if neighbor_c5 < current_c5:
            current_h = neighbor_h
            current_c5 = neighbor_c5
        else:
            # Accept with probability based on temperature
            delta = neighbor_c5 - current_c5
            if np.random.random() < np.exp(-delta / temperature):
                current_h = neighbor_h
                current_c5 = neighbor_c5
                
        # Lower temperature
        temperature *= cooling_rate
        steps += 1
    
    return current_h, current_c5

def gradient_refinement(start_time, budget_s, h_start, best_c5):
    # Try a smoother gradient-based approach with adaptive learning rate
    h_refined = h_start.copy()
    
    try:
        # Create a simple gradient approximation using finite differences
        def gradient_estimate(h):
            eps = 1e-4
            grad = np.zeros_like(h)
            base_c5 = evaluate_c5(h)
            
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_sum_box(h_plus, TARGET_SUM)
                
                c5_plus = evaluate_c5(h_plus)
                grad[i] = (c5_plus - base_c5) / eps
            
            return grad

        lr = 0.1
        for iter in range(500):
            if time.time() - start_time > budget_s - 5:
                break
                
            grad = gradient_estimate(h_refined)
            h_refined -= lr * grad
            
            # Project to constraints
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            
            # Update learning rate based on progress
            new_c5 = evaluate_c5(h_refined)
            if new_c5 < best_c5:
                best_c5 = new_c5
                lr = min(lr * 1.1, 1.0)  # Increase learning rate
            else:
                lr *= 0.9  # Decrease learning rate
            
    except:
        pass
    
    return h_refined, best_c5

        lambda: np.abs(np.linspace(-1, 1, n_points)),
    ]
    
    best_h = h_start.copy()
    best_c5_pattern = best_c5
    
    for pattern_func in patterns:

            h_pattern = h_pattern * TARGET_SUM / np.sum(h_pattern)

            continue
    
    if best_c5_pattern < best_c5:
