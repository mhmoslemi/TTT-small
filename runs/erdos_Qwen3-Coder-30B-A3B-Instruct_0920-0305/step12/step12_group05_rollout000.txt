import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_softmax(h, beta):
    """Compute gradient of smoothed max correlation"""
    c = compute_corr_fft(h)
    c_max = np.max(c)
    w_raw = np.exp(beta * (c - c_max))
    p = w_raw / np.sum(w_raw)
    
    w_rev = p[::-1]
    conv2 = np.convolve(w_rev, 1 - h, mode='full')
    conv3 = np.convolve(w_rev, h, mode='full')
    
    start_idx = n_points - 1
    end_idx = 2 * n_points - 1
    g = dx * (conv2[start_idx:end_idx] - conv3[start_idx:end_idx])
    return g, c

    num_restarts = 5
    iter_per_restart = 1000
    
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Initialize with random perturbation of best or random
        if restart == 0:
            h = h0.copy()
        else:
            h = np.random.rand(n_points)
            h = project_to_sum_box(h, TARGET_SUM)
        
        beta = 1.0
        eta = 0.1
        
        for iteration in range(iter_per_restart):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradient
            g, c = gradient_softmax(h, beta)
            
            # Line search
            step = eta
            improved = False
            for _ in range(5):
                h_new = project_to_sum_box(h - step * g, TARGET_SUM)
                c5_new = evaluate_c5(h_new)
                if c5_new < evaluate_c5(h):
                    h = h_new
                    improved = True
                    break

            if not improved:

                h = project_to_sum_box(h + 0.01 * np.random.randn(n_points), TARGET_SUM)

            c5_curr = evaluate_c5(h)
            if c5_curr < best_c5:

            # Occasionally restart with best solution + noise
