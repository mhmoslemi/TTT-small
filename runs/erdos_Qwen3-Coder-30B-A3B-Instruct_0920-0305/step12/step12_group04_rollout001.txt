import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT approach"""
    corr = correlate(h, 1 - h, mode="full")
    O = corr * dx
    return np.max(O)

def compute_gradient_for_lag(h, lag):
    """Compute gradient of O(k) with respect to h for a given lag k"""
    N = len(h)
    forward = np.zeros(N)
    backward = np.zeros(N)
    
    if lag >= 0:
        forward[:N - lag] = h[lag:]
        backward[lag:] = h[:N - lag]
    else:
        k = -lag
        forward[k:] = h[:N - k]
        backward[:N - k] = h[k:]
    
    grad = np.ones(N) - forward - backward
    return grad

def subgradient_descent(h_init, target_sum, max_iter, time_limit):
    """Run projected subgradient descent"""
    h = h_init.copy()
    best_h = h.copy()
    best_val = evaluate_c5(h)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > time_limit:
            break
            
        # Evaluate current state
        current_val = evaluate_c5(h)
        
        # Update best if current is better
        if current_val < best_val:
            best_val = current_val
            best_h = h.copy()
        
        # Find lag with maximum overlap
        corr = correlate(h, 1 - h, mode="full")
        O = corr * dx
        max_lag_idx = np.argmax(O)
        lag = max_lag_idx - (len(h) - 1)
        
        # Compute gradient for this lag
        grad = compute_gradient_for_lag(h, lag)
        
        # Adaptive step size
        step_size = 0.5 / np.sqrt(iteration + 1) if iteration > 0 else 0.5
        
        # Update step
        h_new = h - step_size * grad
        h_new = project_to_sum_box(h_new, target_sum)
        
        # Evaluate new state
        new_val = evaluate_c5(h_new)
        
        # Accept the update if it improves the objective
        if new_val < current_val:
            h = h_new
        else:
            # Occasionally accept worse updates to escape local minima
            if np.random.random() < 0.01:
                h = h_new
    
    return best_h, best_val

def upsample_h(h_old, n_points_new):
    """Upsample a step function to a new number of points"""
    old_points = np.linspace(0, 2, len(h_old), endpoint=False)
    new_points = np.linspace(0, 2, n_points_new, endpoint=False)
    h_new = np.interp(new_points, old_points, h_old)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    global TARGET_SUM, n_points, dx
    n_points = 200
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    best_c5 = np.inf
    best_h = None
    
    # Use multiple restarts with different initialization strategies
    restarts = 20
    max_iter_per_restart = 2000
    
    for restart in range(restarts):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Strategy 1: Initialize with upscaled previous solution if available
        if restart == 0 and 'initial_h_values' in globals():
            try:
                h_init = upsample_h(np.array(initial_h_values), n_points)
                h_init = project_to_sum_box(h_init, TARGET_SUM)
            except:
                h_init = np.random.rand(n_points)
        elif restart < 5:
            # Strategy 2: Slightly perturbed previous best
            if best_h is not None:
                h_init = best_h + np.random.normal(scale=0.03, size=n_points)
                h_init = project_to_sum_box(h_init, TARGET_SUM)
            else:
                h_init = np.random.rand(n_points)
        else:
            # Strategy 3: Random initialization
            h_init = np.random.rand(n_points)
            h_init = project_to_sum_box(h_init, TARGET_SUM)
        
        # Run subgradient descent for this restart
        time_limit_per_restart = (budget_s - (time.time() - start_time)) / (restarts - restart)
        h_opt, c5_opt = subgradient_descent(
            h_init, TARGET_SUM, max_iter_per_restart, time_limit_per_restart
        )
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Final refinement with multiple iterations
    if best_h is not None:
        h_opt, c5_opt = subgradient_descent(
            best_h, TARGET_SUM, 1000, budget_s - (time.time() - start_time) - 5
        )
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    if best_h is None:
        # Fallback to a simple approach
        best_h = np.ones(n_points) * 0.5
        best_h = project_to_sum_box(best_h, TARGET_SUM)
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
