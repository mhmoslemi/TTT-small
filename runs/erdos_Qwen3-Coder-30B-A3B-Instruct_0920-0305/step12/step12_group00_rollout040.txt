import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort the vector
    sorted_x = np.sort(x)[::-1]
    
    # Binary search for lambda
    def f(lam):
        projected = np.clip(sorted_x - lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    lambda_val = (a + b) / 2
    result = np.clip(x - lambda_val, 0, 1)
    return result

def softmax_weights(corr, alpha):
    """Compute softmax weights for given correlation and alpha"""
    exp_corr = np.exp(alpha * corr)
    return exp_corr / np.sum(exp_corr)

def grad_softmax(corr, alpha, h):
    """Compute gradient of softmax with respect to h"""
    weights = softmax_weights(corr, alpha)
    # Convolution with h (forward and backward)
    L = len(h)
    L_fft = 1 << (L - 1).bit_length()
    W_fft = np.fft.fft(weights, L_fft)
    h_fft = np.fft.fft(h, L_fft)
    forward_conv = np.fft.ifft(W_fft * h_fft).real[:L]
    backward_conv = np.fft.ifft(np.fft.fft(weights[::-1], L_fft) * h_fft).real[:L]
    return 1.0 - forward_conv - backward_conv

def adam_step(h, m, v, grad, t, alpha, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step"""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    new_h = h - alpha * m_hat / (np.sqrt(v_hat) + eps)
    return new_h, m, v

def swap_update(old_corr, S, i, j, h_bin, n):
    """Update correlations after swapping h[i] and h[j]"""
    # Calculate the change in correlation values due to the swap
    delta_S = np.zeros_like(old_corr)
    for l in range(-n+1, n):
        # Handle out-of-bound indices
        if i+l >= 0 and i+l < n and j+l >= 0 and j+l < n:
            delta_S[l+n-1] = -h_bin[i]*h_bin[i+l] - h_bin[i]*h_bin[i-l] - h_bin[j]*h_bin[j+l] - h_bin[j]*h_bin[j-l]
            delta_S[l+n-1] += h_bin[j]*h_bin[i+l] + h_bin[j]*h_bin[i-l] + h_bin[i]*h_bin[j+l] + h_bin[i]*h_bin[j-l]
        elif i+l >= 0 and i+l < n:
            delta_S[l+n-1] = -h_bin[i]*h_bin[i+l] - h_bin[i]*h_bin[i-l] 
            delta_S[l+n-1] += h_bin[j]*h_bin[i+l] + h_bin[j]*h_bin[i-l]
        elif j+l >= 0 and j+l < n:
            delta_S[l+n-1] = -h_bin[j]*h_bin[j+l] - h_bin[j]*h_bin[j-l] 
            delta_S[l+n-1] += h_bin[i]*h_bin[j+l] + h_bin[i]*h_bin[j-l]
            
    return delta_S

def create_balanced_sturmian(n, phi=(np.sqrt(5)-1)/2):
    """Create a balanced Sturmian sequence"""
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1 if (i * phi) % 1 < 0.5 else 0
    return h.astype(float)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    TARGET_SUM = n_points // 2
    
    # Time allocation
    optimisation_time = 0.45 * budget_s
    binary_time = 0.35 * budget_s
    restart_time = 0.15 * budget_s
    final_time = 0.05 * budget_s
    
    # Generate initial solutions
    initial_h = []
    
    # 1. Balanced Sturmian word
    h_sturmian = create_balanced_sturmian(n_points)
    h_sturmian = project_to_sum_box(h_sturmian, TARGET_SUM)
    initial_h.append(("sturmian", h_sturmian.copy()))
    
    # 2. Random initialization
    h_random = np.random.rand(n_points)
    h_random = project_to_sum_box(h_random, TARGET_SUM)
    initial_h.append(("random", h_random.copy()))
    
    # 3. Perturbed balanced word  
    h_perturbed = h_sturmian + 0.05 * (np.random.rand(n_points) - 0.5)
    h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
    initial_h.append(("perturbed", h_perturbed.copy()))
    
    best_c5 = float('inf')
    best_h = None
    
    # Optimisation with Adam
    t_end_opt = start_time + optimisation_time
    for name, h in initial_h:
        if time.time() >= t_end_opt:
            break
            
        # Adam parameters
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        alpha = 0.12  # Learning rate
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        
        # Smooth max parameters
        alpha_min = 4.0
        alpha_max = 200.0
        alpha_growth = 1.5
        t_start = time.time()
        
        current_h = h.copy()
        current_c5 = c5_from_corr(compute_corr_fft(current_h), n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        for t in range(1, 8000):  # Max iterations
            if time.time() >= t_end_opt:
                break
                
            # Compute correlation
            corr = compute_corr_fft(current_h)
            current_c5 = c5_from_corr(corr, n_points)
            
            # Update alpha (soft max smoothness)
            alpha_t = min(alpha_min + (alpha_max - alpha_min) * (1 - np.exp(-0.01 * t)), alpha_max)
            
            # Compute gradient
            grad = grad_softmax(corr, alpha_t, current_h)
            
            # Adam step
            new_h, m, v = adam_step(current_h, m, v, grad, t, alpha)
            
            # Project onto constraints
            new_h = project_to_sum_box(new_h, TARGET_SUM)
            
            # Update current solution
            current_h = new_h.copy()
            current_c5 = c5_from_corr(compute_corr_fft(current_h), n_points)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
    
    # Binary refinement
    t_end_binary = start_time + optimisation_time + binary_time
    if best_h is not None and time.time() < t_end_binary:
        # Round to binary
        median_val = np.median(best_h)
        binary_h = (best_h >= median_val).astype(float)
        
        # Adjust sum
        ones_count = np.sum(binary_h)
        if ones_count > TARGET_SUM:
            ones_indices = np.where(binary_h == 1)[0]
            flip_count = int(ones_count - TARGET_SUM)
            flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
            binary_h[flip_indices] = 0
        elif ones_count < TARGET_SUM:
            zeros_indices = np.where(binary_h == 0)[0]
            flip_count = int(TARGET_SUM - ones_count)
            flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
            binary_h[flip_indices] = 1
        
        binary_h = project_to_sum_box(binary_h, TARGET_SUM)
        current_c5 = c5_from_corr(compute_corr_fft(binary_h), n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = binary_h.copy()
            
        # Swap hill-climb
        iteration = 0
        patience = 0
        max_patience = 5 * n_points
        while time.time() < t_end_binary:
            if patience > max_patience:
                break
                
            # Find 1s and 0s positions
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
            
            if len(ones) == 0 or len(zeros) == 0:
                break
                
            # Pick two random indices - one from 1s and one from 0s
            i_idx = np.random.randint(len(ones))
            j_idx = np.random.randint(len(zeros))
            
            i = ones[i_idx]
            j = zeros[j_idx]
            
            # Apply swap
            temp_h = best_h.copy()
            temp_h[i] = 0
            temp_h[j] = 1
            
            # Calculate new correlations
            new_corr = compute_corr_fft(temp_h)
            new_c5 = c5_from_corr(new_corr, n_points)
            
            if new_c5 < best_c5:
                best_h = temp_h.copy()
                best_c5 = new_c5
                patience = 0  # Reset patience
            else:
                patience += 1
                
            iteration += 1
    
    # Additional random restarts
    t_end_final = start_time + optimisation_time + binary_time + restart_time
    if time.time() < t_end_final:
        for _ in range(10):
            if time.time() >= t_end_final:
                break
                
            # Generate random binary with the right sum
            h_rand = np.random.rand(n_points)
            h_rand = project_to_sum_box(h_rand, TARGET_SUM)
            
            # Round to binary
            median_val = np.median(h_rand)
            binary_h = (h_rand >= median_val).astype(float)
            
            # Adjust sum
            ones_count = np.sum(binary_h)
            if ones_count > TARGET_SUM:
                ones_indices = np.where(binary_h == 1)[0]
                flip_count = int(ones_count - TARGET_SUM)
                flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
                binary_h[flip_indices] = 0
            elif ones_count < TARGET_SUM:
                zeros_indices = np.where(binary_h == 0)[0]
                flip_count = int(TARGET_SUM - ones_count)
                flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
                binary_h[flip_indices] = 1
                
            binary_h = project_to_sum_box(binary_h, TARGET_SUM)
            trial_c5 = c5_from_corr(compute_corr_fft(binary_h), n_points)
            if trial_c5 < best_c5:
                best_c5 = trial_c5
                best_h = binary_h.copy()
    
    if best_h is None:
        # Return initial solution if no better solution was found
        h_sturmian = create_balanced_sturmian(n_points)
        h_sturmian = project_to_sum_box(h_sturmian, TARGET_SUM)
        best_h = h_sturmian.copy()
        best_c5 = c5_from_corr(compute_corr_fft(best_h), n_points)
    
    return (best_h.tolist(), float(best_c5), n_points)
