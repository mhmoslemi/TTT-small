# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Simple projection method
    x = np.array(x)
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x
    
    # Linear programming approach for sum constraint
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase some values
        idx = np.argsort(x)
        for i in range(len(idx)-1, -1, -1):
            if diff <= 0:
                break
            available = 1.0 - x[idx[i]]
            increase = min(available, diff)
            x[idx[i]] += increase
            diff -= increase
    else:
        # Need to decrease some values
        idx = np.argsort(x)
        for i in range(len(idx)):
            if diff >= 0:
                break
            available = x[idx[i]]
            decrease = min(available, -diff)
            x[idx[i]] -= decrease
            diff += decrease
    
    return x
