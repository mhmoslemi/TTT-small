import numpy as np
import time

def compute_correlation(h):
    """Compute full correlation of h with 1-h using numpy.correlate"""
    return np.correlate(h, 1 - h, mode='full')

def evaluate_solution(h, dx):
    """Evaluate the C5 bound for given h"""
    corr = compute_correlation(h)
    return dx * np.max(corr)

def get_binary_pattern(n):
    """Generate a low-discrepancy binary pattern using Beatty sequence"""
    phi = (1 + np.sqrt(5)) / 2
    indices = np.floor(np.arange(n) * phi).astype(int) % n
    indices = np.sort(indices)
    h = np.zeros(n)
    h[indices[:n//2]] = 1
    return h

def update_correlation_delta(C_raw, h, i, j):
    """Update correlation vector when swapping h[i] and h[j]"""
    delta = np.zeros_like(C_raw)
    
    # i changes from 0 to 1
    # For each k where i+k is valid
    for k in range(1, min(n, n - i)):
        delta[k] -= (1 - h[i + k])
        # k in C_raw corresponds to shift k-n+1 in original
    # For each k where i-k is valid  
    for k in range(1, min(i + 1, n)):
        delta[-k] -= h[i - k]
    
    # j changes from 1 to 0
    # For each k where j+k is valid
    for k in range(1, min(n, n - j)):
        delta[k] += (1 - h[j + k])
        # k in C_raw corresponds to shift k-n+1 in original
    # For each k where j-k is valid
    for k in range(1, min(j + 1, n)):
        delta[-k] += h[j - k]
        
    return delta

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 120
    dx = 2.0 / n
    
    # Generate initial patterns
    patterns = generate_initial_patterns(n)
    
    best_h = None
    best_val = float('inf')
    
    # Find the best among initial patterns
    for h in patterns:
        val = evaluate_solution(h, dx)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    
    # Initialize correlation for best pattern
    C_raw = compute_correlation(best_h)
    cur_max = np.max(C_raw)
    
    # Simulated annealing parameters
    T = 0.5
    alpha = 0.995
    T_min = 1e-5
    N_swap = n
    
    swaps = 0
    last_update = 0
    
    while time.time() - start_time < budget_s - 10:
        # Try a random swap
        i = np.random.randint(n)
        # Find a j with opposite value
        j_candidates = np.where(best_h == 1 - best_h[i])[0]
        if len(j_candidates) == 0:
            continue
        j = np.random.choice(j_candidates)
        
        # Compute delta correlation
        delta_C = update_correlation_delta(C_raw, best_h, i, j)
        new_C = C_raw + delta_C
        new_max = np.max(new_C)
        
        # Accept or reject the move
        if new_max < cur_max or np.random.rand() < np.exp(-(new_max - cur_max) / T):
            # Apply swap
            best_h[i], best_h[j] = best_h[j], best_h[i]
            C_raw = new_C
            cur_max = new_max
            
            # Update global best if needed
            if new_max < np.max(compute_correlation(best_h)):
                best_h = best_h.copy()
                C_raw = new_C
                cur_max = new_max
        
        swaps += 1
        
        # Cooling schedule
        if swaps - last_update > N_swap:
            T *= alpha
            last_update = swaps
            if T < T_min:
                break
    
    # Final hill-climbing
    improved = True
    while improved and time.time() - start_time < budget_s - 5:
        improved = False
        for i in range(n):
            if best_h[i] == 0:  # Look for 0 to swap with 1
                for j in range(n):
                    if best_h[j] == 1:  # Swap with 1
                        delta_C = update_correlation_delta(C_raw, best_h, i, j)
                        new_C = C_raw + delta_C
                        new_max = np.max(new_C)

                            # Apply swap
                            best_h[i], best_h[j] = best_h[j], best_h[i]
                            C_raw = new_C
                            cur_max = new_max
                            improved = True
                            break
                if improved:
                    break
    
    # Final evaluation
    final_val = evaluate_solution(best_h, dx)
    
    return (best_h.tolist(), float(final_val), n)
