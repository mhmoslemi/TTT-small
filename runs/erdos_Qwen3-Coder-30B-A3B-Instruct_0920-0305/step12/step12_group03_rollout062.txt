import numpy as np
import time

def compute_correlation(h):
    """Compute full correlation of h with (1-h)"""
    corr = np.correlate(h, 1 - h, mode='full')
    return corr

def init_pattern(n=120):
    """Initialize binary pattern with Beatty sequence"""
    phi = (1 + np.sqrt(5)) / 2
    positions = np.floor(np.arange(n) * phi) % n
    positions = np.sort(positions.astype(int))
    
    h = np.zeros(n)
    h[positions[:n//2]] = 1
    
    return h
