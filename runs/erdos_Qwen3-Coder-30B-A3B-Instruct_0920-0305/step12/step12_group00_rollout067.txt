import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import SCS
from scipy.sparse import diags, csr_matrix
import math

n_points = 128
dx = 2.0 / n_points
TARGET_SUM = n_points / 2
# Precompute Toeplitz matrices for all lags
def compute_toeplitz_matrices(n):
    A_matrices = {}
    for lag in range(1, n):
        # Create Toeplitz matrix for lag
        I = np.arange(n)
        J = np.arange(n)
        rows = np.broadcast_to(I, (n, n))
        cols = np.broadcast_to(J, (n, n)).T
        mask = np.abs(rows - cols) == lag
        A = mask.astype(float)
        A_matrices[lag] = csr_matrix(A)
    return A_matrices

# Precompute once
A_matrices = compute_toeplitz_matrices(n_points)
J = np.ones((n_points, n_points))  # All-ones matrix for trace constraint
J_csr = csr_matrix(J)

def evaluate_c5_exact(h):
    """Evaluate C5 using FFT for exact correlation"""
    # Compute h(x) * (1 - h(x+k)) for all lags k
    h_shifted = np.concatenate([h, np.zeros(n_points-1)])
    ones = np.ones_like(h)
    ones_shifted = np.concatenate([ones, np.zeros(n_points-1)])
    
    # Because we're doing convolution using FFT, we will compute
    # the auto-correlations of the form int h(x)(1-h(x+k))dx = 
    # sum_{i=0..n-1} h_i * (1 - h_{i+k})
    h_convol = correlate(h, ones[::-1], mode='full')  # h * ones
    h_sq_convol = correlate(h, h[::-1], mode='full')   # h * h
    
    # The variance of h(t) (sum of h_i h_{i+k} to construct correlation)
    # C_5 = max_k int h(x)(1-h(x+k)) dx = max_k (N - S_k) where S_k = sum h_i h_{i+k}
    # Using conjugate correlation trick:
    # c_k = sum h_i (1 - h_{i+k}) = N - sum h_i h_{i+k} 
    # So max c_k = max (N - S_k) = N - min S_k
    corr = np.concatenate([np.zeros(n_points-1), h_sq_convol[:n_points] - h_convol[n_points-1:]])
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project to L1 ball ball [0,1] with target sum"""
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-8:
        return x
    
    # Binary search for lambda to shift
    a, b = -1.0, 1.0
    tol = 1e-10
    
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_sdp(n, A_dict, J_csr, target_sum, time_limit):
    # This is a max-min linear SDP
    # maximize m
    # s.t.  trace(A_ell * X) >= m  for all ell
    #       trace(J * X) == target_sum
    #       0 <= X_ij <= 1
    #       X is PSD  
    X = cp.Variable((n, n), symmetric=True)
    m = cp.Variable()
    
    constraints = [
        cp.trace(J_csr @ X) == target_sum,
        X >> 0,
        0 <= X,
        X <= 1
    ]
    
    for i in range(1, n):
        constraints.append(cp.trace(A_dict[i] @ X) >= m)
    
    objective = cp.Maximize(m)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=SCS, max_iters=5000, eps=1e-5, verbose=False,
                   time_limit=time_limit)
    except:
        prob.solve(solver=SCS, max_iters=5000, eps=1e-5, verbose=False,
                   time_limit=time_limit)
    
    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
        return X.value, m.value
    else:
        return X.value, None

def round_sdp_solution(X_val, n_points, target_sum, n_rounds=5):
    if X_val is None:
        return None
    
    # Perform eigenvalue decomposition
    try:
        # Extract eigenpairs with non-negligible eigenvalues
        u, s, vt = np.linalg.svd(X_val, full_matrices=False)
        # Keep only positive eigenvalues
        valid_indices = s > 1e-8
        u_trim = u[:, valid_indices]
        s_trim = s[valid_indices]
        
        if len(s_trim) == 0:
            # Fallback:istribute the total mass evenly 
            h = np.zeros(n_points)
            h[:int(target_sum)] = 1
            return h

        v = u_trim * np.sqrt(s_trim)
        
        best_h = None
        best_c5 = float('inf')
        for i in range(n_rounds):
            # Sample Gaussian vector
            g = np.random.randn(v.shape[1])
            scores = v @ g
            ordered_idx = np.argsort(-scores)  # descending order
            h_round = np.zeros(n_points)
            h_round[ordered_idx[:int(target_sum)]] = 1
            
            #Project to exact sum
            h_round = project_to_sum_box(h_round, target_sum)
            
            c5 = evaluate_c5_exact(h_round)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_round.copy()
                
        return best_h
    except Exception:
        # Fallback to diagonal approach
        scores = np.diag(X_val)
        indices = np.argsort(-scores)
        h = np.zeros(n_points)
        h[indices[:int(target_sum)]] = 1
        h = project_to_sum_box(h, target_sum)
        return h

def perform_swap_hill_climb(h, A_dict, n_points, target_sum, max_swaps=1000, time_limit=None):
    start_time = time.time()
    h = h.copy()
    # Compute initial S values for all lags
    # S_k = Sum_i h_i * h_{i+k}
    # We can compute this using FFT trick:
    # Correlation of h with itself, but also we need to compute all shifts correctly
    
    # Using more efficient representation:
    corr_fft = np.fft.fft(h)
    corr_full = np.fft.ifft(corr_fft * np.conj(corr_fft)).real
    
    # Contribution of h_i * h_{i+k} is in corr_full shifted appropriately
    # To compute for each k = 1..n-1 we take:
    # S_k = \sum_i h_i \cdot h_{i+k} = corr_full(n-1+k) = corr_full[n-1+k]
    # But we want S_k for k=1..n-1 which gives corr_full[n, n+1, ..., 2n-2]
    # But we want to discount wrap-around and handle shifts correctly.
    # Let's directly compute S
    # S[k] = sum_{i=0}^{n-1-k} h[i] * h[i+k]  (for k=0..n-2)
    # Though we only care about k>=1 so ignore k=0
    # S[k] = [h[i] * h[i+k] for i in 0..(n-1-k)]

    S = np.zeros(n_points-1)  # only positive lags 1...n-1
    for k in range(1, n_points):
        if k >= n_points:
            break
        s = 0
        for i in range(n_points - k):
            s += h[i] * h[i+k]
        S[k-1] = s

    min_S = np.min(S)
    if time_limit is not None:
        time_budget = time_limit 
    else:
        time_budget = 100000

    swaps = 0
    improved = True
    
    while improved and swaps < max_swaps:
        if time_limit is not None and (time.time() - start_time) > time_budget:
            break
        improved = False
        for _ in range(20):
            # Pick random i in ones, j in zeros
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Current S values
            old_S = S.copy()
            # Update S vector from the swap of i and j
            # We compute diffusion for all lags k:
            # Change in each lag:
            # For all k: delta_S_k = (h[i]-h[i+k]) - (h[j]-h[j+k])
            # But numbering is confusing, let's make: 
            # Let's just compute exact change in each S_k
                
            # In computing S_k = sum h_i * h_{i+k} 
            # Any change from exchanging position i and j:
            # For k such that i+k < n: 
            #     S_k now gains h_ij if h_i=1 and h_j=0 (i,j were in swap)
            delta_S = np.zeros(n_points-1)
            # One half - change occurring with i
            for k in range(1, n_points):
                if i+k < n_points:
                    delta_S[k-1] += h[i] - h[i+k]  # old before -> new after
                if i-k >= 0:
                    delta_S[k-1] += h[i-k] - h[i]   # current one is wrong indexing
                # Reindex this check for each k...
                # Or smarter way:
                # When we swap i,j:
                # accurate contributions for i: h_i * h_j = 1*0 = 0
                # accurate contributions for j: h_i * h_j = 1*0 = 0 
                # Additionally if k is such that i+k<n (we count that new değeri)
                updated_S = old_S.copy()
                # Update for all compatible k
                for k in range(1, n_points):
                    # S_k = sum h_i h_{i+k}
                    # We affect h_i h_{i+k}, h_j h_{j+k} terms
                    if i+k < n_points:
                        updated_S[k-1] -= h[i]*h[i+k]  # before: h[i]*h[i+k]
                        updated_S[k-1] += h[j]*h[j+k]  # after:  h[j]*h[j+k]
                    if j+k < n_points:
                        updated_S[k-1] -= h[j]*h[j+k]  # before: h[j]*h[j+k]
                        updated_S[k-1] += h[i]*h[i+k]  # after: h[i]*h[i+k]
                    # For back parts:
                    if i-k >= 0:
                        # old contribution of h[i]*h[i-k]
                        updated_S[k-1] -= h[i]*h[i-k]
                        # new: h[j]*h[j-k]
                        updated_S[k-1] += h[j]*h[j-k]
                    if j-k >= 0:
                        updated_S[k-1] -= h[j]*h[j-k] 
                        updated_S[k-1] += h[i]*h[i-k]
                # Update S array and min_S computation
                
                # Redo properly with specific safe updates.
                # Take a framework approach
                
                # One direct way: enumerate exact update for all significant k
                # The update to a lag is: 
                # When we swap positions i and j
                # If k such that i+k < n and j+k < n:
                #     S_k decreases by h[i]*h[i+k] + h[j]*h[j+k] and 
                #     increases by h[j]*h[i+k] + h[i]*h[j+k] 
                # The net change is:
                # (h[i]*h[i+k] + h[j]*h[j+k]) - (h[j]*h[i+k] + h[i]*h[j+k])
                # = h[i]*(h[i+k] - h[j+k]) + h[j]*(h[j+k] - h[i+k])
                # = (h[i] - h[j])*(h[i+k] - h[j+k])
                # For other terms (i-k, j-k), compute analogously.
                #
                # Simplify by focusing on just updating effective shifts
                
                delta_all = np.zeros(n_points-1)  # New problem is really tighter!
                # Direct for overlapping intervals
                for k in range(1, n_points):
                    if i + k < n_points and j + k < n_points:
                        delta_all[k-1] += (h[i]*h[i+k] + h[j]*h[j+k]) - (h[j]*h[i+k] + h[i]*h[j+k])
                    if i - k >= 0 and j - k >= 0:
                        delta_all[k-1] += (h[i]*h[i-k] + h[j]*h[j-k]) - (h[j]*h[i-k] + h[i]*h[j-k])
                
                # If zero delta, skip
                if np.allclose(delta_all, 0.0):
                    continue
                # Update all new values
                new_S = old_S + delta_all
                new_min_S = np.min(new_S) if len(new_S) > 0 else 0.0
                
                # if new_min_S > min_S we accept this swap
                if new_min_S > min_S:
                    # Accept this swap
                    h[i], h[j] = h[j], h[i]
                    S = new_S
                    min_S = new_min_S
                    improved = True
                    swaps += 1
                    break
            if improved:
                break
                
            if time_limit is not None and (time.time() - start_time) > time_budget:
                break  
    return h, evaluate_c5_exact(h)

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    best_c5 = float('inf')
    best_h = None

    # Solve SDP
    TIME_SDP = 0.4 * budget_s  # 240s for 600s
    print("Solving SDP...")
    X_val, m_opt = solve_sdp(n_points, A_matrices, J_csr, TARGET_SUM, TIME_SDP)
    print(f"SDP solved, m_opt = {m_opt}")
    
    # Rounding strategy 
    print("Rounding SDP solution...")
    h_rounded = round_sdp_solution(X_val, n_points, TARGET_SUM, n_rounds=5)
    if h_rounded is not None:
        h_rounded = project_to_sum_box(h_rounded, TARGET_SUM)
        c5_rounded = evaluate_c5_exact(h_rounded)
        print(f"Rounded Solution C5 = {c5_rounded}")
        if c5_rounded < best_c5:
            best_c5 = c5_rounded
            best_h = h_rounded.copy()
    
    # Hoop-cloping using local hill climb
    TIME_HC = max(0.2 * budget_s, 50)  # At least 50
    print("Performing swap hill climb...")
    if best_h is not None:
        h_climbed, c5_climbed = perform_swap_hill_climb(best_h, A_matrices, n_points, TARGET_SUM, max_swaps=5000, time_limit=TIME_HC)
        if c5_climbed < best_c5:
            best_c5 = c5_climbed
            best_h = h_climbed.copy()
    print(f"Best after local search is {best_c5}")
    
    # Random binary restarts
    TIME_RAND = 0.1 * budget_s
    rand_start = time.time() + TIME_SDP + TIME_HC + TIME_RAND
    for i in range(5):
        if rand_start + TIME_RAND < time.time():
            break
        h_random = np.random.rand(n_points)
        h_random = project_to_sum_box(h_random, TARGET_SUM)
        h_random = (h_random > 0.5).astype(float)
        c5_random = evaluate_c5_exact(h_random)
        if c5_random < best_c5:
            best_c5 = c5_random
            best_h = h_random.copy()
            print(f"Improved Random solution C5 = {best_c5}")
    
    return (best_h.tolist(), float(best_c5), n_points)
