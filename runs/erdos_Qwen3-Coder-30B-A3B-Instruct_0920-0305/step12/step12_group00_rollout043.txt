import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
import math

def balanced_sturmian(n, phi=(math.sqrt(5)-1)/2):
    """Generate a balanced Sturmian word with parameter phi."""
    x = np.array([int((i * phi) % 1 < 0.5) for i in range(n)])
    # Adjust to ensure exactly n//2 ones
    current_sum = np.sum(x)
    target_sum = n // 2
    if current_sum < target_sum:
        # Flip zeros to ones
        zero_indices = np.where(x == 0)[0]
        flip_count = target_sum - current_sum
        flip_indices = np.random.choice(zero_indices, flip_count, replace=False)
        x[flip_indices] = 1
    elif current_sum > target_sum:
        # Flip ones to zeros
        one_indices = np.where(x == 1)[0]
        flip_count = current_sum - target_sum
        flip_indices = np.random.choice(one_indices, flip_count, replace=False)
        x[flip_indices] = 0
    return x

def compute_autocorr_fft(x):
    """Compute autocorrelations via FFT."""
    n = len(x)
    x_rev = x[::-1]
    b = 1 - x
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(x_rev, L) * np.fft.fft(b, L)).real
    return corr

def swap_delta(S, i, j, n):
    """Compute the change in autocorrelations for a swap of bits i and j."""
    delta = np.zeros(n)
    # Compute effects for each lag
    for l in range(1, n):
        # If i is within range for lag l
        if i + l < n:
            delta[l] += x[j + l] - x[i + l]
        if i >= l:
            delta[l] += x[j - l] - x[i - l]
        # If j is within range for lag l
        if j + l < n:
            delta[l] -= x[i + l] - x[j + l]
        if j >= l:
            delta[l] -= x[i - l] - x[j - l]
    return delta

def evaluate_c5(h):
    """Evaluate C5 from the given step function."""
    dx = 2.0 / len(h)
    corr = compute_autocorr_fft(h)
    max_corr = np.max(corr)
    return dx * (len(h)/2 - max_corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_list = [128, 144]
    best_c5 = float('inf')
    best_h = None
    best_m = None
    
    for n in n_list:
        dx = 2.0 / n
        N = n // 2
        
        # Build ILP model
        x = cp.Variable(n, boolean=True)
        m = cp.Variable()
        y = {}
        S = {}
        
        # Add auxiliary variables and constraints
        for l in range(1, n):
            for i in range(n - l):
                y[i,l] = cp.Variable(boolean=True)
        # Add McCormick constraints
        for l in range(1, n):
            for i in range(n - l):
                # y[i,l] <= x[i], y[i,l] <= x[i+l]
                # y[i,l] >= x[i] + x[i+l] - 1
                constraints = [
                    y[i,l] <= x[i],
                    y[i,l] <= x[i+l],
                    y[i,l] >= x[i] + x[i+l] - 1
                ]
        
        # Add constraints for S_l
        for l in range(1, n):
            S[l] = cp.sum([y[i,l] for i in range(n-l)])
            constraints.append(S[l] >= m)
            
        # Mass constraint
        mass_constraint = cp.sum(x) == N
        
        # Symmetry-breaking constraints
        symmetry_constraints = [
            x[0] == 1,
            cp.diff(x[:N]) <= 0  # x[0] >= x[1] >= ... >= x[N-1]
        ]
        
        # Objective
        objective = cp.Maximize(m)
        
        prob = cp.Problem(objective, 
                          [mass_constraint] + 
                          [c for l in range(1, n) for i in range(n-l) for c in [y[i,l] <= x[i], y[i,l] <= x[i+l], y[i,l] >= x[i] + x[i+l] - 1]] +
                          [S[l] >= m for l in range(1, n)] +
                          symmetry_constraints)
        
        # Solve with time limit
        try:
            prob.solve(solver=cp.CBC, max_seconds=0.45 * budget_s)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                x_val = np.round(x.value).astype(int)
                if x_val is not None and np.sum(x_val) == N:
                    # Recompute C5 exactly
                    corr_full = compute_autocorr_fft(x_val)
                    max_corr = np.max(corr_full)
                    c5 = dx * (N - max_corr)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = x_val
                        best_m = m.value
        except Exception:
            pass
            
        # If we have a solution or still time, do post-processing
        if best_h is not None and time.time() - start_time < 0.9 * budget_s:
            break
    
    # Fallback to local search if no integer solution from ILP
    if best_h is None:
        # Start with a Sturmian word
        h0 = balanced_sturmian(n)
        best_h = h0.copy()
        best_c5 = evaluate_c5(best_h)
    
    # Post-processing with local search
    t_end = start_time + 0.35 * budget_s
    x = best_h.copy()
    S = compute_autocorr_fft(x)
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    current_m = np.min(S[1:])
    
    # Greedy hill-climb
    while time.time() < t_end:
        for _ in range(5 * n):
            if time.time() >= t_end:
                break
            # Pick random (i in ones, j in zeros)
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            # Compute delta
            delta = swap_delta(S, i, j, n)
            new_S = S + delta  
            new_m = np.min(new_S[1:])
            
            if new_m > current_m:
                # Accept swap
                x[i], x[j] = x[j], x[i]
                S = new_S
                current_m = new_m
                c5 = dx * (N - current_m)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = x.copy()
                # Rebuild support lists (optimization is not needed at each step)
                ones = np.where(x == 1)[0]
                zeros = np.where(x == 0)[0]
            elif np.random.rand() < 0.01:  # Simulated annealing
                # Accept with some probability
                prob = max(0, min(1, np.exp((new_m - current_m) * 0.1)))
                if np.random.rand() < prob:
                    x[i], x[j] = x[j], x[i]
                    S = new_S
                    current_m = new_m
                    c5 = dx * (N - current_m)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = x.copy()
                    ones = np.where(x == 1)[0]
                    zeros = np.where(x == 0)[0]
    
    # Add a final fallback if needed
    if best_h is None:
        # Fallback to random binary with correct sum
        rand_h = np.random.rand(n)
        rand_h = np.round(0.5 * (1 + np.sign(rand_h - 0.5))).astype(int)
        # Ensure correct mass
        ones_count = np.sum(rand_h)
        if ones_count < N:
            zeros_indices = np.where(rand_h == 0)[0]
            flip_indices = np.random.choice(zeros_indices, N - ones_count, replace=False)
            rand_h[flip_indices] = 1
        elif ones_count > N:
            ones_indices = np.where(rand_h == 1)[0]
            flip_indices = np.random.choice(ones_indices, ones_count - N, replace=False)
            rand_h[flip_indices] = 0
        best_h = rand_h
        best_c5 = evaluate_c5(best_h)
        
    return (best_h.tolist(), float(best_c5), n)
