import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Start with a random pattern and optimize using simulated annealing
    h0 = np.random.rand(n_points)
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    # Annealing parameters
    max_iter = 500000
    initial_temp = 0.1
    cooling_rate = 0.9995
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    current_h = h0.copy()
    current_c5 = best_c5
    temp = initial_temp
    
    # Simulated Annealing
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate neighbor solution
        h_new = current_h.copy()
        # Randomly select two indices to swap values
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        # Swap values
        h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
        # Project to maintain constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject based on Metropolis criterion
        if c5_new < current_c5 or np.random.rand() < np.exp(-(c5_new - current_c5) / temp):
            current_h = h_new
            current_c5 = c5_new
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Cool down temperature
        temp *= cooling_rate
        
        # Occasionally do local search improvement
        if i % 1000 == 0 and i > 0:
            # Try several local improvements
            for _ in range(100):
                if time.time() - start_time > budget_s - 5:
                    break
                h_test = current_h.copy()
                # Make small random changes
                for _ in range(5):
                    idx = np.random.randint(0, n_points)
                    h_test[idx] = np.clip(h_test[idx] + np.random.normal(0, 0.02), 0, 1)
                h_test = project_to_sum_box(h_test, TARGET_SUM)
                c5_test = evaluate_c5(h_test)
                if c5_test < current_c5:
                    current_h = h_test
                    current_c5 = c5_test
                    if current_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = current_c5
    
    # Final refinement with optimization
    try:
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        res = minimize(
            objective, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            best_h = res.x
        best_h = project_to_sum_box(best_h, TARGET_SUM)
        best_c5 = evaluate_c5(best_h)
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
