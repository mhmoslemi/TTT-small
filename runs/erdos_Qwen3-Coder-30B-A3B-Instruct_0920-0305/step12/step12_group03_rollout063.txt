def compute_beatty_pattern(n):
    """Generate a Beatty sequence pattern with exactly n/2 ones"""
    phi = (1 + np.sqrt(5)) / 2

    h = np.zeros(n)
    h[sorted_indices[:n//2]] = 1
    return h

def update_correlation_incremental(C_raw, h, i, j):
    """Update correlation vector with a swap of h[i] and h[j]"""

        if i + k < n_points:
            delta[k] -= 1 - h[i + k]
    
    # Update for shift k where i-k is valid
    for k in range(n_points):
        if i - k >= 0:
            delta[n_points - 1 - k] -= h[i - k]
    
    # Apply the effect of changing h[j] from 1 to 0

    for k in range(n_points):
        if j - k >= 0:
            delta[n_points - 1 - k] += h[j - k]
