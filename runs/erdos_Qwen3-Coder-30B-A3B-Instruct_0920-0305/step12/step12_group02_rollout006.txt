```python
import numpy as np
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up problem parameters
    TARGET_SUM = n_points / 2
    x = np.arange(n_points) * dx
    d = np.abs(x - 1.0)  # distance from centre
    
    # Define the objective function to compute C5
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_c5 = np.inf
    best_h = None
    
    # Parametric search over p values
    # Sample p values using log spacing to explore the range effectively
    p_values = np.logspace(np.log10(0.3), np.log10(5), 200)
    
    for p in p_values:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute alpha for the parametric shape
        alpha_raw = (p + 1) / (2 * p)
        alpha = min(alpha_raw, 1.0)
        
        # Compute the base shape function
        ratio = d / alpha
        f = np.where(ratio <= 1, 1 - np.power(ratio, p), 0.0)
        
        # Compute sum of base shape
        sum_f = f.sum()
        
        # Skip if shape sum is zero or very small
        if sum_f < 1e-12:
            continue
            
        # Determine feasible range for c
        c_max1 = TARGET_SUM / sum_f
        
        # The constraint b + c <= 1 where b = (TARGET_SUM - c * sum_f) / N
        # becomes: (TARGET_SUM - c * sum_f) / N + c <= 1
        # Simplifying: TARGET_SUM/N + c * (1 - sum_f/N) <= 1
        # Therefore: c * (1 - sum_f/N) <= 1 - TARGET_SUM/N = 0.5
        denom = 1 - sum_f / n_points
        if abs(denom) < 1e-10:
            c_max2 = np.inf
        elif denom > 0:
            c_max2 = 0.5 / denom
        else:
            # When denom < 0, constraint (1 - sum_f/N) * c <= 0.5 
            # is automatically satisfied for c >= 0, so no upper bound
            c_max2 = np.inf
            
        c_max = min(c_max1, c_max2)
        
        if c_max <= 0:
            continue
            
        # Function to evaluate C5 for given c
        def obj(c):
            if c < 0 or c > c_max:
                return np.inf
            b = (TARGET_SUM - c * sum_f) / n_points
            h = b + c * f
            # Clip to [0,1] to ensure constraints
            h = np.clip(h, 0, 1)
            return compute_c5(h)
        
        # Use ternary search to find optimal c
        lo, hi = 0.0, c_max
        for _ in range(30):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = obj(m1)
            f2 = obj(m2)
            if f1 < f2:
                hi = m2
            else:
                lo = m1
                
        c_opt = (lo + hi) / 2.0
        c5_val = obj(c_opt)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            b = (TARGET_SUM - c_opt * sum_f) / n_points
            best_h = b + c_opt * f
            best_h = np.clip(best_h, 0, 1)
            
            # Ensure exact sum by projection
            best_h = project_to_sum_box(best_h, TARGET_SUM)
            best_c5 = compute_c5(best_h)
    
    # Local refinement if time allows
    if best_h is not None and time.time() - start_time < budget_s - 5:
        def obj_proj(x):
            h = project_to_sum_box(x, TARGET_SUM)
            return compute_c5(h)
        
        # Use L-BFGS-B for refinement
        x0 = best_h.copy()
        try:
            res = minimize(
                obj_proj, 
                x0, 
                method='L-BFGS-B', 
                bounds=[(0,1)]*n_points,
                options={'maxiter': 300, 'ftol': 1e-12}
            )
            if res.success:
                h_opt = project_to_sum_box(res.x, TARGET_SUM)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
```
