import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

# Global variables
n_points = 200
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def build_h_from_params(params, K, n):
    """Convert parameter vector to step-function h"""
    # First K are raw lengths
    raw_lengths = params[:K]
    # Next K are raw heights
    raw_heights = params[K:2*K]
    
    # Compute positive lengths
    lengths = np.exp(raw_lengths)
    lengths = lengths / lengths.sum() * 2.0   # Total interval length = 2
    
    # Compute heights between 0 and 1
    heights = 1.0 / (1.0 + np.exp(-raw_heights))    # sigmoid
    
    # Build h array of size n_points
    x = (np.arange(n) + 0.5) * (2.0 / n)   # sample positions at cell centers
    # Determine segment index for each x
    cum_lengths = np.cumsum(lengths)
    seg_indices = np.searchsorted(cum_lengths, x, side='right')
    h = heights[seg_indices]
    
    return h

def objective(params):
    """Objective function for optimization"""
    h = build_h_from_params(params, K=5, n=n_points)
    c5 = evaluate_c5(h)   # fast
    # area constraint
    sum_h = h.sum()
    penalty = 0.0
    if abs(sum_h - TARGET_SUM) > 1e-8:
        penalty = 1e3 * (sum_h - TARGET_SUM)**2   # penalty large after scaling
    return c5 + penalty

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Differential evolution settings
    K = 5  # number of segments
    bounds = [(-5, 5)] * (2 * K)  # raw variables
    
    # Time management for DE
    max_iter = 200
    remaining_time = budget_s - 20  # leave 20 seconds for polishing
    
    result = differential_evolution(
        objective, 
        bounds,
        maxiter=max_iter,
        popsize=15,
        polish=False,
        seed=seed,
        updating='deferred'
    )
    
    # Convert best parameters to h
    h0 = build_h_from_params(result.x, K=5, n=n_points)
    
    # Project onto sum constraint
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    # Local refinement with SLSQP
    def local_objective(h):
        return evaluate_c5(h)
    
    cons = {'type':'eq','fun': lambda h: np.sum(h) - TARGET_SUM}
    bounds_vec = [(0.0,1.0)] * n_points

    # Check if we have enough time for polishing
    if time.time() - start_time < budget_s - 5:
        try:
            res = minimize(local_objective, h0, 
                           method='SLSQP',
                           bounds=bounds_vec,
                           constraints=cons,
                           options={'maxiter':500, 'ftol':1e-12, 'disp':False})
            if res.success:
                h_opt = res.x
                # Final safeguarding
                h_opt = project_to_sum_box(h_opt, TARGET_SUM)
                final_c5 = evaluate_c5(h_opt)
            else:
                h_opt = h0
                final_c5 = evaluate_c5(h0)
        except:
            h_opt = h0
            final_c5 = evaluate_c5(h0)
    else:
        h_opt = h0
        final_c5 = evaluate_c5(h0)
    
    return (h_opt.tolist(), float(final_c5), n_points)
