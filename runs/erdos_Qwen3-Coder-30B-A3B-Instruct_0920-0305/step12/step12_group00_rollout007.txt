import cvxpy as cp
from scipy.sparse import diags
from scipy.linalg import eigh
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n = 128
    dx = 2.0 / n
    N = n // 2
    t_total = budget_s
    
    # Precompute Toeplitz matrices
    def build_toeplitz_matrix(n, lag):
        # Construct sparse Toeplitz matrix A_lag with 1s on diagonal +lag and -lag
        data = [np.ones(n - abs(lag)), np.ones(n - abs(lag))]
        offsets = [abs(lag), -abs(lag)]
        A = diags(data, offsets, shape=(n, n), format='csr')
        return A
    
    # Precompute all A_lag matrices for lags 1 to n-1
    A_matrices = {}
    for lag in range(1, n):
        A_matrices[lag] = build_toeplitz_matrix(n, lag)
    
    # Build full constraint matrices
    J = np.ones((n, n))
    
    # Utility to evaluate C5 from binary vector
    def compute_c5_from_h(h):
        # Compute correlations via FFT
        h_rev = h[::-1]
        b = 1 - h
        conv = np.convolve(h, b, mode='full')
        max_corr = np.max(conv)
        return dx * max_corr

    start_time = time.time()
    
    # Build the SDP
    X = cp.Variable((n, n), PSD=True)
    m = cp.Variable()
    
    constraints = [cp.trace(J @ X) == N, 0 <= X, X <= 1]
    for lag in range(1, n):
        constraints.append(cp.trace(A_matrices[lag] @ X) >= m)
    
    objective = cp.Maximize(m)
    prob = cp.Problem(objective, constraints)
    
    # Try solving the main SDP
    t_sdp = 0.4 * t_total
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-5, verbose=False, time_limit=t_sdp)
        if prob.status in ['optimal', 'optimal_inaccurate']:
