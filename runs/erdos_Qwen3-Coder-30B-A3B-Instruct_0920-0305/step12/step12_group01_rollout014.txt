def evaluate_c5(h):
    """Fast evaluation of C5 using FFT correlation"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr) * (2.0 / len(h))

def project_to_sum_box(h, target_sum):
    """Project h onto box constraints [0,1] with sum = target_sum"""
    # Binary search for the correct lambda
    low, high = -1.0, 1.0
    while high - low > 1e-12:
        mid = (low + high) / 2
        projected = np.clip(h + mid, 0, 1)
        if np.sum(projected) > target_sum:
            high = mid
        else:
            low = mid
    return np.clip(h + low, 0, 1)
