import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        max_iter = 2000

        c2 = 1.5  # social parameter

        velocities = []
        pbest_positions = []
        pbest_scores = []

            h = np.random.rand(n_points)

            particles.append(h.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
            pbest_positions.append(h.copy())
            pbest_scores.append(evaluate_c5(h))
        
        # Initialize global best
        gbest_idx = np.argmin(pbest_scores)
        gbest_position = pbest_positions[gbest_idx].copy()
        gbest_score = pbest_scores[gbest_idx]
        
        for iter_num in range(max_iter):
            if time.time() - start_time > budget_s - 10:
                break
            
            # Update each particle
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.random(2)
                velocities[i] = (w * velocities[i] + 
                               c1 * r1 * (pbest_positions[i] - particles[i]) + 
                               c2 * r2 * (gbest_position - particles[i]))
                
                # Update position
                particles[i] += velocities[i]
                
                # Apply bounds and projection
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = project_to_sum_box(particles[i], TARGET_SUM)
                
                # Evaluate fitness
                current_score = evaluate_c5(particles[i])
                
                # Update personal best
                if current_score < pbest_scores[i]:
                    pbest_scores[i] = current_score
                    pbest_positions[i] = particles[i].copy()
                    
                    # Update global best
                    if current_score < gbest_score:
                        gbest_score = current_score
                        gbest_position = particles[i].copy()
            
            # Adaptive parameters
            if iter_num % 100 == 0 and iter_num > 0:
                w = max(0.3, w * 0.95)  # decrease inertia
                c1 -= 0.005  # decrease cognitive component
                c2 -= 0.005  # decrease social component
                
        return gbest_position, gbest_score

        max_iterations = 2000

            # Create neighbor solution
            h_new = h.copy()
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.02)
            h_new[idx] += delta
            
            # Project to feasible region
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            # Evaluate
            c5_new = evaluate_c5(h_new)

                h = h_new
                current_c5 = c5_new
            
            # Cool down
            temperature *= cooling_rate
            if temperature < min_temperature:
                temperature = min_temperature
                
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
