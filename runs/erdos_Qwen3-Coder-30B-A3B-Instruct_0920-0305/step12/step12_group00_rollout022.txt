```python
import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time
from math import sqrt

def balanced_sturmian(n):
    """Generate a balanced Sturmian word as initial seed"""
    phi = (sqrt(5) - 1) / 2
    x = np.array([int((i * phi) % 1 < 0.5) for i in range(n)])
    # Adjust to exactly sum to n//2
    current_sum = np.sum(x)
    needed = n // 2 - current_sum
    if needed > 0:
        # Flip zeros to ones
        zeros = np.where(x == 0)[0]
        for i in range(needed):
            x[zeros[i]] = 1
    elif needed < 0:
        # Flip ones to zeros
        ones = np.where(x == 1)[0]
        for i in range(-needed):
            x[ones[i]] = 0
    return x

def compute_autocorr_fft(h):
    """Compute all autocorrelations via FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def swap_delta(S, i, j, n):
    """Compute change to autocorrelations when swapping h[i] and h[j]"""
    delta = np.zeros(n)
    # For each lag ℓ, compute the change in sum due to the swap at positions i,j
    for l in range(1, n):
        # Only consider valid indices
        if i + l < n and j + l < n:
            delta[l] += (h[j+l] - h[i+l])
        if i >= l and j >= l:
            delta[l] += (h[i-l] - h[j-l])
        if i >= l and i + l < n:
            delta[l] += (h[i-l] - h[i+l])
        if j >= l and j + l < n:
            delta[l] += (h[j-l] - h[j+l])
    return delta

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Parameters
    n = 128
    N = n // 2
    dx = 2.0 / n
    best_c5 = float('inf')
    best_h = None
    best_m = 0
    
    # Stage 1: Exact ILP
    print(f"Starting ILP with n={n}")
    try:
        # Variables
        x = cp.Variable(n, boolean=True)
        m = cp.Variable(1)
        
        # Constraints
        constraints = [cp.sum(x) == N]
        
        # Add symmetry breaking
        constraints.append(x[0] == 1)  # Fix first bit
        for i in range(N-1):
            constraints.append(x[i] >= x[i+1])  # Non-increasing first half
            
        # Auxiliary variables for products x[i] * x[i+ℓ]
        y = {}
        S = {}
        for l in range(1, n):
            y[l] = cp.Variable((n-l), boolean=True)
            # McCormick constraints for y[i,l] = x[i] * x[i+l]
            for i in range(n-l):
                constraints.append(y[l][i] <= x[i])
                constraints.append(y[l][i] <= x[i+l])
                constraints.append(y[l][i] >= x[i] + x[i+l] - 1)
            
            # S_ℓ = sum of y[i,ℓ]
            S[l] = cp.sum(y[l])
            constraints.append(S[l] >= m)
        
        # Objective
        objective = cp.Maximize(m)
        prob = cp.Problem(objective, constraints)
        
        # Solve with time limit
        prob.solve(solver=cp.CBC, max_seconds=0.45 * budget_s)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_val = np.round(x.value).astype(int)
            if np.abs(np.sum(x_val) - N) < 1e-6:
                # Compute actual C5 from full correlation
                corr_full = np.correlate(x_val, 1 - x_val, mode='full')
                min_corr = np.min(corr_full[n-1:])  # Exclude lag 0
                current_c5 = dx * (N - min_corr)
                print(f"ILP solution found: C5 = {current_c5:.6f}")
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = x_val.copy()
                    best_m = min_corr
            else:
                print("ILP solution has incorrect sum")
        else:
            print(f"ILP solver status: {prob.status}")
            
    except Exception as e:
        print(f"ILP failed with error: {e}")
    
    # Stage 2: Heuristic refinement if needed
    if best_h is None:
        print("Using fallback: Sturmian word + refinement")
        best_h = balanced_sturmian(n)
        corr_full = np.correlate(best_h, 1 - best_h, mode='full')
        min_corr = np.min(corr_full[n-1:])
        best_c5 = dx * (N - min_corr)
        print(f"Sturmian solution: C5 = {best_c5:.6f}")
        
    # Refinement with local search
    print("Starting local refinement...")
    current_h = best_h.copy()
    corr_full = np.correlate(current_h, 1 - current_h, mode='full')
    min_corr = np.min(corr_full[n-1:])
    current_c5 = dx * (N - min_corr)
    
    ones = np.where(current_h == 1)[0]
    zeros = np.where(current_h == 0)[0]
    
    # Initialize with the best solution found
    best_c5 = current_c5
    best_h = current_h.copy()
    best_m = min_corr
    
    # Greedy hill climbing
    swap_count = 0
    patience = 0
    max_patience = 5 * n
    temp = 0.01 * best_c5
    T = temp
    
    while time.time() - t0 < 0.8 * budget_s:
        if swap_count >= max_patience or patience >= 5000:
            break
            
        # Pick random indices to swap
        i_idx = np.random.randint(len(ones))
        j_idx = np.random.randint(len(zeros))
        i, j = ones[i_idx], zeros[j_idx]
        
        # Compute change in autocorrelations
        delta = swap_delta(corr_full, i, j, n)
        new_corr_full = corr_full + delta
        new_min_corr = np.min(new_corr_full[n-1:])
        new_c5 = dx * (N - new_min_corr)
        
        # Accept or reject
        if new_c5 < current_c5:
            # Accept
            current_h[i], current_h[j] = current_h[j], current_h[i]
            corr_full = new_corr_full
            current_c5 = new_c5
            min_corr = new_min_corr
            patience = 0
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                best_m = min_corr
                print(f"Improved C5 to {best_c5:.6f}")
        else:
            # Simulated annealing acceptance
            if np.random.rand() < np.exp((current_c5 - new_c5) / T):
                # Accept anyway
                current_h[i], current_h[j] = current_h[j], current_h[i]
                corr_full = new_corr_full
                current_c5 = new_c5
                min_corr = new_min_corr
                patience = 0
            else:
                patience += 1
                
        swap_count += 1
        # Cool down temperature
        T = max(1e-8, 0.99 * T)
    
    # Additional restarts if time remains
    if time.time() - t0 < 0.9 * budget_s:
        print("Performing additional restarts...")
        for _ in range(5):
            if time.time() - t0 >= 0.9 * budget_s:
                break
            # Random restart
            h_restart = np.random.rand(n) < 0.5
            h_restart = np.array(h_restart, dtype=int)
            # Adjust sum exactly to N
            current_sum = np.sum(h_restart)
            diff = N - current_sum
            if diff > 0:  # Need to add more 1s
                zero_indices = np.where(h_restart == 0)[0]
                for i in range(diff):
                    if i < len(zero_indices):
                        h_restart[zero_indices[i]] = 1
            elif diff < 0:  # Need to remove 1s
                one_indices = np.where(h_restart == 1)[0]
                for i in range(-diff):
                    if i < len(one_indices):
                        h_restart[one_indices[i]] = 0
            
            # Refine
            corr_full = np.correlate(h_restart, 1 - h_restart, mode='full')
            min_corr = np.min(corr_full[n-1:])
            c5 = dx * (N - min_corr)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_restart.copy()
                best_m = min_corr
                print(f"Restart improved C5 to {best_c5:.6f}")
    
    # Final check
    if best_h is None:
        print("Returning random solution as fallback")
        best_h = balanced_sturmian(n)
        corr_full = np.correlate(best_h, 1 - best_h, mode='full')
        min_corr = np.min(corr_full[n-1:])
        best_c5 = dx * (N - min_corr)
    
    print(f"Final C5 = {best_c5:.6f}")
    return (best_h.tolist(), float(best_c5), n)
```
