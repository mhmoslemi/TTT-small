import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort the vector
    sorted_x = np.sort(x)[::-1]
    
    # Find lambda using binary search
    def f(lam):
        projected = np.clip(sorted_x - lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    lambda_val = (a + b) / 2
    result = np.clip(x - lambda_val, 0, 1)
    return result

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def softmax_weights(corr, alpha):
    """Compute softmax weights"""
    exp_corr = np.exp(alpha * corr)
    return exp_corr / np.sum(exp_corr)

def grad_softmax(corr, alpha, h):
    """Compute gradient of softmax with respect to h"""
    weights = softmax_weights(corr, alpha)
    # Compute (w * h) and (w_rev * h)
    conv1 = np.convolve(weights, h, mode='full')[:len(h)]
    conv2 = np.convolve(weights[::-1], h, mode='full')[:len(h)]
    grad = 1.0 - conv1 - conv2
    return grad

def adam_step(h, m, v, grad, t, lr=0.12, beta1=0.9, beta2=0.999, epsilon=1e-8):
    """Perform Adam update step"""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
    return h_new, m, v

def generate_sturmian(n, phi=(np.sqrt(5)-1)/2):
    """Generate Sturmian word with parameter phi"""
    return (np.arange(n) * phi) % 1 < 0.5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    TARGET_SUM = n_points // 2
    best_c5 = float('inf')
    best_h = None
    
    # Function to evaluate exact C5
    def exact_c5(h):
        corr = compute_corr_fft(h)
        return dx * np.max(corr)
    
    # Generate initial solutions
    initial_solutions = []
    
    # 1. Sturmian word
    sturmian = generate_sturmian(n_points).astype(float)
    sturmian = project_to_sum_box(sturmian, TARGET_SUM)
    initial_solutions.append(("sturmian", sturmian))
    
    # 2. Random solution
    random_h = np.random.rand(n_points)
    random_h = project_to_sum_box(random_h, TARGET_SUM)
    initial_solutions.append(("random", random_h))
    
    # 3. Perturbed sturmian
    perturbed = sturmian + np.random.normal(0, 0.05, n_points)
    perturbed = project_to_sum_box(perturbed, TARGET_SUM)
    initial_solutions.append(("perturbed", perturbed))
    
    # Adam optimization with smooth max
    for init_name, init_h in initial_solutions:
        if time.time() - start_time > 0.45 * budget_s:
            break
            
        h = init_h.copy()
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        alpha = 4.0
        t = 1
        
        max_iter = 8000
        for iter_num in range(max_iter):
            if time.time() - start_time > 0.45 * budget_s:
                break
                
            # Compute correlation
            corr = compute_corr_fft(h)
            
            # Compute gradient of smooth max
            grad = grad_softmax(corr, alpha, h)
            
            # Adam update
            h, m, v = adam_step(h, m, v, grad, t, lr=0.12)
            t += 1
            
            # Project to feasible set
            h = project_to_sum_box(h, TARGET_SUM)
            
            # Update alpha
            if iter_num > 0 and iter_num % 200 == 0:
                alpha = min(alpha * 1.5, 200.0)
            
            # Check if we found a better solution
            current_c5 = exact_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Convert best fractional solution to binary
    if best_h is not None:
        # Round to binary
        binary_h = (best_h >= 0.5).astype(float)
        
        # Adjust to correct sum
        ones_count = np.sum(binary_h)
        if ones_count > TARGET_SUM:
            ones_indices = np.where(binary_h == 1)[0]
            flip_count = int(ones_count - TARGET_SUM)
            flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
            binary_h[flip_indices] = 0
        elif ones_count < TARGET_SUM:
            zeros_indices = np.where(binary_h == 0)[0]
            flip_count = int(TARGET_SUM - ones_count)
            flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
            binary_h[flip_indices] = 1
        
        binary_h = project_to_sum_box(binary_h, TARGET_SUM)
        binary_c5 = exact_c5(binary_h)
        
        if binary_c5 < best_c5:
            best_c5 = binary_c5
            best_h = binary_h.copy()
        
        # Run swap hill-climb
        if time.time() - start_time < 0.8 * budget_s:
            current_h = binary_h.copy()
            current_c5 = binary_c5
            max_swaps = 5 * n_points
            swap_count = 0
            
            while time.time() - start_time < 0.8 * budget_s and swap_count < max_swaps:
                # Get indices of 1s and 0s
                ones = np.where(current_h == 1)[0]
                zeros = np.where(current_h == 0)[0]
                
                if len(ones) == 0 or len(zeros) == 0:
                    break
                    
                # Pick two random indices
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                
                # Simulate the swap
                temp_h = current_h.copy()
                temp_h[i], temp_h[j] = temp_h[j], temp_h[i]
                
                # Compute new correlation
                new_corr = compute_corr_fft(temp_h)
                new_c5 = dx * np.max(new_corr)
                
                # Check if new max correlation is better
                if new_c5 < current_c5:
                    current_h = temp_h.copy()
                    current_c5 = new_c5
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = current_h.copy()
                
                swap_count += 1
            
            # Additional random binary restarts
            for _ in range(5):
                if time.time() - start_time > 0.95 * budget_s:
                    break
                    
                # Create a new random binary sequence with same sum
                rand_h = np.random.rand(n_points)
                rand_h = project_to_sum_box(rand_h, TARGET_SUM)
                # Round to binary
                binary_h = (rand_h >= 0.5).astype(float)
                # Fix sum
                ones_count = np.sum(binary_h)
                if ones_count > TARGET_SUM:
                    ones_indices = np.where(binary_h == 1)[0]
                    flip_count = int(ones_count - TARGET_SUM)
                    flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
                    binary_h[flip_indices] = 0
                elif ones_count < TARGET_SUM:
                    zeros_indices = np.where(binary_h == 0)[0]
                    flip_count = int(TARGET_SUM - ones_count)
                    flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
                    binary_h[flip_indices] = 1
                binary_h = project_to_sum_box(binary_h, TARGET_SUM)
                trial_c5 = exact_c5(binary_h)
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = binary_h.copy()
    
    # Return best solution found
    if best_h is None:
        # Fallback to initial solution
        initial_h = initial_solutions[0][1].copy()
        best_c5 = exact_c5(initial_h)
        best_h = initial_h
    
    return (best_h.tolist(), float(best_c5), n_points)
