def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def upsample_h(h_old, n_new):
    """Upsample h_old to n_new points using linear interpolation"""
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def compute_gradient_for_lag(h, lag):
    """Compute gradient of O(k) with respect to h for a given lag k"""
    N = len(h)
    forward = np.zeros(N)
    backward = np.zeros(N)
    
    if lag >= 0:
        forward[:N - lag] = h[lag:]
        backward[lag:] = h[:N - lag]
    else:
        k = -lag
        forward[k:] = h[:N - k]
        backward[:N - k] = h[k:]
    
    grad = 1.0 - forward - backward
    return grad

        c5_vals = compute_corr_fft(h)
        max_c5 = c5_from_corr(c5_vals, n_points)
        max_idx = np.argmax(c5_vals)
        lag = max_idx - (n_points - 1)
        
        # Compute gradient for this lag
        grad = compute_gradient_for_lag(h, lag)
        
        # Step size schedule
        step_size = 0.2 / np.sqrt(iteration + 1)
        
        # Update step
        h_new = h - step_size * grad
        
        # Project back to feasible region
        h_new = project_to_sum_box(h_new, target_sum)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new)

                best_c5 = new_c5
                best_h = h_new.copy()
        else:
            # Accept with small probability
            if np.random.random() < 0.001:
                h = h_new
                
    return best_h, best_c5
