import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_descent_optimization(h_init, current_best, start_time, budget_s):
    """Enhanced gradient-based optimization"""
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Try multiple restarts with different initializations
    best_h = h_init.copy()
    best_c5 = current_best
    
    for restart in range(5):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Slightly perturb starting point
        h_perturbed = h_init.copy()
        noise_level = 0.05
        for i in range(n_points):
            if np.random.random() < 0.1:
                h_perturbed[i] += np.random.normal(0, noise_level)
        
        h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
        
        try:
            res = minimize(
                objective, 
                h_perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_result = res.x
            else:
                h_result = h_perturbed
            h_result = project_to_sum_box(h_result, TARGET_SUM)
            c5_result = evaluate_c5(h_result)
            
            if c5_result < best_c5:
                best_c5 = c5_result
                best_h = h_result.copy()
        except:
            continue
    
    return best_h, best_c5

def genetic_algorithm_optimization(h_init, current_best, start_time, budget_s):
    """Genetic algorithm with adaptive parameters"""
    population_size = 40
    generations = 300
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 8
    
    # Generate initial population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = h_init.copy()
        # Add diverse perturbations
        for i in range(n_points):
            if np.random.random() < 0.05:
                h[i] += np.random.normal(0, 0.02)
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    
    best_h = h_init.copy()
    best_c5 = current_best
    
    # Evolutionary optimization with adaptive mutation rates
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate fitness
        fitness = [evaluate_c5(h) for h in population]
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        current_best_fitness = fitness[sorted_indices[0]]
        
        if current_best_fitness < best_c5:
            best_c5 = current_best_fitness
            best_h = population[sorted_indices[0]].copy()
        
        # Create new population
        new_population = []
        
        # Keep elite
        for i in range(elite_size):
            new_population.append(population[sorted_indices[i]].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1_idx = sorted_indices[np.random.randint(0, elite_size)]
            parent2_idx = sorted_indices[np.random.randint(0, elite_size)]
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.random() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.random() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.01)
                if np.random.random() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.01)
            
            child1 = project_to_sum_box(child1, TARGET_SUM)
            child2 = project_to_sum_box(child2, TARGET_SUM)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population
    
    return best_h, best_c5

def simulated_annealing_optimization(h_init, current_best, start_time, budget_s):
    """Simulated annealing with adaptive cooling schedule"""
    h_current = h_init.copy()
    c5_current = evaluate_c5(h_current)
    
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # Initial parameters
    temperature = 0.1
    cooling_rate = 0.995
    min_temperature = 1e-6
    
    # Number of iterations per temperature
    iterations_per_temp = 50
    
    iteration = 0
    while temperature > min_temperature and time.time() - start_time < budget_s - 5:
        for _ in range(iterations_per_temp):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Create neighbor solution
            h_neighbor = h_current.copy()
            # Make small random changes
            for i in range(n_points):
                if np.random.random() < 0.03:
                    h_neighbor[i] += np.random.normal(0, 0.01)
            
            h_neighbor = project_to_sum_box(h_neighbor, TARGET_SUM)
            c5_neighbor = evaluate_c5(h_neighbor)
            
            # Accept or reject the neighbor
            if c5_neighbor < c5_current or np.random.random() < np.exp((c5_current - c5_neighbor) / temperature):
                h_current = h_neighbor.copy()
                c5_current = c5_neighbor
                
                if c5_current < best_c5:
                    best_c5 = c5_current
                    best_h = h_current.copy()
        
        temperature *= cooling_rate
        iteration += 1
    
    return best_h, best_c5
