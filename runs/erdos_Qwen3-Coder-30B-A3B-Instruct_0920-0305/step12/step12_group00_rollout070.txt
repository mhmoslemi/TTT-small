import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort the vector
    sorted_x = np.sort(x)[::-1]
    
    # Find lambda using bisection
    def f(lam):
        projected = np.clip(sorted_x - lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x - (a + b)/2, 0, 1)
    return result

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def softmax_weights(corr, alpha):
    """Compute softmax weights for smooth approximation"""
    exp_corr = np.exp(alpha * corr)
    return exp_corr / np.sum(exp_corr)

def grad_softmax(corr, alpha, h):
    """Compute gradient of smooth max with respect to h"""
    weights = softmax_weights(corr, alpha)
    # Forward convolution: w * h
    forward_conv = np.convolve(weights, h, mode='full')
    forward_conv = forward_conv[:len(h)]
    # Backward convolution: w_rev * h
    backward_conv = np.convolve(weights[::-1], h, mode='full')
    backward_conv = backward_conv[:len(h)]
    grad = 1.0 - forward_conv - backward_conv
    return grad

def adam_step(h, m, v, grad, t, alpha_t, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step with projection"""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    h_new = h - alpha_t * m_hat / (np.sqrt(v_hat) + eps)
    return h_new, m, v

def generate_balanced_sturmian(n, phi=(np.sqrt(5)-1)/2):
    """Generate balanced Sturmian word using golden ratio"""
    h = np.array([(i * phi) % 1 < 0.5 for i in range(n)], dtype=float)
    return h

def generate_perturbed_sturmian(n, phi=(np.sqrt(5)-1)/2, noise_level=0.05):
    """Generate perturbed Sturmian word"""
    base = generate_balanced_sturmian(n, phi)
    noise = np.random.normal(0, noise_level, n)
    h = base + noise
    return h

def generate_random_vector(n):
    """Generate random vector and project to feasible set"""
    h = np.random.rand(n)
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Generate initial vectors
    initial_vectors = []
    
    # Balanced Sturmian word
    h0 = generate_balanced_sturmian(n_points)
    h0 = project_to_sum_box(h0, TARGET_SUM)
    initial_vectors.append(("balanced", h0.copy()))
    
    # Perturbed Sturmian word
    h1 = generate_perturbed_sturmian(n_points)
    h1 = project_to_sum_box(h1, TARGET_SUM)
    initial_vectors.append(("perturbed", h1.copy()))
    
    # Random vector
    h2 = generate_random_vector(n_points)
    h2 = project_to_sum_box(h2, TARGET_SUM)
    initial_vectors.append(("random", h2.copy()))
    
    # Run Adam optimization for each initial vector
    max_iter = 8000
    alpha = 4.0
    alpha_growth = 1.5
    alpha_max = 200
    learning_rate = 0.12
    t = 0
    patience_count = 0
    patience_limit = 500
    
    # Track the best solution so far
    for name, h0 in initial_vectors:
        if time.time() - start_time > budget_s * 0.45:
            break
            
        t = time.time()
        h = h0.copy()
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        prev_c5 = evaluate_c5(h)
        
        for iter_num in range(max_iter):
            if time.time() - start_time > budget_s * 0.45:
                break
                
            t = time.time()
            corr = compute_corr_fft(h)
            grad = grad_softmax(corr, alpha, h)
            
            # Adam step
            h_new, m, v = adam_step(h, m, v, grad, iter_num+1, learning_rate)
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            # Evaluate
            c5_new = evaluate_c5(h_new)
            if c5_new < prev_c5:
                h = h_new
                prev_c5 = c5_new
                patience_count = 0
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                patience_count += 1
                
            # Update alpha for smoother approximation
            if iter_num > 0 and iter_num % 200 == 0:
                alpha = min(alpha * alpha_growth, alpha_max)
            
            # Convergence check
            if patience_count > patience_limit:
                break
            
    # Binary refinement if we have a solution
    if best_h is not None:
        # Round to binary
        h_binary = (best_h >= 0.5).astype(float)
        
        # Adjust to correct sum
        ones_count = np.sum(h_binary)
        if ones_count > TARGET_SUM:
            # Find indices with 1s and flip some to 0
            ones_indices = np.where(h_binary == 1)[0]
            flip_count = int(ones_count - TARGET_SUM)
            flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
            h_binary[flip_indices] = 0
        elif ones_count < TARGET_SUM:
            zeros_indices = np.where(h_binary == 0)[0]
            flip_count = int(TARGET_SUM - ones_count)
            flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
            h_binary[flip_indices] = 1
            
        h_binary = project_to_sum_box(h_binary, TARGET_SUM)
        current_c5 = evaluate_c5(h_binary)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_binary.copy()
            
        # Hill-climb swaps
        max_swaps = 5 * n_points
        current_h = h_binary.copy()
        current_c5 = evaluate_c5(current_h)
       
        # Precompute correlation matrix for efficiency
        corr_full = compute_corr_fft(current_h)
        current_max_corr = np.max(corr_full)
        
        # Keep track of indices of 1s and 0s
        ones_indices = set(np.where(current_h == 1)[0])
        zeros_indices = set(np.where(current_h == 0)[0])
        
        # Precompute the localization of 1s and 0s
        def compute_overlap_changes(h, ones_indices, zeros_indices, i, j):
            # For a swap (i, j) where h[i] = 1 and h[j] = 0
            # Compute change in overlaps for all lags
            change = np.zeros(len(h))
            for k in range(len(h)):
                if k in ones_indices and i != k and j != k:
                    change[k] += 1  # Gain 1 overlap with k
                if k in zeros_indices and i != k and j != k:
                    change[k] -= 1  # Lose 1 overlap with k
            change[i] = -1  # Lose overlap with i itself
            change[j] = 1   # Gain overlap with j itself
            return change
                
        # Swap optimization
        for swap_count in range(max_swaps):
            if time.time() - start_time > budget_s * 0.80:
                break
                
            # Pick one 1 and one 0 at random
            if len(ones_indices) == 0 or len(zeros_indices) == 0:
                break
                
            i = np.random.choice(list(ones_indices))
            j = np.random.choice(list(zeros_indices))
            
            # Fast way to compute change in overlaps
            old_c5 = current_c5
            new_h = current_h.copy()
            new_h[i] = 0
            new_h[j] = 1
            
            # Fast update of correlation
            new_corr = compute_corr_fft(new_h)
            new_c5 = np.max(new_corr) * dx
            
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
                ones_indices.remove(i)
                zeros_indices.remove(j)
                ones_indices.add(j)
                zeros_indices.add(i)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
        
        # Multiple restarts for robustness
        for restart in range(10):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Generate a new random binary sequence with correct sum
            rand_h = np.random.rand(n_points)
            rand_h = project_to_sum_box(rand_h, TARGET_SUM)
            # Round to binary
            rand_binary = (rand_h >= 0.5).astype(float)
            # Fix sum
            ones_count = np.sum(rand_binary)
            if ones_count > TARGET_SUM:
                ones_indices = np.where(rand_binary == 1)[0]
                flip_count = int(ones_count - TARGET_SUM)
                flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
                rand_binary[flip_indices] = 0
            elif ones_count < TARGET_SUM:
                zeros_indices = np.where(rand_binary == 0)[0]
                flip_count = int(TARGET_SUM - ones_count)
                flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
                rand_binary[flip_indices] = 1
            rand_binary = project_to_sum_box(rand_binary, TARGET_SUM)
            trial_c5 = evaluate_c5(rand_binary)
            if trial_c5 < best_c5:
                best_c5 = trial_c5
                best_h = rand_binary.copy()
    
    if best_h is None:
        # Fallback to initial balanced solution
        best_h = generate_balanced_sturmian(n_points)
        best_h = project_to_sum_box(best_h, TARGET_SUM)
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
