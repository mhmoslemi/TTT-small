import numpy as np
from scipy.linalg import eigh
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import SCS
import warnings

# Global constants
n_points = 128
dx = 2.0 / n_points
N = n_points // 2
J = np.ones((n_points, n_points))

def build_toeplitz_matrices():
    """Precompute Toeplitz matrices for all lags"""
    A_matrices = []
    for lag in range(1, n_points):
        A = np.zeros((n_points, n_points))
        # Set diagonal elements
        for i in range(n_points - lag):
            A[i, i + lag] = 1
            A[i + lag, i] = 1
        A_matrices.append(A)
    return A_matrices

def sd_solve(A_matrices, max_time=240):
    """Solve the SDP relaxation"""
    # Create CVXPY variables
    X = cp.Variable((n_points, n_points), PSD=True)
    m = cp.Variable()
    
    # Build constraints
    constraints = [
        cp.trace(J @ X) == N,
        0 <= X,
        X <= 1
    ]
    
    # Add autocorrelation constraints
    for A in A_matrices:
        constraints.append(cp.trace(A @ X) >= m)
        
    # Objective
    objective = cp.Maximize(m)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    
    # Use SCS solver with limited time
    try:
        prob.solve(solver=SCS, max_iters=5000, eps=1e-5, time_limit=max_time, verbose=False)
        return X.value, m.value
    except Exception as e:
        print(f"SDP solving failed: {e}")
        return None, None

def random_rounding(X, n_rounds=5):
    """Random hyperplane rounding of SDP solution"""

            corr = correlate(h, 1-h, mode='full')
            c5 = dx * np.max(corr)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        return best_h, best_c5
    except Exception as e:
        print(f"Rounding failed: {e}")
        return None, None

    S = np.array([np.sum(h[i:] * (1-h[:-i])) if i < len(h) else 0 for i in range(1, len(h))])
    m = np.min(S)
    
    max_swaps = 5 * n_points
    swap_count = 0
    best_h = h.copy()
    best_m = m
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while swap_count < max_swaps:
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        # Pick random indices
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Compute change in S for each lag
        delta_S = np.zeros(len(S))
        for k in range(1, len(h)):
            # Check if indices involved in lag k
            if i+k < len(h) and j+k < len(h):
                delta_S[k-1] += (1-h[i+k]) - (1-h[j+k])
            if i >= k and j >= k:
                delta_S[k-1] += (1-h[i-k]) - (1-h[j-k])
            if i+k < len(h) and j >= k:
                delta_S[k-1] += (1-h[j-k]) - (1-h[i+k])
            if i >= k and j+k < len(h):
                delta_S[k-1] += (1-h[i-k]) - (1-h[j+k])
                
        new_S = S + delta_S
        new_m = np.min(new_S)
        
        if new_m > m:
            # Accept swap
            h[i] = 0
            h[j] = 1
            S = new_S
            m = new_m
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            if m > best_m:
                best_m = m
                best_h = h.copy()
                
        swap_count += 1
    
    # Final C5 calculation
    corr = correlate(best_h, 1-best_h, mode='full')
    c5 = dx * np.max(corr)
    return best_h, c5
