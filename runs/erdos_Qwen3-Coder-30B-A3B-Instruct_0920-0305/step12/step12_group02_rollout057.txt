```python
import numpy as np
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Define the objective function for golden section search
    def compute_c5_for_h(h):
        return evaluate_c5(h)
    
    # Precompute the distance array
    x = np.arange(n_points) * dx
    d = np.abs(x - 1.0)
    
    best_c5 = np.inf
    best_h = None
    
    # Try different values of p in log space
    p_values = np.logspace(np.log10(0.3), np.log10(5), 200)
    
    for p in p_values:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute alpha
        alpha_raw = (p + 1) / (2 * p)
        alpha = min(alpha_raw, 1.0)
        
        # Compute raw shape
        ratio = d / alpha
        f = np.where(ratio <= 1, 1 - np.power(ratio, p), 0.0)
        
        sum_f = f.sum()
        
        # Check feasibility
        if sum_f == 0:
            continue
            
        # Compute bounds on c
        c_max1 = TARGET_SUM / sum_f
        denom = 1 - sum_f / n_points
        if denom > 0:
            c_max2 = 0.5 / denom
        else:
            c_max2 = np.inf
            
        c_max = min(c_max1, c_max2)
        if c_max <= 0:
            continue
            
        # Golden section search on c
        def objective(c):
            if c < 0 or c > c_max:
                return np.inf
            b = (TARGET_SUM - c * sum_f) / n_points
            h = b + c * f
            h = np.clip(h, 0, 1)
            return compute_c5_for_h(h)
        
        # Golden section search implementation
        phi = (1 + np.sqrt(5)) / 2
        resphi = 2 - phi
        
        a = 0.0
        b = c_max
        h = b - a
        
        # Initial points
        c = a + resphi * h
        d = b - resphi * h
        
        fc = objective(c)
        fd = objective(d)
        
        while abs(b - a) > 1e-8:
            if time.time() - start_time > budget_s - 5:
                break
                
            if fc < fd:
                b = d
                d = c
                fd = fc
                h = b - a
                c = a + resphi * h
                fc = objective(c)
            else:
                a = c
                c = d
                fc = fd
                h = b - a
                d = b - resphi * h
                fd = objective(d)
        
        c_opt = (a + b) / 2.0
        c5_val = objective(c_opt)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            b = (TARGET_SUM - c_opt * sum_f) / n_points
            best_h = b + c_opt * f
            best_h = np.clip(best_h, 0, 1)
            best_h = project_to_sum_box(best_h, TARGET_SUM)
            best_c5 = compute_c5_for_h(best_h)
    
    # Local refinement with L-BFGS-B
    if best_h is not None and time.time() - start_time < budget_s - 5:
        def obj_proj(x):
            h = project_to_sum_box(x, TARGET_SUM)
            return compute_c5_for_h(h)
            
        x0 = best_h.copy()
        try:
            res = minimize(obj_proj, x0, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                          options={'maxiter':300, 'ftol':1e-12})
            h_opt = project_to_sum_box(res.x, TARGET_SUM)
            c5_opt = compute_c5_for_h(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
```
