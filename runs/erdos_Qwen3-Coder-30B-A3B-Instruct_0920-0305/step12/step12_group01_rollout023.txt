```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

# Global variables
n_points = 200
dx = 2.0 / n_points
TARGET_SUM = n_points / 2
K = 5  # number of segments in step function

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def build_h_from_params(params):
    """Convert parameter vector to discrete step function h"""
    raw_lengths = params[:K]
    raw_heights = params[K:2*K]
    
    # Compute positive lengths
    lengths = np.exp(raw_lengths)
    lengths = lengths / lengths.sum() * 2.0
    
    # Compute heights between 0 and 1
    heights = 1.0 / (1.0 + np.exp(-raw_heights))
    heights = np.clip(heights, 0, 1)
    
    # Build h array
    x = (np.arange(n_points) + 0.5) * dx
    cum_lengths = np.cumsum(lengths)
    seg_indices = np.searchsorted(cum_lengths, x, side='right')
    h = heights[seg_indices]
    
    return h

def objective(params):
    """Objective function for optimization"""
    h = build_h_from_params(params)
    c5 = evaluate_c5(h)
    sum_h = h.sum()
    penalty = 0.0
    if abs(sum_h - TARGET_SUM) > 1e-8:
        penalty = 1e3 * (sum_h - TARGET_SUM)**2
    return c5 + penalty

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define bounds for differential evolution
    bounds = [(-5, 5)] * (2 * K)
    
    # Run differential evolution with adaptive maxiter
    max_iter = 100
    result = differential_evolution(
        objective, 
        bounds,
        maxiter=max_iter,
        popsize=15,
        polish=False,
        seed=seed,
        updating='deferred'
    )
    
    # Convert best parameters to discrete function and project
    h0 = build_h_from_params(result.x)
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    # Local polishing with SLSQP if enough time
    if time.time() - start_time < budget_s - 10:
        cons = {'type':'eq', 'fun': lambda h: np.sum(h) - TARGET_SUM}
        bounds_vec = [(0.0, 1.0)] * n_points
        
        res = minimize(
            evaluate_c5, 
            h0, 
            method='SLSQP',
            bounds=bounds_vec,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12}
        )
        
        if res.success:
            h_opt = res.x
        else:
            h_opt = h0
            
        h_opt = project_to_sum_box(h_opt, TARGET_SUM)
        final_c5 = evaluate_c5(h_opt)
    else:
        # Just use the projected DE result
        final_c5 = evaluate_c5(h0)
        h_opt = h0
    
    return (h_opt.tolist(), float(final_c5), n_points)
```
