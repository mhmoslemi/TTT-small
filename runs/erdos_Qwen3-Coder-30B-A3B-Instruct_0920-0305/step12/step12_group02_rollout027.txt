import numpy as np
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_c5(h):
    """Compute C5 value using correlation"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Generate domain
    x = np.arange(n_points) * dx
    d = np.abs(x - 1.0)  # distance from centre
    
    best_c5 = np.inf
    best_h = None
    
    # Parameter search using golden section search for each p
    p_values = np.linspace(0.3, 5.0, 200)
    
    for p in p_values:
        # Compute alpha
        alpha_raw = (p + 1) / (2 * p)
        alpha = min(alpha_raw, 1.0)
        
        # Compute raw shape
        ratio = d / alpha
        f = np.where(ratio <= 1, 1 - np.power(ratio, p), 0.0)
        
        sum_f = f.sum()
        
        # Avoid division by zero
        if sum_f == 0:
            continue
            
        # Determine feasible range for c
        c_max1 = target_sum / sum_f
        denom = 1 - sum_f / n_points
        if denom > 0:
            c_max2 = 0.5 / denom
        else:
            c_max2 = np.inf
        c_max = min(c_max1, c_max2)
        
        if c_max <= 0:
            continue
            
        # Golden section search for optimal c
        def objective(c):
            if c < 0 or c > c_max:
                return np.inf
            b = (target_sum - c * sum_f) / n_points
            h = b + c * f
            h = np.clip(h, 0, 1)
            return compute_c5(h)
            
        # Golden section search
        phi = (1 + np.sqrt(5)) / 2
        resphi = 2 - phi
        a, b = 0.0, c_max
        h1 = a + resphi * (b - a)
        h2 = a + phi * (b - a)
        f1 = objective(h1)
        f2 = objective(h2)
        
        for _ in range(30):
            if f1 < f2:
                b = h2
                h2 = h1
                f2 = f1
                h1 = a + resphi * (b - a)
                f1 = objective(h1)
            else:
                a = h1
                h1 = h2
                f1 = f2
                h2 = a + phi * (b - a)
                f2 = objective(h2)
                
        c_opt = (a + b) / 2.0
        c5_val = objective(c_opt)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            b_opt = (target_sum - c_opt * sum_f) / n_points
            best_h = b_opt + c_opt * f
            best_h = np.clip(best_h, 0, 1)
            # Project to ensure exact sum constraint
            best_h = project_to_sum_box(best_h, TARGET_SUM)
            best_c5 = compute_c5(best_h)
            
        # Break early if time exhausted
        if time.time() - start_time > budget_s - 1:
            break
    
    # Local refinement with L-BFGS-B
    if best_h is not None and time.time() - start_time < budget_s - 5:
        def obj_proj(x):
            h = project_to_sum_box(x, TARGET_SUM)
            return compute_c5(h)
            
        x0 = best_h.copy()
        res = minimize(obj_proj, x0, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                       options={'maxiter':300, 'ftol':1e-12})
        h_opt = project_to_sum_box(res.x, TARGET_SUM)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            
    return (best_h.tolist(), float(best_c5), n_points)
