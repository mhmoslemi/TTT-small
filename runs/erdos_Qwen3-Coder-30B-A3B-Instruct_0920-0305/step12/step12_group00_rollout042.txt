```python
import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
from math import sqrt

def balanced_sturmian(n):
    """Generate a balanced Sturmian sequence with n/2 ones"""
    phi = (sqrt(5) - 1) / 2
    x = np.array([int(((i * phi) % 1) < 0.5) for i in range(n)])
    # Adjust to have exactly n//2 ones
    current_sum = np.sum(x)
    needed = n // 2 - current_sum
    if needed > 0:
        # Change zeros to ones (from the end)
        zero_indices = np.where(x == 0)[0]
        for i in range(needed):
            x[zero_indices[-(i+1)]] = 1
    elif needed < 0:
        # Change ones to zeros
        one_indices = np.where(x == 1)[0]
        for i in range(-needed):
            x[one_indices[i]] = 0
    return x

def compute_autocorr_fft(h):
    """Compute auto-correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def swap_delta(S, i, j, ones, zeros, n):
    """Compute change in autocorrelations from swapping positions i and j"""
    # Create new S vector
    new_S = S.copy()
    # For each lag, update the change
    for l in range(1, n):
        # Check if i,j are within bounds for this lag
        if i - l >= 0:
            new_S[l] -= h[i - l]
        if i + l < n:
            new_S[l] -= h[i + l]
        if j - l >= 0:
            new_S[l] += h[j - l]
        if j + l < n:
            new_S[l] += h[j + l]
    return new_S

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up parameters
    n = 128
    dx = 2.0 / n
    N = n // 2
    
    best_c5 = float('inf')
    best_h = None
    best_m = 0
    
    # Stage 1: Exact ILP
    try:
        # Create variables
        x = cp.Variable(n, boolean=True)
        m = cp.Variable()
        
        # Constraints
        constraints = [cp.sum(x) == N]
        
        # Add symmetry-breaking constraints
        constraints.append(x[0] == 1)
        for i in range(N-1):
            constraints.append(x[i] >= x[i+1])
        
        # Create auxiliary variables for products
        y = {}
        S = {}
        
        for l in range(1, n):
            # indices for which x[i]*x[i+l] makes sense
            y[l] = cp.Variable((n-l), boolean=True)
            # Add McCormick constraints
            for i in range(n-l):
                constraints.extend([
                    y[l][i] <= x[i],
                    y[l][i] <= x[i+l],
                    y[l][i] >= x[i] + x[i+l] - 1
                ])
            # Sum these to get S_l
            S[l] = cp.sum(y[l])
            constraints.append(S[l] >= m)
        
        # Objective
        objective = cp.Maximize(m)
        
        # Create and solve the problem
        prob = cp.Problem(objective, constraints)
        
        # Solve with time limit
        prob.solve(solver=cp.CBC, max_seconds=0.45*budget_s)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_opt = np.round(x.value).astype(int)
            current_c5 = dx * (N - m.value)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = x_opt
                best_m = m.value
    except Exception:
        pass
    
    # Stage 2: Local refinement with greedy swaps
    if best_h is None:
        # Fallback to Sturmian sequence
        best_h = balanced_sturmian(n)
        current_c5 = dx * (N - 0)  # Will be computed properly below
    
    # Compute initial autocorrelations
    auto_corr = compute_autocorr_fft(best_h)
    current_m = np.min(auto_corr[1:])  # Min over lags ≠ 0
    current_c5 = dx * (N - current_m)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_m = current_m
    
    # Greedy hill-climbing with simulated annealing fallback
    ones = set(np.where(best_h == 1)[0])
    zeros = set(np.where(best_h == 0)[0])
    
    max_swaps = 500 * n
    t_max = start_time + 0.35 * budget_s
    temperature = 0.01 * best_c5
    
    for _ in range(max_swaps):
        if time.time() > t_max:
            break
            
        # Random swap
        i = np.random.choice(list(ones))
        j = np.random.choice(list(zeros))
        
        # Compute new auto-correlations
        new_S = auto_corr.copy()
        for l in range(1, n):
            if i-l >= 0:
                new_S[l] -= best_h[i-l]
            if i+l < n:
                new_S[l] -= best_h[i+l]
            if j-l >= 0:
                new_S[l] += best_h[j-l]
            if j+l < n:
                new_S[l] += best_h[j+l]
        
        new_m = np.min(new_S[1:])
        new_c5 = dx * (N - new_m)
        
        if new_m > current_m:
            # Accept the swap
            best_h[i] = 0
            best_h[j] = 1
            ones.remove(i)
            zeros.remove(j)
            ones.add(j)
            zeros.add(i)
            auto_corr = new_S
            current_m = new_m
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_m = current_m
        else:
            # Accept with probability based on simulated annealing
            if np.random.random() < np.exp((new_m - current_m) / temperature):
                best_h[i] = 0
                best_h[j] = 1
                ones.remove(i)
                zeros.remove(j)
                ones.add(j)
                zeros.add(i)
                auto_corr = new_S
                current_m = new_m
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_m = current_m
        
        # Anneal temperature
        temperature *= 0.99
            
    # Stage 3: Random restarts if needed
    t_end = start_time + budget_s
    for _ in range(5):
        if time.time() > t_end:
            break
        # Random restart
        h_restart = balanced_sturmian(n)
        # Very short hill climb on restart
        auto_corr_restart = compute_autocorr_fft(h_restart)
        m_restart = np.min(auto_corr_restart[1:])
        c5_restart = dx * (N - m_restart)
        if c5_restart < best_c5:
            best_c5 = c5_restart
            best_m = m_restart
            best_h = h_restart
    
    if best_h is None:
        # Fallback to a random balanced solution if needed
        best_h = balanced_sturmian(n)
    
    # Final computation
    final_auto = compute_autocorr_fft(best_h)
    final_m = np.min(final_auto[1:])
    final_c5 = dx * (N - final_m)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_m = final_m
    
    return (best_h.tolist(), float(best_c5), n)
```
