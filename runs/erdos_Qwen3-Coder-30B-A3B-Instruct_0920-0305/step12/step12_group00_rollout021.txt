def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):

        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    # Binary search
    tol = 1e-10
    a, b = -1.0, 1.0

def softmax_weights(corr, alpha):
    """Compute softmax weights"""
    exp_corr = np.exp(alpha * corr)
    return exp_corr / np.sum(exp_corr)

def grad_softmax(corr, alpha, h):
    """Compute gradient of softmax"""
    weights = softmax_weights(corr, alpha)
    # Compute convolutions to get the gradient
    # w * h
    L = 2 * len(h) - 1
    L_fft = 1 << (L - 1).bit_length()
    w_fft = np.fft.fft(weights, L_fft)
    h_fft = np.fft.fft(h, L_fft)
    conv1 = np.fft.ifft(w_fft * h_fft).real[:len(h)]
    
    # w_rev * h
    w_rev_fft = np.fft.fft(weights[::-1], L_fft)
    conv2 = np.fft.ifft(w_rev_fft * h_fft).real[:len(h)]
    
    return 1.0 - conv1 - conv2

def adam_step(h, grad, m, v, t, lr=0.12, beta1=0.9, beta2=0.999, epsilon=1e-8):
    """Adam optimization step"""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
    return h_new, m, v

def create_sturmian_sequence(n, phi=(np.sqrt(5)-1)/2):
    """Create balanced Sturmian sequence"""
    h = np.zeros(n)
    for i in range(n):
        val = (i * phi) % 1
        h[i] = 1 if val < 0.5 else 0
    return h

def create_perturbed_sturmian(n, phi=(np.sqrt(5)-1)/2, noise=0.05):
    """Create perturbed Sturmian sequence"""
    h = np.zeros(n)
    for i in range(n):
        val = (i * phi) % 1
        h[i] = 1 if val < 0.5 else 0
    h = h + np.random.normal(0, noise, n)
    return h
