import numpy as np
from scipy.signal import correlate
from scipy.sparse import diags
from scipy.linalg import eigh
import time
import cvxpy as cp


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Problem parameters
    n = 128  # Use power of two for fast FFT
    dx = 2.0 / n
    N = n // 2  # Target sum

    # Precompute Toeplitz matrices A_ℓ for all lags ℓ = 1,...,n-1
    # Using sparse format for efficiency
    A_matrices = []
    for lag in range(1, n):
        # Construct sparse Toeplitz matrix for lag
        # Elements at positions (i, i+lag) and (i+lag, i)
        # But we only need to store one triangle due to symmetry
        rows = np.arange(n - lag)
        cols = rows + lag
        data = np.ones(n - lag)
        A = diags([data, data], [lag, -lag], shape=(n, n), format='csr')
        A_matrices.append(A)

    # Build entry-wise constraints: all X_ij in [0,1]
    # The constraint 0 ≤ X ≤ 1 means element-wise bounds on matrix

    # Global best tracking
    best_c5 = float('inf')
    best_h = None

    # SDP solver parameters
    t_sdp = 0.4 * budget_s  # 240 seconds for SDP

    # Try solving SDP on n = 128
    try:
        X = cp.Variable((n, n), PSD=True)
        m = cp.Variable()
        J = np.ones((n, n))

        # Constraints
        constraints = [
            cp.trace(J @ X) == N,
            X >= 0,
            X <= 1
        ]

        # For each lag, constrain its autocorrelation >= m
        for A in A_matrices:
            constraints.append(cp.trace(A @ X) >= m)

        # Objective: maximize m
        objective = cp.Maximize(m)
        prob = cp.Problem(objective, constraints)

        # Solve using SCS
        prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-5, verbose=False,
                   time_limit=t_sdp)

        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            X_opt = X.value
            m_opt = m.value
        else:
            # Fallback: try with coarser grid n=64
            n_coarse = 64
            dx_coarse = 2.0 / n_coarse
            N_coarse = n_coarse // 2
            A_coarse = []
            for lag in range(1, n_coarse):
                rows = np.arange(n_coarse - lag)
                cols = rows + lag
                data = np.ones(n_coarse - lag)
                A = diags([data, data], [lag, -lag], shape=(n_coarse, n_coarse), format='csr')
                A_coarse.append(A)

            X_coarse = cp.Variable((n_coarse, n_coarse), PSD=True)
            m_coarse = cp.Variable()
            J_coarse = np.ones((n_coarse, n_coarse))

            constraints_coarse = [
                cp.trace(J_coarse @ X_coarse) == N_coarse,
                X_coarse >= 0,
                X_coarse <= 1
            ]
            for A in A_coarse:
                constraints_coarse.append(cp.trace(A @ X_coarse) >= m_coarse)

            objective_coarse = cp.Maximize(m_coarse)
            prob_coarse = cp.Problem(objective_coarse, constraints_coarse)
            prob_coarse.solve(solver=cp.SCS, max_iters=5000, eps=1e-5, verbose=False, time_limit=t_sdp)

            if prob_coarse.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                X_opt = X_coarse.value
                m_opt = m_coarse.value
            else:
                # If still no good solution, fallback
                raise ValueError("Failed to solve SDP on both scales")

    except:
        # Fallback to random binary construction if SDP fails
        X_opt = None

    # Rounding and hill-climb
    h_best = None
    c5_best = float('inf')

    # Try multiple randomized rounds
    for i in range(5):

                flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
                h[flip_indices] = 0
            elif h_sum < N:
                # Must turn some 0s into 1s
                zeros_indices = np.where(h == 0)[0]
                flip_count = N - h_sum
                flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
                h[flip_indices] = 1
        else:

            # Eigen-decompose
            try:

                # Keep only significant eigenvalues

                w = w[indices]
                v = v[:, indices]

                    h = np.ones(n) * 0.5
                    h[:N] = 1
                    h[N:] = 0

        zeros = np.where(h == 0)[0]

        if len(ones) > 0 and len(zeros) > 0:
            c5 = float(c5)
            best_h_local = h.copy()
            best_c5_local = c5

            # Compute all correlations efficiently
            h_fft = np.fft.fft(h)
            h_fft_conj = np.conj(h_fft)
            corr_full = np.fft.ifft(h_fft * h_fft_conj).real

                iter_count += 1
                if patience > patience_limit:

                j = np.random.choice(zeros)
                # Compute change in each correlation

                temp_h[i], temp_h[j] = temp_h[j], temp_h[i]
                temp_corr = correlate(temp_h, 1 - temp_h, mode='full')
                temp_max_corr = np.max(temp_corr)
                temp_c5 = dx * temp_max_corr

                if temp_c5 < c5:
                    h = temp_h
                    c5 = temp_c5
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    patience = 0
                    if c5 < best_c5_local:
                        best_c5_local = c5
                        best_h_local = h.copy()
                else:
                    patience += 1

        if time.time() - start_time > 0.9 * budget_s:
            break

            h = np.zeros(n)
            ones_indices = np.random.choice(n, N, replace=False)
            h[ones_indices] = 1
            # Compute C5 with FFT
            corr = correlate(h, 1 - h, mode='full')
            c5 = dx * np.max(corr)
            if c5 < c5_best:
                c5_best = c5
