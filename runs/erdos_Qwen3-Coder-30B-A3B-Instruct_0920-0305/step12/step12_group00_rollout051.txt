```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import math

# Global variables
n_points = 256
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort the vector
    sorted_x = np.sort(x)[::-1]
    
    # Find lambda using bisection
    def f(lam):
        projected = np.clip(sorted_x - lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    lambda_val = (a + b) / 2
    result = np.clip(x - lambda_val, 0, 1)
    return result

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def sturmian_sequence(n, phi=(np.sqrt(5)-1)/2):
    """Generate balanced Sturmian sequence"""
    h = np.zeros(n)
    for i in range(n):
        if (i * phi) % 1 < 0.5:
            h[i] = 1
    return h

def random_vector(n, target_sum):
    """Generate random vector with target sum"""
    h = np.random.rand(n)
    return project_to_sum_box(h, target_sum)

def perturbed_sturmian(n, target_sum, noise_std=0.05):
    """Generate perturbed Sturmian sequence"""
    h = sturmian_sequence(n)
    h = h + np.random.normal(0, noise_std, n)
    return project_to_sum_box(h, target_sum)

def softmax_weights(corr, alpha):
    """Compute softmax weights"""
    exp_corr = np.exp(alpha * corr)
    return exp_corr / np.sum(exp_corr)

def grad_softmax(corr, alpha, h):
    """Compute gradient of softmax with respect to h"""
    weights = softmax_weights(corr, alpha)
    # Forward convolution (h * weights)
    forward_conv = np.convolve(h, weights, mode='full')[:len(h)]
    # Backward convolution (weights[::-1] * h)
    backward_conv = np.convolve(weights[::-1], h, mode='full')[:len(h)]
    grad = 1.0 - forward_conv - backward_conv
    return grad

def adam_step(h, grad, m, v, t, lr=0.12, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimization step with projection"""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    
    return h_new, m, v

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Best solution tracking
    best_c5 = float('inf')
    best_h = None
    
    # Initial vectors
    initial_vectors = [
        sturmian_sequence(n_points, (np.sqrt(5)-1)/2),
        random_vector(n_points, TARGET_SUM),
        perturbed_sturmian(n_points, TARGET_SUM, 0.05)
    ]
    
    # Adam optimization
    max_iters = 8000
    alpha = 4.0
    alpha_growth = 1.5
    alpha_max = 200
    t = 1
    
    for init_vec in initial_vectors:
        if time.time() - start_time > 0.45 * budget_s:
            break
            
        h = init_vec.copy()
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        
        for iter_num in range(max_iters):
            if time.time() - start_time > 0.45 * budget_s:
                break
            
            corr = compute_corr_fft(h)
            grad = grad_softmax(corr, alpha, h)
            h, m, v = adam_step(h, grad, m, v, t, lr=0.12)
            t += 1
            
            # Update alpha
            if iter_num % 200 == 0 and alpha < alpha_max:
                alpha = min(alpha * alpha_growth, alpha_max)
            
            # Check for better solution
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Binary refinement
    if best_h is not None:
        # Round to binary
        binary_h = (best_h >= 0.5).astype(float)
        
        # Adjust to correct sum
        ones_count = np.sum(binary_h)
        if ones_count > TARGET_SUM:
            ones_indices = np.where(binary_h == 1)[0]
            flip_count = int(ones_count - TARGET_SUM)
            flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
            binary_h[flip_indices] = 0
        elif ones_count < TARGET_SUM:
            zeros_indices = np.where(binary_h == 0)[0]
            flip_count = int(TARGET_SUM - ones_count)
            flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
            binary_h[flip_indices] = 1
        
        binary_h = project_to_sum_box(binary_h, TARGET_SUM)
        best_binary_c5 = evaluate_c5(binary_h)
        
        # Store current binary for swap operations
        current_binary_h = binary_h.copy()
        current_c5 = best_binary_c5
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_binary_h.copy()
            
        # Swap hill climber
        # Precompute correlation for efficiency
        corr_current = compute_corr_fft(current_binary_h)
        current_max = np.max(corr_current)
        
        max_swaps = 5 * n_points
        swap_count = 0
        
        while time.time() - start_time < 0.8 * budget_s and swap_count < max_swaps:
            # Get indices of 1s and 0s
            ones = np.where(current_binary_h == 1)[0]
            zeros = np.where(current_binary_h == 0)[0]
            
            if len(ones) == 0 or len(zeros) == 0:
                break
                
            # Pick two random indices
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Calculate the effect of swapping
            old_corr = corr_current.copy()
            new_corr = old_corr.copy()
            
            # Update all correlation values affected by the swap
            for l in range(-n_points+1, n_points):
                if l == 0:
                    continue
                # Original contribution of indices i, j to correlation l
                contrib_i = current_binary_h[i] * current_binary_h[(i+l) % n_points] 
                contrib_j = current_binary_h[j] * current_binary_h[(j+l) % n_points]
                
                # New contribution after swap
                new_contrib_i = current_binary_h[j] * current_binary_h[(j+l) % n_points]
                new_contrib_j = current_binary_h[i] * current_binary_h[(i+l) % n_points]
                
                # Update
                new_corr[l+n_points-1] = new_corr[l+n_points-1] - contrib_i - contrib_j + new_contrib_i + new_contrib_j
            
            # Find the new max correlation
            new_max = np.max(new_corr)
            
            if new_max < current_max:
                # Accept the swap
                current_binary_h[i], current_binary_h[j] = current_binary_h[j], current_binary_h[i]
                corr_current = new_corr
                current_max = new_max
                
                # Update best
                temp_c5 = c5_from_corr(corr_current, n_points)
                if temp_c5 < best_c5:
                    best_c5 = temp_c5
                    best_h = current_binary_h.copy()
            else:
                # Simplified parameterization to prevent premature termination
                swap_count += 1
    
    # Multiple binary restarts
    for _ in range(10):
        if time.time() - start_time > 0.95 * budget_s:
            break
        
        # Random binary with correct sum
        rand_vec = np.random.rand(n_points)
        rand_vec = project_to_sum_box(rand_vec, TARGET_SUM)
        rand_binary = (rand_vec >= 0.5).astype(float)
        
        # Correct sum
        ones_count = np.sum(rand_binary)
        if ones_count > TARGET_SUM:
            ones_indices = np.where(rand_binary == 1)[0]
            flip_count = int(ones_count - TARGET_SUM)
            flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
            rand_binary[flip_indices] = 0
        elif ones_count < TARGET_SUM:
            zeros_indices = np.where(rand_binary == 0)[0]
            flip_count = int(TARGET_SUM - ones_count)
            flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
            rand_binary[flip_indices] = 1
        
        rand_binary = project_to_sum_box(rand_binary, TARGET_SUM)
        
        # Hill climb
        corr = compute_corr_fft(rand_binary)
        max_val = np.max(corr)
        current_h = rand_binary.copy()
        current_max = max_val
        
        for _ in range(1000):
            if time.time() - start_time > 0.95 * budget_s:
                break
                
            ones = np.where(current_h == 1)[0]
            zeros = np.where(current_h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
                
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Update correlation values
            old_corr = corr.copy()
            new_corr = old_corr.copy()
            
            for l in range(-n_points+1, n_points):
                if l == 0:
                    continue
                contrib_i = current_h[i] * current_h[(i+l) % n_points] 
                contrib_j = current_h[j] * current_h[(j+l) % n_points]
                
                new_contrib_i = current_h[j] * current_h[(j+l) % n_points]
                new_contrib_j = current_h[i] * current_h[(i+l) % n_points]
                
                new_corr[l+n_points-1] = new_corr[l+n_points-1] - contrib_i - contrib_j + new_contrib_i + new_contrib_j
            
            # Check if beneficial
            new_max = np.max(new_corr)
            if new_max < current_max:
                current_h[i], current_h[j] = current_h[j], current_h[i]
                corr = new_corr
                current_max = new_max
                
                temp_c5 = c5_from_corr(corr, n_points)
                if temp_c5 < best_c5:
                    best_c5 = temp_c5
                    best_h = current_h.copy()
            else:
                # Not improving, and time allows, so continue
                pass
    
    # Return final result
    if best_h is None:
        # Fallback to initial vector
        best_h = initial_vectors[0].copy()
        best_c5 = evaluate_c5(best_h)
        
    return (best_h.tolist(), float(best_c5), n_points)
```
