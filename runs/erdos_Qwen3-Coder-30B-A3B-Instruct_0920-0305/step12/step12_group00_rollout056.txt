import numpy as np
from scipy.signal import correlate
from scipy.linalg import eigh
from cvxpy import *
import time
import cvxpy as cp

# Global variables
n_points = 128
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def build_toeplitz_matrices(n):
    """Build all Toeplitz matrices A_ell for lags 1 to n-1"""
    A_list = []
    for lag in range(1, n):
        A = np.zeros((n, n))
        # Fill diagonal at offset lag
        for i in range(n - lag):
            A[i, i + lag] = 1
            A[i + lag, i] = 1  # symmetric
        A_list.append(A)
    return A_list

def compute_autocorrelations(h, A_list):
    """Fast evaluation of S_ell = h^T A_ell h for all lags"""
    S = np.array([np.dot(h, A @ h) for A in A_list])
    return S

def evaluate_c5_fast(h, A_list):
    """Fast evaluation using provided A_list"""
    S = compute_autocorrelations(h, A_list)
    return dx * (TARGET_SUM - np.min(S))

def solve_sdp(A_list, n, N, max_time=240):
    """Solve the semidefinite programming relaxation"""
    start_time = time.time()
    
    # Variables
    X = Variable((n, n), PSD=True)
    m = Variable()
    
    # Constraints
    constraints = [
        trace(J @ X) == N,
        0 <= X,
        X <= 1,
    ]
    
    for A in A_list:
        constraints.append(trace(A @ X) >= m)
    
    # Objective
    objective = Maximize(m)
    
    # Problem
    prob = Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=SCS, max_iters=5000, eps=1e-5, verbose=False,
                   time_limit=max_time, warm_start=True)
        if prob.status == 'optimal':
            return X.value, m.value
        elif prob.status == 'infeasible' or prob.status == 'unbounded':
            # Fallback: return best primal point
            return X.value, m.value
    except:
        # If solver fails, return None
        pass
    
    # Fallback: return None if SDP cannot be solved
    return None, None

def round_from_sdp(X, n, N, num_rounds=5):
    """Randomized hyperplane rounding from SDP solution"""
    try:
        # Eigen-decomposition to get low-rank factorization
        eigvals, eigvecs = eigh(X)
        # Keep only positive eigenvalues
        idx = eigvals > 1e-8
        V = eigvecs[:, idx] * np.sqrt(eigvals[idx])
        r = V.shape[1]
        
        best_h = None
        best_c5 = float('inf')
        
        for _ in range(num_rounds):
            # Sample random Gaussian vector
            g = np.random.randn(r)
            # Compute scores
            scores = V @ g
            # Threshold to get binary vector
            h = np.zeros(n)
            h[np.argsort(scores)[::-1][:N]] = 1.0
            
            # Ensure exact sum
            current_sum = np.sum(h)
            if current_sum > N:
                # Need to flip some 1's to 0's
                ones_idx = np.where(h == 1)[0]
                flip_idx = np.argsort(scores[ones_idx])[:int(current_sum - N)]
                h[ones_idx[flip_idx]] = 0
            elif current_sum < N:
                # Need to flip some 0's to 1's
                zeros_idx = np.where(h == 0)[0]
                flip_idx = np.argsort(scores[zeros_idx])[::-1][:int(N - current_sum)]
                h[zeros_idx[flip_idx]] = 1
            
            # Evaluate C5
            c5 = evaluate_c5_fast(h, A_list)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        return best_h, best_c5
    except:
        # Fallback to deterministic rounding using diagonal
        scores = np.diag(X)
        h = np.zeros(n)
        h[np.argsort(scores)[::-1][:N]] = 1.0
        return h, evaluate_c5_fast(h, A_list)

def local_hill_climb(h, A_list, max_swaps=5 * n_points):
    """Perform swap-based hill-climb"""
    S = compute_autocorrelations(h, A_list)  # S[ell] = h^T A_ell h
    m = np.min(S[1:])  # max_lag=1 to n-1
    
    ones_indices = np.where(h == 1)[0]
    zeros_indices = np.where(h == 0)[0]
    
    for _ in range(max_swaps):
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            break
            
        # pick random one and zero
        i = np.random.choice(ones_indices)
        j = np.random.choice(zeros_indices)
        
        # compute change
        delta_S = np.zeros_like(S)
        # for each lag ell
        for ell in range(1, n_points):
            delta_S[ell-1] = - (h[(i+ell)%n_points] + h[(i-ell)%n_points]) + \
                              (h[(j+ell)%n_points] + h[(j-ell)%n_points])
        
        new_S = S + delta_S
        new_m = np.min(new_S[1:])
        
        if new_m > m:
            # accept the swap
            h[i], h[j] = h[j], h[i]
            S = new_S
            m = new_m
            
            # update indices
            ones_indices = np.where(h == 1)[0]
            zeros_indices = np.where(h == 0)[0]
            
    return h, evaluate_c5_fast(h, A_list)

def run(seed=42, budget_s=600, **kwargs):
    global n_points, dx, TARGET_SUM, A_list
    
    np.random.seed(seed)
    start_time = time.time()

    # Build Toeplitz matrices
    A_list = build_toeplitz_matrices(n_points)
    J = np.ones((n_points, n_points))

    best_h = None
    best_c5 = float('inf')
    
    # Stage 1: Solve SDP
    X, m_opt = solve_sdp(A_list, n_points, TARGET_SUM, max_time=0.4 * budget_s)
    
    if X is not None:
        # Rounding
        rounded_h, rounded_c5 = round_from_sdp(X, n_points, TARGET_SUM)
        if rounded_c5 < best_c5:
            best_c5 = rounded_c5
            best_h = rounded_h.copy()
            
        # Local optimization
        if best_h is not None:
            refined_h, refined_c5 = local_hill_climb(best_h.copy(), A_list)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h.copy()
    
    # Stage 2: Random restarts
    for _ in range(5):
        if time.time() >= start_time + budget_s * 0.9:
            break
            
        # Create random binary vector
        h = np.random.rand(n_points)
        sorted_idx = np.argsort(h)[::-1]
        h[sorted_idx[:TARGET_SUM]] = 1.0
        h[sorted_idx[TARGET_SUM:]] = 0.0
        
        # Local optimization
        mc_h, mc_c5 = local_hill_climb(h, A_list)
        if mc_c5 < best_c5:
            best_c5 = mc_c5
            best_h = mc_h.copy()
