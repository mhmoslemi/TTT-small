import numpy as np
from scipy.fft import fft, ifft
from scipy.sparse import diags
from scipy.linalg import eigh
import cvxpy as cp
import time
from scipy.signal import correlate

def toeplitz_A(n, lag):
    """Return sparse Toeplitz matrix A_ℓ for lag ℓ"""
    if lag == 0:
        return np.eye(n)
    # Create two diagonals (for ±lag)
    diag = np.ones(n - lag)
    offsets = [lag, -lag]
    return diags([diag, diag], offsets, shape=(n, n)).toarray()

def compute_correlation_fft(h):
    """Fast correlation computation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h, dx):
    """Evaluate C5 = max_k ∫ h(x)(1 - h(x+k)) dx"""
    corr = compute_correlation_fft(h)
    max_corr = np.max(corr)
    return dx * max_corr

def round_sdprank1(X, N):
    """Round a PSD matrix to binary vector using randomized hyperplane rounding"""
    n = X.shape[0]
    
    # Symmetric factorization: X = V V^T (or V^T V = X)
    try:
        # Eigen-decomposition
        lam, V = eigh(X)
        # Keep only positive eigenvalues
        lam = np.maximum(lam, 0)
        mask = lam > 1e-8
        V = V[:, mask]
        lam = lam[mask]
        V = V @ np.diag(np.sqrt(lam))
    except:
        # Fallback to SVD
        try:
            U, s, Vt = np.linalg.svd(X)
            s = np.maximum(s, 0)
            mask = s > 1e-8
            U = U[:, mask]
            s = s[mask]
            V = U @ np.diag(np.sqrt(s))
        except:
            # Final fallback to diagonal approach
            V = np.diag(np.sqrt(np.diag(X)))
    
    # Multiple rounding attempts
    best_h = None
    best_c5 = float('inf')
    
    for _ in range(5):
        # Sample random Gaussian vector
        g = np.random.randn(V.shape[1])
        scores = V @ g
        # Threshold to get binary vector
        sorted_indices = np.argsort(scores)[::-1]  # descending
        h = np.zeros(n)
        h[sorted_indices[:N]] = 1
        c5 = evaluate_c5(h, 2.0/n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    return best_h, best_c5

def hill_climb_swap(h, n_points, dx):
    """Fast local improvement via swapping 1s and 0s"""
    h = h.copy()
    n = n_points
    N = n // 2
    
    # Precompute all autocorrelations via FFT
    corr = compute_correlation_fft(h)
    min_corr = np.max(corr)  # since C5 is max of correlations
    # Actually, we compute: sum h_i * (1 - h_{i+k}) = N - sum h_i * h_{i+k}
    # So for each lag k, S_k = sum_i h_i * h_{i+k}, C5 = dx * max_k (N - S_k)
    # Our objective is to minimize C5, i.e. to maximize min_k S_k.
    
    # for fast calculation we track the autocorrelations directly
    # S_k = (h * h)[k] where * is convolution
    # use FFT to compute all good values
    h_fft = fft(h)
    conv = ifft(h_fft * np.conj(h_fft)).real  # This computes correlation of h with itself
    # But we want sum of h_i * h_{i+k} for all i
    corr_h_h = conv.real[:n]  # first n elements, This is correct
    
    # similarly for computing sum h_i * (1-h_{i+k})
    one_minus_h = 1 - h
    one_minus_h_fft = fft(one_minus_h)
    corr_h_one_minus_h = ifft(h_fft * np.conj(one_minus_h_fft)).real[:n]
    
    # recompute, we do not eliminate code here just for clarity
    corr_h_one_minus_h = compute_correlation_fft(h)
    
    # Now minimize sum h_i (1 - h_{i+k}) = N - sum h_i h_{i+k}
    # So maximize sum h_i h_{i+k}  i.e. minimize per k: N - sum h_i h_{i+k}
    # Which is same as maximize sum h_i h_{i+k}  minX (N - sum h_i h_{i+k}) = maxX sum h_i h_{i+k}
    # So the objective is: max_min sum h_i h_{i+k}
    
    # we keep track of S_k = sum h_i h_{i+k} for all k=1, 2, ..., n-1
    # Compute all correlations
    corr = compute_correlation_fft(h)
    # Now corr has length 2*n-1 but corr[0] ... corr[n-1] have the right values
    # We want corr[1] ... corr[n-1] i.e. the lags k=1..n-1
    # but actually we *don't* compute automatic correlations like this, we do the inverse
    # we compute this from the convolution of h and h
    # corrected versions
    h_conv = correlate(h, h, mode='full')  # negative lags to positive
    h_conv = h_conv[len(h)-1:]  # Take only positive lags
    S = h_conv[:n-1]  # for k=1,...,n-1 we have autocorrelation 
    m = np.min(S)  # this is what we want to *maximize* in the original problem
    
    # Recompute every time
    max_corr = np.max(corr) 
    C5 = dx * max_corr # adjust this order
    
    # Each iteration picks two bytes to swap
    # We can maintain an array where the (negative) change of S_k due to swapping positions i and j
    # Please compute it in the loop.
    
    max_iter = min(500000, 5*n)  # for better samples not at cost of time
    patience = 0
    patience_limit = 2*n
    
    for _ in range(max_iter):
        if patience >= patience_limit:
            break
        # Not changing S_k directly here, instead going directly to original problem
        # Use the precomputed correlations from before and see if we can improve
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        # pick i from ones, j from zeros
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # For swapping check what happens to the lags S_k = sum h_i h_{i+k}
        # Impact of swapping h_i and h_j:
        old_S = S.copy()
        new_S = np.zeros(n-1)
        
        for k in range(1,n):
            # affect at lags k = 1 to n-1
            ni = (i + k) % n
            nj = (j + k) % n
            # This is a little quirk, we should compute:
            # each h_(i-k) should be updated. IF h_{i-k} was 1, now is 0.
            # for i and j limited and lag range, this simplifies
            # If k < n, new S_k = old_S_k - h_{i-k} h_i + h_{i-k} h_j - h_{j-k} h_j + h_{j-k} h_i
            # but only positions with k distance to i and j are affected
            if k < n:
                # affect where index was h_i does not long exist (or j it previously did)
                # we update specifically:
                # replicating with S copy
                new_S[k-1] = old_S[k-1]
                # Only affects term h_i h_{i+k} and h_j h_{j+k} if they exist
                # Expect such check where relevant gonna be insufficient
                # instead compute influence:
                if ni < n and i < n:
                    if h[ni] < 1:
                        #.prev was one, now zero in h_j's position
                        new_S[k-1] = new_S[k-1] - h[i] * h[ni]
                    else:
                        new_S[k-1] = new_S[k-1] - h[i] * h[ni]
                        
                if nj < n and j < n:
                    if h[nj] < 1:
                        #prev was one, now zero in h_i's position
                        new_S[k-1] = new_S[k-1] - h[j] * h[nj]
                    else:
                        new_S[k-1] = new_S[k-1] - h[j] * h[nj]
                # Add h_i h_j replaced (back)
                new_S[k-1] = new_S[k-1] + h[i] * h[j]  # now h[i] and h[j] are 0 and 1 after swap
                
        # select the maximum lag
        lags_changed = new_S != old_S
        if np.any(lags_changed):  
            # [[ * ]* ] WHERE the difference is significant?  
            # There's better way - just compute
            # But it's more insightful down the road
            # this is equivalent to using the S directly
            new_min = np.min(new_S)
            old_min = np.min(old_S)
            if new_min > old_min:  # we prefer better minimum
                h[i], h[j] = h[j], h[i]
                S = new_S.copy()
                m = new_min
                patience = 0
            else:
                patience += 1
        else:
            patience += 1

    # Final evaluation
    corr = compute_correlation_fft(h)
    max_corr = np.max(corr)
    if max_corr == 0:
        max_corr = 0.001  # avoid zero and effectively compute an upper bound not sure 
    C5 = dx * max_corr  
    return h, C5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 128  # power of 2
    dx = 2.0 / n
    N = n // 2
    
    # Matrix A_ℓ
    A_list = [toeplitz_A(n, i) for i in range(1, n)]
    
    # Solve SDP
    X = cp.Variable((n, n), symmetric=True)
    m = cp.Variable()
    
    constraints = [
        cp.trace(X) == N,
        0 <= X,
        X <= 1,
        X >> 0
    ]
    
    # For each non-zero lag, add the constraint
    for i, A in enumerate(A_list):
        lag = i + 1
        constraints.append(cp.trace(A @ X) >= m)
    
    objective = cp.Maximize(m)
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit (rough estimate)
    # Look at time managament to avoid perfectly
    spent_time = time.time() - start_time
    remaining_budget = budget_s - spent_time
    t_sdp = min(200, remaining_budget*0.6)  # leave time for other steps
    
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-4, 
                   time_limit=t_sdp, verbose=False)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            X_val = X.value
            m_opt = m.value
        else:
            # If not solved, return an initial guess
            X_val = np.eye(n) * 0.5
            m_opt = 0
    except Exception as e:
        # Use fallback if error occurs
        X_val = np.eye(n) * 0.5
        m_opt = 0
    
    # Rounding
    h_round, c5_round = round_sdprank1(X_val, N)
    if h_round is None:
        # Fallback to random binary vector
        h = np.random.rand(n)
        h = np.where(h < 0.5, 0, 1)
        h = np.array(h, dtype=float)
        h = h * (N / np.sum(h))
        h = np.round(h).astype(int)
        c5_round = evaluate_c5(h, dx)
    
    best_h = h_round
    best_c5 = c5_round
    
    # Further improvement using hill climb
    improved_h, improved_c5 = hill_climb_swap(h_round, n, dx)
    if improved_c5 < best_c5:
        best_c5 = improved_c5
        best_h = improved_h
    
    # Final time check
    spent_time = time.time() - start_time
    if spent_time < budget_s - 5:
        # Additional random restarts
        for _ in range(3):
            h_rand = np.random.rand(n)
            h_rand = np.where(h_rand < 0.5, 0, 1)
            h_rand = np.array(h_rand, dtype=float)
            h_rand = h_rand * (N / np.sum(h_rand))
            h_rand = np.round(h_rand).astype(int)
            c5_rand = evaluate_c5(h_rand, dx)
            if c5_rand < best_c5:
                best_c5 = c5_rand
                best_h = h_rand
                
            # improvement using hill climb
            h_climb, c5_climb = hill_climb_swap(h_rand, n, dx)
            if c5_climb < best_c5:
                best_c5 = c5_climb
                best_h = h_climb
                
    return (best_h.tolist(), float(best_c5), n)
