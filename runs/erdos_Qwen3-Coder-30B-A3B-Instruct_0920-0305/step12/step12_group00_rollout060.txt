import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort x
    sorted_x = np.sort(x)[::-1]  # descending order

    # Find lambda using binary search
    def f(lam):
        projected = np.clip(sorted_x - lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    lambda_opt = (a + b) / 2
    result = np.clip(x - lambda_opt, 0, 1)
    return result

def compute_softmax_weights(corr, alpha):
    """Computes softmax weights for smooth max approximation"""
    exp_corr = np.exp(alpha * corr)
    sum_exp = np.sum(exp_corr)
    if sum_exp == 0:
        weights = np.ones_like(corr) / len(corr)
    else:
        weights = exp_corr / sum_exp
    return weights

def compute_grad_softmax(corr, alpha, h):
    """Compute gradient of smooth max with respect to h"""
    weights = compute_softmax_weights(corr, alpha)
    # Convolution: weights * h (forward)
    L = len(h) * 2 - 1
    L_fft = 1 << (L - 1).bit_length()
    W_fft = np.fft.fft(weights, L_fft)
    H_fft = np.fft.fft(h, L_fft)
    conv1 = np.fft.ifft(W_fft * H_fft).real[:len(h)]
    
    # Convolution: weights[::-1] * h (backward)
    W_rev_fft = np.fft.fft(weights[::-1], L_fft)
    conv2 = np.fft.ifft(W_rev_fft * H_fft).real[:len(h)]
    
    grad = 1.0 - conv1 - conv2
    return grad

def adam_step(h, grad, m, v, t, lr=0.12, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam step with projection"""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
    return h_new, m, v

def sturmian_sequence(n, phi=(np.sqrt(5)-1)/2):
    """Generate a balanced Sturmian sequence"""
    h = np.zeros(n)
    for i in range(n):
        if (i * phi) % 1 < 0.5:
            h[i] = 1
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Setup
    TARGET_SUM = n_points // 2
    best_c5 = float('inf')
    best_h = None
    
    # Helper functions
    def eval_and_update_best(h, c5):
        nonlocal best_c5, best_h
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Initial starting points
    initial_points = []
    
    # 1. Sturmian sequence
    h_sturm = sturmian_sequence(n_points)
    h_sturm = project_to_sum_box(h_sturm, TARGET_SUM)
    initial_points.append(("Sturmian", h_sturm))
    
    # 2. Random with projection
    for _ in range(3):
        h_rand = np.random.rand(n_points)
        h_rand = project_to_sum_box(h_rand, TARGET_SUM)
        initial_points.append(("Random", h_rand))
    
    # 3. Perturbed Sturmian
    for _ in range(3):
        h_perturb = sturmian_sequence(n_points) + np.random.normal(0, 0.05, n_points)
        h_perturb = project_to_sum_box(h_perturb, TARGET_SUM)
        initial_points.append(("Perturbed", h_perturb))
    
    # Optimization using Adam with smooth max
    max_iters = 8000
    alpha = 4.0
    alpha_growth = 1.5
    alpha_max = 200.0
    t = 0
    
    # Process each initial point
    for name, h_start in initial_points:
        if time.time() - start_time > 0.45 * budget_s:
            break
            
        # Initialize Adam
        current_h = h_start.copy()
        m = np.zeros_like(current_h)
        v = np.zeros_like(current_h)
        
        # Precompute correlations for reference
        ref_corr = compute_corr_fft(current_h)
        ref_c5 = c5_from_corr(ref_corr, n_points)
        eval_and_update_best(current_h, ref_c5)
        
        # Adam iterations
        for iteration in range(max_iters):
            if time.time() - start_time > 0.45 * budget_s:
                break
                
            t += 1
            
            # Compute smooth max gradient
            corr = compute_corr_fft(current_h)
            grad = compute_grad_softmax(corr, alpha, current_h)
            
            # Adam step
            current_h, m, v = adam_step(current_h, grad, m, v, t)
            
            # Project to feasible region
            current_h = project_to_sum_box(current_h, TARGET_SUM)
            
            # Evaluate and update
            c5 = c5_from_corr(compute_corr_fft(current_h), n_points)
            eval_and_update_best(current_h, c5)
            
            # Increase alpha for smooth approximation
            if iteration % 200 == 0:
                alpha = min(alpha * alpha_growth, alpha_max)
    
    # Binary refinement
    if best_h is not None:
        # Convert to binary
        median_val = np.median(best_h)
        binary_h = (best_h >= median_val).astype(float)
        
        # Adjust to correct sum
        ones_count = np.sum(binary_h)
        if ones_count > TARGET_SUM:
            ones_indices = np.where(binary_h == 1)[0]
            flip_count = int(ones_count - TARGET_SUM)
            flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
            binary_h[flip_indices] = 0
        elif ones_count < TARGET_SUM:
            zeros_indices = np.where(binary_h == 0)[0]
            flip_count = int(TARGET_SUM - ones_count)
            flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
            binary_h[flip_indices] = 1
        
        # Final projection
        binary_h = project_to_sum_box(binary_h, TARGET_SUM)
        init_c5 = evaluate_c5(binary_h)
        eval_and_update_best(binary_h, init_c5)
        
        # Fast swap search
        # Maintain overlaps efficiently
        corr = compute_corr_fft(binary_h)
        current_min = np.min(corr)
        current_c5 = dx * np.max(corr)
        eval_and_update_best(binary_h, current_c5)
        
        # Keep track of 1s and 0s for efficient updates
        ones = np.where(binary_h == 1)[0]
        zeros = np.where(binary_h == 0)[0]
        
        # Local search with swaps
        max_swaps = 5 * n_points
        swap_count = 0
        t2 = start_time + 0.85 * budget_s
        
        while time.time() < t2 and swap_count < max_swaps:
            # Pick two random indices: one from ones, one from zeros
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Simulate swap
            temp_h = binary_h.copy()
            temp_h[i], temp_h[j] = temp_h[j], temp_h[i]
            
            # Compute overlap changes
            new_corr = compute_corr_fft(temp_h)
            
            # Check if new minimum overlap is better
            new_min = np.min(new_corr)
            if new_min > current_min:
                binary_h = temp_h
                current_min = new_min
                current_c5 = dx * np.max(new_corr)
                eval_and_update_best(binary_h, current_c5)
                
                # Update indices
                ones = np.where(binary_h == 1)[0]
                zeros = np.where(binary_h == 0)[0]
            
            swap_count += 1
            
        # Additional restarts
        for _ in range(5):
            if time.time() - start_time > 0.95 * budget_s:
                break
            # Create new random binary
            rand_h = np.random.rand(n_points)
            rand_h = project_to_sum_box(rand_h, TARGET_SUM)
            # Round to binary
            median_val = np.median(rand_h)
            rand_binary = (rand_h >= median_val).astype(float)
            # Fix sum
            ones_count = np.sum(rand_binary)
            if ones_count > TARGET_SUM:
                ones_indices = np.where(rand_binary == 1)[0]
                flip_count = int(ones_count - TARGET_SUM)
                flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
                rand_binary[flip_indices] = 0
            elif ones_count < TARGET_SUM:
                zeros_indices = np.where(rand_binary == 0)[0]
                flip_count = int(TARGET_SUM - ones_count)
                flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
                rand_binary[flip_indices] = 1
            rand_binary = project_to_sum_box(rand_binary, TARGET_SUM)
            trial_c5 = evaluate_c5(rand_binary)
            eval_and_update_best(rand_binary, trial_c5)
    
    # Return best solution
    if best_h is None:
        # Fallback: use Sturmian sequence
        h_sturm = sturmian_sequence(n_points)
        h_sturm = project_to_sum_box(h_sturm, TARGET_SUM)
        best_h = h_sturm
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
