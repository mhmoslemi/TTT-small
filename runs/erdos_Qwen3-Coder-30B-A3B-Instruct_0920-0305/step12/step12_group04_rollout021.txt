import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 200
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = correlate(h, 1 - h, mode="full")
    O = corr * dx
    return np.max(O)

def compute_gradient_for_lag(h, lag):
    """Compute gradient of O(k) for a given lag k"""
    N = len(h)
    forward = np.zeros(N)
    backward = np.zeros(N)
    
    if lag >= 0:
        forward[:N - lag] = h[lag:]
        backward[lag:] = h[:N - lag]
    else:
        k = -lag
        forward[k:] = h[:N - k]
        backward[:N - k] = h[k:]
    
    grad = np.ones(N) - forward - backward
    return grad

def gradient_descent(h_init, target_sum, max_iter, time_limit):
    """Run gradient descent with projected subgradients"""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    t_start = time.time()
    iteration = 0
    
    while time.time() - t_start < time_limit and iteration < max_iter:
        # Evaluate current state
        current_c5 = evaluate_c5(h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute correlation to find max O
        corr = correlate(h, 1 - h, mode="full")
        O = corr * dx
        max_idx = np.argmax(O)
        lag = max_idx - (n_points - 1)
        
        # Compute gradient for this lag
        grad = compute_gradient_for_lag(h, lag)
        
        # Adaptive step size
        step_size = 0.3 / (1 + iteration * 0.001)
        
        # Update step
        h_new = h - step_size * grad
        
        # Project to feasible region
        h_new = project_to_sum_box(h_new, target_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new)
        
        # Accept the new solution
        h = h_new
        
        if new_c5 < current_c5:
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        
        iteration += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial solution
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_to_sum_box(h0, n_points / 2)
    else:
        h0 = np.random.rand(n_points)
        h0 = project_to_sum_box(h0, n_points / 2)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Run multiple restarts with gradient descent
    restarts = 20
    iterations_per_restart = 2000
    time_per_restart = (budget_s * 0.9) / restarts
    
    for i in range(restarts):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Random perturbation of initial solution
        h_init = h0 + np.random.normal(0, 0.02, n_points)
        h_init = project_to_sum_box(h_init, n_points / 2)
        
        # Gradient descent
        h_result, c5_result = gradient_descent(
            h_init, 
            n_points / 2, 
            iterations_per_restart, 
            time_per_restart
        )
        
        if c5_result < best_c5:
            best_c5 = c5_result
            best_h = h_result.copy()
    
    # Final refinement
    if time.time() - start_time < budget_s - 5:
        h_refined, c5_refined = gradient_descent(
            best_h, 
            n_points / 2, 
            1000, 
            budget_s - (time.time() - start_time) - 5
        )
        
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
