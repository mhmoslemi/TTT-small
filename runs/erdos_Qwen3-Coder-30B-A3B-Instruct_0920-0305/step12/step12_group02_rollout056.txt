```python
import numpy as np
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Precompute distances
    x = np.arange(n_points) * dx
    d = np.abs(x - 1.0)  # distance from centre
    
    best_c5 = np.inf
    best_h = None
    
    # Try different parameter values
    # Sample p values in log scale for better coverage of the parameter space
    p_values = np.logspace(np.log10(0.3), np.log10(5), 200)
    
    for p in p_values:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute alpha
        alpha_raw = (p + 1) / (2 * p)
        alpha = min(alpha_raw, 1.0)
        
        # Compute raw shape
        ratio = d / alpha
        f = np.where(ratio <= 1, 1 - np.power(ratio, p), 0.0)
        
        sum_f = f.sum()
        
        # Skip if invalid
        if sum_f == 0:
            continue
            
        # Compute feasible range for c
        c_max1 = TARGET_SUM / sum_f
        
        # From constraint b + c <= 1 and b = (TARGET_SUM - c * sum_f) / n_points
        # We get: (TARGET_SUM - c * sum_f) / n_points + c <= 1
        # => TARGET_SUM/n_points + c*(1 - sum_f/n_points) <= 1
        # => c*(1 - sum_f/n_points) <= 1 - TARGET_SUM/n_points = 0.5
        denom = 1 - sum_f / n_points
        if denom > 0:
            c_max2 = 0.5 / denom
        else:
            c_max2 = np.inf
            
        c_max = min(c_max1, c_max2)
        
        if c_max <= 0:
            continue
            
        # Perform simple grid search on c since it's 1D
        c_values = np.linspace(0, min(c_max, 1.0), 20)
        best_c_for_p = None
        best_c5_for_p = np.inf
        
        for c in c_values:
            b = (TARGET_SUM - c * sum_f) / n_points
            if b < 0 or b + c > 1:
                continue
                
            h = b + c * f
            h = np.clip(h, 0, 1)
            
            c5_val = compute_c5(h)
            if c5_val < best_c5_for_p:
                best_c5_for_p = c5_val
                best_c_for_p = c
                
        if best_c5_for_p < best_c5:
            best_c5 = best_c5_for_p
            b = (TARGET_SUM - best_c_for_p * sum_f) / n_points
            best_h = b + best_c_for_p * f
            best_h = np.clip(best_h, 0, 1)
            best_h = project_to_sum_box(best_h, TARGET_SUM)
            best_c5 = compute_c5(best_h)
    
    # Final refinement with local optimization if time permits
    if best_h is not None and time.time() - start_time < budget_s - 5:
        def objective(h):
            return compute_c5(h)
            
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            # Use a more refined local optimization
            x0 = best_h.copy()
            
            # Project initial point
            x0 = project_to_sum_box(x0, TARGET_SUM)
            
            # Try different optimization methods
            methods = ['L-BFGS-B', 'TNC', 'SLSQP', 'trust-constr']
            for method in methods:
                if time.time() - start_time > budget_s - 5:
                    break
                try:
                    res = minimize(
                        objective, 
                        x0, 
                        method=method, 
                        bounds=bounds, 
                        constraints=cons,
                        options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
                    )
                    # Refine with projection
                    if res.success:
                        h_refined = project_to_sum_box(res.x, TARGET_SUM)
                        c5_refined = compute_c5(h_refined)
                        if c5_refined < best_c5:
                            best_c5 = c5_refined
                            best_h = h_refined
                    break
                except:
                    continue
        except:
            pass
            
    return (best_h.tolist(), float(best_c5), n_points)
```
