```python
import numpy as np
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Precompute distances
    x = np.arange(n_points) * dx
    d = np.abs(x - 1.0)  # distance from centre
    
    best_c5 = np.inf
    best_h = None
    
    # Parameter search over p values
    # Sample p values from 0.3 to 5 with log spacing to focus on interesting regions
    p_values = np.logspace(np.log10(0.3), np.log10(5), 200)
    
    for p in p_values:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute alpha parameter
        alpha_raw = (p + 1) / (2 * p)
        alpha = min(alpha_raw, 1.0)
        
        # Compute raw shape function
        ratio = d / alpha
        f = np.where(ratio <= 1, 1 - np.power(ratio, p), 0.0)
        
        # Compute sum of shape
        sum_f = f.sum()
        
        # Skip if shape is degenerate
        if sum_f == 0:
            continue
            
        # Compute feasible range for c
        c_max1 = TARGET_SUM / sum_f
        denom = 1 - sum_f / n_points
        if denom > 0:
            c_max2 = 0.5 / denom
        else:
            c_max2 = np.inf
            
        c_max = min(c_max1, c_max2)
        if c_max <= 0:
            continue
            
        # Perform golden-section search on c in [0, c_max] to minimize C5
        def obj(c):
            b = (TARGET_SUM - c * sum_f) / n_points
            h = b + c * f
            h = np.clip(h, 0, 1)
            return evaluate_c5(h)
        
        # Golden-section search
        gr = (1 + np.sqrt(5)) / 2
        a, b = 0.0, c_max
        res = (gr - 1) * (b - a)
        x1 = a + res
        x2 = b - res
        
        for _ in range(30):
            if time.time() - start_time > budget_s - 10:
                break
            if obj(x1) < obj(x2):
                b = x2
                x2 = x1
                res = (gr - 1) * (b - a)
                x1 = a + res
            else:
                a = x1
                x1 = x2
                res = (gr - 1) * (b - a)
                x2 = b - res
        
        c_opt = (a + b) / 2.0
        c5_val = obj(c_opt)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            h = (TARGET_SUM - c_opt * sum_f) / n_points + c_opt * f
            h = np.clip(h, 0, 1)
            # Apply projection for exact sum constraint
            h = project_to_sum_box(h, TARGET_SUM)
            best_c5 = evaluate_c5(h)
            best_h = h.copy()
    
    # Final refinement using L-BFGS-B if time permits
    if best_h is not None and time.time() - start_time < budget_s - 10:
        def obj_proj(x):
            h = project_to_sum_box(x, TARGET_SUM)
            return evaluate_c5(h)
            
        res = minimize(obj_proj, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                       options={'maxiter': 300, 'ftol': 1e-12})
        
        if res.success:
            h_opt = project_to_sum_box(res.x, TARGET_SUM)
            c5_opt = evaluate_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    
    return (best_h.tolist(), float(best_c5), n_points)
```
