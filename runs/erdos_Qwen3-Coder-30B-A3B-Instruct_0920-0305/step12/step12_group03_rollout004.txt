import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import differential_evolution, minimize
import random

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Simple projection: adjust values to meet sum constraint
    current_sum = np.sum(x)
    if current_sum == target_sum:
        return np.clip(x, 0, 1)
    
    # First clip to [0,1]
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    
    if current_sum > target_sum:
        # Reduce values
        diff = current_sum - target_sum
        indices = np.argsort(x)[::-1]  # Sort by decreasing values
        for i in indices:
            if diff <= 0:
                break
            reduce = min(x[i], diff)
            x[i] -= reduce
            diff -= reduce
    else:
        # Increase values
        diff = target_sum - current_sum
        indices = np.argsort(x)  # Sort by increasing values
        for i in indices:
            if diff <= 0:
                break
            increase = min(1 - x[i], diff)
            x[i] += increase
            diff -= increase
    
    return x
