import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def balanced_sturmian(n):
    """Generate a balanced Sturmian sequence"""
    phi = (np.sqrt(5) - 1) / 2
    x = np.array([int((i * phi) % 1 < 0.5) for i in range(n)])
    # Adjust to have exactly n//2 ones
    ones_count = np.sum(x)
    if ones_count > n // 2:
        # Turn some 1s to 0s
        ones_indices = np.where(x == 1)[0]
        flip_count = ones_count - n // 2
        flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
        x[flip_indices] = 0
    elif ones_count < n // 2:
        # Turn some 0s to 1s
        zeros_indices = np.where(x == 0)[0]
        flip_count = n // 2 - ones_count
        flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
        x[flip_indices] = 1
    return x

def compute_autocorr_fft(x):
    """Compute autocorrelations using FFT"""
    x_rev = x[::-1]
    b = x
    L = 2 * len(x) - 1
    corr = np.fft.ifft(np.fft.fft(x_rev, L) * np.fft.fft(b, L)).real
    return corr[:len(x)]

def compute_delta_S(S, i, j, x, n, ones, zeros):
    """Compute change in autocorrelations due to swapping x[i] and x[j]"""
    delta_S = np.zeros(n)
    # For each lag ℓ, compute change in S_ℓ
    for ℓ in range(1, n):
        # Count how many positions are affected in the autocorrelation
        affected_positions = []
        if i - ℓ >= 0:
            affected_positions.append(i - ℓ)
        if i + ℓ < n:
            affected_positions.append(i + ℓ)
        if j - ℓ >= 0:
            affected_positions.append(j - ℓ)
        if j + ℓ < n:
            affected_positions.append(j + ℓ)
        
        # Get affected points
        affected_i = [pos for pos in affected_positions if pos == i or pos == j]
        affected_ij = [pos for pos in affected_positions if pos in [i, j] and pos != i and pos != j]
        affected_other = [pos for pos in affected_positions if pos not in [i, j]]
        
        # Offsets and contributions
        contrib = 0
        if i in affected_positions:
            if i - ℓ >= 0 and i + ℓ < n:
                # i contributes to both x[i]*x[i-ℓ] and x[i]*x[i+ℓ] 
                if x[i - ℓ]:
                    contrib -= 1  # reduce
                if x[i + ℓ]:
                    contrib -= 1  # reduce
            elif i - ℓ >= 0:
                if x[i - ℓ]:
                    contrib -= 1  # reduce
            elif i + ℓ < n:
                if x[i + ℓ]:
                    contrib -= 1  # reduce
        
        if j in affected_positions:
            if j - ℓ >= 0 and j + ℓ < n:
                # j contributes to both x[j]*x[j-ℓ] and x[j]*x[j+ℓ]
                if x[j - ℓ]:
                    contrib += 1  # increase
                if x[j + ℓ]:
                    contrib += 1  # increase
            elif j - ℓ >= 0:
                if x[j - ℓ]:
                    contrib += 1  # increase
            elif j + ℓ < n:
                if x[j + ℓ]:
                    contrib += 1  # increase

        delta_S[ℓ] = contrib
    return delta_S

def swap_based_refinement(x, S, n, ones, zeros, current_m, dx, N, max_time):
    """Perform hill-climb refinement with swaps"""
    t0 = time.time()
    best_x = x.copy()
    best_m = current_m
    best_c5 = dx * (N - current_m)

    # For faster updates use a precomputed array of all influence
    def get_delta_S_fast(i, j):
        # Compute the change in S_ℓ due to swapping x[i] and x[j]
        delta_S = np.zeros(n)
        # For each lag: 
        for ℓ in range(1, n):
            contrib = 0
            # Check if i and j affect positions due to lag ℓ  
            if i - ℓ >= 0 or i + ℓ < n or j - ℓ >= 0 or j + ℓ < n:
                # For simplicity, multiply contributions by their respective values
                # x_i and x_j. But since we are swapping bits, this logic is simpler:
                # x_i changes from 1 to 0, x_j from 0 to 1
                contrib = 0
                # Consider x[i] in lag ℓ terms: x[i]*x[i-ℓ] and x[i]*x[i+ℓ] 
                # Similarly for x[j]
                # As an approximation:
                if i - ℓ >= 0:
                    contrib += x[i - ℓ]
                if i + ℓ < n:
                    contrib += x[i + ℓ]  
                if j - ℓ >= 0:
                    contrib -= x[j - ℓ]
                if j + ℓ < n:
                    contrib -= x[j + ℓ]
            delta_S[ℓ] = contrib
        return delta_S

    # Main refinement loop
    iters = 0
    while time.time() - t0 < max_time:
        # Propose random swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        if i == j:
            continue

        # Compute change in autocorrelation
        delta_S = get_delta_S_fast(i, j)
        new_S = S + delta_S
        new_m = np.min(new_S[1:])  # exclude lag 0
        
        if new_m > best_m:
            # Accept the swap
            x[i], x[j] = x[j], x[i]
            S = new_S
            best_m = new_m
            best_c5 = dx * (N - best_m)
            ones.remove(i)
            ones.add(j)
            zeros.remove(j)
            zeros.add(i)
            # Update the cached best_x to match current
            best_x = x.copy()
            
        iters += 1
        
        if iters % (10000) == 0:
            # Occasionally recompute best_m to ensure we didn't miss the global
            curr_m = np.min(S[1:])
            if curr_m > best_m:
                best_m = curr_m
                best_c5 = dx * (N - best_m)
                best_x = x.copy()

    return best_x, best_m, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Parameters
    n = 128  # Primary resolution
    dx = 2.0 / n
    N = n // 2
    
    # Variables
    x = cp.Variable(n, boolean=True)
    m = cp.Variable()
    
    # McCormick constraints for products x[i] * x[i+ℓ]
    constraints = []
    y_vars = {}  # Store y[i, ℓ] variables
    
    # Mass constraint
    constraints.append(cp.sum(x) == N)
    
    # Symmetry breaking  
    constraints.append(x[0] == 1)
    
    # Lexicographic ordering on first half (for symmetry breaking)
    for i in range(N-1):
        constraints.append(x[i] >= x[i+1])
    
    # Create y inequalities for all lags and constrained indices
    S_vars = []  # Store S_ℓ variables
    
    for ℓ in range(1, n):
        y_ℓ = cp.Variable((n - ℓ), boolean=True)
        # Store for later use
        y_vars[ℓ] = y_ℓ
        
        # McCormick relaxations
        for i in range(n - ℓ):
            constraints.append(y_ℓ[i] <= x[i])
            constraints.append(y_ℓ[i] <= x[i+ℓ])
            constraints.append(y_ℓ[i] >= x[i] + x[i+ℓ] - 1)
        
        # S_ℓ = sum of y[i, ℓ]  
        S_ℓ = cp.sum(y_ℓ)
        S_vars.append(S_ℓ)
        
        # Epigraph constraint: S_ℓ >= m for all ℓ    
        constraints.append(S_ℓ >= m)
        
    # Objective: maximize m
    objective = cp.Maximize(m)
    prob = cp.Problem(objective, constraints)
    
    # Time allocation
    solve_time = 0.45 * budget_s
    refine_time = 0.35 * budget_s
    fallback_time = 0.15 * budget_s
    
    # Try to solve
    try:
        prob.solve(solver=cp.CBC, max_seconds=solve_time)
        if prob.status in [cp.OPTIMAL, cp.UNBOUNDED]:
            # Extract integer solution
            x_val = np.round(x.value).astype(int)
            if np.allclose(np.sum(x_val), N):
                # Compute exact C5
                corr = correlate(x_val, x_val, mode='full')
                max_corr = np.max(corr[corr != corr[n-1]])  # skip zero lag
                # Compute m using the conversion:
                # If S_ℓ = sum x_i x_{i+ℓ} then for all ℓ ≠ 0, C5 = dx * (N - m)  
                # We also use S_ℓ = x_val * (1 - x_valilde) -> but since both are binar
                # Actually: sum x_i (1-x_{i+ℓ}) = N - sum x_i x_{i+ℓ} = N - S_ℓ 
                # Thus   C5 = dx * max(N - S_ℓ) = dx * (N - min S_ℓ) = dx * (N - m)  
                m_found = np.min(corr[n:-n+1])  # excluding zero lag (and duplicates)
                c5 = dx * (N - m_found)
                best_x = x_val
                best_c5 = c5
                print(f"Found integer solution with m={m_found}, C5={c5}")
            else:
                # Revert to fallback
                best_x = None
                best_c5 = float('inf')
        else:
            # Solver failed or timed out; revert to fallback
            best_x = None
            best_c5 = float('inf')
    except Exception as e:
        best_x = None
        best_c5 = float('inf')
    
    if best_x is None:
        # Fallback: use Sturmian sequence and refine
        x_initial = balanced_sturmian(n)
        # Correct sum to N if needed
        current_sum = np.sum(x_initial)
        if current_sum > N:
            ones_to_zero = np.random.choice(np.where(x_initial)[0], current_sum - N, replace=False)
            x_initial[ones_to_zero] = 0
        elif current_sum < N:
            zeros_to_one = np.random.choice(np.where(~x_initial)[0], N - current_sum, replace=False)
            x_initial[zeros_to_one] = 1
            
        best_x = x_initial.astype(int)
        # Compute the actual minimum autocorrelation
        corr_full = correlate(best_x, best_x, mode='full')
        corr_nonzero = corr_full[n:-n+1]
        best_m = np.min(corr_nonzero)
        best_c5 = dx * (N - best_m)
        
    # Refinement phase with swaps 
    t_refine = time.time() 
    refined_x, refined_m, refined_c5 = swap_based_refinement(
        best_x.copy(), 
        correlate(best_x, best_x, mode='full')[n:-n+1], 
        n, 
        set(np.where(best_x == 1)[0]), 
        set(np.where(best_x == 0)[0]), 
        best_m, 
        dx, 
        N, 
        refine_time
    )
    
    # Even more refinement using solar annealing
    def simulated_annealing_refinement(x_init, dx, N, max_time):
        x = x_init.copy()
        ones = set(np.where(x == 1)[0])
        zeros = set(np.where(x == 0)[0])
        
        # Compute initial matrix
        S = correlate(x, x, mode='full')[n:-n+1]
        m = np.min(S)
        current_c5 = dx * (N - m)
        t0 = time.time()
        
        temp = 0.01 * current_c5  # Initial temperature
        
        while time.time() - t0 < max_time:
            i = np.random.choice(list(ones))
            j = np.random.choice(list(zeros))
            if i == j:
                continue
                
            # Compute delta in S values
            delta_S = np.zeros(n)
            for ℓ in range(1, n):
                contrib = 0
                if i - ℓ >= 0:
                    contrib += x[i - ℓ]
                if i + ℓ < n:
                    contrib += x[i + ℓ]  
                if j - ℓ >= 0:
                    contrib -= x[j - ℓ]
                if j + ℓ < n:
                    contrib -= x[j + ℓ]
                delta_S[ℓ] = contrib
                
            new_S = S + delta_S
            new_m = np.min(new_S) 
            delta_m = new_m - m
            
            if delta_m > 0:  # Improvement
                x[i], x[j] = x[j], x[i]
                ones.remove(i)
                ones.add(j)
                zeros.remove(j)
                zeros.add(i)
                S = new_S
                m = new_m
                current_c5 = dx * (N - m)
            elif np.random.rand() < np.exp(delta_m / temp):  # Accept with probability
                x[i], x[j] = x[j], x[i]
                ones.remove(i)
                ones.add(j)
                zeros.remove(j)
                zeros.add(i)
                S = new_S
                m = new_m
                current_c5 = dx * (N - m) 
            
            temp *= 0.99  # Cool down
            
        return x, m, current_c5
    
    final_x, final_m, final_c5 = simulated_annealing_refinement(
        refined_x,
        dx,
        N,
        fallback_time
    )
    
    # Final verification of best result
    corr_final = correlate(final_x, final_x, mode='full')[n:-n+1]
    final_m_verified = np.min(corr_final)
    final_c5_verified = dx * (N - final_m_verified)
    
    return (final_x.tolist(), float(final_c5_verified), n)
