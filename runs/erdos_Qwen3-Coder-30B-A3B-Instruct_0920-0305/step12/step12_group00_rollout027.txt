def toeplitz_A(lag, n):
    """Generate the Toeplitz matrix A_lag for lag."""
    A = np.zeros((n, n))
    for i in range(n):
        j = (i + lag) % n
        A[i, j] = 1
        if lag != 0:
            A[j, i] = 1
    return A

def compute_autocorrelations(h):
    """Compute all autocorrelations using FFT."""
    h_rev = h[::-1]
    b = 1 - h
    corr = np.convolve(h, b, mode='full')
    return corr[len(h)-1:]

def evaluate_c5(h):
    """Evaluate C5 = max_k ∫ h(x)(1 - h(x+k)) dx."""
    corr = compute_autocorrelations(h)
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto [0,1] box with sum = target_sum."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def round_to_binary(X, N):
    """Round the PSD matrix X to binary vector using randomized rounding."""
    # Eigen-decomposition
    try:
        eigvals, eigvecs = eigh(X)
        # Keep only positive eigenvalues
        idx = eigvals > 1e-8
        eigvals = eigvals[idx]
        eigvecs = eigvecs[:, idx]
        
        # Form V matrix
        V = eigvecs * np.sqrt(eigvals)
        
        best_h = None
        best_c5 = float('inf')
        
        # Try multiple random rounds
        for _ in range(5):
            g = np.random.randn(V.shape[1])
            scores = V @ g
            indices = np.argsort(-scores)  # descending order
            h = np.zeros(len(scores))
            h[indices[:N]] = 1
            h = project_to_sum_box(h, N)
            
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
        return best_h, best_c5
    except Exception as e:
        # Fallback to direct rounding via diagonal
        h = (np.diag(X) > 0.5).astype(float)
        h = project_to_sum_box(h, N)
        return h, evaluate_c5(h)

def hill_climb(h, N, c5_prev):

    # Precompute all autocorrelations once
    corr = compute_autocorrelations(h)
    min_corr = np.max(corr)
    
    # Maintain indices efficiently
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    max_swaps = 5 * len(h)
    swaps_used = 0
    improved = True
    
    while improved and swaps_used < max_swaps:
        improved = False
        # Try a few random swaps
        for _ in range(100):
            if swaps_used >= max_swaps:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Simulate a swap - compute ΔS for all lags
            delta_corr = np.zeros(len(corr))
            
            # Compute the change in correlation for all lags (this is a bit easier in practice)
            # We'll compute the full new correlations for small changes to verify
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
            corr_new = compute_autocorrelations(h_new)
            min_corr_new = np.max(corr_new)

            if min_corr_new < min_corr:
                h = h_new
                corr = corr_new
                min_corr = min_corr_new
                # Update indices
                ones = np.where(h == 1)[0]

                    c5_prev = min_corr
            swaps_used += 1
    
    # Final evaluate C5
    c5_final = evaluate_c5(h)
    return h, c5_final
