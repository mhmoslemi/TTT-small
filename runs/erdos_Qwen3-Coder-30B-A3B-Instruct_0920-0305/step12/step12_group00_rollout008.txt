def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def convolve_fft(a, b):
    """Perform convolution using FFT, returns same length as b"""
    L = len(a) + len(b) - 1
    L_fft = 1 << (L - 1).bit_length()  # next power of 2
    A = np.fft.fft(a, L_fft)
    B = np.fft.fft(b, L_fft)
    result = np.fft.ifft(A * B).real[:len(b)]
    return result

def local_search_mutation(h, step_size, max_iter):
    """Local search with random mutations"""
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    
    for _ in range(max_iter):
        neighbor = current_h.copy()
        idx = np.random.randint(len(neighbor))
        neighbor[idx] += np.random.normal(0, step_size)
        neighbor[idx] = np.clip(neighbor[idx], 0, 1)
        
        neighbor = project_to_sum_box(neighbor, TARGET_SUM)
        new_c5 = evaluate_c5(neighbor)
        
        if new_c5 < current_c5:
            current_h = neighbor
            current_c5 = new_c5
            
    return current_h, current_c5

def local_search_swapping(h, max_swaps):
    """Local search with random pair swapping"""
    current_h = h.copy()

    for _ in range(max_swaps):
        i, j = np.random.choice(len(current_h), 2, replace=False)
        neighbor = current_h.copy()
        neighbor[i], neighbor[j] = neighbor[j], neighbor[i]
        
        neighbor = project_to_sum_box(neighbor, TARGET_SUM)
        new_c5 = evaluate_c5(neighbor)
        
        if new_c5 < current_c5:
            current_h = neighbor
            current_c5 = new_c5
            
    return current_h, current_c5

def local_search_gradient_descent(h, step_size, max_iter):
