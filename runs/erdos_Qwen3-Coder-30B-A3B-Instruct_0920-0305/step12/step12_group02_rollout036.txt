```python
import numpy as np
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_c5(h):
    """Compute C5 using correlation"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    x = np.arange(n_points) * dx
    d = np.abs(x - 1.0)  # distance from centre
    
    best_c5 = np.inf
    best_h = None
    
    # Sample p values logarithmically
    p_values = np.logspace(np.log10(0.3), np.log10(5), 200)
    
    # For each p, compute the optimal c
    for p in p_values:
        if time.time() - start_time > budget_s - 5:
            break
            
        # compute alpha
        alpha_raw = (p + 1) / (2 * p)
        alpha = min(alpha_raw, 1.0)
        
        # compute raw shape
        ratio = d / alpha
        f = np.where(ratio <= 1, 1 - np.power(ratio, p), 0.0)
        
        sum_f = f.sum()
        if sum_f == 0:
            continue
            
        # compute feasible range for c
        # c_max1 from b >=0: c <= target_sum / sum_f
        c_max1 = target_sum / sum_f
        
        # c_max2 from b + c <=1
        # b = (target_sum - c*sum_f) / N
        # condition: b + c <= 1 -> (target_sum - c*sum_f)/N + c <= 1
        # -> target_sum/N + c*(1 - sum_f/N) <= 1
        # -> c*(1 - sum_f/N) <= 1 - target_sum/N = 0.5
        denom = 1 - sum_f / n_points
        if denom > 0:
            c_max2 = 0.5 / denom
        else:
            c_max2 = np.inf  # if denom <= 0, constraint is satisfied for all c
            
        c_max = min(c_max1, c_max2)
        if c_max <= 0:
            continue
            
        # Use simple grid search to find optimal c in [0, c_max]
        best_c = 0
        best_c5_local = np.inf
        
        # Sample a few points in the range [0, c_max]
        c_samples = np.linspace(0, c_max, 20)
        for c in c_samples:
            b = (target_sum - c * sum_f) / n_points
            if b < 0 or b + c > 1:
                continue
            h = b + c * f
            h = np.clip(h, 0, 1)
            c5_val = compute_c5(h)
            if c5_val < best_c5_local:
                best_c5_local = c5_val
                best_c = c
                
        # Refine using golden section search
        def obj(c):
            b = (target_sum - c * sum_f) / n_points
            if b < 0 or b + c > 1:
                return np.inf
            h = b + c * f
            h = np.clip(h, 0, 1)
            return compute_c5(h)
            
        # Golden section search between 0 and c_max
        gr = (np.sqrt(5) - 1) / 2
        a, b = 0, c_max
        c1 = a + (1 - gr) * (b - a)
        c2 = a + gr * (b - a)
        
        for _ in range(30):
            if time.time() - start_time > budget_s - 5:
                break
            if obj(c1) > obj(c2):
                a = c1
                c1 = c2
                c2 = a + gr * (b - a)
            else:
                b = c2
                c2 = c1
                c1 = a + (1 - gr) * (b - a)
                
        c_opt = (a + b) / 2.0
        b_opt = (target_sum - c_opt * sum_f) / n_points
        
        if b_opt < 0 or b_opt + c_opt > 1:
            continue
        
        h = b_opt + c_opt * f
        h = np.clip(h, 0, 1)
        c5_val = compute_c5(h)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
    
    # Final refinement with L-BFGS-B if time allows
    if best_h is not None and time.time() - start_time < budget_s - 5:
        def obj_proj(x):
            h = project_to_sum_box(x, TARGET_SUM)
            return compute_c5(h)
            
        x0 = best_h.copy()
        res = minimize(obj_proj, x0, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                      options={'maxiter': 300, 'ftol': 1e-12})
        if res.success:
            h_opt = project_to_sum_box(res.x, TARGET_SUM)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    
    # Make sure final h is properly projected
    if best_h is not None:
        best_h = project_to_sum_box(best_h, TARGET_SUM)
        best_c5 = compute_c5(best_h)
        
    return (best_h.tolist(), float(best_c5), n_points)
```
