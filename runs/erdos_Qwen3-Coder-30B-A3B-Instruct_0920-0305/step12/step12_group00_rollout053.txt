import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
from scipy.optimize import minimize_scalar

def balanced_sturmian(n):
    """Generate a balanced Sturmian word with n bits, sum=n//2"""
    phi = (np.sqrt(5) - 1) / 2
    x = np.zeros(n, dtype=int)
    for i in range(n):
        if (i * phi) % 1 < 0.5:
            x[i] = 1
    # Adjust to exact sum
    current_sum = np.sum(x)
    target_sum = n // 2
    if current_sum > target_sum:
        ones_indices = np.where(x == 1)[0]
        flips = current_sum - target_sum
        for i in range(flips):
            x[ones_indices[i]] = 0
    elif current_sum < target_sum:
        zeros_indices = np.where(x == 0)[0]
        flips = target_sum - current_sum
        for i in range(flips):
            x[zeros_indices[i]] = 1
    return x

def compute_autocorr_fft(h):
    """Compute autocorrelations efficiently"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr[1:]  # exclude lag 0

def compute_c5_from_corr(corr, dx):
    """Convert correlation to C5 value"""
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def swap_update(S, i, j, ones, zeros, n):
    """Compute change in autocorrelations when swapping bits at i and j"""
    delta_S = np.zeros(n-1)
    for l in range(1, n):
        # Check if indices involved in lag l exist
        if i-l >= 0:
            delta_S[l-1] -= S[i-l]
        if i+l < n:
            delta_S[l-1] -= S[i+l]
        if j-l >= 0:
            delta_S[l-1] -= S[j-l]
        if j+l < n:
            delta_S[l-1] -= S[j+l]
        
        # Add contributions from new positions
        if i-l >= 0:
            delta_S[l-1] += S[i-l] if i-l in ones else 0
        if i+l < n:
            delta_S[l-1] += S[i+l] if i+l in ones else 0
        if j-l >= 0:
            delta_S[l-1] += S[j-l] if j-l in ones else 0
        if j+l < n:
            delta_S[l-1] += S[j+l] if j+l in ones else 0
    
    delta_S = np.array(delta_S)
    return delta_S

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    dx = 2.0 / 128
    n = 128
    N = n // 2
    
    # Stage 1: Exact ILP optimization
    best_c5 = float('inf')
    best_h = None
    best_m = 0
    
    # ILP Model
    x = cp.Variable(n, boolean=True)
    
    # Fix first element to break symmetry
    constraints = [x[0] == 1]
    
    # Enforce sum constraint
    constraints.append(cp.sum(x) == N)
    
    # Enforce lexicographic ordering
    for i in range(N-1):
        constraints.append(x[i] >= x[i+1])
    
    # Build y variables and constraints for products
    big_M = 1
    y = {}
    S = {}
    
    for l in range(1, n):
        for i in range(n-l):
            y[(i,l)] = cp.Variable(boolean=True)
            constraints.extend([
                y[(i,l)] <= x[i],
                y[(i,l)] <= x[i+l],
                y[(i,l)] >= x[i] + x[i+l] - 1,
            ])
        # Compute S_l
        S[l] = cp.sum([y[(i,l)] for i in range(n-l)])
    
    # Add constraints that all S_l should be at least m
    m = cp.Variable()
    for l in range(1, n):
        constraints.append(S[l] >= m)
    
    # Create the problem
    objective = cp.Maximize(m)
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    max_seconds = 0.45 * budget_s
    try:
        prob.solve(solver=cp.CBC, max_seconds=max_seconds)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_val = np.round(x.value).astype(int)
            if np.sum(x_val) == N:
                S_vals = [S[l].value for l in range(1, n)]
                m_val = m.value
                c5_val = dx * (N - m_val)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = x_val
                    best_m = m_val
    except:
        pass
    
    # Stage 2: Binary local search refinement
    if best_h is not None:
        h = best_h.copy()
        current_h = h.copy()
        current_c5 = best_c5
        current_m = best_m
        ones = np.where(current_h == 1)[0]
        zeros = np.where(current_h == 0)[0]
        S = compute_autocorr_fft(current_h)
        max_iter = 10000
        iter_count = 0
        t1 = start_time + 0.8 * budget_s
        while time.time() < t1 and iter_count < max_iter:
            # Pick random swap
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            new_h = current_h.copy()
            new_h[i] = 0
            new_h[j] = 1
            
            new_S = S.copy()
            for l in range(1, n):
                if i-l >= 0:
                    new_S[l-1] -= current_h[i-l]
                if i+l < n:
                    new_S[l-1] -= current_h[i+l]
                if j-l >= 0:
                    new_S[l-1] -= current_h[j-l]
                if j+l < n:
                    new_S[l-1] -= current_h[j+l]
                
                if i-l >= 0:
                    new_S[l-1] += new_h[i-l]
                if i+l < n:
                    new_S[l-1] += new_h[i+l]
                if j-l >= 0:
                    new_S[l-1] += new_h[j-l]
                if j+l < n:
                    new_S[l-1] += new_h[j+l]
            
            new_m = np.min(new_S)
            new_c5 = dx * (N - new_m)
            
            if new_m > current_m:
                current_h = new_h
                current_m = new_m
                current_c5 = new_c5
                ones = np.where(current_h == 1)[0]
                zeros = np.where(current_h == 0)[0]
                S = new_S
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
            iter_count += 1
    
    # Stage 3: Fallback to Sturmian + hill-climb
    if best_h is None:
        h = balanced_sturmian(n)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        S = compute_autocorr_fft(h)
        current_m = np.min(S)
        current_c5 = dx * (N - current_m)
        best_c5 = current_c5
        best_h = h.copy()
        max_iter = 5000
        iter_count = 0
        t2 = start_time + 0.9 * budget_s
        while time.time() < t2 and iter_count < max_iter:
            # Pick random swap
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            new_h = h.copy()
            new_h[i] = 0
            new_h[j] = 1
            
            new_S = S.copy()
            for l in range(1, n):
                if i-l >= 0:
                    new_S[l-1] -= h[i-l]
                if i+l < n:
                    new_S[l-1] -= h[i+l]
                if j-l >= 0:
                    new_S[l-1] -= h[j-l]
                if j+l < n:
                    new_S[l-1] -= h[j+l]
                
                if i-l >= 0:
                    new_S[l-1] += new_h[i-l]
                if i+l < n:
                    new_S[l-1] += new_h[i+l]
                if j-l >= 0:
                    new_S[l-1] += new_h[j-l]
                if j+l < n:
                    new_S[l-1] += new_h[j+l]
            
            new_m = np.min(new_S)
            new_c5 = dx * (N - new_m)
            
            if new_m > current_m:
                h = new_h
                current_m = new_m
                current_c5 = new_c5
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                S = new_S
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            iter_count += 1

    # Final refinement via random restarts
    t_end = start_time + budget_s
    for _ in range(5):
        if time.time() >= t_end:
            break
        # Random initialization
        h = np.random.rand(n)
        h = project_to_sum_box(h, N)
        # Round to binary 
        median_val = np.median(h)
        h = (h >= median_val).astype(int)
        h = project_to_sum_box(h, N)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        S = compute_autocorr_fft(h)
        current_m = np.min(S)
        current_c5 = dx * (N - current_m)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Local search on this init
        max_iter = 1000
        iter_count = 0
        while time.time() < t_end and iter_count < max_iter:
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            new_h = h.copy()
            new_h[i] = 0
            new_h[j] = 1
            
            new_S = S.copy()
            for l in range(1, n):
                if i-l >= 0:
                    new_S[l-1] -= h[i-l]
                if i+l < n:
                    new_S[l-1] -= h[i+l]
                if j-l >= 0:
                    new_S[l-1] -= h[j-l]
                if j+l < n:
                    new_S[l-1] -= h[j+l]
                
                if i-l >= 0:
                    new_S[l-1] += new_h[i-l]
                if i+l < n:
                    new_S[l-1] += new_h[i+l]
                if j-l >= 0:
                    new_S[l-1] += new_h[j-l]
                if j+l < n:
                    new_S[l-1] += new_h[j+l]
            
            new_m = np.min(new_S)
            new_c5 = dx * (N - new_m)
            
            if new_m > current_m:
                h = new_h
                current_m = new_m
                current_c5 = new_c5
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                S = new_S
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            iter_count += 1

    if best_h is None:
        h = balanced_sturmian(n)
        best_h = h.copy() 
        best_c5 = compute_c5_from_corr(compute_autocorr_fft(h), dx)

    return (best_h.tolist(), float(best_c5), n)
