import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 200
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Evaluate C5 using correlation"""
    corr = correlate(h, 1 - h, mode="full")
    O = corr * dx
    return np.max(O)

def compute_gradient_for_lag(h, lag):
    """Compute gradient for the lag that maximizes O"""
    N = len(h)
    forward = np.zeros(N)
    backward = np.zeros(N)
    
    if lag >= 0:
        forward[:N - lag] = h[lag:]
        backward[lag:] = h[:N - lag]
    else:
        k = -lag
        forward[k:] = h[:N - k]
        backward[:N - k] = h[k:]
    
    grad = np.ones(N) - forward - backward
    return grad

def subgradient_descent(h_init, target_sum, max_iter, time_limit):
    """Perform subgradient descent optimization"""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > time_limit:
            break
            
        # Evaluate current state
        current_c5 = evaluate_c5(h)
        
        # Update best solution if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute correlation to find the lag that maximizes O
        corr = correlate(h, 1 - h, mode="full")
        O = corr * dx
        max_lag_idx = np.argmax(O)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient for this lag
        grad = compute_gradient_for_lag(h, max_lag)
        
        # Adaptive step size
        step_size = 0.5 / np.sqrt(iteration + 1)
        
        # Update h
        h_new = h - step_size * grad
        
        # Project onto feasible region
        h_new = project_to_sum_box(h_new, target_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new)
        
        # Accept the new solution
        if new_c5 < current_c5:
            h = h_new
        else:
            h = h_new
            
    return best_h, best_c5

def upsample_h(h_old, n_new):
    """Upsample h from old size to new size using linear interpolation"""
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new
