import numpy as np
from scipy.signal import correlate
from scipy.sparse import diags, eye
import cvxpy as cp
from scipy.linalg import eigh
import time

def build_toeplitz_matrices(n):
    """Build all sparse Toeplitz matrices A_ℓ for ℓ = 1 to n-1"""
    A_list = []
    for lag in range(1, n):
        # Create off-diagonal elements
        diagonals = [np.ones(n - lag)]
        offsets = [lag]
        A = diags(diagonals, offsets, shape=(n, n), format='csr')
        A = A + A.T  # Make symmetric
        A_list.append(A)
    return A_list

def evaluate_c5(h, dx):
    """Fast evaluation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    corr = np.convolve(h_rev, b, mode='full')
    max_corr = np.max(corr)
    return dx * max_corr

def solve_sdp(n, A_list, N, t_sdp):
    """Solve the SDP relaxation"""
    # Variables
    X = cp.Variable((n, n), symmetric=True)
    m = cp.Variable()
    
    # Constraints
    constraints = [
        cp.trace(cp.eye(n) @ X) == N,  # trace(J @ X) == N
        X >> 0,  # PSD
        0 <= X,  # Element-wise constraint
        X <= 1,  # Element-wise constraint
    ]
    
    # Autocorrelation constraints
    for A in A_list:
        constraints.append(cp.trace(A @ X) >= m)
        
    # Objective
    objective = cp.Maximize(m)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-5, verbose=False, time_limit=t_sdp)
    
    return X.value, m.value

            order = np.argsort(-scores)

            # Ensure exact sum
            curr_sum = np.sum(h)
            if curr_sum != N:
                diff = int(N - curr_sum)
                if diff > 0:
                    # Flip zeros to ones
                    zeros = np.where(h == 0)[0]
                    flips = zeros[:diff]
                    h[flips] = 1
                else:
                    # Flip ones to zeros
                    ones = np.where(h == 1)[0]
                    flips = ones[:abs(diff)]
                    h[flips] = 0
            
            c5 = evaluate_c5(h, 2.0 / n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        return best_h, best_c5
    except:
        # Fallback to diagonal rounding if eigen-decomposition fails
        diag_vals = np.diag(X)
        order = np.argsort(-diag_vals)
        h = np.zeros(n)
        h[order[:N]] = 1
        c5 = evaluate_c5(h, 2.0 / n)
        return h, c5

    corr = np.convolve(h, 1-h, mode='full')
    S = corr[n-1:2*n-1]  # Only consider valid lags (1 to n-1)
    m = np.min(S[1:])  # m is min_S for lags 1 to n-1, not including 0
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    best_h = h.copy()
    best_m = m
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < t_hc:
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Compute the change in S_ℓ for all ℓ
        delta_S = np.zeros_like(S)
        # For each non-zero position, update the corresponding lags
        for l in range(1, n):
            if (i + l) < n and (i - l) >= 0:
                delta_S[l] -= h[i + l] + h[i - l]
            if (j + l) < n and (j - l) >= 0:
                delta_S[l] += h[j + l] + h[j - l]
        
        # Update S
        new_S = S + delta_S
        new_m = np.min(new_S[1:])
        
        if new_m > m:
            # Accept the swap
            h[i] = 0
            h[j] = 1
            S = new_S
            m = new_m

            zeros = np.where(h == 0)[0]
            
            if m > best_m:
                best_m = m
                best_h = h.copy()
        
        # Keep track of iterations
        iteration += 1
        
        # Random restart if no progress for a while
        if iteration >= 1000:
            # Highly simplistic recovery - randomize a bit
            pass
            
    return best_h, best_m
