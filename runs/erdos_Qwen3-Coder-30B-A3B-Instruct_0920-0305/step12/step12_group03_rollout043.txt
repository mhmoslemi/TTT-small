import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate initial binary pattern using Beatty sequence
    phi = (1 + np.sqrt(5)) / 2
    positions = [int(np.floor(i * phi)) % n_points for i in range(n_points)]
    positions.sort()
    
    h = np.zeros(n_points, dtype=int)
    for i in range(int(target_sum)):
        h[positions[i]] = 1
    
    # Alternative: use initial_h_values if available

        initial_h_bin = (initial_h > 0.5).astype(int)
        if np.sum(initial_h_bin) == target_sum:

    except:

    # Compute initial correlation
    def compute_correlation(h):
        h_rev = h[::-1]
        b = 1 - h
        corr = np.correlate(h_rev, b, mode='full')
        return corr
    
    C_raw = compute_correlation(h)
    cur_max = np.max(C_raw)
    best_val = cur_max * dx
    best_h = h.copy()
    
    # Simulated Annealing with swap moves
    def update_correlation(C_raw, h, i, j):
        # Prepare delta updates
        delta = np.zeros(2 * n_points - 1)
        # For shift k where i+k is valid
        for k in range(n_points):
            if 0 <= i + k < n_points:
                delta[k] -= (1 - h[i + k])
            if 0 <= j + k < n_points:
                delta[k] += (1 - h[j + k])
        # For shift k where i-k is valid  
        for k in range(1, n_points):
            if 0 <= i - k < n_points:
                delta[2 * n_points - 1 - k] -= h[i - k]
            if 0 <= j - k < n_points:
                delta[2 * n_points - 1 - k] += h[j - k]
        return delta
    
    T = 0.5
    alpha = 0.995
    N_swap = n_points
    swaps = 0
    
    while time.time() - start_time < budget_s - 10 and T > 1e-5:
        # Pick random 0 and 1
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        i = np.random.choice(zeros)
        j = np.random.choice(ones)
        
        # Compute correlation delta
        delta_corr = update_correlation(C_raw, h, i, j)
        new_C_raw = C_raw + delta_corr
        new_max = np.max(new_C_raw)
        
        # Accept or reject
        if new_max < cur_max or np.random.rand() < np.exp(-(new_max - cur_max) / T):
            h[i] = 1
            h[j] = 0
            C_raw = new_C_raw
            cur_max = new_max
            if new_max * dx < best_val:
                best_val = new_max * dx
                best_h = h.copy()
        
        swaps += 1
        if swaps % N_swap == 0:
            T *= alpha
    
    # Hill climb
    improved = True
    while improved and time.time() - start_time < budget_s - 5:
        improved = False
        for i in np.where(best_h == 0)[0]:
            for j in np.where(best_h == 1)[0]:
                delta_corr = update_correlation(C_raw, best_h, i, j)
                new_C_raw = C_raw + delta_corr
                new_max = np.max(new_C_raw)
                if new_max < cur_max:
                    best_h[i] = 1
                    best_h[j] = 0
                    C_raw = new_C_raw
                    cur_max = new_max
                    improved = True
                    break
            if improved:
                break
    
    # Final computation
    C5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h.tolist(), float(C5), n_points)
