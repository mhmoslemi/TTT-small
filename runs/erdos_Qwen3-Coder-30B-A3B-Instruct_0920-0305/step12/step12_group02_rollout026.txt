import numpy as np
from scipy.optimize import minimize
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use 200 points for better resolution
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Precompute distances from centre
    x = np.arange(n_points) * dx
    d = np.abs(x - 1.0)
    
    # Function to compute C5
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_c5 = np.inf
    best_h = None
    
    # Parameter search over p values
    p_values = np.logspace(np.log10(0.3), np.log10(5), 100)
    
    for p in p_values:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute alpha
        alpha_raw = (p + 1) / (2 * p)
        alpha = min(alpha_raw, 1.0)
        
        # Compute raw shape
        ratio = d / alpha
        f = np.where(ratio <= 1, 1 - np.power(ratio, p), 0.0)
        
        sum_f = f.sum()
        
        if sum_f == 0:
            continue
            
        # Compute feasible range for c
        c_max1 = target_sum / sum_f
        
        # Constraint from b + c <= 1 where b = (target_sum - c * sum_f) / n_points
        denom = 1 - sum_f / n_points
        if denom > 0:
            c_max2 = 0.5 / denom
        else:
            c_max2 = np.inf
            
        c_max = min(c_max1, c_max2)
        if c_max <= 0:
            continue
            
        # Use golden-section search to find optimal c
        def objective(c):
            if c > c_max:
                return np.inf
                
            b = (target_sum - c * sum_f) / n_points
            h = b + c * f
            h = np.clip(h, 0, 1)
            return compute_c5(h)
            
        # Golden section search
        phi = (1 + np.sqrt(5)) / 2
        resphi = 2 - phi
        
        a, b = 0.0, c_max
        h1 = a + resphi * (b - a)
        h2 = a + phi * (b - a)
        f1 = objective(h1)
        f2 = objective(h2)
        
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
                
            if f1 < f2:
                b = h2
                h2 = h1
                f2 = f1
                h1 = a + resphi * (b - a)
                f1 = objective(h1)
            else:
                a = h1
                h1 = h2
                f1 = f2
                h2 = a + phi * (b - a)
                f2 = objective(h2)
                
        c_opt = (a + b) / 2.0
        c5_val = objective(c_opt)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            b = (target_sum - c_opt * sum_f) / n_points
            h = b + c_opt * f
            h = np.clip(h, 0, 1)
            # Apply exact projection to ensure sum constraint
            h = project_to_sum_box(h, target_sum)
            best_h = h
            best_c5 = compute_c5(h)
    
    # Final refinement with L-BFGS-B
    if best_h is not None and time.time() - start_time < budget_s - 5:
        def objective(x):
            h = project_to_sum_box(x, target_sum)
            return compute_c5(h)
            
        x0 = best_h.copy()
        res = minimize(objective, x0, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                      options={'maxiter':100, 'ftol':1e-12})
        if res.success:
            h_opt = project_to_sum_box(res.x, target_sum)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    
    return (best_h.tolist(), float(best_c5), n_points)
