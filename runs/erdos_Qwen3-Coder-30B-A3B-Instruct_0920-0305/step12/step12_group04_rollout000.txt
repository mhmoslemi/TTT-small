def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break

            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5_and_corr(h):
    """Evaluate C5 and return correlation array"""
    corr = np.correlate(h, 1 - h, mode="full")
    O = corr * dx
    max_O = np.max(O)
    return max_O, O

def compute_gradient_for_lag(h, lag):
    """Compute gradient of O(k) for specific lag k"""
    N = len(h)
    forward = np.zeros(N)
    backward = np.zeros(N)
    
    if lag >= 0:
        forward[:N - lag] = h[lag:]
        backward[lag:] = h[:N - lag]
    else:
        k = -lag
        forward[k:] = h[:N - k]
        backward[:N - k] = h[k:]
    
    grad = 1.0 - forward - backward
    return grad

        max_O, O_arr = evaluate_c5_and_corr(h)
        
        # Find the lag with maximum overlap
        lag = np.argmax(O_arr) - (n_points - 1)
        
        # Compute gradient for that lag
        grad = compute_gradient_for_lag(h, lag)
        
        # Update step size with diminishing schedule
        step = step_size / np.sqrt(iteration + 1)
        
        # Update h
        h_new = h - step * grad
        
        # Project onto feasible region
        h_new = project_to_sum_box(h_new, target_sum)
        
        # Evaluate new state
        new_val, _ = evaluate_c5_and_corr(h_new)

        if new_val < max_O:
            h = h_new
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
        else:
            # Occasionally accept worse solutions to escape local minima
            if np.random.random() < 0.01:
                h = h_new
        
        iteration += 1
    
    return best_h, best_val
