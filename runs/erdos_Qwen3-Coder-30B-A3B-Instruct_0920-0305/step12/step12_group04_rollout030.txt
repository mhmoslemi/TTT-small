import numpy as np

import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_for_lag(h, lag):
    """Compute gradient of O(k) with respect to h for a given lag k"""
    N = len(h)
    forward = np.zeros(N)
    backward = np.zeros(N)
    
    if lag >= 0:
        # forward shift: h[i] contributes to h[i+k] where i+k < N
        forward[:N - lag] = h[lag:]
        # backward shift: h[i] is shifted backward to h[i-k] where i-k >= 0
        backward[lag:] = h[:N - lag]
    else:
        # lag < 0
        k = -lag
        forward[k:] = h[:N - k]
        backward[:N - k] = h[k:]
    
    # Gradient of O(k) = dx * sum_i h_i * (1 - h_{i+k}) 
    # dO(k)/dh_i = dx * (1 - h_{i+k} - h_{i-k})
    grad = np.ones(N) - forward - backward
    return grad

def subgradient_descent(h_init, target_sum, max_iter, step0, time_limit):
    """Perform subgradient descent to minimize max O(k)"""
    start_time = time.time()
    
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > time_limit:
            break
            
        # Compute all O values
        corr = np.correlate(h, 1 - h, mode="full")
        O = corr * dx
        max_O = np.max(O)
        max_idx = np.argmax(O)
        lag = max_idx - (n_points - 1)  # Convert from full index to actual shift
        
        # Compute gradient for this lag
        grad = gradient_for_lag(h, lag)
        
        # Update step size (diminishing step size)
        step = step0 / np.sqrt(iteration + 1)
        
        # Update h
        h_new = h - step * grad
        
        # Project onto feasible set
        h_new = project_to_sum_box(h_new, target_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new)
        
        # Accept improvement or with probability
        if new_c5 < max_O or np.random.random() < 0.01:
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
    
    return best_h, best_c5
