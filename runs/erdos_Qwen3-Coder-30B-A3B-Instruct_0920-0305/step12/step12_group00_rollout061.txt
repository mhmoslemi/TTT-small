import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort the vector
    sorted_x = np.sort(x)[::-1]
    
    # Find lambda using binary search for sum clipping
    def f(lam):
        projected = np.clip(x - lam, 0, 1)
        return np.sum(projected) - target_sum
    
    # Binary search
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x - (a + b) / 2, 0, 1)
    return result

def softmax_weights(corr, alpha):
    """Compute softmax weights for gradient"""
    exp_corr = np.exp(alpha * corr)
    return exp_corr / np.sum(exp_corr)

def grad_softmax(corr, h, alpha):
    """Compute gradient of smooth max with respect to h"""
    weights = softmax_weights(corr, alpha)
    # Compute forward and backward convolutions
    L = len(h) * 2 - 1
    h_pad = np.pad(h, (0, len(h)-1), mode='constant')
    weights_pad = np.pad(weights, (0, len(weights)-1), mode='constant')
    conv_forward = np.convolve(h_pad, weights_pad, mode='valid')
    conv_backward = np.convolve(weights_pad[::-1], h_pad, mode='valid')[len(h)-1:]
    
    grad = 1.0 - conv_forward - conv_backward
    return grad

def adam_step(h, grad, m, v, t, lr=0.12, beta1=0.9, beta2=0.999, epsilon=1e-8):
    """Adam optimization step"""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
    return h_new, m, v

def build_balanced_word(n, phi=(np.sqrt(5)-1)/2):
    """Build balanced Sturmian word"""
    h = np.array([((i * phi) % 1) < 0.5 for i in range(n)], dtype=float)
    return h

def swap_update(S, h_bin, i, j, n):
    """Update overlap counts when swapping indices i and j"""
    old_S = S.copy()
    for l in range(-n+1, n):
        if l < 0:
            idx_i = i + l
            idx_j = j + l
            if 0 <= idx_i < n and 0 <= idx_j < n:
                old_S[l+n-1] -= h_bin[idx_i] + h_bin[idx_j]
                S[l+n-1] += h_bin[idx_j] + h_bin[idx_i]
        else:
            idx_i = i + l
            idx_j = j + l
            if 0 <= idx_i < n and 0 <= idx_j < n:
                old_S[l+n-1] -= h_bin[idx_i] + h_bin[idx_j]
                S[l+n-1] += h_bin[idx_j] + h_bin[idx_i]
    return old_S, S

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points // 2
    best_c5 = float('inf')
    best_h = None
    
    # Build initial vectors
    h_balanced = build_balanced_word(n_points)
    h_balanced = project_to_sum_box(h_balanced, TARGET_SUM)
    
    h_random = np.random.rand(n_points)
    h_random = project_to_sum_box(h_random, TARGET_SUM)
    
    h_perturbed = h_balanced + np.random.normal(0, 0.05, n_points)
    h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
    
    initial_h_list = [h_balanced, h_random, h_perturbed]
    
    # Run optimization from different starting points
    for i, h0 in enumerate(initial_h_list):
        time_left = budget_s * 0.45
        t_start = time.time()
        
        # Adam optimization with smooth max
        h = h0.copy()
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        t = 1
        alpha = 4.0
        alpha_max = 200.0
        alpha_growth = 1.5
        
        # Early stopping
        patience = 0
        prev_c5 = float('inf')
        max_patience = 200
        
        while time.time() - t_start < time_left and t < 8000:
            corr = compute_corr_fft(h)
            c5_current = c5_from_corr(corr, n_points)
            
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h.copy()
            
            # Gradient of smooth max
            grad = grad_softmax(corr, h, alpha)
            
            # Adam step
            h, m, v = adam_step(h, grad, m, v, t, lr=0.12)
            
            # Project to sum and box constraints
            h = project_to_sum_box(h, TARGET_SUM)
            
            # Increase alpha if we're making progress
            if abs(prev_c5 - c5_current) < 1e-6:
                patience += 1
                if patience > max_patience and alpha < alpha_max:
                    alpha = min(alpha * alpha_growth, alpha_max)
            else:
                patience = 0
            prev_c5 = c5_current
            
            t += 1
            
            # Re-evaluate the current solution with exact C5
            exact_c5 = c5_from_corr(compute_corr_fft(h), n_points)
            if exact_c5 < best_c5:
                best_c5 = exact_c5
                best_h = h.copy()
                
    # Convert to binary
    if best_h is not None:
        binary_h = (best_h >= 0.5).astype(float)
        ones_count = np.sum(binary_h)
        if ones_count > TARGET_SUM:
            ones_indices = np.where(binary_h == 1)[0]
            flip_count = int(ones_count - TARGET_SUM)
            flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
            binary_h[flip_indices] = 0
        elif ones_count < TARGET_SUM:
            zeros_indices = np.where(binary_h == 0)[0]
            flip_count = int(TARGET_SUM - ones_count)
            flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
            binary_h[flip_indices] = 1
        
        binary_h = project_to_sum_box(binary_h, TARGET_SUM)
        current_c5 = c5_from_corr(compute_corr_fft(binary_h), n_points)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = binary_h.copy()
            
        # Do binary swap hill-climb (100k swaps max)
        max_swaps = 100000
        time_left = budget_s * 0.35
        t_start = time.time()
        current_h = binary_h.copy()
        current_c5 = c5_from_corr(compute_corr_fft(current_h), n_points)
        
        for _ in range(max_swaps):
            if time.time() - t_start > time_left:
                break
                
            # Find indices with 1s and 0s
            ones_indices = np.where(current_h == 1)[0]
            zeros_indices = np.where(current_h == 0)[0]
            
            if len(ones_indices) == 0 or len(zeros_indices) == 0:
                break
                
            # Choose swap indices
            i = np.random.choice(ones_indices)
            j = np.random.choice(zeros_indices)
            
            # Simulate the swap
            temp_h = current_h.copy()
            temp_h[i] = 0
            temp_h[j] = 1
            
            # Compute new C5
            new_c5 = c5_from_corr(compute_corr_fft(temp_h), n_points)
            if new_c5 < current_c5:
                current_h = temp_h.copy()
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
                    
        # Additional restarts to ensure we've found a global minimum
        time_left = budget_s * 0.15
        t_start = time.time()
        for _ in range(10):
            if time.time() - t_start > time_left:
                break
                
            # Generate a random binary with sum constraint
            random_h = np.random.rand(n_points)
            random_h = project_to_sum_box(random_h, TARGET_SUM)
            random_binary = (random_h >= 0.5).astype(float)
            
            # Fix sum
            ones_count = np.sum(random_binary)
            if ones_count > TARGET_SUM:
                ones_indices = np.where(random_binary == 1)[0]
                flip_count = int(ones_count - TARGET_SUM)
                flip_indices = np.random.choice(ones_indices, flip_count, replace=False)
                random_binary[flip_indices] = 0
            elif ones_count < TARGET_SUM:
                zeros_indices = np.where(random_binary == 0)[0]
                flip_count = int(TARGET_SUM - ones_count)
                flip_indices = np.random.choice(zeros_indices, flip_count, replace=False)
                random_binary[flip_indices] = 1
            random_binary = project_to_sum_box(random_binary, TARGET_SUM)
            
            # Simple swap hill-climb
            current_h = random_binary.copy()
            current_c5 = c5_from_corr(compute_corr_fft(current_h), n_points)
            
            for _ in range(2000):  # limited swaps to save time
                ones_indices = np.where(current_h == 1)[0]
                zeros_indices = np.where(current_h == 0)[0]
                if len(ones_indices) == 0 or len(zeros_indices) == 0:
                    break
                i = np.random.choice(ones_indices)
                j = np.random.choice(zeros_indices)
                temp_h = current_h.copy()
                temp_h[i] = 0
                temp_h[j] = 1
                
                new_c5 = c5_from_corr(compute_corr_fft(temp_h), n_points)
                if new_c5 < current_c5:
                    current_h = temp_h.copy()
                    current_c5 = new_c5
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = current_h.copy()
            
    if best_h is None:
        best_h = h0.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
