```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def balanced_sturmian(n):
    """Generate a balanced Sturmian sequence with n points and sum=n//2"""
    phi = (np.sqrt(5) - 1) / 2
    x = np.array([int((i * phi) % 1 < 0.5) for i in range(n)])
    # Adjust to have exactly n//2 ones
    current_sum = np.sum(x)
    target_sum = n // 2
    if current_sum > target_sum:
        # Turn some 1s to 0s
        ones = np.where(x == 1)[0]
        to_flip = current_sum - target_sum
        for i in range(to_flip):
            x[ones[i]] = 0
    elif current_sum < target_sum:
        # Turn some 0s to 1s
        zeros = np.where(x == 0)[0]
        to_flip = target_sum - current_sum
        for i in range(to_flip):
            x[zeros[i]] = 1
    return x

def compute_autocorr_fft(x):
    """Compute autocorrelations efficiently using FFT"""
    x_rev = x[::-1]
    b = x
    L = 2 * len(x) - 1
    corr = np.fft.ifft(np.fft.fft(x_rev, L) * np.fft.fft(b, L)).real
    return corr[:len(x)]

def swap_delta(S, x, i, j, n):
    """Compute change in autocorrelations when swapping bits at i and j"""
    delta = np.zeros(n)
    # For each lag ℓ, compute how the swap affects S_ℓ
    for l in range(1, n):
        # x[i] and x[j] are at positions i and j, so x[i+l] and x[j+l] etc
        if i + l < n:
            # x[i] contributes to S_l if l is valid
            # After swap: x[i] becomes x[j], x[j] becomes x[i]
            # Then for lag l: (before) x[i] * x[i+l] and x[j] * x[j+l]
            # (after) x[j] * x[j+l] and x[i] * x[i+l]
            delta[l] -= x[i] * x[i+l] + x[j] * x[j+l]
            delta[l] += x[j] * x[i+l] + x[i] * x[j+l]
        if j + l < n:
            # Similar for backwards indices
            delta[l] -= x[i] * x[j+l] + x[j] * x[i+l]
            delta[l] += x[j] * x[j+l] + x[i] * x[i+l]
    return delta

def hill_climb_swap(x, S, m, n, dx, max_time, t0):
    """Perform greedy and simulated annealing swap hill-climb"""
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    best_x = x.copy()
    best_m = m
    best_c5 = dx * (n // 2 - m)
    
    # Simulated annealing parameters
    T = 0.01 * best_c5
    cooling_rate = 0.99
    
    # Counter to break out of plateaus
    no_improvement_count = 0
    max_no_improvement = 5 * n
    
    time_limit = t0 + max_time
    
    while time.time() < time_limit:
        if no_improvement_count > max_no_improvement:
            # Switch to simulated annealing
            T = max(1e-8, T * cooling_rate)
            
        # Pick two random indices: one from ones, one from zeros
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Compute difference in autocorrelations
        delta = swap_delta(S, x, i, j, n)
        S_new = S + delta
        m_new = np.min(S_new[1:])  # Skip l=0
        
        if m_new > best_m:
            # Accept the swap
            x[i], x[j] = x[j], x[i]
            S = S_new
            best_m = m_new
            best_c5 = dx * (n // 2 - m_new)
            ones = np.where(x == 1)[0]
            zeros = np.where(x == 0)[0]
            no_improvement_count = 0
        elif T > 1e-10 and np.random.rand() < np.exp((m_new - m) / T):
            # Accept with probability exp((m_new - m)/T)
            x[i], x[j] = x[j], x[i]
            S = S_new
            ones = np.where(x == 1)[0]
            zeros = np.where(x == 0)[0]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
    return best_x, best_m, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n = 128
    dx = 2.0 / n
    N = n // 2
    
    best_c5 = float('inf')
    best_h = None
    best_m = 0
    
    # Stage 1: Solve ILP for max-min autocorrelation
    try:
        # Define variables
        x = cp.Variable(n, boolean=True)
        y = cp.Variable((n, n), boolean=True)
        m = cp.Variable()
        
        # Define constraints
        constraints = [
            cp.sum(x) == N,
            x[0] == 1,  # Break symmetry
        ]
        
        # Enforce lexicographic ordering on first half
        for i in range(N - 1):
            constraints.append(x[i] >= x[i + 1])
        
        # Add McCormick constraints for y[i,ℓ] = x[i] * x[i+ℓ]
        for l in range(1, n):
            for i in range(n - l):
                constraints.extend([
                    y[i, l] <= x[i],
                    y[i, l] <= x[i + l],
                    y[i, l] >= x[i] + x[i + l] - 1
                ])
        
        # Define the autocorrelations
        S = [cp.sum(y[:, l]) for l in range(1, n)]
        
        # Add constraints that all S_ℓ >= m
        constraints.extend([S[l - 1] >= m for l in range(1, n)])
        
        # Objective
        objective = cp.Maximize(m)
        
        # Create and solve problem with time limit
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, max_seconds=0.45 * budget_s)
        
        if prob.status == cp.OPTIMAL or prob.status == cp.UNBOUNDED:
            x_sol = np.round(x.value).astype(int)
            
            # Compute actual C5
            corr = correlate(x_sol, x_sol, mode="full")
            # Get overlaps from 1st half of correlation
            overlaps = [corr[len(corr)//2 + 1+i] for i in range(n-1)]
            overlap_max = max(overlaps)
            c5_val = dx * (N - overlap_max)
            
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = x_sol
                best_m = overlap_max
            
    except Exception:
        pass
    
    # Stage 2: Local refinement if needed
    if best_h is None:
        # Fall back to Sturmian sequence
        x0 = balanced_sturmian(n)
        C0 = compute_autocorr_fft(x0)
        m0 = np.min(C0[1:])  # minimum autocorrelation (non-zero lags)
        c5_0 = dx * (n // 2 - m0)
        best_h = x0
        best_m = m0
        best_c5 = c5_0
    
    # Problem is solved once, but we do comprehensive local search to boost the result
    t0 = time.time()
    # Re-compute correlations for efficient delta calculation
    S_full = compute_autocorr_fft(best_h)
    m_current = np.min(S_full[1:])
    
    # Hill climb iterations - use different amounts of time from what's left
    max_time = 0.35 * budget_s
    
    refined_h, ref_m, ref_c5 = hill_climb_swap(
        best_h.copy(), S_full, m_current, n, dx, max_time, t0
    )
    
    if ref_c5 < best_c5:
        best_c5 = ref_c5
        best_h = refined_h
        best_m = ref_m
    
    # Stage 3: Final fallback random restarts
    t_start = time.time()
    if best_h is not None:
        # Try a few random restarts for robustness
        for _ in range(5):
            if time.time() - t_start > budget_s * 0.9:
                break
            # Generate random binary vector with correct sum
            tmp_x = np.random.rand(n)
            tmp_x = np.clip(tmp_x, 0, 1)
            # Project to correct sum
            tmp_x = np.array([1 if x > 0.5 else 0 for x in tmp_x])
            ones_count = np.sum(tmp_x)
            if ones_count > N:
                flip_indices = np.random.choice(np.where(tmp_x == 1)[0], 
                                               ones_count - N, replace=False)
                tmp_x[flip_indices] = 0
            elif ones_count < N:
                flip_indices = np.random.choice(np.where(tmp_x == 0)[0], 
                                               N - ones_count, replace=False)
                tmp_x[flip_indices] = 1
            # Recompute autocorrelations for better refinement
            tmp_S = compute_autocorr_fft(tmp_x)
            tmp_m = np.min(tmp_S[1:])
            tmp_c5 = dx * (N - tmp_m)
            if tmp_c5 < best_c5:
                best_c5 = tmp_c5
                best_h = tmp_x
                best_m = tmp_m
    
    # Final validation - ensure the sum constraint
    actual_sum = np.sum(best_h)
    if np.abs(actual_sum - N) > 0.1:
        # Adjust if needed
        while actual_sum > N:
            pos_ones = np.where(best_h == 1)[0]
            flip_idx = np.random.choice(pos_ones)
            best_h[flip_idx] = 0
            actual_sum -= 1
        while actual_sum < N:
            pos_zeros = np.where(best_h == 0)[0]
            flip_idx = np.random.choice(pos_zeros)
            best_h[flip_idx] = 1
            actual_sum += 1
    
    return (best_h.tolist(), float(best_c5), n)
```
