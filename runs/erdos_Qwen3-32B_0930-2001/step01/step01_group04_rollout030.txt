import numpy as np
import time
import cvxpy as cp

def compute_C5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = h_clipped.sum()
    if np.abs(current_sum) < 1e-10:
        return h_clipped
    h_scaled = h_clipped * target_sum / current_sum
    return h_scaled

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values, dtype=np.float64)

    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h, dx)

    while time.time() - start_time < budget_s:
        # Compute current correlation and find k_max
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_index = np.argmax(corr)

        c = np.zeros(n_points)
