def compute_C5_and_max_k(h, n_points):
    dx = 2.0 / n_points
    shifted = 1.0 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)

        return [i for i in range(m, n_points)]
