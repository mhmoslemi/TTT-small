import numpy as np
import time
import math

def compute_C5(h, n_points):
    dx = 2.0 / n_points
    shifted = 1.0 - h
    corr = np.correlate(h, shifted, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    assert np.all(h >= 0.0) and np.all(h <= 1.0)
    assert np.isclose(np.sum(h), n_points / 2.0)
    best_h = h.copy()
    best_c5 = compute_C5(best_h, n_points)
    start_time = time.time()
    T = 2000.0 * best_c5  # Initial temperature scaled by best_c5
    alpha = 0.995
    current_h = h.copy()
    while time.time() - start_time < budget_s:
        # Perturbation size scales with temperature
        m = max(1, int(0.1 * n_points))
        indices = np.random.choice(n_points, size=m, replace=False)
        perturbation_strength = max(0.05, T / 1000.0)  # Vary perturbation strength with T
        new_h = current_h.copy()
        for idx in indices:
            perturb = np.random.uniform(-perturbation_strength, perturbation_strength)
            new_h[idx] += perturb
        new_h = np.clip(new_h, 0.0, 1.0)
        # Enforce sum constraint
        target_sum = n_points / 2.0
        current_sum = new_h.sum()
        D = target_sum - current_sum
        while abs(D) > 1e-8:
            i = np.random.randint(0, n_points)
            if D > 0:
                max_add = 1.0 - new_h[i]
                if max_add < 1e-10:
                    continue
                add = min(D, max_add)
                new_h[i] += add
                D -= add
            else:
                max_remove = new_h[i]
                if max_remove < 1e-10:
                    continue
                remove = min(-D, max_remove)
                new_h[i] -= remove
                D += remove
        # Evaluate and accept decision
        new_c5 = compute_C5(new_h, n_points)
        delta = new_c5 - best_c5
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            current_h = new_h.copy()
            if delta < 0:
                best_h = new_h.copy()
                best_c5 = new_c5
        # Cooling schedule
        T *= alpha
        # Add occasional reset for global search
        if T < 0.1 * best_c5 and np.random.rand() < 0.05:  # Global reset every few thousand iterations
            current_h = best_h.copy()
            T = 2000.0 * best_c5
    return (best_h.tolist(), best_c5, n_points)
