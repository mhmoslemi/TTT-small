import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    required_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h_vec):
        h = h_vec.reshape(-1)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx

    n_particles = 20
    w = 0.7  # Inertia
    c1 = 1.5  # Cognitive
    c2 = 1.5  # Social
    max_vel = 0.1

    # Initialize positions with perturbations around initial_h
    positions = np.zeros((n_particles, n_points))
    positions[0] = initial_h.copy()
    sigma_perturb = 0.05
    for i in range(1, n_particles):
        perturbed = initial_h + sigma_perturb * (np.random.rand(n_points) - 0.5)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed *= required_sum / np.sum(perturbed) if np.sum(perturbed) > 1e-9 else 1.0
        perturbed = np.clip(perturbed, 0, 1)
        positions[i] = perturbed

    velocities = 0.1 * np.random.randn(n_particles, n_points)

    # Personal best tracking
    personal_best_positions = positions.copy()
    personal_best_scores = np.array([compute_c5(pos) for pos in positions])
    global_best_idx = np.argmin(personal_best_scores)
    best_h = personal_best_positions[global_best_idx].copy()
    best_score = personal_best_scores[global_best_idx]

    # Main PSO loop
    while time.time() - start_time < budget_s:
        r1 = np.random.rand(n_particles, 1)
        r2 = np.random.rand(n_particles, 1)

        for i in range(n_particles):
            # Velocity update
            vel_cognitive = c1 * r1[i] * (personal_best_positions[i] - positions[i])
            vel_social = c2 * r2[i] * (best_h - positions[i])
            velocities[i] = w * velocities[i] + vel_cognitive + vel_social
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)

            # Position update with constraint projection
            new_pos = positions[i] + velocities[i]
            new_pos = np.clip(new_pos, 0, 1)
            pos_sum = np.sum(new_pos)
            if np.isclose(pos_sum, 0):
                new_pos = np.zeros_like(new_pos) + (required_sum / n_points)
            else:
                new_pos = new_pos * (required_sum / pos_sum)
            new_pos = np.clip(new_pos, 0, 1)
            positions[i] = new_pos

            # Objective evaluation

            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = new_pos.copy()
                if score < best_score:
                    best_h[:] = new_pos.copy()
                    best_score = score

    return best_h, best_score, n_points
