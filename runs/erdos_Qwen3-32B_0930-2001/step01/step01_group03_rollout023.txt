def evaluate_individual(h, dx):

def run(seed=42, budget_s=1100, **kwargs):

    while time.time() - start_time < budget_s:
        # Compute gradient
        conv = np.correlate(h, 1 - h, mode='full')
        c_values = conv * dx
        exp_c = np.exp(c_values / epsilon)
        sum_exp = np.sum(exp_c)

        # Gradient step
