import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        result = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(result)

    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    options = {'maxiter': 1000, 'ftol': 1e-8, 'disp': False}

    best_c5 = objective(h_initial)
    best_h = h_initial.copy()

    def optimize_h(h0):
        h_flat_initial = h0.copy().ravel()
        h_flat_initial = np.clip(h_flat_initial, 0, 1)
        sum_initial = h_flat_initial.sum()
        if sum_initial != 0.0:
            scale = target_sum / sum_initial
            h_flat_initial *= scale
        h_flat_initial = np.clip(h_flat_initial, 0, 1)

        result = minimize(
            objective,
            h_flat_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options=options
        )
        if result.success:
            h_opt = result.x.reshape(n_points)
            h_opt = np.clip(h_opt, 0, 1)
            sum_opt = h_opt.sum()
            if sum_opt != 0.0:
                scale = target_sum / sum_opt
                h_opt *= scale
                h_opt = np.clip(h_opt, 0, 1)
            current_c5 = objective(h_opt)
            if current_c5 < best_c5:
                return h_opt, current_c5
        return best_h, best_c5

    h_opt, c5 = optimize_h(h_initial)
    best_h, best_c5 = h_opt, c5

    np.random.seed(seed)
    max_restarts = 20
    initial_noise_scale = 0.1
    for i in range(max_restarts):
        if time.time() - start_time >= budget_s:
            break
        noise_scale = initial_noise_scale * (0.9 ** i)
        perturbation = np.random.randn(n_points) * noise_scale
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = h_perturbed.sum()

        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    return (best_h, best_c5, n_points)
