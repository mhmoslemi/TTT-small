import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points

    def project_feasible(h, n_points):
        h_clipped = np.clip(h, 0, 1)
        target_sum = n_points / 2
        current_sum = h_clipped.sum()
        h_scaled = h_clipped * target_sum / current_sum
        return h_scaled

    h = project_feasible(h, n_points)
    best_h = h.copy()
    best_c5 = float('inf')

    step_size = 0.02
    decay_rate = 0.995
    perturb_interval = 50
    perturb_strength = 0.05
    max_no_improve = 200
    iteration = 0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr, corr

    best_c5, _ = compute_c5(h)
    while time.time() - start_time < budget_s:
        iteration += 1
        current_c5, corr = compute_c5(h)
        k_corr = np.argmax(corr)
        k_max = np.argmax(corr)

        # Compute gradient for k_max
        grad = np.zeros(n_points)
        for i in range(n_points):
            term1 = 0
            partner1 = i + k_max
            if 0 <= partner1 < n_points:
                term1 = 1 - h[partner1]
            term2 = 0
            partner2 = i - k_max
            if 0 <= partner2 < n_points:
                term2 = -h[partner2]
            grad[i] = term1 + term2

        # Take step and project
        h -= step_size * grad
        h = project_feasible(h, n_points)

        # Update best
        current_c5, _ = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h.copy()
            no_improve = 0
            step_size *= decay_rate
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                h = project_feasible(h + np.random.normal(0, perturb_strength, n_points), n_points)
                no_improve = 0
                step_size *= 0.9

        # Periodic perturbation
        if iteration % perturb_interval == 0:
            h = project_feasible(h + np.random.normal(0, perturb_strength * 2, n_points), n_points)
            no_improve = 0

        step_size *= decay_rate

    # Final check
    best_c5, _ = compute_c5(best_h)
    return (best_h.tolist(), best_c5, n_points)
