        smooth_c5 = np.log(sum_exp) / epsilon if sum_exp > 0 else 0.0
        
        grad = np.zeros_like(h)
        for k in range(len(conv_result)):
            s = k - (n_points - 1)
            j_min = max(0, -s)
            j_max = min(n_points - 1, n_points - 1 - s)
            if j_max < j_min:
                continue
            weight = exp_terms[k]
            for j in range(j_min, j_max + 1):
                term = weight * dx
                grad[j] += term * (1 - h[j + s])
                grad[j + s] -= term * h[j]
        grad /= sum_exp
        
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        sum_h = h_new.sum()
        if abs(sum_h) < 1e-10:
            h_new = 0.5 * np.ones_like(h)
        else:
            h_new = h_new * (n_points / 2) / sum_h
        
        current_c5 = evaluate_individual(h_new, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            iter_without_improvement = 0
            alpha = max(0.001, alpha * 1.01)  # Slight increase if improving
        else:
            iter_without_improvement += 1
            alpha = max(0.001, alpha * 0.95)
        
        if iteration % 100 == 0:
            epsilon *= 0.99
            # Perturbation
            h_new += np.random.normal(0, 0.01, h_new.shape)
            h_new = np.clip(h_new, 0, 1)
            sum_h = h_new.sum()
            if abs(sum_h) < 1e-10:
                h_new = 0.5 * np.ones_like(h)
            else:
                h_new = h_new * (n_points / 2) / sum_h
        
        h = h_new.copy()
    
    return (best_h, best_c5, n_points)
