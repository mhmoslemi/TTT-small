            val_plus = smooth_objective(h_plus, dx, smooth_eps)
            h_minus = original_h.copy()
            h_minus[i] -= eps_p
            val_minus = smooth_objective(h_minus, dx, smooth_eps)
            grad[i] = (val_plus - val_minus) / (2 * eps_p)
        
        h = original_h - learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (target_sum / sum_h)
        else:
            h.fill(0.5)
        
        current_c5 = evaluate_individual(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            learning_rate = min(learning_rate * 1.1, 0.5)
        else:
            learning_rate *= 0.95
        
        step += 1
        if step % decay_interval == 0:
            smooth_eps *= 0.9
            smooth_eps = max(smooth_eps, 1e-8)
    
    return (best_h, best_c5, n_points)
