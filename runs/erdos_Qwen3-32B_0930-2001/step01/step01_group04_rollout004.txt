import numpy as np
import time

def project_feasible(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    target_sum = n_points / 2
    current_sum = h_clipped.sum()
    h_scaled = h_clipped * target_sum / current_sum
    return h_scaled

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    current_h = project_feasible(h_initial.copy(), n_points)
    best_h = current_h.copy()

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    best_c5 = current_c5 = compute_c5(current_h)
    T = 0.1
    alpha = 0.9995
    no_improvement = 0

    while time.time() - start_time < budget_s:
        # Perturb with temperature-dependent noise
        noise = np.random.normal(0, 0.05 * T, n_points)
        new_h = project_feasible(current_h + noise, n_points)
        new_c5 = compute_c5(new_h)
        delta = new_c5 - current_c5

        # Metropolis acceptance
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
                no_improvement = 0
            else:
                no_improvement += 1
        else:
            no_improvement += 1

        # Cooling schedule
        T *= alpha

        # Temperature reset if no improvement
        if no_improvement > 1000:
            T = max(0.01, T * 10)
            no_improvement = 0

        # Occasional random restart
        if np.random.rand() < 0.0005 and (time.time() - start_time) < budget_s - 5:
            restart_h = project_feasible(np.random.rand(n_points), n_points)
            restart_c5 = compute_c5(restart_h)
            if restart_c5 < best_c5:
                best_c5 = restart_c5
                best_h = restart_h.copy()
            current_h = restart_h
            current_c5 = restart_c5
            no_improvement = 0

    # Final validation
    best_h = project_feasible(best_h, n_points)
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
