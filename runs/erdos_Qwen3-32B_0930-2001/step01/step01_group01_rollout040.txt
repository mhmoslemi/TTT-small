from scipy.optimize import minimize
import time

def run():
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h_flat):

        result = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(result)

    def project_to_feasible(h):

        current_sum = h_clipped.sum()

        if current_sum == 0:

        scaling_factor = target_sum / current_sum
        h_scaled = h_clipped * scaling_factor
        h_final = np.clip(h_scaled, 0, 1)
        return h_final

    bounds = [(0, 1) for _ in range(n_points)]
    options_l_bfgs = {'maxiter': 150, 'ftol': 1e-8, 'disp': False}

    best_c5 = objective(h_initial)
    best_h = h_initial.copy()

    def optimize_h(h0_flat):
        h0 = h0_flat.reshape(n_points)
        result = minimize(
            objective,
            h0,
            method='L-BFGS-B',
            bounds=bounds,
            jac=None,
            options=options_l_bfgs

    h_opt, c5 = optimize_h(h_initial)
