    perturb_str = 0.05  # Perturbation strength

    while time.time() - start_time < budget_s:
        # Find maximum correlation and corresponding shift
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        k_shift = idx_max - (n_points - 1)

        # Compute gradient for worst-case shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_shift
            if 0 <= j < n_points:
                grad[i] = 1 - h[j]

        # Gradient descent step
        h -= learning_rate * grad

        # Feasibility projection
        h = project_feasible(h, n_points)

        # Track best solution
        corr_current = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr_current) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h.copy()
            no_improvement_steps = 0
        else:
            no_improvement_steps += 1

        # Learning rate adaptation
        if no_improvement_steps > 50:
            learning_rate *= 0.5
            no_improvement_steps = 0
            if learning_rate < 1e-6:
                break  # Terminate if learning rate is too small

        # Perturbations to escape local optima
        if np.random.rand() < perturb_prob or step % perturb_interval == 0:
            pert = np.random.normal(0, perturb_str, n_points)
            h += pert
            h = project_feasible(h, n_points)

        step += 1

    # Final validation
    best_h = project_feasible(best_h, n_points)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    return (best_h.tolist(), final_c5, n_points)
