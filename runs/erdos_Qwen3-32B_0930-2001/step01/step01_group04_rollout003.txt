import numpy as np
import time

def project_feasible(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    target_sum = n_points / 2.0
    current_sum = h_clipped.sum()
    return h_clipped * (target_sum / current_sum)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    
    # Initial solution projection
    current_h = project_feasible(h_initial, n_points)
    best_h = current_h.copy()
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    
    # Simulated Annealing parameters
    T = 0.15  # Initial temperature
    alpha = 0.9995  # Cooling rate
    reset_T = T  # For temperature resets
    no_improve_steps = 0
    max_no_improve = 1000
    
    while time.time() - start_time < budget_s:
        # Generate new candidate by perturbing current_h
        noise_std = 0.1 * T
        perturbation = np.random.normal(0, noise_std, n_points)
        new_h = current_h + perturbation
        new_h = project_feasible(new_h, n_points)
        
        # Evaluate new candidate
        corr_new = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Acceptance decision
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            current_h = new_h.copy()
            no_improve_steps = 0
        else:
            delta = new_c5 - best_c5
            prob = np.exp(-delta / T)
            if np.random.rand() < prob:
                current_h = new_h.copy()
                no_improve_steps = 0
            else:
                no_improve_steps += 1
        
        # Cooling schedule
        T *= alpha
        if no_improve_steps >= max_no_improve:
            T = reset_T
            no_improve_steps = 0
    
    return (best_h.tolist(), best_c5, n_points)
