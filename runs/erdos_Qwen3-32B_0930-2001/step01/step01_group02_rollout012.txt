import numpy as np
import time
import math

def compute_C5(h, n_points):
    dx = 2.0 / n_points
    shifted = 1.0 - h
    corr = np.correlate(h, shifted, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    # Correct the initial h to satisfy constraints
    h = np.clip(h, 0.0, 1.0)
    D = (n_points / 2.0) - h.sum()
    while abs(D) > 1e-8:
        i = np.random.randint(n_points)
        if D > 0:
            adjustment = min(1.0 - h[i], D)
            h[i] += adjustment
            D -= adjustment
        else:
            adjustment = min(h[i], -D)
            h[i] -= adjustment
            D += adjustment
    best_h = h.copy()
    best_c5 = compute_C5(best_h, n_points)
    current_h = h.copy()
    current_c5 = best_c5
    T = 1000.0 * best_c5  # Initial temperature
    alpha = 0.995  # Cooling rate
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Generate new candidate solution
        h_new = current_h.copy()
        m = max(1, int(0.1 * n_points))  # Perturb 10% of the points
        indices = np.random.choice(n_points, size=m, replace=False)
        # Apply perturbations
        for idx in indices:
            perturb = np.random.uniform(-0.1, 0.1)
            h_new[idx] += perturb
        # Clip to [0,1]
        h_new = np.clip(h_new, 0.0, 1.0)
        # Adjust sum to be exactly n_points/2
        D = (n_points / 2.0) - h_new.sum()
        while abs(D) > 1e-8:
            i = np.random.randint(n_points)
            if D > 0:
                max_add = 1.0 - h_new[i]
                if max_add < 1e-10:
                    continue
                add = min(D, max_add)
                h_new[i] += add
                D -= add
            else:
                max_remove = h_new[i]
                if max_remove < 1e-10:
                    continue
                remove = min(-D, max_remove)
                h_new[i] -= remove
                D += remove
        new_c5 = compute_C5(h_new, n_points)
        delta_E = new_c5 - current_c5
        # SA acceptance criterion
        if delta_E < 0 or np.random.rand() < math.exp(-delta_E / T):
            current_h[:] = h_new
            current_c5 = new_c5

                best_c5 = new_c5
        # Reduce temperature
        T *= alpha
    return (best_h.tolist(), best_c5, n_points)
