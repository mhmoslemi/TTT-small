    particles = np.clip(particles, 0, 1)
    for i in range(num_particles):
        s = particles[i].sum()
        if s != 0:
            particles[i] *= T / s
        particles[i] = np.clip(particles[i], 0, 1)
    velocities = np.random.uniform(-0.1, 0.1, size=(num_particles, n_points))
    
    personal_best = particles.copy()
    pb_scores = np.array([compute_c5(p) for p in personal_best])
    global_idx = np.argmin(pb_scores)
    global_best = personal_best[global_idx].copy()
    global_score = pb_scores[global_idx]
    
    max_vel = 0.1
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    while time.time() - start_time < budget_s:
        for i in range(num_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            vel_cog = c1 * r1 * (personal_best[i] - particles[i])
            vel_soc = c2 * r2 * (global_best[np.newaxis, ...] - particles[i])
            velocities[i] = w * velocities[i] + vel_cog + vel_soc
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            
            s = particles[i].sum()
            if s != 0:
                particles[i] *= T / s
            else:
                particles[i] = initial_h.copy()
            particles[i] = np.clip(particles[i], 0, 1)
            
            current_score = compute_c5(particles[i])
            if current_score < pb_scores[i]:
                personal_best[i] = particles[i].copy()
                pb_scores[i] = current_score
                if current_score < global_score:
                    global_best = particles[i].copy()
                    global_score = current_score
    
    return global_best, global_score, n_points
