        mid = (low + high) / 2

        # Constraint to enforce max overlap <= mid
        def constraint_func(w):
            h_temp = w.reshape(n_points)
            corr = np.correlate(h_temp, 1 - h_temp, mode='full')
            max_overlap = np.max(corr) * dx
            return mid - max_overlap  # Mid - overlap >= 0

        constraints = (
            {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2},
            {'type': 'ineq', 'fun': constraint_func}
        )
        bounds = [(0.0, 1.0)] * n_points

        result = optimize.minimize(
            lambda x: 0.0,
            warm_start,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )

        if result.success:
            c5_candidate = compute_c5(result.x)
            if c5_candidate <= mid:
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = result.x.copy()
                high = mid
            else:
                high = mid
            warm_start = result.x.copy()
        else:
            low = mid

        if high - low < tolerance:
            break

    return best_h, best_c5, n_points
