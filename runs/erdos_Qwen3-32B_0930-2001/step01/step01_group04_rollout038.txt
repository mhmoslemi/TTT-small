import numpy as np
import time

def project_feasible(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    target_sum = n_points / 2
    current_sum = h_clipped.sum()
    if current_sum == 0:
        return h_clipped
    h_scaled = h_clipped * target_sum / current_sum
    return h_scaled

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    best_h = project_feasible(h.copy(), n_points)
    best_c5 = compute_c5(best_h)
    current_h = best_h.copy()
    current_c5 = best_c5

    T = 0.1
    alpha = 0.9995
    no_improvement_count = 0
    max_no_improvement = 5000

    while time.time() - start_time < budget_s:
        # Perturb
        noise_scale = max(0.05, T * 0.2)
        new_h = current_h + np.random.normal(0, noise_scale, size=n_points)
        new_h = project_feasible(new_h, n_points)
        new_c5 = compute_c5(new_h)

        # Acceptance
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1

        # Cooling
        T *= alpha

        # Temperature reset
        if no_improvement_count > max_no_improvement:
            T = max(T * 2.0, 0.1)
            no_improvement_count = 0

    return (best_h.tolist(), best_c5, n_points)
