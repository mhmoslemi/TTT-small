import numpy as np
import time

def project_feasible(h, target_sum, n_points):
    h_clipped = np.clip(h.copy(), 0, 1)
    current_sum = h_clipped.sum()
    if abs(current_sum - target_sum) < 1e-10:
        return h_clipped
    delta = target_sum - current_sum
    step = delta / n_points
    h_proj = h_clipped + step
    h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def compute_c5(h, dx):
    result = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(result)

def tournament_selection(population, fitness, tournament_size=2):
    indices = np.random.choice(len(population), tournament_size, replace=False)
    best_idx = indices[0]
    for idx in indices[1:]:
        if fitness[idx] < fitness[best_idx]:
            best_idx = idx
    return best_idx

def crossover(parent1, parent2):
    child1 = (parent1 + parent2) / 2
    child2 = np.where(np.random.rand(len(parent1)) < 0.5, parent1, parent2)
    return (child1, child2)

    indices = np.random.choice(n_points, num_mutation, replace=False)
    mutated[indices] += mutation_magnitude * (2 * np.random.rand(num_mutation) - 1)
    return mutated
