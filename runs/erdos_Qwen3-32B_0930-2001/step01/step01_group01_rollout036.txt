    return np.max(corr)

def run():
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    population_size = 20
    mutation_strength = 0.05
    tournament_size = 3
    
    # Initialize population
    population = [h_initial.copy()]
    for _ in range(population_size - 1):
        perturbed = h_initial + 0.05 * np.random.randn(n_points)
        perturbed = project_to_feasible(perturbed, target_sum)
        population.append(perturbed)
    
    best_c5 = float('inf')
    best_h = None
    np.random.seed(42)
    
    while time.time() - start_time < 1100:
        # Evaluate population
        c5_list = np.array([compute_c5(ind, n_points) for ind in population])
        best_index = np.argmin(c5_list)
        current_best_c5 = c5_list[best_index]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[best_index].copy()
        
        # Create new population
        new_population = []
        while len(new_population) < population_size:
            # Parent selection via tournament
            parent1_indices = np.random.choice(population_size, tournament_size, replace=False)
            parent1_c5 = c5_list[parent1_indices]
            parent1 = population[parent1_indices[np.argmin(parent1_c5)]]
            
            parent2_indices = np.random.choice(population_size, tournament_size, replace=False)
            parent2_c5 = c5_list[parent2_indices]
            parent2 = population[parent2_indices[np.argmin(parent2_c5)]]
            
            # Crossover
            split = np.random.randint(n_points)
            child1 = np.concatenate([parent1[:split], parent2[split:]])
            child2 = np.concatenate([parent2[:split], parent1[split:]])
            
            # Mutation
            child1 += mutation_strength * np.random.randn(n_points)
            child2 += mutation_strength * np.random.randn(n_points)
            
            # Projection
            child1 = project_to_feasible(child1, target_sum)
            child2 = project_to_feasible(child2, target_sum)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = [arr for arr in new_population]
    
    if best_h is None:
        final_c5 = [compute_c5(ind, n_points) for ind in population]
        return (population[np.argmin(final_c5)].copy(), min(final_c5), n_points)
    return (best_h.copy(), best_c5, n_points)
