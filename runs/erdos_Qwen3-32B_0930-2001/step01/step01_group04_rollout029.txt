        valid_corr = corr[n_points-1:]  # Extract right half for shifts k=1..n-1
        return np.max(valid_corr) * dx

    def objective_k(h, k):
        total = 0.0
        for i in range(n_points - k):
            total += h[i] * (1 - h[i + k])
        return total

    def grad_k(h, k):
        grad = np.zeros(n_points)
        dx_val = 2.0 / n_points
        for i in range(n_points - k):
            grad[i] += (1 - h[i + k]) * dx_val
        for i in range(k, n_points):
            grad[i] -= h[i - k] * dx_val
        return grad

    current_h = project_feasible(h_initial, n_points)
    current_c5 = compute_c5(current_h)
    best_c5 = current_c5
    best_h[:] = current_h

    while time.time() - start_time < budget_s:
        # Determine worst shift k
        corr_full = np.correlate(current_h, 1 - current_h, mode='full')
        valid_corr = corr_full[n_points-1:]  # k=1 to n-1
        k_values = np.arange(1, n_points)
        k_max = k_values[np.argmax(valid_corr)] if valid_corr.size else 0

        if k_max == 0:
            break

        # Define optimization target for k_max
        def objective(h_vec):
            return objective_k(h_vec, k_max)

        def grad(h_vec):
            return grad_k(h_vec, k_max)

        bounds = [(0, 1)] * n_points
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]

        # Initial guess
        initial_guess = current_h.copy()

        # Run optimization
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            jac=grad,
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 20, 'ftol': 1e-8, 'disp': False}
        )

        if result.success:
            improved_h = result.x
            improved_h = project_feasible(improved_h, n_points)
            improved_c5 = compute_c5(improved_h)
            if improved_c5 < best_c5:
                best_c5 = improved_c5
                best_h[:] = improved_h
            current_h = improved_h
        else:
            # Perturb and retry
            perturb = np.random.normal(0, 0.02, n_points)
            current_h = project_feasible(current_h + perturb, n_points)
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h[:] = current_h

    return (best_h.tolist(), best_c5, n_points)
