import numpy as np
from scipy.optimize import minimize
import time

def run():
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    T = n_points / 2.0
    bounds = [(0, 1)] * n_points
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        total = h.sum()
        if abs(total - T) > 1e-6:
            scale = T / total if total != 0 else 0
            h_proj = np.clip(h * scale, 0.0, 1.0)
        else:
            h_proj = h
        result = np.correlate(h_proj, 1 - h_proj, mode='full') * dx
        return np.max(result)
    
    def generate_perturbed_h(h, noise_scale=0.05):
        h_perturbed = h.copy()
        noise = np.random.normal(0, noise_scale, n_points)
        h_perturbed += noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        total = h_perturbed.sum()
        if total != 0:
            h_perturbed = np.clip(h_perturbed * (T / total), 0, 1)
        return h_perturbed
    
    best_c5 = objective(h_initial)
    best_h = h_initial.copy()
    
    # Initial optimization
    res = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, options={'maxiter': 200, 'ftol': 1e-8})
    if res.success:
        current_h = res.x
        current_c5 = res.fun
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    # Perturbed restarts with structured sampling
    np.random.seed(42)
    perturbation_epochs = 0
    while time.time() - start_time < 1100 and perturbation_epochs < 25:
        perturbation_epochs += 1
        base_h = best_h if perturbation_epochs % 3 == 0 else h_initial
        h_perturbed = generate_perturbed_h(base_h, 0.05 if perturbation_epochs < 15 else 0.02)
        res = minimize(objective, h_perturbed, method='L-BFGS-B', bounds=bounds, options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            current_h = res.x
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h[:] = current_h
                best_c5 = current_c5
    
    return (best_h, best_c5, n_points)
