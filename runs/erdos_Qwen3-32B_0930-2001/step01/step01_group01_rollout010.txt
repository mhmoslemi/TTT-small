import numpy as np
from scipy.optimize import minimize
import time

def run():
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        result = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(result)

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}

    best_h = h_initial.copy()
    best_c5 = objective(best_h)

    def optimize_step(h0):
        result = minimize(
            objective, h0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 150, 'ftol': 1e-8, 'disp': False}
        )
        if result.success:
            new_h = result.x.reshape(n_points)
            return new_h, objective(new_h)
        return best_h, best_c5

    # Initial optimization
    h_opt, c5 = optimize_step(h_initial)
    if c5 < best_c5:
        best_h, best_c5 = h_opt, c5

    # Perturbed starts with dynamic agitation
    np.random.seed(42)
    while time.time() - start_time < 1100:
        h_pert = h_initial.copy()
        n_agitate = max(15, int(n_points * 0.3))
        agitate_idx = np.random.choice(n_points, n_agitate, replace=False)
        perturbs = np.random.choice([-0.3, 0.3], size=n_agitate)
        h_pert[agitate_idx] += perturbs
        h_pert = np.clip(h_pert, 0, 1)
        # Normalize to maintain sum constraint
        scale = (n_points / 2) / h_pert.sum()
        h_pert = np.clip(h_pert * scale, 0, 1)
        h_opt, c5 = optimize_step(h_pert)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    return (best_h, best_c5, n_points)
