def project_feasible(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    target_sum = n_points / 2
    current_sum = h_clipped.sum()
    if current_sum == 0:
        return h_clipped
    h_scaled = h_clipped * (target_sum / current_sum)
    return h_scaled

import time

def project_feasible(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    target_sum = n_points / 2
    current_sum = h_clipped.sum()
    if current_sum == 0:
        return h_clipped
    h_scaled = h_clipped * (target_sum / current_sum)
    return h_scaled
