import numpy as np
from scipy.optimize import minimize
import time

def run():
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    expected_sum = n_points / 2

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        if sum_h != 0:
            scale_factor = expected_sum / sum_h
            h_scaled = h * scale_factor
        else:
            h_scaled = h.copy()
        h_clipped = np.clip(h_scaled, 0, 1)
        # Handle large shifts due to interpolation
        mode_result = np.correlate(h_clipped, 1 - h_clipped, mode='full') * dx
        return np.max(mode_result)

    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = objective(h_initial)
    best_h = h_initial.copy()

    np.random.seed(42)

    restarts = 100
    for _ in range(restarts):
        if time.time() - start_time >= 1100:
            break

        h_perturbed = h_initial.copy()
        perturb_indices = np.random.choice(n_points, size=max(1, int(n_points * 0.5)), replace=False)
        noise = np.random.normal(0, 0.2, size=len(perturb_indices))
        h_perturbed[perturb_indices] += noise
        h_perturbed = np.clip(h_perturbed, 0, 1)

        # Normalize to preserve expected sum constraint
        sum_perturbed = h_perturbed.sum()
        if sum_perturbed != 0:
            h_perturbed = h_perturbed * (expected_sum / sum_perturbed)
            h_perturbed = np.clip(h_perturbed, 0, 1)

        result = minimize(
            objective,
            h_perturbed, 
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 200, 'ftol': 1e-9, 'gtol': 1e-8}
        )

        if result.success:
            h_candidate = result.x.reshape(n_points)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_h, best_c5 = h_candidate, current_c5

    return (best_h, best_c5, n_points)
