import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # PSO parameters
    swarm_size = 20
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_velocity = 0.1

    # Initialize particles and velocities
    particles = []
    velocities = []
    for i in range(swarm_size):
        if i == 0:
            h = initial_h.copy()
        else:
            h = np.random.uniform(0, 1, n_points)
            current_sum = h.sum()
            desired_sum = n_points / 2
            scaling = desired_sum / current_sum
            h *= scaling
            h = np.clip(h, 0, 1)
            current_sum = h.sum()
            diff = desired_sum - current_sum
            h += diff / n_points
            h = np.clip(h, 0, 1)
        particles.append(h)
        velocities.append(np.random.uniform(-max_velocity, max_velocity, n_points))

    particles = np.array(particles)
    velocities = np.array(velocities)

    # Initialize personal and global bests
    personal_best_positions = particles.copy()
    personal_best_values = np.array([max(np.correlate(p, 1 - p, mode='full') * dx) for p in particles])
    global_best_idx = np.argmin(personal_best_values)
    global_best_position = personal_best_positions[global_best_idx]
    global_best_value = personal_best_values[global_best_idx]

    # Main optimization loop
    while time.time() - start_time < budget_s:
        for i in range(swarm_size):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (inertia * velocities[i] +
                            cognitive * r1 * (personal_best_positions[i] - particles[i]) +
                            social * r2 * (global_best_position - particles[i]))
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)

            # Enforce sum constraint
            current_sum = particles[i].sum()
            desired_sum = n_points / 2
            scaling = desired_sum / current_sum
            particles[i] *= scaling
            particles[i] = np.clip(particles[i], 0, 1)

        # Update personal bests
        for i in range(swarm_size):
            current_value = max(np.correlate(particles[i], 1 - particles[i], mode='full') * dx)
            if current_value < personal_best_values[i]:
                personal_best_values[i] = current_value
                personal_best_positions[i] = particles[i].copy()
                if current_value < global_best_value:
                    global_best_value = current_value
                    global_best_position = particles[i].copy()

    return global_best_position.tolist(), global_best_value, n_points
