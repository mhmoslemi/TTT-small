import numpy as np
import time
import cvxpy as cp

def project_feasible(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    target_sum = n_points / 2
    current_sum = np.sum(h_clipped)
    return h_clipped * (target_sum / current_sum) if current_sum != 0 else h_clipped * 0.5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_initial)
    h = project_feasible(h_initial, n_points)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k_shift = max_index - (n_points - 1)

        valid_indices = []
        for i in range(n_points):
            j = i + k_shift
            if 0 <= j < n_points:
                valid_indices.append(i)

        coeffs = np.zeros(n_points)
        for i in valid_indices:
            coeffs[i] = (1 - h[i + k_shift]) * dx

        h_var = cp.Variable(n_points)
        objective = cp.Minimize(coeffs @ h_var)
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2]
        prob = cp.Problem(objective, constraints)

        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h = project_feasible(h + np.random.normal(0, 0.05, n_points), n_points)
                continue
        except Exception:
            h = project_feasible(h + np.random.normal(0, 0.05, n_points), n_points)
            continue

        h_new = np.array(h_var.value).flatten()
        h = project_feasible(h_new, n_points)

        corr_new = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr_new) * dx

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        elif np.random.rand() < 0.05:
            h = project_feasible(h + np.random.normal(0, 0.05, n_points), n_points)

    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    return (best_h.tolist(), np.max(corr_final) * dx, n_points)
