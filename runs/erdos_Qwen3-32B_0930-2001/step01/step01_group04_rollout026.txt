import numpy as np
import time

def project_feasible(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    target_sum = n_points / 2
    current_sum = h_clipped.sum()
    h_scaled = h_clipped * target_sum / current_sum
    return h_scaled

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    current_h = project_feasible(h_initial.copy(), n_points)
    best_h = current_h.copy()
    current_corr = np.correlate(current_h, 1 - current_h, mode='full')
    best_c5 = current_c5 = np.max(current_corr) * dx

    T = 0.1
    T_initial = T
    T_min = 1e-6
    alpha = 0.9995
    noise_std_base = 0.05
    no_improve_count = 0
    max_no_improve = 1000

    while time.time() - start_time < budget_s:
        # Perturb current_h
        noise = np.random.normal(0, noise_std_base * T, n_points)
        new_h = current_h + noise
        new_h = project_feasible(new_h, n_points)

        # Evaluate new_h
        new_corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(new_corr) * dx

        delta_c5 = new_c5 - current_c5

        # Update best solution
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h[:] = new_h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Accept-reject step
        if delta_c5 < 0:
            current_h = new_h.copy()
            current_c5 = new_c5
        else:
            prob = np.exp(-delta_c5 / max(T, 1e-9))
            if np.random.rand() < prob:
                current_h = new_h.copy()
                current_c5 = new_c5

        # Cool T or reheat
        T *= alpha
        if T < T_min:
            T = max(T_initial, T * 0.1)
        if no_improve_count >= max_no_improve:
            T = max(T_initial, T * 0.1)
            no_improve_count = 0

        # Check time budget
        if time.time() - start_time >= budget_s:
            break

    best_h = project_feasible(best_h, n_points)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    return (best_h.tolist(), np.max(corr) * dx, n_points)
