import numpy as np
from scipy.optimize import minimize
import time

def run():
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    options = {'maxiter': 150, 'ftol': 1e-8, 'disp': False}

    best_h = h_initial.copy()
    current_best = objective(best_h)

    def optimize_h(h0):
        result = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=constraints, options=options)
        if result.success:
            h_opt = result.x.reshape(n_points)
            val = objective(h_opt)
            return h_opt, val
        return h0, objective(h0)

    np.random.seed(42)
    max_attempts = 120
    for attempt in range(max_attempts):
        if time.time() - start_time >= 1100 - 10:
            break
        # Generate perturbed h
        indices = np.random.choice(n_points, size=int(n_points * 0.5), replace=False)
        directions = np.random.choice([-1, 1], len(indices))
        h_pert = best_h.copy()
        h_pert[indices] += directions * 0.15
        h_pert = np.clip(h_pert, 0, 1)
        # Adjust sum
        sum_pert = np.sum(h_pert)
        scaling = (n_points / 2) / sum_pert
        h_pert = np.clip(h_pert * scaling, 0, 1)
        # Optimize
        optimized, val = optimize_h(h_pert)
        if val < current_best:
            best_h[:] = optimized
            current_best = val

    # Final refinement with best solution
    for _ in range(3):
        if time.time() - start_time < 1100 - 5:
            h_refined = best_h + 0.01 * np.random.randn(n_points)
            h_refined = np.clip(h_refined, 0, 1)
            scaling = (n_points / 2) / h_refined.sum()
            h_refined = np.clip(h_refined * scaling, 0, 1)
            optimized, val = optimize_h(h_refined)
            if val < current_best:
                best_h[:] = optimized
                current_best = val

    return (best_h, current_best, n_points)
