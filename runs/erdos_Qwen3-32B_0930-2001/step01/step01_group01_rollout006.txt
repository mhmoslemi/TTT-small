from scipy.optimize import minimize
import time

def run():
    start_time = time.time()
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    def objective(h_flat):
        h_flat = h_flat.reshape(n_points)
        corr = np.correlate(h_flat, 1 - h_flat, mode='full') * dx
        return np.max(corr)

    def project_h(h_flat):
        h_flat = np.clip(h_flat, 0, 1)
        total = np.sum(h_flat)
        if abs(total) < 1e-9:
            return h_flat  # Should not happen with valid input
        target = n_points / 2
        scale = target / total
        h_flat = np.clip(h_flat * scale, 0, 1)
        return h_flat

    best_c5 = objective(h)
    best_h = h.copy()
    max_global_iter = 30
    perturb_frac = 0.5
    perturb_amp = 0.2

    for global_iter in range(max_global_iter):
        if time.time() - start_time >= 1100:
            break

        # Perturbation step
        perturb_indices = np.random.choice(n_points, int(n_points * perturb_frac), replace=False)
        h_perturbed = h.copy()
