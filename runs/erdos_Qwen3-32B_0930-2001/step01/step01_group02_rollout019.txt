        actual_shift = k_index - (n_points - 1)
        return max_corr * dx, actual_shift
    
    def compute_gradient(h, actual_shift, dx):
        grad = np.zeros(n_points)
        if actual_shift >= 0:
            for i in range(n_points - actual_shift):
                j = i + actual_shift
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        else:
            shift_abs = -actual_shift
            for i in range(shift_abs, n_points):
                j = i - shift_abs
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        return grad
    
    current_h = h.copy()
    momentum = 0.0
    gamma = 0.1
    mu = 0.9
    gamma_decay = 0.95
    max_phases = 3
    phase_durations = [0.5, 0.3, 0.2]  # Time fractions for each phase
    
    phase_start = start_time
    for phase in range(max_phases):
        phase_end = start_time + budget_s * sum(phase_durations[:phase + 1])
        if phase_end > start_time + budget_s:
            phase_end = start_time + budget_s
        phase_duration = phase_end - phase_start
        if phase_duration <= 0:
            break
        iters = int(1000 * phase_durations[phase])  # Arbitrary iterations per phase
        for _ in range(iters):
            if time.time() > phase_end:
                break
            try:
                _, actual_shift = compute_C5_and_shift(current_h, n_points)
                grad = compute_gradient(current_h, actual_shift, 2.0 / n_points)
                momentum = mu * momentum + gamma * grad
                h_new = current_h - gamma * momentum
                h_new = adjust_sum(h_new)
                new_c5, _ = compute_C5_and_shift(h_new, n_points)
                if new_c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = new_c5
                    current_h = h_new
                    gamma *= gamma_decay ** 0.5
                else:
                    gamma *= gamma_decay
            except Exception:
                pass
        phase_start = phase_end
    
    return (best_h.tolist(), best_c5, n_points)
