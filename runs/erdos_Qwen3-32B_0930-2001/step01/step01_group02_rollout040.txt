import numpy as np
import time

def compute_C5(h, n_points):
    dx = 2.0 / n_points
    shifted = 1.0 - h
    corr = np.correlate(h, shifted, mode='full')
    return np.max(corr) * dx

def hash_solution(h):
    return tuple(round(val, 5) for val in h.tolist())

def perturb_and_adjust(h, sigma, n_points):
    h_new = h.copy()
    m = max(1, int(0.1 * n_points))
    indices = np.random.choice(n_points, size=m, replace=False)
    h_new[indices] += np.random.uniform(-sigma, sigma, size=len(indices))
    h_new = np.clip(h_new, 0.0, 1.0)
    current_sum = np.sum(h_new)
    target_sum = n_points / 2.0
    D = target_sum - current_sum
    if abs(D) < 1e-8:
        return h_new
    idxs = np.random.permutation(n_points)
    for idx in idxs:
        if D > 0:
            add = min(D, 1.0 - h_new[idx])
            h_new[idx] += add
            D -= add
        else:
            remove = min(-D, h_new[idx])
            h_new[idx] -= remove
            D += remove
        if abs(D) < 1e-8:
            break
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    n_points = len(h)

    best_c5 = compute_C5(best_h, n_points)

    max_tabu_size = 20
    sigma = 0.05

        for _ in range(10):
            candidate = perturb_and_adjust(best_h, sigma, n_points)
