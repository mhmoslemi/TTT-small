import numpy as np
import time

def compute_C5(h, n_points):
    dx = 2.0 / n_points
    shifted = 1.0 - h
    corr = np.correlate(h, shifted, mode='full')
    return np.max(corr) * dx

def perturb_and_adjust(h, sigma, n_points):
    m = max(1, int(0.1 * n_points))
    indices = np.random.choice(n_points, size=m, replace=False)
    h_new = h.copy()
    h_new[indices] += np.random.uniform(-sigma, sigma, m)
    h_new = np.clip(h_new, 0.0, 1.0)
    current_sum = h_new.sum()
    target_sum = n_points / 2.0
    D = target_sum - current_sum
    if abs(D) < 1e-8:
        return h_new
    adjust_indices = np.random.permutation(n_points)
    for idx in adjust_indices:
        if D > 0:
            add = min(D, 1.0 - h_new[idx])
            if add > 0:
                h_new[idx] += add
                D -= add
        else:
            remove = min(-D, h_new[idx])
            if remove > 0:
                h_new[idx] -= remove
                D += remove
        if abs(D) < 1e-8:
            break
    return h_new

def hasheq(a, b):
    return np.allclose(a, b, atol=1e-5)
