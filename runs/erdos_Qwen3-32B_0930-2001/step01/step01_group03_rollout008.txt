```python

import time
import scipy.optimize

def run(seed=42, budget_s=1100, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = initial_h_values.copy()
    current_sum = np.sum(h_initial)
    if not np.allclose(current_sum, target_sum, rtol=1e-5, atol=1e-8):
        h_initial = h_initial * (target_sum / current_sum)

    def objective_func(h_vec):
        h = h_vec.reshape(n_points)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        sum_h = np.sum(h)
        constraint_violation = ((sum_h / target_sum) - 1)
        penalty = 1e7 * (constraint_violation ** 2)
        return max_overlap + penalty

    bounds = [(0, 1) for _ in range(n_points)]

    result = scipy.optimize.minimize(
        objective_func,
        x0=h_initial.flatten(),
        method='Nelder-Mead',
        bounds=bounds,
        options={'maxiter': 5000, 'xatol': 1e-6}
    )

    best_h = result.x
    h = best_h.reshape(n_points)
    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    best_c5 = np.max(overlaps)

    return (best_h, best_c5, n_points)
```
