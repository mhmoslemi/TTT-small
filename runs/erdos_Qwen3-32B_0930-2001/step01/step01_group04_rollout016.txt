import numpy as np
import time

def project_feasible(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    target_sum = n_points / 2.0
    current_sum = h_clipped.sum()
    h_scaled = h_clipped * target_sum / current_sum
    return h_scaled

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    # Initial solution
    h = project_feasible(h_initial.copy(), n_points)
    current_energy = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_energy

    # Simulated Annealing parameters
    temp = 0.1
    alpha = 0.9999  # Slower cooling
    max_no_improve = 10000
    no_improve_count = 0

    while time.time() - start_time < budget_s:
        # Perturbation step
        noise = np.random.normal(0, 0.1 * temp, n_points)  # Larger initial steps
        new_h = h + noise
        new_h = project_feasible(new_h, n_points)
        new_energy = compute_c5(new_h, dx)

        # Acceptance decision
        if new_energy < current_energy:
            h = new_h.copy()
            current_energy = new_energy
            no_improve_count = 0
            if new_energy < best_c5:
                best_h = new_h.copy()
                best_c5 = new_energy
        else:
            delta = new_energy - current_energy
            if np.random.rand() < np.exp(-delta / temp):
                h = new_h.copy()
                current_energy = new_energy
                no_improve_count = 0
            else:
                no_improve_count += 1

        # Cool temperature
        temp *= alpha

        # Reheat if stuck for too long
        if no_improve_count > max_no_improve:
            temp = 0.5 * temp + 0.5 * 2.0  # Blend current temp with higher value
            no_improve_count = 0

        # Time check
        if time.time() - start_time > budget_s - 0.1:
            break

    # Final feasibility check
    best_h = project_feasible(best_h, n_points)
    final_energy = compute_c5(best_h, dx)
    return (best_h.tolist(), final_energy, n_points)
