        if ordered_evals[0] < best_c5:
            best_c5 = ordered_evals[0]
            best_h = ordered_simplex[0].copy()

        # Compute centroid (exclude worst point)
        centroid = ordered_simplex[:-1].mean(axis=0)
        worst = ordered_simplex[-1]
        worst_eval = ordered_evals[-1]

        # Reflection
        xr = centroid + alpha * (centroid - worst)
        xr = project(xr)
        fxr = evaluate_individual(xr, dx, best_c5)

        if fxr < ordered_evals[0]:  # Expand
            xe = centroid + gamma * (xr - centroid)
            xe = project(xe)
            fxe = evaluate_individual(xe, dx, best_c5)
            if fxe < fxr:
                simplex[-1] = xe
            else:
                simplex[-1] = xr
        elif fxr >= ordered_evals[-2]:  # Contract or shrink
            if fxr < worst_eval:  # Outside contraction
                xc = centroid + beta * (xr - centroid)
                xc = project(xc)
                fxc = evaluate_individual(xc, dx, best_c5)
                if fxc < fxr:
                    simplex[-1] = xc
                else:  # Shrink
                    for i in range(1, len(simplex)):
                        simplex[i] = ordered_simplex[0] + sigma * (simplex[i] - ordered_simplex[0])
                        simplex[i] = project(simplex[i])
            else:  # Inside contraction
                xc = centroid - beta * (centroid - worst)
                xc = project(xc)
                fxc = evaluate_individual(xc, dx, best_c5)
                if fxc < worst_eval:
                    simplex[-1] = xc
                else:  # Shrink
                    for i in range(1, len(simplex)):
                        simplex[i] = ordered_simplex[0] + sigma * (simplex[i] - ordered_simplex[0])
                        simplex[i] = project(simplex[i])
        else:  # Accept reflection
            simplex[-1] = xr

    return (best_h, best_c5, n_points)
