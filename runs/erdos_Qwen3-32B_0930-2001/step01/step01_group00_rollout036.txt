                child *= (target_sum / total)
                child = np.clip(child, 0, 1)

            new_pop.append(child)

        # Replace population
        new_fitness = np.array([compute_c5(ind) for ind in new_pop])
        arg = np.argsort(new_fitness)
        new_pop = np.array(new_pop)[arg]
        new_fitness = new_fitness[arg]

        population = np.array(new_pop[:pop_size])
        fitness = new_fitness

        # Update best solution
        current_best = np.min(fitness)
        if current_best < best_c5:
            best_c5 = current_best
            best_h[:] = population[0].copy()
            last_improvement = time.time()

        # Early restart on stagnation
        if time.time() - last_improvement > 60:
            # Re-initialize population better around best_h
            population = []
            for _ in range(pop_size):
                candidate = best_h.copy()
                noise = np.random.uniform(-0.05, 0.05, n_points)
                candidate += noise
                candidate = np.clip(candidate, 0, 1)
                total = candidate.sum()
                if total != 0:
                    candidate *= (target_sum / total)
                    candidate = np.clip(candidate, 0, 1)
                population.append(candidate)
            population = np.array(population)
            fitness = np.array([compute_c5(ind) for ind in population])
            last_improvement = time.time()

    return best_h, best_c5, n_points
