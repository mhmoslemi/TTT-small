import numpy as np

def mutate(h_vec, n_points):
    h = h_vec.copy()
    idx = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
    perturbation = np.random.uniform(-0.1, 0.1, size=len(idx))
    h[idx] += perturbation
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2.0
    current_sum = h.sum()
    delta = target_sum - current_sum
    if abs(delta) > 1e-8:
        adjust_indices = np.where((h > 1e-8) & (h < 1 - 1e-8))[0]
        if len(adjust_indices) > 0:
            adjustment = delta / len(adjust_indices)
            h[adjust_indices] += adjustment
            h = np.clip(h, 0, 1)
    return h
