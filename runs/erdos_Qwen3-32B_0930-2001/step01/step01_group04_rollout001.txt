import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf

    # Hyperparameters
    learning_rate = 0.01
    perturbation_strength = 0.05
    perturbation_interval = 50

    while time.time() - start_time < budget_s:
        # Compute current correlation
        corr = np.correlate(h, 1 - h, mode='full')
        c_idx = np.argmax(corr)
        k_max_shift = c_idx - (n_points - 1)

        # Compute gradient
        indices_i = np.arange(n_points)
        indices_j = indices_i + k_max_shift
        mask = (indices_j >= 0) & (indices_j < n_points)
        grad = np.zeros_like(h)
        grad[mask] = 1.0 - h[indices_j[mask]]

        # Gradient update
        h -= learning_rate * grad

        # Projection to feasible region
        h = np.clip(h, 0.0, 1.0)
        current_sum = h.sum()
        h *= (n_points / 2) / current_sum

        # Evaluate best solution
        corr_current = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr_current) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            learning_rate = 0.01  # Reset learning rate on improvement
        else:
            learning_rate *= 0.95  # Decay learning rate

        # Perturbations to escape local minima
        if np.random.rand() < 0.05 or (time.time() - start_time + perturbation_interval // 2) % perturbation_interval < 1:
            perturbation = np.random.normal(0, perturbation_strength, n_points)
            h += perturbation
            h = np.clip(h, 0.0, 1.0)
            h *= (n_points / 2) / h.sum()

    return (best_h.tolist(), best_c5, n_points)
