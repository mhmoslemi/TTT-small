import numpy as np
import time

def compute_C5(h, n_points):
    dx = 2.0 / n_points
    shifted = 1.0 - h
    corr = np.correlate(h, shifted, mode='full')
    return np.max(corr) * dx

def perturb_and_adjust(h, sigma, n):
    m = max(1, int(n * 0.1))
    indices = np.random.choice(n, size=m, replace=False)
    h_new = h.copy()
    perturb = np.random.uniform(-sigma, sigma, m)
    h_new[indices] += perturb
    h_new = np.clip(h_new, 0.0, 1.0)
    current_sum = h_new.sum()
    target_sum = n / 2.0
    D = target_sum - current_sum
    if abs(D) < 1e-8:
        return h_new
    adjust_order = np.argsort(np.random.rand(n))
    step = 0.01
    for idx in adjust_order:
        if D > 0:
            available = 1.0 - h_new[idx]
            add = min(D, available)
            h_new[idx] += add
            D -= add
        else:
            available = h_new[idx]
            remove = min(-D, available)
            h_new[idx] -= remove
            D += remove
        if abs(D) < 1e-8:
            break
    return h_new

def hash_solution(h, decimals=5):
    return tuple(round(x, decimals) for x in h)

def update_tabu(tabu_list, candidate, max_size):
    if len(tabu_list) >= max_size:
        tabu_list.pop(0)
    tabu_list.append(candidate)
    return tabu_list
