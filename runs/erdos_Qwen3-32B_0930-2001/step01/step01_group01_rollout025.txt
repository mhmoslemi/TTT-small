        # Replace worst in population
        worst_idx = 0
        worst_c5 = compute_c5(population[0])
        for i in range(1, len(population)):
            ci = compute_c5(population[i])
            if ci > worst_c5:
                worst_c5 = ci
                worst_idx = i
        if child_c5 < worst_c5:
            population[worst_idx] = child

        # Update best solution
        if child_c5 < best_c5:
            best_c5 = child_c5
            best_h = child.copy()

    return (best_h, best_c5, n_points)
