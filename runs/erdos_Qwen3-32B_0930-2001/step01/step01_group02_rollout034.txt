import numpy as np
import time

def compute_C5(h, n_points):
    dx = 2.0 / n_points
    shifted = 1.0 - h
    corr = np.correlate(h, shifted, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    best_h = h.copy()
    best_c5 = compute_C5(best_h, n_points)
    current_h = best_h.copy()
    current_c5 = best_c5
    alpha = 0.995
    T = 1000.0 * current_c5  # Initial temperature
    start_time = time.time()

    while time.time() - start_time < budget_s:
        h_new = current_h.copy()
        m = max(1, int(0.1 * n_points))
        indices = np.random.choice(n_points, size=m, replace=False)
        for idx in indices:
            perturb = np.random.uniform(-0.1, 0.1)
            h_new[idx] += perturb
        h_new = np.clip(h_new, 0.0, 1.0)
        
        target_sum = n_points / 2.0
        D = target_sum - h_new.sum()
        while abs(D) > 1e-8:
            i = np.random.randint(0, n_points)
            if D > 0:
                add = min(D, 1.0 - h_new[i])
                if add < 1e-10:
                    continue
                h_new[i] += add
                D -= add
            else:
                remove = min(-D, h_new[i])
                if remove < 1e-10:
                    continue
                h_new[i] -= remove
                D += remove

        new_c5 = compute_C5(h_new, n_points)
        delta_E = new_c5 - current_c5

        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            current_h = h_new.copy()
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5

        T *= alpha

    return (best_h.tolist(), best_c5, n_points)
