def evaluate_individual(h, dx):
    conv = np.correlate(h, 1 - h, mode="full")
    return np.max(conv * dx)

def repair_binary(child, n_ones):
    current_sum = int(np.sum(child))
    if current_sum == n_ones:
        return child
    elif current_sum < n_ones:
        zero_indices = np.where(child == 0)[0]
        np.random.shuffle(zero_indices)
        to_flip = zero_indices[:(n_ones - current_sum)]

        one_indices = np.where(child == 1)[0]
        np.random.shuffle(one_indices)
        to_flip = one_indices[(current_sum - n_ones):]
