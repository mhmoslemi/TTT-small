            no_improvement += 1

        if no_improvement > max_no_improvement:
            break

        grad = np.zeros(n_points)
        for i in range(n_points - k_max):
            j = i + k_max
            grad[i] += (1 - h[j])
            grad[j] += -h[i]

        h -= learning_rate * grad
        h = project_feasible(h, n_points)

        if iteration % perturb_interval == 0:
            noise = np.random.normal(0, perturb_strength, n_points)
            h += noise
            h = project_feasible(h, n_points)

        if no_improvement >= decay_after:
            learning_rate *= learning_decay

    best_h = project_feasible(best_h, n_points)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    relevant_corr = corr[n_points - 1: 2 * n_points - 1]
    final_c5 = np.max(relevant_corr) * (2.0 / n_points)
    return (best_h.tolist(), final_c5, n_points)
