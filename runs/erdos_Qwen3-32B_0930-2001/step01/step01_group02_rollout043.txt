def compute_C5(h, n_points):
    dx = 2.0 / n_points
    shifted = 1.0 - h
    corr = np.correlate(h, shifted, mode='full')
    return np.max(corr) * dx

def adjust_sum(h, target_sum, n_points):
    current_sum = np.sum(h)
    for _ in range(n_points * 10):  # prevent infinite loop
        if abs(current_sum - target_sum) < 1e-8:
            break
        idx = np.random.randint(0, n_points)
        delta = target_sum - current_sum
        if delta > 0:
            max_add = 1.0 - h[idx]

            if max_remove < 1e-10:
                continue

            h[idx] -= remove

    return h

def compute_C5(h, n_points):
    dx = 2.0 / n_points
    shifted = 1.0 - h
    corr = np.correlate(h, shifted, mode='full')
    return np.max(corr) * dx

def adjust_sum(h, target_sum, n_points):
    current_sum = np.sum(h)
    for _ in range(n_points * 10):
        if abs(current_sum - target_sum) < 1e-8:
            break
        idx = np.random.randint(0, n_points)
        delta = target_sum - current_sum
        if delta > 0:
            max_add = 1.0 - h[idx]
            if max_add < 1e-10:
                continue
            add = min(delta, max_add)
            h[idx] += add
            current_sum += add
        else:
            max_remove = h[idx]
            if max_remove < 1e-10:
                continue
            remove = min(-delta, max_remove)
            h[idx] -= remove
            current_sum -= remove
    return h

def compute_gradient(h, k, dx, n_points):
    n = n_points
    gradient = np.zeros(n)

            gradient[i] += (1.0 - h[j]) * dx

    return gradient
