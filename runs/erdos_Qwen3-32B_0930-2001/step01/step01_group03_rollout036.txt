def compute_smooth_objective(h, dx, epsilon):
    # Compute overlaps O
    overlaps = np.correlate(h, 1 - h, mode='full') * dx

    exp_vals = np.exp(epsilon * overlaps)

def compute_smooth_objective(h, dx, epsilon):
    overlaps = np.correlate(h, 1 - h, mode='full') * dx
