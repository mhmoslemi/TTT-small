import numpy as np
import time

def compute_C5(h, n_points):
    dx = 2.0 / n_points
    shifted = 1.0 - h
    corr = np.correlate(h, shifted, mode='full')
    return np.max(corr) * dx

    start_time = time.time()

    alpha = 0.995  # Cooling rate
