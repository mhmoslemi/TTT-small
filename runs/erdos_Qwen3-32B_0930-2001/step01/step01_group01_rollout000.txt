from scipy.optimize import minimize
import time

def run():
    start_time = time.time()

    dx = 2.0 / n_points

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        result = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(result)

    def perturb_and_project(h, delta=0.03):
        h_perturbed = h + delta * np.random.randn(n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = h_perturbed.sum()
        desired_sum = n_points / 2.0
        scale = desired_sum / current_sum
        h_perturbed *= scale
        h_perturbed = np.clip(h_perturbed, 0, 1)
        return h_perturbed

    best_c5 = objective(h_initial)
    best_h = h_initial.copy()

    np.random.seed(42)
    max_iter = 30
    num_restarts = 20

    for _ in range(num_restarts):
        if time.time() - start_time >= 1050:  # early stop margin
            break
        # Perturb and project
        h0 = perturb_and_project(best_h)
        # Optimize
        res = minimize(
