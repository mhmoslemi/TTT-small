import numpy as np
import time

def run():
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    desired_sum = n_points / 2.0

    # Evaluate the C₅ for a candidate solution
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    # Project h to feasible region [0, 1]^n with sum(h) = desired_sum
    def project_individual(h):
        h_clip = np.clip(h, 0, 1)
        current_sum = np.sum(h_clip)
        if np.isclose(current_sum, desired_sum):
            return h_clip
        scale = desired_sum / current_sum
        h_scaled = h_clip * scale
        h_final = np.clip(h_scaled, 0, 1)
        delta = desired_sum - np.sum(h_final)
        h_final += delta / n_points
        return np.clip(h_final, 0, 1)

    # Tournament selection of a single individual
    def tournament_selection(population, fitnesses, size=2):
        indices = np.random.choice(len(population), size, replace=False)
        winner = min(indices, key=lambda i: fitnesses[i])
        return population[winner].copy()

    # Crossover: single-point
    def crossover(parent1, parent2):
        cross_point = np.random.randint(1, n_points)
        child1 = np.concatenate((parent1[:cross_point], parent2[cross_point:]))
        child2 = np.concatenate((parent2[:cross_point], parent1[cross_point:]))
        return [child1, child2]

    # Mutation: Gaussian noise followed by projection
    def mutate(child, mutation_rate=0.1, noise_std=0.05):
        mask = np.random.rand(n_points) < mutation_rate
        child[mask] += np.random.normal(0, noise_std, np.sum(mask))
        return project_individual(child)

    # Initialize population with initial_h_values and perturbed copies
    population_size = 30
    population = [h_initial.copy()]
    for _ in range(population_size-1):
        candidate = h_initial.copy() + 0.1 * np.random.randn(n_points)
        population.append(project_individual(candidate))

    # Evaluate initial population
    fitness = [evaluate_c5(p) for p in population]
    best_index = np.argmin(fitness)
    best_h = population[best_index].copy()
    best_c5 = fitness[best_index]

    # Genetic algorithm main loop
    for generation in range(10000 if n_points < 100 else 500):
        if time.time() - start_time >= 1100:
            break

        offspring = []

        # Create offspring through selection, crossover, and mutation
        for _ in range(population_size):
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            children = crossover(parent1, parent2)
            for child in children:
                mutated_child = mutate(child.copy())
                offspring.append(project_individual(mutated_child))

        # Evaluate all offspring
        offspring_fitness = [evaluate_c5(mutated) for mutated in offspring]
        
        # Combine existing population with offspring and select top performers
        combined = population + offspring
        combined_fitness = fitness + offspring_fitness
        
        # Sort by fitness and keep best population_size solutions
        indices = np.argsort(combined_fitness)
        population = [combined[i] for i in indices[:population_size]]
        fitness = [combined_fitness[i] for i in indices[:population_size]]
        
        # Update best solution
        current_best_index = np.argmin(fitness)
        if fitness[current_best_index] < best_c5:
            best_c5 = fitness[current_best_index]
            best_h = population[current_best_index].copy()

    return (best_h, best_c5, n_points)
