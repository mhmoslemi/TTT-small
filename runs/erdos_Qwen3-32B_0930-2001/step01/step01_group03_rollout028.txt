        reflected = centroid + alpha * (centroid - sorted_population[worst_idx])
        reflected = project(reflected)
        refl_eval = evaluate_individual(reflected, dx)
        
        if refl_eval < sorted_evals[-2]:  # Worse than second worst
            if refl_eval < sorted_evals[worst_idx]:
                population[sorted_indices[worst_idx]] = reflected
                evaluations[sorted_indices[worst_idx]] = refl_eval
        else:
            # Replace worst point with reflection
            population[sorted_indices[worst_idx]] = reflected
            evaluations[sorted_indices[worst_idx]] = refl_eval
            if refl_eval < sorted_evals[0]:  # Expand
                expanded = centroid + gamma * (reflected - centroid)
                expanded = project(expanded)
                exp_eval = evaluate_individual(expanded, dx)
                if exp_eval < refl_eval:
                    population[sorted_indices[worst_idx]] = expanded
                    evaluations[sorted_indices[worst_idx]] = exp_eval
            else:  # Contract
                contracted = centroid + beta * (sorted_population[worst_idx] - centroid)
                contracted = project(contracted)
                cont_eval = evaluate_individual(contracted, dx)
                if cont_eval < evaluations[worst_idx]:
                    population[sorted_indices[worst_idx]] = contracted
                    evaluations[sorted_indices[worst_idx]] = cont_eval
                else:  # Shrink
                    for i in range(1, len(population)):
                        population[i] = sorted_population[0] + sigma * (population[i] - sorted_population[0])
                        population[i] = project(population[i])
                        evaluations[i] = evaluate_individual(population[i], dx)
        
        # Update best solution
        current_best = np.min(evaluations)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[np.argmin(evaluations)].copy()
    
    return (best_h, best_c5, n_points)
