    np.random.seed(seed)

    h = initial_h_values.copy()
    h = project_h(h, target_sum)
    best_h = h.copy()
    best_c5 = evaluate_real_c5(best_h, dx)
    
    eps_perturb = 1e-5
    initial_step_size = 0.1
    epsilon_smooth_start = 0.1
    epsilon_smooth_decay = 0.9
    step_decay = 0.995
    max_iterations = 10000
    current_step_size = initial_step_size
    current_epsilon = epsilon_smooth_start

    start_time = time.time()
    max_time = start_time + budget_s

    iteration = 0

    while time.time() < max_time and iteration < max_iterations:
        current_c5 = evaluate_real_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        grad = np.zeros_like(h)
        for i in range(n_points):
            h_up = h.copy()
            h_up[i] += eps_perturb
            h_up = project_h(h_up, target_sum)
            f_up = evaluate_smooth_c5(h_up, dx, current_epsilon)

            h_down = h.copy()
            h_down[i] -= eps_perturb
            h_down = project_h(h_down, target_sum)
            f_down = evaluate_smooth_c5(h_down, dx, current_epsilon)

            grad[i] = (f_up - f_down) / (2 * eps_perturb)

        h -= current_step_size * grad
        h = project_h(h, target_sum)

        current_step_size *= step_decay
        current_epsilon *= epsilon_smooth_decay
        iteration += 1

    return (best_h, best_c5, n_points)
