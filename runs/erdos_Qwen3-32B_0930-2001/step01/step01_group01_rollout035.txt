import numpy as np
from scipy.optimize import minimize
import time

def run():
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        result = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(result)

    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    options = {'maxiter': 500, 'ftol': 1e-8, 'disp': False}

    best_c5 = objective(h_initial)
    best_h = h_initial.copy()

    def optimize_h(h0):
        h_flat_initial = h0.flatten()
        result = minimize(
            objective,
            h_flat_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options=options
        )
        if result.success:
            h_opt = result.x.reshape(n_points)
            current_c5 = objective(h_opt)
            if current_c5 < best_c5:
                return h_opt, current_c5
        return best_h, best_c5

    def project_feasible(h, target_sum_val):
        h = h.copy()
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum_val) < 1e-8:
            return h
        delta = (target_sum_val - current_sum) / n_points
        h += delta
        h = np.clip(h, 0, 1)
        return h

    # Initial optimization
    h_opt, c5 = optimize_h(h_initial)
    best_h, best_c5 = h_opt, c5

    # Controlled perturbations with projection
    np.random.seed(42)
    max_perturbations = 20
    perturbation_steps = 0
    while perturbation_steps < max_perturbations:
        if time.time() - start_time >= 1100:
            break
        perturbation_steps += 1
        h_perturbed = h_initial.copy()
        n_perturb = max(1, int(0.1 * n_points))
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        delta = 0.05 * np.random.randn(n_perturb)
        h_perturbed[indices] = np.clip(h_perturbed[indices] + delta, 0, 1)
        h_perturbed = project_feasible(h_perturbed, target_sum)
        h_opt, c5 = optimize_h(h_perturbed)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    return (best_h, best_c5, n_points)
