        iteration += 1
        
        # Compute gradient via finite differences
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Forward perturbation
            h_forward = h.copy()

            if h_forward_sum != 0:

            conv_forward = np.correlate(h_forward, 1 - h_forward, mode='full')

            # Backward perturbation
            h_backward = h.copy()
