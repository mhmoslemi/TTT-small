```python
import numpy as np
from scipy.optimize import minimize
import time

def run():
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        result = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(result)

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    options_base = {'maxiter': 80, 'ftol': 1e-8, 'disp': False, 'eps': 1e-4}

    best_h = h_initial.copy()
    best_c5 = objective(best_h)

    np.random.seed(42)
    perturb_mags = [0.1, 0.15, 0.2, 0.25, 0.05]
    n_restarts = 30

    for _ in range(n_restarts):
        if time.time() - start_time >= 1100:
            break
        mag = np.random.choice(perturb_mags)
        h_perturbed = h_initial + mag * np.random.randn(n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        scale = (n_points / 2) / h_perturbed.sum()
        h_perturbed = np.clip(h_perturbed * scale, 0, 1)
        
        result = minimize(
            objective,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options=options_base
        )
        if result.success:
            h_opt = result.x.reshape(n_points)
            current_c5 = objective(h_opt)
            if current_c5 < best_c5:
                best_h[:] = h_opt
                best_c5 = current_c5

    return (best_h, best_c5, n_points)
```
