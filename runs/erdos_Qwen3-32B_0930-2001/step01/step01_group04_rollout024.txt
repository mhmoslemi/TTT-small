def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    np.random.seed(seed)
    start_time = time.time()

    h_initial = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def project_feasible(h):
        h_clipped = np.clip(h, 0, 1)
        target_sum = n_points / 2
        current_sum = h_clipped.sum()
        h_scaled = h_clipped * target_sum / current_sum
        return h_scaled

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    current_h = project_feasible(h_initial)
    current_energy = evaluate(current_h)
    best_h = current_h.copy()
    best_energy = current_energy

    # Simulated Annealing parameters
    T = 0.1
    alpha = 0.9995
    steps_without_improvement = 0
    max_no_improve = 2000
    reset_T_factor = 3.0  # If no improvement, increase T

    while time.time() - start_time < budget_s:
        # Perturb current_h
        noise = np.random.normal(0, 0.1 * T, size=n_points)
        new_h = current_h + noise
        new_h = project_feasible(new_h)
        new_energy = evaluate(new_h)
        delta_energy = new_energy - current_energy

        # Accept or reject the new solution
        if delta_energy < 0:
            current_h = new_h.copy()
            current_energy = new_energy
            if new_energy < best_energy:
                best_h = new_h.copy()
                best_energy = new_energy
                steps_without_improvement = 0
            else:
                steps_without_improvement += 1
        else:
            prob_accept = np.exp(-delta_energy / T)
            if np.random.rand() < prob_accept:
                current_h = new_h.copy()
                current_energy = new_energy
            steps_without_improvement += 1

        # Reset temperature if no improvement for too long
        if steps_without_improvement > max_no_improve:
            T *= reset_T_factor
            steps_without_improvement = 0
            # print(f"Reset T to {T}")  # Not allowed to print

        # Cool down
        T *= alpha

        # Early exit if time is up
        if time.time() - start_time >= budget_s:
            break

    # Feasibility check (redundant but safe)
    best_h = project_feasible(best_h)
    final_energy = evaluate(best_h)

    return (best_h.tolist(), final_energy, n_points)
