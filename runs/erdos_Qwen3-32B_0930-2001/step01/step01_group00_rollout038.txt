        selected = population_sorted[:8]
        current_best = compute_c5(selected[0])
        if current_best < best_c5:
            best_c5 = current_best
            best_h = selected[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1

        offspring = []
        for _ in range(7):
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child = (parent1 + parent2) / 2
            indices = np.random.choice(n_points, int(0.2 * n_points), replace=False)
            mutation = np.random.normal(0, 0.1, len(indices))
            child[indices] += mutation
            child = make_feasible(child)
            offspring.append(child)
        population = selected + offspring

        if no_improvement > 100:
            population = [best_h]
            for _ in range(14):
                pop = np.random.uniform(0, 1, n_points)
                pop = make_feasible(pop)
                population.append(pop)
            no_improvement = 0

    return best_h, best_c5, n_points
