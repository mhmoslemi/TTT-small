import numpy as np
import time
from collections.abc import Callable

def compute_C5(h: np.ndarray, n_points: int) -> float:
    dx = 2.0 / n_points
    shifted = 1.0 - h
    corr = np.correlate(h, shifted, mode='full')
    return np.max(corr) * dx

def hash_solution(h: np.ndarray) -> tuple:
    return tuple(round(x, 5) for x in h)

def perturb_and_adjust(h: np.ndarray, sigma: float, n_points: int) -> np.ndarray:
    h_new = h.copy()
    m = max(1, int(n_points * 0.1))
    indices = np.random.choice(n_points, size=m, replace=False)
    perturbations = np.random.uniform(-sigma, sigma, m)
    h_new[indices] += perturbations
    h_new = np.clip(h_new, 0.0, 1.0)
    
    current_sum = h_new.sum()
    target_sum = n_points / 2.0
    D = target_sum - current_sum
    if abs(D) < 1e-8:
        return h_new
    
    adjust_order = np.random.permutation(n_points)
    for idx in adjust_order:
        if D > 0:
            max_add = 1.0 - h_new[idx]
            if max_add > 1e-8:
                add = min(D, max_add)
                h_new[idx] += add
                D -= add
        else:
            max_remove = h_new[idx]
            if max_remove > 1e-8:
                remove = min(-D, max_remove)
                h_new[idx] -= remove
                D += remove
        if abs(D) < 1e-8:
            break
    return h_new

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    assert np.all(h >= 0.0) and np.all(h <= 1.0), "Initial h out of bounds"
    assert np.isclose(np.sum(h), n_points / 2.0), "Initial h violates mass constraint"
    best_h = h.copy()
    best_c5 = compute_C5(best_h, n_points)
    
    start_time = time.time()
    sigma = 0.05
    tabu_list = []
    tabu_max_size = 20
    
    while time.time() - start_time < budget_s:
        made_improvement = False
        for _ in range(10):  # Batch of 10 candidates
            candidate = perturb_and_adjust(best_h, sigma, n_points)

    return (best_h.tolist(), best_c5, n_points)
