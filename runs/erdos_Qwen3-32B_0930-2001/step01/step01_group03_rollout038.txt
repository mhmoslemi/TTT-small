import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy().astype(float)
    best_h = h.copy()

    start_time = time.time()
    
    epsilon = 0.1
    alpha = 0.01
    iter_count = 0
    
    while time.time() - start_time < budget_s:

        overlaps = np.correlate(h, 1 - h, mode='full') * dx

        exp_vals = np.exp(epsilon * overlaps)
        sum_exp = np.sum(exp_vals)
        weights = exp_vals / sum_exp
        
        # Compute gradient manually
        grad = np.zeros(n_points)
        for k in range(len(overlaps)):
            m = k - (n_points - 1)
            w = weights[k]
            for i in range(n_points):
                i_plus_m = i + m
                if 0 <= i_plus_m < n_points:
                    grad[i] += w * (1 - h[i_plus_m]) * dx
                i_minus_m = i - m
                if 0 <= i_minus_m < n_points:
                    grad[i] += w * (-h[i_minus_m]) * dx

        h -= alpha * grad

        current_c5 = evaluate_individual(h, dx)
        if current_c5 < best_c5:
