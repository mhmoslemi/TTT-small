                expanded = centroid + gamma * (reflected - centroid)
                expanded = project(expanded)
                e_exp = evaluate(expanded, dx, best_c5)
                if e_exp < e_ref:
                    simplex[worst_idx] = expanded
                else:
                    simplex[worst_idx] = reflected
            else:
                # Replace worst with reflected
                simplex[worst_idx] = reflected
        elif e_ref < evaluations[-1]:  # Better than worst
            simplex[-1] = reflected
        else:  # Contraction
            contracted = centroid + beta * (worst_point - centroid)
            contracted = project(contracted)
            e_con = evaluate(contracted, dx, best_c5)
            if e_con < evaluations[-1]:
                simplex[-1] = contracted
            else:
                # Shrink
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[0] + sigma * (simplex[i] - simplex[0])
                    simplex[i] = project(simplex[i])

    return (best_h, best_c5, n_points)
