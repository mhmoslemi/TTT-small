import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    
    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)
    
    def objective_with_penalty(h_vec):
        h = h_vec.reshape(n_points)
        current_sum = h.sum()
        target_sum = n_points / 2
        penalty = 1e4 * (current_sum - target_sum) ** 2
        return compute_c5(h_vec) + penalty
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run differential evolution
    max_de_iter = 30
    result_de = optimize.differential_evolution(
        objective_with_penalty, 
        bounds=bounds,
        strategy='rand1bin',
        maxiter=max_de_iter,
        popsize=10,
        tol=1e-3,
        mutation=(0.5, 1.0),
        recombination=0.8,
        seed=seed
    )
    
    elapsed = time.time() - start_time
    remaining_time = max(0, budget_s - elapsed)
    
    # Project result to feasible region
    def project_to_feasible(h_vec):
        h = h_vec.copy()
        current_sum = np.sum(h)
        target_sum = n_points / 2
        scale = target_sum / current_sum
        h_scaled = h * scale
        # Ensure h is within [0, 1]
        h_clipped = np.clip(h_scaled, 0.0, 1.0)
        current_sum = np.sum(h_clipped)
        scale = target_sum / current_sum
        final_h = h_clipped * scale
        final_h = np.clip(final_h, 0.0, 1.0)
        return final_h
    
    h_proj = project_to_feasible(result_de.x)
    result_de.x = h_proj
    
    best_h = result_de.x
    best_c5 = compute_c5(best_h)
    
    if elapsed < budget_s:
        # Local refinement with SLSQP
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        result_slsqp = optimize.minimize(
            compute_c5, 
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-8}
        )
        if result_slsqp.success:
            h_proj = project_to_feasible(result_slsqp.x)
            c5 = compute_c5(h_proj)
            if c5 < best_c5:
                best_h = h_proj
                best_c5 = c5
    
    return best_h, best_c5, n_points
