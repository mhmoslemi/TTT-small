import numpy as np
import time

def project_feasible(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    target_sum = n_points / 2
    current_sum = h_clipped.sum()
    h_scaled = h_clipped * target_sum / current_sum
    return h_scaled

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points

    T_initial = 0.1
    T = T_initial
    alpha = 0.9995
    current_h = project_feasible(h.copy(), n_points)
    current_corr = np.correlate(current_h, 1 - current_h, mode='full')
    current_c5 = np.max(current_corr) * dx
    best_h = current_h.copy()
    best_c5 = current_c5

    consecutive_no_improve = 0
    max_no_improve = 1000

    while time.time() - start_time < budget_s:
        # Perturb with noise scaled to temperature
        noise_std = 0.05 * (T / T_initial)
        noise = np.random.normal(0, noise_std, size=n_points)
        new_h = current_h + noise
        new_h = project_feasible(new_h, n_points)

        # Evaluate new candidate
        new_corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(new_corr) * dx
        delta = new_c5 - current_c5

        # Metropolis acceptance
        if delta < 0:
            current_h[:] = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h[:] = current_h
                best_c5 = new_c5
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1
        else:
            if np.log(np.random.rand()) < -delta / T:
                current_h[:] = new_h
                current_c5 = new_c5
                consecutive_no_improve += 1

                consecutive_no_improve += 1

        # Temperature control
        if consecutive_no_improve >= max_no_improve:
