```python
import numpy as np
from scipy.optimize import minimize
import time

def run():
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        result = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(result)

    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    options = {'maxiter': 150, 'ftol': 1e-8, 'disp': False}

    best_h = h_initial.copy()
    best_c5 = objective(best_h)

    np.random.seed(42)
    global_count = 0
    while time.time() - start_time < 1100:
        # Perturb current best_h with random subset and thresholding
        h_perturbed = best_h.copy()
        num_perturb = n_points // 2
        perturb_indices = np.random.choice(n_points, num_perturb, replace=False)
        for i in perturb_indices:
            if np.random.rand() < 0.5:
                h_perturbed[i] = max(0.0, min(1.0, h_perturbed[i] - 0.2))
            else:
                h_perturbed[i] = max(0.0, min(1.0, h_perturbed[i] + 0.2))
        # Fix constraint
        current_sum = h_perturbed.sum()
        if current_sum != 0:
            scale = target_sum / current_sum
            h_perturbed *= scale
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)

        # Local refinement
        res = minimize(
            objective,
            h_perturbed.ravel(),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options=options
        )
        if res.success:
            h_candidate = res.x.reshape(n_points)
            current_c5 = objective(h_candidate)
        else:
            h_candidate = h_perturbed.copy()
            current_c5 = objective(h_candidate)

        if current_c5 < best_c5:
            best_h[:] = h_candidate[:]
            best_c5 = current_c5

        global_count += 1
        # Force early exit if time is tight
        if time.time() - start_time > 1000:
            break

    return (best_h, best_c5, n_points)
```
