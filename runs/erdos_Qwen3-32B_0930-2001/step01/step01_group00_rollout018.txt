                if h_sol is not None:
                    # Project to valid constraint surface
                    h_sol = np.clip(h_sol, 0, 1)
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    new_c5 = evaluate_erdos_solution(h_sol)
                    if new_c5 < c5_best:
                        h_best = h_sol
                        c5_best = new_c5
            else:
                # Fallback to ECOS if MOSEK incomplete
                problem.solve(solver=cp.ECOS, verbose=False)
                h_sol = h.value
                if h_sol is not None:
                    h_sol = np.clip(h_sol, 0, 1)
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    new_c5 = evaluate_erdos_solution(h_sol)
                    if new_c5 < c5_best:
                        h_best = h_sol
                        c5_best = new_c5
        except Exception as e:
            pass

    return h_best, c5_best, n_points
