def evaluate_individual(h, dx):
    conv = np.correlate(h, 1 - h, mode="full")
    return np.max(conv * dx)

    dx = 2.0 / n_points

            constraints += [
                z[i, j] <= h[i],
                z[i, j] <= h[j],

                z[i, j] >= 0
            ]
