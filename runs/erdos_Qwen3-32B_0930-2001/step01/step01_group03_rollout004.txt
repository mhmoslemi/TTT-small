import numpy as np
import time

def compute_gradient(h, dx, overlaps, epsilon, n):
    grad = np.zeros_like(h)
    num_overlaps = overlaps.shape[0]
    sum_exp = np.sum(np.exp(epsilon * overlaps))
    if sum_exp == 0:
        return grad
    weights = np.exp(epsilon * overlaps) / sum_exp
    for m in range(num_overlaps):
        weight = weights[m]
        if np.isclose(weight, 0):
            continue
        s = m - (n - 1)
        if s >= 0:
            start_j = 0
            end_j = n - 1 - s
            valid_js = range(start_j, end_j + 1)
        else:
            t = -s
            start_j = t
            end_j = n - 1

        for j in valid_js:
            idx = j
            neighbor = j + s
            grad[idx] += (1 - h[neighbor]) * dx * weight
