    global_best_score = personal_best_scores[global_idx]

    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    velocity_clamp = 0.1

    while time.time() - start_time < budget_s - 1:
        for i in range(n_particles):
            current_h = particles[i]
            current_score = compute_c5(current_h)
            # Update personal best
            if current_score < personal_best_scores[i]:
                personal_best_scores[i] = current_score
                personal_best_positions[i] = current_h.copy()
                if current_score < global_best_score:
                    global_best_score = current_score
                    global_best = current_h.copy()
            
            # Update velocity
            r1 = np.random.rand()
            r2 = np.random.rand()
            pb_diff = personal_best_positions[i] - current_h
            gb_diff = global_best - current_h
            velocities[i] = (inertia * velocities[i] +
                             cognitive * r1 * pb_diff + 
                             social * r2 * gb_diff)
            velocities[i] = np.clip(velocities[i], -velocity_clamp, velocity_clamp)
            
            # Update position
            particles[i] += velocities[i]
            # Clip to bounds
            particles[i] = np.clip(particles[i], 0, 1)
            # Scale to preserve sum
            current_sum = np.sum(particles[i])
            if current_sum != 0:
                particles[i] *= desired_sum / current_sum
            # Re-clip after scaling
            particles[i] = np.clip(particles[i], 0, 1)
        
        if time.time() - start_time > budget_s - 1:
            break

    # Final constraint fix
    h_values = global_best
    current_sum = np.sum(h_values)
    if not np.isclose(current_sum, desired_sum, atol=1e-8):
        excess = desired_sum - np.sum(h_values)
        indices = np.argsort(np.abs(h_values - 0.5))[::-1]
        h_values[indices[:10]] += excess / 10.0
        h_values = np.clip(h_values, 0, 1)
    
    c5 = compute_c5(h_values)
    return h_values, c5, n_points
