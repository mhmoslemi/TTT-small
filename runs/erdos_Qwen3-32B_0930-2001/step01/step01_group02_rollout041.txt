import numpy as np
import time

def compute_C5(h, n_points):
    dx = 2.0 / n_points
    shifted = 1.0 - h
    corr = np.correlate(h, shifted, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    h = np.clip(h, 0.0, 1.0)
    n_points = len(h)
    target_sum = n_points / 2.0
    while abs(h.sum() - target_sum) > 1e-8:
        i = np.random.randint(n_points)
        if h[i] < 1.0 - 1e-6:
            h[i] += 1.0 - h[i]
        elif h[i] > 1e-6:
            h[i] -= h[i]
    best_h = h.copy()
    best_c5 = compute_C5(best_h, n_points)
    start_time = time.time()
    
    tabu_size = 20
    tabu_list = []
    sigma = 0.05
    
    while time.time() - start_time < budget_s:
        improvements = 0
        for _ in range(5):  # Reduced attempts per iteration
            h_perturbed = best_h.copy()
            m = max(1, int(n_points * 0.1))
            idxs = np.random.choice(n_points, size=m, replace=False)
            perturbations = np.random.uniform(-sigma, sigma, size=m)
            h_perturbed[idxs] += perturbations
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            
            # Adjust for sum constraint
            current_sum = h_perturbed.sum()
            D = target_sum - current_sum
            if abs(D) > 1e-8:
                while abs(D) > 1e-8:
                    i = np.random.randint(n_points)
                    if D > 0:
                        add = min(D, 1.0 - h_perturbed[i])
                        h_perturbed[i] += add
                        D -= add
                    else:
                        remove = min(-D, h_perturbed[i])
                        h_perturbed[i] -= remove
                        D += remove
            
            # Generate hash for tabu check
            candidate_hash = tuple(round(h_perturbed[i], 5) for i in range(n_points))
            if candidate_hash not in tabu_list:
                current_c5 = compute_C5(h_perturbed, n_points)
                if current_c5 < best_c5:
                    best_h = h_perturbed.copy()
                    best_c5 = current_c5
                    improvements += 1
                    tabu_list.clear()
                    sigma = max(0.01, sigma * 0.9)
                else:
                    # Add to tabu, expand exploration if no progress
                    if len(tabu_list) >= tabu_size:
                        tabu_list.pop(0)
                    tabu_list.append(candidate_hash)
                    sigma = min(0.5, sigma * 1.15)
        if improvements == 0:
            sigma = min(0.5, sigma * 1.15)
        if time.time() - start_time >= budget_s:
            break
