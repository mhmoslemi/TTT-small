import numpy as np
import time

def run():
    start_time = time.time()
    h = np.array(initial_h_values)  # Ensure h is an array
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    def project_to_feasible(h_vec):
        h = h_vec.reshape(n_points)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-6:
            return h
        # Scale to target sum and re-clip
        scale = target_sum / current_sum
        h *= scale
        h = np.clip(h, 0, 1)
        # Final adjustment if sum is still off
        remaining = target_sum - h.sum()
        if abs(remaining) > 1e-6:
            # Small adjustment to closest elements
            h += remaining / n_points
            h = np.clip(h, 0, 1)
        return h

    pop_size = 20
    population = [np.array(h)]
    for _ in range(pop_size - 1):
        perturbed = h + 0.05 * np.random.randn(n_points)
        perturbed = project_to_feasible(perturbed)
        population.append(perturbed)

    best_c5 = float('inf')
    best_h = h.copy()

    while time.time() - start_time < 1100:
        # Evaluate all in population
        fitness = []
        for indiv in population:
            indiv_proj = project_to_feasible(indiv)
            fitness.append((compute_c5(indiv_proj), indiv_proj))
        fitness.sort(key=lambda x: x[0])
        current_best = fitness[0]
        if current_best[0] < best_c5:
            best_c5 = current_best[0]
            best_h = current_best[1].copy()

        # Tournament selection
        def tournament():
            indexes = np.random.choice(len(fitness), 3, replace=False)
            best_idx = np.argmin([fitness[i][0] for i in indexes])
            return fitness[indexes[best_idx]][1]

        # Generate offspring
        new_offspring = []
        for _ in range(pop_size // 2):
            p1 = tournament()
            p2 = tournament()
            # Crossover: segment-based
            cx_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([p1[:cx_point], p2[cx_point:]])
            # Mutation
            mut_mask = np.random.rand(n_points) < 0.1
            child[mut_mask] += np.random.randn(np.sum(mut_mask)) * 0.1
            child = project_to_feasible(child)
            new_offspring.append(child)

        # Elitism + Offspring
        next_generation = [f[1] for f in fitness[:pop_size - len(new_offspring)]] + new_offspring
        population = next_generation

    return (best_h, best_c5, n_points)
