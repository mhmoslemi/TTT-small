def compute_gradient(h, dx, tau):
    n = len(h)
    # Compute all O_k for each shift
    overlap = np.correlate(h, 1 - h, mode='full') * dx
    k_values = np.arange(-(n-1), n)
    log_sum = np.log(np.sum(np.exp(tau * overlap)))  # Smooth max is log_sum / tau
    weights = np.exp(tau * overlap - log_sum)  # Normalize the exp(tau*O)

    total_gradient = np.zeros(n)
    for k_idx, k in enumerate(k_values):
        w_k = weights[k_idx]
        g_k = np.zeros(n)
        for i in range(n):
            # Compute the two terms for dO_kdh[i]
            term1 = 0.0
            if (i + k) >= 0 and (i + k) < n:
                term1 = (1.0 - h[i + k]) * dx
            term2 = 0.0
            if (i - k) >= 0 and (i - k) < n:
                term2 = h[i - k] * dx
            g_k[i] = term1 - term2
        total_gradient += w_k * g_k
    return total_gradient
