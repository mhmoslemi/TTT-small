import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def compute_c5(h_vec):
        h = h_vec
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    population_size = 15
    population = []
    for _ in range(population_size):
        h = initial_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        norm = np.sum(h)
        h = h * (n_points / 2) / norm if norm != 0 else np.zeros(n_points)
        population.append(h)
    population = np.array(population)

    best_c5 = np.inf
    best_h = population[0].copy()

    while time.time() - start_time < budget_s:
        c5_values = np.array([compute_c5(ind) for ind in population])
        current_idx = np.argmin(c5_values)
        if c5_values[current_idx] < best_c5:
            best_c5 = c5_values[current_idx]
            best_h = population[current_idx].copy()

        selected = population[np.argsort(c5_values)[:population_size//2]]

            child = (p1 + p2) / 2
            child = np.clip(child, 0, 1)
            total = np.sum(child)
            if total != 0:
                child = child * (n_points / 2) / total
            if np.random.rand() < 0.3:
                mask = np.random.rand(n_points) < 0.2
