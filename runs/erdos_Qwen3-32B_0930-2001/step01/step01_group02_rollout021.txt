    return np.max(corr) * dx

    n_points = len(h)
    grad = np.zeros_like(h)
    for j in range(n_points):
        term1 = 0.0
        j_plus_k = j + k
        if 0 <= j_plus_k < n_points:
            term1 = 1.0 - h[j_plus_k]
        term2 = 0.0
        j_minus_k = j - k
        if 0 <= j_minus_k < n_points:
