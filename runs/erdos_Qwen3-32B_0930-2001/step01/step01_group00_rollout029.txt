    n_points = len(h_current)
    dx = 2.0 / n_points

        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)
    
    def constrained_objective(x):
        return compute_c5(x)
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    
    # Differential evolution with penalty-based constraint handling
    def penalized_objective(x, penalty):
        c5 = compute_c5(x)
        constraint_violation = np.sum(x) - n_points / 2
        return c5 + penalty * constraint_violation**2
    
    # First pass: coarse search with lower precision
    if time.time() - start_time < budget_s - 200:

        result_de = optimize.differential_evolution(
            lambda x: penalized_objective(x, initial_penalty),

            maxiter=50,

            mutation=0.7,

        )
        best_x = result_de.x

    # Refinement with adapted penalty
    final_x = best_x.copy()

    remaining_time = max(budget_s - (time.time() - start_time), 50)
