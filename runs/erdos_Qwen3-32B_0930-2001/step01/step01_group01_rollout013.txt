import numpy as np
from scipy.optimize import minimize
import time

def run():
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    desired_sum = n_points / 2.0

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    def constraint_sum(h_flat):
        return np.sum(h_flat) - desired_sum

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    options = {'maxiter': 1000, 'ftol': 1e-9, 'disp': False}

    best_c5 = objective(h_initial)
    best_h = h_initial.copy()

    def optimize_h(h0):
        h_flat_initial = h0.copy().ravel()
        result = minimize(
            objective,
            h_flat_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options=options
        )
        if result.success:
            h_opt = result.x.reshape(n_points)
            current_c5 = objective(h_opt)
            if current_c5 < best_c5:
                return h_opt, current_c5
        return best_h, best_c5

    h_opt, c5 = optimize_h(h_initial)
    best_h, best_c5 = h_opt, c5

    np.random.seed(42)
    max_restarts = 20
    noise_scale = 0.025
    for restart in range(max_restarts):
        if time.time() - start_time >= 1100:
            break
        h_pert = h_initial + noise_scale * np.random.randn(n_points)
        h_pert = np.clip(h_pert, 0, 1)
        s = h_pert.sum()

            scale_factor = desired_sum / s
            h_pert = h_pert * scale_factor

        h_opt, c5 = optimize_h(h_pert)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    return (best_h, best_c5, n_points)
