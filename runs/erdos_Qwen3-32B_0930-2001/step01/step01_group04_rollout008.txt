import numpy as np
import cvxpy as cp
import time

def project_feasible(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    target_sum = n_points / 2.0
    current_sum = h_clipped.sum()
    h_scaled = h_clipped * (target_sum / current_sum) if current_sum > 1e-8 else h_clipped.copy()
    return h_scaled

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    best_c5 = float('inf')

    while time.time() - start_time < budget_s:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        k_phys = max_idx - (n_points - 1)
        
        coeff = np.zeros(n_points)
        for x in range(n_points):
            x_k = x + k_phys
            if 0 <= x_k < n_points:
                coeff[x] += 1 - best_h[x_k]
                coeff[x_k] += -best_h[x]
        
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
        ]
        objective = cp.Minimize(coeff @ h_var)
        prob = cp.Problem(objective, constraints)

        new_c5 = np.max(corr_new) * dx
        if new_c5 < best_c5:

            best_h[:] = h_new
