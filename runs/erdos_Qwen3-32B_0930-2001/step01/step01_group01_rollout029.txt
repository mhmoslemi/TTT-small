import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100):
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        result = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(result)

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    options = {'maxiter': 200, 'ftol': 1e-8, 'disp': False}

    best_c5 = objective(h_initial)
    best_h = h_initial.copy()

    np.random.seed(42)

    def perturb_h(h, perturbation_rate=0.5, delta=0.2):
        h_perturbed = h.copy()
        indices = np.random.choice(n_points, size=int(n_points * perturbation_rate), replace=False)
        for i in indices:
            if np.random.rand() < 0.5:
                h_perturbed[i] = max(0, h_perturbed[i] - delta)
            else:
                h_perturbed[i] = min(1, h_perturbed[i] + delta)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        scale = (n_points / 2) / h_perturbed.sum()
        h_perturbed = np.clip(h_perturbed * scale, 0, 1)
        return h_perturbed

    # Initial optimization
    res = minimize(
        objective,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options=options
    )
    if res.success:
        current_h = res.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h, best_c5 = current_h, current_c5

    # Global perturbations
    global_iter = 0
    max_global_iter = 100
    while global_iter < max_global_iter and time.time() - start_time < budget_s:
        global_iter += 1
        h_perturbed = perturb_h(best_h)
        res = minimize(
            objective,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options=options
        )
        if res.success:
            current_h = res.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h, best_c5 = current_h, current_c5

    return (best_h, best_c5, n_points)
