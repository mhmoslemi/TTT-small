import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_current = initial_h_values.copy()
    n_points = len(h_current)
    dx = 2.0 / n_points

        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    best_h = h_current.copy()

    high = 0.5  # Conservative upper bound

            break

        target_corr = mid / dx

            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) - target_corr
