    required_sum = n_points / 2.0
    
    # Parameters for PSO
    swarm_size = 20
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    velocity_clamp = 0.1
    
    # Initialize particles
    particles = np.zeros((swarm_size, n_points))
    velocities = np.zeros((swarm_size, n_points))
    personal_best_positions = np.zeros((swarm_size, n_points))
    personal_best_scores = np.full(swarm_size, np.inf)
    global_best_position = np.copy(initial_h)
    global_best_score = compute_c5(global_best_position, n_points, dx)
    
    # Initialize the swarm around initial_h with small perturbations
    for i in range(swarm_size):
        perturbation = np.random.normal(0, 0.05, size=n_points)
        particle = np.clip(initial_h + perturbation, 0, 1)
        current_sum = np.sum(particle)
        if current_sum != 0:
            scaling = required_sum / current_sum
            particle = np.clip(particle * scaling, 0, 1)
        particle = np.clip(particle, 0, 1)
        particles[i] = particle
        
        score = compute_c5(particle, n_points, dx)
        if score < personal_best_scores[i]:
            personal_best_scores[i] = score
            personal_best_positions[i] = particle.copy()
            if score < global_best_score:
                global_best_score = score
                global_best_position = particle.copy()
    
    # Main optimization loop
    while time.time() - start_time < budget_s:
        for i in range(swarm_size):
            # Update velocity
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = (
                inertia * velocities[i] +
                cognitive * r1 * (personal_best_positions[i] - particles[i]) +
                social * r2 * (global_best_position - particles[i])
            )
            velocities[i] = np.clip(velocities[i], -velocity_clamp, velocity_clamp)
            
            # Update position and constraints
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            current_sum = np.sum(particles[i])
            if current_sum != 0:
                scaling = required_sum / current_sum
                particles[i] = np.clip(particles[i] * scaling, 0, 1)
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Evaluate new position
            score = compute_c5(particles[i], n_points, dx)
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = particles[i].copy()
                if score < global_best_score:
                    global_best_score = score
                    global_best_position = particles[i].copy()
    
    return global_best_position, global_best_score, n_points
