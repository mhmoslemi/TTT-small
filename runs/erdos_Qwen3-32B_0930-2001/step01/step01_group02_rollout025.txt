def run(seed=42, budget_s=1100, **kwargs):

    def compute_C5(h_vec):

        corr = np.correlate(h_vec, shifted, mode='full') * dx
        return np.max(corr)

    best_c5 = compute_C5(h)
    best_h = h.copy()
    step_size = 0.1

    while time.time() - start_time < budget_s:
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_idx = np.argmax(overlaps)
        k_active = max_idx - (n_points - 1)

        grad = np.zeros(n_points)
        for a in range(n_points):
            if (a + k_active) >= 0 and (a + k_active) < n_points:
                grad[a] += (1.0 - h[a + k_active]) * dx
            if (a - k_active) >= 0 and (a - k_active) < n_points:
                grad[a] -= h[a - k_active] * dx

        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0.0, 1.0)

        target_sum = n_points / 2.0
