import numpy as np
import time

def compute_C5(h, n_points):
    dx = 2.0 / n_points
    shifted = 1.0 - h
    corr = np.correlate(h, shifted, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, k, dx):
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        term1 = 0
        if 0 <= i + k < n:
            term1 = 1 - h[i + k]
        term2 = 0
        if 0 <= i - k < n:
            term2 = h[i - k]
        grad[i] = dx * (term1 - term2)
    return grad

def adjust_sum(h, target_sum):
    n_points = len(h)
    D = target_sum - h.sum()

        i = np.random.randint(0, n_points)
        if D > 0:
            max_add = 1 - h[i]
            if max_add < 1e-10:
                continue
            add = min(D, max_add)
            h[i] += add
            D -= add
        else:
            max_remove = h[i]
            if max_remove < 1e-10:
                continue
            remove = min(-D, max_remove)
            h[i] -= remove
            D += remove
