import numpy as np
import time

def compute_c5(h, dx):
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv * dx)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # PSO parameters
    n_particles = 20
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_velocity = 0.1

    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []

    # Initialize particles
    for _ in range(n_particles):
        h = initial_h.copy()
        h += np.random.normal(0, 0.01, size=n_points)
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if current_sum != 0:
            h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
        particles.append(h)
        velocities.append(np.zeros(n_points))
        pbest_positions.append(h.copy())
        pbest_values.append(compute_c5(h, dx))

    gbest_idx = np.argmin(pbest_values)
    gbest_value = pbest_values[gbest_idx]
    gbest_pos = pbest_positions[gbest_idx].copy()

    while time.time() - start_time < budget_s:

            vel = velocities[i]
            pbest = pbest_positions[i]
