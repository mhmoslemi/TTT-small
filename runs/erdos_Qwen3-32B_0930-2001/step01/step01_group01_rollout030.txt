import numpy as np
from scipy.optimize import minimize
import time

def run():
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        result = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(result)

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    constraints = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    options = {'maxiter': 100, 'ftol': 1e-8, 'disp': False}

    best_h = h_initial.copy()
    best_c5 = objective(best_h)

    def perturb_h(h_base):
        h = h_base.copy()
        n_perturb = max(1, int(0.3 * len(h)))
        idxs = np.random.choice(len(h), n_perturb, replace=False)
        perturb = np.random.choice([-0.2, 0.2], size=n_perturb)
        h_perturbed = h.copy()
        h_perturbed[idxs] += perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        scale = (n_points / 2) / h_perturbed.sum()
        h_perturbed = np.clip(h_perturbed * scale, 0, 1)
        return h_perturbed

    np.random.seed(42)
    max_iter = 80
    for i in range(max_iter):
        if time.time() - start_time >= 1100:
            break
        h_perturbed = perturb_h(h_initial)
        res = minimize(
            objective,
            h_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options=options
        )
        if res.success:
            h_opt = res.x.reshape(n_points)
            c5 = objective(h_opt)
            if c5 < best_c5:
                best_h, best_c5 = h_opt, c5

    return (best_h, best_c5, n_points)
