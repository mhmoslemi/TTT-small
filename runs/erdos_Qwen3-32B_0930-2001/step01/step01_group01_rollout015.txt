import numpy as np
import time

def run():
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project_to_constraints(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return h_clipped
        scale = target_sum / current_sum
        h_scaled = h_clipped * scale
        h_clipped_scaled = np.clip(h_scaled, 0.0, 1.0)
        current_sum_2 = h_clipped_scaled.sum()
        if abs(current_sum_2 - target_sum) < 1e-8:
            return h_clipped_scaled
        free_indices = np.where((h_clipped_scaled > 0) & (h_clipped_scaled < 1))[0]
        if len(free_indices) == 0:
            return h_clipped_scaled
        delta = target_sum - current_sum_2
        adjustment = delta / len(free_indices)
        h_clipped_scaled[free_indices] += adjustment
        return np.clip(h_clipped_scaled, 0.0, 1.0)

    def evaluate(h):
        return np.max(np.correlate(h, 1 - h, mode='full') * dx)

    # Initialize population
    population = [h_initial.copy()]
    for _ in range(19):
        perturbed = h_initial.copy() + 0.05 * np.random.randn(n_points)
        perturbed = project_to_constraints(perturbed)
        population.append(perturbed)

    best_c5 = evaluate(h_initial)
    best_h = h_initial.copy()

    while time.time() - start_time < 1100:
        # Precompute scores for tournament selection
        population_scores = [evaluate(h) for h in population]

        # Generate new individuals
        new_individuals = []
        for _ in range(20):
            # Tournament selection with precomputed scores
            candidates1 = np.random.choice(len(population), 2, replace=False)
            scores1 = [population_scores[i] for i in candidates1]
            parent1 = population[np.argmin(scores1)]

            candidates2 = np.random.choice(len(population), 2, replace=False)
            scores2 = [population_scores[i] for i in candidates2]
            parent2 = population[np.argmin(scores2)]

            # Crossover and mutation
            offspring = (parent1 + parent2) / 2.0
            if np.random.rand() < 0.1:
                offspring += 0.1 * np.random.randn(n_points)
            offspring = project_to_constraints(offspring)
            new_individuals.append(offspring)

        # Evaluate new individuals
        new_scores = [evaluate(h) for h in new_individuals]

        # Combine and select top individuals
        combined = population + new_individuals
        combined_scores = population_scores + new_scores
        sorted_indices = np.argsort(combined_scores)
        population = [combined[i] for i in sorted_indices[:len(population)]]

        # Update best solution
        current_best_idx = np.argmin(combined_scores[:len(population)])
        current_c5 = combined_scores[current_best_idx]
        if current_c5 < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = current_c5

    return (best_h.tolist(), best_c5, n_points)
