from scipy.optimize import minimize
import time

def run():
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        result = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(result)

        current_sum = h_clipped.sum()

            return h_clipped
        scale = target_sum / current_sum
        h_scaled = h_clipped * scale
        h_projected = np.clip(h_scaled, 0, 1)
        return h_projected

    best_h = h_initial.copy()
    best_c5 = objective(best_h)

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': lambda h: h.sum() - target_sum})
    options = {'maxiter': 500, 'ftol': 1e-8, 'disp': False}

    np.random.seed(42)

    for _ in range(50):
        if time.time() - start_time >= 1100:
            break

        h_pert = best_h.copy()
        indices = np.random.choice(n_points, int(n_points * 0.5), replace=False)
        for idx in indices:
            if np.random.rand() < 0.5:

                h_pert[idx] = min(1, h_pert[idx] + 0.2)
        h_pert = project(h_pert)
        # Optimization step
        res = minimize(
            objective,
            h_pert,
            method='SLSQP',
