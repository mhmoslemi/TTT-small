                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                    simplex[i] = project(simplex[i], n_points)
    
    best_h = project(best_h, n_points)
    final_c5 = evaluate_individual(best_h, dx, best_c5)
    return (best_h, final_c5, n_points)
