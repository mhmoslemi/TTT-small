import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    n_particles = 20

    # Initialize particles
    positions = np.random.normal(loc=initial_h, scale=0.05, size=(n_particles, n_points))
    positions = np.clip(positions, 0, 1)
    sum_h = positions.sum(axis=1, keepdims=True)
    target_sum = n_points / 2.0
    positions *= target_sum / sum_h
    positions = np.clip(positions, 0, 1)
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    personal_best_positions = positions.copy()
    personal_best_values = np.array([compute_c5(p) for p in positions])
    global_best_idx = personal_best_values.argmin()
    global_best_position = personal_best_positions[global_best_idx].copy()
    best_c5 = personal_best_values[global_best_idx]
    best_h = global_best_position.copy()

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        dx = 2.0 / n_points
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    velocity_clipping = 0.1
    no_improve_steps = 0
    patience = 150

    while time.time() - start_time < budget_s:
        for i in range(n_particles):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia * velocities[i] +
                             cognitive * r1 * (personal_best_positions[i] - positions[i]) +
                             social * r2 * (global_best_position - positions[i]))
            velocities[i] = np.clip(velocities[i], -velocity_clipping, velocity_clipping)
            positions[i] += velocities[i]
            positions[i] = np.clip(positions[i], 0, 1)
            sum_h = positions[i].sum()
            if not np.isclose(sum_h, target_sum):
                positions[i] *= (target_sum / sum_h)
                positions[i] = np.clip(positions[i], 0, 1)

        current_c5 = np.array([compute_c5(p) for p in positions])
        improved_mask = current_c5 < personal_best_values
        personal_best_values[improved_mask] = current_c5[improved_mask]
        personal_best_positions[improved_mask] = positions[improved_mask].copy()
        global_best_idx = personal_best_values.argmin()
        new_c5 = personal_best_values[global_best_idx]
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = personal_best_positions[global_best_idx].copy()
            no_improve_steps = 0
        else:
            no_improve_steps += 1

            idxs = np.random.choice(n_points, size=(n_particles, 3), replace=True)

            velocities[:] = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
            no_improve_steps = 0

    return best_h, best_c5, n_points
