    start_time = time.time()
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def compute_c5(h_vec):
        ones_minus_h = 1.0 - h_vec
        corr = np.correlate(h_vec, ones_minus_h, mode='full')
        return np.max(corr * dx)

    def project(particle):
        particle_clipped = np.clip(particle, 0.0, 1.0)
        desired_sum = n_points / 2.0
        s = np.sum(particle_clipped)
        if s == 0:
            return np.full_like(particle_clipped, 0.5)
        scale_factor = desired_sum / s
        scaled = particle_clipped * scale_factor
        return np.clip(scaled, 0.0, 1.0)

    swarm_size = 20
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_velocity = 0.1
    positions = np.zeros((swarm_size, n_points))
    velocities = np.zeros((swarm_size, n_points))
    personal_best = np.zeros((swarm_size, n_points))
    personal_best_c5 = np.full(swarm_size, np.inf)

    # Initialize first particle
    positions[0] = initial_h.copy()
    velocities[0] = 0.01 * np.random.randn(n_points)
    positions[0] = project(positions[0])
    current_c5 = compute_c5(positions[0])
    personal_best[0] = positions[0].copy()
    personal_best_c5[0] = current_c5

    # Initialize other particles
    for i in range(1, swarm_size):
        perturbation = np.random.uniform(-0.05, 0.05, n_points)
        new_particle = initial_h.copy() + perturbation
        new_particle = project(new_particle)
        positions[i] = new_particle.copy()
        velocities[i] = 0.01 * np.random.randn(n_points)
        current_c5 = compute_c5(new_particle)
        personal_best[i] = new_particle.copy()
        personal_best_c5[i] = current_c5

    global_best_idx = np.argmin(personal_best_c5)
    global_best = personal_best[global_best_idx].copy()
    global_best_c5 = personal_best_c5[global_best_idx]
    best_h = global_best.copy()
    best_c5 = global_best_c5
    last_improvement_time = start_time

    while time.time() - start_time < budget_s:
        for i in range(swarm_size):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (
                inertia * velocities[i]
                + cognitive * r1 * (personal_best[i] - positions[i])
                + social * r2 * (global_best - positions[i])
            )
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            new_position = positions[i] + velocities[i]
            new_position = project(new_position)
            positions[i] = new_position
            current_c5 = compute_c5(new_position)
            if current_c5 < personal_best_c5[i]:
                personal_best[i] = new_position.copy()
                personal_best_c5[i] = current_c5
                if current_c5 < global_best_c5:
                    global_best = new_position.copy()
                    global_best_c5 = current_c5
                    best_h[:] = new_position.copy()
                    best_c5 = current_c5
                    last_improvement_time = time.time()

        if time.time() - last_improvement_time > 60 and time.time() - start_time + 60 < budget_s:
            for i in range(swarm_size // 2):
                perturbation = np.random.uniform(-0.05, 0.05, n_points)
                new_particle = global_best + perturbation
                new_particle = project(new_particle)
                positions[i] = new_particle.copy()
                current_c5 = compute_c5(new_particle)
                if current_c5 < personal_best_c5[i]:
                    personal_best[i] = new_particle.copy()
                    personal_best_c5[i] = current_c5
                    if current_c5 < global_best_c5:
                        global_best = new_particle.copy()
                        global_best_c5 = current_c5
                        best_h[:] = new_particle.copy()
                        best_c5 = current_c5
                        last_improvement_time = time.time()

    return best_h, best_c5, n_points
