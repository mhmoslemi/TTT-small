        perturbed = np.clip(perturbed, 0.0, 1.0)
        return perturbed

    # Initial optimization
    result = minimize(
        objective_function,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options=options
    )
    if result.success:
        h_candidate = result.x.reshape(n_points)
        c5 = objective_function(h_candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_solution = h_candidate

    # Perturbed restarts
    restarts = 50
    for _ in range(restarts):
        if time.time() - start_time >= 1100:
            break
        try:
            h_start = generate_perturbation(h_initial)
            result = minimize(
                objective_function,
                h_start,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options=options
            )
            if result.success:
                h_candidate = result.x.reshape(n_points)
                c5 = objective_function(h_candidate)
                if c5 < best_c5:
                    best_c5 = c5
                    best_solution = h_candidate
        except:
            continue

    return (best_solution, best_c5, n_points)
