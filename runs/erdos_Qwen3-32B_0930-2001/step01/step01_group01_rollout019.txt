```python
import numpy as np
from scipy.optimize import minimize
import time

def run():
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        result = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(result)
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    options = {'maxiter': 500, 'ftol': 1e-8, 'disp': False}
    
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    
    np.random.seed(42)
    perturbations = 30
    for perturb_idx in range(perturbations):
        if time.time() - start_time >= 1100:
            break
        
        # Generate perturbed initial guess
        h_perturbed = best_h.copy()
        perturb_indices = np.random.choice(n_points, size=int(n_points * 0.3), replace=False)
        noise = 0.2 * (2 * np.random.rand(len(perturb_indices)) - 1)
        h_perturbed[perturb_indices] += noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Adjust sum constraint
        current_sum = np.sum(h_perturbed)
        if not np.isclose(current_sum, target_sum):
            scale = target_sum / current_sum
            h_perturbed = np.clip(h_perturbed * scale, 0, 1)
        
        # Optimize from perturbed guess
        result = minimize(
            objective,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options=options
        )
        if result.success:
            h_opt = result.x.reshape(n_points)
            current_c5 = objective(h_opt)
            if current_c5 < best_c5:
                best_h = h_opt
                best_c5 = current_c5
    
    return (best_h, best_c5, n_points)
```
