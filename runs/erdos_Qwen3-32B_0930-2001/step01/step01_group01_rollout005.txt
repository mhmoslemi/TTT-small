import numpy as np
from scipy.optimize import minimize
import time

def run():
    start_time = time.time()
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        result = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(result)

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    options = {'maxiter': 300, 'ftol': 1e-6, 'disp': False}

    best_c5 = objective(h)
    best_h = h.copy()

    np.random.seed(42)
    max_global_iter = 40
    for _ in range(max_global_iter):
        if time.time() - start_time >= 1100:
            break
        # Perturbation
        h_pert = h.copy()
        indices = np.random.choice(n_points, size=int(n_points * 0.5), replace=False)
        mask = np.random.rand(len(indices)) > 0.5
        h_pert[indices] += np.where(mask, 0.15, -0.15)
        h_pert = np.clip(h_pert, 0, 1)
        # Adjust sum constraint
        s = h_pert.sum()
        if s > 1e-6:
            h_pert = h_pert * (n_points / 2) / s
            h_pert = np.clip(h_pert, 0, 1)
        # Optimize
        result = minimize(
            objective,
            h_pert,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options=options
        )
        if result.success:
            h_opt = result.x.reshape(n_points)
            current_c5 = objective(h_opt)
            if current_c5 < best_c5:
                best_h[:] = h_opt
                best_c5 = current_c5

    return (best_h, best_c5, n_points)
