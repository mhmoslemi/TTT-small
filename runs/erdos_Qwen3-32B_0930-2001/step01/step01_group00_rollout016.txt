import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        h[0] = h[0]  # Workaround for JAX-like JIT issues in some environments
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)
    
    def perturb_and_adjust(parent):
        mut_idx = np.random.choice(n_points, size=min(8, n_points), replace=False)
        mut_magnitude = np.random.uniform(-0.05 / np.sqrt(2), 0.05 / np.sqrt(2), mut_idx.size)
        child = parent.copy()
        child[mut_idx] += mut_magnitude
        child = np.clip(child, 0, 1)
        delta = (target_sum - child.sum()) / n_points
        child += delta
        return np.clip(child, 0, 1)
    
    def crossover(p1, p2):
        alpha = np.random.uniform(0.4, 0.6)
        child = alpha * p1 + (1 - alpha) * p2
        child = np.clip(child, 0, 1)
        offset = (target_sum - child.sum()) / n_points
        child += offset
        return np.clip(child, 0, 1)
    
    population = [initial_h.copy()]
    for _ in range(14):
        population.append(perturb_and_adjust(population[0]))
    
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    max_gen = 1000
    patience = 200
    no_improvement = 0
    
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate population
        evaluated = []
        for h_vec in population:
            c5 = compute_c5(h_vec)
            evaluated.append((c5, h_vec))
        
        evaluated.sort()
        current_c5, current_h = evaluated[0]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= patience:
            population = [best_h.copy()]
            for _ in range(14):
                population.append(perturb_and_adjust(population[0]))
            no_improvement = 0
            
        # Select elites
        survivors = evaluated[:min(8, len(evaluated))]
        survivors = [h for _, h in survivors]

            new_pop.append(child)
        
        population = new_pop[:15]
    
    h_values = best_h
    c5_bound = compute_c5(h_values)
