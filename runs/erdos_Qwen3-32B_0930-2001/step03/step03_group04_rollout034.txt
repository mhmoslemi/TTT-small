def run():

    n_points = len(h)
    dx = 2.0 / n_points
    T = n_points / 2.0

    # Initialize Adam parameters
    learning_rate = 1e-3
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0

    start_time = time.time()

    while time.time() - start_time < 1100:
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(cross_corr)
        # Find indices where cross_corr is max_overlap
        max_indices = np.where(np.abs(cross_corr - max_overlap) < 1e-10)[0]
        grad = np.zeros(n_points)
        for m_idx in max_indices:
            # Convert m_idx to shift_k
            shift_k = m_idx - (n_points - 1)
            # Compute gradient contributions for this shift_k
            if shift_k == 0:
                continue
            elif shift_k > 0:
                # i ranges from 0 to (n-1 - shift_k)
                for i in range(0, n_points - shift_k):
                    j = i + shift_k
                    grad[i] += (1 - h[j]) * dx
                    grad[j] -= h[i] * dx
            else:
                shift_k_abs = -shift_k
                # i ranges from shift_k_abs to n-1
                for i in range(shift_k_abs, n_points):
                    j = i - shift_k_abs
                    grad[i] += (1 - h[j]) * dx
                    grad[j] -= h[i] * dx

        # Adam update: compute m and v
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = learning_rate * m_hat / (np.sqrt(v_hat) + eps)

        # Update h
        h -= step

        # Clip to [0, 1]
        h = np.clip(h, 0, 1)

        # Project back to sum constraint
        current_sum = np.sum(h)
        if abs(current_sum - T) > 1e-6 and current_sum != 0:
            h *= T / current_sum

    return (h, max_overlap, n_points)
