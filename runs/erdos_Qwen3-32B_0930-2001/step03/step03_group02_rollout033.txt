        )

        if result.success:
            candidate = result.x
            c5_val = compute_c5(candidate)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = candidate.copy()

    # Final refinement
    if time.time() - start_time < budget_s - 30:
        result_final = optimize.minimize(
            objective,
            best_h,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 500, 'ftol': 1e-11}
        )
        if result_final.success:
            final_c5 = compute_c5(result_final.x)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = result_final.x.copy()

    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Final C5: {best_c5:.6f}")
    return best_h, best_c5, n_points
