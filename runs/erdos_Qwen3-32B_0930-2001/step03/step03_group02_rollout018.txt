import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    def project(h_unfeasible):
        h_clipped = np.clip(h_unfeasible, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            scaling_factor = 0.0
        else:
            scaling_factor = target_sum / current_sum
        h_scaled = h_clipped * scaling_factor
        return np.clip(h_scaled, 0, 1)

    # PSO Parameters
    n_particles = 30
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    particles = [h_initial + 0.1 * np.random.randn(n_points) for _ in range(n_particles)]
    for i in range(n_particles):
        particles[i] = project(particles[i])
    velocities = [0.01 * np.random.randn(n_points) for _ in range(n_particles)]
    personal_best_positions = [p.copy() for p in particles]
    personal_best_scores = [compute_c5(p) for p in particles]
    global_best_idx = np.argmin(personal_best_scores)
    global_best_position = personal_best_positions[global_best_idx].copy()
    best_c5 = personal_best_scores[global_best_idx]

    # PSO Iterations
    max_iter = 0
    while time.time() - start_time < budget_s - 60:
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            cognitive = c1 * r1 * (personal_best_positions[i] - particles[i])
            social = c2 * r2 * (global_best_position - particles[i])
            velocities[i] = w * velocities[i] + cognitive + social
            particles[i] += velocities[i]

                personal_best_positions[i] = particles[i].copy()
                personal_best_scores[i] = current_score
                if current_score < best_c5:

                    best_c5 = current_score
