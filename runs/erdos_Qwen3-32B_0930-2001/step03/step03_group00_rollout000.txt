import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_current = initial_h_values.copy()
    n_points = len(h_current)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    def finalize_h(h):
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if current_sum == 0:
            h = np.full_like(h, target_sum / len(h))
        else:
            scale = target_sum / current_sum
            h = np.clip(h * scale, 0, 1)
        return h

    h_best = finalize_h(h_current)
    best_c5 = compute_c5(h_best)
    
    while time.time() - start_time < budget_s - 50:
        if np.random.rand() < 0.5:  # Swap-based perturbation
            h_perturbed = h_best.copy()
            sorted_indices = np.argsort(h_perturbed)
            n_swap = max(2, n_points // 20)
            for _ in range(n_swap):
                top_idx = sorted_indices[-1 - _]
                bot_idx = sorted_indices[_]
                h_perturbed[top_idx], h_perturbed[bot_idx] = h_perturbed[bot_idx], h_perturbed[top_idx]
            h_pert = h_perturbed
        else:  # Noise perturbation
            pert = np.random.randn(n_points) * 0.15
            h_pert = np.clip(h_best + pert, 0, 1)
        h_pert = finalize_h(h_pert)
        
        # SLSQP optimization with constraints
        cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})
        result_slsqp = opt.minimize(
            compute_c5, h_pert, method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=cons,
            options={'maxiter': 300, 'ftol': 1e-8}
        )
        if result_slsqp.success:
            c5_candidate = compute_c5(result_slsqp.x)
            if c5_candidate < best_c5:

                best_c5 = c5_candidate
        
        if time.time() - start_time > budget_s - 50:

    elapsed = time.time() - start_time
