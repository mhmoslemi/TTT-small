            if np.random.rand() < 0.1:
                m = max(1, int(0.05 * n_points))  # 5% mutation rate
                ones = np.where(child == 1)[0]
                zeros = np.where(child == 0)[0]
                flip_mask = np.random.choice(ones, m, replace=False)
                flip_zero = np.random.choice(zeros, m, replace=False)
                child[flip_mask] = 0.0
                child[flip_zero] = 1.0
            offspring.append(child)
        # Add diversity through random initialization
        while len(offspring) < pop_size:
            h = np.zeros(n_points, dtype=np.float64)
            ones = np.random.choice(n_points, size=int(target_sum), replace=False)
            h[ones] = 1.0
            offspring.append(repair_binary(h))
        population = np.array(offspring)

        if time.time() - start_time > budget_s - 5:
            break

    return best_h, best_c5, n_points
