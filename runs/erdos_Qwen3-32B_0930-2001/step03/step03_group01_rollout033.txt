import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()
    
    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)
    
    h_best = h.copy()
    c5_best = compute_c5(h_best)

    # Define a perturbation strategy with controlled bit swaps
    def generate_initial_guess(current_h):
        zeros_idcs = np.where(current_h <= 0.5)[0]
        ones_idcs = np.where(current_h >= 0.5)[0]
        num_swaps = np.random.randint(5, 11)
        num_swaps = min(num_swaps, len(zeros_idcs), len(ones_idcs))
        if num_swaps == 0:
            return current_h.copy()
        # Randomly select and swap indices
        selected_zeros = np.random.choice(zeros_idcs, num_swaps, replace=False)
        selected_ones = np.random.choice(ones_idcs, num_swaps, replace=False)
        new_h = current_h.copy()
        new_h[selected_zeros] = 1.0 - new_h[selected_zeros]
        new_h[selected_ones] = 1.0 - new_h[selected_ones]
        return new_h
    
    # Add lightweight line search to escape local basins
    def filter_improvements(x_new, x_old):
        return optimize.minimize(
            compute_c5,
            x0=x_new,
            method='nelder-mead',
            bounds=[(0, 1)] * n_points,
            options={'maxiter': 10}
        ).x
    
    trial_num = 0
    while time.time() - start_time < budget_s - 10:
        trial_num += 1
        # Generate new initial guess with bit swaps
        h0 = generate_initial_guess(h_best)
        h0 = filter_improvements(h0, h_best)
        
        # Constrained optimization with tight tolerances
        result = optimize.minimize(
            compute_c5,
            x0=h0,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 150, 'ftol': 1e-11, 'disp': False}
        )
        
        if result.success:
            c5_candidate = result.fun
            h_candidate = result.x
            if c5_candidate < c5_best:

                c5_best = c5_candidate

                start_time = time.time()
                budget_s = 1100
