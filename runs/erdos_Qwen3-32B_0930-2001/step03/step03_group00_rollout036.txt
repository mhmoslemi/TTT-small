import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    def finalize_h(h):
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if current_sum == 0:
            h = np.full_like(h, target_sum / n_points)
        else:
            scale = target_sum / current_sum
            h = np.clip(h * scale, 0, 1)
        return h

    # PSO Parameters
    num_particles = 50
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    velocity_clip = 0.5

    # Initialize Swarm
    particles = []
    velocities = []
    pbest_positions = []
    pbest_scores = []

    for _ in range(num_particles):
        pert = np.random.randn(n_points) * 0.1
        pos = finalize_h(h_initial + pert)
        particles.append(pos)
        velocities.append(np.random.randn(n_points) * 0.05)
        pbest_positions.append(pos)
        pbest_scores.append(compute_c5(pos))

    # Set global best
    gbest_indices = np.argsort(pbest_scores)
    gbest_score = pbest_scores[gbest_indices[0]]
    gbest_position = particles[gbest_indices[0]].copy()

    iteration = 0
    while time.time() - start_time < budget_s - 50:
        for i in range(num_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            cognitive = c1 * r1 * (pbest_positions[i] - particles[i])
            social = c2 * r2 * (gbest_position - particles[i])
            velocities[i] = w * velocities[i] + cognitive + social
            velocities[i] = np.clip(velocities[i], -velocity_clip, velocity_clip)

            particles[i] += velocities[i]
            particles[i] = finalize_h(particles[i])

            current_score = compute_c5(particles[i])
            if current_score < pbest_scores[i]:
                pbest_scores[i] = current_score
                pbest_positions[i] = particles[i].copy()

        # Update global best
        current_gbest_idx = np.argmin(pbest_scores)
        if pbest_scores[current_gbest_idx] < gbest_score:
            gbest_score = pbest_scores[current_gbest_idx]
            gbest_position = pbest_positions[current_gbest_idx].copy()

        # SLSQP refinement every 20 iterations
        if iteration % 20 == 0:
            indices = np.argsort(pbest_scores)[:5]
            for idx in indices:
                x0 = particles[idx].copy()
                cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})
                result = opt.minimize(
                    compute_c5,
                    x0,
                    method='SLSQP',
                    bounds=[(0, 1)]*n_points,
                    constraints=cons,
                    options={'maxiter': 200, 'ftol': 1e-8}
                )
                if result.success:
                    score = compute_c5(result.x)
                    if score < gbest_score:
                        gbest_score = score
                        gbest_position = result.x.copy()

        iteration += 1

    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Best C5: {gbest_score:.6f}")

    return gbest_position, gbest_score, n_points
