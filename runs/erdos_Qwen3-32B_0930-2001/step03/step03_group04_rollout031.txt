        h = h * T / sum_h

    # Initialize Adam parameters
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 1e-3  # Initial learning rate
    t = 0
    start_time = time.time()
    decay_steps = 100
    decay_rate = 0.9

    while time.time() - start_time < 1100:
        t += 1
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        max_val = np.max(cross_corr)
        # Find indices where value is max_val (with tolerance)
        indices = np.where(np.isclose(cross_corr, max_val, atol=1e-6))[0]
        # Compute the gradient
        grad = np.zeros(n)
        for m_idx in indices:
            k_shift = m_idx - (n - 1)
            for i in range(n):

                    grad[i] += (1 - h[j]) * dx
                    grad[j] += -h[i] * dx
                else:
                    grad[i] += 1.0 * dx
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
