                    gbest_score = new_score
        iteration += 1

    def constraint_sum(x):
        return np.sum(x) - target_sum
    bounds = [(0.0, 1.0) for _ in range(n)]
    result = optimize.minimize(
        evaluate_erdos_solution,
        x0=gbest_position,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 60, 'ftol': 1e-12}
    )
    if result.success:
        best_h = result.x
        final_c5 = evaluate_erdos_solution(best_h)
    else:
        best_h = gbest_position
        final_c5 = gbest_score

    return best_h, final_c5, n
