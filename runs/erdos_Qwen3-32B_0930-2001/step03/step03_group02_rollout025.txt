            child1 = np.concatenate([parents[0][:cp1], parents[1][cp1:cp2], parents[0][cp2:]])
            child2 = np.concatenate([parents[1][:cp1], parents[0][cp1:cp2], parents[1][cp2:]])

        mutated = []
        for child in (child1, child2):
            n_mut = int(n_points * mutation_rate)
            mut_idx = np.random.choice(n_points, n_mut, replace=False)
            child_mut = child.copy()
            noise = np.random.normal(0, mutation_sigma, n_mut)
            child_mut[mut_idx] = np.clip(child_mut[mut_idx] + noise, 0, 1)
            sum_mut = child_mut.sum()
            if sum_mut != 0:
                scale = target_sum / sum_mut
                child_mut = np.clip(child_mut * scale, 0, 1)
            else:
                child_mut[:] = 0.5
            mutated.append(child_mut)

        new_pop = []
        n_elite = int(population_size * elitism_rate)
        elite_idx = np.argsort(fitness_scores)[:n_elite]
        for i in elite_idx:
            new_pop.append(population[i])
        new_pop.extend(mutated)
        while len(new_pop) < population_size:
            new_pop.append(create_individual())
        population = new_pop

        new_scores = [compute_c5(ind) for ind in population]
        current_best_idx = np.argmin(new_scores)
        current_best = new_scores[current_best_idx]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[current_best_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        fitness_scores = new_scores

    if time.time() - start_time < budget_s - 5:
        bounds = [(0, 1)] * n_points
        constraints = {'type': 'eq', 'fun': lambda x: x.sum() - target_sum}
        res = optimize.minimize(compute_c5, best_h, method='SLSQP', bounds=bounds, constraints=constraints,
                               options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            refined = res.x
            refined_c5 = compute_c5(refined)
            if refined_c5 < best_c5:
                best_h, best_c5 = refined, refined_c5

    return best_h, best_c5, n_points
