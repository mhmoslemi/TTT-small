import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    target_sum = n_points / 2.0  # Constraint sum(h) == target_sum
    dx = 2.0 / n_points

    population_size = 50
    elite_size = int(population_size * 0.05)
    mutation_rate = 0.1
    mutation_strength = 0.05
    tournament_size = 5

    # Initialize population with h_initial and randomized individuals
    population = [h_initial.copy()]
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h *= target_sum / h.sum()
        h = np.clip(h, 0, 1)
        population.append(h)
    population = np.array(population)

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    best_c5 = np.inf
    best_h = None

    while time.time() - start_time < budget_s - 50:
        # Evaluate current population
        fitness = np.array([compute_c5(ind) for ind in population])
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()

        # Selection and create new population with elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = population[elite_indices]
        new_population = list(elites)

        while len(new_population) < population_size:
            # Tournament selection for two parents
            parent1 = _tournament_selection(population, fitness, tournament_size)
            parent2 = _tournament_selection(population, fitness, tournament_size)

            # Crossover (one-point or two-point)
            child = _crossover(parent1, parent2)
            # Mutation
            child = _mutate(child, mutation_rate, mutation_strength, target_sum)
            new_population.append(child)

        population = np.array(new_population)

    # Final refinement with SLSQP
    if time.time() - start_time < budget_s and best_h is not None:
        bounds_slsqp = [(0, 1)] * n_points
        constraints_slsqp = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = optimize.minimize(compute_c5, best_h, method='SLSQP', bounds=bounds_slsqp,
                                constraints=constraints_slsqp, options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            refinement_c5 = compute_c5(res.x)
            if refinement_c5 < best_c5:
                best_c5 = refinement_c5
                best_h = res.x

    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Best C5: {best_c5:.6f}")
    return best_h, best_c5, n_points

def _tournament_selection(population, fitness, tournament_size):
    indices = np.random.choice(len(population), tournament_size)
    winner_idx = indices[np.argmin(fitness[indices])]
    return population[winner_idx].copy()

def _crossover(parent1, parent2):
    if np.random.rand() < 0.8:
        if np.random.rand() < 0.6:
            # One-point crossover
            pt = np.random.randint(1, len(parent1))
            return np.concatenate([parent1[:pt], parent2[pt:]])
        else:
            # Two-point crossover
            pts = sorted(np.random.choice(len(parent1), 2, replace=False))
            return np.concatenate([parent1[:pts[0]], parent2[pts[0]:pts[1]], parent1[pts[1]:]])
    return parent1.copy()

def _mutate(individual, mutation_rate, mutation_strength, target_sum):
    mut_mask = np.random.rand(len(individual)) < mutation_rate
    individual[mut_mask] += np.random.randn(np.sum(mut_mask)) * mutation_strength
    individual = np.clip(individual, 0, 1)
    current_sum = individual.sum()
    if current_sum != 0:
        individual *= (target_sum / current_sum)
        individual = np.clip(individual, 0, 1)
    return individual
