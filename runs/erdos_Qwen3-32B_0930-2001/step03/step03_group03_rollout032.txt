import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0
    start_time = time.time()

    h_best = h_initial.copy()

    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)

    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n)]

    while time.time() - start_time < budget_s - 10:
        # Compute progress to adjust perturbation scale
        elapsed = time.time() - start_time
        progress = min(1.0, elapsed / (budget_s - 10))
        # Start with 0.08, reduce to 0.03 linearly
        perturb_scale = 0.08 * (1 - progress) + 0.03 * progress
        # Start with 60, reduce to 10 linearly
        maxiter = 60 - int(50 * progress)
        maxiter = max(10, maxiter)

        h0 = h_best.copy()
        h0 += np.random.normal(0.0, perturb_scale, n)
        h0 = np.clip(h0, 0.0, 1.0)
        # Project to sum constraint while respecting bounds
        current_sum = np.sum(h0)
        adjustment = (target_sum - current_sum) / n
        h0 += adjustment
        h0 = np.clip(h0, 0.0, 1.0)

        result = optimize.minimize(
            compute_c5,
            x0=h0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-10}
        )
        if result.success:
            cand_c5 = result.fun
            if cand_c5 < c5_best:
                h_best[:] = result.x.copy()

    return h_best, c5_best, n
