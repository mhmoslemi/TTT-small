import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = h.copy()
    
    def compute_c5(h_vec):
        return np.max(np.correlate(h_vec, 1 - h_vec, mode='full') * dx)
    
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    T = 1.0  # Initial temperature
    cooling_rate = 0.9995
    perturbation_scale = 0.15
    
    def adjust_sum(h_vec):
        h_vec = np.clip(h_vec, 0.0, 1.0)
        adjustment = (target_sum - np.sum(h_vec)) / n
        h_vec += adjustment
        return np.clip(h_vec, 0.0, 1.0)
    
    # Simulated Annealing Loop
    while time.time() - start_time < budget_s - 30:
        # Generate candidate
        noise = np.random.normal(0.0, perturbation_scale, n)
        candidate = adjust_sum(h + noise)
        current_c5 = compute_c5(h)
        candidate_c5 = compute_c5(candidate)
        
        # Acceptance criterion
        delta_c5 = candidate_c5 - current_c5
        if delta_c5 < 0 or np.exp(-delta_c5 / max(T, 1e-8)) > np.random.rand():
            h = candidate
            current_c5 = candidate_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h[:] = h.copy()
        
        # Cool down
        T *= cooling_rate
        
    # Final refinement with SLSQP
    def objective(x_vec):
        return compute_c5(x_vec.reshape(n))
    
    def constraint_sum(x_vec):
        return np.sum(x_vec.reshape(n)) - target_sum
    
    result = optimize.minimize(
        objective,
        x0=best_h.reshape(-1),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 50, 'ftol': 1e-10}
    )
    
    if result.success:
        final_h = result.x.reshape(n)
        final_c5 = compute_c5(final_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h[:] = final_h[:]
    
    return best_h, best_c5, n
