import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points // 2
    assert np.sum(h) == target_sum, "Initial sum mismatch"
    dx = 2.0 / n_points
    start_time = time.time()

    def create_individual():
        indices = np.random.choice(n_points, size=target_sum, replace=False)
        ind = np.zeros(n_points, dtype=np.float32)
        ind[indices] = 1.0
        return ind

    pop_size = 50
    population = [create_individual() for _ in range(pop_size - 1)]
    population.append(initial_h_values.astype(np.float32).copy())

    def compute_c5(ind):
        conv = np.correlate(ind, 1.0 - ind, mode='full')
        return np.max(conv) * dx

    best_idx = np.argmin([compute_c5(p) for p in population])
    best_h = population[best_idx].copy()
    best_c5 = compute_c5(best_h)

    while time.time() - start_time < budget_s - 20:
        # Select parents
        parent1 = population[np.argmin([compute_c5(p) for p in np.random.choice(population, 2, replace=False)])]
        parent2 = population[np.argmin([compute_c5(p) for p in np.random.choice(population, 2, replace=False)])]

        # Crossover
        if np.random.rand() < 0.7:
            a, b = np.sort(np.random.choice(n_points, 2, replace=False))
            child1 = np.copy(parent1)
            child2 = np.copy(parent2)
            child1[a:b] = parent2[a:b]
            child2[a:b] = parent1[a:b]

            def fix_sum(ind):
                ind = ind.copy()
                s = np.sum(ind)
                delta = int(s - target_sum)
                if delta > 0:
                    idxs = np.random.choice(np.where(ind == 1.0)[0], delta, replace=False)
                    ind[idxs] = 0.0
                elif delta < 0:
                    idxs = np.random.choice(np.where(ind == 0.0)[0], -delta, replace=False)
                    ind[idxs] = 1.0
                return ind

            child1 = fix_sum(child1)
            child2 = fix_sum(child2)

            # Mutation
            def mutate(ind):
                ind = ind.copy()
                swaps = max(1, int(min(0.15 * n_points, n_points - 5)))
                for _ in range(swaps):
                    on = np.where(ind == 1.0)[0]
                    off = np.where(ind == 0.0)[0]
                    if len(on) == 0 or len(off) == 0:
                        break
                    i, j = np.random.choice(on), np.random.choice(off)
                    ind[i], ind[j] = 0.0, 1.0
                return ind

            child1 = mutate(child1)
            child2 = mutate(child2)
        else:
            child1 = mutate(np.copy(parent1))
            child2 = mutate(np.copy(parent2))

        # Evaluate children
        c51 = compute_c5(child1)
        c52 = compute_c5(child2)

        # Merge and sort population
        new_pop = population + [child1, child2]
        scores = [compute_c5(ind) for ind in new_pop]
        sorted_indices = np.argsort(scores)
        population = [new_pop[i] for i in sorted_indices[:pop_size]]

        # Update best
        current_scores = [compute_c5(ind) for ind in population]
        current_idx = np.argmin(current_scores)
        if current_scores[current_idx] < best_c5:
            best_h[:] = population[current_idx].copy()
            best_c5 = current_scores[current_idx]

    return best_h, min(compute_c5(best_h), best_c5), n_points
