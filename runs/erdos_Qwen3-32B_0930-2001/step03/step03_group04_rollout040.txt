import numpy as np
import time

def evaluate_objective(h, n_points):
    dx = 2.0 / n_points
    cross_corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(cross_corr)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2.0

    # Simulated Annealing parameters
    current_C = evaluate_objective(h, n_points)
    best_h = h.copy()
    best_C = current_C
    T = current_C * 2.0  # Initial temperature
    alpha = 0.995  # Cooling rate
    T_min = 1e-4

    current_h = h.copy()
    start_time = time.time()

    while time.time() - start_time < budget_s and T > T_min:
        # Generate a new candidate solution
        h_new = current_h.copy()
        perturbation_rate = 0.1  # 10% of positions to perturb
        mask = np.random.rand(n_points) < perturbation_rate
        delta = np.random.normal(0, T, n_points)
        h_new[mask] += delta[mask]
        h_new = np.clip(h_new, 0, 1)
        total = np.sum(h_new)
        if abs(total - target_sum) > 1e-10 and total != 0:
            h_new *= target_sum / total

        new_C = evaluate_objective(h_new, n_points)
        delta_E = new_C - current_C

        # Acceptance criterion
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            current_h = h_new.copy()
            current_C = new_C
            if new_C < best_C:
                best_h = h_new.copy()
                best_C = new_C

        T *= alpha

    return (best_h, best_C, n_points)
