import numpy as np
import scipy.optimize as optimize
import time
import scipy.interpolate

def evaluate_c5(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(conv)

def enforce_sum_constraint(h, n):
    target_sum = n / 2.0
    current_sum = float(np.sum(h))
    adjustment = (target_sum - current_sum) / n
    h_adj = np.clip(h + adjustment, 0.0, 1.0)
    return h_adj

def tournament_selection(population, fitness, tournament_size=3, k=5):
    selected = []
    indices = np.random.randint(len(population), size=len(population))
    for _ in range(k):
        contestants = indices[np.random.choice(len(indices), tournament_size, replace=False)]
        best_idx = np.argmin(fitness[contestants])
        selected.append(population[contestants[best_idx]])
    return selected

def crossover(h1, h2):
    n = len(h1)
    split = np.random.randint(1, n-1)
    child1 = np.concatenate([h1[:split], h2[split:]])
    child2 = np.concatenate([h2[:split], h1[split:]])
    return child1, child2

def mutate(h):
    h_mut = h.copy()
    idx = np.random.randint(len(h))
    delta = np.random.uniform(-0.3, 0.3)
    h_mut[idx] = np.clip(h_mut[idx] + delta, 0.0, 1.0)
    if np.random.rand() < 0.2 and len(h) > 4:
        split = np.random.randint(1, len(h)-1)
        h_mut = np.concatenate([h_mut[:split], h_mut[split:]:])
    return h_mut

def upsample(h, factor):

    x_old = np.linspace(0, 2, n_old)
    f = scipy.interpolate.interp1d(x_old, h, kind='linear')
    x_new = np.linspace(0, 2, n_old * factor)
    return f(x_new)
