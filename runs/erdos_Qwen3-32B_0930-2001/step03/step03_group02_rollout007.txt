        noise = np.random.randn(n_points) * 0.1
        perturbed = particles[0] + noise
        particles[i] = project(perturbed)
    velocities = 0.01 * np.random.rand(n_particles, n_points)

    # Personal best and global best
    personal_best_positions = particles.copy()
    personal_best_scores = np.array([compute_c5(p) for p in particles])
    best_idx = np.argmin(personal_best_scores)
    global_best_position = personal_best_positions[best_idx]
    best_score = personal_best_scores[best_idx]

    # PSO parameters
    w = 0.729
    c1 = 1.49445
    c2 = 1.49445
    max_iterations = 500

    iteration = 0
    while time.time() - start_time < budget_s - 5:
        if iteration >= max_iterations:
            break
        for i in range(n_particles):
            # Update velocity
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (personal_best_positions[i] - particles[i]) +
                c2 * r2 * (global_best_position - particles[i])
            )
            # Update position and project
            particles[i] = project(particles[i] + velocities[i])
            # Evaluate
            score = compute_c5(particles[i])
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_positions[i] = particles[i].copy()
                personal_best_scores[i] = score
                if score < best_score:
                    global_best_position = particles[i].copy()
                    best_score = score
        iteration += 1

    # Final refinement with SLSQP
    bounds = [(0, 1)] * n_points
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    if time.time() - start_time < budget_s - 5:
        sol = optimize.minimize(
            compute_c5, global_best_position, method='SLSQP',
            bounds=bounds, constraints=constraints, options={'maxiter': 500, 'ftol': 1e-10}
        )
        if sol.success:
            final_h = sol.x
            final_score = compute_c5(final_h)
            if final_score < best_score:
                best_score = final_score
                global_best_position = final_h

    return global_best_position, best_score, n_points
