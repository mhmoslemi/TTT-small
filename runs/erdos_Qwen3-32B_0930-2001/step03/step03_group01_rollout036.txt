        population.append(generate_random_h(n_points))
    population = np.array(population, dtype=np.float32)
    
    best_h = population[0].copy()
    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Evaluate current population
        c5_list = np.array([compute_c5(ind, dx) for ind in population])
        
        # Update best found so far
        current_best_idx = np.argmin(c5_list)
        current_best_c5 = c5_list[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h[:] = population[current_best_idx].copy()
        
        # Tournament selection
        parents = []
        for _ in range(pop_size // 2):
            indices = np.random.choice(pop_size, t_size, replace=False)
            t_c5 = c5_list[indices]
            winner_idx = indices[np.argmin(t_c5)]
            parents.append(population[winner_idx])
        parents = np.array(parents)
        
        # Mutate to generate offspring
        offspring = []
        offspring_c5 = []
        for parent in parents:
            child = mutate(parent)
            child_c5 = compute_c5(child, dx)
            offspring.append(child)
            offspring_c5.append(child_c5)
        offspring = np.array(offspring)
        offspring_c5 = np.array(offspring_c5)
        
        # Merge and select new population
        combined = np.concatenate((population, offspring))
        combined_c5 = np.concatenate((c5_list, offspring_c5))
        sorted_indices = np.argsort(combined_c5)
        population = combined[sorted_indices[:pop_size]]
    
    return best_h, best_c5, n_points
