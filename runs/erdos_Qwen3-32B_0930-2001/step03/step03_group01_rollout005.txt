    else:
        population_size = 100
        population = []
        for _ in range(population_size):
            indices = np.random.permutation(n_points)[:int(target_sum)]
            new_h = np.zeros(n_points, dtype=float)
            new_h[indices] = 1.0
            population.append(new_h)
    population = np.array(population)

    def compute_c5(h_vec):
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(conv * dx)

    evaluation_scores = np.array([compute_c5(ind) for ind in population])
    best_idx = np.argmin(evaluation_scores)
    best_h = population[best_idx].copy()
    best_c5 = evaluation_scores[best_idx]

    while time.time() - start_time < budget_s - 10:
        tournament_size = 5
        tournament_indices = np.random.choice(population_size, tournament_size)
        tournament_scores = evaluation_scores[tournament_indices]
        best_tournament_idx = tournament_indices[np.argmin(tournament_scores)]
        parent = population[best_tournament_idx].copy()

        zero_indices = np.where(parent == 0)[0]
        one_indices = np.where(parent == 1)[0]

        if len(zero_indices) == 0 or len(one_indices) == 0:
            continue

        zero_idx = np.random.choice(zero_indices)
        one_idx = np.random.choice(one_indices)
        child = parent.copy()
        child[zero_idx] = 1.0
        child[one_idx] = 0.0

        child_score = compute_c5(child)
        worst_idx = np.argmax(evaluation_scores)

        if child_score < evaluation_scores[worst_idx]:
            population[worst_idx] = child
            evaluation_scores[worst_idx] = child_score
            if child_score < best_c5:
                best_h[:] = child.copy()
                best_c5 = child_score

    return best_h, best_c5, n_points
