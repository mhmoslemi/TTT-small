        conv = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(conv)

    def scale_and_clip(h_vec, target_sum):
        h_clipped = np.clip(h_vec, 0, 1.0)

            return h_clipped.copy()
        scale_factor = target_sum / current_sum

import scipy.optimize as optimize

        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(conv)

    def scale_and_clip(h_vec, target_sum):
        h_clipped = np.clip(h_vec, 0, 1.0)
        current_sum = np.sum(h_clipped)
        if current_sum == 0.0:
            return h_clipped.copy()
        scale_factor = target_sum / current_sum
        h_scaled = h_clipped * scale_factor
        h_scaled = np.clip(h_scaled, 0.0, 1.0)
        return h_scaled

    population_size = 50
    tournament_size = 5
    elitism_rate = 0.05
    mutation_rate = 0.1
    mutation_sigma = 0.05
    mutation_prob = 0.8
    max_generations = 100

    population = []

        if i == 0:
            candidate = h_initial.copy()
        else:
            noise = np.random.randn(n_points) * 0.1

        candidate = scale_and_clip(candidate, target_sum)
        population.append(candidate)
    population = np.array(population)

    best_c5 = float('inf')
    best_h = None

    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 5:

        c5_values = np.array([compute_c5(candidate) for candidate in population])
        current_min_idx = np.argmin(c5_values)
        current_min_c5 = c5_values[current_min_idx]
        if current_min_c5 < best_c5:
            best_c5 = current_min_c5
            best_h = population[current_min_idx].copy()

        num_elites = max(1, int(elitism_rate * population_size))

        remaining_slots = population_size - num_elites
