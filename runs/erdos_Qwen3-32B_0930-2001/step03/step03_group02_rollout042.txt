import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)

    perturbations = 70
    magnitudes = [0.05, 0.1, 0.2, 0.3]

    for mag in magnitudes:
        if time.time() - start_time > budget_s - 30:
            break
        for _ in range(perturbations):
            if time.time() - start_time > budget_s - 30:
                break
            pert = np.random.randn(n_points) * mag
            h_pert = np.clip(h_initial + pert, 0, 1)
            bounds = [(0, 1)] * n_points
            constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
            res = optimize.minimize(
                fun=compute_c5,
                x0=h_pert,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-8}
            )
            if res.success:
                current_c5 = compute_c5(res.x)
                if current_c5 < best_c5:
                    best_h[:] = res.x[:]
                    best_c5 = current_c5

    if time.time() - start_time < budget_s - 10:
        res_refine = optimize.minimize(
            fun=compute_c5,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 600, 'ftol': 1e-10}
        )
        if res_refine.success:
            current_c5 = compute_c5(res_refine.x)
            if current_c5 < best_c5:
                best_h[:] = res_refine.x[:]
                best_c5 = current_c5

    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Best C5: {best_c5:.8f}")
    return best_h, best_c5, n_points
