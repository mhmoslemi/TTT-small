import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()
    
    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)
    
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    
    current_sum = 21.5
    sum_step = 0.02
    
    while time.time() - start_time < budget_s - 10:
        # Define constraints and bounds
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - current_sum}]
        bounds = [(0., 1.) for _ in range(n_points)]
        
        result = optimize.minimize(
            compute_c5,
            x0=h_best,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'rhobeg': 0.5, 'tol': 1e-8}
        )
        
        if result.success:
            c5_candidate = result.fun
            h_candidate = result.x
            if c5_candidate < c5_best:
                h_best[:] = h_candidate.copy()
                c5_best = c5_candidate
            # Tighten the constraint
            current_sum = max(current_sum - sum_step, 21.0)
    
    return h_best, c5_best, n_points
