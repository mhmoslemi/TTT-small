def compute_gradient(h, k, dx):
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        contribution = 0.0
        # First part: when i is part of the first term h[i] * ...
        if 0 <= i + k < n:

        # Second part: when i is the i +k for another term h[j] * ...
        if 0 <= i - k < n:
            contribution -= h[i - k] * dx
        grad[i] = contribution
    return grad
