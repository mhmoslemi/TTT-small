import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_current = initial_h_values.copy()
    n_points = len(h_current)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    def finalize_h(h):
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if current_sum == 0:
            h = np.full_like(h, target_sum / len(h))
        else:
            scale = target_sum / current_sum
            h = np.clip(h * scale, 0, 1)
        return h

    h_best = finalize_h(h_current)
    best_c5 = compute_c5(h_best)
    
    while time.time() - start_time < budget_s - 50:
        # Perturbation using swaps maintaining integral constraint
        h_pert = h_best.copy()
        num_swaps = 10
        for _ in range(num_swaps):
            i1, i2 = np.random.choice(n_points, 2, replace=False)
            h_pert[i1], h_pert[i2] = h_pert[i2], h_pert[i1]
        h_pert = finalize_h(h_pert)
        
        # Nelder-Mead optimization with bounds
        result_nm = opt.minimize(compute_c5, h_pert, method='Nelder-Mead',
                                 bounds=[(0, 1)] * n_points,
                                 options={'maxiter': 100, 'xatol': 1e-4})
        if result_nm.success:
            h_nm = result_nm.x
        else:
            h_nm = h_pert
        h_nm = finalize_h(h_nm)
        
        # SLSQP optimization with sum constraint
        cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})
        result_slsqp = opt.minimize(compute_c5, h_nm, method='SLSQP',
                                     bounds=[(0, 1)] * n_points,
                                     constraints=cons,
                                     options={'maxiter': 200, 'ftol': 1e-8})
        if result_slsqp.success:
            c5_candidate = compute_c5(result_slsqp.x)
            if c5_candidate < best_c5:
                h_best = result_slsqp.x.copy()
                best_c5 = c5_candidate
        
        if time.time() - start_time > budget_s - 50:
            break
    
    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Best C5: {best_c5:.6f}")
    
    return h_best, best_c5, n_points
