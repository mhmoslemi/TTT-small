import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)

    # Generate diverse candidates with random binary blocks
    candidates = [h_initial]
    num_random_candidates = 50
    for _ in range(num_random_candidates):
        if time.time() - start_time > budget_s - 10:
            break
        on_count = int(np.round(target_sum))
        indices = np.sort(np.random.choice(n_points, size=on_count, replace=False))
        candidate = np.zeros(n_points)
        candidate[indices] = 1.0
        candidates.append(candidate)

    # Optimize each candidate with SLSQP
    for candidate in candidates:
        if time.time() - start_time > budget_s - 5:
            break
        res = optimize.minimize(
            compute_c5, 
            x0=candidate, 
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if res.success:
            current_c5 = compute_c5(res.x)
            if current_c5 < best_c5:
                best_h[:] = res.x
                best_c5 = current_c5

    # Final refinement if time remains
    if time.time() - start_time < budget_s - 5:
        res_refine = optimize.minimize(
            compute_c5, 
            x0=best_h, 
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 500, 'ftol': 1e-10}
        )
        if res_refine.success:
            refined_c5 = compute_c5(res_refine.x)
            if refined_c5 < best_c5:
                best_h[:] = res_refine.x
                best_c5 = refined_c5

    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Best C5: {best_c5:.6f}")
    return best_h, best_c5, n_points
