import numpy as np
import time
import scipy.optimize as optimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0
    start_time = time.time()

    # Parameters for PSO
    num_particles = 20
    w = 0.7  # inertia weight
    c1 = 1.5  # cognitive parameter
    c2 = 1.5  # social parameter
    max_iterations = 1000

    # Initialize particles
    particles = [h_initial.copy()]
    for _ in range(num_particles - 1):
        h = np.random.uniform(0, 1, n)
        sum_h = h.sum()
        delta = target_sum - sum_h
        h += delta / n
        h = np.clip(h, 0, 1)
        particles.append(h)
    particles = np.array(particles)

    # Initialize velocities
    velocities = np.random.uniform(-0.05, 0.05, size=(num_particles, n))

    # Initialize personal bests and global best
    pbest = particles.copy()
    pbest_scores = np.array([compute_c5(p) for p in particles])
    gbest_index = np.argmin(pbest_scores)
    gbest = particles[gbest_index].copy()
    gbest_score = pbest_scores[gbest_index]

    # Track iterations and last improvement
    iteration = 0
    last_improvement = 0

    def compute_c5_h(h_vec):
        h = h_vec
        ones_h = 1.0 - h
        convolution = np.correlate(h, ones_h, mode='full')
        max_conv = np.max(convolution) * dx
        return max_conv

    while time.time() - start_time < budget_s - 10 and iteration < max_iterations:
        for i in range(num_particles):
            # Update velocity
            r1 = np.random.rand(n)
            r2 = np.random.rand(n)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (pbest[i] - particles[i]) + 
                             c2 * r2 * (gbest - particles[i]))
            # Clip velocities to limit step size
            velocities[i] = np.clip(velocities[i], -0.1, 0.1)

            # Update position
            particles[i] += velocities[i]
            # Clip to [0,1]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            # Adjust sum to target_sum
            current_sum = particles[i].sum()
            delta = (target_sum - current_sum) / n
            particles[i] += delta
            # Clip again after adjustment
            particles[i] = np.clip(particles[i], 0.0, 1.0)

        # Compute scores for all particles
        scores = [compute_c5_h(p) for p in particles]
        for i in range(num_particles):
            if scores[i] < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = scores[i]
        current_gbest_index = np.argmin(pbest_scores)
        current_gbest_score = pbest_scores[current_gbest_index]
        if current_gbest_score < gbest_score:
            gbest = pbest[current_gbest_index].copy()
            gbest_score = current_gbest_score
            last_improvement = iteration

        # Adjust parameters if no improvement for 100 iterations
        if (iteration - last_improvement) > 100:
            w = 0.5  # reduce inertia
            velocities = np.random.uniform(-0.05, 0.05, size=(num_particles, n))

        # Local refinement every 500 iterations
        if iteration % 500 == 0 and iteration > 0:
            # Take top 5 particles and refine
            top_indices = np.argsort(pbest_scores)[:5]
            for idx in top_indices:
                h0 = pbest[idx].copy()
                # Constraint: sum(h) == target_sum
                def constraint_sum(x):
                    return np.sum(x) - target_sum
                bounds = [(0.0, 1.0) for _ in range(n)]
                result = optimize.minimize(
                    compute_c5_h,
                    x0=h0,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 60, 'ftol': 1e-10}
                )
                if result.success:
                    improved_h = result.x
                    improved_score = compute_c5_h(improved_h)
                    if improved_score < pbest_scores[idx]:
                        pbest[idx] = improved_h.copy()
                        pbest_scores[idx] = improved_score
                        current_gbest_index = np.argmin(pbest_scores)
                        current_gbest_score = pbest_scores[current_gbest_index]
                        if current_gbest_score < gbest_score:
                            gbest = pbest[current_gbest_index].copy()
                            gbest_score = current_gbest_score

        iteration += 1

    # Final local optimization on best solution
    if iteration < max_iterations:  # If time is left, do more
        h0 = gbest.copy()
        def constraint_sum(x):
            return np.sum(x) - target_sum
        bounds = [(0.0, 1.0) for _ in range(n)]
