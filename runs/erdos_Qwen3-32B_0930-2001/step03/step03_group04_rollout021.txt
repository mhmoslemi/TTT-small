import time

def evaluate_objective(h, n_points):

    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    decay_steps = 100
    decay_rate = 0.9
    start_time = time.time()

    best_h = h.copy()
    best_C5 = evaluate_objective(h, n_points)

    while time.time() - start_time < 1100:
        t += 1

        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        current_max = np.max(cross_corr)

        problematic_shifts = []
        for l in range(len(cross_corr)):
            if abs(cross_corr[l] - current_max) < 1e-8:
                k_shift = l - (n_points - 1)
                problematic_shifts.append(k_shift)
        problematic_shifts = list(set(problematic_shifts))

        grad = np.zeros_like(h)
        for k in problematic_shifts:
