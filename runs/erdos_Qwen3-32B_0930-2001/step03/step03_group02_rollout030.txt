import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # Constraint: sum(h) == target_sum

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)

    # Generate perturbations based on current best_h to refine local optima
    num_perturbations = 60  # Increased number of perturbations
    max_slsqp_iter = 400  # Allow more iterations per SLSQP run
    perturbation_magnitude = 0.15  # Adjusted magnitude for exploration

    for idx in range(num_perturbations):
        if time.time() - start_time > budget_s - 5:
            break
        # Use current best_h as base for perturbations
        h_base = best_h if idx > 0 else h_initial
        # Generate sum-preserving perturbation
        pert = np.random.randn(n_points)
        pert -= pert.mean()  # Zero mean => sum-preserving
        pert *= perturbation_magnitude / np.std(pert) if np.std(pert) != 0 else 1.0
        h_pert = h_base + pert
        # Ensure bounds and sum constraint
        h_pert = np.clip(h_pert, 0, 1)
        sum_pert = h_pert.sum()
        if sum_pert != 0:
            h_pert *= target_sum / sum_pert
            h_pert = np.clip(h_pert, 0, 1)
        # Optimize from perturbed starting point
        res_slsqp = optimize.minimize(
            compute_c5, 
            h_pert, 
            method='SLSQP', 
            bounds=[(0, 1) for _ in range(n_points)], 
            constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}],
            options={'maxiter': max_slsqp_iter, 'ftol': 1e-10}
        )
        if res_slsqp.success and not np.isnan(res_slsqp.fun):
            current_c5 = compute_c5(res_slsqp.x)
            if current_c5 < best_c5:
                best_h = res_slsqp.x.copy()
                best_c5 = current_c5

    # Final refinement with longer run if time allows
    if time.time() - start_time < budget_s - 10:
        refinement_iter = 800
        res_refine = optimize.minimize(
            compute_c5, 
            best_h, 
            method='SLSQP', 
            bounds=[(0, 1) for _ in range(n_points)], 
            constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}],
            options={'maxiter': refinement_iter, 'ftol': 1e-12}
        )
        if res_refine.success and not np.isnan(res_refine.fun):
            refine_c5 = compute_c5(res_refine.x)
            if refine_c5 < best_c5:
                best_h = res_refine.x.copy()
                best_c5 = refine_c5

    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Best C5: {best_c5:.8f}")
    return best_h, best_c5, n_points
