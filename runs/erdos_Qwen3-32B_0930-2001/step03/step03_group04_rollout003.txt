def run():

    h = h_initial.copy()

    def compute_overlaps(h):

        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        return cross_corr

    def compute_gradient(h, max_k_list):
        grads = []
        for k in max_k_list:
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    grad[i] += (1 - h[j]) * dx
                    grad[j] -= h[i] * dx
            grads.append(grad)
        if grads:
            return np.mean(grads, axis=0)
        else:
            return np.zeros(n_points)

    best_h = h.copy()
    best_value = np.max(compute_overlaps(h))
    start_time = time.time()
    max_time = 1100

    # Adam optimizer parameters
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    learning_rate = 1e-3
    t = 0

    while time.time() - start_time < max_time:
        # Compute overlaps and max
        overlaps = compute_overlaps(h)
        current_C = np.max(overlaps)
        max_k_list = np.where(overlaps == current_C)[0]

        # Compute gradient
        grad = compute_gradient(h, max_k_list)

        # Update t
        t += 1
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        update = -learning_rate * m_hat / (np.sqrt(v_hat) + eps)

        # Apply update
        h += update

        # Project onto [0,1] and scale sum to T
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:

        # Check if current solution is better
        overlaps_new = compute_overlaps(h)
