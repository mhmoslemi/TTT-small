        h_corrected = h * corrected_scale
        h_corrected = np.clip(h_corrected, 0, 1)
        conv = np.correlate(h_corrected, 1 - h_corrected, mode='full')
        c5 = np.max(conv * dx)
        return c5

    bounds_slsqp = [(0, 1) for _ in range(n_points)]
    constraints_slsqp = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)

    num_perturbations = 50
    perturbation_magnitude = 0.15

    # Improved sampling of initial perturbations and SQP refinement
    for _ in range(num_perturbations):
        if time.time() - start_time > budget_s - 30:
            break
        pert = np.random.randn(n_points) * perturbation_magnitude
        h_pert = np.clip(h_initial + pert, 0, 1)
        # Use Nelder-Mead for coarse optimization
        res_nelder = optimize.minimize(objective_nelder, h_pert, method='Nelder-Mead',
                                       options={'maxiter': 150, 'xatol': 1e-5, 'fatol': 1e-5})
        if not res_nelder.success:
            continue
        sol = res_nelder.x
        # Use SLSQP for refinement with tighter tolerances
        res_slsqp = optimize.minimize(compute_c5, sol, method='SLSQP',
                                     bounds=bounds_slsqp, constraints=constraints_slsqp,
                                     options={'maxiter': 500, 'ftol': 1e-10})
        if res_slsqp.success:
            current_c5 = compute_c5(res_slsqp.x)
            if current_c5 < best_c5:
                best_h = res_slsqp.x.copy()
                best_c5 = current_c5

    # Adaptive refinement: identify worst lag and constrain it in next optimization
    if time.time() - start_time < budget_s - 20:
        h = best_h.copy()
        h = h.reshape(n_points)
        h = h / h.sum() * target_sum  # enforce constraint
        h = np.clip(h, 0, 1)
        conv = np.correlate(h, 1 - h, mode='full')
        worst_lag_idx = np.argmax(conv)
        worst_value = conv[worst_lag_idx]
        dx_continuous = dx  # step size in continuous domain

        # Create explicit constraint for the worst lag
        def worst_lag_constraint(x):
            x = x.reshape(n_points)
            i_vals = np.arange(n_points)
            k = worst_lag_idx - n_points + 1  # convert full conv index to step offset
            if k < 0:
                i_start = -k
                i_end = n_points
            else:
                i_start = 0
                i_end = n_points - k
            if i_end <= i_start:
                return 0.0  # no overlap
            overlap_sum = np.sum(x[i_start:i_end] * (1 - x[i_start + k:i_end + k]))
            return overlap_sum * dx_continuous - worst_value

        constraints_custom = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                              {'type': 'ineq', 'fun': worst_lag_constraint}]

        res_custom = optimize.minimize(compute_c5, best_h, method='SLSQP',
                                      bounds=bounds_slsqp, constraints=constraints_custom,
                                      options={'maxiter': 500, 'ftol': 1e-10})
        if res_custom.success:
            new_c5 = compute_c5(res_custom.x)
            if new_c5 < best_c5:
                best_h = res_custom.x.copy()
                best_c5 = new_c5

    elapsed = time.time() - start_time
    print(f"Final C₅ bound found: {best_c5:.7f} (Elapsed: {elapsed:.1f}s)")
    return best_h, best_c5, n_points
