            w = max(0.4, w * 0.9)
            no_improvement = 0

        if time.time() - start_time >= budget_s - 10:
            break

    top_indices = np.argsort(pbest_scores)[:5]
    for idx in top_indices:
        if time.time() - start_time < budget_s - 10:

            def objective(x):
                return compute_c5(x)

            def constraint_sum(x):
                return float(np.sum(x) - target_sum)

            bounds = [(0, 1) for _ in range(n)]
            result = optimize.minimize(
                objective,
                x0=particles[idx],
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 60, 'ftol': 1e-10}
            )
            if result.success:
                cand_score = result.fun
                if cand_score < gbest_score:
                    gbest_score = cand_score
                    gbest_position = result.x

    gbest_position = np.clip(gbest_position, 0, 1)
    adjustment = (target_sum - np.sum(gbest_position)) / n
    gbest_position += adjustment
    gbest_position = np.clip(gbest_position, 0, 1)

    return gbest_position, gbest_score, n
