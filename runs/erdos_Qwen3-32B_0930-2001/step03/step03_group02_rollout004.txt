import time
import scipy.optimize as optimize

    tournament_size = 5

    crossover_rate = 0.8
    mutation_std = 0.05
    mutation_rate = 0.1
    
    population = [h_initial.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        population.append(h)
    
    def compute_c5(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        s = h_clipped.sum()
        if s == 0:
            scaled = 0.5 * np.ones(n_points)
        else:
            scaled = h_clipped * (target_sum / s)
        conv = np.correlate(scaled, 1 - scaled, mode='full')
        return np.max(conv * dx)
    
    def evaluate(pop):
        return np.array([compute_c5(h) for h in pop])
    
    fitness = evaluate(population)
    best_idx = np.argmin(fitness)
    best_h, best_c5 = population[best_idx].copy(), fitness[best_idx]
    
    while time.time() - start_time < budget_s - 5:
        parents = []
        for _ in range(2):
            indices = np.random.choice(population_size, tournament_size, replace=False)
            fitness_vals = [fitness[i] for i in indices]
            best_idx = indices[np.argmin(fitness_vals)]
            parents.append(population[best_idx].copy())
        
        p1, p2 = parents
        if np.random.rand() < crossover_rate:
            if np.random.rand() < 0.6:
                cp = np.random.randint(1, n_points - 1)

                cps = np.sort(np.random.randint(1, n_points - 1, size=2))
                cp1, cp2 = cps
                child1 = np.concatenate((p1[:cp1], p2[cp1:cp2], p1[cp2:]))
                child2 = np.concatenate((p2[:cp1], p1[cp1:cp2], p2[cp2:]))
        else:

            x_mut[indices] += np.random.normal(0, mutation_std, size=n_mut)

        child1, child2 = mutate(child1), mutate(child2)
