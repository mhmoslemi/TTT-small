        # Evaluate population
        new_candidates_c5 = []
        for candidate in population:
            c5 = compute_c5(candidate)
            new_candidates_c5.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h[:] = candidate.copy()
        # Update improvement tracking
        if min(new_candidates_c5) < best_c5_prev:
            last_improvement = 0
        else:
            last_improvement += 1
        best_c5_prev = best_c5

        # Generate offspring through tournament selection and mutation
        offspring = []
        for _ in range(population_size // 2):
            # Tournament selection
            contestants = np.random.choice(population_size, 5, replace=False)
            c5_values = [new_candidates_c5[i] for i in contestants]
            best_idx = contestants[np.argmin(c5_values)]
            parent = population[best_idx].copy()
            # Mutate with adaptive rate
            if np.random.rand() < 0.1:
                i, j = np.random.choice(n_points, 2, replace=False)
                parent[i], parent[j] = parent[j], parent[i]
            offspring.append(parent)

        # Sort and replace worst half
        sorted_indices = np.argsort(new_candidates_c5)
        new_population = [population[i] for i in sorted_indices[:population_size//2]]
        new_population.extend(offspring)
        population = new_population

        # Adjust mutation rate
        if last_improvement >= 20:
            pass  # Increase mutation through higher selection pressure
        # Mutation rate implicitly controlled by population diversity

    return best_h, best_c5, n_points
