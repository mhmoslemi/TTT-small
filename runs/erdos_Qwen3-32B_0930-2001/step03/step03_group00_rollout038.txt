import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_init = initial_h_values.copy()
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        h = finalize_h(h)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    def finalize_h(h):
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        scale = target_sum / current_sum
        h = np.clip(h * scale, 0, 1)
        return h
    
    h_best = finalize_h(h_init)
    best_c5 = compute_c5(h_best)
    
    bounds = [(0.0, 1.0)] * n_points
    
    # Global search with Differential Evolution
    if time.time() - start_time < budget_s - 300:
        de_result = opt.differential_evolution(
            lambda x: compute_c5(x), bounds,
            maxiter=50, popsize=10,
            strategy='best1bin', mutation=(0.5, 1.0),
            recombination=0.8, polish=False,
            disp=False, workers=1
        )

        current_de_c5 = compute_c5(h_de)
        if current_de_c5 < best_c5:
            best_c5 = current_de_c5
            h_best = h_de
    
    # Local refinement with SLSQP
    if time.time() - start_time < budget_s - 100:
        slsqp_result = opt.minimize(
            compute_c5, h_best, method='SLSQP',

        )
        if slsqp_result.success and slsqp_result.fun < best_c5:

    # Final local search with perturbation-based exploration

    while time.time() - start_time < budget_s - 20:
        iter_cnt += 1
        pert = np.random.normal(0, 0.1, n_points)
