import numpy as np

def repair_individual(individual, target_sum):
    current = individual.copy()
    s = np.sum(current)
    if s == target_sum:
        return current
    diff = target_sum - s
    if diff > 0:
        zeros_idx = np.where(current == 0)[0]
        np.random.shuffle(zeros_idx)
        for i in range(diff):
            if i < len(zeros_idx):
                current[zeros_idx[i]] = 1
    else:
        ones_idx = np.where(current == 1)[0]
        np.random.shuffle(ones_idx)
        for i in range(-diff):
            if i < len(ones_idx):
                current[ones_idx[i]] = 0
    return current

def mutate_individual(individual, mutation_rate, n_points):
    n_flips = int(np.ceil(mutation_rate * n_points))
    current = individual.copy()
    for _ in range(n_flips):
        ones_idx = np.where(current == 1)[0]
        zeros_idx = np.where(current == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        o = np.random.choice(ones_idx)
        z = np.random.choice(zeros_idx)

        current[z] = 1
