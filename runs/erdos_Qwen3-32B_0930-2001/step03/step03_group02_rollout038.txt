                best_c5_overall = best_c5
                best_overall[:] = global_best.copy()
        if time.time() - start_time > budget_s - 10:
            break

    # Final refinement with SLSQP
    if time.time() - start_time < budget_s - 5:
        def objective(x):
            x_proj = project(x, target_sum)
            return compute_c5(x_proj, dx)
        bounds = [(0, 1) for _ in range(n_points)]
        res = optimize.minimize(
            objective, best_overall.copy(), method='L-BFGS-B', bounds=bounds, 
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        if res.success:
            final_h = project(res.x, target_sum)
            final_c5 = compute_c5(final_h, dx)
            if final_c5 < best_c5_overall:
                best_c5_overall = final_c5
                best_overall = final_h

    h_final = project(best_overall, target_sum)
    c5_final = compute_c5(h_final, dx)
    return h_final, c5_final, n_points
