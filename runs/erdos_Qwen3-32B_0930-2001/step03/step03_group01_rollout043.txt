    winner_idx = np.argmin(selected_fitness)
    return population[indices[winner_idx]]

def crossover(p1, p2, crossover_prob, n_points):
    if np.random.rand() < crossover_prob:
        i, j = np.sort(np.random.choice(n_points, 2, replace=False))
        child1 = p1.copy()
        child2 = p2.copy()
        child1[i:j] = p2[i:j]
        child2[i:j] = p1[i:j]
        def fix_sum_h(h_vec, target_sum):
            s = np.sum(h_vec)
            diff = target_sum - s
            if diff != 0:
                if diff > 0:
                    zeros = np.where(h_vec == 0.0)[0]
                    np.random.shuffle(zeros)
                    h_vec[zeros[:diff]] = 1.0
                else:
                    ones = np.where(h_vec == 1.0)[0]
                    np.random.shuffle(ones)
                    h_vec[ones[:abs(diff)]] = 0.0
            return h_vec
        target_sum = n_points // 2
        child1 = fix_sum_h(child1, target_sum)
        child2 = fix_sum_h(child2, target_sum)
        return child1, child2
    else:
        return p1.copy(), p2.copy()

def mutate(h_vec, mutation_rate, n_points):
    h = h_vec.copy()
    num_mutations = int(mutation_rate * n_points)
    for _ in range(num_mutations):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            continue
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i], h[j] = h[j], h[i]
    return h

def compute_c5(h_vec, dx):
    conv = np.correlate(h_vec, 1 - h_vec, mode='full')
    return np.max(conv) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points // 2
    dx = 2.0 / n_points
    start_time = time.time()
    
    pop_size = 50
    mutation_rate = 0.1
    crossover_prob = 0.8
    
    population = [h.copy()]
    for _ in range(pop_size - 1):
        new_h = np.zeros(n_points, dtype=np.float32)
        indices = np.random.choice(n_points, size=target_sum, replace=False)
        new_h[indices] = 1.0
        population.append(new_h)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    while time.time() - start_time < budget_s - 1:
        fitness = [compute_c5(indiv, dx) for indiv in population]
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        current_best_h = population[current_best_idx].copy()
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h[:] = current_best_h
        
        new_population = []
        for _ in range(pop_size // 2):
            parent1 = tournament_selection(population, fitness, 2)
            parent2 = tournament_selection(population, fitness, 2)
            child1, child2 = crossover(parent1, parent2, crossover_prob, n_points)
            child1 = mutate(child1, mutation_rate, n_points)
            child2 = mutate(child2, mutation_rate, n_points)
            new_population.extend([child1, child2])
        population = new_population[:pop_size]
    
    return best_h, best_c5, n_points
