import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()
    
    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)
    
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    
    current_sum = 21.5
    final_sum = 21.0
    step_size = 0.02
    max_iter = 200
    
    def adjust_sum(h_vec, target_sum):
        delta = target_sum - np.sum(h_vec)
        h_adj = h_vec + delta / n_points
        h_adj = np.clip(h_adj, 0.0, 1.0)
        return h_adj
    
    while time.time() - start_time < budget_s - 10:
        h_initial = adjust_sum(h_best, target_sum=current_sum)
        result = optimize.minimize(
            compute_c5,
            x0=h_initial,
            method='COBYLA',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - current_sum}],
            options={'maxiter': max_iter, 'tol': 1e-8}
        )
        
        if result.success:
            c5_candidate = result.fun
            h_candidate = result.x
            if c5_candidate < c5_best:
                h_best[:] = h_candidate.copy()
                c5_best = c5_candidate
            current_sum = max(current_sum - step_size, final_sum)
        else:
            current_sum = max(current_sum - step_size * 2, final_sum)
        
        if np.isclose(current_sum, final_sum) or time.time() - start_time >= budget_s:
            break
            
    return h_best, c5_best, n_points
