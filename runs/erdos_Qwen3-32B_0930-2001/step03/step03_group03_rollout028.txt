import numpy as np
import time
from scipy.optimize import minimize as optimize_minimize

def compute_c5(h_vec, dx):
    conv = np.correlate(h_vec, 1 - h_vec, mode='full')
    return np.max(conv * dx)

def constraint_sum(x, target_sum):
    return np.sum(x) - target_sum

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')

    # Population and GA parameters
    pop_size = 20
    pop = [h_initial.copy() for _ in range(pop_size)]
    for i in range(pop_size):
        perturb = np.random.normal(0, 0.1, n)
        pop[i] = np.clip(pop[i] + perturb, 0, 1)
        current_sum = np.sum(pop[i])
        delta = (target_sum - current_sum) / n
        pop[i] = np.clip(pop[i] + delta, 0, 1)

    # Genetic Algorithm
    while time.time() - start_time < 0.7 * budget_s:
        # Evaluate population
        c5_scores = [compute_c5(ind, dx) for ind in pop]
        best_idx = np.argmin(c5_scores)
        if c5_scores[best_idx] < best_c5:
            best_c5 = c5_scores[best_idx]
            best_h = pop[best_idx].copy()

        # Selection
        sel_size = pop_size // 2
        tournament_size = 5
        selected = []
        for _ in range(sel_size):
            contestants = np.random.randint(0, pop_size, tournament_size)
            contestant_scores = [c5_scores[i] for i in contestants]
            winner_idx = contestants[np.argmin(contestant_scores)]
            selected.append(pop[winner_idx].copy())

        # Crossover and mutation
        new_pop = selected.copy()
        while len(new_pop) < pop_size:
            p1 = selected[np.random.randint(0, sel_size)].copy()
            p2 = selected[np.random.randint(0, sel_size)].copy()
            # Uniform crossover
            child = np.where(np.random.rand(n) < 0.5, p1, p2)
            # Mutation
            if np.random.rand() < 0.3:
                idxs = np.random.choice(n, int(0.1 * n), replace=False)
                noise = np.random.normal(0, 0.1, len(idxs))
                child[idxs] += noise
                child = np.clip(child, 0, 1)
                current_sum = np.sum(child)
                delta = (target_sum - current_sum) / n
                child += delta
                child = np.clip(child, 0, 1)
            new_pop.append(child)
        pop = new_pop

    if best_h is None:
        best_h = h_initial.copy()
        best_c5 = compute_c5(best_h, dx)

    bounds = [(0.0, 1.0) for _ in range(n)]
    result = optimize_minimize(
