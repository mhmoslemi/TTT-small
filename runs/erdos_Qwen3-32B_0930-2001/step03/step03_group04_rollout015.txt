import numpy as np
import time

def run():
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Adam optimizer parameters
    learning_rate = 1e-3
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()

    while time.time() - start_time < 1100:
        t += 1

        # Compute cross-correlation for gradient
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        current_max = np.max(cross_corr)
        max_indices = np.where(cross_corr == current_max)[0]

        # Compute gradient
        grad = np.zeros_like(h)
        for d in max_indices:
            k = d - (n_points - 1)
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    grad[i] += (1 - h[j]) * dx
                    grad[j] += -h[i] * dx

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        t_float = float(t)
        m_hat = m / (1 - beta1 ** t_float)
        v_hat = v / (1 - beta2 ** t_float)
        step = learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        h -= step

        # Projection onto constraints
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if not np.isclose(s, target_sum) and not np.isclose(s, 0.0):
            h *= target_sum / s
        h = np.clip(h, 0, 1)  # Final clip after scaling

        # Update best solution
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        # Learning rate decay
        if t % 100 == 0:
            learning_rate *= 0.9

    return (best_h, best_c5, n_points)
