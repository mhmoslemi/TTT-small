import numpy as np
import time

def run():
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    T = n_points / 2.0
    h /= np.sum(h)
    h *= T
    h = np.clip(h, 0, 1)

    # Adam optimizer parameters
    learning_rate = 1e-3
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    best_h = h.copy()
    best_c5 = np.inf

    start_time = time.time()

    while time.time() - start_time < 1100:
        t += 1
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(cross_corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        active_indices = np.where(cross_corr == current_c5)[0]
        shifts = active_indices - (n_points - 1)
        grad = np.zeros_like(h)

        for k_shift in shifts:
            for i in range(n_points):
                j = i + k_shift
                if 0 <= j < n_points:
                    grad[i] += (1 - h[j]) * dx
                    grad[j] -= h[i] * dx

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h -= learning_rate * m_hat / (np.sqrt(v_hat) + eps)

        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - T) > 1e-10:
            h *= T / total
            h = np.clip(h, 0, 1)

        if t % 100 == 0:
            learning_rate *= 0.9

    return (best_h, best_c5, n_points)
