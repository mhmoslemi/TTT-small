import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    try:
        initial_h = initial_h_values.copy()
    except:
        n = 42
        initial_h = np.ones(n) * 0.5
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()
    
    # Parameters
    pop_size = 20
    elite_ratio = 0.2
    tournament_size = 5
    generations = 500
    mutation_rate = 0.2
    mutation_strength = 0.1
    refine_interval = 10
    
    def evaluate(h):
        h_flat = h.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv) * dx
    
    def repair(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clipped)
        delta = (target_sum - sum_h) / n_points
        repaired = np.clip(h_clipped + delta, 0.0, 1.0)
        return repaired
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(1, pop_size):
        h = np.random.uniform(0, 1, n_points)
        population.append(repair(h))
    
    best_h, best_c5 = initial_h.copy(), evaluate(initial_h)
    
    while time.time() - start_time < budget_s - 30:
        scores = [evaluate(ind) for ind in population]
        elite_idx = np.argsort(scores)[:int(pop_size * elite_ratio)]
        elites = [population[i].copy() for i in elite_idx]
        new_population = elites.copy()
        
        while len(new_population) < pop_size:
            # Tournament selection
            tournament = np.random.choice(pop_size, tournament_size)
            parent1_idx = tournament[np.argmin([scores[i] for i in tournament])]
            parent1 = population[parent1_idx].copy()
            
            parent2_idx = tournament[np.argmin([scores[i] for i in tournament])]
            parent2 = population[parent2_idx].copy()
            
            # Crossover
            if np.random.rand() < 0.8:
                cross_point = np.random.randint(1, n_points)
                child = np.concatenate([parent1[:cross_point], parent2[cross_point:]])
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mask = np.random.rand(n_points) < 0.2
                child[mask] += np.random.normal(0, mutation_strength, np.sum(mask))
                child = np.clip(child, 0.0, 1.0)
            
            # Repair
            child = repair(child)
            new_population.append(child)
        
        population = new_population
        
        # Local optimization
        if len(population) > 0:
            current_best_idx = np.argmin(scores)
            candidate = population[current_best_idx].copy()
            def obj(x): return evaluate(x)
            def constraint(x): return np.sum(x) - target_sum
            
            bounds = [(0.0, 1.0)] * n_points
            result = optimize.minimize(obj, candidate, method='SLSQP', bounds=bounds,
                                       constraints=[{'type': 'eq', 'fun': constraint}],
                                       options={'maxiter': 30})
            if result.success:

                    best_h, best_c5 = refined, new_c5
    
    return best_h, best_c5, n_points
