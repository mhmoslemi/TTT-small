    particles.append(project_to_feasible(h_initial, target_sum))
    velocities.append(np.zeros(n_points))
    for _ in range(n_particles - 1):
        h_perturbed = h_initial + np.random.randn(n_points) * perturbation_magnitude
        h_projected = project_to_feasible(h_perturbed, target_sum)
        particles.append(h_projected)
        velocities.append(np.zeros(n_points))
    particles = np.array(particles)
    velocities = np.array(velocities)
    
    personal_best_positions = particles.copy()
    personal_best_scores = np.array([compute_c5(p) for p in particles])
    best_idx = np.argmin(personal_best_scores)
    global_best_position = particles[best_idx].copy()
    best_c5 = personal_best_scores[best_idx]

    iteration = 0
    while time.time() - start_time < budget_s - 5:
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (personal_best_positions[i] - particles[i]) +
                             c2 * r2 * (global_best_position - particles[i]))
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_to_feasible(particles[i], target_sum)
            current_score = compute_c5(particles[i])
            if current_score < personal_best_scores[i]:
                personal_best_positions[i] = particles[i].copy()
                personal_best_scores[i] = current_score
        if any(personal_best_scores < best_c5):
            best_idx = np.argmin(personal_best_scores)
            global_best_position[:] = personal_best_positions[best_idx].copy()
            best_c5 = personal_best_scores[best_idx]
        iteration += 1
        if iteration >= max_iterations or time.time() - start_time >= budget_s - 5:
            break

    if time.time() - start_time < budget_s - 5:
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res_slsqp = optimize.minimize(compute_c5, global_best_position, method='SLSQP',
                                      bounds=bounds, constraints=constraints,
                                      options={'maxiter': 500, 'ftol': 1e-10})
        if res_slsqp.success:
            refined_h = res_slsqp.x
            refined_c5 = compute_c5(refined_h)
            if refined_c5 < best_c5:
                global_best_position[:] = refined_h
                best_c5 = refined_c5

    elapsed = time.time() - start_time
    print(f"Total Elapsed: {elapsed:.1f}s, Final C5: {best_c5:.6f}")
    return global_best_position, best_c5, n_points
