        # Inertia reduction and velocity reset if no progress
        if np.allclose(personal_best_scores, personal_best_scores[0], atol=1e-4):
            w = max(0.5, w - 0.01)
            velocities = 0.05 * (2 * np.random.rand(n_particles+1, n) - 1)
    
    # Trust-region refinement
    top_indices = np.argsort(personal_best_scores)[:5]
    for idx in top_indices:
        if time.time() - start_time >= budget_s:
            break
        result = optimize.minimize(
            compute_c5,
            x0=particles[idx],
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 60, 'ftol': 1e-12}
        )
        if result.success and result.fun < best_c5:
            global_best = result.x.copy()
            best_c5 = result.fun
    
    # Final constraint enforcement
    final_h = np.clip(global_best, 0.0, 1.0)
    final_sum = final_h.sum()
    final_h += (target_sum - final_sum) / n
    final_h = np.clip(final_h, 0.0, 1.0)
    
    return (final_h, best_c5, n)
