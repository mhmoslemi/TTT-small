                                      bounds=bounds_slsqp, constraints=constraints_slsqp,
                                      options={'maxiter': 600, 'ftol': 1e-12})
        if res_refine.success:
            final_c5 = compute_c5(res_refine.x)
            if final_c5 < best_c5:
                best_h = res_refine.x.copy()
                best_c5 = final_c5

    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Best C5: {best_c5:.6f}")
    return best_h, best_c5, n_points
