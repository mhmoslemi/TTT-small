import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    T = n_points / 2.0
    np.random.seed(seed)
    
    # Adam parameters
    learning_rate = 1e-3
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Compute current overlaps
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        valid_cross_corr = cross_corr[n_points -1 : 2 * n_points -1]  # shifts 0 to n-1
        current_C5 = np.max(valid_cross_corr)
        max_shifts = np.where(valid_cross_corr == current_C5)[0]
        
        # Compute gradient
        grad = np.zeros(n_points)
        for k in max_shifts:
            for i in range(n_points - k):
                j = i + k
                grad[i] += (1.0 - h[j]) * dx
                grad[j] += (-h[i]) * dx
        if len(max_shifts) > 0:
            grad /= len(max_shifts)
        else:
            grad = np.zeros(n_points)
        
        # Adam update
        t +=1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = m_hat / (np.sqrt(v_hat) + eps)
        h -= learning_rate * step
        
        # Clip and renormalize
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if abs(sum_h - T) > 1e-10:
            h *= T / sum_h
    
    return (h, current_C5, n_points)

import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    T = n_points / 2.0
    
    # Adam optimizer parameters
    learning_rate = 1e-3
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    
    np.random.seed(seed)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Compute overlaps for all valid shifts
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        start_idx = n_points - 1
        valid_cross_corr = cross_corr[start_idx:2 * n_points - 1]
        current_C5 = np.max(valid_cross_corr)
        max_shifts = np.where(valid_cross_corr == current_C5)[0]
        
        # Compute gradient for active max shifts
        grad = np.zeros(n_points)
        for k in max_shifts:
            for i in range(n_points - k):
                j = i + k
                grad[i] += (1.0 - h[j]) * dx
                grad[j] += (-h[i]) * dx
        if len(max_shifts) > 0:
            grad /= len(max_shifts)
        
        # Update Adam variables
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = m_hat / (np.sqrt(v_hat) + eps)
        h -= learning_rate * step
        
        # Project onto feasible region
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if abs(sum_h - T) > 1e-10:
            h *= T / sum_h
    
    return (h, current_C5, n_points)
