def crossover(parent1, parent2):
    # Create a child by swapping some bits between parents
    child = parent1.copy()
    # Find the positions where parent1 and parent2 differ
    diff_indices = np.where(parent1 != parent2)[0]
    if len(diff_indices) == 0:
        return child  # no differences
    # Decide how many to swap; say half
    num_swaps = len(diff_indices) // 2
    swap_indices = np.random.choice(diff_indices, size=num_swaps, replace=False)
    # Swap the bits
    child[swap_indices] = parent2[swap_indices]
    return child

This way, the child will inherit half of the differing positions from parent2, but the total number of 1s remains 21.
