            indices = np.random.choice(m, num_perturb, replace=False)
            for idx in indices:
                delta = np.random.normal(0, 0.05)
                values[idx] += delta
                values[idx] = np.clip(values[idx], 0.0, 1.0)
        else:
            # Split interval
            if m < n - 1:  # Ensure we can split
                split_idx = np.random.randint(0, m)
                s, e = partitions[split_idx]
                if e - s > 1:
                    split_pos = np.random.randint(s + 1, e)
                    # Split partition
                    partitions.pop(split_idx)
                    partitions.insert(split_idx, (s, split_pos))
                    partitions.insert(split_idx + 1, (split_pos, e))
                    m += 1
                    # Duplicate values
                    new_val1 = values[split_idx]
                    new_val2 = values[split_idx]
                    values.pop(split_idx)
                    values.insert(split_idx, new_val1)
                    values.insert(split_idx + 1, new_val2)
                else:
                    # Cannot split, revert
                    partitions = prev_partitions
                    values = prev_values
                    continue
        
        # Compute integral sum S and adjust values
        S = 0.0
        L_list = []
        for i in range(m):
            s, e = partitions[i]
            L = (e - s) * dx
            L_list.append(L)
            S += values[i] * L
        sum_L2 = sum(l**2 for l in L_list)
        a = (1.0 - S) / sum_L2 if sum_L2 != 0 else 0.0
        valid = True
        new_values = values.copy()
        for i in range(m):
            new_v = new_values[i] + a * L_list[i]
            if not 0.0 <= new_v <= 1.0:
                valid = False
                break
        if valid:
            for i in range(m):
                new_values[i] = new_values[i] + a * L_list[i]
        else:
            # Cannot apply correction, revert
            partitions = prev_partitions
            values = prev_values
            continue
        values = new_values
        
        # Compute new_c5
        h_candidate = np.zeros(n)
        for i in range(m):
            s, e = partitions[i]
            h_candidate[s:e] = values[i]
        new_c5 = compute_c5(h_candidate)
        delta_c5 = new_c5 - current_c5
        
        # Acceptance step
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            # Accept, update current state
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_partitions = [p for p in partitions]
                best_values = values.copy()
                best_h = h_candidate.copy()
        else:
            # Reject, revert
            partitions = prev_partitions
            values = prev_values
        
        T *= alpha
    
    return best_h, best_c5, n
