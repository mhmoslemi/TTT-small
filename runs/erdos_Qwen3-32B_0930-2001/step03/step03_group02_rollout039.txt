    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)

        # Final clip in case of overshoot
        h_clipped = np.clip(h_clipped, 0, 1)
        return h_clipped

    w = 0.7  # inertia

    iter_pso = 0
