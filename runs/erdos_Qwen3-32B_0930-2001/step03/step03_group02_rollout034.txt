    start_time = time.time()
    
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    def generate_block_initial(n):
        h = np.zeros(n)
        for _ in range(3):  # 3 blocks
            start = np.random.uniform(0, 2)
            length = np.random.uniform(0.5, 1.5)
            start_idx = int(start * n / 2)
            end_idx = int((start + length) * n / 2)
            start_idx = max(0, start_idx)
            end_idx = min(n, end_idx)
            value = np.random.uniform(0.2, 0.8)
            h[start_idx:end_idx] += value
        h_clipped = np.clip(h, 0, 1)
        s = h_clipped.sum()
        if s == 0:
            return h_initial.copy()
        scale = target_sum / s
        h_scaled = np.clip(h_clipped * scale, 0, 1)
        return h_scaled

    population = [h_initial.copy()]
    for _ in range(20):
        population.append(generate_block_initial(n_points))

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)

    bounds = [(0, 1)] * n_points
    constraints = {'type': 'eq', 'fun': lambda x: x.sum() - target_sum}

    for candidate in population:
        if time.time() - start_time > budget_s - 10:
            break
        res = optimize.minimize(compute_c5, candidate, method='SLSQP', 
                               bounds=bounds, constraints=constraints,
                               options={'maxiter': 50, 'ftol': 1e-6})
        if res.success:
            current_c5 = compute_c5(res.x)
            if current_c5 < best_c5:
                best_h = res.x.copy()
                best_c5 = current_c5

    if time.time() - start_time < budget_s - 5:
        res_refine = optimize.minimize(compute_c5, best_h, method='L-BFGS-B', 
                                      bounds=bounds, 
                                      options={'maxiter': 100, 'ftol': 1e-8})
        if res_refine.success:
            best_h = res_refine.x
            best_c5 = compute_c5(best_h)

    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Best C5: {best_c5:.6f}")
    return best_h, best_c5, n_points
