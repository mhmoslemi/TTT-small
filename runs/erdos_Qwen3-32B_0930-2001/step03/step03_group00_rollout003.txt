            positions[i] = finalize_h(positions[i], target_sum, n_points)

        # Periodic refinement with SLSQP
        if iteration % 20 == 0:
            valid_pbest = pbest_values[np.isfinite(pbest_values)]
            if len(valid_pbest) > 0:
                top_indices = np.argsort(pbest_values)[np.isfinite(pbest_values)][:5]
                for idx in top_indices:
                    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})
                    result = opt.minimize(
                        lambda x: compute_c5(x, n_points, dx),
                        pbest_positions[idx],
                        method='SLSQP',
                        bounds=[(0, 1)] * n_points,
                        constraints=cons,
                        options={'maxiter': 50, 'ftol': 1e-6}
                    )
                    if result.success:
                        candidate_c5 = compute_c5(result.x, n_points, dx)
                        if candidate_c5 < pbest_values[idx]:
                            pbest_positions[idx] = result.x
                            pbest_values[idx] = candidate_c5
                            if candidate_c5 < gbest_value:
                                gbest_value = candidate_c5
                                gbest_position = result.x.copy()

        if time.time() - start_time > budget_s:
            break

    final_c5 = compute_c5(gbest_position, n_points, dx)
    return gbest_position, final_c5, n_points
