import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_current = initial_h_values.copy()
    n_points = len(h_current)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def finalize_h(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if current_sum == 0:
            h = np.full_like(h, target_sum / len(h))
        else:
            scale = target_sum / current_sum
            h = np.clip(h * scale, 0, 1)
        return h

    def compute_c5(h_vec):
        h = finalize_h(h_vec)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    # Differential Evolution with feasibility projection
    x0_de = initial_h_values.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    max_de_iter = min(50, int((budget_s - 300) // 20))
    result_de = opt.differential_evolution(
        compute_c5, 
        bounds,
        strategy='best1bin',
        maxiter=max_de_iter,
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.8,
        polish=False,
        seed=seed,
        workers=1,
        callback=lambda x, c, i: time.time() - start_time > budget_s - 300
    )
    h_de = result_de.x
    h_de = finalize_h(h_de)
    best_c5 = compute_c5(h_de)
    h_best = h_de.copy()
    print(f"DE: {best_c5:.6f}, time: {time.time()-start_time:.1f}s")

    # Iterative refinement with hybrid local optimization
    best_record = [(h_best, best_c5)]
    while time.time() - start_time < budget_s - 50:
        pert = np.random.randn(n_points) * 0.2 + np.random.choice([-0.3, 0.3], size=n_points)
        h_pert = np.clip(h_best * 1.0 + pert, 0, 2)
        h_pert = finalize_h(h_pert)
        
        # Nelder-Mead local search with constraints
        result_nm = opt.minimize(
            lambda x: abs(min(1.0, max(0.0, compute_c5(x) - 0.38))),
            h_pert,
            method='Nelder-Mead',
            bounds=[(0,1)]*n_points,
            options={'maxiter': 200 if best_c5 > 0.38 else 50, 'xatol': 1e-4}
        )
        if result_nm.success:
            h_nm = finalize_h(result_nm.x)
            c5_nm = compute_c5(h_nm)
            if c5_nm < best_c5:
                best_c5, h_best = c5_nm, h_nm.copy()
                print(f"NM: {best_c5:.6f}, time: {time.time()-start_time:.1f}s")
                best_record.append((h_best, c5_nm))
        
        # Constrained gradient-based refinement
        result_cb = opt.minimize(
            lambda x: compute_c5(x),
            h_best,
            method='SLSQP',
            bounds=[(0, 1)]*n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}],
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result_cb.success:
            h_cb = finalize_h(result_cb.x)
            c5_cb = compute_c5(h_cb)
            if c5_cb < best_c5:
                best_c5, h_best = c5_cb, h_cb.copy()
                print(f"CB: {best_c5:.6f}, time: {time.time()-start_time:.1f}s")
                best_record.append((h_best, c5_cb))
                
        if not best_record or best_record[-1][1] == best_c5:
            break
            
    return h_best, best_c5, n_points
