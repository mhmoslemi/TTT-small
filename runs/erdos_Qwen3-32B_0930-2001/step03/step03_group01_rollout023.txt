        h0 = h0 / np.sum(h0) * target_sum  # Ensure sum constraint
        h0 = np.clip(h0, 0, 1)  # Enforce bounds
        result = optimize.minimize(
            compute_c5,
            x0=h0,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        if result.success:
            current_c5 = result.fun
            current_h = np.clip(result.x, 0, 1)
            if current_c5 < c5_best:
                h_best[:] = current_h
                c5_best = current_c5
    
    return h_best, c5_best, n_points
