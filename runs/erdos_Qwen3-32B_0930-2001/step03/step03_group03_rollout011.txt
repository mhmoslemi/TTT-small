import time
import random
import scipy.optimize as optimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_total = len(h_initial)
    target_sum = n_total / 2.0
    dx = 2.0 / n_total

    # Initialize intervals: m=6
    m_initial = 6
    intervals_list = []
    interval_len = n_total // m_initial
    remainder = n_total % m_initial
    start_idx = 0
    for i in range(m_initial):
        il = interval_len + (1 if i < remainder else 0)
        end_idx = start_idx + il - 1
        avg_val = np.mean(h_initial[start_idx:end_idx+1])
        intervals_list.append( (start_idx, end_idx, avg_val) )
        start_idx = end_idx + 1

    # Build h array from intervals
    def build_h(intervals):
        h = np.zeros(n_total)
        for s, e, v in intervals:
            h[s:e+1] = v
        return h

    # Adjust intervals to match target_sum
    def adjust_intervals(intervals):
        total_len = sum(e - s + 1 for s, e, _ in intervals)
        current_sum = sum((e - s + 1) * v for s, e, v in intervals)
        adjustment = (target_sum - current_sum) / total_len
        new_intervals = []
        for s, e, v in intervals:
            new_v = v + adjustment
            new_v = max(0.0, min(1.0, new_v))
            new_intervals.append( (s, e, new_v) )
        return new_intervals

    # Compute C5 value
    def compute_c5(h_vec):
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(conv * dx)

    h_current = build_h(intervals_list)
    c5_current = compute_c5(h_current)
    best_intervals = intervals_list.copy()
    best_c5 = c5_current
    current_intervals = intervals_list.copy()
    start_time = time.time()
    max_budget = budget_s - 20  # Reserve time for final optimization

    T_initial = 0.1
    alpha = 0.9995
    temp = T_initial

    while time.time() - start_time < max_budget:
        # Generate candidate
        if random.random() < 0.5 and len(current_intervals) < 128:
            # Try to split an interval
            candidates = [i for i, (s, e, v) in enumerate(current_intervals) if e - s > 0]
            if not candidates:
                continue
            split_idx = random.choice(candidates)
            s, e, v = current_intervals[split_idx]
            mid = s + (e - s) // 2
            current_intervals[split_idx] = (s, mid, v)
            current_intervals.insert(split_idx + 1, (mid+1, e, v))
        else:
            # Perturb up to 2 intervals
            k = random.randint(1, 2)
            indices = random.sample(range(len(current_intervals)), k)
            new_intervals = current_intervals.copy()
            for idx in indices:
                s, e, v = new_intervals[idx]
                noise = random.gauss(0, 0.2)
                new_v = v + noise
                new_v = max(0.0, min(1.0, new_v))
                new_intervals[idx] = (s, e, new_v)
            current_intervals = adjust_intervals(new_intervals)

        # Build candidate h and evaluate
        h_candidate = build_h(current_intervals)
        c5_candidate = compute_c5(h_candidate)

        # Acceptance criterion
        delta = c5_candidate - c5_current
        if delta < 0:
            # Improvement
            c5_current = c5_candidate
            current_intervals = current_intervals.copy()  # Keep copy
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_intervals = current_intervals.copy()
        else:
            # Metropolis
            metropolis = np.exp(-delta / temp)
            if random.random() < metropolis:
                c5_current = c5_candidate
            else:
                # Revert (but how?)
                # Here, we can reassign to a known good copy. However, in the code above,
                # during split and perturbation, we reassign current_intervals, so to revert, we need to keep a backup.
                # To simplify, just leave it as is (risk of bad move being kept)
                pass

        # Cool down
        temp *= alpha

        # Occasionally force higher resolution
        if len(current_intervals) < 128 and random.random() < 0.05:
            candidates = [i for i, (s, e, v) in enumerate(current_intervals) if e - s > 0]
            if candidates:
                split_idx = random.choice(candidates)
                s, e, v = current_intervals[split_idx]
                mid = s + (e - s) // 2
                current_intervals[split_idx] = (s, mid, v)
                current_intervals.insert(split_idx + 1, (mid+1, e, v))

    # Best h from annealing
    h_best = build_h(best_intervals)

    # Final local optimization
    def objective(x):
        h = x.reshape(n_total)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_total)]
    result = optimize.minimize(
        objective,
        x0=h_best,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    if result.success:
        h_refined = result.x
        c5_refined = objective(h_refined)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            h_best = h_refined

    # Final return
    return h_best, best_c5, n_total
