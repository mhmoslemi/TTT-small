import numpy as np
import time

def evaluate_objective(h, n_points):
    dx = 2.0 / n_points
    cross_corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(cross_corr)

def run():
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 1e-3
    decay_rate = 0.9
    decay_steps = 100

    h = h_initial.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    best_h = h.copy()
    best_c5 = evaluate_objective(h, n_points)

    start_time = time.time()
    max_time = 1100  # seconds

    while time.time() - start_time < max_time:
        t += 1

        # Compute overlaps
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        current_max = np.max(cross_corr)

        # Find indices where overlap is current_max (within tolerance)
        max_indices = np.where(np.isclose(cross_corr, current_max, atol=1e-8))[0]

        # Compute gradient
        grad = np.zeros_like(h)
        for m_index in max_indices:
            actual_shift = m_index - (n_points - 1)
            for i in range(n_points):
                j = (i + actual_shift) % n_points
                grad[i] += (1 - h[j]) * dx
                grad[j] += -h[i] * dx

        # Apply Adam step
        lr_t = lr * (decay_rate ** (t // decay_steps))
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h -= lr_t * m_hat / (np.sqrt(v_hat) + eps)

        # Clip and rescale
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if not np.isclose(sum_clipped, 0):
            h_clipped *= target_sum / sum_clipped
        else:
            # Fallback to initial values if sum is zero (unlikely)
            h_clipped = h_initial.copy()
        h = np.clip(h_clipped, 0, 1)

        # Update best solution
        current_c = evaluate_objective(h, n_points)
        if current_c < best_c5:
            best_h = h.copy()
            best_c5 = current_c

    return (best_h, best_c5, n_points)
