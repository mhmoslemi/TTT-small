    start_time = time.time()
    
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # Constraint: sum(h) == target_sum

    # Precompute overlaps for initial h for all possible lags to select top ones
    def compute_overlap_for_k(h, k):

        sum_hikh = 0
        for i in range(n - k):
            sum_hikh += h[i] * h[i + k]
        overlap = dx * (target_sum - sum_hikh)
        return overlap

    # Compute overlaps for all possible lags k in 1 ... n_points-1
    all_k_candidates = list(range(1, n_points))
    overlaps = []
    for k in all_k_candidates:
        overlap_val = compute_overlap_for_k(h_initial, k)
        overlaps.append( (k, overlap_val) )
    # Sort by descending overlap value
    overlaps.sort(key=lambda x: x[1], reverse=True)
    num_selected = min(20, len(overlaps))  # Select top 20 lags
    selected_lags = [k for k, _ in overlaps[:num_selected]]

    # Solve the QCQP relaxation
    try:
        # Define cvxpy variables
        h = cp.Variable(n_points)
        t = cp.Variable()

        # Define constraints
        constraints = [

            h >= 0,
            h <= 1
        ]

        for k in selected_lags:

        objective = cp.Minimize(t)

        # Create and solve the problem
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.ECOS, verbose=False)
        h_qp = h.value
        t_opt = t.value

        return best_h, best_c5, n_points
