            method='SLSQP',
            bounds=[(0, 1)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}],
            options={'maxiter': 60, 'ftol': 1e-10}
        )
        if result.success and result.fun < personal_best_scores[idx]:
            personal_best_scores[idx] = result.fun
            personal_best[idx] = result.x

    best_idx = np.argmin(personal_best_scores)
    return personal_best[best_idx].flatten(), personal_best_scores[best_idx], n
