        delta = (target_sum - current_sum) / n
        h_clipped = np.clip(h_clipped + delta, 0.0, 1.0)
        return h_clipped
    
    learning_rate = 0.1
    best_h = h.copy()
    best_c5 = np.max(compute_overlaps(h))
    
    while time.time() - start_time < budget_s - 20:
        overlaps = compute_overlaps(h)
        k_max = np.argmax(overlaps) + 1
        
        # Compute gradient
        grad = np.zeros(n)
        i_plus_k = np.arange(n) + k_max
        valid_plus = i_plus_k < n
        grad[valid_plus] = (1 - h[i_plus_k[valid_plus]]) * dx
        i_minus_k = np.arange(n) - k_max
        valid_minus = i_minus_k >= 0
        grad[valid_minus] -= h[i_minus_k[valid_minus]] * dx
        
        # Update step with adaptive clipping
        h = h - learning_rate * grad
        h = project_to_constraints(h)
        
        # Simple learning rate decay
        if (time.time() - start_time) % 100 < 1 and learning_rate > 0.01:
            learning_rate *= 0.95
            
        # Update best found solution
        current_overlaps = compute_overlaps(h)
        current_c5 = np.max(current_overlaps)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return best_h, best_c5, n
