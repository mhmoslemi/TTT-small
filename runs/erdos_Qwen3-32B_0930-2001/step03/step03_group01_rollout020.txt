import numpy as np
import time

def compute_c5(h_vec, n_points, dx):
    h = h_vec.astype(float)
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx

def repair(child, target_sum):
    total = int(np.sum(child))
    if total == target_sum:
        return child
    elif total > target_sum:
        ones = np.where(child == 1)[0]
        np.random.shuffle(ones)
        to_flip = ones[:total - target_sum]
        child[to_flip] = 0
    else:
        zeros = np.where(child == 0)[0]
        np.random.shuffle(zeros)
        to_flip = zeros[:target_sum - total]
        child[to_flip] = 1
    return child

def crossover(p1, p2, target_sum):
    mask = np.random.rand(len(p1)) < 0.5
    child = np.where(mask, p1, p2).copy()
    return repair(child, target_sum)
