import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    def finalize_h(h):
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if current_sum == 0:
            h = np.full_like(h, target_sum / len(h))
        else:
            scale = target_sum / current_sum
            h = np.clip(h * scale, 0, 1)
        return h

    h_best = finalize_h(h_current)
    best_c5 = compute_c5(h_best)
    
    while time.time() - start_time < budget_s - 50:
        # Hybrid perturbation: 50% swaps, 50% noise

            # Perform multiple random swaps

        h_pert = finalize_h(h_pert)
