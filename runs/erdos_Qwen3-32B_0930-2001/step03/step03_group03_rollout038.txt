import numpy as np
import time
import scipy.optimize as optimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    h_initial = initial_h_values.copy()
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0

    def generate_candidate():
        h = np.random.rand(n)
        sum_h = np.sum(h)
        delta = (target_sum - sum_h) / n
        h = np.clip(h + delta, 0, 1)
        sum_h2 = np.sum(h)
        delta2 = (target_sum - sum_h2) / n
        h = np.clip(h + delta2, 0, 1)
        return h

    def compute_c5(h_vec):
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(conv * dx)

    def tournament_selection(population, scores, size=5):
        indices = np.random.choice(len(population), size)
        best_idx = np.argmin(scores[indices])
        return population[indices[best_idx]].copy()

    def crossover(parent1, parent2):
        split_point = np.random.randint(1, n)
        child = np.concatenate((parent1[:split_point], parent2[split_point:]))
        sum_child = np.sum(child)
        delta = (target_sum - sum_child) / n
        child = np.clip(child + delta, 0, 1)
        return child

    def mutate(individual, mr=0.3):
        individual = individual.copy()
        for i in range(n):
            if np.random.rand() < mr:
                delta = np.random.normal(0, 0.1)
                individual[i] += delta
        sum_ind = np.sum(individual)
        delta = (target_sum - sum_ind) / n
        individual = np.clip(individual + delta, 0, 1)
        return individual

    population_size = 30
    population = [h_initial.copy()]
    for _ in range(population_size - 1):
        population.append(generate_candidate())
    population = np.array(population)

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    end_time = start_time + budget_s

    # Main GA loop
    while time.time() < end_time:
        c5_scores = np.array([compute_c5(h) for h in population])
        if np.min(c5_scores) < best_c5:
            indices = np.unravel_index(np.argmin(c5_scores), c5_scores.shape)
            best_h = population[indices[0]].copy()
            best_c5 = c5_scores[indices[0]]

        new_population = []
        for _ in range(population_size):
            parent1 = tournament_selection(population, c5_scores)
            parent2 = tournament_selection(population, c5_scores)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        population = np.array(new_population)

        # Early stopping if time exceeds 90% of budget
        if time.time() > end_time * 0.9:
            break

    # Final refinement with SLSQP
    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = [(0., 1.) for _ in range(n)]
    result = optimize.minimize(
        compute_c5, x0=best_h, method='SLSQP', bounds=bounds,
        constraints=[dict(type='eq', fun=constraint_sum)], options={'maxiter': 80}
    )
    if result.success:
        best_h = np.clip(result.x, 0., 1.)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
