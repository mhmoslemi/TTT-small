import numpy as np
import time

def run():
    h_current = initial_h_values.copy()
    n_points = len(h_current)
    T_sum = n_points / 2.0  # Required integral value

    def evaluate_objective(h):
        dx = 2.0 / n_points
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        return float(np.max(cross_corr))

    current_C = evaluate_objective(h_current)
    best_C = current_C
    best_h = h_current.copy()

    initial_T = current_C * 2.0
    T = initial_T
    alpha = 0.995
    T_min = 1e-4

    start_time = time.time()
    n_perturb = max(1, int(n_points * 0.1))  # Number of positions to perturb each step

    while time.time() - start_time < 1100 and T > T_min:
        # Generate a neighboring solution
        h_prime = h_current.copy()
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        noise = np.random.normal(0, T, size=len(indices))
        h_prime[indices] += noise
        h_prime = np.clip(h_prime, 0, 1)
        
        sum_prime = np.sum(h_prime)
        if np.abs(sum_prime) > 1e-10:
            h_prime *= T_sum / sum_prime
        else:
            h_prime = h_current.copy()  # Fallback to current solution if normalization fails

        new_C = evaluate_objective(h_prime)
        delta_E = new_C - current_C

        # Acceptance criterion
        if delta_E < 0 or np.exp(-delta_E / T) > np.random.rand():
            h_current = h_prime.copy()
            current_C = new_C

            if current_C < best_C:
                best_C = current_C
                best_h[:] = h_current  # Update best solution in-place to minimize memory

        T *= alpha  # Cool down

    return (best_h, best_C, n_points)
