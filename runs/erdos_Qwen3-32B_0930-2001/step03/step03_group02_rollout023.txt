import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)

    candidates = [h_initial]
    # Add shifted patterns
    for shift in np.concatenate((np.arange(1, 5), np.arange(-4, 0))):
        h_candidate = np.roll(h_initial, shift)
        h_candidate = np.clip(h_candidate, 0, 1)
        candidates.append(h_candidate)
    # Add random block candidates
    num_random = 30
    indices = np.random.choice(n_points, size=(num_random, int(target_sum)), replace=False)
    for idx in indices:
        h_candidate = np.zeros(n_points)
        h_candidate[idx] = 1.0
        candidates.append(h_candidate)

    for candidate in candidates:
        if time.time() - start_time > budget_s - 5:
            break
        # Normalize candidate to meet sum constraint
        h0 = np.clip(candidate, 0, 1)
        current_sum = h0.sum()
        if abs(current_sum - target_sum) > 1e-6:
            scale = target_sum / current_sum
            h0 = np.clip(h0 * scale, 0, 1)
        # Optimize
        constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        res = optimize.minimize(compute_c5, h0, method='SLSQP', bounds=[(0, 1)]*n_points,

        if res.success:
            current_c5 = compute_c5(res.x)
            if current_c5 < best_c5:

                best_c5 = current_c5

    # Final refinement if time allows

        if final_res.success:
            best_c5 = compute_c5(final_res.x)
            best_h = final_res.x

    elapsed = time.time() - start_time
    print(f"C5: {best_c5:.8f}, Elapsed: {elapsed:.1f}s")
    return best_h, best_c5, n_points
