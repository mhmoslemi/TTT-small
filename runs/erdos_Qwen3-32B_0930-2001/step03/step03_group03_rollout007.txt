import numpy as np
import time
import scipy.optimize as optimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0

    def compute_c5(h_vec):
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(conv * dx)

    def enforce_sum(h_vec, target):
        h = np.clip(h_vec.copy(), 0, 1)
        current_sum = np.sum(h)
        delta = (target - current_sum) / n
        h += delta
        h = np.clip(h, 0, 1)
        return h

    # Initialize population
    population_size = 20
    population = [h_initial.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n)
        population.append(enforce_sum(h, target_sum))
    fitness = [compute_c5(p) for p in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    start_time = time.time()
    while time.time() - start_time < budget_s - 10:
        # Select top half as parents
        parent_idx = np.argsort(fitness)[:population_size // 2]
        parents = [population[i] for i in parent_idx]
        new_candidates = []

        # Generate offspring via crossover and mutation
        for _ in range(population_size - len(parents)):
            # Crossover
            p1, p2 = parents[np.random.randint(len(parents))], parents[np.random.randint(len(parents))]
            split = np.random.randint(1, n)
            child = np.concatenate((p1[:split], p2[split:]))
            # Mutation
            if np.random.rand() < 0.2:
                idx = np.random.randint(n)
                child[idx] += np.random.normal(0, 0.1)
                child[idx] = np.clip(child[idx], 0, 1)
            child = enforce_sum(child, target_sum)
            new_candidates.append(child)

        # Replace population
        population = parents + new_candidates
        for i in range(len(population)):
            population[i] = enforce_sum(population[i], target_sum)
        fitness = [compute_c5(p) for p in population]
        current_best = np.min(fitness)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[np.argmin(fitness)].copy()

    # Final local optimization
    def objective(x):
        return compute_c5(x)

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0)] * n
    result = optimize.minimize(
        objective,
        x0=best_h,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        method='SLSQP',
        options={'maxiter': 50, 'ftol': 1e-10}
    )
    if result.success:
        optimized = enforce_sum(result.x, target_sum)
        final_c5 = compute_c5(optimized)
        best_h = optimized
        best_c5 = final_c5

    return best_h, best_c5, n
