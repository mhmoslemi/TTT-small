    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv) * dx

    def adjust_feasibility(h_vec, target=target_sum):
        h = h_vec.copy()
        h = np.clip(h, 0.0, 1.0)
        delta = (target - np.sum(h)) / n_points
        h = np.clip(h + delta, 0.0, 1.0)
        return h

    # Genetic Algorithm Phase
    pop_size = 30
    generations = 300
    mutation_rate = 0.15
    population = np.zeros((pop_size, n_points))
    population[0] = adjust_feasibility(h_initial.copy())
    for i in range(1, pop_size):
        h = np.random.uniform(0.0, 1.0, n_points)
        population[i] = adjust_feasibility(h)

    elite_ratio = 0.2
    best_c5 = np.inf
    best_h = h_initial.copy()

    while time.time() - start_time < 0.8 * budget_s:

        elite_idx = np.argsort(c5_list)[:int(pop_size * elite_ratio)]

            best_c5 = np.min(c5_list)
            best_h = population[np.argmin(c5_list)].copy()

        # Crossover and Mutation
        offspring = []

                perturbation = np.random.randn(n_points) * 0.1
                child += perturbation
                child = adjust_feasibility(child)
            offspring.append(child)

    # Bayesian Refinement Phase
    def acq_func(x, model, current_min):
        mean, std = model.predict(x.reshape(1, -1), return_std=True)
        return -np.clip(std, 1e-4, None) * np.maximum(0, current_min - mean)

    top_candidates = np.array([adjust_feasibility(np.random.uniform(0.0, 1.0, n_points))

                res = optimize.minimize(lambda x: acq_func(x, gp, best_c5),

                                        bounds=30 * [(0.0, 1.0)],

                if res.success:
                    prop = adjust_feasibility(res.x)
