import numpy as np

import scipy.optimize as optimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy().astype(float)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()

    def compute_c5(h_vec):
        h_full = np.concatenate([h_vec, np.zeros_like(h_vec)])
        vals = []
        for k in range(1, n_points):
            shifted = np.roll(h_full, -k)
            mask = shifted != shifted[k:]
            shifted = shifted[:-k]
            conv = np.sum(h_vec[:n_points - k] * (1 - shifted[:n_points - k][~mask]))
            vals.append(conv * dx)
        return max(vals)

    population_size = 40
    population = [h_initial.copy()]
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand *= target_sum / h_rand.sum()
        h_rand = np.clip(h_rand, 0.0, 1.0)
        current_sum = h_rand.sum()
        adjustment = (target_sum - current_sum) / n_points
        h_rand = np.clip(h_rand + adjustment, 0.0, 1.0)
        population.append(h_rand)

    best_c5 = compute_c5(population[0])
    best_h = population[0].copy()

    def tournament_selection(pop, pop_fit, k=5):
        indices = np.random.choice(len(pop), k)
        mask = np.argmin([pop_fit[i] for i in indices])
        return pop[indices[mask]].copy()

    gen = 0

        indices = np.argsort(fits)
