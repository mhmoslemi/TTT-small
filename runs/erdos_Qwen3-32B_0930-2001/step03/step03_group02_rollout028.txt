import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    # Particle Swarm Optimization setup
    n_particles = 20
    w = 0.7  # inertia weight
    c1 = 1.5  # cognitive
    c2 = 1.5  # social
    max_iter = 50

    def clip_and_project(position):
        position = np.clip(position, 0.0, 1.0)
        current_sum = position.sum()
        if current_sum < 1e-8:
            return h_initial.copy()
        scale = target_sum / current_sum
        position *= scale
        position = np.clip(position, 0.0, 1.0)
        return position

    particles = [clip_and_project(h_initial.copy()) for _ in range(n_particles)]
    velocities = [np.random.normal(0, 0.05, n_points) for _ in range(n_particles)]
    personal_best_positions = [p.copy() for p in particles]
    personal_best_scores = np.array([compute_c5(p) for p in particles])
    global_best_idx = np.argmin(personal_best_scores)
    global_best_position = personal_best_positions[global_best_idx]
    global_best_score = personal_best_scores[global_best_idx]

    for iter_idx in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        # Evaluate current positions and update bests
        current_scores = [compute_c5(p) for p in particles]
        for i in range(n_particles):
            if current_scores[i] < personal_best_scores[i]:
                personal_best_scores[i] = current_scores[i]
                personal_best_positions[i] = particles[i].copy()
                if current_scores[i] < global_best_score:
                    global_best_score = current_scores[i]
                    global_best_position = particles[i].copy()
        # Update velocities and positions
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (w * velocities[i] +
                            c1 * r1 * (personal_best_positions[i] - particles[i]) +
                            c2 * r2 * (global_best_position - particles[i]))
            particles[i] = particles[i] + velocities[i]
            particles[i] = clip_and_project(particles[i])

    # Final refinement
    best_particle = global_best_position.copy()
    if time.time() - start_time < budget_s - 5:
        result = optimize.minimize(
            compute_c5, best_particle,
            method='SLSQP',
            bounds=[(0, 1)]*n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}],
            options={'maxiter': 300, 'ftol': 1e-10}
        )
        if result.success:
            final_score = compute_c5(result.x)
            if final_score < global_best_score:
                best_particle = result.x
                global_best_score = final_score

    elapsed = time.time() - start_time
    return best_particle, global_best_score, n_points
