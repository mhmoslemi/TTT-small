import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_current = initial_h_values.copy()
    n_points = len(h_current)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    def finalize_h(h):
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if current_sum == 0:
            h = np.full_like(h, target_sum / len(h))
        else:
            scale = target_sum / current_sum
            h = np.clip(h * scale, 0, 1)
        return h

    h_best = finalize_h(h_current)
    best_c5 = compute_c5(h_best)
    
    while time.time() - start_time < budget_s - 50:
        # Aggressive perturbation with increased noise and random feature swaps
        pert = np.random.randn(n_points) * 0.2
        h_pert = np.clip(h_best + pert, 0, 1)
        h_pert = finalize_h(h_pert)
        
        # Quadratic penalty to preserve structure and improve convergence
        result_nm = opt.minimize(
            compute_c5, h_pert, method='Nelder-Mead',
            bounds=[(0, 1)] * n_points,
            options={'maxiter': 150, 'xatol': 1e-4}
        )
        if result_nm.success:
            h_nm = result_nm.x
        else:
            h_nm = h_pert

        cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})

            compute_c5, h_nm, method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=cons,

        )

    elapsed = time.time() - start_time
