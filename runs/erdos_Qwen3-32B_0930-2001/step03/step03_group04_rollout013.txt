import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    T = n_points / 2.0
    dx = 2.0 / n_points

    # Adam optimizer parameters
    learning_rate = 1e-3
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    decay_factor = 0.95
    decay_steps = 200

    # Initialize Adam moments
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0

    start_time = time.time()

    while time.time() - start_time < budget_s:
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        max_val = cross_corr.max()
        mask = np.isclose(cross_corr, max_val)
        max_indices = np.where(mask)[0]

        grad = np.zeros_like(h)
        for idx in max_indices:
            shift = idx - (n_points - 1)
            for i in range(n_points):
                j = i + shift
                if 0 <= j < n_points:
                    grad[i] += (1 - h[j]) * dx
                    grad[j] -= h[i] * dx

        # Adam update step
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        adaptive_lr = learning_rate / (np.sqrt(v_hat) + eps)
        step = adaptive_lr * m_hat

        h -= step

        # Clip and renormalize
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - T) > 1e-10:
            h *= (T / current_sum)

        # Learning rate decay
        if t % decay_steps == 0:
            learning_rate *= decay_factor

        # Early exit check every 50 iterations to save time
        if t % 50 == 0:
            if time.time() - start_time >= budget_s:
                break

    # Final evaluation
    cross_corr_final = np.correlate(h, 1 - h, mode='full') * dx
    c5_bound = cross_corr_final.max()
    return (h, c5_bound, n_points)
