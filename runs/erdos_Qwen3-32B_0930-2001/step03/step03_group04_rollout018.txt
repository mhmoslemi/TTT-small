import numpy as np
import time

def run():
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    target_sum = n_points / 2.0

    # Evaluate initial solution
    def evaluate(h):
        dx = 2.0 / n_points
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(cross_corr)
    current_C = evaluate(h_initial)

    # SA Parameters
    T = current_C * 2.0
    alpha = 0.995
    T_min = 1e-4
    current_h = h_initial.copy()
    best_h = current_h.copy()
    best_C = current_C
    start_time = time.time()
    budget_s = 1100  # time budget

    while time.time() - start_time < budget_s and T > T_min:
        # Generate perturbation
        h_candidate = current_h.copy()
        noise = np.random.normal(0, T, n_points)  # noise scaled by T
        h_candidate += noise

        # Clip to [0, 1]
        h_candidate = np.clip(h_candidate, 0, 1)
        sum_clipped = np.sum(h_candidate)
        if sum_clipped == 0:
            h_candidate = current_h.copy()
            continue
        scaling = target_sum / sum_clipped
        h_candidate *= scaling

        # Check and re-clip if necessary
        clipping_mask = (h_candidate > 1.0) | (h_candidate < 0.0)
        if clipping_mask.any():
            h_candidate_clipped = np.clip(h_candidate, 0.0, 1.0)
            sum_clipped = np.sum(h_candidate_clipped)
            if sum_clipped != 0:
                scaling = target_sum / sum_clipped
                h_candidate = h_candidate_clipped * scaling

        # After possible re-clipping and rescaling, evaluate the new candidate
        new_C = evaluate(h_candidate)

        # Acceptance criterion
        delta_E = new_C - current_C
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            current_h = h_candidate.copy()
            current_C = new_C
            # Update best solution
            if new_C < best_C:
                best_h = current_h.copy()
                best_C = new_C
        # Cool down
        T *= alpha

    return (best_h, best_C, n_points)

import time

def evaluate(h, n_points):
    dx = 2.0 / n_points
    cross_corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(cross_corr)
