import numpy as np
import scipy.optimize as optimize
import time
import copy

def compute_c5(h_vec):
    n = len(h_vec)
    dx = 2.0 / n
    conv = np.correlate(h_vec, 1 - h_vec, mode='full')
    return np.max(conv * dx)

def tournament_selection(population, fitness, tournament_size=5):
    idxs = np.random.randint(len(population), size=tournament_size)
    fitness_vals = [fitness[i] for i in idxs]
    winner_idx = idxs[np.argmin(fitness_vals)]
    return copy.deepcopy(population[winner_idx])

def crossover(parent1, parent2):
    n = len(parent1)
    crossover_point = np.random.randint(1, n)
    child = np.hstack((parent1[:crossover_point], parent2[crossover_point:])).copy()
    return child

def mutate(h, noise_std=0.05, mutation_rate=0.2):
    n = len(h)
    mask = np.random.rand(n) < mutation_rate
    noise = np.random.normal(0, noise_std, n)
    h = h.copy()
    h[mask] += noise[mask]
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum
    target_sum = n / 2.0
    current_sum = h.sum()
    adjustment = (target_sum - current_sum) / n
    h += adjustment
    h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    start_time = time.time()

    pop_size = 50
    max_generations = 500
    population = []
    for _ in range(pop_size):
        if np.random.rand() < 0.5:
            h = h_initial.copy() + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.rand(n_points)
        h = np.clip(h, 0.0, 1.0)
        h_sum = h.sum()
        adjustment = (target_sum - h_sum) / n_points
        h += adjustment
        h = np.clip(h, 0.0, 1.0)
        population.append(h)

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)

    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 10:
            break

        # Evaluate population
        fitness = np.array([compute_c5(np.ravel(ind)) for ind in population])
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]

        # Select parents
        parents = []
        for _ in range(pop_size // 2):
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            parents.append((parent1, parent2))

        # Crossover and mutation
        children = []
        for p1, p2 in parents:
            child = crossover(p1, p2)
            child = mutate(child)
            children.append(child)

        # Form new population
        population = [tournament_selection(population, fitness) for _ in range(pop_size // 3)] + children
        while len(population) < pop_size:
            population.append(mutate(population[np.random.randint(len(population))]))

    # Local optimization
    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h_flat = best_h.flatten()
    result = optimize.minimize(
        compute_c5,
        x0=best_h_flat,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100}
    )
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x

    return best_h, best_c5, n_points
