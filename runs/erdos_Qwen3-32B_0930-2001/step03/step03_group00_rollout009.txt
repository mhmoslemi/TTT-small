    np.random.seed(seed)
    start_time = time.time()

    n_points = len(h_current)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        overlaps = []
        for k in range(n_points):
            shifted = np.roll(h, shift=k)
            sum_prod = np.sum(h * shifted)
            overlap = (target_sum - sum_prod) * dx
            overlaps.append(overlap)
        return overlaps

    def find_worst_ks(overlaps, top=5):
        indices = np.argsort(overlaps)[::-1]
        return indices[:top].tolist()

    h_best = h_current.copy()
    best_c5 = np.max(compute_overlaps(h_best))

        h_var >= 0,
        h_var <= 1
