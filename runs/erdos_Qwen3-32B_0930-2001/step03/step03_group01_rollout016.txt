import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)

    def mutate(h_vec, std=0.15):
        h = h_vec.copy()
        idxs = np.random.choice(n_points, size=4, replace=False)
        for i in idxs[:2]:
            for j in idxs[2:]:
                delta = np.random.normal(0, std)
                max_add = min(1.0 - h[i], h[j])
                max_sub = min(h[i], 1.0 - h[j])
                delta = np.clip(delta, -max_sub, max_add)
                h[i] += delta
                h[j] -= delta
        return np.clip(h, 0, 1)

    pop_size = 80
    population = []
    for _ in range(pop_size):
        if _ == 0:
            population.append(h_initial.copy())
        else:
            alpha = np.ones(n_points)
            d = np.random.dirichlet(alpha)
            h = d * target_sum
            population.append(h)

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s - 5:
        current_c5 = [compute_c5(ind) for ind in population]
        curr_best_idx = np.argmin(current_c5)
        if current_c5[curr_best_idx] < best_c5:
            best_h = population[curr_best_idx].copy()
            best_c5 = current_c5[curr_best_idx]

        new_population = []
        for _ in range(pop_size // 2):
            tournament = np.random.choice(population, size=5, replace=False)
            t_c5 = [compute_c5(t) for t in tournament]
            parent = tournament[np.argmin(t_c5)].copy()

            child = mutate(parent)
            new_population.append(child)

        half_old = [ind for ind in population if np.random.rand() > 0.5]
        population = new_population + half_old
        population.append(population[curr_best_idx].copy())

    return best_h, best_c5, n_points
