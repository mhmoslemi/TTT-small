import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # Constraint: sum(h) == target_sum
    
    def compute_c5(h_vec, dx):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)
    
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Generate diversified block-based candidate pool
    num_population = 50
    candidates = [h_initial.copy()]
    
    for _ in range(num_population - 1):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Block initialization with adjustable parameters
        h_candidate = np.zeros(n_points)
        block_lengths = [10, 11]  # Sum to 21 for n=42
        max_start = n_points - sum(block_lengths)
        if max_start < 0:
            block_lengths[0] = 12
            block_lengths[1] = 9
            max_start = n_points - sum(block_lengths)
        
        start_idx = np.random.randint(0, max_start)
        h_candidate[start_idx:start_idx + block_lengths[0]] = 1.0
        h_candidate[start_idx + block_lengths[0]:start_idx + block_lengths[0] + block_lengths[1]] = 1.0
        candidates.append(h_candidate)
    
    # Gradient-based refinement with strict budget control
    for h_candidate in candidates:
        if time.time() - start_time > budget_s - 10:
            break
            
        def objective(h_vec):
            h = h_vec.reshape(n_points)
            h_clipped = np.clip(h, 0, 1)
            current_sum = np.sum(h_clipped)
            if current_sum == 0:
                return float('inf')
            
            scale_factor = target_sum / current_sum
            h_scaled = np.clip(h_clipped * scale_factor, 0, 1)
            return compute_c5(h_scaled, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(objective, h_candidate, 
                                  method='L-BFGS-B', bounds=bounds,
                                  options={'maxiter': 100, 'ftol': 1e-10})
        
        if not result.success:
            continue
            
        h_opt = result.x.copy()
        h_opt = np.clip(h_opt, 0, 1)
        sum_h = h_opt.sum()
        if not np.isclose(sum_h, target_sum):
            h_opt *= target_sum / sum_h
            h_opt = np.clip(h_opt, 0, 1)
            
        c5 = compute_c5(h_opt, dx)
        if c5 < best_c5:
            best_h = h_opt.copy()

            if sum_h == 0:
                return float('inf')
            h_scaled = h_clipped * (target_sum / sum_h)
            return compute_c5(h_scaled, dx)
        
        try:

    elapsed = time.time() - start_time
    print(f"Final C5 bound: {best_c5:.8f}, elapsed: {elapsed:.1f}s")
    return best_h, best_c5, n_points
