def get_top_lags(h, num_lags=200):
    n_points = len(h)
    dx = 2.0 / n_points
    overlaps = []
    for s in range(n_points):  # shifts from 0 to n-1
        if s >= n_points:
            continue
        if n_points - s < 1:
            continue  # no overlap
        sum_overlap = 0.0

        overlaps.append( (sum_overlap * dx, s) )
    overlaps.sort(reverse=True, key=lambda x: x[0])  # descending
    return [s for _, s in overlaps[:num_lags]]
