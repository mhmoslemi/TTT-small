import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0
    start_time = time.time()

    def compute_c5(h_vec):
        h_flat = np.clip(h_vec, 0.0, 1.0)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)

    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n)]
    h_best = h_initial.copy()
    c5_best = compute_c5(h_best)

    while time.time() - start_time < budget_s - 10:
        h0 = h_initial.copy() if np.random.rand() < 0.1 else h_best.copy()
        # Perturbation with adaptive noise
        noise_scale = 0.1 - 0.05 * (time.time() - start_time) / budget_s
        h0 += np.random.normal(0.0, noise_scale, n)
        h0 = np.clip(h0, 0.0, 1.0)
        # Ensure exact integral constraint
        total = np.sum(h0)
        if abs(total - target_sum) > 1e-6:
            h0 += (target_sum - total) / n
            h0 = np.clip(h0, 0.0, 1.0)
        # Refine with optimization
        result = optimize.minimize(
            compute_c5,
            x0=h0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        if result.success:
            candidate = np.clip(result.x, 0.0, 1.0)
            candidate_c5 = compute_c5(candidate)
            if candidate_c5 < c5_best:
                h_best[:] = candidate
                c5_best = candidate_c5
    return h_best, c5_best, n
