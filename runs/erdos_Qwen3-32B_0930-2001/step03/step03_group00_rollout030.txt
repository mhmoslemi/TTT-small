import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_current = initial_h_values.copy().astype(np.float64)
    n_points = len(h_current)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_var = cp.Variable(n_points, nonneg=True)
    C_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target_sum,
        h_var >= 0.0,
        h_var <= 1.0
    ]

    best_c5 = None
    best_h = h_current.copy()
    iteration = 0

    while time.time() - start_time < budget_s - 5:

        # Compute current overlaps to identify worst shifts
        conv = np.correlate(h_current, 1 - h_current, mode='full') * dx
        top_k_indices = np.argsort(conv)[::-1][:5]
        
        for k_index in top_k_indices:
            k_phys = k_index - (n_points - 1)
            j_array = (np.arange(n_points) - k_phys) % n_points
            linear_expr = sum(h_var[i] * (1.0 - h_current[j_array[i]]) for i in range(n_points))
            linear_expr *= dx
            constraints.append(C_var >= linear_expr)

        objective = cp.Minimize(C_var)
        problem = cp.Problem(objective, constraints)

        try:
            problem.solve(solver=cp.ECOS, max_iter=1000, verbose=False)
            if problem.status == cp.OPTIMAL:
                h_new = h_var.value
                C_new = C_var.value
                if best_c5 is None or C_new < best_c5:
                    best_c5 = C_new
                    best_h = h_new.copy()
                    h_current = h_new  # Update for next iteration
        except:
            pass

        iteration += 1
        if iteration >= 10:
            break

    # Ensure final h meets constraints
    best_h = np.clip(best_h, 0, 1)
    current_sum = best_h.sum()
    scale = target_sum / current_sum
    best_h = best_h * scale

    return best_h, best_c5, n_points
