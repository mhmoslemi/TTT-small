            if 0 <= j_plus < n:
                contrib += (1.0 - h_prev[j_plus])
            j_minus = i - a
            if 0 <= j_minus < n:
                contrib -= h_prev[j_minus]
            grad[i] = dx * contrib
        return grad

        overlaps = []

            overlaps.append((a, compute_overlap(a, h_current)))
        # Sort by overlap in descending order and select top K lags

        h_var = cp.Variable(n_points)

            constraint = overlap_prev + cp.sum(grad * (h_var - h_current)) <= t_var
            constraints.append(constraint)

        objective = cp.Minimize(t_var)
        problem = cp.Problem(objective, constraints)

        h_current = new_h.copy()
