import numpy as np
import time

def run():
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Adam parameters
    learning_rate = 1e-3
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    step = 0

    t0 = time.time()
    max_time = 1100

    while time.time() - t0 < max_time - 1:  # leave buffer
        # Compute cross-correlation
        cor = np.correlate(h, 1 - h, mode='full') * dx
        # Find max and indices
        current_max = np.max(cor)
        max_indices = np.where(cor == current_max)[0]

        # Compute gradient from max indices
        grad = np.zeros_like(h, dtype=np.float64)
        for m_idx in max_indices:
            k = m_idx - (n_points - 1)
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    grad[i] += (1 - h[j]) * dx
                    grad[j] -= h[i] * dx

        # Adam update
        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        delta = -learning_rate * m_hat / (np.sqrt(v_hat) + eps)

        # Update h
        h += delta
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Renormalize sum
        s = np.sum(h)
        if s != 0:
            h *= target_sum / s

        # Adaptive learning rate
        # Decay learning rate every certain steps to fine-tune
        if step % 100 == 0:
            learning_rate *= 0.9

    # Final evaluation
    cor = np.correlate(h, 1 - h, mode='full') * dx
    best_c5 = np.max(cor)
    return (h, best_c5, n_points)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    step = 0

    t0 = time.time()
    while time.time() - t0 < 1100:
        # Compute cross-correlation and max overlap
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_max = np.max(corr)
        max_indices = np.where(corr == current_max)[0]

        # Compute gradient
        grad = np.zeros_like(h)
        for m_idx in max_indices:
            k = m_idx - (n_points - 1)  # Convert index to actual shift
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    grad[i] += (1 - h[j]) * dx
                    grad[j] -= h[i] * dx

        # Adam update step
        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**step)
        v_hat = v / (1 - beta2**step)
        h -= learning_rate * m_hat / (np.sqrt(v_hat) + eps)

        # Enforce constraints
        h = np.clip(h, 0, 1)
        s = h.sum()
        if abs(s - target_sum) > 1e-10 and s != 0:
            h *= target_sum / s

        # Learning rate decay
        if step % 50 == 0:
            learning_rate *= 0.95

    # Final evaluation
    corr = np.correlate(h, 1 - h, mode='full') * dx
    best_c5 = np.max(corr)
    return (h, best_c5, n_points)
