import numpy as np
import time

def run():
    h = initial_h_values.copy().astype(np.float64)
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0

    # Adam parameters
    learning_rate = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr_decay_factor = 0.9
    decay_interval = 100

    # Initialize Adam variables
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    start_time = time.time()
    for iteration in range(5000):  # Large number, time-limited
        if time.time() - start_time > 1100:
            break

        # Compute cross-correlation and max overlaps
        cross_corr = np.correlate(h, 1 - h, 'full') * dx
        current_max = np.max(cross_corr)

        # Find all shifts contributing to current maximum
        max_indices = np.where(np.isclose(cross_corr, current_max, atol=1e-8))[0]

        # Compute gradient from all maximal contributors
        grad = np.zeros_like(h)
        for l in max_indices:
            k_shift = l - (n - 1)
            for i in range(n):
                j = i + k_shift
                if 0 <= j < n:
                    grad[i] += (1 - h[j]) * dx
                    grad[j] -= h[i] * dx

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = -learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)

        h += step
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum > 1e-12:
            h *= target_sum / h_sum

        # Learning rate decay
        if iteration % decay_interval == 0 and iteration > 0:
            learning_rate *= lr_decay_factor

    # Final evaluation
    c5_bound = np.max(np.correlate(h, 1 - h, 'full') * dx)
    return (h, c5_bound, n)
