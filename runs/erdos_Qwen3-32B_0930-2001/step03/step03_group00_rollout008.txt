import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    global initial_h_values
    h_current = initial_h_values.copy()
    n_points = len(h_current)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(x):
        h_trunc = x
        h = np.zeros(n_points)
        h[:n_points - 1] = h_trunc
        h[-1] = target_sum - np.sum(h_trunc)
        if not (0 <= h[-1] <= 1):
            return 1e6  # Penalize invalid solutions
        h_clamped = np.clip(h, 0, 1)
        if not np.allclose(np.sum(h_clamped), target_sum):
            # Re-normalize if necessary
            scale = target_sum / max(1e-8, np.sum(h_clamped))
            h_clamped = np.clip(h_clamped * scale, 0, 1)
        conv = np.correlate(h_clamped, 1 - h_clamped, mode='full')
        return np.max(conv * dx)

    bounds = [(0, 1)] * (n_points - 1)
    result = opt.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        seed=seed,
        maxiter=50,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.7,
        polish=False,
        workers=1
    )

    h_opt = np.zeros(n_points)
    h_opt[:n_points - 1] = result.x
    h_opt[-1] = target_sum - np.sum(result.x)
    h_opt = np.clip(h_opt, 0, 1)
    final_sum = np.sum(h_opt)
    if not np.isclose(final_sum, target_sum):
        h_opt *= target_sum / max(1e-8, final_sum)
    
    # Final validation
    conv = np.correlate(h_opt, 1 - h_opt, mode='full')
    best_c5 = np.max(conv * dx)
    elapsed = time.time() - start_time

    print(f"Elapsed: {elapsed:.1f}s, Best C5: {best_c5:.6f}")
    return h_opt, best_c5, n_points
