import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    def objective_nelder(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        sum_h = h.sum()
        if sum_h == 0:
            return float('inf')
        corrected_scale = target_sum / sum_h
        h_corrected = h * corrected_scale
        h_corrected = np.clip(h_corrected, 0, 1)
        c5_val = compute_c5(h_corrected)
        sum_error = h_corrected.sum() - target_sum
        penalty = 1e5 * sum_error**2
        return c5_val + penalty

    bounds_slsqp = [(0, 1) for _ in range(n_points)]
    constraints_slsqp = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)

    num_perturbations = 50
    max_perturbation = 0.2
    min_perturbation = 0.05

    for i in range(num_perturbations):
        if time.time() - start_time > budget_s - 10:
            break
        # Decreasing perturbation magnitude as iterations progress
        pert_magnitude = max_perturbation - (max_perturbation - min_perturbation) * (i / num_perturbations)
        pert = np.random.randn(n_points) * pert_magnitude
        h_pert = np.clip(h_initial + pert, 0, 1)
        # Nelder-Mead with more iterations
        res_nelder = optimize.minimize(objective_nelder, h_pert, method='Nelder-Mead',
                                       options={'maxiter': 200, 'xatol': 1e-5})
        if not res_nelder.success:
            continue
        # SLSQP with higher iterations and tighter tolerance
        res_slsqp = optimize.minimize(compute_c5, res_nelder.x, method='SLSQP',
                                     bounds=bounds_slsqp, constraints=constraints_slsqp,
                                     options={'maxiter': 300, 'ftol': 1e-10})
        if res_slsqp.success:
            current_c5 = compute_c5(res_slsqp.x)
            if current_c5 < best_c5:
                best_h = res_slsqp.x.copy()
                best_c5 = current_c5

    if time.time() - start_time < budget_s - 10:
        # Extended final refinement
        res_refine = optimize.minimize(compute_c5, best_h, method='SLSQP',
                                      bounds=bounds_slsqp, constraints=constraints_slsqp,
                                      options={'maxiter': 800, 'ftol': 1e-12})
        if res_refine.success and compute_c5(res_refine.x) < best_c5:
            best_h = res_refine.x.copy()
            best_c5 = compute_c5(best_h)

    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Best C5: {best_c5:.8f}")
    return best_h, best_c5, n_points
