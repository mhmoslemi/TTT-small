        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if current_sum == 0:
            return np.zeros_like(h)

        scaled = np.clip(scaled, 0, 1)
        final_sum = scaled.sum()
        if abs(final_sum - target_sum) > 1e-6:
            diff = target_sum - final_sum
            idx = np.argmin(scaled)
            scaled[idx] += diff
            scaled = np.clip(scaled, 0, 1)
        return scaled
    
    n_particles = 30

    personal_best = [p.copy() for p in particles]
