                            + c2 * r2 * (gbest - particles[i]))
            velocities[i] = np.clip(velocities[i], -velocity_clamp, velocity_clamp)
            particles[i] += velocities[i]
            
            # Feasibility constraints
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            current_sum = particles[i].sum()
            delta = (target_sum - current_sum) / n
            particles[i] += delta
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            
            # Update personal best
            score = evaluate_erdos_solution(particles[i])
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
                if score < gbest_score:
                    gbest[:] = particles[i].copy()
                    gbest_score = score
        
        # Update no improvement counter
        if gbest_score == prev_score:
            no_improve_counter += 1
        else:
            no_improve_counter = 0
        
        # Adaptive inertia weight
        if no_improve_counter >= 100:
            w = 0.5
            velocities = np.random.uniform(-0.05, 0.05, (num_particles, n))
            no_improve_counter = 0
        
        # Exit early if budget nearly exhausted
        if time.time() - start_time > budget_s - 10:
            break
    
    # Trust-region refinement on top candidates
    top_idx = np.argsort(pbest_scores)[:5]
    for idx in top_idx:
        if time.time() - start_time > budget_s - 2:
            break
        def objective(x):
            return evaluate_erdos_solution(x.reshape(n))
        x0 = pbest[idx].copy()
        result = opt.minimize(
            objective, x0=x0, method='SLSQP',
            bounds=[(0,1)] * n,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
            options={'maxiter': 60, 'ftol': 1e-12}
        )
        if result.success and result.fun < gbest_score:
            gbest[:] = result.x.reshape(n).copy()
            gbest_score = result.fun
    
    # Final normalization
    gbest = np.clip(gbest, 0.0, 1.0)
    delta = (target_sum - gbest.sum()) / n
    gbest += delta
    gbest = np.clip(gbest, 0.0, 1.0)
    
    return gbest, gbest_score, n
