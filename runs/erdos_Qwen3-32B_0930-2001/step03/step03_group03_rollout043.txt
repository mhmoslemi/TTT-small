    bounds = [(0.0, 1.0) for _ in range(n)]
    h_best = h_initial.copy()
    c5_best = compute_c5(h_best)
    
    # Number of restarts within the budget
    while time.time() - start_time < budget_s - 10:
        # Perturb initial guess
        h0 = h_best.copy()
        h0 += np.random.normal(0.0, 0.05, n)
        h0 = np.clip(h0, 0.0, 1.0)
        current_sum = np.sum(h0)
        adjustment = (target_sum - current_sum) / n
        h0 = np.clip(h0 + adjustment, 0.0, 1.0)
        
        result = optimize.minimize(
            compute_c5,
            x0=h0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 50, 'ftol': 1e-10}
        )
        if result.success:
            cand_c5 = result.fun
            if cand_c5 < c5_best:
                h_best[:] = result.x
                c5_best = cand_c5
    
    return h_best, c5_best, n
