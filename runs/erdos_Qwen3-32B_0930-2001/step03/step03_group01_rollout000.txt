            i, j = np.random.choice(zeros), np.random.choice(ones)
            binary_guess[i], binary_guess[j] = binary_guess[j], binary_guess[i]
        # Add small Gaussian noise
        noise_std = np.random.uniform(0.01, 0.1)
        noise = np.random.normal(0, noise_std, size=n_points)
        perturbed = binary_guess + noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        # Adjust sum constraint
        current_sum = np.sum(perturbed)
        adjustment = (target_sum - current_sum) / n_points
        perturbed += adjustment
        perturbed = np.clip(perturbed, 0.0, 1.0)
        return perturbed
    
    while time.time() - start_time < budget_s - 5:
        h0 = generate_initial_guess(h_best)
        result = optimize.minimize(
            compute_c5,
            x0=h0,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 100, 'ftol': 1e-11, 'disp': False}
        )
        
        if result.success:
            c5_candidate = result.fun
            h_candidate = result.x
            if c5_candidate < c5_best:
                h_best[:] = h_candidate
                c5_best = c5_candidate
    
    return h_best, c5_best, n_points
