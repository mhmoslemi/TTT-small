import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0
    start_time = time.time()

    def adjust_sum(h_vec, target_sum):
        current_sum = np.sum(h_vec)
        delta = (target_sum - current_sum) / len(h_vec)
        h_adj = np.clip(h_vec + delta, 0.0, 1.0)
        return h_adj

    def compute_c5(h_vec):
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(conv * dx)

    def evaluate_candidate(h_vec):
        return compute_c5(h_vec)

    population_size = 20
    mutation_rate = 0.3
    tournament_size = 5
    population = [h_initial.copy()]
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n)
        candidate = adjust_sum(h_rand, target_sum)
        population.append(candidate)

    best_c5 = float('inf')
    best_h = h_initial.copy()

    while time.time() - start_time < budget_s - 50:
        scores = np.array([evaluate_candidate(h) for h in population])
        improved = False
        for i, sc in enumerate(scores):
            if sc < best_c5:
                best_c5 = sc
                best_h[:] = population[i].copy()
                improved = True
        if not improved:
            break

        next_gen = []
        top_indices = np.argsort(scores)[:max(1, int(population_size * 0.1))]
        next_gen.extend([population[i].copy() for i in top_indices])
        while len(next_gen) < population_size:
            p1 = population[np.argmin([scores[i] for i in np.random.choice(len(population), tournament_size)])]
            p2 = population[np.argmin([scores[i] for i in np.random.choice(len(population), tournament_size)])]
            split = np.random.randint(1, n)
            c1 = np.concatenate([p1[:split], p2[split:]])
            c2 = np.concatenate([p2[:split], p1[split:]])
            c1 = adjust_sum(c1, target_sum)
            c2 = adjust_sum(c2, target_sum)
            c1 += np.random.normal(0, 0.1, n) * (np.random.rand(n) < mutation_rate)
            c2 += np.random.normal(0, 0.1, n) * (np.random.rand(n) < mutation_rate)
            c1 = adjust_sum(c1, target_sum)
            c2 = adjust_sum(c2, target_sum)
            next_gen.extend([c1, c2])
        population = next_gen[:population_size]

    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0)] * n
    result = optimize.minimize(
        compute_c5,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 60, 'ftol': 1e-10}
    )
    if result.success:
        h_refined = adjust_sum(result.x, target_sum)
        refined_c5 = compute_c5(h_refined)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h[:] = h_refined

    return best_h, best_c5, n
