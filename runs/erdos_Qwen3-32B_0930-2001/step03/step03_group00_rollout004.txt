import numpy as np
import time
import scipy.optimize as opt

def clip_and_scale(h_vec, target_sum):
    h = np.clip(h_vec, 0, 1)
    n_points = len(h)
    current_sum = h.sum()
    if current_sum == 0:
        h = np.full_like(h, target_sum / n_points)
    else:
        h = np.clip(h * (target_sum / current_sum), 0, 1)
    return h

def compute_c5(h_vec, n_points, dx):
    h = h_vec.reshape(n_points)
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv * dx)

def constraint_sum(x, target_sum):
    return x.sum() - target_sum
