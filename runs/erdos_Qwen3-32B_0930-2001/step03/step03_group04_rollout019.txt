def evaluate_objective(h, n_points):

    cross_corr = np.correlate(h, 1 - h, mode='full') * dx

        max_val = np.max(cross_corr)

        # Update best solution

            best_c5 = current_c5
            best_h = h.copy()

        # Compute active indices
        active_indices = np.where(cross_corr == max_val)[0]

        # Compute gradient
        grad = np.zeros(n_points)
        for idx in active_indices:
            k_shift = idx - (n_points - 1)  # Convert index to shift
            for i in range(n_points):
                ipos = (i + k_shift) % n_points
                jpos = (i - k_shift) % n_points
                contrib = (1 - h[ipos] - h[jpos])

        t += 1
        m = beta1 * m + (1 - beta1) * grad

        h -= learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
