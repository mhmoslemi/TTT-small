            c5 = np.max(conv * dx)
            population_c5.append(c5)
        # Update best solution
        for i in range(len(population_c5)):
            if population_c5[i] < best_c5:
                best_c5 = population_c5[i]
                best_h = population[i].copy()
        # Select parents: top 50% by fitness
        parent_indices = np.argsort(population_c5)[:int(0.5 * population_size)]
        parents = [population[i] for i in parent_indices]
        # Create new population
        new_population = parents.copy()
        while len(new_population) < population_size:
            # Randomly choose parents
            p1, p2 = np.random.choice(parents, 2)
            # Crossover at random point
            split_point = np.random.randint(1, n-1)
            child1 = np.hstack((p1[:split_point], p2[split_point:]))
            child2 = np.hstack((p2[:split_point], p1[split_point:]))
            # Apply mutation
            children = [child1, child2]
            for child in children:
                if np.random.rand() < mutation_rate:
                    start = np.random.randint(0, n)
                    length = np.random.randint(1, min(6, n - start))
                    end = start + length
                    child[start:end] = np.random.rand(length)
                # Adjust for constraints
                child = np.clip(child, 0, 1)
                current_sum = child.sum()
                adjustment = (target_sum - current_sum) / n
                child += adjustment
                child = np.clip(child, 0, 1)
                new_population.append(child)
                if len(new_population) >= population_size:
                    break
        population = new_population[:population_size]

    # Final refinement with SLSQP
    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)
    def constraint_sum(x):
        return np.sum(x) - target_sum
    bounds = [(0.0, 1.0) for _ in range(n)]
    result = optimize.minimize(
        compute_c5,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 60, 'ftol': 1e-10}
    )
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    return best_h, best_c5, n
