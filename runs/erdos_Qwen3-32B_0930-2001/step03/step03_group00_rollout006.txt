import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_init = initial_h_values.copy()
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    bounds = [(0, 1) for _ in range(n_points)]

    def objective_de(h_vec):
        h = h_vec.reshape(n_points)
        current_sum = np.sum(h)
        c5 = compute_c5(h)
        penalty = 1e4 * ((current_sum - target_sum)**2)
        return c5 + penalty

    # Differential Evolution with time check
    de_maxiter = 15
    result_de = opt.differential_evolution(
        objective_de, 
        bounds=bounds,
        popsize=5,
        mutation=(0.5, 1.0),
        recombination=0.9,
        maxiter=de_maxiter,
        tol=1e-4,
        seed=seed
    )
    h_de = result_de.x.reshape(n_points)
    
    # Store best so far
    h_best = h_de.copy()
    best_c5 = compute_c5(h_best)

    # Refine with SLSQP
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    cons = ({'type': 'eq', 'fun': constraint_sum})
    result_slsqp = opt.minimize(
        compute_c5,
        h_best,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 300, 'ftol': 1e-10}
    )
    if result_slsqp.success:
        h_best = result_slsqp.x
        best_c5 = compute_c5(h_best)

    # Final clipping and normalization
    h_best = np.clip(h_best, 0, 1)
    current_sum = np.sum(h_best)
    if abs(current_sum - target_sum) > 1e-6:
        h_best = h_best * target_sum / current_sum
        h_best = np.clip(h_best, 0, 1)

    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Best C5: {best_c5:.8f}")
    return h_best, best_c5, n_points
