        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    best_h = h_current.copy()
    
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target_sum
    ]
    objective = cp.Minimize(C)

        corr = np.correlate(h_current, 1 - h_current, mode='full')
        n_corr = corr.size
        k_list = [i - n_points + 1 for i in range(n_corr)]
        values = corr * dx
        
        # Pair values with k and filter relevant shifts
        paired = sorted(zip(values.tolist(), k_list), key=lambda x: -x[0])
