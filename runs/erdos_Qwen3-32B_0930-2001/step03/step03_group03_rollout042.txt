import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0
    start_time = time.time()

    # Precompute valid shifts
    valid_shifts = []
    for k in range(1, n):
        for i in range(n - k):
            valid_shifts.append((i, k))
    num_shifts = len(valid_shifts)

    # Map each k to corresponding shift indices
    k_to_indices = {}
    for idx, (i, k) in enumerate(valid_shifts):
        if k not in k_to_indices:
            k_to_indices[k] = []
        k_to_indices[k].append(idx)

    # LP variables and constraints
    h = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    z = cp.Variable(num_shifts)
    
    constraints = [
        cp.sum(h) == target_sum,
        h <= 1.0
    ]
    
    for idx, (i, k) in enumerate(valid_shifts):
        j = i + k
        constraints += [
            z[idx] >= h[i],
            z[idx] >= 1 - h[j],
            z[idx] <= 1.0
        ]
    
    for k in k_to_indices:
        indices = k_to_indices[k]
        constraints.append(cp.sum(z[indices]) * dx <= t)
    
    problem = cp.Problem(cp.Minimize(t), constraints)
    
    # Solve LP relaxation
    try:
        problem.solve(solver=cp.GLPK, verbose=False)
        if problem.status == 'optimal':
            h_lp = np.clip(h.value, 0.0, 1.0)
        else:
            h_lp = h_initial.copy()
    except:
        h_lp = h_initial.copy()
    
    # Adjustment to ensure exact integral constraint
    h_lp = np.clip(h_lp, 0.0, 1.0)
    current_sum = np.sum(h_lp)
    adjustment = (target_sum - current_sum) / n
    h_lp = np.clip(h_lp + adjustment, 0.0, 1.0)
    
    # Local optimization
    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)
    
    def constraint_sum(x):
        return np.sum(x) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n)]
    h_best = h_lp.copy()
    c5_best = compute_c5(h_best)
    
    while time.time() - start_time < budget_s - 10:
        if h_best.size == n:
            h0 = h_best.copy()
        else:
            h0 = h_lp.copy()
        h0 += np.random.normal(0.0, 0.02, n)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 += (target_sum - np.sum(h0)) / n
        h0 = np.clip(h0, 0.0, 1.0)
        result = optimize.minimize(
