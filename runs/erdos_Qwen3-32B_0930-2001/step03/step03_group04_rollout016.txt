import numpy as np
import time

def run():
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    T = n_points / 2.0

    # Adam optimizer parameters
    lr = 1e-3
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0

    start_time = time.time()
    best_c5 = np.inf
    best_h = h.copy()

    while time.time() - start_time < 1100:
        t += 1

        # Compute cross-correlation and current max
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(cross_corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Find contributing shifts
        max_mask = np.isclose(cross_corr, current_c5, atol=1e-6)
        contributing_shifts = np.where(max_mask)[0]

        # Compute gradient
        grad = np.zeros(n_points)
        for shift_index in contributing_shifts:
            actual_shift = shift_index - (n_points - 1)
            if actual_shift > 0:
                for i in range(n_points - actual_shift):
                    j = i + actual_shift
                    grad[i] += (1 - h[j]) * dx
                    grad[j] += -h[i] * dx
            elif actual_shift < 0:
                actual_shift_abs = -actual_shift
                for i in range(actual_shift_abs, n_points):
                    j = i - actual_shift_abs
                    grad[i] += (1 - h[j]) * dx
                    grad[j] += -h[i] * dx
            else:
                for i in range(n_points):
                    grad[i] += (1 - 2 * h[i]) * dx

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)

        # Apply step and project
        h += step
        h = np.clip(h, 0, 1)
        sum_h = h.sum()
        if not np.isclose(sum_h, T, rtol=1e-6):
            if sum_h != 0:
                h *= T / sum_h

        # Learning rate decay
        if t % 100 == 0:
            lr *= 0.9

    return (best_h, best_c5, n_points)
