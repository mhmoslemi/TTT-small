import numpy as np
import time

def evaluate_objective(h, n_points):
    dx = 2.0 / n_points
    cross_corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(cross_corr)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    current_h = initial_h_values.copy()
    n_points = len(current_h)
    current_cost = evaluate_objective(current_h, n_points)
    best_h = current_h.copy()
    best_cost = current_cost

    T = 2.0 * current_cost
    alpha = 0.995
    T_min = 1e-4

    start_time = time.time()

    while time.time() - start_time < budget_s and T > T_min:
        # Perturb a subset of indices
        new_h = current_h.copy()
        num_perturb = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        sigma = T
        noise = np.random.normal(0, sigma, num_perturb)
        new_h[indices] += noise
        # Enforce constraints
        new_h = np.clip(new_h, 0, 1)
        total = np.sum(new_h)
        if total != 0:
            new_h *= (n_points / 2.0) / total
        new_cost = evaluate_objective(new_h, n_points)
        delta_E = new_cost - current_cost

        # Metropolis criterion
        if delta_E < 0 or np.exp(-delta_E / T) > np.random.rand():
            current_h = new_h.copy()
            current_cost = new_cost

        # Update best solution
        if current_cost < best_cost:
            best_h = current_h.copy()
            best_cost = current_cost

        T *= alpha

    return (best_h, best_cost, n_points)
