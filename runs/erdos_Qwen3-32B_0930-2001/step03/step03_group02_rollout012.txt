import numpy as np
import scipy.optimize as optimize
import time

def generate_block_candidate(n_points, target_sum):
    h = np.zeros(n_points)
    n_blocks = np.random.randint(1, 4)
    for _ in range(n_blocks):
        start = np.random.randint(0, n_points)
