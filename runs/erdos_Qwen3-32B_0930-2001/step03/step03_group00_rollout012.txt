        return np.max(conv * dx)

    def finalize_h(h):
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if current_sum == 0:
            h = np.full_like(h, target_sum / len(h))
        else:
            scale = target_sum / current_sum
            h = np.clip(h * scale, 0, 1)
        return h

    h_best = finalize_h(h_current)
    best_c5 = compute_c5(h_best)
    
    iteration = 0

        pert = np.random.randn(n_points) * noise_scale
