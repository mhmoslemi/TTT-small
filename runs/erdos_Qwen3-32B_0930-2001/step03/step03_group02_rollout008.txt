import numpy as np
import time
import scipy.optimize as optimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    def project(h):
        h = np.clip(h, 0, 1)
        s = h.sum()
        if s == 0:
            return h
        scale = target_sum / s
        h = h * scale
        h = np.clip(h, 0, 1)
        return h

    # PSO Parameters
    n_particles = 20
    w = 0.7
    c1 = 2.0
    c2 = 2.0

    # Initialize swarm
    positions = np.zeros((n_particles, n_points))
    positions[0] = project(h_initial + 0.1 * np.random.randn(n_points))
    for i in range(1, n_particles):
        positions[i] = project(h_initial + 0.2 * np.random.randn(n_points) * (1 - i / (n_particles * 2)))

    velocities = 0.1 * np.random.randn(n_particles, n_points)
    personal_best_pos = positions.copy()
    personal_best_scores = np.array([compute_c5(p) for p in positions])
    best_idx = np.argmin(personal_best_scores)
    global_best_pos = positions[best_idx].copy()
    global_best_score = personal_best_scores[best_idx]

    max_iter = 100
    iter_time = 5.0  # seconds per iteration
    t_start = time.time()

    while time.time() - t_start < budget_s * 0.8:
        for i in range(n_particles):
            if time.time() - start_time > budget_s - 5:
                break
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (personal_best_pos[i] - positions[i]) +
                             c2 * r2 * (global_best_pos - positions[i]))
            positions[i] += velocities[i]
            positions[i] = project(positions[i])
            current_score = compute_c5(positions[i])
            if current_score < personal_best_scores[i]:
                personal_best_scores[i] = current_score
                personal_best_pos[i] = positions[i].copy()
                if current_score < global_best_score:
                    global_best_score = current_score
                    global_best_pos = positions[i].copy()

    # Final refinement
    best_h = global_best_pos
    if time.time() - start_time < budget_s - 5:
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

                                constraints=constraints, options={'maxiter': 300, 'ftol': 1e-10})

            if refined_score < global_best_score:
                best_h = res.x
                global_best_score = refined_score

    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Best C5: {global_best_score:.6f}")
    return best_h, global_best_score, n_points
