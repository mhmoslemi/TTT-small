import time

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()
    
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

        conv = np.correlate(h, 1 - h, mode='full')

        scale_factor = target_sum / current_sum
        h_scaled = h * scale_factor
        h_scaled = np.clip(h_scaled, 0, 1)
        return h_scaled

    for _ in range(60):
        if time.time() - start_time > budget_s - 15:
            break
        candidates.append(generate_block_candidate())

    bounds = [(0, 1)] * n_points
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})
    best_c5, best_h = float('inf'), h_initial.copy()

    for idx, candidate in enumerate(candidates):
        if time.time() - start_time > budget_s - 5:
            break

            res = optimize.minimize(
