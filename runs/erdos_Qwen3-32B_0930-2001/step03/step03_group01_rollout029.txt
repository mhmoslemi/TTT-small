import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

        return np.max(conv * dx)
    
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    
    current_sum = 21.5  # Initial relaxed constraint
    sum_step = 0.02  # Constraint tightening per iteration
    max_iter_per_cont = 40  # Iterations per continuation stage
    
    while time.time() - start_time < budget_s - 5:
        if current_sum <= target_sum + 1e-6:
            break
            
        def adjust_initial_guess(prev_h, new_sum):
            h0 = prev_h.copy()
            delta = (new_sum - np.sum(h0)) / n_points
            h0 += delta
            h0 = np.clip(h0, 0.0, 1.0)

                h0 *= scale
                h0 = np.clip(h0, 0.0, 1.0)
            return h0
        
        h0 = adjust_initial_guess(h_best, current_sum)
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - current_sum}]
        
        result = optimize.minimize(
            compute_c5,
            x0=h0,
            method='COBYLA',
            bounds=bounds,
            constraints=constraints,

        current_sum = max(current_sum - sum_step, target_sum)
