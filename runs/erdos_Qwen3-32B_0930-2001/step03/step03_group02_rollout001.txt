        val2 = 0.5 * val1
        h[:split] = val1
        h[split:] = val2
        h = np.clip(h, 0, 1)
        candidates.append(h)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    
    for i, candidate in enumerate(candidates):
        if time.time() - start_time > budget_s - 30:
            break
        res = optimize.minimize(compute_c5, candidate, method='SLSQP', 
                               bounds=bounds, constraints=constraints,
                               options={'maxiter': 100, 'ftol': 1e-12})
        if res.success:
            h_test = res.x
            c5_test = compute_c5(h_test)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()
    
    # Final refinement
    if time.time() - start_time < budget_s - 5:
        res_final = optimize.minimize(compute_c5, best_h, method='SLSQP', 
                                    bounds=bounds, constraints=constraints,
                                    options={'maxiter': 200, 'ftol': 1e-14})
        if res_final.success:
            best_h = res_final.x
            best_c5 = compute_c5(best_h)
    
    elapsed = time.time() - start_time
    print(f"Time used: {elapsed:.1f}, Best C₅: {best_c5:.8f}")
    return best_h, best_c5, n_points
