            if current_obj < best_obj:
                best_obj = current_obj
                best_h = res.x.copy()

    if time.time() - start_time < budget_s - 5:
        res_refine = optimize.minimize(objective, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, options={'maxiter': 200})
        if res_refine.success and objective(res_refine.x) < best_obj:
            best_h = res_refine.x.copy()
            best_obj = objective(best_h)

    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Best C5: {best_obj:.6f}")
    return best_h, best_obj, n_points
