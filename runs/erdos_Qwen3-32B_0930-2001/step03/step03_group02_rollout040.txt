import numpy as np
import scipy.optimize as optimize
import time

def project(h_vec, target_sum):
    h = h_vec.copy()
    h_clipped = np.clip(h, 0, 1)
    current_sum = h_clipped.sum()
    if current_sum == 0:
        scale = target_sum / len(h_clipped)
        return np.full_like(h_clipped, scale)
    scale = target_sum / current_sum
    h_scaled = h_clipped * scale
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled

    np.random.seed(seed)
    start_time = time.time()
    
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # Constraint: sum(h) == target_sum

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    # PSO parameters
    n_particles = 20
    w = 0.7
    c1 = 2.0
    c2 = 2.0

    # Initialize particles
    particles = []
    velocities = []
    personal_best = []
    personal_best_c5 = []
    for _ in range(n_particles):
        if _ == 0:
            h = h_initial.copy()
        else:
            h = h_initial + np.random.randn(n_points) * 0.1
        h = project(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.random.randn(n_points) * 0.05)
        personal_best.append(h.copy())
        personal_best_c5.append(compute_c5(h))

    global_best = personal_best[0].copy()
    global_best_c5 = personal_best_c5[0]

    for i in range(1, n_particles):
        if personal_best_c5[i] < global_best_c5:
            global_best = personal_best[i].copy()
            global_best_c5 = personal_best_c5[i]

    max_budget = budget_s * 0.9  # Leave time for final refinement

    while time.time() - start_time < max_budget:
