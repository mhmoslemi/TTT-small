import scipy.optimize as optimize
import time

def adjust_guess_to_sum(h_vec, desired_sum):
    current_sum = np.sum(h_vec)
    if abs(current_sum - desired_sum) < 1e-8:
        return h_vec

    if delta > 0:
        increase_space = 1.0 - h_vec
        increase_amount = np.sum(increase_space)

            return h_vec
        prop = delta / increase_amount
        add = increase_space * prop
        new_h = h_vec + add
    else:

        if abs(decrease_amount) < 1e-8:
            return h_vec
        prop = -delta / decrease_amount
        sub = decrease_space * prop
        new_h = h_vec - sub
    new_h = np.clip(new_h, 0.0, 1.0)
    return new_h
