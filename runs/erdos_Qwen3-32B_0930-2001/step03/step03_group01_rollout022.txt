            h_binary_current[i0], h_binary_current[i1] = 1.0, 0.0

        h0 = h_binary_current.copy()
        # Run optimization
        result = optimize.minimize(
            compute_c5,
            x0=h0,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 150, 'ftol': 1e-11, 'disp': False}
        )

        if result.success:
            c5_candidate = result.fun
            h_candidate = result.x
            if c5_candidate < c5_best:
                h_best[:] = h_candidate.copy()
                c5_best = c5_candidate
                best_h_binary = get_binary_version(h_candidate)

    return h_best, c5_best, n_points
