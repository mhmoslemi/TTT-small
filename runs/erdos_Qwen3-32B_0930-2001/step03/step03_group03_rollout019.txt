import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2.0
    start_time = time.time()

    def compute_c5(h_vec):
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(conv * dx)

    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n)]
    h_best = h_initial.copy()
    c5_best = compute_c5(h_best)

    # Ensure h_best satisfies constraints
    current_sum = np.sum(h_best)
    if not np.isclose(current_sum, target_sum, atol=1e-6):
        adjustment = (target_sum - current_sum) / n
        h_best += adjustment
        h_best = np.clip(h_best, 0.0, 1.0)

    while time.time() - start_time < budget_s - 10:
        h0 = h_best.copy()
        # Perturb with larger std to escape local minima
        h0 += np.random.normal(0.0, 0.2, n)
        h0 = np.clip(h0, 0.0, 1.0)
        current_sum = np.sum(h0)
        adjustment = (target_sum - current_sum) / n
        h0 += adjustment
        h0 = np.clip(h0, 0.0, 1.0)

        result = optimize.minimize(
            compute_c5,
            x0=h0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=({'type': 'eq', 'fun': constraint_sum},),
            options={'maxiter': 50, 'ftol': 1e-12}
        )

        if result.success:
            new_h = np.clip(result.x, 0.0, 1.0)
            new_c5 = compute_c5(new_h)
            if new_c5 < c5_best:
                h_best[:] = new_h
                c5_best = new_c5

    return h_best, c5_best, n
