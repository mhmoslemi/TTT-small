        prob.solve(solver=cp.ECOS, max_iter=1000, eps=1e-12)
    except:
        pass
    
    h_qp_val = np.clip(h_qp.value if h_qp.value is not None else h_initial.copy(), 0, 1)
    h_qp_sum = h_qp_val.sum()
    if h_qp_sum > 0:
        scale = target_sum / h_qp_sum
        h_qp_val = np.clip(h_qp_val * scale, 0, 1)
    
    best_h = h_qp_val.copy()
    best_c5 = compute_c5(best_h)
    
    # SLSQP refinement and perturbations
    bounds = [(0, 1) for _ in range(n_points)]
    constraints_slsqp = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    
    # Initial refinement
    if time.time() - start_time < budget_s - 100:
        res_refine = optimize.minimize(
            lambda x: compute_c5(x),
            best_h,
            bounds=bounds,
            method='SLSQP',
            constraints=constraints_slsqp,
            options={'maxiter': 300, 'ftol': 1e-10}
        )
        if res_refine.success:
            c5_refined = compute_c5(res_refine.x)
            if c5_refined < best_c5:
                best_h = res_refine.x.copy()
                best_c5 = c5_refined
    
    # Perturbative optimization
    pert_magnitude = 0.15
    num_pert = 15
    
    for _ in range(num_pert):
        if time.time() - start_time > budget_s - 20:
            break
        # Symmetric perturbation
        pert = np.random.randn(n_points) * pert_magnitude
        h_pert = np.clip(h_qp_val + pert, 0, 1)
        res = optimize.minimize(
            lambda x: compute_c5(x),
            h_pert,
            bounds=bounds,
            method='SLSQP',
            constraints=constraints_slsqp,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if res.success:
            c5 = compute_c5(res.x)
            if c5 < best_c5:
                best_h = res.x.copy()
                best_c5 = c5
    
    # Final projection to ensure constraints
    def project_h(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum != 0:
            h *= target_sum / h_sum
        return np.clip(h, 0, 1)
    
    best_h = project_h(best_h)
    
    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Best C5: {best_c5:.6f}")
    
    return best_h, best_c5, n_points
