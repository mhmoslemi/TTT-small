import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum_int = int(n_points / 2)  # 21 for n=42
    
    # Convert initial to binary with exact 21 ones
    sorted_indices = np.argsort(h_initial)[::-1]
    h_binary = np.zeros(n_points, dtype=int)
    h_binary[sorted_indices[:target_sum_int]] = 1
    
    population_size = 100
    population = [h_binary]
    for _ in range(population_size - 1):
        idxs = np.random.choice(n_points, size=target_sum_int, replace=False)
        candidate = np.zeros(n_points, dtype=int)
        candidate[idxs] = 1
        population.append(candidate)
    
    def compute_c5(h_vec):
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(conv * dx)
    
    fitness = np.array([compute_c5(ind) for ind in population])
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Tournament selection
        tournament_size = 5
        num_parents = 50
        parents = []
        for _ in range(num_parents):
            indices = np.random.randint(0, population_size, tournament_size)
            idx = indices[np.argmin(fitness[indices])]
            parents.append(population[idx].copy())
        
        # Generate offspring by swapping two bits
        offspring = []
        for parent in parents:
            child = parent.copy()
            idx1, idx2 = np.random.choice(n_points, 2, replace=False)
            child[idx1], child[idx2] = child[idx2], child[idx1]
            offspring.append(child)
        
        # Evaluate offspring
        offspring_fitness = np.array([compute_c5(child) for child in offspring])
        
        # Replace weakest individuals
        num_replacements = min(len(offspring), len(population))
        for i in range(num_replacements):
            if time.time() - start_time >= budget_s:
                break
            weakest_idx = np.argmax(fitness)
            if offspring_fitness[i] < fitness[weakest_idx]:
                population[weakest_idx] = offspring[i]
                fitness[weakest_idx] = offspring_fitness[i]
        
        # Update best solution
        current_best = np.min(fitness)
        if current_best < best_c5:
            best_c5 = current_best
            best_idx = np.argmin(fitness)
            best_h[:] = population[best_idx].copy()
    
    return (best_h, best_c5, n_points)
