            h = np.clip(h_initial + noise, 0, 1)
        h = project(h, target_sum)
        particles.append(h)
        velocities.append(0.01 * np.random.randn(n_points))
    particles = np.array(particles)
    velocities = np.array(velocities)

    personal_best_pos = particles.copy()
    personal_best_scores = np.array([compute_c5(p, dx, n_points) for p in particles])
    best_idx = np.argmin(personal_best_scores)
    global_best = personal_best_pos[best_idx].copy()
    global_best_score = personal_best_scores[best_idx]

    while time.time() - start_time < budget_s - 5:
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = inertia * velocities[i] + c1 * r1 * (personal_best_pos[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            particles[i] = project(particles[i] + velocities[i], target_sum)
            current_score = compute_c5(particles[i], dx, n_points)
            if current_score < personal_best_scores[i]:
                personal_best_pos[i] = particles[i].copy()
                personal_best_scores[i] = current_score

        current_best_idx = np.argmin(personal_best_scores)
        if personal_best_scores[current_best_idx] < global_best_score:
            global_best = personal_best_pos[current_best_idx].copy()
            global_best_score = personal_best_scores[current_best_idx]

    if time.time() - start_time < budget_s - 10:
        bounds = [(0, 1)] * n_points
        constraints = {'type': 'eq', 'fun': lambda x: x.sum() - target_sum}
        res = optimize.minimize(lambda x: compute_c5(x, dx, n_points), x0=global_best, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            final_h = res.x
            final_c5 = compute_c5(res.x, dx, n_points)
        else:
            final_h, final_c5 = global_best, global_best_score
    else:
        final_h, final_c5 = global_best, global_best_score

    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Final C5: {final_c5:.6f}")
    return final_h, final_c5, n_points
