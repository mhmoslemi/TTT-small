        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)
    
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    
    def generate_initial_guess(current_h):
        h_bin = current_h.reshape(-1)
        indices_0 = np.where(h_bin == 0)[0]
        indices_1 = np.where(h_bin == 1)[0]
        num_swaps = np.random.randint(3, 11)
        num_swaps = min(num_swaps, len(indices_0), len(indices_1))
        if num_swaps <= 0:
            return h_bin.copy()
        swap_indices_0 = np.random.choice(indices_0, num_swaps, replace=False)
        swap_indices_1 = np.random.choice(indices_1, num_swaps, replace=False)
        new_h = h_bin.copy()
        new_h[swap_indices_0] = 1
        new_h[swap_indices_1] = 0
        return new_h
    
    trial_num = 0
    while time.time() - start_time < budget_s - 10:
        trial_num += 1
        h0 = generate_initial_guess(h_best)
        result = optimize.minimize(
            compute_c5,
            x0=h0,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
        )
        
        if result.success:
            h_candidate = result.x
            indices = np.argsort(h_candidate)[::-1]
            h_binary = np.zeros_like(h_candidate)
            h_binary[indices[:21]] = 1
            c5_binary = compute_c5(h_binary)
            if c5_binary < c5_best:
                h_best[:] = h_binary.copy()
                c5_best = c5_binary
    
    return h_best, c5_best, n_points
