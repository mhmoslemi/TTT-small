import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    def finalize_h(h):
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if current_sum == 0:
            h = np.full_like(h, target_sum / n_points)
        else:
            scale = target_sum / current_sum
            h = np.clip(h * scale, 0, 1)
        return h

    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_velocity = 0.5

    # Initialize particles
    particles = [finalize_h(h_initial + np.random.randn(n_points) * 0.1) for _ in range(n_particles)]
    particles = np.stack(particles, axis=0)
    velocities = np.zeros_like(particles)
    pbest_positions = particles.copy()
    pbest_scores = np.array([compute_c5(p) for p in particles])
    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]

    iteration = 0
    while time.time() - start_time < budget_s - 5:
        iteration += 1
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest_positions[i] - particles[i]) + c2 * r2 * (gbest_position - particles[i])
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            new_position = particles[i] + velocities[i]
            new_position = finalize_h(new_position)
            particles[i] = new_position
            current_score = compute_c5(new_position)
            if current_score < pbest_scores[i]:
                pbest_positions[i] = new_position.copy()
                pbest_scores[i] = current_score
                if current_score < gbest_score:
                    gbest_position[:] = new_position[:]
                    gbest_score = current_score

        if iteration % 20 == 0:
            top_indices = np.argsort(pbest_scores)[:5]
            for idx in top_indices:
                h_nm = pbest_positions[idx].copy()
                cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})
                result = opt.minimize(compute_c5, h_nm, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, options={'maxiter': 200, 'ftol': 1e-8})
                if result.success:
                    h_new = finalize_h(result.x)
                    new_score = compute_c5(h_new)
                    if new_score < gbest_score:
                        gbest_position[:] = h_new[:]
                        gbest_score = new_score

        if time.time() - start_time > budget_s - 1:
            break

    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Best C5: {gbest_score:.6f}")
    return gbest_position, gbest_score, n_points
