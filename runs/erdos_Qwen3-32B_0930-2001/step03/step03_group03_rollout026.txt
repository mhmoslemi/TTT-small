    global_c5 = personal_best_scores[g_idx]

    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_iterations = 1000
    iteration = 0
    stale_count = 0

    while time.time() - start_time < max_total_time and iteration < max_iterations:
        for i in range(num_particles):
            # Velocity update
            r1 = np.random.rand()
            r2 = np.random.rand()
            vel_update = w * velocities[i] + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            velocities[i] = np.clip(vel_update, -0.1, 0.1)

            # Position update and constraint enforcement
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            current_sum = np.sum(particles[i])
            delta = (target_sum - current_sum) / n
            particles[i] += delta
            particles[i] = np.clip(particles[i], 0.0, 1.0)

            # Update scores and best positions
            c5 = compute_c5(particles[i])
            if c5 < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = c5
                if c5 < global_c5:
                    global_best[:] = particles[i]
                    global_c5 = c5
                    stale_count = 0
            else:
                stale_count += 1

        # Adapt PSO parameters if stagnated
        if stale_count > 100 and iteration < 500:
            w = 0.5
            for i in range(num_particles):
                velocities[i] = 0.5 * velocities[i]
                velocities[i] = np.clip(velocities[i], -0.05, 0.05)
            stale_count = 0

        # Trust-region refinement after mid-optimization
        if iteration == 500 and time.time() - start_time < max_total_time - 10:
            top_idx = np.argsort(personal_best_scores)[:5]
            for idx in top_idx:
                def constraint_sum(x):
                    return np.sum(x) - target_sum
                bounds = [(0.0, 1.0) for _ in range(n)]
                res = optimize.minimize(
                    compute_c5,
                    x0=particles[idx],
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint_sum}],
                    options={'maxiter': 60, 'ftol': 1e-12}
                )
                if res.success:
                    cand_c5 = res.fun
                    if cand_c5 < global_c5:
                        global_best[:] = res.x
                        global_c5 = cand_c5

        iteration += 1

    # Final check and constraint enforcement
    h_best = global_best.copy()
    h_best = np.clip(h_best, 0.0, 1.0)
    current_sum = np.sum(h_best)
    delta = (target_sum - current_sum) / n
    h_best += delta
    h_best = np.clip(h_best, 0.0, 1.0)

    return h_best, global_c5, n
