The number of restarts is adjusted for time (5 in this case). The code ensures that each perturbed initial guess satisfies the sum and bounds constraints.

import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def normalize_h(h):
        target_sum = n_points / 2
        current_sum = h.sum()
        if abs(current_sum - target_sum) > 1e-8:
            scaling = target_sum / current_sum
            h = np.clip(h * scaling, 0.0, 1.0)
            current_sum = h.sum()
        if abs(current_sum - target_sum) > 1e-6:
            diff = target_sum - current_sum
            h[0] += diff
            h[0] = np.clip(h[0], 0.0, 1.0)
            current_sum = h.sum()
            diff = target_sum - current_sum
            if abs(diff) > 1e-6:
                h[1] += diff
                h[1] = np.clip(h[1], 0.0, 1.0)
        return h

    h_initial = normalize_h(h_initial)
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def constraint_eq(h):
        return h.sum() - n_points / 2

    constraints = {'type': 'eq', 'fun': constraint_eq}

    def objective(h):
        h_arr = h.reshape(n_points)
        corr = np.correlate(h_arr, 1 - h_arr, mode='full')
        return corr.max() * dx

    best_c5 = float('inf')
    best_h = h_initial.copy()

    rng = np.random.Generator(np.random.PCG64(seed))
    start_time = time.time()

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break

        perturbation = rng.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_initial + perturbation, 0.0, 1.0)
        h_perturbed = normalize_h(h_perturbed)

        result = scipy.optimize.minimize(
            objective,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-6}
        )

        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()

    result = scipy.optimize.minimize(
        objective,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 50, 'ftol': 1e-6}
    )

    if result.success:
        current_c5 = objective(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x.copy()

    return best_h, best_c5, n_points
