import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()

    delta_radius = 0.1
    alpha_momentum = 0.5
    Q_diag = np.eye(n_points)

    while time.time() - start_time < budget_s:
        overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
        k_max = np.argmax(overlaps)
        shift = k_max - (n_points - 1)

        g = np.zeros(n_points)
        for i in range(n_points):
            j = i + shift
            if 0 <= j < n_points:
                g[i] += (1.0 - h[j])
            m = i - shift
            if 0 <= m < n_points:
                g[i] -= h[m]

        delta_var = cp.Variable(n_points)
        objective = cp.Minimize(0.5 * cp.quad_form(delta_var, Q_diag) + g @ delta_var)
        constraints = [
            h + delta_var <= 1.0,
            h + delta_var >= 0.0,
            cp.sum(delta_var) == 0.0,
            cp.norm(delta_var, 2) <= delta_radius
        ]
        problem = cp.Problem(objective, constraints)
        try:
            problem.solve(solver=cp.ECOS, max_iters=100, verbose=False)
            if problem.status != 'optimal':
                delta_radius *= 0.5
                continue
        except:
            delta_radius *= 0.5
            continue

        h_new = np.clip(h + delta_var.value, 0.0, 1.0)
        current_sum = np.sum(h_new)
        h_new *= target_sum / current_sum
        h_new = np.clip(h_new, 0.0, 1.0)

        c5_new = evaluate_erdos_solution(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            delta_radius *= 1.1

            delta_radius *= 0.5

        current_sum = np.sum(h)
        h *= target_sum / current_sum
        h = np.clip(h, 0.0, 1.0)

    return (best_h, best_c5, n_points)
