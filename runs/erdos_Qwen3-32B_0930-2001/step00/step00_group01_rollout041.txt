def project_to_feasible(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    D = n_points / 2
    current_sum = np.sum(h_clipped)
    if current_sum == 0:

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    convoluted = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(convoluted * dx)
    return c5

def tournament_select(fitness_list, tournament_size):
    contestant_indices = np.random.choice(len(fitness_list), tournament_size)
    contestants = [(fitness_list[i][0], fitness_list[i][1]) for i in contestant_indices]
    best_idx = np.argmin([c for h, c in contestants])
    return contestants[best_idx][0]

def crossover(h1, h2):
    n = len(h1)
    point = np.random.randint(1, n-1)
    child1 = np.concatenate((h1[:point], h2[point:]))
    child2 = np.concatenate((h2[:point], h1[point:]))
    return child1, child2
