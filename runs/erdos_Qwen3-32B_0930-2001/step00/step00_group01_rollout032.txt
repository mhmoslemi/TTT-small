import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    np.random.seed(seed)
    n_points = len(initial_h_values)
    initial_h = initial_h_values.copy()
    dx = 2.0 / n_points
    desired_sum = 0.5 * n_points  # sum(h) must equal this.

    # GA parameters
    population_size = 25
    max_generations = 500
    mutation_strength = 0.05
    tournament_size = 3
    # Initialize population
    population = [initial_h.copy()]

    for _ in range(population_size -1):
        # Perturb the initial_h to generate new candidate
        candidate = initial_h.copy() + np.random.normal(0, 0.1, n_points)
        # Clip and scale
        candidate = np.clip(candidate, 0, 1).copy()
        s = candidate.sum()
        if np.abs(s) < 1e-9:
            scaling = 1.0
        else:
            scaling = desired_sum / s
        candidate *= scaling
        population.append(candidate)

    best_c5 = float('inf')
    best_h = None

    start_time = time.time()
    for gen in range(max_generations):
        # Evaluate population
        c5_list = []
        for h in population:
            # Compute C5 for this h
            # Compute 1 - h
            h1 = 1 - h
            # Compute convolution
            # Note: correlation is equivalent to convolution with reversed kernel
            # But for our case, we compute np.correlate(h, h1, mode='full')
            conv = np.correlate(h, h1, mode='full')
            c5 = np.max(conv * dx)
            c5_list.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        # Check time
        if time.time() - start_time > budget_s * 0.99:  # leave some buffer
            break

        # Tournament selection
        parents = []
        for _ in range(population_size // 2):
            # Randomly select 3 individuals, pick the one with lowest c5
            indices = np.random.choice(population_size, tournament_size, replace=False)
            idx = min(indices, key=lambda i: c5_list[i])
            parents.append(population[idx].copy())

        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            if i+1 < len(parents):
                parent1 = parents[i]
                parent2 = parents[i+1]
                # Single-point crossover
                cx_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:cx_point], parent2[cx_point:]])
                child2 = np.concatenate([parent2[:cx_point], parent1[cx_point:]])
                offspring.append(child1)
                offspring.append(child2)
        # If odd number of parents, maybe duplicate last?

        # Mutation and feasibility
        for child in offspring:
            # Add mutation
            child += np.random.normal(0, mutation_strength, n_points)
            # Clip and scale
            child_clipped = np.clip(child, 0, 1)
            s = child_clipped.sum()
            if np.abs(s) < 1e-9:
                scaling = 1.0
            else:
                scaling = desired_sum / s
            child[:] = child_clipped * scaling

        # Evaluate offspring's c5 to merge into next population
        # But need to compute c5 for them for selection
        offspring_c5 = []
        for child in offspring:
            h1_offspring = 1 - child
            conv = np.correlate(child, h1_offspring, mode='full')
            c5_offspring = np.max(conv * dx)
            offspring_c5.append( (child, c5_offspring) )

        # Merge parents and offspring into a new pool
        all_candidates = population + [c[0] for c in offspring_c5]
        all_c5 = c5_list + [c[1] for c in offspring_c5]
        # Select the best N individuals
        # Sort all_candidates by c5 and take the top N
        sorted_indices = np.argsort(all_c5)
        population = [all_candidates[i] for i in sorted_indices[:population_size]]

    return (best_h, best_c5, n_points)

import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    desired_sum = n_points / 2.0

    population_size = 25
    max_generations = 500
    mutation_strength = 0.05
    tournament_size = 3

    # Initialize population with initial_h and perturbed variants
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        perturb = np.random.normal(0, 0.1, n_points)
        candidate = np.clip(initial_h.copy() + perturb, 0, 1)
        s = candidate.sum()
        scaling = desired_sum / s if s > 1e-9 else 1.0
        candidate *= scaling
        population.append(candidate)

    best_c5 = np.inf
    best_h = None
    start_time = time.time()

    for gen in range(max_generations):
        # Evaluate current population
        c5_list = []
        for h in population:
            h1 = 1 - h
            conv = np.correlate(h, h1, mode='full')
            current_c5 = np.max(conv * dx)
            c5_list.append(current_c5)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

        # Early stopping if time is out
        if time.time() - start_time > budget_s * 0.99:
            break

        # Tournament selection
        parents = []
        for _ in range(population_size // 2):
            indices = np.random.choice(population_size, tournament_size, replace=False)
            selected = min(indices, key=lambda i: c5_list[i])
            parents.append(population[selected].copy())

        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                p1, p2 = parents[i], parents[i+1]
                cx_point = np.random.randint(1, n_points)
                child1 = np.concatenate([p1[:cx_point], p2[cx_point:]])
                child2 = np.concatenate([p2[:cx_point], p1[cx_point:]])
                offspring.append(child1)
                offspring.append(child2)

        # Mutation and feasibility
        for child in offspring:
            noise = np.random.normal(0, mutation_strength, n_points)
            child += noise
            # Clip and scale
            child_clipped = np.clip(child, 0, 1)
            s = child_clipped.sum()
            scaling = desired_sum / s if s > 1e-9 else 1.0
            child[:] = child_clipped * scaling

        # Merge parents and offspring, evaluate offspring
        offspring_c5 = []
        for child in offspring:
            h1_offspring = 1 - child
            conv = np.correlate(child, h1_offspring, mode='full')
            offspring_c5.append((child, np.max(conv * dx)))

        # Create new population from best candidates
        all_candidates = population + [c[0] for c in offspring_c5]
        all_c5 = c5_list + [c[1] for c in offspring_c5]
        sorted_indices = np.argsort(all_c5)
        population = [all_candidates[i] for i in sorted_indices[:population_size]]

    return (best_h, best_c5, n_points)
