def run():

        n_points = n_initial * 2  # Double the resolution
        dx = 2.0 / n_points

        def interpolate_h(h_old, N_new):
            N_old = len(h_old)
            dx_old = 2.0 / N_old
            h_new = np.zeros(N_new)
            for i_new in range(N_new):
                x = i_new * 2.0 / N_new
                j_old = int(np.floor(x / dx_old))
                if j_old >= N_old:
                    j_old = N_old - 1
                h_new[i_new] = h_old[j_old]
            return h_new

        h_initial_new = interpolate_h(h_initial, n_points)

        h = cp.Variable(n_points)
        C = cp.Variable()

        constraints = [
            cp.sum(h) == n_points / 2.0,
            0 <= h,
            h <= 1
        ]

        for k in range(2 * n_points):
            if k < n_points:
                max_idx = n_points - k
                if max_idx <= 0:
                    continue
                sum_expr = 0
                for i in range(max_idx):
                    sum_expr += h[i] * (1 - h[i + k])

                if m >= n_points:

                min_idx = m

            constraints.append(C >= sum_expr * dx)

        prob = cp.Problem(cp.Minimize(C), constraints)
