def constraint_sum(x, n_points):

    def project(h):
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if current_sum == 0:
            return h.copy()
        scale = (n_points / 2.0) / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        return h.copy()

    n_particles = 20
    particles = []
    for _ in range(n_particles):
        particle = initial_h + np.random.uniform(-0.1, 0.1, size=n_points)
        particles.append(project(particle))
    particles = np.array(particles)
    velocities = np.zeros_like(particles)

    pbest_pos = particles.copy()
    pbest_scores = np.array([objective(p, n_points) for p in particles])
    gbest_idx = pbest_scores.argmin()

    gbest_score = pbest_scores[gbest_idx]
