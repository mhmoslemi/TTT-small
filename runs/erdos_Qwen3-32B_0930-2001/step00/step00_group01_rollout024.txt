import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    # Constraints
    lb = 0.0 * np.ones(n_points)
    ub = 1.0 * np.ones(n_points)
    bounds = list(zip(lb, ub))

    def sum_constraint(h):
        return h.sum() - n_points / 2

    cons = {'type': 'eq', 'fun': sum_constraint}
    cons = (cons,)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        return evaluate_erdos_solution(h)

    start_time = time.time()
    result = minimize(
        objective,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-10, 'iprint': 0}
    )

    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0, 1)
    h_sum = h_opt.sum()
    if not np.isclose(h_sum, n_points / 2, rtol=0, atol=1e-8):
        scale = (n_points / 2) / h_sum
        h_opt = h_opt * scale
        h_opt = np.clip(h_opt, 0, 1)

    c5_val = evaluate_erdos_solution(h_opt)
    return (h_opt.tolist(), c5_val, n_points)
