        if abs(sum_h) > 1e-12:
            h *= (n_points / 2) / sum_h
        else:  # Fallback if numerical error
            h = np.zeros_like(h)
            h[n_points//2] = 1.0
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution((best_h, n_points))['c5']
    return (best_h, final_c5, n_points)
