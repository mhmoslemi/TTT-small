import numpy as np
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def run():
    import initial_h_values
    start_time = time.time()
    budget_s = 1100

    h = np.array(initial_h_values)
    n_points = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    
    np.random.seed(42)
    T = 2.0
    alpha = 0.99
    delta = 0.01
    max_iter = int(1e5)
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Random perturbation
        idx = np.random.choice(n_points, size=np.random.randint(1, min(4, n_points)), replace=False)
        mask = np.zeros(n_points)
        mask[idx] = 1.0
        noise = np.random.normal(0, delta, size=n_points) * mask
        
        candidate = best_h.copy() + noise
        candidate = np.clip(candidate, 0, 1)
        current_sum = np.sum(candidate)
        
        if current_sum != 0:
            scaling = (n_points / 2.0) / current_sum
            candidate = np.clip(candidate * scaling, 0, 1)
        
        new_c5 = compute_c5(candidate, n_points)
        delta_c5 = new_c5 - best_c5
        
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            best_h[:] = candidate
            best_c5 = new_c5
        
        T *= alpha
        
        # Safety projection
        while not np.isclose(np.sum(best_h), n_points / 2, atol=1e-5):
            best_h = np.clip(best_h, 0, 1)
            s = np.sum(best_h)
            if abs(s) < 1e-9:
                best_h[:] = 0.5
            else:
                best_h *= (n_points / 2.0) / s
                best_h = np.clip(best_h, 0, 1)

    return (best_h.tolist(), best_c5, n_points)
