import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available, else create a uniform 0.5 initializer
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    
    # Evaluate initial C5 bound
    try:
        best_c5 = evaluate_erdos_solution(best_h, dx)
    except:
        best_c5 = float('inf')
    
    # Simulated Annealing parameters
    initial_temp = 0.2
    final_temp = 1e-5
    current_temp = initial_temp
    
    # Main optimization loop
    while time.time() - start_time < budget_s:
        # Perturb solution and enforce bounds
        perturbation = np.random.uniform(-current_temp, current_temp, size=n_points)
        new_h = np.clip(h + perturbation, 0, 1)
        
        # Re-normalize to satisfy integral constraint
        sum_h = new_h.sum()
        if abs(sum_h) > 1e-10:
            new_h *= (n_points / 2) / sum_h
        
        # Evaluate candidate
        try:
            new_c5 = evaluate_erdos_solution(new_h, dx)
        except:
            new_c5 = float('inf')
        
        # Metropolis acceptance criterion
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / current_temp):
            h = new_h.copy()
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        
        # Exponential cooling
        current_temp *= 0.9999
    
    return best_h, best_c5, n_points
