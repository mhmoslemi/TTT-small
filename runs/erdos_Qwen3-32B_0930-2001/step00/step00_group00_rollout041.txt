        n_points = len(initial_h_values)
        h = initial_h_values.copy()
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h / (sum(h) * (2.0 / n_points)) * 1.0
    required_sum = n_points / 2.0
    h_sum = np.sum(h)
    if not np.isclose(h_sum, required_sum):
        h = h * required_sum / h_sum
    dx = 2.0 / n_points
    start_time = time.time()
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    while time.time() - start_time < budget_s:
        current_h = best_h.copy()
        i, j = np.random.choice(n_points, size=2, replace=False)
        if i != j:
            new_h = current_h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]
            try:
                new_c5 = evaluate_erdos_solution(new_h, dx)
            except:
                new_c5 = float('inf')
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h[:] = new_h[:]
    return best_h, best_c5, n_points
