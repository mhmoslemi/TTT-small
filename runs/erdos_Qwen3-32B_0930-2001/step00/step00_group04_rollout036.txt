    best_idx = np.argmin(personal_best_scores)

    global_best_score = personal_best_scores[best_idx]

    c1 = 1.5
    c2 = 1.5

    idx = 0

            # Update velocity and position
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (personal_bests[i] - positions[i]) +
                             c2 * r2 * (global_best - positions[i]))

            # Update personal best

def numerical_gradient(h):
    eps = 1e-5
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        orig = h[i]
        h[i] += eps
        h_fplus = project(h.copy(), n)
        f_plus = evaluate_erdos_solution(h_fplus)
        
        h[i] -= 2 * eps
        h_fminus = project(h.copy(), n)
        f_minus = evaluate_erdos_solution(h_fminus)
        
        grad[i] = (f_plus - f_minus) / (2 * eps)
        h[i] = orig  # Restore original value
    return grad

def project(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        h[:] = 0.5
        return h
    scale = (n_points / 2.0) / total
    h *= scale
    return np.clip(h, 0.0, 1.0)
