import numpy as np

def compute_overlap(h, dx):
    n = len(h)
    correlate_result = np.correlate(h, 1 - h, mode='full')
    return correlate_result * dx

def find_max_overlap(overlaps, n):
    max_value = np.max(overlaps)
    max_indices = np.where(np.isclose(overlaps, max_value))[0]
    for idx in max_indices:
        lag = idx - (n - 1)
        if - (n - 1) <= lag <= n - 1:
            return max_value, lag
    return max_value, 0

def compute_gradient(h, lag, n):

        if i + lag < n:
            grad[i] += (1.0 - h[i + lag])
        if i - lag >= 0:
            grad[i] -= h[i - lag]
    return grad
