import numpy as np
import time

def project_feasible(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    target_sum = n_points / 2
    current_sum = h_clipped.sum()
    h_scaled = h_clipped * target_sum / current_sum
    return h_scaled

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    population_size = 50
    top_pct = 0.2
    pop = [project_feasible(h_initial.copy(), n_points)]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        pop.append(project_feasible(h, n_points))
    pop = np.array(pop)
    best_c5 = float('inf')
    best_h = h_initial.copy()

    while time.time() - start_time < budget_s:
        C5 = []
        for h in pop:
            corr = np.correlate(h, 1 - h, mode='full')
            C5.append(np.max(corr) * dx)
        C5 = np.array(C5)
        idx = np.argmin(C5)
        if C5[idx] < best_c5:
            best_c5 = C5[idx]
            best_h[:] = pop[idx].copy()

        n_top = max(1, int(population_size * top_pct))
        top_idx = np.argsort(C5)[:n_top]
        elites = pop[top_idx]

        new_pop = []
        while len(new_pop) < population_size - n_top:
            p1 = elites[np.random.choice(n_top)]
            p2 = elites[np.random.choice(n_top)]
            mask = np.random.rand(n_points) < 0.5
            child = np.where(mask, p1, p2)
            noise = np.random.normal(0, 0.05, n_points)
            child = project_feasible(child + noise, n_points)
            new_pop.append(child)
        pop = np.concatenate([elites, new_pop])

        if len(new_pop) > 0 and (len(pop) < population_size or np.random.rand() < 0.1):
            h = np.random.rand(n_points)
            pop[np.random.randint(population_size)] = project_feasible(h, n_points)

    best_h = project_feasible(best_h, n_points)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    return (best_h.tolist(), np.max(corr) * dx, n_points)
