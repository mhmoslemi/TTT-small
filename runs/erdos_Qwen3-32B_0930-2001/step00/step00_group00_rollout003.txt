import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():

            h *= scaling_factor
        else:
            h[:] = 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    initial_temp = 0.2
    final_temp = 1e-5
    current_temp = initial_temp
    
    while time.time() - start_time < budget_s:
        perturbation = np.random.uniform(-current_temp, current_temp, n_points)
        new_h = h + perturbation
        new_h = np.clip(new_h, 0, 1)
        sum_new_h = np.sum(new_h)
        scaling_factor = (n_points / 2.0) / sum_new_h if abs(sum_new_h) > 1e-10 else 1.0
