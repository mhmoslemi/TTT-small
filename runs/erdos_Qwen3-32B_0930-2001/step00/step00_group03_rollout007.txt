import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    h = np.array(initial_h_values, copy=True)
    n_points = len(h)

    while time.time() - start_time < budget_s:
        # Perturb current h
        candidate = h.copy()

        candidate += noise
        candidate = np.clip(candidate, 0, 1)

            h_new = np.zeros_like(h)
        else:
            h_new = candidate * (n_points / 2 / current_sum)

        new_c5 = np.max(new_cc) * dx
