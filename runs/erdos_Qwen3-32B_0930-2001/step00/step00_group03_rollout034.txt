import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2
    assert np.all((0 <= h) & (h <= 1)) and np.isclose(np.sum(h), target_sum)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    sigma_grid = [0.05, 0.1, 0.2, 0.3, 0.4]
    perturbation_magnitude = 0.01

    start_time = time.time()

    for sigma in sigma_grid:
        if time.time() - start_time > budget_s - 30:
            break
        # Gaussian blur
        h_blur = gaussian_filter1d(h, sigma=sigma, mode='reflect', truncate=4)
        h_blur = np.clip(h_blur, 0, 1)
        h_blur *= target_sum / np.sum(h_blur)
        h_blur = np.clip(h_blur, 0, 1)
        # Local optimization
        current_h = h_blur.copy()
        for _ in range(50):
            if time.time() - start_time > budget_s - 30:
                break
            mask = np.random.rand(n) < 0.1
            delta = np.random.uniform(-perturbation_magnitude, perturbation_magnitude, n)
            delta[~mask] = 0
            new_h = current_h + delta
            new_h = np.clip(new_h, 0, 1)
            new_sum = np.sum(new_h)
            scale = target_sum / max(1e-10, new_sum)
            new_h = np.clip(new_h * scale, 0, 1)
            new_c5 = evaluate_erdos_solution(new_h)
            if new_c5 < best_c5:
                best_h[:] = new_h
                best_c5 = new_c5
                current_h = new_h.copy()

    # Simulated Annealing
    temp = 1.0
    cooling_rate = 0.99
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        idx = np.random.randint(n)
        change = np.random.uniform(-perturbation_magnitude, perturbation_magnitude)
        new_h = best_h.copy()
        new_h[idx] = np.clip(new_h[idx] + change, 0, 1)
        new_sum = np.sum(new_h)
        scale = target_sum / max(1e-10, new_sum)
        new_h = np.clip(new_h * scale, 0, 1)
        new_c5 = evaluate_erdos_solution(new_h)
        delta_c5 = best_c5 - new_c5
        if delta_c5 > 0 or np.random.rand() < np.exp(delta_c5 / temp):

            best_c5 = new_c5
        temp *= cooling_rate

    # Final projection
    best_sum = np.sum(best_h)
    scale = target_sum / max(1e-10, best_sum)
    best_h = np.clip(best_h * scale, 0, 1)
