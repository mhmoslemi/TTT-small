import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_c5(h):
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(cross_corr)

    def project_sum(h, target):
        h_clipped = np.clip(h, 0, 1)
        adjustment = target - np.sum(h_clipped)
        if abs(adjustment) < 1e-8:
            return h_clipped
        adjustable = np.where((h_clipped > 1e-6) & (h_clipped < 1 - 1e-6))[0]
        if len(adjustable) == 0:
            return h_clipped
        step = adjustment / len(adjustable)
        h_clipped[adjustable] += step
        return np.clip(h_clipped, 0, 1)

    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    learning_rate = 0.05
    iterations = 0

    while time.time() - start_time < budget_s - 1:
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        max_c5 = np.max(cross_corr)
        max_indices = np.where(np.abs(cross_corr - max_c5) < 1e-8)[0]

        grad = np.zeros_like(h)
        for idx_k in max_indices:
            k = idx_k - n_points + 1
            for i in range(n_points):
                term1 = 0.0
                term2 = 0.0
                if 0 <= i + k < n_points:
                    term1 = 1.0 - h[i + k]
                if 0 <= i - k < n_points:
                    term2 = -h[i - k]
                grad[i] += term1 + term2
        if len(max_indices) > 0:
            grad /= len(max_indices)
        else:
            grad = np.zeros_like(h)

        h -= learning_rate * grad
        h = project_sum(h, target_sum)

        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            learning_rate = 0.05
        else:
            learning_rate *= 0.95

        iterations += 1
        if iterations % 200 == 0 and iterations > 0:
            print(f"Iter {iterations}, C5={best_c5:.6f}, time={time.time()-start_time:.2f}")

    return best_h.tolist(), best_c5, n_points
