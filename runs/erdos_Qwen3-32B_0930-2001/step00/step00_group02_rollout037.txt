def run(seed=42, budget_s=1100, **kwargs):

        h = np.clip(h_candidate, 0, 1)
        current_sum = h.sum()
        if current_sum > 1e-10:
            scaling_factor = target_sum / current_sum
            h *= scaling_factor
        h = np.clip(h, 0, 1)
        return h

    h = project_h(h, target_sum)
