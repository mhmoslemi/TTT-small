    target_sum = n_points / 2.0
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h, n_points)
    start_time = time.time()
    perturb_size = 5
    
    while time.time() - start_time < budget_s - 0.1:
        h_candidate = best_h.copy()
        indices = np.random.choice(n_points, size=np.random.randint(1, min(perturb_size, n_points // 2) + 1), replace=False)
        for i in indices:
            delta = np.random.uniform(-0.1, 0.1)
            new_val = h_candidate[i] + delta
            new_val = max(0., min(1., new_val))
            h_candidate[i] = new_val
        current_sum = h_candidate.sum()
        delta_sum = target_sum - current_sum
        if abs(delta_sum) > 1e-8:
            if len(indices) > 0:
                idx = indices[0]
                adj = h_candidate[idx] + delta_sum
                if 0. <= adj <= 1.:
                    h_candidate[idx] = adj
                else:
                    continue
            else:
                continue
        c5_candidate = evaluate_candidate(h_candidate, n_points)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h[:] = h_candidate[:]
    return best_h, best_c5, n_points
