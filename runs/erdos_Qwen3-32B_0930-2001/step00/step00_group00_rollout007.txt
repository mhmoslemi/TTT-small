import numpy as np
import scipy.optimize as opt
import time

def project_to_constraints(h, n_points):
    h = np.clip(h, 0, 1).copy()
    total = np.sum(h)
    if total == 0:
        h[:] = 0.5
        total = n_points / 2.0
    scaling = (n_points / 2.0) / total
    return h * scaling

def run(seed=42, budget_s=1100, **kwargs):
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)

    start_time = time.time()

    def objective(h_array):
        h = h_array.reshape(n_points)
        correlate = np.correlate(h, 1 - h, mode='full') * dx
        return float(np.max(correlate))

    bounds = [(0, 1)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}]

    h_proj = project_to_constraints(h, n_points)
    try:
        while time.time() - start_time < budget_s:
            result = opt.minimize(
                objective,
                x0=h_proj.reshape(-1),
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-10}
            )
            if not result.success:
                continue
            h_opt = result.x.reshape(n_points)
            h_opt = project_to_constraints(h_opt, n_points)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt.copy(), c5_opt
            if best_c5 <= 0.38085:
                break
    except Exception:
        pass

    finalize_time = time.time() - start_time
    if finalize_time >= budget_s:
        pass  # Already within the budget check
    return (best_h, best_c5, n_points)
