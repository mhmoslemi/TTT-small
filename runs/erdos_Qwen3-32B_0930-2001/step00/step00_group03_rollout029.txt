import time
from numpy import correlate

def run(seed=42, budget_s=1100, **kwargs):
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    assert np.all(h >= 0) and np.all(h <= 1)
    assert abs(np.sum(h) - n_points / 2) < 1e-6
    
    best_h = h.copy()
    best_c5, _ = evaluate_erdos_solution(best_h, n_points)
    np.random.seed(seed)
    start_time = time.time()
    
    T = 1.0
    cooling_alpha = 0.999
    alpha = 0.01
    beta = 0.999
    
    while time.time() - start_time < budget_s:
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr * dx)
        m = np.argmax(corr * dx)
        k_max = m - (n_points - 1)
        
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] += (1 - h[i + k_max])
            if 0 <= i - k_max < n_points:
                grad[i] -= h[i - k_max]
        grad *= dx
        
        h_candidate = h - alpha * grad
        h_candidate = np.clip(h_candidate, 0, 1)
        s = h_candidate.sum()
        if s > 1e-9:
            h_candidate *= (n_points / 2) / s
        else:
            h_candidate = h.copy()
        
        c5_candidate, _ = evaluate_erdos_solution(h_candidate, n_points)
        
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
            h = h_candidate.copy()
        else:
            delta = c5_candidate - c5
            if np.random.rand() < np.exp(-delta / T):
                h = h_candidate.copy()
        
        T *= cooling_alpha
        alpha *= beta
    
    return best_h.tolist(), best_c5, n_points
