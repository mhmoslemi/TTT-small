import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(x):
        h = x.reshape(n_points)
        temp = 1.0 - h
        overlaps = np.correlate(h, temp, mode='full') * dx
        return np.max(overlaps)

    bounds = [(0., 1.) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}

    start_time = time.time()

    def time_callback(xk):
        if time.time() - start_time > budget_s:
            return True
        return False

    result = minimize(
        objective,
        x0=initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        callback=time_callback,
        options={'maxiter': 10000, 'ftol': 1e-8}
    )

    best_h = np.clip(result.x, 0, 1)
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum - n_points / 2) > 1e-6:
        scale = (n_points / 2) / best_h_sum
        best_h = np.clip(best_h * scale, 0, 1)

    c5 = objective(best_h)
    return (best_h, c5, n_points)
