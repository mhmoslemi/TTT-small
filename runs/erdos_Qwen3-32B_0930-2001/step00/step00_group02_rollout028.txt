import numpy as np
from time import time

def run(budget_s=1100):
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time()
    
    best_h = h.copy()
    best_c5 = 1000.0  # Placeholder
    
    def evaluate_c5(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full') * dx
        return np.max(corr)
    
    max_iter = 10000
    alpha = 0.05
    alpha_decay = 0.95
    for iter_num in range(max_iter):
        if time() - start_time >= budget_s:
            break
        current_h = h.copy()
        corr = np.correlate(current_h, 1 - current_h, mode='full') * dx
        c5_current = np.max(corr)
        if c5_current < best_c5:
            best_h = current_h.copy()
            best_c5 = c5_current
        
        m = np.argmax(corr)
        k_star = m - (n_points - 1)
        # Compute gradient for k_star
        grad = np.zeros(n_points)
        for i in range(n_points):
            t_plus = i + k_star
            if t_plus < n_points and t_plus >= 0:
                grad[i] += (1 - current_h[t_plus])
            t_minus = i - k_star
            if t_minus < n_points and t_minus >= 0:
                grad[i] -= current_h[t_minus]
        grad *= dx
        
        h_new = current_h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        sum_h = np.sum(h_new)

                h_new[free] += delta
                h_new = np.clip(h_new, 0, 1)
        h = h_new
        alpha *= alpha_decay
        
    return best_h, best_c5, n_points
