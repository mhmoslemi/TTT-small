import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial h_values if available
    if 'initial_h_values' in globals():
        h = initial_h_values
        n_points = len(h)
    else:
        n_points = 45  # Fallback to 45 if no initial_h_values is available
        h = np.full(n_points, 0.5)  # Start with uniform distribution
    
    dx = 2.0 / n_points
    
    def compute_overlaps(h):
        return np.correlate(h, 1 - h, mode='full') * dx
    
    best_h = h.copy()
    best_c5 = np.max(compute_overlaps(h))
    
    learning_rate = 0.01
    while time.time() - start_time < budget_s:
        overlaps = compute_overlaps(h)
        current_c5 = np.max(overlaps)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        max_idx = np.argmax(overlaps)
        k_shift = max_idx - (n_points - 1)
        
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k_shift
            if 0 <= j < n_points:
                grad[i] += (1.0 - h[j]) * dx
                grad[j] -= h[i] * dx
        
        h -= learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        
        current_sum = h.sum()
        desired_sum = n_points / 2
        if current_sum > 0:
            h = h * (desired_sum / current_sum)
            h = np.clip(h, 0.0, 1.0)
    
    return best_h, best_c5, n_points
