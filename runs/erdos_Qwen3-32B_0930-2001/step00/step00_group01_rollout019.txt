def project_to_feasible(h, required_sum):

    h_clipped = np.clip(h, 0.0, 1.0)

    sum_h = np.sum(h_clipped)

    delta = required_sum - sum_h

    if delta == 0:

        return h_clipped

    h_proj = h_clipped.copy()

    if delta > 0:

        # Need to increase

        # Available increase for each element: 1.0 - h_proj[i]

        available = 1.0 - h_proj

        # Indices sorted by available in descending order

        idx = np.argsort(-available)

        for i in idx:

            add = min(delta, available[i])

            h_proj[i] += add

            delta -= add

            if delta <= 1e-10:

                break

    else: # delta < 0

        # Need to decrease

        available = h_proj

        idx = np.argsort(-available)

        for i in idx:

            sub = min(-delta, available[i])

            h_proj[i] -= sub

            delta += sub

            if delta >= -1e-10:

                break

    return h_proj

    while cur_gen < max_generations:

        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1: #leave some buffer

        # Evaluate all in population
        evaluated_population = []
        for h in population:

            evaluated_population.append( (h, c5) )

        # Select top half as parents

            # Mutate

            # Project to feasible set

    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr * dx)

def project_to_feasible(h, required_sum):
    h_clipped = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h_clipped)
    delta = required_sum - sum_h
    if abs(delta) < 1e-10:
        return h_clipped.copy()
    h_proj = h_clipped.copy()
    if delta > 0:
        available = 1.0 - h_proj
        idx = np.argsort(-available)
        for i in idx:
            add = min(delta, available[i])
            h_proj[i] += add
            delta -= add
            if delta <= 1e-10:
                break
    else:
        available = h_proj
        idx = np.argsort(-available)
        for i in idx:
            sub = min(-delta, available[i])
            h_proj[i] -= sub
            delta += sub
            if delta >= -1e-10:
                break
    return h_proj

def mutate(h, std):
    noise = np.random.normal(0, std, len(h))
    return h + noise

def crossover(h1, h2):
    pt = np.random.randint(1, len(h1)-1)
    c1 = np.concatenate([h1[:pt], h2[pt:]])
    c2 = np.concatenate([h2[:pt], h1[pt:]])
    return c1, c2

def tournament_selection(population, size=3):
    candidates = random.sample(population, size)
    return min(candidates, key=lambda x: x[1])
