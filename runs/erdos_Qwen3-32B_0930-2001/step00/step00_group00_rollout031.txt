        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(np.abs(corr * dx))

    # Ensure initial solution is feasible
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        sum_h = h.sum()
        if not np.isclose(sum_h, n_points / 2.0):
            h *= n_points / 2.0 / sum_h
        return h

    best_h, best_c5 = project_to_feasible(h), objective(h)

    # Optimization using COBYLA (constraint-handling)
    bounds = scipy.optimize.Bounds(0, 1)
    sum_constr = scipy.optimize.LinearConstraint(np.ones(n_points), [n_points / 2.0], [n_points / 2.0])
    
    start_time = time.time()
    result = scipy.optimize.minimize(
        objective,
        x0=best_h,
        method='COBYLA',
        bounds=bounds,
        constraints=sum_constr,
        options={'maxiter': 1000, 'rhobeg': 0.1}
    )
    
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.9 and result.success:
        candidate_h = np.array(result.x)
        candidate_h = project_to_feasible(candidate_h)
        candidate_c5 = objective(candidate_h)
        if candidate_c5 < best_c5:
            best_h, best_c5 = candidate_h, candidate_c5

    # Final projection and evaluation
    best_h = project_to_feasible(best_h)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
