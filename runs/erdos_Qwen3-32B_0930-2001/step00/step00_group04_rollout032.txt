            new_pos = particles[i] + velocities[i]
            new_pos = np.clip(new_pos, 0, 1)
            s = np.sum(new_pos)
            if s > 1e-8:
                scale = (n_points / 2) / s
                new_pos *= scale
            new_pos = np.clip(new_pos, 0, 1)
            particles[i] = new_pos
            score = compute_c5(particles[i])
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = particles[i].copy()
                if score < global_best_score:
                    global_best_score = score
                    global_best_position[:] = particles[i].copy()

    # SLSQP phase
    initial_for_slsqp = global_best_position.copy()
    constraints_eq = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
    bounds = [(0, 1)] * n_points
    remaining_time = budget_s - (time.time() - start_time)
    slsqp_maxiter = min(int(remaining_time * 0.8 // 1), 200)
    if slsqp_maxiter < 1:
        final_h = global_best_position
        final_c5 = global_best_score
    else:
        slsqp_result = minimize(
            compute_c5,
            x0=initial_for_slsqp,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints_eq,
            options={'maxiter': slsqp_maxiter, 'ftol': 1e-6}
        )
        if slsqp_result.success:
            final_h = slsqp_result.x
            final_c5 = slsqp_result.fun
        else:
            final_h = global_best_position
            final_c5 = global_best_score

    assert np.all(final_h >= 0) and np.all(final_h <= 1)
    assert abs(np.sum(final_h) - n_points / 2) < 1e-6
    return (final_h.tolist(), final_c5, n_points)
