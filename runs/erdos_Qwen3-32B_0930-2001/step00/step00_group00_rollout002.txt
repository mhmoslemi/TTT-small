        c5_candidate, _ = evaluate_erdos_solution(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            h = h_candidate.copy()

    return (best_h, best_c5, n_points)
