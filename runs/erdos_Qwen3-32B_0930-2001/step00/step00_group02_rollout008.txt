import numpy as np
import time

def compute_c5(h, dx):
    ones_minus_h = 1.0 - h
    corr = np.correlate(h, ones_minus_h, mode='full')
    return corr.max() * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    try:
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / h.sum()
    dx = 2.0 / n_points
    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    max_iter = 10000
    learning_rate = 0.01
    learning_rate_decay = 0.995
    
    for _ in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        current_c5 = corr.max() * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        max_idx = corr.argmax()
        shift = max_idx - (n_points - 1)
        grad = np.zeros_like(h, dtype=np.float64)
        dx_val = dx
        
        if shift >= 0:
            valid_m_first = np.arange(n_points - shift)
            grad[valid_m_first] += (1 - h[valid_m_first + shift]) * dx_val
            valid_m_second = np.arange(shift, n_points)
            grad[valid_m_second] -= h[valid_m_second - shift] * dx_val
        else:
            l = -shift
            valid_m_first = np.arange(l, n_points)
            grad[valid_m_first] += (1 - h[valid_m_first - l]) * dx_val
            valid_m_second = np.arange(n_points - l)
            grad[valid_m_second] -= h[valid_m_second + l] * dx_val
        
        h -= learning_rate * grad
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_clipped = h_clipped.sum()
        if sum_clipped != 0:
            h[:] = h_clipped * (n_points / 2) / sum_clipped
        else:
            h[:] = 0.0
            h[0] = n_points / 2
        learning_rate *= learning_rate_decay
    
    return best_h, best_c5, n_points
