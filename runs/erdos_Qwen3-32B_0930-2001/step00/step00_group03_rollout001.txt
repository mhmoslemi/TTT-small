import numpy as np

from time import time

def run():
    from evaluate_erdos_solution import evaluate_erdos_solution
    import initial_h_values
    
    # Setup
    h = np.array(initial_h_values)
    n_points = len(h)
    required_sum = n_points / 2.0
    assert np.allclose(np.sum(h), required_sum, atol=1e-6)
    assert np.all((h >= 0.0) & (h <= 1.0))
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time()
    
    # Optimization parameters
    sigma_grid = [0.05, 0.1, 0.2, 0.3, 0.4]
    local_steps = 100
    delta = 0.01
    T_initial = 1.0
    alpha = 0.99
    max_anneal = 500
    
    def project(h):
        h = np.clip(h, 0, 1)
        s = h.sum()
        if s <= 1e-9:
            h[:] = required_sum / n_points
        else:
            h *= required_sum / s
        h = np.clip(h, 0, 1)
        return h
    
    def perturb(h, delta=None):
        h = h.copy()
        idx = np.random.choice(n_points, size=min(3, n_points))
        for i in idx:
            h[i] += np.random.uniform(-delta if delta is not None else -0.05, delta if delta is not None else 0.05)
        return project(h)
    
    # Local optimization with smoothing
    for sigma in sigma_grid:
        if time() - start_time > 1080: break  # 20s margin
        current_h = project(scipy.ndimage.gaussian_filter1d(best_h, sigma, mode='reflect'))
        for _ in range(local_steps):
            if time() - start_time > 1080: break
            for i in range(n_points):
                for d in (+1, -1):
                    temp = current_h.copy()
                    temp[i] += d * delta
                    temp = project(temp)
                    c = evaluate_erdos_solution(temp)
                    if c < best_c5:
                        best_c5 = c
                        best_h[:] = temp
                        current_h[:] = temp
                        break
    
    # Simulated annealing
    T = T_initial
    current_h = best_h.copy()
    current_c = best_c5
    
    for _ in range(max_anneal):
        if time() - start_time > 1080: break
        new_h = perturb(current_h)
        new_c = evaluate_erdos_solution(new_h)
        delta_c = current_c - new_c
        if delta_c > 0 or np.exp(delta_c / T) > np.random.rand():
            current_c = new_c
            current_h[:] = new_h[:]
        T *= alpha
    
    # Final projection
    best_h = project(best_h)
    
    return (best_h, best_c5, n_points)
