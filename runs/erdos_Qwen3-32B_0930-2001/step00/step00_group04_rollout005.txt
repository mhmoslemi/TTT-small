    pbest_scores = np.full(swarm_size, np.inf)
    pbests = particles.copy()
    gbest_score = np.inf
    gbest = particles[0].copy()

    start_time = time.time()

    while time.time() - start_time < budget_s:
        for i in range(swarm_size):
            h = particles[i]
            c5 = evaluate_c5(h, n_points)
            if c5 < pbest_scores[i]:
                pbest_scores[i] = c5
                pbests[i] = h.copy()
                if c5 < gbest_score:
                    gbest_score = c5
                    gbest = h.copy()

        # PSO update
        inertia = 0.7 + 0.3 * npr.rand()  # adaptive inertia
        cognitive = 1.5
        social = 1.5
        for i in range(swarm_size):
            r1, r2 = npr.rand(2)
            vel_update = inertia * velocities[i] + cognitive*r1*(pbests[i] - particles[i]) + social*r2*(gbest - particles[i])
            new_pos = particles[i] + vel_update
            particles[i] = project(new_pos, target_sum)
            velocities[i] = vel_update

    return gbest, gbest_score, n_points
