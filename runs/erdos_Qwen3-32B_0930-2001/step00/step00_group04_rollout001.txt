def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    try:
        h_initial = initial_h_values.copy()
        n_points = len(h_initial)
    except NameError:
        # Fallback with dummy initialization (initial_h_values should be available per task)
        n_points = 42
        h_initial = np.ones(n_points) * 0.5

    target_sum = n_points / 2.0

    swarm_size = 30
    positions = np.zeros((swarm_size, n_points))
    velocities = np.zeros((swarm_size, n_points))

    # Initialize with perturbations of initial_h_values
    positions[0] = h_initial.copy()
    for i in range(1, swarm_size):
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        h = h_initial + perturb
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        positions[i] = h

    pbest_positions = positions.copy()
    pbest_scores = np.array([evaluate_objective(p, n_points) for p in positions])

    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]

    w, c1, c2 = 0.7, 1.5, 1.5

    while time.time() - start_time < budget_s:
        # Update personal bests and global best
        for i in range(swarm_size):
            current = positions[i]
            current_score = evaluate_objective(current, n_points)
            if current_score < pbest_scores[i]:
                pbest_scores[i] = current_score
                pbest_positions[i] = current.copy()
                if current_score < gbest_score:
                    gbest_score = current_score
                    gbest_position = current.copy()

        # Update velocities and positions
        for i in range(swarm_size):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (pbest_positions[i] - positions[i]) +
                             c2 * r2 * (gbest_position - positions[i]))
            positions[i] += velocities[i]
            # Enforce constraints
            positions[i] = np.clip(positions[i], 0, 1)
            sum_pos = np.sum(positions[i])
            if sum_pos != 0:
                positions[i] = positions[i] * target_sum / sum_pos
                positions[i] = np.clip(positions[i], 0, 1)
            else:
                positions[i] = np.clip(positions[i], 0, 1)
                positions[i] = positions[i] * target_sum / np.sum(positions[i]) if np.sum(positions[i]) != 0 else 0

        if time.time() - start_time >= budget_s:
            break

    # Final evaluation with provided function
    c5_bound = evaluate_erdos_solution(gbest_position)
    return gbest_position.tolist(), c5_bound, n_points
