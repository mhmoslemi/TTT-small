def ensure_feasibility(h, n_points):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to make sum(h) = n_points / 2
    current_sum = np.sum(h)
    scaling_factor = (n_points / 2) / current_sum
    h = h * scaling_factor
    # After scaling, clip again in case any values exceed their bounds
    h = np.clip(h, 0, 1)
    return h

        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if current_sum < 1e-9:
            h += 1e-8
            current_sum = h.sum()

        h = np.clip(h, 0, 1)

        return max_c5
    
    def crossover(parent1, parent2):
        point = np.random.randint(1, len(parent1) - 1)

        child2 = np.concatenate((parent2[:point], parent1[point:])).copy()
        return child1, child2
