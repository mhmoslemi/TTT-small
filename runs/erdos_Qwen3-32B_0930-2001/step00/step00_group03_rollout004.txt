import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

        return np.max(corr) * dx

    best_c5 = evaluate_C5(h)
    t0 = time.time()

    cooling_rate = 0.999
    step_size = 0.01
    step_decay = 0.999
    n_eval = 0

    while time.time() - t0 < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k_lag = max_idx - (n_points - 1)
        k = abs(k_lag)

        if k >= n_points:
            continue

        part1 = np.zeros(n_points)
        part1[k:] = -h[:n_points - k]
        part2 = np.zeros(n_points)
