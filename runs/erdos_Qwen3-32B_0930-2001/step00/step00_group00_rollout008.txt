import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    h_init = initial_h_values
    n_points = len(h_init)
    dx = 2.0 / n_points

    def objective(h):
        h = np.asarray(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0., 1.) for _ in h_init]
    constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}

    result = minimize(
        objective,
        h_init,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 200, 'ftol': 1e-7, 'disp': False}
    )

    h_opt = result.x
    c5 = objective(h_opt)

    return h_opt.tolist(), c5, n_points
