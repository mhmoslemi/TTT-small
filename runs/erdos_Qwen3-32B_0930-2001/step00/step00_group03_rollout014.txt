def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h)

    best_C5 = evaluate_C5(h)

    alpha_decay = 0.999

    T_decay = 0.999

    T = T_initial

        if current_time - start_time > budget_s:
            break

        max_corr_idx = np.argmax(correlations)

            grad[i + m] -= h[i]
        elif k_max < 0:
            m = -k_max
            i = np.arange(m, n_points)
            grad[i] += 1 - h[i - m]
            grad[i - m] -= h[i]

        new_max_corr = np.max(new_correlations)

        current_C5 = np.max(correlations) * dx

            current_C5 = new_C5
            if current_C5 < best_C5:
                best_h = h.copy()
                best_C5 = current_C5
        
        alpha *= alpha_decay
        T *= T_decay
