    D = target_sum - current_sum

    if abs(D) < 1e-10:
        return h_clipped

    n_points = len(h_clipped)
    if D > 0:
        mask = h_clipped < 1.0
        available = np.sum(1.0 - h_clipped[mask])
        if available < D - 1e-10:
            return h_clipped
        alpha = D / available
        h_clipped[mask] += (1.0 - h_clipped[mask]) * alpha
    else:
        D = -D
        mask = h_clipped > 1e-10
        available = np.sum(h_clipped[mask])
        if available < D - 1e-10:
            return h_clipped
        alpha = D / available
        h_clipped[mask] -= h_clipped[mask] * alpha
    return h_clipped

def perturb_copy(h):
    new_h = h.copy()
    n_points = len(h)
    n_perturb = np.random.randint(1, min(10, n_points))
    indices = np.random.choice(n_points, size=n_perturb, replace=False)
    for i in indices:
        delta = np.random.uniform(-0.1, 0.1)
        new_h[i] += delta
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    h = np.array(initial_h_values)
    n_points = len(h)
    target_sum = n_points / 2.0
    if not np.isclose(np.sum(h), target_sum):
        raise ValueError("Initial h does not satisfy sum constraint.")

    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)

    while time.time() - start_time < budget_s - 5:
        new_h = perturb_copy(best_h)
        new_h_projected = project_h(new_h, target_sum)
        new_c5 = compute_c5(new_h_projected, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h[:] = new_h_projected.copy()

    return (best_h.tolist(), best_c5, n_points)
