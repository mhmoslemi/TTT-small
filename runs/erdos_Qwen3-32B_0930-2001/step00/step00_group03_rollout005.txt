import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points

    T = 1.0
    alpha_temp = 0.999
    step_size = 0.01
    beta_step = 0.999

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Compute worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        c_values = corr * dx
        k_max_index = np.argmax(c_values)
        k = k_max_index - (n_points - 1)

        # Compute gradient for this shift
        grad = np.zeros(n_points)
        if k >= 0:
            for i in range(n_points - k):
                j = i + k
                grad[i] += (1 - h[j])
                grad[j] += (-h[i])
        else:
            k_abs = -k
            for i in range(k_abs, n_points):
                j = i - k_abs
                grad[i] += (1 - h[j])
                grad[j] += (-h[i])

        # Update candidate
        candidate_h = h - step_size * grad
        candidate_h = np.clip(candidate_h, 0, 1)
        current_sum = np.sum(candidate_h)
        if current_sum > 1e-10:
            scaling_factor = (n_points / 2.0) / current_sum
            candidate_h *= scaling_factor
            candidate_h = np.clip(candidate_h, 0, 1)

        # Evaluate new candidate
        corr_new = np.correlate(candidate_h, 1 - candidate_h, mode='full')
        new_c5 = np.max(corr_new * dx)
        current_c5 = np.max(c_values)

        # Accept or reject
        if new_c5 < current_c5:
            h[:] = candidate_h[:]
            if new_c5 < best_c5:
                best_h[:] = h[:]
                best_c5 = new_c5
        else:
            prob = np.exp((current_c5 - new_c5) / T)

                h[:] = candidate_h[:]

        # Decay parameters
        step_size *= beta_step
        T *= alpha_temp
