import numpy as np
from scipy.optimize import minimize
import time

def run():
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        result = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(result)

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    options = {'maxiter': 500, 'ftol': 1e-8, 'disp': False}

    best_c5 = objective(h_initial)
    best_h = h_initial.copy()

    def optimize_h(h0):
        h_flat_initial = h0.copy()
        result = minimize(
            objective,
            h_flat_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options=options
        )
        if result.success:
            h_opt = result.x.reshape(n_points)
            current_c5 = objective(h_opt)
            if current_c5 < best_c5:
                return h_opt, current_c5
        return best_h, best_c5

    # Initial optimization
    h_opt, c5 = optimize_h(h_initial)
    best_h, best_c5 = h_opt, c5

    # Perturbed restarts
    np.random.seed(42)
    for _ in range(5):
        if time.time() - start_time >= 1100:
            break
        h_perturbed = np.clip(h_initial + 0.05 * np.random.randn(n_points), 0, 1)
        # Adjust for constraint
        scale = (n_points / 2) / h_perturbed.sum()
        h_perturbed = np.clip(h_perturbed * scale, 0, 1)
        h_opt, c5 = optimize_h(h_perturbed)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    return (best_h, best_c5, n_points)
