    target_sum = n_points / 2
    assert np.allclose(np.sum(h_initial), target_sum), "Initial sum is invalid"
    
    def objective(h_flat):
        h = np.clip(h_flat, 0, 1)
        return evaluate_erdos_solution(h)
    
    constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    best_x = h_initial.copy()
    
    def time_check(xk):
        nonlocal best_x
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            best_x[:] = np.clip(xk, 0, 1)
            raise RuntimeError("Time budget exceeded")
    
    try:
        result = scipy.optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-6},
            callback=time_check
        )
        h_opt = result.x
    except RuntimeError as e:
        if str(e) == "Time budget exceeded":
            h_opt = best_x
        else:
            h_opt = h_initial
    
    # Project to feasible region
    h_opt = np.clip(h_opt, 0, 1)
    current_sum = np.sum(h_opt)
    if not np.allclose(current_sum, target_sum, atol=1e-5):
        delta = target_sum - current_sum
        h_opt += delta / n_points
        h_opt = np.clip(h_opt, 0, 1)
    
    c5_bound = evaluate_erdos_solution(h_opt)
    return h_opt, c5_bound, n_points
