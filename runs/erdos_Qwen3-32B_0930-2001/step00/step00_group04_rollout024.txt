def run(seed=42, budget_s=1100, **kwargs):
    # Use initial_h_values
    h_initial = initial_h_values
    n_points = len(h_initial)
    required_sum = n_points / 2.0
    dx = 2.0 / n_points

    # PSO parameters
    swarm_size = 30
    max_iter = 10000  # adjust based on budget
    w = 0.7
    c1 = 1.5
    c2 = 1.5

    # Initialize particles
    np.random.seed(seed)
    # First particle is the initial one
    particles = [h_initial.astype(float)]
    # Add others
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = h * (required_sum / np.sum(h)) if np.sum(h) != 0 else np.full(n_points, required_sum / n_points)
        h = np.clip(h, 0, 1)
        particles.append(h)
    particles = np.array(particles)

    # Velocities initialized to 0
    velocities = np.zeros_like(particles)

    # Evaluate initial best
    def evaluate_particle(h):
        h_constrained = enforce_constraints(h, required_sum)
        correlations = np.correlate(h_constrained, 1 - h_constrained, mode='full')
        return np.max(correlations) * dx

    # Enforce constraints function
    def enforce_constraints(h, required_sum):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            h_scaled = np.full_like(h_clipped, required_sum / len(h_clipped))
        else:
            h_scaled = h_clipped * (required_sum / sum_clipped)
        # Clip again to make sure no precision issues
        h_scaled = np.clip(h_scaled, 0, 1)
        return h_scaled
    
    # Initial evaluation
    personal_best_positions = particles.copy()
    personal_best_scores = np.array([evaluate_particle(p) for p in particles])
    global_best_idx = np.argmin(personal_best_scores)
    global_best_position = personal_best_positions[global_best_idx]
    global_best_score = personal_best_scores[global_best_idx]

    # Timing
    start_time = time()
    iterations = 0
    while time() - start_time < budget_s:
        for i in range(swarm_size):
            # Update velocity
            r1 = np.random.rand()
            r2 = np.random.rand()
            cognitive_vel = c1 * r1 * (personal_best_positions[i] - particles[i])
            social_vel = c2 * r2 * (global_best_position - particles[i])
            velocities[i] = w * velocities[i] + cognitive_vel + social_vel

            # Update position
            particles[i] += velocities[i]

            # Enforce constraints
            particles[i] = enforce_constraints(particles[i], required_sum)

        # Evaluate new particles
        scores = np.array([evaluate_particle(p) for p in particles])

        improved = scores < personal_best_scores

            global_best_position = personal_best_positions[current_global_idx]

        iterations += 1
        # Optional: add iteration limit to prevent infinite loops
        if iterations >= max_iter:

    # Return the best solution found

import numpy as np
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values
    n_points = len(h_initial)
    required_sum = n_points / 2.0
    dx = 2.0 / n_points

    swarm_size = 30
    max_iter = 10000
    w = 0.7
    c1 = 1.5
    c2 = 1.5

    np.random.seed(seed)
    particles = [h_initial.astype(np.float64)]
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = h * (required_sum / np.sum(h)) if np.sum(h) != 0 else np.full(n_points, required_sum / n_points)
        h = np.clip(h, 0, 1)
        particles.append(h)
    particles = np.array(particles)
    velocities = np.zeros_like(particles)

    def enforce_constraints(h, req_sum):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.full_like(h_clipped, req_sum / n_points)
        h_scaled = h_clipped * (req_sum / sum_clipped)
        h_scaled = np.clip(h_scaled, 0, 1)
        return h_scaled

    def evaluate_particle(h):
        h_constrained = enforce_constraints(h, required_sum)
        corr = np.correlate(h_constrained, 1 - h_constrained, mode='full')
        return np.max(corr) * dx

    personal_best_positions = [p.copy() for p in particles]
    personal_best_scores = np.array([evaluate_particle(p) for p in particles])
    global_best_idx = np.argmin(personal_best_scores)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]

    start_time = time()
    iter_count = 0
    while time() - start_time < budget_s:
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (personal_best_positions[i] - particles[i]) +
                             c2 * r2 * (global_best_position - particles[i]))
            particles[i] += velocities[i]
            particles[i] = enforce_constraints(particles[i], required_sum)

        scores = np.array([evaluate_particle(p) for p in particles])
        improved = scores < personal_best_scores
        for i in range(swarm_size):
            if improved[i]:
                personal_best_positions[i] = particles[i].copy()
                personal_best_scores[i] = scores[i]
        current_global_idx = np.argmin(personal_best_scores)
        if personal_best_scores[current_global_idx] < global_best_score:
            global_best_score = personal_best_scores[current_global_idx]
            global_best_position = personal_best_positions[current_global_idx].copy()

        iter_count += 1
        if iter_count >= max_iter:
            break

    return (global_best_position.tolist(), global_best_score, n_points)
