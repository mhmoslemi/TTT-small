import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_init = initial_h_values.copy()
    else:
        n_points = 42
        h_init = np.zeros(n_points)
    
    n_particles = 30
    inertia = 0.7
    c1, c2 = 1.5, 1.5
    dx = 2.0 / n_points
    S = n_points / 2.0
    
    particles = []
    velocities = []
    pb = []
    pb_vals = []
    
    # Initialize with initial_h_values
    def init_particle(h):
        h_clip = np.clip(h, 0, 1)
        total = h_clip.sum()
        if total != 0:
            h_clip = h_clip * (S / total)
        h_clip = np.clip(h_clip, 0, 1)
        return h_clip
    
    particles.append(init_particle(h_init))
    velocities.append(np.zeros(n_points))
    pb.append(particles[0].copy())
    corr = np.correlate(pb[0], 1 - pb[0], mode='full')
    pb_vals.append(np.max(corr * dx))

    for _ in range(1, n_particles):
        h = np.random.rand(n_points)
        h_clip = init_particle(h)
        particles.append(h_clip)
        velocities.append(np.random.uniform(-0.2, 0.2, n_points))
        pb.append(h_clip.copy())
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        pb_vals.append(np.max(corr * dx))
    
    gbest_val = min(pb_vals)
    gbest = pb[pb_vals.index(gbest_val)].copy()
    
    while time.time() - start_time < budget_s - 1:
