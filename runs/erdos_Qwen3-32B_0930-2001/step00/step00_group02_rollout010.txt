        grad = np.zeros(n)
        for i in range(n - k_max):
            grad[i] += (1.0 - h[i + k_max]) * dx
            grad[i + k_max] -= h[i] * dx
        
        # Trust region subproblem: quadratic approximation
        delta = cp.Variable(n)
        objective = cp.Minimize(grad @ delta + 0.5 * cp.sum_squares(delta) * (-2 * dx))
        constraints = [
            cp.norm(delta) <= delta_radius,
            delta <= 1 - h,
            delta >= -h,
            cp.sum(delta) == 0
        ]
        
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if prob.status != cp.OPTIMAL:
            continue
            
        delta_val = delta.value
        h_new = h + delta_val
        h_new = np.clip(h_new, 0, 1)
        h_new += (n/2 - h_new.sum()) / n  # maintain average constraint
        
        # Update best if improved
        new_c5, _ = evaluate_erdos_solution(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            delta_radius = min(delta_radius * 1.1, max_radius)
            alpha = min(alpha + 0.1, 0.9)
        else:
            delta_radius = max(delta_radius * 0.5, min_radius)
            alpha = max(alpha - 0.1, 0.1)
        
        # Momentum step
        h = h + alpha * (h_new - h)
        h = np.clip(h, 0, 1)
        h += (n/2 - h.sum()) / n
    
    return best_h.tolist(), best_c5, n
