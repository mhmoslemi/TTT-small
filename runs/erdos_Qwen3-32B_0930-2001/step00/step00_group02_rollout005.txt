        init_list.append(sin_h)

        # Adjust to meet sum constraint

        scale = (n_points / 2) / current_sum
        perturbed_h *= scale
        perturbed_h = np.clip(perturbed_h, 0, 1)
        init_list.append(perturbed_h)
        return init_list

        patience = 0

            current_c5 = np.max(overlap)
