def project_h(h, n_points):
    target_sum = n_points / 2
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if np.allclose(current_sum, target_sum):
        return h_clipped
    scale_factor = target_sum / current_sum
    h_scaled = h_clipped * scale_factor
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled

import time

def project_h(h, n_points):
    target_sum = n_points / 2
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    scale_factor = target_sum / current_sum if not np.isclose(current_sum, 0) else 1.0
    h_scaled = h_clipped * scale_factor
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled

def compute_gradient(h, n_points):
    dx = 2.0 / n_points
    C_k = []
    # Compute C_k for k=1 to n_points-1
    for k in range(1, n_points):
        overlap = n_points - k

        total *= dx

    for i in range(n_points):
        total_k0 += h[i] * (1 - h[i])
