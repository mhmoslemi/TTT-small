        sum_i = pos.sum()
        if sum_i < 1e-9:
            pos = h0 * (desired_sum / np.sum(h0))
        else:
            pos = pos * (desired_sum / sum_i)
        vel = (np.random.rand(n_points) - 0.5) * max_velocity
        particles.append(pos.copy())
        velocities.append(vel.copy())
        fit = compute_c5(pos, n_points)
        personal_best.append(pos.copy())
        personal_best_fitness.append(fit)
    
    global_best = personal_best[0].copy()
    global_best_fitness = personal_best_fitness[0]
    for i in range(1, n_particles):
        if personal_best_fitness[i] < global_best_fitness:
            global_best = personal_best[i].copy()
            global_best_fitness = personal_best_fitness[i]
    
    max_pso_iter = 300
    for iter_pso in range(max_pso_iter):
        if time.time() - start_time > budget_s - 1:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (velocities[i] * inertia_weight +
                             c1 * r1 * (personal_best[i] - particles[i]) +
                             c2 * r2 * (global_best - particles[i]))
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_i = particles[i].sum()
            if sum_i < 1e-9:
                particles[i] = h0 * (desired_sum / np.sum(h0))
            else:
                particles[i] = particles[i] * (desired_sum / sum_i)
            fit_i = compute_c5(particles[i], n_points)
            
            if fit_i < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fit_i
                if fit_i < global_best_fitness:
                    global_best = particles[i].copy()
                    global_best_fitness = fit_i
    
    def objective(x):
        return compute_c5(x, n_points)
    
    def eq_constraint(x):
        return np.sum(x) - desired_sum
    
    constraints = {'type': 'eq', 'fun': eq_constraint}
    bounds = [(0.0, 1.0)] * n_points
    
    if time.time() - start_time < budget_s - 1:
        res = scipy.optimize.minimize(
            objective, x0=global_best.copy(), method='SLSQP',
            bounds=bounds, constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if res.success:
            final_h = res.x
        else:
            final_h = global_best.copy()
    else:
        final_h = global_best.copy()
    
    final_h = np.clip(final_h, 0.0, 1.0)
    final_sum = final_h.sum()
    final_h = final_h * (desired_sum / final_sum)
    
    final_c5 = compute_c5(final_h, n_points)
    
    return final_h.tolist(), final_c5, n_points
