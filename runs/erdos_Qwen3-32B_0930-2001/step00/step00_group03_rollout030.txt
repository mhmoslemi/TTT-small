            return np.clip(h_clipped, 0, 1)
        h_clipped[eligible] -= (-delta) / num_eligible
    return np.clip(h_clipped, 0, 1)

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = project_feasible(h_initial, n_points)
    
    np.random.seed(seed)
    random.seed(seed)
    
    population_size = 50
    population = [h_initial.copy()]
    for _ in range(population_size - 1):
        h_perturbed = h_initial + np.random.normal(0, 0.1, n_points)
        h_perturbed = project_feasible(h_perturbed, n_points)
        population.append(h_perturbed)
    population = np.array(population)
    
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h, n_points)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        c5_values = np.zeros(population_size)
        for i in range(population_size):
            c5_values[i] = evaluate_c5(population[i], n_points)
        current_best_idx = np.argmin(c5_values)
        current_best_c5 = c5_values[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
        
        n_select = max(1, int(population_size * 0.2))
        selected_indices = np.argsort(c5_values)[:n_select]
        selected_parents = population[selected_indices]
        
        offspring = []
        for _ in range(population_size - n_select):
            p1, p2 = npr.choice(selected_parents, 2)
            child = np.where(npr.rand(n_points) < 0.8, p1, p2)
            child += npr.normal(0, 0.02, n_points)
            child = project_feasible(child, n_points)
            offspring.append(child)
        
        population = np.array(offspring + selected_parents.tolist())
    
    final_c5 = evaluate_c5(best_h, n_points)
    return (best_h.tolist(), final_c5, n_points)
