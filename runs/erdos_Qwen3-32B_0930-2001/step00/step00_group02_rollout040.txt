import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    learning_rate = 0.01
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        corrs = np.correlate(h, 1 - h, mode='full') * dx
        max_idx = np.argmax(corrs)
        shift_k = max_idx - (n_points - 1)
        k = abs(shift_k)
        if k < 0:
            k = 0
        if k >= n_points:
            k = n_points - 1

        grad = np.zeros_like(h)
        for m in range(n_points):
            term1 = 0.0
            if m >= k:

            if m + k < n_points:
                term2 = -h[m + k] * dx
            grad[m] = term1 + term2
        h -= learning_rate * grad
        h_clipped = np.clip(h, 0.0, 1.0)
