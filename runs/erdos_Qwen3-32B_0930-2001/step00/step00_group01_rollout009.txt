import numpy as np
import time

def run(seed=42, budget_s=1100, population_size=25, max_generations=500, mutation_strength=0.05):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
    else:
        n_points = 47  # Fallback if no initial_h_values is provided
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / h_initial.sum() * (n_points / 2)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial / h_initial.sum() * (n_points / 2)
    n_points = len(h_initial)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def generate_individual():
        h = np.random.uniform(0, 1, n_points)
        h = h / h.sum() * target_sum
        h = np.clip(h, 0, 1)
        h = h / h.sum() * target_sum
        return h

    # Initialize population
    population = [h_initial]
    for _ in range(population_size - 1):
        population.append(generate_individual())

    best_h = None
    best_c5 = float('inf')

    while time.time() - start_time < budget_s:
        # Evaluate population
        eval_results = []
        for h in population:
            other = 1 - h
            corr = np.correlate(h, other, mode='full') * dx
            c5 = np.max(corr)
            eval_results.append((h, c5))
        
        # Sort by C5
        eval_results.sort(key=lambda x: x[1])
        if eval_results[0][1] < best_c5:
            best_c5 = eval_results[0][1]
            best_h = eval_results[0][0].copy()

        # Select parents (top half)
        selected = [h for h, _ in eval_results[:population_size // 2]]
        new_population = [eval_results[0][0]]  # Elitism

        while len(new_population) < population_size:
            # Crossover
            idx1, idx2 = np.random.choice(len(selected), 2, replace=False)
            p1, p2 = selected[idx1], selected[idx2]
            split = np.random.randint(1, n_points - 1)
            child = np.concatenate((p1[:split], p2[split:])).copy()
            
            # Mutate
            child += np.random.normal(0, mutation_strength, n_points)
            child = np.clip(child, 0, 1)
            child = child / child.sum() * target_sum

            new_population.append(child)
        
        population = new_population

    # Final evaluation
    other = 1 - best_h
    corr = np.correlate(best_h, other, mode='full') * dx
    final_c5 = np.max(corr)
    return (best_h, final_c5, n_points)
