    best_c5, _ = compute_c5_and_kmax(best_h, n_points)
    current_h = h.copy()
    current_c5, _ = compute_c5_and_kmax(current_h, n_points)
    T = 1.0
    alpha_step = 0.01
    cooling_rate = 0.9995
    step_size_decay = 0.9999
    iteration = 0
    while time.time() - start_time < budget_s:
        current_c5, k_max = compute_c5_and_kmax(current_h, n_points)
        grad = compute_gradient(current_h, k_max, n_points)
        new_h_candidate = current_h - alpha_step * grad
        new_h_candidate = np.clip(new_h_candidate, 0, 1)
        new_h_candidate = adjust_sum(new_h_candidate, n_points / 2.0)
        new_c5, _ = compute_c5_and_kmax(new_h_candidate, n_points)
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0:
            current_h = new_h_candidate
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h_candidate
                best_c5 = new_c5
        else:
            prob = np.exp(-delta_c5 / (T + 1e-8))
            if np.random.rand() < prob:
                current_h = new_h_candidate
                current_c5 = new_c5
        T *= cooling_rate
        alpha_step *= step_size_decay
        iteration += 1
    return best_h, best_c5, n_points
