        current_c5 = np.max(corr)

        if current_c5 < best_c5:
            best_h[:] = h.copy()

        # Identify the shift causing maximum overlap
        max_idx = np.argmax(corr)
        shift_k = max_idx - (n_points - 1)  # Convert correlate index to shift

        # Compute gradient for this shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Contribution from h[i] as numerator in h[i]*(1-h[j])
            if 0 <= i + shift_k < n_points:
