def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time
    np.random.seed(seed)
    
    initial_h = globals().get('initial_h_values', None)
    if initial_h is not None:
        h = np.array(initial_h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    n_points = len(h)
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    current_temp = 0.2
    final_temp = 1e-5
    cooling_rate = 0.99999
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        if best_c5 < 0.38085:
            break
        perturbation = np.random.uniform(-current_temp, current_temp, size=h.shape)
        new_h = h + perturbation
        new_h = np.clip(new_h, 0.0, 1.0)
        sum_h = new_h.sum()
        if abs(sum_h) < 1e-9:
            continue
        scaling = (n_points / 2) / sum_h
        new_h *= scaling
        
        try:
            new_c5 = evaluate_erdos_solution(new_h, dx)
        except:
            continue
        
        delta = new_c5 - best_c5
        if delta < 0 or np.exp(-delta / current_temp) > np.random.rand():
            h = new_h.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                if best_c5 < 0.38085:
                    break
        
        current_temp *= cooling_rate
        current_temp = max(current_temp, final_temp)
    
    return best_h, best_c5, n_points
