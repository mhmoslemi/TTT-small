    return h, evaluate_c5(h, dx)

def simulated_annealing(h, n_points, dx, target_sum, initial_T=1.0, cooling_rate=0.97, max_iter=200):
    h = h.copy()
    current_c5 = evaluate_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    temperature = initial_T
    for _ in range(max_iter):
        num_changes = np.random.randint(1, 5)
        indices = np.random.choice(n_points, num_changes, replace=False)
        delta = 0.05 * np.random.randn(num_changes)
        h_perturbed = h.copy()
        for i, idx in enumerate(indices):
            h_perturbed[idx] += delta[i]
            h_perturbed[idx] = np.clip(h_perturbed[idx], 0.0, 1.0)
        h_proj = project_h(h_perturbed, target_sum)
        new_c5 = evaluate_c5(h_proj, dx)
        if (new_c5 < current_c5 or
            temperature > 1e-6 and
            np.random.rand() < np.exp((current_c5 - new_c5) / temperature)):
            h = h_proj.copy()
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_proj.copy()
        temperature *= cooling_rate
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    sigma_grid = [0.05, 0.1, 0.2, 0.3, 0.4]
    start_time = time.time()

    for sigma in sigma_grid:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        try:
            # Apply Gaussian blur
            b_h = scipy.ndimage.gaussian_filter1d(h, sigma, mode='reflect')
            p_h = project_h(b_h, target_sum)
            # Local optimization
            h_local, c5_local = optimize_local(p_h, delta=0.01, max_iter=50, target_sum=target_sum, dx=dx)
            if c5_local < best_c5:
                best_h, best_c5 = h_local, c5_local
            # Simulated annealing
            if elapsed + 20 < budget_s:  # avoid using entire budget
                sa_h, sa_c5 = simulated_annealing(h_local, n_points, dx, target_sum, max_iter=150)
                if sa_c5 < best_c5:
                    best_h, best_c5 = sa_h, sa_c5
        except:
            continue

    # Final projection and validation
    best_h = project_h(best_h, target_sum)
    return best_h, evaluate_c5(best_h, dx), n_points
