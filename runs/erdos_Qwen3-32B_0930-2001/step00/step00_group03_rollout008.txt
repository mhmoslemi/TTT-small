def run(seed=42, budget_s=1100, **kwargs):
    # Initialize h with initial_h_values
    n = len(initial_h_values)
    h = initial_h_values.copy()
    np.random.seed(seed)
    start_time = time.time()
    best_h = h.copy()
    best_c5 = None
    
    # Parameters for simulated annealing and gradient descent
    T = 1.0  # Initial temperature
    cooling_rate = 0.999
    alpha = 0.01  # Initial step size
    alpha_decay = 0.999
    
    while time.time() - start_time < budget_s:
        # Compute current C5 and find the shift k
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        C5 = max_corr * dx
        
        if best_c5 is None or C5 < best_c5:
            best_h = h.copy()
            best_c5 = C5
        
        # Find the index m where maximum occurs
        m = np.argmax(corr)
        # Compute the corresponding shift k_shift
        k_shift = m - (n - 1)
        k = abs(k_shift)
        # Ensure k is in the valid range [0, n-1]
        if k >= n:
            k = n -1  # edge case?
        
        # Compute gradient for k
        grad = np.zeros(n)
        grad[:n - k] += (1 - h[k:]) * dx
        grad[k:] += (-h[:n - k]) * dx
        
        # Generate candidate h_candidate
        candidate = h - alpha * grad
        # Clip to [0,1]
        candidate = np.clip(candidate, 0, 1)
        # Project to sum to n/2
        current_sum = np.sum(candidate)
        if current_sum != 0:
            candidate = candidate * (n/2) / current_sum
        
        # Compute new_C5
        new_corr = np.correlate(candidate, 1 - candidate, mode='full')
        new_max_corr = np.max(new_corr)
        new_C5 = new_max_corr * dx
        
        # Acceptance criterion
        if new_C5 < C5:
            h = candidate.copy()
            C5 = new_C5
        else:
            # Probability of acceptance
            delta = C5 - new_C5
            p = np.exp(delta / T)
            if np.random.rand() < p:
                h = candidate.copy()
                C5 = new_C5
        
        # Cool down
        T *= cooling_rate
        alpha *= alpha_decay
    
    # Return the best found
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_max = np.max(final_corr)
    c5_bound = final_max * (2.0 / len(best_h))
    return best_h.tolist(), c5_bound, len(best_h)

import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h = initial_h_values.copy()
    n = len(h)
    np.random.seed(seed)
    start_time = time.time()
    best_h = h.copy()
    best_c5 = None
    T = 1.0  # Temperature
    alpha = 0.01  # Step size
    cooling_rate = 0.999
    alpha_decay = 0.9995

    while time.time() - start_time < budget_s:
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        C5 = max_corr * dx

        if best_c5 is None or C5 < best_c5:
            best_h = h.copy()
            best_c5 = C5

        m = np.argmax(corr)
        k_shift = m - (n - 1)
        k = abs(k_shift)
        if k >= n:
            k = n - 1

        grad = np.zeros(n)
        grad[:n - k] += (1 - h[k:]) * dx
        grad[k:] += (-h[:n - k]) * dx

        candidate = h - alpha * grad
        candidate = np.clip(candidate, 0, 1)
        current_sum = np.sum(candidate)
        if current_sum > 1e-10:
            candidate = candidate * (n / 2) / current_sum

        new_corr = np.correlate(candidate, 1 - candidate, mode='full')
        new_max_corr = np.max(new_corr)
        new_C5 = new_max_corr * dx

        if new_C5 < C5:
            h = candidate.copy()
            C5 = new_C5
        else:
            delta = C5 - new_C5
            p = np.exp(delta / T)
            if np.random.rand() < p:
                h = candidate.copy()
                C5 = new_C5

        T *= cooling_rate
        alpha *= alpha_decay

    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_max = np.max(final_corr)
    c5_bound = final_max * (2.0 / len(best_h))
