import numpy as np
from scipy.optimize import minimize, Bounds
import time


def run(seed=42, budget_s=1100, **kwargs):
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    desired_sum = n_points / 2.0

    def objective(h_flat):
        h = h_flat.reshape((n_points,))
        dh = 1.0 - h
        corr = np.correlate(h, dh, mode='full')
        dx = 2.0 / n_points
        return np.max(corr * dx)

    h_flat = h0.copy().flatten()
    bounds = Bounds(0, 1)

    def sum_eq_constraint(x):
        return x.sum() - desired_sum

    cons = ({'type': 'eq', 'fun': sum_eq_constraint})

    best_h = h_flat.copy()
    best_obj = objective(best_h)
    start_time = time.time()

    def wrapped_obj(x_i):
        nonlocal best_h, best_obj
        if time.time() - start_time > 0.99 * budget_s:
            raise RuntimeError("Time budget exhausted")
        current_val = objective(x_i)
        if current_val < best_obj:
            best_h[:] = x_i.copy()
            best_obj = current_val
        return current_val

    options = {'maxiter': 1000, 'ftol': 1e-8}

    try:
        result = minimize(wrapped_obj, h_flat, method='SLSQP', bounds=bounds, constraints=cons, options=options)
    except (RuntimeError, StopIteration):
        pass
    
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = best_h.sum()
    if abs(best_h_sum - desired_sum) > 1e-6:
        diff = desired_sum - best_h_sum
        best_h[0] += diff

    h_final = best_h.reshape(n_points)
    c5_bound = objective(h_final)
    
    return h_final.tolist(), c5_bound, n_points
