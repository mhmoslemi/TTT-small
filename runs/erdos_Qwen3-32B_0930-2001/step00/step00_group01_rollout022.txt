import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    h_clamped = np.clip(h, 0, 1)
    current_sum = h_clamped.sum()
    if np.isclose(current_sum, target_sum):
        return h_clamped.copy()
    scale = target_sum / current_sum
    h_scaled = h_clamped * scale
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled

def run(seed=42, budget_s=1100, **kwargs):
    h0 = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h0)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    hfeas = project_feasible(h0, target_sum)
    while not np.isclose(hfeas.sum(), target_sum, atol=1e-8):
        hfeas = (hfeas * 0.99) + 0.005  # Perturb to avoid 0/1 clipping
        hfeas = project_feasible(hfeas, target_sum)

    constraints = {'type': 'eq', 'fun': lambda h: h.sum() - target_sum}
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h_vec):
        return evaluate_erdos_solution(h_vec)

    best_h = hfeas.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    try:
        result = minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-8}
        )
        if result.success:
            best_h = project_feasible(result.x, target_sum)
            best_c5 = min(best_c5, objective(best_h))
    except:
        pass

    elapsed = time.time() - start_time
    if elapsed < budget_s:
        try:
