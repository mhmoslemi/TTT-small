import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_init = initial_h_values.copy()
    n_points = len(h_init)
    dx = 2.0 / n_points
    population_size = 25

    # Initialize population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        scaling = (n_points / 2.0) / h.sum()
        h = np.clip(h * scaling, 0, 1)
        population.append(h)
    population = np.array(population)

    def evaluate(h):
        h_minus = 1 - h
        corr = np.correlate(h, h_minus, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    best_c5 = evaluate(h_init)
    best_h = h_init.copy()

    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = np.array([evaluate(h) for h in population])
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()

        # Select top 50% parents
        top_indices = np.argsort(fitness)[: population_size // 2]
        parents = population[top_indices]
        parent_fitness = fitness[top_indices]

        # Generate new population
        new_population = []
        for _ in range(population_size):
            # Tournament selection
            t1 = np.random.randint(len(parents), size=3)
