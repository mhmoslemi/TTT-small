import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n_points = len(h)

        h = np.ones(n_points) * 0.5
    dx = 2.0 / n_points

    final_temp = 1e-5
    cooling_factor = (final_temp / initials) ** (1.0 / max_iter)
    time0 = time.time()
    current_temp = initials

    for iter in range(max_iter):
        if time.time() - time0 > budget_s:
            break

            new_c5 = evaluate_erdos_solution(new_h, dx)
