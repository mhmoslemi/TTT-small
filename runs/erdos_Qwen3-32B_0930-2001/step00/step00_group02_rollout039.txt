    # Clip to [0,1]
    h_clipped = np.clip(h, 0, 1)
    # Compute current sum
    current_sum = h_clipped.sum()
    if abs(current_sum - target_sum) < 1e-6:
        return h_clipped
    # If current sum is too low, scale up, but only up to 1
    # This is an approximation
    scale_factor = target_sum / current_sum
    h_scaled = h_clipped * scale_factor
    # Clip again
    h_scaled = np.clip(h_scaled, 0, 1)

    current_sum = h_scaled.sum()
    if abs(current_sum - target_sum) < 1e-6:
