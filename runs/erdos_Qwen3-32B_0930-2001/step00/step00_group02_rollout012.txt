            try:

                if "GLPK" in cp.installed_solvers():

import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    h = initial_h_values.copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        try:
            # Compute current overlap using provided evaluator
            current_corr = np.correlate(h, 1 - h, mode='full') * dx
            max_overlap_idx = np.argmax(current_corr)
            # Map index to lag k
            k = max_overlap_idx - (n - 1)
            
            # Determine valid indices for this lag
            if k >= 0:
                valid_i = np.arange(0, n - k)
            else:
                valid_i = np.arange(-k, n)
            
            # Compute coefficients for linearization
            coeffs = np.zeros(n)
            for i in valid_i:
                j = i + k
                coeffs[j] += h[i]
            
            # Formulate and solve LP
            h_new = cp.Variable(n)
            constraints = [0 <= h_new, h_new <= 1, cp.sum(h_new) == n / 2]
            objective = cp.Maximize(dx * coeffs @ h_new)
            prob = cp.Problem(objective, constraints)
            
            try:
                if "GLPK" in cp.installed_solvers():
                    prob.solve(solver=cp.GLPK)
                else:
                    prob.solve(solver=cp.ECOS)
            except:
                continue
            
            if prob.status == cp.OPTIMAL:
                h[:] = h_new.value
                current_c5 = evaluate_erdos_solution(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        except Exception as e:
            break
    
    return best_h, best_c5, n
