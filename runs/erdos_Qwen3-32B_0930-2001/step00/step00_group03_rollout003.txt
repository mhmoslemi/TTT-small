def run(seed=42, budget_s=1100, **kwargs):
    # Use initial_h_values
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    # Validate initial h
    assert np.allclose(np.sum(h), n_points / 2, atol=1e-8), "Initial h sum incorrect"
    assert np.all((h >= 0.0) & (h <= 1.0)), "Initial h out of bounds"
    
    # Parameters for optimization
    alpha = 0.01  # Step size
    beta = 0.999
    temperature = 1.0
    alpha_temp = 0.999
    max_iter = 10**6  # High upper bound, limited by time
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    for iter_count in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.1:
            break
        
        # Compute current C5
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        k_max_index = np.argmax(corr)
        shift_steps = k_max_index - (n_points - 1)
        
        # Compute gradient for this shift_steps
        gradient = np.zeros_like(h)
        if shift_steps != 0:
            if shift_steps > 0:
                k = shift_steps
                for i in range(n_points - k):
                    gradient[i] += 1 - h[i + k]
                    gradient[i + k] += -h[i]
            else:
                k = -shift_steps
                for i in range(k, n_points):
                    gradient[i] += 1 - h[i - k]
                    gradient[i - k] += -h[i]
        else:
            # shift_steps = 0
            gradient = 1 - 2 * h
            
        # Update h with gradient step
        h_candidate = h - alpha * gradient
        
        # Clip to [0, 1]
        h_candidate = np.clip(h_candidate, 0.0, 1.0)
        
        # Project to sum = n_points / 2
        sum_h = np.sum(h_candidate)
        if np.abs(sum_h) < 1e-10:
            h_candidate = np.zeros_like(h)
        else:
            h_candidate *= (n_points / 2) / sum_h
        
        # Evaluate new_c5
        new_corr = np.correlate(h_candidate, 1 - h_candidate, mode='full') * dx
        new_c5 = np.max(new_corr)
        
        # Acceptance via simulated annealing
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / temperature):
            h = h_candidate
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        # Update temperature and step size
        temperature *= alpha_temp
        alpha *= beta
        
    # Final validation
    assert np.allclose(np.sum(best_h), n_points / 2, atol=1e-8)
    assert np.all((best_h >= 0.0) & (best_h <= 1.0))
    
    return best_h.tolist(), best_c5, n_points

import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    assert np.allclose(np.sum(h), n_points / 2, atol=1e-8)
    assert np.all((h >= 0.0) & (h <= 1.0))
    
    alpha = 0.01
    beta = 0.999
    temperature = 1.0
    alpha_temp = 0.999
    start_time = time.time()
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    
    while time.time() - start_time < budget_s - 0.1:
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        k_max_index = np.argmax(corr)
        shift_steps = k_max_index - (n_points - 1)
        
        gradient = np.zeros_like(h)
        if shift_steps != 0:
            if shift_steps > 0:
                k = shift_steps
                valid_indices = n_points - k
                for i in range(valid_indices):
                    gradient[i] += 1 - h[i + k]
                    gradient[i + k] += -h[i]
            else:
                k = -shift_steps
                for i in range(k, n_points):
                    gradient[i] += 1 - h[i - k]
                    gradient[i - k] += -h[i]
        else:
            gradient = 1 - 2 * h
        
        h_candidate = h - alpha * gradient * dx
        h_candidate = np.clip(h_candidate, 0.0, 1.0)
        sum_h = np.sum(h_candidate)
        if abs(sum_h) > 1e-10:
            h_candidate *= (n_points / 2) / sum_h
        else:
            h_candidate[:] = 0.5
        
        new_corr = np.correlate(h_candidate, 1 - h_candidate, mode='full') * dx
        new_c5 = np.max(new_corr)
        
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / temperature):
            h = h_candidate
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        alpha = alpha * beta
        temperature *= alpha_temp
    
    assert np.allclose(np.sum(best_h), n_points / 2, atol=1e-8)
    assert np.all((best_h >= 0.0) & (best_h <= 1.0))
    return best_h.tolist(), best_c5, n_points
