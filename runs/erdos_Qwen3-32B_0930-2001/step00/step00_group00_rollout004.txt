        new_h = np.clip(new_h, 0, 1)
        sum_h = new_h.sum()
        scaling_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        new_h *= scaling_factor

        try:
            new_c5 = evaluate_erdos_solution(new_h, dx)
        except:
            continue

        delta = new_c5 - best_c5
        if delta < 0 or np.exp(-delta / current_temp) > np.random.rand():
            h = new_h.copy()
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5

        current_temp *= cooling_factor

    return best_h, best_c5, n_points
