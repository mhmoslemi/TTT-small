import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy().astype(np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h *= target_sum / h.sum()
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5, _ = evaluate_erdos_solution(best_h)
    start_time = time.time()

    # Adam parameters
    lr = 0.1
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    restarts = 3
    max_iter = 1000

    for restart in range(restarts):
        if restart > 0:
            # Perturb current best solution
            h = best_h.copy()
            h += np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h *= target_sum / h.sum()

        m = np.zeros(n_points)
        v = np.zeros(n_points)
        t = 0

        for i in range(max_iter):
            if time.time() - start_time > budget_s - 10:
                break
            # Sample shifts
            shifts = np.arange(1, n_points)
            selected_shifts = shifts[np.random.choice(len(shifts), size=max(1, len(shifts) // 10), replace=False)]

            max_overlap = -np.inf
            k_max = 0
            for k in selected_shifts:
                if k >= n_points:
                    continue
                overlap = np.sum(h[:-k] * (1 - h[k:]) * dx)
                if overlap > max_overlap:
                    max_overlap = overlap
                    k_max = k

            # Compute gradient for k_max
            grad = np.zeros(n_points)
            for i in range(n_points):
                if i + k_max < n_points:
                    grad[i] += (1 - h[i + k_max]) * dx
                if i - k_max >= 0:
                    grad[i] -= h[i - k_max] * dx

            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            h -= step
            h = np.clip(h, 0, 1)
            h *= target_sum / h.sum()

            # Evaluate full C5 occasionally
            if i % 20 == 0 or i == max_iter - 1:
                current_c5, _ = evaluate_erdos_solution(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()

    return best_h, best_c5, n_points
