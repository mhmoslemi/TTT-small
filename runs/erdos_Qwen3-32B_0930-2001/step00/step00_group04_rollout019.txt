import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values
    n_points = len(h_initial)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    def compute_c5(h):
        h_clipped = np.clip(h, 0, 1)
        h_clipped = h_clipped / h_clipped.sum() * target_sum
        corr = correlate(h_clipped, 1 - h_clipped, mode="full")
        return corr.max() * dx
    
    def project_h(h):
        h_clipped = np.clip(h, 0, 1)
        s = h_clipped.sum()
        if s == 0:
            h_clipped[np.random.randint(n_points)] = 1.0
            s = 1.0
        return h_clipped / s * target_sum
    
    particles = np.array([project_h(h_initial + np.random.normal(0, 0.1, n_points)) for _ in range(20)])
    velocities = np.zeros_like(particles)
    pbest = particles.copy()
    pbest_scores = np.array([compute_c5(p) for p in particles])
    gbest = particles[pbest_scores.argmin()].copy()
    gbest_score = pbest_scores.min()
    
    w, c1, c2 = 0.7, 1.5, 1.5
    max_vel = 0.1
    
    start_time = time.time()
    while time.time() - start_time < min(600, budget_s - 200):
        for i in range(particles.shape[0]):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] +
                            c1 * r1 * (pbest[i] - particles[i]) +
                            c2 * r2 * (gbest - particles[i]))
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            particles[i] = project_h(particles[i] + velocities[i])
            score = compute_c5(particles[i])
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
                if score < gbest_score:
                    gbest[:] = particles[i]
                    gbest_score = score
    
    def obj(x):
        return compute_c5(x)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(project_h(x)) - target_sum}]
    bounds = [(0, 1)] * n_points
    result = minimize(obj, project_h(gbest), method='SLSQP', bounds=bounds, constraints=cons, 
                     options={'maxiter': 100, 'ftol': 1e-8, 'disp': False})
    
    h_best = project_h(result.x)
    final_score = compute_c5(h_best)
    
    return (h_best, final_score, n_points)
