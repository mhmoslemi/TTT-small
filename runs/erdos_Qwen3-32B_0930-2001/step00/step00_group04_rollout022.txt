import time

def evaluate_particle(h, n_points):
    dx = 2.0 / n_points
    h1 = 1.0 - h
    corr = np.correlate(h, h1, mode="full")
    max_corr = np.max(corr) * dx
    return max_corr

    velocities = np.zeros_like(positions)
