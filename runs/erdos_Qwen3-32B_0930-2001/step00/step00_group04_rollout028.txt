import numpy as np
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h_vec):
        h = h_vec.reshape(-1)
        one_minus_h = 1 - h
        correl = np.correlate(h, one_minus_h, mode='full') * dx
        return np.max(correl)

    sum_constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}
    bounds = [(0.0, 1.0)] * n_points

    options = {'maxiter': 300, 'disp': False}
    start_time = time()

    result = minimize(
        fun=objective,
        x0=initial_h,
        bounds=bounds,
        constraints=[sum_constraint],
        method='SLSQP',
        options=options
    )

    elapsed = time() - start_time
    remaining = max(budget_s - elapsed, 0)
    if remaining > 20:
        result2 = minimize(
            fun=objective,
            x0=result.x,
            bounds=bounds,
            constraints=[sum_constraint],
            method='Nelder-Mead',
            options={'maxiter': 150, 'disp': False}
        )
        result = result2

    h_opt = np.clip(result.x, 0, 1)
    c5_bound = objective(h_opt)
    return h_opt, c5_bound, n_points
