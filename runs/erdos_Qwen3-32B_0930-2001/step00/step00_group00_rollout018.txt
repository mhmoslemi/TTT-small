def compute_C5(h, n_points):
    dx = 2.0 / n_points
    h1 = 1.0 - h
    corr = correlate(h, h1, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def objective_with_scaling(h, n_points):
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    scale_factor = target_sum / current_sum if current_sum != 0 else 1.0
