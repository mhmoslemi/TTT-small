from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()

        nonlocal best_x, best_val
        val = objective(x)

            best_x[0] = x.copy()

            objective,
            h,
            bounds=bounds,
            constraints=constraints,
