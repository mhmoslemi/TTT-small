import numpy as np
import time

def run():
    h = initial_h_values.copy()
    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)[0]

    T = 1.0
    alpha_T = 0.999
    alpha_step_size = 0.999
    step_size = 0.01

    start_time = time.time()
    budget_s = 1100

    while time.time() - start_time < budget_s:
        # Compute current C5 and k_max
        corr = np.correlate(h, 1 - h, mode='full') * dx
        c5_current = np.max(corr)
        idx = np.argmax(corr)
        k_max = idx - (n - 1)

        # Compute gradient
        grad = np.zeros(n)
        if k_max >= 0:
            valid = n - k_max
            grad[0:valid] += (1.0 - h[k_max : k_max + valid]) * dx
            grad[k_max : k_max + valid] -= h[0:valid] * dx
        else:
            m = -k_max
            valid = n - m
            grad[m : m + valid] += (1.0 - h[0:valid]) * dx
            grad[0:valid] -= h[m : m + valid] * dx

        # Update h_candidate
        h_candidate = h - step_size * grad

        # Clip and scale
        h_candidate = np.clip(h_candidate, 0, 1)
        s = np.sum(h_candidate)
        if s > 1e-10:
            scale = target_sum / s
            h_candidate = np.clip(h_candidate * scale, 0, 1)

        # Evaluate new C5
        corr_new = np.correlate(h_candidate, 1 - h_candidate, mode='full') * dx
        c5_new = np.max(corr_new)

        # Acceptance decision
        if c5_new < c5_current:
            h = h_candidate.copy()
            if c5_new < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_new
        else:
            delta = c5_current - c5_new
            if delta > 0:
                prob = np.exp(delta / T)
                if np.random.rand() < prob:
                    h = h_candidate.copy()

        # Update parameters
        T *= alpha_T
        step_size *= alpha_step_size

    return best_h, best_c5, n
