    dx = 2.0 / n_points

    # Initialize
    h = h_initial.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    start_time = time.time()
    max_iter = 10000
    iter_no_improve = 0
    learning_rate = 0.05

    while time.time() - start_time < budget_s and iter_no_improve < max_iter:
        corr = np.correlate(h, 1 - h, mode="full")
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)  # Compute actual lag

        # Gradient computation
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i - lag
            if 0 <= j < n_points:
                grad[i] += (1.0 - h[j])
                grad[j] -= h[i]
        grad *= dx  # Scale by dx for objective function

        # Gradient step
        h = h - learning_rate * grad

        # Project onto feasible set
        h = project(h, n_points)

        # Update best solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            iter_no_improve = 0
        else:
            iter_no_improve += 1

        # Adaptive learning rate
        if iter_no_improve % 200 == 0:
            learning_rate *= 0.95

    return (best_h.tolist(), best_c5, n_points)
