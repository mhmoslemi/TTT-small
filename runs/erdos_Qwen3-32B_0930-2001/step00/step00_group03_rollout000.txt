        for _ in range(perturbation_steps):
            if time.time() - start_time > budget_s - 10:
                break
            i, j = np.random.choice(n_points, 2, replace=False)
            delta = 0.01 * (np.random.rand() - 0.5)
            h_new = h_curr.copy()
            # Ensure move keeps values within bounds
            max_hdl = min(1.0 - h_new[i], h_new[j] - 0.0)
            max_del = np.clip(delta, -max_hdl, max_hdl)
            h_new[i] += max_del
            h_new[j] -= max_del
            h_pert = project_feasible(h_new)
            c5_new = evaluate_c5(h_pert)
            
            if c5_new < c5_curr:
                h_curr, c5_curr = h_pert, c5_new
                if c5_curr < best_c5:
                    best_h, best_c5 = h_curr.copy(), c5_new
                    
    # Simulated Annealing
    if time.time() - start_time < budget_s - 10:
        T = 1.0
        cooling_rate = 0.99
        h_ann = best_h.copy()
        c5_ann = best_c5
        
        for step in range(annealing_steps):
            if time.time() - start_time > budget_s - 10:
                break
            i, j = np.random.choice(n_points, 2, replace=False)
            delta = 0.01 * (np.random.rand() - 0.5)
            h_new = h_ann.copy()
            max_hdl = min(1.0 - h_new[i], h_new[j] - 0.0)
            max_del = np.clip(delta, -max_hdl, max_hdl)
            h_new[i] += max_del
            h_new[j] -= max_del
            h_pert = project_feasible(h_new)
            c5_new = evaluate_c5(h_pert)
            
            if c5_new < c5_ann:
                h_ann, c5_ann = h_pert, c5_new
                if c5_ann < best_c5:
                    best_h, best_c5 = h_ann.copy(), c5_ann
            else:
                if np.random.rand() < np.exp((c5_ann - c5_new) / T):
                    h_ann, c5_ann = h_pert, c5_new
            T *= cooling_rate
    
    h = project_feasible(best_h)
    c5 = evaluate_c5(h)
    
    return h.tolist(), c5, n_points
