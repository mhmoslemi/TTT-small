import numpy as np
import cvxpy as cp
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()
    
    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)
    
    h_best = h_initial.copy()
    c5_best = compute_c5(h_best)
    
    # Solve LP relaxation for structured initial guess
    h_lp = cp.Variable(n_points, nonneg=True)
    t = cp.Variable()
    constraints = [
        cp.sum(h_lp) == target_sum,
        h_lp >= 0,
        h_lp <= 1
    ]
    
    for k in range(1, n_points):
        # Collect indices for positive shift k and negative shift -k
        indices = []
        # Positive shift
        for i in range(n_points - k):
            indices.append((i, i + k))
        # Negative shift
        for i in range(k, n_points):
            indices.append((i, i - k))
        if not indices:
            continue
        z_vars = []
        for i, j in indices:
            expr = h_lp[i] + h_lp[j] - 1
            z = cp.Variable(nonneg=True)
            constraints += [z >= expr]
            z_vars.append(z)
        # Sum for current k's all (i,j) pairs
        constraints += [cp.sum(z_vars) >= t]
    
    try:
        problem = cp.Problem(cp.Maximize(t), constraints)
        problem.solve(solver=cp.ECOS, max_iters=200)
        h_lp_val = np.clip(h_lp.value, 0.0, 1.0) if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE] else h_initial.copy()
    except:
        h_lp_val = h_initial.copy()
    # Ensure sum constraint
    adjustment = (target_sum - np.sum(h_lp_val)) / n_points
    h_lp_val = np.clip(h_lp_val + adjustment, 0.0, 1.0)
    
    # Local refinement with perturbed escape
    def optimize_from_guess(guess):
        def obj_func(x):
            x = x.reshape(n_points)
            return compute_c5(x)
        return optimize.minimize(
            obj_func,
            x0=guess.copy().reshape(n_points),
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{
                'type': 'eq',
                'fun': lambda x: np.sum(x) - target_sum
            }],
            options={'maxiter': 60, 'ftol': 1e-12}
        )
    
    # Alternate between LP-inspired guess and recovery from initial guess
    candidates = []
    
    # First attempt: LP-based solution
    h_candidate1 = h_lp_val.copy().reshape(n_points)
    res1 = optimize_from_guess(h_candidate1)
    if res1.success:
        candidates.append((res1.fun, res1.x))
    
    # Second attempt: random perturbation
    if time.time() - start_time < budget_s - 60:
        h0 = h_initial.copy()
        h0 += np.random.normal(0, 0.15, n_points)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 -= (np.sum(h0) - target_sum) / n_points
        h0 = np.clip(h0, 0.0, 1.0)
        res2 = optimize_from_guess(h0)
        if res2.success:
            candidates.append((res2.fun, res2.x))
    
    # Final refinement with lowest C5
    if candidates:
        candidates.sort()
        best_c5, best_h = candidates[0]
        if best_c5 < c5_best:
            h_best = best_h.copy()
            c5_best = best_c5
    
    return h_best, c5_best, n_points
