import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):

    best_h = current_h.copy()
    best_c5 = current_c5

    temp = 10.0
    alpha = 0.995
    perturb_rate = 0.3
    noise_std = 0.05

    while time.time() - start_time < budget_s:
        h_candidate = current_h.copy()
        indices = np.random.choice(n_points, size=int(n_points * perturb_rate), replace=False)
        noise = np.random.normal(0, noise_std, size=len(indices))
        h_candidate[indices] += noise
        h_candidate = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h_candidate)

            h_candidate = np.zeros(n_points)
