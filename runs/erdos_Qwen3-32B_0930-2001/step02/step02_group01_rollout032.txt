    def compute_gradient(h_vec, k):
        n = len(h_vec)
        valid = n - k
        grad = np.zeros(n)
        grad[:valid] += (1 - h_vec[k:k+valid]) * dx
        grad[k:k+valid] -= h_vec[:valid] * dx
        return grad

    def project(h_vec, target_sum):
        h_proj = np.clip(h_vec, 0, 1)
        sum_h = h_proj.sum()
        if sum_h == 0:
            return np.clip(h_vec, 0, 1)
        scale = target_sum / sum_h
        h_proj *= scale
        h_proj = np.clip(h_proj, 0, 1)
        return h_proj

    # Coarse optimization (n=21)
    n_coarse = 21
    h_coarse = h_current[::2].copy()
    target_coarse = n_coarse / 2.0
    h_coarse = project(h_coarse, target_coarse)
    
    best_c5 = compute_max_shift(h_coarse)[1]
    best_h = h_coarse.copy()
    
    # Coarse grid optimization
    alpha_coarse = 0.002
    n_iter_coarse = 600
    for i in range(n_iter_coarse):
        if time.time() - start_time > budget_s - 200:
            break
        k, _ = compute_max_shift(best_h)
        grad = compute_gradient(best_h, k)
        h_candidate = best_h - alpha_coarse * grad
        h_candidate = project(h_candidate, target_coarse)
        new_c5 = compute_max_shift(h_candidate)[1]
        if new_c5 < best_c5:
            best_h = h_candidate
            best_c5 = new_c5

    # Refine to full resolution (n=42)
    h_fine = np.repeat(best_h, 2).copy()
    h_fine = project(h_fine, target_sum)
    best_h_fine = h_fine.copy()
    best_c5_fine = compute_max_shift(best_h_fine)[1]
    
    alpha_fine = 0.0015
    n_iter_fine = 2000
    for i in range(n_iter_fine):
        if time.time() - start_time > budget_s - 10:
            break
        k, _ = compute_max_shift(best_h_fine)
        grad = compute_gradient(best_h_fine, k)
        h_candidate = best_h_fine - alpha_fine * grad
        h_candidate = project(h_candidate, target_sum)
        new_c5 = compute_max_shift(h_candidate)[1]
        if new_c5 < best_c5_fine:
            best_h_fine = h_candidate
            best_c5_fine = new_c5

    elapsed = time.time() - start_time
    final_c5 = compute_max_shift(best_h_fine)[1]
    print(f"Elapsed: {elapsed:.1f}s, Final C5: {final_c5:.6f}")
    
    return best_h_fine, final_c5, n_points
