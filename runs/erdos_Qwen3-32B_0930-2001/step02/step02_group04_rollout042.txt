import numpy as np
import cvxpy as cp
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()
    
    # Solve relaxed LP to obtain initial guess
    h_var = cp.Variable(n_points)
    t = cp.Variable()
    constraints = [cp.sum(h_var) == target_sum]
    
    # Generate constraints for all possible shifts
    for k in range(-n_points + 1, n_points):
        overlap_indices = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                overlap_indices.append((i, j))
        if not overlap_indices:
            continue
        sum_expr = 0
        for i, j in overlap_indices:
            sum_expr += cp.max(Expression(h_var[i] + h_var[j] - 1), 0)
        constraints.append(sum_expr * dx >= t)
    
    # Formulate and solve LP
    objective = cp.Maximize(t)
    problem = cp.Problem(objective, constraints)
    
    try:
        problem.solve(solver=cp.GLPK, verbose=False)
        h_lp = h_var.value
        if h_lp is None:
            raise ValueError("LP failed")
    except:
        h_lp = initial_h_values.copy()
    
    # Local refinement with SLSQP
    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)
    
    h_current = h_lp.copy()
    h_best = h_current.copy()
    c5_best = compute_c5(h_best)
    
    # Perturbation-free restart with LP solution
    result = optimize.minimize(
        compute_c5,
        x0=h_best,
        method='SLSQP',
        bounds=[(0, 1)] * n_points,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
        options={'maxiter': 500, 'ftol': 1e-12}
    )
    if result.success:
        h_best = result.x
        c5_best = result.fun
    
    # Fallback to initial guess if LP refinement failed
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 20:
        h_initial = initial_h_values.copy()
        h_initial /= np.sum(h_initial) * dx  # Normalize for safety
        result = optimize.minimize(
            compute_c5,
            x0=h_initial,

        )
