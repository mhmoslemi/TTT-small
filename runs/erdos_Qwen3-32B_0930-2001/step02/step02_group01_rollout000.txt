        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    # Alternating optimization between upper/lower halves
    for _ in range(2):
        split = n_points // 2
        lower = h_current[:split]
        upper = h_current[split:]
        sum_lower = np.sum(lower)
        t_upper = target_sum - sum_lower
        def obj_upper(u): return np.max(np.correlate(np.clip(np.concatenate([lower, u]), 0, 1), 1 - np.clip(np.concatenate([lower, u]), 0, 1), 'full') * dx)
        res_upper = optimize.minimize(obj_upper, upper, bounds=[(0, 1)] * (n_points - split), method='SLSQP', constraints=[{'type': 'eq', 'fun': lambda u: np.sum(u) - t_upper}], options={'maxiter': 100})
        if res_upper.success:
            h_current[split:] = res_upper.x
        sum_upper = np.sum(h_current[split:])
