import numpy as np
import time
from scipy.interpolate import interp1d

def compute_overlaps(h, dx):
    return np.correlate(h, 1 - h, mode="full") * dx

def compute_smooth_max(overlaps, smooth_epsilon):
    max_val = np.max(overlaps)
    shifted = overlaps - max_val
    exp_vals = np.exp(shifted / smooth_epsilon)
    return max_val + smooth_epsilon * np.log(np.sum(exp_vals))

def compute_analytic_gradient(h, overlaps, dx, g_overlaps, smooth_epsilon):
    n = len(h)
    grad = np.zeros(n)
    overlaps_array = overlaps / dx  # Remove dx scaling to get back to unscaled overlaps
    # For each l (overlap index), compute s
    for l in range(len(overlaps_array)):
        s = l - (n - 1)  # shift s
        for m in range(n):
            # contribution from term 1: h[m] contributes to overlaps[l]
            # the derivative is (1 - h[m -s]) if m -s is valid
            ind = m - s
            if 0 <= ind < n:
                grad[m] += g_overlaps[l] * (1 - h[ind])
            # contribution from term 2: h[m +s] is involved, derivative is -h[m +s]
            ind2 = m + s
            if 0 <= ind2 < n:
                grad[m] -= g_overlaps[l] * h[ind2]
    return grad

import numpy as np

def compute_overlaps(h, dx):
    return np.correlate(h, 1 - h, mode="full") * dx

def compute_analytic_gradient(h, overlaps, dx, g_overlaps, smooth_epsilon):
    n = len(h)
    grad = np.zeros(n)
    overlaps_array = overlaps / dx  # Remove dx scaling for gradient computation
    for l in range(len(overlaps_array)):
        s = l - (n - 1)  # Compute shift s
        for m in range(n):
            # Term 1: (1 - h[m - s]) if m - s is valid
            ind1 = m - s
            if 0 <= ind1 < n:
                grad[m] += g_overlaps[l] * (1 - h[ind1])
            # Term 2: -h[m + s] if m + s is valid
            ind2 = m + s
            if 0 <= ind2 < n:
                grad[m] -= g_overlaps[l] * h[ind2]
    return grad
