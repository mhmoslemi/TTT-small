import numpy as np
import time

def run():
    start_time = time.time()
    h_init = initial_h_values.copy()
    n_points = len(h_init)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    lower = 0.0
    upper = 0.38085  # Current best bound
    best_upper = upper
    best_h = h_init.copy()

    while time.time() - start_time < 1100:
        mid = (lower + upper) / 2
        h = cp.Variable(n_points, nonneg=True)
        w = cp.Variable((n_points, n_points), nonneg=True)
        constraints = [
            cp.sum(h) == target_sum,
            h <= 1,
        ]

        # McCormick constraints for bilinear terms
        for i in range(n_points):
            for j in range(n_points):
                constraints += [
                    w[i, j] >= h[i] + h[j] - 1,
                    w[i, j] <= h[i],
                    w[i, j] <= h[j],
                ]

        # Constraints for each shift
        for k in range(-n_points + 1, n_points):
            valid_indices = []
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    valid_indices.append((i, j))
            if valid_indices:
                sum_expr = 0
                for i, j in valid_indices:
                    sum_expr += h[i] - w[i, j]
                constraints.append(sum_expr <= mid / dx)

        problem = cp.Problem(cp.Minimize(0), constraints)
        try:
            problem.solve(solver=cp.MOSEK, verbose=False)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                upper = mid
                best_upper = mid
                best_h = h.value.copy()
            else:
                lower = mid
        except:
            lower = mid

        if upper - lower < 1e-6:
            break

    return (best_h, best_upper, n_points)
