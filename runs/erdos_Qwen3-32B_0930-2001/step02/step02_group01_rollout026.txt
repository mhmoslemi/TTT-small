import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # sum(h) must equal this

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    def normalize_h(h):
        h = np.clip(h, 0, 1)
        s = h.sum()
        if s == 0: 
            return h
        h = h * (target_sum / s)
        return np.clip(h, 0, 1)

    best_h = normalize_h(h_initial)
    best_c5 = compute_c5(best_h)

    # Multi-start optimization loop
    num_starts = 20
    for i in range(num_starts):
        if time.time() - start_time >= budget_s - 30:
            break

        # Perturb current best
        seed_i = hash((seed, i)) % 2**32
        np.random.seed(seed_i)
        perturbed = best_h + np.random.normal(0, 0.15, n_points)
        perturbed = normalize_h(perturbed)

        # Nelder-Mead phase (global exploration)
        res_nm = optimize.minimize(
            compute_c5, 
            perturbed, 
            method='Nelder-Mead',
            bounds=[(0, 1)] * n_points,
            options={'maxiter': 80, 'xatol': 1e-4, 'fatol': 1e-5}
        )
        if res_nm.success:
            h_nm = normalize_h(res_nm.x)
        else:
            h_nm = perturbed
        
        # SLSQP refinement (local optimization)
        cons = {'type': 'eq', 'fun': lambda x: x.sum() - target_sum}
        res_sl = optimize.minimize(
            compute_c5, 
            h_nm, 
            method='SLSQP',
            bounds=[(0,1)]*n_points,
            constraints=cons,
            options={'maxiter': 250, 'ftol': 1e-9}
        )
        if res_sl.success:
            h_candidate = normalize_h(res_sl.x)
            c5_candidate = compute_c5(h_candidate)

    # Final polish phase (use tighter tolerances)
    final_t = time.time() - start_time
