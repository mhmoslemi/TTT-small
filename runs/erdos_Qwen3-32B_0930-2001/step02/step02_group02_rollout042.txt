def run():

    dx = 2.0 / n_points

    # Precompute all shifts and their (i,j) pairs
    all_shifts = []
    for k in range(-n_points + 1, n_points):
        pairs = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                pairs.append( (i, j) )
        if pairs:
            all_shifts.append( (k, pairs) )

    lower = 0.0
    upper = 0.38085
    best_c = upper
    best_h = h.copy()

    start_time = time.time()
    timeout = 1100

    while time.time() - start_time < timeout:
        mid = (lower + upper) / 2
        try:
            # Build and solve convex problem
            h_vars = cp.Variable(n_points, nonneg=True)
            w_vars = {}
            w_constraints = []
            for k, pairs in all_shifts:
                for i, j in pairs:
                    if (i, j) not in w_vars:
                        w = cp.Variable()
                        w_vars[(i, j)] = w
                        # McCormick constraints
                        w_constraints.append( w >= h_vars[i] + h_vars[j] - 1 )
                        w_constraints.append( w <= h_vars[i] )
                        w_constraints.append( w <= h_vars[j] )
                        w_constraints.append( w >= 0 )
                        # w_constraints.append( w >= 0 )

            # Constraints from the sum
            constraints = [
                cp.sum(h_vars) == target_sum,
                h_vars >= 0,
                h_vars <= 1
            ] + w_constraints

            # Add the per-shift constraints
            for k, pairs in all_shifts:
                sum_h = sum(h_vars[i] for i, j in pairs)
                sum_w = sum(w_vars[(i, j)] for i, j in pairs)
                constraint_value = sum_h - sum_w <= mid * n_points / 2
                constraints.append( constraint_value )

            # Problem
            objective = cp.Minimize(0)  # Feasibility
            problem = cp.Problem(objective, constraints)

            # Solve with MOSEK
            problem.solve(solver=cp.MOSEK, verbose=False, max_iters=1000)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                # Feasible solution found; update upper bound and save h
                best_c = mid
                best_h = h_vars.value.copy()
                upper = mid
            else:
                # Not feasible; lower the bound
                lower = mid
        except:
            # Avoid getting stuck on exception, just try next
            lower = mid

    return (best_h, best_c, n_points)
