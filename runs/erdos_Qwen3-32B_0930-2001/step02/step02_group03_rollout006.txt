import time

def compute_overlaps(h, dx):
    return np.correlate(h, 1 - h, mode="full") * dx

def compute_smooth_max(overlaps, smooth_epsilon):
    max_val = np.max(overlaps)
    shifted = overlaps - max_val
    exp_vals = np.exp(shifted / smooth_epsilon)
    return max_val + smooth_epsilon * np.log(np.sum(exp_vals))

    c1 = 1.5
    c2 = 1.5
    smooth_epsilon = 0.1
    epsilon_decay = 0.95
    max_iterations = 5000

    particles = np.random.rand(n_particles, n_points)
    velocities = np.zeros((n_particles, n_points))
    personal_best = particles.copy()
    personal_best_values = np.full(n_particles, np.inf)

    # Initialize with initial_h_values if available
    if len(h_init) == n_points:
        particles[0] = h_init.copy()
        particles[0] = np.clip(particles[0], 0, 1)
        h_sum = particles[0].sum()
        if h_sum != 0:
            particles[0] *= target_sum / h_sum
        else:
            particles[0].fill(0.5)
        personal_best[0] = particles[0].copy()

    # Normalize initial particles to satisfy constraints
    for i in range(n_particles):
        h = particles[i]
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (target_sum / h_sum) if h_sum != 0 else h * 0.5

    global_best_idx = 0
