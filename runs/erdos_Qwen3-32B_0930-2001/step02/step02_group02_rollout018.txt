import numpy as np
from scipy.stats import multivariate_normal
import time

def run():
    seed = 42
    budget_s = 1100
    np.random.seed(seed)
    
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    def evaluate(h_vec):
        return np.max(np.correlate(h_vec, 1 - h_vec, mode='full')) * dx
    
    def repair(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return h_clipped
        scaling = target_sum / current_sum
        h_scaled = h_clipped * scaling
        h_scaled = np.clip(h_scaled, 0, 1)
        return h_scaled
    
    num_samples = 60
    elite_proportion = 0.2
    initial_cov_diag = 0.15
    cov_min = 1e-8
    
    mean = h.copy()
    covariance = np.diag([initial_cov_diag] * n_points) + cov_min * np.eye(n_points)
    
    best_c5 = evaluate(mean)
    best_h = mean.copy()
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        samples = multivariate_normal.rvs(mean=mean, cov=covariance, size=num_samples)
        candidates = []
        for sample in samples:
            repaired = repair(sample)
            c5 = evaluate(repaired)
            candidates.append( (c5, repaired) )
        
        candidates.sort()
        num_elite = int(num_samples * elite_proportion)
        elites = candidates[:num_elite]
        
        if not elites:
            break
        
        if elites[0][0] < best_c5:
            best_c5 = elites[0][0]
            best_h = elites[0][1].copy()
        
        elite_samples = np.array([c[1] for c in elites])
        new_mean = np.mean(elite_samples, axis=0)
        new_var = np.var(elite_samples, axis=0) + cov_min
        covariance = np.diag(new_var)
        mean = new_mean
    
    best_h = repair(best_h)
    return (best_h, best_c5, n_points)
