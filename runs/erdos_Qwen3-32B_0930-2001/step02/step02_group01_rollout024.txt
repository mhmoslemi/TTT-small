import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_current = initial_h_values.copy()
    n_points = len(h_current)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    def finalize_h(x):
        h = np.array(x).copy()
        last_val = target_sum - h.sum()
        h = np.append(h, last_val)
        h = np.clip(h, 0, 1)
        scaled_sum = h.sum()
        if scaled_sum == 0:
            return np.clip(h, 0, 1) * target_sum / n_points
        scale = target_sum / scaled_sum
        h = h * scale
        return np.clip(h, 0, 1)

    best_c5 = float('inf')
    best_h = h_current.copy()
    current_c5 = compute_c5(best_h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = best_h.copy()

    n_perturb = 20
    max_nm_iter = 50
    max_slsqp_iter = 100
    perturb_mag = 0.1

    for pid in range(n_perturb):
        if time.time() - start_time > budget_s - 5:
            break

        pert = np.random.randn(n_points) * perturb_mag
        h_pert = h_current + pert
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = finalize_h(h_pert)

        def objective_nm(x):
            h = np.array(x).copy()
            return compute_c5(finalize_h(h))

        result_nm = optimize.minimize(
            objective_nm, h_pert,
            method='Nelder-Mead',
            bounds=[(0, 1)] * n_points,
            options={'maxiter': max_nm_iter}
        )
        if not result_nm.success:
            h_nm = h_pert.copy()
        else:
            h_nm = finalize_h(result_nm.x)

        x_sl_init = h_nm[:n_points - 1]

        def objective_sl(x):
            h = x.copy()
            last_val = target_sum - h.sum()
            h_full = np.append(h, last_val)
            return compute_c5(finalize_h(h_full))

        result_sl = optimize.minimize(
            objective_sl, x_sl_init,
            method='SLSQP',
            bounds=[(0, 1)] * (n_points - 1),
            options={'maxiter': max_slsqp_iter, 'ftol': 1e-8}
        )

        if result_sl.success:
            h_sl = np.append(result_sl.x, target_sum - result_sl.x.sum())
            h_sl = finalize_h(h_sl)
            c5 = compute_c5(h_sl)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_sl.copy()
        else:
            c5 = compute_c5(h_nm)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_nm.copy()

    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Best C5: {best_c5:.6f}")
    return best_h, best_c5, n_points
