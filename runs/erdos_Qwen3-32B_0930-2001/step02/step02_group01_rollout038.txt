import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_mean = initial_h_values.copy()
    n_points = len(h_mean)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def repair(h_vec):
        h = np.clip(h_vec, 0, 1)
        current_sum = h.sum()
        if current_sum == 0:
            h = np.zeros_like(h)
            idx = np.random.randint(n_points)
            h[idx] = target_sum
            h = np.clip(h, 0, 1)
            return h
        scale = target_sum / current_sum
        h = h * scale
        return np.clip(h, 0, 1)
    
    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)
    
    population_size = 100
    elite_frac = 0.1
    n_elites = int(population_size * elite_frac)
    max_iter = 300
    sigma = 0.1
    
    h_history = [h_mean]
    best_c5 = float('inf')
    best_h = h_mean
    
    h_current = repair(h_mean)
    current_c5 = compute_c5(h_current)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_current
    
    while time.time() - start_time < budget_s - 20 and len(h_history) < max_iter:
        population = []
        # Generate population with noise
        for _ in range(population_size):
            pert = np.random.randn(n_points) * sigma
            h_pert = h_current + pert
            h_pert = repair(h_pert)
            population.append(h_pert)
        population = np.array(population)
        
        # Evaluate all
        scores = np.array([compute_c5(p) for p in population])
        # Track new best
        idx_improve = np.argmin(scores)
        if scores[idx_improve] < best_c5:
            best_c5 = scores[idx_improve]
            best_h = population[idx_improve]
        
        # Select elites
        elite_indices = np.argsort(scores)[:n_elites]
        elite_population = population[elite_indices]
        
        # Update mean and sigma
        new_mean = elite_population.mean(axis=0)
        new_sigma = max(0.01, np.std(elite_population - h_current, axis=0).mean() * 0.8)
        
        h_current = new_mean
        sigma = new_sigma
        
        h_history.append(h_current)
    
    # Final refinement with SLSQP
    def objective(x):
        x = x.reshape(n_points)
        return compute_c5(x)
    
    bounds = [(0,1)] * n_points
    constraints_slsqp = {'type': 'eq', 'fun': lambda x: x.sum() - target_sum}
    result = optimize.minimize(
        objective, best_h, method='SLSQP',
        bounds=bounds, constraints=constraints_slsqp,
        options={'maxiter': 200, 'ftol': 1e-8}
    )
    if result.success:
        final_h = result.x
        final_c5 = compute_c5(final_h)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    return best_h, best_c5, n_points
