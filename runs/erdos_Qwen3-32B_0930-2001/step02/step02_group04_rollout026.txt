    def adjust_sum(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_clipped)
        if np.isclose(current_sum, 0):
            return np.full(n_points, target_sum / n_points)
        scaling_factor = target_sum / current_sum
        h_scaled = h_clipped * scaling_factor
        h_clipped = np.clip(h_scaled, 0, 1)
        return h_clipped

    # Final refinement and validation
    if result.success:
        h_best = adjust_sum(result.x)
    else:
        h_best = adjust_sum(h)

    # Recompute final C5 without penalty
    conv = np.correlate(h_best, 1 - h_best, mode='full')
    final_c5 = np.max(conv * dx)

    return h_best, final_c5, n_points
