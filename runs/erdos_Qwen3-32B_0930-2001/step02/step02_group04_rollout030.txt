import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()
    
    # Solve LP relaxation
    h_var = cp.Variable(n_points)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target_sum,
        h_var >= 0,
        h_var <= 1
    ]
    
    for k_shift in range(- (n_points - 1), n_points):
        if k_shift == 0:
            sum_expr = 0.0
            for i in range(n_points):
                expr = h_var[i] + h_var[i] - 1
                sum_expr += cp.pos(expr)
            sum_expr *= dx
            constraints.append(sum_expr >= t_var)
            continue

        valid_i = []
        if k_shift > 0:
            max_i = n_points - k_shift - 1
            valid_i = list(range(0, max_i + 1))
        else:
            m = -k_shift
            valid_i = list(range(m, n_points))
        
        sum_expr = 0.0
        for i in valid_i:
            i_plus_k = i + k_shift
            if 0 <= i_plus_k < n_points:
                expr = h_var[i] + h_var[i_plus_k] - 1
                sum_expr += cp.pos(expr)
        sum_expr *= dx
        constraints.append(sum_expr >= t_var)
    
    objective = cp.Maximize(t_var)
    problem = cp.Problem(objective, constraints)
    try:
        problem.solve(solver=cp.ECOS, max_iters=20, verbose=False)
        h_lp = np.array(h_var.value)
        h_best = h_lp.copy()
    except Exception:
        h_best = h.copy()
    
    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)
    
    c5_best = compute_c5(h_best)
    
    # Local optimization with SLSQP and perturbations
    def objective_func(x):
        return np.max(np.correlate(x, 1 - x, mode='full') * dx)
    
    def constraint_sum(x):
        return np.sum(x) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints_opt = {'type': 'eq', 'fun': constraint_sum}
    
    while time.time() - start_time < budget_s - 10:
        result = optimize.minimize(
            objective_func,
            x0=h_best,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints_opt,
            options={'maxiter': 50}
        )
        if result.success:
            c5_candidate = result.fun
            h_candidate = result.x
            if c5_candidate < c5_best:
                c5_best = c5_candidate
                h_best[:] = h_candidate
        # Perturbation
        noise = np.random.normal(0, 0.05, size=n_points)
        current_sum = np.sum(h_best)
        adjustment = (target_sum - current_sum) / n_points
        h_best = np.clip(h_best + noise + adjustment, 0, 1)
    
    return h_best, c5_best, n_points
