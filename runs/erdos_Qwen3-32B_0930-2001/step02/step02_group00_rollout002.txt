import numpy as np
import time

def run():
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    T = n_points / 2
    dx = 2.0 / n_points

    # Initialize swarm
    swarm_size = 10
    positions = [h_initial.copy()]
    velocities = [np.zeros(n_points)]

    # Generate remaining particles
    for _ in range(swarm_size - 1):
        h = positions[0].copy()
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        # Project to constraint
        s = h.sum()
        if s != 0:
            h = T / s * h
        else:
            h[0] = T  # Avoid zero-case
        positions.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))

    positions = np.array(positions)
    velocities = np.array(velocities)

    def objective(h):
        return np.max(np.correlate(h, 1 - h, mode='full') * dx)

    pbest = positions.copy()
    pbest_scores = np.array([objective(p) for p in positions])
    gbest = positions[pbest_scores.argmin()].copy()
    best_c5 = pbest_scores.min()

    while time.time() - start_time < 1100:
        for i in range(swarm_size):
            # Velocity update
            r1, r2 = np.random.rand(2)
            velocities[i] = (0.7 * velocities[i] +
                           2.0 * r1 * (pbest[i] - positions[i]) +
                           2.0 * r2 * (gbest - positions[i]))
            velocities[i] = np.clip(velocities[i], -1.0, 1.0)

            # Position update
            positions[i] += velocities[i]
            positions[i] = np.clip(positions[i], 0, 1)

            # Constraint projection
            s = positions[i].sum()
            if abs(s) > 1e-8:
                positions[i] *= T / s
            else:
                positions[i][0] = T

            # Update scores
            current_score = objective(positions[i])
            if current_score < pbest_scores[i]:
                pbest[i] = positions[i].copy()
                pbest_scores[i] = current_score
                if current_score < best_c5:
                    gbest = positions[i].copy()
                    best_c5 = current_score

    return (gbest, best_c5, n_points)
