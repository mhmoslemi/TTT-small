import numpy as np
import time

def run():
    start_time = time.time()
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    T = n_points / 2.0

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        return np.max(corr)

    best_h = h.copy()
    best_c5 = objective(best_h)

    initial_temp = 10.0
    cooling_rate = 0.995
    sigma = 0.2
    block_swap_prob = 0.2
    current_temp = initial_temp

    while time.time() - start_time < 1100:
        candidate_h = best_h.copy()

        # Perturbation
        if np.random.rand() < block_swap_prob:
            min_idx = np.random.randint(0, n_points - 1)
            max_idx = np.random.randint(min_idx + 1, n_points)
            candidate_h[min_idx:max_idx + 1] = candidate_h[min_idx:max_idx + 1][::-1]
        else:
            noise = np.random.normal(0, sigma, n_points)
            candidate_h += noise

        # Enforce constraints
        candidate_h = np.clip(candidate_h, 0, 1)
        sum_candidate = np.sum(candidate_h)
        if abs(sum_candidate - T) > 1e-10:
            candidate_h = candidate_h * (T / sum_candidate)
            candidate_h = np.clip(candidate_h, 0, 1)

        candidate_c5 = objective(candidate_h)
        delta = candidate_c5 - best_c5

        # Acceptance criterion
        if delta < 0 or np.random.rand() < np.exp(-delta / current_temp):
            best_h[:] = candidate_h[:]
            best_c5 = candidate_c5

        current_temp *= cooling_rate

    return (best_h, best_c5, n_points)
