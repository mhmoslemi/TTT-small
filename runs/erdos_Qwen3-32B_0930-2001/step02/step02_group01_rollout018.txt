        current_sum = x.sum()
        if current_sum == 0:
            # Fallback to uniform if all clipped to zero
            x = np.ones_like(x) * target_sum / n_points
        else:
            x = x * (target_sum / current_sum)
        x = np.clip(x, 0, 1)
        return x

    # Initialize population from initial guess + random perturbation
    pop_size = 100
    elite_ratio = 0.2
    n_elite = int(pop_size * elite_ratio)
    initial_mean = h_initial * (target_sum / h_initial.sum())
    initial_std = 0.2 * np.ones(n_points)
    population = np.random.normal(loc=initial_mean, scale=initial_std, size=(pop_size, n_points))
    
    # Apply repair to ensure valid initial population
    for i in range(pop_size):
        population[i] = repair_solution(population[i])

    best_solution = None

    while time.time() - start_time < budget_s - 100:

        fitness = np.array([compute_c5(p) for p in population])
