        k_shift = idx_max - (n_points - 1)

        valid_i = []
        for i in range(n_points):
            j = i + k_shift
            if 0 <= j < n_points:
                valid_i.append(i)
        if not valid_i:
            patience_counter += 1
            if patience_counter >= max_patience:
                break
            continue

        contributions = []
        for i in valid_i:
            j = i + k_shift
            contrib = h_current[i] * (1.0 - h_current[j])
            contributions.append((-contrib, i))
        contributions.sort()
        top_count = max(1, int(0.1 * len(contributions)))
        top_indices = [i for (neg_c, i) in contributions[:top_count]]

        for i in top_indices:
            j = i + k_shift
            delta = min(h_current[i], 0.1)
            h_current[i] -= delta
            if 0 <= j < n_points:
                h_current[j] = min(1.0, h_current[j] + delta)

        h_current = np.clip(h_current, 0.0, 1.0)
        current_sum = np.sum(h_current)
        if abs(current_sum) > 1e-10:
            scaling = T / current_sum
            h_current *= scaling
            h_current = np.clip(h_current, 0.0, 1.0)

        current_c5 = compute_c5(h_current)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h_current[:]
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter >= max_patience:
                break

    return (best_h, best_c5, n_points)
