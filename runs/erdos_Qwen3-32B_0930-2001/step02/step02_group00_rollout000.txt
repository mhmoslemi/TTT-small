import numpy as np
import time

def run():
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    T = n_points / 2.0

    def objective(h):
        return np.max(np.correlate(h, 1 - h, mode='full') * dx)

    # PSO Parameters
    n_particles = 10
    inertia = 0.7
    cognitive = 2.0
    social = 2.0
    max_velocity = 1.0

    # Initialize swarm
    swarm_positions = np.zeros((n_particles, n_points))
    swarm_velocities = np.zeros((n_particles, n_points))
    pbest_positions = np.zeros((n_particles, n_points))
    pbest_values = np.full(n_particles, np.inf)
    gbest = h_initial.copy()
    gbest_value = objective(gbest)

    swarm_positions[0] = h_initial.copy()
    pbest_positions[0] = h_initial.copy()
    pbest_values[0] = objective(h_initial)

    for i in range(1, n_particles):
        # Perturb initial with Gaussian noise
        particle = h_initial + np.random.normal(0, 0.1, n_points)
        # Enforce constraints
        particle = np.clip(particle, 0, 1)
        s = np.sum(particle)
        if s > 1e-8:
            particle = particle * (T / s)
        particle = np.clip(particle, 0, 1)
        # Update swarm
        swarm_positions[i] = particle
        val = objective(particle)
        pbest_positions[i] = particle.copy()
        pbest_values[i] = val
        if val < gbest_value:
            gbest[:] = particle
            gbest_value = val

    # Initialize velocities
    swarm_velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))

    while time.time() - start_time < 1100:
        for i in range(n_particles):
            # Velocity update
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            cognitive_term = cognitive * r1 * (pbest_positions[i] - swarm_positions[i])
            social_term = social * r2 * (gbest - swarm_positions[i])
            swarm_velocities[i] = inertia * swarm_velocities[i] + cognitive_term + social_term
            np.clip(swarm_velocities[i], -max_velocity, max_velocity, out=swarm_velocities[i])

            # Position update
            swarm_positions[i] += swarm_velocities[i]
            particle = swarm_positions[i]

            # Constraint projection
            particle = np.clip(particle, 0, 1)
            s = np.sum(particle)
            if s > 1e-8:
                particle = particle * (T / s)
            particle = np.clip(particle, 0, 1)
            swarm_positions[i] = particle

            # Update personal best
            current_val = objective(particle)
            if current_val < pbest_values[i]:
                pbest_positions[i] = particle.copy()
                pbest_values[i] = current_val
                if current_val < gbest_value:
                    gbest[:] = particle
                    gbest_value = current_val

    return (gbest, gbest_value, n_points)
