import numpy as np

def project(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    s = h_clipped.sum()
    if s == 0:
        return h_clipped.copy()
    a = target_sum / s
    h_scaled = h_clipped * a
    return np.clip(h_scaled, 0, 1)

def compute_gradient(h, k):
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        j = i + k
        if j < n:
            grad[i] += 1 - h[j]
        j = i - k
        if j >= 0:
            grad[i] -= h[j]
    return grad

def compute_overlaps(h, n_points):
    dx = 2.0 / n_points
    return np.correlate(h, 1 - h, mode='full') * dx
