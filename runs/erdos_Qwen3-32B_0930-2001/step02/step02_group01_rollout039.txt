import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_current = initial_h_values.copy()
    n_points = len(h_current)
    T = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h_vec):
        h_clipped = np.clip(h_vec, 0., 1.)
        sum_h = np.sum(h_clipped)
        penalty = 1e6 * (sum_h - T)**2
        conv = np.correlate(h_clipped, 1.0 - h_clipped, mode='full')
        c5 = np.max(conv * dx)
        return c5 + penalty

    def repair_individual(x):
        x_clipped = np.clip(x, 0., 1.)
        sum_clipped = np.sum(x_clipped)
        last = T - sum_clipped
        h_repaired = np.append(x_clipped, last)
        return h_repaired

    # Initialization parameters
    pop_size = 80
    elite_fraction = 0.2
    max_iter = 200
    alpha = 0.9  # Covariance damping factor
    mean_ce = h_current[:-1].copy()
    cov_ce = np.diag((0.1 * np.ones_like(mean_ce))**2)
    
    best_h = None
    best_c5_value = float('inf')

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Generate population
        population = np.random.multivariate_normal(mean_ce, cov_ce, size=pop_size)
        repaired_population = [repair_individual(p) for p in population]
        c5_list = [compute_c5(h) for h in repaired_population]
        
        # Select elites
        elite_idx = np.argsort(c5_list)[:int(elite_fraction * pop_size)]
        elite_vectors = [repaired_population[i] for i in elite_idx]
        elite_first_n_minus_1 = np.array([h[:-1] for h in elite_vectors])
        
        # Update mean and covariance
        mean_ce = np.mean(elite_first_n_minus_1, axis=0)
        if elite_first_n_minus_1.shape[0] > 1:
            cov_update = np.cov(elite_first_n_minus_1, rowvar=False, bias=True)
            cov_ce = alpha * cov_ce + (1 - alpha) * cov_update
        
        # Check for improvement
        for h in elite_vectors:
            c5 = compute_c5(h)
            if c5 < best_c5_value:
                best_c5_value = c5
                best_h = h.copy()

    # Final SLSQP refinement
    if best_h is not None:
        bounds = [(0, 1)] * n_points
        constraints_slsqp = {'type': 'eq', 'fun': lambda x: np.sum(x) - T}
        result_slsqp = optimize.minimize(
            compute_c5, best_h, method='SLSQP',
            bounds=bounds, constraints=constraints_slsqp,
            options={'maxiter': 150, 'ftol': 1e-8}
        )
        if result_slsqp.success:
            h_slsqp = result_slsqp.x
            c5_slsqp = compute_c5(h_slsqp)
            if c5_slsqp < best_c5_value:
                best_c5_value = c5_slsqp
                best_h = h_slsqp

    elapsed = time.time() - start_time
    print(f"CE Done in {elapsed:.1f}s, Best C5: {best_c5_value:.6f}")
    
    return best_h, best_c5_value, n_points
