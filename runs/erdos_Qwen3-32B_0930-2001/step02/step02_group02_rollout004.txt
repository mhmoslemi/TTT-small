import numpy as np
import time

def run():
    start_time = time.time()
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # PSO Parameters
    swarm_size = 50
    c1, c2 = 2.0, 2.0
    w, max_vel = 0.7, 0.5

    def repair(h_vec):
        h_repaired = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_repaired)
        if current_sum == 0:
            return initial_h_values.copy()
        scaling = target_sum / current_sum
        return np.clip(h_repaired * scaling, 0, 1)

    def evaluate_c5(h_vec):
        return np.max(np.correlate(h_vec, 1 - h_vec, mode="full") * dx)

    # Initialize swarm
    np.random.seed(42)
    swarm = np.random.normal(initial_h_values, 0.1, (swarm_size, n_points))
    for i in range(swarm_size):
        swarm[i] = repair(swarm[i]).copy()
    
    velocities = np.zeros((swarm_size, n_points))
    particle_best = swarm.copy()
    particle_best_scores = np.array([evaluate_c5(p) for p in swarm])
    global_best_idx = np.argmin(particle_best_scores)
    global_best = particle_best[global_best_idx].copy()
    global_best_score = particle_best_scores[global_best_idx]

    # PSO loop
    while time.time() - start_time < 1100:
        for i in range(swarm_size):
            current = swarm[i].copy()
            current = repair(current)
            score = evaluate_c5(current)

            if score < particle_best_scores[i]:
                particle_best[i] = current.copy()
                particle_best_scores[i] = score
                if score < global_best_score:
                    global_best[:] = current[:]
                    global_best_score = score

            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = w * velocities[i] + c1 * r1 * (particle_best[i] - swarm[i]) + c2 * r2 * (global_best - swarm[i])
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            swarm[i] = repair(swarm[i] + velocities[i]).copy()

    # Final validation
    best_h = repair(global_best)
    c5 = evaluate_c5(best_h)
    assert np.allclose(np.sum(best_h), target_sum, atol=1e-6)
    return (best_h, c5, n_points)
