import numpy as np
import time

def run():
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    T_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):

        return np.max(correlations)

    def project_h(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            h_clipped[:] = 0.5
        else:
            scale = T_sum / sum_h
            h_scaled = h_clipped * scale

        return h_clipped

    current_h = project_h(h_initial.copy())
    current_c5 = compute_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5

    start_time = time.time()
    budget = 1100

    # Simulated Annealing parameters
    temp = 10.0
    temp_rate = 0.995
    min_temp = 1e-4
    sigma = 0.1

    while time.time() - start_time < budget and temp > min_temp:
        # Perturb and project
        noise = np.random.normal(0, sigma, n_points)
        candidate_h = current_h + noise
        candidate_h = project_h(candidate_h)

        candidate_c5 = compute_c5(candidate_h)
        delta = candidate_c5 - current_c5

        if delta < 0 or np.exp(-delta / temp) > np.random.rand():
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            if candidate_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = candidate_c5

        temp *= temp_rate

    return (best_h, best_c5, n_points)
