import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()

    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    minimizer_kwargs = {
        'method': 'SLSQP',
        'bounds': bounds,
        'constraints': constraints,
        'options': {'maxiter': 50, 'ftol': 1e-10}
    }

    def basinhopping_callback(x, f, context):
        if time.time() - start_time > budget_s - 5:
            raise StopIteration()

    result = optimize.basinhopping(
        fun=compute_c5,
        x0=h.copy(),
        niter=300,
        T=1.0,
        stepsize=0.5,
        minimizer_kwargs=minimizer_kwargs,
        callback=basinhopping_callback,
        interval=50,
        disp=False
    )

    h_best = result.x.copy()
    h_best = np.clip(h_best, 0.0, 1.0)
    current_sum = np.sum(h_best)
    if not np.isclose(current_sum, target_sum):
        adjustment = (target_sum - current_sum) / n_points
        h_best += adjustment
        h_best = np.clip(h_best, 0.0, 1.0)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, n_points
