import time

    target_sum = n_points / 2.0

    c1, c2 = 2.0, 2.0
    w = 0.7
    max_vel = 0.5

    # Initialize swarm
    swarm = np.random.rand(n_particles, n_points) * 0.2 + (h - 0.1)
    swarm = np.clip(swarm, 0, 1)
    velocities = np.random.uniform(-max_vel, max_vel, size=(n_particles, n_points))
    
    # Normalize initial positions
    for p in range(n_particles):
        s = np.sum(swarm[p])
        if s != 0 and s != n_points:
            swarm[p] = np.clip(swarm[p] * (target_sum / s), 0, 1)

    personal_best_positions = swarm.copy()
    personal_best_scores = np.array([np.max(np.correlate(p, 1 - p, mode='full') * dx) for p in swarm])
    global_best_index = np.argmin(personal_best_scores)
    global_best_position = personal_best_positions[global_best_index].copy()
    global_best_score = personal_best_scores[global_best_index]

    while time.time() - start_time < 1100:
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)

                            c1 * r1 * (personal_best_positions[i] - swarm[i]) +
                            c2 * r2 * (global_best_position - swarm[i]))
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)

            # Update position
            swarm[i] += velocities[i]
            swarm[i] = np.clip(swarm[i], 0, 1)

            # Normalize to constraint
            s = np.sum(swarm[i])
            if s != 0 and s != n_points:
                swarm[i] = np.clip(swarm[i] * (target_sum / s), 0, 1)

            # Evaluate fitness
            score = np.max(np.correlate(swarm[i], 1 - swarm[i], mode='full') * dx)

            # Update personal best
            if score < personal_best_scores[i]:

                personal_best_positions[i] = swarm[i].copy()

                # Update global best

                    global_best_position[:] = swarm[i].copy()
