import numpy as np
import time

def compute_overlaps(h, dx):
    return np.correlate(h, 1 - h, mode="full") * dx

def compute_smooth_max(overlaps, smooth_epsilon):
    max_val = np.max(overlaps)
    shifted = overlaps - max_val
    exp_vals = np.exp(shifted / smooth_epsilon)
    return max_val + smooth_epsilon * np.log(np.sum(exp_vals))

def compute_objective(h, dx, smooth_epsilon):
    overlaps = compute_overlaps(h, dx)
    return compute_smooth_max(overlaps, smooth_epsilon)

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum == 0:
        h.fill(0.5)
    else:
        h *= (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_available = 'initial_h_values' in globals()
    if initial_available:
        n_points = len(initial_h_values)
    else:
        n_points = 47  # standard parent construction
    dx = 2.0 / n_points
    target_sum = n_points / 2

    n_particles = 50
    particles = []
    velocities = []
    p_best_positions = []
    p_best_values = []
    if initial_available:
        h_init = initial_h_values.copy()
        h_init_sum = h_init.sum()
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (target_sum / h_init.sum()) if h_init_sum != 0 else np.full(n_points, 0.5)
        particles.append(h_init)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        p_best_positions.append(h_init.copy())
        p_best_values.append(compute_objective(h_init, dx, 0.1))
    else:
        for _ in range(n_particles):
            h = np.random.uniform(0, 1, n_points)
            h = project_to_feasible(h, n_points)
            particles.append(h)
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
            p_best_positions.append(h.copy())
            p_best_values.append(compute_objective(h, dx, 0.1))
    particles = np.array(particles)
    velocities = np.array(velocities)
    p_best_positions = np.array(p_best_positions)
    p_best_values = np.array(p_best_values)
    g_best_idx = np.argmin(p_best_values)
    g_best_position = p_best_positions[g_best_idx].copy()
    g_best_value = p_best_values[g_best_idx]

    smooth_epsilon = 0.1
    w = 0.7
    c1, c2 = 2.0, 2.0
    max_iter = 1000000
    decay_rate = 0.9
    decay_steps = 50
    start_time = time.time()

    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            # Update position
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (p_best_positions[i] - particles[i]) +
                             c2 * r2 * (g_best_position - particles[i]))
            particles[i] += velocities[i]
            particles[i] = project_to_feasible(particles[i], n_points)
            # Compute objective
            current_value = compute_objective(particles[i], dx, smooth_epsilon)
            # Update personal best
            if current_value < p_best_values[i]:
                p_best_positions[i] = particles[i].copy()
                p_best_values[i] = current_value
                # Update global best
                if current_value < g_best_value:
                    g_best_position = particles[i].copy()
                    g_best_value = current_value
        # Decay smoothing
        if iter_count % decay_steps == 0:
            smooth_epsilon = max(1e-5, smooth_epsilon * decay_rate)

    overlaps_best = compute_overlaps(g_best_position, dx)
    final_c5 = np.max(overlaps_best)
    return (g_best_position, final_c5, n_points)
