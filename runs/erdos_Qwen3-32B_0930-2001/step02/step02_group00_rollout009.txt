import numpy as np
import cvxpy as cp
import time

def run():
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)

    T = n_points / 2.0
    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    iteration = 0

    while time.time() - start_time < 1100:
        iteration += 1
        O = np.correlate(best_h, 1 - best_h, mode='full') * dx
        m_max = np.argmax(O)
        k = m_max - (n_points - 1)

        # Compute coefficients
        c = np.zeros(n_points)
        for i in range(n_points):
            shifted = i + k
            if 0 <= shifted < n_points:
                c[i] = 1.0 - best_h[shifted]
            else:
                c[i] = 1.0

        # Solve LP
        h = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(c * h))
        constraints = [
            cp.sum(h) == T,
            0 <= h,
            h <= 1
        ]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.GLP, verbose=False)
        except:
            continue

        if prob.status in ['optimal', 'optimal_inaccurate']:
            new_h = np.clip(h.value, 0, 1)
            current_c5 = evaluate_erdos_solution(new_h)
            if current_c5 < best_c5:
                best_h = new_h.copy()

        # Early stopping if no improvement for 20 iterations
