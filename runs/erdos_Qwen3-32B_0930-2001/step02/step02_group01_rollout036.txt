import numpy as np
import scipy.optimize as optimize
import time
import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_current = initial_h_values.copy()
    n_points = len(h_current)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    def finalize_h(x):
        h = np.array(x).copy()
        last_val = target_sum - h.sum()
        h = np.append(h, last_val)
        h = np.clip(h, 0, 1)
        scaled_sum = h.sum()
        if scaled_sum == 0:
            return np.clip(h, 0, 1) * target_sum / n_points
        scale = target_sum / scaled_sum
        h = h * scale
        return np.clip(h, 0, 1)

    best_h = h_current.copy()
    best_c5 = compute_c5(best_h)
    cons_slsqp = {'type': 'eq', 'fun': lambda x: x.sum() - target_sum}

    while time.time() - start_time < budget_s - 60:
        if (time.time() - start_time) > budget_s - 30:
            break

        # Create perturbed starting point
        x0 = h_current[:n_points-1].copy()
        noise = np.random.randn(n_points - 1) * 0.1
        x0 = np.clip(x0 + noise, -100, 100)  # Prevent numerical issues
        x0 = np.clip(x0, 0, 1)

        # Nelder-Mead first pass
        res_nm = optimize.minimize(
            lambda x: compute_c5(finalize_h(x)),
            x0,
            method='Nelder-Mead',
            bounds=[(0,1)]*(n_points-1),
            options={'maxiter': 100, 'xatol': 1e-5}
        )

        if res_nm.success:
            x_nm = res_nm.x
            h_nm = finalize_h(x_nm)
            c5_nm = compute_c5(h_nm)
            if c5_nm < best_c5:
                best_c5, best_h = c5_nm, h_nm.copy()

            # SLSQP refinement
            res_slsqp = optimize.minimize(
                compute_c5,
                h_nm,
                method='SLSQP',
                bounds=[(0,1)]*n_points,
                constraints=cons_slsqp,
                options={'maxiter': 300, 'ftol': 1e-8}
            )
            if res_slsqp.success:
                c5_sq = compute_c5(res_slsqp.x)
                if c5_sq < best_c5:
                    best_c5, best_h = c5_sq, res_slsqp.x.copy()

        # Add random perturbation for diversity
        h_current = best_h.copy()

    # Final refinement if time remains
    remaining = budget_s - (time.time() - start_time)
    if remaining > 50:
        pert = np.random.randn(n_points) * 0.05
        h_pert = np.clip(best_h + pert, 0, 1.0)
        res_final = optimize.minimize(
            compute_c5,
            h_pert,
            method='SLSQP',
            bounds=[(0,1)]*n_points,
            constraints=cons_slsqp,
            options={'maxiter': 200}
        )
        if res_final.success and compute_c5(res_final.x) < best_c5:
            best_h = res_final.x
            best_c5 = compute_c5(best_h)

    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Best C5: {best_c5:.6f}")
    return best_h, best_c5, n_points
