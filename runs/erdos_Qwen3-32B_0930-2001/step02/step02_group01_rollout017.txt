import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # sum(h) dx = 1 => sum(h) = n_points/2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    def repair_h(h_vec):
        h = np.clip(h_vec, 0, 1)
        current_sum = h.sum()
        if current_sum == 0:
            h = np.clip(h_vec, 0, 1)
            h = h / (current_sum + 1e-10)
        scale_factor = target_sum / current_sum
        h = h * scale_factor
        h = np.clip(h, 0, 1)
        return h

    initial_h_repair = repair_h(initial_h)
    best_h = initial_h_repair
    best_c5 = compute_c5(best_h)

    mu = initial_h_repair.copy()
    sigma = 0.1 * np.ones(n_points)
    population_size = 60
    elite_fraction = 0.2
    damping = 0.8
    max_iterations = 200
    time_check_interval = 10

    # Initial population
    population = np.random.normal(mu, sigma, (population_size, n_points))
    repaired = np.array([repair_h(h) for h in population])
    c5_population = np.array([compute_c5(h) for h in repaired])
    best_idx = np.argmin(c5_population)
    if c5_population[best_idx] < best_c5:
        best_h = repaired[best_idx].copy()
        best_c5 = c5_population[best_idx]

    for iter in range(max_iterations):
        current_pop = np.random.normal(mu, sigma, (population_size, n_points))
        repaired = np.array([repair_h(h) for h in current_pop])
        c5_vals = np.array([compute_c5(h) for h in repaired])
        n_elite = max(1, int(population_size * elite_fraction))
        elite_idx = np.argsort(c5_vals)[:n_elite]
        elites = repaired[elite_idx]
        mu = elites.mean(axis=0)
        sigma = damping * sigma + (1 - damping) * elites.std(axis=0)
        best_pop_idx = np.argmin(c5_vals)
        if c5_vals[best_pop_idx] < best_c5:

            break

    # Final refinement
    constraints_slsqp = optimize.LinearConstraint(np.ones(n_points), target_sum, target_sum)

            best_c5 = refined_c5

    return best_h, best_c5, n_points
