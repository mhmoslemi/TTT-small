import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()

        adjustment = (target_sum - current_sum) / n_points
        return np.clip(x + adjustment, 0.0, 1.0)

    def compute_c5(h_vec):
        h_repaired = repair(h_vec)
        conv = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(conv * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Phase 1: Differential Evolution
    de_maxiter = 50
    de_result = opt.differential_evolution(
        compute_c5,
        bounds,

        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.7,
        strategy='best1bin',
        polish=False,
        disp=False
    )
    de_end_time = time.time()
    best_h = repair(de_result.x)
    best_c5 = compute_c5(best_h)

    remaining_time = max(0, budget_s - (de_end_time - start_time) - 10)
    if remaining_time > 0:

            compute_c5,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,

            candidate_c5 = compute_c5(candidate)

                best_c5 = candidate_c5
