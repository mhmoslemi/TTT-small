import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()

    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    def generate_candidate(base_h):
        noise = np.random.normal(0, 0.15, n_points)
        h = base_h + noise
        h = np.clip(h, 0, 1)
        adjustment = (target_sum - np.sum(h)) / n_points
        h += adjustment
        h = np.clip(h, 0, 1)
        return h

    # Initialize population
    population = [generate_candidate(h) for _ in range(100)]

    while time.time() - start_time < budget_s - 10:
        # Evaluate population
        scores = np.array([compute_c5(p) for p in population])
        sorted_indices = np.argsort(scores)
        top_idx = sorted_indices[:20]
        parents = [population[i] for i in top_idx]

        # Generate offspring
        offspring = []
        for _ in range(50):
            p1, p2 = parents[np.random.choice(20, 2, replace=False)]
            mask = np.random.rand(n_points) < 0.5
            child = np.where(mask, p1, p2)
            # Mutation
            m_mask = np.random.rand(n_points) < 0.1
            child[m_mask] += np.random.normal(0, 0.1, size=m_mask.sum())
            child = np.clip(child, 0, 1)
            # Normalize sum
            adjustment = (target_sum - np.sum(child)) / n_points
            child += adjustment
            child = np.clip(child, 0, 1)
            offspring.append(child)

        # Combine and select top candidates
        combined = population + offspring
        all_scores = np.array([compute_c5(p) for p in combined])
        top_indices = np.argsort(all_scores)[:100]
        population = [combined[i] for i in top_indices]

        # Update best
        current_best = min(all_scores)
        if current_best < best_c5:
            best_c5 = current_best
            best_h[:] = combined[top_indices[0]].copy()

        # Local refinement
        for candidate in population[:5]:
            if time.time() - start_time > budget_s - 10:
                break
            res = optimize.minimize(
                compute_c5,
                x0=candidate,
                method='SLSQP',
                bounds=[(0, 1)] * n_points,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                options={'maxiter': 40, 'ftol': 1e-10}
            )
            if res.success and res.fun < best_c5:
                best_h[:] = res.x
                best_c5 = res.fun

    return best_h, best_c5, n_points
