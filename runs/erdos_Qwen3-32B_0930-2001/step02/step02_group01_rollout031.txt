def project(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = h.sum()
    if current_sum == 0:
        return h.copy()
    scale = target_sum / current_sum
    h = h * scale
    h = np.clip(h, 0, 1)
    return h.copy()

def compute_c5(h, dx):
