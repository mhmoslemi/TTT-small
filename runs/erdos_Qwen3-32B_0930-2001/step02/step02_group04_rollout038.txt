import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()
    
    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)
    
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    
    def adjust_sum(h, target):
        delta = (target - np.sum(h)) / n_points
        h += delta
        np.clip(h, 0, 1, out=h)
        return h
    
    def generate_candidate(base_h):
        candidate = base_h + np.random.normal(0, 0.1, n_points)
        candidate = np.clip(candidate, 0, 1)
        adjust_sum(candidate, target_sum)
        return candidate
    
    pop_size = 100
    mutation_rate = 0.15
    generations = 0
    current_pop = [generate_candidate(h) for _ in range(pop_size)]
    
    while time.time() - start_time < budget_s - 30:
        scores = [compute_c5(p) for p in current_pop]
        best_idx = np.argmin(scores)
        if scores[best_idx] < c5_best:
            h_best[:] = current_pop[best_idx].copy()
            c5_best = scores[best_idx]
        
        parent_indices = np.argsort(scores)[:max(20, int(0.2 * pop_size))]
        parents = [current_pop[i] for i in parent_indices]
        
        new_pop = parents.copy()
        while len(new_pop) < pop_size:
            if np.random.rand() < 0.5 and len(parents) >= 2:
                p1, p2 = np.random.choice(parents, 2, replace=False)
                child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
            else:
                parent = np.random.choice(parents)
                child = parent.copy()
                mutate_idx = np.random.rand(n_points) < mutation_rate
                child[mutate_idx] += np.random.normal(0, 0.15, size=sum(mutate_idx))
                child = np.clip(child, 0, 1)
            adjust_sum(child, target_sum)
            new_pop.append(child)
        
        current_pop = new_pop
        generations += 1
    
    # Local refinement
    def optimize_local(h0):
        result = optimize.minimize(
            compute_c5, h0, method='SLSQP',

        )

        if refined_score < c5_best:
            c5_best = refined_score
            h_best[:] = refined
    
    return h_best, c5_best, n_points
