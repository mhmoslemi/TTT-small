import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_current = initial_h_values.copy()
    n_points = len(h_current)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # Correct sum for [0,1] constraint
    
    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)
    
    def repair(samples):
        samples_clipped = np.clip(samples, 0, 1)
        current_sums = np.sum(samples_clipped, axis=1)
        scale_factors = np.where(current_sums != 0, target_sum / current_sums, 1.0)
        scaled_samples = samples_clipped * scale_factors[:, np.newaxis]
        return np.clip(scaled_samples, 0, 1)
    
    # Initialize CE parameters
    initial_mean = repair(initial_h_values[np.newaxis, :].copy()).flatten()
    mean = initial_mean
    std_dev = 0.1 * np.ones(n_points)
    best_h = mean.copy()
    best_c5 = compute_c5(best_h)
    
    population_size = 100
    elite_fraction = 0.1
    damping_factor = 0.7
    min_std = 1e-3
    
    # CE loop
    while time.time() - start_time < budget_s - 10:
        samples = np.random.normal(loc=mean, scale=std_dev, size=(population_size, n_points))
        repaired_samples = repair(samples)
        
        c5_values = [compute_c5(repaired_samples[i]) for i in range(population_size)]
        
        # Select elites
        elite_indices = np.argsort(c5_values)[:int(population_size * elite_fraction)]
        elites = repaired_samples[elite_indices]
        
        # Update best solution
        for i in elite_indices:
            current_c5 = c5_values[i]
            if current_c5 < best_c5:

        # Update distribution parameters
        new_mean = np.mean(elites, axis=0)

        std_dev = np.maximum(std_dev, min_std)  # Maintain diversity
