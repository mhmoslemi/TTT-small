        current_global_idx = np.argmin(particle_best_scores)
        if particle_best_scores[current_global_idx] < best_score:
            best_score = particle_best_scores[current_global_idx]
            np.copyto(global_best, particle_best[current_global_idx])

    final_h = repair(global_best, target_sum)
    final_c5 = evaluate(final_h, dx)
    return (final_h, final_c5, n_points)
