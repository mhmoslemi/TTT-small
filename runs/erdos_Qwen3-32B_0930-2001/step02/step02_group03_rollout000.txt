    param_step = learning_rate * m_hat / (sqrt(v_hat) + epsilon)

                h_coarse[i] = h_high[0]
            elif j >= len(h_high) - 1:
                h_coarse[i] = h_high[-1]
            else:
                x0, x1 = x_high[j], x_high[j+1]
                w = (x - x0) / (x1 - x0)
                h_coarse[i] = (1 - w) * h_high[j] + w * h_high[j+1]
        return h_coarse
    
    def upsample(h_coarse, n_fine):
        x_coarse = np.linspace(0, 2, len(h_coarse))
        x_fine = np.linspace(0, 2, n_fine)
        h_fine = np.zeros(n_fine)
        for i in range(n_fine):

            j = np.searchsorted(x_coarse, x) - 1
            if j < 0:
                h_fine[i] = h_coarse[0]
            elif j >= len(h_coarse) - 1:
                h_fine[i] = h_coarse[-1]
            else:
                x0, x1 = x_coarse[j], x_coarse[j+1]
                w = (x - x0) / (x1 - x0)
                h_fine[i] = (1 - w) * h_coarse[j] + w * h_coarse[j+1]
        return h_fine
    
    # Step 1: Coarse grid optimization
    h_coarse = downsample(h, n_coarse)
    current_time = time.time()
    remaining_time = budget_s - (current_time - start_time)
    coarse_max_iterations = int(remaining_time * 0.18)  # Allocate 180s for coarse phase
    if coarse_max_iterations > 0:
        m_coarse = np.zeros(n_coarse)
        v_coarse = np.zeros(n_coarse)
        t_coarse = 0
        
        for iteration in range(coarse_max_iterations):
            current_time = time.time()
            if current_time - start_time > budget_s:
                break
            t_coarse += 1
            overlaps_coarse = compute_overlaps(h_coarse, 2.0 / n_coarse)
            grad_coarse = compute_analytical_gradient(h_coarse, overlaps_coarse, 2.0 / n_coarse, smooth_epsilon)
            
            # Constraint handling: project to feasible set after computing gradients
            # But for this phase, we clip to [0,1] first, then normalize
            h_coarse_clipped = np.clip(h_coarse, 0, 1)
            h_sum = h_coarse_clipped.sum()
            if h_sum == 0:
                h_coarse = np.full(n_coarse, 0.5)
            else:
                h_coarse = h_coarse_clipped * (n_coarse / 2.0 / h_sum)
            
            # Adam update step
            m_coarse = beta1 * m_coarse + (1 - beta1) * grad_coarse
            v_coarse = beta2 * v_coarse + (1 - beta2) * (grad_coarse ** 2)
            m_hat_coarse = m_coarse / (1 - beta1 ** t_coarse)
            v_hat_coarse = v_coarse / (1 - beta2 ** t_coarse)
            step_coarse = learning_rate * m_hat_coarse / (np.sqrt(v_hat_coarse) + eps_adam)
            h_coarse -= step_coarse
            
            # Project constraints again
            h_coarse = np.clip(h_coarse, 0, 1)
            h_sum = h_coarse.sum()
            if h_sum == 0:
                h_coarse = np.full(n_coarse, 0.5)
            else:
                h_coarse = h_coarse * (n_coarse / 2.0 / h_sum)
            
            # Decay smoothing parameter
            if iteration % 50 == 0 and iteration > 0:
                smooth_epsilon = max(1e-5, smooth_epsilon * 0.95)
    
    # Step 2: Upsample to fine grid
    h_fine = upsample(h_coarse, n_points)
    # Also project to constraints
    h_fine = np.clip(h_fine, 0, 1)
    h_sum = h_fine.sum()
    if h_sum == 0:
        h_fine = np.full(n_points, 0.5)
    else:
        h_fine = h_fine * (n_points / 2.0 / h_sum)

    v_fine = np.zeros(n_points)

        current_time = time.time()
        if current_time - start_time > budget_s:
            break
        t_fine += 1

        current_c5 = np.max(overlaps_fine)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        grad_fine = compute_analytical_gradient(h, overlaps_fine, dx, smooth_epsilon)
        
        # Adam update step
        m_fine = beta1 * m_fine + (1 - beta1) * grad_fine
        v_fine = beta2 * v_fine + (1 - beta2) * (grad_fine ** 2)
        m_hat_fine = m_fine / (1 - beta1 ** t_fine)
        v_hat_fine = v_fine / (1 - beta2 ** t_fine)
        step_fine = learning_rate * m_hat_fine / (np.sqrt(v_hat_fine) + eps_adam)
        h -= step_fine

        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum == 0:
            h = np.full(n_points, 0.5)
        else:
            h = h * (n_points / 2.0 / h_sum)
        
        # Decay learning rate and smoothing parameter

            learning_rate = max(1e-4, learning_rate * 0.99)
            smooth_epsilon = max(1e-5, smooth_epsilon * 0.95)
    
    return (best_h, best_c5, n_points)

1. Computes the analytical gradient of the objective function with respect to h using the derived formula.

3. Starts with a coarse grid optimization phase to explore the solution space.
4. Transfers the solution to the fine grid and continues optimization with a smaller learning rate.
5. Enforces constraints by clipping and normalizing h after each step.
6. Ensulates time budget by stopping when the remaining time is exhausted.

The code uses interpolation to transfer solutions between coarse and fine grids, ensuring a smooth transition of the solution. By starting with a coarser grid, the algorithm can explore a larger portion of the solution space, avoiding local optima. import numpy as np
import time

def compute_overlaps(h, dx):
    return np.correlate(h, 1 - h, mode="full") * dx

def compute_analytical_gradient(h, overlaps, dx, smooth_epsilon):
    n = len(h)
    max_val = np.max(overlaps)
    shifted = overlaps - max_val
    exp_vals = np.exp(shifted / smooth_epsilon)
    weights = exp_vals / np.sum(exp_vals)
    
    grads = np.zeros(n)
    for i in range(len(overlaps)):
        weight = weights[i]
        k = i - (n - 1)
        for m in range(n):
            term1 = 0.0
            if 0 <= m + k < n:
                term1 = 1 - h[m + k]
            term2 = 0.0
            if 0 <= m - k < n:
                term2 = h[m - k]
            grads[m] += weight * dx * (term1 - term2)
    return grads

def downsample(h_high, n_coarse):
    x_high = np.linspace(0, 2, len(h_high))
    x_coarse = np.linspace(0, 2, n_coarse)
    h_coarse = np.zeros(n_coarse)
    for i in range(n_coarse):
        x = x_coarse[i]
        j = np.searchsorted(x_high, x) - 1
        if j < 0:
            h_coarse[i] = h_high[0]
        elif j >= len(h_high) - 1:
            h_coarse[i] = h_high[-1]
        else:

            w = (x - x0) / (x1 - x0)
            h_coarse[i] = (1 - w) * h_high[j] + w * h_high[j+1]
    return h_coarse
