import numpy as np
from scipy.stats import multivariate_normal
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    def evaluate(h_vec):
        return np.max(np.correlate(h_vec, 1 - h_vec, mode='full')) * dx
    
    def repair(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        s = h_vec.sum()
        if s == 0:
            return h_vec
        h_vec *= target_sum / s
        h_vec = np.clip(h_vec, 0, 1)
        return h_vec
    
    mean = h.astype(float)
    cov = 0.1 * np.eye(n_points)
    elite_prop = 0.2
    num_samples = 100
    best_h = mean.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        samples = []
        repaired = []
        while len(samples) < num_samples:
            try:
                s = multivariate_normal.rvs(mean, cov)
                samples.append(s)
                repaired.append(repair(s))
            except np.linalg.LinAlgError:
                cov += 1e-6 * np.eye(n_points)
        candid = np.array(repaired)
        scores = [evaluate(c) for c in candid]
        idx = np.argsort(scores)
        elite = candid[idx[:max(1, int(num_samples*elite_prop))]]
        if len(elite) < 2:
            new_mean = elite.mean(axis=0)
            new_cov = np.diag((elite.std(axis=0)*0.1)**2) + 1e-6 * np.eye(n_points)
        else:
            new_mean = elite.mean(axis=0)
            new_cov = np.cov(elite, rowvar=False) + 1e-6 * np.eye(n_points)
        mean = new_mean
        cov = new_cov
        if scores[idx[0]] < best_c5:
            best_h = elite[0].copy()
            best_c5 = scores[idx[0]]
    
    best_h = repair(best_h)
    return (best_h, best_c5, n_points)
