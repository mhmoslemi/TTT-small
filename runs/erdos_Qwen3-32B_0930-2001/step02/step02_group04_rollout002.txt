    while time.time() - start_time < budget_s - 10:
        base_choice = np.random.choice(["original", "best", "binary"], p=[0.5, 0.3, 0.2])
        if base_choice == "original":
            base = initial_h_values.copy()
        elif base_choice == "best":
            base = h_best.copy()
        else:
            base = np.zeros(n_points)
            num_ones = int(target_sum)
            idx = np.random.choice(n_points, size=num_ones, replace=False)
            base[idx] = 1.0

        noise_scale = np.random.uniform(0.1, 0.3)
        h0 = generate_initial_guess(base, noise_scale, target_sum, n_points)
        current_sum_h0 = np.sum(h0)
        adjustment = (target_sum - current_sum_h0) / n_points
        h0 += adjustment
        h0 = np.clip(h0, 0.0, 1.0)

        result = optimize.minimize(
            compute_c5,
            x0=h0,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 30, 'ftol': 1e-8}
        )

        if result.success:
            c5_candidate = result.fun
            h_candidate = result.x
            if c5_candidate < c5_best:
                h_best[:] = h_candidate.copy()
                c5_best = c5_candidate

    return h_best, c5_best, n_points
