import numpy as np
import time
import scipy.optimize as optimize
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()
    
    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)
    
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    
    def generate_initial_guess(guess):
        noise = np.random.normal(0, 0.15, size=n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        current_sum = np.sum(perturbed)
        adjustment = (target_sum - current_sum) / n_points
        perturbed += adjustment
        perturbed = np.clip(perturbed, 0.0, 1.0)
        return perturbed
    
    # Linear programming relaxation
    try:
        h_lp = cp.Variable(n_points)
        t = cp.Variable()
        constraints = [0 <= h_lp, h_lp <= 1]
        constraints += [cp.sum(h_lp) == target_sum]
        
        for k in range(-n_points + 1, n_points):
            # Create a shifted version of h_lp for offset k
            shifted_h = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = (i + k) % n_points
                shifted_h[i, j] = 1
            offset = np.zeros((n_points, 1))
            offset[0, 0] = k * dx
            overlaps = h_lp * shifted_h @ h_lp + offset
            constraints += [cp.sum(overlaps - (n_points - 1) * (1 - t)) <= 0]
        
        objective = cp.Maximize(t)
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.CVXOPT, verbose=False, kktsolver="robust")
        
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_candidate = h_lp.value
            if h_candidate is not None:
                h_candidate = np.clip(h_candidate, 0.0, 1.0)
                current_sum = np.sum(h_candidate)
                if not np.isclose(current_sum, target_sum, atol=1e-6):
                    adjustment = (target_sum - current_sum) / n_points
                    h_candidate += adjustment
                    h_candidate = np.clip(h_candidate, 0.0, 1.0)
                c5_candidate = compute_c5(h_candidate)
                if c5_candidate < c5_best:
                    h_best = h_candidate.copy()
                    c5_best = c5_candidate
        else:
            print("Warning: LP relaxation failed to solve")
    except Exception as e:
        print(f"Warning: LP relaxation failed with error: {str(e)}")
    
    # Local optimization refinement
    h0 = generate_initial_guess(h_best)
    result = optimize.minimize(
        compute_c5,
        x0=h0.flatten(),
        method='SLSQP',
        bounds=[(0, 1)] * n_points,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
        options={'maxiter': 100, 'ftol': 1e-10}
    )
    
    if result.success:
        h_final = result.x.reshape(n_points)
        c5_final = compute_c5(h_final)
        if c5_final < c5_best:
            return h_final, c5_final, n_points
    
    return h_best, c5_best, n_points
