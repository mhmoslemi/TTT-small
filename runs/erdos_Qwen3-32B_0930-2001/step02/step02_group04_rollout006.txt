import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()
    
    def compute_c5(h_vec):
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(conv * dx)
    
    def generate_candidate(base):
        noise = np.random.normal(0, 0.1, size=n_points)
        candidate = np.clip(base + noise, 0, 1)
        adjustment = (target_sum - candidate.sum()) / n_points
        return np.clip(candidate + adjustment, 0, 1)

    population_size = 100
    population = [generate_candidate(h) for _ in range(population_size)]
    best_h = min(population, key=compute_c5)
    best_c5 = compute_c5(best_h)
    
    while time.time() - start_time < budget_s - 10:
        # Evaluate and sort
        scores = [compute_c5(ind) for ind in population]
        sorted_indices = np.argsort(scores)
        top_indices = sorted_indices[:50]
        top_population = [population[i] for i in top_indices]
        
        # Generate offspring
        offspring = []
        for _ in range(50):
            p1, p2 = np.random.choice(len(top_population), 2)
            parent1 = top_population[p1]
            parent2 = top_population[p2]
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            child = np.clip(child, 0, 1)
            adj = (target_sum - child.sum()) / n_points
            child = np.clip(child + adj, 0, 1)
            # Mutation
            mut_indices = np.random.choice(n_points, size=int(0.1*n_points), replace=False)
            child[mut_indices] += np.random.uniform(-0.1, 0.1, size=len(mut_indices))
            child = np.clip(child, 0, 1)
            adj = (target_sum - child.sum()) / n_points
            child = np.clip(child + adj, 0, 1)
            offspring.append(child)
        
        # Expand population with offspring
        population = top_population + offspring
        # Keep only top 100
        scores = [compute_c5(ind) for ind in population]
        idxs = np.argsort(scores)
        population = [population[i] for i in idxs[:population_size]]
        
        # Local optimization
        for candidate in population[:5]:
            if time.time() - start_time > budget_s - 5:
                break
            result = optimize.minimize(
                compute_c5,
                x0=candidate,
                method='SLSQP',
                bounds=[(0,1)]*n_points,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                options={'maxiter': 50}
            )
            if result.success:
                new_candidate = np.clip(result.x, 0, 1)
                adj = (target_sum - np.sum(new_candidate)) / n_points
                new_candidate = np.clip(new_candidate + adj, 0, 1)
                population.append(new_candidate)
        
        # Update global best
        current_best = min(population, key=compute_c5)
        current_c5 = compute_c5(current_best)
        if current_c5 < best_c5:
            best_h, best_c5 = current_best, current_c5
    
    return best_h, best_c5, n_points
