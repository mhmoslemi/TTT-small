import numpy as np
import time

def run():
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    swarm_size = 50
    c1 = 2.0
    c2 = 2.0
    w_init = 0.9
    w_min = 0.4
    max_vel = 0.5

    np.random.seed(42)
    swarm = np.random.randn(swarm_size, n_points) * 0.1 + h_initial
    velocities = np.zeros((swarm_size, n_points))

    def repair(h):
        h = np.clip(h, 0, 1)
        s = h.sum()
        if s == 0:
            return np.full_like(h, 0.5)
        scale = target_sum / s
        h = np.clip(h * scale, 0, 1)
        return h

    for i in range(swarm_size):
        swarm[i] = repair(swarm[i])

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    scores = np.array([evaluate(p) for p in swarm])
    pb = np.copy(swarm)
    pbest_scores = scores.copy()
    gbest_idx = pbest_scores.argmin()
    gbest = pb[gbest_idx].copy()
    best_score = pbest_scores[gbest_idx]

    while time.time() - start_time < 1100:
        w = max(w_min, w_init - 0.3 * (time.time() - start_time) / 1100)
        for i in range(swarm_size):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (w * velocities[i] +
                            c1 * r1 * (pb[i] - swarm[i]) +
                            c2 * r2 * (gbest - swarm[i]))
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            swarm[i] = swarm[i] + velocities[i]
            swarm[i] = repair(swarm[i])
            current_score = evaluate(swarm[i])
            if current_score < pbest_scores[i]:
                pbest_scores[i] = current_score
                pb[i] = swarm[i].copy()

        gbest_idx = pbest_scores.argmin()
        if pbest_scores[gbest_idx] < best_score:
            best_score = pbest_scores[gbest_idx]
            gbest = pb[gbest_idx].copy()

    return (gbest, best_score, n_points)
