import numpy as np
import time

def compute_overlaps(h, dx):
    return np.correlate(h, 1 - h, mode="full") * dx

def compute_smooth_max(overlaps, smooth_epsilon):
    max_val = np.max(overlaps)
    shifted = overlaps - max_val
    exp_vals = np.exp(shifted / smooth_epsilon)
    return max_val + smooth_epsilon * np.log(np.sum(exp_vals))

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    for _ in range(3):  # 3 iterations to stabilize constraints
        total = h.sum()
        if total == 0:
            h.fill(0.5)
        else:
            h = h * (n_points / 2 / total)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    n_particles = 50
    particles = np.zeros((n_particles, n_points))
    velocities = 0.2 * np.random.randn(n_particles, n_points)  # small initial velocities

    # Initialize particles with initial solution and random feasible ones
    particles[0] = h_initial
    for i in range(1, n_particles):
        h = np.random.rand(n_points)
        particles[i] = project_to_feasible(h, n_points)

    p_best = particles.copy()
    p_best_obj = np.full(n_particles, np.inf)
    g_best = particles[0].copy()
    g_best_obj = np.inf
    smooth_eps = 0.1
    decay_rate = 0.95
    decay_steps = 50
    c1, c2 = 2.0, 2.0
    w = 0.7

    start_time = time.time()
    iteration = 0

    while time.time() - start_time < budget_s:
        iteration += 1

        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            overlaps = compute_overlaps(h, dx)
            obj = compute_smooth_max(overlaps, smooth_eps)

                p_best[i] = h.copy()
                p_best_obj[i] = obj
