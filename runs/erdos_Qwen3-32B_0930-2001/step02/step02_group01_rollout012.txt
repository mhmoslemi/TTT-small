                new_c5 = compute_c5(new_h)
                if new_c5 < best_c5:
                    h_best = new_h
                    best_c5 = new_c5
                    h_current = h_best.copy()

        # Optimize upper half
        if time.time() - start_time < budget_s - 200:
            def objective_upper(x):
                h_new = np.clip(x, 0, 1)
                h_full = np.concatenate([h_current[:mid], h_new])
                h_full = np.clip(h_full, 0, 1)
                return compute_c5(h_full)
            
            upper_target = target_sum - h_current[:mid].sum()
            cons = ({'type': 'eq', 'fun': lambda x: x.sum() - upper_target}) if upper_target >= 0 else None
            if cons:
                bounds_upper = [(0, 1)] * (n_points - mid)
                result = optimize.minimize(objective_upper, x0=h_current[mid:], method='SLSQP', 
                                           bounds=bounds_upper, constraints=cons, options={'maxiter': 100})
                if result.success:
                    h_upper = np.clip(result.x, 0, 1)
                    new_h = np.concatenate([h_current[:mid], h_upper])
                    new_h = np.clip(new_h, 0, 1)
                    new_c5 = compute_c5(new_h)
                    if new_c5 < best_c5:
                        h_best = new_h
                        best_c5 = new_c5
                        h_current = h_best.copy()

    for _ in range(15):  # perturbation-based multi-start
        if time.time() - start_time > budget_s - 50:
            break
        pert = np.random.randn(n_points) * 0.05
        h_pert = h_best + pert
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (target_sum / h_pert.sum())
        h_pert = np.clip(h_pert, 0, 1)
        result = optimize.minimize(compute_c5, h_pert, method='SLSQP', bounds=[(0,1)]*n_points,
                                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}],
                                   options={'maxiter': 50})
        if result.success:
            new_c5 = compute_c5(result.x)
            if new_c5 < best_c5:
                h_best = result.x
                best_c5 = new_c5

    if time.time() - start_time < budget_s - 30:
        refine_pert = np.random.randn(n_points) * 0.02
        h_ref = h_best + refine_pert
        h_ref = np.clip(h_ref, 0, 1)
        h_ref = h_ref * (target_sum / h_ref.sum())
        h_ref = np.clip(h_ref, 0, 1)
        result = optimize.minimize(compute_c5, h_ref, method='SLSQP', bounds=[(0,1)]*n_points,
                                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}],
                                   options={'maxiter': 30})
        if result.success:
            new_c5 = compute_c5(result.x)
            if new_c5 < best_c5:
                h_best = result.x
                best_c5 = new_c5

    return h_best, best_c5, n_points
