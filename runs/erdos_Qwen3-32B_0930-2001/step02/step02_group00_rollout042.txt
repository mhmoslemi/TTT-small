        for i in valid_indices:
            j = i + k
            c[i] = (1.0 - h_prev[j]) * dx
        
        # Define and solve convex subproblem
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(c @ h_var)
        constraints = [cp.sum(h_var) == T, h_var >= 0, h_var <= 1]
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h = h_var.value
                # Update best solution
                current_c5 = evaluate_erdos_solution(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        except Exception:
            pass  # Skip on solver exception
        
    return (best_h, best_c5, n_points)
