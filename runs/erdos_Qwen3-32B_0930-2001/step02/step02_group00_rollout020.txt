import numpy as np
import cvxpy as cp
import time

def run():
    start_time = time.time()
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    T = n_points / 2.0

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)

    while time.time() - start_time < 1100:
        # Compute worst-case overlap
        corrs_full = np.correlate(h, 1 - h, mode='full') * dx
        idx_max = np.argmax(corrs_full)
        worst_overlap = corrs_full[idx_max]
        if worst_overlap < best_c5:
            best_c5 = worst_overlap
            best_h = h.copy()

        # Map corr index to lag
        lag = idx_max - (n_points - 1)

        # Build convexified cost vector
        c = np.zeros(n_points)
        for i in range(n_points):
            j = i - lag
            if 0 <= j < n_points:
                c[i] = (1 - h[j]) * dx

        # Solve convex approximation
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(c @ h_var)
        constraints = [
            cp.sum(h_var) == T,
            h_var >= 0,
            h_var <= 1
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)

        if prob.status == 'optimal':
            h[:] = h_var.value.copy()

    return (best_h, best_c5, n_points)
