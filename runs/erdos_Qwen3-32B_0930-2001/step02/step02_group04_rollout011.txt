                h_rand[indices] = np.clip(h_rand[indices] - delta/len(indices), 0, 1)
        return h_rand
    
    while time.time() - start_time < budget_s - 10:
        if np.random.rand() < 0.5:
            noise_std = np.random.uniform(0.1, 0.3)
            h0 = generate_initial_guess(h_best, noise_std)
        else:
            if np.random.rand() < 0.5:
                noise_std = np.random.uniform(0.1, 0.3)
                h0 = generate_initial_guess(initial_h_values, noise_std)
            else:
                h0 = generate_random_initial()
        
        result = optimize.minimize(
            compute_c5,
            x0=h0,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        
        if result.success:
            c5_candidate = result.fun
            h_candidate = result.x
            if c5_candidate < c5_best:
                h_best[:] = h_candidate.copy()
                c5_best = c5_candidate
    
    return h_best, c5_best, n_points
