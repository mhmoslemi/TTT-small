        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed,
        maxiter=maxiter,
        polish=True,
        disp=False
    )

    if result.success:
        candidate = np.clip(result.x, 0, 1)
        # Final repair to satisfy constraint exactly
        sum_current = np.sum(candidate)
        delta = (target_sum - sum_current) / n_points
        h_repaired = np.clip(candidate + delta, 0.0, 1.0)
        final_sum = np.sum(h_repaired)
        # Safety fallback if sum is off
        if abs(final_sum - target_sum) > 1e-6:
            adjustment = (target_sum - final_sum) / n_points
            h_repaired += adjustment
            h_repaired = np.clip(h_repaired, 0, 1)
        
        conv = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        c5_val = np.max(conv * dx)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_repaired

    return best_h, best_c5, n_points
