            new_offspring.append(child)

        combined_pop = population + new_offspring
        combined_fit = fitness + new_fitness
        combined = list(zip(combined_pop, combined_fit))
        combined.sort(key=lambda x: x[1])
        population = [x[0] for x in combined[:100]]
        fitness = [x[1] for x in combined[:100]]

        if min(fitness) < c5_best:
            h_best = population[0].copy()
            c5_best = min(fitness)

        best_candidates = [x[0] for x in combined[:10]]
        for bc in best_candidates:
            result = optimize.minimize(
                compute_c5,
                x0=bc,
                method='SLSQP',
                bounds=[(0, 1)] * n_points,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                options={'maxiter': 80, 'ftol': 1e-10}
            )
            if result.success:
                c5_candidate = result.fun
                h_candidate = result.x
                if c5_candidate < c5_best:
                    h_best[:] = h_candidate.copy()
                    c5_best = c5_candidate

        if time.time() - start_time > budget_s - 5:
            break

    return h_best, c5_best, n_points
