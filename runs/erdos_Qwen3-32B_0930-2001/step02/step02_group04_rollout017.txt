        top_percent = int(0.2 * len(population_c5))
        parents = [ind for (c5, ind) in population_c5[:top_percent]]

        # Generate 50 offspring
        offspring = []
        while len(offspring) < 50:
            if not parents:
                break
            p1 = parents[np.random.randint(len(parents))]
            p2 = parents[np.random.randint(len(parents))]
            crossover_point = np.random.randint(1, n_points)
            child = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
            
            # Mutation
            n_mutate = max(1, int(0.1 * n_points))
            indices = np.random.choice(n_points, n_mutate, replace=False)
            for i in indices:
                perturbation = np.random.normal(0, 0.1)
                child[i] = np.clip(child[i] + perturbation, 0, 1)
            
            # Normalize sum
            current_sum = np.sum(child)
            if current_sum != 0:
                child = target_sum / current_sum * child
                child = np.clip(child, 0, 1)
            offspring.append(child)

        # Combine population and offspring
        offspring_c5 = [(compute_c5(ind), ind) for ind in offspring]
        combined = population_c5 + offspring_c5
        combined.sort(key=lambda x: x[0])
        population = [ind for (c5, ind) in combined[:100]]
        
        # Update best solution
        if combined and combined[0][0] < c5_best:
            h_best = combined[0][1].copy()
            c5_best = combined[0][0]

    # Final refinement with SLSQP
    top_candidates = [ind for (c5, ind) in combined[:10]]
    for candidate in top_candidates:
        if time.time() - start_time >= budget_s - 5:
            break
        result = optimize.minimize(
            compute_c5,
            x0=candidate,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 80, 'ftol': 1e-10}
        )
        if result.success and result.fun < c5_best:
            h_best[:] = result.x.copy()
            c5_best = result.fun

    return h_best, c5_best, n_points
