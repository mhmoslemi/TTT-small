    c2 = 2.0
    w = 0.7
    max_vel = 0.5

    # Initialize particles
    np.random.seed(42)
    swarm = np.random.randn(swarm_size, n_points) * 0.1 + initial_h_values
    velocities = np.zeros((swarm_size, n_points))

    # Apply repair
    def repair(h_vec):

        s = h_vec.sum()

    # Repair the initial swarm
    for i in range(swarm_size):
        swarm[i] = repair(swarm[i])

    # Initialize personal best and global best
    particle_best = [swarm[i].copy() for i in range(swarm_size)]
    particle_best_scores = [evaluate(swarm[i]) for i in range(swarm_size)]

    while time.time() - start_time < 1100:

import numpy as np
import time

def run(seed=42, budget_s=1100):
    # Start time tracking
    start_time = time.time()
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # PSO parameters
    swarm_size = 50
    c1 = 2.0  # Cognitive coefficient
    c2 = 2.0  # Social coefficient
    w = 0.7   # Inertia weight
    max_vel = 0.25  # Max velocity

    # Initialize swarm and velocities
    np.random.seed(seed)
    swarm = np.random.randn(swarm_size, n_points) * 0.1 + initial_h_values
    velocities = np.zeros((swarm_size, n_points))

    def repair(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        s = h_vec.sum()
        if s == 0:
            return h_vec
        return h_vec * (target_sum / s)

    # Apply repair to initialize particles
    for i in range(swarm_size):
        swarm[i] = repair(swarm[i])

    # Initialize personal best and global best
    # Evaluate all particles to get initial scores
    particle_best_scores = []
    particle_best = []
    for i in range(swarm_size):
        pb = swarm[i].copy()
        particle_best.append(pb)
        score = np.max(np.correlate(pb, 1 - pb, mode='full') * dx)
        particle_best_scores.append(score)
    global_best_index = np.argmin(particle_best_scores)
    global_best = particle_best[global_best_index].copy()
    global_best_score = particle_best_scores[global_best_index]

    # PSO main loop
    while time.time() - start_time < budget_s:
        # Update personal best and global best
        for i in range(swarm_size):
            # Check if current particle is better than personal best
            current_score = np.max(np.correlate(swarm[i], 1 - swarm[i], mode='full') * dx)
            if current_score < particle_best_scores[i]:
                particle_best[i] = swarm[i].copy()
                particle_best_scores[i] = current_score
        # Update global best
        current_global_index = np.argmin(particle_best_scores)
        if particle_best_scores[current_global_index] < global_best_score:
            global_best = particle_best[current_global_index].copy()
            global_best_score = particle_best_scores[current_global_index]
        # Update velocities and positions
        for i in range(swarm_size):

        s = h_vec.sum()
        if s == 0:
            return h_vec
        return h_vec * (target_sum / s)
    
    for i in range(swarm_size):
        swarm[i] = repair(swarm[i])

    global_best = particle_best[np.argmin(particle_best_scores)].copy()
    global_best_score = min(particle_best_scores)
