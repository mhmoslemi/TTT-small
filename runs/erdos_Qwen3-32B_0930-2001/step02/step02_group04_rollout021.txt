import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()
    
    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)
    
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    
    try:
        h_var = cp.Variable(n_points, nonneg=True)
        t = cp.Variable()
        constraints = []
        constraints.append(cp.sum(h_var) == target_sum)
        constraints.append(h_var <= 1)
        
        for k in range(n_points):
            valid_i = [i for i in range(n_points - k)]
            if not valid_i:
                continue
            sum_z = 0
            for i in valid_i:
                z = cp.Variable()
                constraints.append(z >= h_var[i] + h_var[i + k] - 1)
                constraints.append(z >= 0)
                sum_z += z
            constraints.append(sum_z >= t)
        
        objective = cp.Maximize(t)
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.ECOS, verbose=False)
        if problem.status == cp.OPTIMAL:
            h_best = h_var.value.copy()
            c5_best = compute_c5(h_best)
    except:
        pass
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time < 10:
        return h_best, c5_best, n_points
    
    def generate_initial_guess(guess):
        noise = np.random.normal(0, 0.15, size=n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        current_sum = np.sum(perturbed)
        adjustment = (target_sum - current_sum) / n_points
        perturbed += adjustment
        perturbed = np.clip(perturbed, 0.0, 1.0)
        return perturbed
    
    attempts = 0
    while time.time() - start_time < budget_s - 10:
        if attempts == 0:
            h0 = h_best.copy()
        else:
            h0 = generate_initial_guess(h_best)
        result = optimize.minimize(
            compute_c5,
            x0=h0,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if result.success:
            c5_candidate = result.fun
            h_candidate = result.x
            if c5_candidate < c5_best:
                h_best[:] = h_candidate.copy()
                c5_best = c5_candidate
        
        attempts += 1
    
    return h_best, c5_best, n_points
