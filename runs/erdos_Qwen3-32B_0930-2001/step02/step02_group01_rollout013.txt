import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    h_current = initial_h_values.copy()
    n_points = len(h_current)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # 21.0 for n=42

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    def finalize_h(x):
        h = np.array(x).copy()
        last_val = target_sum - h.sum()
        h = np.append(h, last_val)
        h = np.clip(h, 0, 1)
        scaled_sum = h.sum()
        if scaled_sum == 0:
            return np.clip(h, 0, 1) * target_sum / n_points
        scale = target_sum / scaled_sum
        h = h * scale
        return np.clip(h, 0, 1)

    # Initial evaluation
    h_current = finalize_h(h_current)
    best_c5 = compute_c5(h_current)

    # Alternating optimization
    if n_points % 2 == 0:
        lower_indices = np.arange(n_points // 2)
        upper_indices = np.arange(n_points // 2, n_points)
        max_alt = 3
        for _ in range(max_alt):
            # Optimize upper half given lower
            fixed_lower = h_current[lower_indices]
            target_upper_sum = target_sum - fixed_lower.sum()
            x0_upper = h_current[upper_indices]
            res_upper = optimize.minimize(
                lambda x: compute_c5(np.concatenate([fixed_lower, x])), 
                x0_upper, 
                method='SLSQP',
                bounds=[(0,1)] * len(upper_indices),
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_upper_sum}],
                options={'maxiter': 50}
            )
            if res_upper.success:
                h_current[upper_indices] = res_upper.x
        
            # Optimize lower half given upper
            fixed_upper = h_current[upper_indices]
            target_lower_sum = target_sum - fixed_upper.sum()
            x0_lower = h_current[lower_indices]
            res_lower = optimize.minimize(

                x0_lower,

                bounds=[(0,1)] * len(lower_indices),
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_lower_sum}],
                options={'maxiter': 50}
            )
            if res_lower.success:
                h_current[lower_indices] = res_lower.x
    
    # Differential evolution with initial guess
    h_de_best = h_current.copy()
    best_c5 = compute_c5(h_de_best)
