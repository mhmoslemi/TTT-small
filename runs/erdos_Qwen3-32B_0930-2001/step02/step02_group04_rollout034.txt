import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()
    
    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        valid_conv = conv[n_points - 1 : 2 * n_points - 1]
        return np.max(valid_conv * dx)
    
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    
    def generate_initial_guess(guess):
        noise = np.random.uniform(-0.2, 0.2, size=n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        np.clip(perturbed, 0.0, 1.0, out=perturbed)
        current_sum = np.sum(perturbed)
        adjustment = (target_sum - current_sum) / n_points
        perturbed += adjustment
        perturbed = np.clip(perturbed, 0.0, 1.0)
        return perturbed
    
    while time.time() - start_time < budget_s - 1:
        h0 = generate_initial_guess(h_best)
        result = optimize.minimize(
            compute_c5,
            x0=h0,
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        
        if result.success:
            c5_candidate = result.fun
            h_candidate = result.x
            if c5_candidate < c5_best:
                h_best[:] = h_candidate.copy()
                c5_best = c5_candidate
    
    return h_best, c5_best, n_points
