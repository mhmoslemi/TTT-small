import numpy as np
import time

def compute_overlaps(h, dx):
    return np.correlate(h, 1 - h, mode="full") * dx

def compute_smooth_max(overlaps, smooth_epsilon):
    max_val = np.max(overlaps)
    shifted = overlaps - max_val
    exp_vals = np.exp(shifted / smooth_epsilon)
    return max_val + smooth_epsilon * np.log(np.sum(exp_vals))

def project_h(h, n_points):
    for _ in range(3):
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum != 0:
            h = h * (n_points / 2 / h_sum)
        else:
            h[:] = 0.5
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_init = initial_h_values.copy() if 'initial_h_values' in globals() else None
    n_points = len(h_init) if h_init is not None else 47
    dx = 2.0 / n_points

    n_particles = 50
    particles = []
    if h_init is not None:
        particles.append(h_init.copy())
        for _ in range(n_particles - 1):
            h = np.random.rand(n_points)
            project_h(h, n_points)
            particles.append(h)
        particles = np.array(particles)
    else:
        raise ValueError("Initial_h_values must be available")

    velocities = 0.1 * (2 * np.random.rand(n_particles, n_points) - 1)
    p_best = particles.copy()
    p_best_obj = np.array([compute_smooth_max(compute_overlaps(p, dx), 0.1) for p in particles])
    g_best_idx = np.argmin(p_best_obj)
    g_best = particles[g_best_idx].copy()
    g_best_obj = p_best_obj[g_best_idx]

    best_h = g_best.copy()
    best_c5 = float('inf')
    start_time = time.time()
    max_iterations = 1000000
    smooth_epsilon = 0.1
    w, c1, c2 = 0.7, 2.0, 2.0
    decay_steps = 50

    for it in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

        for i in range(n_particles):
            overlaps = compute_overlaps(particles[i], dx)
            obj = compute_smooth_max(overlaps, smooth_epsilon)
            if obj < p_best_obj[i]:

                p_best_obj[i] = obj

        current_g_best_idx = np.argmin(p_best_obj)
        if p_best_obj[current_g_best_idx] < g_best_obj:
            g_best[:] = particles[current_g_best_idx].copy()
            g_best_obj = p_best_obj[current_g_best_idx]

            r1 = np.random.rand(n_points)

            velocities[i] += c1 * r1 * (p_best[i] - particles[i])

    return (best_h, best_c5, n_points)
