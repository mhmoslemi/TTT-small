import numpy as np
import time

def run():
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    T = n_points / 2.0  # Target sum

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5

    temp = 10.0
    cool_rate = 0.995
    min_temp = 1e-6
    noise_sigma = 0.1
    block_swap_prob = 0.2
    start_time = time.time()
    budget = 1100

    while time.time() - start_time < budget and temp > min_temp:
        # Generate perturbed candidate
        idxs = np.arange(n_points)
        if np.random.rand() < block_swap_prob:
            # Block swap: reverse a segment
            a, b = np.sort(np.random.choice(idxs, 2, replace=False))
            new_h = current_h.copy()
            new_h[a:b+1] = new_h[a:b+1][::-1]
        else:
            # Add Gaussian noise
            noise = np.random.normal(0, noise_sigma, n_points)
            new_h = np.clip(current_h + noise, 0, 1)

        # Enforce sum constraint
        s = np.sum(new_h)
        if not np.isclose(s, T):
            scale = T / s
            new_h = np.clip(new_h * scale, 0, 1)

        new_c5 = compute_c5(new_h)
        delta = new_c5 - current_c5

        # Acceptance criterion
        if delta < 0 or np.exp(-delta / temp) > np.random.rand():
            current_h, current_c5 = new_h, new_c5

        # Update best solution
        if current_c5 < best_c5:
            best_h, best_c5 = current_h.copy(), current_c5

        # Cool down
        temp *= cool_rate

    return (best_h, best_c5, n_points)
