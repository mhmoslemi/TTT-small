def run():

    global_best_idx = personal_best_scores.argmin()
    global_best = personal_best[global_best_idx].copy()

def _repair(h_vec, n_points, target_sum):
    h_clipped = np.clip(h_vec, 0, 1)
    current_sum = h_clipped.sum()
    if current_sum == 0:
        return np.full(n_points, 0.5)
    scale_factor = target_sum / current_sum
    scaled = h_clipped * scale_factor
    repaired = np.clip(scaled, 0, 1)
    sum_repaired = repaired.sum()
    diff = target_sum - sum_repaired
    alter_indices = np.where((repaired > 1e-6) & (repaired < 1 - 1e-6))[0]
    if len(alter_indices) > 0:
        delta = diff / len(alter_indices)
        for idx in alter_indices:
            repaired[idx] += delta
        repaired = np.clip(repaired, 0, 1)
    return repaired

def _evaluate(h_vec, n_points, dx):
    corrs = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
    return np.max(corrs)
