import numpy as np
from scipy.stats import multivariate_normal
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def evaluate(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def repair(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = h_vec.sum()
        if current_sum == 0 or current_sum == n_points:
            return h_vec
        scaling = target_sum / current_sum
        h_vec *= scaling
        h_vec = np.clip(h_vec, 0, 1)
        return h_vec

    mean = h.copy()
    cov = 0.1 * np.eye(n_points)
    best_h = h.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    num_samples = 100
    elite_proportion = 0.2
    eps = 1e-6

    while time.time() - start_time < budget_s:
        samples = multivariate_normal.rvs(mean, cov, num_samples)
        candidates = []
        for s in samples:
            repaired = repair(s)
            c5 = evaluate(repaired)
            candidates.append((c5, repaired))

        candidates.sort()
        num_elite = max(1, int(num_samples * elite_proportion))
        elites = candidates[:num_elite]

        elite_arr = np.array([h for _, h in elites])
        new_mean = elite_arr.mean(axis=0)
        new_cov = np.cov(elite_arr, rowvar=False) + eps * np.eye(n_points)
        mean, cov = new_mean, new_cov

        if best_c5 > elites[0][0]:
            best_h[:] = elites[0][1].copy()
            best_c5 = elites[0][0]

    return (best_h, best_c5, n_points)
