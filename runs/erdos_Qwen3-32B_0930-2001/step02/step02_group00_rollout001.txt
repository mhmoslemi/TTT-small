import numpy as np
import time

def run():

    dx = 2.0 / n_points
    T = n_points / 2.0

    swarm_size = 10
    omega = 0.7  # Inertia
    phi_p = 2.0  # Cognitive
    phi_g = 2.0  # Social
    max_vel = 1.0

    # Initialize swarm
    particles = np.zeros((swarm_size, n_points))
    velocities = np.zeros_like(particles)
    particles[0] = h_initial.astype(float)
    for i in range(1, swarm_size):
        perturbed = particles[0].copy() + np.random.normal(0, 0.2, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (T / np.sum(perturbed))  # Enforce constraint
        particles[i] = np.clip(perturbed, 0, 1)

    # Initialize best positions
    p_best = particles.copy()
    p_best_objective = [np.max(np.correlate(p, 1 - p, mode='full') * dx) for p in particles]
    g_best_idx = np.argmin(p_best_objective)
    g_best = particles[g_best_idx].copy()

    while time.time() - start_time < 1100:

            # Update velocity
            r_p, r_g = np.random.rand(), np.random.rand()

                            phi_g * r_g * (g_best - particles[i]))
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)

            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)

            # Enforce constraint

                particles[i] *= (T / s)
                particles[i] = np.clip(particles[i], 0, 1)

        # Evaluate objectives
        current_objective = [np.max(np.correlate(p, 1 - p, mode='full') * dx) for p in particles]
        current_objective = np.array(current_objective)

        # Update personal best
        for i in range(swarm_size):
            if current_objective[i] < p_best_objective[i]:
                p_best[i] = particles[i].copy()
                p_best_objective[i] = current_objective[i]

        # Update global best
