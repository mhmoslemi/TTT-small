        new_correlation = np.correlate(h_current, 1 - h_current, mode='full')
        new_c5 = np.max(new_correlation) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_current.copy()

    return (best_h, best_c5, n_points)
