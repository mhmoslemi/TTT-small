def compute_overlaps(h, dx):

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    best_c5 = np.inf
    start_time = time.time()
    
    # CVXPY variables and constraints
    h_var = cp.Variable(n_points)
    constraints = [
        0 <= h_var,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2
    ]
    
    # Generate overlap expressions
    overlaps = []
    for k in range(1, n_points):  # excluding k=0 to avoid self-overlap
        shifted = cp.roll(h_var, shift=-k % n_points)
        overlap_k = cp.sum(h_var * (1 - shifted)) * dx
        overlaps.append(overlap_k)
    
    # Parameters for regularization
    lambda_param = cp.Parameter(nonneg=True, value=0.1)
    gamma_param = cp.Parameter(nonneg=True, value=0.01)
    
    overlap_expr = cp.hstack(overlaps)
    log_sum_exp = lambda_param * cp.log_sum_exp(overlap_expr / lambda_param)
    entropy = gamma_param * cp.sum(cp.entr(h_var))
    
    # Combine objective
    objective = log_sum_exp - entropy  # Entropy encourages diversity in h
    problem = cp.Problem(cp.Minimize(objective), constraints)
    
    # Iterative optimization with parameter decay
    for i in range(100):
        if time.time() - start_time > budget_s - 5:

        # Adjust regularization parameters

        lambda_param.value = lambda_val
        gamma_param.value = gamma_val
        
        # Solve with ECOS for convex relaxation
        problem.solve(solver=cp.ECOS, verbose=False, warm_start=(i > 0))
