import numpy as np
import time

def run():
    start_time = time.time()
    h_current = initial_h_values.copy()
    n_points = len(h_current)
    dx = 2.0 / n_points
    T = n_points / 2.0

    # Initial evaluation
    overlaps = np.correlate(h_current, 1 - h_current, mode='full') * dx
    best_c5 = np.max(overlaps)
    best_h = h_current.copy()

    patience = 0
    max_patience = 100

    while time.time() - start_time < 1100:
        overlaps = np.correlate(h_current, 1 - h_current, mode='full') * dx
        max_overlap_idx = np.argmax(overlaps)
        m = max_overlap_idx - (n_points - 1)

        valid_i = []
        for i in range(n_points):
            if 0 <= (i + m) < n_points:
                valid_i.append(i)
        
        if not valid_i:
            patience += 1
            continue

        contributions = []
        for i in valid_i:
            contrib = h_current[i] * (1.0 - h_current[i + m])
            contributions.append((contrib, i))
        contributions.sort(reverse=True, key=lambda x: x[0])

        top_pct = 0.1
        num_top = max(1, int(len(valid_i) * top_pct))
        top_is = [i for (c, i) in contributions[:num_top]]

        # Apply update
        for i in top_is:
            i_shifted = i + m
            delta = min(0.1, h_current[i])
            h_current[i] -= delta
            if 0 <= i_shifted < n_points:
                if h_current[i_shifted] < 1.0:
                    add = min(delta, 1.0 - h_current[i_shifted])
                    h_current[i_shifted] += add

        # Maintain constraints
        h_current = np.clip(h_current, 0, 1)
        sum_h = np.sum(h_current)
        if abs(sum_h - T) > 1e-10 and sum_h > 1e-10:
            scale = T / sum_h
            h_current = np.clip(h_current * scale, 0, 1)

        # Update best if improved
        current_c5 = np.max(np.correlate(h_current, 1 - h_current, mode='full') * dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            break

    return (best_h, best_c5, n_points)
