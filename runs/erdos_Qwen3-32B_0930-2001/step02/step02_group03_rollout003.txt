        # Projection to constraints
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum != 0:
            h = h * (n_points / 2 / h_sum)
        else:
            h.fill(0.5)
        
        if iteration % decay_steps == 0 and iteration > 0:
            smooth_eps = max(1e-5, smooth_eps * decay_rate)
    
    return (best_h, best_c5, n_points)
