```python
import numpy as np
import cvxpy as cp
import time

def run():
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    T = n_points / 2.0
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)
    start_time = time.time()
    
    h = best_h.copy()
    
    while time.time() - start_time < 1100:
        # Compute current correlation
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_index = np.argmax(corr)
        k_shift = max_index - n_points + 1
        
        # Compute coefficients for LP
        c = np.zeros(n_points)
        for i in range(n_points):
            shifted_index = i - k_shift
            if 0 <= shifted_index < n_points:
                c[i] = 1.0 - h[shifted_index]
            else:
                c[i] = 1.0  # because 1 - 0
        c *= dx
        
        # Solve LP using CVXPY
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(c @ h_var)
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == T]
        problem = cp.Problem(objective, constraints)
        
        try:
            problem.solve(solver=cp.GLOP, verbose=False)
            if problem.status != cp.OPTIMAL:
                continue
            new_h = h_var.value.copy()
            new_h = np.clip(new_h, 0.0, 1.0)
            h = new_h
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        except Exception as e:
            continue
    
    return (best_h, best_c5, n_points)
```
