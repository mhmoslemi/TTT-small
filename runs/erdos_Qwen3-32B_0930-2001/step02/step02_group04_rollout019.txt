    h_best = h.copy()
    c5_best = np.max(np.correlate(h_best, 1 - h_best, mode='full') * dx)
    
    # Solve LP relaxation for initial guess
    try:
        h_var = cp.Variable(n_points, 0, 1)
        t_var = cp.Variable()
        constraints = [cp.sum(h_var) == target_sum]
        
        for k in range(1, n_points):
            sum_z = 0.0
            for i in range(n_points - k):
                j = i + k
                term = h_var[i] + h_var[j] - 1.0
                z = cp.Variable(nonneg=True)
                constraints.extend([z >= term])
                sum_z += z
            if sum_z > 0:
                constraints.append(sum_z <= t_var)
        
        objective = cp.Minimize(t_var)
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.GLPK if 'GLPK' in cp.installed_solvers() else cp.CVXOPT, verbose=False)
        h_lp = h_var.value.copy()
    except:
        h_lp = h.copy()
    
    h_best = h_lp.copy()
    c5_best = np.max(np.correlate(h_best, 1 - h_best, mode='full') * dx)
    
    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx), conv
    
    # Local optimization with LP solution as starting point
    def objective(x):
        return compute_c5(x)[0]
    
    def constraint_func(x):
        return np.sum(x) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints_opt = [{'type': 'eq', 'fun': constraint_func}]
    
    result = optimize.minimize(
        objective,
        x0=h_best,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints_opt,
        options={'maxiter': 500, 'ftol': 1e-12}
    )
    
    if result.success:
        h_candidate = result.x
        c5_candidate = result.fun
        if c5_candidate < c5_best:
            h_best[:] = h_candidate
            c5_best = c5_candidate
    
    # Perturbation refinement with time budget
    remaining_time = budget_s - (time.time() - start_time)
    while remaining_time > 20 and time.time() - start_time < budget_s - 5:
        h_start = np.clip(h_best + np.random.normal(0, 0.05, n_points), 0.0, 1.0)
        h_start *= target_sum / np.sum(h_start)
        result = optimize.minimize(
            objective,
            x0=h_start,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints_opt,
            options={'maxiter': 200}
        )
        if result.success and result.fun < c5_best:
            h_best[:] = result.x
            c5_best = result.fun
        remaining_time = budget_s - (time.time() - start_time)
    
    return h_best, c5_best, n_points
