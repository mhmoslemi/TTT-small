import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlaps = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlaps)

    # Ensure initial guess satisfies constraint

    constraints = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})

    result = minimize(
        fun=lambda x: objective(x, n_points, dx),
        x0=initial_h,

        constraints=constraints,
        method='L-BFGS-B',

    )
