import numpy as np
import time

def compute_overlaps(h, dx):
    return np.correlate(h, 1 - h, mode="full") * dx

def compute_smooth_max(overlaps, smooth_epsilon):
    max_val = np.max(overlaps)
    shifted = overlaps - max_val
    exp_vals = np.exp(shifted / smooth_epsilon)
    return max_val + smooth_epsilon * np.log(np.sum(exp_vals))

def compute_analytical_gradient(h, overlaps, smooth_epsilon, dx):
    m = len(overlaps)
    n = len(h)
    smooth_grad = np.exp((overlaps - np.max(overlaps)) / smooth_epsilon) / np.sum(
        np.exp((overlaps - np.max(overlaps)) / smooth_epsilon)
    )
    smooth_grad = smooth_grad * smooth_epsilon

    x_coarse = np.linspace(0.0, 2.0, n)
    x_fine = np.linspace(0.0, 2.0, m)

    # Compute gradient from h[i] -> (1 - h[j])
    indices = (np.tile(x_coarse, (m, 1)) <= np.tile(x_fine.reshape(-1, 1), (1, n))) * \
              (np.tile(x_coarse, (m, 1)) > np.tile(np.roll(x_fine, -1).reshape(-1, 1), (1, n)))
    indices[:, -1] = np.tile(x_coarse >= 2 - 1e-9, m)
    indices_tile = np.tile(x_coarse.reshape(1, -1), (m, 1))
    j_indices = np.argmin(np.abs(indices_tile - x_fine.reshape(-1, 1)), axis=0)

    grad1 = np.zeros_like(h)
    for i in range(n):
        valid_k = x_fine <= x_coarse[i] + 1e-9
        k_ind = np.where(valid_k)
        j_ind = j_indices[k_ind]
        grad1[i] += np.sum(smooth_grad[k_ind] * (1 - h[j_ind])) * dx

    # Compute gradient from h[j] -> -h[i]
    indices = (np.roll(x_fine, 1).reshape(-1, 1) <= np.tile(x_coarse.reshape(1, -1), (m, 1))) * \
              (np.tile(x_fine.reshape(-1, 1), (1, n)) > np.tile(x_coarse.reshape(1, -1), (1, n)))
    indices[0, :] = np.tile(x_coarse >= 0.0, 1)
    j_indices2 = np.argmin(np.abs(x_fine.reshape(-1, 1) - x_coarse.reshape(1, -1)), axis=0)

    grad2 = np.zeros_like(h)
    for i in range(n):
        valid_k = x_fine >= x_coarse[i] - 1e-9
        k_ind = np.where(valid_k)
        j_ind = j_indices2[k_ind]
        grad2[i] += -np.sum(smooth_grad[k_ind] * h[j_ind]) * dx

    return grad1 + grad2

def downsample(h_fine, n_coarse):
    n_fine = len(h_fine)
    x_fine = np.linspace(0.0, 2.0, n_fine, endpoint=True)
    x_coarse = np.linspace(0.0, 2.0, n_coarse, endpoint=True)
    indices = np.searchsorted(x_fine, x_coarse, side='right') - 1
    indices = np.clip(indices, 0, n_fine - 1)
