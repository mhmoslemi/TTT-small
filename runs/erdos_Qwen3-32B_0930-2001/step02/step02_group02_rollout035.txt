import numpy as np
import time

def run():
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    n_particles = 50
    c1, c2 = 2.0, 2.0
    w = 0.7
    max_vel = 0.5

    # Initialize swarm
    positions = np.tile(h_initial, (n_particles, 1))
    noise = np.random.uniform(-0.05, 0.05, positions.shape)
    positions += noise
    for i in range(n_particles):
        positions[i] = np.clip(positions[i], 0.0, 1.0)
        s = positions[i].sum()
        if 0 < s < n_points:
            positions[i] = np.clip(positions[i] * (target_sum / s), 0.0, 1.0)

    velocities = np.zeros_like(positions)

    # Objective function
    def calc_c5(p):
        corr = np.correlate(p, 1.0 - p, mode='full')
        return np.max(corr) * dx

    # Personal best and global best
    pbest_positions = positions.copy()
    pbest_values = np.array([calc_c5(p) for p in positions])
    gbest_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]

    best_h = gbest_position.copy()
    best_c5 = gbest_value

    while time.time() - start_time < 1100:
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (pbest_positions[i] - positions[i]) +
                             c2 * r2 * (gbest_position - positions[i]))
            velocities[i] = np.clip(velocities[i], -max_vel, +max_vel)
            positions[i] += velocities[i]
            positions[i] = np.clip(positions[i], 0.0, 1.0)
            s = positions[i].sum()
            if 0 < s < n_points:
                positions[i] = np.clip(positions[i] * (target_sum / s), 0.0, 1.0)

        current_values = np.array([calc_c5(p) for p in positions])
        for i in range(n_particles):
            if current_values[i] < pbest_values[i]:
                pbest_positions[i] = positions[i].copy()
                pbest_values[i] = current_values[i]

        new_gbest_idx = np.argmin(pbest_values)
        if pbest_values[new_gbest_idx] < gbest_value:
            gbest_position = pbest_positions[new_gbest_idx].copy()
            gbest_value = pbest_values[new_gbest_idx]

        if gbest_value < best_c5:
            best_h[:] = gbest_position.copy()
            best_c5 = gbest_value

    return (best_h, best_c5, n_points)
