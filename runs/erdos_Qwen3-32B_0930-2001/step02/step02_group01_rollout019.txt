def project(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    sum_current = h_clipped.sum()
    if abs(sum_current - target_sum) < 1e-8:
        return h_clipped
    # Need to adjust to reach target_sum
    # Find the variables that are not at their bounds (0 < h < 1)
    active = np.where((h_clipped > 0) & (h_clipped < 1))[0]
    if active.size == 0:

        return h_clipped

    # Clip again in case any values went out of bounds due to delta

    return h_adjusted
