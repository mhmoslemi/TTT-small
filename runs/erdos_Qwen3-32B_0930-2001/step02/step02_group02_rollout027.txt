import numpy as np
from scipy.stats import multivariate_normal
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    def evaluate(h_vec):
        return np.max(np.correlate(h_vec, 1 - h_vec, mode='full')) * dx
    
    def repair(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        s = np.sum(h_vec)
        if s == 0:
            return h_vec
        scaling = target_sum / s
        h_vec *= scaling
        h_vec = np.clip(h_vec, 0, 1)
        return h_vec
    
    mean = h.copy()
    covariance = 0.1 * np.eye(n_points)
    
    num_samples = 200
    elite_proportion = 0.1
    best_c5 = evaluate(repair(h))
    best_h = h.copy()
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        samples = multivariate_normal.rvs(mean, covariance, size=num_samples)
        candidates = []
        for sample in samples:
            repaired = repair(sample)
            c5 = evaluate(repaired)
            candidates.append((c5, repaired))
        candidates.sort()
        elite_candidates = candidates[:max(1, int(num_samples * elite_proportion))]
        if not elite_candidates:
            break
        
        elite_h = np.array([c[1] for c in elite_candidates])
        new_mean = np.mean(elite_h, axis=0)
        new_cov = np.cov(elite_h, rowvar=False) + 1e-6 * np.eye(n_points)
        
        mean = new_mean
        covariance = new_cov
        
        if elite_candidates and elite_candidates[0][0] < best_c5:
            best_c5 = elite_candidates[0][0]
            best_h = elite_candidates[0][1].copy()
    
    best_h = repair(best_h)
    return (best_h, best_c5, n_points)
