import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()

    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)

    def generate_candidate(guess):
        noise = np.random.normal(0, 0.1, n_points)
        new_h = np.clip(guess + noise, 0.0, 1.0)
        adj = (target_sum - np.sum(new_h)) / n_points
        new_h += adj
        return np.clip(new_h, 0.0, 1.0)

    population_size = 100
    population = np.array([generate_candidate(h0) for _ in range(population_size)])

    while time.time() - start_time < budget_s - 30:
        c5_scores = np.array([compute_c5(ind) for ind in population])
        best_indices = np.argsort(c5_scores)
        parents = population[best_indices[:20]]

        offspring = []
        for _ in range(80):
            p1, p2 = parents[np.random.choice(20, 2)]
            cp = np.random.randint(1, n_points - 1)
            child = np.concatenate([p1[:cp], p2[cp:]])
            mask = np.random.rand(n_points) < 0.1
            child += mask * np.random.normal(0, 0.1, n_points)
            child = np.clip(child, 0, 1)
            child += (target_sum - child.sum()) / n_points
            child = np.clip(child, 0, 1)
            offspring.append(child)
        population = np.vstack((parents, np.array(offspring)))

    c5_scores = [compute_c5(ind) for ind in population]
    best_indices = np.argsort(c5_scores)
    top_candidates = population[best_indices[:10]]

    h_best = population[best_indices[0]]
    c5_best = c5_scores[best_indices[0]]

    for h0_refine in top_candidates:
        def constraint(x):
            return np.sum(x) - target_sum

        res = optimize.minimize(
            compute_c5,
            x0=h0_refine,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        if res.success:
            c5_candidate = res.fun
            if c5_candidate < c5_best:
                h_best[:] = res.x
                c5_best = c5_candidate

    return h_best, c5_best, n_points
