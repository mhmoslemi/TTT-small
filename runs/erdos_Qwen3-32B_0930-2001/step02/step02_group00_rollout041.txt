import cvxpy as cp
import time

def run():
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    T = n_points / 2.0
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    start_time = time.time()
    budget_s = 1100

    while time.time() - start_time < budget_s - 1:  # leave 1s buffer
        # Compute current overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        max_over_value = corr[max_index] * dx
        if max_over_value < best_c5:
            best_c5 = max_over_value
            best_h = h.copy()

        k_shift = max_index - (n_points - 1)

        # Compute shifted_h
        shifted_h = np.zeros(n_points)
        for i in range(n_points):
            shifted_index = i - k_shift
            if 0 <= shifted_index < n_points:
                shifted_h[i] = h[shifted_index]
            else:
                shifted_h[i] = 0.0  # zero-padding as per full mode

        # Define LP variables
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(h_var * (1 - shifted_h)) * dx)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == T
        ]
        problem = cp.Problem(objective, constraints)
        # Solve using ECOS, which is available in most environments
        problem.solve(solver=cp.ECOS)

        if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            break

        h_candidate = h_var.value

        # Ensure feasibility: sometimes ECOS might have small violations
        h_candidate = np.clip(h_candidate, 0, 1)
        sum_h_candidate = np.sum(h_candidate)
        if abs(sum_h_candidate - T) > 1e-8:
            # Adjust to meet the constraint
            h_candidate *= T / sum_h_candidate
            h_candidate = np.clip(h_candidate, 0, 1)

        h = h_candidate.copy()

    # Final evaluation of best_h
    final_corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(final_corr)
    return (best_h, final_c5, n_points)

import numpy as np
import cvxpy as cp
import time

def run():
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    T = n_points / 2.0
    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full') * dx
    best_c5 = np.max(corr)
    start_time = time.time()
    budget_s = 1100

    while time.time() - start_time < budget_s - 1:
        # Find the current maximum overlap and corresponding shift
        corr_current = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr_current)
        k_shift = max_index - (n_points - 1)

        # Compute shifted_h for the current k_shift
        shifted_h = np.zeros(n_points)
        for i in range(n_points):
            shifted_index = i + k_shift
            if 0 <= shifted_index < n_points:
                shifted_h[i] = h[shifted_index]
            else:
                shifted_h[i] = 0.0  # zero-padding for out-of-bounds

        # Formulate and solve the LP
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(h_var * (1 - shifted_h)) * dx)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == T
        ]
        problem = cp.Problem(objective, constraints)
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
        except:
            break

        if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            break

        h_candidate = h_var.value

        # Ensure feasibility
        h_candidate = np.clip(h_candidate, 0, 1)
        sum_h_candidate = np.sum(h_candidate)
        if abs(sum_h_candidate - T) > 1e-8:
            h_candidate *= T / sum_h_candidate
            h_candidate = np.clip(h_candidate, 0, 1)

        h = h_candidate.copy()

        # Update best solution
        corr_current = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr_current)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    return (best_h, best_c5, n_points)
