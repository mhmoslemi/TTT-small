        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)
    
    best_c5 = compute_c5(h_current)
    best_h = h_current.copy()

    # Alternating optimization between halves
    half = n_points // 2
    part_first = np.arange(half)
    part_second = np.arange(half, n_points)

    for alt_iter in range(8):
        # Optimize first half while keeping second half fixed
        fixed_part = h_current[part_second]
        sum_fixed = fixed_part.sum()
        sum_part = target_sum - sum_fixed
        var_initial = h_current[part_first].copy()
        # Ensure initial guess meets sum constraint approximately
        var_sum = var_initial.sum()
        if var_sum != 0:
            var_initial *= (sum_part / var_sum)
        bounds = [(0, 1)] * half
        
        def objective(x):
            x = np.clip(x, 0, 1)
            full_h = np.concatenate([x, fixed_part], axis=0)
            full_h = np.clip(full_h, 0, 1)
            conv = np.correlate(full_h, 1 - full_h, mode='full')
            return np.max(conv * dx)
        
        def constraint_sum(x):
            return x.sum() - sum_part
        
        def final_adjust(x):
            x = np.clip(x, 0, 1)
            full_h = np.concatenate([x, fixed_part], axis=0)
            full_h_sum = full_h.sum()
            if full_h_sum != 0:
                full_h *= (target_sum / full_h_sum)
            full_h = np.clip(full_h, 0, 1)
            return full_h[part_first]
        
        result = optimize.minimize(objective, var_initial, method='SLSQP',
                                   bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': constraint_sum}],
                                   options={'maxiter': 40})
        if result.success:
            new_var = result.x
            h_current = final_adjust(new_var)
        
        # Check if C5 improved
        current_c5 = compute_c5(h_current)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()

        # Optimize second half while keeping first half fixed
        fixed_part = h_current[part_first]
        sum_fixed = fixed_part.sum()
        sum_part = target_sum - sum_fixed
        var_initial = h_current[part_second].copy()
        var_sum = var_initial.sum()
        if var_sum != 0:
            var_initial *= (sum_part / var_sum)
        bounds = [(0, 1)] * (n_points - half)
        
        def objective_second(x):
            x = np.clip(x, 0, 1)
            full_h = np.concatenate([fixed_part, x], axis=0)
            full_h = np.clip(full_h, 0, 1)
            conv = np.correlate(full_h, 1 - full_h, mode='full')
            return np.max(conv * dx)
        
        def constraint_sum_second(x):
            return x.sum() - sum_part
        
        def final_adjust_second(x):
            x = np.clip(x, 0, 1)
            full_h = np.concatenate([fixed_part, x], axis=0)
            full_h_sum = full_h.sum()
            if full_h_sum != 0:
                full_h *= (target_sum / full_h_sum)
            full_h = np.clip(full_h, 0, 1)
            return full_h[part_second]
        
        result = optimize.minimize(objective_second, var_initial, method='SLSQP',
                                   bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': constraint_sum_second}],
                                   options={'maxiter': 40})
        if result.success:
            new_var = result.x
            h_current = final_adjust_second(new_var)
        
        current_c5 = compute_c5(h_current)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()

        if time.time() - start_time > budget_s - 300:
            break
    
    # Final refinement with DE and perturbed SLSQP
    bounds = [(0, 1)] * (n_points - 1)
    def objective_de(x):
        h = np.append(x, target_sum - sum(x))
        h = np.clip(h, 0, 1)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)
    
    if time.time() - start_time < budget_s - 100:
        result = optimize.differential_evolution(objective_de, bounds, strategy='best1bin',
                                                 maxiter=10, polish=False)
        x_de = result.x
        h_de = np.append(x_de, target_sum - x_de.sum())
        h_de = np.clip(h_de, 0, 1)
        h_de_sum = h_de.sum()
        if h_de_sum != 0:
            h_de *= (target_sum / h_de_sum)
        h_de = np.clip(h_de, 0, 1)
        c5_de = compute_c5(h_de)
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de.copy()

    # Perturbed refinements
    for _ in range(8):
        if time.time() - start_time > budget_s - 10:
            break
        pert = np.random.rand(n_points) * 0.1
        h_trial = best_h.copy()
        h_trial += pert
        h_trial = np.clip(h_trial, 0, 1)
        h_trial_sum = h_trial.sum()
        if h_trial_sum != 0:
            h_trial *= (target_sum / h_trial_sum)
        h_trial = np.clip(h_trial, 0, 1)
        
        result = optimize.minimize(compute_c5, h_trial, method='SLSQP',
                                   bounds=[(0, 1)] * n_points,
                                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}],
                                   options={'maxiter': 200})
        if result.success:
            c5_new = compute_c5(result.x)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = result.x
    
    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Best C5: {best_c5:.6f}")
    
    return best_h, best_c5, n_points
