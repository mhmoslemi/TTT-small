import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()
    
    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)
    
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    
    while time.time() - start_time < budget_s - 10:
        if np.random.rand() < 0.3:
            # Generate random initial guess
            h0 = np.random.uniform(0.0, 1.0, size=n_points)
            current_sum = np.sum(h0)
            adjustment = (target_sum - current_sum) / n_points
            h0 += adjustment
            h0 = np.clip(h0, 0.0, 1.0)
        else:
            # Perturb current best
            noise = np.random.normal(0, 0.15, size=n_points)
            perturbed = h_best + noise
            perturbed = np.clip(perturbed, 0.0, 1.0)
            current_sum = np.sum(perturbed)
            adjustment = (target_sum - current_sum) / n_points
            perturbed += adjustment
            perturbed = np.clip(perturbed, 0.0, 1.0)
            h0 = perturbed
        
        result = optimize.minimize(
            compute_c5,
            x0=h0,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            c5_candidate = result.fun
            h_candidate = result.x
            if c5_candidate < c5_best:
                h_best[:] = h_candidate.copy()
                c5_best = c5_candidate
    
    return h_best, c5_best, n_points
