import numpy as np
import cvxpy as cp
import time

def compute_C5(h):
    dx = 2.0 / len(h)
    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(overlaps)

def interpolate_coarse_to_fine(h_coarse, n_coarse, n_fine):
    x_coarse = np.linspace(0, 2, n_coarse, endpoint=False)
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    h_fine *= (n_fine / 2) / h_fine.sum()
    return h_fine

def optimize_grid(h_initial, n, dx, smooth_eps, gamma):
    h_var = cp.Variable(n)
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2
    ]
    
    O = []
    for k in range(n):
        indices = [(i + k) % n for i in range(n)]
        shifted_h = h_var[indices]
        sum_S = cp.sum(h_var * shifted_h)
        O_k = 1 - dx * sum_S
        O.append(O_k)
    
    log_sum_exp = smooth_eps * cp.log_sum_exp(cp.hstack(O) / smooth_eps)
    entropy = gamma * cp.sum(cp.entr(h_var))
    objective = log_sum_exp - entropy
    
    problem = cp.Problem(cp.Minimize(objective), constraints)
    h_var.value = h_initial.copy()
    problem.solve(solver=cp.ECOS, max_iter=150, verbose=False)
    
    h_opt = h_var.value
    h_opt = np.clip(h_opt, 0, 1)
    if h_opt.sum() != 0:
        h_opt *= (n / 2) / h_opt.sum()
    else:
        h_opt = h_initial.copy()
    return h_opt
