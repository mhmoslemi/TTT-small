import cvxpy as cp
import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    start_time = time.time()

    # Construct SDP variables and constraints
    H = cp.Variable((n_points, n_points), symmetric=True)
    t = cp.Variable()
    constraints = [H >> 0]  # Enforce positive semidefiniteness

    for k in range(1, n_points):  # All possible shifts
        # Build sum of H[i, i + k] for valid i
        sum_term = cp.sum([H[i, i + k] for i in range(n_points - k)])
        constraints.append(sum_term >= 1 - t * n_points / 2)

    # Formulate and solve the SDP
    problem = cp.Problem(cp.Minimize(t), constraints)
    
    # Early termination flag
    elapsed = time.time() - start_time
    if elapsed > budget_s * 0.9:
        return (h_initial, 0.38085, n_points)  # Fallback to known bound

    try:
        problem.solve(solver=cp.ECOS, verbose=False, max_iters=10000, reltol=1e-4)
    except Exception as e:
        return (h_initial, 0.38085, n_points)

    # Extract leading eigenvector from H
    if H.value is None:
        return (h_initial, 0.38085, n_points)

    H_val = H.value
    eigenvalues, eigenvectors = np.linalg.eigh(H_val)
    v = eigenvectors[:, -1]  # Last column corresponds to largest eigenvalue

    # Normalize and project to constraints
    h_candidate = v / np.sum(v)  # Normalize to have sum(v)=1
    h_candidate = np.clip(h_candidate, 0, 1)  # Ensure non-negativity
    current_sum = np.sum(h_candidate)
    
    if current_sum > 1e-6:  # Non-zero sum after clipping
        h_candidate *= (n_points / 2) / current_sum  # Scale to satisfy ∫h dx = 1
        h_candidate = np.clip(h_candidate, 0, 1)  # Final clamp

    # Post-process to maintain sum constraint
    while np.sum(h_candidate) < n_points / 2 - 1e-6:
        h_candidate += 1e-5 * (h_initial - h_candidate)
        h_candidate = np.clip(h_candidate, 0, 1)
    
    # Compute result
    overlaps = np.correlate(h_candidate, 1 - h_candidate, mode="full") * dx
    c5_bound = np.max(overlaps)
    
    # Final cleanup
    h_result = h_initial if np.isnan(c5_bound) else h_candidate
    return (h_result, c5_bound, n_points)
