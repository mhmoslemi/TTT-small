import numpy as np
import cvxpy as cp
import time

def run():
    start_time = time.time()
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h_initial.copy()
    best_c5 = 0.38085  # current upper bound

    lower = 0.0
    upper = 0.38085
    precision = 1e-5

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        return np.max(corr)

    while time.time() - start_time < 1100:
        mid = (lower + upper) / 2
        h = cp.Variable(n_points, nonneg=True)
        w = cp.Variable((n_points, n_points), nonneg=True)
        constraints = [
            cp.sum(h) == target_sum,
            h >= 0,
            h <= 1
        ]

        for i in range(n_points):
            for j in range(n_points):
                constraints += [
                    w[i, j] >= h[i] + h[j] - 1,
                    w[i, j] <= h[i],
                    w[i, j] <= h[j],
                    w[i, j] >= 0
                ]

        for k in range(1, n_points):
            # Positive shift
            valid = range(n_points - k)
            sum_pos = sum(h[i] - w[i, i + k] for i in valid)
            constraints.append(sum_pos * dx <= mid)
            # Negative shift
            valid = range(k, n_points)
            sum_neg = sum(h[i] - w[i, i - k] for i in valid)
            constraints.append(sum_neg * dx <= mid)

        # k=0 case
        sum_zero = sum(h[i] - w[i, i] for i in range(n_points))
        constraints.append(sum_zero * dx <= mid)

        problem = cp.Problem(cp.Minimize(0), constraints)
        try:

                         mosek_params={'MSK_DPAR_OPTIMIZER_MAX_TIME': 60.0})
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h.value
