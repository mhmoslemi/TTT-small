import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()

        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)
    
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    
    def generate_initial_guess(guess):
        if np.random.rand() < 0.35:
            indices = np.random.choice(n_points, size=int(target_sum), replace=False)
            new_guess = np.zeros(n_points)
            new_guess[indices] = 1.0
        else:
            new_guess = guess.copy()
        noise = np.random.normal(0, 0.25, size=n_points)
        perturbed = new_guess + noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        current_sum = np.sum(perturbed)
        adjustment = (target_sum - current_sum) / n_points
        perturbed += adjustment
        perturbed = np.clip(perturbed, 0.0, 1.0)
        return perturbed
    
    while time.time() - start_time < budget_s - 10:
        h0 = generate_initial_guess(h_best)
        result = optimize.minimize(
            compute_c5,
            x0=h0,
            method='L-BFGS-B',
