import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy().astype(np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()

    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)

    h_best = h.copy()
    c5_best = compute_c5(h_best)

    def generate_initial_guess(base, scale=0.1):
        noise = np.random.normal(0, scale, size=n_points)
        perturbed = base + noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        current_sum = np.sum(perturbed)
        adjustment = (target_sum - current_sum) / n_points
        perturbed += adjustment
        perturbed = np.clip(perturbed, 0.0, 1.0)
        return perturbed

    # Genetic Algorithm parameters
    pop_size = 100
    elite_frac = 0.2
    n_survivors = 50
    crossover_frac = 0.5
    mutation_rate = 0.1
    generations = 0
    pop = [generate_initial_guess(h) for _ in range(pop_size)]

    while time.time() - start_time < budget_s - 20:
        # Compute fitness
        pop_c5 = np.array([compute_c5(ind) for ind in pop])
        elite_idx = pop_c5.argsort()[:int(pop_size * elite_frac)]
        elites = [pop[i] for i in elite_idx]

        # Crossover and mutation
        children = []
        for _ in range(n_survivors):
            p1, p2 = np.random.choice(elites, 2, replace=False)
            cross = np.random.rand(n_points) < crossover_frac
            child = p1 * cross + p2 * (1 - cross)
            children.append(child)

        for child in children:
            mutation_mask = np.random.rand(n_points) < mutation_rate
            child[mutation_mask] += np.random.normal(0, 0.15, size=mutation_mask.sum())
            child = np.clip(child, 0.0, 1.0)
            child_sum = np.sum(child)
            adjustment = (target_sum - child_sum) / n_points
            child += adjustment
            child = np.clip(child, 0.0, 1.0)
            pop.append(child)

        # Feedback and truncate
        pop = sorted(pop, key=lambda x: compute_c5(x))[:pop_size]
        current_best_c5 = compute_c5(pop[0])
        if current_best_c5 < c5_best:
            h_best[:] = pop[0].copy()
            c5_best = current_best_c5
        generations += 1

        # Local refinement on top candidates
        for candidate in pop[:int(pop_size * 0.1)]:
            result = optimize.minimize(
                compute_c5,
                x0=candidate,
                method='SLSQP',
                bounds=[(0, 1)] * n_points,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                options={'maxiter': 80, 'ftol': 1e-10}
            )
            if result.success:
                c5_candidate = result.fun
                h_candidate = result.x
                if c5_candidate < c5_best:
                    h_best[:] = h_candidate.copy()
                    c5_best = c5_candidate

        if generations % 5 == 0:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 0:
                break

    return h_best, c5_best, n_points
