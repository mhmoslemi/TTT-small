import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_current = initial_h_values.copy()
    n_points = len(h_current)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    def normalize_h(h_vec):
        h = np.clip(h_vec, 0, 1).copy()
        current_sum = h.sum()
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        return h

    def objective_nm(x):
        h = normalize_h(x)
        return compute_c5(h) + 1e4 * (h.sum() - target_sum)**2

    def objective_slsqp(x):
        return compute_c5(x)

    h_initial = normalize_h(h_current)
    best_c5 = compute_c5(h_initial)
    best_h = h_initial.copy()

    # Perturbation parameters
    n_perts = 20
    max_pert = 0.15

    for _ in range(n_perts):
        pert = (np.random.rand(n_points) * 2 - 1) * max_pert
        h_pert = h_current + pert
        h_pert = normalize_h(h_pert)

        res_nm = optimize.minimize(
            objective_nm, h_pert, method='Nelder-Mead', bounds=[(0,1)]*n_points,
            options={'maxiter': 100, 'xatol': 1e-5, 'fatol': 1e-5}
        )
        if res_nm.success:
            h_nm = res_nm.x
            h_nm = normalize_h(h_nm)

            cons_slsqp = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
            res_slsqp = optimize.minimize(
                objective_slsqp, h_nm, method='SLSQP', bounds=[(0,1)]*n_points,
                constraints=cons_slsqp, options={'maxiter': 200, 'ftol': 1e-8}
            )
            if res_slsqp.success:
                h_slsqp = normalize_h(res_slsqp.x)
                c5 = compute_c5(h_slsqp)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_slsqp

        if time.time() - start_time > budget_s - 30:
            break

    # Final refinement
    if best_c5 < float('inf'):
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res_refine = optimize.minimize(
            objective_slsqp, best_h, method='SLSQP', bounds=[(0,1)]*n_points,
            constraints=cons, options={'maxiter': 500}
        )
        if res_refine.success:
            best_h = normalize_h(res_refine.x)
            best_c5 = compute_c5(best_h)

    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Best C5: {best_c5:.6f}")
    
    return best_h, best_c5, n_points
