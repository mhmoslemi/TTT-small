import numpy as np
import time

def evaluate_c5(h, dx):
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def run():
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    T_required = n_points / 2.0

    h_current = h_initial.copy()
    best_h = h_current.copy()
    best_c5 = evaluate_c5(h_current, dx)

    temp = 10.0
    cooling_rate = 0.995
    sigma = 0.1
    swap_prob = 0.2  # Probability of block swap vs Gaussian noise

    start_time = time.time()

    while time.time() - start_time < 1100:
        h_candidate = h_current.copy()

        if np.random.rand() < swap_prob:
            # Reverse a random segment (block swap)
            i, j = np.sort(np.random.randint(0, n_points, 2))
            h_candidate[i:j] = h_candidate[i:j][::-1]
        else:
            # Add Gaussian noise to random subset of points
            mask = np.random.rand(n_points) < 0.5
            noise = np.random.normal(0, sigma, n_points)
            h_candidate[mask] += noise[mask]

        # Enforce bounds and sum constraint
        h_candidate = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h_candidate)
        if sum_h != 0:
            scale = T_required / sum_h
            h_candidate = np.clip(h_candidate * scale, 0, 1)

        # Compute C5 values
        current_c5 = evaluate_c5(h_current, dx)
        candidate_c5 = evaluate_c5(h_candidate, dx)
        delta = candidate_c5 - current_c5

        # Metropolis acceptance criterion
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h_current = h_candidate.copy()
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h[:] = h_current.copy()

        # Cool down
        temp *= cooling_rate

    return (best_h, best_c5, n_points)
