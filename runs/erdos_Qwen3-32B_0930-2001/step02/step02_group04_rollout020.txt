import time
import numpy as np
import cvxpy as cp
import scipy.optimize as optimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()
    
    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)
    
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    
    # Try to solve the LP to get a better initial guess
    try:
        # CVXPY variables
        h_var = cp.Variable(n_points)
        t_var = cp.Variable()
        constraints = []
        # Basic constraints
        constraints += [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
        
        # For each shift k from 1 to n_points -1
        for k in range(1, n_points):
            z_list = []
            for i in range(n_points - k):
                j = i + k
                z = cp.Variable()
                z_list.append(z)
                constraints += [z >= h_var[i] + h_var[j] - 1, z >= 0]
            constraints += [cp.sum(z_list) >= t_var]
        
        # Maximize t_var
        problem = cp.Problem(cp.Maximize(t_var), constraints)
        problem.solve(solver=cp.GLOP, verbose=False)
        if problem.status == cp.OPTIMAL:
            h_lp = h_var.value
            if h_lp is not None:
                # Clip to [0,1] in case of numerical issues
                h_lp = np.clip(h_lp, 0.0, 1.0)
                # Adjust the sum to be exactly target_sum in case of minor errors
                current_sum = np.sum(h_lp)
                if not np.isclose(current_sum, target_sum, atol=1e-6):
                    adjustment = (target_sum - current_sum) / n_points
                    h_lp += adjustment
                    h_lp = np.clip(h_lp, 0.0, 1.0)
                h_initial = h_lp
            else:
                h_initial = h.copy()
        else:
            h_initial = h.copy()
    except Exception as e:
        h_initial = h.copy()
    
    # Now perform local optimization using this h_initial
    # Define the optimization function
    def optimize_wrapper(x):
        h_flat = x.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)
    
    def constraint_sum(x):
        return np.sum(x) - target_sum
    
    # Initial guess
    h0 = h_initial
    # Local optimization
    while time.time() - start_time < budget_s - 10:
        # Add some noise to h_initial for better exploration
        noise = np.random.normal(0, 0.15, size=n_points)
        perturbed = h_best + noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        current_sum = np.sum(perturbed)
        adjustment = (target_sum - current_sum) / n_points
        perturbed += adjustment
        perturbed = np.clip(perturbed, 0.0, 1.0)
        h0 = perturbed
        result = optimize.minimize(
            optimize_wrapper,
            x0=h0,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 80, 'ftol': 1e-10}
        )
        if result.success:
            c5_candidate = result.fun
            h_candidate = result.x
            if c5_candidate < c5_best:
                h_best[:] = h_candidate.copy()
                c5_best = c5_candidate
    
    return h_best, c5_best, n_points

        h_flat = h_vec.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv * dx)
    
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    
    # Try to solve the LP to get a better initial guess
    h_initial = h.copy()
    try:
        h_var = cp.Variable(n_points)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
        
        for k in range(1, n_points):
            z_list = []
            for i in range(n_points - k):
                j = i + k
                z = cp.Variable()
                z_list.append(z)
                constraints += [z >= h_var[i] + h_var[j] - 1, z >= 0]
            if z_list:
                constraints.append(cp.sum(z_list) >= t_var)
        
        problem = cp.Problem(cp.Maximize(t_var), constraints)
        problem.solve(solver=cp.GLOP, verbose=False)
        if problem.status == cp.OPTIMAL:
            h_lp = np.clip(h_var.value, 0, 1)
            h_sum = np.sum(h_lp)
            if not np.isclose(h_sum, target_sum, atol=1e-6):
                adj = (target_sum - h_sum) / n_points
