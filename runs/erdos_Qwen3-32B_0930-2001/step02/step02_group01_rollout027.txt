            new_variance = damping_factor * variance + (1 - damping_factor) * np.var(elites, axis=0) + min_variance
            mean = new_mean
            variance = np.clip(new_variance, 1e-6, 1)

        if len(c5_values) > 0 and c5_values[sorted_indices[0]] < best_c5:
            best_c5 = c5_values[sorted_indices[0]]
            best_h = samples[sorted_indices[0]].copy()

    def objective_slsqp(h_vec):
        return compute_c5(h_vec)

    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    result = optimize.minimize(
        objective_slsqp, best_h, method='SLSQP', bounds=[(0, 1)]*n_points,
        constraints=constraints, options={'maxiter': 500, 'ftol': 1e-8}
    )
    if result.success:
        h_refined = result.x
        c5_refined = compute_c5(h_refined)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined

    remaining = budget_s - (time.time() - start_time)
    if remaining > 0:
        for _ in range(int(remaining // 10)):
            if time.time() - start_time >= budget_s:
                break
            pert = np.random.randn(n_points) * 0.2
            h_candidate = repair_h(best_h + pert)
            result = optimize.minimize(
                objective_slsqp, h_candidate, method='SLSQP', bounds=[(0, 1)]*n_points,
                constraints=constraints, options={'maxiter': 200}
            )
            if result.success and compute_c5(result.x) < best_c5:
                best_c5 = compute_c5(result.x)
                best_h = result.x

    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Best C5: {best_c5:.6f}")
    
    return best_h, best_c5, n_points
