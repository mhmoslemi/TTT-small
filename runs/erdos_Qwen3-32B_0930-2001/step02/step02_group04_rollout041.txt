    h_lp = cp.Variable(n_points)
    t = cp.Variable()
    constraints = [
        h_lp >= 0,
        h_lp <= 1,
        cp.sum(h_lp) == target_sum
    ]
    for k in range(1, n_points):
        m = n_points - k
        z_k = cp.Variable(m, nonnegative=True)
        for i in range(m):
            constraints.append(z_k[i] >= h_lp[i] + h_lp[i + k] - 1)
        constraints.append(cp.sum(z_k) >= t)
    obj = cp.Maximize(t)
    problem = cp.Problem(obj, constraints)

    initial_guess = h.copy()
    try:
        time_left = budget_s - (time.time() - start_time) - 60
        if time_left > 30:
            problem.solve(solver=cp.GLPK, verbose=False)
            if problem.status in {cp.OPTIMAL, cp.OPTIMAL_INACCURATE}:
                h_candidate = np.clip(np.array(h_lp.value).ravel(), 0, 1)
                if abs(np.sum(h_candidate) - target_sum) < 1e-6:

        return np.max(conv * dx)
