import numpy as np
import time

def run():
    start_time = time.time()
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def feasibility_repair(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.full_like(h_clipped, 0.5)
        else:
            scale = (target_sum / current_sum)
            h_scaled = h_clipped * scale
            return np.clip(h_scaled, 0, 1)

    def evaluate_c5(h):
        kernel = 1.0 - h
        corr = np.correlate(h, kernel, mode='full')
        return np.max(corr) * dx

    swarm_size = 50
    c1, c2, w = 2.0, 2.0, 0.7
    max_vel = 0.5

    # Initialize swarm and velocities
    np.random.seed(int(time.time()))
    swarm = np.random.randn(swarm_size, n_points) * 0.1 + h0
    swarm = np.array([feasibility_repair(p) for p in swarm])
    velocities = np.random.rand(swarm_size, n_points) * max_vel * 0.1
    pbest = np.copy(swarm)
    pbest_obj = np.array([evaluate_c5(p) for p in pbest])
    gbest_idx = np.argmin(pbest_obj)
    gbest = np.copy(pbest[gbest_idx])
    gbest_obj = pbest_obj[gbest_idx]

    while time.time() - start_time < 1100:
        # Update personal and global bests
        for i in range(swarm_size):
            swarm[i] = feasibility_repair(swarm[i])
            obj = evaluate_c5(swarm[i])
            if obj < pbest_obj[i]:
                pbest[i] = swarm[i].copy()
                pbest_obj[i] = obj
        current_gbest_idx = np.argmin(pbest_obj)
        if pbest_obj[current_gbest_idx] < gbest_obj:
            gbest = pbest[current_gbest_idx].copy()
            gbest_obj = pbest_obj[current_gbest_idx]

        # Update velocities and positions
        r1 = np.random.rand(swarm_size, 1)
        r2 = np.random.rand(swarm_size, 1)
        cognitive = c1 * r1 * (pbest - swarm)
        social = c2 * r2 * (gbest - swarm)
        velocities = w * velocities + cognitive + social
        velocities = np.clip(velocities, -max_vel, max_vel)
        swarm += velocities
        swarm = np.array([feasibility_repair(p) for p in swarm])

    # Final validation and return
    final_h = feasibility_repair(gbest)
    final_obj = evaluate_c5(final_h)
    final_h = np.clip(final_h, 0, 1)
    assert np.isclose(np.sum(final_h), target_sum, atol=1e-6)
    return (final_h, final_obj, n_points)
