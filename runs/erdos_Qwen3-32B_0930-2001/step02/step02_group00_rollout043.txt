import numpy as np
import cvxpy as cp
import time

def run():
    start_time = time.time()
    h_prev = initial_h_values.copy()
    n_points = len(h_prev)
    dx = 2.0 / n_points
    T = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    best_c5 = compute_c5(h_prev)
    best_h = h_prev.copy()

    while time.time() - start_time < 1100:
        # Compute current overlaps to find worst shift k
        corr = np.correlate(h_prev, 1 - h_prev, mode='full')
        m_max = np.argmax(corr)
        k = m_max - (n_points - 1)

        # Create coefficient array for current shift
        coeff = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                coeff[i] = 1.0 - h_prev[j]
            else:
                coeff[i] = 1.0

        # CVXPY problem to minimize overlap at current worst shift
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1, cp.sum(h) == T]
        objective = cp.Minimize(cp.sum(coeff * h) * dx)
        prob = cp.Problem(objective, constraints)

        # Solve with ECOS solver
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, verbose=False)
        except:
            pass

        # Update h_prev if solution is valid
        if h.value is not None:
            h_new = np.clip(h.value, 0, 1)
            sum_h = np.sum(h_new)
            if abs(sum_h - T) > 1e-10:
                h_new = np.clip(h_new * (T / sum_h), 0, 1)
            h_prev = h_new

            current_c5 = compute_c5(h_prev)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_prev.copy()

    return (best_h, best_c5, n_points)
