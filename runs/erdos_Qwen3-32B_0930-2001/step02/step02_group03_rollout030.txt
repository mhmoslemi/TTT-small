        overlaps = np.correlate(h_values, 1 - h_values, mode="full") * dx
        return np.max(overlaps)

    h_init_sum = np.sum(h_init)

    stages = [

    ]

        overlaps = []

            indices = np.arange(n_points - k)
            term = h[indices] * (1 - h[indices + k])
            overlaps.append(cp.sum(term))
