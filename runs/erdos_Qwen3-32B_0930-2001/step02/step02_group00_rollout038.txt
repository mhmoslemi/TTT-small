import numpy as np
import time

def run(seed=42, budget_s=1100):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    T = n_points / 2.0

    best_h = h_initial.copy()
    best_c5 = max(np.correlate(best_h, 1 - best_h, mode='full') * dx)

    start_time = time.time()

    while time.time() - start_time < budget_s:
        h_current = best_h.copy()
        overlaps = np.correlate(h_current, 1 - h_current, mode='full')
        relevant_overlaps = overlaps[n_points - 1:]
        k_max_index = np.argmax(relevant_overlaps)
        k_max = k_max_index

        valid_x = np.arange(n_points - k_max)
        contributions = h_current[valid_x] * (1 - h_current[valid_x + k_max])
        top_indices = np.argsort(contributions)[::-1]
        num_top = max(1, len(top_indices) // 10)
        top_x = top_indices[:num_top]

        for x in top_x:
            delta = min(h_current[x], 0.1)
            h_current[x] -= delta
            if x + k_max < n_points:
                h_current[x + k_max] = np.clip(h_current[x + k_max] + delta, 0, 1)

        h_current = np.clip(h_current, 0, 1)
        current_sum = np.sum(h_current)
        if abs(current_sum - T) > 1e-10:
            scale_factor = T / current_sum
            h_current = np.clip(h_current * scale_factor, 0, 1)

        overlaps_new = np.correlate(h_current, 1 - h_current, mode='full') * dx
        new_c5 = np.max(overlaps_new)
