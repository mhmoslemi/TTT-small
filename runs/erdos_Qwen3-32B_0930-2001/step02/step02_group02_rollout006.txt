import numpy as np
from scipy.stats import multivariate_normal
import time

def evaluate(h_vec, dx):
    return np.max(np.correlate(h_vec, 1 - h_vec, mode='full')) * dx

def repair(h_vec, target_sum):
    h_vec = np.clip(h_vec, 0, 1)
    current_sum = np.sum(h_vec)
    if current_sum == 0:
        return h_vec
    scaling = target_sum / current_sum
    h_vec *= scaling
    h_vec = np.clip(h_vec, 0, 1)
    return h_vec

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    num_samples = 100
    elite_proportion = 0.2
    cov_factor = 0.1
    min_cov = 1e-6

    mean = h.copy()
    covariance = cov_factor * np.eye(n_points)
    best_h = h.copy()
    best_c5 = evaluate(best_h, dx)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        samples = multivariate_normal.rvs(mean, covariance, size=num_samples)
        candidates = []
        for sample in samples:
            repaired = repair(sample, target_sum)
            c5 = evaluate(repaired, dx)
            candidates.append((c5, repaired))

        candidates.sort()
        num_elite = max(1, int(num_samples * elite_proportion))
        elite_candidates = candidates[:num_elite]

        elite_h = np.array([c[1] for c in elite_candidates])
        new_mean = np.mean(elite_h, axis=0)
        try:
            new_cov = np.cov(elite_h, rowvar=False) + min_cov * np.eye(n_points)
        except np.linalg.LinAlgError:
            new_cov = covariance

        covariance = new_cov

            best_h[:] = current_best_h[:]
            best_c5 = current_best_c5

    best_h = repair(best_h, target_sum)
    return (best_h, best_c5, n_points)
