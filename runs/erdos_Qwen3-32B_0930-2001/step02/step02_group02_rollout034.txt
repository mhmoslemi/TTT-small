    for i in range(n_particles):
        s = np.sum(positions[i])
        if s != 0.0 and s != n_points:
            positions[i] *= (target_sum / s)
            positions[i] = np.clip(positions[i], 0, 1)

    velocities = np.random.uniform(-0.05, 0.05, size=positions.shape)

    pbest = positions.copy()
    pbest_values = np.zeros(n_particles)
    for i in range(n_particles):
        current_clipped = np.clip(pbest[i], 0, 1)
        s = np.sum(current_clipped)
        if s != 0.0 and s != n_points:
            current_clipped = np.clip(current_clipped * (target_sum / s), 0, 1)
        corr = np.correlate(current_clipped, 1 - current_clipped, mode='full')
        pbest_values[i] = np.max(corr) * dx

    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_values[gbest_idx]

    start_time = time.time()

    while time.time() - start_time < budget_s:
        for i in range(n_particles):
            current = positions[i]
            current_clipped = np.clip(current, 0, 1)
            s = np.sum(current_clipped)
            if s != 0.0 and s != n_points:
                current_clipped = np.clip(current_clipped * (target_sum / s), 0, 1)
            corr = np.correlate(current_clipped, 1 - current_clipped, mode='full')
            current_value = np.max(corr) * dx
            if current_value < pbest_values[i]:
                pbest[i] = current_clipped.copy()
                pbest_values[i] = current_value
                if current_value < gbest_c5:
                    gbest = pbest[i].copy()
                    gbest_c5 = current_value

        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            vel_cognitive = c1 * r1 * (pbest[i] - positions[i])
            vel_social = c2 * r2 * (gbest - positions[i])
            velocity = w * velocities[i] + vel_cognitive + vel_social
            velocity = np.clip(velocity, -max_vel, max_vel)
            velocities[i] = velocity
            positions[i] += velocity

        for i in range(n_particles):
            positions[i] = np.clip(positions[i], 0, 1)
            s = np.sum(positions[i])
            if s != 0.0 and s != n_points:
                positions[i] = np.clip(positions[i] * (target_sum / s), 0, 1)

    return (gbest, gbest_c5, n_points)
