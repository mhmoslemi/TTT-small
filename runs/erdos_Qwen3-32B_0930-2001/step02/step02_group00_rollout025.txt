import numpy as np
import time
import math

def run():
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    T = n_points / 2.0  # sum(h) constraint

    def objective(h):
        correlate = np.correlate(h, 1.0 - h, mode='full')
        return np.max(correlate * dx)

    best_h = h_initial.copy()
    current_h = h_initial.copy()
    best_c5 = objective(best_h)
    current_c5 = best_c5

    start_time = time.time()
    temperature = 10.0
    cooling_rate = 0.995
    block_swap_prob = 0.2
    patience = 500
    no_improvement_steps = 0

    while time.time() - start_time < 1100:
        candidate = current_h.copy()

        # Generate candidate
        if np.random.rand() < block_swap_prob:
            # Block swap/reverse
            i, j = np.random.choice(n_points, 2, replace=False)
            i, j = sorted([i, j])
            candidate[i:j+1] = candidate[i:j+1][::-1]
        else:
            # Two-point perturbation
            i, j = np.random.choice(n_points, 2, replace=False)
            x, y = candidate[i], candidate[j]
            min_delta = max(-x, y - 1.0)
            max_delta = min(1.0 - x, y)
            if min_delta <= max_delta:
                delta = np.random.uniform(min_delta, max_delta)
                candidate[i] += delta
                candidate[j] -= delta

        # Evaluate candidate
        new_c5 = objective(candidate)
        delta_energy = new_c5 - current_c5

        # Acceptance criterion
        if delta_energy < 0 or math.exp(-delta_energy / temperature) > np.random.rand():
            current_h = candidate.copy()
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                no_improvement_steps = 0
            else:
                no_improvement_steps += 1
        else:
            no_improvement_steps += 1

        temperature *= cooling_rate
        if no_improvement_steps > patience:
            break

    return (best_h, best_c5, n_points)
