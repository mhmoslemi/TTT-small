        noise_scale = 0.2 * (1 - frac) + 0.05 * frac
        noise_scale = max(noise_scale, 0.05)
        
        if np.random.rand() < 0.25:
            guess = initial_h_values.copy()
        else:
            guess = h_best.copy()
        
        h0 = generate_initial_guess(guess, noise_scale)
        
        result = optimize.minimize(
            compute_c5,
            x0=h0,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 40, 'ftol': 1e-6}
        )
        
        if result.success:
            c5_candidate = result.fun
            h_candidate = result.x
            if c5_candidate < c5_best:
                h_best[:] = h_candidate.copy()
                c5_best = c5_candidate
    
    return h_best, c5_best, n_points
