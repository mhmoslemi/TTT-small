import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h_current = initial_h_values.copy()
    n_points = len(h_current)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv * dx)

    # Normalize initial solution
    current_sum = h_current.sum()
    if not np.isclose(current_sum, target_sum):
        h_current = (h_current / current_sum) * target_sum
        h_current = np.clip(h_current, 0, 1)

    POPULATION_SIZE = 60
    ELITE_FRACTION = 0.15
    n_elites = int(POPULATION_SIZE * ELITE_FRACTION)
    MEAN_DAMPING = 0.7
    COV_DAMPING = 0.7
    INIT_VAR = 0.03

    mean = h_current.copy()
    cov = INIT_VAR * np.eye(n_points)

    best_h = h_current.copy()
    best_c5 = compute_c5(best_h)

    while time.time() - start_time < budget_s - 50:
        population = []
        for _ in range(POPULATION_SIZE):
            sample = np.random.multivariate_normal(mean, cov)
            sample = np.clip(sample, 0, 1)
            sample_sum = sample.sum()
            if sample_sum == 0:
                repaired = best_h.copy()
            else:
                scale = target_sum / sample_sum
                sample *= scale
                sample = np.clip(sample, 0, 1)
                repaired = sample
            population.append(repaired)
        
        population = np.array(population)
        c5_list = np.array([compute_c5(p) for p in population])
        
        elite_indices = np.argsort(c5_list)[:n_elites]
        elites = population[elite_indices]
        
        new_mean = elites.mean(axis=0)
        new_cov = np.cov(elites.T) + 1e-4 * np.eye(n_points)
        
        mean = MEAN_DAMPING * mean + (1 - MEAN_DAMPING) * new_mean
        cov = COV_DAMPING * cov + (1 - COV_DAMPING) * new_cov

        improved_idx = np.argmin(c5_list)
        if c5_list[improved_idx] < best_c5:
            best_h = population[improved_idx].copy()
            best_c5 = c5_list[improved_idx]
            print(f"New best C5: {best_c5:.6f}")

    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    result = optimize.minimize(compute_c5, best_h, method='SLSQP',
                               bounds=[(0, 1)] * n_points,
                               constraints=constraints,
                               options={'maxiter': 800, 'ftol': 1e-7})

    if result.success:
        final_c5 = compute_c5(result.x)
        if final_c5 < best_c5:
            best_h = result.x
            best_c5 = final_c5

    return best_h, best_c5, n_points
