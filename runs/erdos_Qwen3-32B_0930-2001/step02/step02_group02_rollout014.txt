import numpy as np
import time

def _repair(h):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if current_sum == 0:
        return h_clipped
    target_sum = 0.5 * len(h_clipped)
    scaling_factor = target_sum / current_sum
    h_scaled = h_clipped * scaling_factor
    return np.clip(h_scaled, 0, 1)

def _evaluate(h):
    n = len(h)
    dx = 2.0 / n
    ones_minus_h = 1 - h
    corr = np.correlate(h, ones_minus_h, mode='full')
    return np.max(corr) * dx

def run():
    start_time = time.time()
    h = initial_h_values.copy()
    n_points = len(h)

    swarm_size = 50
    c1 = 2.0
    c2 = 2.0
    w = 0.7
    max_vel = 0.5

    np.random.seed(42)
    swarm = np.random.randn(swarm_size, n_points) * 0.1 + initial_h_values
    for i in range(swarm_size):
        swarm[i] = _repair(swarm[i])
    velocities = np.zeros((swarm_size, n_points))
    particle_best = np.copy(swarm)
    particle_best_scores = np.array([_evaluate(p) for p in particle_best])
    global_best = particle_best[particle_best_scores.argmin()].copy()
    global_best_score = particle_best_scores.min()

    while time.time() - start_time < 1100:
        for i in range(swarm_size):
            current = _repair(swarm[i])
            score = _evaluate(current)
            if score < particle_best_scores[i]:
                particle_best[i] = current.copy()
                particle_best_scores[i] = score
            if score < global_best_score:
                global_best[:] = current[:]
                global_best_score = score

        for i in range(swarm_size):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = w * velocities[i] + c1 * r1 * (particle_best[i] - swarm[i]) + c2 * r2 * (global_best - swarm[i])
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            swarm[i] += velocities[i]
            swarm[i] = _repair(swarm[i])

    final_h = _repair(global_best)
    final_c5 = _evaluate(final_h)
    return (final_h, final_c5, n_points)
