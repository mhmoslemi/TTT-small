        top_count = max(1, int(0.1 * len(idx_terms)))
        top_indices = [idx for term, idx in idx_terms[:top_count]]

        # Adjust h
        for i in top_indices:
            delta = min(h[i], 0.1)
            original = h[i]
            h[i] = max(0.0, original - delta)
            delta_used = original - h[i]
            j = i + k_worst
            if 0 <= j < n_points:
                h[j] = min(1.0, h[j] + delta_used)

        # Project back to sum T
        sum_h = np.sum(h)
        if abs(sum_h - T) > 1e-10:
            scaling = T / sum_h
            h = np.clip(h * scaling, 0, 1)

    return (best_h, best_c5, n_points)
