import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    dx = 2.0 / n_points

    population_size = 100
    population = []
    for _ in range(population_size):
        noisy = np.random.normal(0, 0.15, n_points)

        candidate += (target_sum - np.sum(candidate)) / n_points
        candidate = np.clip(candidate, 0.0, 1.0)
        population.append(candidate)
    
    best_c5 = np.inf
    best_h = np.empty(n_points)
    
    while time.time() - start_time < budget_s - 30:
        # Evaluate population
        scores = [compute_c5(ind) for ind in population]
        elite_idx = np.argsort(scores)[:int(population_size * 0.2)]

            # Mutation

            # Feasibility
