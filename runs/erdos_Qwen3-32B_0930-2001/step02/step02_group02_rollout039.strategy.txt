def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def evaluate(h_vec):
        return np.max(np.correlate(h_vec, 1 - h_vec, mode='full')) * dx

    def repair(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum == 0:
            return h_vec
        scaling = target_sum / current_sum
        h_vec *= scaling
        h_vec = np.clip(h_vec, 0, 1)
        return h_vec

    # Initialize distribution
    mean = h.copy()
    covariance = 0.1 * np.eye(n_points)
    elite = []
    best_h = h.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Sample candidates
        samples = multivariate_normal.rvs(mean, covariance, size=num_samples)
        candidates = []
        for sample in samples:
            repaired = repair(sample)
            c5 = evaluate(repaired)
            candidates.append( (c5, repaired) )

        # Select elite candidates
        candidates.sort()
        elite_candidates = candidates[:int(num_samples * elite_proportion)]
        if not elite_candidates:
            break

        # Update distribution
        elite_h = np.array([c[1] for c in elite_candidates])
        new_mean = np.mean(elite_h, axis=0)
        new_cov = np.cov(elite_h, rowvar=False)

    best_h = repair(best_h)
