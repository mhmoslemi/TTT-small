        return np.max(conv * dx)
    
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    
    def generate_perturbed_guess(guess):
        noise = np.random.normal(0, 0.25, size=n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        current_sum = np.sum(perturbed)
        adjustment = (target_sum - current_sum) / n_points
        perturbed += adjustment
        perturbed = np.clip(perturbed, 0.0, 1.0)
        return perturbed
    
    while time.time() - start_time < budget_s - 10:
        current_time = time.time()
        if current_time - start_time >= budget_s - 10:
            break

        # Generate initial guess
        if np.random.rand() < 0.5 and n_points % 2 == 0:
            # Binary initial guess with exactly target_sum ones
            num_ones = int(target_sum)
            indices = np.random.choice(n_points, size=num_ones, replace=False)
            h0 = np.zeros(n_points)
            h0[indices] = 1.0
        else:
            # Perturb current best
            h0 = generate_perturbed_guess(h_best)
        
        result = optimize.minimize(
            compute_c5,
            x0=h0,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            c5_candidate = result.fun
            h_candidate = result.x
            if c5_candidate < c5_best:
                h_best[:] = h_candidate.copy()
                c5_best = c5_candidate
    
    return h_best, c5_best, n_points
