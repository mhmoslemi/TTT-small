import numpy as np
import time

def compute_overlaps(h, dx):
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_smooth_max(overlaps, smooth_epsilon):
    max_val = np.max(overlaps)
    shifted = overlaps - max_val
    exp_vals = np.exp(shifted / smooth_epsilon)
    sum_exp = np.sum(exp_vals)
    return max_val + smooth_epsilon * np.log(sum_exp)

def compute_smooth_max_gradient(overlaps, smooth_epsilon):
    max_val = np.max(overlaps)
    shifted = overlaps - max_val
    exp_vals = np.exp(shifted / smooth_epsilon)
    sum_exp = np.sum(exp_vals)
    return exp_vals / sum_exp

    n = len(h)
    full_len = len(grad_corr)

    grad = grad_corr[start_idx:start_idx + n]
    return grad
