    positions = initial_h_values + np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    positions = np.clip(positions, 0, 1)

    # Normalize to target sum
    for i in range(n_particles):
        s = np.sum(positions[i])
        if s != 0 and s != n_points:
            positions[i] *= (target_sum / s)
        positions[i] = np.clip(positions[i], 0, 1)

    velocities = np.zeros((n_particles, n_points))
    personal_best_positions = positions.copy()
    personal_best_eval = np.apply_along_axis(
        lambda x: np.max(np.correlate(x, 1 - x, 'full') * dx),
        1, positions)
    global_idx = np.argmin(personal_best_eval)
    global_best_pos = personal_best_positions[global_idx].copy()
    global_best_eval = personal_best_eval[global_idx]

    def evaluate_c5(h_vec):
        return np.max(np.correlate(h_vec, 1 - h_vec, 'full') * dx)

    while time() - start_time < 1100:
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)

            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (personal_best_positions[i] - positions[i]) +
                             c2 * r2 * (global_best_pos - positions[i]))
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)

            positions[i] += velocities[i]
            positions[i] = np.clip(positions[i], 0, 1)

            s = np.sum(positions[i])
            if s != 0 and s != n_points:
                positions[i] *= (target_sum / s)
                positions[i] = np.clip(positions[i], 0, 1)

        current_eval = np.apply_along_axis(evaluate_c5, 1, positions)

        improved = current_eval < personal_best_eval
        personal_best_positions[improved] = positions[improved]
        personal_best_eval[improved] = current_eval[improved]

        current_min_idx = np.argmin(current_eval)
        if current_eval[current_min_idx] < global_best_eval:
            global_best_eval = current_eval[current_min_idx]
            global_best_pos = positions[current_min_idx].copy()

    return (global_best_pos, global_best_eval, n_points)
