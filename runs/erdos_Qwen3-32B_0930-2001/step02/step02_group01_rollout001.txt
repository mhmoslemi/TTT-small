            pass

        # Optimize upper half
        fixed_lower = h_current[lower_indices]
        sum_fixed = fixed_lower.sum()
        required_sum_upper = target_sum - sum_fixed

        variables_upper = h_current[upper_indices]

        def objective_upper(x):
            x_clipped = np.clip(x, 0, 1)
            h_temp = np.concatenate((fixed_lower, x_clipped))
            conv = np.correlate(h_temp, 1 - h_temp, mode='full')
            return np.max(conv * dx)
        
        cons_upper = [{'type': 'eq', 'fun': lambda x: np.sum(x) - required_sum_upper}]
        bounds_upper = [(0, 1) for _ in variables_upper]
        
        try:
            res_upper = optimize.minimize(objective_upper, variables_upper, method='SLSQP', 
                                          bounds=bounds_upper, constraints=cons_upper, 
                                          options={'maxiter': 30})
            if res_upper.success:
                h_current[upper_indices] = np.clip(res_upper.x, 0, 1)
        except:
            pass

        # Post-optimization normalization
        current_sum = h_current.sum()
        h_current *= (target_sum / current_sum)
        h_current = np.clip(h_current, 0, 1)

    # Multi-start with perturbations
    best_c5 = compute_c5(h_current)
    best_h = h_current.copy()

    while time.time() - start_time < budget_s - 50:
        pert = np.random.randn(n_points) * 0.08
        h_pert = np.clip(h_current + pert, 0, 1)
        current_sum = h_pert.sum()
        if current_sum > 0:
            h_pert *= (target_sum / current_sum)
        h_pert = np.clip(h_pert, 0, 1)
        
        try:
            result = optimize.minimize(compute_c5, h_pert, method='SLSQP', 
                                       bounds=[(0, 1)] * n_points,
                                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], 
                                       options={'maxiter': 150})
            if result.success:
                c5_trial = compute_c5(result.x)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = result.x
        except:
            pass
        if time.time() - start_time > budget_s - 50:
            break

    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.1f}s, Best C5: {best_c5:.6f}")
    
    return best_h, best_c5, n_points
