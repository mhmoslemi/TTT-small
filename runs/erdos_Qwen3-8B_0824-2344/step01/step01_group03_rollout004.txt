    # Try different numbers of points
    n_points_list = [100, 200, 300, 400, 500]
    best_c5 = float('inf')
    best_h_vals = None
    best_n_points = 0
    
    start_time = time.time()
    
    # Optimization parameters
    max_iter = 100
    tol = 1e-5
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Resample initial construction if needed
        if n_points_initial != n_points:
            dx_initial = 2.0 / n_points_initial
            x_initial = np.arange(0, 2, dx_initial)
            x_new = np.linspace(0, 2, n_points)
            h_new = interp.interp1d(x_initial, h_initial, kind='linear', fill_value="extrapolate")(x_new)
            h_new = np.clip(h_new, 0, 1)
        else:
            h_new = h_initial
        
        # Define the variables
        x = np.linspace(0, 2, n_points)
        dx = 2.0 / n_points
        sum_h = n_points / 2
        
        # Objective function
        def objective(h):
            h = np.clip(h, 0, 1)
            h = h.reshape(n_points)
            c5 = np.max(correlate(h, 1 - h, mode='full')) * dx
            return c5
        
        # Constraint: sum(h) == n_points / 2
        def constraint_sum(h):
            return np.sum(h) - sum_h
        
        # Bounds
        bounds = Bounds(0, 1)
        
        # Initial guess
        h_initial_guess = h_new.copy()
        
        # Optimization with L-BFGS-B
        result = minimize(objective, h_initial_guess, bounds=bounds, method='L-BFGS-B', 
                          constraints={'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)},

        if result.success:
            h_vals = result.x
            c5_bound = result.fun
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h_vals = h_vals
                best_n_points = n_points
        
        # Early exit if time is up
        if time.time() - start_time > budget_s:
            break
    
    # Return the best solution found
    return (best_h_vals, best_c5, best_n_points)
