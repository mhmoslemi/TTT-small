    n_points = 256  # Optimal for speed and resolution
    dx = 2.0 / n_points
    
    # Initialize h from the provided initial guess and normalize
    h = initial_h_values[:n_points].copy()
    h = h / np.sum(h) * (n_points / 2)
    
    # Convert to a CVXPy variable
    h_var = cp.Variable(n_points, lower=0, upper=1)
    
    # Define the objective function: minimize the maximum overlap
    def evaluate_overlap(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return cp.Max(corr * dx)

    volume_constraint = cp.sum(h_var) == n_points / 2
    
    # Optimization problem
    problem = cp.Problem(cp.Minimize(evaluate_overlap(h_var)), [volume_constraint])
