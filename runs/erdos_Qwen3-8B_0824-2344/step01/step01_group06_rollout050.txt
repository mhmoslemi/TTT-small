def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 256
    dx = 2.0 / n_points
    
    # Generate several initial configurations
    initial_h_values_list = [
        initial_h_values[:n_points].copy(),
        np.random.rand(n_points) * 2 - 1,  # Random values scaled to [0, 1]
        np.kaiser(n_points, 5),  # Kaiser window
        np.zeros(n_points),  # All zeros with normalization
        np.ones(n_points),  # All ones with normalization
        np.zeros(n_points) + 0.5,  # All 0.5
        np.sin(np.linspace(0, 2 * np.pi, n_points)) * 0.5 + 0.5,
        np.cos(np.linspace(0, 2 * np.pi, n_points)) * 0.5 + 0.5
    ]
    
    # Normalize the initial configurations
    initial_h_values_list = [h / np.sum(h) * (n_points / 2) for h in initial_h_values_list]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = n_points
    
    # Evaluate and optimize each initial configuration
    for h in initial_h_values_list:
        h = np.clip(h, 0, 1)
    
        # Define the optimization problem
        h_var = cp.Variable(n_points, name='h')
        constraints = [cp.sum(h_var) == n_points / 2]
        for i in range(n_points):
            constraints.append(0 <= h_var[i])
            constraints.append(h_var[i] <= 1)
        
        # Objective: minimize the maximum value of the correlation
        correlation = np.correlate(h_var.value, 1 - h_var.value, mode="full")
        max_correlation = cp.Max(correlation)
        
        # Setup optimization
        prob = cp.Problem(cp.Minimize(max_correlation), constraints)
        
        # Run the optimization
        start_time = time.time()
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000, verbose=False)
            if prob.status == 'OPTIMAL':
                current_h = h_var.value
                current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode="full")) * dx
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    best_n_points = n_points
        except:
            pass
        
        # Check time limit and break if exceeded
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, best_n_points)
