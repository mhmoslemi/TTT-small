import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
from scipy.optimize import minimize

# Pre-imported functions
# evaluate_erdos_solution() and initial_h_values are assumed to be available

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial construction
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    assert np.isclose(sum_h * dx, 1.0), "Initial h does not satisfy the integral constraint"
    
    # Function to evaluate C5
    def evaluate_c5(h):
        h = np.array(h)
        if len(h) != n_points:
            raise ValueError("h must have length n_points")
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        # Ensure integral constraint
        if not np.isclose(np.sum(h) * dx, 1.0):
            h = h / np.sum(h) * (n_points / 2)
        # Compute cross-correlation
        corr = signal.correlate(h, 1 - h, mode='full')
        # Compute C5
        c5 = np.max(corr * dx)
        return c5
    
    # Function to optimize
    def objective(h):
        return evaluate_c5(h)
    
    # Function to perform optimization with different initial points
    def optimize(initial_guess):
        # Bounds for h
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        # Optimization settings
        options = {
            'disp': False,
            'ftol': 1e-6,
            'maxiter': 1000
        }
        # Perform optimization
        result = minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            options=options
        )
        return result
    
    # Try multiple initializations to avoid local optima
    best_h = initial_h
    best_c5 = evaluate_c5(initial_h)
    
    # First optimization with initial guess
    result = optimize(initial_h)
    if result.fun < best_c5:
        best_h = result.x
        best_c5 = result.fun
    
    # Try random initialization
    random_h = np.random.uniform(0, 1, n_points)
    random_h = random_h / np.sum(random_h) * (n_points / 2)
    result = optimize(random_h)
    if result.fun < best_c5:
        best_h = result.x
        best_c5 = result.fun
    
    # Try a simple "checkerboard" pattern
    checkerboard_h = np.zeros(n_points)
    for i in range(n_points):
        checkerboard_h[i] = 0.5 if i % 2 == 0 else 0.5
    checkerboard_h = checkerboard_h / np.sum(checkerboard_h) * (n_points / 2)
    result = optimize(checkerboard_h)
    if result.fun < best_c5:
        best_h = result.x
        best_c5 = result.fun
    
    # Try a piecewise constant function with more segments
    piecewise_h = np.zeros(n_points)
    for i in range(n_points):

    piecewise_h = piecewise_h / np.sum(piecewise_h) * (n_points / 2)
    result = optimize(piecewise_h)
    if result.fun < best_c5:
        best_h = result.x
        best_c5 = result.fun
    
    # Try reducing the number of points to find a better solution
    for n in range(20, n_points, -1):

        # Initial guess for new n_points
        h_new = np.zeros(n)
        h_new[:n_points] = best_h[:n_points] * (n / n_points)

        h_new = h_new / np.sum(h_new) * (n / 2)
        # Ensure h_new is in [0, 1]
        h_new = np.clip(h_new, 0, 1)
        # Evaluate
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            n_points = n
            dx = 2.0 / n_points
    
    # Final check to see if we can reduce n_points further
    for n in range(10, n_points, -1):
        dx_new = 2.0 / n
        # Initial guess for new n_points
        h_new = np.zeros(n)
        h_new[:n_points] = best_h[:n_points] * (n / n_points)
        # Normalize
        h_new = h_new / np.sum(h_new) * (n / 2)
        # Ensure h_new is in [0, 1]
        h_new = np.clip(h_new, 0, 1)
        # Evaluate
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            n_points = n
            dx = 2.0 / n_points
    
    # Ensure we don't exceed the time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return (best_h.tolist(), best_c5, n_points)
    
    # If we can't reduce n_points further, return the best result
    return (best_h.tolist(), best_c5, n_points)
