import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points, can be increased if needed
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    
    # Define initial constructions
    initial_constructions = [
        lambda n: np.concatenate([np.ones(n // 2), np.zeros(n // 2)]),  # Alternating
        lambda n: np.random.rand(n) * 0.5,  # Randomized
        lambda n: np.random.choice([0, 1], size=n, p=[0.5, 0.5]),  # Binary
        lambda n: np.zeros(n),  # All zeros (used as a base)
        lambda n: np.random.rand(n),  # Random uniform
        lambda n: np.zeros(n) + 0.5  # Uniform
    ]
    
    def compute_c5(h_values):
        h = h_values.copy()
        h_padded = np.zeros(2 * len(h) - 1)
        h_padded[:len(h)] = h
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def objective(x):
        return compute_c5(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    def constraint_low(x):
        return x
    
    def constraint_high(x):
        return 1 - x
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    for init_func in initial_constructions:

        sum_h = np.sum(h)
        if sum_h != n_points / 2:
            h = h * (n_points / 2) / sum_h
        x0 = h.copy()
        
        start_time = time.time()
        
        # Use SLSQP for constrained optimization
        result = optimize.minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[
                {'type': 'eq', 'fun': constraint_sum},
                {'type': 'ineq', 'fun': constraint_low},
                {'type': 'ineq', 'fun': constraint_high}
            ],
            tol=1e-6,
            options={'maxiter': 100, 'ftol': 1e-8}
        )

        else:
            print("Optimization failed for this construction")
    
    # Final check for constraints

        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h, best_c5, n_points)
