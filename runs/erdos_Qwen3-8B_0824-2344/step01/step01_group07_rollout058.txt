# Function to compute C5 for a given h
def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h_expanded = np.zeros(2 * n_points)
    h_expanded[:n_points] = h
    h_expanded[n_points:] = h
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

# Function to evaluate a solution
def evaluate_solution(h_values, n_points):
    return compute_c5(h_values, n_points)

# Function to generate a random initial guess
def generate_initial_guess(n_points):
    h = np.random.rand(n_points)
    total = np.sum(h)
    h = h * (n_points / 2) / total
    return h

# Strategy: Use a combination of random search, gradient descent, and constrained optimization
def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    n_points = len(initial_h_values)

    best_c5 = evaluate_solution(best_h, n_points)

    h = best_h.copy()

    h = np.clip(h, 0.0, 1.0)

        if time.time() - start_time > budget_s:
            break

        x = cp.Variable(n_points)
        # Objective: Minimize max correlation
        obj = cp.Minimize(cp.Max(cp.sum(cp.multiply(x, 1 - x[cp.arange(n_points) + 1]))))
        # Constraints
        constraints = [
            cp.sum(x) == n_points / 2,
            0 <= x,
            x <= 1
        ]
        # Solve the problem
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-3)
            if prob.status == 'optimal':
                h = x.value
                h = np.clip(h, 0.0, 1.0)
                c5 = evaluate_solution(h, n_points)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
        except:
            pass
    
    # Step 3: Use gradient descent with random restarts
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        h = generate_initial_guess(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0.0, 1.0)
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            # Compute gradient using finite differences
            grad = np.zeros(n_points)
            epsilon = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += epsilon
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = evaluate_solution(h_plus, n_points)
                
                h_minus = h.copy()
                h_minus[i] -= epsilon
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                c5_minus = evaluate_solution(h_minus, n_points)
                
                grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
            
            # Update h
            h = h - 0.1 * grad
            h = np.clip(h, 0.0, 1.0)
            h = h * (n_points / 2) / np.sum(h)
            
            c5 = evaluate_solution(h, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
    
    # Step 4: Perturb the best solution to escape local optima
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break

        if c5 < best_c5:
            best_c5 = c5
            best_h = perturbed_h
    
    return (best_h, best_c5, n_points)
