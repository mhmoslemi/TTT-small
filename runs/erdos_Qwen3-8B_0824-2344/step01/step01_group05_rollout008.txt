import scipy.optimize as optimize
import time

def run(seed=42, budget_s=60, **kwargs):

    n_points = 100  # Use 100 points for speed and reasonable resolution
    dx = 2.0 / n_points

    sum_h = np.sum(h)
    if sum_h != n_points / 2:
        h = h * (n_points / 2) / sum_h

        h = h_values.copy()
        # Pad h with zeros to handle full correlation
        h_padded = np.zeros(2 * len(h) - 1)
        h_padded[:len(h)] = h
        # Compute the full correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Multiply by dx and take maximum
        return np.max(corr * dx)
    
    # Objective: minimize C5
    def objective(x):
        return compute_c5(x)
    
    # Constraints: sum(x) == n_points / 2, 0 <= x[i] <= 1
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    def constraint_low(x):
        return x - np.zeros_like(x)
    
    def constraint_high(x):
        return np.ones_like(x) - x

    bounds = [(0, 1) for _ in range(n_points)]

        objective,

            {'type': 'ineq', 'fun': constraint_low},
            {'type': 'ineq', 'fun': constraint_high}

        maxiter=50,
        tol=1e-6,

        disp=False
