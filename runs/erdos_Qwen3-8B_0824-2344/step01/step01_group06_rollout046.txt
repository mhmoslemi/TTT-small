Our strategy will be grounded in an evolutionary algorithm approach, where we iteratively improve a population of candidate solutions through selection, mutation, and crossover. This method is particularly effective for optimization under complex constraints and is known to escape local optima better than gradient-based methods.

### Key Strategy Elements

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points = 256  # Good balance between speed and resolution
    n_half = n_points // 2
    dx = 2.0 / n_points
    
    # Initial population
    population_size = 50
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    
    # Normalize to satisfy volume constraint
    for i in range(population_size):
        population[i] = population[i] / np.sum(population[i]) * (n_points / 2)
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def crossover(parent1, parent2):
        child = np.random.choice([parent1, parent2], size=n_points)
        return child
    
    def mutate(h, mutation_rate=0.05):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] = np.clip(h[i] + np.random.uniform(-0.1, 0.1), 0, 1)
        return h
    
    def select_top(population, num_parents):
        population_with_fitness = [(h, evaluate_c5(h)) for h in population]
        population_with_fitness.sort(key=lambda x: x[1])
        return [h for h, _ in population_with_fitness[:num_parents]]
    
    best_c5 = float('inf')
    best_h = initial_h_values[:n_points].copy()
    
    start_time = time.time()
    
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness and select parents
        population_with_fitness = [(h, evaluate_c5(h)) for h in population]
        population_with_fitness.sort(key=lambda x: x[1])
        parents = [h for h, _ in population_with_fitness[:population_size // 2]]
        
        # Generate new offspring
        new_population = []
        for _ in range(population_size):
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            child = crossover(parent1, parent2)
            child = mutate(child)
            child = child / np.sum(child) * (n_points / 2)
            new_population.append(child)
        
        # Replace old population
        population = new_population
        
        # Update best solution
        for h in population:
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Fine-tune the best solution with local optimization
    def optimize_best_h(h):
        for _ in range(100):
            current_c5 = evaluate_c5(h)
            if current_c5 < 0.3808:
                break
            corr = np.correlate(h, 1 - h, mode="full")
            max_corr_index = np.argmax(corr)
            shift = max_corr_index - n_points + 1
            h[shift:shift + n_points // 2] = np.clip(h[shift:shift + n_points // 2] - 0.01, 0, 1)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
        return h
    
    best_h = optimize_best_h(best_h)
    final_c5 = evaluate_c5(best_h)
    
    return (best_h, final_c5, n_points)
