def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 256
    h = initial_h_values[:n_points].copy()
    h = h / np.sum(h) * (n_points / 2)

    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        return c5

    x = cp.Variable(n_points)
    constraints = [cp.sum(x) == n_points / 2, 0 <= x, x <= 1]
    objective = cp.Minimize(cp.Max(cp.sum(cp.multiply(x, 1 - x[1:])) / (n_points / 2)))
    prob = cp.Problem(objective, constraints)
    start_time = time.time()
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == 'optimal':
            h_opt = x.value
        else:
            h_opt = h
    except:
        h_opt = h

    best_h = h_opt
    best_c5 = evaluate_c5(h_opt)
    return (best_h, best_c5, n_points)
