    n_points = 256
    dx = 2.0 / n_points

        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    def objective(x):
        h = x.copy()
        return evaluate_c5(h)

    def constraint(x):
        h = x.copy()
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint}]
    result = opt.minimize(objective, h, bounds=bounds, constraints=constraints, method='L-BFGS-B', tol=1e-6)
    best_h = result.x
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, n_points)
