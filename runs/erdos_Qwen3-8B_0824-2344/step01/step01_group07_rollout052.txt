# Function to compute C5 for a given h
def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h_expanded = np.zeros(2 * n_points)
    h_expanded[:n_points] = h
    h_expanded[n_points:] = h
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

# Function to evaluate a solution
def evaluate_solution(h_values, n_points):
    return compute_c5(h_values, n_points)
