import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
# evaluate_erdos_solution() is used to compute the overlap integral
# initial_h_values is a starting point for the optimization

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define parameter sweeps
    n_points_list = [45, 60, 75, 90, 105, 120]
    best_h = None
    best_c5 = float('inf')
    best_n_points = 45
    
    for n_points in n_points_list:
        if time.time() - start_time >= budget_s - 5:
            break
        
        # Start with a diverse initial configuration
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 if i % 3 == 0 else 0.2 if i % 3 == 1 else 0.8
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        
        # Define the objective function
        def objective(x):
            x = np.clip(x, 0, 1)
            x = x / np.sum(x) * (n_points / 2)
            corr = np.correlate(x, 1 - x, mode="full")
            c5_val = np.max(corr) * (2.0 / n_points)
            return c5_val
        
        # Optimization with constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        )
        
        # Check if this is the best solution so far
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
            best_n_points = n_points
    
    return (best_h, best_c5, best_n_points)
