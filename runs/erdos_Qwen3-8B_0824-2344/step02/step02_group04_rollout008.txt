import numpy as np
import scipy.optimize as optimize
import time
import random
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points = len(initial_h_values)
    best_h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    
    # Function to normalize h
    def normalize(h):
        return h * (len(h) / 2.0) / np.sum(h)
    
    # Function to evaluate and return the C5 value
    def objective_func(x):
        h = x.copy()
        h = np.clip(h, 0.0, 1.0)
        h = normalize(h)
        return evaluate_erdos_solution(h)
    
    # Random perturbations
    for _ in range(5):
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        h = np.clip(initial_h_values + perturb, 0.0, 1.0)
        h = normalize(h)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Global optimization using differential evolution
    if time.time() - start_time < budget_s * 0.6:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = optimize.differential_evolution(
            objective_func,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            disp=False
        )

            h_values = result.x

            if c5 < best_c5:

                best_c5 = c5
    
    # Local refinement using L-BFGS-B
    if time.time() - start_time < budget_s * 0.8:
        result = optimize.minimize(
            objective_func,
            best_h,
            method='L-BFGS-B',
            options={'maxiter': 100}
        )

            h_values = np.clip(h_values, 0.0, 1.0)

    best_h = normalize(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, len(best_h))
