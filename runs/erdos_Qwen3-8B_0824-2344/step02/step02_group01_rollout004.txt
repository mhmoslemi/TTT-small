def run(seed=42, budget_s=60, **kwargs):

    dx = 2.0 / n_points

    def generate_initial_h(n_points):
        h = np.zeros(n_points)
        h[0:n_points//3] = 1.0

    def compute_c5(h_vals):
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        c5_val = np.max(corr) * dx
        return c5_val
