import numpy as np
import scipy.optimize as optimize
import time
import random
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with initial solution
    n_points = 45
    h = initial_h_values.copy()
    h = h * (n_points / 2.0) / np.sum(h)
    best_h = h
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()
    
    # Try multiple random perturbations of the initial solution
    for _ in range(3):
        perturb = np.random.uniform(-0.2, 0.2, size=len(h))
        h = np.clip(best_h + perturb, 0.0, 1.0)
        h = h * (n_points / 2.0) / np.sum(h)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Try refining with local optimization
    def objective_func(h_flat):
        h = h_flat.reshape(-1)
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2.0) / np.sum(h)
        return evaluate_erdos_solution(h)
    
    result = optimize.minimize(objective_func, best_h, method='L-BFGS-B', options={'maxiter': 100})
    if result.success:
        h_values = result.x
        h_values = np.clip(h_values, 0.0, 1.0)
        h_values = h_values * (n_points / 2.0) / np.sum(h_values)
        c5 = evaluate_erdos_solution(h_values)
        if c5 < best_c5:
            best_h = h_values.copy()
            best_c5 = c5
    
    # Try different n_points (e.g., 60, 90) for better resolution
    for n_points in [60, 90]:
        h_new = np.zeros(n_points)
        if n_points == 60:
            # Interpolate from 45-point to 60-point
            for i in range(n_points):
                h_new[i] = np.mean(best_h[2 * i // 3:2 * i // 3 + 2])
        elif n_points == 90:
            # Interpolate from 45-point to 90-point
            for i in range(n_points):
                h_new[i] = np.mean(best_h[2 * i // 1 :2 * i // 1 + 2])
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        c5 = evaluate_erdos_solution(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
    
    # Try global optimization with differential evolution
    if time.time() - start_time < budget_s * 0.7:
        bounds = [(0.0, 1.0) for _ in range(len(best_h))]
        result = optimize.differential_evolution(objective_func, bounds, strategy='best1bin', maxiter=50, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, disp=False)
        if result.success:
            h_values = result.x
            h_values = np.clip(h_values, 0.0, 1.0)
            h_values = h_values * (len(h_values) / 2.0) / np.sum(h_values)
            c5 = evaluate_erdos_solution(h_values)
            if c5 < best_c5:
                best_h = h_values.copy()
                best_c5 = c5
    
    # Final check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (len(best_h) / 2.0) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, len(best_h))
