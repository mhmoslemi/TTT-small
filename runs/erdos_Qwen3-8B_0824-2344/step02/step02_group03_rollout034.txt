    n_points = 60  # Using the initial_h_values with 60 samples

    def evaluate(individual):
        h_vals = np.array(individual)
        # Normalize to ensure sum(h_vals) == n_points / 2
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        c5_val = np.max(corr) * dx
        return (c5_val,)

    # Initialize toolbox

    toolbox.register("evaluate", evaluate)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0.0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Start genetic algorithm
    population = toolbox.population(n=20)
    h_initial = [float(x) for x in h_initial]
    population.append(creator.Individual(h_initial))

    stats.register("avg", np.mean)
    stats.register("min", np.min)
    stats.register("max", np.max)
    
    # Start timer and run genetic algorithm
    start_time = time.time()
    for gen in range(100):

    return (best_h_values, best_c5, n_points)
