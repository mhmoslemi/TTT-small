import time

from deap import base, creator, tools, algorithms

    start_time = time.time()

    def evaluate_c5(h, n_points):
        h = np.array(h)
        if len(h) != n_points:
            raise ValueError("h must have length n_points")
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        dx = 2.0 / n_points
        # Ensure integral constraint
        if not np.isclose(sum_h * dx, 1.0):
            h = h / sum_h * (n_points / 2)
        # Compute cross-correlation
        corr = signal.correlate(h, 1 - h, mode='full')
        # Compute C5
        c5 = np.max(corr * dx)
        return c5

    def objective(h, n_points):
        return evaluate_c5(h, n_points)

        return creator.Individual(ind)

        n_points = len(ind)
        return (objective(ind, n_points),)

        size = len(ind1)
        # Single-point crossover
        cx_point = np.random.randint(1, size)
        ind1[cx_point:], ind2[cx_point:] = ind2[cx_point:], ind1[cx_point:]
        return ind1, ind2
