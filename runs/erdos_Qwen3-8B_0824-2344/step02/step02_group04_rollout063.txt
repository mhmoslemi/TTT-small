import numpy as np
import scipy.optimize as optimize
import time
import random
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with initial solution
    n_points = 45
    h = initial_h_values.copy()
    h = h * (n_points / 2.0) / np.sum(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    elapsed_time = 0
    
    # First, perform a local optimization with SLSQP to refine the initial solution
    def objective_func(x):
        x = x.reshape(-1)
        x = np.clip(x, 0.0, 1.0)
        x = x * (len(x) / 2.0) / np.sum(x)
        return evaluate_erdos_solution(x)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = optimize.minimize(objective_func, best_h, method='SLSQP', constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}],
                               options={'maxiter': 100})
    if result.success:
        best_h = result.x
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = best_h * (len(best_h) / 2.0) / np.sum(best_h)
        best_c5 = evaluate_erdos_solution(best_h)
        elapsed_time = time.time() - start_time
    
    # Try a larger grid size with fewer points for efficiency
    n_points = 90
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = np.mean(best_h[2 * i:2 * i + 2])
    h = h * (n_points / 2.0) / np.sum(h)
    c5 = evaluate_erdos_solution(h)
    if c5 < best_c5:
        best_h = h.copy()
        best_c5 = c5
    elapsed_time = time.time() - start_time
    
    # If time remains, perform a global optimization with differential evolution
    if elapsed_time < budget_s * 0.7:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = optimize.differential_evolution(objective_func, bounds, strategy='best1bin', maxiter=50, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, disp=False)
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0.0, 1.0)
            best_h = best_h * (len(best_h) / 2.0) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h)
    
    # Final cleanup and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (len(best_h) / 2.0) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, len(best_h))
