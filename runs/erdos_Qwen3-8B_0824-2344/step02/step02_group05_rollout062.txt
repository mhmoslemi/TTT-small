import numpy as np
import scipy.optimize as opt
import time
from scipy.signal import correlate
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)

    # Initial constructions
    initial_h_values = np.array([0.0]*100 + [1.0]*100)
    random_h_values = np.random.rand(100) * 2 - 1
    random_h_values = np.clip(random_h_values, 0, 1)
    random_h_values = random_h_values * (100 / np.sum(random_h_values))
    initial_constructions = [initial_h_values, random_h_values]

    best_c5 = float('inf')
    best_h_vals = None
    best_n_points = 0
    start_time = time.time()

    for n_points in range(100, 501, 100):
        if time.time() - start_time > budget_s:
            break

        for h_initial in initial_constructions:
            if n_points != len(h_initial):
                dx_initial = 2.0 / len(h_initial)
                x_initial = np.arange(0, 2.0, dx_initial)
                x_new = np.linspace(0, 2.0, n_points)
                h_new = np.interp(x_new, x_initial, h_initial)
                h_new = np.clip(h_new, 0, 1)
                h_initial = h_new

            x = np.linspace(0, 2, n_points)
            dx = 2.0 / n_points
            sum_h = n_points / 2

            def objective(h):
                h = np.clip(h, 0, 1)
                c5 = np.max(correlate(h, 1 - h, mode='full')) * dx
                return c5

            def constraint_sum(h):
                return np.sum(h) - sum_h

            bounds = Bounds(0, 1)

            # Try different optimization methods
            try:
                result = minimize(objective, h_initial, bounds=bounds, method='SLSQP',
                                  constraints={'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)},
                                  options={'maxiter': 100, 'ftol': 1e-5, 'disp': False})
                if result.success:
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h_vals = result.x
                        best_n_points = n_points
            except:
                pass

            try:
                result = minimize(objective, h_initial, bounds=bounds, method='COBYLA',
                                  constraints={'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)},
                                  options={'maxiter': 100, 'ftol': 1e-5, 'disp': False})
                if result.success:
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h_vals = result.x
                        best_n_points = n_points
            except:
                pass

            if time.time() - start_time > budget_s:
                break

    return (best_h_vals, best_c5, best_n_points)
