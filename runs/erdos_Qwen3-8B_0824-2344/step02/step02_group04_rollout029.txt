import numpy as np
import scipy.optimize as optimize
import time
import random
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points = 100  # Optimal number of points based on computational budget
    h_values = np.zeros(n_points)
    for i in range(n_points):
        h_values[i] = 0.5  # Start with uniform distribution
    
    # Normalize to satisfy the integral constraint
    h_values = h_values * (n_points / 2.0) / np.sum(h_values)
    
    best_h = h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    
    # Define objective function with constraints
    def objective_func(h_vec):
        h = h_vec.reshape(-1)
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2.0) / np.sum(h)
        return evaluate_erdos_solution(h)
    
    # Local optimization using L-BFGS-B
    result = optimize.minimize(objective_func, best_h, method='L-BFGS-B', options={'maxiter': 100})
    if result.success:
        h_values = result.x
        h_values = np.clip(h_values, 0.0, 1.0)
        h_values = h_values * (n_points / 2.0) / np.sum(h_values)
        c5 = evaluate_erdos_solution(h_values)
        if c5 < best_c5:
            best_h = h_values.copy()
            best_c5 = c5
    
    # Global optimization using differential evolution
    if time.time() - start_time < budget_s * 0.5:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = optimize.differential_evolution(objective_func, bounds, strategy='best1bin', 
                                                maxiter=50, popsize=15, tol=1e-6, mutation=(0.5, 1), 
                                                recombination=0.8, disp=False)
        if result.success:
            h_values = result.x
            h_values = np.clip(h_values, 0.0, 1.0)
            h_values = h_values * (n_points / 2.0) / np.sum(h_values)
            c5 = evaluate_erdos_solution(h_values)
            if c5 < best_c5:
                best_h = h_values.copy()
                best_c5 = c5
    
    # Refinement with simulated annealing
    if time.time() - start_time < budget_s * 0.7:
        result = optimize.basinhopping(objective_func, best_h, niter=50, T=1.0, stepsize=0.1, 
                                       minimizer_kwargs={"method": "L-BFGS-B", "options": {"maxiter": 20}})
        if result.success:
            h_values = result.x
            h_values = np.clip(h_values, 0.0, 1.0)
            h_values = h_values * (n_points / 2.0) / np.sum(h_values)
            c5 = evaluate_erdos_solution(h_values)
            if c5 < best_c5:
                best_h = h_values.copy()
                best_c5 = c5
    
    # Final check and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (n_points / 2.0) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
