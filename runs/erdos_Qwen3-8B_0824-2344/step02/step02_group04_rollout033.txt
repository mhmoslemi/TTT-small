import numpy as np
import scipy.optimize as optimize
import time
import random
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with initial solution
    h = initial_h_values.copy()
    n_points = len(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    
    # Generate a few random perturbations
    for _ in range(5):
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        h = np.clip(best_h + perturb, 0.0, 1.0)
        h = h * (n_points / 2.0) / np.sum(h)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Try coarse grid optimization
    n_points_coarse = int(n_points / 2)
    h_coarse = np.zeros(n_points_coarse)
    for i in range(n_points_coarse):
        h_coarse[i] = np.mean(best_h[2 * i:2 * i + 2])
    h_coarse = h_coarse * (n_points_coarse / 2.0) / np.sum(h_coarse)
    c5_coarse = evaluate_erdos_solution(h_coarse)
    if c5_coarse < best_c5:
        best_h = h_coarse.copy()
        best_c5 = c5_coarse
    
    # Local optimization with L-BFGS-B
    def objective_func(h_vec):
        h = h_vec.reshape(-1)
        h = np.clip(h, 0.0, 1.0)
        h = h * (len(h) / 2.0) / np.sum(h)
        return evaluate_erdos_solution(h)
    
    result = optimize.minimize(objective_func, best_h, method='L-BFGS-B', options={'maxiter': 100})
    if result.success:
        h_values = result.x
        h_values = np.clip(h_values, 0.0, 1.0)
        h_values = h_values * (len(h_values) / 2.0) / np.sum(h_values)
        c5 = evaluate_erdos_solution(h_values)
        if c5 < best_c5:
            best_h = h_values.copy()
            best_c5 = c5
    
    # Use differential evolution for global search
    if time.time() - start_time < budget_s * 0.7:
        bounds = [(0.0, 1.0) for _ in range(len(best_h))]
        result = optimize.differential_evolution(objective_func, bounds, strategy='best1bin', maxiter=50, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, disp=False)
        if result.success:
            h_values = result.x
            h_values = np.clip(h_values, 0.0, 1.0)
            h_values = h_values * (len(h_values) / 2.0) / np.sum(h_values)
            c5 = evaluate_erdos_solution(h_values)
            if c5 < best_c5:
                best_h = h_values.copy()
                best_c5 = c5
    
    # Further refinement with simulated annealing
    if time.time() - start_time < budget_s * 0.8:
        result = optimize.basinhopping(objective_func, best_h, niter=20, T=1.0, stepsize=0.5, minimizer_kwargs={"method": "L-BFGS-B"})
        if result.success:
            h_values = result.x
            h_values = np.clip(h_values, 0.0, 1.0)
            h_values = h_values * (len(h_values) / 2.0) / np.sum(h_values)
            c5 = evaluate_erdos_solution(h_values)
            if c5 < best_c5:
                best_h = h_values.copy()
                best_c5 = c5
    
    # Final check and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (len(best_h) / 2.0) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, len(best_h))
