def run(seed=42, budget_s=60, **kwargs):

        return h_initial

    def objective(h, n_points):
        h = np.array(h)

        h = np.clip(h, 0, 1)
        # Enforce the integral constraint
        sum_h = np.sum(h)
        if not np.isclose(sum_h * (2.0 / n_points), 1.0):
            h = h / sum_h * (n_points / 2)
        # Compute cross-correlation
        corr = signal.correlate(h, 1 - h, mode='full')
        # Compute C5
        c5 = np.max(corr * (2.0 / n_points))
        return c5

    pop_size = 20
    n_points = 40
