import numpy as np
import scipy.optimize as optimize
import time
import random
import cvxpy as cp
import copy

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial solution
    h_values = copy.deepcopy(initial_h_values)
    n_points = len(h_values)
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h_values)
    best_h = h_values
    
    # Function to clip and normalize h_values
    def clip_and_normalize(h):
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        scale = (n_points / 2.0) / total if total > 0 else 1.0
        return h * scale
    
    # Try different initializations with more diversity
    for _ in range(5):
        # Perturb with random noise
        perturb = np.random.uniform(-0.2, 0.2, size=n_points)
        h_values = np.clip(h_values + perturb, 0.0, 1.0)
        h_values = clip_and_normalize(h_values)
        c5 = evaluate_erdos_solution(h_values)
        if c5 < current_c5:
            current_c5 = c5
            best_h = h_values.copy()
    
    # Try a coarser grid for faster optimization
    for n_points in [45, 30, 20, 15, 10]:
        h_values = np.zeros(n_points)
        for i in range(n_points):
            h_values[i] = np.mean(initial_h_values[2*i:2*i+2])
        h_values = clip_and_normalize(h_values)
        c5 = evaluate_erdos_solution(h_values)
        if c5 < current_c5:
            current_c5 = c5
            best_h = h_values.copy()
    
    # Gradient-based optimization with constraint handling
    def objective_func(h_flat):
        h = h_flat.reshape(-1)
        h = clip_and_normalize(h)
        return evaluate_erdos_solution(h)
    
    # Try multiple initializations for better exploration
    for _ in range(3):
        # Initialize with a random perturbation
        h_init = np.random.uniform(0.0, 1.0, size=n_points)
        h_init = clip_and_normalize(h_init)
        result = optimize.minimize(objective_func, h_init, method='L-BFGS-B', options={'maxiter': 100})
        if result.success:
            h_values = result.x
            h_values = clip_and_normalize(h_values)
            c5 = evaluate_erdos_solution(h_values)
            if c5 < current_c5:
                current_c5 = c5
                best_h = h_values.copy()
    
    # Final optimization step with more iterations
    h_values = best_h.copy()
    h_values = clip_and_normalize(h_values)
    result = optimize.minimize(objective_func, h_values, method='L-BFGS-B', options={'maxiter': 150})
    if result.success:
        h_values = result.x
        h_values = clip_and_normalize(h_values)
        c5 = evaluate_erdos_solution(h_values)
        if c5 < current_c5:
            current_c5 = c5
            best_h = h_values.copy()
    
    # Check if time is running out
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s - 2:
        return (best_h, current_c5, len(best_h))
    
    # Try another refinement with different initialization
    h_values = np.random.uniform(0.0, 1.0, size=n_points)
    h_values = clip_and_normalize(h_values)
    result = optimize.minimize(objective_func, h_values, method='L-BFGS-B', options={'maxiter': 150})
    if result.success:
        h_values = result.x
        h_values = clip_and_normalize(h_values)
        c5 = evaluate_erdos_solution(h_values)
        if c5 < current_c5:
            current_c5 = c5
            best_h = h_values.copy()
    
    # Final check and output
    best_h = clip_and_normalize(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    return (best_h, best_c5, len(best_h))
