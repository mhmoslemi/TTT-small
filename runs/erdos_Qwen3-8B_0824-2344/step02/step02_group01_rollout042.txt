import scipy.optimize as optimize
import time
import random

# evaluate_erdos_solution() is used to compute the overlap integral

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 200
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    # Initial population for genetic algorithm
    pop_size = 50
    pop = np.random.uniform(0, 1, (pop_size, n_points))
    for i in range(pop_size):
        pop[i] = pop[i] / np.sum(pop[i]) * (n_points / 2)
        pop[i] = np.clip(pop[i], 0, 1)

    # Genetic algorithm parameters
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 5
    for gen in range(generations):
        if time.time() - start_time >= budget_s - 5:
            break
        
        # Evaluate fitness
        fitness = np.zeros(pop_size)
        for i in range(pop_size):
            h_vals = pop[i]
            corr = np.correlate(h_vals, 1 - h_vals, mode="full")
            c5_val = np.max(corr) * dx
            fitness[i] = c5_val

        # Select elite
        elite_indices = np.argsort(fitness)[:elite_size]
        elite_pop = pop[elite_indices]

        # Generate new population
        new_pop = np.zeros((pop_size, n_points))
        for i in range(pop_size):
            if i < elite_size:
                new_pop[i] = elite_pop[i - elite_size]
            else:
                # Crossover
                parent1 = np.random.choice(elite_indices)
                parent2 = np.random.choice(elite_indices)
                child = np.zeros(n_points)
                for j in range(n_points):
                    if np.random.rand() < crossover_rate:
                        child[j] = pop[parent1][j]
                    else:
                        child[j] = pop[parent2][j]
                # Mutation

                    if np.random.rand() < mutation_rate:
                        child[j] = np.random.uniform(0, 1)
                # Normalize and clip
                child = child / np.sum(child) * (n_points / 2)
                child = np.clip(child, 0, 1)
                new_pop[i] = child

        pop = new_pop

    # Take the best solution from the genetic algorithm and refine it
    best_h = pop[np.argmin(fitness)]
    best_c5 = np.min(fitness)

    # Local refinement with SLSQP
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = optimize.minimize(
        lambda x: np.max(np.correlate(x, 1 - x, mode="full")) * dx,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]

    best_h = result.x
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx

            corr = np.correlate(perturbed_h, 1 - perturbed_h, mode="full")
            c5_val = np.max(corr) * dx
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = perturbed_h

    return (best_h, best_c5, n_points)
