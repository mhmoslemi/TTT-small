import numpy as np
import scipy.optimize as optimize
import time
import random
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Starting from the initial solution
    h_values = initial_h_values.copy()
    n_points = len(h_values)
    best_c5 = evaluate_erdos_solution(h_values)
    best_h = h_values.copy()
    
    # Function to evaluate candidate solution
    def objective_func(h_vec):
        h = h_vec.reshape(-1)
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2.0) / np.sum(h)
        return evaluate_erdos_solution(h)
    
    # Function to evaluate and update best solution
    def update_best(h_vec):
        nonlocal best_h, best_c5
        h = h_vec.reshape(-1)
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2.0) / np.sum(h)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            return True
        return False
    
    # Start timing
    start_time = time.time()
    
    # Perturb initial solution with small random changes
    for _ in range(5):
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        h = np.clip(initial_h_values + perturb, 0.0, 1.0)
        h = h * (n_points / 2.0) / np.sum(h)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Try refining with local optimization
    result = optimize.minimize(objective_func, best_h, method='L-BFGS-B', options={'maxiter': 100})
    if result.success:
        h_values = result.x
        h_values = np.clip(h_values, 0.0, 1.0)
        h_values = h_values * (n_points / 2.0) / np.sum(h_values)
        c5 = evaluate_erdos_solution(h_values)
        if c5 < best_c5:
            best_h = h_values.copy()
            best_c5 = c5
    
    # Try global optimization with differential evolution
    if time.time() - start_time < budget_s * 0.7:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = optimize.differential_evolution(objective_func, bounds, strategy='best1bin', maxiter=50, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, disp=False)
        if result.success:
            h_values = result.x
            h_values = np.clip(h_values, 0.0, 1.0)
            h_values = h_values * (n_points / 2.0) / np.sum(h_values)
            c5 = evaluate_erdos_solution(h_values)
            if c5 < best_c5:
                best_h = h_values.copy()
                best_c5 = c5
    
    # Try refining with random restarts
    for _ in range(3):
        if time.time() - start_time >= budget_s:
            break
        # Generate a new random solution
        h = np.random.uniform(0.0, 1.0, size=n_points)
        h = h * (n_points / 2.0) / np.sum(h)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        # Local optimization from this point
        result = optimize.minimize(objective_func, h, method='L-BFGS-B', options={'maxiter': 50})
        if result.success:
            h_values = result.x
            h_values = np.clip(h_values, 0.0, 1.0)
            h_values = h_values * (n_points / 2.0) / np.sum(h_values)
            c5 = evaluate_erdos_solution(h_values)
            if c5 < best_c5:
                best_h = h_values.copy()
                best_c5 = c5
    
    # Final check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (n_points / 2.0) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
