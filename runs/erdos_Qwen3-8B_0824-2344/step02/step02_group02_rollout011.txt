def run(seed=42, budget_s=60, **kwargs):

    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 45

    n_points = 50
    dx = 2.0 / n_points
    pop_size = 50
    n_gen = 100
    cx_prob = 0.8
    mut_prob = 0.1
    mut_rate = 0.1
    
    # Define the objective function
    def objective(h_values):
        h = np.array(h_values)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if not np.isclose(sum_h * dx, 1.0):
            h = h / sum_h * (n_points / 2)
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr * dx)
        return c5
    
    # Create the genetic algorithm
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    # Initialize the toolbox
    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0.0, 1.0)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    
    # Define the genetic operators
    toolbox.register("evaluate", objective)
    toolbox.register("mate", tools.cxTwoPoint)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=mut_rate, indpb=mut_prob)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Initialize the population
    pop = toolbox.population(n=pop_size)
    
    # Run the genetic algorithm
    for gen in range(n_gen):

        # Check for early stopping

    best_ind = min(pop, key=lambda x: x.fitness.values[0])
