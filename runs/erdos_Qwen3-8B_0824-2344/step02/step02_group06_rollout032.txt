```python
import numpy as np
import scipy.optimize as optimize
import time
import random
import cvxpy as cp
import copy

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 45
    h_values = np.zeros(n_points)
    for i in range(n_points):
        h_values[i] = np.mean(initial_h_values[2*i:2*i+2])
    h_values = np.clip(h_values, 0.0, 1.0)
    total = np.sum(h_values)
    scale = (n_points / 2.0) / total if total > 0 else 1.0
    h_values = h_values * scale
    best_h = h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    elapsed_time = time.time() - start_time
    
    def objective_func(h):
        h = h.reshape(-1)
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        scale = (n_points / 2.0) / total if total > 0 else 1.0
        h = h * scale
        return evaluate_erdos_solution(h)
    
    # Try several different n_points
    for n_points in [45, 30, 20, 15, 10]:
        h_values = np.zeros(n_points)
        for i in range(n_points):
            h_values[i] = np.mean(initial_h_values[2*i:2*i+2])
        h_values = np.clip(h_values, 0.0, 1.0)
        total = np.sum(h_values)
        scale = (n_points / 2.0) / total if total > 0 else 1.0
        h_values = h_values * scale
        c5 = evaluate_erdos_solution(h_values)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_values.copy()
        if elapsed_time >= budget_s - 5:
            break
    
    # Optimization with L-BFGS-B
    for _ in range(3):
        h_init = np.random.uniform(0.0, 1.0, size=n_points)
        h_init = np.clip(h_init, 0.0, 1.0)
        total = np.sum(h_init)
        scale = (n_points / 2.0) / total if total > 0 else 1.0
        h_init = h_init * scale
        result = optimize.minimize(objective_func, h_init, method='L-BFGS-B', options={'maxiter': 100})
        if result.success:
            h_values = result.x
            h_values = np.clip(h_values, 0.0, 1.0)
            total = np.sum(h_values)
            scale = (n_points / 2.0) / total if total > 0 else 1.0
            h_values = h_values * scale
            c5 = evaluate_erdos_solution(h_values)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_values.copy()
        if elapsed_time >= budget_s - 5:
            break
    
    # Try another refinement
    h_values = best_h.copy()
    h_values = np.clip(h_values, 0.0, 1.0)
    total = np.sum(h_values)
    scale = (n_points / 2.0) / total if total > 0 else 1.0
    h_values = h_values * scale
    result = optimize.minimize(objective_func, h_values, method='L-BFGS-B', options={'maxiter': 100})
    if result.success:
        h_values = result.x
        h_values = np.clip(h_values, 0.0, 1.0)
        total = np.sum(h_values)
        scale = (n_points / 2.0) / total if total > 0 else 1.0
        h_values = h_values * scale
        c5 = evaluate_erdos_solution(h_values)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_values.copy()
    
    # Final check
    if elapsed_time < budget_s - 2:
        # Try a fine-tuning heuristic
        for i in range(n_points):
            perturb = np.random.uniform(-0.05, 0.05, size=n_points)
            h_values = best_h + perturb
            h_values = np.clip(h_values, 0.0, 1.0)
            total = np.sum(h_values)
            scale = (n_points / 2.0) / total if total > 0 else 1.0
            h_values = h_values * scale
            c5 = evaluate_erdos_solution(h_values)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_values.copy()
    
    return (best_h, best_c5, len(best_h))
```
