import scipy.optimize as opt
import scipy.signal as signal
import time
from deap import algorithms, base, creator, tools

# Pre-imported functions

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 40

    def evaluate_c5(h, n_points):
        h = np.array(h)
        if len(h) != n_points:
            raise ValueError("h must have length n_points")
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        dx = 2.0 / n_points
        # Ensure integral constraint
        if not np.isclose(sum_h * dx, 1.0):
            h = h / sum_h * (n_points / 2)
        # Compute cross-correlation
        corr = signal.correlate(h, 1 - h, mode='full')
        # Compute C5
        c5 = np.max(corr * dx)
        return c5

    def objective(h, n_points):
        return evaluate_c5(h, n_points)

    def optimize_with_bounds(h_initial, n_points):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [
            {
                'type': 'eq',
                'fun': lambda h: np.sum(h) * (2.0 / n_points) - 1.0
            }
        ]
        result = opt.minimize(
            lambda h: objective(h, n_points),
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={
                'ftol': 1e-9,
                'maxiter': 1000
            }
        )
        return result.x
