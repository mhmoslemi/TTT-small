import scipy.signal as signal
import time
from scipy.optimize import minimize

# Pre-imported functions

def run(seed=42, budget_s=60, **kwargs):

    best_h = None
    best_n_points = 45
    
    # Try a range of n_points, starting from the initial one

        initial_guesses = []

        if hasattr(initial_h_values, "__len__") and len(initial_h_values) <= n_points:
            structured_guess = np.zeros(n_points)

            structured_guess = structured_guess / np.sum(structured_guess) * (n_points / 2)
            structured_guess = np.clip(structured_guess, 0, 1)
            initial_guesses.append(structured_guess)
        # Random initial guess
        random_guess = np.random.uniform(0, 1, n_points)
        random_guess = random_guess / np.sum(random_guess) * (n_points / 2)

        initial_guesses.append(random_guess)
        
        # Use a combination of initial guesses to avoid local optima
        for h_initial in initial_guesses:

            def objective(h):

                if not np.isclose(sum_h * dx, 1.0):
                    h = h / sum_h * (n_points / 2)
                # Compute cross-correlation
                corr = signal.correlate(h, 1 - h, mode='full')
                # Compute C5
                c5 = np.max(corr * dx)
                return c5
