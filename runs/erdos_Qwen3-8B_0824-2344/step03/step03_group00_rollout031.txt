def evaluate_individual(h, n_points, dx):
    h = np.array(h)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if not np.isclose(sum_h * dx, 1.0):
        h = h / sum_h * (n_points / 2)
    corr = signal.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr * dx)
    return c5

def select_parents(population, fitness):
    # Tournament selection
    indices = np.random.choice(len(population), size=3, replace=False)
    best_index = np.argmin(fitness[indices])
    parent1 = population[indices[best_index]]
    indices = np.random.choice(len(population), size=3, replace=False)
    best_index = np.argmin(fitness[indices])
    parent2 = population[indices[best_index]]
    return parent1, parent2

def crossover(parent1, parent2):
    # Blend crossover
    alpha = np.random.rand()
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(child, n_points):
    # Bounded mutation
    for i in range(n_points):
        if np.random.rand() < 0.1:
            child[i] += np.random.uniform(-0.1, 0.1)
            child[i] = np.clip(child[i], 0, 1)
    return child

def clip_and_normalize(h, n_points, dx):
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if not np.isclose(sum_h * dx, 1.0):
        h = h / sum_h * (n_points / 2)
    return h
