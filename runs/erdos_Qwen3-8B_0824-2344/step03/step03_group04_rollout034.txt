import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

# Pre-imported functions
# evaluate_erdos_solution() and initial_h_values are assumed to be available

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 40
    
    def evaluate_c5(h, n_points):
        h = np.array(h)
        if len(h) != n_points:
            raise ValueError("h must have length n_points")
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        dx = 2.0 / n_points
        # Ensure integral constraint
        if not np.isclose(sum_h * dx, 1.0):
            h = h / sum_h * (n_points / 2)
        # Compute cross-correlation
        corr = signal.correlate(h, 1 - h, mode='full')
        # Compute C5
        c5 = np.max(corr * dx)
        return c5
    
    def objective(h, n_points):
        return evaluate_c5(h, n_points)
    
    def optimize_with_bounds(h_initial, n_points):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [
            {
                'type': 'eq',
                'fun': lambda h: np.sum(h) * (2.0 / n_points) - 1.0
            }
        ]
        result = opt.minimize(
            lambda h: objective(h, n_points),
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={
                'ftol': 1e-9,
                'maxiter': 1500,
                'disp': False
            }
        )
        return result.x
    
    # Start with the initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    h_opt = optimize_with_bounds(initial_h, n_points)
    current_c5 = evaluate_c5(h_opt, n_points)
    if current_c5 < best_c5:
        best_h = h_opt
        best_c5 = current_c5
        best_n_points = n_points
    
    # Try perturbed versions of the initial guess
    for n in [40, 35, 30, 25, 20]:
        dx = 2.0 / n
        h_new = np.array(initial_h_values)
        h_new = h_new[:n]
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n / 2)
        h_new += np.random.uniform(-0.1, 0.1, n)
        h_new = np.clip(h_new, 0, 1)
        h_opt = optimize_with_bounds(h_new, n)
        current_c5 = evaluate_c5(h_opt, n)
        if current_c5 < best_c5:
            best_h = h_opt
            best_c5 = current_c5
            best_n_points = n
    
    # Try a uniform distribution initialization
    for n in [40, 35, 30, 25, 20]:
        dx = 2.0 / n
        h_new = np.full(n, 0.5)
        h_new = h_new / np.sum(h_new) * (n / 2)
        h_new = np.clip(h_new, 0, 1)
        h_opt = optimize_with_bounds(h_new, n)
        current_c5 = evaluate_c5(h_opt, n)
        if current_c5 < best_c5:
            best_h = h_opt
            best_c5 = current_c5
            best_n_points = n
    
    # Try random initialization for smaller n
    for n in [20, 15, 10]:
        h_new = np.random.uniform(0, 1, n)
        h_new = h_new / np.sum(h_new) * (n / 2)
        h_new = np.clip(h_new, 0, 1)
        h_opt = optimize_with_bounds(h_new, n)
        current_c5 = evaluate_c5(h_opt, n)
        if current_c5 < best_c5:
            best_h = h_opt
            best_c5 = current_c5
            best_n_points = n
    
    # Try local search around the best found so far
    if best_n_points >= 20:
        h_local = best_h.copy()
        for _ in range(5):
            perturbation = np.random.uniform(-0.1, 0.1, best_n_points)
            h_perturbed = np.clip(h_local + perturbation, 0, 1)
            h_perturbed = h_perturbed / np.sum(h_perturbed) * (best_n_points / 2)
            h_opt = optimize_with_bounds(h_perturbed, best_n_points)
            current_c5 = evaluate_c5(h_opt, best_n_points)
            if current_c5 < best_c5:
                best_h = h_opt
                best_c5 = current_c5
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return (best_h.tolist(), best_c5, best_n_points)
    
    return (best_h.tolist(), best_c5, best_n_points)
