import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
from scipy.optimize import minimize

# Pre-imported functions
# evaluate_erdos_solution() and initial_h_values are assumed to be available

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 41
    
    n_points = best_n_points
    dx = 2.0 / n_points

    # Initial guess
    if hasattr(initial_h_values, "__len__") and len(initial_h_values) == n_points:
        h_initial = np.array(initial_h_values, dtype=np.float64)
    else:
        h_initial = np.random.uniform(0, 1, n_points)
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        h_initial = np.clip(h_initial, 0, 1)

    # Define the objective function
    def objective(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if not np.isclose(sum_h * dx, 1.0):
            h = h / sum_h * (n_points / 2)
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr * dx)
        return c5

    # Optimization settings
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': lambda h: np.sum(h) * dx - 1.0}
    ]
    
    # Optimization
    result = minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'disp': False, 'ftol': 1e-8, 'maxiter': 200}
    )
    
    if result.success:

            best_h = result.x
            best_n_points = n_points
    
    # Early exit if time budget is exceeded

        pass
    
    return (best_h.tolist(), best_c5, best_n_points)
