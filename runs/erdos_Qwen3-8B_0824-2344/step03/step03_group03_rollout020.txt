    def compute_c5(h, n_points):

        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        h_normalized = h * (n_points / 2) / sum_h
        conv = np.correlate(h_normalized, 1 - h_normalized, mode="full")
        max_overlap = np.max(conv)
        return max_overlap * (2.0 / n_points)

    def objective(h, n_points):
        return compute_c5(h, n_points)

    def constraint(h, n_points):
        return np.sum(h) - n_points / 2  # Ensure the integral is 1
