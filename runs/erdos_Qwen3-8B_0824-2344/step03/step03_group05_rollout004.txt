def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points_list = [45, 60, 75]

            break

        h = np.random.uniform(0, 1, n_points)

        def compute_c5(h_vals):

            corr = np.correlate(h_vals, 1 - h_vals, mode="full")
            c5_val = np.max(corr) * (2.0 / n_points)
            return c5_val
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = optimize.minimize(
            compute_c5,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        )
