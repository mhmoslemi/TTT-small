        h = np.clip(h, 0, 1)

        # Compute C5
        c5 = np.max(corr * dx)

    ]

    result = minimize(

        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,

    for n_points in range(41, 101):
