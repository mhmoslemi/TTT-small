import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

# Pre-imported functions
# evaluate_erdos_solution() is used to compute the overlap integral
# initial_h_values is a starting point for the optimization

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different grid sizes and starting points
    n_points_list = [75, 90, 105, 120]  # Focus on larger grids to get better approximations
    best_c5 = float('inf')
    best_h = None
    best_n_points = 75
    
    # Function to evaluate the overlap integral given h
    def evaluate_overlap(h_vals, dx):
        # Ensure sum(h_vals) is exactly n_points / 2
        h_vals = h_vals / np.sum(h_vals) * (len(h_vals) / 2)
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        return np.max(corr) * dx
    
    # Define bounds for each h[i]
    def bounds_func(x):
        return np.clip(x, 0, 1)
    
    # Optimization settings
    max_iter = 100
    ftol = 1e-8
    perturb_count = 3
    perturb_strength = 0.1
    
    for n_points in n_points_list:
        if time.time() - start_time >= budget_s - 5:
            break
        
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Initialize h_vals
        if n_points == len(initial_h_values):
            h_initial = np.copy(initial_h_values)
        else:
            h_initial = np.random.rand(n_points)
        
        # Set up bounds and constraints
        bounds = Bounds(0, 1)
        constraint_sum = lambda h_vals: np.sum(h_vals) - n_points / 2
        
        # Define the objective function
        def objective(h_vals):
            return evaluate_overlap(h_vals, dx)
        
        # Use a local optimizer with multiple iterations and perturbations
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': max_iter, 'ftol': ftol}
        )
        
        # Evaluate the best solution
        h_vals = result.x
        c5_val = evaluate_overlap(h_vals, dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_vals
            best_n_points = n_points
        
        # Perturb the solution for further exploration
        for _ in range(perturb_count):
            perturb = np.random.uniform(-perturb_strength, perturb_strength, size=n_points)
            perturb = np.clip(perturb, -0.5, 0.5)
            perturb_h = h_vals + perturb
            perturb_h = np.clip(perturb_h, 0, 1)
            perturb_h = perturb_h / np.sum(perturb_h) * (n_points / 2)
            c5_val = evaluate_overlap(perturb_h, dx)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = perturb_h
                best_n_points = n_points
    
    return (best_h, best_c5, best_n_points)
