import numpy as np
import scipy.signal as signal
import time
from scipy.optimize import differential_evolution

# Pre-imported functions
# evaluate_erdos_solution() and initial_h_values are assumed to be available

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 45

    # Try a range of n_points, starting from the initial one
    for n_points in range(20, 101):
        dx = 2.0 / n_points

        # Generate diverse initial guesses
        initial_population = []
        # Use initial_h_values if available
        if hasattr(initial_h_values, "__len__") and len(initial_h_values) <= n_points:
            initial_guess = np.zeros(n_points)
            initial_guess[:len(initial_h_values)] = initial_h_values
            initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
            initial_guess = np.clip(initial_guess, 0, 1)
            initial_population.append(initial_guess)
        # Add random initial guess
        random_guess = np.random.uniform(0, 1, n_points)
        random_guess = random_guess / np.sum(random_guess) * (n_points / 2)
        random_guess = np.clip(random_guess, 0, 1)
        initial_population.append(random_guess)
        # Add some structured guesses for diversity
        for i in range(3):
            structured_guess = np.zeros(n_points)
            structured_guess[i::4] = 1.0
            structured_guess = structured_guess / np.sum(structured_guess) * (n_points / 2)
            structured_guess = np.clip(structured_guess, 0, 1)
            initial_population.append(structured_guess)

        # Genetic algorithm setup
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        def objective(h):

            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if not np.isclose(sum_h * dx, 1.0):
                h = h / sum_h * (n_points / 2)
            corr = signal.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr * dx)
            return c5

        result = differential_evolution(
            objective,
            bounds,
            popsize=15,

            break

    return (best_h.tolist(), best_c5, best_n_points)
