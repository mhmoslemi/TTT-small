import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

# Pre-imported functions
# evaluate_erdos_solution() and initial_h_values are assumed to be available

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 45
    n_points = 60  # Starting with 60 points
    
    # Define genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Function to evaluate and normalize a candidate solution
    def evaluate_and_normalize(h):
        h_norm = h / np.sum(h) * (n_points / 2)
        h_norm = np.clip(h_norm, 0, 1)
        return h_norm
    
    # Function to compute C5
    def compute_c5(h):
        dx = 2.0 / n_points
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr * dx)
        return c5
    
    # Initial population
    population = []
    if hasattr(initial_h_values, "__len__") and len(initial_h_values) <= n_points:
        initial_guess = np.zeros(n_points)
        initial_guess[:len(initial_h_values)] = initial_h_values
        initial_guess = evaluate_and_normalize(initial_guess)
        population.append(initial_guess)
    # Add random initial guess
    random_guess = np.random.uniform(0, 1, n_points)
    random_guess = evaluate_and_normalize(random_guess)
    population.append(random_guess)
    
    # Fill the rest of the population with random guesses
    for _ in range(population_size - 2):
        guess = np.random.uniform(0, 1, n_points)
        guess = evaluate_and_normalize(guess)
        population.append(guess)
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate all candidates
        fitness = [compute_c5(individual) for individual in population]
        best_index = np.argmin(fitness)
        best_h_current = population[best_index]
        best_c5_current = fitness[best_index]
        
        # Update best solution if needed
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_h_current.copy()
            best_n_points = n_points
        
        # Selection: choose top half of the population
        selected_indices = np.argsort(fitness)[:population_size // 2]
        selected = [population[i] for i in selected_indices]
        
        # Crossover

        while len(new_population) < population_size:

            # Normalize and clip
            child = evaluate_and_normalize(child)
            new_population.append(child)
        
        population = new_population
    
    return (best_h.tolist(), best_c5, best_n_points)
