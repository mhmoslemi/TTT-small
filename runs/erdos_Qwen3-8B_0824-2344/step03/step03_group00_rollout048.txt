import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
from scipy.optimize import minimize

# Pre-imported functions
# evaluate_erdos_solution() and initial_h_values are assumed to be available

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 45
    
    # Function to evaluate the overlap integral
    def evaluate_c5(h, dx):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        # Ensure integral constraint is satisfied
        sum_h = np.sum(h)
        if not np.isclose(sum_h * dx, 1.0):
            h = h / sum_h * (n_points / 2)
        # Compute cross-correlation
        corr = signal.correlate(h, 1 - h, mode='full')
        # Compute C5
        c5 = np.max(corr * dx)
        return c5
    
    # Define a function to generate a smooth initial guess based on initial_h_values
    def get_initial_guess(n_points):
        if n_points <= len(initial_h_values):
            h_initial = np.zeros(n_points)
            h_initial[:n_points] = initial_h_values[:n_points]
        else:
            h_initial = np.zeros(n_points)
            h_initial[:len(initial_h_values)] = initial_h_values
            # Fill the rest with random values
            h_initial[len(initial_h_values):] = np.random.uniform(0, 1, n_points - len(initial_h_values))
        # Normalize
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        h_initial = np.clip(h_initial, 0, 1)
        # Add jitter
        h_initial += np.random.uniform(-0.05, 0.05, size=n_points)
        h_initial = np.clip(h_initial, 0, 1)
        return h_initial
    
    # Try a range of n_points
    for n_points in range(35, 101):
        dx = 2.0 / n_points
        
        # Generate multiple initial guesses
        initial_guesses = []
        for _ in range(3):
            h_initial = get_initial_guess(n_points)
            initial_guesses.append(h_initial)
        
        for h_initial in initial_guesses:
            # Define the objective function
            def objective(h):
                return evaluate_c5(h, dx)
            
            # Optimization settings
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            
            # Optimization
            result = minimize(
                objective,
                h_initial,
                method='L-BFGS-B',
                bounds=bounds,
                options={'disp': False, 'ftol': 1e-8, 'maxiter': 150}
            )
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
                    best_n_points = n_points
        
        # Early exit if time budget is exceeded
        if time.time() - start_time > budget_s - 1:
            break
    
    return (best_h.tolist(), best_c5, best_n_points)
