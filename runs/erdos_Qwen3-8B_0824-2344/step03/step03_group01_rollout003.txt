import numpy as np
import scipy.signal as signal
import time
from deap import base, creator, tools, algorithms

# Pre-imported functions
# evaluate_erdos_solution() and initial_h_values are assumed to be available

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 45
    
    # Set up genetic algorithm
    n_points = 60  # Start with 60 points for better resolution
    dx = 2.0 / n_points
    
    # Define the problem
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    # Initialize toolbox
    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0.0, 1.0)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    
    # Define the objective function
    def evaluate(individual):
        h = np.array(individual)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if not np.isclose(sum_h * dx, 1.0):
            h = h / sum_h * (n_points / 2)
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr * dx)
        return (c5,)
    
    toolbox.register("evaluate", evaluate)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0.0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Initialize population
    pop = toolbox.population(n=50)
    
    # Genetic algorithm parameters
    generations = 100
    cxpb = 0.5
    mutpb = 0.1
    
    # Run genetic algorithm
    for gen in range(generations):
        # Evaluate the entire population
        offspring = algorithms.varAnd(pop, toolbox, cxpb, mutpb)
        fits = toolbox.map(toolbox.evaluate, offspring)
        # Select the best individuals
        fits = list(zip(fits, offspring))
        fits.sort()
        fits = fits[:len(pop)]
        # Replace population
        pop = [ind for _, ind in fits]
        # Update best solution
        for ind in pop:
            if ind.fitness.values[0] < best_c5:
                best_c5 = ind.fitness.values[0]
                best_h = np.array(ind)
                best_n_points = n_points
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Return the best solution found
    return (best_h.tolist(), best_c5, best_n_points)
