import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
from scipy.optimize import minimize

# Pre-imported functions
# evaluate_erdos_solution() and initial_h_values are assumed to be available

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 45
    
    # Start from the initial construction and consider nearby n_points
    initial_n_points = len(initial_h_values)
    n_points_range = range(40, 101)
    
    for n_points in n_points_range:
        dx = 2.0 / n_points
        
        # Generate initial guess based on initial_h_values or uniform distribution
        if n_points >= initial_n_points:
            h_initial = np.zeros(n_points)
            h_initial[:initial_n_points] = initial_h_values
            # Normalize and add jitter
            h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        else:
            h_initial = np.random.rand(n_points)
            h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        
        # Add some jitter to avoid local optima
        h_initial += np.random.uniform(-0.05, 0.05, size=n_points)
        h_initial = np.clip(h_initial, 0, 1)
        
        # Define the objective function
        def objective(h):
            h = np.array(h)
            # Ensure h is in [0, 1] and satisfies the integral constraint
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if not np.isclose(sum_h * dx, 1.0):
                h = h / sum_h * (n_points / 2)
            # Compute cross-correlation
            corr = signal.correlate(h, 1 - h, mode='full')
            # Compute C5
            c5 = np.max(corr * dx)
            return c5
        
        # Optimization settings
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [
            {'type': 'eq', 'fun': lambda h: np.sum(h) * dx - 1.0}
        ]
        
        # Optimization
        result = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'disp': False, 'ftol': 1e-8, 'maxiter': 150}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                best_n_points = n_points
        
        # Early exit if time budget is exceeded
        if time.time() - start_time > budget_s - 1:
            break
    
    return (best_h.tolist(), best_c5, best_n_points)
