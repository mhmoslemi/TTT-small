```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

# Pre-imported functions
# evaluate_erdos_solution() and initial_h_values are assumed to be available

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 45
    n_points_max = 100
    
    def evaluate(h_values, n_points):
        dx = 2.0 / n_points
        h = np.array(h_values)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if not np.isclose(sum_h * dx, 1.0, atol=1e-6):
            h = h / sum_h * (n_points / 2)
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr * dx)
        return c5
    
    def optimize(n_points):
        dx = 2.0 / n_points
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}]
        
        result = minimize(
            lambda x: evaluate(x, n_points),
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        if result.success:
            return result.x, result.fun
        return h, float('inf')
    
    for n_points in range(40, n_points_max + 1):
        if time.time() - start_time > budget_s - 1:
            break
        
        h_initial = np.zeros(n_points)
        if hasattr(initial_h_values, "__len__") and len(initial_h_values) <= n_points:
            h_initial[:len(initial_h_values)] = initial_h_values
            h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
            h_initial = np.clip(h_initial, 0, 1)
        
        h, c5 = optimize(n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            best_n_points = n_points
    
    return (best_h.tolist(), best_c5, best_n_points)
```
