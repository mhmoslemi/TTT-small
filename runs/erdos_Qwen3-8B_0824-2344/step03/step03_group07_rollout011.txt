import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
# evaluate_erdos_solution() is used to compute the overlap integral
# initial_h_values is a starting point for the optimization

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different starting points
    n_points_list = [90, 105, 120, 135, 150]  # Try different grid sizes
    best_c5 = float('inf')
    best_h = None

    for n_points in n_points_list:
        if time.time() - start_time >= budget_s - 5:
            break
        
        # Define the grid
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Generate initial points: use the given initial_h_values if available, otherwise random
        if n_points == len(initial_h_values):
            h_initial = np.copy(initial_h_values)
        else:
            h_initial = np.random.rand(n_points)
        
        # Define the objective function for optimization
        def objective(h_vals):
            # Normalize to ensure sum(h_vals) == n_points / 2
            h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
            h_vals = np.clip(h_vals, 0, 1)
            # Compute the overlap integral
            corr = np.correlate(h_vals, 1 - h_vals, mode="full")
            c5_val = np.max(corr) * dx
            return c5_val
        
        # Define the constraints
        def constraint_sum(h_vals):
            return np.sum(h_vals) - n_points / 2
        
        # Define bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use a local optimizer to refine the solution
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        # Evaluate the best solution
        h_vals = result.x
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        c5_val = np.max(corr) * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_vals
        
        # Perturb the solution for further exploration
        for _ in range(3):
            perturb = np.random.uniform(-0.1, 0.1, size=n_points)
            perturb = np.clip(perturb, -0.5, 0.5)
            perturb_h = h_vals + perturb
            perturb_h = np.clip(perturb_h, 0, 1)
            perturb_h = perturb_h / np.sum(perturb_h) * (n_points / 2)
            corr = np.correlate(perturb_h, 1 - perturb_h, mode="full")
            c5_val = np.max(corr) * dx
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = perturb_h
    
    return (best_h, best_c5, len(best_h))
