        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        c5_val = np.max(corr) * dx
        return c5_val

    def fitness(individual):
        return -evaluate(individual)  # Minimize C₅
