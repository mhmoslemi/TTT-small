import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Pre-defined initial solution from initial_h_values
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Objective: Minimize the maximum overlap integral C5
    
    def objective_function(h):
        # Normalize h to ensure the integral is 1
        h_normalized = h.copy() / (np.sum(h) * dx)
        # Compute the correlation
        corr = correlate(h_normalized, 1 - h_normalized, mode="full")
        # Compute the maximum overlap integral
        c5 = np.max(corr) * dx
        return c5
    
    def evaluate_h(h):
        # Ensure h is valid
        if np.any(h < 0) or np.any(h > 1):
            return float('inf')
        # Check the integral constraint
        if np.abs(np.sum(h) * dx - 1) > 1e-6:
            return float('inf')
        return objective_function(h)
    
    # Initial evaluation
    initial_c5 = evaluate_h(initial_h)
    best_h = initial_h.copy()
    best_c5 = initial_c5
    
    # Optimization settings
    max_iter = 50
    tolerance = 1e-4
    
    for _ in range(max_iter):
        # Use local optimization around the current best solution
        # Define the optimization variables
        h = cp.Variable(n_points, lower=0, upper=1)
        objective = cp.Minimize(objective_function(h))
        
        # Define constraints
        constraint = [cp.sum(h) * dx == 1]
        
        # Setup and solve the problem
        prob = cp.Problem(objective, constraint)
        try:
            prob.solve(solver=cp.SCS, eps=1e-4, verbose=False)
            if prob.status == 'optimal':
                current_h = h.value
                current_c5 = evaluate_h(current_h)
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
        except:
            pass
        
        # Early stopping if the improvement is below a threshold
        if np.abs(best_c5 - initial_c5) < tolerance:
            break
    
    # Return the best solution found
    return (best_h, best_c5, n_points)
