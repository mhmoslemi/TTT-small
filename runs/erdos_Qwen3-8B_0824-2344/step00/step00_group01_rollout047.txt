```python
import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import scipy.sparse.linalg as spla
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    elapsed = 0
    
    # Initial construction
    n_points = 47
    h_initial = initial_h_values.copy()
    
    # Try to refine the solution by increasing n_points
    for n_points in [64, 128, 256, 512, 1024]:
        if elapsed >= budget_s:
            break
        
        # Initialize h based on previous solution
        h = np.zeros(n_points)
        h[::2] = 1
        h[1::2] = 0
        h = h / (n_points / 2) * n_points / 2
        
        # Use the initial solution as a starting point
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
        
        # Objective: minimize max correlation
        def objective(x):
            h = x.reshape(-1)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr) * (2.0 / n_points)
        
        # Constraints: sum(h) = n_points / 2
        def constraint(x):
            return np.sum(x) - n_points / 2
        
        # Bounds: 0 <= h[i] <= 1
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimization
        result = optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        h = result.x
        c5_bound = result.fun
        elapsed = time.time() - start_time
        
        if elapsed < budget_s:
            # Try to refine further with a finer grid
            h_fine = np.zeros(2 * n_points)
            h_fine[::2] = h[:n_points]
            h_fine[1::2] = h[:n_points]
            h_fine = h_fine / (2 * n_points / 2) * (2 * n_points / 2)
            h_fine = np.interp(np.linspace(0, 2, 2 * n_points), np.linspace(0, 2, n_points), h)
            
            n_points = 2 * n_points
            h = h_fine
            h = h / (n_points / 2) * n_points / 2
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
        
        if elapsed >= budget_s:
            break
    
    # Final solution
    h_values = h
    c5_bound = evaluate_erdos_solution(h_values)
    
    return h_values, c5_bound, len(h_values)
```
