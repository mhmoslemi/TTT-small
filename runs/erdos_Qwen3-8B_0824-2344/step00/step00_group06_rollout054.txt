import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points (adjust if needed)
    dx = 2.0 / n_points
    h = np.copy(initial_h_values[:n_points])
    h = h / np.sum(h) * (n_points / 2)

    # Objective function to minimize C5
    def objective(h_vec):
        h = h_vec.reshape(-1)
        # Ensure h is in [0, 1] and sum is correct
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if abs(sum_h - (n_points / 2)) > 1e-6:
            return np.inf
        # Compute the overlap integral
        h_expanded = np.repeat(h, 2)
        shifted = np.roll(h_expanded, 1)
        overlap = np.convolve(h, 1 - h, mode="full") * dx
        c5 = np.max(overlap)
        return c5

    # Optimization: use a local optimizer for speed
    result = optimize.minimize(
        objective,
        h,
        method="L-BFGS-B",
        bounds=[(0, 1) for _ in range(n_points)],
        options={"maxiter": 100, "ftol": 1e-6, "gtol": 1e-6}
    )

    h_opt = result.x
    h_opt = np.clip(h_opt, 0, 1)
    sum_h_opt = np.sum(h_opt)
    if abs(sum_h_opt - (n_points / 2)) > 1e-6:
        h_opt = h_opt / sum_h_opt * (n_points / 2)

    # Evaluate C5
    c5 = objective(h_opt)

    # If time allows, increase n_points and refine
    if budget_s > 10:
        n_points = 200
        dx = 2.0 / n_points
        h_refined = np.clip(h_opt, 0, 1)
        h_refined = h_refined / np.sum(h_refined) * (n_points / 2)
        result_refined = optimize.minimize(
            lambda h_vec: objective(h_vec),
            h_refined,
            method="L-BFGS-B",
            bounds=[(0, 1) for _ in range(n_points)],
            options={"maxiter": 50, "ftol": 1e-6, "gtol": 1e-6}
        )
        h_opt = result_refined.x
        h_opt = np.clip(h_opt, 0, 1)
        sum_h_opt = np.sum(h_opt)
        if abs(sum_h_opt - (n_points / 2)) > 1e-6:
            h_opt = h_opt / sum_h_opt * (n_points / 2)
        c5 = objective(h_opt)

    return (h_opt, c5, n_points)
