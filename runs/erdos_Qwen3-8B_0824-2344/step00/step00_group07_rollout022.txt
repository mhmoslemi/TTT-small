def run(seed=42, budget_s=60, **kwargs):
    n_points = 200  # Use 200 points for a good balance between speed and accuracy
    dx = 2.0 / n_points
    h = initial_h_values[:n_points].copy()  # Start from initial values
    h = h / (dx * n_points) * n_points / 2  # Normalize to meet the area constraint

    # Function to evaluate the C5 bound
    def evaluate(h):
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)

    # Function to optimize the function using gradient descent
    def optimize_h(h, max_iter=1000, learning_rate=0.01, tolerance=1e-5):
        for i in range(max_iter):
            # Compute current C5
            current_c5 = evaluate(h)
            if current_c5 < tolerance:
                break

            # Compute gradient approximation
            grad = np.zeros_like(h)
            for j in range(len(h)):
                # Perturb one point at a time
                h_perturbed = h.copy()
                h_perturbed[j] += 1e-5
                h_perturbed = np.clip(h_perturbed, 0, 1)
                perturbed_c5 = evaluate(h_perturbed)
                grad[j] = (perturbed_c5 - current_c5) / 1e-5

            # Update h
            h -= learning_rate * grad
            h = np.clip(h, 0, 1)

            # Ensure area constraint is satisfied
            h_sum = np.sum(h)
            if h_sum != n_points / 2:
                h = h * (n_points / 2) / h_sum

        return h

    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate(best_h)
    n_points_used = n_points

    # Attempt some local search and refinement
    for _ in range(3):
        h = optimize_h(h, max_iter=100, learning_rate=0.01)
        current_c5 = evaluate(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    # Final evaluation
    final_c5 = evaluate(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)

    # Ensure we stay within the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        print("Time budget exceeded. Returning best found so far.")
        return (best_h, final_c5, n_points_used)

    return (best_h, final_c5, n_points_used)
