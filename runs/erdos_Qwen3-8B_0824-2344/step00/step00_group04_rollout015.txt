def run(seed=42, budget_s=60, **kwargs):
    # Initial parameters
    n_points = 100  # Start with 100 points for better accuracy
    dx = 2.0 / n_points
    h_initial = initial_h_values()
    h_initial = np.resize(h_initial, n_points)
    h_initial = h_initial[:n_points]  # Ensure it's of correct length
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)  # Normalize to satisfy the integral constraint
    
    # Objective: minimize the maximum overlap with shifted 1 - h
    def objective_function(h):
        h = np.array(h, dtype=np.float64)
        shifted_h = np.zeros_like(h)
        for k in range(1, len(h)):
            shifted_h = np.roll(h, k)
            overlap = correlate(h, 1 - shifted_h, mode="full")
            c5 = np.max(overlap) * dx
        return c5
    
    # Optimization: use scipy.optimize to minimize the objective function
    def optimize_h():
        x0 = h_initial.copy()
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = optimize.minimize(objective_function, x0, bounds=bounds, method='L-BFGS-B', tol=1e-5)

    # Try different n_points
    best_h = h_initial
