import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute the overlap integral
    # For each k, integrate h(x)(1 - h(x+k)) over x in [0, 2]
    # We'll use a convolution approach
    # h_1_minus_h = 1 - h_values
    # Use np.correlate with mode="full" to get all shifted versions
    correlation = np.correlate(h_values, 1 - h_values, mode="full")
    # Only consider shifts k where 0 <= k <= 2 (since the domain is [0, 2])
    # The valid k indices are between n_points - 2 and n_points + 2
    valid_indices = np.arange(n_points - 2, n_points + 2 + 1)
    valid_correlation = correlation[valid_indices]
    # Compute max overlap
    max_overlap = np.max(valid_correlation) * dx
    return max_overlap

def initial_h_values():
    # A rough initial guess with 80 points
    n_points = 80
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if 0 <= x < 0.4:
            h[i] = 0.0
        elif 0.4 <= x < 0.6:
            h[i] = 1.0
        elif 0.6 <= x < 0.8:
            h[i] = 0.0
        elif 0.8 <= x < 1.0:
            h[i] = 1.0
        elif 1.0 <= x < 1.2:
            h[i] = 0.0
        elif 1.2 <= x < 1.4:
            h[i] = 1.0
        elif 1.4 <= x < 1.6:
            h[i] = 0.0
        elif 1.6 <= x < 1.8:
            h[i] = 1.0
        elif 1.8 <= x < 2.0:
            h[i] = 0.0
    # Ensure the integral is exactly 1
    total = np.sum(h) * dx
    if total != 1:
        h = h * (1 / total)
    return h

# Strategy:
# 1. Start from the initial guess with 80 points
# 2. Use a local optimization method to minimize the maximum overlap
# 3. Try to improve the solution by varying the number of points
# 4. Use a heuristic or grid search to explore different configurations
# 5. Ensure the solution is within the time budget

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    
    # Initial setup
    initial_h = initial_h_values()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to evaluate the maximum overlap for a given h
    def evaluate(h):
        return evaluate_erdos_solution(h)
    
    # Define the optimization problem
    # We aim to minimize the maximum overlap
    # We use a local optimization method such as SLSQP
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        return evaluate(h)
    
    # Bounds for h
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    # Constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    # Initial guess
    initial_guess = initial_h.copy()
    
    # Optimization
    result = optimize.minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    # Check if we have a feasible solution
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        best_h = initial_h
        best_c5 = evaluate(best_h)
    
    # Try a slightly larger number of points to see if we can improve further
    # (we'll use double the points for this)
    n_points_new = n_points * 2
    h_new = np.zeros(n_points_new)
    dx_new = 2.0 / n_points_new
    
    # Interpolate the current best solution to a finer grid
    # This will give us a starting point for the new number of points
    h_interpolated = np.interp(np.linspace(0, 2, n_points_new), np.linspace(0, 2, n_points), best_h)
    
    # Use the interpolated values as an initial guess
    initial_guess_new = h_interpolated.copy()
    
    # Setup for the new optimization problem
    def objective_new(h_flat):
        h = h_flat.reshape(n_points_new)
        return evaluate_erdos_solution(h)
    
    bounds_new = [(0.0, 1.0) for _ in range(n_points_new)]
    # Constraint: sum(h) * dx_new = 1 => sum(h) = n_points_new / 2
    def constraint_new(h_flat):
        h = h_flat.reshape(n_points_new)
        return np.sum(h) - n_points_new / 2
    
    # Optimization
    result_new = optimize.minimize(
        objective_new,
        initial_guess_new,
        method='SLSQP',
        bounds=bounds_new,
        constraints={'type': 'eq', 'fun': constraint_new},
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    # Check if we have a feasible solution
    if result_new.success:
        best_h_new = result_new.x
        best_c5_new = result_new.fun
    else:
        best_h_new = h_interpolated
        best_c5_new = evaluate_erdos_solution(best_h_new)
    
    # Choose the better of the two solutions
    if best_c5_new < best_c5:
        best_h = best_h_new
        best_c5 = best_c5_new
        n_points = n_points_new
    
    # Final check to ensure we are within the time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
