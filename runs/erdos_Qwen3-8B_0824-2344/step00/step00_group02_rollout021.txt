import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions and variables
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    h_values = copy.deepcopy(initial_h_values)
    n_points = len(h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(h_values)
    # Ensure the sum matches the requirement
    if abs(sum_h - n_points / 2) > 1e-6:
        h_values = (h_values * (n_points / 2) / sum_h).clip(0, 1)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = h.clip(0, 1)
        h = np.concatenate([h, np.zeros_like(h)])
        shifted = np.roll(h, 1)
        overlap = np.correlate(h, 1 - shifted, mode='full') * dx
        return np.max(overlap)
    
    # Function to evaluate the solution
    def evaluate(h):
        nonlocal n_points
        n_points = len(h)
        dx = 2.0 / n_points
        if dx < 1e-6:
            return float('inf')
        h = h.clip(0, 1)
        sum_h = np.sum(h)
        if abs(sum_h - n_points / 2) > 1e-6:
            return float('inf')
        h = np.concatenate([h, np.zeros_like(h)])
        shifted = np.roll(h, 1)
        overlap = np.correlate(h, 1 - shifted, mode='full') * dx
        c5 = np.max(overlap)
        return c5
    
    # Optimization settings
    max_iter = 100
    tol = 1e-4
    best_h = h_values.copy()
    best_c5 = evaluate(best_h)
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Perturb the current solution slightly
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        
        # Check the sum constraint
        sum_perturbed = np.sum(perturbed_h)
        if abs(sum_perturbed - n_points / 2) > 1e-6:
            # Adjust to maintain the sum
            scale = (n_points / 2) / sum_perturbed
            perturbed_h = perturbed_h * scale
        
        current_c5 = evaluate(perturbed_h)
        
        # Accept the perturbation if it improves the solution
        if current_c5 < best_c5:
            best_h = perturbed_h.copy()
            best_c5 = current_c5
    
    # Refinement step using local optimization
    def objective(h):
        return evaluate(h)
    
    # Use L-BFGS-B optimizer for refinement
    result = optimize.minimize(
        objective,
        best_h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    
    best_h = result.x
    best_c5 = evaluate(best_h)
    
    # Final evaluation
    final_c5 = best_c5
    final_h = best_h
    final_n_points = len(final_h)
    
    # Ensure the final solution meets the constraints
    final_h = np.clip(final_h, 0, 1)
    sum_final = np.sum(final_h)
    if abs(sum_final - final_n_points / 2) > 1e-6:
        scale = (final_n_points / 2) / sum_final
        final_h = final_h * scale
    
    # Final evaluation
    final_c5 = evaluate(final_h)
    
    # Ensure we don't exceed the time budget
    if time.time() - start_time > budget_s - 1:
        return (final_h, final_c5, final_n_points)
    
    return (final_h, final_c5, final_n_points)
