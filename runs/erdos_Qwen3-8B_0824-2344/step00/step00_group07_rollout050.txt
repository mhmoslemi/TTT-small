import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions and variables
initial_h_values = np.load('initial_h_values.npy')  # Assume this is preloaded with an initial construction

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    n_points = len(initial_h_values)
    h_values = initial_h_values.copy()
    dx = 2.0 / n_points
    
    def evaluate_c5(h):
        h = np.array(h)
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        # Compute overlap integral
        # Pad h to double length for full convolution
        h_padded = np.pad(h, (0, len(h)), mode='constant')
        # Compute correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        # Scale by dx
        c5 = np.max(correlation) * dx
        return c5
    
    def objective_function(h_values):
        return evaluate_c5(h_values)
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Use a local optimization method to refine the initial solution
    result = optimize.minimize(
        objective_function,
        x0=h_values,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    
    # Check if we have time to run a more sophisticated optimization
    elapsed = time.time() - start_time
    if elapsed < budget_s - 5:
        # Try a small convex optimization
        h = cp.Variable(n_points)
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        objective = cp.Maximize(cp.max(cp.correlate(h.value, 1 - h.value, mode='full') * dx))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.SCS, max_iter=100)
        h_values = h.value
    
    # Final evaluation
    final_c5 = evaluate_c5(h_values)
    
    # Return the best solution found
    return (h_values, final_c5, n_points)
