import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1.0 - h
    corr = signal.correlate(h, one_minus_h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    # Placeholder for initial construction
    return [0.5] * 95

# Strategy:
# 1. Start with the initial_h_values as a baseline.
# 2. Perform a local optimization to improve the construction.
# 3. Use a simple gradient-based method with constraints to update the h_values.
# 4. Evaluate the overlap integral and track the best solution within the time budget.

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 95
    dx = 2.0 / n_points
    
    # Initial guess based on initial_h_values
    h_initial = initial_h_values()
    h = np.array(h_initial, dtype=np.float64)
    
    # Function to compute the overlap integral
    def objective(x):
        h = x.reshape((n_points,))
        one_minus_h = 1.0 - h
        corr = signal.correlate(h, one_minus_h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Function to compute constraints
    def constraints(x):
        h = x.reshape((n_points,))
        return [np.sum(h) - n_points / 2]  # Ensure the integral of h(x) is 1
    
    # Optimization problem: minimize the overlap integral
    # We use a local optimization method with constraints
    x = cp.Variable(n_points)
    obj = cp.Minimize(objective(x))
    cons = [cp.sum(x) == n_points / 2]
    prob = cp.Problem(obj, cons)
    
    # Run the optimization
    start_time = time.time()
    result = prob.solve(solver=cp.SCS, max_iter=100, verbose=False)
    elapsed_time = time.time() - start_time
    
    if result is not None:
        h_opt = x.value
        c5_bound = evaluate_erdos_solution(h_opt)
        return (h_opt.tolist(), c5_bound, n_points)
    else:
        # Fallback to initial values if optimization fails
        return (h_initial, evaluate_erdos_solution(h_initial), n_points)
